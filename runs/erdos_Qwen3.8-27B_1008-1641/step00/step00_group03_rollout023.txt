import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def proj_sum(h, n):
    """Project to sum = n/2, 0 <= h <= 1."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Simple uniform adjustment
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # Second pass for any clipping issues
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if np.sum(free) > 0:
                h[free] += diff2 / np.sum(free)
                h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, n, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    if temp < 1e-10:
        return m, None
    e = np.exp((scaled - m) / temp)
    val = m + temp * np.log(np.sum(e))
    # Gradient
    if temp >= 1e-10:
        w = e / np.sum(e)  # softmax weights, shape (2n-1,)
        # w[k] corresponds to lag = k - (n-1)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            wk = w[k] * dx

                continue
            # d(corr_k)/d(h[j]) = (1-h[j+lag])*[0<=j+lag<n] - h[j-lag]*[0<=j-lag<n]
            for j in range(n):

                jm = j - lag
                contrib = 0.0
                if 0 <= jp < n:
                    contrib += (1.0 - h[jp])
                if 0 <= jm < n:
                    contrib -= h[jm]
                grad[j] += wk * contrib
        return val, grad
    return val, None

where A_k = sum of all terms not involving h[j], and B_k = coefficient of h[j].

From the derivative: B_k = (1-h[j+lag_k])*[0<=j+lag_k<n] - h[j-lag_k]*[0<=j-lag_k<n]

So we want to minimize max_k (A_k + B_k * t) * dx subject to 0 <= t <= 1 and the sum constraint.

The sum constraint: sum(h) = n/2 means t = n/2 - sum(h without j). So t is actually determined! We can't freely choose t.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def proj_sum(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            free = (h > 1e-12) & (h < 1.0 - 1e-12)
            if np.any(free):
                h[free] += diff2 / np.sum(free)
                h = np.clip(h, 0.0, 1.0)
    return h


def sobj(h, n, dx, temp):
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    if temp < 1e-12:
        return m
    e = np.exp((scaled - m) / temp)
    s = np.sum(e)
    w = e / s
    val = m + temp * np.log(s)
    grad = np.zeros(n)
    # Vectorized gradient: grad[j] = dx * sum_k w[k] * ((1-h[j+lag_k])*1[j+lag_k valid] - h[j-lag_k]*1[j-lag_k valid])
    # For each lag, accumulate contribution to each j
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k] * dx
        if abs(wk) < 1e-18:
            continue
        # h[j+lag] valid: 0<=j+lag<n => max(0,-lag) <= j <= min(n-1, n-1-lag)
        j_lo1 = max(0, -lag)
        j_hi1 = min(n - 1, n - 1 - lag)
        if j_lo1 <= j_hi1:
            grad[j_lo1:j_hi1 + 1] += wk * (1.0 - h[j_lo1 + lag:j_hi1 + 1 + lag])
        # h[j-lag] valid: 0<=j-lag<n => max(0,lag) <= j <= min(n-1, n-1+lag)
        j_lo2 = max(0, lag)
        j_hi2 = min(n - 1, n - 1 + lag)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= wk * h[j_lo2 - lag:j_hi2 + 1 - lag]
    return val, grad
