import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-10) if diff > 0 else (h > 1e-10)
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_step(h, j, n):
    """Find optimal h[j] given others fixed, via 1D convex optimization."""
    dx = 2.0 / n
    hs = h[j]
    h[j] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode="full")
    h[j] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode="full")
    h[j] = hs
    
    a = c0 * dx
    b = (c1 - c0) * dx
    
    # Minimize max(a + b*t) for t in [0,1] - convex piecewise linear
    ts = np.linspace(0.0, 1.0, 401)
    vals = a + b * ts[:, None]
    fs = np.max(vals, axis=1)
    i_best = np.argmin(fs)
    
    # Refine
    if 0 < i_best < 400:
        lo, hi = ts[max(0, i_best-1)], ts[min(400, i_best+1)]
    else:
        lo, hi = 0.0, 1.0
    ts2 = np.linspace(lo, hi, 401)
    vals2 = a + b * ts2[:, None]
    fs2 = np.max(vals2, axis=1)
    t_opt = ts2[np.argmin(fs2)]
    
    return t_opt

    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = project_feasible(np.array(initial_h_values, dtype=float), n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    
    # Phase 1: Coordinate descent (main workhorse)
    order = list(range(n))
    sweep = 0
    while time.time() - t0 < budget_s * 0.45:
        sweep += 1
        # periodically reshuffle
        if sweep % 10 == 0:
            rng.shuffle(order)
        for j in order:
            h[j] = coord_step(h, j, n)
        h = project_feasible(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 2: SLSQP with smooth max
    bounds = Bounds(np.zeros(n), np.ones(n))
    lin_con = LinearConstraint(np.ones(n), n/2.0, n/2.0)
    
    if time.time() - t0 < budget_s * 0.7:
        for temp in [0.5, 0.1, 0.03, 0.01]:
            if time.time() - t0 > budget_s * 0.7:
                break
            def obj(h_in, t=temp):
                dx_val = 2.0 / n
                corr = np.correlate(h_in, 1.0 - h_in, mode="full") * dx_val
                m = np.max(corr)
                return m + t * np.log(np.sum(np.exp((corr - m) / t)))
            
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=[lin_con], tol=1e-10,
                          options={'maxiter': 3000, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_feasible(res.x, n)
                c5 = compute_c5(cand, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = cand.copy()
                # Also try from res.x without projection
                c5_raw = compute_c5(np.clip(res.x, 0, 1), n)
                if c5_raw < best_c5:
                    best_c5 = c5_raw
                    best_h = project_feasible(res.x, n)
    
    # Phase 3: Nelder-Mead on random coordinate subsets
    if time.time() - t0 < budget_s * 0.88:

            k_sub = rng.randint(5, min(15, n))
            idx = rng.choice(n, size=k_sub, replace=False)
            
            def obj_sub(sv, idx=idx):
                ht = best_h.copy()
                ht[list(idx)] = np.clip(sv, 0, 1)
                return compute_c5(project_feasible(ht, n), n)

            if res_nm.x is not None:

                ht[list(idx)] = np.clip(res_nm.x, 0, 1)
