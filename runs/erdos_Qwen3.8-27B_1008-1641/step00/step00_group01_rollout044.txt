def _project(v, target):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=target} via bisection."""
    # Find lambda such that sum(clip(v - lambda, 0, 1)) = target
    s0 = np.sum(np.clip(v, 0.0, 1.0))

import numpy as np
from scipy.optimize import differential_evolution, minimize

_ctx = {}

def _project(v, target):
    s = np.sum(np.clip(v, 0.0, 1.0))
    if s <= target:
        lo, hi = -1.0, 0.0
    else:
        lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
