    k_max = np.argmax(corr)
    val = corr[k_max]

    tol = 1e-6

            if 0 <= jm < n:
                val_j -= h[jm]
            g[j] = val_j
        grad += g
    grad /= len(top_indices)

        t_seg = min(t_end_all, t0 + budget_s * (0.45 if n == (len(parent) if parent is not None else 0) else 0.88))
        frac = 0.08
        t_end_seg = t0 + (time.time() - t0) + budget_s * frac
        t_end_seg = min(t_end_seg, t_end_all)
        h_opt, v_opt = pgd_optimize(h_init, n, t0, t_end_seg, rng, lr0=0.05, mom0=0.65)
        if v_opt < best_val:
            best_val = v_opt
            best_h = h_opt.copy()

        # Quick NM refine if time allows
        if time.time() < t_end_all - 30:
            h_r, v_r = nm_refine(h_opt, n, t0, min(t_end_all, time.time() + 30), rng)
            if v_r < best_val:
                best_val = v_r
                best_h = h_r.copy()

    # Phase 3: Final local search on best
    if best_h is not None and time.time() < t0 + budget_s * 0.96:
        n_best = len(best_h)
        t_end3 = t0 + budget_s * 0.96
        h_f, v_f = nm_refine(best_h, n_best, t0, t_end3, rng)
        if v_f < best_val:
            best_val = v_f
            best_h = h_f.copy()

    # Fallback
    if best_h is None:
        n = 45
        best_h = np.full(n, 0.5)
        best_h = project_h(best_h, n)
        best_val = obj_max(best_h, n)

    # Validate
    n_final = len(best_h)
    sum_check = np.sum(best_h)
    assert abs(sum_check - n_final / 2.0) < 1.0, f"Sum constraint violated: {sum_check}"
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_h(best_h, n_final)
    final_val = obj_max(best_h, n_final)

    return best_h, final_val, n_final
