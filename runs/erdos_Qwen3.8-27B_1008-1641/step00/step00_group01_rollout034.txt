```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)  # the actual lag
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j + k < n:
            grad[j] += (1.0 - h[j + k])
        if 0 <= j - k < n:
            grad[j] += (-h[j - k])
    return dx * grad

def project_onto_simplex_box(p, target_sum):
    """Project p onto {h: 0<=h<=1, sum(h)=target_sum} using bisection on lambda."""
    lo, hi = p.min() - 1.0, p.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(p - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune: adjust one element to exactly hit target
    diff = target_sum - h.sum()
    # Find element with most room
    if diff > 0:
        rooms = 1.0 - h
        idx = np.argmax(rooms)
        h[idx] += min(diff, rooms[idx])
    elif diff < 0:
        rooms = h
        idx = np.argmax(rooms)
        h[idx] += max(diff, -rooms[idx])
    return h

def optimize_h(h0, n, max_iters=5000, lr_init=0.01):
    """Projected subgradient method."""
    h = h0.copy().astype(float)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c = c5_value(h)
    lr = lr_init
    
    for it in range(max_iters):
        grad = c5_subgrad(h)
        h_new = h - lr * grad
        h_new = project_onto_simplex_box(h_new, target_sum)
        
        c = c5_value(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            # Slightly increase lr (or keep same)
            lr = min(lr * 1.01, 0.1)
        else:
            h_new = h  # revert
            lr *= 0.95
        
        h = h_new
    
    return best_h, best_c
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # softmax weights

    beta_corr -= beta_corr.max()  # for numerical stability
    exp_bc = np.exp(beta_corr)
    w = exp_bc / exp_bc.sum()

    return grad, dx * corr.max()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * corr
    bc -= bc.max()
    exp_bc = np.exp(bc)
    w = exp_bc / exp_bc.sum()
    grad = np.zeros(n)
    for m in range(len(corr)):
        k = m - (n - 1)
        wk = w[m] * dx
        if wk < 1e-12:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] += wk * (-h[:n - k])
        else:
            kn = -k
            grad[kn:] += wk * (1.0 - h[:n - kn])
            grad[:n - kn] += wk * (-h[kn:])
    return grad, dx * corr.max()

def project_feasible(p, target_sum):
    lo = p.min() - 1.0
    hi = p.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        ss = np.clip(p - mid, 0.0, 1.0).sum()
        if ss > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(p - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def optimize_smooth(h0, n, iters, beta, lr, anneal_beta=False):
    h = h0.copy().astype(float)
    ts = n / 2.0
    best_h = h.copy()
    best_c = c5_value(h)
    for it in range(iters):
        b = beta if not anneal_beta else beta * (1.0 + 3.0 * it / iters)
        g, _ = smooth_grad(h, b)
        step = lr * (1.0 - 0.5 * it / iters)
        h_new = project_feasible(h - step * g, ts)
        c = c5_value(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new
        else:
            h = best_h.copy()
    return best_h, best_c

def random_init(n, rng):
    h = rng.random(n)
    h = project_feasible(h, n / 2.0)
    return h

def cosine_init(n, freq):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.cos(np.pi * freq * x)
    h = project_feasible(h, n / 2.0)
    return h

def interp_init(h_ref, n_new):
    n_old = len(h_ref)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h = np.interp(x_new, x_old, h_ref)
    return project_feasible(h, n_new / 2.0)

def polish_slsqp(h0, n, max_iter=500):
    ts = n / 2.0
    dx = 2.0 / n

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        bc = 20.0 * corr
        bc -= bc.max()
        return (1.0 / 20.0) * (bc.max() + np.log(np.exp(bc).sum())) * dx

    def obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        bc = 20.0 * corr
        bc -= bc.max()
        exp_bc = np.exp(bc)
        w = exp_bc / exp_bc.sum()
        grad = np.zeros(n)
        for m in range(len(corr)):
            k = m - (n - 1)
            wk = w[m] * dx
            if wk < 1e-12:
                continue
            if k == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif k > 0:
                grad[:n - k] += wk * (1.0 - h[k:])
                grad[k:] += wk * (-h[:n - k])
            else:
                kn = -k
                grad[kn:] += wk * (1.0 - h[:n - kn])
                grad[:n - kn] += wk * (-h[kn:])
        return grad

    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - ts, 'jac': lambda h: np.ones(n)}]
    bounds = [(0, 1)] * n
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    if res.success:
        h_best = project_feasible(res.x, ts)
        return h_best, c5_value(h_best)
    return h0, c5_value(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c = np.inf
    best_n = 0

    def elapsed():
        return time.time() - t0

    # Use initial_h_values if available
    h_start = None
    try:
        h_start = np.array(initial_h_values, dtype=float)
    except:
        pass

    if h_start is not None:
        n_s = len(h_start)
        c_s = c5_value(h_start)
        if c_s < best_c:
            best_c, best_h, best_n = c_s, h_start.copy(), n_s

    # Try n values with multiple initializations
    n_list = [20, 25, 30, 35, 40, 47, 50, 60, 80]
    if h_start is not None:
        n_list.insert(0, len(h_start))

    for n in n_list:
        if elapsed() > budget_s * 0.85:
            break
        inits = [random_init(n, rng), cosine_init(n, 1), cosine_init(n, 2), cosine_init(n, 3)]
        if best_h is not None and len(best_h) != n:
            inits.append(interp_init(best_h, n))
        if h_start is not None and len(h_start) != n:
            inits.append(interp_init(h_start, n))

        for h0 in inits:
            if elapsed() > budget_s * 0.9:
                break
            for beta in [5.0, 15.0, 50.0]:
                if elapsed() > budget_s * 0.9:
                    break
                h_opt, c_opt = optimize_smooth(h0, n, iters=1500, beta=beta, lr=0.02, anneal_beta=True)
                if c_opt < best_c:
                    best_c, best_h, best_n = c_opt, h_opt.copy(), n

        # Polish best for this n if time remains
        if elapsed() < budget_s * 0.95:
            h_pol, c_pol = polish_slsqp(best_h, best_n, max_iter=200)
            if c_pol < best_c:
                best_c, best_h = c_pol, h_pol.copy()

    # Final verification
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_feasible(best_h, best_n / 2.0)
        best_c = c5_value(best_h)

    if best_h is None:
        # Fallback: constant 0.5
        best_n = 20
        best_h = np.full(best_n, 0.5)
        best_c = c5_value(best_h)

    return (best_h, best_c, best_n)
```
