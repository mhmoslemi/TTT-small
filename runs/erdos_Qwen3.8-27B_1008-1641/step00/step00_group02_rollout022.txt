            h_alt, c5_alt = coord_descent(h_alt, an_target, 50, 0.84, t0, budget_s)
        if c5_alt < best_c5:
            best_c5, best_h, best_n = c5_alt, h_alt, an

    # Phase 5: Random restarts
    while time.time() - t0 < budget_s * 0.92:
        h_r = np.random.rand(best_n)
        h_r = project_h(h_r, best_n / 2.0)
        h_r, c5_r = subgradient_optimize(h_r, best_n / 2.0, 2000, 0.3, 0.92, t0, budget_s)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()

    # Final polish
    if time.time() - t0 < budget_s * 0.97:
        h_f, c5_f = coord_descent(best_h, best_n / 2.0, 100, 0.97, t0, budget_s)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
