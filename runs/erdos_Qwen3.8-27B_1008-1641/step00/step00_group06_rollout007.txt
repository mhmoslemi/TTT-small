import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def subgradient_at_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for ka in active:
        off = ka - (n - 1)
        for m in range(n):
            i1 = m - off
            if 0 <= i1 < n:
                grad[m] += (1.0 - h[i1])
            j = m + off
            if 0 <= j < n:
                grad[m] -= h[j]
    grad /= len(active)
    return dx * grad, dx * max_val

def project_box_sum(h, S):
    n = len(h)
    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
