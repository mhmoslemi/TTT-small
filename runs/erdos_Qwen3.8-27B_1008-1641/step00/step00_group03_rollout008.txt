    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP from initial
    if rem() > 40:
        h_new, c5_new = slsqp_refine(best_h, n_init, maxiter=600)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Binary SA
    if rem() > 50:
        h_bin, c5_bin = binary_sa(best_h, n_init, t0, budget_s, rng, max_iter=10000)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin

    # Phase 3: SLSQP again from best (including binary)
    if rem() > 40:
        h_new, c5_new = slsqp_refine(best_h, n_init, maxiter=600, alpha=120.0)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: Fractional local search
    if rem() > 40:
        h_new, c5_new = local_frac_search(best_h, n_init, t0, budget_s, rng)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Try multiple random restarts with SLSQP
    for restart in range(6):
        if rem() < 50:
            break
        # Interpolate + noise
        noise = rng.uniform(-0.15, 0.15, n_init)
        h_try = project_feasible(best_h + noise, n_init)
        h_new, c5_new = slsqp_refine(h_try, n_init, maxiter=300)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Try different n values
    for n_try in [50, 80, 100, 140]:
        if rem() < 50:
            break
        x_old = np.linspace(0, 2, n_init, endpoint=False) + 1.0 / n_init
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_new, c5_new = slsqp_refine(h_interp, n_try, maxiter=400)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 7: Final local polish
    if rem() > 20:
        h_new, c5_new = local_frac_search(best_h, len(best_h), t0, budget_s, rng)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(np.sum(best_h) - len(best_h) / 2.0) < 1e-8
    return best_h, best_c5, len(best_h)
