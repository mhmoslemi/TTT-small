    except Exception:
        return h_start

def do_slsqp(h_start, bnds, maxiter):
    try:
        cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
        res = minimize(pobj, h_start, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h_start

def run(seed=42, budget_s=1100, **kwargs):
    global _N, _TARGET, _DX, _PEN
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=np.float64)
    n = len(h0)
    _N = n
    _TARGET = n / 2.0
    _DX = 2.0 / n
    _PEN = 500.0
    bnds = [(0.0, 1.0)] * n

    best = h0.copy()
    best_c = eval_c5(best)

    # Phase 1: L-BFGS-B from initial
    h_try = do_lbfgs(h0.copy(), bnds, 2000)
    h_try = project(h_try, _TARGET)
    c_try = eval_c5(h_try)
    if c_try < best_c:
        best, best_c = h_try, c_try

    # Phase 2: SLSQP with equality constraint
    if time.time() - t0 < budget_s - 30:
        h_try = do_slsqp(best.copy(), bnds, 500)
        h_try = project(h_try, _TARGET)
        c_try = eval_c5(h_try)
        if c_try < best_c:
            best, best_c = h_try, c_try

    # Phase 3: Random restarts with L-BFGS-B
    for i in range(400):
        if time.time() - t0 > budget_s - 60:
            break
        scale = 0.015 + 0.08 * rng.random()
        h_try = best + rng.standard_normal(n) * scale
        h_try = project(h_try, _TARGET)
        if i % 5 == 0:
            h_try = do_slsqp(h_try, bnds, 150)
        else:
            h_try = do_lbfgs(h_try, bnds, 200)
        h_try = project(h_try, _TARGET)
        c_try = eval_c5(h_try)
        if c_try < best_c - 1e-13:
            best, best_c = h_try, c_try

    # Phase 4: Coordinate descent on exact objective
    steps = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005]
    stagnant = 0
    for it in range(20000):
        if time.time() - t0 > budget_s - 10:
            break
        improved = False
        for ss in steps:
            for j in range(n):
                found = False
                for sgn in (1, -1):
                    h_try = best.copy()
                    h_try[j] += sgn * ss
                    h_try = project(h_try, _TARGET)
                    c_try = eval_c5(h_try)
                    if c_try < best_c - 1e-13:
                        best, best_c = h_try, c_try
                        improved = True
                        found = True
                        break
                if found:
                    break
            if improved:
                break
        if not improved:
            stagnant += 1
            if stagnant >= 3:
                steps = [s * 0.5 for s in steps]
                stagnant = 0
            if steps[0] < 2e-7:
                break
        else:
            stagnant = 0

    # Phase 5: Random perturbation search (simulated annealing lite)
    temp = 0.001
    for it in range(50000):
        if time.time() - t0 > budget_s - 3:
            break
        j = rng.integers(n)
        delta = rng.normal(0, 0.01)
        h_try = best.copy()
        h_try[j] += delta
        if 0 <= h_try[j] <= 1:
            h_try = project(h_try, _TARGET)
            c_try = eval_c5(h_try)
            if c_try < best_c:
                best, best_c = h_try, c_try
            elif rng.random() < np.exp((best_c - c_try) / temp):
                best = h_try
        temp *= 0.99999
        if temp < 1e-6:
            temp = 0.0005

    # Final projection
    best = project(best, _TARGET)
    best_c = eval_c5(best)

    return (best, best_c, n)
