    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

import numpy as np
import time
import math
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(v, n):
    target = n / 2.0
    v = np.asarray(v, dtype=float)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def make_obj(n):
    def obj(v):
        h = project_feasible(v, n)
        return compute_c5(h)
    return obj

def smooth_obj(v, n, T):
    h = project_feasible(v, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = np.max(corr)
    centered = corr - mx
    return float(mx + (1.0 / T) * np.log(np.sum(np.exp(T * centered))))

def make_smooth_obj(n, T):
    def obj(v):
        return smooth_obj(v, n, T)
    return obj

def nelder_mead_local(v0, n, maxiter=3000):
    obj_fn = make_obj(n)
    res = minimize(
        obj_fn, v0,
        method='Nelder-Mead',
        options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True}
    )
    return project_feasible(res.x, n)

def powell_local(v0, n, maxiter=2000):
    obj_fn = make_obj(n)
    res = minimize(
        obj_fn, v0,
        method='Powell',
        options={'maxiter': maxiter, 'xtol': 1e-8, 'ftol': 1e-11}
    )
    return project_feasible(res.x, n)

def randomized_start(n, rng):
    v = rng.random(n)
    return project_feasible(v, n)

def structured_start(n, rng):
    # Create a block-like pattern with random widths
    v = np.zeros(n)
    pos = 0
    while pos < n:
        width = rng.integers(1, max(2, n // 4))
        v[pos:pos + width] = 1.0
        pos += width
        gap = rng.integers(0, max(1, n // 5))
        pos += gap
    return project_feasible(v, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_base = len(h_init)
    except NameError:
        h_init = None
        n_base = 45

    if h_init is not None:
        h_init = project_feasible(h_init, n_base)

    best_h = None
    best_val = 1.0
    best_n = n_base

    def elapsed():
        return time.time() - t0

    def update_best(h, val, n):
        nonlocal best_h, best_val, best_n
        if val < best_val:
            best_val = val
            best_h = h.copy()
            best_n = n

    # Evaluate initial
    if h_init is not None:
        init_val = compute_c5(h_init)
        update_best(h_init, init_val, n_base)

    # Main optimization loop: Nelder-Mead + Powell with restarts
    n_opt = n_base
    v_best = h_init.copy() if h_init is not None else project_feasible(np.full(n_opt, 0.5) + 0.15 * np.sin(2 * np.pi * np.arange(n_opt) / n_opt), n_opt)

    total_budget = budget_s - 30  # reserve 30s for safety
    max_restarts = 80

    for i in range(max_restarts):
        if elapsed() > total_budget:
            break

        # Choose noise level (annealing)
        frac = i / max_restarts
        noise = 0.12 * (1.0 - 0.7 * frac)

        # Starting point
        if i == 0 and h_init is not None:
            v0 = v_best.copy()
        elif i % 7 == 5:
            # Random restart
            v0 = randomized_start(n_opt, rng)
        elif i % 7 == 6:

            v0 = structured_start(n_opt, rng)

            # Upsample/downsample comparison: just compare raw C5 values
            if val_cand < best_val:

        best_n = n_base
        best_val = compute_c5(best_h)

    # Ensure exact constraint satisfaction
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h)

    return best_h, best_val, best_n
