    """Compute gradient of all c[m] w.r.t. h[p]. Returns array of shape (2n-1, n)."""
    grad = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        lag = m - (n-1)  # lag ranges from -(n-1) to n-1
        for p in range(n):

def project_polytope(h, target):

    # Actually lambda range: h - lambda in [0,1] means h-1 <= lambda <= h

        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_single_lag(h, k):
    """Gradient of S[k] (unscaled correlation) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # S[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        p1 = np.arange(k, n)
        g[p1] += (1.0 - h[p1 - k])
        p2 = np.arange(0, n - k)
        g[p2] -= h[p2 + k]

        g[p1] += (1.0 - h[p1 + q])
        p2 = np.arange(q, n)
        g[p2] -= h[p2 - q]
    return g

import numpy as np
import time


def project_polytope(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} via bisection on shift."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def eval_c5(h):
    """Compute C5 = max_k correlate(h,1-h,'full')[k] * dx."""
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def grad_lag(h, k):
    """Gradient of unscaled correlation S[k] w.r.t. h. k is the lag (neg or pos)."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        p1 = np.arange(k, n)
        g[p1] += 1.0 - h[p1 - k]
        if n - k > 0:
            p2 = np.arange(0, n - k)
            g[p2] -= h[p2 + k]
    else:
        q = -k
        if n - q > 0:
            p1 = np.arange(0, n - q)
            g[p1] += 1.0 - h[p1 + q]
        if q < n:
            p2 = np.arange(q, n)
            g[p2] -= h[p2 - q]
    return g


def active_subgrad_step(h, target, lr, rng):
    """One active-set subgradient step with optional noise."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top = np.argsort(corr)[-5:]
    k_active = int(top[rng.integers(0, len(top))]) - (n - 1)
    g = grad_lag(h, k_active)
    g -= g.mean()
    h_new = project_polytope(h - lr * g, target)
    return h_new, corr[k_active + (n - 1)]

        return (time.time() - t0) / budget_s

    # Phase 1: Active-set subgradient with adaptive step

    patience = 0

            patience = 0
        else:
            patience += 1
        if patience > 400:
            lr *= 0.7
            patience = 0
        lr = max(0.03, lr * 0.99998)

    # Phase 2: Softmax-smoothed gradient descent
    h = best_h.copy()
    beta = 8.0
    lr = 0.4
