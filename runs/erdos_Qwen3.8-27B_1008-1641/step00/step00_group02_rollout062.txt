    best_c5 = c5_value(best_h, n)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    bounds = [(0.0, 1.0)] * n
    
    def penalized_obj(h):
        return c5_value(h, n) + 100.0 * (np.sum(h) - target_sum) ** 2
    
    # Phase 1: L-BFGS-B
    if time_left() > 10:

            res = minimize(penalized_obj, best_h.copy(), method='L-BFGS-B',

            hc = project_feasible(res.x, target_sum)
            vc = c5_value(hc, n)
            if vc < best_c5:
                best_h, best_c5 = hc, vc
        except Exception:
            pass
    
    # Phase 2: Projected subgradient
    if time_left() > 10:
        h_curr = best_h.copy()
        lr = 0.01
        no_improve = 0
        while time_left() > 10 and lr > 1e-6:
            grad = c5_subgrad(h_curr, n)
            h_curr = project_feasible(h_curr - lr * grad, target_sum)
            vc = c5_value(h_curr, n)
            if vc < best_c5 - 1e-12:
                best_h, best_c5 = h_curr.copy(), vc
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 200:
                lr *= 0.9
                no_improve = 0

    if time_left() > 20:
        for _ in range(50):
            if time_left() < 15:
                break
            h_try = project_feasible(best_h + rng.normal(0, 0.03, n), target_sum)

                              bounds=bounds, options={'maxiter': 500, 'ftol': 1e-15})
                hc = project_feasible(res.x, target_sum)

            if vc < best_c5:

            lr *= 0.999
    
    return best_h, best_c5, n

import time

    n = _OPT['n']
    ts = _OPT['ts']
    return c5_value(h, n) + 500.0 * (np.sum(h) - ts) ** 2

def project_feasible(h, target_sum):
    n = len(h)
    h = h.copy()
    for _ in range(80):
        lo, hi = -1.5, 1.5
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
                lo = mid
            else:
                hi = mid
        new_h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
        if np.max(np.abs(new_h - h)) < 1e-14:
            return new_h
        h = new_h
    return h

def c5_value(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * (2.0 / n))

def c5_subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    mx = np.max(corr)
    act = np.where(corr >= mx - 1e-12)[0]
    g = np.zeros(n)
    for k in act:
        sh = k - (n - 1)
        a = max(0, -sh)
        b = min(n, n - sh)
        if b > a:
            g[a:b] += 1.0 - h[a+sh:b+sh]
        c = max(0, sh)
        d = min(n, n + sh)
        if d > c:
            g[c:d] -= h[c-sh:d-sh]
    return g * dx / len(act)

_OPT = {}

    return c5_value(h, _OPT['n']) + 1000.0 * (np.sum(h) - _OPT['ts']) ** 2

def project_feasible(h, ts):
    h = h.copy()
    for _ in range(100):
        lo, hi = -1.5, 1.5
        for _ in range(50):
            m = 0.5 * (lo + hi)
            if np.sum(np.clip(h - m, 0.0, 1.0)) > ts:
                lo = m
            else:
                hi = m
        nh = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
        if np.max(np.abs(nh - h)) < 1e-14:
            return nh
        h = nh
    return h

def c5_value(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * (2.0 / n))

def c5_subgrad(h, n):
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, 'full')
    g = np.zeros(n)
    mx = np.max(cr)
    act = np.where(cr >= mx - 1e-12)[0]
    for k in act:
        sh = k - n + 1
        a, b = max(0, -sh), min(n, n - sh)
        if b > a:
            g[a:b] += 1.0 - h[a + sh:b + sh]
        c, d = max(0, sh), min(n, n + sh)
        if d > c:
            g[c:d] -= h[c - sh:d - sh]
    return g * dx / len(act)
