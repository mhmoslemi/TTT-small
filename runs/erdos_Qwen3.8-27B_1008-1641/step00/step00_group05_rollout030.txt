        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        if not np.any(m):
            break
        s = d / np.sum(m)
        if d > 0:
            s = min(s, (1.0 - h[m]).min())
        else:
            s = max(s, h[m].min())
        h[m] += s
    return np.clip(h, 0.0, 1.0)


def _obj(h, dx, t, pen, n):
    hc = np.clip(h, 0.0, 1.0)
    c = np.correlate(hc, 1.0 - hc, mode='full')
    return _smooth_max(c, t) * dx + pen * (np.sum(hc) - 0.5 * n) ** 2


def _grad(h, dx, t, pen, n):
    e = 1e-7
    b = _obj(h, dx, t, pen, n)
    g = np.empty(n)
    for i in range(n):
        h2 = h.copy()
        h2[i] += e
        g[i] = (_obj(h2, dx, t, pen, n) - b) / e
    return g


def _run_one(h0, dx, t, pen, n, mi):
    c = _project(h0, 0.5 * n)
    r = minimize(_obj, c, args=(dx, t, pen, n), method='L-BFGS-B',
                 jac=_grad, bounds=[(0.0, 1.0)] * n,
                 options={'maxiter': mi, 'ftol': 1e-14, 'gtol': 1e-9})
    return _project(r.x, 0.5 * n)


def _shake(h, rng, n, ts, trials=2000):
    best = h.copy()
    best_c = _eval_c5(best, 2.0 / n)
    for _ in range(trials):
        i, j = rng.choice(n, 2, replace=False)
        delta = rng.uniform(-0.1, 0.1)
        h2 = h.copy()
        h2[i] += delta
        h2[j] -= delta
        h2 = np.clip(h2, 0.0, 1.0)
        if abs(np.sum(h2) - ts) > 1e-10:
            continue
        c2 = _eval_c5(h2, 2.0 / n)
        if c2 < best_c:
            best_c = c2
            best = h2.copy()
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    dx = 2.0 / n
    ts = 0.5 * n
    h0 = _project(h0, ts)

    bh = h0.copy()
    bc = _eval_c5(bh, dx)

    # Phase 1: Multi-temperature optimization from best
    for t in [10, 20, 35, 55, 80, 120, 200, 350]:
        if time.time() - t0 > budget_s * 0.5:
            break
        hr = _run_one(bh, dx, t, 100.0, n, 250)
        c = _eval_c5(hr, dx)
        if c < bc:
            bc, bh = c, hr.copy()

    # Phase 2: Random restarts with varying perturbation
    max_r = 200
    for r in range(max_r):
        if time.time() - t0 > budget_s * 0.85:
            break
        s = 0.02 + 0.20 * (r / float(max_r))
        hr0 = bh + rng.normal(0.0, s, n)
        tv = [20, 40, 60, 100, 150][r % 5]
        hr = _run_one(hr0, dx, tv, 100.0, n, 150)
        c = _eval_c5(hr, dx)
        if c < bc:
            bc, bh = c, hr.copy()

    # Phase 3: Shake (pair-swap local search)
    if time.time() - t0 < budget_s * 0.92:
        bh2, c2 = _shake(bh, rng, n, ts, trials=5000)
        if c2 < bc:
            bc, bh = c2, bh2.copy()
        # Re-optimize from shaken solution
        hr = _run_one(bh, dx, 80, 100.0, n, 200)
        c = _eval_c5(hr, dx)
        if c < bc:
            bc, bh = c, hr.copy()

    # Phase 4: High-temperature refinement (near-exact max)
    for t in [500, 1000, 2000]:
        if time.time() - t0 > budget_s * 0.96:
            break
        hr = _run_one(bh, dx, t, 500.0, n, 300)
        c = _eval_c5(hr, dx)
        if c < bc:
            bc, bh = c, hr.copy()

    # Final: ensure exact feasibility
    bh = _project(bh, ts)
    bc = _eval_c5(bh, dx)

    return (bh, bc, n)
