                lambda x: smooth_obj(x, alpha, n_points),
                h_cur, method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-14}
            )
            h_cand = fix_sum(res.x, target_sum)
            c5_new = compute_c5(h_cand)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cand.copy()
            h_cur = h_cand
        except Exception:
            pass

    # Phase 2: Hill climbing local search (pair swaps)
    h_ls = best_h.copy()
    best_ls = compute_c5(h_ls)
    max_local = int(budget_s * 0.75)
    it_count = 0
    while elapsed() < max_local:
        i = rng.integers(0, n_points)
        j = rng.integers(0, n_points)
        if i == j:
            continue
        delta = rng.uniform(0.03, 0.9)
        old_i, old_j = h_ls[i], h_ls[j]
        if rng.random() < 0.5:
            h_ls[i], h_ls[j] = old_i + delta, old_j - delta
        else:
            h_ls[i], h_ls[j] = old_i - delta, old_j + delta
        if h_ls[i] < -1e-9 or h_ls[i] > 1 + 1e-9 or h_ls[j] < -1e-9 or h_ls[j] > 1 + 1e-9:
            h_ls[i], h_ls[j] = old_i, old_j
            continue
        h_ls[i] = np.clip(h_ls[i], 0.0, 1.0)
        h_ls[j] = np.clip(h_ls[j], 0.0, 1.0)
        c5_new = compute_c5(h_ls)
        it_count += 1
        if c5_new < best_ls - 1e-12:
            best_ls = c5_new
            if best_ls < best_c5:
                best_c5 = best_ls
                best_h = h_ls.copy()
        else:
            h_ls[i], h_ls[j] = old_i, old_j

    # Phase 3: Simulated annealing for escape from local optima
    if elapsed() < budget_s * 0.95:
        h_sa = best_h.copy()
        best_sa = compute_c5(h_sa)
        T = 0.008
        T_min = 0.0003
        cooling = 0.9997
        while elapsed() < budget_s * 0.95:
            i = rng.integers(0, n_points)
            j = rng.integers(0, n_points)
            if i == j:
                continue
            delta = rng.uniform(0.05, 1.0)
            old_i, old_j = h_sa[i], h_sa[j]
            if rng.random() < 0.5:
                h_sa[i], h_sa[j] = old_i + delta, old_j - delta
            else:
                h_sa[i], h_sa[j] = old_i - delta, old_j + delta
            if h_sa[i] < 0 or h_sa[i] > 1 or h_sa[j] < 0 or h_sa[j] > 1:
                h_sa[i], h_sa[j] = old_i, old_j
                continue
            c5_new = compute_c5(h_sa)
            if c5_new < best_sa:
                best_sa = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_sa.copy()
            else:
                if rng.random() < np.exp(-(c5_new - best_sa) / T):
                    pass  # accept worse
                else:
                    h_sa[i], h_sa[j] = old_i, old_j
            T = max(T * cooling, T_min)

    # Final: ensure exact sum and bounds
    best_h = fix_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_points)
