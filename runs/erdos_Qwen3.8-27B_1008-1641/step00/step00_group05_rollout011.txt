import numpy as np
import time

def compute_c5(h, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def gradient_at_max(h, n):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    k_max = int(np.argmax(corr))
    s = k_max - (n - 1)
    # grad[t] = (1-h[t-s]) * I[0<=t-s<n] - h[t+s] * I[0<=t+s<n]
    grad = np.zeros(n)
    for t in range(n):
        if 0 <= t - s < n:
            grad[t] += (1.0 - h[t - s])
        if 0 <= t + s < n:
            grad[t] -= h[t + s]
    return grad, s

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    # Fix sum
    target = n / 2.0
    for _ in range(50):
        excess = np.sum(h) - target
        if abs(excess) < 1e-10:
            break
        if excess > 0:
            mask = h > 0
            if not np.any(mask):
                break
            h[mask] -= excess / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h < 1
            if not np.any(mask):
                break
            h[mask] += (-excess) / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def subsampled_evaluate(h):
    """Return (c5, argmax_shift) for the full correlation."""
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    dx = 2.0 / n
    return float(np.max(corr) * dx), int(np.argmax(corr))

    h_best, c5_best = subgradient_descent(h_try, n_try, dx_try, 3000, lr_init=0.2, temp=0.01)
    if time.time() - start_time > budget_s * 0.8:
        return h_best, c5_best / (c5_best if c5_best > 0 else 1)
    h_c, c5_c = coordinate_refine(h_best, n_try, dx_try, max_rounds=10, seed=seed)
    if c5_c < c5_best:
        h_best, c5_best = h_c, c5_c
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # Get initial solution
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        pass
    except Exception:
        pass

    if h0 is not None and len(h0) > 0:
        n = len(h0)
        h0 = project_feasible(h0, n)
    else:
        n = 70
        h0 = np.full(n, 0.5)
        # Add some structure: alternating blocks
        for i in range(n):
            h0[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / 14.0)
        h0 = project_feasible(h0, n)

    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Subgradient descent from initial
    if time.time() - start_time < budget_s * 0.35:
        h1, c5_1 = subgradient_descent(h0, n, dx, 8000, lr_init=0.4, temp=0.003)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Coordinate refinement on best
    if time.time() - start_time < budget_s * 0.6:
        h2, c5_2 = coordinate_refine(best_h, n, dx, max_rounds=40, seed=seed)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: More subgradient with lower temp (sharper)
    if time.time() - start_time < budget_s * 0.7:
        h3, c5_3 = subgradient_descent(best_h, n, dx, 6000, lr_init=0.2, temp=0.001)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        h4, c5_4 = coordinate_refine(best_h, n, dx, max_rounds=30, seed=seed + 1)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 4: Random restarts at different n
    for n_try in [70, 50, 84, 100]:
        if time.time() - start_time > budget_s * 0.88:
            break
        dx_try = 2.0 / n_try
        # Start from best_h interpolated
        if n_try != n:
            xs = np.linspace(0, 2, n)
            xt = np.linspace(0, 2, n_try)
            h_try = np.interp(xt, xs, best_h)
        else:
            h_try = best_h.copy()
        # Add small noise
        h_try += rng.normal(0, 0.05, n_try)
        h_try = project_feasible(h_try, n_try)
        h_r, c5_r = subgradient_descent(h_try, n_try, dx_try, 3000, lr_init=0.25, temp=0.005)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            if time.time() - start_time < budget_s * 0.92:
                h_cr, c5_cr = coordinate_refine(best_h, n_try, dx_try, max_rounds=20, seed=seed+n_try)
                if c5_cr < best_c5:
                    best_c5 = c5_cr
                    best_h = h_cr.copy()

    # Phase 5: Final polish on the best
    final_n = len(best_h)
    final_dx = 2.0 / final_n
    if time.time() - start_time < budget_s * 0.97:
        h_f, c5_f = coordinate_refine(best_h, final_n, final_dx, max_rounds=50, seed=seed + 99)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final verification
    h_final = project_feasible(best_h, final_n)
    final_c5 = compute_c5(h_final, 2.0 / final_n)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_final
    else:
        best_c5 = final_c5

    return best_h, best_c5, final_n
