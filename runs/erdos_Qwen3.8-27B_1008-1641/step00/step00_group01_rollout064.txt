import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corrs) * dx)

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corrs)
    exp_vals = np.exp(tau * (corrs - m))
    sum_exp = np.sum(exp_vals)
    val = m + np.log(sum_exp) / tau
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        d = idx - (n - 1)
        w = weights[idx]
        if w < 1e-14:
            continue
        if d >= 0:
            t1 = np.zeros(n)
            t1[d:] = 1.0 - h[:n - d]
            t2 = np.zeros(n)
            t2[:n - d] = h[d:]
        else:
            e = -d
            t1 = np.zeros(n)
            t1[:n - e] = 1.0 - h[e:]
            t2 = np.zeros(n)
            t2[e:] = h[:n - e]
        grad += w * (t1 - t2)
    grad *= dx
    return val, grad

def sum_eq_fun(x, n):
    return np.sum(x) - n / 2.0

def sum_eq_jac(x, n):
    return np.ones_like(x)

def project_feasible(h, n):

    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        diff = s - target
        order = np.argsort(-h)
        for i in range(n):
            idx = int(order[i])
            r = min(h[idx], diff)
            h[idx] -= r
            diff -= r
            if diff < 1e-13:
                break
    else:
        diff = target - s
        order = np.argsort(h)
        for i in range(n):
            idx = int(order[i])
            inc = min(1.0 - h[idx], diff)
            h[idx] += inc
            diff -= inc
            if diff < 1e-13:
                break
    return h

def projected_gradient_step(h, n, lr, max_lag_idx=None):
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    if max_lag_idx is None:
        max_lag_idx = int(np.argmax(corrs))
    d = max_lag_idx - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        t1 = np.zeros(n)
        t1[d:] = 1.0 - h[:n - d]
        t2 = np.zeros(n)
        t2[:n - d] = h[d:]
    else:
        e = -d
        t1 = np.zeros(n)
        t1[:n - e] = 1.0 - h[e:]
        t2 = np.zeros(n)
        t2[e:] = h[:n - e]
    grad = (t1 - t2) * dx
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, max_lag_idx

def slsqp_optimize(h, n, tau, maxiter, ftol):

        smooth_obj_and_grad, h, args=(tau,),
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0}
    )
    h_new = project_feasible(res.x, n)
    return h_new
