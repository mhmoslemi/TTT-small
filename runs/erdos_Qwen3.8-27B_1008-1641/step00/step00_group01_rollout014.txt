        if time.time() - t0 > budget_s * 0.35:
            break
        ht, c5t = _improve(best_h, n, alpha, 600, 1e-14)
        if ht is not None and c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: Multi-start perturbations
    n_trials = 40
    for i in range(n_trials):
        deadline = t0 + budget_s * 0.72
        if time.time() > deadline:
            break
        scale = 0.015 + 0.1 * (i / n_trials)
        h_pert = best_h + rng.randn(n) * scale
        h_start = repair(h_pert, n)
        alpha = 30.0 + 500.0 * rng.rand()
        ht, c5t = _improve(h_start, n, alpha, 300, 1e-12)
        if ht is not None and c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2b: Structured starts
    deadline = t0 + budget_s * 0.82
    for h_struct in make_structured(n, rng):
        if time.time() > deadline:
            break
        ht, c5t = _try_all(h_struct, n, rng, t0, deadline)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: Try nearby n values
    deadline = t0 + budget_s * 0.94
    n_best = len(best_h)
    for dn in [1, -1, 2, -2, 3, 5, 10, 20, 50]:
        n_try = n_best + dn
        if n_try < 20 or n_try > 500 or time.time() > deadline:
            continue
        h_interp = repair(interp_step(best_h, n_try), n_try)
        c5_i = compute_c5(h_interp)
        if c5_i < best_c5:
            best_h, best_c5 = h_interp, c5_i
        ht, c5t = _improve(h_interp, n_try, 300.0, 400, 1e-12)
        if ht is not None and c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Final polish with very high smoothness
    if time.time() < t0 + budget_s * 0.98:
        n_f = len(best_h)
        ht, c5t = _improve(best_h, n_f, 10000.0, 1000, 1e-15)
        if ht is not None and c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Verify feasibility
    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
