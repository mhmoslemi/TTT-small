    dx = 2.0 / n

def smooth_max_obj(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(vals - m))) / beta

def smooth_max_grad(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    w = np.exp(vals - m)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        shift = k - (n - 1)
        jp = np.arange(n) + shift
        jm = np.arange(n) - shift
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        t1[vp] = 1 - h[np.clip(jp, 0, n-1)[vp]]
        t2[vm] = h[np.clip(jm, 0, n-1)[vm]]
        grad += w[k] * (t1 - t2)

def optimize_h(h0, n, budget_for_this=100):
    h = h0.copy()
    # Ensure constraints
    h = np.clip(h, 0, 1)
    # Fix sum
    target = n / 2.0
    current_sum = np.sum(h)
    # Simple projection: scale
    if abs(current_sum - target) > 1e-10:
        h = h * (target / current_sum)
        h = np.clip(h, 0, 1)
        # Re-fix sum with water-filling
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            # Add/subtract from entries with room
            if diff > 0:
                room = 1 - h
                idx = np.argsort(-room)
                for i in idx:
                    add = min(room[i], diff)
                    h[i] += add
                    diff -= add
                    if diff < 1e-10:
                        break
            else:
                room = h
                idx = np.argsort(-room)
                for i in idx:
                    sub = min(room[i], -diff)
                    h[i] -= sub
                    diff += sub
                    if diff > -1e-10:
                        break
    
    def obj(x):
        return smooth_max_obj(x, beta=50.0)
    
    def grad(x):
        return smooth_max_grad(x, beta=50.0)

                   options={'maxiter': 200, 'ftol': 1e-10})
    
    if res.success or res.fun < obj(h):
        h = res.x
        h = np.clip(h, 0, 1)
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-10 and abs(diff) < 0.5:
        idx = np.argmax(1.0 - x) if diff > 0 else np.argmax(x)
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def smooth_c5(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(vals - m))) / beta

def smooth_c5_grad(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    w = np.exp(vals - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        shift = k - (n - 1)
        ips = np.arange(n) + shift
        ims = np.arange(n) - shift
        vp = (ips >= 0) & (ips < n)
        vm = (ims >= 0) & (ims < n)
        term1 = np.where(vp, 1.0 - h[np.clip(ips, 0, n - 1)], 0.0)
        term2 = np.where(vm, h[np.clip(ims, 0, n - 1)], 0.0)
        grad += wk * (term1 - term2)
    grad *= dx
    return grad

def optimize_slsqp(h0, n, max_iter=400, beta=50.0):
    h = project_constraint(h0.copy(), n)
    orig_val = smooth_c5(h, beta)

    def obj(x):
        return smooth_c5(x, beta)

    def grad(x):
        return smooth_c5_grad(x, beta)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        h_new = project_constraint(res.x, n)
        if smooth_c5(h_new, beta) < orig_val:
            return h_new
    except Exception:
        pass
    return h

def optimize_subgrad(h0, n, iterations=2000, lr_init=0.01):
    h = project_constraint(h0.copy(), n)
    best_copy = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(iterations):
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode="full")
        vals = corr * dx
        k_star = np.argmax(vals)
        shift = k_star - (n - 1)
        ips = np.arange(n) + shift
        ims = np.arange(n) - shift
        vp = (ips >= 0) & (ips < n)
        vm = (ims >= 0) & (ims < n)
        term1 = np.where(vp, 1.0 - h[np.clip(ips, 0, n - 1)], 0.0)
        term2 = np.where(vm, h[np.clip(ims, 0, n - 1)], 0.0)
        g = dx * (term1 - term2)
        h_new = project_constraint(h - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_copy = h_new.copy()
            h = h_new
            lr *= 1.02
        else:
            lr *= 0.9
            if lr < 1e-8:
                break
    return best_copy
