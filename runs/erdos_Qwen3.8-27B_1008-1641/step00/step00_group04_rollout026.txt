def c5_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def local_search(h, max_sweeps=500):
    n = len(h)
    h = h.copy().astype(float)
    cur = c5_score(h)
    for _ in range(max_sweeps):
        bd = 0.0
        bi = bj = -1
        oi = np.where(h > 0.5)[0]
        zi = np.where(h < 0.5)[0]
        for i in oi:
            for j in zi:
                h[i] = 0.0; h[j] = 1.0
                d = cur - c5_score(h)
                if d > bd:
                    bd = d; bi = i; bj = j
                h[i] = 1.0; h[j] = 0.0
        if bi >= 0:
            h[bi] = 0.0; h[bj] = 1.0
            cur = c5_score(h)
        else:
            break
    return h, cur

def sa_search(h, iters, rng):

    cur = c5_score(h)

        else:
            h[idx[:tgt - no]] = 1
    return h

    no = int(hb.sum()); tgt = n_dst // 2

        for k in np.argsort(hi)[::-1][:no - tgt]:
            hb[k] = 0
    elif no < tgt:
        for k in np.argsort(hi)[:tgt - no]:
            hb[k] = 1
