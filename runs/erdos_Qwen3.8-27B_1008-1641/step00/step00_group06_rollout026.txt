def compute_c5(h, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def adjust_sum(h, target):
    h = np.clip(h, 0, 1).astype(float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-13
    else:
        mask = h > 1e-13
    if mask.any():
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        h += diff / len(h)
        h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argmax(h if diff < 0 else 1 - h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

        elif h[i] > 0.25:

        else:

        ones.pop()
    while len(ones) < needed_ones:
        ones.add(zeros.pop())
    if half < 0:
        half = zeros.pop() if zeros else n // 2
    # Verify sum
    cur = len(ones) * 1.0 + 0.5
    if abs(cur - target) > 0.51:
        # Fallback to continuous
        return adjust_sum(h, target), compute_c5(h, dx)
    
    def make_h():
        hh = np.zeros(n)
        for i in ones:
            hh[i] = 1.0
        if half >= 0 and half not in ones:
            hh[half] = 0.5
            zeros.discard(half)
        return hh
    
    hv = make_h()
    hv = adjust_sum(hv, target)
    c5 = compute_c5(hv, dx)
    
    while time.time() < deadline:
        improved = False
        best_c5 = c5
        best_move = None
        # Try: swap one 1 with one 0
        for i in list(ones):
            for j in list(zeros):
                ones.discard(i)
                zeros.add(i)
                zeros.discard(j)
                ones.add(j)
                hv = make_h()
                hv = adjust_sum(hv, target)
                c = compute_c5(hv, dx)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_move = ('swap', i, j)
                # Undo
                ones.discard(j)
                zeros.add(j)
                zeros.discard(i)
                ones.add(i)
        # Try: move half
        old_half = half
        for j in list(zeros):
            half = j
            hv = make_h()
            hv = adjust_sum(hv, target)
            c = compute_c5(hv, dx)
            if c < best_c5 - 1e-14:
                best_c5 = c
                best_move = ('half', old_half, j)
        half = old_half
        # Apply best move
        if best_move is not None:
            if best_move[0] == 'swap':
                ones.discard(best_move[1])
                zeros.add(best_move[1])
                zeros.discard(best_move[2])
                ones.add(best_move[2])
            else:
                half = best_move[2]
                zeros.discard(best_move[2])
            hv = make_h()
            hv = adjust_sum(hv, target)
            c5 = best_c5
            improved = True
        if not improved:
            break
    return hv, c5
