import numpy as np
from scipy import optimize
import time


def project(x, n):
    x = np.clip(np.asarray(x, dtype=np.float64), 0.0, 1.0)
    t = 0.5 * n
    for _ in range(2000):
        d = x.sum() - t
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = x > 1e-14
            c = int(m.sum())
            if c == 0:
                break
            s = min(d / c, x[m].min() - 1e-14)
            if s <= 0:
                break
            x[m] -= s
        else:
            m = x < 1.0 - 1e-14
            c = int(m.sum())
            if c == 0:
                break
            s = min(-d / c, (1.0 - x[m]).min() - 1e-14)
            if s <= 0:
                break
            x[m] += s
    return x


def c5_of(h):
    return float(np.max(np.correlate(h, 1.0 - h, 'full'))) * 2.0 / len(h)


def run_es(x0, n, seed, t_deadline, sigma0=0.25, max_gen=400000):
    rng = np.random.RandomState(seed)
    lam = 4 * n + 4
    mu = lam // 2
    w = np.log(np.arange(mu, 0, -1, dtype=np.float64))
    w /= w.sum()
    mean = x0.copy()
    sigma = sigma0
    best = x0.copy()
    bf = c5_of(x0)
    kids = np.empty((lam, n))
    fits = np.empty(lam)
    for gen in range(max_gen):
        if time.time() > t_deadline:
            break
        for i in range(lam):
            if rng.random() < 0.02:
                kids[i] = project(rng.rand(n), n)
            else:
                kids[i] = project(mean + sigma * rng.randn(n), n)
            fits[i] = c5_of(kids[i])
        idx = np.argsort(fits)
        if fits[idx[0]] < bf:
            bf = fits[idx[0]]
            best = kids[idx[0]].copy()
        nm = np.zeros(n)
        for j in range(mu):
            nm += w[j] * kids[idx[j]]
        mean = project(nm, n)
        if gen % 40 == 39 and gen > 0:
            dv = np.array([np.linalg.norm(kids[idx[j]] - mean)
                           for j in range(min(10, mu))])
            med = float(np.median(dv))
            if med > 1e-8:
                sigma = max(1e-7, min(1.5, 0.75 * med))
    return best, bf


def run_nm(x0, n, t_deadline, step=0.004):
    def obj(x):
        return c5_of(project(x, n))
    simplex = [x0.copy()]
    for i in range(n):
        vi = x0.copy()
        vi[i] += step
        simplex.append(project(vi, n))
    sim = np.array(simplex)
    if time.time() > t_deadline:
        return x0.copy(), c5_of(x0)
    try:
        res = optimize.minimize(obj, x0.copy(), method='Nelder-Mead',
                                initial_simplex=sim,
                                options={'maxiter': 30000, 'fatol': 1e-14,
                                         'xatol': 1e-14})
        hr = project(res.x, n)
        return hr, c5_of(hr)
    except Exception:
        return x0.copy(), c5_of(x0)
