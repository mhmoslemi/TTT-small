from scipy.optimize import minimize

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_grad(h, n):
    """Compute C5 and subgradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    C5 = corr[max_idx] * dx
    
    # Subgradient: gradient of the active correlation
    # corr[k] = sum_i h[i+k-(n-1)] * (1-h[i]) for valid i
    # d corr[k] / d h[j] = (1-h[j+k-(n-1)]) (if j+k-(n-1) valid) 
    #                      + h[j] * (-1) (if j valid in the (1-h) term)
    # This is complex; let's just do finite differences or use the analytical form.
    
    # corr[k] = sum_{i: valid} h[i+k-(n-1)] * (1-h[i])
    # Let m = k - (n-1), so corr[k] = sum_i h[i+m] * (1-h[i])
    # d/dh[j] corr[k]:
    #   Term 1: if j = i+m for some valid i, contribution is (1-h[i]) = (1-h[j-m])
    #   Term 2: if j = i for some valid i, contribution is -h[i+m] = -h[j+m]
    # So d/dh[j] corr[k] = (1-h[j-m]) * [j+m valid] - h[j+m] * [j valid in range]
    
    return C5

import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n, rng):
    h = (h >= 0.5).astype(float)
    target = n // 2
    cnt = h.sum()
    if cnt > target:
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, size=int(cnt - target), replace=False)
        h[idx] = 0
    elif cnt < target:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(zeros, size=int(target - cnt), replace=False)
        h[idx] = 1
    return h

def local_search(h, n, max_rounds=10000, time_limit=15.0, rng=None):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    start = time.time()
    n_stall = 0
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            improved_inner = False
            for j in zeros:

                h[j] = 1.0
                c5 = compute_c5(h, n)
                h[i] = 1.0
                h[j] = 0.0
                if c5 < best_c5 - 1e-12:
                    h[i] = 0.0
                    h[j] = 1.0
                    best_c5 = c5
                    # Recompute ones and zeros
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    improved = True
                    improved_inner = True
                    break
            if improved_inner:
                continue
        
        if not improved:
            n_stall += 1
            if n_stall >= 3:
                break
        else:
            n_stall = 0
    
    return h, best_c5
