```python
        if 0 <= i - k < n and n_lo <= i - k <= n_hi:
            val += (1.0 - h[i - k])
        # Second term: -h[i+k] if i in [n_lo, n_hi]
        if 0 <= i + k < n and n_lo <= i <= n_hi:
            val -= h[i + k]
        grad[i] = dx * val
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def c5_grad_active(h, n):
    """Compute C5 and gradient of the active (max) constraint."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    n_lo = max(0, -k)
    n_hi = min(n - 1, n - 1 - k)
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        if 0 <= i - k < n and n_lo <= (i - k) <= n_hi:
            val += (1.0 - h[i - k])
        if 0 <= i + k < n and n_lo <= i <= n_hi:
            val -= h[i + k]
        grad[i] = dx * val
    return float(c[m_star]), grad

def c5_smooth_max(h, n, eps):
    """Smooth max of correlation*dx."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(c)
    return cmax + eps * np.log(np.sum(np.exp((c - cmax) / eps)))

def c5_smooth_grad(h, n, eps):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(c)
    p = np.exp((c - cmax) / eps)
    p /= p.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-12:
            continue
        k = m - (n - 1)
        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)
        for i in range(n):
            val = 0.0
            if 0 <= i - k < n and n_lo <= (i - k) <= n_hi:
                val += (1.0 - h[i - k])
            if 0 <= i + k < n and n_lo <= i <= n_hi:
                val -= h[i + k]
            grad[i] += p[m] * dx * val
    return grad

def slsqp_optimize(h0, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        h = project_feasible(x, n)
        return compute_c5(h, n)
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
    h_opt = project_feasible(res.x, n)
    return h_opt, compute_c5(h_opt, n)

def smooth_descent(h0, n, t_end, rng):
    h = project_feasible(h0, n.copy(), n)
    h = project_feasible(h0, n)
    eps = 0.05
    lr = 0.02
    t0 = time.time()
    best = compute_c5(h, n)
    best_h = h.copy()
    while time.time() - t0 < t_end:
        for _ in range(200):
            f = c5_smooth_max(h, n, eps)
            g = c5_smooth_grad(h, n, eps)
            h_new = h - lr * g
            h_new += h - project_feasible(h, n)  # small projection correction
            h = project_feasible(h_new, n)
            c5v = compute_c5(h, n)
            if c5v < best:
                best = c5v
                best_h = h.copy()
            eps = max(0.005, eps * 0.999)
            lr = max(0.001, lr * 0.9995)
    # Final SLSQP polish
    h_f, c5_f = slsqp_optimize(best_h, n, maxiter=200)
    if c5_f < best:
        best = c5_f
        best_h = h_f
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(47) * 0.5
    n = len(h0)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(h0, n)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: SLSQP from initial
    if remaining() > 20:
        h_new, c5_new = slsqp_optimize(best_h, n, maxiter=500)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # Phase 2: Smooth gradient descent
    if remaining() > 30:
        h_new, c5_new = smooth_descent(best_h, n, remaining() * 0.3, rng)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # Phase 3: Random restarts with SLSQP
    restarts = 15
    while remaining() > 15 and restarts > 0:
        restarts -= 1
        noise = rng.randn(n) * 0.08
        h_pert = project_feasible(best_h + noise, n)
        # Also try from a simple 0-1 pattern
        if restarts % 5 == 0:
            h_rand = np.zeros(n)
            half = n // 2
            idx = rng.choice(n, size=half, replace=False)
            h_rand[idx] = 1.0
            if n % 2 == 1:
                h_rand[rng.randint(n)] = 0.5
            h_pert = project_feasible(h_rand, n)
        h_new, c5_new = slsqp_optimize(h_pert, n, maxiter=300)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # Phase 4: Try a couple other n values
    for n_alt in [48, 50, 94]:
        if remaining() < 25:
            break
        # Simple starting pattern: alternating blocks
        h_alt = np.zeros(n_alt)
        h_alt[:n_alt // 2] = 1.0
        if n_alt % 2 == 1:
            h_alt[n_alt // 2] = 0.5
        h_alt = project_feasible(h_alt, n_alt)
        h_new, c5_new = slsqp_optimize(h_alt, n_alt, maxiter=400)
        if remaining() > 15:
            h_new2, c5_new2 = smooth_descent(h_new, n_alt, remaining() * 0.15, rng)
            if c5_new2 < c5_new:
                h_new, c5_new = h_new2, c5_new2
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new
            n = n_alt

    # Final SLSQP polish
    if remaining() > 5:
        h_final, c5_final = slsqp_optimize(best_h, n, maxiter=500)
        if c5_final < best_c5:
            best_c5, best_h = c5_final, h_final

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
```
