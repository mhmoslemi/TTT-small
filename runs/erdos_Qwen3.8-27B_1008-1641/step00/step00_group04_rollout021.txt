import numpy as np

from scipy.optimize import minimize

def project_to_constraint(h, target_sum):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def compute_c5(h):
    n = len(h)

def smooth_c5(h, tau=50.0):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m)) / len(corr)) / tau

        return np.ones(n)

        res = minimize(objective, h0, method='SLSQP', bounds=bounds,
