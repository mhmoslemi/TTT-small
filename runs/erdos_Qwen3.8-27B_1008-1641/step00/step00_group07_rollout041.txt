import numpy as np
from scipy.optimize import minimize
import time

def repair(h, target_sum):
    h = np.clip(h.astype(float), 0, 1)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-12
            if mask.sum() == 0:
                break
            h[mask] -= (-diff) / mask.sum()
        h = np.clip(h, 0, 1)
    return h

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def smooth_max(vals, t):
    m = np.max(vals)
    if t < 1e-6:
        return m
    return m + np.log(np.sum(np.exp(t * (vals - m)))) / t

def _objective_smo(h, dx, t, target_sum):
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return smooth_max(corr, t)

def _run_slqpgp(h0, dx, target_sum, t, maxiter, tol):
    n = len(h0)
    def obj(h):
        return _objective_smo(h, dx, t, target_sum)
    def cons_eq(h):
        return h.sum() - target_sum
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':cons_eq}],
                   options={'maxiter':maxiter,'ftol':tol})
    if res.success or (res.x is not None and np.all(np.isfinite(res.x))):
        return repair(res.x, target_sum)
    return h0

def _coord_descent(h, dx, target_sum, n_iter, rng, step_size):
    n = len(h)
    curr = h.copy()
    curr_c5 = c5_eval(curr)
    for it in range(n_iter):
        corr = overlaps(curr)
        hot_lags = np.where(corr > curr_c5 - 1e-6)[0]
        lag = hot_lags[rng.randint(len(hot_lags))] if len(hot_lags) > 0 else rng.randint(len(corr))
        actual_lag = lag - (n - 1)
        improved = False
        for _ in range(30):
            idx = rng.randint(n)
            for direction in [1, -1]:
                delta = direction * step_size * rng.uniform(0.5, 1.5)
                h_try = curr.copy()
                h_try[idx] = np.clip(h_try[idx] + delta, 0, 1)
                h_try = repair(h_try, target_sum)
                c5_try = c5_eval(h_try)
                if c5_try < curr_c5 - 1e-10:
                    curr = h_try
                    curr_c5 = c5_try
                    improved = True
                    break
        if not improved:
            step_size *= 0.7
            if step_size < 1e-5:
                break
    return curr, curr_c5

    best_c5 = c5_eval(best_h)
