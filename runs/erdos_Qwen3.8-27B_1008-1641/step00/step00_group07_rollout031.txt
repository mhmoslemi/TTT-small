    t0 = time.time()
    
    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        h0 = None
    
    n = len(h0) if h0 is not None else 80
    dx = 2.0 / n
    half = n // 2
    
    best = None
    best_c5 = np.inf
    
    # Evaluate initial
    if h0 is not None:
        c5 = _c5(h0, dx)
        if c5 < best_c5:
            best_c5 = c5
            best = h0.copy()
    
    # Phase 1: binary local search from initial
    if h0 is not None:
        hb = _to_bin(h0, half)
        bh, c5 = _search(hb, dx, t0 + budget_s * 0.25, 3.0)
        if c5 < best_c5:
            best_c5 = c5
            best = bh
    
    # Phase 2: structured initializations
    for sh in [_alt_bin(n, half), _block_bin(n,half,2), _block_bin(n,half,3),
               _block_bin(n,half,4), _block_bin(n,half,7), _rand_bin(n,half)]:
        if time.time() > t0 + budget_s * 0.65:
            break
        bh, c5 = _search(sh, dx, min(t0+budget_s*0.04, t0+budget_s*0.65), 3.0)
        if c5 < best_c5:
            best_c5 = c5
            best = bh
    
    # Phase 3: random restarts
    while time.time() < t0 + budget_s * 0.82:
        hr = _rand_bin(n, half)
        dl = min(time.time() + budget_s * 0.02, t0 + budget_s * 0.82)
        bh, c5 = _search(hr, dx, dl, 3.0)
        if c5 < best_c5:
            best_c5 = c5
            best = bh
    
    # Phase 4: polish
    if time.time() < t0 + budget_s * 0.92 and best is not None:
        hb = _to_bin(best, half)

        best_c5 = _c5(best, dx)
