    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_f(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    if t * mc < 500:
        return mc + np.log(np.sum(np.exp(t * corr - t * mc))) / t
    return mc

def smooth_g(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    tc = t * corr - t * mc
    tc[np.argmax(tc)] = 0  # numerical stability
    e = np.exp(tc)
    w = e / np.sum(e)
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)

        k = int(shifts[j])
        wj = w[j]

        # h[m] in second factor: m=i, need 0<=m<n, 0<=m+k<n

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_f(x, t):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    if t * mc < 500:
        shifted = t * corr - t * mc
        return mc + np.log(np.sum(np.exp(shifted))) / t
    return mc

def smooth_g(x, t):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    shifted = t * corr - t * mc
    e = np.exp(shifted)
    w = e / np.sum(e)
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for j in range(2 * n - 1):
        k = int(shifts[j])
        wj = w[j]
        if wj < 1e-14:
            continue
        ms = max(0, k)
        me = min(n, n + k)
        if me > ms:
            grad[ms:me] += wj * (1.0 - x[np.arange(ms, me) - k])
        ms2 = max(0, -k)
        me2 = min(n, n - k)
        if me2 > ms2:
            grad[ms2:me2] -= wj * x[np.arange(ms2, me2) + k]

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    shifted = t * corr - t * mc
    e = np.exp(shifted)
    e_sum = np.sum(e)
    w = e / e_sum
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for j in range(2 * n - 1):
        k = int(shifts[j])
        wj = w[j]
        if wj < 1e-14:
            continue
        ms = max(0, k)
        me = min(n, n + k)
        if me > ms:
            idx = np.arange(ms, me)
            grad[ms:me] += wj * (1.0 - x[idx - k])
        ms2 = max(0, -k)
        me2 = min(n, n - k)
        if me2 > ms2:
            idx = np.arange(ms2, me2)
            grad[ms2:me2] -= wj * x[idx + k]
    return dx * grad

def penalized_g(x, t, target, lam):

    return smooth_g(x, t) + 2.0 * lam * s * np.ones(len(x))

def project(x, target):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(500):
        d = target - np.sum(x)
        if abs(d) < 1e-14:

        if d > 0:
            m = x < 1.0
        else:

        cnt = np.sum(m)
        if cnt == 0:
            break
        x[m] += d / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def compute_c5(h):
