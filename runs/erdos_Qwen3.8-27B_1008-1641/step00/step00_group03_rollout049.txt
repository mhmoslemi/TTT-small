    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if not np.any(mask):
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if not np.any(mask):
                break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        nr = n - k
        g[:nr] += (1.0 - h[k:k+nr])  # h[i] term: d/dh[i] of h[i]*(1-h[i+k])
        g[k:] -= h[:nr]  # h[i+k] term: d/dh[j] of h[j-k]*(1-h[j])

        kk = -k
        nr = n - kk
        g[kk:] += (1.0 - h[:nr])  # h[i] term
        g[:-kk] -= h[kk:]  # h[i-kk] term
    return g

    h_init = project_sum(h_init, ts)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with decreasing temperature
    temps = [0.8, 0.5, 0.3, 0.2, 0.12, 0.07, 0.04, 0.02]
    for t in temps:
        if time.time() - t_start > 0.45 * budget_s:
            break
        h_new, c5_new = slqp_run(best_h, n, t, maxiter=250)
        if h_new is not None and c5_new < best_c5 - 1e-8:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Binary rounding + SA
    if time.time() - t_start < 0.75 * budget_s:
        h_bin, c5_bin = best_binary_round(best_h, n)
        if h_bin is not None and c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
        h_sa, c5_sa = binary_sa(best_h, n, max_iters=300000, time_limit=25)
        if c5_sa < best_c5:
            best_h = h_sa
            best_c5 = c5_sa

    # Phase 3: Random restarts with SLSQP
    for r in range(40):
        if time.time() - t_start > 0.88 * budget_s:
            break
        noise = np.random.uniform(-0.12, 0.12, n)
        hp = np.clip(best_h + noise, 0.0, 1.0)
        hp = project_sum(hp, ts)
        c5p = compute_c5(hp)
        if c5p < best_c5 + 0.008:
            h_new, c5_new = slqp_run(hp, n, 0.08, maxiter=180)
            if h_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                # Quick binary check
                if time.time() - t_start < 0.88 * budget_s:
                    hb, cb = best_binary_round(best_h, n)
                    if hb is not None and cb < best_c5:
                        best_h = hb
                        best_c5 = cb

    # Phase 4: Final binary SA polish if time remains
    if time.time() - t_start < 0.92 * budget_s:
        remaining = 0.92 * budget_s - (time.time() - t_start)
        if remaining > 5:
            h_final, c5_final = binary_sa(best_h, n,
                                          max_iters=int(remaining * 10000),
                                          time_limit=min(remaining * 0.9, 30))
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
