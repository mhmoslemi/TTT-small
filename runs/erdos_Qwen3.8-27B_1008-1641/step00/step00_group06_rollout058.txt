import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * corr.max())

def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if mask.sum() == 0:
            if diff > 0:
                mask = (h < 1.0 - 1e-13)
            else:
                mask = (h > 1e-13)
        if mask.sum() == 0:
            break
        h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
    return h

def subgrad_vec(h, k_idx):
    n = len(h)
    s = k_idx - (n - 1)
    grad = np.zeros(n)
    j1 = max(0, s)
    j2 = min(n - 1, n - 1 + s)
    if j1 <= j2:
        grad[j1:j2 + 1] += (1.0 - h[j1 - s:j2 - s + 1])
    j3 = max(0, -s)
    j4 = min(n - 1, n - 1 - s)
    if j3 <= j4:
        grad[j3:j4 + 1] -= h[j3 + s:j4 + s + 1]
    return grad
