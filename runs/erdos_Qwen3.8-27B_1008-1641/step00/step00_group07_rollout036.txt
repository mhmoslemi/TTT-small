```python
def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_val = corr.max()
    tol = max(1e-8, 1e-6 * max_val)
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)
    for m in active:
        k = m - n + 1
        # Term 1: (1-h[j+k]) for valid j+k
        # Term 2: -h[j-k] for valid j-k
        g1 = np.zeros(n)
        g2 = np.zeros(n)
        if k >= 0:
            # j+k valid when 0 <= j+k < n, i.e., 0 <= j < n-k
            g1[:n-k] = one_minus_h[k:]
        else:
            # j+k valid when 0 <= j+k < n, i.e., -k <= j < n
            g1[-k:] = one_minus_h[:n+k]
        
        if k >= 0:
            # j-k valid when 0 <= j-k < n, i.e., k <= j < n
            g2[k:] = h[:n-k]
        else:
            # j-k valid when 0 <= j-k < n, i.e., 0 <= j < n+k
            g2[:n+k] = h[-k:]
        
        grad += (g1 - g2)
    
    grad *= dx / len(active)
    return grad, max_val * dx
```

This is O(n * n_active) per step, fully vectorized. Very fast.

```python
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            return h
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == len(h):
            h += diff / len(h)
        elif free.sum() > 0:
            h[free] += diff / free.sum()

            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
```

```python
def compute_c5(h):

        if abs(diff) < 1e-14:

        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == len(h):
            h += diff / len(h)
        elif nf > 0:
            h[free] += diff / nf
        else:
            idx = np.argmin(h) if diff > 0 else np.argmax(h)
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
```
