def smooth_c5_analytical(h, temp, dx):

    s = e.sum()

    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)  # (2n-1,)
    
    for idx, lag in enumerate(lags):
        if w[idx] < 1e-15:
            continue

        # term1: (1-h[k-lag]) if 0<=k-lag<n
        # term2: -h[k+lag] if 0<=k+lag<n

        k_min1 = max(0, lag)
        k_max1 = min(n-1, lag + n - 1)
        if k_min1 <= k_max1:
            ks = np.arange(k_min1, k_max1 + 1)
            grad[ks] += w[idx] * (1.0 - h[ks - lag])

        k_min2 = max(0, -lag)
        k_max2 = min(n-1, n - 1 - lag)
        if k_min2 <= k_max2:
            ks = np.arange(k_min2, k_max2 + 1)
            grad[ks] -= w[idx] * h[ks + lag]
    
    grad *= dx

def thue_morse(n, k):
    """Generate Thue-Morse-like arrangement of k ones in n positions."""
    # Use the van der Corput / bit-reversal approach
    positions = []
    for i in range(n):
        # Count bits in i
        pos = 0
        x = i
        while x:
            pos += x & 1
            x >>= 1
        if pos % 2 == (k % 2):  # Rough alternation
            positions.append(i)
    # This won't be exact, let me just use a different approach
    # Use golden ratio distribution
    phi = (1 + 5**0.5) / 2

    # Map to positions
    result = np.zeros(n)
    for p in points:
        result[int(p * n)] = 1.0
    return result

                    excess = 0
                if abs(excess) < 1e-9:

            if abs(h_adj.sum() - target_sum) < 1e-6:
                yield h_adj

    h_mid = np.where(h > 0.5, 1.0, 0.0)
    h_mid = project_onto_feasible(h + 0.1 * np.sign(h_mid - 0.5), target_sum)
    yield h_mid
