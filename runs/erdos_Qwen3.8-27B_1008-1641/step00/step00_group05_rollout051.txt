import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def c5_and_grad(h):
    """Compute C5 and subgradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    c5 = dx * max_val
    
    # Subgradient: gradient of the max component
    grad = np.zeros(n)
    # Find active components (within tolerance of max)
    tol = 1e-4 * max_val if max_val > 0 else 1e-8
    active = np.where(np.abs(corr - max_val) <= tol)[0]
    
    n_active = len(active)
    for idx in active:
        lag = idx - (n - 1)
        if lag >= 0:
            # sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
            # grad wrt h[i+lag]: (1-h[i])
            # grad wrt h[i]: -h[i+lag]
            for i in range(n - lag):
                grad[i + lag] += (1.0 - h[i])
                grad[i] += (-h[i + lag])
        else:
            lag = -lag
            # sum_{i=lag}^{n-1} h[i] * (1-h[i-lag])
            # grad wrt h[i]: (1-h[i-lag])
            # grad wrt h[i-lag]: -h[i]
            for i in range(lag, n):
                grad[i] += (1.0 - h[i - lag])
                grad[i - lag] += (-h[i])
    
    grad = dx * grad / n_active
    return c5, grad

def project(h, n):
    """Project to feasible set: 0<=h<=1, sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    s = np.sum(h)
    # Water-filling style projection
    for _ in range(10):
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:

        if not np.any(mask):

        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0, 1)
        s = np.sum(h)
    # Final adjustment

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(np.max(corr))

def project_exact(h, n):
    target = n / 2.0
    lo, hi = -2.0 / n, 2.0 / n
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def c5_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    tol = 1e-3 * max(max_val, 1e-10)
    active = np.where(np.abs(corr - max_val) <= tol)[0]
    grad = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                grad[i + lag] += (1.0 - h[i])
                grad[i] -= h[i + lag]
        else:
            lag = -lag
            for i in range(lag, n):
                grad[i] += (1.0 - h[i - lag])
                grad[i - lag] -= h[i]
    return dx * grad / len(active)

def subgradient_descent(h, n, steps=5000, lr0=0.1):
    best = h.copy()
    best_c5 = compute_c5(h)
    for t in range(steps):
        g = c5_subgradient(h)
        lr = lr0 / (1.0 + t * 0.01)
        h_new = h - lr * g
        h_new = project_exact(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        h = h_new
        if t % 500 == 0:
            # Increase active set tolerance for diversity
            pass
    return best, best_c5

    """Local search for binary solutions: try all swaps."""
    h = h_binary.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    improved = True

        max_iters -= 1

            for j in zeros:

                h[j] = 1

                if c5_new < best_swap_c5 - 1e-12:

                    best_swap = (i, j)
                h[i] = 1
                h[j] = 0

            h[i] = 0
            h[j] = 1
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(np.max(corr))

def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2} using bisection."""
    target = n / 2.0
    lo, hi = -2.0 / n, 2.0 / n
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def c5_grad(h):
    """Subgradient of C5 at h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    tol = 1e-6 * max(mx, 1.0)
    active = np.abs(corr - mx) <= tol
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx + lag] += (1.0 - h[idx])
            grad[idx] -= h[idx + lag]
        else:
            lag = -lag
            idx = np.arange(lag, n)
            grad[idx] += (1.0 - h[idx - lag])
            grad[idx - lag] -= h[idx]
    grad *= (dx / np.sum(active))
    return grad

def projected_subgrad(h, n, steps=5000):
    """Projected subgradient descent."""
    best = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()

        lr = 0.1 / (1.0 + t * 0.005)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best = h_new.copy()
            best_c5 = c5_new
        h_curr = h_new
    return best, best_c5
