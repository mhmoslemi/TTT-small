    # Final: try one more continuous refinement from best
    if elapsed() < budget_s * 0.98:
        ch, cc5 = slsqp_run(best_h, n, time_limit=min(budget_s * 0.02, 20.0))
        if cc5 < best_c5:
            best_h, best_c5 = ch, cc5

    best_h = project_h(best_h, n, target)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
