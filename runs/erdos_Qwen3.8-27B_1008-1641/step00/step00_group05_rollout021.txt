import numpy as np
from scipy.optimize import minimize
import time

def _overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h):
    return float(np.max(_overlaps(h)))

def _project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return _project_box_sum(h, target)

def _smooth_obj(x, n, alpha):
    ov = _overlaps(x)
    m = ov.max()
    e = np.exp(alpha * (ov - m))
    return float(m + np.log(e.sum()) / alpha)

def _smooth_grad(x, n, alpha):
    h = x
    dx = 2.0 / n
    ov = _overlaps(h)
    m = ov.max()
    e = np.exp(alpha * (ov - m))
    p = e / e.sum()
    shifts = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for ki, s in enumerate(shifts):
        w = p[ki]
        if w < 1e-14:
            continue
        wdx = dx * w
        for j in range(n):
            t = 0.0
            js = j + s
            jm = j - s
            if 0 <= js < n:
                t += 1.0 - h[js]
            if 0 <= jm < n:
                t -= h[jm]
            grad[j] += wdx * t
    return grad

def _opt_scipy(h, n, alpha, max_iter=200):
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(_smooth_obj, h.copy(), jac=lambda x: _smooth_grad(x, n, alpha),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-11})
    return _project_box_sum(res.x, target)

def _bin_c5_fast(h):
    """Fast C5 for binary h using integer arithmetic where possible."""
    return _c5(h)

def _bin_local_search(h, budget, sweep_size=500):
    n = len(h)
    h = h.copy()
    t0 = time.time()
    best = _c5(h)
    best_h = h.copy()
    no_improve = 0
    while time.time() - t0 < budget and no_improve < 40:
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        if not ones or not zeros:
            break
        improved = False
        # Shuffle and try swaps
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones[:sweep_size]:
            for j in zeros[:sweep_size]:
                if time.time() - t0 > budget:
                    break
                h[i], h[j] = 0.0, 1.0
                c = _c5(h)
                if c < best - 1e-12:
                    best = c
                    best_h = h.copy()
                    improved = True
                    no_improve = 0
                    ones = list(np.where(h == 1.0)[0])

def _proj_grad_descent(h, n, budget, alpha=100):
    target = n / 2.0
    h = _project_box_sum(h.copy(), target)
    best_c5 = _c5(h)
    best_h = h.copy()

        h_new = h - lr * g
        h_new = _project_box_sum(h_new, target)

        if c_new < best_c5 - 1e-12:

            h = h_new

            h = h_new  # still move (for exploration)
            stall += 1
            if stall > 200:
                lr *= 0.5
                stall = 0
        if lr < 1e-6:
            break
    return best_h, best_c5
