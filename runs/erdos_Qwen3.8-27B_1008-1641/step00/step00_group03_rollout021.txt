    """Compute C5 = max_k correlate(h, 1-h, full)[k] * dx"""
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def compute_gradient(h, dx, shifts=None):
    """Compute subgradient of C5 w.r.t. h"""
    n = len(h)
    c = np.correlate(h, 1 - h, mode="full")
    c_normalized = c * dx
    
    if shifts is None:
        # Use top few shifts
        top_k = max(1, min(len(c), 5))
        idx = np.argsort(c_normalized)[-top_k:]
    else:
        idx = shifts
    
    grad = np.zeros(n)
    for k in idx:
        # k in full correlate: k=0 corresponds to shift 0, k<0 to negative shifts
        # In full mode, output has length 2n-1, indexed 0 to 2n-2
        # shift = k - (n-1) in terms of actual shift
        shift = k - (n - 1)
        weight = c_normalized[k]  # weight by how active this shift is
        
        for j in range(n):
            # gradient component: dx * [(1 - h[j-shift]) - h[j+shift]] (valid indices only)
            g = 0.0
            j_plus = j + shift
            j_minus = j - shift
            if 0 <= j_plus < n:
                g -= h[j_plus]
            if 0 <= j_minus < n:
                g += (1 - h[j_minus])
            grad[j] += weight * g
    
    # Normalize gradient by max weight to keep scale reasonable
    max_w = np.max(c_normalized[idx]) if len(idx) > 0 else 1.0
    if max_w > 0:
        grad /= max_w
    
    return grad

def project_onto_constraints(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0, 1)
    # Adjust to hit target sum
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if diff > 0:
        # Need to increase: add to elements with most room
        room = 1 - h
        order = np.argsort(-room)
        for i in order:
            if diff <= 0:
                break
            add = min(diff, room[i])
            h[i] += add
            diff -= add
    elif diff < 0:
        # Need to decrease: subtract from elements with most excess
        excess = h
        order = np.argsort(-excess)
        for i in order:
            if diff >= 0:
                break
            sub = min(-diff, excess[i])
            h[i] -= sub
            diff += sub
    
    return h

def subgradient_descent(h0, dx, n_evals_budget, n_top=3):

            best_h = h.copy()
