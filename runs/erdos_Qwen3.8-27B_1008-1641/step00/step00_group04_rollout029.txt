import numpy as np

def compute_corr(h):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return corr

def coordinate_step(h, i, n):
    # b_k = h[j(i,k)] for each lag k
    # j(i,k) = i + k - (n-1), valid when 0 <= j < n
    n_lags = 2*n - 1
    b = np.zeros(n_lags)
    a = np.zeros(n_lags)
    
    # Compute the "rest" part of a_k: sum_{p≠i} h[p]*h[j(p,k)]
    # For lag k, j(p,k) = p + k - (n-1)
    # a_k = (n/2) - sum_{p≠i} h[p]*h[p+k-(n-1)]
    # But sum_p h[p]*h[p+k-(n-1)] = corr_autocorr[k] (autocorrelation)
    # Actually, let me just compute it directly.
    
    # For each lag k, the valid range of p is:
    # 0 <= p < n and 0 <= p+k-(n-1) < n
    # => n-1 <= p+k < 2n-1 and 0 <= p < n
    # => max(0, n-1-k) <= p <= min(n-1, 2n-2-k)
    
    # b_k: j(i,k) = i+k-(n-1), valid if 0 <= i+k-(n-1) < n
    for k in range(n_lags):
        j = i + k - (n - 1)
        if 0 <= j < n:
            b[k] = h[j]
        p_lo = max(0, n-1-k)
        p_hi = min(n-1, 2*n-2-k)
        if p_lo <= p_hi:
            # Sum h[p]*h[p+k-(n-1)] for p in [p_lo, p_hi], p != i
            valid_p = np.arange(p_lo, p_hi+1)
            if i in valid_p:
                mask = valid_p != i
                products = h[valid_p]*h[valid_p+k-(n-1)]
                a[k] = n/2.0 - products[mask].sum()
            else:
                products = h[valid_p]*h[valid_p+k-(n-1)]
                a[k] = n/2.0 - products.sum()
        else:
            a[k] = n/2.0  # No valid terms, so rest = 0
    
    # Minimize max_k (a_k - t*b_k) over t in [0,1]
    # This is a 1D LP. Find the optimal t.
    # f(t) = max_k (a_k - t*b_k)
    # Breakpoints: t = (a_k - a_l) / (b_l - b_k) for all pairs
    # Also t = 0, t = 1
    
    # For efficiency with 83 lags, find the optimal t by checking candidate points.
    # The optimal t minimizes the upper envelope of lines.
    # Use binary search or direct enumeration of breakpoints.
    
    # Simpler: since f is convex and piecewise linear, the minimum is at a breakpoint or boundary.
    # Number of breakpoints is O(n_lags^2) = ~7000, manageable.
    
    candidates = set()
    candidates.add(0.0)
    candidates.add(1.0)

    for k in range(n_lags):

            continue
        for l in range(n_lags):
            if k == l or b[l] == 0:
                continue
            denom = b[l] - b[k]
            if abs(denom) > 1e-12:
                t = (a[k] - a[l]) / denom
                if 0 <= t <= 1:
                    candidates.add(t)
    
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        t = max(0.0, min(1.0, t))
        val = np.max(a - t * b)
        if val < best_val:
            best_val = val
            best_t = t

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def coord_step(h, i, n, autocorr):
    n_lags = 2 * n - 1
    lags = np.arange(n_lags)
    j = i + lags - (n - 1)
    b = np.zeros(n_lags)
    mask = (j >= 0) & (j < n)
    b[mask] = h[j[mask]]
    a = n * 0.5 - autocorr + h[i] * b
    
    if np.all(b < 1e-15):
        return h[i]
    
    active = np.where(b > 1e-15)[0]
    na = len(active)
    if na <= 1:
        return float(np.clip(h[i], 0, 1))
    
    # Generate all pairwise intersection candidates
    k_arr, l_arr = np.meshgrid(active, active, indexing='ij')
    k_flat = k_arr.ravel()
    l_flat = l_arr.ravel()
    valid = (k_flat != l_flat)
    k_flat = k_flat[valid]
    l_flat = l_flat[valid]
    denom = b[l_flat] - b[k_flat]
    valid2 = np.abs(denom) > 1e-15
    k_flat = k_flat[valid2]
    l_flat = l_flat[valid2]
    denom = denom[valid2]
    t_vals = (a[k_flat] - a[l_flat]) / denom
    in_range = (t_vals >= 0) & (t_vals <= 1)
    t_vals = t_vals[in_range]
    
    candidates = np.concatenate(([0.0, 1.0], t_vals))
    candidates = np.clip(candidates, 0.0, 1.0)
    
    matrix = a[None, :] - candidates[:, None] * b[None, :]
    vals = np.max(matrix, axis=1)
    best_idx = int(np.argmin(vals))
    return float(candidates[best_idx])

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def coord_step(h, i, n, autocorr):
    n_lags = 2 * n - 1
    lags = np.arange(n_lags)
    j = i + lags - (n - 1)
    b = np.zeros(n_lags)
    mask = (j >= 0) & (j < n)
    b[mask] = h[j[mask]]
    a = n * 0.5 - autocorr + h[i] * b
    if np.all(b < 1e-15):
        return h[i]
    active = np.where(b > 1e-15)[0]
    na = len(active)
    if na <= 1:
        return float(np.clip(h[i], 0, 1))
    k_a, l_a = np.meshgrid(active, active, indexing='ij')
    k_f = k_a.ravel()
    l_f = l_a.ravel()
    vp = k_f != l_f
    k_f = k_f[vp]
    l_f = l_f[vp]
    denom = b[l_f] - b[k_f]
    v2 = np.abs(denom) > 1e-15
    k_f = k_f[v2]
    l_f = l_f[v2]
    denom = denom[v2]
    t_vals = (a[k_f] - a[l_f]) / denom
    in_r = (t_vals >= 0) & (t_vals <= 1)
    t_vals = t_vals[in_r]
    candidates = np.concatenate(([0.0, 1.0], t_vals))
    candidates = np.clip(candidates, 0.0, 1.0)
    mat = a[None, :] - candidates[:, None] * b[None, :]
    vals = np.max(mat, axis=1)
    return float(candidates[int(np.argmin(vals))])

def optimize_cd(h, max_time):
    n = len(h)
    h = h.astype(np.float64).copy()
    start = time.time()
    best_obj = np.inf
    for _ in range(200):
        if time.time() - start > max_time:
            break
        autocorr = np.correlate(h, h, mode='full')
        obj_before = float(np.max(n * 0.5 - autocorr))
        changed = False
        for i in range(n):
            nv = coord_step(h, i, n, autocorr)
            if abs(nv - h[i]) > 1e-13:
                h[i] = nv
                changed = True
        autocorr = np.correlate(h, h, mode='full')
        obj_after = float(np.max(n * 0.5 - autocorr))
        if obj_after < best_obj:
            best_obj = obj_after
        if not changed:
            break
    return h

def binary_swap_search(h, max_time):
    n = len(h)
    h = (h > 0.5).astype(np.float64).copy()
    cnt = int(h.sum())
    target = n // 2
    if cnt > target:
        ones = np.where(h == 1)[0]
        ones = ones[np.argsort(h[ones])]
        for idx in ones[:cnt - target]:
            h[idx] = 0.0
    elif cnt < target:
        zeros = np.where(h == 0)[0]
        zeros = zeros[np.argsort(-h[zeros])]
        for idx in zeros[:target - cnt]:
            h[idx] = 1.0
    best_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < max_time:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if h[i] == h[j]:
                    continue
                h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)
    t0 = time.time()
    n = 42

    else:

    elapsed = lambda: time.time() - t0

    t_rem = budget_s - elapsed()

    return (best_h, final_c5, n)

def project_feasible(h, n, target_sum):
    h = np.clip(h.astype(np.float64), 0, 1)
    diff = target_sum - h.sum()
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # Fine adjustment: distribute residual to non-saturated entries
    residual = target_sum - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            room = residual / (len(order) + 1)
            amt = min(room, 1.0 - h[idx]) if residual > 0 else max(room, -h[idx])
            h[idx] += amt
            residual -= amt
            if abs(residual) < 1e-14:
                break
    return h

def make_starts(h_parent, h_best, n, seed):
    rng = np.random.RandomState(seed + 1)
    starts = []
    target_sum = n / 2.0
    # Start 1: parent + noise
    s1 = h_parent + rng.randn(n) * 0.12
    starts.append(project_feasible(s1, n, target_sum))
    # Start 2: random binary
    s2 = np.zeros(n)
    s2[rng.choice(n, int(target_sum), replace=False)] = 1.0
    starts.append(s2.copy())
    # Start 3: best + larger noise
    s3 = h_best + rng.randn(n) * 0.2
    starts.append(project_feasible(s3, n, target_sum))
    # Start 4: two-block structure
    s4 = np.zeros(n)
    s4[:n // 4] = 1.0
    s4[n // 2:3 * n // 4] = 1.0
    starts.append(project_feasible(s4, n, target_sum))
    # Start 5: golden ratio
    s5 = np.zeros(n)
    for i in range(int(target_sum)):
        pos = int((i * 1.618033988749895) % n) % n
        s5[pos] = 1.0
    cnt = int(s5.sum())
    if cnt > int(target_sum):
        ones = np.where(s5 == 1)[0][int(target_sum):]
        s5[ones] = 0
    starts.append(project_feasible(s5, n, target_sum))
    # Start 6: random continuous beta
    s6 = rng.beta(3, 3, n)
    starts.append(project_feasible(s6, n, target_sum))
    # Start 7: parent flipped
    s7 = 1.0 - h_parent
    starts.append(project_feasible(s7, n, target_sum))
    return starts
