    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]

    best_h = None
    best_c5 = float('inf')

    # Evaluate all starting points
    for h_s in starts:
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 1: Coordinated ascent on best start
    h_co, c5_co = coordinated_ascent(best_h.copy(), n, steps=200)
    if c5_co < best_c5:
        best_c5 = c5_co
        best_h = h_co.copy()

    # Phase 2: SLSQP from best
    elapsed = time.time() - t0
    if budget_s - elapsed > 20:
        h_opt = try_slsqp(best_h.copy(), n, obj_a, grad_a, bounds, cons, 300)
        if h_opt is not None:
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: Multiple perturbations + coordinated ascent + SLSQP
    elapsed = time.time() - t0
    attempt = 0
    while budget_s - elapsed > 50:
        scales = [0.06, 0.03, 0.015, 0.08]
        scale = scales[attempt % len(scales)]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n)

        h_ca, c5_ca = coordinated_ascent(h_pert, n, steps=80)
        candidate = h_ca if c5_ca < compute_c5(h_pert) else h_pert

        elapsed = time.time() - t0
        if budget_s - elapsed > 25:
            h_opt = try_slsqp(candidate, n, obj_b, grad_b, bounds, cons, 150)
            if h_opt is not None:
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    # Coordinated ascent on improved solution
                    h_co2, c5_co2 = coordinated_ascent(best_h.copy(), n, steps=100)
                    if c5_co2 < best_c5:
                        best_c5 = c5_co2
                        best_h = h_co2.copy()

        # Also try random starts occasionally
        if attempt % 4 == 3:
            h_rand = make_random_balanced(n, seed + attempt * 7)
            h_cr, c5_cr = coordinated_ascent(h_rand, n, steps=100)
            if c5_cr < best_c5:
                best_c5 = c5_cr
                best_h = h_cr.copy()

        attempt += 1
        elapsed = time.time() - t0

    # Phase 4: Fine-tune with small temperature
    elapsed = time.time() - t0
    if budget_s - elapsed > 25:
        h_opt = try_slsqp(best_h.copy(), n, obj_c, grad_c, bounds, cons, 200)
        if h_opt is not None:
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_co, c5_co = coordinated_ascent(best_h.copy(), n, steps=100)
                if c5_co < best_c5:
                    best_c5 = c5_co
                    best_h = h_co.copy()

    # Phase 5: Final coordinated ascent
    elapsed = time.time() - t0
    if budget_s - elapsed > 10:
        h_final, c5_final = coordinated_ascent(best_h.copy(), n, steps=200, step_size=0.01)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
