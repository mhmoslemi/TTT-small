    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        # Distribute to interior points
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, alpha, n):
    """One subgradient step for the max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    shift = k_star - (n - 1)
    
    g = np.zeros(n)
    # d/dh[j] corr[k_star] = (1-h[j-shift])*[valid(j-shift)] - h[j+shift]*[valid(j+shift)]

        if 0 <= j - shift < n:
            val += (1.0 - h[j - shift])
        if 0 <= j + shift < n:
            val -= h[j + shift]
        g[j] = val

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * corr.max()


def fix_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-7) & (h < 1.0 - 1e-7)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def subgradient_step(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    shift = k_star - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= j - shift < n:
            val += (1.0 - h[j - shift])
        if 0 <= j + shift < n:
            val -= h[j + shift]
        g[j] = val
    dx = 2.0 / n
    return h - alpha * dx * g


def obj_smooth(h, tau):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = corr * tau
    scaled -= scaled.max()
    m = scaled.max()
    logsum = np.log(np.exp(scaled).sum())
    return (logsum + m) / tau


def obj_direct(h):
    return compute_c5(h)
