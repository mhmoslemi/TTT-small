def project_feasible(h):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) < 1e-10:
        return h
    if diff > 0:  # need to decrease
        # decrease from largest values first (or proportionally)
        order = np.argsort(-h)
        remaining = diff
        for i in order:
            if remaining <= 1e-10:
                break
            to_decrease = min(remaining, h[i])
            h[i] -= to_decrease
            remaining -= to_decrease
    else:  # need to increase
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            if remaining <= 1e-10:
                break
            to_increase = min(remaining, 1.0 - h[i])
            h[i] += to_increase
            remaining -= to_increase
    return h

    n = len(h)
    target = n / 2.0
    
    def sum_clamped(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0))
    
    # Bisection on λ
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_clamped(mid) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

def coordinate_descent(h, max_iter=100):
    n = len(h)
    dx = 2.0 / n
    best = compute_c5(h)
    
    for iteration in range(max_iter):
        improved = False
        for i in range(n):
            # Try a range of values for h[i]
            old_val = h[i]
            # Compute optimal value via ternary search or grid
            best_v = old_val
            best_obj = best
            for trial in np.linspace(0, 1, 21):
                h[i] = trial
                h_proj = project_feasible(h)
                obj = compute_c5(h_proj)
                if obj < best_obj:
                    best_obj = obj
                    best_v = trial
            h[i] = best_v
            h = project_feasible(h)
            obj = compute_c5(h)
            if obj < best - 1e-10:
                best = obj
                improved = True
        if not improved:
            break
    return h, best

    #   Wait: second term: h[i+m]*(1-v) - h[i+m]*(1-h[i]) = h[i+m]*(h[i]-v) = -(v-h[i])*h[i+m]

    # corr at lag -m: sum_{j=m}^{n-1} h[j]*(1-h[j-m])
    # Replace h[i] with v:
    #   Term j=i: valid if m<=i<=n-1. Was h[i]*(1-h[i-m]), now v*(1-h[i-m]). Change: (v-h[i])*(1-h[i-m])
    #   Term j-m=i (j=i+m): valid if 0<=i<=n-1-m. Was h[i+m]*(1-h[i]), now h[i+m]*(1-v). Change: -(v-h[i])*h[i+m]

    # corr_new[k] = corr_old[k] + (v - h[i]) * alpha_k
    # where alpha_{n-1} is special (quadratic): corr_new[n-1] = corr_old[n-1] + v*(1-v) - h[i]*(1-h[i])
    # and for k != n-1: alpha_k = (1-h[i-m])*[i>=m] - h[i+m]*[i+m<n] where m = |k-(n-1)|

    # Currently sum of others = n/2 - h[i]. So I need to redistribute the difference.
    # This is where projection comes in.

    # For a better approach: don't enforce sum constraint in the 1D optimization, just find best v, then project.
    
    # Let's compute the objective as function of v (ignoring sum constraint for now):
    # f(v) = dx * max_k corr_new[k]

    # This is the max of (2n-2) affine functions and one concave quadratic in v.
    # The max is convex (max of affine is convex; adding a concave function to the max... hmm).
    # Actually, max of affine functions is convex, and the quadratic is concave. The max of a convex and concave function is not necessarily convex.
    # But we can still find the minimum by checking critical points.
    
    # For the affine part: f_affine(v) = max_k (corr_old[k] + (v-h[i]) * alpha_k) for k != n-1
    # This is piecewise linear, convex in v.
    # For the quadratic: f_quad(v) = corr_old[n-1] + v*(1-v) - h[i]*(1-h[i])
    # The overall f(v) = dx * max(f_affine(v), f_quad(v))
    
    # To minimize: find where f_affine(v) = f_quad(v) (breakpoint) and check minima of each piece.
    
    # This is doable but complex. Let me just do a fine grid search in [0,1] for v.

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    result = np.clip(h - lam, 0.0, 1.0)

    diff = result.sum() - target

        if diff > 0:
            idx = np.argmax(result)
        else:
            idx = np.argmin(result)
        result[idx] -= diff
