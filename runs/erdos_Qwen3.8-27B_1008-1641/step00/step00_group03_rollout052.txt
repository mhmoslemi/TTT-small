import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_and_active_lag(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    return float(corr[max_idx] * dx), max_idx, corr

def project_onto_constraint(v, n):
    target = n / 2.0
    # Binary search on shift t: h_i = clip(v_i - t, 0, 1), find t s.t. sum = target
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(v - t, 0.0, 1.0)
    # Fix rounding
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference to interior points
        interior = (h > 1e-6) & (h < 1 - 1e-6)
        if interior.any():
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, weights, n):
    """Compute weighted gradient of correlations.
    weights: array of length 2n-1 corresponding to the full correlation indices.
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = lag
            # overlap = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
            # d/dh[i] = (1-h[i+m]) for 0<=i<=n-1-m
            #           -h[i-m] for m<=i<=n-1
            grad[:n-m] += w * one_minus_h[m:]
            grad[m:] -= w * h[:n-m]
        else:
            p = -lag
            # overlap = sum_{i=p}^{n-1} h[i]*(1-h[i-p])
            # d/dh[i] = (1-h[i-p]) for p<=i<=n-1
            #           -h[i+p] for 0<=i<=n-1-p
            grad[p:] += w * one_minus_h[:n-p]
            grad[:n-p] -= w * h[p:]
    
    return grad * dx

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(v, n):
    target = n * 0.5
    lo, hi = -1.5, 2.5
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    h = np.clip(v - t, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.any():
            h[mask] += diff / mask.sum()
        elif n > 0:
            h[0] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_weighted(h, w, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = lag
            if m < n:
                grad[:n - m] += wk * omh[m:]
                grad[m:] -= wk * h[:n - m]
        else:
            p = -lag
            if p < n:
                grad[p:] += wk * omh[:n - p]
                grad[:n - p] -= wk * h[p:]
    return grad * dx

def smooth_weights(corr, tau):
    shifted = corr - np.max(corr)
    w = np.exp(shifted / tau)
    s = w.sum()
    if s > 1e-300:
        w /= s
    else:
        w[:] = 1.0 / len(w)
    return w

    return compute_c5(x)
