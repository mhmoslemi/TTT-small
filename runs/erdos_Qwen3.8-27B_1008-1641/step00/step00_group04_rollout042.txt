import numpy as np
import time
import math

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, n):
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c_sqrt_inv(C):
    evals, evecs = np.linalg.eigh(C)
    evals = np.maximum(evals, 1e-12)
    return evecs @ np.diag(1.0 / np.sqrt(evals)) @ evecs.T
