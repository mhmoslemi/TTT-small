```python
def compute_gradient(h, w):
    n = len(h)
    dx = 2.0 / n
    # grad[m] = dx * sum_k w[k] * dc[k]/dh[m]
    # dc[k]/dh[m] = (1-h[m-k+n-1]) if 0<=m-k+n-1<n - h[m+k-n+1] if 0<=m+k-n+1<n
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        d = k - n + 1
        # First term: (1-h[m-d]) for m where 0<=m-d<n
        # m-d in [0,n-1] means m in [d, d+n-1], intersect [0,n-1]
        lo1 = max(0, d)
        hi1 = min(n-1, d+n-1)
        if lo1 <= hi1:
            indices = np.arange(lo1, hi1+1)
            grad[indices] += w[k] * (1.0 - h[indices - d])
        # Second term: -h[m+d] for m where 0<=m+d<n
        # m+d in [0,n-1] means m in [-d, n-1-d], intersect [0,n-1]
        lo2 = max(0, -d)
        hi2 = min(n-1, n-1-d)
        if lo2 <= hi2:
            indices = np.arange(lo2, hi2+1)
            grad[indices] += w[k] * (-h[indices + d])
    grad *= dx
    return grad
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-8
        else:
            mask = h > 1e-8
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def pg_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - n + 1
        lo1 = max(0, d)
        hi1 = min(n - 1, d + n - 1)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wk * (1.0 - h[idx - d])
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] += -wk * h[idx + d]
    return grad * dx

def optimize_pg(h_init, n, temp_init=0.05, n_iter=3000, time_limit=120):
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    temp = temp_init
    lr = 0.02
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        exp_c = np.exp((corr - m) / temp)
        w = exp_c / exp_c.sum()
        grad = pg_gradient(h, w, dx)
        h = h - lr * grad
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Anneal
        if it % 200 == 199:
            temp *= 0.85
            lr *= 0.97
    return best_h, best_c5

    dx = 2.0 / n
    t0 = time.time()
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
             'jac': lambda h: np.ones(n)}]

        return compute_c5(h)
    
    result = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-9})
```

```python
            grad[idx] -= wk * h[idx + d]
    return grad * dx

def optimize_pg(h_init, n, temp_init=0.05, lr_init=0.02, n_iter=5000, time_limit=120):
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    temp = temp_init
    lr = lr_init
    t0 = time.time()
    no_improve = 0
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        exp_c = np.exp((corr - m) / temp)
        w = exp_c / exp_c.sum()
        grad = pg_gradient(h, w, dx, n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if it % 200 == 199:
            temp = max(0.003, temp * 0.82)
            lr *= 0.97
        if no_improve > 500:
            # Perturb to escape local min
            h = h + 0.02 * np.random.randn(n)
            h = project_feasible(h, n)
            no_improve = 0
    return best_h, best_c5

def optimize_slsqp(h_init, n, maxiter=300, time_limit=90):
    t0 = time.time()
    target = n / 2.0

    def obj_func(x):
        return compute_c5(x)

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    from scipy.optimize import minimize
    result = minimize(obj_func, h_init, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-9})
    return result.x, float(result.fun)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_h = None
    best_c5 = np.inf
    n0 = 70

    # Start from initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        if len(h0) == n0:
            h0 = project_feasible(h0, n0)
            c5_0 = compute_c5(h0)
            best_h = h0.copy()
            best_c5 = c5_0

    if best_h is None:
        h0 = np.full(n0, 0.5)
        angles = np.linspace(0, 4 * np.pi, n0)
        h0 += 0.15 * np.sin(angles) + 0.1 * np.cos(2 * angles)
        best_h = project_feasible(h0, n0)
        best_c5 = compute_c5(best_h)

    # Phase 1: PG with moderate temperature
    remaining = budget_s - (time.time() - t_start)
    if remaining > 60:
        h_opt, c5_opt = optimize_pg(best_h, n0, temp_init=0.08, lr_init=0.03,
                                     n_iter=4000, time_limit=min(150, remaining - 20))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: SLSQP refinement
    remaining = budget_s - (time.time() - t_start)
    if remaining > 60:
        try:
            h_opt, c5_opt = optimize_slsqp(best_h, n0, maxiter=400,
                                           time_limit=min(90, remaining - 20))
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
        except Exception:
            pass

    # Phase 3: PG with low temperature (fine-tuning)
    remaining = budget_s - (time.time() - t_start)
    if remaining > 40:
        h_opt, c5_opt = optimize_pg(best_h, n0, temp_init=0.01, lr_init=0.01,
                                     n_iter=4000, time_limit=min(100, remaining - 20))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Restart with different initializations
    remaining = budget_s - (time.time() - t_start)
    for trial in range(4):
        if remaining < 35:
            break
        h_rand = np.full(n0, 0.5)
        h_rand += np.random.uniform(-0.25, 0.25, n0)
        h_rand = project_feasible(h_rand, n0)
        try:
            h_opt, c5_opt = optimize_pg(h_rand, n0, temp_init=0.06, lr_init=0.025,
                                         n_iter=2000, time_limit=min(50, remaining - 10))
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
        except Exception:
            pass
        remaining = budget_s - (time.time() - t_start)

    # Phase 5: Try structured initializations
    remaining = budget_s - (time.time() - t_start)
    if remaining > 40:
        # Sawtooth pattern
        h_struct = (np.arange(n0) % 7) / 6.0
        h_struct = project_feasible(h_struct, n0)
        h_opt, c5_opt = optimize_pg(h_struct, n0, temp_init=0.04, lr_init=0.02,
                                     n_iter=2000, time_limit=min(40, remaining - 10))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        remaining = budget_s - (time.time() - t_start)

    # Phase 6: Try n=50 grid
    remaining = budget_s - (time.time() - t_start)
    if remaining > 50:
        h_interp = np.interp(np.linspace(0, 1, 50), np.linspace(0, 1, n0), best_h)
        h_interp = project_feasible(h_interp, 50)
        try:
            h_opt, c5_opt = optimize_pg(h_interp, 50, temp_init=0.05, lr_init=0.02,
                                         n_iter=3000, time_limit=min(50, remaining - 10))
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
        except Exception:
            pass

    # Final: ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
