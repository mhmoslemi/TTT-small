import time
import numpy as np
from scipy.optimize import minimize

_CFG = {}


def _project(v, target):
    x = np.asarray(v, dtype=float).copy()
    if x.size == 0:
        return x
    x = np.clip(x, 0.0, 1.0)
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(34):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        diff = target - x.sum()
        if abs(diff) <= 1e-12:
            break
        room = 1.0 - x if diff > 0 else x
        mask = room > 1e-12
        if not np.any(mask):
            break
        x[mask] += diff * room[mask] / room[mask].sum()
        x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13 and x.size:
        idx = np.where(x < 1.0)[0] if diff > 0 else np.where(x > 0.0)[0]
        if idx.size:
            x[idx[0]] = float(np.clip(x[idx[0]] + diff, 0.0, 1.0))
    return x


def _mirror(v, n):
    m = n // 2
    x = np.empty(n)
    x[:m] = v
    x[m:] = v[::-1]
    return x


def _c5_full(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return 1e9
    return float((2.0 / h.size) * np.max(np.correlate(h, 1.0 - h, mode="full")))


def _c5_v(v, n):
    m = n // 2
    x = np.empty(n)
    x[:m] = v
    x[m:] = v[::-1]
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    mid = n - 1
    cabs = 0.5 * (c[mid:] + c[:mid][::-1])
    return float(np.max(cabs))


def _smooth_value_v(v, n, beta):
    x = _mirror(v, n)
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    mid = n - 1
    cabs = 0.5 * (c[mid:] + c[:mid][::-1])
    cmax = float(np.max(cabs))
    z = np.exp(beta * (cabs - cmax))
    zsum = z[0] + 2.0 * z[1:].sum()
    return float(cmax + np.log(zsum) / beta)


def _grad_v(v, n, beta, top_k):
    x = _mirror(v, n)
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    mid = n - 1
    cabs = 0.5 * (c[mid:] + c[:mid][::-1])
    cmax = float(np.max(cabs))
    z = np.exp(beta * (cabs - cmax))
    zsum = z[0] + 2.0 * z[1:].sum()
    w = np.empty(n)
    w[0] = z[0] / zsum
    w[1:] = 2.0 * z[1:] / zsum
    if top_k <= 0 or top_k >= n:
        active = np.arange(n)
    else:
        active = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for d in active:
        wd = float(w[d])
        if wd <= 1e-18:
            continue
        if d == 0:
            g += wd * (1.0 - 2.0 * x)
        else:
            g[:n - d] += wd * (1.0 - x[d:])
            g[d:] += -wd * x[:-d]
    m = n // 2
    return g[:m] + g[n - m:]


def _slsqp_fun(v):
    cfg = _CFG
    return _smooth_value_v(v, cfg["n"], cfg["beta"])


def _slsqp_jac(v):
    cfg = _CFG
    return _grad_v(v, cfg["n"], cfg["beta"], cfg["top_k"])


def _slsqp_con(v):
    return np.array([np.sum(v) - _CFG["target"]])


def _slsqp_con_jac(v):
    return np.ones((1, v.size))


def _slsqp_refine(v, n, deadline, beta, maxiter, maxfun):
    target = n / 4.0
    v = _project(v, target)
    if time.time() + 5.0 > deadline:
        return v
    _CFG["n"] = n
    _CFG["beta"] = beta
    _CFG["target"] = target
    _CFG["top_k"] = min(60, max(1, n))
    bounds = [(0.0, 1.0)] * v.size
    cons = [{"type": "eq", "fun": _slsqp_con, "jac": _slsqp_con_jac}]
    try:
        res = minimize(
            _slsqp_fun,
            v,
            method="SLSQP",
            jac=_slsqp_jac,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "maxfun": maxfun, "ftol": 1e-12},
        )
    except Exception:
        return v
    x = np.asarray(res.x, dtype=float)
    if not np.all(np.isfinite(x)):

    x = _project(x, target)
    c1 = _c5_full(_mirror(x, n))
    c0 = _c5_full(_mirror(v, n))
    if c1 + 1e-14 < c0:

    return v

    best_c = c
    pairs = []
    for i in range(m):
        for j in range(i + 1, m):
            pairs.append((i, j))
    no_improve_cycles = 0
    while time.time() < deadline - 1.0:
        rng.shuffle(pairs)
        any_improve = False
        cnt = 0
        for i, j in pairs:
            if time.time() > deadline - 0.5:
                break
            if max_pairs is not None and cnt >= max_pairs:
                break
            cnt += 1
            a = v[i]
            b = v[j]
            smin = max(0.0, a + b - 1.0)
            smax = min(1.0, a + b)
            if smax - smin < 1e-12:
                continue
            step = (smax - smin) / max(1, coarse)
            vt = v.copy()
            best_s = 0.0
            best_pc = c
            for k in range(coarse + 1):
                s = smin + k * step
                vt[i] = a + s
                vt[j] = b - s
                pc = _c5_v(vt, n)
                if pc + 1e-15 < best_pc:
                    best_pc = pc
                    best_s = s
            if best_pc + 1e-13 < c:
                lo = max(smin, best_s - step)
                hi = min(smax, best_s + step)
                if hi > lo:
                    for k in range(refine):
                        s = lo + (hi - lo) * k / max(1, refine - 1)
                        vt[i] = a + s
                        vt[j] = b - s
                        pc = _c5_v(vt, n)
                        if pc + 1e-15 < best_pc:
                            best_pc = pc
                            best_s = s
                newi = a + best_s
                newj = b - (newi - a)
                if 0.0 <= newi <= 1.0 and 0.0 <= newj <= 1.0:
                    v[i] = newi
                    v[j] = float(np.clip(newj, 0.0, 1.0))
                    c = best_pc
                    any_improve = True
                    if c + 1e-14 < best_c:
                        best_c = c
                        best_v = v.copy()
        if any_improve:
            no_improve_cycles = 0
        else:
            no_improve_cycles += 1
            if no_improve_cycles >= 3:
                break
    best_v = _project(best_v, target)
    best_c = _c5_v(best_v, n)
    return best_v, best_c


def _sym_v_from_h(h, n):
    if n % 2 or n < 2:
        return None
    m = n // 2
    return _project(0.5 * (h[:m] + h[n - m:][::-1]), n / 4.0)


def _random_v(n, rng):
    m = n // 2
    i = np.arange(m)
    mode = int(rng.integers(0, 4))
    if mode == 0:
        v = 0.5 + 0.25 * np.tanh(1.2 * rng.normal(size=m))
    elif mode == 1:
        f = float(rng.integers(1, max(2, m // 3 + 1))) + rng.uniform(0.0, 0.9)
        v = 0.5 + 0.28 * np.sin(2.0 * np.pi * f * i / m + rng.uniform(0.0, 2.0 * np.pi))
    elif mode == 2:
        f1 = float(rng.integers(1, max(2, m // 3 + 1)))
        f2 = float(rng.integers(1, max(2, m // 2 + 1)))
        v = (
            0.5
            + 0.20 * np.sin(2.0 * np.pi * f1 * i / m)
            + 0.12 * np.sin(2.0 * np.pi * f2 * i / m + rng.uniform(0.0, 2.0 * np.pi))
        )
    else:
        v = 0.5 + 0.3 * (2.0 * rng.uniform(size=m) - 1.0)
    v = np.clip(v, 0.05, 0.95)
    return _project(v, n / 4.0)


def _upsample_sym(h, n2):
    n = h.size
    x0 = np.linspace(0.0, 2.0, n, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x1, x0, h, left=h[0], right=h[-1])
    m2 = n2 // 2
    v2 = _project(0.5 * (h2[:m2] + h2[n2 - m2:][::-1]), n2 / 4.0)
    return _mirror(v2, n2), v2


def run(seed=42, budget_s=1100, **kwargs):
    scale = float(budget_s)
    t0 = time.time()
    end = t0 + max(2.0, scale - 4.0)
    rng = np.random.default_rng(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).reshape(-1)
    except Exception:
        h0 = _mirror(_random_v(70, rng), 70)

    if h0.size < 4:
        h0 = _mirror(_random_v(70, rng), 70)
    if h0.size % 2:
        ne = h0.size + 1
        x0 = np.linspace(0.0, 2.0, h0.size, endpoint=False)
        x1 = np.linspace(0.0, 2.0, ne, endpoint=False)
        h0 = np.interp(x1, x0, h0, left=h0[0], right=h0[-1])
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project(h0, h0.size / 2.0)

    best_h = h0
    best_c = _c5_full(best_h)
    best_n = best_h.size
    best_v = _sym_v_from_h(best_h, best_n)
    if best_v is None:
        best_v = _random_v(best_n, rng)

    hs = _mirror(best_v, best_n)
    cs = _c5_full(hs)
    if cs + 1e-15 < best_c:
        best_h = hs.copy()
        best_c = cs

    d1 = t0 + min(0.22 * scale, end - t0 - 5.0)
    if time.time() < d1:
        v, _ = _projected_gradient_refine(best_v, best_n, 25.0, 0.08, 100000, d1, 20, rng)
        hh = _mirror(v, best_n)
        cc = _c5_full(hh)
        if cc + 1e-15 < best_c:
            best_h = hh.copy()
            best_c = cc
            best_v = v.copy()

    d2 = t0 + min(0.38 * scale, end - t0 - 5.0)
    if time.time() < d2:
        v, _ = _projected_gradient_refine(best_v, best_n, 90.0, 0.02, 100000, d2, 15, rng)
        hh = _mirror(v, best_n)
        cc = _c5_full(hh)
        if cc + 1e-15 < best_c:
            best_h = hh.copy()
            best_c = cc
            best_v = v.copy()

    d3 = t0 + min(0.48 * scale, end - t0 - 10.0)
    if time.time() + 10.0 < d3:
        v2 = _slsqp_refine(best_v, best_n, d3, 130.0, 220, 2500)
        hh = _mirror(v2, best_n)
        cc = _c5_full(hh)
        if cc + 1e-15 < best_c:
            best_h = hh.copy()
            best_c = cc
            best_v = v2.copy()

    d4 = t0 + min(0.68 * scale, end - t0 - 10.0)
    if time.time() + 10.0 < d4:
        v2, _ = _pair_refine_v(best_v, best_n, d4, 25, 15, rng, None)
        hh = _mirror(v2, best_n)
        cc = _c5_full(hh)
        if cc + 1e-15 < best_c:
            best_h = hh.copy()
            best_c = cc
            best_v = v2.copy()

    d5 = t0 + min(0.82 * scale, end - 160.0)
    attempts = 0
    while attempts < 5 and time.time() + 100.0 < d5:
        attempts += 1
        base = best_v
        rand = _random_v(best_n, rng)
        if base is None or base.size != rand.size:
            vv = rand
        else:
            vv = _project(0.55 * base + 0.45 * rand, best_n / 4.0)
        di = min(d5, time.time() + 70.0)
        vv, _ = _projected_gradient_refine(vv, best_n, 70.0, 0.05, 100000, di, 18, rng)
        dpr = min(di, time.time() + 25.0)
        vv, _ = _pair_refine_v(vv, best_n, dpr, 15, 11, rng, 80)
        hh = _mirror(vv, best_n)
        cc = _c5_full(hh)
        if cc + 1e-15 < best_c:
            best_h = hh.copy()
            best_c = cc
            best_v = vv.copy()

    if time.time() + 150.0 < end and best_n < 140:
        n_hi = 140
        h_hi, v_hi = _upsample_sym(best_h, n_hi)
        c_hi = _c5_full(h_hi)
        if c_hi + 1e-15 < best_c:
            best_h = h_hi.copy()
            best_c = c_hi
            best_v = v_hi.copy()
            best_n = n_hi
        dh = min(end - 120.0, time.time() + 120.0)
        v_hi, _ = _projected_gradient_refine(v_hi, n_hi, 80.0, 0.05, 100000, dh, 25, rng)
        hh = _mirror(v_hi, n_hi)
        cc = _c5_full(hh)
        if cc + 1e-15 < best_c:
            best_h = hh.copy()
            best_c = cc
            best_v = v_hi.copy()
            best_n = n_hi
        ds = min(end - 80.0, time.time() + 35.0)
        if time.time() + 10.0 < ds and best_n == n_hi:
            v_hi = _slsqp_refine(v_hi, n_hi, ds, 160.0, 140, 1500)
            hh = _mirror(v_hi, n_hi)
            cc = _c5_full(hh)
            if cc + 1e-15 < best_c:
                best_h = hh.copy()
                best_c = cc
                best_v = v_hi.copy()
                best_n = n_hi
        dp = min(end - 10.0, time.time() + 45.0)
        if time.time() + 10.0 < dp and best_n == n_hi:
            v_hi, _ = _pair_refine_v(v_hi, n_hi, dp, 17, 11, rng, 400)
            hh = _mirror(v_hi, n_hi)
            cc = _c5_full(hh)
            if cc + 1e-15 < best_c:
                best_h = hh.copy()
                best_c = cc
                best_v = v_hi.copy()
                best_n = n_hi

    if time.time() + 110.0 < end and best_n < 210 and scale > 900.0:
        n_hi = 210
        h_hi, v_hi = _upsample_sym(best_h, n_hi)
        c_hi = _c5_full(h_hi)
        if c_hi + 1e-15 < best_c:
            best_h = h_hi.copy()
            best_c = c_hi
            best_v = v_hi.copy()
            best_n = n_hi
        dh = min(end - 70.0, time.time() + 70.0)
        v_hi, _ = _projected_gradient_refine(v_hi, n_hi, 90.0, 0.04, 100000, dh, 25, rng)
        hh = _mirror(v_hi, n_hi)
        cc = _c5_full(hh)
        if cc + 1e-15 < best_c:
            best_h = hh.copy()
            best_c = cc
            best_v = v_hi.copy()
            best_n = n_hi

    if best_n % 2 == 0 and best_n > 0:
        target = int(best_n // 2)
        b = np.zeros(best_n)
        b[np.argsort(-best_h)[:target]] = 1.0
        cb = _c5_full(b)
        if cb + 1e-15 < best_c:
            best_h = b
            best_c = cb
        hc = 1.0 - best_h
        ch = _c5_full(hc)
        if ch + 1e-15 < best_c:
            best_h = hc
            best_c = ch

    h_out = best_h.copy()
    target = h_out.size / 2.0
    diff = target - h_out.sum()
    if abs(diff) > 1e-12:
        idx = np.where(h_out < 1.0)[0] if diff > 0 else np.where(h_out > 0.0)[0]
        if idx.size:
            h_out[idx[0]] = float(np.clip(h_out[idx[0]] + diff, 0.0, 1.0))
        if abs(h_out.sum() - target) > 1e-8:
            h_out = _project(h_out, target)

    c_out = _c5_full(h_out)
    if np.isfinite(c_out):
        best_c = float(c_out)

    return h_out, float(best_c), int(len(h_out))
