def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient_max_corr(h, n, top_k=5):
    """Compute gradient of max(corr) w.r.t. h, using top_k active lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find top_k largest lags
    sorted_idx = np.argsort(corr)[::-1]
    active_idx = sorted_idx[:top_k]
    
    grad = np.zeros(n)
    for p in active_idx:
        lag = p - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j - lag <= n - 1:
                g += 1.0 - h[j - lag]
            if 0 <= j + lag <= n - 1:
                g -= h[j + lag]
            grad[j] += g
    grad /= len(active_idx)
    return grad, max_val

def project_to_feasible(h, target_sum):
    """Project h to {x: 0≤x≤1, sum(x)=target_sum} using bisection on threshold."""
    n = len(h)
    def f(tau):
        return np.sum(np.clip(h - tau, 0, 1)) - target_sum
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    result = np.clip(h - tau, 0, 1)
    # Fine-tune to get exact sum
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute the difference
        mask = (result > 0) & (result < 1)
        if np.sum(mask) > 0:
            adj = diff / np.sum(mask)
            result[mask] += adj
            result = np.clip(result, 0, 1)
    return result

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_grad_max(h, n, top_k=3):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for p in idx:
        lag = p - (n - 1)
        # Vectorized:
        # d(corr)/dh[j] = (1-h[j-lag])*[0<=j-lag<n] - h[j+lag]*[0<=j+lag<n]
        j = np.arange(n)
        term1 = np.zeros(n)
        mask1 = (j - lag >= 0) & (j - lag < n)
        term1[mask1] = 1.0 - h[j[mask1] - lag]
        term2 = np.zeros(n)
        mask2 = (j + lag >= 0) & (j + lag < n)
        term2[mask2] = h[j[mask2] + lag]
        grad += (term1 - term2)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_grad_max(h, n, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    idx = np.argsort(corr)[::-1][:top_k]
    j = np.arange(n)
    grad = np.zeros(n)
    for p in idx:
        lag = int(p - (n - 1))
        m1 = (j - lag >= 0) & (j - lag < n)
        m2 = (j + lag >= 0) & (j + lag < n)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        t1[m1] = 1.0 - h[j[m1] - lag]
        t2[m2] = h[j[m2] + lag]
        grad += t1 - t2
    grad /= top_k
    return grad

def slsqp_refine(h_init, n, dx, target_sum, deadline):
