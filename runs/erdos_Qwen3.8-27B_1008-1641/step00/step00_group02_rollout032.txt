def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(y, n):
    S = 0.5 * n
    y = np.asarray(y, dtype=float).copy()
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(y - mid, 0.0, 1.0))) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def lse_obj(x, alpha, n):
    h = np.clip(x, 0.0, 1.0)
    pen = 200.0 * (float(h.sum()) - 0.5 * n) ** 2
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = float(c.max())
    return m + float(np.log(np.sum(np.exp(alpha * (c - m))))) / alpha + pen

def _obj_wrap(x, alpha, n):
    return lse_obj(x, alpha, n)

def nm_opt(h, alpha, n, maxiter):
    res = minimize(_obj_wrap, h.copy(), method='Nelder-Mead',
                   args=(alpha, n),
                   options={'maxiter': maxiter, 'xatol': 1e-10,
                            'fatol': 1e-13, 'adaptive': True})
    return project(res.x, n)

def cd_opt(h, n, rng, t_end):
    bs = h.copy()
    bc5 = compute_c5(bs)
    steps = [0.02, -0.02, 0.01, -0.01, 0.05, -0.05, 0.005, -0.005]
    it = 0
    while time.time() < t_end:
        it += 1
        i = int(rng.randint(n))
        old = float(bs[i])
        for dv in steps:
            v = old + dv
            if v < 0.0 or v > 1.0:
                continue
            bs[i] = v
            p = project(bs, n)
            c = compute_c5(p)
            if c < bc5:
                bs = p
                bc5 = c
                old = v
                break
            bs[i] = old
        if it % 300 == 0 and int(rng.rand() * 8) == 0:
            j = int(rng.randint(n))
            bs[j] = float(rng.rand())
            bs = project(bs, n)
    return bs, bc5

def cd_cyclic(h, n, t_end):
    bs = h.copy()
    bc5 = compute_c5(bs)
    steps = [0.015, -0.015, 0.008, -0.008, 0.03, -0.03]
    while time.time() < t_end:
        for i in range(n):
            if time.time() > t_end:
                break
            old = float(bs[i])
            for dv in steps:
                v = old + dv
                if v < 0.0 or v > 1.0:
                    continue
                bs[i] = v
                p = project(bs, n)
                c = compute_c5(p)
                if c < bc5:
                    bs = p
                    bc5 = c
                    old = v
                    break
                bs[i] = old
    return bs, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 45
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    except Exception:
        h0 = np.zeros(n)
        for i in range(n):
            h0[i] = 1.0 if i % 3 == 0 else 0.0

    best = project(h0, n)
    best_c5 = compute_c5(best)

    # Phase 1: Nelder-Mead with increasing sharpness of log-sum-exp
    alpha_iters = [(20.0, 3000), (50.0, 2500), (100.0, 2000)]
    for a, mi in alpha_iters:
        if time.time() - t0 > budget_s * 0.22:
            break
        hc = nm_opt(best, a, n, mi)
        cc = compute_c5(hc)
        if cc < best_c5:
            best, best_c5 = hc, cc

    # Phase 2: Random coordinate descent
    t_end = t0 + budget_s * 0.50
    if time.time() < t_end:
        best, best_c5 = cd_opt(best, n, rng, t_end)

    # Phase 3: Cyclic coordinate descent (systematic pass)
    t_end = min(t0 + budget_s * 0.62, time.time() + budget_s * 0.15)
    if time.time() < t_end:
        best, best_c5 = cd_cyclic(best, n, t_end)

    # Phase 4: NM restarts from perturbed incumbent
    for r in range(10):
        if time.time() - t0 > budget_s * 0.82:
            break
        hs = project(best + rng.randn(n) * 0.05, n)
        hc = nm_opt(hs, 60.0, n, 1500)
        cc = compute_c5(hc)
        if cc < best_c5:
            best, best_c5 = hc, cc
            te = min(t0 + budget_s * 0.85, time.time() + 15.0)
            if time.time() < te:
                best, best_c5 = cd_opt(best, n, rng, te)

    # Phase 5: Final polish – random CD
    te = t0 + budget_s * 0.93
    if time.time() < te:
        best, best_c5 = cd_opt(best, n, rng, te)

    # Phase 6: Final polish – cyclic CD
    te = min(t0 + budget_s * 0.97, time.time() + budget_s * 0.05)
    if time.time() < te:
        best, best_c5 = cd_cyclic(best, n, te)

    best = project(best, n)
    best_c5 = compute_c5(best)
    return best, best_c5, n
