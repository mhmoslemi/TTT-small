def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def c5_max_lag(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = int(np.argmax(c))
    return float(c[idx]), idx

def project_feasible(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1} via bisection on shift."""
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))

            lo = mid
        else:
            hi = mid

import numpy as np
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_smooth_max(h, n, temp):
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    m = np.max(c)
    ec = np.exp((c - m) / temp)
    w = ec / ec.sum()
    j = np.arange(n)
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-8:
            continue
        k = idx - (n - 1)
        jk = j - k
        jkp = j + k
        m1 = (jk >= 0) & (jk < n)
        m2 = (jkp >= 0) & (jkp < n)
        g = np.zeros(n)
        g[m1] += (1.0 - h[jk[m1]])
        g[m2] -= h[jkp[m2]]
        grad += w[idx] * g
    return grad * dx

    stall = 0

        if time.time() - t0 > time_limit:
            break
        c_val = compute_c5(h, dx)
        if c_val < best_c5 - 1e-13:
            best_c5 = c_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
