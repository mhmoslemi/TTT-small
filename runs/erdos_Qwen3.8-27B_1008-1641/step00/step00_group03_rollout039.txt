def project_feasible(v, n, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}"""
    # Bisection on shift
    lo, hi = -1.0 - np.max(np.abs(v)) + target_sum/n, 1.0 + np.max(np.abs(v))
    # Simple: lo = min(v) - 1, hi = max(v)

        proj = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # Fine adjustment for exact sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:

    h = np.clip(h, 0.0, 1.0)
    # Scale to get exact sum (tiny correction)
    h = h * target_sum / max(np.sum(h), 1e-15)
    h = np.clip(h, 0.0, 1.0)
    return h

    lo, hi = -1.0, 1.0

        else:
