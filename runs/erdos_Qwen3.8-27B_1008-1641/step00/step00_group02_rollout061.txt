To minimize the max, we want |S ∩ (S+lag)| to be as large as possible for all lags. This is the "maximum intersection" problem for a set of size m in {0,...,n-1}.

For a "concentrated" set like S = {0, 1, ..., m-1}, the intersections decay quickly.
For a "spread out" set, the intersections are more uniform.

The optimal structure for this is related to difference sets. For n = 2m, a (n, m, λ) difference set would give constant intersection λ for all non-zero shifts. But such sets don't generally exist for arbitrary m.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def dcordb_h(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        if n - L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
    else:
        Lp = -L
        if n - Lp > 0:
            g[:n - Lp] += (1.0 - h[Lp:])
            g[Lp:] -= h[:n - Lp]
    return g

def degrad_pnorm(h, p):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    g = raw * dx
    g = np.maximum(g, 1e-15)
    K = len(g)
    f = float(np.sum(g**p) / K)**(1.0 / p)
    if f < 1e-14:
        return np.zeros(n)
    scale = (f**(1.0 - p)) / K
    grad = np.zeros(n)
    for k in range(K):
        L = k - (n - 1)
        w = (g[k])**(p - 1) * scale
        if abs(w) < 1e-20:
            continue
        grad += w * dcordb_h(h, L)
    grad *= dx
    return grad

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def pnorm_descent(h, target, p, deadline, lr_init, max_iter, decay, decay_every):
    h_cur = h.copy()
    best = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    lr = lr_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = degrad_pnorm(h_cur, p)
        h_new = h_cur - lr * grad
        h_cur = project_sum(h_new, target)
        c5v = compute_c5(h_cur)
        if c5v < best_c5:
            best = h_cur.copy()
            best_c5 = c5v
        if (it + 1) % decay_every == 0:
            lr *= decay
    return best, best_c5

    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        rem = np.setdiff1d(np.arange(n), idx)
        h[rng.choice(rem, size=1)] = 0.5

    """Try random swaps (preserving sum) and keep improvements."""
    best = h.copy()
    best_c5 = compute_c5(h)

        # Simple: try setting h[i] and h[j] to 1- h[j] and 1-h[i] (swap values)
        h[j], h[i] = h[i], h[j]

        if c5v < best_c5:
            best = h.copy()
            best_c5 = c5v
        else:
            h[j], h[i] = h[i], h[j]  # undo
    return best, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def dcordb_h(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        if n - L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
    else:
        Lp = -L
        if n - Lp > 0:
            g[:n - Lp] += (1.0 - h[Lp:])
            g[Lp:] -= h[:n - Lp]
    return g

def degrad_pnorm(h, p):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    g = np.maximum(raw * dx, 1e-15)
    K = len(g)
    f = float(np.sum(g**p) / K)**(1.0 / p)
    if f < 1e-14:
        return np.zeros(n)
    scale = (f**(1.0 - p)) / K
    grad = np.zeros(n)
    for k in range(K):
        L = k - (n - 1)
        w = g[k]**(p - 1) * scale
        if abs(w) < 1e-22:
            continue
        grad += w * dcordb_h(h, L)
    grad *= dx
    return grad

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def pnorm_descent(h, target, p, deadline, lr_init, max_iter, decay, decay_every):
    h_cur = h.copy()
    best = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    lr = lr_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = degrad_pnorm(h_cur, p)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        cur_c5 = compute_c5(h_cur)
        improved = False
        for alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_try = project_sum(h_cur - alpha * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < cur_c5:
                h_cur = h_try
                cur_c5 = c5_try
                improved = True
                if c5_try < best_c5:
                    best = h_cur.copy()
                    best_c5 = c5_try
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-6:
                break
        elif (it + 1) % decay_every == 0:
            lr *= decay
    return best, best_c5

def make_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        rem = np.setdiff1d(np.arange(n), idx)
        h[rng.choice(rem, size=1)] = 0.5
    return h

def binarize(h, target):
    n = len(h)
    hb = np.where(h > 0.5, 1.0, 0.0)
    diff = target - float(np.sum(hb))
    if abs(diff) < 1e-10:
        return hb
    if diff > 0:
        zeros = np.where(hb == 0)[0]
        order = np.argsort(-h[zeros])[:int(np.ceil(diff))]
        hb[zeros[order]] = 1.0
    else:
        ones = np.where(hb == 1)[0]
        order = np.argsort(h[ones])[:int(np.ceil(-diff))]
        hb[ones[order]] = 0.0
    return hb
