def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def smooth_obj(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp(alpha * (corr - m)))
    return float((m + np.log(s) / alpha) * dx)

def proj_sum(h, target):
    h = h.astype(float).copy()
