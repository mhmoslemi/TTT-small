    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * corr.max()

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Ensure exact sum
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        if free.sum() > 0:
            result[free] += diff / free.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def subgradient_for_lag(h, m):
    """Gradient of the correlation at lag m (where m = k - (n-1))"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # Vectorized:
    # Contribution 1: (1-h[i+m]) for valid i
    valid1 = (np.arange(n) + m >= 0) & (np.arange(n) + m < n)
    indices1 = np.arange(n)[valid1]
    grad[indices1] += (1.0 - h[indices1 + m])
    # Contribution 2: -h[i-m] for valid i
    valid2 = (np.arange(n) - m >= 0) & (np.arange(n) - m < n)
    indices2 = np.arange(n)[valid2]
    grad[indices2] -= h[indices2 - m]
    return dx * grad

def get_subgradient(h, top_k=3):
    """Get subgradient as average of top-k active lags"""
    n = len(h)
    corr = compute_corr(h)
    # Get top k lags
    top_indices = np.argsort(corr)[::-1][:top_k]
    subgrad = np.zeros(n)
    for k in top_indices:
        m = k - (n - 1)
        subgrad += subgradient_for_lag(h, m)
    subgrad /= top_k
    return subgrad

def grad_corr_at_lag(h, m):
    n = len(h)
    idx = np.arange(n)
    grad = np.zeros(n)
    mask1 = (idx + m >= 0) & (idx + m < n)
    grad[mask1] += (1.0 - h[idx[mask1] + m])
    mask2 = (idx - m >= 0) & (idx - m < n)
    grad[mask2] -= h[idx[mask2] - m]
    return grad

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

        m = k - (n - 1)

    deltas = (0.005, 0.01, 0.02, 0.04, 0.06, 0.1, 0.15)
    for _ in range(rounds):
        imp = False
        for i in range(n):
            for j in range(n):
                if i == j: continue
                for d in deltas:
                    if h[i] < d or h[j] > 1 - d: continue
                    h2 = h.copy()
                    h2[i] -= d
                    h2[j] += d
                    v = compute_c5(h2)
                    if v < best_c5 - 1e-13:
                        h, best_c5, imp = h2, v, True
                        break
        if not imp: break
    return h, best_c5
