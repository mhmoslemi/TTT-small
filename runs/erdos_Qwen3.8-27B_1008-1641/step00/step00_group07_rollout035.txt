import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def corr_subgrad(h, k):
    n = len(h)
    sg = np.zeros(n)
    if k == 0:
        sg[:] = 1.0 - 2.0 * h
    elif k > 0:
        sg[k:n] += 1.0 - h[0:n-k]
        sg[0:n-k] -= h[k:n]
    else:
        kp = -k
        sg[0:n-kp] += 1.0 - h[kp:n]
        sg[kp:n] -= h[0:n-kp]
    return sg
