    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < (1.0 - 1e-15)
        else:

            break
        h[m] += diff / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, dx, target, max_iter, step_init, time_limit, t0):
    n = len(h)
    c5 = compute_c5(h, dx)
    step = step_init
    for it in range(max_iter):

            break
        if step < 1e-9:

                c5 = v

                break
