def fix_ones(h, n):
    target = n // 2
    cnt = int(np.sum(h > 0.5))
    if cnt > target:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:cnt - target]] = 0.0
    elif cnt < target:
        idx = np.where(h <= 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:target - cnt]] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def hill_climb(h, rounds=200):
    best = compute_c5(h)
    for _ in range(rounds):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bv, bm = best, None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                v = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
                if v < bv - 1e-14:
                    bv, bm = v, (i, j)
        if bm is None:
            break
        i, j = bm
        h[i], h[j] = 0.0, 1.0
        best = bv
    return h, best

    bh = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_ones(h, n):
    target = n // 2
    h = np.where(h >= 0.5, 1.0, 0.0).copy()
    cnt = int(np.sum(h))
    if cnt > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:cnt - target]] = 0.0
    elif cnt < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - cnt]] = 1.0
    return h

def hill_climb(h, rounds=200):
    best = compute_c5(h)
    for _ in range(rounds):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bv, bm = best, None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                v = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
                if v < bv - 1e-14:
                    bv, bm = v, (i, j)
        if bm is None:
            break
        i, j = bm
        h[i], h[j] = 0.0, 1.0
        best = bv
    return h, best

def sa_search(h, iters=5000, T0=0.005, T1=0.0003):
    h = h.copy()
    bc5, bh = compute_c5(h), h.copy()
    cc5 = bc5
    for it in range(iters):
        T = T0 * (T1 / T0) ** (it / max(iters - 1, 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nc5 = compute_c5(h)
        d = nc5 - cc5
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-15)):
            cc5 = nc5
            if nc5 < bc5:
                bc5, bh = nc5, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return bh, bc5

def resample(h_old, n_new):
    if len(h_old) == n_new:
        return h_old.copy()
    xo = np.arange(len(h_old)) / len(h_old)
    xn = np.arange(n_new) / n_new
    return np.interp(xn, xo, h_old)

def fix_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    order = np.argsort(np.abs(h - 0.5))
    for i in order:
        step = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += step
        diff -= step
        if abs(diff) < 1e-12:
            break
    return h
