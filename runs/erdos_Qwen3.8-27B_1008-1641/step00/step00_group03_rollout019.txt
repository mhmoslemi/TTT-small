            h_cur = best_h + rng.uniform(-0.25, 0.25, n)
        else:
            h_cur = rng.uniform(0.0, 1.0, n)
        h_cur = project_h(h_cur, n)
        h_new, c_new = do_slsqp(h_cur, n, maxiter=250)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new

    # Phase 5: 0-1 local search
    if time_left() > 20:
        h01 = (best_h >= 0.5).astype(float)
        n1 = int(np.sum(h01))
        tgt = n // 2
        if n1 > tgt:
            nz = np.where(h01 == 1.0)[0]
            order = np.argsort(best_h[nz])
            for idx in order[:n1 - tgt]:
                h01[nz[idx]] = 0.0
        elif n1 < tgt:
            nz = np.where(h01 == 0.0)[0]
            order = np.argsort(-best_h[nz])
            for idx in order[:tgt - n1]:
                h01[nz[idx]] = 1.0
        h01 = project_h(h01, n)
        c01 = compute_c5(h01)
        if c01 < best_c5:
            best_h, best_c5 = h01, c01

        if time_left() > 10:
            h01_new, c01_new = zero_one_local_search(h01, n, rng, max_iters=5000)
            if c01_new < best_c5:
                best_h, best_c5 = h01_new, c01_new

    # Phase 6: Final polish with soft SLSQP on best
    if time_left() > 30:
        h_new, c_new = do_soft_slsqp(best_h, n, maxiter=200)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new
        h_new, c_new = do_slsqp(best_h, n, maxiter=200)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new

    # Final validation
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
