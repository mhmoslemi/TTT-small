import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h[np.isnan(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def smooth_max_obj(h, dx, alpha, target_sum, lam):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    s = corr / alpha
    m = np.max(s)
    lse = m + np.log(np.sum(np.exp(s - m)))
    return alpha * lse + lam * (np.sum(h) - target_sum) ** 2

def binary_c5(h_bin, n):
    dx = 2.0 / n
    return compute_c5(h_bin.astype(float), dx)

    cur_c5 = best_c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T_init = 0.01
    T_final = 0.0005
    for it in range(n_iters):
        if time.time() > t_end:
            break
        frac = it / max(n_iters, 1)

        h[i], h[j] = 0, 1
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best, best_c5
