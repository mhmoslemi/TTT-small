import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h, dx):
    h = np.asarray(h, dtype=np.float64)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, target_sum):
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        rem = diff
        for i in order:
            if rem > 0:
                a = min(rem, 1.0 - h[i])
                h[i] += a
                rem -= a
            else:
                a = min(-rem, h[i])
                h[i] -= a
                rem += a
            if abs(rem) < 1e-13:
                break
    return h
