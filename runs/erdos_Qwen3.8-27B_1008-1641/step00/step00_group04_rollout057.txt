def cma_es.Evaluate(func, x0, sigma, n_points, budget, rng):
    n = len(x0)
    # Parameters
    lam = max(4, int(4 + 3*np.log(n)))  # population size
    mu = lam // 2
    # Weights
    w = np.log(mu + 0.5) - np.log(np.arange(1, mu+1))
    w = w / w.sum()
    mu_eff = 1.0 / (w @ w)
    c_c = (4 + mu_eff/n) / (n + 4 + 2*mu_eff/n)
    c_1 = 2 / (n + 1.3)**2 + 2*np.sqrt(mu_eff/n - 1)/(n + 2)
    c_mu = min(1 - c_1, 2 * (mu_eff - 2 + 1/mu_eff) / ((n + 2)**2 + mu_eff))
    d_sigma = 1 + 2*max(0, np.sqrt((mu_eff - 1)/(n + 1)) - 1) + c_sigma
    c_sigma = np.sqrt((mu_eff + 2)/(n + 2))
    chi_n = np.sqrt(n) * (1 - 1/(4*n) + 1/(21*n**2))
    
    # State
    mean = x0.copy().astype(float)
    C = np.eye(n)
    Cs = np.zeros((n, n))
    
    f_val = func(mean)
    best = mean.copy()
    best_val = f_val
    
    start_time = budget[0]  # time when started
    deadline = start_time + budget[1]  # deadline
    
    while time.time() < deadline:
        # Sample
        B, D = np.linalg.cholesky(C)
        Bt = B.T
        ys = []
        xs = []
        for k in range(lam):
            z = rng.standard_normal(n)
            y = sigma * (B @ z)  # Actually C^(1/2) @ z
            x = mean + y
            x = project_to_constraints(x, n_points)
            xs.append(x)
            ys.append(y)
        
        # Evaluate
        fitness = [(func(x), x, y) for x, y in zip(xs, ys)]
        fitness.sort(key=lambda t: t[0])
        
        # Update
        y_w = sum(w[i] * fitness[i][2] for i in range(mu))
        x_w = sum(w[i] * fitness[i][1] for i in range(mu))
        
        # Update sigma
        sigma *= np.exp((c_sigma / d_sigma) * (np.linalg.norm(y_w) / chi_n - 1))
        
        # Update mean
        mean = x_w
        
        # Update covariance
        diff = y_w / sigma
        Cs = (1 - mu_eff*c_1*c_mu)*Cs + c_mu * (np.outer(diff, diff) - C)
        C = (1 - c_1 - c_mu*min(1, np.trace(Cs)/n))*C + c_1 * np.outer(diff, diff)
        
        # Ensure C is positive definite
        eigvals, eigvecs = np.linalg.eigh(C)
        if eigvals.min() < 1e-10:
            C = eigvecs @ np.diag(np.maximum(eigvals, 1e-8)) @ eigvecs.T
        
        if fitness[0][0] < best_val:
            best_val = fitness[0][0]
            best = fitness[0][1].copy()
    
    return best, best_val

    best_f = c5_of(best, dx)
    sigma = 0.1
