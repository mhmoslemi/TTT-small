        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nf = float(free.sum())
        if nf < 1e-12:
            m = int(np.argmin(np.abs(h - 0.5)))
            h[m] = np.clip(h[m] + diff, 0.0, 1.0)
        else:
            h[free] += diff / nf
            np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(80, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    h = repair(h0.copy(), target)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    # Phase 1: Adaptive hill climbing with segment perturbations
    step = 0.12
    stuck = 0
    for it in range(600000):
        if time.time() - t0 > budget_s * 0.45:
            break
        k = int(rng.integers(1, max(2, n // 6) + 1))
        idx = rng.choice(n, size=k, replace=False)
        best_trial_c5 = best_c5
        best_trial_h = h.copy()
        for _ in range(3):
            h_t = h.copy()
            h_t[idx] += rng.uniform(-step, step, k)
            h_t = repair(h_t, target)
            c5 = compute_c5(h_t, dx)
            if c5 < best_trial_c5:
                best_trial_c5 = c5
                best_trial_h = h_t.copy()
        if best_trial_c5 < best_c5 - 1e-14:
            best_c5 = best_trial_c5
            best_h = best_trial_h.copy()
            h = best_trial_h.copy()
            stuck = 0
            step = min(step * 1.03, 0.35)
        else:
            stuck += 1
            if stuck > 500:
                step = max(step * 0.7, 0.01)
                stuck = 0
            elif stuck > 2000:
                # Diversification: blend with random
                h_rand = repair(rng.uniform(0, 1, n), target)
                h = repair(0.8 * best_h + 0.2 * h_rand, target)
                best_c5 = compute_c5(h, dx)
                best_h = h.copy()
                step = 0.1
                stuck = 0

    # Phase 2: Coordinate descent with adaptive step
    h = best_h.copy()
    cs = 0.04
    for sweep in range(100):
        if time.time() - t0 > budget_s * 0.72:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - t0 > budget_s * 0.72:
                break
            for d in (cs, -cs, 1.5 * cs, -1.5 * cs):
                h_t = h.copy()
                h_t[i] += d
                h_t = repair(h_t, target)
                c5 = compute_c5(h_t, dx)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    best_h = h_t.copy()
                    h = h_t.copy()
                    any_imp = True
                    break
        if not any_imp:
            cs *= 0.55
            if cs < 5e-5:
                break
        else:
            cs = min(cs * 1.1, 0.2)

    # Phase 3: Fine stochastic search with SA acceptance
    h = best_h.copy()
    T = 0.005
    for it in range(1000000):
        if time.time() - t0 > budget_s * 0.94:
            break
        k = int(rng.integers(1, 4))
        idx = rng.choice(n, size=k, replace=False)
        h_t = h.copy()
        h_t[idx] += rng.uniform(-T, T, k)
        h_t = repair(h_t, target)
        c5 = compute_c5(h_t, dx)
        if c5 <= best_c5:
            h = h_t.copy()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_t.copy()
        elif rng.random() < np.exp((best_c5 - c5) / max(T, 1e-8)):
            h = h_t.copy()
        T = max(T * 0.99995, 2e-4)

    # Phase 4: Final polish - try binarization-inspired moves
    h = best_h.copy()
    for _ in range(20):
        if time.time() - t0 > budget_s * 0.98:
            break
        h_t = h.copy()
        for i in range(n):
            if h_t[i] > 0.6:
                h_t[i] = 1.0
            elif h_t[i] < 0.4:
                h_t[i] = 0.0
        h_t = repair(h_t, target)
        c5 = compute_c5(h_t, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_t.copy()
            h = h_t.copy()
        else:
            break
        # Gradual: only threshold the most extreme values
        if time.time() - t0 > budget_s * 0.98:
            break
        h_t = h.copy()
        thresh = 0.5 + 0.3 * (1.0 / (_ + 1.0))
        for i in range(n):
            if h_t[i] > thresh:
                h_t[i] = min(1.0, h[t[i] if False else i] + 0.1)
            elif h_t[i] < (1.0 - thresh):
                h_t[i] = max(0.0, h[i] - 0.1)
        h_t = repair(h_t, target)
        c5 = compute_c5(h_t, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_t.copy()
            h = h_t.copy()

    # Final validation
    best_h = repair(best_h, target)
    best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, n
