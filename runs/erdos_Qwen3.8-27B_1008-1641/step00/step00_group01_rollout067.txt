    if c5 < _state['best_c5']:
        _state['best_c5'] = c5
        _state['best_h'] = h.copy()
    _state['eval_count'] += 1
    return c5

def _local_objective(x):
    n = _state['n']
    h = _repair(x, n)
    c5 = _compute_c5(h, n)
    if c5 < _state['best_c5']:
        _state['best_c5'] = c5
        _state['best_h'] = h.copy()
    return c5

def _de_callback(xk, convergence):
    if time.time() - _state['t0'] > _state['budget'] * 0.65:
        raise _Timeout()

def _run_de(bounds, n, seed, budget_s, t0):
    global _state
    _state['t0'] = t0
    _state['budget'] = budget_s
    try:
        differential_evolution(
            _de_objective, bounds, seed=seed,
            maxiter=500000, popsize=40,
            mutation=(0.5, 1.2), recombination=0.8,
            tol=None, polish=False, init='latinhypercube',
            callback=_de_callback
        )
    except _Timeout:
        pass
    except Exception:
        pass

def _run_local(x0, n, seed, budget_s, t0):
    global _state
    try:
        minimize(_local_objective, x0, method='L-BFGS-B',
                 bounds=[(0.0, 1.0)] * n,
                 options={'maxiter': 1000, 'maxfun': 10000})
    except Exception:
        pass
    if time.time() - t0 > budget_s - 15:
        return
    try:
        h = _state['best_h'].copy()
        minimize(_local_objective, h, method='SLSQP',
                 bounds=[(0.0, 1.0)] * n,
                 options={'maxiter': 300, 'iprint': 0})
    except Exception:
        pass

def _coordinate_descent(h, n, budget_s, t0, max_rounds=500):
    global _state
    best = _state['best_c5']
    dx = 2.0 / n
    target = n / 2.0
    for rnd in range(max_rounds):
        if time.time() - t0 > budget_s - 10:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > budget_s - 10:
                break
            for delta in [0.05, 0.1, 0.2, 0.5, -0.05, -0.1, -0.2, -0.5]:
                h2 = h.copy()
                h2[i] = np.clip(h[i] + delta, 0.0, 1.0)
                s = h2.sum()
                if abs(s - target) < 1e-12:
                    pass
                else:
                    j = next(j2 for j2 in range(n) if j2 != i and
                             (h2[j2] < 1.0 if s < target else h2[j2] > 0.0))
                    h2[j] += (target - s)
                    if h2[j] < 0.0 or h2[j] > 1.0:
                        continue
                c5 = _compute_c5(h2, n)
                if c5 < best - 1e-12:
                    best = c5
                    h = h2.copy()
                    improved = True
                    break
        _state['best_c5'] = best
        _state['best_h'] = h.copy()
        if not improved:
            break

def run(seed=42, budget_s=1100, **kwargs):
    global _state
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    except (NameError, Exception):
        n = 47
        h0 = np.full(n, 0.5)
    h0 = _repair(h0, n)

    _state = {
        'n': n, 'best_h': h0.copy(),
        'best_c5': _compute_c5(h0, n),
        'eval_count': 0, 't0': t0, 'budget': budget_s
    }

    bounds = [(0.0, 1.0)] * n

    # Phase 1: DE
    _run_de(bounds, n, seed, budget_s, t0)

    # Phase 2: Local optimization from best + perturbations
    rng = np.random.RandomState(seed + 100)
    for attempt in range(5):
        if time.time() - t0 > budget_s - 30:
            break
        x0 = _state['best_h'].copy()
        if attempt > 0:
            scale = 0.03 * attempt
            x0 = np.clip(x0 + scale * rng.randn(n), 0.0, 1.0)
            x0 = _repair(x0, n)
        _run_local(x0, n, seed + attempt, budget_s, t0)

    # Phase 3: Coordinate descent for fine tuning
    _coordinate_descent(_state['best_h'].copy(), n, budget_s, t0, max_rounds=1000)

    # Final repair and evaluation
    h_final = _repair(_state['best_h'], n)
    c5_final = _compute_c5(h_final, n)
    if c5_final < _state['best_c5']:
        _state['best_c5'] = c5_final
        _state['best_h'] = h_final.copy()

    h_final = np.clip(_state['best_h'], 0.0, 1.0)
    c5_final = _compute_c5(h_final, n)

    return (h_final, c5_final, n)
