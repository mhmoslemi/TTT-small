import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project_feasible(v, target_sum):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_obj_grad(h, beta, n, mu, dx):
    corr = compute_corr(h)
    shifted = beta * (corr - corr.max())
    ev = np.exp(shifted)
    p = ev / ev.sum()
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            d = k - (n - 1)
            if j + d >= 0 and j + d < n:
                s += p[k] * (1.0 - h[j + d])
            if j - d >= 0 and j - d < n:
                s -= p[k] * h[j - d]
        grad[j] = s / beta
    grad = grad * dx + 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def smooth_max_value(h, beta, n, mu, dx):
    corr = compute_corr(h)
    shifted = beta * (corr - corr.max())
    ev = np.exp(shifted)
    val = (corr.max() + np.log(ev.sum()) / beta) * dx
    val += mu * (h.sum() - n / 2.0) ** 2
    return val

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.zeros(45)
        h0[:22] = 1.0
        h0[22] = 0.5

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0

    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def remaining():
        return budget_s - (time.time() - t0)

    def lbfgsb_optimize(start_h, maxit=500):
        if remaining() < 20:
            return None
        results = []
        for beta in [15.0, 40.0, 100.0]:
            for mu in [50.0, 200.0]:
                if remaining() < 15:
                    break
                res = minimize(
                    lambda h: smooth_max_value(h, beta, n, mu, dx),
                    start_h,
                    jac=lambda h: smooth_max_obj_grad(h, beta, n, mu, dx),
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n,
                    options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10}
                )
                h_new = project_feasible(res.x, target_sum)
                c5_new = compute_c5(h_new)
                results.append((c5_new, h_new.copy()))
        results.sort(key=lambda x: x[0])
        return results[0] if results else None

    # Phase 1: L-BFGS-B from initial
    r = lbfgsb_optimize(h0)
    if r and r[0] < best_c5:
        best_c5, best_h = r
    if remaining() < 60:
        return best_h, best_c5, n

    # Phase 2: Multiple random restarts with L-BFGS-B
    for restart in range(25):
        if remaining() < 60:
            break
        scale = 0.08 + 0.04 * np.exp(-restart / 8.0)
        pert = rng.randn(n) * scale
        h_try = project_feasible(best_h + pert, target_sum)
        r = lbfgsb_optimize(h_try, maxit=300)
        if r and r[0] < best_c5:
            best_c5, best_h = r

    if remaining() > 40:

        def slsqp_obj(x):
            return x[n]

        def slsqp_jac(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        cons = []

            cons.append({'type': 'ineq', 'fun': cfun})
        cons.append({'type': 'eq', 'fun': lambda x: x[:n].sum() - target_sum})

        try:
            res = minimize(slsqp_obj, x0, method='SLSQP', jac=slsqp_jac,
                          constraints=cons, bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_feasible(res.x[:n], target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new
        except Exception:
            pass

    # Phase 4: Coordinate descent refinement
    if remaining() > 60:
        for iteration in range(3000):
            if remaining() < 30:
                break
            improved = False
            i = (iteration) % n
            j = (iteration // n) % n
            if i == j:
                continue
            old_i = best_h[i]
            old_j = best_h[j]

                nj = np.clip(old_j - (ni - old_i), 0.0, 1.0)

                best_h[i], best_h[j] = ni, nj
                c = compute_c5(best_h)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    improved = True
                    break
                best_h[i], best_h[j] = old_i, old_j
            if not improved and iteration % 500 == 0:
                # Try L-BFGS-B quick polish
                r = lbfgsb_optimize(best_h, maxit=100)
                if r and r[0] < best_c5:
                    best_c5, best_h = r

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
