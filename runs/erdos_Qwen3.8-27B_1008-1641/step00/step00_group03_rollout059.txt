        zeros = np.where(h < 0.5)[0]
        if n_ones_actual > n_ones:
            to_flip = rng.choice(ones, n_ones_actual - n_ones, replace=False)
            h[to_flip] = 0.0
        else:
            to_flip = rng.choice(zeros, n_ones - n_ones_actual, replace=False)
            h[to_flip] = 1.0
    return project_constraint(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t_start = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    best_h = project_constraint(h0, n).copy()
    best_c5 = compute_c5(best_h, dx)
    print(f"Initial: C5={best_c5:.6f}, n={n}")

    rng = np.random.default_rng(seed)

    # Phase 1: SLSQP on initial
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.15:
        h_s, c5_s = slsqp_opt(best_h, n, maxiter=400)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
            print(f"After SLSQP1: C5={best_c5:.6f}")

    # Phase 2: Multiple GD restarts with varied T and perturbations
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    n_restarts = max(4, min(16, int(remaining / 12)))
    T_schedules = [(5, 50), (20, 200), (50, 500), (10, 100), (100, 1000), (1, 300), (30, 800), (5, 300)]

    for i in range(n_restarts):
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed
        if remaining < 20:
            break
        if i == 0:
            h_init = best_h.copy()
            noise_scale = 0.0
        elif i == 1:
            h_init = project_constraint(1.0 - best_h, n)
            noise_scale = 0.0
        else:
            h_init = best_h + rng.standard_normal(n) * (0.08 + 0.03 * (i % 5))
            h_init = project_constraint(h_init, n)
            if i % 3 == 2:
                h_init = random_struct_init(n, rng)

        T0, T1 = T_schedules[i % len(T_schedules)]
        alloc = min(remaining / max(n_restarts - i, 1), 40.0)
        alloc = max(alloc, 5.0)
        h_new, c5_new = gd_smooth(h_init, dx, T0, T1, max_iter=400, time_budget=alloc)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            print(f"  GD {i} (T:{T0}->{T1}): C5={best_c5:.6f}")

    # Phase 3: SLSQP refinement
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 30:
        h_s, c5_s = slsqp_opt(best_h, n, maxiter=600)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
            print(f"After SLSQP2: C5={best_c5:.6f}")

    # Phase 4: High-T GD for sharpening
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 25:
        h_new, c5_new = gd_smooth(best_h, dx, 300, 2000, max_iter=600, time_budget=remaining * 0.5)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed
        if remaining > 15:
            h_s, c5_s = slsqp_opt(best_h, n, maxiter=400)
            if c5_s < best_c5:
                best_h, best_c5 = h_s, c5_s

    # Phase 5: Final micro-refinement with very high T
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 10:
        h_new, c5_new = gd_smooth(best_h, dx, 1000, 5000, max_iter=300, time_budget=remaining * 0.6)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = project_constraint(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    assert np.min(best_h) >= -1e-10 and np.max(best_h) <= 1.0 + 1e-10
    assert abs(np.sum(best_h) - n / 2.0) < 1e-8
    elapsed = time.time() - t_start
    print(f"Final: C5={best_c5:.6f}, n={n}, time={elapsed:.1f}s")
    return best_h, best_c5, n
