```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_grad_c5(h):
    """Compute subgradient of C5 at h (using the active lag)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    k_star = np.argmax(corr)
    # Gradient of corr[k_star] wrt h
    # corr[k] = sum_j h[j+k] * (1-h[j]) for valid j, j+k
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        if 0 <= m - k_star <= n - 1:
            g += (1 - h[m - k_star])
        if 0 <= m + k_star <= n - 1:
            g -= h[m + k_star]
        grad[m] = g
    return dx * grad

def project_feasible(h, n_points):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n_points / 2.0
    # Bisection for t such that sum(clip(h+t, 0, 1)) = target
    # f(t) = sum(clip(h+t, 0, 1)) is increasing in t
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) / 2.0
    result = np.clip(h + t, 0, 1)
    # Fix small numerical errors
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute the difference
        idx = np.argsort(result)[::-1] if diff > 0 else np.argsort(result)
        for i in idx:
            if diff > 0:
                add = min(1.0 - result[i], diff)
                result[i] += add
                diff -= add
            else:
                sub = min(result[i], -diff)
                result[i] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return result

def subgradient_optimize(h0, n_points, max_iter=5000, tol=1e-10):
    """Projected subgradient method for minimizing C5"""
    h = h0.copy()
    dx = 2.0 / n_points
```

```python
def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr_scaled(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_corr_k(h, k):
    n = len(h)
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        if m - k >= 0 and m - k < n:
            g += 1.0 - h[m - k]
        if m + k >= 0 and m + k < n:
            g -= h[m + k]
        grad[m] = g
    return grad

def project_feasible(h_val, n_points):
    target = n_points * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h_val + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    t = 0.5 * (lo + hi)
    result = np.clip(h_val + t, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += diff / cnt
    return np.clip(result, 0.0, 1.0)

def _smooth_c5(h, eps):
    corr = get_corr_scaled(h)
    m = np.max(corr)
    s = np.clip(eps * (corr - m), -500.0, 0.0)
    return m + np.log(np.exp(s).sum()) / eps

def _smooth_c5_grad(h, eps, penalty):
    corr = get_corr_scaled(h)
    m = np.max(corr)
    s = np.clip(eps * (corr - m), -500.0, 0.0)
    e = np.exp(s)
    w = e / e.sum()
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    top_idx = np.argsort(e)[-25:]
    for k in top_idx:
        if w[k] < 1e-10:
            continue
        grad += w[k] * grad_corr_k(h, k) * dx
    # Penalty gradient
    penalty_grad = 2.0 * penalty * (np.sum(h) - n * 0.5)
    grad += penalty_grad
    val = m + np.log(e.sum()) / eps + penalty * (np.sum(h) - n * 0.5) ** 2
    return val, grad

    return h_new, v_new
```

```python
def grad_corr_k(h, k):
    n = len(h)
    l = k - (n - 1)  # effective lag
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        ml = m + l
        if 0 <= ml < n:
            g += 1.0 - h[ml]
        mnl = m - l
        if 0 <= mnl < n:
            g -= h[mnl]
        grad[m] = g
    return grad
```
