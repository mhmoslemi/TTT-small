def project_feasible(h, target_sum):
    x = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(x)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        if not np.any(mask):
            break
        adj = diff / float(np.sum(mask))
        x[mask] = np.clip(x[mask] + adj, 0.0, 1.0)
    return x

def smooth_max(c, T):
    c_max = np.max(c)
    shifted = (c - c_max) / T
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    return c_max + T * np.log(s), exp_s / s

def _eq_fun(x, target):
    return np.sum(x) - target

def _eq_jac(x, target):
    return np.ones(len(x))

def _slsqp_obj(x, T, target):
    c = np.correlate(x, 1.0 - x, mode="full")
    val, _ = smooth_max(c, T)
    return val

def grad_smooth_max(x, T):
    n_loc = len(x)
    c = np.correlate(x, 1.0 - x, mode="full")
    _, w = smooth_max(c, T)
    grad = np.zeros(n_loc)
    for idx in range(len(c)):
        wi = w[idx]
        if wi < 1e-10:
            continue
        kp = idx - (n_loc - 1)
        if kp >= 0:
            grad[kp:] += wi * (1.0 - x[:n_loc - kp])
            grad[:n_loc - kp] -= wi * x[kp:]
        else:
            m = -kp
            grad[:n_loc - m] += wi * (1.0 - x[m:])
            grad[m:] -= wi * x[:n_loc - m]
    return grad

def binary_local_search(h, target, max_rounds=20, time_fn=None, budget=9999):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for r in range(max_rounds):
        if time_fn and time_fn() > budget:
            break
        improved = False
        best_delta = 0
        best_pair = None
        for i in range(n):
            for j in range(i + 1, n):
                if abs(best[i] - best[j]) < 1e-6:
                    continue
                test = best.copy()
                test[i], test[j] = test[j], test[i]
                c5_t = compute_c5(test)
                delta = c5_t - best_c5
                if delta < -1e-12:
                    best_delta = delta
                    best_pair = (i, j)
                    if delta < -0.001:
                        break
            if best_pair and best_delta < -0.001:
                break
        if best_pair:
            best[best_pair[0]], best[best_pair[1]] = best[best_pair[1]].copy(), best[best_pair[0]].copy()
            best_c5 += best_delta
            improved = True
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    import time
    t0 = time.time()
    np.random.seed(seed)
    n = 70
    target = float(n // 2)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (target,)}]

    try:
        h0 = np.array(initial_h_values, dtype=float)
        if len(h0) != n:
            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h0)), h0)
    except:
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP with decreasing temperature
    for T in [30.0, 15.0, 8.0, 4.0, 2.5, 1.5, 1.0, 0.5]:
        if time.time() - t0 > budget_s - 120:
            break
        res = minimize(_slsqp_obj, best_h, args=(T, target),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12})
        if res.x is not None:
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 2: Projected gradient with smooth max
    lr = 0.08
    no_improve = 0
    for it in range(30000):
        if time.time() - t0 > budget_s - 60:
            break
        g = grad_smooth_max(best_h, 0.3)
        x_new = best_h - lr * g
        x_new = project_feasible(x_new, target)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5 - 1e-14:
            best_h = x_new
            best_c5 = c5_new
            no_improve = 0
            if lr < 0.3:
                lr *= 1.02
        else:
            lr *= 0.85
            no_improve += 1
            if no_improve > 2000:
                break

    # Phase 3: Binary rounding + local search
    h_bin = (best_h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    diff_ones = int(target) - n_ones
    if diff_ones > 0:
        idx = np.where(h_bin == 0)[0]
        idx_sorted = idx[np.argsort(-best_h[idx])][:diff_ones]
        h_bin[idx_sorted] = 1.0
    elif diff_ones < 0:
        idx = np.where(h_bin == 1)[0]
        idx_sorted = idx[np.argsort(best_h[idx])][:(-diff_ones)]
        h_bin[idx_sorted] = 0.0

    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin

    # Best-pair binary local search
    h_ls, c5_ls = binary_local_search(
        (best_h >= 0.5).astype(float), target, max_rounds=30,
        time_fn=time.time, budget=t0 + budget_s - 20)
    if c5_ls < best_c5:
        best_h = h_ls
        best_c5 = c5_ls

    # Phase 4: Random restarts if time allows
    remaining = budget_s - (time.time() - t0)
    n_restarts = min(int(remaining / 80), 5)
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 30:
            break
        # Random 0-1 starting point
        h_rand = np.zeros(n)
        ones_idx = np.random.choice(n, int(target), replace=False)
        h_rand[ones_idx] = 1.0
        h_rand = project_feasible(h_rand, target)
        c5_rand = compute_c5(h_rand)
        if c5_rand < best_c5:
            best_h = h_rand.copy()
            best_c5 = c5_rand
        # Quick gradient on the random start
        lr2 = 0.05
        for it2 in range(2000):
            if time.time() - t0 > budget_s - 15:
                break
            g2 = grad_smooth_max(h_rand, 0.5)
            h_rand = project_feasible(h_rand - lr2 * g2, target)
            c5_2 = compute_c5(h_rand)
            if c5_2 < c5_rand - 1e-14:
                c5_rand = c5_2
            else:
                lr2 *= 0.8
        c5_rand = compute_c5(h_rand)
        h_bin2 = (h_rand >= 0.5).astype(float)
        n2_ones = int(np.sum(h_bin2))
        d2 = int(target) - n2_ones
        if d2 > 0:
            idx2 = np.where(h_bin2 == 0)[0]
            h_bin2[idx2[np.argsort(-h_rand[idx2])][:d2]] = 1.0
        elif d2 < 0:
            idx2 = np.where(h_bin2 == 1)[0]
            h_bin2[idx2[np.argsort(h_rand[idx2])][:(-d2)]] = 0.0
        c5_b2 = compute_c5(h_bin2)
        if c5_b2 < c5_rand:
            h_rand = h_bin2
            c5_rand = c5_b2
        if c5_rand < best_c5:
            best_h = h_rand.copy()
            best_c5 = c5_rand

    return (best_h, float(best_c5), n)
