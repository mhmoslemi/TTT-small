    for n in [n_init if n_init % 2 == 0 else n_init + 1,
              n_init + 2, n_init + 4, 50, 60, 64, 80, 90]:
        if time.time() - t0 > budget_s * 0.40:
            break
        n = max(n, 10)
        h_b, c5_b = binary_search(n, 4, t0, budget_s, seed + n)
        if h_b is not None and c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 3: Fourier parametric search
    for n in [n_init, n_init + 10, 60, 90, 120]:
        if time.time() - t0 > budget_s * 0.65:
            break
        for nf in [1, 2, 3, 4]:
            if time.time() - t0 > budget_s * 0.65:
                break
            h_f, c5_f = fourier_search(n, nf, t0, budget_s, seed + 1000 + n * 10 + nf)
            if h_f is not None and c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    # Phase 4: Random restarts at best n and nearby
    n_best = len(best_h)
    for n in [n_best, n_best + 2, n_best - 2 if n_best > 20 else n_best + 4]:
        if time.time() - t0 > budget_s * 0.80:
            break
        n = max(n, 10)
        h_r, c5_r = random_restart_search(n, 3, t0, budget_s, seed + 2000 + n)
        if h_r is not None and c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 5: Final polish
    if time.time() - t0 < budget_s * 0.90:
        n_f = len(best_h)
        h_pol, c5_pol = try_local(best_h, n_f, maxfev=100000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Ensure feasibility
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_out
