    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def compute_grad_single(h, k_idx):
    """Gradient of c[k_idx] * dx where k_idx is index in full correlation"""
    n = len(h)
    dx = 2.0 / n
    k = k_idx - (n - 1)  # actual lag
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        i1 = j - k
        if 0 <= i1 < n:
            g += (1.0 - h[i1])
        i2 = j + k
        if 0 <= i2 < n:
            g -= h[i2]
        grad[j] = g
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-14)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        for j in range(n):
            g = 0.0
            i1 = j - k
            if 0 <= i1 < n:
                g += (1.0 - h[i1])
            i2 = j + k
            if 0 <= i2 < n:
                g -= h[i2]
            grad[j] += g
    grad *= dx / len(active)
    return dx * max_val, grad

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -1.5, 2.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    if globals().get('initial_h_values') is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()

    h = project_feasible(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

            break

            best_c5 = c5_v

            t *= 0.4
