        h = np.clip(h, 0.0, 1.0)
    # Final exact fix on floating point
    residual = target - h.sum()
    if abs(residual) > 1e-15:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + residual)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def coord_descent(h, n, max_sweeps=5, tol=1e-10):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(max_sweeps):
        improved = False
        for i in range(n):
            orig = best[i]
            for trial in [orig - 0.05, orig - 0.02, orig - 0.005, orig + 0.005,
                          orig + 0.02, orig + 0.05, 0.0, 1.0]:
                trial = np.clip(trial, 0.0, 1.0)
                if trial == orig:
                    continue
                candidate = best.copy()
                candidate[i] = trial
                candidate = repair(candidate, n)
                c5 = compute_c5(candidate)
                if c5 < best_c5 - tol:
                    best = candidate
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 47
        h0 = np.zeros(n)
        h0[:23] = 1.0
        h0[23] = 0.5

    h0 = repair(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    nm_opts = {'maxiter': 4000, 'xatol': 1e-9, 'fatol': 1e-13, 'adaptive': True,
               'initial_simplex': None}

    def make_obj(nn):
        def _obj(x):
            return compute_c5(repair(x, nn))
        return _obj

    # Initial Nelder-Mead
    if time.time() - t0 < budget_s - 30:
        res = minimize(make_obj(n), h0, method='Nelder-Mead', options=nm_opts)
        h_nm = repair(res.x, n)
        c5_nm = compute_c5(h_nm)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm

    # Coordinate descent polish
    if time.time() - t0 < budget_s - 30:
        best_h, best_c5 = coord_descent(best_h, n, max_sweeps=3)

    # Multiple random restarts
    n_restarts = 50
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 120:
            break
        if i == 0:
            h_start = best_h + rng.randn(n) * 0.08
        elif i % 5 == 1:
            h_start = best_h + rng.randn(n) * 0.04
        elif i % 5 == 2:
            h_start = best_h.copy()
            k = rng.randint(2, max(3, n // 4))
            flip_idx = rng.choice(n, size=k, replace=False)
            h_start[flip_idx] = 1.0 - h_start[flip_idx]
        elif i % 5 == 3:
            h_start = best_h.copy()
            bs = rng.randint(3, min(12, n // 3))
            si = rng.randint(0, n - bs)
            h_start[si:si + bs] = rng.uniform(0, 1, bs)
        else:
            h_start = repair(rng.uniform(0, 1, n), n)

        h_start = repair(h_start, n)
        opts = nm_opts.copy()
        opts['maxiter'] = 2500
        res = minimize(make_obj(n), h_start, method='Nelder-Mead', options=opts)
        h_new = repair(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Coordinate descent on best
    if time.time() - t0 < budget_s - 60:
        best_h, best_c5 = coord_descent(best_h, n, max_sweeps=5)

    # Try nearby n values by interpolation
    for n_try in [46, 48, 49, 50, 51, 52, 54, 56, 60]:
        if time.time() - t0 > budget_s - 90:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h) + 1)
        x_new = np.linspace(0.0, 2.0, n_try + 1)
        h_interp = np.interp(x_new[:-1], x_old, best_h)
        h_interp = repair(h_interp, n_try)

        opts = nm_opts.copy()
        opts['maxiter'] = 2000
        res = minimize(make_obj(n_try), h_interp, method='Nelder-Mead', options=opts)
        h_new = repair(res.x, n_try)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            n = n_try
        # Polynomial refinement on this n
        if time.time() - t0 < budget_s - 60:
            best_h, best_c5 = coord_descent(best_h, len(best_h), max_sweeps=2)

    # Final polish
    if time.time() - t0 < budget_s - 10:
        best_h, best_c5 = coord_descent(best_h, len(best_h), max_sweeps=3)
        best_h = repair(best_h, len(best_h))
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
