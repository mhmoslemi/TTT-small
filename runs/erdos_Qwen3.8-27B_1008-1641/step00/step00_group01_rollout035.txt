        h_t = np.interp(x_new_p, x_old, best_h)
        h_t = project_constraint(h_t, target_t)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
        if time.time() - start_time < budget_s * 0.88:
            x_opt = _do_slsqp(best_h, target_t, 200, 300)
            c5_opt = compute_c5(x_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = x_opt

    # Phase 6: Random binary starts at even n
    if time.time() - start_time < budget_s * 0.75:
        for n_try in (n_init + 1, n_init + 3):
            if n_try % 2 != 0:
                continue
            target_t = n_try / 2.0
            for s in range(15):
                if time.time() - start_time > budget_s * 0.88:
                    break
                h_b = _binary_init(n_try, seed + 200 + s)
                x_opt = _do_slsqp(h_b, target_t, 200, 400)
                c5_opt = compute_c5(x_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = x_opt

    # Final: ensure feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_constraint(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_final)
