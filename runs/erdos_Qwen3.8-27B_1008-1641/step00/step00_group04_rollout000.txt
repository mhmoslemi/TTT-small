import time
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _project_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)


def _lag_values(h):
    n = int(len(h))
    off = n - 1
    lags = np.empty(2 * n - 1, dtype=float)
    one = 1.0 - h
    for d in range(-off, off + 1):
        t = d + off
        if d >= 0:
            m = n - d
            lags[t] = float(np.dot(h[:m], one[d:d + m]))
        else:
            m = -d
            lags[t] = float(np.dot(h[m:], one[:n - m]))
    return lags


def _lag_max(h):
    return float(_lag_values(h).max())


def score_array(h):
    n = int(len(h))
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _pair_coeffs(h, base, p, q):
    n = int(len(h))
    off = n - 1
    A = np.zeros(2 * n - 1, dtype=float)
    B = np.zeros(2 * n - 1, dtype=float)
    C = base.copy()

    a = float(h[p])
    b = float(h[q])
    s = a + b

    for j in range(n):
        t = j - p + off
        if j == p:
            old = a * (1.0 - a)
            C[t] -= old
            A[t] -= 1.0
            B[t] += 1.0
        elif j == q:
            old = a * (1.0 - b)
            C[t] -= old
            A[t] += 1.0
            B[t] += 1.0 - s
        else:
            r = float(h[j])
            C[t] -= a * (1.0 - r)
            B[t] += 1.0 - r

    for i in range(n):
        if i == p:
            continue
        t = p - i + off
        if i == q:
            old = b * (1.0 - a)
            C[t] -= old
            A[t] += 1.0
            B[t] -= s + 1.0
            C[t] += s
        else:
            r = float(h[i])
            C[t] -= r * (1.0 - a)
            B[t] -= r
            C[t] += r

    for j in range(n):
        t = j - q + off
        if j == q:
            old = b * (1.0 - b)
            C[t] -= old
            A[t] -= 1.0
            B[t] += 2.0 * s - 1.0
            C[t] += s - s * s
        elif j == p:
            continue
        else:
            r = float(h[j])
            C[t] -= b * (1.0 - r)
            B[t] += r - 1.0
            C[t] += (1.0 - r) * s

    for i in range(n):
        if i == q:
            continue
        t = q - i + off
        if i == p:
            continue
        else:
            r = float(h[i])
            C[t] -= r * (1.0 - b)
            B[t] += r
            C[t] += r * (1.0 - s)

    return A, B, C


def _eval_poly(A, B, C, xs):
    x = np.asarray(xs, dtype=float)[:, None]
    vals = C[None, :] + B[None, :] * x + A[None, :] * (x * x)
    return vals.max(axis=1)


def _line_search(A, B, C, lo, hi, x0, cur_max, grid, levels):
    lo = float(lo)
    hi = float(hi)
    if hi - lo < 1e-13:
        return float(x0), float(cur_max)

    x0 = float(np.clip(x0, lo, hi))
    grid = max(3, int(grid))
    levels = max(0, int(levels))

    xs = np.linspace(lo, hi, grid)
    xs = np.unique(np.concatenate((xs, np.array([x0]))))
    vals = _eval_poly(A, B, C, xs)

    best_idx = int(np.argmin(vals))
    best_val = float(vals[best_idx])
    best_x = float(xs[best_idx])

    if best_val < cur_max - 1e-14:
        arr = xs
        f = vals
        idx = best_idx
        current_best = best_val
        for _ in range(levels):
            m = len(arr)
            left = float(arr[idx - 1]) if idx > 0 else lo
            right = float(arr[idx + 1]) if idx + 1 < m else hi
            if right - left < 1e-13:
                break
            arr = np.linspace(left, right, grid)
            f = _eval_poly(A, B, C, arr)
            idx = int(np.argmin(f))
            current_best = float(f[idx])
        best_val = current_best
        best_x = float(arr[idx])

    return best_x, best_val


def optimize_pairs(h, deadline, rng=None, grid=None, levels=None):
    n = int(len(h))
    h = _project_sum(h, 0.5 * n)
    if n < 2:
        return h, score_array(h)

    if grid is None:
        grid = 65 if n <= 70 else 33
    if levels is None:
        levels = 3 if n <= 70 else 2

    best_h = h.copy()
    best_score = score_array(h)

    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    if not pairs:
        return best_h, best_score

    dx = 2.0 / n
    pass_counter = 0

    while time.time() < deadline:
        base = _lag_values(h)
        cur_max = float(base.max())
        cur_score = cur_max * dx
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_h = h.copy()

        if rng is not None:
            order = pairs[:]
            rng.shuffle(order)
        else:
            order = pairs

        improved_pass = 0.0
        updates = 0

        for k in range(len(order)):
            if (k % 32 == 0) and time.time() >= deadline:
                break
            p, q = order[k]
            s = h[p] + h[q]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                continue

            A, B, C = _pair_coeffs(h, base, p, q)
            bx, bv = _line_search(A, B, C, lo, hi, h[p], cur_max, grid, levels)

            if bv < cur_max - 1e-13:
                old = cur_max
                h[p] = bx
                h[q] = s - bx
                base = C + B * bx + A * bx * bx
                cur_max = float(base.max())
                updates += 1
                if old - cur_max > improved_pass:
                    improved_pass = old - cur_max
                if cur_max * dx < best_score - 1e-12:
                    best_score = cur_max * dx
                    best_h = h.copy()

                if updates % 32 == 0:
                    base = _lag_values(h)
                    cur_max = float(base.max())
                    if cur_max * dx < best_score - 1e-12:
                        best_score = cur_max * dx
                        best_h = h.copy()

        if improved_pass < 1e-12:
            break
        pass_counter += 1
        if pass_counter > 500:
            break

    best_h = _project_sum(best_h, 0.5 * len(best_h))
    return best_h, score_array(best_h)


def _lag_gradient(h, d):
    n = int(len(h))
    g = np.zeros(n, dtype=float)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        g[:m] += 1.0 - h[d:d + m]
        g[d:] += -h[:m]
    else:
        m = -d
        L = n - m
        g[m:] += 1.0 - h[:L]
        g[:L] += -h[m:]
    return g


def linearized_polish(h, deadline, active_tol=0.08):
    n = int(len(h))
    target = 0.5 * n
    h0 = _project_sum(h, target)
    best_h = h0.copy()
    best_T = _lag_max(h0)

    if linprog is None:
        return best_h, score_array(best_h)

    radii = []
    r = 0.25
    for _ in range(12):
        radii.append(r)
        r *= 0.5

    for radius in radii:
        if time.time() > deadline - 1.0:
            break
        radius_improved = False

        for _ in range(40):
            if time.time() > deadline - 1.0:
                break

            base = _lag_values(h0)
            T = float(base.max())
            if T < best_T - 1e-12:
                best_T = T
                best_h = h0.copy()

            active = base >= T - active_tol
            active_idx = np.flatnonzero(active)
            if active_idx.size == 0:
                break

            A_rows = []
            b_rows = []
            for t in active_idx:
                d = int(t - (n - 1))
                g = _lag_gradient(h0, d)
                c0 = float(base[t])
                rhs = c0 - float(np.dot(g, h0))
                row = np.zeros(n + 1, dtype=float)
                row[0] = -1.0
                row[1:] = g
                A_rows.append(row)
                b_rows.append(-rhs)

            if not A_rows:
                break

            A = np.asarray(A_rows, dtype=float)
            b = np.asarray(b_rows, dtype=float)
            Aeq = np.zeros((1, n + 1), dtype=float)
            Aeq[0, 1:] = 1.0
            beq = np.array([target], dtype=float)

            lb = np.maximum(0.0, h0 - radius)
            ub = np.minimum(1.0, h0 + radius)
            bounds = [(0.0, None)]
            for i in range(n):
                bounds.append((float(lb[i]), float(ub[i])))

            c = np.zeros(n + 1, dtype=float)
            c[0] = 1.0

            try:
                res = linprog(c, A, b, Aeq, beq, bounds=bounds)
            except Exception:
                break

            if res is None or not getattr(res, "success", False):
                break

            sol = np.asarray(res.x[1:], dtype=float)
            if not np.all(np.isfinite(sol)):
                break
            sol = _project_sum(sol, target)
            actual = _lag_max(sol)

            if actual < T - 1e-5:
                h0 = sol
                radius_improved = True
            else:
                break

    final_T = _lag_max(h0)
    if final_T < best_T:
        best_h = h0
    best_h = _project_sum(best_h, target)
    return best_h, score_array(best_h)


def combined_polish(h, time_limit, rng=None):
    h = _project_sum(h, 0.5 * len(h))
    t_end = time.time() + float(max(0.0, time_limit))

    d1 = min(t_end, time.time() + 0.45 * float(time_limit))
    h, sc = optimize_pairs(h, d1, rng)

    d2 = t_end - 2.0
    if d2 > time.time() + 1.0:
        h2, sc2 = linearized_polish(h, d2)
        if sc2 < sc:
            h, sc = h2, sc2

    d3 = t_end - 3.0
    if d3 > time.time() + 2.0:
        h3, sc3 = optimize_pairs(h, d3, rng)
        if sc3 < sc:
            h, sc = h3, sc3

    h = _project_sum(h, 0.5 * len(h))
    return h, score_array(h)


def _make_checker(n):
    h = np.zeros(int(n), dtype=float)
    h[0::2] = 1.0
    return _project_sum(h, 0.5 * int(n))


def _make_thue(n):
    n = int(n)
    h = np.zeros(n, dtype=float)
    for i in range(n):
        if (int(bin(i).count("1")) & 1) != 0:
            h[i] = 1.0

    if n % 2 == 0:
        m = n // 2
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        extra = int(len(ones)) - m
        if extra > 0:
            for i in range(extra):
                h[ones[i]] = 0.0
        elif extra < 0:
            for i in range(-extra):
                h[zeros[i]] = 1.0

    return _project_sum(h, 0.5 * n)


def _make_random_balanced(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    target = n // 2
    if target > 0:
        idx = rng.permutation(n)[:target]
        h[idx] = 1.0
    return _project_sum(h, 0.5 * n)


def _upsample(h, n):
    m = int(len(h))
    n = int(n)
    if m == 1:
        h2 = np.full(n, float(h[0]), dtype=float)
    else:
        old = np.linspace(0.0, 1.0, m)
        new = np.linspace(0.0, 1.0, n)
        h2 = np.interp(new, old, h)
    return _project_sum(h2, 0.5 * n)

    primary = min(0.5 * budget_s, 450.0, remain)
    if primary > 10.0:
        cand, sc = combined_polish(best_h, primary, rng)
        if sc < best_score:
            best_h, best_score = cand, sc

    noises = (0.02, 0.05, 0.10, 0.20, 0.35)
    for noise in noises:
        remain = budget_s - (time.time() - t0)
        if remain < 35.0:
            break
        limit = min(90.0, 0.20 * remain)
        nh = best_h + rng.normal(0.0, float(noise), len(best_h))
        nh = _project_sum(nh, 0.5 * len(nh))
        cand, sc = combined_polish(nh, limit, rng)
        if sc < best_score:
            best_h, best_score = cand, sc

    n_alt = int(len(best_h))
    if n_alt % 2 != 0:
        n_alt += 1

    patterns = []
    patterns.append(_make_checker(n_alt))
    patterns.append(_make_thue(n_alt))
    base_seed = int(seed)
    for s in range(3):
        patterns.append(_make_random_balanced(n_alt, np.random.default_rng(base_seed + 101 + s)))
    if parent is not None and int(parent.size) != n_alt:
        patterns.append(_upsample(parent, n_alt))

    for cand0 in patterns:
        remain = budget_s - (time.time() - t0)
        if remain < 35.0:
            break
        limit = min(70.0, 0.15 * remain)
        cand, sc = combined_polish(cand0, limit, rng)
        if sc < best_score:
            best_h, best_score = cand, sc

    remain = budget_s - (time.time() - t0)
    if remain > 30.0:
        final_deadline = t0 + budget_s - 3.0
        if len(best_h) <= 70:
            cand, sc = optimize_pairs(best_h, final_deadline, rng, grid=97, levels=4)
        else:
            cand, sc = optimize_pairs(best_h, final_deadline, rng)
        if sc < best_score:
            best_h, best_score = cand, sc

        if time.time() < final_deadline - 4.0:
            cand2, sc2 = linearized_polish(best_h, final_deadline)
            if sc2 < best_score:
                best_h, best_score = cand2, sc2

    best_h = _project_sum(best_h, 0.5 * len(best_h))
    if not np.all(np.isfinite(best_h)):
        best_h = _make_random_balanced(42, rng)

    best_score = score_array(best_h)
    return best_h, float(best_score), int(len(best_h))
