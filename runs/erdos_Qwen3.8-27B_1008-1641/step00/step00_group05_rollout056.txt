import numpy as np
from scipy.optimize import minimize
import time


class ObjFunc:
    def __init__(self, dx, T):
        self.dx = dx
        self.T = T

    def __call__(self, h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        return float((m + np.log(np.sum(np.exp(corr - m))) / self.T) * self.dx)


class SumFun:
    def __init__(self, target):
        self.target = target

    def __call__(self, h):
        return float(h.sum() - self.target)


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h0, n, T, maxiter=500, ftol=1e-13):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    obj = ObjFunc(dx, T)
    cons = SumFun(target)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)


def lbfgs_optimize(h0, n, T, maxiter=1000, ftol=1e-14):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    pen = 1e3
    dx_arr = 2.0 / n

    class PenObj:
        def __init__(self, dx_val, T_val, target_val, pen_val):
            self.dx = dx_val
            self.T = T_val
            self.target = target_val
            self.pen = pen_val

        def __call__(self, h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            m = np.max(corr)
            base = (m + np.log(np.sum(np.exp(corr - m))) / self.T) * self.dx
            c = h.sum() - self.target
            return base + self.pen * c * c

    obj = PenObj(dx_arr, T, target, pen)
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)
