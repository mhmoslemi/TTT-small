    current = h.copy()
    best = h.copy()
    best_c5 = compute_c5(best, n)
    current_c5 = best_c5

    iter_count = 0

        iter_count += 1
        T = start_T * (end_T / start_T) ** ((time.time() - start) / time_limit)

def scipy_optimize(h, n, rng, duration):
    from scipy.optimize import minimize
    
    target = n / 2
    
    def smooth_max(x, p=50):
        c = np.correlate(x, 1.0 - x, mode="full")
        val = c.max()  # Use actual max for final evaluation
        # Smooth version for optimization
        c_pos = np.maximum(c, 1e-10)
        return (np.sum(c_pos**p))**(1.0/p) * (2.0/n)

        h_val = np.clip(x, 0.0, 1.0)
        c = np.correlate(h_val, 1.0 - h_val, mode="full")
        c_max = c.max()
        penalty = (h_val.sum() - target)**2
        return c_max * (2.0/n) + 1000 * penalty
    
    best = h.copy()
    best_val = compute_c5(best, n)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Scale
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 0.05) & (h < 0.95)
        if free.any():
            h[free] = np.clip(h[free] + diff / free.sum(), 0.0, 1.0)
    return h

    lag = k_star - (n - 1)
    g = np.zeros(n)
    if lag == 0:

        L = -lag
        g[:n-L] += (1.0 - h[L:])
        g[L:] += -h[:n-L]

    dur = max(p5_end - time.time(), 0)
    if dur > 2:
        r, c5r = cont_subgrad(best_h, n, rng, dur, 0.008, 0.001, 0.99995)
        if c5r < best_c5:
            best_h, best_c5 = r, c5r

    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
