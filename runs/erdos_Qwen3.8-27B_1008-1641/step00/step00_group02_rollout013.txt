        np.clip(buf, 0.0, 1.0, out=buf)
        buf = repair_sum(buf, target_sum)
        new_c5 = compute_c5(buf, dx)
        if new_c5 <= c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - c5) / T)):
            h = buf.copy()
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_c5, best_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(45, 0.5)

    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = repair_sum(h, target_sum)

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    # Phase 1: Intensive local search
    best_c5, best_h = local_search(
        h, n, dx, target_sum, rng,
        400000, best_c5 * 0.006, 0.992,
        best_c5, best_h, t0, budget_s
    )

    # Phase 2: Iterated local search with perturbations
    ils_count = 0
    while time.time() - t0 < budget_s - 40:
        n_pert = rng.randint(3, 12)
        h_pert = best_h.copy()
        idx = rng.choice(n, size=n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.3, 0.3, size=n_pert)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = repair_sum(h_pert, target_sum)

        best_c5, best_h = local_search(
            h_pert, n, dx, target_sum, rng,
            60000, best_c5 * 0.004, 0.985,
            best_c5, best_h, t0, budget_s
        )
        ils_count += 1

    # Phase 3: Try rounding to grid and polish
    for step in [0.25, 0.1, 0.05]:
        if time.time() - t0 > budget_s - 15:
            break
        h_round = np.round(best_h / step) * step
        h_round = repair_sum(h_round, target_sum)
        c5_r = compute_c5(h_round, dx)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_round.copy()
            best_c5, best_h = local_search(
                best_h, n, dx, target_sum, rng,
                40000, best_c5 * 0.003, 0.99,
                best_c5, best_h, t0, budget_s
            )

    # Phase 4: Try alternate n values briefly
    for n_trial in [44, 46, 50]:
        if time.time() - t0 > budget_s - 15:
            break
        dx_t = 2.0 / n_trial
        ts_t = n_trial / 2.0
        h_t = np.zeros(n_trial)
        n_ones = int(ts_t)
        h_t[:n_ones] = 1.0
        if ts_t != n_ones:
            h_t[n_ones] = ts_t - n_ones
        rng.shuffle(h_t)
        h_t = repair_sum(h_t, ts_t)
        c5_t = compute_c5(h_t, dx_t)
        bh_t = h_t.copy()
        best_t, bh_t = local_search(
            h_t, n_trial, dx_t, ts_t, rng,
            30000, c5_t * 0.005, 0.988,
            c5_t, bh_t, t0, budget_s
        )
        if best_t < best_c5:
            best_c5 = best_t
            best_h = bh_t
            n = n_trial

    best_h = repair_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
