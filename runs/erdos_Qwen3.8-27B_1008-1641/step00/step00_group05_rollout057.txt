import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        count = np.sum(mask)
        if count == 0:
            break
        h[mask] += diff / count
        h = np.clip(h, 0, 1)
    return h

def soft_max_c5(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(alpha*(corr-m))))/alpha) * (2.0/len(h))

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 70
        h0 = np.full(n, 0.5)
    
    n = len(h0)
    h_best = project_feasible(h0.copy(), n)
    c5_best = compute_c5(h_best)
    
    # Phase 1: Binary search - find good binary patterns
    # Convert to binary
    h_bin = np.where(h_best > 0.5, 1.0, 0.0)

    if num_ones > n//2:
        # Remove excess ones (those with lowest h_best value)
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h_best[one_idx])
        remove = one_idx[order[:num_ones - n//2]]
        h_bin[remove] = 0
    elif num_ones < n//2:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h_best[zero_idx])
        add = zero_idx[order[:n//2 - num_ones]]
        h_bin[add] = 1
    
    c5_bin = compute_c5(h_bin)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()
    
    # Phase 2: Local search on binary patterns (swap ones and zeros)
    if time_left() > 60:
        improved = True
        while improved and time_left() > 30:
            improved = False
            one_idx = np.where(h_best == 1)[0]
            zero_idx = np.where(h_best == 0)[0]

            for _ in range(n_swaps):
                if time_left() < 10:

                j = zero_idx[np.random.randint(len(zero_idx))]
                h_try = h_best.copy()
                h_try[i] = 0
                h_try[j] = 1
                c5_try = compute_c5(h_try)
                if c5_try < c5_best - 1e-10:
                    c5_best = c5_try
                    h_best = h_try
                    one_idx = np.where(h_best == 1)[0]
                    zero_idx = np.where(h_best == 0)[0]
                    improved = True
    
    # Phase 3: Continuous optimization starting from best binary
    if time_left() > 60:
        for alpha in [20, 50, 100]:
            if time_left() < 30:
                break

                h = project_feasible(hf.copy(), n)
                return soft_max_c5(h, alpha)

                h_cand = project_feasible(res.x.copy(), n)

                if c5_cand < c5_best:
                    c5_best = c5_cand
                    h_best = h_cand
            except:

            h_pert = project_feasible(h_pert, n)
            
            def obj_s(hf):
                h = project_feasible(hf.copy(), n)
                return soft_max_c5(h, 50)

                add = zi[np.argsort(-h_best[zi])[:n//2 - num_ones]]

        c5_bin2 = compute_c5(h_bin2)
        if c5_bin2 < c5_best:
            c5_best = c5_bin2
            h_best = h_bin2
    
    # Final: try a few more binary swaps
    if time_left() > 10:
        for _ in range(500):
            if time_left() < 5:
                break
            oi = np.where(h_best == 1)[0]
            zi = np.where(h_best == 0)[0]
            if len(oi) == 0 or len(zi) == 0:
                break
            i = oi[np.random.randint(len(oi))]
            j = zi[np.random.randint(len(zi))]
            h_try = h_best.copy()
            h_try[i], h_try[j] = 0, 1
            c5_try = compute_c5(h_try)
            if c5_try < c5_best - 1e-10:
                c5_best = c5_try
                h_best = h_try
    
    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_c5(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    return sm * (2.0 / n)

def make_binary(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    num_ones = int(np.sum(h_b))
    target = n // 2
    if num_ones > target:
        oi = np.where(h_b == 1.0)[0]
        rem = oi[np.argsort(h[oi])[:num_ones - target]]
        h_b[rem] = 0.0
    elif num_ones < target:
        zi = np.where(h_b == 0.0)[0]
        add = zi[np.argsort(-h[zi])[:target - num_ones]]
        h_b[add] = 1.0
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 70
        h0 = np.full(n, 0.5)
    n = len(h0)
    target_ones = n // 2

    h_best = project_feasible(h0.copy(), n)
    c5_best = compute_c5(h_best)

    # Phase 1: Threshold to binary
    h_bin = make_binary(h_best, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()

    # Phase 2: Simulated annealing on binary patterns
    if time_left() > 60:
        T = 0.02
        T_min = 0.0001
        rate = (T_min / T) ** (1.0 / 50000.0)
        for it in range(200000):
            if time_left() < 15:
                break
            if it % 1000 == 0:
                T = max(T * rate, T_min)
            oi = np.where(h_best == 1.0)[0]
            zi = np.where(h_best == 0.0)[0]
            if len(oi) == 0 or len(zi) == 0:
                break
            i = oi[np.random.randint(len(oi))]
            j = zi[np.random.randint(len(zi))]
            h_try = h_best.copy()
            h_try[i] = 0.0
            h_try[j] = 1.0
            c5_try = compute_c5(h_try)
            d = c5_try - c5_best
            if d < -1e-12:
                c5_best = c5_try
                h_best = h_try

            if time_left() < 8:
                break
            oi = np.where(h_best == 1.0)[0]
            zi = np.where(h_best == 0.0)[0]
            if len(oi) == 0 or len(zi) == 0:

            h_try[i] = 0.0

            c5_try = compute_c5(h_try)

    # Final projection and verification

    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)
