from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h, target_sum):
    # Project onto {x: 0<=x<=1, sum=x_target}
    # Using bisection on the Lagrange multiplier
    def sum_at_lambda(lam):
        return np.sum(np.clip(h - lam, 0, 1))
    
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = sum_at_lambda(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0, 1)
    # Fix small numerical errors
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-12:
        # Adjust elements not at bounds
        free = (h_proj > 1e-6) & (h_proj < 1.0 - 1e-6)
        nfree = np.sum(free)
        if nfree > 0:
            h_proj[free] += diff / nfree
            h_proj = np.clip(h_proj, 0, 1)
        else:
            # All at bounds, perturb slightly
            h_proj[0] = np.clip(h_proj[0] + diff, 0, 1)
    return h_proj

def subgradient(h, k):
    """Subgradient of g_k(h) = sum_i h[i+k](1-h[i]) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # For each j, grad[j] = sum over contributions
    # h[j] appears in h[i+k] when i = j-k (contribution: 1-h[j-k])
    # h[j] appears in (1-h[i]) when i = j (contribution: -h[j+k])
    for j in range(n):
        i1 = j - k
        if 0 <= i1 < n:
            grad[j] += (1.0 - h[i1])
        i2 = j + k
        if 0 <= i2 < n:
            grad[j] += (-h[i2])
    return grad

    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)
    return k, float(corr[max_idx])

    """Projected subgradient + adaptive step size."""
    n = len(h)
    h = project_to_feasible(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    
    for it in range(max_iters):
        k, _ = get_active_lag(h)
        grad = subgradient(h, k)
        # Project gradient: subtract mean to maintain sum constraint
        grad_centered = grad - np.mean(grad)
        # Optional: project gradient to feasible direction
        h_new = h - step * grad
        h_new = project_to_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.9
            h = best_h.copy()  # revert
            if step < 1e-6:
                break
    
    return best_h, best_c5

def nelder_mead_optimize(h, target_sum, max_iters=2000):
    """Nelder-Mead on the reduced problem (using the equality constraint via projection)."""
    n = len(h)
    
    def objective(x):

        return compute_c5(x)

    res = minimize(objective, h, method='Nelder-Mead',

    h_opt = project_to_feasible(res.x, target_sum)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

            best_h, best_c5 = h_bs, c5_bs
    else:
        # Adjust to get correct count
        sorted_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sorted_idx[:int(target_sum)]] = 1.0
        h_bs, c5_bs = binary_local_search(h_bin, target_sum, max_flips=300)
        if c5_bs < best_c5:
            best_h, best_c5 = h_bs, c5_bs
    if time.time() - t0 > budget_s * 0.55:
        best_h = project_feasible(best_h, target_sum)
        return best_h, compute_c5(best_h), n

    # Phase 4: Multi-start perturbations
    for restart in range(12):
        if time.time() - t0 > budget_s * 0.8:
            break
        scale = 0.12 + 0.03 * restart
        noise = scale * rng.randn(n)
        hp = project_feasible(best_h + noise, target_sum)
        # Quick gradient
        h_gd, c5_gd = grad_descent(hp, target_sum, 1000, 8.0)
        if c5_gd < best_c5:
            hp, c5_gd = h_gd, c5_gd
        # Nelder-Mead refinement
        h_nm, c5_nm = nm_optimize(hp, target_sum, 1500)
        if c5_nm < best_c5:
            best_h, best_c5 = h_nm, c5_nm

    # Phase 5: Try higher resolution n=100
    if time.time() - t0 < budget_s * 0.88:
        n2 = 100
        ts2 = float(n2 // 2)
        h_interp = np.interp(np.linspace(0, n - 1, n2), np.arange(n), best_h)
        h_interp = project_feasible(h_interp, ts2)
        h_gd, c5_gd = grad_descent(h_interp, ts2, 2000, 8.0)
        h_nm, c5_nm = nm_optimize(h_gd, ts2, 2000)
        if c5_nm < best_c5:
            best_h, best_c5, n = h_nm, c5_nm, n2

    # Phase 6: Final NJ refinement
    if time.time() - t0 < budget_s * 0.97:
        h_nm, c5_nm = nm_optimize(best_h, float(n // 2), 3000)
        if c5_nm < best_c5:
            best_h, best_c5 = h_nm, c5_nm

    best_h = project_feasible(best_h, float(n // 2))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
