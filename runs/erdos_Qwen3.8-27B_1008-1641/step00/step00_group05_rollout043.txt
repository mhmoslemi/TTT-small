import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h


def golden_binary(n):
    n_ones = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    for i in range(n_ones):
        positions.add(int((i * phi) % n))
    h = np.zeros(n)
    for p in positions:
        h[p] = 1.0
    return project_feasible(h, n)


def local_search(h, n, target, budget_fn, max_time=60.0):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.12
    no_improve = 0
    t0 = time.time()

    while budget_fn() and (time.time() - t0) < max_time:
        if no_improve > 600:
            step *= 0.88
            no_improve = 0
            if step < 0.002:
                break
        k = np.random.randint(1, 4)
        idx = np.random.choice(n, size=k, replace=False)
        h_new = best_h.copy()
        h_new[idx] += np.random.uniform(-step, step, size=k)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5


def targeted_search(h, n, budget_fn, max_time=30.0):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    step = 0.08

    while budget_fn() and (time.time() - t0) < max_time:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        worst = np.argmax(corr)
        offset = worst - (n - 1)
        n_affected = n - abs(offset)
        if n_affected < 2:
            break
        lo = max(0, -offset)
        hi = lo + n_affected
        candidates = np.arange(lo, hi)
        k = np.random.randint(1, 3)
        idx = np.random.choice(candidates, size=k, replace=False)
        h_new = best_h.copy()
        h_new[idx] += np.random.uniform(-step, step, size=k)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            step = max(0.003, step * 0.995)
    return best_h, best_c5
