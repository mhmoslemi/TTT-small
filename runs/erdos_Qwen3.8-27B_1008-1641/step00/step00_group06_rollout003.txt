def smooth_obj(x,beta):
    n=_N; h=np.asarray(x,dtype=float)
    if h.size!=n: h=np.resize(h,n)
    if not np.all(np.isfinite(h)): return (1e100,np.zeros(n))
    d=_DX
    C=np.correlate(h,h,'full')[n-1:]
    pref=np.empty(n+1); pref[0]=0.0; np.cumsum(h,out=pref[1:])
    lidx=np.arange(n)
    Tpos=(pref[n-lidx]-C)*d
    Tneg=(pref[n]-pref[lidx]-C)*d
    T=np.empty(2*n-1)
    T[0]=Tpos[0]
    T[1:n]=Tpos[1:]
    T[n:2*n-1]=Tneg[1:]
    m=float(T.max())
    if beta <= 1e-14:

    else:

       if s<=0: ...; else w=ez/s; val=m+beta*math.log(s)
    g=np.zeros(n)
    w0=w[0]
    if w0>0.0: g += d*w0*(1.0-2.0*h)

import time, math
import numpy as np
from scipy.optimize import minimize

_N = 0
_DX = 0.0
_TARGET_SUM = 0.0
_MIN_STOP = None
OBJ_SCALE = 10.0
DEFAULT_BETAS = (0.05, 0.01, 0.002, 0.0004, 0.0001)
COORD_GRID = np.linspace(0.0, 1.0, 21)
COORD_MULTS = np.array((-3.0, -2.0, -1.0, 1.0, 2.0, 3.0))


def box_bounds(n):
    return [(0.0, 1.0)] * int(n)


def symmetrize(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    return 0.5 * (h + h[::-1])


def project_to_box_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = int(x.size)
    if n == 0:
        return x
    target = float(np.clip(target, 0.0, float(n)))
    x = np.clip(x, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) < 1e-13:
        return x
    sh = diff / float(n)
    if np.all(x + sh >= -1e-12) and np.all(x + sh <= 1.0 + 1e-12):
        return np.clip(x + sh, 0.0, 1.0)
    for _ in range(int(max(130, n + 30))):
        diff = target - float(x.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            inc = diff / float(cnt)
            maxinc = float(np.min(1.0 - x[mask]))
            if inc >= maxinc:
                x[mask] += maxinc
            else:
                x[mask] += inc
                break
        else:
            mask = x > 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            dec = -diff / float(cnt)
            minval = float(np.min(x[mask]))
            if dec >= minval:
                x[mask] -= minval
            else:
                x[mask] -= dec
                break
    for _ in range(40):
        diff = target - float(x.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - x))
            add = min(diff, 1.0 - x[idx])
            if add <= 1e-15:
                break
            x[idx] += add
        else:
            idx = int(np.argmin(x))
            dec = min(-diff, float(x[idx]))
            if dec <= 1e-15:
                break
            x[idx] -= dec
    return np.clip(x, 0.0, 1.0)


def true_c5(x):
    x = np.asarray(x, dtype=float).reshape(-1)
    n = int(x.size)
    if n == 0:
        return 0.0
    y = np.clip(x, 0.0, 1.0)
    m = float(np.max(np.correlate(y, 1.0 - y, mode='full')))
    if m < 0.0:
        m = 0.0
    return m * (2.0 / float(n))


def make_pattern_h(n, kind):
    n = int(n)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if kind == 0:
        h = 0.5 * (1.0 - np.cos(np.pi * x))
    elif kind == 1:
        h = 0.5 * (1.0 + np.cos(np.pi * x))
    elif kind == 2:
        h = 1.0 - np.abs(x - 1.0)
    else:
        h = np.where((x < 0.5) | (x > 1.5), 1.0, 0.0)
    return project_to_box_sum(h, 0.5 * float(n))


def resample_h(h, n_dst):
    h = np.asarray(h, dtype=float).reshape(-1)
    n_src = int(h.size)
    n_dst = int(n_dst)
    if n_src == n_dst:
        return h.copy()
    if n_src < 2:
        return np.full(n_dst, float(h[0]) if n_src else 0.5)
    xs = (np.arange(n_src) + 0.5) * (2.0 / float(n_src))
    xd = (np.arange(n_dst) + 0.5) * (2.0 / float(n_dst))
    return np.interp(xd, xs, h, left=float(h[0]), right=float(h[-1]))


def sum_eq(x):
    return float(np.sum(x) - _TARGET_SUM)


def sum_eq_jac(x):
    return np.ones_like(x, dtype=float)


def minimize_callback(xk):
    if _MIN_STOP is not None and time.time() > _MIN_STOP:
        raise RuntimeError("time limit")


def smooth_obj(x, beta):
    n = _N
    h = np.asarray(x, dtype=float).reshape(-1)
    if h.size != n:
        h = np.resize(h, n)
    if not np.all(np.isfinite(h)):
        return (1e100, np.zeros(n, dtype=float))
    d = float(_DX)
    C = np.correlate(h, h, mode='full')[n - 1:]
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    lidx = np.arange(n)
    Tpos = (pref[n - lidx] - C) * d
    Tneg = (pref[n] - pref[lidx] - C) * d
    T = np.empty(2 * n - 1, dtype=float)
    T[0] = Tpos[0]
    T[1:n] = Tpos[1:]
    T[n:2 * n - 1] = Tneg[1:]
    m = float(np.max(T))
    beta = float(beta)
    if beta <= 1e-14:
        w = np.zeros(2 * n - 1, dtype=float)
        act = T >= m - 1e-12
        w[act] = 1.0
        w[act] /= float(act.sum())
        val = m
    else:
        z = np.clip((T - m) / beta, -700.0, 0.0)
        ez = np.exp(z)
        s = float(ez.sum())
        if s <= 0.0:
            w = np.zeros(2 * n - 1, dtype=float)
            w[0] = 1.0
            val = m
        else:
            w = ez / s
            val = m + beta * math.log(s)
    g = np.zeros(n, dtype=float)
    w0 = float(w[0])
    if w0 > 0.0:
        g += d * w0 * (1.0 - 2.0 * h)
    for l in range(1, n):
        wp = float(w[l])
        wn = float(w[n - 1 + l])
        if wp <= 1e-18 and wn <= 1e-18:
            continue
        a = n - l
        ws = wp + wn
        g[:a] += d * (wp - ws * h[l:])
        g[l:] += d * (wn - ws * h[:a])
    val *= OBJ_SCALE
    g *= OBJ_SCALE
    return (float(val), g)


def optimize_from(start, deadline, max_calls=5, maxiter=None):
    global _N, _DX, _TARGET_SUM, _MIN_STOP
    h = np.asarray(start, dtype=float).reshape(-1)
    n = int(h.size)
    if n == 0:
        return None, np.inf
    target = 0.5 * float(n)
    _N = n
    _DX = 2.0 / float(n)
    _TARGET_SUM = target

    best = h.copy()

    if maxiter is None:
        maxiter = 45 if n <= 64 else (35 if n <= 128 else 28)
    bounds = box_bounds(n)

    start = h.copy()
    for beta in DEFAULT_BETAS[:int(max_calls)]:
        if time.time() >= deadline - 10.0:
            break
        _MIN_STOP = deadline - 3.0
        try:
            res = minimize(
                smooth_obj, start, args=(float(beta),), method='SLSQP',
                bounds=bounds, constraints=cons, jac=True,
                callback=minimize_callback,
                options={'maxiter': int(maxiter), 'ftol': 1e-9, 'disp': False}
            )
        except Exception:
            res = None
        _MIN_STOP = None
        if res is None or res.x is None:
            continue
        cand = project_to_box_sum(res.x, target)
        c = true_c5(cand)
        if not np.isfinite(c):
            continue
        if c < best_c:
            best = cand
            best_c = c
            start = cand
        elif c <= best_c + 2e-7:
            start = cand
        else:
            start = best
    return best, best_c

                    continue

                shift = (old - y) / float(n)
                if np.all(h2 + shift >= -1e-12) and np.all(h2 + shift <= 1.0 + 1e-12):
                    h2 = np.clip(h2 + shift, 0.0, 1.0)
                else:
                    h2 = project_to_box_sum(h2, target)

                best_h = h.copy()
