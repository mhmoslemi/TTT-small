```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_subgrad(h, n_points):
    dx = 2.0 / n_points
    n = n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    tol = 1e-6 * max(1.0, abs(max_corr))
    active = corr >= max_corr - tol
    cnt = np.sum(active)
    if cnt == 0:
        active[np.argmax(corr)] = True
        cnt = 1
    w = 1.0 / cnt
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if not active[k_idx]:
            continue
        k = k_idx - (n - 1)
        m = abs(k)
        if m == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[m:n] += w * (1.0 - h[0:n-m])
            grad[0:n-m] += w * (-h[m:n])
        else:
            grad[0:n-m] += w * (1.0 - h[m:n])
            grad[m:n] += w * (-h[0:n-m])
    return grad * dx

def project_sum(h, n, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} approximately."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Simple: distribute evenly among non-saturated variables
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    n_mask = np.sum(mask)
    if n_mask == 0:
        return h
    h[mask] += diff / n_mask
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Try multiple n_points values
    n_options = [70, 50, 100]
    
    best_h = None
    best_c5 = float('inf')
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
        n_init = len(init)
        c5_init = compute_c5(init, n_init)
        best_h = init.copy()
        best_c5 = c5_init
        n_options.insert(0, n_init)
    
    for n in set(n_options):
        if time.time() - t0 > budget_s * 0.8:
            break
        
        dx = 2.0 / n
        target = n / 2.0
        
        # Create initial points for this n
        if n == (len(best_h) if best_h is not None else -1):
            inits = [best_h.copy()]
        else:
            # Interpolate from best or create random
            if best_h is not None and len(best_h) != n:
                from scipy.interpolate import interp1d
                x_old = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n)
                f = interp1d(x_old, best_h, kind='linear')
                h_interp = f(x_new)
                h_interp = project_sum(h_interp, n, target)
                inits = [h_interp]
            else:
                h_interp = None
                inits = []
            
            # Add random starts
            for _ in range(3):
                h_rand = rng.uniform(0, 1, n)
                h_rand = project_sum(h_rand, n, target)
                inits.append(h_rand)
        
        # Optimize from each initial point
        for init_pt in inits:
            if time.time() - t0 > budget_s * 0.9:
                break
            
            try:
                res = minimize(
                    compute_c5,
                    init_pt.copy(),
                    args=(n,),
                    method='SLSQP',
                    jac=lambda h, n_: c5_subgrad(h, n_),
                    bounds=[(0, 1)] * n,
                    constraints=[{'type': 'eq', 'fun': lambda h, n_=n: np.sum(h) - n_/2.0,
                                  'jac': lambda h, n_=n: np.ones(n)}],
                    options={'maxiter': 300, 'ftol': 1e-10}
                )
                if res.fun < best_c5:
                    best_c5 = res.fun
                    best_h = res.x.copy()
            except Exception:
                pass
        
        # Also try projected subgradient descent
        if time.time() - t0 > budget_s * 0.85:
            break
        
        h_cur = init_pt.copy() if len(inits) > 0 else rng.uniform(0, 1, n)
        h_cur = project_sum(h_cur, n, target)
        
        lr = 0.01
        for step in range(2000):
            if time.time() - t0 > budget_s * 0.9:
                break
            g = c5_subgrad(h_cur, n)
            h_new = h_cur - lr * g
            h_new = project_sum(h_new, n, target)
            # Adaptive step
            if compute_c5(h_new, n) < compute_c5(h_cur, n):
                h_cur = h_new
                lr = min(lr * 1.05, 0.1)
            else:
                lr *= 0.8
            
            if compute_c5(h_cur, n) < best_c5:
                best_c5 = compute_c5(h_cur, n)
                best_h = h_cur.copy()
    
    # Final polish with SLSQP
```

- c[k] = sum_{t} a[t+k] * conj(v[t]) for valid t
- k ranges from -(len(v)-1) to (len(a)-1), i.e., -(n-1) to (n-1)
- Index in output: k + (n-1), ranging from 0 to 2n-2
