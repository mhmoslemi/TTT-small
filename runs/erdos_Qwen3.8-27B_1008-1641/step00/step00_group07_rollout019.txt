import numpy as np
import time

def compute_c5(h):
    """Compute C5 value for a given h vector."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def local_search_binary(h, iterations, seed=42):
    """Local search for binary h with fixed number of ones."""
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    # Ensure binary
    h = (h > 0.5).astype(float)
    # Ensure correct number of ones
    num_ones = int(round(n/2))
    while h.sum() > num_ones:
        idx = rng.choice(np.where(h==1)[0])
        h[idx] = 0
    while h.sum() < num_ones:
        idx = rng.choice(np.where(h==0)[0])
        h[idx] = 1
    
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    best_h = h.copy()
    
    for it in range(iterations):
        # Pick a 1 to flip to 0 and a 0 to flip to 1
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        if new_c5 <= current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def simulated_annealing_binary(h, iterations, start_temp, end_temp, seed=42):
    """Simulated annealing for binary h."""
    rng = np.random.RandomState(seed)
    n = len(h)
    h = (h > 0.5).astype(float)
    num_ones = int(round(n/2))
    while h.sum() > num_ones:
        idx = rng.choice(np.where(h==1)[0])
        h[idx] = 0
    while h.sum() < num_ones:
        idx = rng.choice(np.where(h==0)[0])
        h[idx] = 1
    
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    best_h = h.copy()
    
    for it in range(iterations):
        T = start_temp * (end_temp/start_temp) ** (it/iterations)
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.rand() < np.exp(-delta/T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_binary(h, seed=0):
    n = len(h)
    target = n // 2
    h_b = np.zeros(n, dtype=np.float64)
    order = np.argsort(-h)

    return h_b
