import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def local_search(h, rng, n_iter, step_size, time_check, budget_f):
    """Sum-preserving local search. Returns (h, best_c5, n_updates)."""
    n = len(h)
    h_work = h.copy()
    best = h.copy()
    best_c5 = np.max(np.correlate(best, 1.0 - best, mode='full')) * (2.0 / n)
    updates = 0
    for _ in range(n_iter):
        if time_check() < budget_f:
            break
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        hi_lim = min(1.0 - h_work[i], h_work[j])
        lo_lim = max(-h_work[i], -(1.0 - h_work[j]))
        lo = max(-step_size, lo_lim)
        hi = min(step_size, hi_lim)
        if hi - lo < 1e-12:
            continue
        delta = rng.uniform(lo, hi)
        h_work[i] += delta
        h_work[j] -= delta
        c5 = np.max(np.correlate(h_work, 1.0 - h_work, mode='full')) * (2.0 / n)
        if c5 < best_c5:
            c5_best_save = best_c5
            best_c5 = c5
            best = h_work.copy()
            updates += 1
        else:
            h_work[i] -= delta
            h_work[j] += delta
    return best, best_c5, updates

def sa_search(h, rng, n_iter, t_start, t_end, time_check, budget_f):
    """Simulated annealing for h with sum-preserving moves."""
    n = len(h)
    h_work = h.copy()
    best = h.copy()
    best_c5 = np.max(np.correlate(best, 1.0 - best, mode='full')) * (2.0 / n)
    for it in range(n_iter):
        if time_check() < budget_f:
            break
        temp = t_start + (t_end - t_start) * it / max(1, n_iter)
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        hi_lim = min(1.0 - h_work[i], h_work[j])
        lo_lim = max(-h_work[i], -(1.0 - h_work[j]))
        lo = max(-0.05, lo_lim)
        hi = min(0.05, hi_lim)
        if hi - lo < 1e-12:
            continue
        delta = rng.uniform(lo, hi)
        h_work[i] += delta
        h_work[j] -= delta
        c5 = np.max(np.correlate(h_work, 1.0 - h_work, mode='full')) * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best = h_work.copy()
        elif rng.random() < np.exp(-(c5 - best_c5) / max(temp, 1e-8)):
            pass  # keep the move
        else:
            h_work[i] -= delta
            h_work[j] += delta
    return best, best_c5

def subgradient_step(h, step_lr, time_check, budget_f, n_iter=50000):
    """Subgradient method focusing on the worst lag."""
    n = len(h)
    h_work = h.copy()
    best = h.copy()
    best_c5 = np.max(np.correlate(best, 1.0 - best, mode='full')) * (2.0 / n)
    for it in range(n_iter):
        if time_check() < budget_f:
            break
        corr = np.correlate(h_work, 1.0 - h_work, mode='full')
        k_star = int(np.argmax(corr))
        # Gradient of corr[k_star] w.r.t. h[i]
        # corr[k_star] = sum_j h[j] * (1-h[j + k_star - (n-1)])  for valid j
        # Let offset = k_star - (n-1)
        # corr = sum_j h[j+k_off] * (1-h[j]) for valid j
        off = k_star - (n - 1)
        grad = np.zeros(n)
        if off >= 0:
            # corr = sum_{j=0}^{n-1-off} h[j+off]*(1-h[j])
            # d/dh[j]: from h[j+off] term: +1 if j+off in range, i.e., j in [0, n-1-off]
            #          from -h[j+off]*h[j]: -h[j] and -h[j+off]
            for j in range(n - off):
                idx1 = j + off  # position of h[j+off]
                idx2 = j        # position of h[j]
                grad[idx1] += 1.0 - h_work[idx2]
                grad[idx2] += -h_work[idx1]
                # Actually: d/dh[idx1] of h[idx1]*(1-h[idx2]) = (1-h[idx2])
                # d/dh[idx2] of h[idx1]*(1-h[idx2]) = -h[idx1]
                # But we already have: h[j+off]*(1-h[j])
                # d/dh[j+off] = (1-h[j])
                # d/dh[j] = -h[j+off]
                # But each h[i] can appear in multiple terms!
        else:
            off = -off
            for j in range(n - off):
                idx1 = j  # h[j]
                idx2 = j + off  # h[j+off]
                # h[j+off]*(1-h[j]): d/dh[j+off]=(1-h[j]), d/dh[j]=-h[j+off]
                grad[idx2] += 1.0 - h_work[idx1]
                grad[idx1] += -h_work[idx2]
        # Project gradient to preserve sum: subtract mean
        grad -= np.mean(grad)
        # Take step
        h_work = np.clip(h_work + step_lr * grad, 0.0, 1.0)
        # Fix sum
        diff = n / 2.0 - h_work.sum()
        h_work += diff / n
        h_work = np.clip(h_work, 0.0, 1.0)
        c5 = compute_c5(h_work)
        if c5 < best_c5:
            best_c5 = c5
            best = h_work.copy()
        else:
            h_work = best.copy()
    return best, best_c5

            h_best = h_try
            c5_best = c5_try
