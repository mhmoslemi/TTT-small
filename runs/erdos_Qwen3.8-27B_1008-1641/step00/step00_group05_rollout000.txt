import numpy as np

def eval_c5(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n <= 0:
        return 1.0
    vals = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(vals)) * (2.0 / n)


def project_box_sum(y, S):
    y = np.asarray(y, dtype=float).copy()
    n = int(y.size)
    if n == 0:
        return y
    y = np.clip(y, 0.0, 1.0)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(100):
        r = S - float(np.sum(z))
        if abs(r) < 1e-13:
            break
        if r > 0.0:
            idx = np.where(z < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            step = r / float(idx.size)
            if step < float(np.min(1.0 - z[idx])):
                z[idx] += step
            else:
                for i in idx:
                    if r <= 0.0:
                        break
                    cap = 1.0 - z[i]
                    add = min(cap, r)
                    z[i] += add
                    r -= add
        else:
            idx = np.where(z > 1e-12)[0]
            if idx.size == 0:
                break
            step = -r / float(idx.size)
            if step < float(np.min(z[idx])):
                z[idx] -= step
            else:
                for i in idx:
                    if r >= 0.0:
                        break
                    cap = z[i]
                    sub = min(cap, -r)
                    z[i] -= sub
                    r += sub
    return z


def normalize_h(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return h.copy()
    return project_box_sum(h, 0.5 * float(h.size))


def smooth_max_value(h, beta):
    h = np.asarray(h, dtype=float)
    vals = np.correlate(h, 1.0 - h, mode="full")
    vals = vals * (2.0 / float(h.size))
    m = float(np.max(vals))
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    ez = np.exp(shifted)
    s = float(np.sum(ez))
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / beta


def smooth_max_grad(h, beta):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode="full")
    vals = vals * dx
    m = float(np.max(vals))
    shifted = np.clip(beta * (vals - m), -700.0, 0.0)
    ez = np.exp(shifted)
    s = float(np.sum(ez))
    if s <= 0.0:
        return np.zeros(n, dtype=float)
    w = ez / s
    g = np.zeros(n, dtype=float)
    base = n - 1
    for lag in range(-(n - 1), n):
        wi = float(w[lag + base])
        if wi <= 1e-15:
            continue
        if lag >= 0:
            a = slice(lag, n)
            b = slice(0, n - lag)
        else:
            a = slice(0, n + lag)
            b = slice(-lag, n)
        g[a] += dx * wi * (1.0 - h[b])
        g[b] -= dx * wi * h[a]
    return g


def penalty_smooth_value(x, beta, lam, target):
    v = smooth_max_value(x, beta)
    d = float(np.sum(x) - target)
    return v + lam * d * d


def penalty_smooth_grad(x, beta, lam, target):
    g = smooth_max_grad(x, beta)
    d = float(np.sum(x) - target)
    return g + (2.0 * lam * d / float(x.size)) * np.ones_like(g)


def sum_eq(x, target):
    return float(np.sum(x) - target)


def exact_obj(z):
    return float(z[-1])


def exact_obj_grad(z):
    g = np.zeros(z.size, dtype=float)
    g[-1] = 1.0
    return g


def exact_eq(z, target):
    return float(np.sum(z[:-1]) - target)


def exact_ineq_values(z):
    h = z[:-1]
    t = float(z[-1])
    vals = np.correlate(h, 1.0 - h, mode="full")
    vals = vals * (2.0 / float(h.size))
    return t - vals


def exact_ineq_jac(z):
    h = z[:-1]
    n = int(h.size)
    dx = 2.0 / float(n)
    L = 2 * n - 1
    J = np.zeros((L, n + 1), dtype=float)
    base = n - 1
    for lag in range(-(n - 1), n):
        row = J[lag + base]
        row[-1] = 1.0
        if lag >= 0:
            a = slice(lag, n)
            b = slice(0, n - lag)
        else:
            a = slice(0, n + lag)
            b = slice(-lag, n)
        row[a] -= dx * (1.0 - h[b])
        row[b] += dx * h[a]
    return J


def exact_slqp(h, target, maxiter):
    if minimize is None:
        return h, eval_c5(h)
    h = normalize_h(h)
    v = eval_c5(h)
    t = min(1.5, max(1e-4, v + 1e-3))
    z = np.concatenate([h, np.array([t], dtype=float)])
    bnds = [(0.0, 1.0)] * h.size + [(0.0, 1.5)]
    cons = [
        {"type": "eq", "fun": exact_eq, "args": (target,)},
        {"type": "ineq", "fun": exact_ineq_values, "jac": exact_ineq_jac},
    ]
    try:
        res = minimize(
            exact_obj,
            z,
            method="SLSQP",
            jac=exact_obj_grad,
            bounds=bnds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
        )
        x = np.asarray(res.x[:-1], dtype=float)
        if not np.all(np.isfinite(x)):
            return h, v
        x = normalize_h(x)
        return x, eval_c5(x)
    except Exception:
        return h, v

def active_positions(h, tol=1e-5):
    n = int(h.size)
    vals = np.correlate(h, 1.0 - h, mode="full")
    vals = vals * (2.0 / float(n))

                b = slice(0, n - lag)
            else:
                a = slice(0, n + lag)
                b = slice(-lag, n)
            act[a] = True
            act[b] = True
    return act

def make_alternating(n):

def random_binary_h(n, seed):
    n = int(n)
    rng = np.random.default_rng(seed)
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=float)
    k = int(0.5 * n)
    if k > 0:

    return normalize_h(h)


def resample_h(h, n):
    h = np.asarray(h, dtype=float)
    n0 = int(h.size)
    n = int(n)
    if n <= 0 or n0 <= 0:
        return h.copy()
    if n0 == n:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    g = np.interp(x1, x0, h, left=float(h[0]), right=float(h[-1]))
    return normalize_h(g)


def noise_h(h, scale, seed):

    rng = np.random.default_rng(seed)
    z = h + float(scale) * rng.standard_normal(h.size)
    return normalize_h(z)


def roll_h(h, k):

        parent = None

    if parent is None:
        parent = make_alternating(70)

    specs = []
    if parent.size >= 4:
        specs.append((parent, True, 90.0, 0))
        specs.append((noise_h(parent, 0.04, seed + 5), True, 55.0, 5))
        specs.append((noise_h(parent, 0.10, seed + 11), True, 45.0, 11))
        specs.append((resample_h(parent, 98), False, 45.0, 23))
        specs.append((resample_h(parent, 140), False, 30.0, 29))
        specs.append((resample_h(parent, 56), True, 40.0, 37))
        specs.append((noise_h(roll_h(parent, 2), 0.05, seed + 41), True, 35.0, 41))
        specs.append((noise_h(parent, 0.22, seed + 47), False, 30.0, 47))
    else:
        specs.append((parent, True, 90.0, 0))
        specs.append((random_binary_h(70, seed), True, 60.0, 2))
        specs.append((random_binary_h(98, seed + 7), False, 40.0, 7))
        specs.append((random_binary_h(56, seed + 13), True, 40.0, 13))

    phase_ends = (0.34, 0.52, 0.66, 0.78, 0.88, 0.96, 1.00, 1.00)

    for idx, spec in enumerate(specs):
        now = time.time()
        if now > deadline - 35:
            break
        h0, use_slqp, local_sec, seed_off = spec
        if idx < len(phase_ends):
            end_abs = t_start + float(phase_ends[idx]) * budget_s
        else:
            end_abs = deadline - 15.0
        end_abs = min(end_abs, deadline - 20.0)
        if end_abs <= now + 12.0:
            continue

        n0 = int(np.asarray(h0).size)
        if n0 <= 80:
            betas = (10.0, 30.0, 80.0, 160.0, 320.0)
            if n0 <= 60:
                maxiter = 100
            else:
                maxiter = 70
        elif n0 <= 120:
            betas = (20.0, 60.0, 140.0, 260.0)
            maxiter = 45
        else:
            betas = (30.0, 80.0, 160.0)
            maxiter = 30

        local_sec = min(float(local_sec), max(0.0, end_abs - now - 18.0))
        h1, c1 = optimize_candidate(h0, end_abs, betas, maxiter, use_slqp, local_sec, seed + int(seed_off))
        best_h, best_c = update_best(best_h, best_c, h1, c1)

        if best_h is not None and best_c < 0.37999 and time.time() > t_start + 0.25 * budget_s:
            extra = (normalize_h(best_h), True, 30.0, 900 + idx)
            if idx + 1 < len(specs):
                specs[idx + 1:0] = []
                specs.insert(0, extra)

    if best_h is None:
        best_h = normalize_h(parent)
        best_c = eval_c5(best_h)
    else:
        best_h = normalize_h(best_h)
        best_c = eval_c5(best_h)

    if time.time() < deadline - 12:
        left = max(1.0, min(25.0, deadline - time.time() - 8.0))
        h1, c1 = mass_local(best_h, seconds=left, seed=seed + 555)
        best_h, best_c = update_best(best_h, best_c, h1, c1)

    best_h = normalize_h(best_h)
    best_c = eval_c5(best_h)
    return (best_h, float(best_c), int(best_h.size))
