from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    n = len(h)
    target = float(np.clip(target, 0, n))
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return np.clip(h, 0, 1)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def smooth_max_obj_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    ec = np.exp(tau * (corr - m))
    w = ec / np.sum(ec)
    obj_val = m + np.log(np.sum(ec)) / tau
    obj_val *= dx
    
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - n + 1
            if 0 <= i + s < n:
                g += wk * (1.0 - h[i + s])
            if 0 <= i - s < n:
                g -= wk * h[i - s]
        grad[i] = dx * g
    return obj_val, grad

    h = project_box_sum(h0, target)

    best_c5 = compute_c5(h, dx)
    
    tau = 50.0
    lr = 0.01
    n_iter = 0
    stall = 0
    
    while time.time() - t0 < max_time:
        n_iter += 1
        val, grad = smooth_max_obj_grad(h, dx, tau)
        
        # Gradient step with projection
        h_new = project_box_sum(h - lr * grad, target)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stall = 0

                tau = min(tau * 1.5, 200)

            h = h  # keep current

            if stall > 50:
                # Reduce learning rate
                lr *= 0.7
                stall = 0
                if lr < 1e-6:
                    break
