import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_ls(h, t=50.0):
    """Smoothed version using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Shift to avoid overflow
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / t

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)

        if diff > 0:
            order = np.argsort(h)  # low values first
        else:

                h[i] += d
            else:
                d = min(h[i], -diff)
                h[i] -= d
            diff = target - h.sum()
            if abs(diff) < 1e-14:
                break
    return h

def slsqp_opt(h0, n, maxiter=300, tol=1e-9):
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - n/2,
             'jac': lambda x: np.ones(n)},)
    res = minimize(compute_c5, h0, method='SLSQP',
                   bounds=[(0,1)]*n, constraints=cons,
                   tol=tol, options={'maxiter': maxiter})
    if np.all(np.isfinite(res.x)):
        res.x = project_feasible(res.x, n)
        return res.x, compute_c5(res.x)
    return h0, compute_c5(h0)

def bit_hill_climb(h, n, max_steps=20000, rng=None):
    """Local search: flip pairs of bits to maintain sum"""
    if rng is None:
        rng = np.random.RandomState(0)
    h = np.clip(h, 0, 1)

    best_v = compute_c5(best)

        j = rng.randint(n)
        if i == j:
            continue
        # Try swap-like perturbation: decrease h[i], increase h[j]
        # Or flip: if h[i]>0.5 and h[j]<0.5, swap
        new_h = best.copy()
        if new_h[i] > 0.5 and new_h[j] < 0.5:
            new_h[i] = 0.0
            new_h[j] = 1.0
        elif new_h[i] < 0.5 and new_h[j] > 0.5:
            new_h[i] = 1.0
            new_h[j] = 0.0
        else:
            # Partial swap
            amt = 0.5
            new_h[i] = max(0, new_h[i] - amt)
            new_h[j] = min(1, new_h[j] + amt)
        new_v = compute_c5(new_h)
        if new_v < best_v - 1e-14:
            best = new_h
            best_v = new_v
    return best, best_v

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(h) if diff > 0 else np.argsort(-h)
        for i in idx:
            if diff > 0:
                d = min(1.0 - h[i], diff)
            else:
                d = min(h[i], -diff)
            h[i] += d if diff > 0 else -d
            diff = target - h.sum()
            if abs(diff) < 1e-14:
                break
    return h

def slsqp_opt(h0, n, maxiter=300, tol=1e-9):
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - n / 2,
             'jac': lambda x: np.ones(n)},)
    try:
