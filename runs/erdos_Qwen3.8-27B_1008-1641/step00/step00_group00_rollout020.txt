```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_to_sum(h, target_sum):
    h = h.copy()
    for _ in range(200):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) == 0:
            mask = np.ones(len(h), dtype=bool)
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_to_sum(h_new, new_n / 2.0)
    return h_new

def subgradient_h(h, k_star):
    """Analytical subgradient of c_{k_star} = sum_n h[n+k]*(1-h[n]) w.r.t. h.
    Using numpy correlate convention: c_k = sum_n a[n+k]*v[n] where a=h, v=1-h.
    So c_k = sum_n h[n+k]*(1-h[n]).
    dc_k/dh[i]:
      - i appears in h[n+k] when n = i-k: contributes (1-h[i-k]) if 0<=i-k<n
      - i appears in h[n] when n = i: contributes -h[i+k] if 0<=i+k<n
    """
    n = len(h)
    grad = np.zeros(n)
    # Vectorized:
    # Term 1: for i where 0<=i-k<n, grad[i] += (1-h[i-k])
    # Term 2: for i where 0<=i+k<n, grad[i] += -h[i+k]
    
    lo1 = max(0, k_star)
    hi1 = min(n, n + k_star)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        j = idx - k_star
        grad[idx] += (1.0 - h[j])
    
    lo2 = max(0, -k_star)  # i >= 0
    hi2 = min(n, n - k_star)  # i+k < n
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        j = idx + k_star
        grad[idx] += (-h[j])
    
    return grad

def projected_gradient_step(h, alpha, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = np.argmax(corr)
    g = subgradient_h(h, k_star)
    # Project gradient onto sum=0 subspace
    g = g - np.mean(g)
    h_new = h - alpha * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

                if s_rand < best_score:
                    best = h_rand.copy()
                    best_score = s_rand

            alpha = 0.01
    
    return best, best_score
```

```python
def compute_c5(h):

        diff = target_sum - s
        if abs(diff) < 1e-14:
            break

        cnt = np.sum(mask)
        if cnt == 0:
            mask = np.ones(len(h), dtype=bool)
            cnt = len(h)
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_to_sum(h_new, new_n / 2.0)
    return h_new

def subgradient_h(h, k_star):
    n = len(h)
    grad = np.zeros(n)
    # c_k = sum_n h[n+k]*(1-h[n])
    # dc_k/dh[i] = (1-h[i-k_star]) [if 0<=i-k_star<n] + (-h[i+k_star]) [if 0<=i+k_star<n]
    lo = max(0, k_star)
    hi = min(n, n + k_star)
    if lo < hi:
        idx = np.arange(lo, hi)
        grad[idx] += (1.0 - h[idx - k_star])
    lo2 = max(0, -k_star)
    hi2 = min(n, n - k_star)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[idx] += (-h[idx + k_star])
    return grad

    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)

    min_alpha = 1e-6

        # Compute subgradient at worst lag
        corr = np.correlate(best, 1.0 - best, mode='full')
        k_star = np.argmax(corr)
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_to_sum(h, target_sum):
    h = h.copy()
    for _ in range(300):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt == 0:
            mask = np.ones(len(h), dtype=bool)
            cnt = len(h)
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_to_sum(h_new, new_n / 2.0)
    return h_new

def subgradient_h(h, k_star):
    n = len(h)
    L = n - 1 - k_star
    grad = np.zeros(n)
    lo1 = max(0, L)
    hi1 = min(n, L + n)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[idx] += (1.0 - h[idx - L])
    lo2 = max(0, -L)
    hi2 = min(n, n - L)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[idx] += (-h[idx + L])
    return grad

def pair_moves(h, target_sum, rng, n_trials):
    n = len(h)
    best = h.copy()
    best_s = compute_c5(best)
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = (best[j] - best[i]) * rng.uniform(0.3, 1.0)
        h_try = best.copy()
        h_try[i] += delta
        h_try[j] -= delta
        if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
            continue
        s = compute_c5(h_try)
        if s < best_s:
            best = h_try
            best_s = s
    return best, best_s

def optimize_gradient(h, target_sum, time_limit, rng):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    alpha = 0.05
    min_alpha = 1e-7
    iter_count = 0

    while time.time() - start < time_limit:
        corr = np.correlate(best, 1.0 - best, mode='full')
        k_star = np.argmax(corr)
        g = subgradient_h(best, k_star)
        g = g - g.mean()
        g_norm = np.sqrt(np.sum(g * g))
        if g_norm < 1e-12:
            break
        g = g / g_norm
        h_new = best - alpha * g_norm * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        score = compute_c5(h_new)
        iter_count += 1
        if score < best_score - 1e-14:
            best = h_new
            best_score = score
            alpha = min(alpha * 1.02, 0.5)
        else:
            alpha *= 0.85
            if alpha < min_alpha:
                h_p, s_p = pair_moves(best, target_sum, rng, 200)
                if s_p < best_score:
                    best = h_p
                    best_score = s_p
                    alpha = 0.05
                    continue
                break
        if iter_count % 300 == 0:
            h_rand = best + 0.015 * rng.randn(n)
            h_rand = np.clip(h_rand, 0, 1)
            h_rand = project_to_sum(h_rand, target_sum)
            s_rand = compute_c5(h_rand)
            if s_rand < best_score:
                best = h_rand
                best_score = s_rand
                alpha = 0.05
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    if 'initial_h_values' in globals():

    # Phase 1: optimize at native resolution
    t_lim = min(90, budget_s * 0.12)
    h_opt, s_opt = optimize_gradient(h_best, len(h_best) / 2.0, t_lim, rng)
    if s_opt < best_score:

        h_best = h_opt.copy()
        best_n = len(h_best)

    # Pair moves at native resolution
    h_pm, s_pm = pair_moves(h_best, best_n / 2.0, rng, 3000)
    if s_pm < best_score:
        best_score = s_pm
        h_best = h_pm.copy()

    # Phase 2: upscale and refine
    for target_n in [90, 180, 360]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 50:
            break
        if target_n <= best_n:
            continue
        h_up = upsample(h_best, target_n)
        h_cur = h_up.copy()
        cur_score = compute_c5(h_up)
        n_attempts = min(5, max(2, int((remaining - 40) / 30)))
        for attempt in range(n_attempts):
            remaining = budget_s - (time.time() - t0)
            if remaining < 35:

            t_lim = min(25, remaining - 30)
            if t_lim < 5:

            if attempt == 0:
                h_try = h_up.copy()
            else:
                h_try = h_cur + 0.008 * rng.randn(target_n)
                h_try = np.clip(h_try, 0, 1)
                h_try = project_to_sum(h_try, target_n / 2.0)
            h_res, s_res = optimize_gradient(h_try, target_n / 2.0, t_lim, rng)
            h_res, s_res2 = pair_moves(h_res, target_n / 2.0, rng, 1500)
            if s_res2 < s_res:
                h_res = h_res.copy()  # already best

            if s_res2 < cur_score:

        if cur_score < best_score:
            best_score = cur_score

            best_n = target_n

    return h_best, final_score, best_n
```
