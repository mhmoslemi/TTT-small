import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.special import logsumexp
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * temp * logsumexp(corr / temp)

def project_h(h):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = int(np.sum(m))
        if c == 0:
            break
        h[m] += diff / c
        h = np.clip(h, 0.0, 1.0)
    return h

def h_to_t(h, n):
    return (h[:n-1] - 0.5).copy()

def t_to_h(t, n):
    h = np.empty(n)
    h[:n-1] = 0.5 + t
    h[n-1] = 0.5 - np.sum(t)
    return h

        h = t_to_h(t, n)

            return evaluate_c5(hc)
        return smooth_c5(hc, temp)
