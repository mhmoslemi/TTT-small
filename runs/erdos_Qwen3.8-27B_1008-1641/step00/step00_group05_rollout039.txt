    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff < 1e-14:
                break
    else:
        order = np.argsort(h)
        for i in order:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff > -1e-14:
                break
    return h

def smax_val_grad(h, dx, alpha):
    n = len(h)
    c = compute_c(h, dx)
    m = np.max(c)
    exp_c = np.exp(alpha * (c - m))
    w = exp_c / exp_c.sum()
    val = m + (1.0 / alpha) * np.log(np.mean(exp_c))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if lag >= 0:
            t1[lag:] = 1.0 - h[:n - lag]
            t2[:n - lag] = h[lag:]
        else:
            L = -lag
            t1[:n - L] = 1.0 - h[L:]
            t2[L:] = h[:n - L]
        grad += w[k] * dx * (t1 - t2)
    return val, grad

def lbfgs_opt(h0, n, alpha=50):
    dx = 2.0 / n
    target = n / 2.0
    pen = 200.0

    def obj(h):
        v, g = smax_val_grad(h, dx, alpha)
        v += pen * (h.sum() - target) ** 2
        g = g + 2.0 * pen * (h.sum() - target)
        return v, g

    res = minimize(obj, h0, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 800, 'ftol': 1e-14, 'maxls': 60})
    return make_feasible(np.clip(res.x, 0, 1), n)

def slsqp_opt(h0, n, alpha=40):
    dx = 2.0 / n
    target = n / 2.0

    def obj(h):
        c = compute_c(h, dx)
        m = np.max(c)
        return m + (1.0 / alpha) * np.log(np.mean(np.exp(alpha * (c - m))))

    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': 500, 'ftol': 1e-10})
    return make_feasible(np.clip(res.x, 0, 1), n)

def binary_project(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 70

    h_parent = None
    n_parent = 0
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except Exception:
        pass

    def elapsed():
        return time.time() - t0

    def update(h, n):
        nonlocal best_h, best_c5, best_n
        c = c5_value(h)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
            best_n = n

    def try_opt(h0, n, method='lbfgs', alpha=50):
        if elapsed() > budget_s * 0.92:
            return
        h0 = make_feasible(h0, n)
        c0 = c5_value(h0)
        if c0 < best_c5:
            update(h0, n)
        try:
            if method == 'lbfgs':
                ho = lbfgs_opt(h0, n, alpha)
            else:
                ho = slsqp_opt(h0, n, alpha)
            c_new = c5_value(ho)
            if c_new < best_c5:
                update(ho, n)
            return ho
        except Exception:
            return None

    # Determine grids
    grids = set([40, 50, 60, 70, 80, 100])
    if n_parent > 0 and 20 <= n_parent <= 200:
        grids.add(n_parent)
    grids = sorted(grids)

    for n in grids:
        if elapsed() > budget_s * 0.9:
            break
        dx = 2.0 / n
        starts = []

        if h_parent is not None and n_parent > 0:
            if n == n_parent:
                starts.append(h_parent.copy())
            else:
                xo = np.linspace(0, 2, n_parent, endpoint=False)
                xn = np.linspace(0, 2, n, endpoint=False)
                for i in range(n):
                    j = int(i * n_parent / n)
                    j = min(j, n_parent - 1)
                xp = np.linspace(0, 2, n, endpoint=False)
                from scipy.interpolate import CubicSpline
                try:
                    cs = CubicSpline(xo, h_parent)
                    starts.append(make_feasible(cs(xp), n))
                except Exception:
                    f = np.interp(xp, xo, h_parent)
                    starts.append(make_feasible(f, n))

        starts.append(0.5 * np.ones(n))
        if n % 2 == 0:
            hb = np.zeros(n)
            hb[::2] = 1.0
            starts.append(hb)
        if n % 4 == 0:
            hb2 = np.zeros(n)
            hb2[:n // 4] = 1.0
            hb2[n // 2: 3 * n // 4] = 1.0
            starts.append(hb2)
        for _ in range(3):
            starts.append(make_feasible(rng.uniform(0.25, 0.75, n), n))

        for h0 in starts:
            if elapsed() > budget_s * 0.9:
                break
            ho = try_opt(h0, n, 'lbfgs', 50)
            if ho is not None and elapsed() < budget_s * 0.85:
                try_opt(ho, n, 'lbfgs', 100)
            if elapsed() < budget_s * 0.8:
                try_opt(h0, n, 'slsqp', 40)

        # Multi-start perturbation from best at this n
        if best_n == n and elapsed() < budget_s * 0.85:
            for _ in range(4):
                if elapsed() > budget_s * 0.9:
                    break
                hp = best_h + rng.normal(0, 0.06, n)
                try_opt(hp, n, 'lbfgs', 60)

        # Binary projection attempt
        if best_n == n and elapsed() < budget_s * 0.85:
            hb = binary_project(best_h, n)
            try_opt(hb, n, 'lbfgs', 40)

    # Final SLSQP polish on best
    if best_h is not None and elapsed() < budget_s * 0.9:
        ho = try_opt(best_h, best_n, 'slsqp', 50)
        if ho is not None and elapsed() < budget_s * 0.9:
            try_opt(ho, best_n, 'lbfgs', 80)

    if best_h is None:
        best_h = 0.5 * np.ones(70)
        best_n = 70
        best_c5 = c5_value(best_h)

    return best_h, best_c5, best_n
