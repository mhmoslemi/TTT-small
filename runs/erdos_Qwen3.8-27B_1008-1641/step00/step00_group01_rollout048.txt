from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def obj_wrapper(x, n_points):
    return compute_c5(x, n_points)

def constraint_fn(x, target_sum):
    return np.sum(x) - target_sum

def repair_sum(h, target_sum):
    """Repair h to have sum = target_sum while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s == 0:
        h = np.full(len(h), target_sum / len(h))
        return h
    if abs(s - target_sum) < 1e-12:
        return h
    # Iteratively adjust
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        adj = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

    rng = np.random.RandomState(seed)
    
    n = len(initial_h_values)
