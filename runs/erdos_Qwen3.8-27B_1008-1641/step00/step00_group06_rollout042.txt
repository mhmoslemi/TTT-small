import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(95, 0.5)

    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h = fix_sum(h0, target)
    best_c5 = compute_c5(best_h)

    con = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
            'jac': lambda x: np.ones(n)}]

    def remaining(frac=1.0):
        return t0 + budget_s * frac - time.time()

    # Phase 1: SLSQP from initial
    if remaining(0.10) > 5:
        res = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': 3000, 'ftol': 1e-15})
        if res.x is not None:
            h_c = fix_sum(res.x, target)
            v = compute_c5(h_c)
            if v < best_c5:
                best_h, best_c5 = h_c, v

    # Phase 2: SLSQP with random perturbations
    n_phase2 = 0

        progress = 1.0 - remaining(0.70) / (budget_s * 0.60)
        sigma = 0.02 + 0.12 * (1.0 - progress)
        h_p = best_h + rng.normal(0.0, sigma, n)
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = fix_sum(h_p, target)

        method = 'SLSQP' if n_phase2 % 3 != 0 else 'trust-constr'
        try:

                lc = LinearConstraint(np.ones((1, n)), target, target)
                res = minimize(compute_c5, h_p, method='trust-constr',
                               bounds=bounds, constraints=[lc],
                               options={'maxiter': 300, 'gtol': 1e-8})
        except Exception:
            res = None

        if res is not None and res.x is not None:

    # Phase 3: Local search via pairwise transfers

                h_k = fix_sum(np.clip(h_k, 0.0, 1.0), target)

                best_c5 = compute_c5(best_h)
                stall = 0

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
