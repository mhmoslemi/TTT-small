    """Compute subgradient of C5 at h for given active lags"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag_idx in active_lags:
        lag = lag_idx - (n - 1)
        if lag >= 0:
            for m in range(lag, n):
                grad[m] += 1 - h[m - lag]
            for m in range(0, n - lag):
                grad[m] -= h[m + lag]
        else:

                grad[m] += 1 - h[m + m_lag]
            for m in range(m_lag, n):
                grad[m] -= h[m - m_lag]
    # Average over active lags
    grad /= len(active_lags)

    return grad

Actually, the numpy documentation says:
c_k = ∑_n a_{n+k} · v̄_n

For 'full' mode, the length is M+N-1 where M, N are the lengths. For same length n: 2n-1.

The indexing: result has indices 0 to 2n-2.
result[k] corresponds to the correlation at lag (k - (n-1)).

For lag l = k - (n-1):
- If l ≥ 0: result[k] = ∑_{n=0}^{n-1-l} h[n+l] * (1-h[n])
- If l < 0: result[k] = ∑_{n=-l}^{n-1} h[n] * (1-h[n+l])

Let me verify with a small example. h = [h0, h1, h2], n=3.
np.correlate(h, 1-h, 'full') has 5 elements (indices 0-4).

Index 0 (lag = -2): ∑ h[n] * (1-h[n-2]) for n=2 to 2: h[2]*(1-h[0])
Index 1 (lag = -1): ∑ h[n] * (1-h[n-1]) for n=1 to 2: h[1]*(1-h[0]) + h[2]*(1-h[1])
Index 2 (lag = 0): h[0]*(1-h[0]) + h[1]*(1-h[1]) + h[2]*(1-h[2])
Index 3 (lag = 1): ∑ h[n+1] * (1-h[n]) for n=0 to 1: h[1]*(1-h[0]) + h[2]*(1-h[1])
Index 4 (lag = 2): h[2]*(1-h[0])

So for lag l ≥ 0:
result = ∑_{n=0}^{n-1-l} h[n+l] * (1-h[n])

∂result/∂h[m]:
- h[m] appears as h[n+l] when n = m-l, valid if 0 ≤ m-l ≤ n-1-l, i.e., l ≤ m ≤ n-1
  Contribution: (1 - h[m-l])
- h[m] appears in (1-h[n]) when n = m, valid if 0 ≤ m ≤ n-1-l
  Contribution: -h[m+l]

So: ∂result/∂h[m] = (1-h[m-l]) * I[l≤m≤n-1] - h[m+l] * I[0≤m≤n-1-l]

For lag l < 0, let m = -l > 0:
result = ∑_{n=m}^{n-1} h[n] * (1-h[n-m])

∂result/∂h[m']:
- h[m'] appears as h[n] when n = m', valid if m ≤ m' ≤ n-1
  Contribution: (1 - h[m'-m])
- h[m'] appears in (1-h[n-m]) when n-m = m', i.e., n = m'+m, valid if m ≤ m'+m ≤ n-1, i.e., 0 ≤ m' ≤ n-1-m
  Contribution: -h[m'+m]

So: ∂result/∂h[m'] = (1-h[m'-m]) * I[m≤m'≤n-1] - h[m'+m] * I[0≤m'≤n-1-m]

This is symmetric to the l ≥ 0 case with roles of l and -l swapped (as expected).

def _subgrad_step(h, step_size, nt):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    mx = np.max(corr)
    active = np.where(corr >= mx - 1e-8)[0]
    
    grad = np.zeros(n)
    for idx in active:
        l = idx - (n - 1)
        if l >= 0:
            for m in range(l, n):
                grad[m] += 1 - h[m - l]
            for m in range(0, n - l):
                grad[m] -= h[m + l]
        else:
            ml = -l
            for m in range(ml, n):
                grad[m] += 1 - h[m - ml]
            for m in range(0, n - ml):
                grad[m] -= h[m + ml]
    
    grad /= len(active)
    grad *= dx
    
    # Gradient step (descending, so subtract)
    h_new = h - step_size * grad
    h_new = _proj(h_new, nt)
    return h_new

def _proj(h, nt):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = nt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            c = m.sum()
        else:
            m = h > 0.0
            c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    d = nt - h.sum()
    if abs(d) > 1e-15:
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = m.sum()
        if c:
            h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _active_lags(h, tol=1e-10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, 'full')
    mx = np.max(corr)

            for m in range(0, n - ml):
                g[m] -= h[m + ml]
    g /= max(1, len(active_idx))
