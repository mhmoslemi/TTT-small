import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full") * dx
    cm = np.max(corr)
    ex = np.exp(beta * (corr - cm))
    S = ex.sum()
    w = ex / S
    f = cm + np.log(S) / beta
    mags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            mk = mags[k]
            j1 = i + mk
            j2 = i - mk
            if 0 <= j1 < n:
                s += w[k] * (1.0 - h[j1])
            if 0 <= j2 < n:
                s -= w[k] * h[j2]
        grad[i] = dx * s
    return f, grad


def obj_pen(h, beta, n, pen):
    f, g = obj_grad(h, beta)
    d = h.sum() - n * 0.5
    f += pen * d * d
    g = g + 2.0 * pen * d
    return f, g


def optimize_lbfgs(h0, beta, n, pen=1e4, maxiter=400):
    x0 = project_feasible(h0.astype(float).copy(), n * 0.5)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, x0, args=(beta, n, pen), method='L-BFGS-B',
                       jac=True, bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        out = project_feasible(res.x, n * 0.5)
    except Exception:
        out = x0
    return out


def random_feasible(n, rng):
    return project_feasible(rng.uniform(0.0, 1.0, n), n * 0.5)


def local_polish(h, n, n_iter=50):
    """Coordinate descent polish: nudge to reduce max correlation."""
    dx = 2.0 / n
    ts = n * 0.5
    h = h.copy()
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = np.argmax(corr)
        m = k_star - (n - 1)
        improved = False
        for j in range(n):
            j1 = j + m
            j2 = j - m
            for delta in [0.001, -0.001, 0.005, -0.005]:
                h[j] += delta
                if h[j] < 0.0 or h[j] > 1.0:
                    h[j] -= delta
                    continue
                h2 = project_feasible(h, ts)
                c5_new = compute_c5(h2)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-9:
                    h = h2
                    improved = True
                    break
                h[j] -= delta
            if improved:
                break
        if not improved:
            break
    return project_feasible(h, ts)
