def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff2 / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k):
    """Gradient of c_k (before multiplying by dx)."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        m = -k
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

    """Log-sum-exp smoothed objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

The objective is: min_h max_k sum_i h[i]*(1-h[i+k]) * dx

subject to: 0 ≤ h[i] ≤ 1, sum h[i] = n/2

Note that sum_i h[i]*(1-h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k] = n/2 - sum_i h[i]*h[i+k] (when the sums are over valid ranges... actually not quite, because the valid ranges differ for different k).

For k=0: sum h[i](1-h[i]) = n/2 - sum h[i]^2
For k>0: sum_{i=0}^{n-1-k} h[i](1-h[i+k])

The n/2 - sum h[i]^2 term for k=0 is always ≥ 0 (by Cauchy-Schwarz, since sum h = n/2, sum h^2 ≥ (n/2)^2/n = n/4, so n/2 - sum h^2 ≤ n/4). Wait, actually n/2 - sum h^2 could be negative... no. sum h^2 ≤ (sum h)^2 / n = n/4 * ... hmm.

By Cauchy: (sum h)^2 ≤ n * sum(h^2), so sum(h^2) ≥ (n/2)^2/n = n/4.
So n/2 - sum(h^2) ≤ n/2 - n/4 = n/4.
And n/2 - sum(h^2) ≥ n/2 - sum(h) (since h[i]^2 ≤ h[i] for h[i] in [0,1]) = n/2 - n/2 = 0.

So lag 0 contribution is in [0, n/4]. Multiplied by dx = 2/n: [0, 1/2].

    _g_tau = tau
    bd = [(0.0, 1.0)] * n
    cs = ({'type': 'eq', 'fun': _c_fun, 'jac': _c_jac},)
    try:
        r = minimize(_obj, np.clip(h0, 0, 1), method='SLSQP',
                     jac=_grad, bounds=bd, constraints=cs,
                     options={'maxiter': mx, 'ftol': 1e-12})
        hh = project_h(r.x, n)
        return hh, compute_c5(hh)
    except Exception:
        return h0, compute_c5(h0)

def pgd_anneal(h0, n, iters, tk, lt, rng):
    h = h0.astype(float).copy()
    bh = h.copy()
    bc = compute_c5(h)
    lr = 0.01
    st = 0
    for i in range(iters):
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        ti = np.argsort(c)[-tk:]
        tv = c[ti] - float(c.max())
        w = np.exp(tv / lt)
        w /= w.sum()
        g = np.zeros(n)
        for j in range(tk):
            g += w[j] * lag_grad(h, int(ti[j]) - (n - 1))
        g *= dx
        step = lr * g
        if i % 200 == 199:
            step += rng.normal(0, 0.02, n)
        h = project_h(h - step, n)
        cv = compute_c5(h)
        if cv < bc:
            bc, bh, st = cv, h.copy(), 0
        else:
            st += 1
        if st > 120:
            lr = max(lr * 0.6, 0.0005)
            st = 0
        if i % 300 == 299:
            lr *= 0.96
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    bh = h0.copy()
    bc = compute_c5(h0)

    # Phase 1: PGD from initial
    if budget_s - (time.time() - t0) > 150:
        h, c = pgd(h0, n, 5000, 3, 0.01)
        if c < bc:
            bh, bc = h, c

    # Phase 2: SLSQP polish
    if budget_s - (time.time() - t0) > 50:
        h, c = slsqp(bh, n, 0.03, 120)
        if c < bc:
            bh, bc = h, c

    # Phase 3: PGD with wider top-k
    if budget_s - (time.time() - t0) > 150:
        h, c = pgd(bh, n, 5000, 5, 0.03)
        if c < bc:
            bh, bc = h, c

    # Phase 4: SLSQP tighter
    if budget_s - (time.time() - t0) > 40:
        h, c = slsqp(bh, n, 0.01, 150)
        if c < bc:
            bh, bc = h, c

    # Phase 5: PGD with annealing from perturbed best
    if budget_s - (time.time() - t0) > 120:
        hp = project_h(bh + rng.normal(0, 0.05, n), n)
        h, c = pgd_anneal(hp, n, 3000, 4, 0.02, rng)
        if c < bc:
            bh, bc = h, c

    # Phase 6: SLSQP
    if budget_s - (time.time() - t0) > 30:
        h, c = slsqp(bh, n, 0.008, 200)
        if c < bc:
            bh, bc = h, c

    # Phase 7: Random restarts
    while budget_s - (time.time() - t0) > 120:
        hr = project_h(rng.uniform(0, 1, n), n)
        h, c = pgd(hr, n, 2000, 3, 0.02)
        if c < bc:
            bh, bc = h, c
        # Also try structured init
        if budget_s - (time.time() - t0) > 100:
            hs = np.zeros(n)
            phi = 0.6180339887
            for i in range(n // 2):
                hs[int((i * phi * n + rng.uniform(0, 1)) % n)] = 1.0
            hs = project_h(hs + rng.normal(0, 0.1, n), n)
            h, c = pgd(hs, n, 1500, 3, 0.02)
            if c < bc:
                bh, bc = h, c

    # Final polish
    if budget_s - (time.time() - t0) > 20:
        h, c = slsqp(bh, n, 0.005, 200)
        if c < bc:
            bh, bc = h, c

    if budget_s - (time.time() - t0) > 10:
        h, c = pgd(bh, n, 2000, 5, 0.005)
        if c < bc:
            bh, bc = h, c

    return bh, bc, n
