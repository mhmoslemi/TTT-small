    idx_plus = np.arange(n) + lag
    valid_plus = (idx_plus >= 0) & (idx_plus < n)
    grad[valid_plus] += 1.0 - h[idx_plus[valid_plus]]
    
    idx_minus = np.arange(n) - lag
    valid_minus = (idx_minus >= 0) & (idx_minus < n)
    grad[valid_minus] -= h[idx_minus[valid_minus]]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def c5_grad(h):
    """Objective and subgradient of C5"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    c5 = dx * corr[k_star]
    
    axes = np.arange(n)
    grad = np.zeros(n)
    
    idx_m = axes - lag
    vm = (idx_m >= 0) & (idx_m < n)
    grad[vm] += 1.0 - h[idx_m[vm]]
    
    idx_p = axes + lag
    vp = (idx_p >= 0) & (idx_p < n)
    grad[vp] -= h[idx_p[vp]]
    
    grad *= dx
    return c5, grad

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def optimize_slsqp(h0, target_sum, time_budget):
    n = len(h0)
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    best_h = h0.copy()
    best_c = compute_c5(h0)
    
    start = time.time()
    x = h0.copy()
    
    for attempt in range(3):
        if time.time() - start > time_budget:
            break
        try:
            res = minimize(c5_grad, x, jac=True, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-10})
            x = project_feasible(res.x, target_sum)
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
        except:
            pass
        
        # Perturb and retry if stuck
        if time.time() - start > time_budget * 0.6:
            x = best_h + np.random.normal(0, 0.02, n)
            x = project_feasible(x, target_sum)
    
    return best_h, best_c

def optimize_subgrad(h0, target_sum, time_budget):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0.05
    no_improve = 0
    start = time.time()
    
    while time.time() - start < time_budget:
        c, g = c5_grad(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            step *= 0.8
            if step < 1e-6:
                break
            no_improve = 0
        
        h_new = h - step * g
        h_new = project_feasible(h_new, target_sum)
        
        if compute_c5(h_new) <= c + 1e-10:
            h = h_new
        else:
            step *= 0.95
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    
    best_h = None
    best_c = float('inf')
    
    n_init = 95
    try:
        n_init = len(initial_h_values)

def compute_c5(h):
    n = len(h)

def project_feasible(h, ts):
    h = h.astype(float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > ts:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_and_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    val = dx * corr[k_star]
    axes = np.arange(n)
    grad = np.zeros(n)
    idx_m = axes - lag
    vm = (idx_m >= 0) & (idx_m < n)
    grad[vm] += 1.0 - h[idx_m[vm]]
    idx_p = axes + lag
    vp = (idx_p >= 0) & (idx_p < n)
    grad[vp] -= h[idx_p[vp]]
    grad *= dx
    return val, grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,

    best_h = project_feasible(h0.copy(), ts)

                break
            tb_sqp = min(40.0, rem * 0.2)
            h2, c2_val = slsqp_optimize(h1, ts, tb_sqp)
            if c2_val < best_c:
                best_c = c2_val
                best_h = h2.copy()

            # Second round of coord descent from SLSQP result
            rem = budget_s - (time.time() - t0)
            if rem < 15:
                break
            tb_cd2 = min(40.0, rem * 0.15)
            h3, c3_val = coord_descent_full(h2, ts, tb_cd2)
            if c3_val < best_c:
                best_c = c3_val
                best_h = h3.copy()

    if best_h is None:
        best_h = np.full(n_init, 0.5)
        best_c = compute_c5(best_h)

    return best_h, best_c, len(best_h)
