import numpy as np
import time

def proj_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    return (2.0 / n) * np.max(np.correlate(h, 1.0 - h, mode="full"))

def compute_grad(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    e = np.exp(np.clip((c - m) * beta, -200, 200))
    p = e / e.sum()
    grad = np.zeros(n)
    for s in range(n):
        d_lo = s - (n - 1)
        d_hi = s
        vd = np.arange(d_lo, d_hi + 1)
        t1 = np.dot(p[n - 1 + vd], 1.0 - h[s - vd])
        d_lo2 = -s
        d_hi2 = n - 1 - s
        vd2 = np.arange(d_lo2, d_hi2 + 1)
        t2 = np.dot(p[n - 1 + vd2], h[s + vd2])
        grad[s] = dx * (t1 - t2)
    return grad

def pgd_optimize(h, n, max_iter, t_start, t_budget, seed):

    h = proj_feasible(h.copy(), n)

        beta = 50.0 + 500.0 * min(1.0, it / (0.85 * max_iter))
        g = compute_grad(h, n, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-12:

        step = min(1.0, 0.5 / (gn + 1e-8))

    h_bin = (h > 0.5).astype(float)

    if n_ones > target:
