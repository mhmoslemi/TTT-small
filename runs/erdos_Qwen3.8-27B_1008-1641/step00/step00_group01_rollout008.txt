    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # Clip for numerical stability

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj_func(x, n, alpha):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500.0, 0.0)
    lse = m + np.log(np.sum(np.exp(s))) / alpha
    return dx * lse

def pen_obj_func(x, n, alpha, lam):
    val = smooth_obj_func(x, n, alpha)
    viol = x.sum() - n / 2.0
    return val + lam * viol * viol

def con_eq(x):
    return x.sum() - len(x) / 2.0

def con_eq_jac(x):
    return np.ones(len(x))

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-14:
        return h
    idx = np.argsort(np.abs(h - 0.5))
    for i in range(min(5, n)):
        j = idx[i]
        h[j] = np.clip(h[j] + diff, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) < 1e-14:
            break
    return np.clip(h, 0.0, 1.0)

def polish_slsqp(h, n, alpha, maxiter=150):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
    res = minimize(smooth_obj_func, h, args=(n, alpha),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return fix_sum(res.x, n)

                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return fix_sum(res.x, n)
