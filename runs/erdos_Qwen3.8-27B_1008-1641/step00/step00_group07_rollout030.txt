from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1-h, mode="full")) * dx)

def proj(h, n):
    h = np.clip(h, 0, 1)
    t = n / 2.0
    for _ in range(30):
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
        if s == 0:
            h[:] = 0.5
            break
        h = np.clip(h * t / s, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-13:
        m = (h > 1e-10) & (h < 1 - 1e-10)
        if m.any():
            h[m] += d / m.sum()
            h = np.clip(h, 0, 1)
    return h

def sg(h, alpha):
    """Smooth objective and gradient"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full")
    m = c.max()
    sh = alpha * (c - m)
    eu = np.exp(sh).sum()
    obj = dx * m + dx / alpha * np.log(eu)
    p = np.exp(sh) / eu
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        g[idx] += pk * (1 - h[idx + s])
        np.add.at(g, idx + s, -pk * h[idx])
    return obj, dx * g

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bounds = [(0, 1)] * n
    
    # Phase 1: SLSQP with increasing alpha
    for alpha in [30, 60, 120, 240, 480]:
        if time.time() - t0 > budget_s - 60:
            break

            return sg(x, alpha)[0]
        def objGrad(x):
            return sg(x, alpha)[1]

            c5 = compute_c5(h)

                best_h = h.copy()
    
    # Phase 2: Nelder-Mead on actual max
