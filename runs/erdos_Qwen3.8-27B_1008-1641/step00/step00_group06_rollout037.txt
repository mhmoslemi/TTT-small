import numpy as np
from scipy.optimize import minimize
import time

def _proj(x, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if np.clip(x + mid, 0.0, 1.0).sum() < tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def _lse(c, a):
    m = c.max()
    if m > 10.0:
        return 1e5
    return m + float(np.log(np.exp((c - m) * a).sum() / len(c))) / a

def _obj(x, tgt, dx, alpha):
    hh = _proj(x, tgt)
    cv = np.correlate(hh, 1.0 - hh, mode='full') * dx
    return _lse(cv, alpha)
