def project_to_simplex(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)

    diff = target_sum - current_sum

        if abs(diff) < 1e-12:

        # Add to elements with most room

        if diff > 0:

        if not np.any(mask):

    # Log-sum-exp for numerical stability

    # Softmax weights

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_grad(h, tau):
    """Returns (smooth_objective, smooth_gradient)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_c = np.max(corr)
    # logsumexp
    z = (corr - max_c) / tau
    z_max = np.max(z)
    log_sum = z_max + np.log(np.sum(np.exp(z - z_max)))
    obj = (max_c + tau * log_sum) * dx
    
    # softmax weights
    w = np.exp(z - z_max)
    w /= np.sum(w)
    
    # gradient
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        k = idx - (n - 1)
        wi = w[idx]
        for i in range(n):
            v = 1.0
            j1 = i + k
            if 0 <= j1 < n:
                v -= h[j1]
            j2 = i - k
            if 0 <= j2 < n:
                v -= h[j2]
            grad[i] += wi * v
    grad *= dx
    return obj, grad

def _pgd_optimize(h0, n, budget_s=300, verbose=False):
    """Projected gradient descent optimization"""
    h = h0.copy().astype(float)
    h = _project(h, n)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    t0 = time.time()
    
    # Adaptive temperature and step size
    tau = 0.05
    lr = 0.01
    patience = 500
    no_improve = 0
    
    it = 0
    while time.time() - t0 < budget_s:
        obj, grad = _smooth_obj_grad(h, tau)
        # Gradient step
        h_new = h - lr * grad
        h_new = _project(h_new, n)
        
        # Check improvement
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        # Anneal
        if no_improve > patience and tau > 0.001:
            tau *= 0.7
            lr *= 0.7
            no_improve = 0
        
        it += 1
        if it % 1000 == 0 and verbose:
            elapsed = time.time() - t0
            print(f"  it={it}, c5={best_c5:.6f}, tau={tau:.4f}, elapsed={elapsed:.1f}s")
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    mc = np.max(corr)
    z = (corr - mc) / tau
    zm = np.max(z)
    lse = zm + np.log(np.sum(np.exp(z - zm)))
    obj = (mc + tau * lse) * dx
    w = np.exp(z - zm)
    s = np.sum(w)
    if s < 1e-300:
        return obj, np.zeros(n)
    w /= s
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            m = k
            if m < n:
                grad[:n - m] += wi * (1.0 - h[m:])
                grad[m:] -= wi * h[:n - m]
        else:
            m = -k
            if m < n:
                grad[m:] += wi * (1.0 - h[:n - m])
                grad[:n - m] -= wi * h[m:]
    grad *= dx
    return obj, grad

def _pgd_optimize(h0, budget_s=200.0):
    n = len(h0)
    h = _project(h0.copy().astype(float), n)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    t0 = time.time()
    tau = 0.05
    lr = 0.02
    no_improve = 0
    it = 0
    while time.time() - t0 < budget_s:
        obj, grad = _smooth_obj_grad(h, tau)
        h_new = h - lr * grad
        h_new = _project(h_new, n)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5 - 1e-9:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 400 and tau > 0.002:
            tau *= 0.75
            lr *= 0.75
            no_improve = 0
        elif no_improve > 1500:
            tau = max(tau * 0.5, 0.001)
            lr *= 0.5
            no_improve = 0
        it += 1
    return best_h, best_c5

def _scipy_smooth_minimize(h0, tau, budget_s=60.0):
    n = len(h0)
    h0 = _project(h0.copy().astype(float), n)
    t0 = time.time()
    calls = [0]
    
    def obj_wrapper(x):
        calls[0] += 1
        if time.time() - t0 > budget_s:
            return 1e10
        x = _project(x, n)
        o, _ = _smooth_obj_grad(x, tau)
        return o
    
    res = minimize(obj_wrapper, h0, method='Nelder-Mead',
                   options={'maxiter': 5000, 'xatol': 1e-5, 'fatol': 1e-7})
    h_res = _project(res.x.astype(float), n)
    c5_res = _compute_c5(h_res)
    return h_res, c5_res

def run(seed=42, budget_s=1100, **kwargs):

        hh, cc = _pgd_optimize(h_pert, budget_s=tb)

            best_h = hh
            best_c5 = cc
            n = n_small
            x_orig = np.linspace(0, 2, n, endpoint=False)
        rem = budget_s - (time.time() - t0) - 30.0

    # Final: make sure best is valid
    best_h = _project(best_h, len(best_h))
    final_c5 = _compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
