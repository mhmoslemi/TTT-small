import time

import numpy as np

_SLQ_N = 0
_SLQ_TARGET = 0.0


def count_max(h):
    n = h.shape[0]
    oth = 1.0 - h
    best = float(np.dot(h, oth))
    for k in range(1, n):
        v = float(np.dot(h[:n-k], oth[k:]))
        if v > best:
            best = v
        v = float(np.dot(h[k:], oth[:n-k]))
        if v > best:
            best = v
    return best


def lag_counts_array(h):
    n = h.shape[0]
    oth = 1.0 - h
    c = np.empty(2*n - 1)
    p = 0
    for k in range(1, n):
        c[p] = float(np.dot(h[k:], oth[:n-k]))
        p += 1
    c[p] = float(np.dot(h, oth))
    p += 1
    for k in range(1, n):
        c[p] = float(np.dot(h[:n-k], oth[k:]))
        p += 1
    return c


def project_box_sum(y, target):
    x = np.array(y, dtype=float)
    if x.ndim != 1:
        x = x.ravel()
    n = x.size
    if n == 0:
        return x
    x = np.clip(x, 0.0, 1.0)
    fixed = np.zeros(n, dtype=bool)
    for _ in range(2*n + 10):
        free = ~fixed
        cnt = int(np.count_nonzero(free))
        if cnt == 0:
            break
        need = target - float(x[fixed].sum())
        s = float(x[free].sum())
        delta = (need - s) / float(cnt)
        if delta > 1e-14:
            over = x[free] + delta > 1.0
            if np.any(over):
                idx = np.where(free)[0][over]
                x[idx] = 1.0
                fixed[idx] = True
                continue
        elif delta < -1e-14:
            under = x[free] + delta < 0.0
            if np.any(under):
                idx = np.where(free)[0][under]
                x[idx] = 0.0
                fixed[idx] = True
                continue
        x[free] += delta
        break
    x = np.clip(x, 0.0, 1.0)
    for _ in range(4):
        diff = target - float(x.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idxs = np.where(x < 1.0 - 1e-12)[0]
            cap = 1.0 - x[idxs]
        else:
            idxs = np.where(x > 1e-12)[0]
            cap = x[idxs]
        if idxs.size == 0:
            break
        j = int(np.argmax(cap))
        x[idxs[j]] = min(1.0, max(0.0, x[idxs[j]] + diff))
    if abs(float(x.sum()) - target) > 1e-7:
        x = np.full(n, target / float(n))
    return np.clip(x, 0.0, 1.0)


def gradient_mask(h, active):
    n = h.shape[0]
    g = np.zeros(n)
    active = np.asarray(active)
    if active[n-1]:
        g += (1.0 - 2.0*h)
    for k in range(1, n):
        if active[n-1-k]:
            g[k:] += 1.0 - h[:-k]
            g[:-k] -= h[k:]
        if active[n-1+k]:
            g[:n-k] += 1.0 - h[k:]
            g[k:] -= h[:n-k]
    s = float(active.sum())
    if s > 0:
        g /= s
    return g


def gradient_improve(h, best_h, best_c, deadline, rng):
    n = len(h)
    target = 0.5 * float(n)
    fracs = (1e-6, 1e-5, 2e-5, 5e-5, 1e-4, 2e-4, 5e-4, 1e-3, 2e-3, 5e-3, 1e-2)
    stall = 0
    while time.time() < deadline:
        cvals = lag_counts_array(h)
        C = float(np.max(cvals))
        if C < best_c - 1e-13:
            best_c = C
            best_h = h.copy()
        argmax = int(np.argmax(cvals))
        active_all = cvals >= C - max(1e-9, 1e-4 * C)
        if not np.any(active_all):
            active_all[argmax] = True
        active_arg = np.zeros_like(active_all)
        active_arg[argmax] = True
        directions = (gradient_mask(h, active_all), gradient_mask(h, active_arg))
        improved = False
        for g in directions:
            g = g - g.mean()
            scale = float(np.max(np.abs(g)))
            if scale < 1e-15:
                continue
            gdir = g / scale
            dir_best_c = best_c
            dir_best_h = None
            for frac in fracs:
                h2 = project_box_sum(h - frac * gdir, target)
                c2 = count_max(h2)
                if c2 < dir_best_c - 1e-12:
                    dir_best_c = c2
                    dir_best_h = h2.copy()
            if dir_best_h is not None:
                best_c = dir_best_c
                best_h = dir_best_h.copy()
                h = dir_best_h.copy()
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return h, best_h, best_c

def binary_threshold(h):
    n = len(h)
    target = int(round(0.5 * float(n)))
    if n % 2 == 1:
        target = n // 2
    bits = np.zeros(n, dtype=float)
    idx = np.argsort(h)[::-1][:target]
    bits[idx] = 1.0
    return bits


def interpolate_parent(p, n):
    p = np.asarray(p, dtype=float).ravel()
    if n == p.size:
        return project_box_sum(p, 0.5 * float(n))
    xp = np.linspace(0.0, 1.0, p.size)
    xn = np.linspace(0.0, 1.0, n)
    return project_box_sum(np.interp(xn, xp, p), 0.5 * float(n))


def make_sine_candidates(n, rng, limit=4):
    target = 0.5 * float(n)
    x = np.linspace(0.0, 2.0, n)
    out = []
    idx = 0
    freqs = (0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 7.0, 10.0)

    amps = (0.25, 0.40, 0.50)
    for f in freqs:
        for ph in phases:
            for amp in amps:
                h = 0.5 + amp * np.sin(2.0 * math.pi * f * x + ph)
                h = project_box_sum(h, target)

        h = project_box_sum(h, target)
        out.append((count_max(h), idx, h))
        idx += 1
    out.sort()
    res = []
    for item in out[:limit]:

def make_random_binary(n, rng, limit=3, parent=None):

        return []

    idx = 0
    base = None

def _slq_obj(x):
    return count_max(x)


def _slq_con(x):
    return float(np.sum(x) - _SLQ_TARGET)


def try_slsqp(h, maxiter=40):
    global _SLQ_N, _SLQ_TARGET
    n = len(h)
    target = 0.5 * float(n)
    x0 = project_box_sum(h, target)
    _SLQ_N = n
    _SLQ_TARGET = target
    try:
        from scipy.optimize import minimize
        res = minimize(
            _slq_obj, x0, method='SLSQP',
            constraints=[{'type': 'eq', 'fun': _slq_con}],
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': int(maxiter), 'ftol': 1e-12, 'disp': False}
        )
        if hasattr(res, 'x') and res.x.size == n:
            h2 = project_box_sum(res.x, target)
            h2[target * 0]  # no-op
        else:
            h2 = x0
    except Exception:
        h2 = x0
    return h2, count_max(h2)


def refine_pairwise(h, deadline, rng):
    n = len(h)
    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c = count_max(h)
    sweep = 0

        sweep += 1
        improved = False

            for j in range(i + 1, n):

            if span < 1e-13:

            vals = [a, low, high, 0.0, 1.0, 0.5]
            for f in frac_set:
                vals.append(low + f * span)
            vals.append(rng.uniform(low, high))
            vals.append(rng.uniform(low, high))
            local_best_c = best_c
            local_val = a
            for val in vals:
                val = float(val)

def random_perturb(h, rng, moves=8):
    n = len(h)
    h = np.array(h, dtype=float).copy()
    for _ in range(moves):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        a = float(h[i])
        b = float(h[j])
        s = a + b
        low = max(0.0, s - 1.0)
        high = min(1.0, s)
        if high - low < 1e-13:
            continue
        val = low + (high - low) * rng.random()
        h[i] = val
        h[j] = s - val
    return project_box_sum(h, 0.5 * float(n))


def anneal_polish(h, deadline, rng):

    c = count_max(h)

        j = int(rng.integers(0, n))

            val = low + span * rng.random()

        c2 = count_max(h)
        diff = c2 - c

        if diff <= 0.0:

        elif diff < 100.0 * temp:

        if accept:
            c = c2
            if c2 < best_c - 1e-14:
                best_c = c2
                best_h = h.copy()
                no_improve = 0

                no_improve += 1
        else:
            h[i] = a
            h[j] = b
            no_improve += 1
        if no_improve > 25000:
            moves = max(3, n // 8)
            h = random_perturb(best_h, rng, moves)
            c = count_max(h)
            no_improve = 0
            temp = max(0.5 * temp, 1e-6)
        temp *= 0.999
    return best_h, best_c

        n0 = int(parent.size)
        if n0 < 2:
            n0 = 42

    if budget < 5.0:
        if parent is not None:
            h = project_box_sum(parent, 0.5 * float(n0))
        else:
            h = project_box_sum(0.5 + 0.4 * np.sin(2.0 * math.pi * np.linspace(0, 2, n0)), 0.5 * float(n0))
        c5 = count_max(h) * 2.0 / float(len(h))
        return h, c5, len(h)

    best_h = None
    best_c5 = 1e100

    if parent is not None:
        h = project_box_sum(parent, 0.5 * float(n0))
    else:
        cands0 = make_sine_candidates(n0, rng, 1)
        h = cands0[0][1] if cands0 else project_box_sum(np.full(n0, 0.5), 0.5 * float(n0))

    c5 = count_max(h) * 2.0 / float(len(h))
    best_h = h.copy()
    best_c5 = c5

    if parent is not None and deadline - time.time() > 90.0:
        lim = time.time() + min(0.45 * budget, deadline - time.time() - 30.0)
        hh, cc = local_optimize(project_box_sum(parent, 0.5 * float(n0)), lim, rng)
        c5 = cc * 2.0 / float(len(hh))
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    if parent is not None and parent.size % 2 == 0 and time.time() < deadline - 60.0:
        bits = binary_threshold(project_box_sum(parent, 0.5 * float(n0)))
        lim = time.time() + min(0.12 * budget, deadline - time.time() - 20.0)
        hh, cc = local_optimize(bits, lim, rng)
        c5 = cc * 2.0 / float(len(hh))
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    if time.time() < deadline - 80.0:
        cands = make_sine_candidates(n0, rng, 3)
        for _c, hc in cands:
            if time.time() < deadline - 60.0:
                lim = time.time() + min(0.05 * budget, deadline - time.time() - 10.0)
                hh, cc = local_optimize(hc, lim, rng)
                c5 = cc * 2.0 / float(len(hh))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hh.copy()

    if n0 % 2 == 0 and time.time() < deadline - 70.0:
        bcands = make_random_binary(n0, rng, 2, parent)
        for _c, hc in bcands:
            if time.time() < deadline - 50.0:
                lim = time.time() + min(0.04 * budget, deadline - time.time() - 10.0)
                hh, cc = refine_pairwise(hc, lim, rng)
                c5 = cc * 2.0 / float(len(hh))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hh.copy()

    if parent is not None:
        alt_ns = []
        for nn in range(n0, n0 + 16):
            if nn % 2 == 0 and nn != n0:
                alt_ns.append(nn)
        for nn in alt_ns:
            if time.time() < deadline - 55.0:
                lh = interpolate_parent(parent, nn)
                lim = time.time() + min(0.025 * budget, deadline - time.time() - 10.0)
                hh, cc = local_optimize(lh, lim, rng)
                c5 = cc * 2.0 / float(len(hh))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hh.copy()
            if time.time() < deadline - 45.0:
                bc = make_random_binary(nn, rng, 1, None)
                if bc:
                    lim = time.time() + min(0.02 * budget, deadline - time.time() - 10.0)
                    hh, cc = refine_pairwise(bc[0][1], lim, rng)
                    c5 = cc * 2.0 / float(len(hh))
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hh.copy()

    if best_h is not None and time.time() < deadline - 25.0:
        if time.time() < deadline - 35.0:
            lim = time.time() + min(0.25 * (deadline - time.time()), 240.0)
            hh, cc = refine_pairwise(best_h, lim, rng)
            c5 = cc * 2.0 / float(len(hh))
            if c5 < best_c5:
                best_c5 = c5
                best_h = hh.copy()

        if time.time() < deadline - 35.0:
            hh, cc = try_slsqp(best_h, 70)
            c5 = cc * 2.0 / float(len(hh))
            if c5 < best_c5:
                best_c5 = c5
                best_h = hh.copy()

        stall = 0
        while time.time() < deadline - 80.0 and stall < 14:
            nlen = len(best_h)
            moves = max(2, int(nlen * (0.04 + 0.12 * rng.random())))
            hp = random_perturb(best_h, rng, moves)
            chunk = min(35.0, 0.25 * (deadline - time.time() - 20.0))
            if chunk < 5.0:
                break
            hh, cc = refine_pairwise(hp, time.time() + chunk, rng)
            c5 = cc * 2.0 / float(len(hh))
            if c5 < best_c5:
                best_c5 = c5
                best_h = hh.copy()
                stall = 0
            else:
                stall += 1

        if time.time() < deadline - 30.0:
            hh, cc = anneal_polish(best_h, deadline - 18.0, rng)
            c5 = cc * 2.0 / float(len(hh))
            if c5 < best_c5:
                best_c5 = c5
                best_h = hh.copy()

        if time.time() < deadline - 8.0:
            hh, cc = refine_pairwise(best_h, deadline - 2.0, rng)
            c5 = cc * 2.0 / float(len(hh))
            if c5 < best_c5:
                best_c5 = c5
                best_h = hh.copy()

    if best_h is None:
        best_h = project_box_sum(np.full(n0, 0.5), 0.5 * float(n0))

    h_fin = project_box_sum(best_h, 0.5 * float(len(best_h)))
    n_fin = len(h_fin)
    dx = 2.0 / float(n_fin)
    c1 = float(count_max(h_fin)) * dx
    c2 = float(np.max(np.correlate(h_fin, 1.0 - h_fin, mode="full") * dx))
    c5 = max(c1, c2)
    return h_fin, c5, n_fin
