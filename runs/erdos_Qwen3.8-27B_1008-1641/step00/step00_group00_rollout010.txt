import numpy as np
import time
from scipy.optimize import differential_evolution

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = proj < 1.0 - 1e-7
        else:
            mask = proj > 1e-7
        cnt = np.sum(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
    np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def grad_corr_lag(proj, t, n):
    """d(c_t)/d(h) where c_t = sum_i h[i](1-h[i+t])"""
    grad = np.zeros(n)
    j = np.arange(n)
    v1 = (j + t >= 0) & (j + t < n)
    if np.any(v1):
        grad[v1] = (1.0 - proj[j[v1] + t])
    v2 = (j - t >= 0) & (j - t < n)
    if np.any(v2):
        grad[v2] -= proj[j[v2] - t]
    return grad

def soft_descent(h0, target_sum, dx, beta, max_iter, max_step=0.15):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    step = 0.005
    no_imp = 0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        eb = np.exp(beta * (corr - cmax))
        w = eb / np.sum(eb)
        grad = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-8:
                continue
            t = (n - 1) - k
            grad += w[k] * grad_corr_lag(h, t, n)
        grad *= dx
        improved = False
        for s in [step, step * 0.5, step * 0.25, step * 1.5, step * 2.0]:
            if s > max_step:
                continue
            h_new = project_feasible(h - s * grad, target_sum)
            val = compute_c5(h_new, dx)
            if val < best_val - 1e-13:
                h = h_new
                best_h = h_new.copy()
                best_val = val
                improved = True
                break
        if improved:
            no_imp = 0
            step = min(step * 1.08, max_step)
        else:
            no_imp += 1
            step *= 0.85
            if no_imp > 300 or step < 1e-9:
                break
    return best_h, best_val
