def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr), corr

    for _ in range(100):

        if abs(diff) < 1e-10:

        else:
            mask = h > 0.0

                break

            h[mask] += adj

    diff = target - h.sum()

            idx = np.where(h < 1.0)[0]
            h[idx[0]] = min(1.0, h[idx[0]] + diff)
        else:
            idx = np.where(h > 0.0)[0]
            h[idx[0]] = max(0.0, h[idx[0]] + diff)
    return h
