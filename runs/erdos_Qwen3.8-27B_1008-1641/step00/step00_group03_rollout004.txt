                if cnt > half:
                    idx = np.argsort(hb)[:cnt - half]
                    hbin[idx] = 0
                else:
                    idx = np.argsort(-hb)[:half - cnt]
                    hbin[idx] = 1
        elif attempt <= 8:
            # Perturbed binary from best
            hbin = (hb > 0.5).astype(np.int8)
            flip_k = min(3, n // 10)
            for _ in range(flip_k):
                ones = np.where(hbin == 1)[0]
                zeros = np.where(hbin == 0)[0]
                if len(ones) and len(zeros):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    hbin[i], hbin[j] = hbin[j], hbin[i]
        else:
            hbin = np.zeros(n, dtype=np.int8)
            hbin[rng.choice(n, n // 2, replace=False)] = 1

        seg = min(t2, time.time() + budget_s * 0.07)
        res_b, res_v = bin_hill(hbin, n, seg, rng)
        res_c5 = res_v * dx
        if res_c5 < cb:
            cb = res_c5
            hb = res_b.astype(float)

    # Final projection
    hf = project_feasible(hb, n)
    cf = compute_c5(hf, 2.0 / n)
    return hf, cf, n
