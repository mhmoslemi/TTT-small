from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            np.minimum(h, 1.0, out=h)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            np.maximum(h, 0.0, out=h)
    return h

def subgradient(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mv = np.max(corr)
    thresh = mv - 1e-9
    active = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] -= h[s:]
        else:
            u = -s
            if n - u > 0:
                grad[:n-u] += (1.0 - h[u:])
                grad[u:] -= h[:n-u]
    grad /= len(active)
    grad *= dx
    return grad

def pgd_step(h, grad, lr, target_sum):
    h_new = h - lr * grad
    return repair_h(h_new, target_sum)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            np.minimum(h, 1.0, out=h)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            np.maximum(h, 0.0, out=h)
    return h

def subgradient(h, dx):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    mv = float(corr.max())
    thresh = mv - 1e-9
    active = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    for m in active:
        t = m - (n - 1)
        if t >= 0:
            if n - t > 0:
                grad[t:] += omh[:n - t]
                grad[:n - t] -= h[t:]
        else:
            u = -t
            if n - u > 0:
                grad[:n - u] += omh[u:]
                grad[u:] -= h[:n - u]
    grad /= len(active)
    grad *= dx
    return grad

def run_pgd(h0, dx, target_sum, max_iter=8000, lr0=0.06, time_limit=200):

    best_h = h.copy()
    best_score = evaluate_c5(h, dx)
    t0 = time.time()
    lr = lr0
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        grad = subgradient(h, dx)
        h_new = repair_h(h - lr * grad, target_sum)
        sc = evaluate_c5(h_new, dx)
        if sc < best_score:

            best_h = h_new.copy()
            stall = 0
        else:

        h = h_new

            stall = 0
            if lr < 1e-7:
                break
        elif it % 300 == 299:
            lr *= 0.97
    return best_h, best_score

def nm_polish(h0, dx, target_sum, max_iter=1500, time_limit=80):
    t0 = time.time()

    def objective(x):
        if time.time() - t0 > time_limit:
            return 999.0
        hh = repair_h(x, target_sum)
        return evaluate_c5(hh, dx)

    opts = {'maxiter': max_iter, 'xatol': 1e-10, 'fatol': 1e-12, 'adaptive': True}
    res = minimize(objective, h0.copy(), method='Nelder-Mead', options=opts)
    hh = repair_h(res.x, target_sum)
    return hh, evaluate_c5(hh, dx)

    # Phase 3: NM polish
    rem = budget_s - (time.time() - t_start) - 80
    if rem > 5:
        h3, s3 = nm_polish(best_h, dx, target_sum, max_iter=1200, time_limit=max(5, rem * 0.35))
        if s3 < best_score:
            best_h, best_score = h3, s3

    # Phase 4: Perturbation restarts
    for attempt in range(8):
        rem = budget_s - (time.time() - t_start) - 40
        if rem < 25:
            break
        sigmas = [0.04, 0.08, 0.12, 0.06, 0.10, 0.05, 0.15, 0.03]
        h_p = perturb(best_h, target_sum, sigmas[attempt % len(sigmas)])
        h_opt, sc_opt = run_pgd(h_p, dx, target_sum, max_iter=2000, lr0=0.04, time_limit=max(5, rem * 0.3))
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
        rem = budget_s - (time.time() - t_start) - 35
        if rem > 10:
            h_nm, sc_nm = nm_polish(best_h, dx, target_sum, max_iter=800, time_limit=max(3, rem * 0.3))
            if sc_nm < best_score:
                best_h, best_score = h_nm, sc_nm

    # Final polish
    rem = budget_s - (time.time() - t_start) - 5
    if rem > 8:
        h_f, s_f = nm_polish(best_h, dx, target_sum, max_iter=2000, time_limit=rem)
        if s_f < best_score:
            best_h, best_score = h_f, s_f

    best_h = repair_h(best_h, target_sum)
    best_score = evaluate_c5(best_h, dx)
    return best_h, best_score, n
