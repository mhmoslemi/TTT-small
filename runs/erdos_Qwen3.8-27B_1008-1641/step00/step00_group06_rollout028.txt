import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Find the right shift
        # If diff > 0, need to increase some h[i] that are < 1
        # If diff < 0, need to decrease some h[i] that are > 0
        # Use bisection
        lo, hi = -1, 1
        for _ in range(60):
            mid = (lo + hi) / 2
            h_test = np.clip(h + mid, 0, 1)
            if np.sum(h_test) > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo+hi)/2, 0, 1)
    return h

    best_h = initial_h_values.copy()
    n = len(best_h)
    best_c5 = compute_c5(best_h)
    
    # Optimization loop
    from scipy.optimize import minimize
    
    def obj(h_flat):
        h = np.clip(h_flat, 0, 1)
        return compute_c5(h)
    
    from scipy.optimize import NonlinearConstraint
    
    # Try SLSQP
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0,

    bounds = [(0,1) for _ in range(n)]
    
    # Multiple restarts

            h0 = project_feasible(h0, n)

def compute_subgrad(h, n, k_star):
    dx = 2.0 / n
    s = n - 1 - k_star
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    else:
        p = -s
        grad[:n - p] += (1.0 - h[p:])
        grad[p:] -= h[:n - p]
    return grad * dx

            break
        h_try = interpolate_to_n(best_h, best_n, n_try)
        bounds_t = [(0.0, 1.0)] * n_try
        target_t = n_try / 2.0

        def _obj_t(x):
            x = np.clip(x, 0, 1)
            return compute_c5(x) + 200.0 * (np.sum(x) - target_t) ** 2

        for i in range(10):
            if time.time() > time_limit - 10:
                break
            if i == 0:
                h_s = h_try.copy()
            else:
                h_s = project_feasible(h_try + rng.normal(0, 0.02, n_try), n_try)
            res = minimize(_obj_t, h_s, method='L-BFGS-B', bounds=bounds_t,
                           options={'maxiter': 400, 'gtol': 1e-9})
            h_new = project_feasible(res.x, n_try)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_try
                h_try = h_new.copy()

        if time.time() < time_limit - 10:
            h_s2, c5_s2 = projected_subgrad(best_h, best_n, 20000, rng, time_limit - 10)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s2

    # Final: borrow best from initial_h_values if it's still beating us
    c5_init = compute_c5(project_feasible(initial_h_values.copy().astype(float), len(initial_h_values)))
    if c5_init < best_c5:
        best_h = project_feasible(initial_h_values.copy().astype(float), len(initial_h_values))
        best_c5 = c5_init
        best_n = len(best_h)

    # Final refinement
    if time.time() < time_limit - 5:
        h_f, c5_f = coordinate_refine(best_h, best_n, rng, time_limit, max_rounds=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    return best_h, best_c5, best_n
