import numpy as np
import time


def finalize_sum(y, target):
    y = np.asarray(y, dtype=float).copy()
    for _ in range(8):
        d = float(target - np.sum(y))
        if abs(d) < 1e-12:
            break
        m = (y > 1e-10) & (y < 1.0 - 1e-10)
        if m.any():
            y[m] += d / float(np.count_nonzero(m))
        else:
            y[int(np.argmax(y * (1.0 - y)))] += d
        y = np.clip(y, 0.0, 1.0)
    return y


def make_feasible(v, target):
    v = np.asarray(v, dtype=float).ravel().copy()
    if v.size == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    target = float(target)
    if float(v.min()) >= 0.0 and float(v.max()) <= 1.0:
        d = target - float(np.sum(v))
        if float(v.min()) + d >= -1e-12 and float(v.max()) + d <= 1.0 + 1e-12:
            v += d
            return finalize_sum(np.clip(v, 0.0, 1.0), target)
    lo = -float(np.max(v)) - 1.0
    hi = 1.0 - float(np.min(v)) + 1.0
    for _ in range(38):
        mid = 0.5 * (lo + hi)
        ys = np.clip(v + mid, 0.0, 1.0)
        if float(np.sum(ys)) > target:
            hi = mid
        else:
            lo = mid
    return finalize_sum(np.clip(v + 0.5 * (lo + hi), 0.0, 1.0), target)


def exact_c5(h):
    n = int(h.size)
    if n <= 0:
        return 1.0
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(n)))


def lse_val(s, temp):
    m = float(np.max(s))
    if temp <= 1e-12:
        return m
    return float(m + temp * np.log(np.sum(np.exp((s - m) / temp))))


def lse_grad_h(h, temp):
    n = int(h.size)
    s = np.correlate(h, 1.0 - h, mode="full")
    val = lse_val(s, temp)
    m = float(np.max(s))
    w = np.exp((s - m) / temp)
    ws = float(np.sum(w))
    if ws > 0.0:
        w /= ws
    g = np.zeros(n, dtype=float)
    th = 1e-10
    if n > 0:
        w0 = w[n - 1]
        if w0 > th:
            g += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[n - 1 - d]
        wn = w[n - 1 + d]
        if wp <= th and wn <= th:
            continue
        left = h[:-d]
        right = h[d:]
        if wp > th:
            g[:-d] += wp * (1.0 - right)
            g[d:] += -wp * left
        if wn > th:
            g[:-d] += -wn * right
            g[d:] += wn * (1.0 - left)
    return g, val


def gradient_phase(h, n, deadline, temp, seed):
    n = int(n)
    target = 0.5 * n
    h = make_feasible(h, target)
    best_h = h.copy()
    best_c = exact_c5(h)
    cur_c = best_c
    rng = np.random.default_rng(seed)
    cur_temp = max(0.01, float(temp))
    step = 0.08
    it = 0
    fail = 0
    last_imp = 0

    while time.time() < deadline:
        g, val = lse_grad_h(h, cur_temp)
        g -= float(np.mean(g))
        norm = float(np.linalg.norm(g))
        if norm < 1e-12:
            break
        direction = -g / norm
        fcur = val
        accepted = False
        sq = step
        for _ in range(10):
            cand = h + sq * direction
            if float(cand.min()) >= -1e-12 and float(cand.max()) <= 1.0 + 1e-12:
                cand = np.clip(cand, 0.0, 1.0)
                d = target - float(np.sum(cand))
                if abs(d) > 1e-11:
                    cand += d / float(n)
                    if float(cand.min()) < 0.0 or float(cand.max()) > 1.0:
                        cand = make_feasible(cand, target)
                    else:
                        cand = np.clip(cand, 0.0, 1.0)
            else:
                cand = make_feasible(cand, target)
            sc = np.correlate(cand, 1.0 - cand, mode="full")
            fc = lse_val(sc, cur_temp)
            cc = float(np.max(sc)) * (2.0 / float(n))
            if fc <= fcur - 1e-12 or cc <= cur_c - 1e-12:
                h = cand
                val = fc
                cur_c = cc
                accepted = True
                if cur_c < best_c - 1e-12:
                    best_h = h.copy()
                    best_c = cur_c
                    last_imp = it
                step = min(0.25, step * 1.15)
                break
            sq *= 0.5
        it += 1
        if not accepted:
            fail += 1
            step = max(0.004, step * 0.75)
            cur_temp = max(0.01, cur_temp * 0.92)
            if fail > 100:
                h = make_feasible(best_h + 0.02 * rng.normal(size=n), target)
                cur_c = exact_c5(h)
                if cur_c < best_c - 1e-12:
                    best_h = h.copy()
                    best_c = cur_c
                fail = 0
                step = 0.08
                cur_temp = max(0.05, cur_temp)
        else:
            fail = 0
        if it - last_imp > 400:
            cur_temp = max(0.01, cur_temp * 0.88)
        if it % 200 == 0 and cur_temp > 0.02:
            cur_temp = max(0.02, cur_temp * 0.97)

    cur_c = exact_c5(h)
    if cur_c < best_c - 1e-12:
        best_h = h.copy()
        best_c = cur_c
    return best_h, best_c


def interp_initial(ref, n):
    if ref is None:
        return None
    ref = np.asarray(ref, dtype=float)
    n0 = int(ref.size)
    if n0 == 0:
        return None
    if n0 == int(n):
        return ref.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    return np.interp(x1, x0, ref, left=float(ref[0]), right=float(ref[-1]))


def binary_starts(n, ref, seed, k):
    n = int(n)
    target = 0.5 * n
    out = []
    for m in range(max(1, int(k))):
        rng = np.random.default_rng(seed + m)
        if m == 0 and ref is not None:
            vals = interp_initial(ref, n)
        elif m == 0:
            vals = np.array([float(i.bit_count() & 1) for i in range(n)], dtype=float)
        elif m == 1:
            vals = rng.uniform(size=n)
        else:
            bl = int(rng.integers(2, max(3, n // 4 + 3)))
            sh = int(rng.integers(bl))
            vals = ((np.arange(n) + sh) % bl) < (bl // 2)
            vals = vals.astype(float)
        if vals is None:
            vals = rng.uniform(size=n)
        b = (vals + 0.05 * rng.normal(size=n) > 0.5).astype(float)
        while float(b.sum()) > target and np.flatnonzero(b > 0.5).size:
            b[int(rng.choice(np.flatnonzero(b > 0.5)))] = 0.0
        while float(b.sum()) < target and np.flatnonzero(b < 0.5).size:
            b[int(rng.choice(np.flatnonzero(b < 0.5)))] = 1.0
        out.append(b)
    return out


def binary_local(n, deadline, seed, init_bits):
    n = int(n)
    target = 0.5 * n
    dx = 2.0 / float(n)
    rng = np.random.default_rng(seed)
    if init_bits is None:
        init_bits = binary_starts(n, None, seed, 1)[0]
    h = (np.asarray(init_bits, dtype=float) > 0.5).astype(float)
    while float(h.sum()) > target and np.flatnonzero(h > 0.5).size:
        h[int(rng.choice(np.flatnonzero(h > 0.5)))] = 0.0
    while float(h.sum()) < target and np.flatnonzero(h < 0.5).size:
        h[int(rng.choice(np.flatnonzero(h < 0.5)))] = 1.0
    h = make_feasible(h, target)
    cur = exact_c5(h)
    best_h = h.copy()
    best_c = cur
    stag = 0
    while time.time() < deadline:
        s = np.correlate(h, 1.0 - h, mode="full")
        mv = float(np.max(s))
        idxs = np.where(s >= mv - 1e-7)[0]
        m = int(rng.choice(idxs)) if idxs.size else n - 1
        tlag = n - 1 - m
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        p = int(rng.choice(ones))
        q = int(rng.choice(zeros))
        if tlag > 0:
            gg = np.where((h[:-tlag] > 0.5) & (h[tlag:] < 0.5))[0]
            if gg.size:
                p = int(rng.choice(gg))
                q = p + tlag if rng.random() < 0.65 else int(rng.choice(zeros))
        elif tlag < 0:
            d = -tlag
            gg = np.where((h[d:] > 0.5) & (h[:-d] < 0.5))[0]
            if gg.size:
                left = int(rng.choice(gg))
                p = left + d
                q = left if rng.random() < 0.65 else int(rng.choice(zeros))
        h2 = h.copy()
        h2[p] = 0.0
        h2[q] = 1.0
        s2 = np.correlate(h2, 1.0 - h2, mode="full")
        c2 = float(np.max(s2)) * dx
        improved = c2 < cur - 1e-12
        if improved:
            h = h2
            cur = c2
            stag = 0
            if cur < best_c - 1e-12:
                best_h = h.copy()
                best_c = cur
        elif c2 <= cur + 1e-12:
            if rng.random() < 0.22:
                h = h2
                cur = c2
        if not improved:
            stag += 1
        if stag > 4000:
            h = best_h.copy()
            for _ in range(5):
                ones = np.flatnonzero(h > 0.5)
                zeros = np.flatnonzero(h < 0.5)
                if ones.size and zeros.size:
                    p = int(rng.choice(ones))
                    q = int(rng.choice(zeros))
                    h[p] = 0.0
                    h[q] = 1.0
            h = make_feasible(h, target)
            cur = exact_c5(h)
            stag = 0
    if cur < best_c - 1e-12:
        best_h = h.copy()
        best_c = cur
    return best_h, best_c


def pair_polish(h, n, deadline, seed):
    n = int(n)
    target = 0.5 * n
    h = make_feasible(h, target)
    cur = exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(seed)
    stag = 0
    while time.time() < deadline:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        e = 0.02 + 0.12 * rng.random()
        if rng.random() < 0.5:
            a, b = i, j
        else:
            a, b = j, i
        cand = None
        for dlt in (e, -e):
            na = h[a] + dlt
            nb = h[b] - dlt
            if 0.0 <= na <= 1.0 and 0.0 <= nb <= 1.0:
                cand = h.copy()
                cand[a] = na
                cand[b] = nb
                break
        if cand is None:
            stag += 1
            continue
        c = exact_c5(cand)
        improved = c < cur - 1e-12
        if improved or (c <= cur + 1e-12 and rng.random() < 0.3):
            h = cand
            cur = c
            if cur < best_c - 1e-12:
                best_h = h.copy()
                best_c = cur
        if improved:
            stag = 0
        else:
            stag += 1
        if stag > 5000:
            h = make_feasible(best_h + 0.02 * rng.normal(size=n), target)
            cur = exact_c5(h)
            stag = 0
    if cur < best_c - 1e-12:
        best_h = h.copy()
        best_c = cur
    return best_h, best_c


def construct_simple(n, seed):
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * x) + 0.15 * np.sin(4.0 * np.pi * x + 0.7)
    return make_feasible(h, 0.5 * n)
