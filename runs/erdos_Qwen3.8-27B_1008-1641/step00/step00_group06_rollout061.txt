```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def make_feasible(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        mask = (h > 0) & (h < 1) if diff != 0 else np.ones(n, dtype=bool)
        if diff > 0:
            # Need to increase sum: increase non-1 values
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() == 0:
            break
        step = diff / mask.sum()
        h[mask] += step
        h = np.clip(h, 0, 1)
    return h

    def obj(h):
        return compute_c5(h)
    def obj_grad(h):
```

```python
import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h


def thue_morse(n):
    t = np.zeros(n)
    i = 1
    while i < n:
        t[i:2 * i] = 1.0 - t[:i]
        i *= 2
    return t


def balance_01(h, n):
    """Adjust a 0-1 array so it has exactly n//2 ones."""
    h = (h > 0.5).astype(float)
    s = int(h.sum())
    if s != n // 2:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        if s > n // 2:
            h[oi[:s - n // 2]] = 0
        else:
            h[zi[:n // 2 - s]] = 1
    return h


def slsqp_opt(h0, n, max_iter=300):
    def obj(x):
        return compute_c5(x)
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        return make_feasible(res.x, n)
    except Exception:
        return h0

    h = h.copy()

    return h
```
