    return float(np.max(corr)) * dx

def project_feasible(h, n, n_iter=50):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(n_iter):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            space = 1.0 - h[mask]
            total = float(np.sum(space))
            if total < 1e-15:

        else:

            space = h[mask]
            total = float(np.sum(space))
            if total < 1e-15:
