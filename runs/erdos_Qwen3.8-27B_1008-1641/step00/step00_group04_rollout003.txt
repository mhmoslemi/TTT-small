def S_arrays_from_h(h):
    n = int(h.size)
    m = 1
    need = 2*n
    while m < need: m <<= 1
    f = np.fft.rfft(h, m)
    c = np.fft.irfft(f * np.conj(f), m)
    c = c[:n]
    c = np.where(c > 0.0, c, 0.0)
    cum = np.empty(n+1, dtype=float)
    cum[0] = 0.0
    cum[1:] = np.cumsum(h)
    total = cum[-1]
    spos = cum[n:0:-1] - c
    sneg = total - cum[:n] - c
    sall = np.empty(2*n - 1, dtype=float)
    sall[:n] = spos
    sall[n:] = sneg[1:]
    smax = float(np.max(sall))
    return sall, smax

import time
import numpy as np


def feasible_update(h, direction, alpha, target):
    h = np.asarray(h, dtype=float)
    d = np.array(direction, dtype=float, copy=True)
    d -= np.mean(d)
    x = h + float(alpha) * d
    np.clip(x, 0.0, 1.0, out=x)
    target = float(target)

    for _ in range(8):
        diff = target - float(x.sum())
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            slack = 1.0 - x
            s = float(slack.sum())
            if s <= 1e-14:
                x.fill(1.0)
                break
            x += diff * slack / s
        else:
            s = float(x.sum())
            if s <= 1e-14:
                x.fill(0.0)
                break
            x += diff * x / s
        np.clip(x, 0.0, 1.0, out=x)

    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x


def project_l2_box_sum(y, target):
    y_arr = np.asarray(y, dtype=float)
    n = int(y_arr.size)
    target = float(target)

    x = np.clip(y_arr, 0.0, 1.0)
    if abs(float(x.sum()) - target) <= 1e-8 * max(1.0, float(n)):
        return feasible_update(x, np.zeros(n, dtype=float), 0.0, target)

    lo = float(np.min(y_arr)) - 1.0
    hi = float(np.max(y_arr)) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y_arr - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(y_arr - 0.5 * (lo + hi), 0.0, 1.0)
    return feasible_update(h, np.zeros(n, dtype=float), 0.0, target)


def get_parent_h():
    try:
        val = initial_h_values
    except NameError:
        return None
    arr = np.asarray(val, dtype=float)
    if arr.ndim != 1 or arr.size < 2 or not np.all(np.isfinite(arr)):
        return None
    return arr


def make_initial_h(n, rng):
    target = 0.5 * n
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.40 * np.cos(np.pi * x) + 0.15 * np.cos(2.0 * np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    return project_l2_box_sum(h, target)


def symmetrize_feasible(h):
    h2 = 0.5 * (h + h[::-1])
    return project_l2_box_sum(h2, 0.5 * h.size)


def resample_h(h_old, n_new):
    n_old = int(h_old.size)
    n_new = int(n_new)
    if n_new < 2:
        n_new = 2
    if n_old == n_new:
        h = np.array(h_old, dtype=float, copy=True)
    elif n_old == 1:
        h = np.full(n_new, float(h_old[0]))
    else:
        x_old = np.linspace(0.0, 1.0, n_old)
        x_new = np.linspace(0.0, 1.0, n_new)
        h = np.interp(x_new, x_old, h_old)
    return project_l2_box_sum(h, 0.5 * n_new)


def S_arrays_from_h(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    m = 1
    need = 2 * n
    while m < need:
        m <<= 1

    f = np.fft.rfft(h, m)
    c = np.fft.irfft(f * np.conj(f), m)
    c = c[:n]
    c = np.where(c > 0.0, c, 0.0)

    cum = np.empty(n + 1, dtype=float)
    cum[0] = 0.0
    cum[1:] = np.cumsum(h)
    total = cum[-1]

    spos = cum[n:0:-1] - c
    sneg = total - cum[:n] - c

    sadd = np.empty(2 * n - 1, dtype=float)
    sadd[:n] = spos
    sadd[n:] = sneg[1:]

    return sadd, float(np.max(sadd))


def objective_c5(h):
    _, smax = S_arrays_from_h(h)
    return 2.0 * smax / float(h.size)


def evaluate_direct(h):
    n = int(h.size)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / float(n)))


def compute_weights(h, active_c):
    sadd, smax = S_arrays_from_h(h)
    n = int(h.size)
    t = max(1e-6, float(active_c) * 0.5 * float(n))
    expv = np.exp((sadd - smax) / t)
    s = float(expv.sum())
    if s <= 0.0:
        w = np.zeros_like(sadd)
        w[0] = 1.0
    else:
        w = expv / s

    wpos = w[:n].copy()
    wneg = np.zeros(n, dtype=float)
    wneg[1:] = w[n:]
    return smax, wpos, wneg


def gradient_from_weights(h, wpos, wneg):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    g = (1.0 - 2.0 * h) * float(wpos[0])
    thr = 1e-12

    for lag in range(1, n):
        wp = float(wpos[lag])
        wn = float(wneg[lag])
        q = wp + wn
        if q <= thr:
            continue
        length = n - lag
        tail = h[lag:]
        head = h[:length]
        g[:length] += wp - q * tail
        g[lag:] += wn - q * head
    return g


def local_refine(h, n, time_s, rng, active_c_initial=2e-3):
    n = int(n)
    target = 0.5 * float(n)
    h = project_l2_box_sum(h, target)
    c = objective_c5(h)
    best_h = h.copy()
    best_c = float(c)

    if time_s <= 0.0 or n < 2:
        return best_h, best_c

    deadline = time.time() + float(time_s)
    active_c = float(active_c_initial)
    alpha = 0.15
    no_accept = 0

    while time.time() < deadline and alpha > 1e-6:
        _, wpos, wneg = compute_weights(h, active_c)
        grad = gradient_from_weights(h, wpos, wneg)
        direction = -grad
        direction -= np.mean(direction)
        norm = float(np.linalg.norm(direction, ord=np.inf))

        if norm < 1e-9:
            h = feasible_update(h, rng.normal(size=n), 0.02, target)
            c = objective_c5(h)
            active_c = min(active_c * 2.0, 5.0e-2)
            no_accept += 1
            if no_accept > 30:
                alpha *= 0.5
            continue

        direction /= norm
        accepted = False

        for a in (alpha, 0.5 * alpha, 0.25 * alpha, 0.125 * alpha, 0.0625 * alpha, 0.005):
            if time.time() > deadline:
                break
            x = feasible_update(h, direction, a, target)
            cx = objective_c5(x)
            if np.isfinite(cx) and cx < c - 1e-12:
                h = x
                c = float(cx)
                if a < alpha:
                    alpha = max(a, 1e-6)
                else:
                    alpha = min(0.5, 1.25 * a)
                active_c = max(1e-6, active_c * 0.9)
                no_accept = 0
                if c < best_c - 1e-12:
                    best_h = x.copy()
                    best_c = c
                accepted = True
                break

        if not accepted:
            no_accept += 1
            alpha *= 0.5
            active_c = min(active_c * 1.5, 5.0e-2)

            if no_accept > 12:
                base = best_h if (no_accept % 2 == 1 or c > best_c + 1e-4) else h
                step = 0.02 + 0.06 * float(rng.random())
                h = feasible_update(base, rng.normal(size=n), step, target)
                c = objective_c5(h)
                if c < best_c - 1e-12:
                    best_h = h.copy()
                    best_c = c
                alpha = 0.15
                active_c = max(2.0e-4, 0.5 * active_c)
                no_accept = 0

        if c > best_c + 1e-4 and no_accept > 25:
            h = best_h.copy()
            c = best_c
            no_accept = 0

    return best_h, best_c


def quantize_dyadic(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    scale = 1 << 30
    c = np.rint(h * float(scale)).astype(np.int64)
    c = np.clip(c, 0, scale)

    target = (n * scale) // 2
    diff = int(target - int(c.sum()))

    if diff != 0:
        if diff > 0:
            order = np.argsort(-(scale - c).astype(np.int64))
        else:
            order = np.argsort(-c.astype(np.int64))

        i = 0
        while diff != 0 and i < 2 * n + 10:
            idx = int(order[i % n])
            if diff > 0:
                avail = int(scale - int(c[idx]))
                if avail > 0:
                    take = min(avail, diff)
                    c[idx] += take
                    diff -= take
            else:
                avail = int(c[idx])
                if avail > 0:
                    take = min(avail, -diff)
                    c[idx] -= take
                    diff += take
            i += 1

        if diff != 0:
            if diff > 0:
                for idx in range(n):
                    if int(c[idx]) < scale:
                        take = min(int(scale - int(c[idx])), diff)
                        c[idx] += take
                        diff -= take
                        if diff == 0:
                            break
            else:
                for idx in range(n):
                    if int(c[idx]) > 0:
                        take = min(int(c[idx]), -diff)
                        c[idx] -= take
                        diff += take
                        if diff == 0:
                            break

    return c.astype(float) / float(scale)


def snapshot_add(snapshots, h):
    if len(snapshots) >= 12 or h is None:
        return
    arr = np.asarray(h, dtype=float)
    if arr.size < 2 or not np.all(np.isfinite(arr)):
        return
    snapshots.append(arr.copy())


def consider_candidate(snapshots, h, c, best_h, best_c):
    if h is None:
        return best_h, best_c
    arr = np.asarray(h, dtype=float)
    if arr.size < 2:
        return best_h, best_c
    try:
        c = float(c)
    except Exception:
        return best_h, best_c
    if not np.isfinite(c) or not np.all(np.isfinite(arr)):
        return best_h, best_c

    if c < best_c:
        best_h = arr.copy()
        best_c = c
        snapshot_add(snapshots, best_h)
        return best_h, best_c

    best_h_copy = best_h
    if best_h_copy is not None:
        try:
            best_c = float(best_c)
        except Exception:
            pass
    return best_h, best_c
