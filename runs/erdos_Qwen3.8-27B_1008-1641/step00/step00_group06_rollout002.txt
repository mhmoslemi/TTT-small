def project_simplex(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = len(x)
    target = float(target)
    lo = float(np.min(x) - 1.0) - 1e-12
    hi = float(np.max(x) + 1.0) + 1e-12
    # function sum clip(x - lam,0,1) decreasing in lam
    # If target outside [0,n], clip target.
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    for _ in range(80):
       lam = 0.5*(lo+hi)
       y = np.clip(x - lam, 0.0, 1.0)
       s = y.sum()
       if s > target: lo = lam
       else: hi = lam
    lam=0.5*(lo+hi)
    y=np.clip(x-lam,0,1)
    diff = target - y.sum()
    if diff != 0:
       # distribute tiny residual among entries with slack
       slack = (y < 1.0) & (y > 0.0)
       if np.any(slack):
           cnt = int(np.sum(slack))

    for _ in range(70):

        if s > target:
            lo = lam
        else:
            hi = lam

            idx = int(np.argmax(y)) if diff > 0 else int(np.argmin(y))
            y[idx] += diff
        y = np.clip(y, 0.0, 1.0)

def smooth_max_grad(h, tau):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0, np.zeros(0), 0.0
    cfull = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cmax = float(np.max(cfull))
    tau = max(float(tau), 1e-12)
    if n == 1:
        w0 = 1.0
        Z = 1.0
    else:
        zneg = np.exp((cfull[:n-1] - cmax) / tau)
        zpos = np.exp((cfull[n:] - cmax) / tau)
        z0 = math.exp((cfull[n-1] - cmax) / tau)
        Z = float(zneg.sum() + z0 + zpos.sum())
        if Z < 1e-300:
            Z = 1.0
        wneg = zneg / Z
        wpos = zpos / Z
        w0 = z0 / Z
    grad = np.zeros(n)
    if w0 != 0.0:
        grad += w0 * (1.0 - 2.0 * h)
    if n > 1:

        for d in range(1, n):

            if wm > 1e-15:

                grad[:-d] -= wp * h[d:]
                grad[d:] += wp * omh[:-d]
        grad *= 2.0 / n
    else:
        grad *= 2.0
    sm = cmax + tau * math.log(max(Z, 1e-300))

def erdos_c(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_simplex(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    target = float(target)
    if n == 0:
        return x
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(70):
        lam = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - lam, 0.0, 1.0)))
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = 0.5 * (lo + hi)
    y = np.clip(x - lam, 0.0, 1.0)
    for _ in range(10):
        diff = target - float(y.sum())
        if abs(diff) < 1e-12:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if bool(np.any(mask)):
            y[mask] += diff / float(np.count_nonzero(mask))
        else:
            idx = int(np.argmax(y)) if diff > 0.0 else int(np.argmin(y))
            y[idx] += diff
        y = np.clip(y, 0.0, 1.0)
    return y

def smooth_max_grad(h, tau):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0, np.zeros(0), 0.0
    dx = 2.0 / n
    cfull = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(np.max(cfull))
    tau = max(float(tau), 1e-12)
    if n == 1:
        Z = 1.0
        w0 = 1.0
        grad = w0 * (1.0 - 2.0 * h)
        grad *= dx
        return float(cmax), grad, cmax
    zneg = np.exp((cfull[:n - 1] - cmax) / tau)
    zpos = np.exp((cfull[n:] - cmax) / tau)
    z0 = math.exp((cfull[n - 1] - cmax) / tau)
    Z = float(zneg.sum() + z0 + zpos.sum())
    if Z < 1e-300:
        Z = 1.0
    wneg = zneg / Z
    wpos = zpos / Z
    w0 = z0 / Z
    grad = w0 * (1.0 - 2.0 * h)
    omh = 1.0 - h
    for d in range(1, n):
        wm = wneg[n - 1 - d]
        if wm > 1e-15:
            grad[:-d] += wm * omh[d:]
            grad[d:] -= wm * h[:-d]
        wp = wpos[d - 1]
        if wp > 1e-15:
            grad[:-d] -= wp * h[d:]
            grad[d:] += wp * omh[:-d]
    grad *= dx
    sm = cmax + tau * math.log(max(Z, 1e-300))
    return float(sm), grad, cmax

def slsqp_objective(x):
    if time.time() >= _deadline_slsqp:
        raise TimeLimit()
    sm, _, _ = smooth_max_grad(x, _tau_slsqp)
    return float(sm)

def slsqp_jacobian(x):
    if time.time() >= _deadline_slsqp:
        raise TimeLimit()
    _, grad, _ = smooth_max_grad(x, _tau_slsqp)
    return grad

def mass_eq(x):
    return float(np.sum(np.asarray(x)) - _target_slsqp)

def mass_eq_jac(x):
    return np.ones_like(np.asarray(x), dtype=float)

def run_slsqp(h, deadline, tau, maxiter):
    global _target_slsqp, _tau_slsqp, _deadline_slsqp
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(h.size)
    h = project_simplex(h, target)
    if time.time() >= deadline:
        return h
    _target_slsqp = target
    _tau_slsqp = float(tau)
    safe_deadline = deadline - 1.0 if deadline > time.time() + 2.0 else deadline
    _deadline_slsqp = float(safe_deadline)
    bounds = [(0.0, 1.0)] * h.size
    cons = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_jac}]
    try:
        res = optimize.minimize(slsqp_objective, h, method='SLSQP', jac=slsqp_jacobian,
                                bounds=bounds, constraints=cons,
                                options={'maxiter': int(maxiter), 'ftol': 1e-10,
                                         'disp': False, 'maxls': 50})
        if hasattr(res, 'x') and np.asarray(res.x).size == h.size:
            h = project_simplex(res.x, target)
    except Exception:
        pass
    return h

def random_refine(h, deadline, seed=0):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * float(n)
    h = project_simplex(h, target)
    c = erdos_c(h)

    while time.time() < deadline:
        k = int(rng.integers(2, min(n, 15)))
        idx = rng.choice(n, size=k, replace=False)
        delta = rng.normal(size=k)
        delta -= delta.mean()
        scale = rng.uniform(0.002, 0.05)
        trial = h.copy()
        trial[idx] += delta * scale
        trial = project_simplex(trial, target)
        ct = erdos_c(trial)
        if ct < bestc:
            best = trial
            bestc = ct
            h = trial.copy()

def make_smooth_start(n, seed):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)

    for m in range(1, 6):
        amp = rng.uniform(0.02, 0.18)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(np.pi * m * x + ph)
    return project_simplex(h, 0.5 * float(n))

def resample_h(h, n_new):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return h.astype(float).copy()
    if n == n_new:
        return project_simplex(h, 0.5 * float(n))
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = np.linspace(0.0, 2.0, n_new, endpoint=False)
    hnew = np.interp(y, x, h, left=h[0], right=h[-1])
    return project_simplex(hnew, 0.5 * float(n_new))

    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return 0.0

def project_simplex(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    target = float(target)
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(70):
        lam = 0.5 * (lo + hi)
        if float(np.sum(np.clip(x - lam, 0.0, 1.0))) > target:
            lo = lam

def smooth_max_grad(h, tau):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(np.max(c))
    tau = max(float(tau), 1e-12)
    if n == 1:
        g = (1.0 - 2.0 * h) * dx
        return float(cmax), g, cmax
    zn = np.exp((c[:n - 1] - cmax) / tau)
    zp = np.exp((c[n:] - cmax) / tau)
    z0 = math.exp((c[n - 1] - cmax) / tau)
    Z = float(zn.sum() + z0 + zp.sum())
    zn /= Z
    zp /= Z
    z0 /= Z
    g = z0 * (1.0 - 2.0 * h)
    om = 1.0 - h
    for d in range(1, n):
        w = zn[n - 1 - d]
        if w > 1e-15:
            g[:-d] += w * om[d:]
            g[d:] -= w * h[:-d]
        w = zp[d - 1]
        if w > 1e-15:
            g[:-d] -= w * h[d:]
            g[d:] += w * om[:-d]

import time, math
import numpy as np
from scipy import optimize

class TimeLimit(Exception): pass
_tau = 0.0
_target = 0.0
_ddl = 0.0

def erdos_c(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / h.size)

def project_simplex(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    target = float(target)
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(70):
        lam = 0.5 * (lo + hi)
        if float(np.sum(np.clip(x - lam, 0.0, 1.0))) > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - float(y.sum())
    m = (y > 1e-12) & (y < 1.0 - 1e-12)
    if np.any(m):
        y[m] += d / float(np.count_nonzero(m))
    else:
        y[int(np.argmax(y) if d > 0 else np.argmin(y))] += d
    return np.clip(y, 0.0, 1.0)

def smooth_max_grad(h, tau):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(np.max(c))
    tau = max(float(tau), 1e-12)
    if n == 1:
        g = (1.0 - 2.0 * h) * dx
        return float(cmax), g, cmax
    zn = np.exp((c[:n - 1] - cmax) / tau)
    zp = np.exp((c[n:] - cmax) / tau)
    z0 = math.exp((c[n - 1] - cmax) / tau)
    Z = float(zn.sum() + z0 + zp.sum())
    zn /= Z
    zp /= Z
    z0 /= Z
    g = z0 * (1.0 - 2.0 * h)
    om = 1.0 - h
    for d in range(1, n):
        w = zn[n - 1 - d]
        if w > 1e-15:
            g[:-d] += w * om[d:]
            g[d:] -= w * h[:-d]
        w = zp[d - 1]
        if w > 1e-15:
            g[:-d] -= w * h[d:]
            g[d:] += w * om[:-d]
    g *= dx
    sm = cmax + tau * math.log(max(Z, 1e-300))
    return float(sm), g, cmax

def slsqp_fun(x):
    if time.time() >= _ddl:
        raise TimeLimit()
    return float(smooth_max_grad(x, _tau)[0])

def slsqp_jac(x):
    if time.time() >= _ddl:
        raise TimeLimit()
    return smooth_max_grad(x, _tau)[1]

def mass_fun(x):
    return float(np.sum(np.asarray(x)) - _target)

def mass_jac(x):
    return np.ones_like(np.asarray(x), dtype=float)

def run_slsqp(h, deadline, tau, maxiter):

    if time.time() >= deadline:

    _target = 0.5 * float(h.size)

                    best, bestc = h2, c2
    if time.time() + 1.0 < deadline:
        h2, c2 = run_pg(best, deadline, 0.02 if n < 150 else 0.035, seed)
        if c2 < bestc:
            best, bestc = h2, c2
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)
    init = globals().get("initial_h_values", None)
    best_h = None
    best_c = 1e100
    if init is not None:
        try:
            arr = np.asarray(init, dtype=float).ravel().copy()
            if arr.size > 1:
                best_h = project_simplex(arr, 0.5 * float(arr.size))
                best_c = erdos_c(best_h)
        except Exception:
            best_h = None
    if best_h is None:
        best_h = construct_start(None, 120, seed)
        best_c = erdos_c(best_h)

    phase = min(deadline, t0 + max(5.0, 0.30 * float(budget_s)))
    h, c = optimize_start(best_h.copy(), phase, True, seed)
    if c < best_c:
        best_h, best_c = h, c

    n1 = int(min(240, max(120, 2 * int(best_h.size))))
    if n1 != int(best_h.size) and time.time() < deadline:
        ph2 = min(deadline, max(phase, t0 + 0.68 * float(budget_s)))
        h, c = optimize_start(resample_h(best_h, n1), ph2, True, seed + 1)
        if c < best_c:
            best_h, best_c = h, c
        phase = ph2

    if time.time() < deadline - 60.0 and best_c > 0.3808:
        n_alt = int(min(240, max(120, 2 * int(best_h.size))))
        alt_ph = min(deadline, max(phase, t0 + 0.80 * float(budget_s)))
        h, c = optimize_start(construct_start(best_h, n_alt, seed + 7), alt_ph, True, seed + 7)
        if c < best_c:
            best_h, best_c = h, c
        phase = alt_ph

    if time.time() < deadline - 15.0:
        n2 = int(min(420, max(200, 2 * int(best_h.size))))
        start3 = resample_h(best_h, n2) if n2 != int(best_h.size) else best_h.copy()
        ph3 = min(deadline, max(phase, t0 + 0.90 * float(budget_s)))
        h, c = optimize_start(start3, ph3, False, seed + 2)
        if c < best_c:
            best_h, best_c = h, c
        phase = ph3

    final = min(deadline, t0 + max(0.0, float(budget_s) - 2.0))
    if time.time() < final:
        h, c = optimize_start(best_h.copy(), final, False, seed + 3)
        if c < best_c:
            best_h, best_c = h, c

    best_h = project_simplex(best_h, 0.5 * float(best_h.size))
    best_c = erdos_c(best_h)
    return best_h, float(best_c), int(best_h.size)
