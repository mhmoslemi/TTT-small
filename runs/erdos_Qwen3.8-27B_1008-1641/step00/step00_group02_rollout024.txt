from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full') * dx
    return np.max(corr), corr

def init_random(n, seed=42):
    rng = np.random.default_rng(seed)
    h = rng.random(n)
    h = (h >= 0.5).astype(float)  # binary
    # Fix sum
    target = n // 2
    while h.sum() != target:
        if h.sum() > target:
            idx = np.where(h == 1)[0]
            h[rng.integers(len(idx))] = 0 if idx else 0
            # pick random 1 to flip
            ones = np.where(h == 1)[0]
            h[rng.integers(len(ones))] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[rng.integers(len(zeros))] = 1
    return h

def init_sinusoidal(n, freq=1, seed=42):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
    # Adjust to have exact sum
    h = np.clip(h, 0, 1)
    # Fix sum
    diff = n/2 - h.sum()
    h += diff / n
    h = np.clip(h, 0, 1)
    # Final fix
    diff = n/2 - h.sum()
    # Distribute
    h += diff / n
    return h

def repair(h, n):
    h = np.clip(h, 0, 1)
    diff = n/2 - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        h += diff / n
        h = np.clip(h, 0, 1)
        # If still off, adjust specific elements
        diff = n/2 - h.sum()
        if diff > 0:
            # Need to increase some elements
            order = np.argsort(-h)  # most room first
            idx = order[:int(diff)+1]
            for i in idx:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff <= 1e-10:

        else:

    """Local optimization using SLSQP with smooth max approximation."""
    start_time = time.time()
    
    def objective(params):
        h = params
        dx = 2.0 / n
        comp = 1.0 - h
        corr = np.correlate(h, comp, mode='full') * dx
        # Smooth max using log-sum-exp
        t = 200  # temperature for smoothness
        max_corr = np.max(corr)
        return np.log(np.sum(np.exp(t * (corr - max_corr)))) / t + max_corr
    
    def gradient(params):
        # Numerical gradient (or analytical)
        # For now, use numerical
        eps = 1e-7
        g = np.zeros_like(params)
        for i in range(n):

            params[i] -= eps
            g[i] = (max_plus - objective(params)) / eps  # rough

    return np.max(compute_corrs(h))

    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:

        else:
            s = h.sum()
            if s > 1e-14:

        h = np.clip(h, 0.0, 1.0)

            else:

    a = 1.0 - 2.0 * h
    grad = dx * np.correlate(w, a, mode='valid')
    return obj, grad
