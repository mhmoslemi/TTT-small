        h = _slsqp(best_h, nb, alpha=alpha, maxiter=350)
        v = _c5(h)
        if v < best_c5:
            best_h, best_c5 = h, v

    # Phase 2: Local search
    if time.time() - t0 < budget_s * 0.25:
        best_h = _local_search(best_h, nb, iters=400, rng=rng)
        best_c5 = _c5(best_h)

    # Phase 3: SLSQP from perturbations + random restarts
    n_restarts = 120
    for r in range(n_restarts):
        if time.time() - t0 > budget_s * 0.65:
            break
        if r < 10:
            h0 = best_h + rng.normal(0, 0.015, nb)
        elif r < 25:
            h0 = best_h + rng.normal(0, 0.04, nb)
        elif r < 40:
            h0 = best_h + rng.normal(0, 0.10, nb)
            if r % 5 == 0:
                h0 = _binary_round(h0, nb)
        else:
            h0 = _fix_sum(rng.random(nb), nb / 2.0)
            h0 = h0 * (nb / 2.0) / max(1e-9, h0.sum())
            h0 = _fix_sum(h0, nb / 2.0)
        h = _slsqp(h0, nb, alpha=60, maxiter=300)
        v = _c5(h)
        if v < best_c5:
            best_h, best_c5 = h, v
        if r % 30 == 29 and time.time() - t0 < budget_s * 0.5:
            best_h = _local_search(best_h, nb, iters=100, rng=rng)
            best_c5 = _c5(best_h)

    # Phase 4: Local search refinement
    if time.time() - t0 < budget_s * 0.80:
        best_h = _local_search(best_h, nb, iters=500, rng=rng)
        best_c5 = _c5(best_h)
        h2 = _slsqp(best_h, nb, alpha=150, maxiter=400)
        v2 = _c5(h2)
        if v2 < best_c5:
            best_h, best_c5 = h2, v2

    # Phase 5: Try different n values
    for nt in [nb * 2, nb + 24, nb * 3]:
        if time.time() - t0 > budget_s * 0.93:
            break
        if nt == nb or nt > 800:
            continue
        hr = _fix_sum(_resample(best_h, nt), nt / 2.0)
        hr = _slsqp(hr, nt, alpha=60, maxiter=350)
        v = _c5(hr)
        if v < best_c5:
            best_h, best_c5 = hr, v
            nb = nt
        hr2 = _fix_sum(rng.random(nt), nt / 2.0)
        hr2 = hr2 * (nt / 2.0) / max(1e-9, hr2.sum())
        hr2 = _slsqp(hr2, nt, alpha=80, maxiter=300)
        v2 = _c5(hr2)
        if v2 < best_c5:
            best_h, best_c5 = hr2, v2
            nb = nt

    # Final polish
    nf = len(best_h)
    if time.time() - t0 < budget_s * 0.97:
        best_h = _slsqp(best_h, nf, alpha=200, maxiter=500)
    best_h = _local_search(best_h, nf, iters=100, rng=rng)
    best_h = _fix_sum(best_h, nf / 2.0)
    best_c5 = _c5(best_h)

    return best_h, best_c5, nf
