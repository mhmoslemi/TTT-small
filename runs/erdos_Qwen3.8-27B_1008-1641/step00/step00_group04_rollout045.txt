def project_to_constraints(h, n_points):
    """Project h onto {0<=h[i]<=1, sum(h)=n_points/2}"""
    h = np.clip(h, 0, 1)
    target = n_points / 2.0
    total = np.sum(h)
    if abs(total - target) < 1e-10:
        return h
    # Scale
    alpha = target / total if total > 0 else 1.0
    h_proj = h * alpha
    # Handle overflow
    if np.max(h_proj) > 1:
        # Water-filling
        h_proj = np.where(h_proj > 1, 1.0, h_proj)
        # Redistribute
        deficit = target - np.sum(h_proj)
        free_mask = h_proj < 1.0
        n_free = np.sum(free_mask)
        if n_free > 0:
            h_proj[free_mask] += deficit / n_free
            # Re-clip
            h_proj = np.clip(h_proj, 0, 1)
            # Iterate if needed
            for _ in range(100):
                total_now = np.sum(h_proj)
                if abs(total_now - target) < 1e-10:
                    break
                alpha = target / total_now
                h_proj = h_proj * alpha
                h_proj = np.clip(h_proj, 0, 1)
    elif np.min(h_proj) < 0:
        # Similar for underflow
        h_proj = np.where(h_proj < 0, 0.0, h_proj)
        surplus = np.sum(h_proj) - target
        free_mask = h_proj > 0.0
        n_free = np.sum(free_mask)
        if n_free > 0:
            h_proj[free_mask] -= surplus / n_free
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def project(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, target / n)
    if abs(s - target) < 1e-12:
        return h
    if s < target:
        free = h < 1.0
        cap = (1.0 - h[free]).sum()
        if cap > 0:
            frac = min(1.0, (target - s) / cap)
            h[free] += frac * (1.0 - h[free])
    else:
        free = h > 0.0
        frac = min(1.0, (s - target) / s)
        h[free] -= frac * h[free]
    # Final squeeze for exactness
    s = h.sum()
    if s > 0 and abs(s - target) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.sum() > 0:
            h[free] += (target - s) / free.sum()
    return np.clip(h, 0.0, 1.0)
