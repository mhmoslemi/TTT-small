        h[perms[k-1]] = 1.0
    return project(h, n / 2.0)

def coord_descent(h, n, target, best_c5, max_sweeps=100, time_fn=None):
    for sweep in range(max_sweeps):
        if time_fn is not None and time_fn() < 5:
            break
        improved = False
        for i in range(n):
            old = h[i]
            for c in [0.0, 1.0, old - 0.05, old + 0.05, old - 0.1, old + 0.1]:
                c = np.clip(c, 0, 1)
                if abs(c - old) < 1e-8:
                    continue
                h[i] = c
                h = project(h, target)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
                    break
                h[i] = old
                h = project(h, target)
        # swap moves
        for i in range(n):
            if time_fn is not None and time_fn() < 5:
                break
            for j in range(i + 1, n):
                if h[i] < 1e-8 or h[j] < 1e-8:
                    continue
                if h[i] > 1.0 - 1e-8:
                    continue
                h[i] += 0.05
                h[j] -= 0.05
                h = project(h, target)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
                else:
                    h[i] -= 0.05
                    h[j] += 0.05
        if not improved:
            break
    return h, best_c5

def slsqp_optimize(h0, n, target, beta, maxiter=200):
    def f(x):
        hc = np.clip(x, 0, 1)
        return smooth_max(np.correlate(hc, 1.0 - hc, mode="full") * (2.0 / n), beta)

    def fp(x):
        grad = np.zeros(n)
        eps = 1e-5
        f0 = f(x)
        for i in range(n):
            x[i] += eps
            f1 = f(x)
            x[i] -= eps
            grad[i] = (f1 - f0) / eps
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=fp,
                       cons=cons, bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project(res.x, target)
    except Exception:
        pass
    return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        h0 = make_binary_start(95, seed)

    n = len(h0)
    target = n / 2.0
    h0 = project(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP with log-sum-exp smoothing
    for beta in [30, 60, 120, 250]:
        if rem() < 40:
            break
        h_new = slsqp_optimize(h0, n, target, beta, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h0 = h_new.copy()

    # Phase 2: Coordinate descent (deterministic)
    h0, best_c5 = coord_descent(h0, n, target, best_c5, max_sweeps=80, time_fn=rem)
    if all(abs(h0[i] - best_h[i]) < 1e-10 for i in range(n)):
        pass
    else:
        # Re-check
        c5_check = compute_c5(h0)
        if c5_check < best_c5:
            best_c5 = c5_check
        best_h = h0.copy() if c5_check < best_c5 else best_h

    # Phase 3: Try alternate starts with n=96 (binary possible)
    if rem() > 120:
        n2 = 96
        target2 = n2 / 2.0
        for trial in range(5):
            if rem() < 60:
                break
            h2 = make_binary_start(n2, seed + 100 + trial * 7)
            h2 = slsqp_optimize(h2, n2, target2, 50, maxiter=100)
            h2, c5_2 = coord_descent(h2, n2, target2, compute_c5(h2), max_sweeps=40, time_fn=rem)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                n = n2
                target = target2
                h0 = h2.copy()

    # Phase 4: Fine-tuning with very small steps
    if rem() > 30:
        for _ in range(3):
            if rem() < 10:
                break
            h0, best_c5 = coord_descent(h0, n, target, best_c5, max_sweeps=50, time_fn=rem)
            c5_check = compute_c5(h0)
            if c5_check < best_c5:
                best_c5 = c5_check
            best_h = h0.copy()

    # Phase 5: Random perturbations + local repair
    if rem() > 20:
        rng = np.random.RandomState(seed + 999)
        for it in range(5000):
            if rem() < 5:
                break
            h_try = best_h.copy()
            idx = rng.randint(0, n)
            h_try[idx] = np.clip(h_try[idx] + rng.uniform(-0.2, 0.2), 0, 1)
            h_try = project(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h0 = h_try.copy()

    # Final validation
    best_h = project(best_h, target)
    best_c5 = compute_c5(best_h)
    assert abs(best_h.sum() - target) < 1e-8
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
