import numpy as np
import time

def project_feasible(x, n):
    target = n / 2.0
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def cma_optimize(x0, n, sigma0, budget_fn, rng, max_gen=8000):
    lam = min(4 + int(3 * np.log(n)), 2 * n)
    mu = lam // 2
    w = np.log(mu + 0.5) - np.log(1 + np.arange(mu))
    w /= w.sum()
    m_eff = 1.0 / np.sum(w ** 2)
    c_s = (m_eff + 2.0) / (n + m_eff + 5.0)
    c_c = (4.0 + m_eff / n) / (n + 4.0 + 2.0 * m_eff / n)
    d_s = 1.0 + 2.0 * max(0.0, np.sqrt((m_eff - 1.0) / (n + 1.0)) - 1.0) + c_s
    c_1 = 2.0 / ((n + 1.3) ** 2 + m_eff)
    c_mu = min(1.0 - c_1, 2.0 * ((m_eff + 2.0) / (n + 2.0) - 1.0))
    chi_n = np.sqrt(n) * (1.0 - 1.0 / (4.0 * n) + 1.0 / (21.0 * n ** 2))

    x = x0.copy()
    sigma = sigma0
    C = np.eye(n)
    eB = np.eye(n)
    eD = np.ones(n)
    p_s = np.zeros(n)
    p_c = np.zeros(n)
    best_f = c5_eval(x)
    best_x = x.copy()

    for gen in range(max_gen):
        if budget_fn() < 20:
            break
        zs = rng.standard_normal((lam, n))
        xs = np.zeros((lam, n))
        fs = np.zeros(lam)
        for i in range(lam):
            xs[i] = project_feasible(x + sigma * (eB @ (eD * zs[i])), n)
            fs[i] = c5_eval(xs[i])
        idx = np.argsort(fs)
        xs = xs[idx]
        fs = fs[idx]
        if fs[0] < best_f:
            best_f = fs[0]
            best_x = xs[0].copy()
        old = x.copy()
        x = sum(w[i] * xs[i] for i in range(mu))
        Ch = eB @ np.diag(1.0 / eD)
        dm = x - old
        if np.linalg.norm(dm) > 1e-16:
            p_s = (1.0 - c_s) * p_s + np.sqrt(c_s * (2.0 - c_s) * m_eff) * (Ch @ (dm / sigma))
            sigma *= np.exp(c_s / d_s * (np.linalg.norm(p_s) / chi_n - 1.0))
        h_s = 1.0 if np.linalg.norm(p_s) < (1.0 + c_c / 2.0) * chi_n else 0.0
        p_c = (1.0 - c_c) * p_c + h_s * np.sqrt(c_c * (2.0 - c_c) * m_eff) * (dm / sigma)

            rm += w[i] * np.outer(v, v)
        C = (1.0 - c_mu + c_1 * (1.0 - h_s)) * C + c_1 * np.outer(p_c, p_c) + c_mu * rm

def coord_refine(h, n, budget_fn):
    best = h.copy()
    best_f = c5_eval(h)
    eps = 0.005
    for _round in range(200):
        if budget_fn() < 15:
            break
        improved = False
        for j in range(n):

                trial[j] = np.clip(trial[j] + sign * eps, 0.0, 1.0)
                trial = project_feasible(trial, n)
                f = c5_eval(trial)
                if f < best_f - 1e-12:
                    best = trial
                    best_f = f
                    improved = True
        if not improved:
            break
    return best, best_f
