- If sum(v_clipped) = n/2, done
- If sum > n/2, we need to reduce some values. The closest point means we reduce the values that are "highest" (since we're projecting onto a simplex-like constraint). Actually, this is a projection onto the intersection of a box and a hyperplane.

2. Compute the gap = target_sum - sum(clipped)
3. If gap > 0: we need to increase the sum. Add gap/n to all non-saturated values (those < 1). If some get pushed to 1, redistribute.
4. If gap < 0: we need to decrease the sum. Subtract |gap|/n from all non-saturated values (those > 0). If some get pushed to 0, redistribute.

This is the standard "Dempster's formula" or water-filling algorithm.

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = alpha * corr / dx
    # For numerical stability
    vals -= np.max(vals)
    log_sum = np.log(np.sum(np.exp(vals))) + np.max(vals)
    f = dx / alpha * log_sum
    # Gradient: weighted average of subgradients
    weights = np.exp(vals - np.max(vals))
    weights /= np.sum(weights)
    grad = np.zeros(n)

                    g[j] += (1.0 - h[j - lag])
                if 0 <= j + lag < n:
                    g[j] -= h[j + lag]

import numpy as np
import time

def project_feasible(v, target_sum):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr
    scaled_max = np.max(scaled)
    exp_vals = np.exp(scaled - scaled_max)
    log_sum_exp = np.log(np.sum(exp_vals)) + scaled_max
    f = log_sum_exp / alpha
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            for j in range(n):
                g_j = 0.0
                if 0 <= j - lag < n:
                    g_j += (1.0 - h[j - lag])
                if 0 <= j + lag < n:
                    g_j -= h[j + lag]
                grad[j] += wk * g_j
    return f, grad

def pgd_optimize(h0, max_iter, alpha, target_sum, time_limit, start_time):
    n = len(h0)
    h = project_feasible(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.003
    stall = 0

    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        f, grad = smooth_obj_and_grad(h, alpha)
        h_new = project_feasible(h - step * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            step = min(step * 1.08, 0.03)
        else:
            stall += 1
            if stall > 30:
                step *= 0.85
                stall = 0
        h = h_new
    return best_h, best_c5

def try_binary_polish(h, n, target_sum, time_budget, start_time):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(200):
        if time.time() - start_time > time_budget:
            break
        improved = False
        for j in range(n):
            if 0.05 < h[j] < 0.95:
                for target in [0.0, 1.0]:
                    h[j] = target
                    h = project_feasible(h, target_sum)
                    c5_t = compute_c5(h)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h.copy()
                        improved = True
                    else:
                        h = best_h.copy()
                        h[j] = target
                        h = project_feasible(h, target_sum)
        if not improved:
            break
    return best_h, best_c5
