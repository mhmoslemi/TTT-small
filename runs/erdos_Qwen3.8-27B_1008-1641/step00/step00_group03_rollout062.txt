                r = minimize(obj_smooth_h, best_h, jac=grad_smooth_h,
                             method='SLSQP', bounds=bounds, constraints=con,
                             tol=1e-12, options={'maxiter': 400})
                if r.x is not None:
                    hd = project_feas(r.x, _target)
                    c = compute_c5(hd, _dx)
                    if c < best_c5:
                        best_c5, best_h = c, hd
            except Exception:
                pass
    return best_h, best_c5

def phase_exact(t0, budget, h_best, c5_best, n_tries=5):
    bounds = [(0.0, 1.0)] * _n
    con = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
    for _ in range(n_tries):
        if time.time() - t0 > budget - 20:
            break
        try:
            r = minimize(obj_exact_h, h_best,
                         method='SLSQP', bounds=bounds, constraints=con,
                         tol=1e-14, options={'maxiter': 3000})
            if r.x is not None:
                hd = project_feas(r.x, _target)
                c = compute_c5(hd, _dx)
                if c < c5_best:
                    c5_best, h_best = c, hd
        except Exception:
            pass
    return h_best, c5_best

def phase_subgrad(t0, budget, h_best, c5_best):
    hc = h_best.copy()
    for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() - t0 > budget - 10:
            break
        stuck = 0
        for _ in range(15000):
            if time.time() - t0 > budget - 5:
                break
            hc = subgrad_step(hc, step)
            cn = compute_c5(hc, _dx)
            if cn < c5_best:
                c5_best, h_best = cn, hc.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck > 3000:
                break
    return h_best, c5_best

def phase_coord_search(t0, budget, h_best, c5_best):
    n = _n
    eps = [0.001, 0.005, 0.02]
    for e in eps:
        if time.time() - t0 > budget - 5:
            break
        for i in range(n):
            if time.time() - t0 > budget - 3:
                break
            for delta in [e, -e]:
                trial = h_best.copy()
                if 0.0 <= trial[i] + delta <= 1.0:
                    trial[i] += delta
                    trial = project_feas(trial, _target)
                    c = compute_c5(trial, _dx)
                    if c < c5_best:
                        c5_best, h_best = c, trial
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    global _dx, _n, _target, _tau_cur
    t0 = time.time()
    _n = 70
    _dx = 2.0 / _n
    _target = _n / 2.0
    _tau_cur = 100.0

    best_h, best_c5 = phase_smooth(t0, budget_s)
    best_h, best_c5 = phase_exact(t0, budget_s, best_h, best_c5, n_tries=4)
    best_h, best_c5 = phase_subgrad(t0, budget_s, best_h, best_c5)
    best_h, best_c5 = phase_exact(t0, budget_s, best_h, best_c5, n_tries=3)
    best_h, best_c5 = phase_coord_search(t0, budget_s, best_h, best_c5)
    best_h, best_c5 = phase_exact(t0, budget_s, best_h, best_c5, n_tries=3)
    best_h, best_c5 = phase_subgrad(t0, budget_s, best_h, best_c5)

    best_h = project_feas(best_h, _target)
    best_c5 = compute_c5(best_h, _dx)
    return best_h, best_c5, _n
