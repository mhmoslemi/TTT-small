    n = len(h)
    return np.correlate(h, 1 - h, mode="full")

    c = compute_correlations(h)

def project_onto_simplex_box(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2} using water-filling."""
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

    n = len(h)
    dx = 2.0 / n
    c = compute_correlations(h) * dx
    # Smooth max
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    log_sum = c_max + np.log(np.sum(exp_c)) / beta

    grad = np.zeros(n)

            if 0 <= j - l < n:
                val -= h[j - l]

import numpy as np
import time


def project_feasible(h, n):
    target = 0.5 * n
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        if mask.any():
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj


def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _get_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    c_all = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = np.max(c_all)
    ec = np.exp(beta * (c_all - cm))
    Z = np.sum(ec)
    p = ec / Z
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        l = k - n1
        if 0 <= l < n:
            grad[:n - l] += pk * (1.0 - h[l:])
            grad[l:] -= pk * h[:n - l]
        elif -n < l < 0:
            lp = -l
            grad[lp:] += pk * (1.0 - h[:n - lp])
            grad[:n - lp] -= pk * h[lp:]
    grad *= dx
    return grad


def pgd_schedule(h, time_limit=60.0):
    h = project_feasible(h.copy(), len(h))
    n = len(h)
    best_c5 = c5_value(h)
    best_h = h.copy()
    t0 = time.time()
    step = 0.15
    betas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        for _ in range(8000):
            if time.time() - t0 > time_limit:
                break
            grad = _get_gradient(h, beta)
            h_new = project_feasible(h - step * grad, n)
            c5_new = c5_value(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                step = min(step * 1.08, 0.5)
            else:
                step = max(step * 0.75, 0.003)
    return best_h, best_c5


def local_search(h, time_limit=30.0, seed=42):
    h = h.copy()
    n = len(h)
    best_c5 = c5_value(h)
    best_h = h.copy()
    t0 = time.time()
    rng = np.random.default_rng(seed)
    step = 0.05
    it = 0
    while time.time() - t0 < time_limit:
        it += 1
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = step * (1.0 if rng.random() < 0.5 else -1.0)
        if h[i] + d > 1.0 or h[j] - d < 0.0:
            continue
        h[i] += d
        h[j] -= d
        c5_new = c5_value(h)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h.copy()
            step = min(step * 1.04, 0.3)
        else:
            h[i] -= d
            h[j] += d
            if it % 200 == 0:
                step = max(step * 0.92, 0.002)
                h = best_h.copy()
    return best_h, best_c5


def _upscale(h_src, n_dst):
    x_src = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)


def _time_left(t_start, budget):
    return budget - (time.time() - t_start)
