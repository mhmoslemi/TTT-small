                cons_r = [{'type': 'eq', 'fun': sum_fun_r, 'jac': sum_jac_r}]
                try:
                    res = minimize(obj_r, h_work, method='SLSQP',
                                   bounds=bounds_r, constraints=cons_r,

                    if res.x is not None:

    best_h = make_feasible(best_h, n_f)
    c5_final = compute_c5(best_h)
