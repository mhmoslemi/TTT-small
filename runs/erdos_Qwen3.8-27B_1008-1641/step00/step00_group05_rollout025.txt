import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def fix_binary_sum(h, target):
    while h.sum() > target:
        ones = np.flatnonzero(h)
        if len(ones) == 0:
            break
        h[rng.choice(ones)] = 0
    while h.sum() < target:
        zeros = np.flatnonzero(1 - h)
        if len(zeros) == 0:
            break
        h[rng.choice(zeros)] = 1

def binary_sa(h, T0, T_min, cooling, max_time, start_time, budget_frac, best_h_ref, best_c5_ref):
    n = len(h)
    target = n // 2
    T = T0
    c5_cur = compute_c5(h)
    best_c5_local = c5_cur
    best_h_local = h.copy()
    end_time = start_time + budget_frac * (budget_s - 0)
    while time.time() < end_time:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0; h[j] = 1
        c5_new = compute_c5(h)
        if c5_new <= c5_cur or (T > T_min and rng.rand() < np.exp(-(c5_new - c5_cur) / T)):
            c5_cur = c5_new
            if c5_new < best_c5_local:
                best_c5_local = c5_new
                best_h_local = h.copy()
            T = max(T_min, T * cooling)
        else:
            h[i] = 1; h[j] = 0
    return best_h_local, best_c5_local

def continuous_refine(h_init, max_iters, start_time, end_time, n):
    target = n // 2
    alpha = 25.0
    h_c = h_init.astype(float).copy()
    best_c5_local = compute_c5(h_c)
    best_h_local = h_c.copy()
    dx_ = 2.0 / n
    step = 5e-7
    lr = 0.003
    
    for it in range(max_iters):
        if time.time() > end_time:
            break
        corr = np.correlate(h_c, 1.0 - h_c, mode="full") * dx_
        m = np.max(corr)
        exps = np.exp(alpha * (corr - m))
        base_val = m + np.log(np.sum(exps)) / alpha
        
        g = np.zeros(n)
        for i in range(n):
            h_c[i] += step
            corr2 = np.correlate(h_c, 1.0 - h_c, mode="full") * dx_
            m2 = np.max(corr2)
            exps2 = np.exp(alpha * (corr2 - m2))
            val2 = m2 + np.log(np.sum(exps2)) / alpha
            h_c[i] -= step
            g[i] = (val2 - base_val) / step
        
        g -= np.mean(g)
        h_c -= lr * g
        h_c = np.clip(h_c, 0, 1)
        diff = h_c.sum() - target
        if abs(diff) > 1e-10:
            free = np.flatnonzero((h_c > 1e-4) & (h_c < 1 - 1e-4))
            if len(free) > 0:
                h_c[free] -= diff / len(free)
                h_c = np.clip(h_c, 0, 1)
            d2 = h_c.sum() - target
            if abs(d2) > 1e-12:
                idx = int(np.argmax(h_c)) if d2 > 0 else int(np.argmin(h_c))
                h_c[idx] = max(0.0, min(1.0, h_c[idx] - d2))
        
        c5_val = compute_c5(h_c)
        if c5_val < best_c5_local:
            best_c5_local = c5_val
            best_h_local = h_c.copy()
        if it % 2000 == 0 and it > 0:
            lr = max(0.0005, lr * 0.92)
    return best_h_local, best_c5_local

    global rng
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    n = len(initial_h_values)
    target = n // 2
    h_best = np.array(initial_h_values, dtype=float)
    best_c5 = compute_c5(h_best)
    
    # Phase 1: Binary SA from initial (long run)
    h_s = np.where(h_best > 0.5, 1.0, 0.0)
    fix_binary_sum(h_s, target)
    h_s, c5_s = binary_sa(h_s, 0.003, 0.00001, 0.99997, 20000000, t_start, budget_s * 0.35, None, None)
    if c5_s < best_c5:
        best_c5 = c5_s
        h_best = h_s.copy()
    
    # Phase 2: Continuous refinement
    end_p2 = t_start + budget_s * 0.65
    h_c, c5_c = continuous_refine(h_best, 200000, t_start, end_p2, n)
    if c5_c < best_c5:
        best_c5 = c5_c
        h_best = h_c.copy()
    
    # Phase 3: Multiple SA restarts from perturbed best
    t_restart_start = time.time()
    restart_end = t_start + budget_s * 0.95
    attempt = 0
    while time.time() < restart_end:
        attempt += 1
        h_r = np.where(h_best > 0.5, 1.0, 0.0)
        perturb = max(2, n // (4 + attempt % 5))
        for _ in range(perturb):
            idx = rng.randint(n)
            h_r[idx] = 1.0 - h_r[idx]
        fix_binary_sum(h_r, target)
        
        T_init = max(0.0005, 0.003 / (1 + attempt * 0.1))
        h_r, c5_r = binary_sa(h_r, T_init, 0.00001, 0.99998, 500000, t_start, 0, None, None)
        if c5_r < best_c5:
            best_c5 = c5_r
            h_best = h_r.copy()
    
    # Phase 4: Final continuous polish
    end_p4 = t_start + budget_s * 0.98
    h_c, c5_c = continuous_refine(h_best, 50000, t_start, end_p4, n)
    if c5_c < best_c5:
        best_c5 = c5_c
        h_best = h_c.copy()
    
    return (h_best, best_c5, n)
