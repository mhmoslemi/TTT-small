import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def obj_fn(h, n):
    return compute_c5(h, n)

def cons_fun(h, n):
    return np.sum(h) - n / 2.0

def cons_jac_fn(h, n):
    return np.ones(n)

def smooth_obj_fn(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mu = 0.005
    c_max = np.max(corr)
    return float(c_max + mu * np.log(np.sum(np.exp((corr - c_max) / mu))))

def pen_obj_fn(h, n):
    c5 = compute_c5(h, n)
    pen = 100.0 * (np.sum(h) - n / 2.0) ** 2
    return c5 + pen

def interpolate_h(h_src, n_new):
    n_src = len(h_src)
    if n_src == n_new:
        return h_src.copy()
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def coordinate_descent(h, n, max_iter=400, time_limit=None, t0=None):
    h = project_feasible(h.copy(), n)
    deltas = [0.15, -0.15, 0.08, -0.08, 0.04, -0.04, 0.02, -0.02]
    c5_curr = compute_c5(h, n)
    for it in range(max_iter):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        improved = False
        order = np.arange(n)
        for i in order:
            if time_limit is not None and t0 is not None:
                if time.time() - t0 > time_limit:
                    break
            for d in deltas:
                hv = h[i] + d
                if hv < -1e-9 or hv > 1 + 1e-9:
                    continue
                h_trial = h.copy()
                h_trial[i] = hv
                diff = h[i] - hv
                if abs(diff) > 1e-8:
                    mask = (h_trial < 1.0) if diff > 0 else (h_trial > 0.0)
                    mask[i] = False
                    cnt = int(np.sum(mask))
                    if cnt == 0:
                        continue
                    h_trial[mask] = np.clip(h_trial[mask] + diff / cnt, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial, n)
                if c5_t < c5_curr - 1e-10:
                    h = h_trial
                    c5_curr = c5_t
                    improved = True
                    break
        if not improved:
            break
    return h, c5_curr

def binary_push(h, n, time_limit=None, t0=None):
    h = project_feasible(h.copy(), n)
    c5_curr = compute_c5(h, n)
    for _ in range(100):
