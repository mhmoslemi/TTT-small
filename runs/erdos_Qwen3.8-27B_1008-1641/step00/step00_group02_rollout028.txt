    """Fix sum of h to target while keeping h in [0,1]."""
    h = h.copy()
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        step = diff / mask.sum()
        step = max(min(step, 1.0), -1.0)  # clamp
        h[mask] += step
        h = np.clip(h, 0, 1)
    return h

    dx = 2.0 / n

    return np.max(corr) * dx

    n_old = len(h)

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
