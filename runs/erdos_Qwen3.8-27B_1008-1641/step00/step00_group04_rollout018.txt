def compute_c5(h):
    """Compute C5 score for binary or continuous h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    """Fix h to be binary with exactly target_sum ones."""
    h = np.round(h).astype(int)

    if current_sum > target_sum:
        # Need to flip some 1s to 0s

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, rng):
    n = len(h)
    h = (h > 0.5).astype(float)
    target = n // 2
    s = int(h.sum())
    if s > target:
        excess = s - target
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, excess, replace=False)] = 0
    elif s < target:
        deficit = target - s
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, deficit, replace=False)] = 1
    return h

def one_swap_ls(h, time_limit, t_start):
    n = len(h)
    h = h.astype(np.int8).copy()
    best = compute_c5(h.astype(float))
    while time.time() - t_start < time_limit:
        ones = np.where(h > 0)[0]
        zeros = np.where(h < 1)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_s = best
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] -= 1
                h[j] += 1
                s = compute_c5(h.astype(float))
                h[j] -= 1
                h[i] += 1
                if s < best_s - 1e-15:
                    best_s = s
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] -= 1
        h[j] += 1
        best = best_s
    return h.astype(float), best

def sim_anneal(h, rng, time_limit, t_start):
    n = len(h)
    h = h.astype(np.int8).copy()
    cur = compute_c5(h.astype(float))
    best = cur
    best_h = h.copy()
    T = 0.4
    while time.time() - t_start < time_limit:
        ones = np.where(h > 0)[0]
        zeros = np.where(h < 1)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] -= 1
        h[j] += 1
        new_s = compute_c5(h.astype(float))
        delta = new_s - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new_s
            if new_s < best:
                best = new_s
                best_h = h.copy()
        else:
            h[j] -= 1
            h[i] += 1
        T = max(0.005, T * 0.9997)
        if T <= 0.005 and delta >= 0:
            T = 0.3
    return best_h.astype(float), best

def upsample(h, n_target, rng):
    n_src = len(h)
    h_new = np.zeros(n_target)
    for i in range(n_src):
        s = int(round(i * n_target / n_src))
        e = int(round((i + 1) * n_target / n_src))
        h_new[s:e] = h[i]
    target = n_target // 2
    s = int(h_new.sum())
    if s > target:
        ones = np.where(h_new == 1)[0]
        h_new[rng.choice(ones, s - target, replace=False)] = 0
    elif s < target:
        zeros = np.where(h_new == 0)[0]
        h_new[rng.choice(zeros, target - s, replace=False)] = 1
    return h_new

def two_swap_ls(h, rng, time_limit, t_start):
    n = len(h)
    h = h.astype(np.int8).copy()
    best = compute_c5(h.astype(float))
    while time.time() - t_start < time_limit:
        improved = False
        ones = np.where(h > 0)[0]
        zeros = np.where(h < 1)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        for _ in range(min(500, len(ones) * len(zeros) // 4)):
            pick1 = rng.choice(ones, 2, replace=False)
            pick0 = rng.choice(zeros, 2, replace=False)
            a, b = pick1
            c, d = pick0
            h[a] -= 1; h[b] -= 1
            h[c] += 1; h[d] += 1
            s = compute_c5(h.astype(float))
            if s < best - 1e-15:
                best = s
                improved = True
            else:
                h[c] -= 1; h[d] -= 1
                h[a] += 1; h[b] += 1
            if time.time() - t_start > time_limit:
                return h.astype(float), best
        if not improved:
            break
    return h.astype(float), best

def continuous_refine(h, rng, time_limit, t_start):
    from scipy.optimize import minimize
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best = h.copy()
    best_s = compute_c5(h)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        pen = 1e3 * (np.sum(x) - n / 2) ** 2
        return float(np.max(corr) * dx + pen)

    bounds = [(0.0, 1.0)] * n
    for _ in range(4):
        if time.time() - t_start > time_limit:
            break
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 200, 'maxfun': 3000})
        h_opt = np.clip(res.x, 0.0, 1.0)
        err = h_opt.sum() - n / 2
        span = h_opt.max() - h_opt.min()
        if span > 0.01:
            h_opt -= err / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        s_opt = compute_c5(h_opt)
        if s_opt < best_s:
            best_s = s_opt
            best = h_opt.copy()
        h = np.clip(res.x, 0.0, 1.0)
        if time.time() - t_start > time_limit:
            break
    return best, best_s
