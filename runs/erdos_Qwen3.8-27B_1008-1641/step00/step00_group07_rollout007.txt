import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def repair_h(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 0.9999
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0001
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def make_penalty_obj(n):
    target_sum = n / 2.0
    def obj(z):
        h = np.clip(z, 0, 1)
        pen = 50.0 * (np.sum(h) - target_sum) ** 2
        return evaluate_c5(h) + pen
    return obj

def slsqp_optimize(h0, n, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    obj = make_penalty_obj(n)
    res = minimize(obj, h0.copy(), method='SLSQP', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    cand = repair_h(res.x, n)
    return cand, evaluate_c5(cand)

def coordinate_descent(h, n, max_rounds=200, time_limit_fn=None):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    steps = [0.1, 0.05, 0.02, 0.01]
    for rnd in range(max_rounds):
        if time_limit_fn is not None and time_limit_fn():
            break
        improved = False
        for i in range(n):
            for s in steps:
                for sign in (1, -1):
                    nv = best[i] + sign * s
                    if nv < 0 or nv > 1:
                        continue
                    trial = best.copy()
                    trial[i] = nv
                    trial = project_feasible(trial, n)
                    c5 = evaluate_c5(trial)
                    if c5 < best_c5 - 1e-10:
                        best = trial
                        best_c5 = c5
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best, best_c5

def genetic_algorithm(h_init, n, pop_size, n_gen, rng, time_limit_fn, best_c5_ref):
    pop = [h_init.copy()]
    for _ in range(pop_size - 1):
        hp = h_init + rng.randn(n) * 0.3
        pop.append(repair_h(hp, n))
    pop_c5 = [evaluate_c5(p) for p in pop]
    best_h = pop[np.argmin(pop_c5)].copy()
    best_c5 = min(pop_c5)
    elite_k = max(3, pop_size // 6)
    for gen in range(n_gen):
        if time_limit_fn():
            break
        order = np.argsort(pop_c5)
        pop = [pop[i] for i in order]
        pop_c5 = [pop_c5[i] for i in order]
        new_pop = [pop[i].copy() for i in range(elite_k)]
        while len(new_pop) < pop_size:
                i1, i2 = rng.randint(0, pop_size, 2)
                p1 = pop[i1] if pop_c5[i1] < pop_c5[i2] else pop[i2]
                j1, j2 = rng.randint(0, pop_size, 2)
                p2 = pop[j1] if pop_c5[j1] < pop_c5[j2] else pop[j2]
                cp = rng.randint(1, n - 1)
                child = np.concatenate([p1[:cp], p2[cp:]])
                mut_rate = 0.15 + 0.1 * (gen / max(n_gen, 1))
                child += rng.randn(n) * mut_rate
                child = repair_h(child, n)
                new_pop.append(child)
        pop = new_pop
        pop_c5 = [evaluate_c5(p) for p in pop]
        if pop_c5[0] < best_c5:
            best_c5 = pop_c5[0]
            best_h = pop[0].copy()
    return best_h, best_c5

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old + 1)[:-1]
    x_new = np.linspace(0, 2, n_new + 1)[:-1]
    h_new = np.interp(x_new, x_old, h, period=None)
    return repair_h(h_new, n_new)

    def time_up(frac=1.0):

    # Get initial values
    n = 80

    # Phase 1: SLSQP from initial
    try:
        cand, c5 = slsqp_optimize(best_h, n, maxiter=300)

            best_h, best_c5 = cand, c5
    except Exception:
        pass

    # Phase 2: Multi-start SLSQP
    n_restarts = 25

        h_pert = best_h + rng.randn(n) * scale

            cand, c5 = slsqp_optimize(h_pert, n, maxiter=120)

        except Exception:
            pass

    # Phase 3: GA for global search
    if not time_up(0.65):

                                        best_c5_ref=best_c5)

            best_h, best_c5 = ga_h, ga_c5

    # Phase 4: Coordinate descent refinement
    if not time_up(0.85):
        best_h, best_c5 = coordinate_descent(

    best_h = np.clip(best_h, 0.0, 1.0)

    if abs(diff) > 1e-6:
        best_h = project_feasible(best_h, n)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n)
