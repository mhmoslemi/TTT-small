            t1 = 1.0 - h[j - lag]
        t2 = 0.0
        if 0 <= j + lag < n:
            t2 = h[j + lag]
        grad[j] = dx * (t1 - t2)
    return vals[k_max], grad

def subgradient_opt(h0, n, max_iter, lr, seed, time_limit):
    rng = np.random.RandomState(seed)
    h = project_feasible(h0.copy(), n)
    best = compute_c5(h)
    best_h = h.copy()
    stale = 0
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        c5v, grad = c5_grad(h, n)
        step = h - lr * grad
        step = project_feasible(step, n)
        c5s = compute_c5(step)
        if c5s < best - 1e-14:
            best = c5s
            best_h = step.copy()
            h = step
            stale = 0
        else:
            stale += 1
            if stale > 30:
                lr *= 0.9
                if lr < 1e-6:
                    break
            h = project_feasible(h - 0.3 * lr * grad, n)
        if it % 100 == 99 and it > 0:
            noise = rng.randn(n) * (0.02 + 0.01 * (it // 100))
            noise -= noise.mean()
            h = project_feasible(h + noise, n)
            cn = compute_c5(h)
            if cn < best:
                best = cn
                best_h = h.copy()
    return best_h, best

def multi_scale_opt(h0, n, time_limit, seed):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lrs = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
    iters = [3000, 3000, 4000, 4000, 5000, 5000]
    for i, (lr, mi) in enumerate(zip(lrs, iters)):
        if time.time() > time_limit:
            break
        h, c5 = subgradient_opt(best_h, n, mi, lr, seed + i * 7, time_limit)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    return best_h, best_c5

def fine_tune(h, n, n_steps, time_limit):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for step in range(n_steps):
        if time.time() > time_limit:
            break
        c5v, grad = c5_grad(best_h, n)
        improved = False
        for alpha in [0.002, 0.001, 0.0005, 0.0002]:
            h_try = project_feasible(best_h - alpha * grad, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-14:
                best_c5 = c_try
                best_h = h_try
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 45
        h_init = np.zeros(n)
        h_init[:22] = 1.0
        h_init[22] = 0.5

    n = len(h_init)
    h_init = project_feasible(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Main optimization from initial
    h, c5 = multi_scale_opt(h_init, n, min(deadline, t0 + 300), seed)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    # Phase 2: Multiple restarts
    rng = np.random.RandomState(seed + 100)
    n_restarts = 15
    for r in range(n_restarts):
        if time.time() > deadline - 120:
            break
        perturb = 0.05 + 0.15 * rng.rand()
        noise = rng.randn(n) * perturb
        noise -= noise.mean()
        h_p = project_feasible(best_h + noise, n)
        h, c5 = multi_scale_opt(h_p, n, min(deadline - 60, t0 + 60 * (r + 1)), seed + 200 + r)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 3: Fine-tuning
    if time.time() < deadline - 20:
        remaining_steps = int((deadline - time.time()) / 0.005)
        best_h, best_c5 = fine_tune(best_h, n, max(remaining_steps, 500), deadline)

    # Final verification
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
