import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def c5_gradient(h, dx, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        for i in range(n - shift):
            grad[i] += (1.0 - h[i + shift])
            grad[i + shift] -= h[i]
    else:
        s = -shift
        for i in range(n - s):
            grad[i] -= h[i + s]
            grad[i + s] += (1.0 - h[i])
    return grad * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if diff > 0:
        # Need to increase sum; increase values farthest from 1
        order = np.argsort(1.0 - h)  # ascending, so farthest from 1 first
        for i in order:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
            if diff < 1e-15:
                break
    elif diff < 0:
        # Need to decrease sum; decrease values farthest from 0
        order = np.argsort(h)  # ascending, so farthest from 0 first (descending)
        for i in order[::-1]:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
            if diff > -1e-15:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_subgrad(h, dx, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mi = int(np.argmax(corr))
    shift = mi - (n - 1)
    g = np.zeros(n)
    if shift == 0:
        g[:] = 1.0 - 2.0 * h
    elif shift > 0:
        s, m = shift, n - shift
        g[s:] = 1.0 - h[:m]
        g[:m] -= h[s:n]
    else:
        s, m = -shift, n + shift
        g[:m] = 1.0 - h[s:n]
        g[s:] -= h[:m]
    return g * dx

def smooth_obj(h, dx, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(e)) / alpha

def smooth_grad(h, dx, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    m = np.max(f)
    e = np.exp(alpha * (f - m))
    w = e / np.sum(e)
    g = np.zeros(n)
    for i in range(len(w)):
        wi = w[i]
        if wi < 1e-12:
            continue
        shift = i - (n - 1)
        if shift == 0:
            g += wi * (1.0 - 2.0 * h)
        elif shift > 0:
            s, mm = shift, n - shift
            g[s:] += wi * (1.0 - h[:mm])
            g[:mm] -= wi * h[s:n]
        else:
            s, mm = -shift, n + shift
            g[:mm] += wi * (1.0 - h[s:n])
            g[s:] -= wi * h[:mm]
    return g * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        if diff > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
                if diff < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
                if diff > -1e-14:
                    break
        if abs(diff) < 1e-14:
            break
    return h

def slsqp_optim(h0, dx, n, budget_left):
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    try:
        res = minimize(
            lambda x: compute_c5(x, dx), h0,
            method='SLSQP', jac=lambda x: c5_subgrad(x, dx, n),
            bounds=bounds, constraints=cons,
            options={'maxiter': 600, 'ftol': 1e-14})
        if res.fun < compute_c5(h0, dx):
            return fix_sum(res.x, target)
    except Exception:
        pass
    return h0

def smooth_optim(h0, dx, n, budget_left):
    target = n / 2.0
    h = h0.copy()
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    for alpha in [5.0, 10.0, 20.0, 50.0, 100.0, 200.0]:
        if time.time() - budget_left < 0:
            break
        try:
            res = minimize(
                lambda x: smooth_obj(x, dx, n, alpha), h,
                method='SLSQP', jac=lambda x: smooth_grad(x, dx, n, alpha),
                bounds=bounds, constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-13})
            if res.fun < smooth_obj(h, dx, n, alpha) + 1e-15:
                h = fix_sum(res.x, target)
        except Exception:
            pass
    return h

def make_starts(n, x_grid, rng, h_init):
    cands = []
    if h_init is not None and len(h_init) == n:
        cands.append(h_init.copy())
    cands.append(np.full(n, 0.5))
    for freq in range(1, 6):
        for amp in [0.15, 0.3, 0.42]:
            cands.append(0.5 + amp * np.cos(2 * np.pi * freq * x_grid / 2.0))
    for _ in range(4):
        cands.append(rng.beta(2, 2, n))
    cands.append(rng.beta(8, 8, n))
    # Block patterns
    for nb in [3, 5, 7]:
        bs = max(1, n // nb)
        h = np.zeros(n)
        for b in range(nb):
            if b % 2 == 0:
                h[b*bs:min((b+1)*bs, n)] = 1.0
        cands.append(h)
    return cands

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    best_c5 = 0.5
    best_h = None
    best_n = 70

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    n_list = [70, 100, 140, 200]
    total_budget = budget_s

    for n in n_list:
        if time.time() - t_start > total_budget * 0.88:
            break
        dx = 2.0 / n
        target = n / 2.0
        x_grid = np.linspace(0, 2, n, endpoint=False)

        cands = make_starts(n, x_grid, rng, h_init)
        for h0 in cands:
            elapsed = time.time() - t_start
            if elapsed > total_budget * 0.93:
                break
            h0 = fix_sum(h0.copy(), target)
            c0 = compute_c5(h0, dx)

            # Phase 1: SLSQP on max
            h1 = slsqp_optim(h0, dx, n, total_budget - elapsed)
            c1 = compute_c5(fix_sum(h1, target), dx)
            if c1 > c0:
                h1 = h0
                c1 = c0

            # Phase 2: Smooth max refinement
            if time.time() - t_start < total_budget * 0.90:
                h2 = smooth_optim(fix_sum(h1, target), dx, n,
                                  total_budget - (time.time() - t_start))
                c2 = compute_c5(h2, dx)
                if c2 < c1:
                    h1, c1 = h2, c2

            # Phase 3: Final SLSQP polish
            if time.time() - t_start < total_budget * 0.92:
                h3 = slsqp_optim(fix_sum(h1, target), dx, n,
                                 total_budget - (time.time() - t_start))
                c3 = compute_c5(fix_sum(h3, target), dx)
                if c3 < c1:
                    h1, c1 = h3, c3

            if c1 < best_c5:
                best_c5 = c1
                best_h = fix_sum(h1, target)
                best_n = n

    # Final verification
    if best_h is None:
        best_n = 70
        best_h = np.full(best_n, 0.5)
        best_c5 = 0.5
    dx = 2.0 / best_n
    best_h = fix_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, best_n
