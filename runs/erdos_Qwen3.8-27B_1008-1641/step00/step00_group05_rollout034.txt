import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1-h, mode='full')
    return np.max(overlaps) * dx

def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    diff = n/2 - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference uniformly
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # Iterate to converge
    for _ in range(100):
        diff = n/2 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1)
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def subgradient_c5(h):
    """Compute subgradient of C5(h) = dx * max_s overlap(s)"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    # overlap(s) for s from -(n-1) to n-1
    # correlate gives index 0 to 2n-2, where index k corresponds to shift s = k-(n-1)
    corr = np.correlate(h, 1-h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr == max_val)[0]
    # Pick the worst (or average all active)
    # For each active index k, shift s = k - (n-1)
    grad = np.zeros(n)
    for k in active:

            jm = j - s
            if 0 <= js < n:
                g += (1 - h[js])
            if 0 <= jm < n:
                g -= h[jm]
            grad[j] += g
    grad /= len(active)
    return dx * grad

    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_overlap_s(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        sp = -s
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def subgradient_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = 1e-8
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        grad += grad_overlap_s(h, s)
    grad /= len(active)
    return dx * grad
