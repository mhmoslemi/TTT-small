def obj_smooth(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    beta = 80.0
    mc = np.max(corr)
    return mc + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - mc))))

def constraint_sum(h):
    return float(np.sum(h) - len(h) * 0.5)

def constraint_grad(h):
    return np.ones(len(h))

    h = h.copy()
    n = len(h)
    h += (n * 0.5 - np.sum(h)) / n
