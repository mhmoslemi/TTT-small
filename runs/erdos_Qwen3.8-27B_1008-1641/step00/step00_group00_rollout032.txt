import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_opt(h0, T, pen=200.0, maxiter=1000):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        val = (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
        s = h.sum() - target
        return val + pen * s * s
    try:
        res = minimize(obj, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)

def coord_descent(h, max_rounds=200):
    n = len(h)
    best = compute_c5(h)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for _ in range(max_rounds):
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                for d in deltas:
                    for s in (1, -1):
                        hi, hj = h[i] - s * d, h[j] + s * d
                        if hi < -1e-9 or hi > 1.0 + 1e-9 or hj < -1e-9 or hj > 1.0 + 1e-9:
                            continue
                        hn = h.copy()
                        hn[i] = max(0.0, min(1.0, hi))
                        hn[j] = max(0.0, min(1.0, hj))
                        c = compute_c5(hn)
                        if c < best - 1e-13:
                            h, best, imp = hn, c, True
        if not imp:
            break
    return h, best

def upsample(h, n_dst):

    if r > 0 and r * len(h) == n_dst:
        return np.tile(h, r)
    idx = np.clip(np.arange(n_dst) * len(h) // n_dst, 0, len(h) - 1)
    return h[idx].copy()
