    #                     = 0 for h+alpha ≤ 0  => h ≤ -alpha
    #                     = 1 for h+alpha ≥ 1  => h ≥ 1-alpha
    # sum = sum_{h+alpha in (0,1)} (h+alpha) + count_{h ≥ 1-alpha} - count_{h ≤ -alpha} * 0
    #     = sum_{mid} h + alpha * count_mid + count_high
    # where mid = {h: -alpha < h < 1-alpha}, high = {h: h ≥ 1-alpha}
    
    # Bisection on alpha
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid_alpha = (lo + hi) / 2
        vals = np.clip(h + mid_alpha, 0, 1)
        s = np.sum(vals)
        if s < target:
            lo = mid_alpha
        else:
            hi = mid_alpha
    h = np.clip(h + (lo+hi)/2, 0, 1)
    # Final adjustment (should be very close)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute among non-bound elements
        mask = (h > 0) & (h < 1)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    z = tau * (c - c_max)
    z_max = z.max()

    denom = exp_z.sum()
    w = exp_z / denom

import time

def _project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.any():
            h[m] += diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _obj_grad(h, n, tau):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    z = tau * (c - cm)
    zm = z.max()
    ez = np.exp(z - zm)
    S = ez.sum()
    w = ez / S
    obj = dx * (cm + (zm + np.log(S)) / tau)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-12:
            continue
        d = j - (n - 1)
        if d >= 0:
            a1 = np.arange(d, n)
            h1 = np.arange(0, n - d)
            grad[a1] += wj * (1.0 - h[h1])
            if d > 0:
                a2 = np.arange(0, n - d)
                h2 = np.arange(d, n)
                grad[a2] -= wj * h[h2]
            else:
                grad[:] -= wj * h  # d=0: term2 is h itself
        else:
            ad = -d
            a1 = np.arange(0, n - ad)
            h1 = np.arange(ad, n)
            grad[a1] += wj * (1.0 - h[h1])
            a2 = np.arange(ad, n)
            h2 = np.arange(0, n - ad)
            grad[a2] -= wj * h[h2]
    grad *= dx
    return obj, grad
