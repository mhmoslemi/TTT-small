    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def grad_lag(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1 - h[k:])
            g[k:] -= h[:m]
    else:
        kk = -k
        m = n - kk
        if m > 0:
            g[kk:] += (1 - h[:m])
            g[:m] -= h[kk:]
    return g

def subgrad_refine(h, n, dx, max_time, top_k=1, lr_init=0.02):
    h = h.astype(float).copy()
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    t0 = time.time()
    lr = lr_init
    for it in range(100000):
        if time.time() - t0 > max_time:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        if top_k > 1:
            thresh = np.max(corr) - top_k * 0.5 * dx
            active = np.where(corr >= thresh)[0]
            active = active[:min(len(active), top_k * 2)]
            grad = np.zeros(n)
            for li in active:
                grad += grad_lag(h, li - (n - 1), n)
            grad /= max(len(active), 1)
        else:
            max_idx = np.argmax(corr)
            grad = grad_lag(h, max_idx - (n - 1), n)
        h -= lr * grad
        h = project_sum(h, target)
        val = compute_c5(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        if it % 300 == 299:
            lr = max(0.0003, lr * 0.88)
    return best_h, best_val

def bin_local_search(h, n, dx, max_time):
    h_b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(h_b))
    while s > target:
        ones = np.where(h_b == 1)[0]
        i = ones[np.argmin(h_b[ones])] if len(ones) else 0
        h_b[i] = 0
        s -= 1
    while s < target:
        zeros = np.where(h_b == 0)[0]
        i = zeros[np.argmax(h_b[zeros])] if len(zeros) else 0
        h_b[i] = 1
        s += 1
    best_val = compute_c5(h_b, dx)
    t0 = time.time()
    while time.time() - t0 < max_time:
        ones = np.where(h_b == 1)[0].tolist()
        zeros = np.where(h_b == 0)[0].tolist()
        improved = False
        for i in ones:
            for j in zeros:
                h_b[i] = 0; h_b[j] = 1
                v = compute_c5(h_b, dx)
                if v < best_val - 1e-12:
                    best_val = v
                    improved = True
                    ones = np.where(h_b == 1)[0].tolist()
                    zeros = np.where(h_b == 0)[0].tolist()
                    break
                else:
                    h_b[i] = 1; h_b[j] = 0
            if improved:
                break
        if not improved:
            # Try double-swap (2 ones -> 2 zeros and vice versa)
            improved = False
            for _ in range(200):
                if time.time() - t0 > max_time:
                    break
                idx1 = np.random.randint(len(ones))
                idx2 = np.random.randint(len(ones) - 1)
                if idx2 >= idx1: idx2 += 1
                idx3 = np.random.randint(len(zeros))
                idx4 = np.random.randint(len(zeros) - 1)
                if idx4 >= idx3: idx4 += 1
                a, b = ones[idx1], ones[idx2]
                c, d = zeros[idx3], zeros[idx4]
                h_b[a]=0; h_b[b]=0; h_b[c]=1; h_b[d]=1
                v = compute_c5(h_b, dx)
                if v < best_val - 1e-12:
                    best_val = v
                    improved = True
                    ones = np.where(h_b == 1)[0].tolist()
                    zeros = np.where(h_b == 0)[0].tolist()
                else:
                    h_b[a]=1; h_b[b]=1; h_b[c]=0; h_b[d]=0
            if not improved:
                break
    return h_b, best_val

def make_patterns(n, seed):
    rng = np.random.default_rng(seed)
    target = n // 2
    pats = []
    # Periodic patterns
    for period in range(3, 13):
        for ones_per in range(1, period):
            base = np.concatenate([np.ones(ones_per), np.zeros(period - ones_per)])
            pat = np.tile(base, n // period + 1)[:n]
            while np.sum(pat) > target:
                idx = np.where(pat > 0.5)[0]
                pat[idx[rng.integers(len(idx))]] = 0
            while np.sum(pat) < target:
                idx = np.where(pat < 0.5)[0]
                pat[idx[rng.integers(len(idx))]] = 1
            if int(np.sum(pat)) == target:
                pats.append(pat.copy())
    # Block: ones in center
    pat = np.zeros(n)
    start = n // 4
    pat[start:start + target] = 1
    pats.append(pat)
    # Random
    for _ in range(3):
        pat = np.zeros(n)
        idx = rng.permutation(n)[:target]
        pat[idx] = 1
        pats.append(pat)
    return pats[:8]

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)
    try:
        n = len(initial_h_values)
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 70
        h_init = (np.arange(n) % 2).astype(float)
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, dx)

    def time_left():
        return budget_s - (time.time() - t_start) - 25

    # Phase 1: Subgradient refinement (multiple configs)
    if time_left() > 30:
        alloc = min(180, time_left() * 0.25)
        for tk in [1, 2, 3]:
            if time_left() < 15: break
            h1, c1 = subgrad_refine(h_init, n, dx, alloc / 3.0, top_k=tk, lr_init=0.02)
            if c1 < best_c5:
                best_c5, best_h = c1, h1.copy()
            h1, c1 = subgrad_refine(h_init, n, dx, alloc / 3.0, top_k=tk, lr_init=0.05)
            if c1 < best_c5:
                best_c5, best_h = c1, h1.copy()

    # Phase 2: Binary local search from best
    if time_left() > 25:
        alloc = min(200, time_left() * 0.3)
        h2, c2 = bin_local_search(best_h, n, dx, alloc)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Try alternate starting patterns
    if time_left() > 40:
        patterns = make_patterns(n, seed)
        for pat in patterns:
            if time_left() < 20: break
            alloc = min(50, time_left() * 0.1)
            h3, c3 = subgrad_refine(pat, n, dx, alloc * 0.6, top_k=2, lr_init=0.03)
            if c3 < best_c5:
                best_c5, best_h = c3, h3.copy()
            h3b, c3b = bin_local_search(best_h, n, dx, alloc * 0.4)
            if c3b < best_c5:
                best_c5, best_h = c3b, h3b.copy()

    # Phase 4: Final polish with subgradient
    if time_left() > 15:
        h4, c4 = subgrad_refine(best_h, n, dx, time_left(), top_k=1, lr_init=0.005)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()
        # Also try binary refinement of the relaxed solution
        if time_left() > 10:
            h4b, c4b = bin_local_search(best_h, n, dx, time_left() * 0.7)
            if c4b < best_c5:
                best_c5, best_h = c4b, h4b.copy()

    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, n / 2.0)
    final_c5 = compute_c5(best_h, dx)
    if final_c5 > best_c5:
        best_c5 = final_c5
    return (best_h, final_c5, n)
