import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    val = float(corr[k_max] * dx)
    s = k_max - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    else:
        sp = -s
        grad[sp:] += (1.0 - h[:n - sp])
        grad[:n - sp] -= h[sp:]
    return val, grad * dx


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def _cfun(x):
    return float(np.sum(x) - len(x) / 2.0)


def _cjac(x):
    return np.ones(len(x))


def slsqp_optimize(h_start, n_pts, max_iter, ftol):
    bounds = [(0.0, 1.0)] * n_pts
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        res = minimize(compute_c5_grad, h_start, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start


def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.arange(n_new, dtype=float) * n_old / n_new
    return np.interp(x_new, x_old, h)


def make_structured_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    candidates = []
    candidates.append(np.full(n, 0.5))
    for freq in [2, 3, 4, 5, 7]:
        p = 0.5 + 0.5 * np.cos(np.pi * freq * x)
        candidates.append(np.clip(p, 0, 1))
    block_size = max(1, n // 6)
    for phase in range(3):
        p = np.zeros(n)
        for i in range(n):
            if ((i + phase) // block_size) % 2 == 0:
                p[i] = 1.0
        candidates.append(p)
    candidates.append(rng.uniform(0, 1, n))
    return candidates
