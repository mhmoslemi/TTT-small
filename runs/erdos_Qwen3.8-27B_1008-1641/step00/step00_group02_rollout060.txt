import numpy as np
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(y, n):
    target = n * 0.5
    y = np.clip(y, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def grad_c5(h, dx, n):
    eps = 1e-7
    g = np.zeros(n)
    for i in range(n):
        hp = h.copy()
        hp[i] += eps
        hm = h.copy()
        hm[i] -= eps
        g[i] = (compute_c5(hp, dx) - compute_c5(hm, dx)) / (2.0 * eps)
    return g

def gradient_descent(h_init, n, dx, deadline, rng):
    h = h_init.copy()
    c5_cur = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5_cur
    lr = 0.04
    stalled = 0
    while time.time() < deadline:
        g = grad_c5(h, dx, n)
        h_new = project_feasible(h - lr * g, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5_cur - 1e-13:
            h = h_new
            c5_cur = c5_new
            lr = min(lr * 1.01, 0.15)
            stalled = 0
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            stalled += 1
            lr *= 0.94
            if lr < 1e-8 or stalled > 300:
                jump_size = 0.05 + 0.1 * min(stalled / 1000.0, 1.0)
                h = project_feasible(h + rng.standard_normal(n) * jump_size, n)
                c5_cur = compute_c5(h, dx)
                lr = 0.04
                stalled = 0
    return best_h, best_c5

def perturbation_search(h_init, n, dx, deadline, rng):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    while time.time() < deadline:
        size = int(rng.integers(1, min(4, n)))
        idx = rng.choice(n, size=size, replace=False)
        hp = h.copy()
        hp[idx] += rng.uniform(-0.2, 0.2, size=size)
        hp = project_feasible(hp, n)
        c5p = compute_c5(hp, dx)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()
            h = hp.copy()
    return best_h, best_c5

def make_block(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    ones = rng.choice(n, size=half, replace=False)
    h[ones] = 1.0
    if n % 2 == 1:
        remaining = np.where(h == 0)[0]
        h[int(rng.choice(remaining))] = 0.5
    return project_feasible(h, n)

def try_binary_rounding(h, n, dx, rng, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    half = n // 2
    while time.time() < deadline:
        h_try = np.zeros(n)

    return best_h, best_c5
