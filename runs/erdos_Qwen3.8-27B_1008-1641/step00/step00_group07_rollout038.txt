import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum} via bisection."""
    lo = h.min() - target_sum / h.size - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_c5(h):
    """Compute C5 value and subgradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    k_star = int(np.argmax(corr))
    d = k_star - (n - 1)
    
    grad = np.zeros(n)
    # Vectorized computation
    # Term 1: [0 <= m-d < n] * (1-h[m-d])
    # Term 2: -[0 <= m+d < n] * h[m+d]
    if d >= 0:
        # m-d in [0,n-1] means m in [d, n-1]
        # m+d in [0,n-1] means m in [0, n-1-d]
        grad[d:] += (1.0 - h[:n-d])
        grad[:n-d] -= h[d:]
    else:
        dd = -d
        # m-d = m+dd in [0,n-1] means m in [0, n-1-dd]
        # m+d = m-dd in [0,n-1] means m in [dd, n-1]
        grad[:n-dd] += (1.0 - h[dd:])
        grad[dd:] -= h[:n-dd]
    
    return corr[k_star] * dx, grad * dx

def optimize_pgd(h0, n_iters, lr, rtol=1e-10):
    """Projected subgradient descent."""
    h = h0.copy()
    target_sum = len(h) / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    
    lr_cur = lr
    for it in range(n_iters):
        val, grad = subgrad_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # Staleness check
        if it > 100 and val >= best_val * (1 + 1e-8):

            h[mask] += np.random.randn(len(mask)) * 0.01
            h = project(h, target_sum)
