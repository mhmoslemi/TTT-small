def gradient_at_lag(h, t):
    N = len(h)
    grad = np.zeros(N)
    lo1 = max(0, t)
    hi1 = min(N, t + N)
    if hi1 > lo1:
        idx_src = np.arange(lo1 - t, hi1 - t)
        grad[lo1:hi1] += (1.0 - h[idx_src])
    lo2 = max(0, -t)
