import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def perturb(h, rng, scale, target):
    return project_sum(h + rng.normal(0, scale, len(h)), target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:

    n = len(h0)
    target = n / 2.0
    h_best = project_sum(h0, target)
    c5_best = compute_c5(h_best)

    def tleft():
        return budget_s - (time.time() - t0) - 5.0

    nm_opts = {'maxiter': 600, 'xatol': 1e-6, 'fatol': 1e-8, 'adaptive': True}

    def nm_obj(x):
        return compute_c5(project_sum(x, target))

    # Phase 1: Nelder-Mead with restarts
    for r in range(25):
        if tleft() < 25:
            break
        if r == 0:
            x0 = h_best.copy()
        else:
            sc = 0.02 + 0.015 * min(r, 8)
            x0 = perturb(h_best, rng, sc, target)
        try:
            res = optimize.minimize(nm_obj, x0, method='Nelder-Mead', options=nm_opts)
            xo = project_sum(res.x, target)
            v = compute_c5(xo)
            if v < c5_best:
                c5_best = v
                h_best = xo.copy()

            pass

    # Phase 2: SLSQP with constraints

        for r in range(6):
            if tleft() < 40:
                break
            x0 = perturb(h_best, rng, 0.03 * (r + 1), target) if r > 0 else h_best.copy()
            try:

                                        constraints=cons,
                                        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False})
                xo = project_sum(res.x, target)
                v = compute_c5(xo)
                if v < c5_best:
                    c5_best = v
                    h_best = xo.copy()
            except:
                pass

    # Phase 3: Coordinate-wise local search
    if tleft() > 30:
        deltas = np.array([-0.15, -0.1, -0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15])
        step_mult = 1.0

        for i in range(50000):

            if stagnant > 500:

                stagnant = 0
