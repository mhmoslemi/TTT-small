import time

def project_feasible(v, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v + mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def obj_func(h, dx):
    return compute_c5(h, dx)

    return np.sum(h) - target_sum

def slsqp_solve(x0, target_sum, dx, maxiter=300):

    bnds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (target_sum,)}
    res = minimize(obj_func, x0, args=(dx,), method='SLSQP',

                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res
