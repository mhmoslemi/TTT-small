        step = eps if old + eps <= 1.0 else -eps

        h[j] = old

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def constraint_eq(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_eq_jac(h):
    return np.ones(len(h))

def objective_fn(h):
    return eval_c5(h)

def objective_grad(h):
    n = len(h)
    eps = 1e-6
    grad = np.zeros(n)
    f0 = eval_c5(h)
    for j in range(n):
        old = h[j]
        step = eps if old + eps <= 1.0 else (-eps if old - eps >= 0.0 else eps)
        h[j] = old + step
        f1 = eval_c5(h)
        h[j] = old
        grad[j] = (f1 - f0) / step
    return grad

def make_bounds(n):
    return [(0.0, 1.0)] * n

def fix_sum(h, n):
    h = h.copy()
    for _ in range(50):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / int(mask.sum())
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / int(mask.sum())
        h = np.clip(h, 0.0, 1.0)
    # Final uniform micro-correction
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h, constraints, bounds, maxiter=400):
    try:
        res = minimize(objective_fn, h, jac=objective_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return res
    except Exception:
        return None

        return None
