    best_c5 = compute_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    def obj(x):
        return compute_c5(project(x, n))

    # Phase 1: Nelder-Mead from initial and perturbed starts
    perturbs = [0.0, 0.02, 0.05, 0.1, 0.15, 0.25, 0.4]
    for ps in perturbs:
        if rem() < 15:
            break
        x0 = best_h + rng.normal(0, ps, n) if ps > 0 else best_h
        x0 = project(x0, n)
        res = minimize(obj, x0, method='Nelder-Mead',
                      options={'maxiter': 100000, 'fatol': 1e-16,
                               'xatol': 1e-16, 'adaptive': True})
        fh = project(res.x, n)
        fc = compute_c5(fh)
        if fc < best_c5:
            best_c5 = fc
            best_h = fh

    # Phase 2: Powell refinement
    if rem() > 20:
        res = minimize(obj, best_h, method='Powell',
                      options={'maxiter': 200000, 'ftol': 1e-16, 'xtol': 1e-16})
        fh = project(res.x, n)
        fc = compute_c5(fh)
        if fc < best_c5:
            best_c5 = fc
            best_h = fh

    # Phase 3: More random restarts with Nelder-Mead
    for i in range(30):
        if rem() < 15:
            break
        scale = 0.05 + 0.1 * (i % 3)
        x0 = best_h + rng.normal(0, scale, n)
        x0 = project(x0, n)
        res = minimize(obj, x0, method='Nelder-Mead',
                      options={'maxiter': 80000, 'fatol': 1e-16,
                               'xatol': 1e-16, 'adaptive': True})
        fh = project(res.x, n)
        fc = compute_c5(fh)
        if fc < best_c5:
            best_c5 = fc
            best_h = fh

    # Phase 4: Differential evolution (global search)
    if rem() > 80:
        def de_obj(x):
            if rem() < 3:
                return 999.0
            return compute_c5(project(x, n))
        try:
            res = differential_evolution(de_obj, [(0.0, 1.0)] * n, seed=seed,
                                        maxiter=200, popsize=12, tol=1e-14,
                                        mutation=(0.3, 1.0), recombination=0.8)
            fh = project(res.x, n)
            fc = compute_c5(fh)
            if fc < best_c5:
                best_c5 = fc
                best_h = fh
        except Exception:
            pass

    # Phase 5: Final local polish (NM + Powell)
    if rem() > 15:
        res = minimize(obj, best_h, method='Nelder-Mead',
                      options={'maxiter': 100000, 'fatol': 1e-18,
                               'xatol': 1e-18, 'adaptive': True})
        fh = project(res.x, n)
        fc = compute_c5(fh)
        if fc < best_c5:
            best_c5 = fc
            best_h = fh

    if rem() > 10:
        res = minimize(obj, best_h, method='Powell',
                      options={'maxiter': 100000, 'ftol': 1e-18, 'xtol': 1e-18})
        fh = project(res.x, n)
        fc = compute_c5(fh)
        if fc < best_c5:
            best_c5 = fc
            best_h = fh

    # Try snapping: quantize to improve (fractional values at edges)
    for snap_frac in [0.25, 0.5, 0.75]:
        if rem() < 5:
            break
        hs = best_h.copy()
        hs[hs < snap_frac * 0.5] = 0.0
        hs[hs > 1.0 - snap_frac * 0.5] = 1.0
        hs = project(hs, n)
        fc = compute_c5(hs)
        if fc < best_c5:
            best_c5 = fc
            best_h = hs
            # Re-polish after snap
            res = minimize(obj, best_h, method='Nelder-Mead',
                          options={'maxiter': 50000, 'fatol': 1e-16, 'adaptive': True})
            fh = project(res.x, n)
            fc2 = compute_c5(fh)
            if fc2 < best_c5:
                best_c5 = fc2
                best_h = fh

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
