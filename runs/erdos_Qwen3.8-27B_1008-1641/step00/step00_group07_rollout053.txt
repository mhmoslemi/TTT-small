import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)

                   constraints=cons, options={'maxiter': maxit, 'ftol': 1e-14})
    return res.x if res.x is not None else h

def _local_search(h, target, rng, time_limit, best_c5_init):
    """Pairwise transfer local search."""
    n = len(h)
    h = h.copy()
    best_c5 = best_c5_init
    timeout = time.time() + time_limit
    deltas = np.array([0.02, 0.05, 0.1, 0.2, 0.4, 0.7])
    iters = 0
    while time.time() < timeout:
        iters += 1
        if iters % 5000 == 0:
            c5 = _c5(h)
            if c5 < best_c5:
                best_c5 = c5
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            for direction in (1, -1):
                if direction == 1:
                    if h[i] < d or h[j] > 1.0 - d:
                        continue
                    h_try = h.copy()
                    h_try[i] -= d
                    h_try[j] += d
                else:
                    if h[j] < d or h[i] > 1.0 - d:
                        continue
                    h_try = h.copy()
                    h_try[i] += d
                    h_try[j] -= d
                c5_try = _c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    h = h_try
                    best_c5 = c5_try
    h = _repair(h, target)
    c5 = _c5(h)
    if c5 < best_c5:
        best_c5 = c5
    return h, best_c5

def _binary_round(h, target):
    """Push h toward binary values while maintaining sum."""
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)
    # Assign 1s to top n/2 positions
    h_bin[order[n//2:]] = 1.0
    # Adjust: ensure sum is exactly target
    s = np.sum(h_bin)
    if s != target:
        # Find fractional adjustment
        if s > target:
            idx = order[:int(s - target)]
            h_bin[idx] = 0.0
        elif s < target:
            idx = order[::-1][:int(target - s)]
            h_bin[idx] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def elapsed():
        return time.time() - t0

    # Initialize
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except:
        h_best = np.ones(80) * 0.5
    n = len(h_best)
    target = n / 2.0
    h_best = _repair(h_best, target)
    best_c5 = _c5(h_best)

    # Also try binary rounding of initial
    h_bin = _binary_round(h_best, target)
    c5_bin = _c5(h_bin)
    if c5_bin < best_c5:
        h_best = h_bin.copy()
        best_c5 = c5_bin

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    # Phase 1: SLSQP with increasing smoothness (40% budget)
    p1_time = budget_s * 0.35
    for beta in [8, 15, 30, 60, 100, 150, 300]:
        if elapsed() > p1_time:
            break
        maxit = max(100, int((p1_time - elapsed()) * 8))
        x_new = _slsqp_opt(h_best, target, bounds, cons, beta, maxit)
        x_new = _repair(x_new, target)
        c5 = _c5(x_new)
        if c5 < best_c5:
            best_c5 = c5
            h_best = x_new

    # Try from binary starting point too
    if elapsed() < p1_time * 0.7:
        for beta in [30, 80, 150]:
            if elapsed() > p1_time:
                break
            x_new = _slsqp_opt(h_bin, target, bounds, cons, beta, 300)
            x_new = _repair(x_new, target)
            c5 = _c5(x_new)
            if c5 < best_c5:
                best_c5 = c5
                h_best = x_new

    # Phase 2: Local pairwise search (30% budget)
    p2_time = budget_s * 0.30
    if elapsed() < t0 + p2_time + p1_time:
        remaining_p2 = max(10, (t0 + p1_time + p2_time) - time.time())
        h_best, best_c5 = _local_search(h_best, target, rng, remaining_p2, best_c5)

    # Phase 3: Random restarts with SLSQP (remaining time)
    p3_start = time.time()
    p3_limit = t0 + budget_s * 0.92
    restart_count = 0
    while time.time() < p3_limit:
        restart_count += 1
        # Perturb best solution
        noise_scale = max(0.02, 0.15 - 0.01 * restart_count)
        h_try = h_best + rng.randn(n) * noise_scale
        h_try = _repair(h_try, target)
        # Quick SLSQP
        beta = 100 if restart_count < 5 else 60
        maxit = max(50, int(20))
        x_new = _slsqp_opt(h_try, target, bounds, cons, beta, maxit)
        x_new = _repair(x_new, target)
        c5 = _c5(x_new)
        if c5 < best_c5:
            best_c5 = c5
            h_best = x_new
            # Do local search on new best
            if time.time() + 10 < p3_limit:
                h_best, best_c5 = _local_search(h_best, target, rng, 8, best_c5)

    # Final: ensure feasibility
    h_best = _repair(h_best, target)
    final_c5 = _c5(h_best)

    return h_best, final_c5, n
