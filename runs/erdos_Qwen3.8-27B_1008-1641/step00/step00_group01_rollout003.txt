        h0 = None
    if h0 is not None and len(h0) > 0:
        best_h = h0.copy()
        best_n = len(h0)
    else:
        best_n = 47
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h, best_n)

    # --- Helper: try SLSQP from a start ---
    def try_slsqp(h_start, n_pts, alpha, maxiter=500):
        target = n_pts / 2.0
        bnds = [(0.0, 1.0)] * n_pts
        cnstr = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
        res = minimize(obj_fn, h_start.copy(), args=(n_pts, alpha),
                       method='SLSQP', bounds=bnds, constraints=cnstr,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return repair(res.x, n_pts)

    # --- Strategy 1: Local refinement from initial with increasing alpha ---
    n = best_n
    h_work = repair(best_h.copy(), n)
    for alpha in [10, 30, 50, 100, 200, 500, 1000, 2000, 5000]:
        if remaining() < 5:
            break
        try:
            h_new = try_slsqp(h_work, n, alpha, 800)
            c5 = compute_c5(h_new, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h_work = h_new.copy()
        except Exception:
            pass

    # --- Strategy 2: Multi-start local optimization ---
    n = best_n
    starts = []
    # Perturbations of best
    for scale in [0.02, 0.05, 0.1, 0.15]:
        for _ in range(5):
            hp = best_h + rng.normal(0, scale, n)
            starts.append(repair(hp, n))
    # Block patterns
    for w in range(4, n, 2):
        hb = np.zeros(n)
        mid = n // 2
        lo = max(0, mid - w // 2)
        hi = min(n, mid + w // 2 + 1)
        hb[lo:hi] = 1.0
        s = hb.sum()
        if s > 0:
            hb *= (n / 2.0) / s
        starts.append(repair(np.clip(hb, 0, 1), n))
    # Periodic
    for p in range(2, min(n, 30)):
        hp = np.array([1.0 if (i % p) < p // 2 else 0.0 for i in range(n)])
        s = hp.sum()
        if s > 0:
            hp *= (n / 2.0) / s
        starts.append(repair(np.clip(hp, 0, 1), n))
    # Random
    for _ in range(15):
        hr = (rng.random(n) > 0.5).astype(float)
        s = hr.sum()
        if s == 0:
            hr = np.ones(n) * 0.5
            s = n
        hr *= (n / 2.0) / s
        starts.append(repair(np.clip(hr, 0, 1), n))

    for h_s in starts:
        if remaining() < 8:
            break
        try:
            h_new = try_slsqp(h_s, n, 200, 300)
            c5 = compute_c5(h_new, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass

    # --- Strategy 3: Try different n_points ---
    for n_try in [31, 37, 41, 43, 47, 49, 53, 59, 61, 67, 73, 79, 83, 97]:
        if remaining() < 15:
            break
        if n_try == best_n:
            continue
        # Interpolate
        h_interp = np.interp(
            np.linspace(0, 1, n_try, endpoint=False),
            np.linspace(0, 1, best_n, endpoint=False),
            best_h
        )
        h_interp = repair(h_interp, n_try)
        try:
            h_new = try_slsqp(h_interp, n_try, 200, 400)
            c5 = compute_c5(h_new, n_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_try
        except Exception:
            pass

    # --- Strategy 4: COBYLA-based refinement (derivative-free) ---
    n = best_n
    target = n / 2.0
    h_work = repair(best_h.copy(), n)
    # COBYLA with penalty for constraint
    def cobyla_obj(x):
        x = np.clip(x, 0, 1)
        diff = x.sum() - target
        penalty = diff * diff * 1e6
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr * dx, 500) + penalty

    if remaining() > 30:
        try:
            res = minimize(cobyla_obj, h_work.copy(), method='COBYLA',
                           options={'maxiter': 500, 'rhobeg': 0.01})
            h_new = repair(res.x, n)
            c5 = compute_c5(h_new, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass

    # --- Strategy 5: Final high-alpha refinement ---
    n = best_n
    h_work = repair(best_h.copy(), n)
    for alpha in [2000, 5000, 10000, 20000]:
        if remaining() < 3:
            break
        try:
            h_new = try_slsqp(h_work, n, alpha, 1000)
            c5 = compute_c5(h_new, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h_work = h_new.copy()
        except Exception:
            pass

    # Final validation
    best_h = repair(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return (best_h, final_c5, best_n)
