                h = np.clip(h, 0, 1)

        temp = max(0.003, temp * 0.99995)

    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, ts)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)
