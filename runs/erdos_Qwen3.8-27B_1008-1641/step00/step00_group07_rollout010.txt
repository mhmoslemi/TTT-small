import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_bounds_sum(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_worst_lag(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        p = np.arange(n)
        mask1 = p >= k
        mask2 = p <= n - 1 - k
        grad[mask1] += (1.0 - h[p[mask1] - k])
        grad[mask2] -= h[p[mask2] + k]
    else:
        kk = -k
        p = np.arange(n)
        mask1 = (p < kk) & (p + kk < n)
        mask2 = (p >= kk) & (p - kk < kk)
        grad[mask1] += (1.0 - h[p[mask1] + kk])
        grad[mask2] -= h[p[mask2] - kk]
    return grad, corr

def binary_ls(h, time_limit, rng, anneal=True):
    n = len(h)
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    start = time.time()
    c5 = compute_c5(h)
    T = 0.005
    T_min = 1e-6
    no_improve = 0
    while time.time() - start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        order_i = rng.permutation(len(ones))
        order_j = rng.permutation(len(zeros))
        for ai in order_i:
            if time.time() - start > time_limit:
                break
            i = ones[ai]
            for aj in order_j:
                j = zeros[aj]
                h[i] = 0.0
                h[j] = 1.0
                c5_new = compute_c5(h)
                delta = c5_new - c5
                accept = delta < -1e-15
                if not accept and anneal:
                    accept = rng.random() < np.exp(-delta / max(T, 1e-15))
                if accept:
                    c5 = c5_new
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if improved:
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            T = max(T * 0.9, T_min)
            if no_improve > 20 and T <= T_min:
                break
            if no_improve > 20:
                o = np.where(h == 1.0)[0]
                z = np.where(h == 0.0)[0]
                h[rng.choice(o)] = 0.0
                h[rng.choice(z)] = 1.0
                c5 = compute_c5(h)
    return h, compute_c5(h)

def fractional_refine(h, time_limit, lr_init=0.08):
    n = len(h)
    target = n / 2.0
    h = project_bounds_sum(h.copy(), target)
    start = time.time()
    c5 = compute_c5(h)
    lr = lr_init
    stuck = 0
    while time.time() - start < time_limit:
        grad, _ = grad_at_worst_lag(h)
        h_new = project_bounds_sum(h - lr * grad, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-15:
            h = h_new
            c5 = c5_new
            lr = min(lr * 1.03, 0.5)
            stuck = 0
        else:
            lr *= 0.7
            stuck += 1
            if lr < 1e-8 or stuck > 50:
                break
    return h, compute_c5(h)

def make_random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def upsample(h, n_new):
    n_old = len(h)
    ratios = (n_new - 1) / (n_old - 1) if n_old > 1 else 0
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h, left=0, right=0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    budget = budget_s - 10  # safety margin

    best_h = None
    best_c5 = 1.0
    best_n = 0

    # Use initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        c0 = compute_c5(h0)
        best_h = h0.copy()
        best_c5 = c0
        best_n = n0
    else:
        h0 = make_random_binary(80, rng)
        best_h = h0.copy()
        best_c5 = compute_c5(h0)
        best_n = 80

    # Phase 1: Binary local search on initial (n=80)
    t_phase = min(300, budget * 0.3)
    h_bin, c5_bin = binary_ls(best_h, t_phase, rng, anneal=True)
    if c5_bin < best_c5:
        best_h, best_c5, best_n = h_bin, c5_bin, len(h_bin)

    elapsed = time.time() - t0
    remaining = budget - elapsed

    # Phase 2: Fractional refinement
    t_frac = min(remaining * 0.4, 250)
    if t_frac > 10:
        h_frac, c5_frac = fractional_refine(best_h, t_frac)
        if c5_frac < best_c5:
            best_h, best_c5, best_n = h_frac, c5_frac, len(h_frac)

    elapsed = time.time() - t0
    remaining = budget - elapsed

    # Phase 3: More binary restarts on n=80
    if remaining > 30:
        t_restarts = remaining * 0.5
        chunk = t_restarts / 4
        for r in range(4):
            if time.time() - t0 > budget:
                break
            h_rand = make_random_binary(80, rng)
            h_r, c5_r = binary_ls(h_rand, chunk, rng, anneal=True)
            if c5_r < best_c5:
                best_h, best_c5, best_n = h_r, c5_r, len(h_r)

    elapsed = time.time() - t0
    remaining = budget - elapsed

    # Phase 4: Try upsampled n=100, 120
    if remaining > 60:
        for n_try in [100, 120]:
            if time.time() - t0 > budget - 30:
                break
            t_sub = min(remaining * 0.4, 180)
            h_up = upsample(best_h, n_try)
            h_up = project_bounds_sum(h_up, n_try / 2.0)
            h_up_r, c5_up = binary_ls(h_up, t_sub, rng, anneal=True)
            if c5_up < best_c5:
                best_h, best_c5, best_n = h_up_r, c5_up, n_try
            h_fr, c5_fr = fractional_refine(h_up_r, min(30, t_sub * 0.3))
            if c5_fr < best_c5:
                best_h, best_c5, best_n = h_fr, c5_fr, n_try
            remaining = budget - (time.time() - t0)

    # Phase 5: Final refinement on best
    elapsed = time.time() - t0
    if time.time() - t0 < budget - 5:
        t_final = min(60, budget - elapsed)
        h_fin, c5_fin = fractional_refine(best_h, t_final, lr_init=0.02)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    # Ensure constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_bounds_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
