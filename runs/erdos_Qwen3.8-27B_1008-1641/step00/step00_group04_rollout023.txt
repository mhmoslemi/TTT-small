def make_feasible(h):
    """Project h to satisfy constraints: 0<=h<=1, sum(h)*dx=1."""
    n = len(h)

    current_sum = np.sum(h)

        h = h + (target_sum - current_sum) / n

import numpy as np
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def thue_morse(n):
    tm = np.zeros(n, dtype=np.float64)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 else 0.0
    return tm

def fix_sum(h, n):
    """Adjust h so that sum(h) = n/2, maintaining 0<=h<=1."""
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the adjustment
    if diff > 0:
        # Add to elements with most room (closest to 1 but not at 1)
        room = 1.0 - h
        order = np.argsort(-room)
        for idx in order:
            add = min(diff, room[idx])
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    else:
        # Remove from elements with most value
        order = np.argsort(-h)
        for idx in order:
            rem = min(-diff, h[idx])
            h[idx] -= rem
            diff += rem
            if abs(diff) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def local_search_binary(h, n, max_rounds, rng, deadline):
    """Local search for binary h: flip individual bits and fix sum."""
    h = h.copy()

    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = fix_sum(h, n)
    best_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
