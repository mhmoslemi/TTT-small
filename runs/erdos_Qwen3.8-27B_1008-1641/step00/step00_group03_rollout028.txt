```python
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)

def smooth_max(c, t):
    m = np.max(c)
    cs = c - m
    return m + np.log(np.sum(np.exp(t * cs))) / t

def project_sum_bounds(h, target_sum):
    """Project h onto {sum = target_sum, 0 <= h <= 1}"""
    for _ in range(200):
        h = np.clip(h, 0, 1)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(mask) == 0:
            # Adjust all
            h = np.clip(h + diff / len(h), 0, 1)
            continue
        h[mask] += diff / np.sum(mask)
    return h

def slsqp_solve(h0, n, t=100, maxiter=1000):
    dx = 2.0 / n
    target_sum = n / 2.0
    
    def obj(x):
        x_c = np.clip(x, 0, 1)
        c = np.correlate(x_c, 1.0 - x_c, mode="full")
        return smooth_max(c, t)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n
    
    result = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
    h_new = project_sum_bounds(result.x, target_sum)
    return h_new

    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_sum_bounds(h, target_sum)
```

```python
    # Phase 4: Subgradient again (finer)
    if elapsed_frac() < 0.93:
        h_sg, c5_sg = subgradient_search(best_h, n, ts, 15000, 0.02, rng)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg

    # Phase 5: Final SLSQP
    if elapsed_frac() < 0.98:
        h_new = slsqp_opt(best_h, n, ts, 1000, 2000)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # Final projection and verification
    best_h = project(best_h, ts)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
```
