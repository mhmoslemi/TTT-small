```python
def c5_objective(h, dx):
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def c5_grad(h, dx, eps=1e-6):

    f0 = c5_objective(h, dx)
```

```python
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    g = np.zeros(len(h))

    # Valid i: 0 <= i < n and 0 <= i + k_star - n + 1 < n
    # i.e., max(0, n-1-k_star) <= i <= min(n-1, n-1) = n-1
    # and 0 <= k_star - n + 1 + i <= n - 1
    # i >= n - 1 - k_star and i <= n - 1 (since k_star - n + 1 + i <= n-1 → i <= 2n-2-k_star, but i < n anyway)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_ts = 0.0
_dxg = 0.0

def c5v(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def c5grad(h, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    ko = int(np.argmax(c))
    k = ko - n + 1
    g = np.zeros(n)
    lo = max(0, -k)
    hi = min(n - 1, n - 1 - k)
    for i in range(lo, hi + 1):
        g[i + k] += 1.0 - h[i]
        g[i] -= h[i + k]
    return g * dx

def _of(x):
    return c5v(x, _dxg)

def _cf(x):
    return float(np.sum(x) - _ts)

def _cj(x):
    return np.ones(len(x))

def pbase(h, s):
    h = np.asarray(h, dtype=float)
    lo, hi = -1.0, 1.0
    for _ in range(64):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(h + m, 0.0, 1.0)) > s:
            hi = m
        else:
            lo = m
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def pgo(h0, n, dx, s, nit, lr0, sd):
    h = h0.copy()
    mom = np.zeros(n)
    lr = lr0
    bc = c5v(h, dx)
    bh = h.copy()
    for it in range(nit):
        g = c5grad(h, dx)
        mom = 0.9 * mom - lr * g
        h = pbase(h + mom, s)
        c = c5v(h, dx)
        if c < bc:
            bc = c
            bh = h.copy()
        if c > bc * 1.1 + 1e-10:
            lr = max(lr * 0.5, 1e-7)
            mom *= 0.5
        elif it % 500 == 499:
            lr *= 0.95
    return bh, bc

def sls(h0, n, dx, s, mi):
    global _ts, _dxg
    _ts = s
    _dxg = dx
    try:
        r = minimize(_of, h0, method="SLSQP",
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{"type": "eq", "fun": _cf, "jac": _cj}],
                     options={"maxiter": mi, "ftol": 1e-12})
        if r.x is not None:
            return np.asarray(r.x, dtype=float), c5v(r.x, dx)
    except Exception:
        pass
    return None, None

def gst(n, rng):
    s = n / 2.0
    L = []
    try:
        iv = np.array(initial_h_values, dtype=float)
        if len(iv) == n:
            L.append(iv.copy())
    except Exception:
        pass
    for _ in range(12):
        v = (rng.random(n) < 0.5).astype(float)
        v += 0.1 * rng.standard_normal(n)
        L.append(pbase(v, s))
    for p in range(2, min(12, n)):
        f = max(1, p // 2)
        v = np.array([1.0 if i % p < f else 0.0 for i in range(n)])
        L.append(pbase(v, s))
    L.append(pbase(np.linspace(0.0, 1.0, n), s))
    L.append(pbase(1.0 - np.linspace(0.0, 1.0, n), s))
    L.append(pbase(np.full(n, 0.5), s))
    # Alternating blocks
    for bl in range(1, min(8, n // 2)):
        v = np.tile([1.0] * bl + [0.0] * bl, n // (2 * bl) + 1)[:n]
        L.append(pbase(v, s))
    return L

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    bh, bc, bn = None, 1e18, 0
    nl = [45, 44, 46, 48, 42, 40, 50]
    try:
        iv = np.array(initial_h_values, dtype=float)
        nl = list(dict.fromkeys([len(iv)] + nl))
    except Exception:
        pass
    for n in nl:
        if time.time() - t0 > budget_s - 30:
            break
        dx = 2.0 / n
        s = n / 2.0
        for h0 in gst(n, rng):
            if time.time() - t0 > budget_s - 15:
                break
            h0 = pbase(h0, s)
            # Projected gradient
            h1, c1 = pgo(h0, n, dx, s, 3000, 0.01, seed)
            # SLSQP
            h2, c2 = sls(h1, n, dx, s, 400)
            if h2 is not None and c2 < c1:
                h1, c1 = h2.copy(), c2
            # Snap
            h3 = h1.copy()
            h3[h3 < 0.2] = 0.0
            h3[h3 > 0.8] = 1.0
```

```python
        iv = np.array(initial_h_values, dtype=float)
        ln = len(iv)
        if ln in nl:
            nl.remove(ln)
        nl.insert(0, ln)
    except Exception:
        pass
    for n in nl:
        if time.time() > _deadline + 15:
            break
        dx = 2.0 / n
        s = n / 2.0
        starts = gst(n, rng)
        for h0 in starts:
            if time.time() > _deadline:
                break
            h0 = pbase(h0, s)
            h1, c1 = pgo(h0, n, dx, s, 2500)
            h2, c2 = sls(h1, n, dx, s, 500)
            if h2 is not None and c2 < c1:
                h1, c1 = h2, c2
            h3, c3 = snap_opt(h1, n, dx, s)
            if h3 is not None and c3 < c1:
                h1, c1 = h3, c3
            h2b, c2b = sls(h1, n, dx, s, 400)
            if h2b is not None and c2b < c1:
                h1, c1 = h2b, c2b
            if c1 < bc:
                bc = c1
                bh = h1.copy()
                bn = n
            for _ in range(3):
                if time.time() > _deadline:
                    break
                hp = pbase(h1 + 0.06 * rng.standard_normal(n), s)
                h5, c5 = sls(hp, n, dx, s, 250)
                if h5 is not None and c5 < bc:
                    bc = c5
                    bh = h5.copy()
                    bn = n
    if bh is None:
        bn = 45
        bh = np.full(bn, 0.5)
        bc = c5v(bh, 2.0 / bn)
    # final polish if time remains
    if time.time() < _deadline + 10:
        try:
            hf, cf = sls(bh, bn, 2.0 / bn, bn / 2.0, 500)
            if hf is not None and cf < bc:
                bh, bc = hf, cf
        except Exception:
            pass
    return bh, float(bc), int(bn)
```
