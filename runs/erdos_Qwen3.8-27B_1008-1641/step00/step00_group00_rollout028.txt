import numpy as np

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
        else:
            mask = h < 1.0 - 1e-15
        cnt = mask.sum()
        if cnt == 0:
            h[:] = target / n
            break
        h[mask] -= diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
