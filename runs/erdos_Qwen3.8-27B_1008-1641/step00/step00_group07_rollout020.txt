                best_c5 = c5_new
        except Exception:
            pass

    # Phase 3: Powell coordinate search from best
    if time.time() - t0 < budget_s - 150:
        try:
            res = minimize(obj_nm, best_h, method='Powell',
                         options={'maxiter': 300, 'xtol': 1e-14, 'ftol': 1e-15,
                                  'maxfev': 60000})
            h_new = project_feasible(res.x, n, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 4: SLSQP with hard objective from best
    if time.time() - t0 < budget_s - 80:
        try:
            res = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds,
                         constraints=[sum_constraint],
                         options={'maxiter': 400, 'ftol': 1e-14, 'eps': 1e-7})
            h_new = project_feasible(res.x, n, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 5: Final Nelder-Mead polish
    if time.time() - t0 < budget_s - 40:
        try:
            res = minimize(obj_nm, best_h, method='Nelder-Mead',
                         options={'maxiter': 50000, 'adaptive': True,
                                  'xatol': 1e-13, 'fatol': 1e-15,
                                  'maxfev': 120000})
            h_new = project_feasible(res.x, n, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 6: Round-up SLSQP if time remains
    if time.time() - t0 < budget_s - 20:
        try:
            res = minimize(soft_c5, best_h, method='SLSQP', bounds=bounds,
                         constraints=[sum_constraint],
                         options={'maxiter': 200, 'ftol': 1e-14, 'eps': 1e-8})
            h_new = project_feasible(res.x, n, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass

    # Final feasibility enforcement
    best_h = project_feasible(best_h, n, target_sum)
    best_c5 = compute_c5(best_h)

    # Verify with pre-imported evaluator if available
    try:
        if 'evaluate_erdos_solution' in globals() and evaluate_erdos_solution is not None:
            verified = evaluate_erdos_solution(best_h)
            if verified is not None:
                best_c5 = float(min(best_c5, verified))
    except Exception:
        pass

    return best_h, best_c5, n
