import numpy as np
from scipy.optimize import minimize
import time

_TARGET_SUM = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        sr = np.sum(room)
        if sr > 1e-12:
            h += room * (diff / sr)
    else:
        room = h.copy()
        sr = np.sum(room)
        if sr > 1e-12:
            h += room * (diff / sr)
    return np.clip(h, 0.0, 1.0)

def _sum_con(x):
    return np.sum(x) - _TARGET_SUM

def _sum_con_jac(x):
    return np.ones(len(x))

def _c5_obj(x):
    return compute_c5(np.clip(x, 0, 1))

def _slsqp_phase(h_best, target, rng, n_restarts, t0, deadline, maxiter):
    global _TARGET_SUM
    _TARGET_SUM = target
    n = len(h_best)
    bounds_list = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}]
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        if r == 0:
            x0 = best_h.copy()
        else:
            scale = max(0.04, 0.25 * (1.0 - r / float(max(n_restarts, 1))))
            x0 = best_h + rng.randn(n) * scale
        x0 = repair_sum(x0, target)
        try:
            res = minimize(_c5_obj, x0, method='SLSQP', bounds=bounds_list,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
            hn = repair_sum(np.clip(res.x, 0, 1), target)
            c5n = compute_c5(hn)
            if c5n < best_c5:
                best_c5, best_h = c5n, hn.copy()
        except Exception:
            pass
    return best_h, best_c5

def _coord_descent(h, target, rng, max_iter, deadline):
    n = len(h)
    deltas = [0.01, 0.02, 0.05, 0.1]
    best = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        for d in deltas:
            if h[i] + d <= 1.0 and h[j] >= d:
                si, sj = h[i], h[j]
                h[i], h[j] = si + d, sj - d
                c = compute_c5(h)
                if c < best:
                    best = c
                else:
                    h[i], h[j] = si, sj
                    break
            if h[j] + d <= 1.0 and h[i] >= d:
                si, sj = h[i], h[j]
                h[i], h[j] = si - d, sj + d
                c = compute_c5(h)
                if c < best:
                    best = c
                else:
                    h[i], h[j] = si, sj
                    break
    h = repair_sum(h, target)
    return h, best

def _binary_local_search(h, target, rng, max_iter, deadline):
    n = len(h)
    best = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        found_improve = False
        # Try to move mass from i to a random j (make h[i] smaller, h[j] larger)
        if h[i] > 0.02:
            j = int(rng.randint(n))
            if j != i:
                amt = min(h[i] * 0.5, 1.0 - h[j])
                if amt > 0.01:
                    si, sj = h[i], h[j]
                    h[i], h[j] = si - amt, sj + amt

                        found_improve = True
                    else:

                            found_improve = True
                        else:
