    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    """Systematic hill climbing: try all swaps, pick best improvement."""
    n = len(h)
    h = h.astype(float).copy()
    while (time.time() - t0) < budget_s * 0.85:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0
        best_i, best_j = None, None
        for i in ones:
            h[i] = 0.0

                delta = s - score

                    best_i, best_j = i, j

            h[best_i] = 0.0
            h[best_j] = 1.0
            score = score + best_delta
            improved = True
        if not improved:
            break
    return h, score
