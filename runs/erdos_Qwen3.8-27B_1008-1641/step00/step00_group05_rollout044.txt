import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 0:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def _resize_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

def _smooth_max(corr, t):
    m = np.max(corr)
    if t < 1e-8:
        return m, np.zeros_like(corr)
    shifted = t * (corr - m)
    exps = np.exp(shifted)
    sw = np.sum(exps)
    sm_val = m + np.log(sw) / t
    weights = exps / sw
    return sm_val, weights

def _obj_fn(h, t, pen):
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, _ = _smooth_max(corr, t)
    viol = np.sum(h) - len(h) / 2.0
    return val + pen * viol * viol

def _grad_fn(h, t, pen):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, weights = _smooth_max(corr, t)
    grad = np.zeros(n)
    nk1 = n - 1
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        lag = j - nk1
        for pos in range(n):
            v1 = pos - lag
            v2 = pos + lag
            term = 0.0
            if 0 <= v1 < n:
                term += 1.0 - h[v1]
            if 0 <= v2 < n:
                term -= h[v2]
            grad[pos] += w * term
    viol = np.sum(h) - n / 2.0
    grad[:] += 2.0 * pen * viol
    return grad

def _eq_constraint(h):
    return np.sum(h) - len(h) / 2.0

def _optimize_lbfgs(h0, t, pen, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_fn, h0, jac=_grad_fn, method='L-BFGS-B',
                   bounds=bounds, args=(t, pen),
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    h = _project_feasible(res.x, n)
    return h

def _optimize_slqp(h0, t, maxiter):

    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_fn, h0, jac=_grad_fn, method='SLSQP',
                   bounds=bounds, args=(t, 0.0),
                   constraints=[{'type': 'eq', 'fun': _eq_constraint}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    h = _project_feasible(res.x, n)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmax((h > 1e-14) & (h < 1.0 - 1e-14))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _resize_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

def _smooth_max_parts(corr, t):
    m = np.max(corr)
    if t < 1e-8:
        sm = m
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
        return sm, weights
    shifted = t * (corr - m)
    exps = np.exp(shifted)
    sw = np.sum(exps)
    sm = m + np.log(sw) / t
    weights = exps / sw
    return sm, weights

def _obj_fn(h, t, pen):
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, _ = _smooth_max_parts(corr, t)
    viol = np.sum(h) - len(h) / 2.0
    return val + pen * viol * viol

def _grad_fn(h, t, pen):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, weights = _smooth_max_parts(corr, t)
    grad = np.zeros(n)
    nk1 = n - 1
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        lag = j - nk1
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] += w * (-h[lag:])
        else:
            m_abs = -lag
            grad[:n - m_abs] += w * (1.0 - h[m_abs:])
            grad[m_abs:] += w * (-h[:n - m_abs])
    viol = np.sum(h) - n / 2.0
    grad[:] += 2.0 * pen * viol
    return grad

def _eq_con_fun(h):
    return np.sum(h) - len(h) / 2.0

def _opt_lbfgs(h0, t, pen, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_fn, h0, jac=_grad_fn, method='L-BFGS-B',
                   bounds=bounds, args=(t, pen),
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    return _project_feasible(res.x, n)

def _opt_slqp(h0, t, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_fn, h0, jac=_grad_fn, method='SLSQP',
                   bounds=bounds, args=(t, 0.0),
                   constraints=[{'type': 'eq', 'fun': _eq_con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return _project_feasible(res.x, n)
