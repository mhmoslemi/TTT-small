import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        interior = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        for idx in interior:
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            step = min(abs(diff), room)
            h[idx] += np.sign(diff) * step
            diff -= np.sign(diff) * step
            if abs(diff) < 1e-14:
                break
    return h

def upsample_arr(h_src, n_tgt):
    n_src = len(h_src)
    x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_tgt, endpoint=False)
    h = np.interp(xt, x, h_src)
    return make_feasible(h, n_tgt)

def coord_descent(h, max_sweeps=None, grid_size=30, t0=None, budget_left=120.0):
    n = len(h)
    dx = 2.0 / n
    c5 = compute_c5(h)
    full_offset = n - 1
    if max_sweeps is None:
        max_sweeps = 50

    for sweep in range(max_sweeps):
        if t0 is not None and (time.time() - t0) > budget_left:
            break
        improved = False
        for i in range(n):
            j_lo = max(i - full_offset, -i)
            j_hi = min(i, full_offset - i)
            if j_lo > j_hi:
                continue
            corr = np.correlate(h, 1.0 - h, mode="full")
            max_unaff = 0.0
            for m in range(len(corr)):
                j = m - full_offset
                if j_lo <= j <= j_hi:
                    continue
                if corr[m] > max_unaff:
                    max_unaff = corr[m]
            grid = np.linspace(0.0, 1.0, grid_size)
            cur = h[i]
            if 1e-8 < cur < 1.0 - 1e-8:
                grid = np.unique(np.concatenate([grid, [cur]]))
            best_t = cur
            best_obj = np.inf
            for t in grid:
                max_aff = 0.0
                for j in range(j_lo, j_hi + 1):
                    m = j + full_offset
                    if j == 0:
                        rest = corr[m] - cur * (1.0 - cur)
                        val = rest + t * (1.0 - t)
                    else:
                        a = 1.0 - h[i - j] - h[i + j]
                        rest = corr[m] - (h[i + j] + cur * a)
                        val = rest + t * a
                    if val > max_aff:
                        max_aff = val
                obj = max(max_unaff, max_aff)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            if abs(best_t - cur) > 1e-10:
                h[i] = best_t
                improved = True
        new_c5 = compute_c5(h)
        if new_c5 < c5 - 1e-11:
            c5 = new_c5
            h = make_feasible(h, n)
        elif not improved:
            break
    return h, c5

def targeted_repair(h, n, t0, budget_left):
    dx = 2.0 / n
    for _ in range(20):
        if time.time() - t0 > budget_left:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst_k = int(np.argmax(corr))
        j = worst_k - (n - 1)
        for i in range(n):
            if i - j < 0 or i - j >= n:
                continue
            partner = h[i - j]
            if partner > 0.3:
                h[i] = max(0.0, h[i] - 0.15)
            elif partner < 0.5:
                h[i] = min(1.0, h[i] + 0.08)
        h = make_feasible(h, n)
    return h

    if parent_c5 < 0.3809:

    # Coordinate descent at parent resolution
    if time.time() < deadline:

        h_try, c5_try = coord_descent(h_try, max_sweeps=40, t0=t0, budget_left=deadline - t0)

    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12), "Bound constraint violated"
