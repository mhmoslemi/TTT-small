def evaluate_c5(h):
    """Compute C5 for given h values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def subgradient_step(h, lr, lags):
    """One subgradient step for minimizing max_k f_k(h)."""
    n = len(h)
    # Compute all f_k values
    corr = np.correlate(h, 1-h, mode="full")
    max_val = np.max(corr)
    
    # Subgradient of max is the gradient of the active constraint
    # f_k(h) = sum h_i (1-h_{i+k}) for valid i
    # df_k/dh_j involves h_{j-k} and h_j
    
    # Actually, let's compute the gradient of the worst lag
    # f(lag) = sum_{valid i} h[i] * (1 - h[i+lag])
    # df/dh[j] = (1 - h[j+lag]) for i=j (if valid) 
    #           - h[j-lag] for i=j-lag (if valid)
    
    active_lags = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    grad = np.zeros(n)
    for lag_idx in active_lags:
        lag = lag_idx - (n - 1)
        # f = sum_{i: valid} h[i] * (1 - h[i+lag])
        # d/dh[j] of term where i=j: (1 - h[j+lag]) if j+lag valid
        # d/dh[j] of term where i+lag=j, i=j-lag: -h[j-lag] if j-lag valid
        if 0 <= lag < n:
            for i in range(n - lag):
                grad[i] += (1 - h[i + lag])
                grad[i + lag] -= h[i]
        elif -n < lag <= 0:
            for i in range(n + lag):
                grad[i] += (1 - h[i - lag])
                grad[i - lag] -= h[i]
    
    # Project: h + lr * grad, clip to [0,1], then fix sum
    h_new = np.clip(h + lr * grad, 0, 1)
    # Fix sum constraint
    target_sum = n / 2
    current_sum = np.sum(h_new)
    if np.abs(current_sum - target_sum) > 1e-10:
        h_new = h_new + (target_sum - current_sum) / n
        h_new = np.clip(h_new, 0, 1)
    
    return h_new

import numpy as np
import time

def compute_corr(h):
    """Compute the full correlation of h with 1-h."""
    return np.correlate(h, 1-h, mode="full")

def c5_objective(h):
    """Compute C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx

def c5_smooth(h, beta=50):
    """Smoothed C5 objective using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    # Softmax-like
    max_c = np.max(corr)
    log_sum = max_c + np.log(np.sum(np.exp(corr - max_c)))
    return (1.0/beta) * log_sum * dx

def project_to_constraint(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}."""
    h = np.clip(h, 0, 1)
    target = n / 2
    diff = target - np.sum(h)
    if np.abs(diff) > 1e-12:
        # Distribute evenly
        h = h + diff / n
        h = np.clip(h, 0, 1)
    return h

def optimize_continuous(h0, n, max_iter=5000, beta=50):
    """Optimize using projected subgradient on smoothed objective."""
    h = h0.copy().astype(float)
    best_c5 = c5_objective(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        corr = compute_corr(h)
        # Subgradient of sum h[i](1-h[i+k]) w.r.t. h[j]
        # For lag k: contributes (1-h[j+k]) to h[j] and -h[j-k] to h[j]
        
        # Use softmax weights for subgradient of max
        max_c = np.max(corr)
        weights = np.exp(cor * max_c) / np.sum(np.exp(corr - max_c))
        
        grad = np.zeros(n)
        for j in range(n):
            k = j - (n - 1)
            if 0 <= k < n:
                valid_i = range(n - k)
                for i in valid_i:
                    grad[i] += weights[j] * (1 - h[i+k])
                    grad[i+k] -= weights[j] * h[i]
            elif -n < k <= 0:
                valid_i = range(n + k)
                for i in valid_i:
                    grad[i] += weights[j] * (1 - h[i-k])
                    grad[i-k] -= weights[j] * h[i]
        
        # Learning rate
        lr = 1.0 / (it + 10)
        h_new = h - lr * grad * (1/beta)  # scaled by beta since we're using smoothed obj
        h_new = project_to_constraint(h_new, n)
        
        c5_val = c5_objective(h_new)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def compute_delta(h, i, j, n):
    """Compute delta in correlation when swapping h[i]:1→0 and h[j]:0→1."""
    lags = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    # Term 1: -(1-h[i+L]) for valid L
    valid1 = (lags >= -i) & (lags <= n-1-i)
    delta[valid1] -= (1 - h[i + lags[valid1]])
    
    # Term 2: +(1-h[j+L]) for valid L
    valid2 = (lags >= -j) & (lags <= n-1-j)
    delta[valid2] += (1 - h[j + lags[valid2]])
    
    # Term 3: +h[i-L] for valid L
    valid3 = (lags >= i-n+1) & (lags <= i)
    delta[valid3] += h[i - lags[valid3]]
    
    # Term 4: -h[j-L] for valid L
    valid4 = (lags >= j-n+1) & (lags <= j)
    delta[valid4] -= h[j - lags[valid4]]
    
    return delta

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def compute_delta(h, i, j, n):
    """Delta in full correlation after swapping h[i](1→0) and h[j](0→1)."""
    lags = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    v1 = (lags >= -i) & (lags <= n-1-i)
    delta[v1] -= (1.0 - h[i + lags[v1]])
    v2 = (lags >= -j) & (lags <= n-1-j)
    delta[v2] += (1.0 - h[j + lags[v2]])
    v3 = (lags >= i-n+1) & (lags <= i)
    delta[v3] += h[i - lags[v3]]
    v4 = (lags >= j-n+1) & (lags <= j)
    delta[v4] -= h[j - lags[v4]]
    return delta

def local_search_binary(h, time_limit=60.0):
    """Intensive local search via pair flips."""
    n = len(h)
    h = (h > 0.5).astype(np.float64)
    target = n // 2
    sc = int(np.sum(h))
    if sc > target:
        for idx in np.where(h==1)[0][:sc-target]:
            h[idx] = 0
    elif sc < target:
        for idx in np.where(h==0)[0][:target-sc]:
            h[idx] = 1
    
    corr = np.correlate(h, 1-h, mode="full")
    best_c5 = float(np.max(corr) * 2.0/n)
    start = time.time()
    
    while time.time() - start < time_limit:
        improved = False
        i_vals = list(np.where(h==1)[0])
        j_vals = list(np.where(h==0)[0])
        np.random.shuffle(i_vals)
        np.random.shuffle(j_vals)
        
        for i in i_vals:
            if improved:
                break
            for j in j_vals:
                delta = compute_delta(h, i, j, n)
                new_corr = corr + delta
                new_max = float(np.max(new_corr))
                if new_max < np.max(corr) - 1e-14:
                    h[i], h[j] = h[j], h[i]
                    corr = new_corr
                    best_c5 = new_max * 2.0/n

                    break

        if not improved:
            # Random perturbation

            for _ in range(200):

                j = int(np.random.choice(j_vals))

                    h[i], h[j] = h[j], h[i]
                    corr = new_corr
                    found = True
                    break

                break
