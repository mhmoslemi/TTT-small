        s = np.sum(np.clip(y - tau, 0.0, 1.0))
        if s > S:
            lo = tau
        else:
            hi = tau
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h):
    n = len(h)
    return float(np.max(corr_full(h)) * 2.0 / n)

def subgrad_c5(h, k=None):
    n = len(h)
    dx = 2.0 / n
    c = corr_full(h)
    if k is None:
        k = int(np.argmax(c))
    m = k - n + 1
    L = max(0, n - 1 - k)
    U = min(n - 1, 2 * n - k - 2)
    grad = np.zeros(n)
    for i in range(n):
        v = 0.0
        if L <= i <= U:
            im = i + m
            if 0 <= im < n:
                v += 1.0 - h[im]
        im2 = i - m
        if L <= im2 <= U and 0 <= im2 < n:
            v -= h[im2]
        grad[i] = dx * v
    return grad

def smooth_c5(h, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cx = c * 2.0 / n
    mx = np.max(cx)
    if beta <= 0:
        return mx, None
    shifted = cx - mx
    weights = np.exp(beta * shifted)
    wsum = np.sum(weights)
    return float(mx + np.log(wsum) / beta), weights / wsum

def smooth_grad_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = corr_full(h)
    cx = c * dx
    mx = np.max(cx)
    shifted = cx - mx
    weights = np.exp(beta * shifted)
    wsum = np.sum(weights)
    val = mx + np.log(wsum) / beta
    grad = np.zeros(n)
    for k_idx, w in enumerate(weights):
        if w < 1e-15:
            continue
        m = k_idx - n + 1
        L = max(0, n - 1 - k_idx)
        U = min(n - 1, 2 * n - k_idx - 2)
        for i in range(n):
            v = 0.0
            if L <= i <= U:
                im = i + m
                if 0 <= im < n:
                    v += 1.0 - h[im]
            im2 = i - m
            if L <= im2 <= U and 0 <= im2 < n:
                v -= h[im2]
            grad[i] += w * dx * v
    grad /= wsum
    return val, grad

def projected_subgrad(h, S, iters, beta_mom, step_init, t_end):
    n = len(h)
    h = h.copy()
    best = h.copy()
    best_v = c5_of(h)
    mom = np.zeros(n)
    step = step_init
    beta_so = 20.0
    for it in range(iters):
        if time.time() > t_end:
            break
        if it < iters // 3:
            beta_so = 20.0
        elif it < 2 * iters // 3:
            beta_so = 50.0
        else:
            beta_so = 200.0
        if beta_so < 100:
            _, g = smooth_grad_c5(h, beta_so)
        else:
            g = subgrad_c5(h)
        mom = beta_mom * mom + (1.0 - beta_mom) * g
        for _ in range(20):
            h_try = h - step * mom
            h_try = project_capped_simplex(h_try, S)
            v_try = c5_of(h_try)
            if v_try <= best_v + 1e-12:
                break
            step *= 0.65
        else:
            step = max(step, 1e-5)
            continue
        if v_try < best_v:
            h = h_try
            best = h.copy()
            best_v = v_try
        else:
            step *= 0.85
        step = min(step, 0.1)
    return best, best_v

def coordinate_descent(h, S, n_sweeps, t_end):
    n = len(h)
    h = h.copy()
    best = h.copy()
    best_v = c5_of(h)
    deltas_big = (-0.1, -0.05, -0.02, 0.02, 0.05, 0.1)
    deltas_small = (-0.02, -0.01, -0.005, 0.005, 0.01, 0.02)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            j = (i + 1) % n
            vi, vj = h[i], h[j]
            for delta in deltas_big:
                ni, nj = vi + delta, vj - delta
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                v = c5_of(h)
                if v < best_v:
                    best = h.copy()
                    best_v = v
                    improved = True
                else:
                    h[i], h[j] = vi, vj
            if not improved:
                for delta in deltas_small:
                    ni, nj = vi + delta, vj - delta
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    v = c5_of(h)
                    if v < best_v:
                        best = h.copy()
                        best_v = v
                        improved = True
                    else:
                        h[i], h[j] = vi, vj
        h = best.copy()
        if not improved:
            break
    return best, best_v

def random_perturb(h, S, rng, n_iters, t_end):
    n = len(h)
    best = h.copy()
    best_v = c5_of(h)
    temp = 0.003
    for it in range(n_iters):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (i + 1) % n
        delta = rng.uniform(-0.08, 0.08)
        ni, nj = best[i] + delta, best[j] - delta
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        trial = best.copy()
        trial[i], trial[j] = ni, nj
        v = c5_of(trial)
        if v < best_v or (v - best_v < temp and rng.rand() < 0.15):
            best = trial
            if v < best_v:
                best_v = v
        temp = max(temp * 0.999, 1e-5)
    return best, best_v

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 80
    S = n / 2.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n:
            old_n = len(h0)
            x_old = np.linspace(0.0, 2.0, old_n)
            x_new = np.linspace(0.0, 2.0, n)
            idx = np.clip(np.searchsorted(x_old, x_new, side='right') - 1, 0, old_n - 1)
            frac = (x_new - x_old[idx]) / np.diff(np.concatenate([x_old, [x_old[-1] + 0.01]]))
            h0 = h0[idx] * (1 - frac) + h0[np.clip(idx + 1, 0, old_n - 1)] * frac
            h0 = np.clip(h0, 0, 1)
    except Exception:
        h0 = 0.5 * np.ones(n)

    h0 = project_capped_simplex(h0, S)
    best_h = h0.copy()
    best_c5 = c5_of(h0)

    # Phase 1: Projected subgradient (35% budget)
    t1 = t0 + budget_s * 0.35
    best_h, best_c5 = projected_subgrad(h0, S, 5000, 0.75, 0.05, t1)

    # Phase 2: Coordinate descent (30% budget)
    t2 = t0 + budget_s * 0.65
    best_h, best_c5 = coordinate_descent(best_h, S, 800, t2)

    # Phase 3: Random perturbations (20% budget)
    t3 = t0 + budget_s * 0.85
    best_h, best_c5 = random_perturb(best_h, S, rng, 8000, t3)

    # Phase 4: Multi-restart from perturbed points (15% budget)
    t4 = t0 + budget_s * 0.97
    for restart in range(6):
        if time.time() > t4:
            break
        pert = best_h + rng.normal(0, 0.03, n)
        pert = project_capped_simplex(pert, S)
        v = c5_of(pert)
        if v < best_c5:
            best_h = pert.copy()
            best_c5 = v
        sub_t = min(t4, time.time() + (t4 - time.time()) / (7 - restart))
        cand, cv = projected_subgrad(pert, S, 2000, 0.7, 0.03, sub_t)
        if cv < best_c5:
            best_h = cand.copy()
            best_c5 = cv

    # Final polish
    best_h = project_capped_simplex(best_h, S)
    best_c5 = c5_of(best_h)
    # Verify feasibility
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(np.sum(best_h) - S) < 1e-3
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
