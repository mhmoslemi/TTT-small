import numpy as np
from scipy.optimize import minimize
import time


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def clip_and_fix(h, n):
    h = np.array(h, dtype=float, copy=True)
    target = n / 2.0
    for _ in range(20):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    return h


def eval_c5_constrained(h):
    h = clip_and_fix(h, len(h))
    return eval_c5(h)


def do_nelder(x0, maxiter=2000, fatol=1e-10):
    res = minimize(eval_c5_constrained, x0, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': fatol,
                            'adaptive': True})
    return res.x


def do_powell(x0, maxiter=500):
    res = minimize(eval_c5_constrained, x0, method='Powell',
                   options={'maxiter': maxiter, 'xtol': 1e-9, 'ftol': 1e-11})
    return res.x
