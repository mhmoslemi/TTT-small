        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 1.5: Binary push refinement
    el = time.time() - t_start
    if budget_s - el > 10:
        h_bp = binary_push(best_h, n / 2.0, n)
        c5_bp = c5_value(h_bp)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp.copy()
        # Quick CD after binary push
        h1b, c5_1b = coord_descend(best_h, min(10.0, budget_s - el - 5), n_grid=80)
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h1b.copy()

    # Phase 2: Perturbation + CD
    n_pert = max(5, n // 6)
    while (budget_s - (time.time() - t_start)) > 30:
        h_p = best_h.copy()
        idx = rng.choice(n, size=n_pert, replace=False)
        h_p[idx] += rng.randn(n_pert) * 0.12
        h_p = project_feasible(h_p, n / 2.0)
        cd_t = min(15.0, budget_s - (time.time() - t_start) - 35)
        if cd_t < 2:
            break
        h2, c5_2 = coord_descend(h_p, cd_t, n_grid=80)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Final CD with fine grid
    remaining = budget_s - (time.time() - t_start) - 3
    if remaining > 5:
        h3, c5_3 = coord_descend(best_h, remaining, n_grid=200)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, n / 2.0)
    best_c5 = c5_value(best_h)
    return best_h, best_c5, n
