import numpy as np
from scipy.optimize import minimize, nelder_mead
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_c5_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = 1e-8
    active = corr >= max_val - tol
    n_active = float(np.sum(active))
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)  # lags for full correlation
    for idx in np.where(active)[0]:
        lag = lags[idx]
        # For each j:
        # Term 1: (1-h[j-lag]) if 0<=j-lag<n
        # Term 2: -h[j+lag] if 0<=j+lag<n
        # Use vectorized form
        # Term 1 contribution to grad[j]: sum over active lags where j-lag in [0,n)
        # This means lag in [j-n+1, j], i.e., lag in [-(n-1), j] (for j>=0)
        # and lag in [j-n+1, (n-1)] (for j<n)
        
    # Vectorized approach:
    # For a given lag, the contribution to grad is:
    #   shift(1-h, -lag) - shift(h, lag) (with zero padding)
    
    for idx in np.where(active)[0]:
        lag = lags[idx]
        # Term 1: h[j] as first factor (h[i+lag]=h[j], i=j-lag)
        # grad gets (1-h[j-lag]) for j where 0<=j-lag<n
        if lag >= 0:
            # j-lag >= 0 => j >= lag; j-lag < n => j < n+lag (always since j<n)
            # So j ranges from lag to n-1
            grad[lag:] += (1.0 - h[:n-lag])
        else:
            # j-lag >= 0 => j >= lag (negative, so j>=0 always)
            # j-lag < n => j < n+lag
            # So j ranges from 0 to n+lag-1
            grad[:n+lag] += (1.0 - h[-lag:])
        
        # Term 2: h[j] as second factor (h[i]=h[j], i=j)
        # grad gets -h[j+lag] for j where 0<=j+lag<n
        if lag >= 0:
            # j+lag < n => j < n-lag
            # j >= 0 always
            grad[:n-lag] -= h[lag:]
        else:
            # j+lag >= 0 => j >= -lag
            # j+lag < n => j < n+lag (always since j<n and lag<0 means n+lag<n)
            # Wait: j < n and j >= -lag
            grad[-lag:] -= h[:n+lag]
    
    grad /= n_active
    return dx * grad, dx * max_val

import numpy as np
from scipy.optimize import minimize, nelder_mead
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_c5_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = 1e-7
    active = corr >= max_val - tol
    n_active = float(np.sum(active))
    grad = np.zeros(n)
    lags = np.arange(-(n - 1), n)
    for idx in np.where(active)[0]:
        lag = lags[idx]
        if lag >= 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        else:
            grad[:n + lag] += (1.0 - h[-lag:])
            grad[-lag:] -= h[:n + lag]
    grad /= n_active
    return dx * grad, dx * max_val

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = np.sum(h)
        diff = (target_sum - s) / len(h)
        if abs(diff) < 1e-14:
            break
        h += diff
        np.clip(h, 0.0, 1.0, out=h)
    return h

def interpolate_to_n(h, n_src, n_dst):
    if n_src == n_dst:
        return h.copy()
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    return np.interp(x_dst, x_src, h)
