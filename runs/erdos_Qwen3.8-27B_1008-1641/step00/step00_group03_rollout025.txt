import numpy as np
from scipy import optimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx), corr

def fix_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h + diff / n
    h = np.clip(h, 0, 1)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] = np.clip(h[mask] + diff2 / cnt, 0, 1)
    return h

def smooth_obj_grad(x, n, dx, beta):
    x = np.clip(x, 0, 1)
    corr = np.correlate(x, 1 - x, mode="full")
    o = corr * dx
    m = np.max(o)
    exps = np.exp(beta * (o - m))
    tot = np.sum(exps)
    w = exps / tot
    obj = m + np.log(tot) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            L = n - s
            grad[:L] += w[k] * dx * (1 - x[s:s + L])
            grad[s:s + L] += w[k] * (-dx) * x[:L]
        else:
            t = -s
            L = n - t
            grad[t:t + L] += w[k] * dx * (1 - x[:L])
            grad[:L] += w[k] * (-dx) * x[t:t + L]
    return obj, grad

def slsqp_optimize(h0, n, max_iter=100, beta=35):
    dx = 2.0 / n
    h0 = fix_sum(h0.copy(), n)

    def obj(x):
        return smooth_obj_grad(x, n, dx, beta)[0]

    def grad(x):
        return smooth_obj_grad(x, n, dx, beta)[1]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = optimize.minimize(obj, h0, method='SLSQP', jac=grad,
                                bounds=bounds, constraints=cons,
                                options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.x is not None:
            return fix_sum(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h0

    while time.time() - t0 < budget_s - 40 and attempts < 15:
        n_r = int(np.random.choice([50, 60, 70, 80, 90, 100]))
        h_r = np.random.random(n_r)
        thr = np.sort(h_r)[n_r // 2] + np.random.uniform(-0.05, 0.05)
        h_r = (h_r > thr).astype(float)
        if np.sum(h_r) != n_r / 2:
            idx = np.argsort(np.abs(h_r - 0.5))
            need = int(n_r / 2) - int(np.sum(h_r))
            for idx_i in idx[:abs(need)]:
                if need > 0:
                    h_r[idx_i] = 1.0
                else:
                    h_r[idx_i] = 0.0
        h_r = fix_sum(h_r, n_r)
        if time.time() - t0 < budget_s - 40:
            try_solution(h_r, n_r, alloc=50)
        attempts += 1

    # Phase 4: Periodic constructions
    for p in [7, 11, 13, 17, 19]:
        if time.time() - t0 > budget_s - 30:
            break
        n_p = p * max(1, best_n // p)
        if n_p < 30:
            continue
        block = np.zeros(p)
        block[:p // 2] = 1.0
        h_p = np.tile(block, n_p // p)
        h_p = fix_sum(h_p, n_p)
        try_solution(h_p, n_p, alloc=30)

    # Phase 5: Final polish
    if time.time() - t0 < budget_s - 15:
        h_f = slsqp_optimize(best_h, best_n, max_iter=50, beta=80)
        h_f, c5_f = local_search(h_f, best_n, max_iters=200,
                                 time_limit=budget_s - 5, t0=t0)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    best_h = fix_sum(best_h, best_n)
    best_c5, _ = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
