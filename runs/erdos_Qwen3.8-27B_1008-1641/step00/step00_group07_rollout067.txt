import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def compute_grad(corr, h, thresh=0.98):
    n = len(h)
    dx = 2.0 / n
    mx = np.max(corr)
    if mx < 1e-12:
        return np.zeros(n)
    active = np.where(corr >= mx * thresh)[0]
    grad = np.zeros(n)
    idx = np.arange(n)
    for m in active:
        k = m - (n - 1)
        w = corr[m] / mx
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            v1 = (idx - k >= 0) & (idx - k < n)
            grad[v1] += w * (1.0 - h[idx[v1] - k])
            v2 = (idx + k >= 0) & (idx + k < n)
            grad[v2] -= w * h[idx[v2] + k]
    return grad * dx

def project_h(h, target_sum):
    h = h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        hp = np.clip(h + mid, 0.0, 1.0)
        if hp.sum() < target_sum:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    d = target_sum - hp.sum()
    if abs(d) > 1e-12:
        free = (hp > 1e-4) & (hp < 1.0 - 1e-4)
        cnt = free.sum()
        if cnt > 0:
            hp[free] += d / cnt
            np.clip(hp, 0.0, 1.0, out=hp)
    return hp

def grad_phase(h, target_sum, t_end, lr_init=0.004):
    n = len(h)
    best = h.copy()
    best_c5 = eval_c5(best)[0]
    lr = lr_init
    vel = np.zeros(n)
    beta = 0.85
    stall = 0
    while time.time() < t_end:
        c5v, corr = eval_c5(best)
        if c5v < best_c5:
            best_c5 = c5v
        grad = compute_grad(corr, best, 0.98)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-8:
            break
        grad = grad / gnorm
        vel = beta * vel - lr * grad
        cand = best + vel
        cand = project_h(cand, target_sum)
        c5c = eval_c5(cand)[0]
        if c5c < best_c5:
            best_c5 = c5c
            best = cand
            stall = 0
            lr = min(lr * 1.015, 0.015)
        else:
            stall += 1
            lr *= 0.98
            if stall > 120:
                best = best + np.random.randn(n) * 0.04
                best = project_h(best, target_sum)
                best_c5 = eval_c5(best)[0]
                vel = np.zeros(n)
                lr = lr_init
                stall = 0
    return best, best_c5

def coord_phase(h, target_sum, t_end):
    n = len(h)
    best = h.copy()
    best_c5 = eval_c5(best)[0]
    while time.time() < t_end:
        improved = False
        order = np.random.permutation(n)
        for j in order:
            cur = best[j]
            candidates = [0.0, 1.0, max(0.0, cur - 0.15), min(1.0, cur + 0.15), 0.5]
            for v in candidates:
                if abs(v - cur) < 0.02:
                    continue
                trial = best.copy()
                trial[j] = v
                trial = project_h(trial, target_sum)
                c5t = eval_c5(trial)[0]
                if c5t < best_c5 - 1e-6:
                    best_c5 = c5t
                    best = trial
                    cur = v
                    improved = True
                    break
        if not improved and np.random.random() < 0.3:
            break
    return best, best_c5

def random_phase(h, target_sum, t_end, sigma=0.02):
    n = len(h)
    best = h.copy()
    best_c5 = eval_c5(best)[0]
    while time.time() < t_end:
        cand = best + np.random.randn(n) * sigma
        cand = project_h(cand, target_sum)
        c5c = eval_c5(cand)[0]
        if c5c < best_c5:
            best_c5 = c5c
            best = cand
    return best, best_c5
