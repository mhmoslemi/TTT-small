    n = len(h)
    target = n // 2
    # Sort by value, keep top 'target' as 1
    idx = np.argsort(h)[::-1]

    n = len(h)
    h = h.copy()
    start = time.time()
    current_score = eval_c5(h, dx)

    return h, score


def simulate_anneal(h, time_limit, seed):
    rng = np.random.default_rng(seed)
    n = len(h)
    h = h.copy()
    start = time.time()
    score = eval_c5(h)
    best_h = h.copy()
    best_s = score
    temp = 0.04
    cool = 0.9998
    while time.time() - start < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        ns = eval_c5(h)
        delta = ns - score
        if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            score = ns
            if ns < best_s:
                best_s = ns
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        temp *= cool
    return best_h, best_s


def iterated_search(h, time_limit, seed):
    rng = np.random.default_rng(seed)
    h = h.copy()
    start = time.time()
    best_h = h.copy()
    best_s = eval_c5(h)
    kick_size = 2
    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        if remaining < 2:
            break
        tc = min(remaining * 0.6, 8.0)
        h, s = hill_climb(h, tc)
        if s < best_s:
            best_s = s
            best_h = h.copy()
        nk = min(kick_size, 4)
        for _ in range(nk):
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            if i != j:
                h[i], h[j] = h[j], h[i]
        kick_size = min(kick_size + 1, 6)
        if time.time() - start > time_limit * 0.8:
            kick_size = 5
    return best_h, best_s


def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def even_spread(n):
    h = np.zeros(n)
    k = n // 2
    step = n / k
    for i in range(k):
        h[int(i * step) % n] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial solution
    try:
        best_h = np.array(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 42
        best_h = even_spread(n0)

    n = len(best_h)
    # Ensure binary
    if np.max(best_h) - np.min(best_h) > 0.1:
        best_h = make_binary(best_h)
    # Fix sum
    target_sum = n // 2
    cur_sum = int(np.sum(best_h > 0.5))
    if cur_sum != target_sum:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if cur_sum > target_sum:
            for k in range(cur_sum - target_sum):
                best_h[ones[k]] = 0.0
        else:
            for k in range(target_sum - cur_sum):
                best_h[zeros[k]] = 1.0

    best_score = eval_c5(best_h)
    best_n = n

    # Phase 1: Hill climb on initial
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        h1, s1 = hill_climb(best_h, min(remaining * 0.2, 120))
        if s1 < best_score:
            best_h, best_score, best_n = h1, s1, n

    # Phase 2: Iterated local search
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        h2, s2 = iterated_search(best_h, min(remaining * 0.35, 350), seed + 10)
        if s2 < best_score:
            best_h, best_score, best_n = h2, s2, n

    # Phase 3: SA restarts
    remaining = budget_s - (time.time() - t0)
    for attempt in range(5):
        if time.time() - t0 > budget_s - 60:
            break
        sa_time = min(remaining * 0.08, 80)
        if sa_time < 5:
            break
        h3, s3 = simulate_anneal(best_h, sa_time, seed + 100 + attempt * 7)
        if s3 < best_score:
            best_h, best_score, best_n = h3, s3, n
        remaining = budget_s - (time.time() - t0)

    # Phase 4: Try alternative n values
    for n_try in [40, 44, 46, 48, 50, 54, 56]:
        if time.time() - t0 > budget_s - 80:
            break
        if n_try == best_n:
            continue
        remaining = budget_s - (time.time() - t0)
        t_alloc = min(remaining * 0.12, 100)
        if t_alloc < 10:
            break

        # Multiple starts
        h_cand = even_spread(n_try)
        s_cand = eval_c5(h_cand)
        for _ in range(4):
            h_r = random_binary(n_try, rng)
            s_r = eval_c5(h_r)
            if s_r < s_cand:
                h_cand, s_cand = h_r, s_r

        # Hill climb
        hc_time = min(t_alloc * 0.5, 40)
        h_cand, s_cand = hill_climb(h_cand, hc_time)

        # Iterated search
        rem2 = t_alloc - (time.time() - t0 - (budget_s - remaining))
        it_time = min(remaining * 0.06, 50)
        if it_time > 3:
            h_it, s_it = iterated_search(h_cand, it_time, seed + n_try * 13)
            if s_it < s_cand:
                h_cand, s_cand = h_it, s_it

        if s_cand < best_score:
            best_h, best_score, best_n = h_cand.copy(), s_cand, n_try

    # Phase 5: Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        h_fin, s_fin = hill_climb(best_h, min(remaining * 0.5, 60))
        if s_fin < best_score:
            best_h, best_score = h_fin, s_fin

    # Ensure valid output
    final_n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Snap to binary if close
    best_h = (best_h > 0.5).astype(np.float64)
    # Fix sum
    target = final_n // 2
    cur = int(best_h.sum())
    if cur > target:
        idx = np.where(best_h == 1)[0]
        for k in range(cur - target):
            best_h[idx[k]] = 0.0
    elif cur < target:
        idx = np.where(best_h == 0)[0]
        for k in range(target - cur):
            best_h[idx[k]] = 1.0

    final_score = eval_c5(best_h)
    return best_h, final_score, final_n
