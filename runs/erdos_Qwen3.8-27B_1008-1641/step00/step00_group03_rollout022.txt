from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_onto_feasible(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}"""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    diff = target_sum - h.sum()
    if diff > 0:
        # Need to increase some values
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        increments = diff / mask.sum()
        h[mask] = np.minimum(h[mask] + increments, 1.0)
    elif diff < 0:
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        decrements = (-diff) / mask.sum()
        h[mask] = np.maximum(h[mask] - decrements, 0.0)
    return h

        alpha = 80.0
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -40, 40)
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    best_obj = objective(h0)

                         constraints=cons,

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} (approximately)."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 0.9999999
        else:
            m = h > 1e-7
        cnt = int(np.count_nonzero(m))
        if cnt == 0:
            break
        h[m] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h
