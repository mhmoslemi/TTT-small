```python
            h2[i] = min(h2[i] + eps, 1.0)
            grad[i] = (compute_c5(h2) - compute_c5(h)) / eps
        h -= lr * grad
        h = fix_sum(h, target)
    return h
```

Actually, for f(h) = max_k g_k(h), the subgradient at h is:
conv{∇g_k(h) : k in argmax}

For the one-sided difference (h + eps*e_i):
If the max is at k* and remains at k*: ∂f/∂h_i ≈ (g_{k*}(h+eps*e_i) - g_{k*}(h)) / eps = ∂g_{k*}/∂h_i
If the max switches: the difference gives a different (but valid) subgradient.

```python
import numpy as np
from scipy.optimize import minimize
import time

_TGT = 0.0
_PEN = 1e4
_CON_N = 0

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _f_pen(x):
    return compute_c5(x) + _PEN * (x.sum() - _TGT) ** 2

def _con_fun(x):
    return x.sum() - _TGT

def _con_jac(x):
    return np.ones(_CON_N)

def fix_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h < 1.0 - 1e-14) if d > 0 else (h > 1e-14)
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def make_golden(n, target):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    pos = np.array([(i * phi) % 1.0 for i in range(n)])
    order = np.argsort(pos)
    h = np.zeros(n)
    k = int(target)
    h[order[:k]] = 1.0
    if target - k > 0.01:
        h[order[k]] = target - k
    return fix_sum(h, target)

def make_alternating(n, target):
    h = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    return fix_sum(h, target)

def make_random(n, target, rng):
    return fix_sum(rng.uniform(0.0, 1.0, n), target)

def run(seed=42, budget_s=1100, **kwargs):
    global _TGT, _PEN, _CON_N
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 7)
    n = len(initial_h_values)
    _TGT = n / 2.0
    _CON_N = n
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}

    best_h = fix_sum(initial_h_values.astype(float), _TGT)
    best_c5 = compute_c5(best_h)
    t0 = time.time()

    starts = [best_h.copy(), make_golden(n, _TGT), make_alternating(n, _TGT)]
    for _ in range(4):
        starts.append(make_random(n, _TGT, rng))

    it = 0
    stall = 0
    while time.time() - t0 < budget_s - 5:
        it += 1

        if stall >= 10:
            si = it % len(starts)
            h0 = starts[si].copy()
            kp = max(5, n // 3)
            h0[rng.choice(n, kp, replace=False)] += rng.uniform(-0.25, 0.25, kp)
            h0 = fix_sum(h0, _TGT)
            stall = 0
        else:
            h0 = best_h.copy()
            frac = 0.05 + 0.15 * (stall / 10.0)
            kp = max(3, int(n * frac))
            h0[rng.choice(n, kp, replace=False)] += rng.uniform(-0.1, 0.1, kp)
            h0 = fix_sum(h0, _TGT)

        improved = False

        try:
            r = minimize(_f_pen, h0, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-8})
            rh = fix_sum(r.x, _TGT)
            v = compute_c5(rh)
            if v < best_c5 - 1e-14:
                best_h, best_c5, improved = rh, v, True
        except Exception:
            pass

        if time.time() - t0 < budget_s - 20:
            try:
                r2 = minimize(compute_c5, best_h.copy(), method='SLSQP',
                              bounds=bounds, constraints=[con],
                              options={'maxiter': 150, 'ftol': 1e-14})
                rh2 = fix_sum(r2.x, _TGT)
                v2 = compute_c5(rh2)
                if v2 < best_c5 - 1e-14:
                    best_h, best_c5, improved = rh2, v2, True
            except Exception:
                pass

        if not improved:
            stall += 1

    # Polish: a few tight SLSQP runs on the best
    for _ in range(3):
        if time.time() - t0 >= budget_s - 2:
            break
        try:
            rp = minimize(compute_c5, best_h.copy(), method='SLSQP',
                          bounds=bounds, constraints=[con],
                          options={'maxiter': 300, 'ftol': 1e-15})
            hp = fix_sum(rp.x, _TGT)
            vp = compute_c5(hp)
            if vp < best_c5:
                best_h, best_c5 = hp, vp
        except Exception:
            pass

    best_h = fix_sum(best_h, _TGT)
    return best_h, compute_c5(best_h), n
```
