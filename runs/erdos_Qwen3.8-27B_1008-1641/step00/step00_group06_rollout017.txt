            n_r += n_r % 2  # make even
            target_r = n_r / 2.0
            h_r = smooth_init(n_r, int(rng.integers(1, 5)))
            rem = min(10.0, budget_s * 0.96 - (time.time() - t0))
            if rem < 2:
                break
            h_r, c5_r = subgrad_optimize(h_r, target_r, time.time() - t0 + rem, t0)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_r

    if best_h is None:
        n_fb = 96
        best_h = smooth_init(n_fb, 1)
        best_c5 = compute_c5(best_h)
        best_n = n_fb

    return best_h, best_c5, best_n
