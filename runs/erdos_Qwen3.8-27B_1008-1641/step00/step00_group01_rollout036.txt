_DE_S = 0.0
def de_objective(x):
    return c5_value(project_box_sum(x, _DE_S))
def de_global(n, maxiter=50, popsize=10, seed=1):
    global _DE_S
    n=int(n)
    if n < 4 or maxiter <=0: return None
    try:
        from scipy.optimize import differential_evolution
    except Exception:
        return None
    _DE_S = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    try:
        res = differential_evolution(de_objective, bounds, maxiter=maxiter, popsize=popsize, tol=1e-5, seed=seed, polish=False, atol=1e-7)
    except Exception:
        return None
    if res is None or res.x is None:
        return None

import numpy as np
import math
import time

_DE_S = 0.0

def c5_value(h):
    n = len(h)
    if n <= 1:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, S):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x
    if not np.isfinite(S):
        S = 0.5 * n
    S = float(S)
    if S < 0.0:
        S = 0.0
    elif S > float(n):
        S = float(n)
    x = np.clip(np.where(np.isfinite(x), x, 0.5), 0.0, 1.0)
    lo = min(0.0, S - n)
    hi = max(0.0, S)
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        y = np.clip(x + mid, 0.0, 1.0)
        if float(np.sum(y)) < S:
            lo = mid
        else:
            hi = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - float(np.sum(x))
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - x)
        else:
            order = np.argsort(-x)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - x[idx])
            else:
                add = max(diff, -x[idx])
            x[int(idx)] += add
            diff -= add
            if abs(diff) <= 1e-14:
                break
    return np.clip(x, 0.0, 1.0)

def assess(h):
    arr = np.asarray(h, dtype=float).ravel()
    n = arr.size
    if n < 2:
        return None, np.inf, n
    S = 0.5 * n
    arr = project_box_sum(arr, S)
    c = c5_value(arr)
    if not np.isfinite(c):
        return None, np.inf, n
    return arr, c, n

def random_binary(n, rng):
    h = np.zeros(n)
    S = 0.5 * n
    k = int(math.floor(S))
    if k > 0:
        h[:k] = 1.0
    rem = S - k
    if rem > 1e-12 and k < n:
        h[k] = rem
    rng.shuffle(h)
    return h

def smooth_candidate(n, phase, amp=0.5):
    x = np.linspace(0.0, 1.0, n)
    return 0.5 + amp * np.cos(2.0 * np.pi * (x + phase))

def resample(h0, n0, n):
    if n0 == n:
        return np.array(h0, dtype=float)
    if n0 < 2:
        return np.full(n, 0.5)
    xs = np.linspace(0.0, 1.0, n0)
    xn = np.linspace(0.0, 1.0, n)
    return np.interp(xn, xs, h0)

def pair_eval(t, h, i, j, n):
    hi = h[i]
    hj = h[j]
    h[i] = min(1.0, max(0.0, hi + t))
    h[j] = min(1.0, max(0.0, hj - t))
    v = c5_value(h)
    h[i] = hi
    h[j] = hj
    return v

def golden_min_1d(f, a, b, args, tol=1e-9, maxit=32):
    if b <= a:
        return float(a), float(f(a, *args))
    phi = 0.5 * (math.sqrt(5.0) - 1.0)
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = f(c, *args)
    fd = f(d, *args)
    for _ in range(maxit):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = f(c, *args)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = f(d, *args)
    t = 0.5 * (a + b)
    ft = f(t, *args)
    best, bestf = t, ft
    if fc < bestf:
        best, bestf = c, fc
    if fd < bestf:
        best, bestf = d, fd
    return float(best), float(bestf)

def attempt_pair(h, i, j, n, val, maxit, tol):
    a = h[i]
    b = h[j]
    lo = max(-a, b - 1.0)
    hi = min(1.0 - a, b)
    if hi - lo <= 1e-13:
        return val, False
    t, f = golden_min_1d(pair_eval, lo, hi, (h, i, j, n), tol, maxit)
    if f + 1e-12 < val:
        h[i] = min(1.0, max(0.0, a + t))
        h[j] = min(1.0, max(0.0, b - t))
        return f, True
    return val, False

def refine_pairs(h, rng, deadline, maxit=32, tol=1e-9):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    S = 0.5 * n
    val = c5_value(h)
    if deadline <= time.time() or n < 2:
        return h, val
    if n <= 80:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        while time.time() < deadline - 0.5:
            order = pairs.copy()
            rng.shuffle(order)
            improved = False
            for i, j in order:
                if time.time() > deadline - 0.2:
                    break
                val, ok = attempt_pair(h, i, j, n, val, maxit, tol)
                improved = improved or ok
            if not improved:
                break
    else:
        sweeps = 0
        sample = max(500, 4 * n)
        while time.time() < deadline - 0.5:
            improved = False
            for _ in range(sample):
                if time.time() > deadline - 0.2:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                if i > j:
                    i, j = j, i
                val, ok = attempt_pair(h, i, j, n, val, maxit, tol)
                improved = improved or ok
            sweeps += 1
            if sweeps > 12 and not improved:
                break
    h = project_box_sum(h, S)
    val = c5_value(h)
    return h, val

def de_objective(x):
    return c5_value(project_box_sum(x, _DE_S))

def de_global(n, maxiter=50, popsize=10, seed=1):
    global _DE_S
    n = int(n)
    if n < 4 or n > 80 or maxiter <= 0:
        return None
    try:
        from scipy.optimize import differential_evolution
    except Exception:
        return None
    _DE_S = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    try:
        res = differential_evolution(
            de_objective, bounds, maxiter=maxiter, popsize=popsize,
            tol=1e-5, seed=seed, polish=False, atol=1e-7, workers=1
        )
    except Exception:
        return None
    if res is None or res.x is None:
        return None
    return project_box_sum(res.x, _DE_S)
