def repair(x):
    x = np.array(x, dtype=float, copy=True)
    n = x.size
    if n == 0: return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(20):
        err = target - np.sum(x)
        if abs(err) < 1e-12:
            break
        if err > 0:
            room = 1.0 - x
            tot = room.sum()
            if tot <= 0:
                break
            add = room * (err / tot)
            s = add.sum()
            if s > err:
                add *= err / s
            x += add
        else:
            room = x
            tot = room.sum()
            if tot <= 0:
                break
            sub = room * (-err / tot)
            s = sub.sum()
            if s > -err:
                sub *= -err / s
            x -= sub
        np.clip(x, 0.0, 1.0, out=x)
    for idx in np.argsort(-x * (1.0 - x)):
        val = target - (np.sum(x) - x[idx])
        if -1e-12 <= val <= 1.0 + 1e-12:
            x[idx] = float(np.clip(val, 0.0, 1.0))
            break
    return x

    if n==0: return None,best_c

import time
import math
import numpy as np
from scipy.optimize import minimize

G_BETA = 80.0


def repair(x):
    x = np.array(x, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(24):
        err = target - float(np.sum(x))
        if abs(err) < 1e-12:
            break
        if err > 0.0:
            room = 1.0 - x
            tot = float(room.sum())
            if tot <= 0.0:
                break
            add = room * (err / tot)
            s = float(add.sum())
            if s > err:
                add *= err / s
            x += add
        else:
            room = x
            tot = float(room.sum())
            if tot <= 0.0:
                break
            sub = room * (-err / tot)
            s = float(sub.sum())
            if s > -err:
                sub *= -err / s
            x -= sub
        np.clip(x, 0.0, 1.0, out=x)
    order = np.argsort(-x * (1.0 - x))
    for idx in order:
        val = target - (float(np.sum(x)) - float(x[idx]))
        if -1e-12 <= val <= 1.0 + 1e-12:
            x[idx] = float(np.clip(val, 0.0, 1.0))
            break
    return x


def quantize_feasible(x):
    x = np.array(x, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    D = 1 << 30
    raw = np.clip(x, 0.0, 1.0) * float(D)
    cnt = np.rint(raw).astype(np.int64)
    cnt = np.clip(cnt, 0, D)
    target_count = int(np.rint(target * float(D)))
    diff = target_count - int(cnt.sum())
    if diff > 0:
        order = np.argsort(-(raw - cnt))
        for idx in order:
            if diff <= 0:
                break
            if cnt[idx] < D:
                add = int(min(diff, D - cnt[idx]))
                cnt[idx] += add
                diff -= add
    elif diff < 0:
        dd = -diff
        order = np.argsort(raw - cnt)
        for idx in order:
            if dd <= 0:
                break
            if cnt[idx] > 0:
                sub = int(min(dd, cnt[idx]))
                cnt[idx] -= sub
                dd -= sub
    if diff > 0:
        for idx in range(n):
            if diff <= 0:
                break
            if cnt[idx] < D:
                add = int(min(diff, D - cnt[idx]))
                cnt[idx] += add
                diff -= add
    elif diff < 0:
        dd = -diff
        for idx in range(n):
            if dd <= 0:
                break
            if cnt[idx] > 0:
                sub = int(min(dd, cnt[idx]))
                cnt[idx] -= sub
                dd -= sub

def exact_c5(x):
    n = x.size
    if n == 0:
        return 0.0
    c = np.correlate(x, 1.0 - x, mode="full")
    return float(np.max(c) * (2.0 / float(n)))

    cand = repair(cand)
    val = exact_c5(cand)
    if val < best_c - 1e-15:
        return cand, val
    return best_h, best_c


def lag_gradient(x, s):
    n = x.size
    g = np.zeros(n, dtype=float)
    if s == 0:
        return 1.0 - 2.0 * x
    if s > 0:
        m = n - s
        g[:m] -= x[s:s + m]
        g[s:] += 1.0 - x[:m]
    else:
        d = -s
        m = n - d
        g[:m] += 1.0 - x[d:d + m]
        g[d:] -= x[:m]
    return g


def smooth_obj(x):
    if x.size < 2:
        return 0.0
    C = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(C))
    b = float(G_BETA)
    if b <= 1.0 or m == 0.0:
        return m
    z = b * (C - m)
    return float(m + math.log(float(np.exp(z).sum())) / b)


def smooth_jac(x):
    n = x.size
    if n < 2:
        return np.zeros(n, dtype=float)
    C = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(C))
    b = float(G_BETA)
    if m == 0.0:
        return np.zeros(n, dtype=float)
    if b <= 1.0:
        idx = int(np.argmax(C))
        return lag_gradient(x, idx - (n - 1))
    z = b * (C - m)
    e = np.exp(z)
    w = e / float(e.sum())
    g = np.zeros(n, dtype=float)
    active = np.flatnonzero(w > 1e-14)
    off = n - 1
    for idx in active:
        wt = float(w[idx])
        s = int(idx) - off
        if s == 0:
            g += wt * (1.0 - 2.0 * x)
        elif s > 0:
            m = n - s
            g[:m] -= wt * x[s:s + m]
            g[s:] += wt * (1.0 - x[:m])
        else:
            d = -s
            m = n - d
            g[:m] += wt * (1.0 - x[d:d + m])
            g[d:] -= wt * x[:m]
    return g


def sum_fun(x):
    return float(np.sum(x) - 0.5 * x.size)


def sum_jac(x):
    return np.ones_like(x, dtype=float)


def optimize_smooth(x0, beta, maxiter):
    global G_BETA
    x0 = repair(x0)
    G_BETA = float(beta)
    bounds = [(0.0, 1.0)] * x0.size
    cons = [{"type": "eq", "fun": sum_fun, "jac": sum_jac}]
    try:
        res = minimize(
            smooth_obj,
            x0,
            jac=smooth_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False},
        )
        if getattr(res, "x", None) is not None and res.x.size == x0.size:
            return repair(res.x)
    except Exception:
        pass
    return x0


def try_line_search_direction(x, g, best_c, steps):
    n = x.size
    if n == 0:
        return None, best_c
    g = np.array(g, dtype=float, copy=True)
    g -= float(np.mean(g))
    nrm = float(np.linalg.norm(g))
    if nrm < 1e-12:
        return None, best_c
    g /= nrm
    for st in steps:
        cand = repair(x - float(st) * g)
        val = exact_c5(cand)
        if val < best_c - 1e-14:
            return cand, val
    return None, best_c


def exact_local_search(x, limit_s, rng):
    global G_BETA
    x = repair(np.array(x, dtype=float, copy=True))
    n = x.size
    if n < 2 or limit_s <= 0.0:
        return x, exact_c5(x)
    G_BETA = 80.0
    best = exact_c5(x)
    maxC = best / (2.0 / float(n))
    C = np.correlate(x, 1.0 - x, mode="full")
    t_end = time.time() + float(limit_s)
    steps = (0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005)
    noise = 0.035
    stagnant = 0
    while time.time() < t_end and stagnant < 120:
        cand, val = try_line_search_direction(x, smooth_jac(x), best, steps)
        if cand is not None:
            x = cand
            best = val
            maxC = best / (2.0 / float(n))
            C = np.correlate(x, 1.0 - x, mode="full")
            stagnant = 0
            continue
        tol = 1e-7 * max(1.0, maxC)
        active = np.flatnonzero(C >= maxC - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(C))])
        k = min(12, int(active.size))
        top = active[np.argsort(-C[active])[:k]]
        rng.shuffle(top)
        improved = False
        for idx in top:
            s = int(idx) - (n - 1)
            cand, val = try_line_search_direction(x, lag_gradient(x, s), best, steps)
            if cand is not None:
                x = cand
                best = val
                maxC = best / (2.0 / float(n))
                C = np.correlate(x, 1.0 - x, mode="full")
                improved = True
                break
        if improved:
            stagnant = 0
            continue
        for _ in range(25):
            if time.time() >= t_end:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            c2 = float(x[i] + x[j])
            lo = max(0.0, c2 - 1.0)
            hi = min(1.0, c2)
            if hi <= lo + 1e-12:
                continue
            tv = float(x[i] + rng.normal(0.0, noise))
            tv = min(hi, max(lo, tv))
            if abs(tv - float(x[i])) < 1e-8:
                continue
            cand = x.copy()
            cand[i] = tv
            cand[j] = c2 - tv
            val = exact_c5(cand)
            if val < best - 1e-14:
                cand = repair(cand)
                val = exact_c5(cand)
                if val < best - 1e-14:
                    x = cand
                    best = val
                    maxC = best / (2.0 / float(n))
                    C = np.correlate(x, 1.0 - x, mode="full")
                    improved = True
                    break
        if improved:
            stagnant = 0
            noise = max(0.001, noise * 0.85)
        else:
            stagnant += 1
            noise = max(0.0005, noise * 0.75)
    return x, best


def resample_h(x, n):
    x = np.asarray(x, dtype=float)
    n = int(n)
    m = x.size
    if n <= 0 or m == 0:
        return repair(np.array([], dtype=float))
    if m == n:
        return repair(x.copy())
    if m == 1:
        return repair(np.full(n, float(x[0])))
    xo = (np.arange(m) + 0.5) * (2.0 / float(m))
    xn = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(xn, xo, x)
    return repair(h)


def wave_function(n, amp, ph):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = 0.5 + float(amp) * np.cos(math.pi * x + float(ph))
    return repair(h)


def make_binary_balanced(x):
    x = np.asarray(x, dtype=float)
    n = x.size
    b = np.zeros(n, dtype=float)
    if n == 0:
        return b
    order = np.argsort(-x)
    if n % 2 == 0:
        target = n // 2
        if target > 0:
            b[order[:target]] = 1.0
    else:
        t = n // 2
        if t > 0:
            b[order[:t]] = 1.0
        if t < n:
            b[order[t]] = 0.5
    return b


def random_binary(n, rng):
    n = int(n)
    b = np.zeros(n, dtype=float)
    if n == 0:
        return b
    if n % 2 == 0:
        k = n // 2
        if k > 0:
            b[rng.choice(n, k, replace=False)] = 1.0
    else:
        k = n // 2
        if k > 0:
            b[rng.choice(n, k, replace=False)] = 1.0
        zeros = np.flatnonzero(b == 0.0)
        if zeros.size > 0:
            b[int(rng.integers(zeros.size))] = 0.5
    return b


def thue_morse_balanced(n, rng):
    n = int(n)
    if n == 0:
        return np.array([], dtype=float)
    t = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        t[i] = int(t[i >> 1]) ^ int(i & 1)
    b = t.astype(float)
    if n % 2 == 0:
        target = n // 2
        cnt = int(b.sum())
        if cnt > target:
            ones = np.flatnonzero(b == 1.0)
            need = min(cnt - target, int(ones.size))
            if need > 0:
                b[rng.choice(ones, need, replace=False)] = 0.0
        elif cnt < target:
            zeros = np.flatnonzero(b == 0.0)
            need = min(target - cnt, int(zeros.size))
            if need > 0:
                b[rng.choice(zeros, need, replace=False)] = 1.0
    else:
        target = n // 2
        cnt = int(b.sum())
        if cnt > target:
            ones = np.flatnonzero(b == 1.0)
            need = min(cnt - target, int(ones.size))
            if need > 0:
                b[rng.choice(ones, need, replace=False)] = 0.0
        elif cnt < target:
            zeros = np.flatnonzero(b == 0.0)
            need = min(target - cnt, int(zeros.size))
            if need > 0:
                b[rng.choice(zeros, need, replace=False)] = 1.0
        zeros = np.flatnonzero(b == 0.0)
        if zeros.size > 0:
            b[int(rng.integers(zeros.size))] = 0.5
    return b


def maxc_corr_int(b):
    return int(np.max(np.correlate(b, 1 - b, mode="full")))


def binary_sa(start_b, limit_s, rng):
    n = int(np.asarray(start_b).size)
    b = np.zeros(n, dtype=np.int16)
    start_b = np.asarray(start_b)
    b[start_b > 0.5] = 1
    if n % 2 == 0 and n > 0:
        target = n // 2
        cnt = int(b.sum())
        if cnt > target:
            ones = np.flatnonzero(b == 1)
            for idx in ones[cnt - target:]:
                b[idx] = 0
        elif cnt < target:
            zeros = np.flatnonzero(b == 0)
            for idx in zeros[:target - cnt]:
                b[idx] = 1
    if n == 0:
        return b, 0
    curC = maxc_corr_int(b)
    bestC = curC
    best = b.copy()
    t_end = time.time() + float(limit_s)
    temp = 5.0
    since_best = 0
    kick = 0
    while time.time() < t_end:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        bi = int(b[i])
        bj = int(b[j])
        if bi == bj:
            continue
        b[i] = 1 - bi
        b[j] = 1 - bj
        newC = maxc_corr_int(b)
        accept = newC <= curC
        if not accept:
            delta = newC - curC
            if delta <= int(temp) + 2:
                if rng.random() < math.exp(-float(delta) / max(temp, 1e-3)):
                    accept = True
        if accept:
            curC = newC
            if curC < bestC:
                bestC = curC
                best = b.copy()
                since_best = 0
            else:
                since_best += 1
        else:
            b[i] = bi
            b[j] = bj
            since_best += 1
        if since_best > max(2000, int(500 * (temp + 1.0))):
            if kick % 4 == 3:
                b = best.copy()
            else:
                b = best.copy()
                ksteps = 2 + int(temp)
                for _ in range(ksteps):
                    ii = int(rng.integers(n))
                    jj = int(rng.integers(n))
                    if ii != jj and b[ii] != b[jj]:
                        b[ii] = 1 - b[ii]
                        b[jj] = 1 - b[jj]
            curC = maxc_corr_int(b)
            since_best = 0
            kick += 1
        temp = max(0.4, temp * 0.9995)
        if curC > bestC + 6:
            b = best.copy()
            curC = bestC
            since_best = 0
    return best, bestC


def binary_greedy_polish(start_b, limit_s, rng):
    n = int(np.asarray(start_b).size)
    b = np.asarray(start_b, dtype=np.int16).copy()
    if n == 0:
        return b, 0
    curC = maxc_corr_int(b)
    t_end = time.time() + float(limit_s)
    improved = True
    passes = 0
    while improved and passes < 24 and time.time() < t_end:
        improved = False
        passes += 1
        ones = np.flatnonzero(b == 1)
        zeros = np.flatnonzero(b == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        o_order = rng.permutation(ones.size)
        z_order = rng.permutation(zeros.size)
        for ii in o_order:
            if time.time() > t_end:
                break
            i = int(ones[ii])
            for jj in z_order:
                j = int(zeros[jj])
                b[i] = 0
                b[j] = 1
                newC = maxc_corr_int(b)
                if newC < curC:
                    curC = newC
                    improved = True
                    break
                b[i] = 1
                b[j] = 0
            if improved:
                break
    return b, curC


def run(seed=42, budget_s=1100, **kwargs):
    global G_BETA
    t0 = time.time()
    total = float(budget_s)
    rng = np.random.default_rng(seed)
    G_BETA = 80.0

    best_h = None
    init = globals().get("initial_h_values", None)
    if init is not None:
        p = np.asarray(init, dtype=float).ravel()
        if p.size > 0 and np.all(np.isfinite(p)):
            best_h = repair(p)
        elif p.size > 0:
            best_h = repair(np.full(p.size, 0.5))
    if best_h is None:
        best_h = repair(np.full(80, 0.5))
    best_c = exact_c5(best_h)
    best_n = int(best_h.size)

    variants = [best_h.copy()]
    if best_n >= 8:
        variants.append(repair(1.0 - best_h))
        variants.append(repair(best_h[::-1]))
        variants.append(repair(best_h + 0.02 * rng.normal(size=best_n)))
        variants.append(make_binary_balanced(best_h))
        variants.append(wave_function(best_n, 0.5, 0.0))
        variants.append(wave_function(best_n, 0.5, math.pi / 2.0))

    for v in variants:
        if time.time() - t0 > total - 60.0:
            break
        best_h, best_c = consider_solution(v, best_h, best_c)
        best_n = int(best_h.size)
        h = v
        for beta, maxit in ((20.0, 70), (60.0, 70), (120.0, 50)):
            rem = total - (time.time() - t0)
            if rem < 25.0:
                break
            maxit_eff = min(maxit, max(10, int(rem / 5.0)))
            h = optimize_smooth(h, beta, maxit_eff)
            best_h, best_c = consider_solution(h, best_h, best_c)
            best_n = int(best_h.size)

    rem = total - (time.time() - t0)
    if rem > 80.0:
        pol_s = min(240.0, max(20.0, rem * 0.20))
        h2, c2 = exact_local_search(best_h, pol_s, rng)
        best_h, best_c = consider_solution(h2, best_h, best_c)
        best_n = int(best_h.size)

    bin_ns = []
    for m in (best_n, best_n + 20, best_n + 40, 80, 100, 120, 160, 200):
        m = int(m)
        if 20 <= m <= 240 and m % 2 == 0 and m not in bin_ns:
            bin_ns.append(m)

    for n in bin_ns:
        rem = total - (time.time() - t0)
        if rem < 90.0:
            break
        per_sa = min(40.0, max(10.0, rem / 15.0))
        per_gr = min(8.0, per_sa / 4.0)
        starts = [
            make_binary_balanced(resample_h(best_h, n)),
            thue_morse_balanced(n, rng),
            random_binary(n, rng),
            make_binary_balanced(wave_function(n, 0.5, 0.0)),
            make_binary_balanced(wave_function(n, 0.5, math.pi / 2.0)),
        ]
        for s in starts:
            rem = total - (time.time() - t0)
            if rem < 20.0:
                break
            s = repair(s)
            best_h, best_c = consider_solution(s, best_h, best_c)
            best_n = int(best_h.size)
            if rem > per_sa + per_gr + 10.0:
                bb, bc = binary_sa(s, per_sa, rng)
                val = float(bc) * (2.0 / float(n))
                if val < best_c - 1e-15:
                    best_h = bb.astype(float)
                    best_c = val
                    best_n = n
                rem2 = total - (time.time() - t0)
                if rem2 > per_gr + 10.0:
                    bp, cp = binary_greedy_polish(bb, per_gr, rng)
                    val = float(cp) * (2.0 / float(n))
                    if val < best_c - 1e-15:
                        best_h = bp.astype(float)
                        best_c = val
                        best_n = n

    cont_ns = []
    for m in (best_n, best_n + 20, best_n + 40, 120, 160):
        m = int(m)
        if 20 <= m <= 200 and m % 2 == 0 and m not in cont_ns:
            cont_ns.append(m)

    for n in cont_ns:
        rem = total - (time.time() - t0)
        if rem < 160.0:
            break
        h = resample_h(best_h, n)
        best_h, best_c = consider_solution(h, best_h, best_c)
        best_n = int(best_h.size)
        for beta, maxit in ((30.0, 60), (80.0, 50)):
            rem = total - (time.time() - t0)
            if rem < 60.0:
                break
            h = optimize_smooth(h, beta, max(10, min(maxit, int(rem / 6.0))))
            best_h, best_c = consider_solution(h, best_h, best_c)
            best_n = int(best_h.size)
        rem = total - (time.time() - t0)
        if rem > 120.0:
            pol_s = min(80.0, rem * 0.25)
            h2, c2 = exact_local_search(h, pol_s, rng)
            best_h, best_c = consider_solution(h2, best_h, best_c)
            best_n = int(best_h.size)

    rem = total - (time.time() - t0)
    if rem > 40.0:
        pol_s = max(5.0, min(rem - 20.0, 180.0))
        h2, c2 = exact_local_search(best_h, pol_s, rng)
        best_h, best_c = consider_solution(h2, best_h, best_c)
        best_n = int(best_h.size)

    best_h = quantize_feasible(best_h)
    best_c = exact_c5(best_h)
    return best_h, float(best_c), int(best_h.size)
