    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            free = h < 1.0
            if not free.any(): break
            step = diff / free.sum()
            step = min(step, (1.0 - h[free]).min())
            h[free] += step
        else:
            free = h > 0.0
            if not free.any(): break
            step = -diff / free.sum()

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(corr) * 2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            return h
        if diff > 0:
            free = h < 1.0
            if not free.any():
                return h
            step = min(diff / free.sum(), (1.0 - h[free]).min())
            h[free] += step
        else:
            free = h > 0.0
            if not free.any():
                return h
            step = min(-diff / free.sum(), h[free].min())
            h[free] -= step

    s = i - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        m = -s
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    return g

def smooth_grad(h, temperature):
    n = len(h)
    corr = compute_corrs(h)
    # Soft-max weights
    m = corr.max()
    exps = np.exp((corr - m) / temperature)
    w = exps / exps.sum()
    # Accumulate gradient
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] > 1e-10:
            g += w[k] * grad_lag(h, k)
    return g, corr.max() * (2.0/n), np.argmax(corr)

import numpy as np
import time

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    corr = compute_corrs(h)
    return float(np.max(corr) * 2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, i):
    n = len(h)
    s = i - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    else:
        m = -s
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    return g

def weighted_grad(h, temperature):
    n = len(h)
    corr = compute_corrs(h)
    cmax = corr.max()
    exps = np.exp((corr - cmax) / temperature)
    w = exps / exps.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            g += w[k] * grad_lag(h, k)
    return g, float(cmax * (2.0 / n))
