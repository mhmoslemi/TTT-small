                    if compute_c5(h_try) < compute_c5(h_cd) - 1e-13:
                        h_cd = h_try
                        improved = True
                        break
            if improved:
                c_new = compute_c5(h_cd)
                if c_new < best_c5:
                    best_h = h_cd.copy()
                    best_c5 = c_new

    # Phase 3: Fine SLSQP polish
    if time.time() - t0 < t_limit * 0.8:
        try:
            res = minimize(
                compute_c5, best_h,
                jac=lambda x: numerical_gradient(x),
                method='SLSQP', bounds=bounds, constraints=[con_slsqp],
                options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10, 'disp': False}
            )
            if res.x is not None:
                candidate = fix_sum(np.clip(res.x, 0.0, 1.0), target_sum)
                if compute_c5(candidate) < best_c5:
                    best_h = candidate
                    best_c5 = compute_c5(best_h)
        except Exception:
            pass

    # Phase 4: Try binary/ternary relaxation
    if time.time() - t0 < t_limit * 0.7:
        # Try thresholding at various levels
        for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
            h_bin = (best_h > thresh).astype(float)
            h_bin = fix_sum(h_bin, target_sum)
            try:
                res = minimize(
                    compute_c5, h_bin,
                    jac=lambda x: numerical_gradient(x),
                    method='SLSQP', bounds=bounds, constraints=[con_slsqp],
                    options={'maxiter': 200, 'ftol': 1e-13, 'disp': False}
                )
                if res.x is not None:
                    candidate = fix_sum(np.clip(res.x, 0.0, 1.0), target_sum)
                    if compute_c5(candidate) < best_c5:
                        best_h = candidate
                        best_c5 = compute_c5(best_h)
            except Exception:
                pass

    # Final: ensure feasibility
    best_h = fix_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
