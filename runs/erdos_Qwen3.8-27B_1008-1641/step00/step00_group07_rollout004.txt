import time

def project_feasible(h, n, target_sum):
    """Project h to {0≤h≤1, sum(h)=target_sum}"""
    h = np.clip(h, 0, 1)
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        # Adjust uniformly or proportionally
        h += diff / n
        h = np.clip(h, 0, 1)
    # Final adjustment
    s = np.sum(h)
    diff = target_sum - s
    if diff > 0:
        # Need to increase: add to elements not at 1
        mask = h < 1 - 1e-10
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    elif diff < 0:
        # Need to decrease: remove from elements not at 0
        mask = h > 1e-10
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        mask_hi = h > 1e-12
        if diff > 0:
            mask_lo = h < 1.0 - 1e-12
            cnt = np.sum(mask_lo)
            if cnt > 0:
                h[mask_lo] += diff / cnt
            else:
                break
        else:
            mask_hi = h > 1e-12
            cnt = np.sum(mask_hi)
            if cnt > 0:
                h[mask_hi] += diff / cnt
            else:
                break
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target_sum - s
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-6)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
    elif diff < 0:
        idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def c5_and_corr(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx), c
