        hi = min(n - 1, m)
        info.append((m, lo, hi))

    scaled = corr / tau
    M = float(np.max(scaled))
    exp_s = np.exp(scaled - M)
    sw = float(np.sum(exp_s))
    weights = exp_s / sw

def projected_gradient_descent(h, n, dx, target_sum, lag_info, tau_seq, max_iter, time_limit, t0):
    h = h.copy()
    best_h = h.copy()
    best_f = compute_c5(h, dx)
    alpha = 1.0
    
    for tau in tau_seq:

                break
