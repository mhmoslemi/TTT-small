            h[i] -= sub
            diff += sub
            if diff >= -1e-14:
                break
    return h

def upsample(h_old, n_old, n_new):
    if n_new == n_old:
        return h_old.copy()
    idx = (np.arange(n_new) * n_old) // n_new
    idx = np.clip(idx, 0, n_old - 1)
    h_new = h_old[idx].copy()
    return project_to_box_sum(h_new, n_new / 2.0)

def optimize_slsqp(h_init, n, dx, max_iter=300):
    target_sum = n / 2.0
    corr_len = 2 * n - 1
    c5_init = compute_C5(h_init, dx)

    def obj(v):
        return v[n]

    cons = [{'type': 'eq', 'fun': lambda v: np.sum(v[:n]) - target_sum}]
    for k in range(corr_len):
        def mkcon(kk):
            def con(v):
                hv = v[:n]
                c = np.correlate(hv, 1.0 - hv, mode='full')
                return v[n] - c[kk] * dx
            return con
        cons.append({'type': 'ineq', 'fun': mkcon})

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    v0 = np.concatenate([h_init, [c5_init]])

    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    h_opt = project_to_box_sum(res.x[:n], target_sum)
    return h_opt, compute_C5(h_opt, dx)

def optimize_lbfgsb(h_init, n, dx, max_iter=500, beta=150.0):
    target_sum = n / 2.0
    pen_coeff = 1000.0

    def obj(v):
        h = np.clip(v, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(corr)
        smax = cmax + np.log(np.sum(np.exp(beta * (corr - cmax)))) / beta
        sum_dev = np.sum(h) - target_sum
        return smax + pen_coeff * sum_dev * sum_dev

    res = minimize(obj, h_init, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50})
    h_opt = project_to_box_sum(res.x, target_sum)
    return h_opt, compute_C5(h_opt, dx)

def coordinate_descent(h, n, dx, max_passes=30):
    target_sum = n / 2.0
    best_c5 = compute_C5(h, dx)
    for _ in range(max_passes):
        improved = False
        for i in range(n):
            for new_val in (0.0, 0.5, 1.0):
                if abs(new_val - h[i]) < 1e-8:
                    continue
                h_trial = h.copy()
                h_trial[i] = new_val
                h_trial = project_to_box_sum(h_trial, target_sum)
                c5_trial = compute_C5(h_trial, dx)
                if c5_trial < best_c5 - 1e-12:
                    h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h_curr = np.array(initial_h_values, dtype=np.float64).copy()
    h_curr = project_to_box_sum(h_curr, n0 / 2.0)
    best_h = h_curr.copy()
    best_c5 = compute_C5(best_h, 2.0 / n0)
    best_n = n0

    def elapsed():
        return time.time() - start_time

    # Phase 1: SLSQP at n=45
    if elapsed() < budget_s * 0.15:
        h_opt, c5_opt = optimize_slsqp(h_curr, n0, 2.0 / n0, max_iter=400)
        h_opt, c5_opt = coordinate_descent(h_opt, n0, 2.0 / n0, max_passes=50)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n0
        h_curr = best_h.copy()

    # Phase 2: Multi-resolution optimization
    for mult in [2, 4, 8]:
        n = n0 * mult
        if n > 720:
            break
        if elapsed() > budget_s * 0.55:
            break
        dx = 2.0 / n
        h_up = upsample(best_h, best_n, n)

        if n <= 180 and elapsed() < budget_s * 0.45:
            h_opt, c5_opt = optimize_slsqp(h_up, n, dx, max_iter=200)
        else:
            h_opt, c5_opt = optimize_lbfgsb(h_up, n, dx, max_iter=400)
        h_opt, c5_opt = coordinate_descent(h_opt, n, dx,
                                           max_passes=20 if n > 90 else 50)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n

    # Phase 3: Random restarts at best resolution
    for attempt in range(30):
        if elapsed() > budget_s * 0.92:
            break
        noise_scale = 0.12 / (1.0 + attempt * 0.3)
        h_pert = best_h + rng.randn(best_n) * noise_scale
        h_pert = project_to_box_sum(h_pert, best_n / 2.0)
        dx_cur = 2.0 / best_n
        if best_n <= 90:
            h_opt, c5_opt = optimize_slsqp(h_pert, best_n, dx_cur, max_iter=100)
        else:
            h_opt, c5_opt = optimize_lbfgsb(h_pert, best_n, dx_cur, max_iter=200)
        h_opt, c5_opt = coordinate_descent(h_opt, best_n, dx_cur, max_passes=20)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Binary rounding + coordinate descent
    h_bin = (best_h > 0.5).astype(np.float64)
    h_bin = project_to_box_sum(h_bin, best_n / 2.0)
    h_bin, c5_bin = coordinate_descent(h_bin, best_n, 2.0 / best_n, max_passes=80)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin

    # Phase 5: Final SLSQP polish if time remains
    if elapsed() < budget_s * 0.95 and best_n <= 90:
        dx_cur = 2.0 / best_n
        h_opt, c5_opt = optimize_slsqp(best_h, best_n, dx_cur, max_iter=300)
        h_opt, c5_opt = coordinate_descent(h_opt, best_n, dx_cur, max_passes=30)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    final_c5 = compute_C5(best_h, 2.0 / best_n)
    return best_h, final_c5, best_n
