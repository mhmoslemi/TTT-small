    n = len(h)

                    if val < best_val - 1e-12:
                        best_val = val
                        improved = True
                    else:
                        h = project_sum(h, S)
                        # revert
                        # recompute from best
                        break
                if improved:
                    break
    return h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n_init = 95
        h_init = np.zeros(n_init)
        h_init[::2] = 1.0
        h_init = project_sum(h_init, n_init / 2.0)

    n_init = len(h_init)
    h_init = np.clip(h_init, 0.0, 1.0)
    h_init = project_sum(h_init, n_init / 2.0)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient at original n (35%)
    t1 = budget_s * 0.35
    h1, c5_1 = optimize_subgrad(best_h, max_iter=50000, t0=6.0,
                                 decay=0.9995, time_limit=t1, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Bit-flip local search (15%)
    t2 = budget_s * 0.15
    h2, c5_2 = local_search_bitflip(best_h, time_limit=t2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2

    # Phase 3: Multi-n exploration (30%)
    n_current = len(best_h)
    n_candidates = [50, 75, 95, 100, 125, 150]
    t3_total = budget_s * 0.30
    for n_new in n_candidates:
        if time.time() - t_start > t_start + budget_s * 0.85:
            break
        if n_new == n_current:
            continue
        t3 = t3_total / max(len(n_candidates) - 1, 1)
        h_trial = interp_h(best_h, n_new)
        h3, c5_3 = optimize_subgrad(h_trial, max_iter=15000, t0=4.0,
                                     decay=0.999, time_limit=t3, seed=seed + n_new)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    # Phase 4: Final refinement (15%)
    elapsed = time.time() - t_start
    remaining = max(5.0, budget_s * 0.95 - elapsed)
    h4, c5_4 = optimize_subgrad(best_h, max_iter=80000, t0=2.0,
                                 decay=0.9998, time_limit=remaining, seed=seed + 777)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4

    # Final constraint enforcement
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_final)
