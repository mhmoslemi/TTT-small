def subgradient(h, k):

    s = k - (n - 1)
    g = np.zeros(n)
    for p in range(n):

        if 0 <= p - s <= n - 1:
            val += 1.0 - h[p - s]
        if 0 <= p + s <= n - 1:

        g[p] = val
    return g

def project_sum(h, target, lower=0.0, upper=1.0):
    # Project h onto {lower ≤ h ≤ upper, sum(h) = target}
    n = len(h)
    lo = -1e6
    hi = 1e6
    for _ in range(60):

        h_proj = np.clip(h + mid, lower, upper)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2, lower, upper)
    # Fine-tune
    diff = target - np.sum(h_proj)
    idx = np.argsort(-h_proj if diff > 0 else h_proj)[:max(1, abs(int(diff)))]
    for i in idx:
        if diff > 0:
            adj = min(diff, upper - h_proj[i])
            h_proj[i] += adj
            diff -= adj
        elif diff < 0:
            adj = min(-diff, h_proj[i] - lower)
            h_proj[i] -= adj
            diff += adj
    return h_proj

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def subgradient_corr(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    for p in range(n):
        val = 0.0
        if 0 <= p - s <= n - 1:
            val += 1.0 - h[p - s]
        if 0 <= p + s <= n - 1:
            val -= h[p + s]
        g[p] = val
    return g


def project_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        hp = np.clip(h + mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        order = np.argsort(-hp) if diff > 0 else np.argsort(hp)
        for i in order:
            if diff > 1e-12:
                a = min(diff, 1.0 - hp[i])
                hp[i] += a
                diff -= a
            elif diff < -1e-12:
                a = min(-diff, hp[i])
                hp[i] -= a
                diff += a
            else:
                break
    return hp


def fix_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    n = len(h)
    perm = np.random.permutation(n)

            h[i] += a
            diff -= a
        elif diff < -1e-12:

            h[i] -= a
            diff += a
        else:

    return h

        idx = np.where(b == 0)[0]
        np.random.shuffle(idx)
        for i in idx[:int(target - ones)]:
            b[i] = 1.0
    return b
