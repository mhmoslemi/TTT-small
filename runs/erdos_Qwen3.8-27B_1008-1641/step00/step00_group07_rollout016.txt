def project_onto_constraint(h, n_pts):
    """Project h onto {x: 0<=x_i<=1, sum(x)=n_pts/2}"""
    target = n_pts / 2.0
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2.0, 0, 1)
    # Fine-tune to get exact sum
    diff = target - np.sum(proj)
    # Distribute among non-saturated coordinates
    unsat = (proj > 1e-10) & (proj < 1 - 1e-10)

from scipy.optimize import minimize
import time

def project_feasible(x, n_pts):
    target = n_pts / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(x - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        unsat = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if np.sum(unsat) > 0:
            proj[unsat] += diff / np.sum(unsat)
            proj = np.clip(proj, 0.0, 1.0)

    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def compute_subgrad(h):
    n = len(h)

            for i in range(-k, n):
                grad[i + k] += (1.0 - h[i])
                grad[i] -= h[i + k]

    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode="full")  # 2n-1 values

    # Softmax weights
    c_max = np.max(c_scaled)
    exps = np.exp(beta * (c_scaled - c_max))
    weights = exps / np.sum(exps)
    # Objective value (smoothed)
    obj = c_max + np.log(np.sum(exps)) / beta
    # Gradient: weighted average of subgradients
    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, n_pts):
    target = n_pts / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(x - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-14:
        unsat = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if np.any(unsat):
            proj[unsat] += diff / np.sum(unsat)
            proj = np.clip(proj, 0.0, 1.0)
    diff2 = target - np.sum(proj)
    if abs(diff2) > 1e-15:
        idx = np.argmin(np.abs(proj - 0.5))
        proj[idx] = np.clip(proj[idx] + diff2, 0.0, 1.0)
    return proj

def compute_c(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_obj(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_c(h))) * dx

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c_full = compute_c(h)
    c_scaled = c_full * dx
    c_max = np.max(c_scaled)
    exps = np.exp(beta * (c_scaled - c_max))
    total = np.sum(exps)
    obj = c_max + np.log(total) / beta
    weights = exps / total
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            kp = -k
            grad[:n - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n - kp]
    grad *= dx
    return obj, grad

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(80, 0.5)

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    best_val = compute_obj(h_best)

    beta = 40.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    def obj_func(x):
        return smooth_obj_grad(x, beta)[0]

    def grad_func(x):
        return smooth_obj_grad(x, beta)[1]

    # Phase 1: SLSQP with smoothed objective, multiple restarts
    max_phase1 = budget_s * 0.55
    restarts_done = 0
    stagnation = 0

    for restart in range(200):
        if time.time() - t0 > max_phase1:
            break
        if restart == 0:
            x0 = h_best.copy()
        else:
            sigma = 0.03 + 0.25 * min(restart, 8) / 8.0
            x0 = project_feasible(h_best + np.random.randn(n) * sigma, n)

        try:
            res = minimize(obj_func, x0, jac=grad_func, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-12})
            restarts_done += 1
            x_new = project_feasible(res.x, n)
            v = compute_obj(x_new)
            if v < best_val - 1e-12:
                h_best = x_new
                best_val = v
                stagnation = 0
                beta = min(beta * 1.05, 150.0)
            else:
                stagnation += 1
        except Exception:
            pass

    # Phase 2: SLSQP with raw (unsmoothed) objective for final polish
    max_phase2 = budget_s * 0.80
    for restart in range(30):
        if time.time() - t0 > max_phase2:
            break
        if restart == 0:
            x0 = h_best.copy()
        else:
            sigma = 0.02 * (restart + 1) / 30.0
            x0 = project_feasible(h_best + np.random.randn(n) * sigma, n)
        try:
            res = minimize(compute_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-12})
            x_new = project_feasible(res.x, n)
            v = compute_obj(x_new)
            if v < best_val - 1e-14:
                h_best = x_new
                best_val = v
        except Exception:
            pass

    # Phase 3: Coordinate descent with fine steps
    max_phase3 = budget_s - 10
    h_curr = h_best.copy()
    cur_val = compute_obj(h_curr)

    for cd_round in range(200):
        if time.time() - t0 > max_phase3:
            break
        improved = False
        deltas = np.linspace(-0.02, 0.02, 17)
        for i in range(n):
            if time.time() - t0 > max_phase3:
                break
            best_d = 0.0
            best_v = cur_val
            for d in deltas:
                trial = h_curr.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                v = compute_obj(trial)
                if v < best_v - 1e-14:
                    best_v = v
                    best_d = d
            if best_d != 0.0:
                h_curr[i] = np.clip(h_curr[i] + best_d, 0.0, 1.0)
                h_curr = project_feasible(h_curr, n)
                cur_val = best_v
                improved = True
        if not improved:
            break
        if cur_val < best_val:
            h_best = h_curr.copy()
            best_val = cur_val

    # Phase 4: Very fine local search (Nelder-Mead on a subset)
    max_phase4 = budget_s - 5

            res = minimize(compute_obj, nm_x0, method='Nelder-Mead',

            x_new = project_feasible(res.x, n)
            v = compute_obj(x_new)
            if v < best_val:
                h_best = x_new
                best_val = v
        except Exception:
            pass

    h_best = project_feasible(h_best, n)
    best_val = compute_obj(h_best)
    return h_best, best_val, n
