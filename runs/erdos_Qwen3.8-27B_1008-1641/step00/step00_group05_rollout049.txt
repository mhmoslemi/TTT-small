import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute C5 = max_k dx * corr[k]"""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project(h, target_sum):
    """Project h onto {g : 0 ≤ g ≤ 1, sum(g) = target_sum}"""
    # Bisection on threshold
    lo, hi = -1, 2
    for _ in range(100):
        mid = (lo + hi) / 2
        g = np.clip(h - mid, 0, 1)
        if g.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    g = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine-tune to get exact sum
    diff = target_sum - g.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        idx = np.argsort(g)
        if diff > 0:
            for i in range(idx[-1], idx[0]-1, -1):
                rem = min(diff, 1-g[i])
                g[i] += rem
                diff -= rem
                if abs(diff) < 1e-10: break
        else:
            for i in range(idx[0], idx[-1]+1):
                rem = min(-diff, g[i])
                g[i] -= rem
                diff += rem
                if abs(diff) < 1e-10: break
    return g

def gradient_c5(h, dx):
    """Compute subgradient of max_k corr[k]"""
    N = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    k_star = np.argmax(corr)  # index in the full correlation
    # k_star ranges from 0 to 2N-2, corresponding to shift k = k_star - (N-1)
    shift = k_star - (N - 1)
    
    grad = np.zeros(N)
    if shift >= 0:
        # corr = sum_{n=0}^{N-1-shift} h[shift+n]*(1-h[n])
        for n in range(N - shift):
            j = shift + n  # index of h in first factor
            grad[j] += (1 - h[n])
            grad[n] -= h[j]
    else:
        p = -shift
        # corr = sum_{n=p}^{N-1} h[n-p]*(1-h[n])
        for n in range(p, N):
            j = n - p  # index of h in first factor
            grad[j] += (1 - h[n])
            grad[n] -= h[j]
    
    return grad * dx

def _obj(h):
    N = _global_N
    dx = _global_dx
    beta = _global_beta
    corr = np.correlate(h, 1-h, mode="full")
    vals = beta * dx * corr
    m = np.max(vals)
    logsumexp = m + np.log(np.exp(vals - m).sum())

def _grad(h):
    N = _global_N
    dx = _global_dx
    beta = _global_beta
    corr = np.correlate(h, 1-h, mode="full")
    vals = beta * dx * corr
    m = np.max(vals)
    w = np.exp(vals - m)
    w /= w.sum()
    
    grad = np.zeros(N)
    for k in range(2*N-1):
        shift = k - (N-1)
        if shift >= 0:
            for n in range(N - shift):
                j = shift + n
                grad[j] += w[k] * (1.0 - h[n])
                grad[n] -= w[k] * h[j]
        else:
            p = -shift
            for n in range(p, N):
                j = n - p
                grad[j] += w[k] * (1.0 - h[n])
                grad[n] -= w[k] * h[j]
    return grad * dx

    return h.sum() - _global_target_sum

def polish(h, dx, target_sum, max_iter=50):

    best = h.copy()
    best_c5 = compute_c5(best, dx)
    for _ in range(max_iter):
        improved = False
        for i in range(N):
            for delta in [0.01, -0.01, 0.005, -0.005]:

                    j = (j + 1) % N

                        improved = True
