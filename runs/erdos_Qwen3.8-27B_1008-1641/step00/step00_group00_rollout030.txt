import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(v, target_sum):
    n = len(v)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    r = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(r)
    if abs(diff) > 1e-12:
        mask = (r > 1e-12) & (r < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            r[mask] += diff / cnt
    return np.clip(r, 0.0, 1.0)

def c5_eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = int(np.argmax(corr))
    s = k - n + 1
    g = np.zeros(n)
    if s >= 0:
        g[s:] += dx * (1.0 - h[:n - s])
        g[:n - s] -= dx * h[s:]
    else:
        ss = -s
        g[:n - ss] += dx * (1.0 - h[ss:])
        g[ss:] -= dx * h[:n - ss]
    return g

def subgrad_descent(h, n, tlim, seed):

    h = project_feasible(h.copy(), ts)
    best_h, best_v = h.copy(), c5_eval(h)
    t0 = time.time()
    rng = np.random.default_rng(seed)
    lr, stall, max_stall = 0.8, 0, 400
    while time.time() - t0 < tlim:
        g = subgrad(h)
        step, ok = lr, False
        for _ in range(25):
            h2 = project_feasible(h - step * g, ts)
            v2 = c5_eval(h2)
            if v2 < best_v - 1e-11:

                ok = True

def smooth_obj(x, n, ts, temp):
    x2 = project_feasible(x, ts)
    corr = np.correlate(x2, 1.0 - x2, mode='full') * (2.0 / n)
    cmax = np.max(corr)
    exps = np.exp((corr - cmax) / temp)
    z = np.sum(exps)
    val = cmax + temp * np.log(z)
    w = exps / z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-8:
            continue
        s = k - n + 1
        if s >= 0:
            grad[s:] += w[k] * (2.0 / n) * (1.0 - x2[:n - s])
            grad[:n - s] -= w[k] * (2.0 / n) * x2[s:]
        else:
            ss2 = -s
            grad[:n - ss2] += w[k] * (2.0 / n) * (1.0 - x2[ss2:])
            grad[ss2:] -= w[k] * (2.0 / n) * x2[:n - ss2]
    return val, grad

    elapsed = time.time() - t_start
    remaining = budget_s * 0.85 - elapsed
    if remaining > 60:
        tlim_each = max(10, remaining / len(ntries))
        for nt in ntries:
            elapsed = time.time() - t_start
            if elapsed > budget_s * 0.85:
                break
            if time.time() - t_start + 10 > budget_s:
                break
            hi = interp_to_n(best_h, nt)
            hi = project_feasible(hi, nt / 2.0)
            h4, v4 = subgrad_descent(hi, nt, tlim_each, seed + 200 + nt)
            if v4 < best_v:
                best_h, best_v, best_n = h4, v4, nt

    # Phase 5: final polish
    elapsed = time.time() - t_start
    tlim = max(5, (budget_s - elapsed) * 0.8)
    if tlim > 5:
        h5, v5 = hill_climb(best_h, best_n, tlim, seed + 999)
        if v5 < best_v:
            best_h, best_v = h5, v5

    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_v = c5_eval(best_h)
    # Sanity check
    if final_v > best_v + 1e-8:
        final_v = best_v
    return best_h, final_v, len(best_h)
