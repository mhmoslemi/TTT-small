import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum_box(h, S):
    h = np.array(h, dtype=np.float64)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = S - proj.sum()
    if abs(diff) > 1e-10:
        free = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        if free.any():
            proj[free] += diff / free.sum()
    return np.clip(proj, 0.0, 1.0)

def smooth_obj(h_flat, dx, beta):
    n = len(h_flat)
    corr = np.correlate(h_flat, 1 - h_flat, mode="full")
    x = corr * dx
    m = x.max()
    e = np.exp(beta * (x - m))
    return m + np.log(e.sum()) / beta

def smooth_obj_grad(h_flat, dx, beta):
    n = len(h_flat)
    corr = np.correlate(h_flat, 1 - h_flat, mode="full")
    x = corr * dx
    m = x.max()
    exps = np.exp(beta * (x - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        t = k - (n - 1)
        for j in range(n):
            if 0 <= j + t < n:
                grad[j] += w[k] * dx * (1.0 - h_flat[j + t])
            if 0 <= j - t < n:
                grad[j] -= w[k] * dx * h_flat[j - t]
    return grad

def optimize_n(h_init_n, n, S, dx, t0, budget_s):
    best_local_h = h_init_n.copy()
    best_local_c5 = compute_c5(h_init_n)
    fracs = [0.60, 0.55, 0.50, 0.45, 0.40, 0.35]
    for frac in fracs:
        if time.time() - t0 > budget_s * 0.88:
            break
        h_cur = h_init_n + frac * np.random.randn(n)
        h_cur = project_sum_box(h_cur, S)
        for beta in [40.0, 80.0, 150.0]:
            if time.time() - t0 > budget_s * 0.88:
                break
            def obj(z):
                return smooth_obj(z, dx, beta)
            def grad(z):
                return smooth_obj_grad(z, dx, beta)
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 2000, 'ftol': 1e-14, 'maxfun': 8000})
            h_opt = project_sum_box(res.x, S)
            # SLSQP polish with constraint
            def obj2(z):
                return smooth_obj(z, dx, 200.0)
            cons = [{'type': 'eq', 'fun': lambda z: z.sum() - S,
                     'jac': lambda z: np.ones(n)}]
            res2 = minimize(obj2, h_opt, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                            constraints=cons, options={'maxiter': 500, 'ftol': 1e-12})
            h_opt2 = project_sum_box(res2.x, S)
            for cand in [h_opt, h_opt2]:
                c5 = compute_c5(cand)
                if c5 < best_local_c5:
                    best_local_c5 = c5
                    best_local_h = cand.copy()
    return best_local_h, best_local_c5

def local_refine(h, S, t0, budget_s):
    n = len(h)
    dx = 2.0 / n
    c5 = compute_c5(h)
    step = 0.02
    for it in range(3000):
        if time.time() - t0 > budget_s * 0.95:
            break
        improved = False
        idx = np.random.permutation(n)
        for i in idx:
            if time.time() - t0 > budget_s * 0.95:
                break
            old = h[i]
            for delta in [-step, step, -step * 0.3, step * 0.3]:
                nv = old + delta
                if nv < 0 or nv > 1:
                    continue
                trial = h.copy()
                trial[i] = nv
                trial = project_sum_box(trial, S)
                c5_t = compute_c5(trial)
                if c5_t < c5 - 1e-10:
                    h = trial
                    c5 = c5_t
                    improved = True
                    break
        if not improved:
            step *= 0.7
            if step < 1e-5:
                break
    return h, c5
