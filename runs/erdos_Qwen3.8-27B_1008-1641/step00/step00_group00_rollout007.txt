            sig = np.random.uniform(0.05, 0.5)

        elif r < 0.55:
            ht = best_h.copy()
            k = np.random.randint(2, max(3, n0 // 4))
            idx = np.random.choice(n0, k, replace=False)
            ht[idx] = 1.0 - ht[idx]
        elif r < 0.75:
            ht = np.random.rand(n0)
        else:
            ht = np.zeros(n0)
            ht[np.random.choice(n0, int(ts0), replace=False)] = 1.0
        ht = project_feasible(ht, ts0)

        if tl() < 8:
            c5_try = compute_c5(ht, n0)
            if c5_try < best_c5:
                best_h, best_c5 = ht, c5_try
            continue

        tau = np.random.choice([0.015, 0.02, 0.03, 0.05])
        ho, c5 = do_slsqp(ht, n0, tau, 150)
        if c5 < best_c5:
            best_h, best_c5 = ho, c5

    # Phase 3: Try even n for binary-friendly optimization
    for n_alt in [46, 48, 50, 52]:
        if tl() < 25:
            break
        ts_alt = n_alt / 2.0
        # Structured init: evenly spaced
        ha = np.zeros(n_alt)
        pos = np.round(np.linspace(0.5, n_alt - 1.5, int(ts_alt))).astype(int)
        ha[np.unique(pos)] = 1.0
        # Ensure exact count
        n_ones = int(np.sum(ha))
        if n_ones < int(ts_alt):
            for i in np.random.permutation(n_alt):
                if n_ones >= int(ts_alt):
                    break
                if ha[i] == 0:
                    ha[i] = 1.0
                    n_ones += 1
        elif n_ones > int(ts_alt):
            ones_idx = np.where(ha == 1)[0]
            for i in ones_idx[:n_ones - int(ts_alt)]:
                ha[i] = 0.0

        for tau in [0.04, 0.02]:
            if tl() < 10:
                break
            ho, c5 = do_slsqp(ha, n_alt, tau, 200)
            if c5 < best_c5:
                best_h, best_c5, best_n = ho, c5, n_alt
                ha = ho.copy()

        # Binary local search for this n
        if tl() > 15:
            hb = np.zeros(n_alt)
            hb[np.random.choice(n_alt, int(ts_alt), replace=False)] = 1.0
            hb, c5b = local_search_binary(hb, n_alt, steps=100)
            if c5b < best_c5:
                hb, c5b2 = do_slsqp(hb, n_alt, 0.02, 200)
                if c5b2 < best_c5:
                    best_h, best_c5, best_n = hb, c5b2, n_alt

    # Phase 4: Final polish on best solution
    if tl() > 10:
        for tau in [0.01, 0.005]:
            if tl() < 5:
                break
            ho, c5 = do_slsqp(best_h, best_n, tau, 200)
            if c5 < best_c5:
                best_h, best_c5 = ho, c5

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
