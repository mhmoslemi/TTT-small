def gradient_attempt(h, f=None, S=None, max_active=8):
    if f is None: f = np.correlate(h,1.0-h,mode='full')
    M = float(f.max())
    tol = max(1e-9, 1e-5 * max(1.0,M))
    idx_all = np.where(f >= M - tol)[0]
    if idx_all.size == 0:
       idx_all = np.array([int(np.argmax(f))])
    if idx_all.size > max_active:
       vals=f[idx_all]
       # choose largest
       order=np.argsort(vals)
       idx_all=idx_all[order[-max_active:]]
    g=np.zeros_like(h)
    n=len(h)
    for idx in idx_all:
       d=idx-(n-1)
       if d==0:
          g += 1.0 - 2.0*h
       elif d>0:
          a=np.arange(n-d)
          b=a+d
          g[a] += 1.0 - h[b]
          g[b] -= h[a]
       else:
          r=-d
          a=np.arange(r,n)
          b=a-r
          g[a] += 1.0 - h[b]
          g[b] -= h[a]
    g = g / float(idx_all.size)
    g -= g.mean()
    gn = float(np.linalg.norm(g))
    if not np.isfinite(gn) or gn < 1e-14:

    g = g / gn
    # try modules; descending active max. We can also use f_new.max()

def project_box_sum(z, S):
    x = np.asarray(z, dtype=float)
    x = np.clip(x, 0.0, 1.0)
    x = np.where(np.isfinite(x), x, 0.5)
    s = float(np.sum(x))
    if abs(s - S) < 1e-14:
        return x
    if s < S:
        lo, hi = 0.0, 1.0
        for _ in range(45):
            mid = 0.5*(lo+hi)
            y = np.clip(x+mid,0.0,1.0)
            if float(np.sum(y)) < S:

def quantize_int(h, Q):
    h=np.asarray(h,dtype=float).reshape(-1)
    n=h.size
    Q=int(Q)
    T=(n*Q)//2
    x=np.where(np.isfinite(h), h, 0.5)
    x=np.clip(x,0.0,1.0)
    raw=x*float(Q)
    base=np.floor(raw + 1e-9).astype(np.int64)
    base=np.clip(base,0,Q).astype(np.int64)
    rem=T - int(base.sum(dtype=np.int64))
    if rem > 0:
       frac=raw - base.astype(float)
       order=np.argsort(-frac, kind='stable')
       added=0
       for idx in order:
          if added >= rem: break
          if base[idx] < Q:
             base[idx] += 1; added += 1
       if added < rem:
          order=np.argsort(base, kind='stable')
          for idx in order:
             if added >= rem: break
             if base[idx] < Q:
                base[idx] += 1; added +=1
    elif rem < 0:
       frac=raw - base.astype(float)
       order=np.argsort(frac, kind='stable')
       need=-rem; removed=0
       for idx in order:
          if removed >= need: break
          if base[idx] > 0:
             base[idx] -= 1; removed +=1
       if removed < need:
          order=np.argsort(-base, kind='stable')
          for idx in order:
             if removed >= need: break
             if base[idx] > 0:
                base[idx] -=1; removed+=1
    # final adjustment for robustness (could loop multiple)
    diff=T - int(base.sum(dtype=np.int64))
    idx=0
    guard=0
    while diff > 0 and guard < 10*n:
       if base[idx] < Q:
          base[idx] += 1; diff -= 1

          base[idx] -= 1; diff +=1

import numpy as np
import time


def project_box_sum(z, S):
    x = np.asarray(z, dtype=float).reshape(-1)
    x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    s = float(np.sum(x))
    if abs(s - float(S)) <= 1e-14:
        return x
    lo, hi = 0.0, 1.0
    if s < float(S):
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            y = np.clip(x + mid, 0.0, 1.0)
            if float(np.sum(y)) < float(S):
                lo = mid
            else:
                hi = mid
        return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(np.sum(y)) > float(S):
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)


def feasible_h(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size < 2 or not np.all(np.isfinite(h)):
        return False
    if float(np.min(h)) < -1e-13 or float(np.max(h)) > 1.0 + 1e-13:
        return False
    return abs(float(np.sum(h)) - 0.5 * h.size) <= 1e-14


def max_corr(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def c5_value(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    return max_corr(h) * (2.0 / float(h.size))


def c5_int(a, Q):
    a = np.asarray(a, dtype=np.int64).reshape(-1)
    n = a.size
    b = int(Q) - a
    c = np.correlate(a, b, mode="full")
    return float(np.max(c)) * (2.0 / (float(n) * float(Q) * float(Q)))


def quantize_int(h, Q):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    Q = int(Q)
    T = (n * Q) // 2
    h = np.where(np.isfinite(h), h, 0.5)
    h = np.clip(h, 0.0, 1.0)
    raw = h * float(Q)
    base = np.floor(raw + 1e-9).astype(np.int64)
    base = np.clip(base, 0, Q).astype(np.int64)
    rem = T - int(base.sum(dtype=np.int64))
    if rem > 0:
        frac = raw - base.astype(float)
        order = np.argsort(-frac, kind="stable")
        added = 0
        for idx in order:
            if added >= rem:
                break
            if base[idx] < Q:
                base[idx] += 1
                added += 1
        if added < rem:
            order = np.argsort(base, kind="stable")
            for idx in order:
                if added >= rem:
                    break
                if base[idx] < Q:
                    base[idx] += 1
                    added += 1
    elif rem < 0:
        need = -rem
        frac = raw - base.astype(float)
        order = np.argsort(frac, kind="stable")
        removed = 0
        for idx in order:
            if removed >= need:
                break
            if base[idx] > 0:
                base[idx] -= 1
                removed += 1
        if removed < need:
            order = np.argsort(-base, kind="stable")
            for idx in order:
                if removed >= need:
                    break
                if base[idx] > 0:
                    base[idx] -= 1
                    removed += 1
    diff = T - int(base.sum(dtype=np.int64))
    guard = 0
    idx = 0
    while diff > 0 and guard < 20 * n:
        if base[idx] < Q:
            base[idx] += 1
            diff -= 1
        idx = (idx + 1) % max(1, n)
        guard += 1
    guard = 0
    idx = n - 1
    while diff < 0 and guard < 20 * n:
        if base[idx] > 0:
            base[idx] -= 1
            diff += 1
        idx = (idx - 1) % max(1, n)
        guard += 1
    return base


def make_default_h(n, rng):
    n = int(n)
    S = 0.5 * n
    h = np.zeros(n)
    k = int(np.floor(S))
    frac = S - k
    idx = rng.permutation(n)
    if k > 0:
        h[idx[:k]] = 1.0
    if k < n:
        h[idx[k]] = frac
    return h


def make_random_feasible(n, rng, mode):
    n = int(n)
    S = 0.5 * n
    mode = int(mode) % 3
    if mode == 0:
        return project_box_sum(rng.uniform(0.0, 1.0, size=n), S)
    if mode == 1:
        h = np.zeros(n)
        k = int(np.floor(S))
        frac = S - k
        idx = rng.permutation(n)
        if k > 0:
            h[idx[:k]] = 1.0
        if k < n:
            h[idx[k]] = frac
        return h
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    k = int(1 + rng.integers(0, 4))
    amp = 0.6 + 0.4 * float(rng.random())
    phase = float(rng.uniform(0.0, 6.283185307179586))
    h = np.clip(0.5 + amp * np.sin(2.0 * 3.141592653589793 * k * x + phase), 0.0, 1.0)
    return project_box_sum(h, S)


def upsample_h(h, m):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    m = int(m)
    S = 0.5 * m
    if m == n:
        return project_box_sum(h, S)
    xold = np.linspace(0.0, 1.0, n, endpoint=False)
    xnew = np.linspace(0.0, 1.0, m, endpoint=False)
    y = np.interp(xnew, xold, h)
    return project_box_sum(y, S)


def replicate_h(h, m):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    m = int(m)
    S = 0.5 * m
    if m % n == 0:
        y = np.repeat(h, m // n)
        return project_box_sum(y, S)
    return upsample_h(h, m)


def normalize_centered(g):
    g = np.asarray(g, dtype=float)
    if not np.all(np.isfinite(g)):
        return None
    g = g - float(np.mean(g))
    nrm = float(np.linalg.norm(g))
    if nrm <= 1e-14:
        return None
    return g / nrm


def grad_actual_offset(h, k):
    n = h.size
    g = np.zeros(n, dtype=float)
    k = int(k)
    if k == 0:
        return 1.0 - 2.0 * h
    m = n - abs(k)
    if m <= 0:
        return g
    a = np.arange(m)
    b = a + abs(k)
    if k > 0:
        g[a] -= h[b]
        g[b] += 1.0 - h[a]
    else:
        g[a] += 1.0 - h[b]
        g[b] -= h[a]
    return g


def grad_from_offsets(h, offsets):
    offsets = np.asarray(offsets, dtype=int).reshape(-1)
    if offsets.size == 0:
        return None
    g = np.zeros(h.size, dtype=float)
    for k in offsets:
        g += grad_actual_offset(h, int(k))
    return normalize_centered(g / float(offsets.size))


def smooth_grad_direction(h, beta, f):
    M = float(np.max(f))
    w = np.exp((f - M) / float(beta))
    wsum = float(np.sum(w))
    if not np.isfinite(wsum) or wsum <= 0.0:
        return None
    g = np.zeros(h.size, dtype=float)
    center = h.size - 1
    thresh = 1e-5 * wsum
    for idx in range(f.size):
        wi = float(w[idx])
        if wi > thresh:
            g += (wi / wsum) * grad_actual_offset(h, int(idx - center))
    return normalize_centered(g)


def gradient_attempt(h, S, f=None):
    if f is None:
        f = np.correlate(h, 1.0 - h, mode="full")
    M = float(np.max(f))
    n = h.size
    center = n - 1
    idx = np.where(f >= M - 1e-4)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(f))])
    if idx.size > 8:
        idx = idx[np.argsort(f[idx], kind="stable")[-8:]]
    offsets = (idx - center).astype(int)
    directions = []
    g = grad_from_offsets(h, offsets)
    if g is not None:
        directions.append(g)
    for off in offsets[:4]:
        g = grad_from_offsets(h, np.array([off]))
        if g is not None:
            directions.append(g)
    for beta in (0.10, 0.05):
        g = smooth_grad_direction(h, beta, f)
        if g is not None:
            directions.append(g)
    best = M
    for g in directions:
        if g is None:
            continue
        for step in (0.4, 0.2, 0.1, 0.05, 0.025, 0.0125, 0.00625, 0.003125):
            hn = project_box_sum(h - step * g, S)
            m = max_corr(hn)
            if m < best - 1e-12:
                return hn, m
    return h, best


def swap_pass(h, S, current_c, deadline, rng):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    best_h = h.copy()
    best_c = float(current_c)
    work = h.copy()
    checks = 0
    if n <= 80:
        for i in range(n):
            vi = work[i]
            for j in range(i + 1, n):
                vj = work[j]
                if abs(vi - vj) > 1e-11:
                    work[i], work[j] = vj, vi
                    c = max_corr(work)
                    if c < best_c:
                        best_c = c
                        best_h = work.copy()
                    work[i], work[j] = vi, vj
                checks += 1
                if (checks & 511) == 0 and time.time() > deadline:
                    return best_h, best_c
    else:
        num = min(1500, 40 * n)
        for _ in range(num):
            if time.time() > deadline:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            vi = work[i]
            vj = work[j]
            if abs(vi - vj) <= 1e-11:
                continue
            work[i], work[j] = vj, vi
            c = max_corr(work)
            if c < best_c:
                best_c = c
                best_h = work.copy()
            work[i], work[j] = vi, vj
    return best_h, best_c


def random_transfers(h, S, current_c, num_eval, deadline, rng):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    best_h = h.copy()
    best_c = float(current_c)
    alphas = (1.0, 0.5, 0.25, 0.10, 0.05)
    for _ in range(int(num_eval)):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        work = h.copy()
        if rng.random() < 0.5:
            live = float(min(work[i], 1.0 - work[j]))
            if live <= 1e-12:
                continue
            a = live * float(alphas[int(rng.integers(len(alphas)))])
            work[i] -= a
            work[j] += a
        else:
            live = float(min(work[j], 1.0 - work[i]))
            if live <= 1e-12:
                continue
            a = live * float(alphas[int(rng.integers(len(alphas)))])
            work[i] += a
            work[j] -= a
        c = max_corr(work)
        if c < best_c:
            best_c = c
            best_h = work.copy()
    return best_h, best_c


def local_refine_continuous(h0, S, deadline, rng):
    h = project_box_sum(h0, S)
    best_h = h.copy()
    best_c = max_corr(h)
    cur_h = h.copy()
    cur_c = best_c
    patience = 0
    while time.time() < deadline:
        improved = False
        for _ in range(25):
            if time.time() >= deadline:
                break
            f = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            hn, cn = gradient_attempt(cur_h, S, f)
            if cn < cur_c - 1e-12:
                cur_h = hn
                cur_c = cn
                improved = True
                if cn < best_c:
                    best_c = cn
                    best_h = cur_h.copy()
                continue
            break
        if improved:
            patience = 0
            continue
        hl, cl = swap_pass(cur_h, S, cur_c, deadline, rng)
        if cl < cur_c - 1e-12:
            cur_h = hl
            cur_c = cl
            improved = True
            if cl < best_c:
                best_c = cl
                best_h = cur_h.copy()
        if not improved:
            hr, cr = random_transfers(cur_h, S, cur_c, 220, deadline, rng)
            if cr < cur_c - 1e-12:
                cur_h = hr
                cur_c = cr
                improved = True
                if cr < best_c:
                    best_c = cr
                    best_h = cur_h.copy()
        if not improved:
            patience += 1
            if patience > 8:
                patience = 0
                cur_h = best_h.copy()
                cur_c = best_c
            scale = 0.03 + 0.02 * float(min(patience, 8))
            cur_h = project_box_sum(best_h + rng.normal(0.0, scale, best_h.size), S)
            cur_c = max_corr(cur_h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c


def discrete_best_move_full(a, Q, deadline):
    n = a.size
    best_c = c5_int(a, Q)
    best = a.copy()
    work = a.copy()
    checks = 0
    for i in range(n):
        vi = work[i]
        for j in range(i + 1, n):
            vj = work[j]
            if vi != vj:
                work[i], work[j] = vj, vi
                c = c5_int(work, Q)
                if c < best_c:
                    best_c = c
                    best = work.copy()
                work[i], work[j] = vi, vj
            checks += 1
            if (checks & 1023) == 0 and time.time() > deadline:
                return best, best_c
    for i in range(n):
        vi = work[i]
        for j in range(n):
            if i == j:
                continue
            vj = work[j]
            if vi > 0 and vj < Q:
                work[i] = vi - 1
                work[j] = vj + 1
                c = c5_int(work, Q)
                if c < best_c:
                    best_c = c
                    best = work.copy()
                work[i] = vi
                work[j] = vj
            checks += 1
            if (checks & 1023) == 0 and time.time() > deadline:
                return best, best_c
    return best, best_c


def discrete_random_best_move(a, Q, deadline, rng, num_evals):
    n = a.size
    best_c = c5_int(a, Q)
    best = a.copy()
    work = a.copy()
    for _ in range(int(num_evals)):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if rng.random() < 0.35:
            vi = work[i]
            vj = work[j]
            if vi != vj:
                work[i], work[j] = vj, vi
                c = c5_int(work, Q)
                if c < best_c:
                    best_c = c
                    best = work.copy()
                work[i], work[j] = vi, vj
        else:
            if work[i] > 0 and work[j] < Q:
                work[i] -= 1
                work[j] += 1
                c = c5_int(work, Q)
                if c < best_c:
                    best_c = c
                    best = work.copy()
                work[i] += 1
                work[j] -= 1
            elif work[j] > 0 and work[i] < Q:
                work[i] += 1
                work[j] -= 1
                c = c5_int(work, Q)
                if c < best_c:
                    best_c = c
                    best = work.copy()
                work[i] -= 1
                work[j] += 1
    return best, best_c


def one_int_move(a, Q, rng):
    n = a.size
    for _ in range(60):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if a[i] != a[j]:
            w = a.copy()
            w[i], w[j] = a[j], a[i]
            return w
        if a[i] > 0 and a[j] < Q:
            w = a.copy()
            w[i] -= 1
            w[j] += 1
            return w
        if a[j] > 0 and a[i] < Q:
            w = a.copy()
            w[i] += 1
            w[j] -= 1
            return w
    for i in range(n):
        for j in range(i + 1, n):
            if a[i] != a[j]:
                w = a.copy()
                w[i], w[j] = a[j], a[i]
                return w
            if a[i] > 0 and a[j] < Q:
                w = a.copy()
                w[i] -= 1
                w[j] += 1
                return w
            if a[j] > 0 and a[i] < Q:
                w = a.copy()
                w[i] += 1
                w[j] -= 1
                return w
    return a.copy()


def discrete_kick(a, Q, rng, up_tol, force=False):
    base_c = c5_int(a, Q)
    work = a.copy()
    n = work.size
    steps = int(rng.integers(3, 11))
    deltas = (1, 2, 4, 8, 16, 32)
    for _ in range(steps):
        if rng.random() < 0.4:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and work[i] != work[j]:
                work[i], work[j] = work[j], work[i]
        else:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = int(deltas[int(rng.integers(len(deltas)))])
            if rng.random() < 0.5:
                dd = min(d, int(work[i]), int(Q - work[j]))
                if dd > 0:
                    work[i] -= dd
                    work[j] += dd
            else:
                dd = min(d, int(work[j]), int(Q - work[i]))
                if dd > 0:
                    work[i] += dd
                    work[j] -= dd
    if force and np.array_equal(work, a):
        work = one_int_move(work, Q, rng)
    c = c5_int(work, Q)
    if force or c <= base_c + float(up_tol):
        return work, c
    return a.copy(), base_c


def discrete_local(a0, Q, deadline, rng):
    a0_arr = np.asarray(a0)
    if a0_arr.dtype.kind == "f":
        a = quantize_int(a0_arr, Q)
    else:
        a = a0_arr.astype(np.int64).copy()
        T = (a.size * int(Q)) // 2
        if int(a.sum(dtype=np.int64)) != T:
            a = quantize_int(a.astype(float) / float(Q), Q)
    best_a = a.copy()
    best_c = c5_int(a, Q)
    cur_a = a.copy()
    cur_c = best_c
    full = a.size <= 64
    up = 0.0004
    stagn = 0
    while time.time() < deadline:
        if full:
            mv_a, mv_c = discrete_best_move_full(cur_a, Q, deadline)
        else:
            mv_a, mv_c = discrete_random_best_move(cur_a, Q, deadline, rng, 900)
        if mv_c < cur_c - 1e-15:
            cur_a, cur_c = mv_a, mv_c
            if cur_c < best_c:
                best_a, best_c = cur_a.copy(), cur_c
            stagn = 0
        else:
            old_a = cur_a
            old_c = cur_c
            force = stagn >= 6
            wa, wc = discrete_kick(cur_a, Q, rng, up, force)
            if wc != old_c or not np.array_equal(wa, old_a):
                cur_a, cur_c = wa, wc
                if cur_c < best_c:
                    best_a, best_c = cur_a.copy(), cur_c
                stagn = 0
            else:
                wa = one_int_move(cur_a, Q, rng)
                wc = c5_int(wa, Q)
                cur_a, cur_c = wa, wc
                if cur_c < best_c:
                    best_a, best_c = cur_a.copy(), cur_c
                stagn = 0
            up = min(0.003, up * 1.3)
            if cur_c > best_c + 0.003:
                cur_a = best_a.copy()
                cur_c = best_c
                up = 0.0004
    return best_a, best_c


def update_cont_best(h, c, best_h, best_c):
    c = float(c)
    if c < float(best_c):
        return np.asarray(h, dtype=float), c
    return best_h, float(best_c)


def update_int_best(a, q, c, best_a, best_q, best_c):
    c = float(c)
    if c < float(best_c):
        return np.asarray(a, dtype=np.int64), int(q), c
    return best_a, int(best_q), float(best_c)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + max(1.0, float(budget_s))
    span = max(1.0, hard - t0)
    rng = np.random.default_rng(int(seed))

    init = globals().get("initial_h_values", None)
    if init is not None:
        hpar = np.asarray(init, dtype=float).copy().reshape(-1)
    else:
        hpar = make_default_h(47, rng)
    if hpar.size < 2:
        hpar = make_default_h(47, rng)
    npar = hpar.size
    Spar = 0.5 * npar
    hpar = project_box_sum(hpar, Spar)

    best_feas_h = None
    best_feas_c = float("inf")
    if feasible_h(hpar):
        best_feas_h = hpar.copy()
        best_feas_c = c5_value(hpar)

    QF = 1 << 16
    QD = 1 << 12
    best_a = quantize_int(hpar, QF)
    best_q = QF
    best_c = c5_int(best_a, QF)

    best_cont_h = hpar.copy()
    best_cont_c = max_corr(hpar) * (2.0 / float(npar))

    e1 = min(hard - 250.0, t0 + min(220.0, 0.20 * span))
    e2 = min(hard - 250.0, t0 + min(500.0, 0.55 * span))
    e3 = min(hard - 140.0, t0 + min(900.0, 0.80 * span))
    e4 = min(hard - 90.0, t0 + min(1000.0, 0.92 * span))

    if time.time() < e1 - 5.0 and npar <= 128:
        dl = min(e1, time.time() + 220.0)
        if dl > time.time():
            h1,c1 = local_refine_continuous(hpar, Spar, dl, rng)
            best_cont_h, best_cont_c = update_cont_best(h1, c1 * (2.0 / h1.size), best_cont_h, best_cont_c)
            a1 = quantize_int(h1, QF)
            best_a, best_q, best_c = update_int_best(a1, QF, c5_int(a1, QF), best_a, best_q, best_c)

    if time.time() < e2 - 10.0:
        rem = max(0.0, e2 - time.time())
        d1 = min(e2 - (120.0 if rem > 200.0 else 0.0), time.time() + max(10.0, 0.6 * rem))
        astart = quantize_int(hpar, QD)
        ares, cres = discrete_local(astart, QD, d1, rng)
        best_a, best_q, best_c = update_int_best(ares, QD, cres, best_a, best_q, best_c)
        if time.time() < e2 - 5.0:
            astart = quantize_int(best_cont_h, QD)
            ares, cres = discrete_local(astart, QD, e2, rng)
            best_a, best_q, best_c = update_int_best(ares, QD, cres, best_a, best_q, best_c)

    if time.time() < e3 - 30.0 and best_cont_h.size <= 50:
        m = 2 * best_cont_h.size
        y1 = upsample_h(best_cont_h, m)
        y2 = replicate_h(best_cont_h, m)
        start = y2 if c5_value(y2) < c5_value(y1) else y1
        dl = min(e3, time.time() + 220.0)
        if dl > time.time():
            hup, cup = local_refine_continuous(start, 0.5 * m, dl, rng)
            best_cont_h, best_cont_c = update_cont_best(hup, cup * (2.0 / hup.size), best_cont_h, best_cont_c)
            aq = quantize_int(hup, QF)
            best_a, best_q, best_c = update_int_best(aq, QF, c5_int(aq, QF), best_a, best_q, best_c)

    if time.time() < e4 - 10.0:
        ares = quantize_int(best_cont_h, QD)
        ares, cres = discrete_local(ares, QD, e4, rng)
        best_a, best_q, best_c = update_int_best(ares, QD, cres, best_a, best_q, best_c)

    mode = 0
    while time.time() < hard - 80.0 and mode < 4:
        ntrial = 47
        if mode % 2 == 0:
            htry = make_random_feasible(ntrial, rng, mode % 3)
        else:
            htry = project_box_sum(hpar + 0.10 * rng.normal(0.0, 1.0, hpar.size), Spar)
        dl = min(hard - 80.0, time.time() + 30.0)
        if dl <= time.time():
            break
        htry, ctry = local_refine_continuous(htry, 0.5 * ntrial, dl, rng)
        aq = quantize_int(htry, QF)
        best_a, best_q, best_c = update_int_best(aq, QF, c5_int(aq, QF), best_a, best_q, best_c)
        best_cont_h, best_cont_c = update_cont_best(htry, ctry * (2.0 / htry.size), best_cont_h, best_cont_c)
        mode += 1

    aq = quantize_int(best_cont_h, QF)
    best_a, best_q, best_c = update_int_best(aq, QF, c5_int(aq, QF), best_a, best_q, best_c)

    if int(best_q) != QF:
        if QF % int(best_q) == 0:
            aF = (best_a * (QF // int(best_q))).astype(np.int64)
            cF = c5_int(aF, QF)
            best_a, best_q, best_c = update_int_best(aF, QF, cF, best_a, best_q, best_c)
        else:
            aF = quantize_int(best_a.astype(float) / float(int(best_q)), QF)
            cF = c5_int(aF, QF)
            best_a, best_q, best_c = update_int_best(aF, QF, cF, best_a, best_q, best_c)
    else:
        aF = best_a.astype(np.int64).copy()

    final_dl = min(hard - 1.0, time.time() + min(20.0, max(1.0, 0.5 * max(0.0, hard - time.time()))))
    if final_dl > time.time():
        aF, cF = discrete_local(aF, QF, final_dl, rng)
        best_a, best_q, best_c = update_int_best(aF, QF, cF, best_a, best_q, best_c)

    if best_feas_h is not None and float(best_feas_c) < float(best_c):
        hout = np.asarray(best_feas_h, dtype=float)
        return hout, float(best_feas_c), int(hout.size)

    hout = best_a.astype(float) / float(int(best_q))
    return hout, float(best_c), int(hout.size)
