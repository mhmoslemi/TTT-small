import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    s = k_max - (n - 1)
    grad = np.zeros(n)
    js = np.arange(n)
    # First term: (1-h[j+s]) for valid j+s
    valid1 = (js + s >= 0) & (js + s < n)
    grad[valid1] += (1.0 - h[js[valid1] + s])
    # Second term: -h[j-s] for valid j-s
    valid2 = (js - s >= 0) & (js - s < n)
    grad[valid2] -= h[js[valid2] - s]
    return grad * dx

def smoothed_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Log-sum-exp for numerical stability
    shifted = corr * tau
    m = np.max(shifted)
    return (m + np.log(np.sum(np.exp(shifted - m)))) / tau

def smoothed_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr * tau
    m = np.max(shifted)
    weights = np.exp(shifted - m) / np.sum(np.exp(shifted - m))
    # Weighted average of individual gradients
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if weights[idx] > 1e-10:
            s = idx - (n - 1)
            js = np.arange(n)
            valid1 = (js + s >= 0) & (js + s < n)
            grad[valid1] += weights[idx] * (1.0 - h[js[valid1] + s])
            valid2 = (js - s >= 0) & (js - s < n)
            grad[valid2] -= weights[idx] * h[js[valid2] - s]
    return grad * dx

def fix_sum(h, target):

    h += diff / len(h)
    return np.clip(h, 0, 1)

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x, float(res.fun)
    except Exception:
        return h0, c5_value(h0)

    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    s = k_max - (n - 1)
    js = np.arange(n)
    grad = np.zeros(n)
    v1 = (js + s >= 0) & (js + s < n)
    grad[v1] += (1.0 - h[js[v1] + s])
    v2 = (js - s >= 0) & (js - s < n)
    grad[v2] -= h[js[v2] - s]
    return grad * dx

def smoothed_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sh = corr * tau
    m = np.max(sh)
    return (m + np.log(np.sum(np.exp(sh - m)))) / tau

def smoothed_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sh = corr * tau
    m = np.max(sh)
    w = np.exp(sh - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    js = np.arange(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-12:
            continue
        s = idx - (n - 1)
        v1 = (js + s >= 0) & (js + s < n)
        grad[v1] += w[idx] * (1.0 - h[js[v1] + s])
        v2 = (js - s >= 0) & (js - s < n)
        grad[v2] -= w[idx] * h[js[v2] - s]
    return grad * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def run_slsqp(h0, n, maxiter=2000, ftol=1e-12):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(c5_value, h0, jac=c5_subgrad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return res.x, float(res.fun)
    except Exception:
        pass
    return h0.copy(), c5_value(h0)

def run_smoothed(h0, n, tau=10.0, maxiter=2000):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: smoothed_obj(x, tau), h0,
                      jac=lambda x: smoothed_grad(x, tau),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return res.x, c5_value(res.x)
    except Exception:
        pass
    return h0.copy(), c5_value(h0)

def make_initial(n, base_h=None):
    if base_h is not None and len(base_h) != n:
        x_old = np.linspace(0, 2, len(base_h))
        x_new = np.linspace(0, 2, n)
        h = np.interp(x_new, x_old, base_h)
    elif base_h is not None:
        h = base_h.copy()
    else:
        h = np.full(n, 0.5)
    return fix_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - 5.0 - elapsed()

    # Load initial
    h_base = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_base = np.array(initial_h_values, dtype=float)

    best_h = h_base.copy() if h_base is not None else np.full(45, 0.5)
    best_n = len(best_h)
    best_c5 = c5_value(best_h)

    # Phase 1: SLSQP on initial
    h_opt, v_opt = run_slsqp(best_h, best_n, maxiter=3000)
    if v_opt < best_c5:
        best_h, best_c5 = h_opt, v_opt

    # Phase 2: Smoothed max with increasing tau
    for tau in [5.0, 10.0, 20.0, 40.0]:
        if remaining() < 20:
            break
        h_sm, v_sm = run_smoothed(best_h, best_n, tau=tau, maxiter=2000)
        if v_sm < best_c5:
            best_h, best_c5 = h_sm, v_sm

    # Phase 3: Try other n values
    n_candidates = [44, 46, 48, 50, 55, 60, 65]
    for n in n_candidates:
        if remaining() < 40:
            break
        if n == best_n:
            continue
        h_init_n = make_initial(n, h_base)
        h_trial, v_trial = run_slsqp(h_init_n, n, maxiter=2000)
        if remaining() < 20:
            continue
        for tau in [10.0, 25.0]:
            if remaining() < 15:
                break
            h_t2, v_t2 = run_smoothed(h_trial, n, tau=tau, maxiter=1500)
            if v_t2 < v_trial:
                h_trial, v_trial = h_t2, v_t2
        if v_trial < best_c5:
            best_h, best_n, best_c5 = h_trial, n, v_trial

    # Phase 4: Random restarts around best
    n_restarts = 0
    while remaining() > 30 and n_restarts < 80:
        scale = 0.02 + 0.04 * np.random.rand()
        h_try = best_h + np.random.normal(0, scale, best_n)
        h_try = fix_sum(h_try, best_n / 2.0)
        h_r, v_r = run_slsqp(h_try, best_n, maxiter=1500, ftol=1e-11)
        if v_r < best_c5:
            best_h, best_c5 = h_r, v_r
        n_restarts += 1

    # Phase 5: Structured restarts (near-binary)
    if remaining() > 30:
        for pattern in range(3):
            if remaining() < 15:
                break
            h_bin = np.zeros(best_n)
            step = best_n // 11
            for b in range(0, best_n, step * 2):
                h_bin[b:min(b + step, best_n)] = 1.0
            if pattern == 1:
                h_bin = np.where(h_bin > 0, 0.0, 1.0)
            elif pattern == 2:
                h_bin = np.where(np.arange(best_n) % 2 == 0, 1.0, 0.0)
            h_bin = fix_sum(h_bin, best_n / 2.0)
            # Add noise
            h_bin = fix_sum(h_bin + np.random.normal(0, 0.1, best_n), best_n / 2.0)
            h_r, v_r = run_slsqp(h_bin, best_n, maxiter=1500)
            if v_r < best_c5:
                best_h, best_c5 = h_r, v_r

    # Phase 6: Final polish
    if remaining() > 10:
        h_p, v_p = run_slsqp(best_h, best_n, maxiter=5000, ftol=1e-14)
        if v_p < best_c5:
            best_h, best_c5 = h_p, v_p
        for tau in [30.0, 60.0]:
            if remaining() < 10:
                break
            h_p2, v_p2 = run_smoothed(best_h, best_n, tau=tau, maxiter=3000)
            if v_p2 < best_c5:
                best_h, best_c5 = h_p2, v_p2

    # Validate and fix
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = best_n / 2.0 - np.sum(best_h)
    best_h += diff / best_n
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = c5_value(best_h)

    return best_h, best_c5, best_n
