def _grad(h, alpha):
    """Analytical gradient of smooth max objective"""
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full')*dx
    m = np.max(corr)
    w = np.exp(alpha*(corr-m))/np.sum(np.exp(alpha*(corr-m)))
    
    # grad of corr[k] w.r.t. h[j]:
    # corr index m corresponds to lag = m-(n-1)
    # d(corr[m])/d(h[j]) = dx * [(1-h[j+lag])*I(0<=j+lag<n) - h[j-lag]*I(0<=j-lag<n)]
    # where lag = m-(n-1)
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to (n-1)
    
    for m in range(2*n-1):
        k = shifts[m]
        # (1-h[j+k]) for valid j+k
        # j+k in [0,n-1] => j in [-k, n-1-k], intersect [0,n-1] => j in [max(0,-k), min(n-1, n-1-k)]
        if k >= 0:
            # term 1: (1-h[j+k]) for j=0..n-1-k
            # term 2: -h[j-k] for j=k..n-1
            grad[:n-k] += w[m] * dx * (1 - h[k:])
            grad[k:] += w[m] * dx * (-h[:n-k])
        else:
            # k < 0, let k' = -k > 0
            kp = -k
            # term 1: (1-h[j-k]) for j=kp..n-1
            # term 2: -h[j+k] for j=0..n-1-kp
            grad[kp:] += w[m] * dx * (1 - h[:n-kp])
            grad[:n-kp] += w[m] * dx * (-h[kp:])
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def _eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha


def _smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        k = m_idx - (n - 1)
        if k >= 0:
            grad[:n - k] += w[m_idx] * dx * (1.0 - h[k:])
            grad[k:] += w[m_idx] * dx * (-h[:n - k])
        else:
            kp = -k
            grad[kp:] += w[m_idx] * dx * (1.0 - h[:n - kp])
            grad[:n - kp] += w[m_idx] * dx * (-h[kp:])
    return grad


def _project_sum(h, n, iters=200):
    for _ in range(iters):
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - n / 2.0
        if abs(diff) < 1e-12:
            break
        h -= diff / n
    return h


def _make_init(n, rng, mode=0):
    if mode == 0:
        h = rng.random(n)
    elif mode == 1:
        h = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        h += rng.random(n) * 0.1
    elif mode == 2:
        h = 0.5 + 0.3 * np.sin(np.pi * np.arange(n) / n * 3.0 + rng.random())
    else:
        h = rng.random(n)
        h = np.maximum(h, 0.5)
    h = _project_sum(h, n)
    return h

    t0 = time.time()

        try:
            res = minimize(
                _smooth_max_obj, h, args=(alpha,),
                jac=lambda x, a: _smooth_max_grad(x, a),
                method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
            )
            xc = _project_sum(np.clip(res.x, 0, 1), n)
            c5 = _eval(xc)
            if c5 < best_c5:
                best_c5 = c5
                best_h = xc.copy()
            elif c5 < _eval(h) + 1e-8:
                h = xc.copy()
                best_h = h.copy()
                best_c5 = c5
            else:
                h = _project_sum(np.clip(res.x, 0, 1), n)
                hc5 = _eval(h)
