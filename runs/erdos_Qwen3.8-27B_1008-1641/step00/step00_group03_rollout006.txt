import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def obj_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = int(np.argmax(corr))
    val = corr[max_idx] * dx
    offset = max_idx - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        valid = n - offset
        for i in range(valid):
            grad[i] += 1.0 - h[i + offset]
            grad[i + offset] -= h[i]
    else:
        valid = n + offset
        for i in range(valid):
            grad[i - offset] -= h[i]
            grad[i] += 1.0 - h[i - offset]
    return val, grad * dx


def fix_constraint(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def slsqp_optimize(h0, n, maxiter=300):
    h0 = fix_constraint(h0.copy(), n)
    constraint = {'type': 'eq',
                  'fun': lambda x: np.sum(x) - n / 2.0,
                  'jac': lambda x: np.ones(n)}
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return obj_subgrad(x)[0]

    def g(x):
        return obj_subgrad(x)[1]

    res = minimize(f, h0, jac=g, method='SLSQP', bounds=bounds,
                   constraints=[constraint],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return fix_constraint(res.x, n)


def binary_local_search(h, rng, max_iter=20000):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_iter):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            h[i], h[j] = h[j], h[i]
    return h, c5_cur


def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def interpolated_to(h_src, n_dst):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h = np.interp(x_dst, x_src, h_src)
    return fix_constraint(h, n_dst)
