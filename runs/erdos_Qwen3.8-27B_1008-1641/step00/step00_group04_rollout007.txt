import numpy as np
import time

def _corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def _c5(h):
    return float(_corr_dx(h).max())

def _project(h, target):
    h = np.array(h, dtype=np.float64, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(2000):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        if not m.any():
            break
        h[m] += d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_sm(h, T):
    n = len(h)
    dx = 2.0 / n
    cd = _corr_dx(h)
    s = cd / T
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        L = k - (n - 1)
        a = w[k]
        if L >= 0:
            m = n - L
            if m > 0:
                g[:m] += a * (1.0 - h[L:])
                g[L:] -= a * h[:m]
        else:
            m = n + L
            if m > 0:
                g[-L:] += a * (1.0 - h[:m])
                g[:m] -= a * h[-L:]
    return dx * g

def _grad_hd(h):
    n = len(h)
    dx = 2.0 / n
    cd = _corr_dx(h)
    mx = cd.max()
    act = cd >= mx - 1e-8
    na = max(1, int(np.sum(act)))
    g = np.zeros(n)
    for k in np.where(act)[0]:
        L = k - (n - 1)
        if L >= 0:
            m = n - L
            if m > 0:
                g[:m] += 1.0 - h[L:]
                g[L:] -= h[:m]
        else:
            m = n + L
            if m > 0:
                g[-L:] += 1.0 - h[:m]
                g[:m] -= h[-L:]
    return dx * g / na

def _gd(h, target, n_iter, T0, T1, lr0, lr1, mom_decay=0.8):
    n = len(h)
    h = h.copy()
    best = _c5(h)
    bh = h.copy()
    mom = np.zeros(n)
    for it in range(n_iter):
        f = (it + 1) / n_iter
        T = T0 * (T1 / T0) ** f
        lr = lr0 * (lr1 / lr0) ** f
        g = _grad_sm(h, T) if T > 0.01 else _grad_hd(h)
        mom = mom_decay * mom - lr * g
        h = _project(h + mom, target)
        c = _c5(h)
        if c < best:
            best, bh = c, h.copy()
    return bh, best

def _polish(h, target, n_iter=40000, s0=0.03):
    n = len(h)
    h = h.copy()
    best = _c5(h)
    fj = 0.00005 / s0
    for it in range(n_iter):
        f = it / n_iter
        step = s0 * fj ** f
        i = it % n
        for sgn in (1, -1):
            for mult in (1.0, 0.5, 0.25, 2.0, 0.1):
                v = sgn * step * mult
                h2 = h.copy()
                h2[i] += v
                if h2[i] < -0.05 or h2[i] > 1.05:
                    continue
                h2 = _project(h2, target)
                c = _c5(h2)
                if c < best - 1e-15:
                    h, best = h2, c
                    break
    return h, best
