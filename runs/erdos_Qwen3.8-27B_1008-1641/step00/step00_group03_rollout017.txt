import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        mask = (h > 1e-7) & (h < 1.0 - 1e-7)
        if mask.sum() > 0:
            h[mask] += d / mask.sum()
        else:
            h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h_init, n_opt, maxit=5000):
    dx_o = 2.0 / n_opt
    tgt = n_opt / 2.0
    def obj(h):
        c = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(c) * dx_o)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt, 'jac': lambda x: np.ones(n_opt)}]
    bnds = [(0.0, 1.0)] * n_opt
    res = minimize(obj, h_init, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxit, 'ftol': 1e-15})
    h_new = fix_sum(res.x, tgt)
    return h_new, compute_c5(h_new, dx_o)

def smooth_opt(h_init, n_opt, T_vals=(20.0, 50.0, 150.0)):
    dx_o = 2.0 / n_opt

    best = h_init.copy()
    best_v = compute_c5(h_init, dx_o)

        def obj_s(h):
            c = np.correlate(h, 1.0 - h, mode="full") * dx_o
            m = np.max(c)
            pen = 500.0 * (h.sum() - tgt) ** 2
            return float(m + np.log(np.sum(np.exp(c - m))) / T + pen)
        res = minimize(obj_s, best, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_opt,
                       options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = fix_sum(res.x, tgt)
        v = compute_c5(h_new, dx_o)
        if v < best_v:
            best, best_v = h_new, v
    return best, best_v
