import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((corr * dx).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    m = c.max()
    ec = np.exp((c - m) / t)
    S = ec.sum()
    f = m + t * np.log(S)
    w = ec / S
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        j = n - 1 + k
        wj = w[j]
        if wj < 1e-14:
            continue
        i_lo1 = max(0, -k)
        i_hi1 = min(n, n - k)
        if i_hi1 > i_lo1:
            idx = np.arange(i_lo1, i_hi1)
            grad[idx] += dx * wj * (1.0 - h[idx + k])
        i_lo2 = max(0, k)
        i_hi2 = min(n, n + k)
        if i_hi2 > i_lo2:
            idx2 = np.arange(i_lo2, i_hi2)
            grad[idx2] += -dx * wj * h[idx2 - k]
    return f, grad

def lbfbs_optimize(h0, n, t, max_iter):
    h = project_feasible(h0, n)
    def fg(x):
        hp = np.clip(x, 0.0, 1.0)
        return obj_grad(hp, n, t)
    try:
        res = minimize(fg, h, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-14,
                                'gtol': 1e-10, 'maxls': 30})
        h = project_feasible(res.x, n)
    except Exception:
        pass
    return h

def slsqp_optimize(h0, n, t, max_iter):
    h = project_feasible(h0, n)
    target = n * 0.5
    def fg(x):
        return obj_grad(x, n, t)
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fg, h, method='SLSQP', jac=True, constraints=cons,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        h = project_feasible(res.x, n)
    except Exception:
        pass
    return h

def binary_local_search(h0, n, n_iter, time_limit):
    t0 = time.time()
    h = (h0 > 0.5).astype(float)
    num_ones = int(np.round(h.sum()))
    target_ones = n // 2
    if num_ones > target_ones:
        idx = np.where(h == 1)[0]
        h[idx[:num_ones - target_ones]] = 0
    elif num_ones < target_ones:
        idx = np.where(h == 0)[0]
        h[idx[:target_ones - num_ones]] = 1
    c5_cur = compute_c5(h)
    rng = np.random.RandomState(999)
    for _ in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        i, j = rng.randint(0, n, 2)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            h[i], h[j] = h[j], h[i]
    return h, c5_cur

def coordinate_descent(h, n, n_iter, time_limit):
    t0 = time.time()
    c5_cur = compute_c5(h)
    rng = np.random.RandomState(777)
    for _ in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        i = rng.randint(0, n)
        step = 0.005
        h[i] = np.clip(h[i] + step, 0.0, 1.0)
        h = project_feasible(h, n)
        c5_new = compute_c5(h)
        if c5_new >= c5_cur:
            h[i] = np.clip(h[i] - 2.0 * step, 0.0, 1.0)
            h = project_feasible(h, n)
            c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            h[i] = np.clip(h[i] + 2.0 * step, 0.0, 1.0)
            h = project_feasible(h, n)
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    # Get initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.ones(70) * 0.5
    n_init = len(h_init)
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: L-BFGS-B at initial n with decreasing temperature
    t_list1 = [0.05, 0.02, 0.005, 0.001, 0.0003]
    h_cur = best_h.copy()
    for t in t_list1:
        if time.time() - start_time > budget_s * 0.2:
            break
        h_cur = lbfbs_optimize(h_cur, n_init, t, 300)
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_c5, best_h = c5, h_cur.copy()

    # SLSQP refinement at initial n
    if time.time() - start_time < budget_s * 0.3:
        h_sqp = slsqp_optimize(best_h, n_init, 0.0005, 200)
        c5 = compute_c5(h_sqp)
        if c5 < best_c5:
            best_c5, best_h = c5, h_sqp.copy()

    # Binary local search at initial n
    if time.time() - start_time < budget_s * 0.5:
        h_bin, c5_bin = binary_local_search(best_h, n_init, 80000, budget_s * 0.08)
        if c5_bin < best_c5:
            best_c5, best_h = c5_bin, h_bin.copy()
        # Also try SLSQP from binary solution
        if time.time() - start_time < budget_s * 0.55:
            h_mix = slsqp_optimize(h_bin, n_init, 0.0003, 200)
            c5_mix = compute_c5(h_mix)
            if c5_mix < best_c5:
                best_c5, best_h = c5_mix, h_mix.copy()

    # Phase 2: Try other resolutions by interpolation
    n_targets = [nn for nn in [50, 80, 100, 140, 200, 300, 500] if nn < 1000]
    for n_try in n_targets:
        if time.time() - start_time > budget_s * 0.8:
            break
        x_old = (np.arange(len(best_h)) + 0.5) * (2.0 / len(best_h))
        x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
        h0 = np.interp(x_new, x_old, best_h)
        h0 = project_feasible(h0, n_try)
        for t in [0.02, 0.003, 0.0005]:
            if time.time() - start_time > budget_s * 0.8:

            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5, best_h = c5_t, h_t.copy()
            h0 = h_t if c5_t < compute_c5(h0) else h0

    # Phase 3: Multi-start at best resolution
    n_best = len(best_h)
    np_starts = 3
    for s in range(np_starts):
        if time.time() - start_time > budget_s * 0.9:

        h_s = best_h + 0.03 * rng.randn(n_best)
        h_s = project_feasible(h_s, n_best)
        h_s = lbfbs_optimize(h_s, n_best, 0.002, 200)
        h_s = lbfbs_optimize(h_s, n_best, 0.0003, 200)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()

    if time.time() - start_time < budget_s * 0.95:
        h_cd, c5_cd = coordinate_descent(best_h, n_best, 20000, budget_s * 0.03)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Final SLSQP polish

        c5_final = compute_c5(h_final)

    best_c5 = compute_c5(best_h)
    n_final = len(best_h)
    print(f"Result: n={n_final}, C5={best_c5:.6f}")
    return best_h, best_c5, n_final
