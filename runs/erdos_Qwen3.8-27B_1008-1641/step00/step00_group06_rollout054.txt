def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                room = 1.0 - h[i]
                add = min(diff, room)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:
            idx = np.argsort(h)[::-1]
            for i in idx:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if -diff < 1e-12:
                    break
    return h

def smooth_gradient(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cs = corr / t
    cs -= cs.max()
    w = np.exp(cs)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-8:
            continue
        s = m - (n - 1)
        wm = w[m]
        if s >= 0:

                grad[:n - s] -= wm * h[s:]
        else:
            tt = -s

                grad[:n - tt] += wm * (1.0 - h[tt:])
                grad[tt:] -= wm * h[:n - tt]
