            h2[p2], h2[q2] = h2[q2], h2[p2]
            nc2 = evaluate_c5(h2, n)
            if nc2 <= ct or rng.random() < math.exp(-(nc2 - ct) / T2):
                ct = nc2
                ones2[ii], zeros2[jj] = q2, p2
                if nc2 < best_c5:
                    best_c5 = nc2
                    best_h = h2.copy()
            else:
                h2[p2], h2[q2] = h2[q2], h2[p2]

    # Try other n values
    for n_tr in [60, 80, 100, 120, 140]:
        if tleft() < 40:
            break
        for rr in range(2):
            if tleft() < 20:
                break
            ht = np.zeros(n_tr)
            ht[rng.choice(n_tr, size=n_tr // 2, replace=False)] = 1.0
            ones_t = np.where(ht == 1.0)[0].tolist()
            zeros_t = np.where(ht == 0.0)[0].tolist()
            ct = evaluate_c5(ht, n_tr)
            T2 = 0.05
            dl = time.time() + max(15, tleft() // 6 - 10)
            while time.time() < dl and tleft() > 15:
                T2 = max(T2 * 0.99999, 0.002)
                ii = int(rng.integers(len(ones_t)))
                jj = int(rng.integers(len(zeros_t)))
                p2, q2 = ones_t[ii], zeros_t[jj]
                ht[p2], ht[q2] = ht[q2], ht[p2]
                nc2 = evaluate_c5(ht, n_tr)
                if nc2 <= ct or rng.random() < math.exp(-(nc2 - ct) / T2):
                    ct = nc2
                    ones_t[ii], zeros_t[jj] = q2, p2
                    if nc2 < best_c5:
                        best_c5 = nc2
                        best_h = ht.copy()
                        best_n = n_tr
                else:
                    ht[p2], ht[q2] = ht[q2], ht[p2]

    # Steepest-descent polish on best
    if tleft() > 10 and best_h is not None:
        h = best_h.copy()
        n = best_n
        ones = np.where(h == 1.0)[0].tolist()
        zeros = np.where(h == 0.0)[0].tolist()
        curr = best_c5
        while tleft() > 3:
            found = False
            bd = 0.0
            bsw = None
            for i in range(len(ones)):
                pi = ones[i]
                for j in range(len(zeros)):
                    qj = zeros[j]
                    h[pi] = 0.0
                    h[qj] = 1.0
                    nc = evaluate_c5(h, n)
                    d = nc - curr
                    if d < bd:
                        bd = d
                        bsw = (i, j, pi, qj)
                    h[pi] = 1.0
                    h[qj] = 0.0
            if bsw is None:
                break
            i, j, pi, qj = bsw
            h[pi] = 0.0
            h[qj] = 1.0
            ones[i] = qj
            zeros[j] = pi
            curr += bd
            if curr < best_c5:
                best_c5 = curr
                best_h = h.copy()
            found = True

    return best_h, best_c5, len(best_h)
