        pc = (1 - cc) * pc + np.sqrt(cc * (2 - cc) * mueff) * yw
        mean = project(mean + sigma * yw, n)
        norm_ratio = np.linalg.norm(ps) / chi_n
        delta = 1.0 if norm_ratio > (1.4 + 5.0 / (n + 1)) else 0.0
        if delta:
            sigma *= np.exp(cs / ds * (norm_ratio - 1))
        sigma = np.clip(sigma, 1e-6, 5.0)
        rank_mu = np.zeros((n, n))
        for i in range(mu_a):
            rank_mu += w[i] * np.outer(ys[order[i]], ys[order[i]])
        C = (1.0 + c1 * (2.0 * delta - 1.0) + cmu) * C + c1 * np.outer(pc, pc) + cmu * rank_mu
        C = (C + C.T) / 2.0
        try:
            eigvals, eigvecs = np.linalg.eigh(C)
            eigvals = np.maximum(eigvals, 1e-12)
            D = np.sqrt(eigvals)
            B = eigvecs
        except Exception:
            C = np.eye(n)
            B, D = np.eye(n), np.ones(n)
        gen += 1
    return best_h, best_c5

def local_refine(h, n, budget, rng):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = c5_eval(h)
    step = 0.15
    while time.time() - t_start < budget and step > 0.001:
        improved = False
        for _ in range(300):
            if time.time() - t_start > budget:
                break
            idx = rng.randint(n)
            delta = rng.uniform(-step, step)
            h_new = best_h.copy()
            h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
            h_new = project(h_new, n)
            c_new = c5_eval(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new
                improved = True
        if not improved:
            step *= 0.6
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1e9
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    if h0 is None or len(h0) < 4:
        h0 = np.full(42, 0.5)
    n0 = len(h0)
    h0 = project(h0, n0)
    c0 = c5_eval(h0)
    best_h = h0.copy()
    best_c5 = c0
    # Main CMA-ES on parent dimension
    budget_cma = budget_s * 0.55
    h_best, c_best = cma_optimize(h0, n0, budget_cma, rng, sigma_init=0.35)
    if c_best < best_c5:
        best_c5 = c_best
        best_h = h_best
    # Local refinement
    remaining = budget_s - (time.time() - t0) - 30
    if remaining > 10:
        hr, cr = local_refine(best_h, len(best_h), max(remaining * 0.3, 5), rng)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr
    # Try other n values
    remaining = budget_s - (time.time() - t0) - 10
    for n_try in [8, 10, 12, 16, 20, 24, 28, 32, 36, 48, 56, 64, 80]:
        if remaining < 8 or time.time() - t0 > budget_s - 10:
            break
        for attempt in range(2):
            if time.time() - t0 > budget_s - 10:
                break
            h_r = rng.uniform(0.2, 0.8, n_try)
            h_r = project(h_r, n_try)
            b_r = budget_s - (time.time() - t0) - 12
            b_r = min(b_r, 30.0)
            if b_r < 2:
                break
            hh, cc = cma_optimize(h_r, n_try, b_r, rng, sigma_init=0.3)
            if cc < best_c5:
                best_c5 = cc
                best_h = hh
                remaining = budget_s - (time.time() - t0) - 10
    # Final local refinement on best
    remaining = budget_s - (time.time() - t0) - 2
    if remaining > 3:
        hr, cr = local_refine(best_h, len(best_h), max(remaining, 1), rng)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = c5_eval(best_h)
    return best_h, final_c5, len(best_h)
