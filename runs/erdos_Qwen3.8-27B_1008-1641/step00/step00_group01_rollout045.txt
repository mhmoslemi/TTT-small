import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def c5_of(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def enforce_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h += (diff / tr) * room
        else:
            w = h.copy()
            tw = w.sum()
            if tw < 1e-15:
                break
            h += (diff / tw) * w
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-18:
        for i in range(n):
            room_i = (1.0 - h[i]) if diff > 0 else h[i]
            if room_i > 1e-15:
                adj = min(abs(diff), room_i)
                h[i] += np.sign(diff) * adj
                break
    return h

def local_search(h, n, rng, max_iter=5000):
    """Perturbation-based local search."""
    best = h.copy()
    best_c5 = c5_of(best)
    for it in range(max_iter):
        i = rng.randint(n)
        delta = rng.choice([-0.5, -0.2, -0.1, 0.1, 0.2, 0.5]) * 0.5
        trial = best.copy()
        trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
        trial = enforce_sum(trial, n)
        c5t = c5_of(trial)
        if c5t < best_c5:
            best = trial
            best_c5 = c5t
        elif rng.random() < 0.1:
            # Occasionally try bigger perturbation
            j = rng.randint(n)
            trial[j] = rng.random()
            trial = enforce_sum(trial, n)
            c5t = c5_of(trial)
            if c5t < best_c5:
                best = trial
                best_c5 = c5t
    return best, best_c5

def binary_local_search(h, n, rng, max_iter=20000):
    """Local search restricted near binary values."""
    best = h.copy()
    best_c5 = c5_of(best)
    for it in range(max_iter):
        i = rng.randint(n)
        # Push toward 0 or 1
        if best[i] > 0.5:
            trial = best.copy()
            trial[i] = 1.0
        else:
            trial = best.copy()
            trial[i] = 0.0
        trial = enforce_sum(trial, n)
        c5t = c5_of(trial)
        if c5t < best_c5:
            best = trial
            best_c5 = c5t
    return best, best_c5
