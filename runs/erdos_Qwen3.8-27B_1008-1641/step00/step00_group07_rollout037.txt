import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    """Project h onto {0<=h[i]<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        # Distribute diff among non-saturated variables
        free = (h > 0) & (h < 1)
        if np.sum(free) == 0:
            break
        adj = diff / np.sum(free)
        h[free] += adj
    return np.clip(h, 0, 1)

def compute_c5(h, n):
    """Compute C5 = max correlate * dx"""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def subgradient(h, n, k_active):
    """Compute subgradient of c_k * dx w.r.t. h for given active lag k_active.
    k_active is the index in the full correlation output."""
    dx = 2.0 / n
    grad = np.zeros(n)
    # k_active in [0, 2n-2], lag = k_active - (n-1)
    lag = k_active - (n - 1)

    # d c_k / d h[j] = (1-h[j-lag]) * [j-lag valid] - h[j+lag] * [j+lag valid]

        # h[j] appears as h[i+lag] when i = j-lag, valid if 0 <= j-lag <= n-1-lag i.e., lag <= j <= n-1
        # h[j] appears as h[i] when i = j, valid if 0 <= j <= n-1-lag i.e., j <= n-1-lag
        for j in range(n):
            g = 0.0
            if lag <= j <= n-1:  # j-lag is valid i
                g += (1.0 - h[j-lag])
            if j <= n-1-lag:  # j is valid i
                g -= h[j+lag]
            grad[j] = g

import math

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, int(np.argmax(corr))

def compute_c5_only(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def subgrad_for_lag(h, n, lag):
    """Subgradient of c_lag * dx with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:])
        grad[l:] -= h[:n - l]
    return grad * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    for _ in range(500):
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = free.sum()
        if cnt == 0:
            break
        h[free] += diff / cnt
    return h

def top_k_subgradient(h, n, k=5):
    """Weighted subgradient using top-k active lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k indices
    top_idx = np.argsort(corr)[-k:][::-1]
    top_vals = corr[top_idx]
    max_val = top_vals[0]
    # Weights: exponential decay from max
    weights = np.exp(-0.5 * (max_val - top_vals) / (max_val + 1e-10))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        lag = idx - (n - 1)
        g = subgrad_for_lag(h, n, lag)
        grad += w * g
    return grad

    return best_h, best_c5

                            'maxls': 60, 'disp': False})
    h_out = project_feasible(res.x, n)
    return h_out, eval_c5(h_out, n)

def subgrad_descent(h0, n, deadline, n_iter=20000):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = eval_c5(h0, n)
    vx = np.zeros(n)
    rho = 0.9
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c5, k_act = eval_c5(h, n), int(np.argmax(np.correlate(h, 1.0 - h, mode='full')))
        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            best_h = h.copy()
        g = subgrad_for_lag(h, n, k_act - (n - 1))
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        step = 0.02 * n / (np.sqrt(it + 10.0))
        vx = rho * vx - step * g / g_norm
        h_try = project_feasible(h + vx, n)
        c5_try = eval_c5(h_try, n)
        if c5_try <= c5:
            h = h_try
        else:
            vx *= 0.5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_start = np.array(initial_h_values, dtype=float)
        n = len(h_start)
    except NameError:
        n = 80
        h_start = np.full(n, 0.5)
    h_start = project_feasible(h_start, n)
    best_h = h_start.copy()
    best_c5 = eval_c5(h_start, n)
    
    # Phase 1: Continuation L-BFGS-B from initial
    phases = [
        (10.0, 10.0, 500),
        (50.0, 25.0, 500),
        (200.0, 60.0, 500),
        (800.0, 120.0, 800),
        (3000.0, 250.0, 800),
    ]
    for mu, lam, maxit in phases:
        if time.time() > deadline - 60:
            break
        try:
            h_new, c5_new = lbfgs_opt(best_h, n, deadline, mu, lam, maxit)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
    
    # Phase 2: Subgradient descent refinement
    if time.time() < deadline - 60:
        try:
            h_sg, c5_sg = subgrad_descent(best_h, n, deadline, n_iter=15000)
            if c5_sg < best_c5:
                best_h, best_c5 = h_sg, c5_sg
        except Exception:
            pass
    
    # Phase 3: Structured and random restarts
    starts = []
    for freq in [3, 5, 7, 11, 17, 23]:
        h_s = 0.5 + 0.35 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        starts.append(project_feasible(h_s, n))
    for _ in range(2):
        starts.append(project_feasible(rng.uniform(0.0, 1.0, n), n))
    
    for h_s in starts:
        if time.time() > deadline - 30:
            break
        try:
            h_new, c5_new = lbfgs_opt(h_s, n, deadline, 200.0, 60.0, 400)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
    
    # Phase 4: Final high-precision polish
    if time.time() < deadline - 15:
        try:
            h_new, c5_new = lbfgs_opt(best_h, n, deadline, 5000.0, 300.0, 1500)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
    
    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, n)
    return best_h, best_c5, n
