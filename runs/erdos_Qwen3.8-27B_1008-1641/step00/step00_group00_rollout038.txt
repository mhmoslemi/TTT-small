    x = x_in.astype(np.int32).copy()
    if x.sum() != n//2:

    raw = int(np.max(r)); score = score_from_r(r)
    best_x = x.copy(); best_raw = raw; best_score = score
    while time.time() < t_end:
        phase_end = min(t_end - 1e-6, time.time() + greedy_block)
        # greedy sweeps until no improvement or phase_end
        while time.time() < phase_end:
            ones = np.flatnonzero(x).tolist()

                if x[a] != 1: continue

def project_feasible(h, target=None):
    a=np.asarray(h,dtype=float).copy()
    a=np.clip(a,0.0,1.0)
    n=len(a)
    if target is None: target=0.5*n
    s=float(np.sum(a)); diff=target-s
    if abs(diff)>1e-12:
        if diff>0:
            order=np.argsort(-a)
            rem=diff
            for i in range(n):
                idx=order[i]
                if rem<=1e-15: break
                add=min(rem,1.0-a[idx])
                a[idx]+=add; rem-=add
        else:
            order=np.argsort(a)
            rem=-diff
            for i in range(n):
                idx=order[i]
                if rem<=1e-15: break
                sub=min(rem,a[idx])
                a[idx]-=sub; rem-=sub
    s=float(np.sum(a)); d=target-s
    if abs(d)>1e-14:
        if d>0:
            for idx in range(n):
                if a[idx] < 1.0 - 1e-14:
                    a[idx]=min(1.0,a[idx]+d); break
        else:
            for idx in range(n):
                if a[idx] > 1e-14:
                    a[idx]=max(0.0,a[idx]+d); break

def c5_of(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, "full")) * (2.0 / n))


def project_feasible(h, t=None):
    a = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    n = len(a)
    if t is None:
        t = 0.5 * n
    d = t - float(np.sum(a))
    if abs(d) > 1e-12:
        if d > 0:
            for i in np.argsort(-a):
                if d <= 1e-15:
                    break
                add = min(d, 1.0 - a[i])
                a[i] += add
                d -= add
        else:
            for i in np.argsort(a):
                if d >= -1e-15:
                    break
                sub = min(-d, a[i])
                a[i] -= sub
                d += sub
    for _ in range(4):
        d = t - float(np.sum(a))
        if abs(d) <= 1e-13:
            break
        if d > 0:
            for i in range(n):
                if a[i] < 1.0 - 1e-14:
                    add = min(d, 1.0 - a[i])
                    a[i] += add
                    break
        else:
            for i in range(n):
                if a[i] > 1e-14:
                    sub = min(-d, a[i])
                    a[i] -= sub
                    break
    return np.clip(a, 0.0, 1.0)


def upsample_to(parent, n):
    p = np.asarray(parent, dtype=float)
    m = len(p)
    if m == n:
        return p.copy()
    if n % m == 0:
        return np.repeat(p, n // m)
    x0 = np.arange(m, dtype=float)
    x1 = np.linspace(0.0, m - 1.0, n)
    return np.interp(x1, x0, p)


def make_binary(h, n, rng):
    h = project_feasible(h, 0.5 * n)
    x = np.zeros(n, dtype=np.int32)
    k = n // 2
    if 0 < k < n:
        x[np.argsort(h + rng.normal(0.0, 1e-8, n))[-k:]] = 1
    return x


def score_from_r(r):
    m = int(np.max(r))
    L = len(r)
    high = float(np.count_nonzero(r >= m - 1))
    rr = r.astype(np.int64)
    return float(m) + 0.02 * (high / float(L)) + 1e-5 * float(np.sum(rr * rr)) / float(L)


def binary_search(x, n, rng, t_end, gtime=30.0, stime=20.0):
    if x.sum() != n // 2:
        x = make_binary(x, n, rng)
    else:
        x = x.astype(np.int32).copy()
    gx = 1 - x
    r = np.correlate(x, gx, "full")
    m = int(np.max(r))
    sc = score_from_r(r)
    xb = x.copy()
    bm, bs = m, sc

    while time.time() < t_end:
        pe = min(t_end, time.time() + gtime)
        while time.time() < pe:
            ones = np.flatnonzero(x).tolist()
            zeros = np.flatnonzero(gx).tolist()
            acc = False
            for a in ones:
                if x[a] != 1:
                    continue
                for b in zeros:
                    if a == b or x[a] != 1 or x[b] != 0:
                        continue
                    x[a] = 0; gx[a] = 1; x[b] = 1; gx[b] = 0
                    r = np.correlate(x, gx, "full")
                    nm = int(np.max(r))
                    ns = score_from_r(r)
                    if ns < sc - 1e-12:
                        sc = ns; m = nm; acc = True
                        if nm < bm or (nm == bm and ns < bs):
                            bm = nm; bs = ns; xb = x.copy()
                    else:
                        x[a] = 1; gx[a] = 0; x[b] = 0; gx[b] = 1
            if not acc:
                break
            if time.time() >= pe:
                break

        if time.time() >= t_end:
            break
        se = min(t_end, time.time() + stime)
        se0 = time.time()
        dur = max(1.0, se - se0)
        while time.time() < se:
            T = 0.01 + 1.0 * max(0.0, (se - time.time()) / dur)
            a = int(rng.integers(n))
            while x[a] == 0:
                a = int(rng.integers(n))
            b = int(rng.integers(n))
            while x[b] == 1:
                b = int(rng.integers(n))
            if a == b:
                continue
            x[a] = 0; gx[a] = 1; x[b] = 1; gx[b] = 0
            r = np.correlate(x, gx, "full")
            nm = int(np.max(r))
            ns = score_from_r(r)
            ok = ns < sc or (ns > sc and rng.random() < math.exp(-min(50.0, (ns - sc) / T)))
            if ok:
                sc = ns; m = nm
                if nm < bm or (nm == bm and ns < bs):
                    bm = nm; bs = ns; xb = x.copy()
            else:
                x[a] = 1; gx[a] = 0; x[b] = 0; gx[b] = 1
    return xb, bm


def cont_smooth_penalty(y):
    n = _CONT_N
    h = np.clip(y, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, "full") * (2.0 / n)
    mx = float(np.max(c))
    b = _SMOOT_BETA
    s = float(np.sum(h) - 0.5 * n)
    return mx + float(np.log(np.sum(np.exp(b * (c - mx)))) / b + _CONT_MU * s * s


def cont_objective(y):
    n = _CONT_N
    h = np.clip(y, 0.0, 1.0)
    return c5_of(h)


def cont_sum_constraint(y):
    return float(np.sum(y) - 0.5 * _CONT_N)


def optimize_continuous(h0, n, t_end, rng):
    global _CONT_N, _CONT_MU
    _CONT_N = n
    best = project_feasible(np.asarray(h0, dtype=float).copy(), 0.5 * n)
    cb = c5_of(best)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": cont_sum_constraint}]
    att = 0
    no = 0
    while time.time() < t_end - 3.0:
        if att == 0:
            x0 = best.copy()
        else:
            x0 = np.clip(best + rng.normal(0.0, 0.02 + 0.005 * (att % 5), n), 0.0, 1.0)
        x0 = project_feasible(x0, 0.5 * n)
        mi = 12 if n > 100 else (20 if n > 80 else 40)
        old = cb
        if time.time() < t_end - 3.0:
            _CONT_MU = 200.0
            try:
                res = so.minimize(cont_smooth_penalty, x0, method="L-BFGS-B",
                                  bounds=bounds, options={"maxiter": mi, "ftol": 1e-12})
                if res is not None and res.x is not None:
                    h1 = project_feasible(res.x, 0.5 * n)
                    c1 = c5_of(h1)
                    if c1 < cb - 1e-12:
                        cb = c1
                        best = h1.copy()
            except Exception:
                pass
        if time.time() < t_end - 3.0:
            starts = (best,) if att % 2 == 0 else (best, x0)
            for st in starts:
                if time.time() >= t_end - 3.0:
                    break
                try:
                    res = so.minimize(cont_objective, st, method="SLSQP",
                                      bounds=bounds, constraints=cons,
                                      options={"maxiter": mi, "ftol": 1e-12})
                    if res is not None and res.x is not None:
                        h2 = project_feasible(res.x, 0.5 * n)
                        c2 = c5_of(h2)
                        if c2 < cb - 1e-12:
                            cb = c2
                            best = h2.copy()
                except Exception:
                    pass
        att += 1
        if att > 1 and cb >= old - 1e-11:
            no += 1
        else:
            no = 0
        if no > 6:
            break
    return best, cb


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + float(budget_s)
    rng = np.random.default_rng(seed)

    parent = globals().get("initial_h_values")
    best_h = None
    best_c = float("inf")
    best_n = 0
    parent_c0 = float("inf")

    if parent is not None and len(parent) > 0:
        p = project_feasible(np.asarray(parent, dtype=float), 0.5 * len(parent))
        best_h = p.copy()
        best_c = c5_of(best_h)
        best_n = len(best_h)
        parent_c0 = best_c
    else:
        n = 90
        x = make_binary(np.full(n, 0.5), n, rng)
        best_h = x.astype(float)
        best_c = c5_of(best_h)
        best_n = n
        parent_c0 = best_c

    if float(budget_s) < 25:
        best_h = project_feasible(best_h, 0.5 * best_n)
        return best_h, float(c5_of(best_h)), int(best_n)

    if best_n <= 60 and dl - time.time() > 80:
        te = min(dl - max(120.0, 0.55 * float(budget_s)),
                 t0 + min(45.0, 0.25 * float(budget_s)))
        if te > time.time() + 8.0:
            h2, c2 = optimize_continuous(best_h, best_n, te, rng)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
                best_n = len(h2)

    ns = set()
    if best_n % 2 == 0 and 20 <= best_n <= 200:
        ns.add(best_n)
    ns.add(46)
    ns.add(90)
    if float(budget_s) > 600:
        ns.add(120)
    ns = sorted(ns)

    reserve = max(30.0, 0.20 * float(budget_s))
    proc = dl - reserve

    for N in ns:
        if time.time() >= proc - 5.0:
            break
        src = parent if (parent is not None and best_c >= parent_c0 - 1e-12) else best_h
        if src is None or len(src) == 0:
            src = best_h
        h0 = upsample_to(src, N)

        rem = max(0.0, proc - time.time())
        if N <= 120 and rem > 25.0:
            x0 = make_binary(h0, N, rng)
            chunk = min(70.0 if N <= 60 else 100.0, rem * 0.40)
            if chunk > 8.0:
                xb, raw = binary_search(x0, N, rng, time.time() + chunk)
                c = raw * (2.0 / N)
                if c < best_c:
                    best_h = xb.astype(float)
                    best_c = c
                    best_n = N
                    h0 = xb.astype(float)

        if time.time() < proc - 15.0:
            chunk = min(90.0 if N <= 90 else 120.0, max(20.0, (proc - time.time()) * 0.55))
            if chunk > 8.0:
                h2, c2 = optimize_continuous(h0, N, time.time() + chunk, rng)
                if c2 < best_c:
                    best_h = h2.copy()
                    best_c = c2
                    best_n = N

        if time.time() < proc - 15.0 and float(budget_s) > 400:
            xr = make_binary(np.full(N, 0.5), N, rng).astype(float)
            hr = project_feasible(xr + rng.normal(0.0, 0.03, N), 0.5 * N)
            if time.time() < proc - 10.0:
                h2, c2 = optimize_continuous(hr, N, min(proc, time.time() + 60.0), rng)
                if c2 < best_c:
                    best_h = h2.copy()
                    best_c = c2
                    best_n = N

    if best_h is not None and time.time() < dl - 2.0:
        n = best_n
        r = np.round(best_h)
        if n % 2 == 0 and np.sum(r) == n // 2 and np.max(np.abs(best_h - r)) < 1e-7:
            best_h = r.astype(float)
        if n % 2 == 0 and len(np.unique(best_h)) <= 2 and np.max(np.abs(best_h - np.round(best_h))) < 1e-7:
            xb, raw = binary_search(best_h.astype(np.int32), n, rng, dl)
            c = raw * (2.0 / n)
            if c < best_c:
                best_h = xb.astype(float)
                best_c = c
                best_n = n
        else:
            h2, c2 = optimize_continuous(best_h, n, dl, rng)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
                best_n = len(h2)

    best_h = project_feasible(best_h, 0.5 * best_n)
    return best_h, float(c5_of(best_h)), int(best_n)
