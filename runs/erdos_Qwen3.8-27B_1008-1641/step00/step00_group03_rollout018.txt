import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_gradient_component(h, k, n):
    """Gradient of correlate(h, 1-h, 'full')[k] w.r.t. h"""
    # k is the index in the full correlation output
    # offset = k - (n-1)
    offset = k - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        # h[i] * (1 - h[i + offset])
        j = i + offset
        if 0 <= j < n:
            grad[i] += (1.0 - h[j])
            grad[j] += (-h[i])
    return grad

def logsumexp_c5(h, alpha, n):
    """Smooth approximation of max of correlate values * dx"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (1.0/alpha) * logsumexp(alpha * corr)

def logsumexp_c5_grad(h, alpha, n):
    """Gradient of logsumexp approximation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = np.exp(alpha * (corr - np.max(corr)))
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] > 1e-10:
            grad += weights[k] * c5_gradient_component(h, k, n)
    return grad / alpha  # chain rule for (1/alpha) * logsumexp

def optimize_h(h_init, n, alpha, max_iter=500, tol=1e-8):
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, dx)

    def objective_grad(h):

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2, 'jac': lambda h: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(objective, h_init, jac=objective_grad,
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': tol})
        if res.success or res.fun < best_c5:

            h_new = np.clip(h_new, 0, 1)

from scipy.special import logsumexp
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_c5_grad(h, n):
    """Subgradient of max_k correlate(h,1-h,'full')[k]"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)

                grad[j] += (-h[i])

    max_c = np.max(corr)

    w = exp_vals / np.sum(exp_vals)

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_objective(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((1.0 / alpha) * logsumexp(alpha * corr))

def smooth_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    ev = np.exp(alpha * (corr - mc))
    w = ev / ev.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            n_terms = n - d
            if n_terms <= 0:
                continue
            i1 = np.arange(n_terms)
            i2 = i1 + d
            grad[i1] += w[k] * (1.0 - h[i2])
            grad[i2] -= w[k] * h[i1]
        else:
            q = -d
            n_terms = n - q
            if n_terms <= 0:
                continue
            i1 = np.arange(q, n)
            i2 = np.arange(n_terms)
            grad[i1] += w[k] * (1.0 - h[i2])
            grad[i2] -= w[k] * h[i1]
    return grad / alpha

def subgradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = np.max(corr)
    active = np.where(np.abs(corr - mv) < 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        if d >= 0:
            n_terms = n - d
            if n_terms <= 0:
                continue
            i1 = np.arange(n_terms)
            i2 = i1 + d
            grad[i1] += (1.0 - h[i2])
            grad[i2] -= h[i1]
        else:
            q = -d
            n_terms = n - q
            if n_terms <= 0:
                continue
            i1 = np.arange(q, n)
            i2 = np.arange(n_terms)
            grad[i1] += (1.0 - h[i2])
            grad[i2] -= h[i1]
    return grad / len(active)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 0:
        idx = np.argsort(h)[::-1] if diff > 0 else np.argsort(h)
        for i in idx:
            step = min(abs(diff), h[i]) if diff > 0 else min(abs(diff), 1.0 - h[i])
            h[i] -= step if diff > 0 else (h[i] += step)
            diff += step if diff > 0 else -step
            if abs(diff) < 1e-14:
                break
    return h

def slsqp_optimize(h0, n, alpha, maxiter=200):
    def obj(h):
        return smooth_objective(h, alpha)
    def g(h):
        return smooth_gradient(h, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def sg_polish(h0, n, lr=0.01, n_iter=5000):
    h = h0.copy()
    best = h.copy()
    bc = compute_c5(h, n)
    for i in range(n_iter):
        g = subgradient(h)
        h = project_feasible(h - lr * g, n)
        c = compute_c5(h, n)
        if c < bc:
            bc = c
            best = h.copy()
        lr *= 0.9995
        if lr < 1e-10:
            break
    return best, bc

def gen_random(n, seed):

    return project_feasible(rng.uniform(0, 1, n), n)

Now let me also fix the project_feasible function. The issue is ensuring sum(h) = n/2 exactly (or very close).

Let me use a simpler projection:
1. Clip to [0,1]
2. Compute diff = n/2 - sum(h)
3. If diff > 0: need to increase some values. Add diff/n to all, clip, repeat.
4. If diff < 0: need to decrease some values. Subtract |diff|/n from all, clip, repeat.
5. Final: use a more careful redistribution for the remaining error.

Actually, a cleaner approach: use the bisection method to find threshold τ such that sum(clip(h - τ, 0, 1)) = n/2. But that's the projection onto the simplex with box constraints, which is what we want.

Let me implement it properly:

```python
def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    # Bisection for tau: sum(clip(h - tau, 0, 1)) = target
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    result = np.clip(h - tau, 0.0, 1.0)
    # Fix any remaining error
    diff = target - result.sum()
    if abs(diff) > 1e-14:
        free = (result > 0) & (result < 1)
        if free.any():
            result[free] += diff / free.sum()
            result = np.clip(result, 0.0, 1.0)
    return result
```

This is the Euclidean projection onto the box-simplex, which is the correct projection for our constraint set {h: 0≤h≤1, sum(h)=n/2}.

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_objective(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((1.0 / alpha) * logsumexp(alpha * corr))

def smooth_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    ev = np.exp(alpha * (corr - mc))
    w = ev / ev.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            nt = n - d
            if nt <= 0:
                continue
            i1 = np.arange(nt)
            i2 = i1 + d
            grad[i1] += w[k] * (1.0 - h[i2])
            grad[i2] -= w[k] * h[i1]
        else:
            q = -d
            nt = n - q
            if nt <= 0:
                continue
            i1 = np.arange(q, n)
            i2 = np.arange(nt)
            grad[i1] += w[k] * (1.0 - h[i2])
            grad[i2] -= w[k] * h[i1]
    return grad / alpha

def subgradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = corr.max()
    active = np.where(np.abs(corr - mv) < 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        if d >= 0:
            nt = n - d
            if nt <= 0:
                continue
            i1 = np.arange(nt)
            i2 = i1 + d
            grad[i1] += (1.0 - h[i2])
            grad[i2] -= h[i1]
        else:
            q = -d
            nt = n - q
            if nt <= 0:
                continue
            i1 = np.arange(q, n)
            i2 = np.arange(nt)
            grad[i1] += (1.0 - h[i2])
            grad[i2] -= h[i1]
    return grad / len(active)

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    res = np.clip(h - tau, 0.0, 1.0)
    diff = target - res.sum()
    if abs(diff) > 1e-13:
        free = (res > 1e-12) & (res < 1.0 - 1e-12)
        if free.any():
            res[free] += diff / free.sum()
            res = np.clip(res, 0.0, 1.0)
    return res

def slsqp_opt(h0, n, alpha, maxiter=200):
    def obj(x):
        return smooth_objective(x, alpha)
    def g(x):
        return smooth_gradient(x, alpha)
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h0, jac=g, method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        if r.x is not None:
            ho = project_feasible(r.x, n)
            return ho, compute_c5(ho, n)
    except Exception:
        pass
    return h0, compute_c5(h0, n)

def sg_polish(h0, n, lr=0.005, niter=5000):
    h = h0.copy()
    best = h.copy()
    bc = compute_c5(h, n)
    for i in range(niter):
        g = subgradient(h)
        h = project_feasible(h - lr * g, n)
        c = compute_c5(h, n)
        if c < bc:
            bc = c
            best = h.copy()
        lr *= 0.9997
        if lr < 1e-11:
            break
    return best, bc

def gen_random(n, seed):
    rng = np.random.RandomState(seed)
    return project_feasible(rng.uniform(0, 1, n), n)

def gen_cosine(n, seed, freq=1):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2.0 * np.pi * freq, n)
    ph = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.4 * np.cos(x + ph)
    return project_feasible(h, n)

def gen_bump(n, seed, nb=2):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for _ in range(nb):
        c = rng.uniform(0.3, 1.7)
        w = rng.uniform(0.15, 0.55)
        h += np.exp(-((np.arange(n) * 2.0 / n - c) ** 2) / (2 * w * w))
    h = project_feasible(h, h, )
    return project_feasible(h, n)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)
