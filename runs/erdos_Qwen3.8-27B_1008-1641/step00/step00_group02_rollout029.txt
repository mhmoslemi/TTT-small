        np.clip(h, 0.0, 1.0, out=h)

        nf = np.sum(free)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

import numpy as np
import math
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, np.argmax(corr) - (n - 1)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = np.sum(h) - target
        if abs(diff) < 1e-13:
            break
        h = h - diff / n
        np.clip(h, 0.0, 1.0, out=h)
    res = target - np.sum(h)
    if abs(res) > 1e-13:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf > 0:
            h[free] += res / nf
            np.clip(h, 0.0, 1.0, out=h)
    return h


def grad_single_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    kk = abs(lag)
    g = np.full(n, dx)
    if kk < n:
        g[:n - kk] -= dx * h[kk:]
        g[kk:] -= dx * h[:n - kk]
    return g


def soft_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / np.sum(e)
    c5_val = m + np.log(np.sum(e)) / beta
    active = np.where(w > 1e-8)[0]
    grad = np.zeros(n)
    for k_idx in active:
        lag = k_idx - (n - 1)
        grad += w[k_idx] * grad_single_lag(h, lag)
    grad -= np.mean(grad)
    return c5_val, grad


def optimize_local(h, n, time_limit, lr_init=0.05, beta=300):
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    c5, _ = compute_c5(h)
    while time.time() < time_limit and lr > 1e-8:

            h = h_new
            c5 = c5_new

            lr = min(lr * 1.02, 1.0)
