    starts.append(proj(hr, n, ts))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.92
    rng = np.random.default_rng(seed)

    try:
        hi = np.asarray(initial_h_values, dtype=float)
        n_init = len(hi)
    except Exception:
        hi = None
        n_init = 42

    best_h = hi.copy() if hi is not None else np.ones(n_init) * 0.5
    best_c5 = eval_c5(best_h)

    n_list = sorted(set([n_init, 48, 56, 64, 80, 96, 120]))
    n_list = [x for x in n_list if 20 <= x <= 200]

    for n in n_list:
        if time.time() > deadline:
            break
        dx = 2.0 / n
        ts = n / 2.0

        # Interpolate best to this n for a warm start
        if n != n_init:
            h_interp = np.interp(np.linspace(0, 2, n, endpoint=False),
                                 np.linspace(0, 2, len(best_h), endpoint=False), best_h)
            starts = gen_starts(n, ts, rng, None, n_init)
            starts.insert(0, proj(h_interp, n, ts))
        else:
            starts = gen_starts(n, ts, rng, hi, n_init)

        # Score all starts
        scored = []
        for hs in starts:
            hs = proj(hs, n, ts)
            scored.append((eval_c5(hs), hs))
        scored.sort(key=lambda x: x[0])

        # Optimize top candidates
        top_k = 5
        for c5_init, h0 in scored[:top_k]:
            if time.time() > deadline:
                break
            h_opt, c5_opt = slsqp_opt(h0, n, dx, ts, deadline)
            if time.time() > deadline:
                break
            h_ls, c5_ls = local_search(h_opt, n, ts, rng, deadline)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Final refinement on best solution
    if time.time() < deadline:
        n = len(best_h)
        dx = 2.0 / n
        ts = n / 2.0
        deadline2 = t0 + budget_s * 0.95
        best_h, best_c5 = slsqp_opt(best_h, n, dx, ts, deadline2)
        best_h, best_c5 = local_search(best_h, n, ts, rng, deadline2, max_it=600)

    return (best_h, best_c5, len(best_h))
