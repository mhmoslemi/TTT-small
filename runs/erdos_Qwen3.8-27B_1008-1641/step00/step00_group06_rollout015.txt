    return mx + T * np.log(np.sum(np.exp((vals - mx) / T)) / len(vals))

        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_all_shifts(h):
    n = len(h)
    n_corr = 2 * n - 1
    grads = np.zeros((n_corr, n))
    for idx in range(n_corr):
        s = idx - (n - 1)
        g = np.ones(n)

            mp = m + s
            if 0 <= ms < n:
                g[m] -= h[ms]
            if 0 <= mp < n:
                g[m] -= h[mp]
        grads[idx] = g
    return grads
