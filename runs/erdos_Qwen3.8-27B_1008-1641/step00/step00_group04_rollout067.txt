def project(h, n):
    """Project h to satisfy 0<=h[i]<=1 and sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase sum
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
        else:
            # Need to decrease sum
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

import numpy as np
import time

def project_h(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    return h

def c5_of(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def _initial_h():
    if 'initial_h_values' in globals() and initial_h_values is not None:
        return np.array(initial_h_values, dtype=float)
    h = np.zeros(42)
    h[::2] = 1.0
    return h

def cmaes_step(xmean, C, pc, ps, sigma, N, lam, mu, w, mueff, cc, c1, cmu, cs, ds, hn, rng):
    jitter = 1e-14
    try:
        L = np.linalg.cholesky(C + jitter * np.eye(N))
    except np.linalg.LinAlgError:
        ev, evc = np.linalg.eigh(C)
        C = evc @ np.diag(np.maximum(ev, 1e-12)) @ evc.T
        C = (C + C.T) / 2
        L = np.linalg.cholesky(C + jitter * np.eye(N))

    Z = rng.standard_normal((lam, N))
    Y = Z @ L.T
    pops = np.empty((lam, N))
    for i in range(lam):
        raw = xmean + sigma * Y[i]
        pops[i] = project_h(raw, N)
    vals = np.empty(lam)
    for i in range(lam):
        vals[i] = c5_of(pops[i])

    order = np.argsort(vals)
    pops = pops[order]
    vals = vals[order]

    m = np.sum(w[:mu, None] * pops[:mu], axis=0)
    ysig = (m - xmean) / sigma
    nysig = np.linalg.norm(ysig)
    if nysig > 1e-14:
        d = ysig / nysig
    else:
        d = np.zeros(N)

    hlam = 1.0 if nysig < (1 + 2 * max(0, np.sqrt((mueff - 1) / (N + 1)) - 1)) * hn else 0.0

    pc = (1 - cc) * pc + hlam * np.sqrt(cc * (2 - cc) * mueff) * d
    ps = (1 - cs) * ps + hlam * np.sqrt(cs * (2 - cs) * mueff) * d

    delta_hlam = c1 * (1 - hlam) * cc * (2 - cc)
    C = (1 - c1 - cmu + delta_hlam) * C + c1 * np.outer(pc, pc) + delta_hlam * np.eye(N)
    for i in range(mu):
        yi = (pops[i] - xmean) / sigma
        C += cmu * w[i] * np.outer(yi, yi)
    C = (C + C.T) / 2

    xmean_new = m
    sigma *= np.exp((cs / ds) * (np.linalg.norm(ps) / hn - 1))
    sigma = np.clip(sigma, 1e-8, 10.0)

    return xmean_new, C, pc, ps, sigma, pops, vals, hlam

def run(seed=42, budget_s=1100, **kw):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    time_limit = budget_s - 3

    h0 = _initial_h()
    N = len(h0)
    h_best = project_h(h0, N).copy()
    c5_best = c5_of(h_best)

    # CMA-ES parameters
    lam = max(4 + int(np.ceil(3 * np.log(N))), 30)
    mu = max(lam // 2, 5)
    w = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    w = w / w.sum()
    mueff = 1.0 / np.sum(w ** 2)
    cc = (4 + mueff / N) / (N + 4 + 2 * mueff / N)
    c1 = 2.0 / ((N + 1.3) ** 2 + mueff)
    cmu = min(1 - c1, 2 * (mueff - 2 + 1 / mueff) / ((N + 2) ** 2 + mueff))
    cs = 5.0 / (N + 5) ** 2
    ds = 1 + 2 * max(0, np.sqrt((mueff - 1) / (N + 1)) - 1) + cs
    hn = np.sqrt(N) * (1 - 1 / (4 * N) + 1 / (21 * N ** 2))

    xmean = h_best.copy()
    C = np.eye(N)
    pc = np.zeros(N)
    ps = np.zeros(N)
    sigma = 0.25

    # Phase 1: Global CMA-ES search
    while time.time() - t0 < time_limit - 60:
        xmean, C, pc, ps, sigma, pops, vals, hlam = cmaes_step(
            xmean, C, pc, ps, sigma, N, lam, mu, w, mueff, cc, c1, cmu, cs, ds, hn, rng
        )
        if vals[0] < c5_best:
            c5_best = vals[0]
            h_best = pops[0].copy()
        # Periodic sigma reset if stuck
        if sigma < 0.001:
            sigma = 0.15
            C = np.eye(N)
            pc = np.zeros(N)
            ps = np.zeros(N)
            xmean = h_best.copy()

    # Phase 2: Local refinement with multi-scale perturbations
    scales = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for s in scales:
        if time.time() - t0 > time_limit:
            break
        for it in range(2000):
            if time.time() - t0 > time_limit:
                break
            z = rng.standard_normal(N)
            h_try = project_h(h_best + s * z, N)
            v = c5_of(h_try)
            if v < c5_best:
                c5_best = v
                h_best = h_try.copy()

    # Phase 3: Coordinate descent polish
    if time.time() - t0 < time_limit:
        for sweep in range(200):
            if time.time() - t0 > time_limit:
                break
            improved = False
            for i in range(N):
                if time.time() - t0 > time_limit:
                    break
                for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]:
                    h_try = h_best.copy()
                    h_try[i] += delta
                    h_try = project_h(h_try, N)
                    v = c5_of(h_try)
                    if v < c5_best:
                        c5_best = v
                        h_best = h_try.copy()
                        improved = True
                        break
            if not improved and sweep > 20:
                break

    h_best = project_h(h_best, N)
    c5_best = c5_of(h_best)
    return h_best, c5_best, N
