from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(x, target_sum):
    h = np.clip(x, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)

def make_bounds(n):
    return [(0.0, 1.0)] * n

    def cons_fun(x):
        return x.sum() - target_sum
    def cons_jac(x):
        return np.ones_like(x)
    return [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
