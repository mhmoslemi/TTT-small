import numpy as np
import time
from scipy.optimize import minimize, differential_evolution

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(h)
    else:
        order = np.argsort(-h)
    rem = abs(diff)
    for i in order:
        if rem <= 1e-14:
            break
        if diff > 0:
            a = min(rem, 1.0 - h[i])
            h[i] += a
        else:
            a = min(rem, h[i])
            h[i] -= a
        rem -= a
    return h

def _soft_c5_high(x):
    return _soft_c5(x, 80.0)

def _soft_c5_low(x):
    return _soft_c5(x, 50.0)

def _soft_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    return float(m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha)

def _sum_con(x):
    return np.sum(x) - len(x) / 2.0

def _de_obj(x):
    x = project_feasible(x)
    return compute_C5(x)

def resample_to(h, n_tgt):
    n_src = len(h)
    dw = 2.0 / n_src
    x_new = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    idx = np.clip(np.floor(x_new / dw).astype(int), 0, n_src - 1)
    return project_feasible(h[idx])

def local_search(h, rng, n_iter=300):
    h = h.copy()
    best = compute_C5(h)
    n = len(h)
    step = 0.1
    for _ in range(n_iter):
        n_p = max(1, n // 10)
        idx = rng.choice(n, n_p, replace=False)
        h2 = h.copy()
        h2[idx] += rng.uniform(-step, step, n_p)
        h2 = project_feasible(h2)
        s = compute_C5(h2)
        if s < best - 1e-13:
            h, best = h2, s
        else:
            step *= 0.97
        if step < 0.002:
            step = 0.08
    return h, best

def slsqp_opt(h0, n):
    mx = 400 if n <= 90 else (200 if n <= 200 else 100)
    fm = _soft_c5_high if n <= 100 else _soft_c5_low
    try:
        r = minimize(fm, h0.copy(), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _sum_con}],
                     options={'maxiter': mx, 'ftol': 1e-12})
        hx = project_feasible(np.array(r.x, dtype=float))
        return hx, compute_C5(hx)
    except Exception:
        return h0.copy(), compute_C5(h0)

    rng = np.random.default_rng(seed)

    bs = compute_C5(bh)
