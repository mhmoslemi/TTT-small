def set_opts(target, beta):

def fix_sum_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    r = float(target - h.sum())
    if abs(r) > 1e-14:
        if r > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                if r <= 1e-15:
                    break
                add = min(1.0 - h[idx], r)
                if add > 0:
                    h[idx] += add
                    r -= add
        else:
            r = -r
            order = np.argsort(h)
            for idx in order:
                if r <= 1e-15:
                    break
                sub = min(h[idx], r)
                if sub > 0:
                    h[idx] -= sub
                    r -= sub
    return h


def project_sum_box(y, target):
    y = np.asarray(y, dtype=float).ravel()
    n = y.size
    if n == 0:
        return y
    if not np.all(np.isfinite(y)):
        y = np.where(np.isfinite(y), y, 0.5)
    target = float(np.clip(target, 0.0, float(n)))
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    return fix_sum_exact(p, target)


def make_feasible(v, n, target):
    arr = np.asarray(v, dtype=float).ravel()
    if arr.size != n:
        arr = upsample_arr(arr, n)
    arr = np.clip(arr, 0.0, 1.0)
    if not np.all(np.isfinite(arr)):

    return fix_sum_exact(project_sum_box(arr, target), target)


def upsample_arr(arr, n):
    arr = np.asarray(arr, dtype=float).ravel()
    m = arr.size
    if m == 0:
        return np.full(n, 0.5)
    if m == n:
        return arr.copy()
    idx = (np.arange(n) * m) // n
    return arr[idx]


def upsample_smooth(arr, n):
    arr = np.asarray(arr, dtype=float).ravel()
    m = arr.size
    if m == 0:
        return np.full(n, 0.5)
    if m == n:
        return arr.copy()
    if m == 1:
        return np.full(n, float(arr[0]))
    xo = (np.arange(m) + 0.5) / float(m)
    xn = (np.arange(n) + 0.5) / float(n)
    return np.interp(xn, xo, arr)


def random_smooth_start(n, seed):

def compute_c5(h, n):

    if n == 0 or h.size == 0:
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")

def lse_value(h, beta):
    h = np.asarray(h, dtype=float).ravel()

    corr = np.correlate(h, 1.0 - h, mode="full")

    e = np.exp((corr - m) / beta)
    s = float(e.sum())
    lse = m + beta * np.log(s) if s > 0 else m
    return lse, m

    if x.size == 0 or not np.all(np.isfinite(x)):
        return 1e100
    v, _ = lse_value(x, _OPT_BETA)
    if not np.isfinite(v):
        return 1e100
    return v


def gradient_slsqp(x):
    x = np.asarray(x, dtype=float).ravel()
    if x.size == 0 or not np.all(np.isfinite(x)):
        return np.zeros(x.size, dtype=float)
    return lse_grad(x, _OPT_BETA)


def refine_slsqp(h0, n, target, t_limit):
    h = make_feasible(h0, n, target)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": constraint_sum, "jac": constraint_sum_jac}]
    maxiter = 150 if n <= 80 else (90 if n <= 120 else 60)
    betas = [0.1, 0.04, 0.015, 0.006, 0.003]
    set_opts(target, betas[0])
    for beta in betas:
        if time.time() > t_limit:
            break
        set_opts(target, beta)
        x0 = make_feasible(cur_h, n, target)
        try:
            res = minimize(
                objective_slsqp,
                x0,
                method="SLSQP",
                jac=gradient_slsqp,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-8, "maxls": 50, "disp": False},
            )
            cand = np.asarray(res.x, dtype=float)
        except Exception:
            cand = x0
        cand = make_feasible(cand, n, target)
        c = compute_c5(cand, n)
        if c <= best_c + 0.02:
            cur_h = cand
        if c < best_c - 1e-12:
            best_h = cand.copy()
            best_c = c
    return best_h, best_c


def pgrad_refine(h0, n, target, t_limit):
    h = make_feasible(h0, n, target)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    betas = [0.08, 0.03, 0.012, 0.005, 0.002]
    for beta in betas:
        if time.time() > t_limit:
            break
        alpha = 0.1
        stall = 0
        for _ in range(160):
            if time.time() > t_limit:
                break
            lse, maxq, g = lse_value_grad(h, beta)
            c_exact = float(maxq * (2.0 / float(n)))
            if c_exact < best_c - 1e-12:
                best_c = c_exact
                best_h = h.copy()
            gt = g - float(np.mean(g))
            gn = float(np.linalg.norm(gt))
            if gn < 1e-12:
                break
            step = min(0.25, alpha / max(gn, 0.5))
            fcur = lse
            improved = False
            for _ in range(25):
                y = project_sum_box(h - step * gt, target)
                fnew, _ = lse_value(y, beta)
                if fnew < fcur - 1e-9:
                    h = y
                    improved = True
                    break
                step *= 0.5
            if improved:
                stall = 0
                alpha = min(0.35, alpha * 1.05)
            else:
                stall += 1
                alpha *= 0.6
                if stall > 8:
                    break
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def binary_polish(h0, n, target, t_limit):
    if n > 160:
        return None, None
    h = make_feasible(h0, n, target)
    order = np.argsort(-h)
    b = np.zeros(n, dtype=float)
    cnt = int(round(target))
    if cnt > 0:
        b[order[:cnt]] = 1.0
    best_b = b.copy()
    best_c = compute_c5(b, n)
    local_c = best_c
    for _ in range(30):
        if time.time() > t_limit:
            break
        ones = np.nonzero(b == 1.0)[0]
        zeros = np.nonzero(b == 0.0)[0]
        best_pair = None
        best_pair_c = local_c
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                b[i] = 0.0
                b[j] = 1.0
                c = compute_c5(b, n)
                b[i] = 1.0
                b[j] = 0.0
                if c < best_pair_c - 1e-13:
                    best_pair_c = c
                    best_pair = (i, j)
        if best_pair is None:
            break
        b[best_pair[0]] = 0.0
        b[best_pair[1]] = 1.0
        local_c = best_pair_c
        if local_c < best_c - 1e-12:
            best_c = local_c
            best_b = b.copy()
    return best_b, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(1.0, float(budget_s)) - 2.0
    seed = int(seed)
    parent = globals().get("initial_h_values", None)

    best_h = None
    best_c = float("inf")
    best_n = 0

    if parent is not None:
        p = np.asarray(parent, dtype=float).ravel()
        if p.size > 0:
            n0 = int(p.size)
            if n0 % 2 == 1:
                n0 += 1
            target0 = n0 // 2
            if n0 == p.size:
                h0 = make_feasible(p, n0, target0)
            else:
                h0 = make_feasible(upsample_arr(p, n0), n0, target0)
            c0 = compute_c5(h0, n0)
            if np.isfinite(c0):
                best_h = h0.copy()
                best_c = c0
                best_n = n0

    if best_h is None:
        best_n = 70
        best_h = random_smooth_start(best_n, seed)
        best_c = compute_c5(best_h, best_n)

    target0 = best_n // 2
    if time.time() < deadline - 40.0:
        limit = min(deadline - 40.0, time.time() + (180.0 if best_n <= 90 else 120.0))
        h, c = refine_slsqp(best_h, best_n, target0, limit)
        if c < best_c:
            best_c = c
            best_h = h

    if time.time() < deadline - 20.0:
        limit = min(deadline - 20.0, time.time() + 150.0)
        h, c = pgrad_refine(best_h, best_n, target0, limit)
        if c < best_c:
            best_c = c
            best_h = h

    if best_n <= 160 and time.time() < deadline - 20.0:
        limit = min(deadline - 20.0, time.time() + 120.0)
        h, c = binary_polish(best_h, best_n, target0, limit)
        if c is not None and c < best_c:
            best_c = c
            best_h = h

    n_opts = []
    for mult in (2, 3):
        nh = best_n * mult
        if nh % 2 == 1:
            nh += 1
        if nh > best_n and 60 <= nh <= 240:
            if nh not in n_opts:
                n_opts.append(nh)
    for nh in (100, 140, 180):
        if nh > best_n and nh <= 240:
            if nh not in n_opts:
                n_opts.append(nh)
    n_opts.sort()

    for nh in n_opts[:2]:
        if deadline - time.time() < 220.0:
            break
        target = nh // 2
        starts = []
        if best_h is not None:
            starts.append(make_feasible(upsample_arr(best_h, nh), nh, target))
            starts.append(make_feasible(upsample_smooth(best_h, nh), nh, target))
            starts.append(perturb_h(best_h, nh, target, 0.03, seed + nh))
        if parent is not None:
            parr = np.asarray(parent, dtype=float).ravel()
            starts.append(make_feasible(upsample_smooth(parr, nh), nh, target))
        if nh <= 180:
            starts.append(random_smooth_start(nh, seed + nh + 11))
            starts.append(random_binary_start(nh, seed + nh + 23))

        top0 = (float("inf"), None)
        top1 = (float("inf"), None)
        for st in starts:
            c = compute_c5(st, nh)
            if not np.isfinite(c):
                continue
            if c < top0[0]:
                top1 = top0
                top0 = (c, st)
            elif c < top1[0]:
                top1 = (c, st)

        for titem in (top0, top1):
            if titem[1] is None:
                continue
            if deadline - time.time() < 120.0:
                break
            st = titem[1]
            slsqp_time = 100.0 if nh > 120 else 140.0
            h, c = refine_slsqp(st, nh, target, min(deadline - 60.0, time.time() + slsqp_time))
            if c < best_c:
                best_c = c
                best_h = h
                best_n = nh
            if deadline - time.time() > 80.0:
                h, c = pgrad_refine(h, nh, target, min(deadline - 20.0, time.time() + 120.0))
                if c < best_c:
                    best_c = c
                    best_h = h
                    best_n = nh
            if nh <= 160 and deadline - time.time() > 70.0:
                h, c = binary_polish(h, nh, target, min(deadline - 20.0, time.time() + 80.0))
                if c is not None and c < best_c:
                    best_c = c
                    best_h = h
                    best_n = nh

    restart = 0
    while deadline - time.time() > 180.0 and restart < 6:
        seed = (seed * 1103515245 + 12345) % 2147483648
        target = best_n // 2
        if best_h is not None:
            st = perturb_h(best_h, best_n, target, 0.04 + 0.02 * (restart % 3), seed)
        else:
            st = random_smooth_start(best_n, seed)
        h, c = refine_slsqp(st, best_n, target, min(deadline - 60.0, time.time() + 100.0))
        if c < best_c:
            best_c = c
            best_h = h
        if deadline - time.time() > 70.0:
            h, c = pgrad_refine(h, best_n, target, min(deadline - 20.0, time.time() + 90.0))
            if c < best_c:
                best_c = c
                best_h = h
        restart += 1

    target = best_n // 2
    best_h = make_feasible(best_h, best_n, target)
    final_c = compute_c5(best_h, best_n)
    if not np.isfinite(final_c):
        final_c = best_c
    return best_h, float(final_c), int(best_n)
