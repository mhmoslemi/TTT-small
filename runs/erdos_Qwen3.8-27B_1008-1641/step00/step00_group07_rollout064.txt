import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_to_set(h, S):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    shifted = alpha * (corr - mc)
    ev = np.exp(shifted)
    s = np.sum(ev)
    val = mc + np.log(s) / alpha
    w = ev / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        j1s, j1e = max(0, lag), min(n, n + lag)
        if j1s < j1e:
            idx = np.arange(j1s, j1e)
            grad[idx] += w[k] * (1.0 - h[idx - lag])
        j2s, j2e = max(0, -lag), min(n, n - lag)
        if j2s < j2e:
            idx = np.arange(j2s, j2e)
            grad[idx] += w[k] * (-h[idx + lag])
    grad *= dx
    return val, grad

def pgd_soft(h0, S, max_iter, alpha0, alpha1, step0):
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    step = step0
    n_iter = max_iter
    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        alpha = alpha0 + (alpha1 - alpha0) * frac ** 2
        val, grad = smooth_obj_grad(h, alpha)
        tv = evaluate_c5(h)
        if tv < best_val:
            best_val = tv
            best_h = h.copy()
        s = step
        for _ in range(15):
            h_new = project_to_set(h - s * grad, S)
            nv = evaluate_c5(h_new)
            if nv < tv or nv < best_val:
                h = h_new
                best_val_check = nv
                if nv < best_val:
                    best_val = nv
                    best_h = h.copy()
                break
            s *= 0.5
        else:
            step *= 0.5
        step = min(step * 1.02, step0)
    return best_h, best_val

def binary_local_search(h0, S, max_flips):
    n = len(h)
    S = int(round(S))
    h = (h0 > 0.5).astype(float)
    h = np.sort(h)[::-1]
    # fix to exactly S ones
    h = np.zeros(n)
    h[:S] = 1.0
    best = evaluate_c5(h)
    for _ in range(max_flips):
        improved = False
        for i in range(n):
            if h[i] == 1.0:
                for j in range(n):
                    if h[j] == 0.0:
                        h[i] = 0.0
                        h[j] = 1.0
                        v = evaluate_c5(h)
                        if v < best:
                            best = v
                            improved = True
                        h[j] = 0.0
                        h[i] = 1.0
            if improved:
                break
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.random.rand(80)

    n = len(h0)
    S = n / 2.0
    h0 = project_to_set(h0, S)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    def elapsed():
        return time.time() - t0

    # Phase 1: Soft-max projected gradient on initial resolution
    if elapsed() < budget_s * 0.35:
        h1, v1 = pgd_soft(h0, S, 4000, 5.0, 80.0, 0.04)
        if v1 < best_c5:
            best_c5, best_h = v1, h1

    # Phase 2: Restart perturbations at same resolution
    while elapsed() < budget_s * 0.60:
        for scale in [0.08, 0.15, 0.25]:
            if elapsed() >= budget_s * 0.60:
                break
            hp = best_h + scale * np.random.randn(n)
            hp = project_to_set(hp, S)
            h2, v2 = pgd_soft(hp, S, 1500, 10.0, 60.0, 0.03)
            if v2 < best_c5:
                best_c5, best_h = v2, h2

    # Phase 3: Binary local search as starting point for continuous refinement
    if elapsed() < budget_s * 0.70:
        hb, vb = binary_local_search(best_h, S, 3000)
        if vb < best_c5:
            best_c5, best_h = vb, hb
        h3, v3 = pgd_soft(hb, S, 2000, 10.0, 100.0, 0.02)
        if v3 < best_c5:
            best_c5, best_h = v3, h3

    # Phase 4: Try higher resolution via interpolation
    tried_higher = False
    while elapsed() < budget_s * 0.88 and not tried_higher:
        n2 = n + 20
        if n2 > 160:
            break
        S2 = n2 / 2.0
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_set(h_interp, S2)
        h4, v4 = pgd_soft(h_interp, S2, 3000, 8.0, 80.0, 0.03)
        if v4 < best_c5:
            best_c5, best_h, n, S = v4, h4, n2, S2
        else:
            tried_higher = True

    # Phase 5: Final fine refinement
    iters_left = max(100, int((budget_s * 0.95 - elapsed()) * 200))
    if iters_left > 100:
        h5, v5 = pgd_soft(best_h, S, min(iters_left, 5000), 50.0, 200.0, 0.01)
        if v5 < best_c5:
            best_c5, best_h = v5, h5

    # Final feasibility check
    best_h = project_to_set(best_h, S)
    final_c5 = evaluate_c5(best_h)

    return (best_h, final_c5, len(best_h))
