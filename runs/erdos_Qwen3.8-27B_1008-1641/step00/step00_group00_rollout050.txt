            delta = rng.choice([-0.02, -0.01, -0.005, 0.005, 0.01, 0.02])
            h_try = best_local.copy()
            h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
            h_try = project_onto_simplex(h_try, target_sum)
            c5_try = np.max(compute_correlation(h_try)) * dx
            if c5_try < best_local_c5 - 1e-14:
                best_local_c5 = c5_try
                best_local = h_try
        if best_local_c5 < best_c5:
            best_c5, best_h = best_local_c5, best_local

    # Final projection
    best_h = project_onto_simplex(best_h, target_sum)
    best_c5 = np.max(compute_correlation(best_h)) * dx

    return best_h, best_c5, n
