import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    s = np.sum(np.clip(h, 0.0, 1.0))
    if abs(s - target_sum) < 1e-10:
        return np.clip(h, 0.0, 1.0).copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def coord_desc(h, target, rng, t_limit, max_rounds=300):
    c5 = eval_c5(h)
    n = len(h)
    for r in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            for d in (0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.03, -0.03):
                t = h.copy()
                t[i] += d
                t = project_feasible(t, target)
                tc = eval_c5(t)
                if tc < c5 - 1e-13:
                    h = t
                    c5 = tc
                    improved = True
                    break
        if not improved:
            break
    return h, c5

def stochastic_search(h, target, rng, t_limit, sigma_init):

            h = trial
            c5 = tc
            sigma = min(0.1, sigma * 1.003)
