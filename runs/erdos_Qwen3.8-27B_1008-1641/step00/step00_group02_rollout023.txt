import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_feasible(h, target):
    """Project onto {0<=h<=1, sum(h)=target}"""
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny numerical errors
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        free = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def corr_grads(h, weights):
    """Weighted sum of gradients of corr[k] for all k"""
    n = len(h)
    grad = np.zeros(n)
    nk1 = n - 1
    for k in range(2 * n - 1):

        if w < 1e-12:

            if n - r > 0:
                grad[:n-r] += w * (1.0 - h[r:])

        h = project_feasible(h, target)
    elif s == 3:
        h = np.zeros(n)
        blk = max(1, n // 8)
        for i in range(0, n - blk + 1, 2 * blk):
            h[i:i + blk] = 1.0
        h += rng.randn(n) * 0.1
        h = project_feasible(h, target)
    elif s == 4:
        h = np.zeros(n)
        n_ones = int(target)
        remainder = target - n_ones
        idx = np.arange(n)
        rng.shuffle(idx)
        h[idx[:n_ones]] = 1.0
        if remainder > 0.01:
            h[idx[n_ones]] = remainder
        h = project_feasible(h, target)
    elif s == 5:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.2 * np.sin(3 * np.pi * x) + 0.15 * np.cos(5 * np.pi * x)
        h += rng.randn(n) * 0.05
        h = project_feasible(h, target)
    else:
        h = np.ones(n) * 0.5 + rng.randn(n) * 0.2
        h = project_feasible(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    best_c5 = [float('inf'), None]

    n_list = [45, 50, 60, 80, 100, 128, 160]
    for n in n_list:
        if time.time() > t_end:
            break
        target = n / 2.0
        n_starts = 7
        for s in range(n_starts):
            if time.time() > t_end:
                break
            h = make_start(n, s, target, rng, h_init)
            h_opt, c5_opt = gradient_optimize(h, n, target, t_end)
            if c5_opt < best_c5[0]:
                best_c5[0] = c5_opt
                best_c5[1] = h_opt.copy()
            h_opt, c5_p = perturb_refine(h_opt, n, target, rng, t_end, best_c5)
            if c5_p < best_c5[0]:
                best_c5[0] = c5_p
                best_c5[1] = h_opt.copy()
            # Second round of gradient from refined point
            if time.time() < t_end:
                h_opt2, c5_opt2 = gradient_optimize(h_opt, n, target, t_end)
                if c5_opt2 < best_c5[0]:
                    best_c5[0] = c5_opt2
                    best_c5[1] = h_opt2.copy()

    if best_c5[1] is None:
        n = 45
        h = np.ones(n) * 0.5
        best_c5[1] = project_feasible(h, n / 2.0)
        best_c5[0], _ = compute_c5(best_c5[1])

    return best_c5[1], best_c5[0], len(best_c5[1])
