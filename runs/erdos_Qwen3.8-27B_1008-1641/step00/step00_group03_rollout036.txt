import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
import time


def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def _make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total > 0:
                h += diff * room / total
        else:
            total = h.sum()
            if total > 0:
                h += diff * h / total
    return np.clip(h, 0.0, 1.0)


def _obj_func(x, n, dx):
    x = np.clip(x, 0, 1)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    T = 0.003
    cs = c / T
    cm = np.max(cs)
    return T * (cm + np.log(np.sum(np.exp(cs - cm))))


def _binary_ls(h_init, max_iter, timeout, rng):
    n = len(h_init)
    t_s = time.time()
    si = np.argsort(-h_init)
    hb = np.zeros(n)
    hb[si[:n // 2]] = 1.0
    c5 = _eval_c5(hb)
    oi = np.where(hb == 1)[0]
    zi = np.where(hb == 0)[0]
    for _ in range(max_iter):
        if time.time() - t_s > timeout:
            break
        i = oi[rng.integers(len(oi))]
        j = zi[rng.integers(len(zi))]
        hb[i] = 0.0
        hb[j] = 1.0
        c5n = _eval_c5(hb)
        if c5n <= c5:
            c5 = c5n
            oi = np.where(hb == 1)[0]
            zi = np.where(hb == 0)[0]
        else:
            hb[i] = 1.0
            hb[j] = 0.0
    return hb, c5


def _sa_bin(h_init, max_iter, timeout, rng):
    n = len(h_init)
    t_s = time.time()
    si = np.argsort(-h_init)
    hb = np.zeros(n)
    hb[si[:n // 2]] = 1.0
    c5 = _eval_c5(hb)
    best_h = hb.copy()
    best_c5 = c5
    oi = np.where(hb == 1)[0]
    zi = np.where(hb == 0)[0]
    T0, T1 = 0.04, 0.003
    for it in range(max_iter):
        if time.time() - t_s > timeout:
            break
        frac = float(it) / max(max_iter - 1, 1)
        T = T0 * (T1 / T0) ** frac
        i = oi[rng.integers(len(oi))]
        j = zi[rng.integers(len(zi))]
        hb[i] = 0.0
        hb[j] = 1.0
        c5n = _eval_c5(hb)
        d = c5n - c5
        if d < 0 or (T > 1e-8 and rng.uniform() < np.exp(-d / T)):
            c5 = c5n
            oi = np.where(hb == 1)[0]
            zi = np.where(hb == 0)[0]
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hb.copy()
        else:
            hb[i] = 1.0
            hb[j] = 0.0
    return best_h, best_c5


def _cont_opt(h_init, n, max_iter):
    dx = 2.0 / n
    x = np.clip(h_init, 0, 1).copy()
    cons = [LinearConstraint(np.ones((1, n)), n / 2.0, n / 2.0)]
    bnds = Bounds(0.0, 1.0)
    try:
        res = minimize(_obj_func, x, args=(n, dx), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        return _make_feasible(res.x, n)
    except Exception:
        return h_init
