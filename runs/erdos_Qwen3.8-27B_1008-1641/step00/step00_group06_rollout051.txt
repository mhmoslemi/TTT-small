    # First term: (1-h[j+lag]) for j where j+lag in [0,n-1]
    if lag >= 0:
        idx1 = np.arange(0, n - lag)  # j from 0 to n-1-lag
        grad[idx1] += (1.0 - h[idx1 + lag])
        idx2 = np.arange(lag, n)  # j from lag to n-1
        grad[idx2] += (-h[idx2 - lag])
    else:
        nl = -lag
        idx1 = np.arange(nl, n)  # j from nl to n-1
        grad[idx1] += (1.0 - h[idx1 - nl])
        idx2 = np.arange(0, n - nl)  # j from 0 to n-1-nl
        grad[idx2] += (-h[idx2 + nl])
    grad *= dx
    return c5, grad, mi

Now, one issue: the objective is the MAX over all lags. The subgradient at the max is the gradient of the active lag. But there could be multiple lags achieving the max. In that case, the subgradient is any convex combination of their gradients. Using just one is OK for a subgradient method, but it might oscillate.

A better approach: use the average gradient over the top few lags, or use the gradient of the smooth approximation (log-sum-exp).

Let me use the top-2 or top-3 lags average for a more stable gradient:

def smooth_grad(h, top_k=3):

    corr = np.correlate(h, 1.0 - h, mode='full')

            idx2 = np.arange(0, n - nl)
            grad[idx2] += w * (-h[idx2 + nl])
    c5 = float(dx * corr.max())
    grad *= dx
    return c5, grad

f_β(h) = (1/β) * log(sum(exp(β * corr_k)))

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[::-1][:top_k]

    grad = np.zeros(n)
    for mi in top_indices:
        lag = int(mi) - (n - 1)
        if lag >= 0:
            idx1 = np.arange(0, n - lag)
            grad[idx1] += (1.0 - h[idx1 + lag])
            idx2 = np.arange(lag, n)

            grad[idx1] += (1.0 - h[idx1 - nl])
            idx2 = np.arange(0, n - nl)
            grad[idx2] += (-h[idx2 + nl])
    grad /= float(top_k)
    grad *= dx

def soft_grad(h, beta=30.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = float(corr.max())
    expv = np.exp(beta * (corr - c5))
    w = expv / expv.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx1 = np.arange(0, n - lag)
            grad[idx1] += wk * (1.0 - h[idx1 + lag])
            idx2 = np.arange(lag, n)
            grad[idx2] += wk * (-h[idx2 - lag])

            nl = -lag

            grad[idx1] += wk * (1.0 - h[idx1 - nl])
            idx2 = np.arange(0, n - nl)
            grad[idx2] += wk * (-h[idx2 + nl])
