    """Project x onto {0<=h<=1, sum(h)=target_sum}"""

        if s > target_sum:
            lo = mid
        else:
            hi = mid

        if n_free.any():

import numpy as np
import time

def project_feasible(x, target_sum):
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def subgradient_max_corr(h, n):
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    max_c = corr.max()
    active = np.where(corr >= max_c - 1e-9)[0]
    grad = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if hi1 > lo1:
            grad[lo1:hi1] += g[lo1 + shift:hi1 + shift]
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad / len(active)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    dx = 2.0 / n

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    def elapsed():
        return time.time() - t0

    def optimize_phase(h_start, alpha_init, n_iters, mu):
        h_local = h_start.copy()
        v = np.zeros(n)
        local_best = compute_c5(h_local, dx)
        local_best_h = h_local.copy()
        alpha = alpha_init
        no_improve = 0

            gn = np.linalg.norm(grad)

            c5_trial = compute_c5(h_trial, dx)

                local_best = c5_trial
                local_best_h = h_trial.copy()
                h_local = h_trial
                no_improve = 0

                h_local = project_feasible(h_local - 0.5 * v, target_sum)
                no_improve += 1
            alpha *= 0.9998
            if no_improve > 80:
                alpha *= 0.85
                v *= 0.3
                no_improve = 0
        return local_best_h, local_best

    n_splits = 25
    num_restarts = 40
