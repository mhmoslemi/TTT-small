    t0 = time.time()

    hv = globals().get('initial_h_values', None)
    if hv is not None:
        h_init = np.array(hv, dtype=float)
    else:
        h_init = np.full(47, 0.5)
        h_init += 0.3 * np.sin(2.0 * np.pi * np.arange(47) / 47.0)
        h_init = repair(h_init, 47)

    n0 = len(h_init)
    dx0 = 2.0 / n0
    best_h = h_init.copy()
    best_c5 = c5_score(best_h, dx0)

    # Phase 1: SLSQP local optimization from initial
    if budget_s - (time.time() - t0) > 30:
        h1, c5_1 = slsqp_refine(best_h, n0, maxiter=600)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    # Phase 2: Differential Evolution (multiple runs)
    for de_seed in [seed, seed + 100, seed + 200]:
        elapsed = time.time() - t0
        if budget_s - elapsed < 30:
            break
        rep_obj = RepairedObj(n0)
        bounds = [(0.0, 1.0)] * n0
        max_evals = min(int((budget_s - elapsed) * 80), 80000)
        try:
            res = differential_evolution(
                rep_obj, bounds, seed=de_seed,
                maxiter=2000, popsize=40,
                mutation=(0.3, 1.0), recombination=0.8,
                tol=1e-12, max_evals=max_evals,
                polish=False, init='latinhypercube'
            )
            h_de = repair(res.x, n0)
            c5_de = c5_score(h_de, dx0)
            if c5_de < best_c5:
                best_h, best_c5 = h_de, c5_de
        except Exception:
            pass

    # Phase 3: SLSQP refinement of best
    if budget_s - (time.time() - t0) > 30:
        nb = len(best_h)
        h3, c5_3 = slsqp_refine(best_h, nb, maxiter=800)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: Try other n values
    for n_try in [47, 50, 64, 80]:
        if n_try == n0 and budget_s - (time.time() - t0) < 45:
            continue
        if budget_s - (time.time() - t0) < 45:
            break
        dx_t = 2.0 / n_try
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate')
        h_t = f(x_new) + 0.08 * rs.randn(n_try)
        h_t = repair(h_t, n_try)

        if budget_s - (time.time() - t0) > 45:
            rep_obj = RepairedObj(n_try)
            bounds = [(0.0, 1.0)] * n_try
            max_evals = min(int((budget_s - (time.time() - t0)) * 60), 40000)
            try:
                res = differential_evolution(
                    rep_obj, bounds, seed=seed + n_try,
                    maxiter=1000, popsize=30,
                    mutation=(0.3, 1.0), recombination=0.8,
                    tol=1e-12, max_evals=max_evals,
                    polish=False, init='latinhypercube'
                )
                h_de = repair(res.x, n_try)
                c5_de = c5_score(h_de, dx_t)
                if c5_de < best_c5:
                    best_h, best_c5 = h_de, c5_de
                    h_t = h_de
            except Exception:
                pass

        if budget_s - (time.time() - t0) > 25:
            h_r, c5_r = slsqp_refine(h_t, n_try, maxiter=400)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r

    # Phase 5: Final polish
    if budget_s - (time.time() - t0) > 15:
        nb = len(best_h)
        h5, c5_5 = slsqp_refine(best_h, nb, maxiter=300)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    best_h = repair(best_h, len(best_h))
    best_c5 = c5_score(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
