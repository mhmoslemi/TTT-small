import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    return np.max(corr) * dx

def smooth_max(val, alpha):
    m = np.max(val)
    if m < -1e30:
        return m
    return m + np.log(np.mean(np.exp(val - m))) / alpha

def smooth_c5_obj(h, dx, alpha):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    return smooth_max(corr, alpha)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = np.sum(mask)
        if cnt > 0:
            step = diff / cnt
            step = min(step, (1.0 - h[mask]).max() if cnt > 0 else 0)
            h[mask] = np.minimum(h[mask] + step, 1.0)
    else:
        mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            step = -diff / cnt  # diff is negative, so step is positive... wait
            # We need to decrease sum by |diff|
            step = -diff / cnt  # this is the amount to subtract per element
            step = min(step, h[mask].min() if cnt > 0 else 0)
            h[mask] = np.maximum(h[mask] - step, 0.0)

    m = np.max(val)
    return m + np.log(np.sum(np.exp(val - m))) / (alpha * len(val))

    return smooth_max_obj(compute_corr(x, dx), alpha)

    n = len(x)
    corr = compute_corr(x, dx)
    shifted = corr - np.max(corr)
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    # Compute grad of each correlation, then weighted sum

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_corr(h, dx):
    omh = 1.0 - h
    return np.correlate(h, omh, mode='full') * dx

def compute_c5(h, dx):
    return float(np.max(compute_corr(h, dx)))

def smooth_max_obj(val, alpha):
    m = np.max(val)
    return m + np.log(np.sum(np.exp(alpha * (val - m)))) / alpha

def smooth_obj(x, dx, alpha):
    return smooth_max_obj(compute_corr(x, dx), alpha)

def smooth_grad(x, dx, alpha):
    n = len(x)
    corr = compute_corr(x, dx)
    shifted = alpha * (corr - np.max(corr))
    w = np.exp(shifted)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if w[k_idx] < 1e-14:
            continue
        shift = k_idx - (n - 1)
        wk = w[k_idx]
        if shift == 0:
            grad += wk * (1.0 - 2.0 * x) * dx
        elif shift > 0:
            grad[shift:] += wk * (1.0 - x[:-shift]) * dx
            grad[:-shift] -= wk * x[shift:] * dx
        else:
            s = -shift
            grad[:-s] += wk * (1.0 - x[s:]) * dx
            grad[s:] -= wk * x[:-s] * dx
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, (1.0 - h[mask]).max())
            h[mask] += step
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(-diff / cnt, h[mask].min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def sum_constraint(x):
    return np.sum(x) - x.shape[0] / 2.0

def make_fun_and_jac(dx, alpha):
    def fun(x):
        return smooth_obj(x, dx, alpha)
    def jac(x):
        return smooth_grad(x, dx, alpha)
    return fun, jac

def opt_lbfgsb(h, dx, alpha, maxiter, tol):
    n = len(h)
    fun, jac = make_fun_and_jac(dx, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol,
                            'maxls': 40})
    return project_feasible(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    n = 70
    dx = 2.0 / n

    h0 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    if h0 is None:
        h0 = np.full(n, 0.5)
    if len(h0) != n:
        xo = np.linspace(0, 2, len(h0), endpoint=False)
        xn = np.linspace(0, 2, n, endpoint=False)
        h0 = interp1d(xo, h0, kind='linear', bounds_error=False, fill_value=0.5)(xn)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    for alpha in [20.0, 50.0, 100.0, 200.0, 400.0, 800.0]:
        if rem() < 25:
            break
        h_new = opt_lbfgsb(best_h, dx, alpha, maxiter=3000, tol=1e-13)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    n_starts = 14
    for s in range(n_starts):
        if rem() < 35:
            break
        if s == 0:
            h_cur = best_h + 0.08 * np.random.randn(n)
        elif s % 4 == 1:
            h_cur = np.zeros(n)
            idx = np.random.choice(n, n // 2, replace=False)
            h_cur[idx] = 1.0
        elif s % 4 == 2:
            h_cur = best_h + 0.25 * np.random.randn(n)
        elif s % 4 == 3:
            h_cur = np.zeros(n)
            bsz = np.random.randint(4, 13)
            for j in range(0, n, bsz):
                if np.random.rand() > 0.4:
                    h_cur[j:min(j + bsz, n)] = 1.0
        else:
            h_cur = best_h * np.random.uniform(0.5, 1.5, n)
        h_cur = project_feasible(h_cur, n)
        for a in [60.0, 200.0]:
            if rem() < 20:
                break
            h_new = opt_lbfgsb(h_cur, dx, a, maxiter=2500, tol=1e-12)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    for n2 in [100, 140, 200]:
        if rem() < 70:
            break
        dx2 = 2.0 / n2
        x1 = np.linspace(0, 2, len(best_h), endpoint=False)
        x2 = np.linspace(0, 2, n2, endpoint=False)
        h_fine = interp1d(x1, best_h, kind='linear', bounds_error=False, fill_value=0.5)(x2)
        h_fine = project_feasible(h_fine, n2)

            if rem() < 25:

                best_h = h_new

        nf = len(best_h)
        dxf = 2.0 / nf
        for a in [600.0, 1200.0]:
            if rem() < 12:

    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
