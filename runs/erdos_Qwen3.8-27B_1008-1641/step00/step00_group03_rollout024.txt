import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_obj(n):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    return obj

def make_obj_smooth(n, alpha):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1 - h, mode="full")
        v = corr * dx * alpha
        v = np.clip(v, -300, 300)
        m = np.max(v)
        return (m + np.log(np.sum(np.exp(v - m)))) / alpha
    return obj

def try_slsqp(h_start, n, bounds, cons, maxiter=400, ftol=1e-12):
    obj = make_obj(n)
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None

def try_smooth(h_start, n, bounds, cons, alpha, maxiter=300):
    obj = make_obj_smooth(n, alpha)

        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
