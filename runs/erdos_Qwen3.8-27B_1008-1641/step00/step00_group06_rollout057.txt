import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_c5_smooth(h, t):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # log-sum-exp for numerical stability
    max_v = np.max(corr)
    if t < 1e-6:
        return dx * max_v
    return dx * (max_v + np.log(np.sum(np.exp(t * (corr - max_v)))) / t)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if not mask.any():

        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)

def c5_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_full = np.argmax(corr)
    k = k_full - (n - 1)  # actual lag
    
    grad = np.zeros(n)
    # For c[k] = sum_i h[i+k]*(1-h[i]):
    # ∂/∂h[j] = (1-h[j-k]) * 1{0≤j-k<n, valid i range} - h[j+k] * 1{0≤j+k<n, valid i range}
    
    # Valid i: max(0,-k) ≤ i ≤ min(n-1, n-1-k)
    i_lo = max(0, -k)
    i_hi = min(n-1, n-1-k)

            val += (1.0 - h[j - k])
        if i_lo <= j <= i_hi:
            val -= h[j + k]
        grad[j] = dx * val
    
    return grad, dx * corr[k_full]

    amp = rng.uniform(0.3, 0.55)
    h = 0.5 + amp * np.sin(np.pi * f * x + phase)
    return make_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values, evaluate_erdos_solution
    rng = np.random.default_rng(seed)
    t0 = time.time()
    best_h = None
    best_c5 = np.inf

    def time_left(frac=1.0):
        return (time.time() - t0) < budget_s * frac

    # Phase 1: SLSQP from initial_h_values
    if initial_h_values is not None:
        n0 = len(initial_h_values)
        h0 = make_feasible(initial_h_values.astype(float), n0)
        h1 = optimize_slqp(h0, n0, maxiter=1000)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

        # Phase 2: Projected gradient refinement
        if time_left(0.45):
            h2, c2 = optimize_proj_grad(h1, n0, max_iters=50000)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

        # Phase 3: Perturbation restarts
        if time_left(0.60):
            for _ in range(60):
                if not time_left(0.60):
                    break
                ns = float(rng.uniform(0.03, 0.15))
                hp = make_feasible(best_h + rng.normal(0, ns, n0), n0)
                hr = optimize_slqp(hp, n0, maxiter=300)
                cr = compute_c5(hr)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()

    # Phase 4: Try other n values
    n_list = [100, 120, 200, 300, 500, 998]
    for n_try in n_list:
        if not time_left(0.85):
            break
        n = n_try
        if best_h is not None and len(best_h) != n:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n)
            h0 = np.interp(x_new, x_old, best_h)
        elif best_h is not None:
            h0 = best_h.copy()
        else:
            h0 = make_sine_init(n, rng)
        h0 = make_feasible(h0, n)
        maxiter = 500 if n <= 200 else 300
        h_opt = optimize_slqp(h0, n, maxiter=maxiter)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()
        # proj grad for larger n
        if n > 200 and time_left(0.85):
            h_pg, c_pg = optimize_proj_grad(h_opt, n, max_iters=30000)
            if c_pg < best_c5:
                best_c5 = c_pg
                best_h = h_pg.copy()
        # sine init alternative
        if time_left(0.85):
            h_sine = make_sine_init(n, rng)
            h_s = optimize_slqp(h_sine, n, maxiter=300)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s.copy()

    # Phase 5: Binary polish (only for even n)
    if time_left(0.95):
        n_b = len(best_h)
        if n_b % 2 == 0:
            h_bin, c_bin = binary_polish(best_h, n_b, rng, max_iters=200000)
            if c_bin < best_c5:
                best_c5 = c_bin
                best_h = h_bin.copy()
        # Also try binary at even n values not yet tried
        for n_ev in [100, 200, 500, 998]:
            if not time_left(0.95):
                break
            if n_ev == n_b:
                continue
            h_init = make_sine_init(n_ev, rng)
            h_fin = optimize_slqp(h_init, n_ev, maxiter=300)
            h_bin2, c_bin2 = binary_polish(h_fin, n_ev, rng, max_iters=200000)
            if c_bin2 < best_c5:
                best_c5 = c_bin2
                best_h = h_bin2.copy()

    # Final SLSQP polish on best solution
    if time_left(0.99):
        n_f = len(best_h)
        h_final = optimize_slqp(best_h, n_f, maxiter=500)
        c_final = compute_c5(h_final)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final.copy()

    if best_h is None:
        n_fallback = 100
        best_h = make_feasible(np.full(n_fallback, 0.5), n_fallback)
        best_c5 = compute_c5(best_h)

    try:
        evaluate_erdos_solution(best_h)
    except Exception:
        pass

    return best_h, best_c5, len(best_h)
