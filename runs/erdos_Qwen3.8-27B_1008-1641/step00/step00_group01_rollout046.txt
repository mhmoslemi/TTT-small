```python
def repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            # Need to increase; distribute to elements below 1
            mask = h < 1.0
            if not np.any(mask):
                break  # Can't increase, all at 1
            # Distribute proportionally to headroom
            headroom = 1.0 - h[mask]
            add = diff / np.sum(headroom) * headroom
            h[mask] += add
            h[mask] = np.minimum(h[mask], 1.0)
        else:
            # Need to decrease; distribute from elements above 0
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(h[mask]) * h[mask]
            h[mask] = np.maximum(h[mask], 0.0)
    return h
```

```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

_GN = 47

def repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def de_objective(x):
    h = repair(x, _GN)
    return compute_c5(h, _GN)

def smooth_obj(x):
    n = _GN
    h = repair(x, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    I = corr * dx
    m = np.max(I)
    vals = I - m
    return m + np.log(np.sum(np.exp(vals))) / 80.0

def smooth_obj_200(x):
    n = _GN
    h = repair(x, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    I = corr * dx
    m = np.max(I)
    vals = I - m
    return m + np.log(np.sum(np.exp(vals))) / 200.0

def cons_eq(x):
    return np.sum(x) - _GN / 2.0

def cons_jac(x):
    return np.ones(_GN)

def local_refine(h, n, max_time):
    global _GN
    _GN = n
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}
    for obj in [smooth_obj, smooth_obj_200]:
        if time.time() - t0 > max_time - 5:
            break
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 500, 'ftol': 1e-15})
            h_new = repair(res.x, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best_h

def run(seed=42, budget_s=1100, **kwargs):
    global _GN
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(47, 0.5)

    n = len(h_init)
    _GN = n
    h_init = repair(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Differential Evolution (global search)
    de_budget = min(budget_s * 0.45, 400.0)
    mi = max(80, int(de_budget * 2.5))

    result = differential_evolution(
        de_objective, [(0.0, 1.0)] * n,
        seed=seed, maxiter=mi, popsize=35,
        tol=None, mutation=(0.4, 1.2), recombination=0.8,
        init='latinhypercube', polish=False
    )
    h_de = repair(result.x, n)
    c5_de = compute_c5(h_de, n)
    if c5_de < best_c5:
        best_h = h_de
        best_c5 = c5_de

    # Phase 2: Second DE with perturbation around best
    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        h_pert = best_h + np.random.RandomState(seed + 1).uniform(-0.1, 0.1, n)
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = repair(h_pert, n)
        # Add to initial population by running DE again with different seed
        result2 = differential_evolution(
            de_objective, [(0.0, 1.0)] * n,
            seed=seed + 100, maxiter=150, popsize=25,
            tol=None, mutation=(0.3, 0.9), recombination=0.7,
            init='latinhypercube', polish=False
        )
        h_de2 = repair(result2.x, n)
        c5_de2 = compute_c5(h_de2, n)
        if c5_de2 < best_c5:
            best_h = h_de2
            best_c5 = c5_de2

    # Phase 3: Local refinement
    elapsed = time.time() - start_time
    if budget_s - elapsed > 30:
        local_budget = min(budget_s - elapsed - 15, 200.0)
        best_h = local_refine(best_h, n, local_budget)
        best_c5 = compute_c5(best_h, n)

    # Phase 4: Try nearby n values
    elapsed = time.time() - start_time
    for n_try in [n + 1, n - 1, n + 2, 50, 60]:
        if n_try < 20 or n_try >= 200:
            continue
        if time.time() - start_time > budget_s - 30:
            break
        _GN = n_try
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = repair(h_interp, n_try)
        res_n = differential_evolution(
            de_objective, [(0.0, 1.0)] * n_try,
            seed=seed + n_try, maxiter=120, popsize=25,
            tol=None, mutation=(0.5, 1.0), recombination=0.8,
            init='latinhypercube', polish=False
        )
        h_try = repair(res_n.x, n_try)
        c5_try = compute_c5(h_try, n_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            n = n_try
        _GN = n

    # Final: polish the best solution
    elapsed = time.time() - start_time
    if budget_s - elapsed > 15:
        _GN = n
        best_h = local_refine(best_h, n, min(budget_s - elapsed - 5, 60.0))
        best_c5 = compute_c5(best_h, n)

    # Ensure exact constraint
    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, n)

    return (best_h, best_c5, n)
```
