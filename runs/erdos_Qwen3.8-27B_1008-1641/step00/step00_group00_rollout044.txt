import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return dx * np.max(compute_corr(h))

def corr_grad_matrix(h, n):
    """Return G of shape (2n-1, n) where G[i,m] = d(corr[i])/d(h[m])"""
    num = 2 * n - 1
    G = np.zeros((num, n))
    for i in range(num):
        off = i - (n - 1)
        j_low = max(0, -off)
        j_high = n - 1 - max(0, off)
        lo1, hi1 = max(0, j_low + off), min(n - 1, j_high + off)
        idx1 = np.arange(lo1, hi1 + 1)
        G[i, idx1] += 1.0 - h[idx1 - off]
        idx2 = np.arange(j_low, j_high + 1)
        valid = (idx2 + off >= 0) & (idx2 + off < n)
        G[i, idx2[valid]] -= h[idx2[valid] + off]
    return G

def project_to_set(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - p.sum()
    if abs(diff) > 1e-14:
        free = (p > 1e-12) & (p < 1.0 - 1e-12)
        if free.any():
            p[free] += diff / free.sum()
            np.clip(p, 0.0, 1.0, out=p)
    return p

def smoothed_obj_grad(h, dx, n, alpha):
    """Log-sum-exp approximation to max(dx*corr). Returns (value, gradient)."""
    corr = compute_corr(h)
    s = alpha * dx * corr
    s_max = np.max(s)
    exp_s = np.exp(s - s_max)
    w = exp_s / np.sum(exp_s)
    val = s_max / alpha + np.log(np.sum(exp_s)) / alpha
    G = corr_grad_matrix(h, n)
    grad = dx * (G.T @ w)
    return val, grad

def pgd_optimize(h, dx, n, target_sum, alpha, lr, max_iter, rng, time_fn):
    """Projected gradient descent with backtracking."""
    c5_cur = compute_c5(h, dx)
    for it in range(max_iter):
        if time_fn() < 10:
            break
        _, grad = smoothed_obj_grad(h, dx, n, alpha)
        step = lr
        for _ in range(10):
            h_new = project_to_set(h - step * grad, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new <= c5_cur + 1e-12:
                break
            step *= 0.5
        h = h_new
        c5_cur = c5_new
    return h, c5_cur

def slsqp_optimize(h0, dx, n, target_sum, rng, max_iter, time_fn):
    alpha = 100.0

    def obj(x):
        hp = project_to_set(x, target_sum)
        corr = compute_corr(hp)
        s = dx * corr
        m = np.max(s)
        return m + np.log(np.sum(np.exp(alpha * (s - m)))) / alpha

    def obj_grad(x):
        hp = project_to_set(x, target_sum)
        corr = compute_corr(hp)
        s = alpha * dx * corr
        m = np.max(s)
        w = np.exp(s - m) / np.sum(np.exp(s - m))
        G = corr_grad_matrix(hp, n)
        return dx * (G.T @ w)

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', jac=obj_grad,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    hp = project_to_set(res.x, target_sum)
    return hp, compute_c5(hp, dx)
