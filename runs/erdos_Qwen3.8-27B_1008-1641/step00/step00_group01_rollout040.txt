            old = best_h[k]
            for delta in [1e-4, -1e-4, 5e-4, -5e-4]:
                trial = best_h.copy()
                trial[k] = np.clip(old + delta, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c5_try = c5_objective(trial, dx)
                if c5_try < best_c5 - 1e-13:
                    best_h, best_c5 = trial.copy(), c5_try
                    break

    best_h = project_feasible(best_h, n)
    final_c5 = c5_objective(best_h, dx)
    return (best_h, final_c5, n)
