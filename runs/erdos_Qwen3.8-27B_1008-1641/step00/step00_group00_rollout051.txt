            break
        scale = 0.06 * (0.8 ** (r // 50))
        h0 = project_feasible(best_h + scale * rng.randn(n), S)
        try:
            res = minimize(_slsqp_obj, h0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 600, 'ftol': 1e-14})
            ht = project_feasible(res.x, S)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
        except Exception:
            pass

    # Phase 4: Final polish - subgradient with tiny steps
    h = best_h.copy()
    alpha2 = 0.005
    for it in range(100000):
        if not ok(3.0):
            break
        cv, g = subgrad_max(h)
        C5_cur = cv * (2.0 / n)
        h_try = project_feasible(h - alpha2 * g, S)
        C5_try = compute_c5(h_try)
        if C5_try < C5_cur:
            h = h_try
            alpha2 = min(alpha2 * 1.01, 0.05)
            if C5_try < best_c5:
                best_c5 = C5_try
                best_h = h.copy()
        else:
            alpha2 *= 0.9
            if alpha2 < 1e-6:
                break

    best_h = project_feasible(best_h, S)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
