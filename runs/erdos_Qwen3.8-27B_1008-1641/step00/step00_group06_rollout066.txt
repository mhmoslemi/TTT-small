    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))


def compute_c5_smooth(h, alpha=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return float(dx * (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha))


def fix_sum(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                break
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        order = np.argsort(h)
        for i in order:
            sub = min(-diff, h[i])
            if sub <= 0:
                break
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h


def _make_constraints(n):
    target = n / 2.0
    def eq_con(x):
        return np.sum(x) - target
    def eq_jac(x):
        return np.ones(n)
    return ({'type': 'eq', 'fun': eq_con, 'jac': eq_jac},)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    parent = np.array(initial_h_values, dtype=float)
    n = len(parent)
    parent = fix_sum(parent, n)

    best_h = parent.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    cons = _make_constraints(n)

    def deadline():
        return time.time() - t0 > budget_s - 20

    # Phase 1: SLSQP from parent
    if not deadline():
        res = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
        cand = fix_sum(res.x, n)
        c5v = compute_c5(cand)
        if c5v < best_c5:
            best_h, best_c5 = cand, c5v

    # Phase 2: Smoothed objective
    if not deadline():
        res = minimize(lambda x: compute_c5_smooth(x, 100.0), best_h,
                       method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
        cand = fix_sum(res.x, n)
        c5v = compute_c5(cand)
        if c5v < best_c5:
            best_h, best_c5 = cand, c5v

    # Phase 3: Multi-start perturbations
    attempt = 0
    while not deadline():
        if attempt < 15:
            scale = 0.15
        elif attempt < 30:
            scale = 0.08
        else:
            scale = 0.04

        h0 = best_h + np.random.randn(n) * scale
        h0 = fix_sum(h0, n)

        use_smooth = (attempt % 3 != 0)
        if use_smooth:
            func = lambda x: compute_c5_smooth(x, 100.0)
        else:
            func = compute_c5

        try:
            res = minimize(func, h0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 150, 'ftol': 1e-12})
            cand = fix_sum(res.x, n)
            c5v = compute_c5(cand)
            if c5v < best_c5:
                best_h, best_c5 = cand, c5v
        except Exception:
            pass
        attempt += 1

    # Phase 4: Coordinate descent refinement
    if not deadline():
        target = n / 2.0
        for _cd in range(50):
            if deadline():
                break
            improved = False
            order = np.random.permutation(n)
            for i in order:
                if deadline():
                    break
                old = best_h[i]
                # Try sweeping h[i] while adjusting h[j] to maintain sum
                # Pick j with most room in the needed direction
                for val in np.linspace(0.0, 1.0, 21):
                    delta = val - old
                    if abs(delta) < 1e-8:
                        continue
                    # Compensate on the coordinate with most available room
                    if delta > 0:
                        # Need to decrease another coord
                        j = int(np.argmin(best_h))
                        room = min(delta, best_h[j])
                        if room <= 1e-8:
                            continue
                    else:
                        j = int(np.argmin(1.0 - best_h))
                        room = min(-delta, 1.0 - best_h[j])
                        if room <= 1e-8:
                            continue
                    trial = best_h.copy()
                    trial[i] = val
                    trial[j] -= delta if trial[i] > old else -delta
                    # More careful compensation
                    trial = best_h.copy()
                    trial[i] = val
                    comp = old - val  # amount to subtract from j
                    trial[j] -= comp
                    if trial[j] < 0 or trial[j] > 1:
                        continue
                    c5v = compute_c5(trial)
                    if c5v < best_c5 - 1e-10:
                        best_h = trial
                        best_c5 = c5v
                        improved = True
                        break
        # Final SLSQP polish
        if not deadline():
            res = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 400, 'ftol': 1e-14})
            cand = fix_sum(res.x, n)
            c5v = compute_c5(cand)
            if c5v < best_c5:
                best_h, best_c5 = cand, c5v

    best_h = fix_sum(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
