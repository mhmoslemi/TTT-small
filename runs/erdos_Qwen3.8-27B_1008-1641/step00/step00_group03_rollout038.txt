import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n_pts):
    target = n_pts / 2.0
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        free = (h > 0.005) & (h < 0.995)
        nf = np.sum(free)
        if nf > 0 and nf < n_pts:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            h += diff2 / n_pts
            h = np.clip(h, 0.0, 1.0)
        diff3 = target - np.sum(h)
        guard = 0
        while abs(diff3) > 1e-12 and guard < n_pts * 2:
            if diff3 > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff3)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff3)
            diff3 = target - np.sum(h)
            guard += 1
    return h

def get_worst_shift(h):
    n = len(h)
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, corr, idx

def local_search_shift(h, n_pts, max_iters=5000, tol=1e-10):
    """Custom local search: move mass to reduce worst-shift overlap."""
    target = n_pts / 2.0
    best = h.astype(np.float64).copy()
    best_c5 = compute_c5(best)
    for it in range(max_iters):
        shift, corr, cidx = get_worst_shift(best)
        if abs(shift) >= n_pts:
            break
        # For shift s, overlap = sum h[i+s]*(1-h[i]) for valid i
        # Find pairs contributing most to overlap
        s = shift
        if s >= 0:
            pairs = [(i + s, i) for i in range(n_pts - s)]
        else:
            sp = -s
            pairs = [(i, i + sp) for i in range(n_pts - sp)]
        # Compute contribution of each pair
        contribs = np.array([best[p[0]] * (1.0 - best[p[1]]) for p in pairs])
        # Sort by contribution descending
        order = np.argsort(contribs)[::-1]
        improved = False
        for oi in order[:30]:
            i0, i1 = pairs[oi]
            # Move mass: decrease h[i0], increase h[i1] (reduces h[i0]*(1-h[i1]))
            for delta in [0.02, 0.05, 0.1]:
                h_try = best.copy()
                h_try[i0] -= delta
                h_try[i1] += delta
                if h_try[i0] < 0 or h_try[i1] > 1:
                    continue
                c = compute_c5(h_try)
                if c < best_c5 - tol:
                    best = h_try
                    best_c5 = c
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Try reverse: increase h[i0], decrease h[i1]
            for oi in order[:30]:
                i0, i1 = pairs[oi]
                for delta in [0.02, 0.05]:
                    h_try = best.copy()
                    h_try[i0] += delta
                    h_try[i1] -= delta
                    if h_try[i0] > 1 or h_try[i1] < 0:
                        continue
                    c = compute_c5(h_try)
                    if c < best_c5 - tol:
                        best = h_try
                        best_c5 = c
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best, best_c5
