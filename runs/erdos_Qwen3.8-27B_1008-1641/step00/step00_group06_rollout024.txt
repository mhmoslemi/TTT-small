import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(95, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0

    def project_feasible(h):
        h = np.clip(h, 0.0, 1.0)
        for _ in range(300):
            s = h.sum()
            diff = target_sum - s
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                mask = h < 1.0 - 1e-14
            else:
                mask = h > 1e-14
            if not mask.any():
                break
            room = (1.0 - h[mask]).min() if diff > 0 else h[mask].min()
            step = diff / mask.sum()
            if diff > 0:
                step = min(step, room * 0.999)
            else:
                step = max(step, -room * 0.999)
            h[mask] += step
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-13:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)
        return h

    def eval_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * dx)

    h0 = project_feasible(h0)
    best_h = h0.copy()
    best_c5 = eval_c5(h0)
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(x, temp):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr * dx
        m = m / temp
        m -= np.max(m)
        return temp * np.log(np.sum(np.exp(m))) + np.max(corr) * dx - np.max(corr) * dx

    def smooth_obj_simple(x, temp):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr * dx
        m_norm = m / temp
        m_norm -= np.max(m_norm)
        return temp * np.log(np.sum(np.exp(m_norm))) + np.max(m) - np.max(m)

    # Simpler: just use logsumexp directly
    def sobj(x, temp):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        mx = np.max(corr)
        return mx + temp * np.log(np.mean(np.exp((corr - mx) / temp)))

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]

    elapsed = time.time() - start_time
    n_rounds = 60

    for iteration in range(n_rounds):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.82:
            break

        # Generate perturbed starting point
        if iteration == 0:
            h_pert = best_h.copy()
        else:
            ptype = iteration % 5
            if ptype == 0:
                h_pert = best_h + rng.normal(0, 0.04, n)
            elif ptype == 1:
                h_pert = best_h.copy()
                k = rng.randint(3, 25)
                idx = rng.choice(n, size=k, replace=False)
                h_pert[idx] = rng.uniform(0, 1, k)
            elif ptype == 2:
                h_pert = best_h.copy()
                sh = rng.randint(-10, 11)
                h_pert = np.roll(h_pert, sh)
                h_pert += rng.normal(0, 0.02, n)
            elif ptype == 3:
                h_rand = rng.uniform(0, 1, n)
                h_rand = h_rand / h_rand.sum() * target_sum
                alpha = rng.uniform(0.2, 0.6)
                h_pert = alpha * best_h + (1 - alpha) * h_rand
            else:
                h_pert = best_h + rng.normal(0, 0.1, n)

        h_pert = project_feasible(h_pert)

        # SLSQP with log-sum-exp smoothing
        temp = max(0.003, 0.01 * np.exp(-0.03 * iteration))
        try:
            res = minimize(lambda x: sobj(x, temp), h_pert, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 200, 'ftol': 1e-14})
            h_opt = project_feasible(res.x)
            c5_opt = eval_c5(h_opt)
        except Exception:
            h_opt = h_pert
            c5_opt = eval_c5(h_pert)

        if c5_opt < best_c5 - 1e-12:
            best_h = h_opt
            best_c5 = c5_opt

        # Local refinement: active-set gradient step
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.82:
            break
        h_active = best_h.copy()
        for _ in range(30):
            corr = np.correlate(h_active, 1.0 - h_active, mode='full') * dx
            k_star = int(np.argmax(corr))
            # Numerical gradient of corr[k_star]
            eps = 1e-7
            grad = np.zeros(n)
            for j in range(n):
                h_active[j] += eps
                c_plus = float(np.correlate(h_active, 1.0 - h_active, mode='full')[k_star] * dx)
                h_active[j] -= 2 * eps
                c_minus = float(np.correlate(h_active, 1.0 - h_active, mode='full')[k_star] * dx)
                h_active[j] += eps
                grad[j] = (c_plus - c_minus) / (2 * eps)
            # Subgradient of max includes all near-max lags too
            top_k = np.argsort(corr)[-3:]
            for kk in top_k:
                if kk == k_star:
                    continue
                g2 = np.zeros(n)
                for j in range(n):
                    h_active[j] += eps
                    cp = float(np.correlate(h_active, 1.0 - h_active, mode='full')[kk] * dx)
                    h_active[j] -= 2 * eps
                    cm = float(np.correlate(h_active, 1.0 - h_active, mode='full')[kk] * dx)
                    h_active[j] += eps
                    g2[j] = (cp - cm) / (2 * eps)
                grad += 0.5 * g2

            # Projected gradient step
            lr = 0.01
            step = -lr * grad
            h_new = h_active + step
            h_new = project_feasible(h_new)
            c_new = eval_c5(h_new)
            if c_new < eval_c5(h_active) - 1e-14:
                h_active = h_new
                if c_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c_new
            else:
                lr *= 0.5
                if lr < 1e-6:
                    break
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.82:
                break

    best_h = project_feasible(best_h)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, n
