    """Compute smooth max of correlate values and its gradient."""

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def sla_objective(h, dx, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    s = np.sum(np.exp((corr - cmax) / K))
    return (cmax + K * np.log(s)) * dx

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

Let me restructure to avoid closures and lambdas.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def sla_obj(h, dx, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    Z = np.sum(np.exp((corr - cmax) / K))
    return (cmax + K * np.log(Z)) * dx

def sla_jac(h, dx, K):
    n = len(h)
    oh = 1.0 - h
    corr = np.correlate(h, oh, mode="full")
    cmax = np.max(corr)
    exps = np.exp((corr - cmax) / K)
    Z = np.sum(exps)
    w = exps / Z
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        L = idx - (n - 1)
        if L >= 0:
            js1 = np.arange(0, n - L)
            grad[js1] += wi * oh[js1 + L]
            js2 = np.arange(L, n)
            grad[js2] -= wi * h[js2 - L]
        else:
            LA = -L
            js2 = np.arange(LA, n)
            grad[js2] += wi * oh[js2 - LA]
            js1 = np.arange(0, n - LA)
            grad[js1] -= wi * h[js1 + LA]
    
    return grad * dx

def project_sum(h, target):
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(free):
            h[free] += diff2 / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

    return h.sum() - target
