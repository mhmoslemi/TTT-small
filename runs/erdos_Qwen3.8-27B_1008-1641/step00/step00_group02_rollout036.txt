import time
from scipy.optimize import minimize

    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(h):
        c = np.correlate(h, 1.0 - h, 'full') * dx
        return c.max()
    
    def proj(h):
        lo, hi = -1.0, 2.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        return np.clip(h - (lo + hi) / 2.0, 0, 1)
    
    def grad_c5(h):
        c = np.correlate(h, 1.0 - h, 'full')
        k = int(np.argmax(c))
        s = k - (n - 1)
        g = np.zeros(n)
        # First term: (1-h[j-s]) where 0<=j-s<n
        j1 = max(0, s)
        j1e = min(n, n + s)
        if j1 < j1e:
            g[j1:j1e] += (1.0 - h[j1-s:j1e-s])
        # Second term: -h[j+s] where 0<=j+s<n
        j2 = max(0, -s)
        j2e = min(n, n - s)
        if j2 < j2e:
            g[j2:j2e] -= h[j2+s:j2e+s]
        return g * dx
    
    h_best = proj(h0)
    best_c5 = obj(h_best)
    
    # Subgradient descent
    for restart in range(20):
        if time.time() - t0 > budget_s - 60:
            break
        if restart == 0:
            h = h_best.copy()
        else:
            h = proj(h_best + rng.normal(0, 0.1, n))
        
        alpha = 0.1  # initial step size
        for it in range(5000):
            if time.time() - t0 > budget_s - 60:
                break
            g = grad_c5(h)
            # Ensure gradient is not zero
            gn = np.linalg.norm(g)
            if gn < 1e-12:
                break
            # Step
            h_new = proj(h - alpha * g)
            c5_new = obj(h_new)
            c5_old = obj(h)
            if c5_new < c5_old:

                alpha = min(alpha * 1.05, 1.0)  # increase step

                alpha *= 0.7  # decrease step

            if c5_new < best_c5:

                h_best = h_new.copy()

            h_opt = proj(res.x)

                best_c5 = obj(h_opt)
                h_best = h_opt.copy()
        except:
            pass
    
    return h_best, best_c5, n

d/dh [ (1/ε) log(sum_k exp(ε * f_k(h))) ] = sum_k [ε * f_k'(h) * exp(ε * f_k) / sum_j exp(ε * f_j)]
= sum_k [f_k'(h) * softmax_k]

where softmax_k = exp(ε * f_k) / sum_j exp(ε * f_j).
