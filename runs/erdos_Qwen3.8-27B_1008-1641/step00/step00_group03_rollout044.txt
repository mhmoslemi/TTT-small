    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (target,)}]
    res = minimize(exact_obj, h_start, args=(target,),
                   method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0

    h0 = np.array(initial_h_values, dtype=float)
    h0 = project_sum(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Smooth softmax optimization with multiple restarts
    alphas = [15, 40, 100, 250]
    for r in range(6):
        if budget_s - (time.time() - t0) < 80:
            break
        if r == 0:
            h_cur = best_h.copy()
        else:
            h_cur = best_h + 0.03 * rng.randn(n)
            h_cur = project_sum(h_cur, target)
        for alpha in alphas:
            if budget_s - (time.time() - t0) < 40:
                break
            try:
                res = run_smooth(h_cur, target, alpha, bounds, 300)
                if res.x is not None:
                    h_new = project_sum(res.x, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h_cur = h_new.copy()
            except Exception:
                continue

    # Phase 2: Direct optimization of exact max (no smoothing)
    if budget_s - (time.time() - t0) > 30:
        try:
            res = run_exact(best_h, target, bounds, 200)
            if res.x is not None:
                h_new = project_sum(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except Exception:
            pass

    # Phase 3: Fine-tuning restarts
    for r in range(5):
        if budget_s - (time.time() - t0) < 25:
            break
        h_cur = best_h + 0.015 * rng.randn(n)
        h_cur = project_sum(h_cur, target)
        for alpha in [100, 250, 500]:
            if budget_s - (time.time() - t0) < 15:
                break
            try:
                res = run_smooth(h_cur, target, alpha, bounds, 250)
                if res.x is not None:
                    h_new = project_sum(res.x, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
            except Exception:
                continue

    # Phase 4: Final exact polish
    if budget_s - (time.time() - t0) > 20:
        try:
            res = run_exact(best_h, target, bounds, 150)
            if res.x is not None:
                h_new = project_sum(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except Exception:
            pass

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
