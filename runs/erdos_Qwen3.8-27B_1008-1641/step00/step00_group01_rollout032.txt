import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    order = np.argsort(h) if diff > 0 else np.argsort(-h)
    for i in order:
        if abs(diff) < 1e-14:
            break
        if diff > 0 and h[i] < 1.0:
            a = min(diff, 1.0 - h[i])
            h[i] += a; diff -= a
        elif diff < 0 and h[i] > 0.0:
            s = min(-diff, h[i])
            h[i] -= s; diff += s
    return h

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def eval_c5(h, n):
    dx = 2.0 / n
    c = corr_full(h)
    return dx * float(c.max())

def smooth_obj(h, n, beta):
    dx = 2.0 / n
    c = corr_full(h)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    return dx * (cm + np.log(e.sum()) / beta)

def subgrad_corr(h, n, k_star):
    tau = k_star - (n - 1)
    g = np.zeros(n)
    if tau >= 0:
        idx1 = np.arange(0, n - tau)
        g[idx1] += (1.0 - h[idx1 + tau])
        idx2 = np.arange(tau, n)
        g[idx2] -= h[idx2 - tau]
    else:
        t = -tau
        idx1 = np.arange(t, n)
        g[idx1] += (1.0 - h[idx1 - t])
        idx2 = np.arange(0, n - t)
        g[idx2] -= h[idx2 + t]
    return g

    n_candidates = [48, 49, 50, 52, 54, 56, 60, 64, 72, 80, 96, 100, 128]
    for n in n_candidates:
        h_b = np.where((np.arange(n) * alpha_gold) % 1.0 < 0.5, 1.0, 0.0)
        h_b = project_feasible(h_b, n)
        h_i = np.interp(np.linspace(0, 2.0, n, endpoint=False),

        h_i = project_feasible(h_i, n)

    # Optimize each initialization

                        h_bin[flip[_]] = 0.0
                else:
                    flip = np.where(h_bin == 0.0)[0]
                    for _ in range(n // 2 - n_ones):
                        h_bin[flip[_]] = 1.0
            c5_bin = eval_c5(h_bin, n)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()

    # Final polish on best solution
    if rem() > 5:
        n_b = len(best_h)
        for ss in [0.008, 0.004, 0.002, 0.001]:
            for _ in range(400):
                if rem() < 3:
                    break
                h_p = best_h + rng.normal(0.0, ss, n_b)
                h_p = project_feasible(h_p, n_b)
                c5_p = eval_c5(h_p, n_b)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p

    return best_h, float(best_c5), int(len(best_h))
