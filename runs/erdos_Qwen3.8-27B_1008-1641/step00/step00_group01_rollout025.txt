from scipy.optimize import minimize

def project_h(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2} via bisection on shift."""
    target = n / 2.0
    lo = target / n - v.max() - 1.0
    hi = target / n - v.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) / 2.0
    return np.clip(v + t, 0.0, 1.0)

def c5_eval(h, n):
    """Return C5 = max(correlate(h,1-h,full)*dx)."""
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()

def c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def c5_subgrad(h, n, n_active=5):
    """Subgradient from top-n_active active constraints."""
    vals = c5_all(h, n)
    max_v = vals.max()
    active = np.where(vals >= max_v - 1e-10)[0]
    if len(active) > n_active:
        active = np.argsort(vals)[-n_active:]
    j = np.arange(n)
    grad = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        v1 = (j + shift >= 0) & (j + shift < n)
        v2 = (j - shift >= 0) & (j - shift < n)
        term = np.zeros(n)
        term[v1] += 1.0 - h[j[v1] + shift]
        term[v2] -= h[j[v2] - shift]
        grad += term
    grad *= (2.0 / n) / len(active)
    return grad, max_v

        term[v1] += 1.0 - x[j[v1] + shift]
        term[v2] -= x[j[v2] - shift]
        grad += w[k] * term
    grad *= dx
    return obj, grad

    """Nesterov-accelerated subgradient descent."""
    y = h.copy()
    beta = 2.0 / (t_start + 3.0)
    h_best = h.copy()
    c5_best = c5_eval(h, n)
    for t in range(1, n_iter + 1):
        grad, _ = c5_subgrad(y, n)
        step = 1.0 / (np.sqrt(t) + t_start)
        h_new = project_h(y - step * grad, n)
        beta = 2.0 / (t + 3.0)
        y = h_new + beta / (beta_prev) * (h_new - h)
        y = project_h(y, n)
        h = h_new
        c5_cur = c5_eval(h, n)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()
        beta_prev = beta
    return h_best, c5_best
