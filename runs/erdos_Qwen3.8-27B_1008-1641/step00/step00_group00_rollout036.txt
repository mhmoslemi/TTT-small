def soft_lse_grad(x, tau):
    x = np.asarray(x, dtype=float).ravel()
    n=x.size
    if not np.all(np.isfinite(x)):
        return np.zeros(n)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    if tau <= 1e-8:
       p = np.zeros(corr.size)
       mask = np.abs(corr - m) <= 1e-10
       p[mask] = 1.0 / max(1, int(np.count_nonzero(mask)))
    else:
       z = tau * (corr - m)
       p = np.exp(z)
       ps = float(np.sum(p))
       if ps > 0.0:
          p /= ps
       else:
          p = np.zeros(corr.size); p[int(np.argmax(corr))]=1.0
    g=np.zeros(n)
    for k in range(corr.size):
       pk=float(p[k])
       if pk > 1e-12:
          grad_lag_add(g, x, n - 1 - k, pk)
    return (2.0 / n) * g

import time
import numpy as np
import scipy.optimize as sco


def c5_eval(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return 0.0
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def enforce_sum(h, target=None):
    x = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    if target is None:
        target = 0.5 * n
    for _ in range(4):
        res = target - float(np.sum(x))
        if abs(res) < 1e-13:
            break
        if res > 0.0:
            slack = 1.0 - x
            idx = np.argsort(-slack)
            for ii in idx:
                if res <= 0.0:
                    break
                add = min(res, float(slack[ii]))
                if add > 0.0:
                    x[ii] += add
                    res -= add
        else:
            idx = np.argsort(x)
            for ii in idx:
                if res >= 0.0:
                    break
                sub = min(-res, float(x[ii]))
                if sub > 0.0:
                    x[ii] -= sub
                    res += sub
    res = target - float(np.sum(x))
    if abs(res) > 1e-15:
        if res > 0.0:
            ii = int(np.argmax(1.0 - x))
            if 1.0 - x[ii] >= abs(res):
                x[ii] += res
        else:
            ii = int(np.argmax(x))
            if x[ii] >= abs(res):
                x[ii] += res
    return x


def project_box_sum(x, target=None):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x
    if target is None:
        target = 0.5 * n
    y = np.clip(x, 0.0, 1.0)
    if abs(float(np.sum(y)) - target) <= 1e-12:
        return y
    lo, hi = -1.0, 1.0
    slo = float(np.sum(np.clip(x + lo, 0.0, 1.0)))
    shi = float(np.sum(np.clip(x + hi, 0.0, 1.0)))
    while slo > target and lo > -1e6:
        lo *= 2.0
        slo = float(np.sum(np.clip(x + lo, 0.0, 1.0)))
    while shi < target and hi < 1e6:
        hi *= 2.0
        shi = float(np.sum(np.clip(x + hi, 0.0, 1.0)))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        sm = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if sm > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)


def grad_lag_add(out, h, s, w):
    n = h.size
    if s >= 0:
        m = n - s
        if m <= 0:
            return
        out[:m] += w
        out[:m] -= w * h[s:]
        out[s:] -= w * h[:m]
    else:
        t = -s
        m = n - t
        if m <= 0:
            return
        out[t:] += w
        out[t:] -= w * h[:-t]
        out[:-t] -= w * h[t:]


def soft_lse_grad(x, tau):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    if tau <= 1e-8:
        mask = np.abs(corr - m) <= 1e-10
        p = np.zeros(corr.size, dtype=float)
        cnt = max(1, int(np.count_nonzero(mask)))
        p[mask] = 1.0 / cnt
    else:
        z = tau * (corr - m)
        p = np.exp(z)
        ps = float(np.sum(p))
        if ps > 0.0:
            p /= ps
        else:
            p = np.zeros(corr.size, dtype=float)
            p[int(np.argmax(corr))] = 1.0
    g = np.zeros(n, dtype=float)
    for k in range(corr.size):
        pk = float(p[k])
        if pk > 1e-12:
            grad_lag_add(g, x, n - 1 - k, pk)
    return (2.0 / n) * g


def slsqp_fun(x, tau):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return 0.0
    if not np.all(np.isfinite(x)):
        return 1e100
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    dx = 2.0 / n
    if tau <= 1e-8:
        return m * dx
    z = tau * (corr - m)
    return float((m + float(np.log(float(np.exp(z).sum()))) / tau) * dx)


def slsqp_jac(x, tau):
    return soft_lse_grad(x, tau)


def slsqp_sum(x):
    x = np.asarray(x, dtype=float).ravel()
    return float(np.sum(x) - 0.5 * x.size)


def slsqp_optimize(start, tau, maxiter):
    x0 = enforce_sum(start)
    n = x0.size
    if n == 0 or maxiter <= 0:
        return x0
    bounds = [(0.0, 1.0)] * n
    try:
        res = sco.optimize.minimize(
            slsqp_fun,
            x0,
            method='SLSQP',
            jac=slsqp_jac,
            args=(tau,),
            bounds=bounds,
            constraints={'type': 'eq', 'fun': slsqp_sum},
            options={'maxiter': int(maxiter), 'ftol': 1e-10, 'disp': False}
        )
        if hasattr(res, 'x') and res.x is not None:
            return np.asarray(res.x, dtype=float).ravel()
    except Exception:
        pass
    return x0


def scan_exchange(h, time_s, tol=1e-14):
    h = enforce_sum(np.array(h, dtype=float, copy=True))
    n = h.size
    if n < 2 or time_s <= 0.0:
        return h, c5_eval(h)
    cur = c5_eval(h)
    besth = h.copy()
    bestc = cur
    limit = time.time() + float(time_s)
    fracs = (1.0, 0.5, 0.25, 0.1, 0.03)
    while time.time() < limit - 0.05:
        improved = False
        for i in range(n - 1):
            if time.time() > limit - 0.1:
                return besth, bestc
            for j in range(i + 1, n):
                if time.time() > limit - 0.05:
                    return besth, bestc
                hi, hj = h[i], h[j]
                if hi >= hj:
                    a, b = i, j
                else:
                    a, b = j, i
                did = False
                maxd = min(float(h[a]), float(1.0 - h[b]))
                if maxd > 1e-12:
                    oa, ob = float(h[a]), float(h[b])
                    for fr in fracs:
                        d = maxd * fr
                        if d <= 1e-13:
                            continue
                        h[a] = oa - d
                        h[b] = ob + d
                        f = c5_eval(h)
                        if f < cur - tol:
                            cur = f
                            improved = True
                            did = True
                            if cur < bestc:
                                bestc = cur
                                besth = h.copy()
                            break
                    if not did:
                        h[a] = oa
                        h[b] = ob
                if not did:
                    a2, b2 = b, a
                    maxd = min(float(h[a2]), float(1.0 - h[b2]))
                    if maxd > 1e-12:
                        oa, ob = float(h[a2]), float(h[b2])
                        for fr in (0.25, 0.1, 0.03):
                            d = maxd * fr
                            if d <= 1e-13:
                                continue
                            h[a2] = oa - d
                            h[b2] = ob + d
                            f = c5_eval(h)
                            if f < cur - tol:
                                cur = f
                                improved = True
                                did = True
                                if cur < bestc:
                                    bestc = cur
                                    besth = h.copy()
                                break
                        if not did:
                            h[a2] = oa
                            h[b2] = ob
                if did:
                    pass
        if not improved:
            break
    return besth, bestc


def projected_gradient_local(h, time_s, rng):
    h = enforce_sum(np.array(h, dtype=float, copy=True))
    cur = c5_eval(h)
    besth = h.copy()
    bestc = cur
    limit = time.time() + float(time_s)
    eta = 0.08
    tau = 20.0
    stag = 0
    pfails = 0
    while time.time() < limit - 0.2:
        g = soft_lse_grad(h, tau)
        d = -(g - float(np.mean(g)))
        nd = float(np.linalg.norm(d))
        if nd < 1e-12:
            break
        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.1, 2.0, 4.0):
            step = (eta / nd) * factor
            maxstep = 0.25 / nd
            if step > maxstep:
                step = maxstep
            if step <= 0.0:
                continue
            v = project_box_sum(h + step * d, 0.5 * h.size)
            vf = c5_eval(v)
            if vf < cur - 1e-13:
                h = v
                cur = vf
                accepted = True
                if factor >= 1.0:
                    eta = min(0.25, eta * 1.15)
                else:
                    eta = max(0.002, eta * 0.85)
                break
        if accepted:
            stag = 0
            pfails = 0
            tau = min(150.0, tau * 1.15)
            if cur < bestc:
                bestc = cur
                besth = h.copy()
        else:
            stag += 1
            eta = max(0.002, eta * 0.75)
            tau = max(3.0, tau * 0.8)
            if stag > 30:
                noise = rng.normal(scale=0.03, size=h.size)
                noise -= float(np.mean(noise))
                v = project_box_sum(h + noise, 0.5 * h.size)
                vf = c5_eval(v)
                if vf <= bestc + 0.03:
                    h = v
                    cur = vf
                    eta = 0.05
                    tau = 8.0
                    stag = 0
                else:
                    pfails += 1
                    if pfails > 3:
                        break
    return besth, bestc


def random_exchange(h, time_s, rng):
    h = enforce_sum(np.array(h, dtype=float, copy=True))
    n = h.size
    if n < 2 or time_s <= 0.0:
        return h, c5_eval(h)
    cur = c5_eval(h)
    besth = h.copy()
    bestc = cur
    limit = time.time() + float(time_s)
    choices = (0.01, 0.02, 0.05, 0.1, 0.2, 0.5)
    noimp = 0
    fails = 0
    while time.time() < limit - 0.05:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            j = (j + 1) % n
        if h[i] >= h[j]:
            a, b = i, j
        else:
            a, b = j, i
        if rng.random() < 0.25:
            a, b = b, a
        maxd = min(float(h[a]), float(1.0 - h[b]))
        if maxd <= 1e-12:
            continue
        if rng.random() < 0.1:
            d = maxd * float(rng.uniform(0.001, 0.2))
        else:
            d = maxd * choices[int(rng.integers(0, len(choices)))]
        if d > maxd:
            d = maxd
        if d <= 1e-12:
            continue
        oa, ob = float(h[a]), float(h[b])
        h[a] = oa - d
        h[b] = ob + d
        f = c5_eval(h)
        if f < cur - 1e-14:
            cur = f
            noimp = 0
            fails = 0
            if cur < bestc:
                bestc = cur
                besth = h.copy()
        else:
            h[a] = oa
            h[b] = ob
            noimp += 1
        if noimp > 1000:
            noise = rng.normal(scale=0.04, size=n)
            noise -= float(np.mean(noise))
            v = project_box_sum(besth + noise, 0.5 * n)
            vf = c5_eval(v)
            if vf <= bestc + 0.02:
                h = v.copy()
                cur = vf
            else:
                h = besth.copy()
                cur = bestc
                fails += 1
                if fails > 4:
                    break
            noimp = 0
    return besth, bestc


def local_optimizer(start, time_s, rng, use_slsqp=True):
    t_start = time.time()
    h = enforce_sum(np.array(start, dtype=float, copy=True))
    besth = h.copy()
    bestc = c5_eval(besth)

    rem = max(0.0, time_s - (time.time() - t_start))
    t1 = min(20.0, 0.3 * rem)
    if t1 > 2.0:
        h1, c1 = scan_exchange(besth, t1)
        if c1 < bestc:
            besth = h1
            bestc = c1

    rem = max(0.0, time_s - (time.time() - t_start))
    t2 = min(15.0, 0.25 * rem)
    if t2 > 2.0:
        h2, c2 = projected_gradient_local(besth, t2, rng)
        if c2 < bestc:
            besth = h2
            bestc = c2

    rem = max(0.0, time_s - (time.time() - t_start))
    if use_slsqp and rem > 60 and besth.size <= 300:
        n = besth.size
        if n <= 60:
            maxit = 70 if rem > 160 else 40 if rem > 80 else 25
        elif n <= 150:
            maxit = 45 if rem > 160 else 28 if rem > 80 else 18
        else:
            maxit = 25 if rem > 160 else 16 if rem > 80 else 10
        for tau in (12.0, 50.0):
            rem = max(0.0, time_s - (time.time() - t_start))
            if rem < 40:
                break
            x1 = slsqp_optimize(besth, tau, maxit)
            x1 = enforce_sum(x1)
            c1 = c5_eval(x1)
            if c1 < bestc:
                besth = x1
                bestc = c1
            rem = max(0.0, time_s - (time.time() - t_start))
            if rem > 8:
                t3 = min(8.0, 0.25 * rem)
                h3, c3 = scan_exchange(besth, t3)
                if c3 < bestc:
                    besth = h3
                    bestc = c3

    rem = max(0.0, time_s - (time.time() - t_start))
    if rem > 8 and besth.size <= 500:
        tr = min(15.0, 0.3 * rem)
        hr, cr = random_exchange(besth, tr, rng)
        if cr < bestc:
            besth = hr
            bestc = cr

    rem = max(0.0, time_s - (time.time() - t_start))
    if rem > 5:
        ts = min(30.0, max(1.0, rem - 2.0))
        hs, cs = scan_exchange(besth, ts)
        if cs < bestc:
            besth = hs
            bestc = cs

    rem = max(0.0, time_s - (time.time() - t_start))
    if rem > 8 and besth.size <= 200:
        tg = min(12.0, max(1.0, rem - 2.0))
        hg, cg = projected_gradient_local(besth, tg, rng)
        if cg < bestc:
            besth = hg
            bestc = cg

    besth = enforce_sum(besth)
    return besth, c5_eval(besth)


def binaryize_from_h(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    target = 0.5 * n
    k = int(np.floor(target))
    r = target - k
    b = np.zeros(n, dtype=float)
    order = np.argsort(-h)
    pos = 0
    if r > 1e-12:
        b[order[pos]] = r
        pos += 1
    if k > pos:
        b[order[pos:pos + k]] = 1.0
    return b


def random_binary_start(n, rng):
    target = 0.5 * n
    k = int(np.floor(target))
    r = target - k
    h = np.zeros(n, dtype=float)
    order = rng.permutation(n)
    pos = 0
    if r > 1e-12:
        h[order[pos]] = r
        pos += 1
    if k > pos:
        h[order[pos:pos + k]] = 1.0
    return h


def random_smooth_start(n, rng):
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = 0.5 * np.ones(n, dtype=float)
    nhar = int(rng.integers(2, 5))
    for m in range(1, nhar + 1):
        amp = float(rng.uniform(0.05, 0.45))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        if rng.random() < 0.5:
            h += amp * np.sin(m * np.pi * x + ph)
        else:
            h += amp * np.cos(m * np.pi * x + ph)
    h = 0.5 + 0.5 * np.tanh(h - 0.5)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, 0.5 * n)


def make_ordered_starts(parent_h, rng):
    n0 = parent_h.size if parent_h is not None else 45
    raw = []
    if parent_h is not None:
        raw.append(parent_h)
        for scale in (0.02, 0.05, 0.1, 0.2, 0.4):
            raw.append(enforce_sum(parent_h + scale * rng.normal(size=n0)))
        raw.append(binaryize_from_h(parent_h))
    else:
        raw.append(random_binary_start(n0, rng))
    for _ in range(5):
        raw.append(random_binary_start(n0, rng))
    for _ in range(5):
        raw.append(random_smooth_start(n0, rng))

    ordered = []
    if parent_h is not None:
        ordered.append(parent_h)
    for h in raw:
        if h.size != n0:
            continue
        dup = False
        for q in ordered:
            if q.size == h.size and float(np.sum((h - q) ** 2)) < 1e-12:
                dup = True
                break
        if not dup:
            ordered.append(h)

    if parent_h is not None:
        others = []
        for idx, h in enumerate(ordered[1:]):
            c = c5_eval(h)
            if np.isfinite(c):
                others.append((float(c), idx, h))
        others.sort()
        others_h = [hh for c_i, idx, hh in others[:12]]
        return [parent_h], others_h

    scores = []
    for idx, h in enumerate(ordered):
        c = c5_eval(h)
        if np.isfinite(c):
            scores.append((float(c), idx, h))
    scores.sort()
    if not scores:
        return [random_binary_start(n0, rng)], []
    first = [hh for c_i, idx, hh in scores[:1]][0]
    rest = [hh for c_i, idx, hh in scores[1:13]]
    return [first], rest


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s)) - 5.0
    rng = np.random.default_rng(seed)

    parent = globals().get('initial_h_values')
    parent_h = None
    if parent is not None:
        parent_h = enforce_sum(np.asarray(parent, dtype=float))
        if parent_h.size == 0:
            parent_h = None

    first_list, other_starts = make_ordered_starts(parent_h, rng)
    if not first_list:
        first_list = [random_binary_start(45, rng)]
        other_starts = []

    besth = first_list[0].copy()
    bestc = c5_eval(besth)
    for h in other_starts:
        c0 = c5_eval(h)
        if c0 < bestc:
            besth = h.copy()
            bestc = c0

    rem = max(0.0, deadline - time.time())
    if rem > 40:
        alloc = min(rem - 40.0, 280.0)
        rh, rc = local_optimizer(first_list[0], alloc, rng, True)
        if rc < bestc:
            besth = rh
            bestc = rc

    for pos, h in enumerate(other_starts):
        rem = max(0.0, deadline - time.time())
        if rem < 50:
            break
        if pos == 0:
            alloc = min(rem - 40.0, 160.0)
            use_sl = True
        else:
            alloc = min(rem - 30.0, 45.0)
            use_sl = False
        if alloc <= 10.0:
            break
        rh, rc = local_optimizer(h, alloc, rng, use_sl)
        if rc < bestc:
            besth = rh
            bestc = rc

    rem = max(0.0, deadline - time.time())
    n = int(besth.size)
    if rem > 150 and n <= 100 and n * 2 <= 480:
        h2 = enforce_sum(np.repeat(besth, 2))
        c2 = c5_eval(h2)
        if c2 < bestc:
            besth = h2
            bestc = c2
        alloc = min(rem - 40.0, 240.0)
        rh, rc = local_optimizer(h2, alloc, rng, n * 2 <= 150)
        if rc < bestc:
            besth = rh
            bestc = rc

    rem = max(0.0, deadline - time.time())
    if rem > 20:
        rh, rc = local_optimizer(besth, min(rem - 5.0, 60.0), rng, rem > 180)
        if rc < bestc:
            besth = rh
            bestc = rc

    rem = max(0.0, deadline - time.time())
    if rem > 5:
        rh, rc = scan_exchange(besth, min(rem - 2.0, 30.0))
        if rc < bestc:
            besth = rh
            bestc = rc

    besth = enforce_sum(besth)
    bestc = c5_eval(besth)
    return besth, float(bestc), int(besth.size)
