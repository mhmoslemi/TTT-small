import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(vals, t=0.01):
    m = np.max(vals)
    if t < 1e-14:
        return m
    return m + t * np.log(np.sum(np.exp((vals - m) / t)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        if step > 0:
            max_step = 1.0 - h[mask].max()
            if max_step <= 0:
                break
            step = min(step, max_step)
        else:
            min_step = -h[mask].min()
            if min_step <= 0:
                break
            step = max(step, -min_step)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def make_obj(t_smooth, n):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(corr, t_smooth) * dx
    return obj
