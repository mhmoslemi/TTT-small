import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
        else:
            free = h > 1e-12
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        js = j - lag
        jsp = j + lag
        if 0 <= js < n:
            g += 1.0 - h[js]
        if 0 <= jsp < n:
            g -= h[jsp]
        grad[j] = g
    return grad * dx

        return compute_c5(x, dx)
    
    def jac(x):
        return gradient_c5(x, n)
    
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=[cons], jac=jac,

    if res.success or res.fun < compute_c5(h0, dx):
        return make_feasible(res.x, n), res.fun
    return h0, obj(h0)

import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

_t0 = 0.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
        else:
            free = h > 1e-14
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def vec_gradient_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)
    first = np.zeros(n)
    second = np.zeros(n)
    if s > 0:
        first[s:] = 1.0 - h[:n - s]
        second[:n - s] = h[s:]
    elif s < 0:
        first[:n + s] = 1.0 - h[-s:]
        second[-s:] = h[:n + s]
    else:
        first[:] = 1.0 - h
        second[:] = h
    grad = (first - second) * dx
    return grad

def subgradient_c5(h, n):
    """Average gradient over near-active lags for smoother descent."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    active = (corr >= mx - 1e-8)
    grad = np.zeros(n)
    for idx in np.where(active)[0]:
        s = int(idx) - (n - 1)
        first = np.zeros(n)
        second = np.zeros(n)
        if s > 0:
            first[s:] = 1.0 - h[:n - s]
            second[:n - s] = h[s:]
        elif s < 0:
            first[:n + s] = 1.0 - h[-s:]
            second[-s:] = h[:n + s]
        else:
            first[:] = 1.0 - h
            second[:] = h
        grad += (first - second)
    grad /= (np.sum(active) * 1.0)
    return grad * dx

def project_to_sum(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    rem = target - np.sum(h_new)
    if abs(rem) > 1e-14:
        free = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if np.any(free):
            h_new[free] += rem / np.sum(free)
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def slsqp_optimize(h0, n, maxiter=300):
    dx = 2.0 / n
    bounds = Bounds(0.0, 1.0)
    A = np.ones((1, n))
    cons = LinearConstraint(A, n / 2.0, n / 2.0)

    def obj(x):
        return compute_c5(x, dx)

    def jac(x):
        return vec_gradient_c5(x, n)

    c0 = compute_c5(h0, dx)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[cons], jac=jac,
                       tol=1e-12, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_res = make_feasible(res.x, n)
        c_res = compute_c5(h_res, dx)
        if c_res < c0:
            return h_res, c_res
    except Exception:
        pass
    return h0, c0

def pgdescent(h0, n, lr=0.03, steps=3000, t_deadline=None):
    dx = 2.0 / n
    best_h = h0.copy()
    best_c = compute_c5(best_h, dx)
    h = h0.copy()
    for step in range(steps):
        if t_deadline is not None and time.time() > t_deadline:
            break
        g = subgradient_c5(h, n)
        h_new = h - lr * g
        h_new = project_to_sum(h_new, n)
        c = compute_c5(h_new, dx)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_new.copy()
            h = h_new.copy()
            lr = min(lr * 1.02, 0.15)
        else:
            lr *= 0.7
            if lr < 1e-7:
                break
    return best_h, best_c

def random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def random_fuzzy(n, rng, center, sigma):
    h = center + rng.normal(0, sigma, n)
    return make_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0
    _t0 = time.time()
    rng = np.random.RandomState(seed)
    n_pts = 70
    dx = 2.0 / n_pts
    deadline = _t0 + budget_s

    # Initial point
    try:
        h_init = np.array(initial_h_values, dtype=float)
        if len(h_init) != n_pts:
            h_init = np.interp(np.linspace(0, len(h_init) - 1, n_pts),
                               np.arange(len(h_init)), h_init)
        h_init = make_feasible(h_init, n_pts)
    except Exception:
        h_init = np.ones(n_pts) * 0.5

    best_h = h_init.copy()
    best_c = compute_c5(best_h, dx)

    def remaining():
        return deadline - time.time()

    # Phase 1: SLSQP from initial
    if remaining() > 120:
        h_new, c_new = slsqp_optimize(h_init, n_pts, maxiter=400)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Phase 2: Projected gradient from best
    if remaining() > 100:
        h_new, c_new = pgdescent(best_h, n_pts, lr=0.05, steps=3000,
                                  t_deadline=deadline - 80)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Phase 3: SLSQP + PG restarts
    n_restarts = 40
    for r in range(n_restarts):
        if remaining() < 60:
            break
        # Diverse starting points
        if r == 0:
            h_start = h_init.copy()
        elif r % 4 == 1:
            h_start = best_h + rng.uniform(-0.08, 0.08, n_pts)
            h_start = make_feasible(h_start, n_pts)
        elif r % 4 == 2:
            h_start = random_binary(n_pts, rng)
        elif r % 4 == 3:
            h_start = random_fuzzy(n_pts, rng, best_h, 0.06)
        else:
            # Shift-modulated binary
            h_start = random_binary(n_pts, rng)
            h_start[:(rng.randint(1, n_pts))] += rng.uniform(0, 0.3)
            h_start = make_feasible(h_start, n_pts)

        # SLSQP
        h_new, c_new = slsqp_optimize(h_start, n_pts, maxiter=250)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

        # PG follow-up if improving
        if c_new < best_c + 0.005 and remaining() > 50:
            h_new2, c_new2 = pgdescent(h_new, n_pts, lr=0.02, steps=1500,
                                        t_deadline=deadline - 40)
            if c_new2 < best_c:
                best_h, best_c = h_new2, c_new2

    # Phase 4: Fine SLSQP polish on best
    if remaining() > 40:
        h_new, c_new = slsqp_optimize(best_h, n_pts, maxiter=500)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Phase 5: Final PG polish
    if remaining() > 20:
        h_new, c_new = pgdescent(best_h, n_pts, lr=0.005, steps=2000,
                                  t_deadline=deadline - 2)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Ensure feasibility
    best_h = make_feasible(best_h, n_pts)
    best_c = compute_c5(best_h, dx)

    return best_h, best_c, n_pts
