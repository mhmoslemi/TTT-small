import numpy as np
import time
from scipy.optimize import minimize


def c5_of_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())


def make_feasible(h, n, max_iter=20):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_1d_eval(h, j, x, dx, target_sum, lam):
    h[j] = x
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max() + lam * (h.sum() - target_sum) ** 2


def coord_descent(h, n, dx, target_sum, max_sweeps, lam, seed, deadline):
    rng = np.random.RandomState(seed)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    best_c5 = c5_of_h(h)
    best = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for j in order:
            a, b = 0.0, 1.0
            c = b - invphi * (b - a)
            d = a + invphi * (b - a)
            fc = obj_1d_eval(h, j, c, dx, target_sum, lam)
            fd = obj_1d_eval(h, j, d, dx, target_sum, lam)
            for _ in range(28):
                if b - a < 1e-10:
                    break
                if fc <= fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - invphi * (b - a)
                    fc = obj_1d_eval(h, j, c, dx, target_sum, lam)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + invphi * (b - a)
                    fd = obj_1d_eval(h, j, d, dx, target_sum, lam)
            h[j] = (a + b) * 0.5

        h = make_feasible(h, n)
        c5 = c5_of_h(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break
    return best, best_c5


def perturb_h(h, n, rng, mag):
    hp = h.copy()
    k = max(1, n // 5)
    idx = rng.choice(n, size=k, replace=False)
    hp[idx] += rng.uniform(-mag, mag, size=k)
    return make_feasible(hp, n)


def smooth_obj(x, dx, t_smooth, target_sum, pen):
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = corr - m
    lse = m + np.log(np.exp(t_smooth * shifted).sum()) / t_smooth
    return lse + pen * (x.sum() - target_sum) ** 2

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
