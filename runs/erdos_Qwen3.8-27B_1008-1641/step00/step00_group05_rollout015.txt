def get_correlation_shifts(h):
    """Returns array of c_k * dx for all shifts."""
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    c = np.correlate(h, v, mode='full')
    return c * dx

def compute_c5(h):
    c = get_correlation_shifts(h)
    return np.max(c)

def smooth_obj(h, beta):
    c = get_correlation_shifts(h)
    return logsumexp(beta * c) / beta

    dx = 2.0 / n
    v = 1.0 - h
    c = np.correlate(h, v, mode='full') * dx
    # weights
    if beta > 50:
        max_c = np.max(c)
        exp_bc = np.exp(beta * (c - max_c))
        w = exp_bc / np.sum(exp_bc)
    else:
        exp_bc = np.exp(beta * c)
        w = exp_bc / np.sum(exp_bc)
    
    # c has 2n-1 elements, indexed 0 to 2n-2
    # c[n-1+k] corresponds to shift k
    grad = np.zeros(n)

- c_{-k} = (number of i where h[i]=1, h[i-k]=0) * dx for k>0

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return np.max(get_corr(h))

def smooth_objective(h, beta, pen, ts):
    c = get_corr(h)
    obj = logsumexp(beta * c) / beta
    obj += pen * (np.sum(h) - ts) ** 2
    return obj

def smooth_gradient(h, beta, pen, ts):
    n = len(h)
    dx = 2.0 / n
    c = get_corr(h)
    bc = beta * c
    bc -= np.max(bc)
    w = np.exp(bc)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wi = w[n - 1 + k] * dx
        if wi < 1e-16:
            continue
        if k >= 0:
            if n - k > 0:
                m1 = np.arange(0, n - k)
                grad[m1] += wi * (1.0 - h[m1 + k])
            if k < n:
                m2 = np.arange(k, n)
                grad[m2] += wi * (-h[m2 - k])
        else:
            l = -k
            if l < n:
                m1 = np.arange(l, n)
                grad[m1] += wi * (1.0 - h[m1 - l])
            if n - l > 0:
                m2 = np.arange(0, n - l)
                grad[m2] += wi * (-h[m2 + l])
    grad += 2.0 * pen * (np.sum(h) - ts)
    return grad

def project_sum(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        d = ts - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not np.any(m):
            break
        h[m] = np.clip(h[m] + d / m.sum(), 0.0, 1.0)
    return h

def optimize_phase(h_init, beta, pen, ts, bounds, max_iter=3000):
    res = minimize(
        smooth_objective, smooth_gradient, h_init,
        args=(beta, pen, ts),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': 8000}
    )
    return project_sum(res.x, ts)

    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive smoothing on initial n
    bounds = [(0.0, 1.0)] * n
    configs = [(3, 500), (5, 1000), (10, 5e3), (20, 1e4), (50, 1e4), (100, 1e5), (300, 1e5), (1000, 1e6)]
    for beta, pen in configs:

    # Phase 2: Try Different n values with random starts
