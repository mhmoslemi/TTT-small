            if f_try < f_smooth - 1e-12:
                h = h_try
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-8:
                    best_c5 = c5_try
                    best_h = h.copy()
                accepted = True
                break
        if accepted:
            stuck_count = 0
        else:
            stuck_count += 1
            step *= 0.7
            if step < 1e-7:
                beta = min(beta * 1.8, 200.0)
                step = 0.08
                stuck_count = 0
            if stuck_count > 40:
                h = best_h + rng.randn(n) * 0.02
                h = project_onto_constraints(h, n)
                step = 0.05
                stuck_count = 0
        if int(time.time() - start_t) % 3 == 0:
            beta = min(beta * 1.02, 200.0)
    return best_h, best_c5

def smooth_obj_eval(h, beta, n, offsets, npmax):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr
    mx = np.max(scaled)
    exp_s = np.exp(scaled - mx)
    Z = np.sum(exp_s)
    f = dx * (mx / beta + np.log(Z) / beta)
    return f, None

def make_periodic_init(n, period, seed):
    rng = np.random.RandomState(seed)
    base = np.array([1.0, 0.3, 0.7, 0.4, 0.8, 0.2, 0.6, 0.5][:period])
    base = base / base.sum() * (n / period * (n / 2.0) / n)
    reps = n // period + 1
    h = np.tile(base, reps)[:n]
    h += rng.randn(n) * 0.05
    return project_onto_constraints(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    n_parent = 47
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(n_parent, 0.5)
    h_parent = project_onto_constraints(h_parent, n_parent)
    c5_parent = compute_c5(h_parent)
    best_h = h_parent.copy()
    best_c5 = c5_parent
    best_n = n_parent

    remaining = budget_s - (time.time() - t0) - 60
    if remaining < 60:
        return best_h, best_c5, best_n

    # Phase 1: optimize from parent with various betas
    alloc1 = int(remaining * 0.45)
    h1, c5_1 = optimize_from(h_parent, n_parent, alloc1, seed)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, n_parent

    remaining = budget_s - (time.time() - t0) - 60
    if remaining < 60:
        return best_h, best_c5, best_n

    # Phase 2: try different n values
    for n_try in [64, 97, 128]:
        remaining = budget_s - (time.time() - t0) - 60
        if remaining < 80:
            break
        dx_try = 2.0 / n_try
        # Upsample parent to n_try
        indices = np.linspace(0, n_parent - 1, n_try)
        lo_idx = np.floor(indices).astype(int)
        hi_idx = np.minimum(lo_idx + 1, n_parent - 1)
        frac = indices - lo_idx
        h_try = h_parent[lo_idx] * (1 - frac) + h_parent[hi_idx] * frac
        h_try = project_onto_constraints(h_try, n_try)
        alloc2 = int(min(remaining * 0.35, 300))
        h2, c5_2 = optimize_from(h_try, n_try, alloc2, seed + n_try)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n_try

    remaining = budget_s - (time.time() - t0) - 30
    if remaining < 60:
        return best_h, best_c5, best_n

    # Phase 3: random restarts from perturbed best
    n_use = best_n
    rng = np.random.RandomState(seed + 777)
    alloc3 = int(remaining * 0.7)
    n_restarts = min(4, max(1, alloc3 // 80))
    for r in range(n_restarts):
        remaining = budget_s - (time.time() - t0) - 30
        if remaining < 70:
            break
        h_pert = best_h + rng.randn(n_use) * 0.08
        h_pert = project_onto_constraints(h_pert, n_use)
        aa = int(min(remaining * 0.4, 200))
        hr, cr = optimize_from(h_pert, n_use, aa, seed + 1000 + r)
        if cr < best_c5:
            best_h, best_c5, best_n = hr, cr, n_use

    # Phase 4: final fine-tuning on best
    remaining = budget_s - (time.time() - t0) - 10
    if remaining > 40 and best_c5 < 0.99:
        hf, cf = optimize_from(best_h + np.random.RandomState(seed+1).randn(best_n) * 0.01,
                               best_n, int(remaining), seed + 2000)
        if cf < best_c5:
            best_h, best_c5, best_n = hf, cf, best_n

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraints(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if not np.isfinite(final_c5):
        best_h = h_parent
        final_c5 = c5_parent
        best_n = n_parent
    return best_h, final_c5, best_n
