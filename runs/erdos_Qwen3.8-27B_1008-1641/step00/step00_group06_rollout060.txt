import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(np.max(corr))

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lag_gradient(m, h):
    """Gradient of CORR[m] with respect to h, where CORR[m] = sum_i h[i]*(1-h[i+m])"""
    n = len(h)
    g = np.zeros(n)
    # Terms where j is the "i" index: need j+m valid
    # j+m in [0,n) => j in [-m, n-1-m], intersect [0,n-1]
    j_lo = max(0, -m)
    j_hi = min(n-1, n-1-m)
    if j_lo <= j_hi:
        # Contribution: (1 - h[j+m])
        g[j_lo:j_hi+1] += (1.0 - h[j_lo+m:j_hi+m+1])
    # Terms where j is the "i+m" index (i = j-m): need j-m valid
    # j-m in [0,n) => j in [m, m+n-1], intersect [0,n-1]
    j_lo2 = max(0, m)
    j_hi2 = min(n-1, m+n-1)
    if j_lo2 <= j_hi2:
        # Contribution: -h[j-m]
        g[j_lo2:j_hi2+1] -= h[j_lo2-m:j_hi2-m+1]
    return g

def project_feasible(h, target_sum):
    """Project onto {0<=h_i<=1, sum(h)=target_sum}"""
    n = len(h)
    if abs(h.sum() - target_sum) < 1e-10:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)

    if abs(diff) > 1e-12:

        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(np.max(corr))


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def lag_gradient(m, h):
    n = len(h)
    g = np.zeros(n)
    lo1 = max(0, -m)
    hi1 = min(n, n - m)
    if lo1 < hi1:
        g[lo1:hi1] += (1.0 - h[lo1 + m:hi1 + m])
    lo2 = max(0, m)
    hi2 = min(n, n + m)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 - m:hi2 - m]
    return g


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target_sum) < 1e-10:
        return h.copy()
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-9) & (hp < 1.0 - 1e-9)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            np.clip(hp, 0.0, 1.0, out=hp)
    return hp


def _eq_fun(x, target):
    return np.sum(x) - target


def _eq_jac(x, target):
    return np.ones(len(x))


def _quad_obj(x, m):
    n = len(x)
    total = 0.0
    for i in range(max(0, -m), min(n, n - m)):
        total += x[i] * (1.0 - x[i + m])
    return total


def _quad_grad(x, m):
    n = len(x)
    g = np.zeros(n)
    for i in range(max(0, -m), min(n, n - m)):
        g[i] += (1.0 - x[i + m])
        g[i + m] -= x[i]
    return g


def polish_lag(h, m, deadline):
    n = len(h)
    target = n / 2.0
    if time.time() > deadline:
        return h, compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (target,)}]
    try:
        res = minimize(_quad_obj, h, args=(m,), jac=_quad_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 80, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            hp = project_feasible(res.x, target)
            return hp, compute_c5(hp)
    except Exception:
        pass
    return h, compute_c5(h)


def psg_optimize(h_start, max_iters, deadline):
    h = h_start.copy()
    n = len(h)
    tgt = n / 2.0
    local_best = h.copy()
    local_val = compute_c5(h)
    alpha = 4.0 / n
    stuck = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr_full(h)
        max_c = float(corr.max())
        tol = max(1e-5, 0.002 * max_c)
        active_idx = np.where(corr >= max_c - tol)[0]
        lags = active_idx - (n - 1)

        grad = np.zeros(n)
        for m in lags:
            grad += lag_gradient(int(m), h)
        if len(lags) > 0:
            grad /= len(lags)
        grad *= 2.0 / n

        h_new = project_feasible(h - alpha * grad, tgt)
        v = compute_c5(h_new)

        if v < local_val - 1e-11:
            local_best = h_new.copy()
            local_val = v
            stuck = 0
            alpha = min(alpha * 1.01, 5.0)
        else:
            stuck += 1
            alpha *= 0.96
            if stuck > 60:
                noise = 0.03 * np.random.randn(n)
                h = project_feasible(local_best + noise, tgt)
                stuck = 0
                alpha = 4.0 / n
        h = h_new

    return local_best, local_val
