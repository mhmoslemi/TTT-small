def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0: return x
    target = float(target)
    if target <= 0.0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = -float(np.max(x)) - 1.0

    for _ in range(70):
       mid = 0.5*(lo+hi)
       s = float(np.sum(np.clip(x+mid,0.0,1.0)))
       if s < target: lo=mid
       else: hi=mid
    y = np.clip(x + 0.5*(lo+hi),0.0,1.0)
    diff = target - float(np.sum(y))
    if diff > 0.0:

          if diff <= 1e-15: break
          room=1.0-y[i]
          if room > 0.0:

    elif diff < 0.0:

          i=int(np.argmin(y))

def quantize_exact(h, target, scale=4096):
    h = np.asarray(h, dtype=float)
    q = np.rint(h * float(scale)).astype(np.int64)

    if need > 0:

    return q.astype(float) / float(scale)

import warnings
import numpy as np
from scipy.optimize import minimize

warnings.simplefilter("ignore")

_P = 16.0
_TARGET = 0.0
_CALL_DEADLINE = 0.0
_CALL_X = None
_PAIR_OFFSETS = (0.005, 0.01, 0.02, 0.05, 0.1, 0.2)


class SLSQPTimeout(Exception):
    pass


def default_start(n):
    h = np.zeros(int(n), dtype=float)
    h[::2] = 1.0
    return h


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return np.empty(0, dtype=float)
    target = float(target)
    if not np.all(np.isfinite(x)):
        x = np.full(n, 0.5, dtype=float)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = -float(np.max(x)) - 1.0
    hi = 1.0 - float(np.min(x)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(y))
    if diff > 0.0:
        for i in range(n):
            if diff <= 1e-15:
                break
            room = 1.0 - y[i]
            if room > 0.0:
                add = min(room, diff)
                y[i] += add
                diff -= add
    elif diff < 0.0:
        for i in range(n):
            if diff >= -1e-15:
                break
            room = y[i]
            if room > 0.0:
                add = min(room, -diff)
                y[i] -= add
                diff += add

    s = float(np.sum(y))
    if abs(target - s) > 1e-15:
        if target > s:
            i = int(np.argmax(1.0 - y))
        else:
            i = int(np.argmin(y))
        val = target - (s - y[i])
        if val < 0.0:
            val = 0.0
        elif val > 1.0:
            val = 1.0
        y[i] = val
    return y


def c5_true(h):
    n = int(len(h))
    if n == 0:
        return 1e100
    return float(np.max(np.correlate(h, 1.0 - h, "full")) * (2.0 / float(n)))


def correlate_c(h):
    n = int(len(h))
    return np.correlate(h, 1.0 - h, "full") * (2.0 / float(n))


def smooth_compute(x, p):
    C = correlate_c(x)
    nc = C.size
    if nc == 0:
        return 0.0, C, np.empty(0, dtype=float)
    m = float(np.max(C))
    if m <= 1e-15:
        return 0.0, C, np.zeros(nc, dtype=float)
    t = np.clip(C / m, 0.0, 1.0)
    tp = t ** float(p)
    S = float(tp.sum())
    if S <= 0.0:
        return m, C, np.zeros(nc, dtype=float)
    val = m * (S ** (1.0 / float(p)))
    coeff = S ** (-(float(p) - 1.0) / float(p))
    w = coeff * (t ** (float(p) - 1.0))
    return val, C, w


def gradient_from_CW(h, C, w):
    n = int(len(h))
    if n == 0:
        return np.empty(0, dtype=float)
    dx = 2.0 / float(n)
    mid = n - 1
    g = np.zeros(n, dtype=float)
    for idx in range(C.size):
        wi = float(w[idx])
        if wi <= 1e-17:
            continue
        d = mid - idx
        wdx = wi * dx
        if d == 0:
            g += wdx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            g[:m] += wdx * (1.0 - h[d:d + m])
            g[d:] -= wdx * h[:m]
        else:
            s = -d
            m = n - s
            g[s:] += wdx * (1.0 - h[:m])
            g[:m] -= wdx * h[s:]
    return g


def _smooth_fun(x):
    x = np.asarray(x, dtype=float)
    val, C, w = smooth_compute(x, _P)
    return float(val)


def _smooth_jac(x):
    x = np.asarray(x, dtype=float)
    val, C, w = smooth_compute(x, _P)
    return gradient_from_CW(x, C, w)


def _sum_fun(x):
    return float(np.sum(x) - _TARGET)


def _sum_jac(x):
    return np.ones_like(np.asarray(x, dtype=float))


def _slsqp_callback(xk):
    global _CALL_X, _CALL_DEADLINE
    xk = np.asarray(xk, dtype=float)
    _CALL_X = xk.copy()
    if time.time() > _CALL_DEADLINE:
        raise SLSQPTimeout
    return False


def slsqp_smooth(h, p, target, maxiter, ftol, call_limit):
    global _P, _TARGET, _CALL_DEADLINE, _CALL_X
    h = project_box_sum(h, target)
    n = int(len(h))
    if n == 0:
        return h
    _P = float(p)
    _TARGET = float(target)
    _CALL_X = h.copy()
    _CALL_DEADLINE = time.time() + max(1.0, float(call_limit))
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": _sum_fun, "jac": _sum_jac}]
    try:
        res = minimize(
            _smooth_fun,
            h,
            jac=_smooth_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(max(1, maxiter)), "ftol": float(max(1e-14, ftol))},
            callback=_slsqp_callback,
        )
        x = np.asarray(res.x, dtype=float)
    except SLSQPTimeout:
        x = _CALL_X if _CALL_X is not None else h
    except Exception:
        x = _CALL_X if _CALL_X is not None else h
    return project_box_sum(x, target)


def _add_candidate(vals, v, lo, hi):
    try:
        v = float(v)
    except Exception:
        return
    if not np.isfinite(v):
        return
    if v < lo - 1e-9 or v > hi + 1e-9:
        return
    if v < lo:
        v = lo
    if v > hi:
        v = hi
    for u in vals:
        if abs(v - u) < 1e-8:
            return
    vals.append(v)


def pair_candidates(lo, hi, old):
    if hi - lo < 1e-12:
        return []
    vals = []
    for off in _PAIR_OFFSETS:
        _add_candidate(vals, old + off, lo, hi)
        _add_candidate(vals, old - off, lo, hi)
    _add_candidate(vals, 0.5 * (lo + hi), lo, hi)
    _add_candidate(vals, lo, lo, hi)
    _add_candidate(vals, hi, lo, hi)
    return vals


def pair_local_search(h, target, deadline, max_sweeps=8):
    dl = float(deadline)
    h = project_box_sum(h, target).copy()
    n = int(len(h))
    if n < 2:
        return h, c5_true(h)

        return h, c5_true(h)

    best = c5_true(h)
    sweeps = 0

        sweeps += 1
        improved = False

                break
            for j in range(i + 1, n):
                if time.time() > dl:
                    break

                cur_j = float(h[j])

                if hi - lo < 1e-12:
                    continue
                cands = pair_candidates(lo, hi, cur_i)
                if not cands:
                    continue
                best_pair = best
                best_x = None
                best_y = None
                for x in cands:
                    y = s - float(x)
                    if y < -1e-9 or y > 1.0 + 1e-9:

                    h[i] = best_x
                    h[j] = best_y

            break

def random_neighborhood(h, target, scale):

    z = np.random.normal(0.0, float(scale), len(h))
    z -= z.mean()
    return project_box_sum(h + z, target)


def snap_to_binary(h, target):
    h = np.asarray(h, dtype=float)
    n = int(len(h))
    order = np.argsort(-h)
    h_bin = np.zeros(n, dtype=float)
    k = int(round(float(target)))
    k = max(0, min(n, k))
    h_bin[order[:k]] = 1.0
    return project_box_sum(h_bin, target)


def random_binary_h(n, target):
    n = int(n)
    h = np.zeros(n, dtype=float)
    idx = np.random.permutation(n)
    k = int(round(float(target)))
    k = max(0, min(n, k))
    h[idx[:k]] = 1.0
    return project_box_sum(h, target)


def resample_h(h, n_new):
    h = np.asarray(h, dtype=float)
    n_old = int(len(h))
    n_new = int(n_new)
    if n_old == 0 or n_new == 0:
        return np.empty(n_new, dtype=float)
    if n_new == n_old:
        return project_box_sum(h, 0.5 * float(n_new))
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    y = np.interp(x_new, x_old, h)
    return project_box_sum(y, 0.5 * float(n_new))


def quantize_exact(h, target, scale=4096):
    h = np.asarray(h, dtype=float)
    scale = int(scale)
    q = np.rint(h * float(scale)).astype(np.int64)
    q = np.clip(q, 0, scale)
    need = int(round(float(target) * float(scale))) - int(q.sum())
    if need > 0:
        err = h * float(scale) - q.astype(float)
        while need > 0:
            mask = q < scale
            if not np.any(mask):
                break
            idx = int(np.argmax(np.where(mask, err, -np.inf)))
            q[idx] += 1
            need -= 1
    elif need < 0:
        err = q.astype(float) - h * float(scale)
        while need < 0:
            mask = q > 0
            if not np.any(mask):
                break
            idx = int(np.argmax(np.where(mask, err, np.inf)))
            q[idx] -= 1
            need += 1
    return q.astype(float) / float(scale)
