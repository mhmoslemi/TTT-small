import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    free = (h > 1e-4) & (h < 1.0 - 1e-4)
    cnt = free.sum()
    if cnt > 0:
        h[free] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-10:
        h[0] = np.clip(h[0] + diff2, 0.0, 1.0)
    return h

def _sa_loop(h, c5, c5_best, h_best, rng, n, T, T_min, rate, deadline):
    """Run SA until deadline, return updated state."""
    while time.time() < deadline:
        i, j = rng.choice(n, 2, replace=False)
        oi, oj = h[i], h[j]
        if rng.random() < 0.30:
            h[i], h[j] = oj, oi
        else:
            s = oi + oj
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                T = max(T_min, T * rate)
                continue
            t = rng.uniform(lo, hi)
            h[i], h[j] = t, s - t
        c5_new = evaluate_c5(h)
        if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / T):
            c5 = c5_new
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
        else:
            h[i], h[j] = oi, oj
        T = max(T_min, T * rate)
    return c5, c5_best, h_best, T

def _hill_climb(h, c5, c5_best, h_best, rng, n, deadline):
    """Pure hill climbing with periodic kicks."""
    stuck = 0
    while time.time() < deadline:
        i, j = rng.choice(n, 2, replace=False)

            stuck = 0
