       idx=np.flatnonzero(y<1-1e-12)

import numpy as np
import time

def pbox(v, m):
    v = np.asarray(v, dtype=float)
    if v.size == 0:
        return v
    v = np.clip(v, 0.0, 1.0)
    s = float(v.sum())
    if abs(s - m) < 1e-10:
        return v
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    d = m - float(y.sum())
    if abs(d) > 1e-12:
        idx = np.flatnonzero((y > 1e-12) & (y < 1.0 - 1e-12))
        if idx.size:
            y[idx] += d / idx.size
            y = np.clip(y, 0.0, 1.0)
            d = m - float(y.sum())
        if d > 0.0:
            idx = np.flatnonzero(y < 1.0)
        elif d < 0.0:
            idx = np.flatnonzero(y > 0.0)
        else:
            idx = np.empty(0, dtype=int)
        if idx.size:
            y[idx] += d / idx.size
            y = np.clip(y, 0.0, 1.0)
    d = m - float(y.sum())
    if d > 1e-13:
        k = int(np.argmax(1.0 - y))
        y[k] = min(1.0, y[k] + d)
    elif d < -1e-13:
        k = int(np.argmax(y))
        y[k] = max(0.0, y[k] + d)
    return y

def mc(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def lags(h):
    n = h.size
    om = 1.0 - h
    vals = np.empty(2 * n - 1)
    off = n - 1
    for l in range(-n + 1, n):
        if l >= 0:
            m = n - l
            vals[off + l] = float(np.dot(h[l:], om[:m]))
        else:
            d = -l
            m = n - d
            vals[off + l] = float(np.dot(h[:m], om[d:]))
    return vals

def agrad(h):
    n = h.size
    vals = lags(h)
    mv = float(vals.max())
    act = np.flatnonzero(vals > mv - 1e-4)
    if act.size == 0:
        act = np.array([n - 1])
    g = np.zeros(n)
    for idx in act:
        l = int(idx - (n - 1))
        if l >= 0:
            a = h[l:]
            b = h[:n - l]
            if a.size:
                g[l:] += 1.0 - b
                g[:n - l] -= a
        else:
            d = -l
            a = h[:n - d]
            b = h[d:]
            if a.size:
                g[:n - d] += 1.0 - b
                g[d:] -= a
    return g / act.size, mv

def suggest(h, k, rng):
    n = h.size
    if n < 2 or k <= 0:
        return np.empty((0, 2), dtype=int)
    g, _ = agrad(h)
    score = g[:, None] - g[None, :]
    mask = (h[:, None] < 1.0 - 1e-10) & (h[None, :] > 1e-10)
    np.fill_diagonal(mask, False)
    score[~mask] = np.inf
    flat = score.ravel()
    if rng is not None:
        flat = flat + rng.random(flat.size) * 1e-9
    fin = np.isfinite(flat)
    cnt = int(fin.sum())
    if cnt == 0:
        return np.empty((0, 2), dtype=int)
    if cnt <= k:
        idx = np.flatnonzero(fin)
    else:
        idx = np.argpartition(flat, k - 1)[:k]
    idx = idx[fin[idx]]
    if idx.size:
        idx = idx[np.argsort(flat[idx])]
    if idx.size == 0:
        return np.empty((0, 2), dtype=int)
    return np.column_stack(np.unravel_index(idx, (n, n)))

def rpair(h, k, rng):
    n = h.size
    up = np.flatnonzero(h < 1.0 - 1e-10)
    down = np.flatnonzero(h > 1e-10)
    if up.size == 0 or down.size == 0:
        return np.empty((0, 2), dtype=int)
    k = int(min(k, max(1, up.size * down.size)))
    i = rng.choice(up, size=k)
    j = rng.choice(down, size=k)
    ok = i != j
    if not ok.any():
        return np.empty((0, 2), dtype=int)
    return np.column_stack((i[ok], j[ok]))

def bround(h):
    n = h.size
    m = int(0.5 * n + 0.5)
    if m <= 0:
        return np.zeros(n)
    if m >= n:
        return np.ones(n)
    x = np.zeros(n)
    idx = np.argsort(-h, kind="stable")[:m]
    x[idx] = 1.0
    return x

def thue(n):
    L = 1
    while L < 2 * n:
        L <<= 1
    seq = np.zeros(L)
    half = 1
    while half < L:
        seq[half:2 * half] = 1.0 - seq[:half]
        half <<= 1
    return bround(seq[:n].copy())

def rbin(n, seed):
    m = int(0.5 * n + 0.5)
    x = np.zeros(n)
    if m <= 0 or m >= n:
        return x
    rng = np.random.default_rng(seed)
    idx = rng.permutation(n)[:m]
    x[idx] = 1.0
    return x

def res(base, n):
    base = np.asarray(base, dtype=float).ravel()
    N = base.size
    if N == 0 or n <= 0:
        return np.zeros(1)
    base = np.clip(base, 0.0, 1.0)
    total = float(base.sum()) * 2.0 / N
    if not np.isfinite(total) or total <= 1e-12:
        base = np.full(N, 0.5)
        total = 1.0
    base = base / total
    xo = np.linspace(0.0, 2.0, N + 1)
    c = np.empty(N + 1)
    c[0] = 0.0
    np.cumsum(base, out=c[1:])
    c *= 2.0 / N
    xn = np.linspace(0.0, 2.0, n + 1)
    cn = np.interp(xn, xo, c)
    cn[0] = 0.0
    cn[-1] = 1.0
    h = np.diff(cn) / (2.0 / n)
    return pbox(np.clip(h, 0.0, 1.0), 0.5 * n)

def clean(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return np.zeros(1)
    h = np.clip(h, 0.0, 1.0)
    m = 0.5 * h.size
    if abs(float(h.sum()) - m) > 1e-8:
        h = pbox(h, m)
    return h

def ep(h, i, j, t, f0):
    if abs(t) < 1e-15:
        return f0
    h2 = h.copy()
    vi = h2[i] + t
    if vi < 0.0:
        vi = 0.0
    elif vi > 1.0:
        vi = 1.0
    h2[i] = vi
    vj = h2[j] - t
    if vj < 0.0:
        vj = 0.0
    elif vj > 1.0:
        vj = 1.0
    h2[j] = vj
    return mc(h2)

def pmin(h, i, j, f0, mit):
    a = -h[i]
    b = 1.0 - h[j]
    if b - a <= 1e-12:
        return 0.0, f0
    gr = 0.6180339887498949
    c = a + (1.0 - gr) * (b - a)
    d = b - (1.0 - gr) * (b - a)
    fc = ep(h, i, j, c, f0)
    fd = ep(h, i, j, d, f0)
    bt = 0.0
    bf = f0
    if fc < bf:
        bf, bt = fc, c
    if fd < bf:
        bf, bt = fd, d
    for _ in range(mit):
        if abs(b - a) < 1e-7:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = a + (1.0 - gr) * (b - a)
            fc = ep(h, i, j, c, f0)
            if fc < bf:
                bf, bt = fc, c
        else:
            a = c
            c = d
            fc = fd
            d = b - (1.0 - gr) * (b - a)
            fd = ep(h, i, j, d, f0)
            if fd < bf:
                bf, bt = fd, d
    for t in (a, b):
        ft = ep(h, i, j, t, f0)
        if ft < bf:
            bf, bt = ft, t
    if bf >= f0 and abs(bt) < 1e-12:
        return 0.0, f0
    return bt, bf

def gstep(h, f, m):
    g, _ = agrad(h)
    base = h.copy()
    for eta in (0.05, 0.02, 0.01, 0.003):
        y = pbox(base - eta * g, m)
        fy = mc(y)
        if fy < f - 1e-13:
            return y, fy, 1
    return base, f, 0

                h[j] -= tn
                if h[i] < 0.0:
                    h[i] = 0.0
                elif h[i] > 1.0:
                    h[i] = 1.0
                if h[j] < 0.0:
                    h[j] = 0.0
                elif h[j] > 1.0:
                    h[j] = 1.0
                f = fn
                if f < bf - 1e-12:
                    bf = f
                    bh = h.copy()
                    bt = time.time()
                imp = True
        if imp:
            no = 0
        else:
            no += 1
            if no >= 2:
                h, f, ok = gstep(h, f, m)
                if ok and f < bf - 1e-12:
                    bf = f
                    bh = h.copy()
                    bt = time.time()
                    no = 0
                    continue
                base = bh.copy()
                sig = 0.02 + 0.02 * (no // 2)
                if sig > 0.25:
                    sig = 0.25
                h = pbox(base + sig * rng.standard_normal(n), m)
                f = mc(h)
                if n % 2 == 0 and rng.random() < 0.35:
                    hb = bround(base + sig * rng.standard_normal(n))
                    fb = mc(hb)
                    if fb < f:
                        h = hb
                        f = fb
                no = 0
            if time.time() - bt > max(30.0, 0.2 * limit):
                break
    if n % 2 == 0:
        hb = bround(bh)
        fb = mc(hb)
        if fb < bf:
            bf = fb
            bh = hb
    return clean(bh), bf

def sumcon(x):
    return float(np.sum(x) - 0.5 * x.size)

def lo(x):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    return float(lags(x).max())

def lj(x):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    g, _ = agrad(x)
    return g

def polish(h0, limit):
    h = clean(h0)
    f0 = mc(h)
    try:
        import scipy.optimize as opt
        n = h.size
        bnds = []
        for _ in range(n):
            bnds.append((0.0, 1.0))
        maxit = int(max(10, min(80, 0.4 * float(limit))))
        r = opt.minimize(lo, h, method="SLSQP", jac=lj, bounds=bnds,
                         constraints=[{"type": "eq", "fun": sumcon}],
                         options={"maxiter": maxit, "ftol": 1e-12})
        x = pbox(np.clip(np.asarray(r.x, dtype=float), 0.0, 1.0), 0.5 * n)
        f = mc(x)
        if f < f0:
            return x, f
    except Exception:
        pass
    return h, f0

def get_base():
    base = globals().get("initial_h_values")
    if base is not None:
        try:
            arr = np.asarray(base, dtype=float).ravel()
            if arr.size > 0:
                return arr
        except Exception:
            pass
    return thue(70)

def better(h, f, best_h, best_c):
    if best_h is None or f * (2.0 / h.size) < best_c - 1e-14:
        return h, f * (2.0 / h.size)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + budget - 5.0
    base = get_base()
    bh = clean(base)
    bf = mc(bh)
    besth = bh
    bestc = bf * (2.0 / bh.size)
    rem = deadline - time.time()
    if rem > 30.0:
        lim = min(0.45 * budget, rem - 10.0)
        h2, f2 = search(bh, lim, int(seed))
        besth, bestc = better(h2, f2, besth, bestc)
    n0 = bh.size
    tasks = []
    if n0 >= 10:
        nc = max(20, n0 // 2)
        if nc != n0 and nc >= 10:
            tasks.append((res(base, nc), 0.10, 101))
    if n0 != 70:
        tasks.append((res(base, 70), 0.08, 102))
    nf = min(140, max(2 * n0, 100))
    if nf != n0 and nf >= 10:
        tasks.append((res(base, nf), 0.07, 103))
    if n0 >= 8:
        tasks.append((thue(n0), 0.06, 104))
    if n0 % 2 == 0:
        tasks.append((rbin(n0, int(seed) + 201), 0.05, 105))
        tasks.append((rbin(n0, int(seed) + 301), 0.05, 106))
    tasks.append((pbox(bh + 0.05 * np.random.default_rng(seed + 401).standard_normal(bh.size), 0.5 * bh.size), 0.03, 107))
    for idx in range(len(tasks)):
        rem = deadline - time.time()
        if rem < 15.0:
            break
        hst, frac, sd = tasks[idx]
        hst = clean(hst)
        limit = min(frac * budget, rem - 5.0)
        if limit < 5.0:
            qf = mc(hst)
            besth, bestc = better(hst, qf, besth, bestc)
            continue
        h2, f2 = search(hst, limit, int(sd))
        besth, bestc = better(h2, f2, besth, bestc)
    if besth is not None:
        rem = deadline - time.time()
        if rem > 25.0:
            lim = min(rem - 10.0, 0.08 * budget)
            h2, f2 = search(clean(besth), lim, int(seed) + 501)
            besth, bestc = better(h2, f2, besth, bestc)
        rem = deadline - time.time()
        if rem > 20.0:
            lim = min(rem - 5.0, 0.06 * budget)
            h2, f2 = polish(besth, lim)
            besth, bestc = better(h2, f2, besth, bestc)
    if besth is None:
        besth = bh
    final = clean(besth)
    finalc = mc(final) * (2.0 / final.size)
    return final, float(finalc), int(final.size)
