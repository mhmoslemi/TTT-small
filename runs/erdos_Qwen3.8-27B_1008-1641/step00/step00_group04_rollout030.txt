import numpy as np
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _to_binary(hv, n):
    h = (hv >= 0.5).astype(np.float64)
    half = n // 2
    cnt = int(h.sum())
    if cnt > half:
        idxs = np.random.permutation(np.where(h == 1)[0])[:cnt - half]
        h[idxs] = 0
    elif cnt < half:
        idxs = np.random.permutation(np.where(h == 0)[0])[:half - cnt]
        h[idxs] = 1
    return h

def _greedy(h, n, t_max):
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    cur = _c5(h, n)
    for _ in range(300):
        if time.time() > t_max:
            break
        best_c = cur
        best_i = -1
        best_j = -1
        for ii in range(len(ones)):
            for jj in range(len(zeros)):
                i = ones[ii]
                j = zeros[jj]
                h[i] = 0
                h[j] = 1
                c = _c5(h, n)
                if c < best_c:
                    best_c = c
                    best_i = i
                    best_j = j
                h[i] = 1
                h[j] = 0
        if best_i < 0:
            break
        h[best_i] = 0
        h[best_j] = 1
        cur = best_c
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()
    return h, cur

def _sa(h, n, t_end):
    sh = h.copy()
    sc5 = _c5(sh, n)
    os_l = np.where(sh == 1)[0].tolist()
    zs_l = np.where(sh == 0)[0].tolist()
    best_c = sc5
    best_h = sh.copy()
    t_sa = time.time()
    dur = max(t_end - t_sa, 1.0)
    while time.time() < t_end:
        elapsed = time.time() - t_sa
        frac = max(0.0, 1.0 - elapsed / dur)
        temp = 0.004 * frac * frac + 1e-12
        if temp < 1e-13:
            break
        ii = np.random.randint(len(os_l))
        jj = np.random.randint(len(zs_l))
        i_ = os_l[ii]
        j_ = zs_l[jj]
        sh[i_] = 0
        sh[j_] = 1
        os_l[ii] = j_
        zs_l[jj] = i_
        nc = _c5(sh, n)
        delta = nc - sc5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            sc5 = nc
            if sc5 < best_c:
                best_c = sc5
                best_h = sh.copy()
        else:
            sh[i_] = 1
            sh[j_] = 0
            os_l[ii] = i_
            zs_l[jj] = j_
        if best_c < sc5 - 0.03 and np.random.rand() < 0.01:
            sh[:] = best_h
            sc5 = best_c
            os_l = np.where(sh == 1)[0].tolist()
            zs_l = np.where(sh == 0)[0].tolist()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_limit = budget_s - 3
    n = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=np.float64).clip(0, 1)

    # Phase 1: Greedy from parent 0-1 rounding
    h_par = _to_binary(h0, n)
    h_par, c5_par = _greedy(h_par, n, t0 + 60)

    # Phase 2: SA from greedy result (65% of budget)
    t_sa_end = t0 + (t_limit - t0) * 0.65
    h_par, c5_par = _sa(h_par, n, t_sa_end)
    # Quick polish
    h_par, c5_par = _greedy(h_par, n, time.time() + 5)

    best_h = h_par.copy()
    best_c5 = c5_par

    # Phase 3: Random restarts with independent SA
    n_restarts = 5
    t_remain = t_limit - time.time()
    if t_remain > 20:
        t_per = t_remain / n_restarts
        for r in range(n_restarts):
            if time.time() >= t_limit:
                break
            t_r_end = min(time.time() + t_per, t_limit)
            h_r = np.zeros(n)
            h_r[np.random.permutation(n)[:n // 2]] = 1
            h_r, c5_r = _greedy(h_r, n, min(t_r_end, time.time() + 8))
            if c5_r < best_c5:
                best_h = h_r.copy()
                best_c5 = c5_r
            if time.time() < t_r_end - 5:
                h_r2, c5_r2 = _sa(h_r, n, t_r_end)
                if c5_r2 < best_c5:
                    best_h = h_r2.copy()
                    best_c5 = c5_r2

    # Final polish
    if time.time() < t_limit:
        best_h, best_c5 = _greedy(best_h, n, t_limit)

    final_c5 = _c5(best_h, n)
    return best_h, final_c5, n
