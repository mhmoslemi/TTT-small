import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-8) & (h < 1.0 - 1e-8)
    cnt = np.sum(mask)
    if cnt == 0:
        mask = np.ones(len(h), dtype=bool)
        cnt = len(h)
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - 15.0 - (time.time() - t0)

    # Load initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(47, 0.5)

    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0

    best = h0.copy()
    best_c = compute_c5(best, dx)

    # Build pair list
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    def pair_optimizer(i, j, s, lo, hi):
        """Optimize one pair (i,j) with fixed sum s."""
        h_i0, h_j0 = best[i], best[j]
        def f1d(t):
            best[i] = t
            best[j] = s - t
            return compute_c5(best, dx)
        res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                             options={'maxiter': 100, 'xatol': 1e-13})
        if res.fun < best_c - 1e-15:
            best[i] = res.x
            best[j] = s - res.x
            return res.fun
        else:
            best[i] = h_i0
            best[j] = h_j0
            return best_c

    # Phase 1: Cyclic paired coordinate descent
    no_improve = 0
    for epoch in range(20000):
        if rem() < 3.0:
            break
        epoch_start_c = best_c
        # Shuffle pairs for diversity
        perm = rng.permutation(len(pairs))
        for idx in perm:
            i, j = pairs[idx]
            s = best[i] + best[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            span = hi - lo
            if span < 1e-11:
                continue
            before = best_c
            best_c = pair_optimizer(i, j, s, lo, hi)
            if best_c < before - 1e-14:
                no_improve = 0
                break  # restart order after improvement
            no_improve += 1
        if epoch_start_c - best_c < 1e-15:
            no_improve += 100
        else:
            no_improve = 0

        # If stuck, try perturbations
        if no_improve > 5000:
            no_improve = 0
            for _ in range(200):
                if rem() < 3.0:
                    break
                i, j = rng.choice(n, 2, replace=False)
                s = best[i] + best[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-6:
                    continue
                t_new = rng.uniform(lo, hi)
                h_si, h_sj = best[i], best[j]
                best[i] = t_new
                best[j] = s - t_new
                c = compute_c5(best, dx)

            return compute_c5(x, dx)

            if res.fun < best_c:

        except Exception:
            pass

    best = np.clip(best, 0.0, 1.0)
    best = fix_sum(best, tgt)
    best_c = compute_c5(best, dx)

    return best, best_c, n
