    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_obj(h, dx, beta):

    corr = compute_corr(h)
    vals = dx * corr
    m = np.max(vals)
    # Numerically stable log-sum-exp
    shifted = vals - m
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

    corr = compute_corr(h)
    vals = dx * corr
    m = np.max(vals)

        clamped = np.clip(v - mid, 0.0, 1.0)

        else:

        if np.any(can_add):
            clamped[can_add] += diff / np.sum(can_add)
            clamped = np.clip(clamped, 0.0, 1.0)

def project_feasible(v, target):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=target} via water-filling."""
    n = len(v)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - result.sum()

            mask = result < 1.0
        else:

        cnt = mask.sum()

            result[mask] += diff / cnt
            np.clip(result, 0.0, 1.0, out=result)

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = dx * corr
    m = vals.max()
    ex = np.exp(beta * (vals - m))
    total = ex.sum()
    w = ex / total
    obj = m + np.log(total) / beta
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        s = (n - 1) - k  # shift in (1-h)
        c = w[k] * dx
        if s >= 0:
            # First term: x = 0..n-s-1, (1-h[x+s])
            grad[:n-s] += c * (1.0 - h[s:])
            # Second term: x = s..n-1, -h[x-s]
            grad[s:] -= c * h[:n-s]
        else:
            sp = -s  # positive
            # First term: x = sp..n-1, (1-h[x-sp])
            grad[sp:] += c * (1.0 - h[:n-sp])
            # Second term: x = 0..n-sp-1, -h[x+sp]
            grad[:n-sp] -= c * h[sp:]
    return obj, grad

import numpy as np
import time

def project_feasible(v, target):
    n = len(v)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        mask = result < 1.0 if diff > 0 else result > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            result[mask] += diff / cnt
            np.clip(result, 0.0, 1.0, out=result)
    return result

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = dx * corr
    m = vals.max()
    ex = np.exp(beta * (vals - m))
    total = ex.sum()
    w = ex / total
    obj = m + np.log(total) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        s = (n - 1) - k
        c = w[k] * dx
        if s >= 0:
            if s < n:
                grad[:n - s] += c * (1.0 - h[s:])
                grad[s:] -= c * h[:n - s]
        else:
            sp = -s
            if sp < n:
                grad[sp:] += c * (1.0 - h[:n - sp])
                grad[:n - sp] -= c * h[sp:]
    return obj, grad

def pgd_optimize(h, dx, beta, target, max_iter, lr_init):
    n = len(h)
    lr = lr_init
    obj, grad = smooth_obj_grad(h, dx, beta)
    for _ in range(max_iter):
        step_h = project_feasible(h - lr * grad, target)
        step_obj, step_grad = smooth_obj_grad(step_h, dx, beta)
        if step_obj < obj - 1e-15:
            h, obj, grad = step_h, step_obj, step_grad
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.5
            if lr < 1e-12:
                break
    return h

def binary_local_search(h, dx, target, best_c5, max_rounds):
    n = len(h)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            if h[i] > 0.01:
                h_try = project_feasible(h - 0.01, target)
                h_try[i] = 0.0
                h_try = project_feasible(h_try, target)
                c5 = compute_c5(h_try, dx)
                if c5 < best_c5 - 1e-12:
                    h, best_c5 = h_try, c5
                    improved = True
                    break
            if h[i] < 0.99:
                h_try = h.copy()
                h_try[i] = 1.0
                h_try = project_feasible(h_try, target)
                c5 = compute_c5(h_try, dx)
                if c5 < best_c5 - 1e-12:
                    h, best_c5 = h_try, c5
                    improved = True
                    break
            for d in [0.05, 0.02, -0.05, -0.02]:
                h_try = h.copy()
                h_try[i] = min(1.0, max(0.0, h[i] + d))
                h_try = project_feasible(h_try, target)
                c5 = compute_c5(h_try, dx)
                if c5 < best_c5 - 1e-12:
                    h, best_c5 = h_try, c5
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c5
