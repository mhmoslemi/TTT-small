```
def finalize_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    diff = float(target - x.sum())
    if diff > 1e-14:
        order = np.argsort(-(1.0 - x))
        for i in order:
            if diff <= 1e-14: break
            add = min(1.0 - float(x[i]), diff)
            if add <= 0.0: continue
            x[i] = x[i] + add
            diff -= add
    elif diff < -1e-14:
        order = np.argsort(-x)
        for i in order:
            if diff >= -1e-14: break
            sub = min(float(x[i]), -diff)
            if sub <= 0.0: continue
            x[i] = x[i] - sub
            diff += sub
    diff = float(target - x.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            i = int(np.argmax(1.0 - x))
            x[i] = min(1.0, float(x[i]) + diff)
        else:
            i = int(np.argmax(x))
            x[i] = max(0.0, float(x[i]) + diff)
    return x
```

```
def quad_roots(a,b,c):
    out=[]
    if abs(a) > 1e-14:
        d = b*b - 4.0*a*c
        if d >= 0.0:
            r = np.sqrt(d)
            out.append((-b - r)/(2.0*a))
            out.append((-b + r)/(2.0*a))
    elif abs(b) > 1e-14:
        out.append(-c/b)
    return out
```

```
def one_d_best(A,B,C,t0):
    t0 = float(np.clip(t0, 0.0, 1.0))
    cand = np.linspace(0.0, 1.0, 97)
    extra = np.array([t0, 0.0, 1.0])
    mask = np.abs(C) > 1e-13
    if np.any(mask):
        tv = -B[mask] / (2.0 * C[mask])
        tv = tv[(tv >= 0.0) & (tv <= 1.0)]
        cand = np.concatenate((cand, tv, extra))
    else:
        cand = np.concatenate((cand, extra))
    vals = envelope_max(A,B,C,cand)
    i = int(np.argmin(vals))
    best = float(cand[i]); best_v = float(vals[i])
    radii = (0.02, 0.008, 0.003, 0.001, 4.0e-4)
    for radius in radii:
        pts = np.concatenate((best + np.linspace(-radius, radius, 33), np.array([best])))
        pts = np.clip(pts, 0.0, 1.0)
        v = envelope_max(A,B,C,pts)
        j = int(np.argmin(v))
        if v[j] < best_v - 1e-15:
            best = float(pts[j]); best_v = float(v[j])
        elif v[j] <= best_v + 1e-14:
            best = float(pts[j])
        if radius <= 1.0e-3:
            break
    vals_best = A + B*best + C*best*best
    maxv = float(np.max(vals_best))
    topn = min(12, vals_best.size)
    if topn > 1:
        idx = np.argpartition(vals_best, -topn)[-topn:]
        for a in range(topn):
            ia = idx[a]
            for b in range(a+1, topn):
                ib = idx[b]
                aa = C[ia] - C[ib]
                bb = B[ia] - B[ib]
                cc = A[ia] - A[ib]
                for tt in quad_roots(aa, bb, cc):
                    if -1e-10 <= tt <= 1.0 + 1e-10:
                        tt = float(np.clip(tt, 0.0, 1.0))
```

```
def coordinate_polish(h, seed, time_limit, stagnate_limit=45.0):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if n < 2:
        return h,c5_raw(h)
    target = 0.5 * n
    h = project_box_sum(h, target)
    cur = c5_raw(h)
    start = time.time()
    deadline = start + time_limit
    last_imp = start
    rng = np.random.default_rng(seed)
    sweeps = 0
    upd = 0
    half = max(1, n // 2)
    while time.time() < deadline:
        improved_sweep = False
        order = rng.permutation(n)
        for i in order:
            if time.time() >= deadline: break
            p = int(i); q = (p + 1) % n
            old = cur
            cur, imp = pair_update(h,p,q,cur)
            if imp and old - cur > 1e-10:
                improved_sweep = True
                last_imp = time.time()
                upd += 1
                if upd % 250 == 0:
                    if abs(float(h.sum()) - target) > 1e-8:
                        h = project_box_sum(h, target)
                        cur = c5_raw(h)
        if time.time() >= deadline: break
        for i in order[:half]:
            if time.time() >= deadline: break
            p=int(i); q=(p+half)%n
            old=cur; cur,imp=pair_update(h,p,q,cur)
            if imp and old-cur>1e-10:
```

```python
import time
import numpy as np


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    if x.size == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(60):
        mu = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mu, 0.0, 1.0))
        if s > target:
            lo = mu
        else:
            hi = mu
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    return finalize_sum(y, target)


def finalize_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    d = float(target - x.sum())
    if d > 1e-14:
        for i in np.argsort(-(1.0 - x)):
            if d <= 1e-14:
                break
            a = min(1.0 - x[i], d)
            if a <= 0.0:
                continue
            x[i] = x[i] + a
            d -= a
    elif d < -1e-14:
        for i in np.argsort(-x):
            if d >= -1e-14:
                break
            a = min(x[i], -d)
            if a <= 0.0:
                continue
            x[i] = x[i] - a
            d += a
    d = float(target - x.sum())
    if abs(d) > 1e-12:
        if d > 0.0:
            i = int(np.argmax(1.0 - x))
            x[i] = min(1.0, float(x[i]) + d)
        else:
            i = int(np.argmax(x))
            x[i] = max(0.0, float(x[i]) + d)
    return x


def c5_raw(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def c5_value(h):
    return c5_raw(h) * (2.0 / float(h.size))


def project_eval(h):
    target = 0.5 * float(h.size)
    h = project_box_sum(h, target)
    h = finalize_sum(h, target)
    return h, c5_raw(h)


def envelope_max(A, B, C, t):
    t = np.asarray(t, dtype=float).reshape(-1)
    tt = t[:, None]
    return np.max(A[None, :] + B[None, :] * tt + C[None, :] * tt * tt, axis=1)


def quad_scalar(A, B, C, t):
    return float(np.max(A + B * t + C * t * t))


def quad_roots(a, b, c):
    out = []
    if abs(a) > 1e-14:
        d = b * b - 4.0 * a * c
        if d < -1e-11:
            return out
        if d < 0.0:
            d = 0.0
        r = np.sqrt(d)
        out.append((-b - r) / (2.0 * a))
        out.append((-b + r) / (2.0 * a))
    elif abs(b) > 1e-14:
        out.append(-c / b)
    return out


def one_d_best(A, B, C, t0):
    t0 = float(np.clip(t0, 0.0, 1.0))
    cand = np.linspace(0.0, 1.0, 81)
    extra = np.array([t0, 0.0, 1.0])
    mask = np.abs(C) > 1e-13
    if np.any(mask):
        tv = -B[mask] / (2.0 * C[mask])
        tv = tv[(tv >= 0.0) & (tv <= 1.0)]
        cand = np.concatenate((cand, tv, extra))
    else:
        cand = np.concatenate((cand, extra))
    vals = envelope_max(A, B, C, cand)
    i = int(np.argmin(vals))
    best = float(cand[i])
    best_v = float(vals[i])

    radii = (0.02, 0.008, 0.003, 0.001)
    for radius in radii:
        pts = np.concatenate((best + np.linspace(-radius, radius, 33), np.array([best])))
        pts = np.clip(pts, 0.0, 1.0)
        v = envelope_max(A, B, C, pts)
        j = int(np.argmin(v))
        if v[j] < best_v - 1e-15:
            best = float(pts[j])
            best_v = float(v[j])

    vals_best = A + B * best + C * best * best
    maxv = float(np.max(vals_best))
    topn = min(12, vals_best.size)
    if topn > 1:
        idx = np.argpartition(vals_best, -topn)[-topn:]
        for aa in range(topn):
            ia = idx[aa]
            for bb in range(aa + 1, topn):
                ib = idx[bb]
                ca = C[ia] - C[ib]
                cb = B[ia] - B[ib]
                cc = A[ia] - A[ib]
                for tt in quad_roots(ca, cb, cc):
                    if -1e-10 <= tt <= 1.0 + 1e-10:
                        tt = float(np.clip(tt, 0.0, 1.0))
                        vt = quad_scalar(A, B, C, tt)
                        if vt < best_v - 1e-15:
                            best = tt
                            best_v = vt

    for _ in range(6):
        vals_best = A + B * best + C * best * best
        maxv = float(np.max(vals_best))
        active = vals_best >= maxv - 1e-10
        if not np.any(active):
            break
        deriv = B[active] + 2.0 * C[active] * best
        dmin = float(np.min(deriv))
        dmax = float(np.max(deriv))
        if dmin > 1e-8:
            dirn = -1.0
        elif dmax < -1e-8:
            dirn = 1.0
        else:
            break
        step = 0.02
        moved = False
        for _ in range(10):
            tt = best + dirn * step

                if vt < best_v - 1e-13:
                    best = tt
                    best_v = vt
                    moved = True
                    break
            step *= 0.5
        if not moved:
            break

    return float(np.clip(best, 0.0, 1.0)), float(best_v)


def pair_raw_at(h, p, q, s, x):
    x = min(1.0, max(0.0, float(x)))
    y = float(s) - x
    if y < 0.0:
        y = 0.0
    elif y > 1.0:
        y = 1.0
    h[p] = x
    h[q] = y
    return np.correlate(h, 1.0 - h, mode="full")


def compute_pair_coeffs(h, p, q):
    oldp = float(h[p])
    oldq = float(h[q])
    s = oldp + oldq
    if s < 0.0:
        s = 0.0
    elif s > 2.0:
        s = 2.0
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-13:
        return None
    x1 = 0.5 * (lo + hi)
    v0 = pair_raw_at(h, p, q, s, lo)
    v1 = pair_raw_at(h, p, q, s, x1)
    v2 = pair_raw_at(h, p, q, s, hi)
    h[p] = oldp
    h[q] = oldq
    C = 2.0 * (v2 - 2.0 * v1 + v0)
    B = v2 - v0 - C
    return v0, B, C, lo, hi, oldp, oldq


def pair_update(h, p, q, cur):
    if p == q:
        return cur, False
    cf = compute_pair_coeffs(h, p, q)
    if cf is None:
        return cur, False
    A, B, C, lo, hi, oldp, oldq = cf
    width = hi - lo
    if width <= 0.0:
        return cur, False
    t0 = (oldp - lo) / width
    tb, _ = one_d_best(A, B, C, t0)
    if not np.isfinite(tb) or abs(tb - t0) < 1e-14:
        return cur, False
    ns = oldp + oldq
    newp = lo + tb * width
    if newp < 0.0:
        newp = 0.0
    elif newp > 1.0:
        newp = 1.0
    newq = ns - newp
    if newq < 0.0:
        newq = 0.0
    elif newq > 1.0:
        newq = 1.0
    h[p] = newp
    h[q] = newq
    raw = c5_raw(h)
    if raw < cur - 1e-15:
        return raw, True
    h[p] = oldp
    h[q] = oldq
    return cur, False


def coordinate_polish(h, seed, time_limit, stagnate_limit=45.0):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if n < 2:
        return h, c5_raw(h)
    target = 0.5 * n
    h = project_box_sum(h, target)
    cur = c5_raw(h)
    start = time.time()
    deadline = start + max(0.0, time_limit)
    last_imp = start
    rng = np.random.default_rng(seed)
    sweeps = 0
    half = max(1, n // 2)

    while time.time() < deadline:
        improved_sweep = False
        order = rng.permutation(n)
        pairs = []
        for i in order:
            p = int(i)
            pairs.append((p, (p + 1) % n))
        for i in order[:half]:
            p = int(i)
            pairs.append((p, (p + half) % n))
        for _ in range(half):
            p = int(rng.integers(n))
            q = int(rng.integers(n))
            if p == q:
                q = (q + 1) % n
            pairs.append((p, q))

        for p, q in pairs:
            if time.time() >= deadline:
                break
            old = cur
            cur, imp = pair_update(h, p, q, cur)
            if imp and old - cur > 1e-10:
                improved_sweep = True
                last_imp = time.time()

        sweeps += 1
        if sweeps % 20 == 0 and abs(float(h.sum()) - target) > 1e-8:
            h = project_box_sum(h, target)
            cur = c5_raw(h)
        if not improved_sweep and time.time() - last_imp > stagnate_limit:
            break

    return h, cur


def perturb_h(h, sigma, rng):
    z = h + sigma * rng.normal(size=h.size)
    return project_box_sum(z, 0.5 * h.size)


def binary_restart_h(h, rng):
    n = h.size
    noise = 0.05 * rng.normal(size=n)
    q = float(np.median(h))
    b = np.where(h + noise > q, 1.0, 0.0)
    return project_box_sum(b, 0.5 * n)


def generate_restart(best, orig, attempt, rng):
    base = orig if (attempt % 3 == 2) else best
    if attempt % 6 == 5:
        return binary_restart_h(base, rng)
    sigmas = (5.0e-4, 2.0e-3, 8.0e-3, 3.0e-2, 1.0e-1, 3.0e-1)
    return perturb_h(base, sigmas[attempt % 6], rng)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)

    h0 = None
    try:
        arr = np.asarray(initial_h_values, dtype=float)
        if arr.size > 1:
            h0 = arr.reshape(-1).copy()
    except Exception:
        h0 = None

    if h0 is None:
        n = 80
        x = np.linspace(0.0, 2.0, n)
        h0 = 0.5 + 0.25 * np.cos(np.pi * x) + 0.1 * np.cos(2.0 * np.pi * x)

    h0, r0 = project_eval(h0)
    best_h = h0.copy()
    best_c = r0 * 2.0 / h0.size
    h_orig = h0.copy()

    bs = float(budget_s)
    reserve = max(30.0, 0.3 * bs)
    lim = min(0.45 * bs, 450.0)
    lim = max(5.0, min(lim, max(5.0, bs - reserve)))

    h1, rc = coordinate_polish(h0, seed, lim, 45.0)
    h1, rc = project_eval(h1)
    cc = rc * 2.0 / h1.size
    if cc < best_c:
        best_h = h1
        best_c = cc

    rng = np.random.default_rng(seed + 999)
    attempt = 0
    while time.time() < deadline - 45.0 and attempt < 12:
        z = generate_restart(best_h, h_orig, attempt, rng)
        z, rz = project_eval(z)
        cz = rz * 2.0 / z.size
        if cz < best_c:
            best_h = z
            best_c = cz

        rem = deadline - time.time()
        chunk = min(75.0, rem - 35.0)
        if chunk > 20.0:
            z2, rz2 = coordinate_polish(z, seed + 2000 + attempt, chunk, 25.0)
            z2, rz2 = project_eval(z2)
            cz2 = rz2 * 2.0 / z2.size
            if cz2 < best_c:
                best_h = z2
                best_c = cz2
        attempt += 1

    rem = deadline - time.time()
    if rem > 60.0:
        zf, rf = coordinate_polish(best_h, seed + 9000, min(45.0, rem - 20.0), 20.0)
        zf, rf = project_eval(zf)
        cf = rf * 2.0 / zf.size
        if cf < best_c:
            best_h = zf
            best_c = cf

    backup = best_h.copy()
    backup_c = best_c
    hfin, rfin = project_eval(backup)
    cfin = rfin * 2.0 / hfin.size
    if cfin <= backup_c + 1e-12:
        best_h = hfin
        best_c = cfin

    return best_h, best_c, best_h.size
```
