def project_feasible(h, S):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h - t, 0.0, 1.0)

def smooth_max_c5(h, tau=50.0):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return m + np.log(np.sum(np.exp((corr - m) * tau))) / tau

def smooth_max_grad(h, tau=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp((corr - m) * tau)
    w /= w.sum()
    d_offsets = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    j = np.arange(n)
    for idx in range(len(d_offsets)):
        if w[idx] < 1e-14:
            continue
        d = d_offsets[idx]
        jd = j + d
        jnd = j - d
        g_d = np.zeros(n)
        mask1 = (jd >= 0) & (jd < n)
        g_d[mask1] += (1.0 - h[jd[mask1]])
        mask2 = (jnd >= 0) & (jnd < n)
        g_d[mask2] -= h[jnd[mask2]]
        grad += w[idx] * g_d
    grad *= dx
    return grad

def slsqp_optimize(h0, n, S, maxiter=500, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: np.sum(x) - S, 'jac': lambda x: np.ones(n)}
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < compute_c5(h0):
            h_new = project_feasible(res.x, S)
            return h_new
    except Exception:
        pass
    return h0

def grad_optimize(h0, n, S, budget_frac, start_time, budget_s):
    tau_seq = [5.0, 15.0, 50.0, 150.0, 400.0]
    for tau in tau_seq:
        if time.time() - start_time > budget_s * budget_frac:
            break
        n_iter = 300
        eta = 0.5
        h = h0.copy()
        best_h = h0.copy()
        best_val = compute_c5(h0)
        for it in range(n_iter):
            if time.time() - start_time > budget_s * budget_frac:
                break
            g = smooth_max_grad(h, tau)
            lr = eta / (1.0 + 0.02 * it)
            h = project_feasible(h - lr * g, S)
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
        h0 = best_h
    return h0

def random_perturb(h, n, scale):
    h_new = h + scale * np.random.randn(n)
    return project_feasible(h_new, np.sum(h))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(95, 0.5)

    n = len(h_init)
    S = n / 2.0
    h_init = project_feasible(h_init, S)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: SLSQP from initial
    if time.time() - start_time < budget_s * 0.3:
        h_t = slsqp_optimize(h_init, n, S, maxiter=800)
        c5 = compute_c5(h_t)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_t.copy()

    # Phase 2: Gradient-based refinement
    if time.time() - start_time < budget_s * 0.6:
        h_t = grad_optimize(best_h, n, S, 0.6, start_time, budget_s)
        c5 = compute_c5(h_t)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_t.copy()

    # Phase 3: Random restarts with SLSQP + gradient
    n_restarts = 40
    attempt = 0
    while attempt < n_restarts and time.time() - start_time < budget_s * 0.9:
        attempt += 1
        scale = 0.05 + 0.1 * np.random.random()
        h_r = random_perturb(best_h, n, scale)
        c5_r = compute_c5(h_r)

        # Quick SLSQP
        if time.time() - start_time < budget_s * 0.9:
            h_r = slsqp_optimize(h_r, n, S, maxiter=300)
            c5_r = compute_c5(h_r)

        # Gradient polish
        if c5_r < best_c5 and time.time() - start_time < budget_s * 0.9:
            h_r = grad_optimize(h_r, n, S, 0.9, start_time, budget_s)
            c5_r = compute_c5(h_r)

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Fine gradient polish on best
    if time.time() - start_time < budget_s * 0.97:
        h_t = grad_optimize(best_h, n, S, 0.97, start_time, budget_s)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    best_h = project_feasible(best_h, S)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
