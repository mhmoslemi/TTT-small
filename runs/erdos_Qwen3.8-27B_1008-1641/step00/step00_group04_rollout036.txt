import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n, target=None):
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += min(diff / cnt, 1.0 - h[mask] if np.any(mask) else 0)
            # Simpler: just distribute
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    return h

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return dx * (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    weights /= np.sum(weights)
    # Gradient of sum_k weights[k] * corr[k]
    # corr[k] = sum_i h[i] * (1-h[i + (k-n+1)])
    # Let lag[k] = k - n + 1
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-15:
            continue
        lag = k - n + 1
        # d corr[k] / d h[m]:
        # (1-h[m-lag]) * [0<=m-lag<n] - h[m+lag] * [0<=m+lag<n]
        # Vectorized:
        # For the first term: m ranges 0..n-1, m-lag ranges 0..n-1
        #   means m ranges max(0,lag)..min(n-1, n-1+lag)
        # For the second term: m ranges 0..n-1, m+lag ranges 0..n-1
        #   means m ranges max(0,-lag)..min(n-1, n-1-lag)
        
        # First term: (1 - h[m-lag]) for valid m
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)  # exclusive
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            grad[idx] += weights[k] * (1.0 - h[idx - lag])
        
        # Second term: -h[m+lag] for valid m
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)  # exclusive
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= weights[k] * h[idx + lag]
    
    return dx * grad

def projected_gradient_step(h, grad, lr, n):
    h_new = h - lr * grad
    return project_feasible(h_new, n)

        h_current = np.interp(x_new % 2.0, x_init, initial_h)
        h_current = project_feasible(h_current, n)
        
        # Projected gradient descent with line search
        alpha = 50.0  # softness parameter
        lr = 0.01
        patience = 0
        
        for it in range(500):
            if time.time() - start_time > budget_s - 20:

            obj = smooth_obj(h_current, alpha)
            grad = smooth_obj_grad(h_current, alpha)

        hi2 = min(n, n - d)
        if hi2 > lo2:

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def smooth_obj(x, alpha):
    h = np.asarray(x, dtype=float)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return dx * (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def upsample_periodic(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_per = np.concatenate([x_src - 2.0, x_src, x_src + 2.0])
    h_per = np.concatenate([h_src, h_src, h_src])
    return np.interp(x_dst, x_per, h_per)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Optimize at original resolution with multiple SLSQP restarts
    n = n0
    h_cur = h0.copy()
    max_iter_n0 = 800
    for restart in range(30):
        if time.time() - t0 > budget_s - 60:
            break
        if restart == 0:
            h_try = h_cur.copy()
        else:
            scale = 0.015 * (1.0 + restart * 0.3)
            h_try = h_cur + rng.normal(0.0, scale, n)
            h_try = project_feasible(h_try, n)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                smooth_obj, h_try, method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': sum_con}],
                args=(60.0,),
                options={'maxiter': max_iter_n0, 'ftol': 1e-14}
            )
            h_new = project_feasible(np.asarray(res.x), n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur = h_new.copy()
        except Exception:
            pass

    # Phase 2: Try higher resolutions
    for mult in [2, 3, 4]:
        n = n0 * mult
        remaining = budget_s - 40 - (time.time() - t0)
        if remaining < 30:
            break

        # Upsample
        h_cur = upsample_periodic(h0, n0, n)
        h_cur = project_feasible(h_cur, n)

        n_restarts = max(3, int(remaining / 40))
        for restart in range(n_restarts):
            if time.time() - t0 > budget_s - 40:
                break
            if restart == 0:
                h_try = h_cur.copy()
            elif restart == 1:
                h_try = project_feasible(h_cur + rng.normal(0.0, 0.02, n), n)
            else:
                h_try = project_feasible(h_cur + rng.normal(0.0, 0.01, n), n)
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(
                    smooth_obj, h_try, method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': sum_con}],
                    args=(40.0,),
                    options={'maxiter': 300, 'ftol': 1e-12}
                )
                h_new = project_feasible(np.asarray(res.x), n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
            except Exception:
                pass

    # Phase 3: Coordinate descent polish on best solution
    if time.time() - t0 < budget_s - 15:
        n = len(best_h)
        h_pol = best_h.copy()
        for _ in range(5):
            if time.time() - t0 > budget_s - 10:
                break
            improved = False
            for i in range(n):
                if time.time() - t0 > budget_s - 10:
                    break
                for delta in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02]:
                    h_tmp = h_pol.copy()
                    h_tmp[i] = np.clip(h_tmp[i] + delta, 0.0, 1.0)
                    # Adjust another coordinate to maintain sum
                    diff = (n / 2.0) - np.sum(h_tmp)
                    if abs(diff) < 1e-14:
                        pass
                    else:
                        # Spread the correction
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        if diff > 0:
                            mask2 = mask & (h_tmp < 1.0 - 1e-12)
                        else:
                            mask2 = mask & (h_tmp > 1e-12)
                        cnt = int(np.sum(mask2))
                        if cnt == 0:
                            continue
                        h_tmp[mask2] += diff / cnt
                        h_tmp = np.clip(h_tmp, 0.0, 1.0)
                    c5_tmp = compute_c5(h_tmp)
                    if c5_tmp < compute_c5(h_pol) - 1e-10:
                        h_pol = h_tmp
                        improved = True
                        break
            if not improved:
                break
        c5_pol = compute_c5(project_feasible(h_pol, n))
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = project_feasible(h_pol, n)

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, float(best_c5), len(best_h))
