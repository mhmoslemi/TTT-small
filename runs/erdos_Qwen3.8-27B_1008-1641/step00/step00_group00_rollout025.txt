def corr_gradient(h, weights):
    n = len(h)
    n_corr = 2 * n - 1

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def compute_c5(h, dx):
    full_corr = np.correlate(h, 1 - h, mode="full")
    return np.max(full_corr) * dx

def smooth_max_val(y, beta):
    m = np.max(y)
    if beta * (m - np.min(y)) > 400:
        return m
    e = np.exp(beta * (y - m))
    return m + np.log(np.sum(e)) / beta

def smooth_max_weights(y, beta):
    m = np.max(y)
    if beta * (m - np.min(y)) > 400:
        w = np.zeros_like(y)
        w[np.argmax(y)] = 1.0
        return w
    e = np.exp(beta * (y - m))
    return e / np.sum(e)

def corr_gradient(h, weights):
    n = len(h)
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for m in range(n_corr):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
        else:
            kk = -k
            if kk < n:
                grad[:n - kk] += w * (1.0 - h[kk:])
                grad[kk:] -= w * h[:n - kk]
    return grad

def project_box_sum(v, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def upscale(h, new_n):
    n = len(h)
    x_o = (np.arange(n) + 0.5) * (2.0 / n)
    x_n = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_n, x_o, h)
    return project_box_sum(h_new, new_n / 2.0)

def subgrad_opt(h0, n, max_iter=300, time_limit=60.0):
    dx = 2.0 / n
    h = project_box_sum(np.array(h0, dtype=float), n / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    t0 = time.time()
    beta = 12.0
    step = 0.04
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-12:
            best_c5, best_h, stall = c5, h.copy(), 0
        else:
            stall += 1
        if stall > 60:
            beta = min(beta * 2.0, 300.0)
            step = max(step * 0.35, 5e-6)
            stall = 0
            if step < 1e-5:
                break
        sw = smooth_max_weights(corr, beta)
        grad = corr_gradient(h, sw) * dx
        if np.linalg.norm(grad) < 1e-11:
            break
        smooth_curr = smooth_max_val(corr, beta) * dx
        s = step
        for _ in range(12):
            h_new = project_box_sum(h - s * grad, n / 2.0)
            cn = np.correlate(h_new, 1 - h_new, mode="full")
            sv = smooth_max_val(cn, beta) * dx
            if sv < smooth_curr - 1e-12:
                h = h_new
                break
            s *= 0.45
        step = s
        if s < 1e-7:
            break
    return best_h, best_c5

def scipy_opt(h0, n, beta, maxiter=150):
    dx = 2.0 / n

    def obj(x):
        return smooth_max_val(np.correlate(x, 1 - x, mode="full"), beta) * dx

    def jac(x):
        c = np.correlate(x, 1 - x, mode="full")
        return corrGradient(x, smooth_max_weights(c, beta)) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = sp_min(obj, np.array(h0, dtype=float), method='SLSQP',
                     jac=jac, bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        hx = project_box_sum(res.x, n / 2.0)
        return hx, compute_c5(hx, dx)
    except Exception:
        return np.array(h0, dtype=float), compute_c5(h0, dx)

def corrGradient(h, weights):
    return corr_gradient(h, weights)

def rand_h(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)

def struct_h(n, rng, m):
    if m == 0:
        a = rng.randint(n // 3, 2 * n // 3)
        h = np.zeros(n)
        h[:a] = 1.0
    elif m == 1:
        c = rng.uniform(0.3, 1.7)
        w = rng.uniform(0.2, 0.8)
        x = np.linspace(0, 2, n)
        h = 0.5 * (1 + np.tanh((x - c) / w))
    else:
        h = np.tile(np.array([1.0, 0.6, 0.4])[:n % 3 + 1], n // 3 + 1)[:n]
    return project_box_sum(h, n / 2.0)

    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 6.0 - (time.time() - t_global))

    best_h, best_c5 = None, float('inf')

    # Get initial
    h_init = None
