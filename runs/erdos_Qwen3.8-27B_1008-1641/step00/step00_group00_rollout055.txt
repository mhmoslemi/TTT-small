import numpy as np
from scipy.optimize import minimize
import time

def project_to_feasible(v, target_sum):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def gradient_active(h, dx, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)
    m1 = (j - lag >= 0) & (j - lag < n)
    grad[m1] += (1.0 - h[j[m1] - lag])
    m2 = (j + lag >= 0) & (j + lag < n)
    grad[m2] -= h[j[m2] + lag]
    return grad * dx

def _slsqp_optimize(h0, f_grad, bounds, cons, maxiter, ftol):
    try:
        r = minimize(f_grad, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if hasattr(r, 'fun') and np.isfinite(r.fun):
            return r.x, float(r.fun)
    except Exception:
        pass
    return h0, None
