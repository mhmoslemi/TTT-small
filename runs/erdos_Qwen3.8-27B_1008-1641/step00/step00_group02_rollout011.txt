import numpy as np
import time

MULTIPLIERS = (0.25, 0.5, 1.0, 2.0, 4.0)
_LAG_SIGN_CACHE = {}


def get_lag_sign(n):
    v = _LAG_SIGN_CACHE.get(n)
    if v is not None:
        return v
    n = int(n)
    if n <= 1:
        _LAG_SIGN_CACHE[n] = 1
        return 1
    a = np.zeros(n, dtype=float)
    a[0] = 1.0
    c = np.correlate(a, 1.0 - a, mode='full')
    mid = n - 1
    left = float(np.sum(c[:mid]))
    right = float(np.sum(c[mid + 1:]))
    s = 1 if right >= left else -1
    _LAG_SIGN_CACHE[n] = s
    return s


def c_values(h, dx):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def smooth_value(c, beta):
    if beta <= 1e-8:
        return float(np.max(c))
    m = float(np.max(c))
    e = np.exp(beta * (c - m))
    s = float(np.sum(e))
    if s <= 1e-300:
        return m
    return m + float(np.log(s)) / beta


def softmax_weights(c, beta):
    m = float(np.max(c))
    if beta <= 1e-8:
        w = np.ones_like(c, dtype=float)
        w /= float(c.size)
        return w
    e = np.exp(beta * (c - m))
    s = float(np.sum(e))
    if s <= 1e-300:
        w = np.zeros_like(c, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def raw_grad_weights(h, W, n):
    n = int(n)
    W = np.asarray(W, dtype=float)
    if W.size != 2 * n - 1:
        W = np.resize(W, 2 * n - 1)
    mid = n - 1
    csum = np.empty(2 * n, dtype=float)
    csum[0] = 0.0
    csum[1:] = np.cumsum(W)
    upper = csum[mid + n : mid : -1]
    lower = csum[mid::-1]
    A = upper - lower
    Wrev = W[::-1]
    B = np.convolve(h, Wrev, 'full')[mid : mid + n]
    C = np.convolve(h, W, 'full')[mid : mid + n]
    return A - B - C

                idx = int(idxs[int(np.argmax(1.0 - h[idxs]))])
            else:
                idx = int(idxs[int(np.argmin(h[idxs]))])
        else:
            idx = int(np.argmax(1.0 - h)) if r > 0.0 else int(np.argmin(h))
        nv = float(target) - (s - h[idx])
        if nv < 0.0:
            nv = 0.0
        elif nv > 1.0:
            nv = 1.0
        h[idx] = nv
    return h


def project_feasible(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = int(x.size)
    if n == 0:
        return x
    if not np.all(np.isfinite(x)):
        x.fill(0.5)
    mn = float(np.min(x))
    mx = float(np.max(x))
    if mn >= -1e-13 and mx <= 1.0 + 1e-13:
        xc = x.copy()
        np.clip(xc, 0.0, 1.0, out=xc)
        if abs(float(np.sum(xc)) - float(target)) <= 1e-10:
            return exact_sum_fix(xc, target)
    lo = -mx - 1e-12
    hi = 1.0 - mn + 1e-12
    for _ in range(45):
        midv = 0.5 * (lo + hi)
        y = np.clip(x + midv, 0.0, 1.0)
        if float(np.sum(y)) < float(target):
            lo = midv
        else:
            hi = midv
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    return exact_sum_fix(h, target)


def zero_mean_gaussian(n, rng, scale=1.0):
    n = int(n)
    v = rng.standard_normal(n)
    v -= float(v.mean())
    v *= float(scale)
    return v


def random_binary_feasible(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    half = n // 2
    if half > 0:
        h[:half] = 1.0
    if n % 2 == 1:
        h[half] = 0.5
    if n > 1:
        h = h[rng.permutation(n)]
    return h


def resample_h(h_old, old_n, new_n):
    h_old = np.asarray(h_old, dtype=float)
    old_n = int(old_n)
    new_n = int(new_n)
    if new_n == old_n:
        return h_old.copy()
    if new_n % old_n == 0:
        return np.repeat(h_old, new_n // old_n)
    x_old = (np.arange(old_n, dtype=float) + 0.5) * (2.0 / float(old_n))
    x_new = (np.arange(new_n, dtype=float) + 0.5) * (2.0 / float(new_n))
    return np.interp(x_new, x_old, h_old)


def interp_upsample(h_old, old_n, new_n):
    h_old = np.asarray(h_old, dtype=float)
    old_n = int(old_n)
    new_n = int(new_n)
    if new_n == old_n:
        return h_old.copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) * (2.0 / float(old_n))
    x_new = (np.arange(new_n, dtype=float) + 0.5) * (2.0 / float(new_n))
    return np.interp(x_new, x_old, h_old)


def local_refine(h, n, target, dx, rng, time_limit,
                 max_iter=30000, beta_init=20.0, beta_final=15000.0,
                 step_init=0.12):
    n = int(n)
    h = project_feasible(h, target)
    dx = float(dx)
    t_start = time.time()
    if time_limit <= 0.05:
        return h, float(np.max(c_values(h, dx)))

    lag = get_lag_sign(n)
    cvals = c_values(h, dx)
    best_h = h.copy()
    best_c = float(np.max(cvals))
    beta = float(beta_init)
    step = float(step_init)
    stall_soft = 0
    stall_hard = 0
    it = 0

    while it < int(max_iter):
        if time.time() - t_start >= time_limit:
            break

        hard = float(np.max(cvals))
        if hard < best_c - 1e-12:
            best_c = hard
            best_h = h.copy()
            stall_hard = 0
        else:
            stall_hard += 1

        if stall_hard > 80:
            if beta < beta_final:
                beta = min(float(beta_final), beta * 1.3 + 1.0)
        else:
            if beta < beta_final:
                beta = min(float(beta_final), beta * 1.01 + 0.05)

        w = softmax_weights(cvals, beta)
        W = w if lag > 0 else w[::-1]
        g = dx * raw_grad_weights(h, W, n)
        g -= float(np.mean(g))
        if not np.all(np.isfinite(g)):
            break

        norm = float(np.linalg.norm(g))
        if norm < 1e-9:
            scale = 0.03 + min(0.30, step * 2.0)
            h = project_feasible(h + scale * zero_mean_gaussian(n, rng), target)
            cvals = c_values(h, dx)
            step = max(0.03, step * 0.7)
            stall_soft = 0
            it += 1
            continue

        direction = -g / norm
        Fcur = smooth_value(cvals, beta)
        best_F = Fcur
        best_hard = hard
        best_hc = None
        best_cvals = None

        for mult in MULTIPLIERS:
            if time.time() - t_start >= time_limit:
                break
            s2 = step * float(mult)
            if s2 < 1e-5 or s2 > 0.85:
                continue
            hp = project_feasible(h + s2 * direction, target)
            cn = c_values(hp, dx)
            fn = smooth_value(cn, beta)
            hn = float(np.max(cn))
            if fn < best_F - 1e-11 or (abs(fn - best_F) < 1e-10 and hn < best_hard - 1e-10):
                best_F = fn
                best_hard = hn
                best_hc = hp
                best_cvals = cn

        if best_hc is not None and (best_F < Fcur - 1e-12 or best_hard < hard - 1e-10):
            h = best_hc
            cvals = best_cvals
            step = min(0.65, step * 1.12)
            stall_soft = 0
        else:
            step *= 0.75
            stall_soft += 1
            if stall_soft > 35:
                scale = 0.04 + min(0.35, step * 2.0)
                h = project_feasible(h + scale * zero_mean_gaussian(n, rng), target)
                cvals = c_values(h, dx)
                step = min(0.35, max(0.04, scale))
                beta = max(float(beta_init), beta * 0.5)
                stall_soft = 0
            if step < 1e-5:
                break
        it += 1

    if np.all(np.isfinite(h)):
        hard = float(np.max(c_values(h, dx)))
        if hard < best_c:
            best_c = hard
            best_h = h.copy()

    return project_feasible(best_h, target), float(best_c)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    budget = float(budget_s)
    if budget <= 1.0:
        budget = 1.0

    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)

    if init is not None:
        h0 = np.asarray(init, dtype=float).ravel()
    else:
        h0 = random_binary_feasible(45, rng)

    if h0.size == 0:
        h0 = random_binary_feasible(45, rng)

    n0 = int(h0.size)
    target0 = 0.5 * float(n0)
    best_h = project_feasible(h0, target0)
    best_n = n0
    best_c = float(np.max(c_values(best_h, 2.0 / float(best_n))))

    if not np.isfinite(best_c):
        best_h = random_binary_feasible(n0, rng)
        best_c = float(np.max(c_values(best_h, 2.0 / float(best_n))))

    if budget < 120.0:
        max_multiplier = 2
    elif budget < 300.0:
        max_multiplier = 3
    elif budget < 650.0:
        max_multiplier = 5
    else:
        max_multiplier = 10

    base_factors = (1, 2, 3, 5, 8, 10)
    stage_ns = []
    stage_ws = []
    for f in base_factors:
        if f > max_multiplier:
            break
        ns = int(n0 * f)
        if ns < 1 or ns > 999:
            continue
        if ns in stage_ns:
            continue
        stage_ns.append(ns)
        stage_ws.append(1.0 + 0.2 * len(stage_ns))

    if not stage_ns:
        stage_ns = [n0]
        stage_ws = [1.0]

    for si in range(len(stage_ns)):
        ns = int(stage_ns[si])
        rem = budget - (time.time() - t_start)
        if rem < 25.0:
            break

        rem_w = 0.0
        for sj in range(si, len(stage_ns)):
            rem_w += stage_ws[sj]
        stage_time = rem * (stage_ws[si] / rem_w) * 0.98
        if stage_time < 4.0:
            continue

        targetn = 0.5 * float(ns)
        dxn = 2.0 / float(ns)

        h_main = resample_h(best_h, best_n, ns)

        if si == 0:
            t_main = stage_time * 0.55
            t_pert = stage_time * 0.30
            t_rand = stage_time * 0.15
        else:
            t_main = stage_time * 0.78
            t_pert = stage_time * 0.22
            t_rand = 0.0

        beta_i = 20.0 + 0.05 * float(ns)
        beta_f = 15000.0
        step_i = 0.12 if ns <= 225 else 0.09

        if t_main > 3.0:
            hc, cc = local_refine(h_main, ns, targetn, dxn, rng, t_main,
                                  30000, beta_i, beta_f, step_i)
            if cc < best_c:
                best_c = float(cc)
                best_h = hc
                best_n = ns

        rem = budget - (time.time() - t_start)

        if t_pert > 3.0 and rem > 35.0:
            h_base = interp_upsample(best_h, best_n, ns)
            scale = 0.10 if ns <= 225 else 0.06
            h_p = project_feasible(h_base + scale * zero_mean_gaussian(ns, rng), targetn)
            hc, cc = local_refine(h_p, ns, targetn, dxn, rng, t_pert,
                                  30000, max(30.0, beta_i), beta_f, min(0.10, step_i))
            if cc < best_c:
                best_c = float(cc)
                best_h = hc
                best_n = ns

        if t_rand > 3.0 and rem > 35.0 and ns <= 225:
            h_r = random_binary_feasible(ns, rng)
            hc, cc = local_refine(h_r, ns, targetn, dxn, rng, t_rand,
                                  30000, 15.0 + 0.02 * float(ns), 5000.0, 0.15)
            if cc < best_c:
                best_c = float(cc)
                best_h = hc
                best_n = ns

    rem = budget - (time.time() - t_start)
    if rem > 30.0:
        targetb = 0.5 * float(best_n)
        dxb = 2.0 / float(best_n)
        h_b = project_feasible(best_h, targetb)
        t_pol = min(rem * 0.6, 180.0)
        hc, cc = local_refine(h_b, best_n, targetb, dxb, rng, t_pol,
                              30000, 80.0 + 0.05 * float(best_n), 30000.0, 0.05)
        if cc < best_c:
            best_c = float(cc)
            best_h = hc

        rem = budget - (time.time() - t_start)
        if rem > 15.0 and best_n <= 400:
            h_p = project_feasible(best_h + 0.05 * zero_mean_gaussian(best_n, rng), targetb)
            hc, cc = local_refine(h_p, best_n, targetb, dxb, rng,
                                  min(rem * 0.7, 120.0),
                                  30000, 100.0 + 0.05 * float(best_n), 30000.0, 0.04)
            if cc < best_c:
                best_c = float(cc)
                best_h = hc

    saved_h = best_h.copy()
    saved_c = float(best_c)
    saved_n = int(best_n)

    if np.min(best_h) < -1e-9 or np.max(best_h) > 1.0 + 1e-9:
        best_h = np.clip(best_h, 0.0, 1.0)

    best_c = float(np.max(c_values(best_h, 2.0 / float(best_n))))
    if not np.isfinite(best_c) or best_c > saved_c + 1e-8:
        best_h = saved_h
        best_c = saved_c
        best_n = saved_n

    return best_h, float(best_c), int(best_n)
