    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    m1s = max(0, -shift)
    m1e = min(n, n - shift)
    if m1e > m1s:
        idx = np.arange(m1s, m1e)
        g[idx] += 1.0 - h[idx + shift]
    m2s = max(0, shift)

    if m2e > m2s:
        idx = np.arange(m2s, m2e)
        g[idx] -= h[idx - shift]
