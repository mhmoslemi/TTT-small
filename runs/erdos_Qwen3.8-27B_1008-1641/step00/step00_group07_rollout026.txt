def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        res = target - h.sum()
        if abs(res) < 1e-12:
            break
        if res > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += res / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += res / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def soft_max_obj_and_p(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = corr.max()
    exp_c = np.exp((corr - max_c) / T)
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    obj = max_c + T * np.log(sum_exp)
    return obj, p

    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)

        valid1 = (j + lags >= 0) & (j + lags < n)
        valid2 = (j - lags >= 0) & (j - lags < n)
        grad[j] = dx * (np.dot(p[valid1], 1.0 - h[j + lags[valid1]]) -
                        np.dot(p[valid2], h[j - lags[valid2]]))
    return grad
