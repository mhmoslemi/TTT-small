            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_try = h_new.copy()
            else:
                h_try = h_new

    # Phase 3: Fine refinement at best resolution
    if elapsed() < budget_s * 0.85:
        n_fin = len(best_h)
        temps_fine = [0.08, 0.04, 0.02, 0.01]
        for t in temps_fine:
            if elapsed() > budget_s * 0.95:
                break
            h_new = optimize_step(best_h, n_fin, t, 50000.0, 150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Coordinate-descent style local search
    if elapsed() < budget_s * 0.93:
        n_cd = len(best_h)
        dx = 2.0 / n_cd
        improved = True
        while improved and elapsed() < budget_s * 0.95:
            improved = False
            for i in range(n_cd):
                for delta in [0.02, -0.02, 0.01, -0.01, 0.05, -0.05]:
                    h_test = best_h.copy()
                    h_test[i] = best_h[i] + delta
                    # Compensate to maintain sum
                    comp = (n_cd / 2.0 - h_test.sum()) / (n_cd - 1)
                    for j in range(n_cd):
                        if j != i:
                            h_test[j] += comp
                    if np.all(h_test >= 0) and np.all(h_test <= 1):
                        c5_t = compute_c5(h_test)
                        if c5_t < best_c5 - 1e-10:
                            best_c5 = c5_t
                            best_h = h_test.copy()
                            improved = True
                            break

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
