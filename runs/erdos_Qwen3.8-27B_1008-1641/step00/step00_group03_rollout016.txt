import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def compute_c5_values(h):
    """Return the full correlation * dx array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_c5(h):
    return np.max(compute_c5_values(h))

def smooth_obj(h, alpha):
    vals = compute_c5_values(h)
    m = np.max(vals)
    s = np.sum(np.exp(alpha * (vals - m)))
    return m + np.log(s) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    vals = compute_c5_values(h)
    m = np.max(vals)
    shifted = alpha * (vals - m)
    max_shifted = np.max(shifted)
    exps = np.exp(shifted - max_shifted)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for m_idx in range(n_lags):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            # i-k valid: i in [k, n-1], contribution (1-h[i-k])
            # i+k valid: i in [0, n-1-k], contribution -h[i+k]
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] += w * (-h[k:])
        else:
            ak = -k
            # i+ak valid: i in [0, n-1-ak], contribution (1-h[i+ak])
            # i-ak valid: i in [ak, n-1], contribution -h[i-ak]
            grad[:n-ak] += w * (1.0 - h[ak:])
            grad[ak:] += w * (-h[:n-ak])
    return grad * dx

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda: find lambda such that sum(clamp(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fine-tune: adjust sum exactly
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-10:
        # Distribute among free variables
        free = (proj > 1e-8) & (proj < 1 - 1e-8)
        if np.any(free):
            proj[free] += diff / np.sum(free)
            proj = np.clip(proj, 0.0, 1.0)
            # Final correction
            diff2 = target_sum - np.sum(proj)
            if abs(diff2) > 1e-10:
                all_free = (proj > 0) & (proj < 1)
                if np.any(all_free):
                    proj[all_free] += diff2 / np.sum(all_free)
    return proj

def proj_gradient_step(h, target_sum, alpha, lr, n_inner):
    """Projected gradient descent."""
    for _ in range(n_inner):
        g = smooth_grad(h, alpha)
        h_new = h - lr * g
        h = project_feasible(h_new, target_sum)
    return h

def slsqp_optimize(h0, n, alpha):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return smooth_obj(x, alpha)
    
    def grad(x):
        return smooth_grad(x, alpha)
    
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 200, 'ftol': 1e-10})
    if res.success or res.fun < compute_c5(h0):
        return project_feasible(res.x, target)
    return h0

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h):
    return float(np.max(compute_c5_vals(h)))


def smooth_obj(h, alpha):
    vals = compute_c5_vals(h)
    m = np.max(vals)
    max_s = alpha * (vals - m)
    return m + np.log(np.sum(np.exp(max_s))) / alpha


def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    vals = compute_c5_vals(h)
    m = np.max(vals)
    sh = alpha * (vals - m)
    sh -= np.max(sh)
    exps = np.exp(sh)
    tw = np.sum(exps)
    weights = exps / tw
    grad = np.zeros(n)
    nlags = 2 * n - 1
    for m_idx in range(nlages):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] += w * (-h[k:])
        else:
            ak = -k
            grad[:n - ak] += w * (1.0 - h[ak:])
            grad[ak:] += w * (-h[:n - ak])
    return grad * dx


def project_feasible(h, target):

            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        free = (proj > 1e-10) & (proj < 1.0 - 1e-10)
