def project(h, n):
    """Project h to feasible set: 0<=h[i]<=1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    target = n / 2.0
    if s > 0:
        # Adjust to meet sum constraint
        diff = target - s
        if diff > 0:
            # Need to increase: add to elements not at 1
            can_add = 1.0 - h
            if can_add.sum() > 0:
                if can_add.sum() >= diff:
                    h = h + diff * can_add / can_add.sum()
                else:
                    h = h + can_add  # all at 1
        else:
            # Need to decrease: reduce elements not at 0
            can_sub = h
            if can_sub.sum() > 0:
                if can_sub.sum() >= -diff:
                    h = h + diff * can_sub / can_sub.sum()
                else:
                    h = np.zeros_like(h)
    h = np.clip(h, 0, 1)
    # Final fix: if still off, distribute evenly
    s = np.sum(h)

        h = h + (target - s) / n
        h = np.clip(h, 0, 1)
