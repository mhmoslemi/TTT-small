from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def proj_sum(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(free):

        h[free] += diff / max(1, np.sum(free))
