    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new)

def binary_local_search(h_bin, n, max_passes):
    best_c5 = compute_c5(h_bin)
    for _ in range(max_passes):
        improved = False
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
        for i in ones:
            hit = False
            for j in zeros:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c = compute_c5(h_bin)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    improved = True
                    hit = True
                else:
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
            if hit:
                zeros = np.where(h_bin < 0.5)[0]
        if not improved:
            break
    return h_bin, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Annealed gradient from initial
    temps = [15.0, 40.0, 100.0, 250.0, 500.0, 1000.0]
    for T in temps:
        if time.time() - t0 > budget_s - 150:
            break
        h_new, c5_new = gradient_optimize(best_h, n, T, 5000, 25000)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Multi-start perturbation
    for r in range(15):
        if time.time() - t0 > budget_s - 200:
            break
        scale = 0.35 - 0.015 * r
        h_start = best_h + np.random.randn(n) * scale
        h_start = project_feasible(h_start, n)
        for T in [150.0, 400.0]:
            if time.time() - t0 > budget_s - 80:
                break
            h_new, c5_new = gradient_optimize(h_start, n, T, 2500, 12000)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new

    # Phase 3: Binary local search
    if time.time() - t0 < budget_s - 90:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.round(np.sum(h_bin)))
        target = n // 2
        if n_ones != target:
            if n_ones > target:
                idx = np.where(h_bin > 0.5)[0]
                order = np.argsort(best_h[idx])
                for k in range(n_ones - target):
                    h_bin[idx[order[k]]] = 0.0
            else:
                idx = np.where(h_bin < 0.5)[0]
                order = np.argsort(best_h[idx])[::-1]
                for k in range(target - n_ones):
                    h_bin[idx[order[k]]] = 1.0
        h_bin, c5_bin = binary_local_search(h_bin, n, 300)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 3b: Simulated annealing on binary
    if time.time() - t0 < budget_s - 60:
        h_sa = best_h.copy()
        if np.all((h_sa > 0.5) | (h_sa < 0.5)):
            h_sa = (h_sa > 0.5).astype(float)
            n_ones = int(np.sum(h_sa))
            target = n // 2
            if n_ones != target:
                if n_ones > target:
                    idx = np.where(h_sa > 0.5)[0]
                    for k in range(n_ones - target):
                        h_sa[idx[np.random.randint(len(idx))]] = 0.0
                else:
                    idx = np.where(h_sa < 0.5)[0]
                    for k in range(target - n_ones):
                        h_sa[idx[np.random.randint(len(idx))]] = 1.0
            c5_sa = compute_c5(h_sa)
            T_sa = 0.01
            for step in range(200000):
                if time.time() - t0 > budget_s - 30:
                    break
                ones = np.where(h_sa > 0.5)[0]
                zeros = np.where(h_sa < 0.5)[0]
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h_sa[i] = 0.0
                h_sa[j] = 1.0
                c_new = compute_c5(h_sa)
                if c_new < c5_sa or np.random.rand() < np.exp(-(c_new - c5_sa) / T_sa):
                    c5_sa = c_new
                else:
                    h_sa[i] = 1.0
                    h_sa[j] = 0.0
                T_sa *= 0.9998
                if T_sa < 1e-6:
                    T_sa = 1e-6
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 4: Other grid sizes
    for n_try in [50, 56, 64, 72, 80, 90, 100]:
        if time.time() - t0 > budget_s - 25:
            break
        if n_try == n:
            continue
        old_n = len(best_h)
        old_x = (np.arange(old_n) + 0.5) * 2.0 / old_n
        new_x = (np.arange(n_try) + 0.5) * 2.0 / n_try
        h_interp = np.interp(new_x, old_x, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_new, c5_new = gradient_optimize(h_interp, n_try, 300.0, 3000, 15000)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
