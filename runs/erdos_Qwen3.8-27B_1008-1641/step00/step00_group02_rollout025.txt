def project_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = float(target - h.sum())
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(h)
    else:
        order = np.argsort(h)[::-1]
    for idx in order:
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h[idx]
            if room <= 0: continue
            add = min(room, diff)
            h[idx] += add
            diff -= add
        else:
            room = h[idx]
            if room <= 0: continue
            sub = min(room, -diff)
            h[idx] -= sub
            diff += sub
    # Final tiny fix: if still diff due to float, apply to an eligible
    if abs(diff) > 1e-15:
        if diff > 0:
           idx = int(np.argmin(h))
           h[idx] = min(1.0, h[idx] + diff)
        else:
           idx = int(np.argmax(h))
           h[idx] = max(0.0, h[idx] + diff)
    return h

def solve_binary_milp(n, time_limit):
    from scipy.optimize import milp, LinearConstraint
    from scipy.sparse import coo_matrix
    m = 2*n - 1
    p = n*(n-1)//2
    V = n + p + 1
    u, v = np.triu_indices(n, 1)
    pid = np.full((n,n), -1, dtype=np.int64)
    pid[u,v] = np.arange(p)
    pid[v,u] = np.arange(p)
    rows=[]; cols=[]; vals=[]; rhs=[]
    # y <= h_i
    for idd in range(p):
       i=int(u[idd]); j=int(v[idd]); y = n + idd
       r = len(rhs)
       rows.append(r); cols.append(y); vals.append(1.0); rhs.append(0.0)
       rows.append(r); cols.append(i); vals.append(-1.0)
       # y <= h_j
       r = len(rhs)
       rows.append(r); cols.append(y); vals.append(1.0); rhs.append(0.0)
       rows.append(r); cols.append(j); vals.append(-1.0)
       # h_i + h_j - y <= 1
       r = len(rhs)
       rows.append(r); cols.append(i); vals.append(1.0); rhs.append(1.0)
       rows.append(r); cols.append(j); vals.append(1.0)
       rows.append(r); cols.append(y); vals.append(-1.0)
    pairs=build_pairs(n)
    t_var=V-1
    for ii_idx, (iarr, jarr) in enumerate(pairs):
       r = len(rhs)
       for i,j in zip(iarr, jarr):
          i=int(i); j=int(j)
          if i == j: continue
          rows.append(r); cols.append(i); vals.append(1.0)
          iddel=int(pid[i,j])
          rows.append(r); cols.append(n+iddel); vals.append(-1.0)
       rows.append(r); cols.append(t_var); vals.append(-1.0)
       rhs.append(0.0)
    A_ub = coo_matrix((vals,(rows,cols)), shape=(len(rhs), V)).tocsr()
    eq_rows = [0]*n
    eq_cols = list(range(n))
    eq_vals = [1.0]*n
    A_eq = coo_matrix((eq_vals, (eq_rows, eq_cols)), shape=(1,V)).tocsr()
    bounds = [(0.0,1.0)]*n + [(0.0,1.0)]*p + [(0.0, float('inf'))]
    integrality = np.zeros(V, dtype=int); integrality[:n]=1
    c = np.zeros(V); c[-1]=1.0
    cons = [LinearConstraint(A_ub, -np.inf, np.asarray(rhs, dtype=float)),
            LinearConstraint(A_eq, float(n/2.0), float(n/2.0))]
    options = {'time_limit': float(time_limit), 'mip_rel_gap': 0.0, 'presolve': True}
    try:

def vertex_from_q(q, target):

    k = int(np.floor(target + 1e-12))

    if k > 0:

    frac = target - k
    if frac > 1e-12 and k < n:
        e[order[k]] = frac
    return e

def project_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    diff = float(target - h.sum())
    if abs(diff) < 1e-13:
        return h
    order = np.argsort(h) if diff > 0 else np.argsort(h)[::-1]
    for idx in order:
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h[idx]
            if room > 0:
                a = min(room, diff); h[idx] += a; diff -= a
        else:
            room = h[idx]
            if room > 0:
                a = min(room, -diff); h[idx] -= a; diff += a
    if abs(diff) > 1e-15:
        if diff > 0:
            i = int(np.argmin(h)); h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmax(h)); h[i] = max(0.0, h[i] + diff)
    return h

        from scipy.optimize import milp, LinearConstraint

        if p:
            pid[u, v] = np.arange(p)
            pid[v, u] = np.arange(p)

            P = np.zeros((m, p), dtype=float)

```python
import numpy as np
import time

_PAIR_CACHE = {}
_SMALL_ALPHAS = np.array([
    0.0, 1e-8, 1e-7, 1e-6, 1e-5, 2e-5, 5e-5,
    1e-4, 2e-4, 5e-4, 1e-3, 2e-3, 5e-3, 1e-2, 2e-2, 5e-2
])
_COARSE_ALPHAS = np.unique(np.concatenate((
    _SMALL_ALPHAS,
    np.linspace(0.0, 1.0, 41),
    np.logspace(-5.0, 0.0, 16)
)))
_COARSE_ALPHAS = _COARSE_ALPHAS[(_COARSE_ALPHAS >= 0.0) & (_COARSE_ALPHAS <= 1.0)]


def eval_c5_fast(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    diff = float(target - h.sum())
    if abs(diff) < 1e-13:
        return h
    order = np.argsort(h) if diff > 0 else np.argsort(h)[::-1]
    for i in order:
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h[i]
            if room > 0.0:
                a = min(room, diff)
                h[i] += a
                diff -= a
        else:
            room = h[i]
            if room > 0.0:
                a = min(room, -diff)
                h[i] -= a
                diff += a
    if abs(diff) > 1e-15:
        i = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[i] = min(1.0, max(0.0, h[i] + diff))
    return h


def make_start(n, rng, kind, phase):
    if kind == 0:
        h = rng.random(n)
    else:
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h = 0.5 + 0.5 * np.sin(2.0 * np.pi * float(kind) * x + phase)
        h = np.clip((h - 0.32) / 0.36, 0.0, 1.0)
    return project_sum(h, n / 2.0)


def resample_step(h, n):
    m = len(h)
    if m == n:
        return project_sum(h, n / 2.0)
    x0 = (np.arange(m) + 0.5) * (2.0 / m)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    hh = np.interp(x, x0, h, left=float(h[0]), right=float(h[-1]))
    return project_sum(hh, n / 2.0)


def sort_starts(starts):
    n = len(starts)
    for i in range(n):
        m = i
        for j in range(i + 1, n):
            if starts[j][2] < starts[m][2]:
                m = j
        if m != i:
            starts[i], starts[m] = starts[m], starts[i]
    return starts


def get_dynamic_pairs(n):
    if n in _PAIR_CACHE:
        return _PAIR_CACHE[n]
    eye = np.eye(n)
    m = 2 * n - 1
    groups = [[] for _ in range(m)]
    for a in range(n):
        ea = eye[a]
        for b in range(n):
            idx = int(np.argmax(np.correlate(ea, eye[b], mode='full')))
            groups[idx].append((a, b))
    out = []
    for idx in range(m):
        if groups[idx]:
            ia = np.array([z[0] for z in groups[idx]], dtype=int)
            ja = np.array([z[1] for z in groups[idx]], dtype=int)
        else:
            ia = np.empty(0, dtype=int)
            ja = np.empty(0, dtype=int)
        out.append((ia, ja))
    _PAIR_CACHE[n] = out
    return out


def line_search(h, e, f0, dx, target):
    d = e - h
    d = d - d.mean()
    if float(np.max(np.abs(d))) < 1e-15:
        return h, f0
    alphas = _COARSE_ALPHAS
    best_f = f0
    best_a = 0.0
    best_idx = 0
    for ii, a in enumerate(alphas):
        fa = eval_c5_fast(h + float(a) * d, dx)
        if fa < best_f - 1e-15:
            best_f = fa
            best_a = float(a)
            best_idx = ii
    if best_f < f0 - 1e-15:
        if best_idx == 0:
            left, right = 0.0, float(alphas[1])
        elif best_idx == len(alphas) - 1:
            left, right = float(alphas[-2]), 1.0
        else:
            left, right = float(alphas[best_idx - 1]), float(alphas[best_idx + 1])
        if right > left + 1e-10:

                if fa < best_f - 1e-15:
                    best_f = fa
                    best_a = float(a)

        fnew = eval_c5_fast(hnew, dx)
        if fnew < f0 - 1e-15:

            if ff > 1e-12:
                rem = np.nonzero(mask)[0]
                if rem.size:
                    er[int(rng.choice(rem))] = ff
            hnew, fnew = line_search(best, er, bf, dx, target)
            if fnew < bf - 1e-15:
                best = hnew
                bf = fnew
                bi = 0
                stall = 0
                continue

        bi += 1
        if stall > 80:
            hp = project_sum(best + 0.02 * (rng.random(n) - 0.5), target)
            fp = eval_c5_fast(hp, dx)
            if fp < bf:
                best = hp
                bf = fp
            bi = 0
            stall = 0

    best = project_sum(best, target)
    return best, eval_c5_fast(best, dx)


def solve_binary_milp(n, time_limit):
    if n < 2 or time_limit <= 0.0:
        return None
    try:
        from scipy.optimize import milp, LinearConstraint
    except Exception:
        return None

    try:
        p = n * (n - 1) // 2
        V = n + p + 1
        m = 2 * n - 1
        pairs = get_dynamic_pairs(n)
        u, v = np.triu_indices(n, 1)
        pid = np.full((n, n), -1, dtype=int)
        if p:
            pid[u, v] = np.arange(p)
            pid[v, u] = np.arange(p)

        L = np.zeros((m, n), dtype=float)
        P = np.zeros((m, p), dtype=float) if p else np.zeros((m, 0), dtype=float)
        for idx, (ia, ja) in enumerate(pairs):
            for a, b in zip(ia, ja):
                a = int(a)
                b = int(b)
                if a != b:
                    L[idx, a] += 1.0
                    P[idx, int(pid[a, b])] += 1.0

        A_ub = np.zeros((3 * p + m, V), dtype=float)
        b_ub = np.zeros(3 * p + m, dtype=float)
        r = 0
        for idd in range(p):
            i = int(u[idd])
            j = int(v[idd])
            y = n + idd
            A_ub[r, y] = 1.0
            A_ub[r, i] = -1.0
            r += 1
            A_ub[r, y] = 1.0
            A_ub[r, j] = -1.0
            r += 1
            A_ub[r, i] = 1.0
            A_ub[r, j] = 1.0
            A_ub[r, y] = -1.0
            b_ub[r] = 1.0
            r += 1

        for rr in range(m):
            A_ub[r, :n] = L[rr]
            if p:
                A_ub[r, n:n + p] = -P[rr]
            A_ub[r, V - 1] = -1.0
            r += 1

        A_eq = np.zeros((1, V), dtype=float)
        A_eq[0, :n] = 1.0
        bounds = [(0.0, 1.0)] * (V - 1) + [(0.0, float('inf'))]
        integrality = np.zeros(V, dtype=int)
        integrality[:n] = 1
        c = np.zeros(V, dtype=float)
        c[-1] = 1.0
        cons = [
            LinearConstraint(A_ub, -np.inf, b_ub),
            LinearConstraint(A_eq, n / 2.0, n / 2.0)
        ]
        options = {'time_limit': float(time_limit), 'presolve': True}
        res = milp(c, constraints=cons, integrality=integrality, bounds=bounds, options=options)
    except Exception:
        return None

    if res is None or res.x is None:
        return None
    hx = np.rint(np.array(res.x[:n], dtype=float))
    hx = project_sum(hx, n / 2.0)
    fx = eval_c5_fast(hx, 2.0 / n)
    if not np.isfinite(fx):
        return None
    return hx, float(fx)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    if budget <= 0.0:
        budget = 1100.0
    deadline = t0 + budget
    rng = np.random.default_rng(seed)

    starts = []
    n0 = 0
    try:
        parent = initial_h_values
    except NameError:
        parent = None

    if parent is not None:
        h0 = np.array(parent, dtype=float)
        n0 = int(h0.size)
        if n0 >= 4 and np.all(np.isfinite(h0)):
            h0 = project_sum(h0, n0 / 2.0)
            f0 = eval_c5_fast(h0, 2.0 / n0)
            if np.isfinite(f0):
                starts.append((h0.copy(), n0, f0))
            for nr in (46, 44, 48, 50):
                if nr != n0:
                    hr = resample_step(h0, nr)
                    fr = eval_c5_fast(hr, 2.0 / nr)
                    if np.isfinite(fr):
                        starts.append((hr, nr, fr))

    n_list = []
    if parent is not None and n0 > 0:
        n_list.append(n0)
    for nr in (46, 45, 44, 50):
        if nr not in n_list:
            n_list.append(nr)

    for nr in n_list:
        for kind in (0, 1, 2, 3):
            h = make_start(nr, rng, kind, 0.3 * float(kind))
            f = eval_c5_fast(h, 2.0 / nr)
            starts.append((h, nr, f))

    if not starts:
        nr = 46
        h = make_start(nr, rng, 0, 0.0)
        starts.append((h, nr, eval_c5_fast(h, 2.0 / nr)))

    sort_starts(starts)
    best_h = starts[0][0].copy()
    best_f = float(starts[0][2])
    best_n = int(starts[0][1])

    if time.time() < deadline - 25.0:
        hc, nc, fc = starts[0]
        alloc = min(90.0, max(15.0, 0.25 * (deadline - time.time())))
        end = min(deadline - 25.0, time.time() + alloc)
        hnew, fnew = improve_local(hc, nc, end, rng)
        if fnew < best_f:
            best_f = fnew
            best_h = hnew
            best_n = nc

        if len(starts) > 1 and starts[1][2] <= best_f + 1e-3 and time.time() < deadline - 35.0:
            hc, nc, fc = starts[1]
            end = min(deadline - 35.0, time.time() + 25.0)
            hnew, fnew = improve_local(hc, nc, end, rng)
            if fnew < best_f:
                best_f = fnew
                best_h = hnew
                best_n = nc

    for nm in (46, 44, 50):
        if time.time() > deadline - 60.0:
            break
        remain = deadline - time.time()
        tl = min(25.0, max(8.0, 0.15 * remain))
        tl = min(tl, max(5.0, deadline - 40.0 - time.time()))
        if tl <= 1.0:
            break
        res = solve_binary_milp(nm, tl)
        if res is not None:
            hm, fm = res
            if fm < best_f:
                best_f = fm
                best_h = hm
                best_n = nm
            starts.append((hm, nm, fm))
            sort_starts(starts)

    if time.time() < deadline - 15.0:
        remain = deadline - time.time()
        end = min(deadline - 15.0, time.time() + max(30.0, 0.65 * remain))
        hnew, fnew = improve_local(best_h, best_n, end, rng)
        if fnew < best_f:
            best_f = fnew
            best_h = hnew
            best_n = best_n

    if time.time() < deadline - 20.0:
        tried = 0
        for idx in range(min(5, len(starts))):
            if tried >= 2 or time.time() > deadline - 25.0:
                break
            hc, nc, fc = starts[idx]
            if fc > best_f + 1e-3:
                continue
            if nc == best_n and abs(fc - best_f) < 1e-6:
                continue
            end = min(deadline - 25.0, time.time() + 25.0)
            hnew, fnew = improve_local(hc, nc, end, rng)
            if fnew < best_f:
                best_f = fnew
                best_h = hnew
                best_n = nc
            tried += 1

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_f = eval_c5_fast(best_h, 2.0 / len(best_h))
    return best_h, float(best_f), int(len(best_h))
```
