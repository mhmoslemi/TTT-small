    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')

    c = corr * dx
    c_max = np.max(c)
    c_clipped = c - c_max
    soft_max = c_max + (1/tau) * np.log(np.sum(np.exp(tau * c_clipped)))

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase: distribute to elements with room
        order = np.argsort(h)  # smallest first (most room)
        for idx in order:
            add = min(1 - h[idx], diff)
            h[idx] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # Need to decrease: take from elements with value
        order = np.argsort(h)[::-1]  # largest first
        diff = -diff
        for idx in order:
            sub = min(h[idx], diff)
            h[idx] -= sub
            diff -= sub
            if diff <= 1e-12:
                break
    return h

def subgradient_max_corr(h, top_k=5):
    """Compute subgradient of max_k corr_k, using top-k lags"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Get top-k lags
    top_indices = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for k_idx in top_indices:
        k = k_idx - (n - 1)
        if k >= 0:
            for j in range(n - k):
                grad[j] += (1 - h[j + k])
                grad[j + k] += (-h[j])
        else:
            m = -k
            for j in range(m, n):
                grad[j] += (1 - h[j - m])
                grad[j - m] += (-h[j])
    return grad, corr[top_indices].max()

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        for idx in np.argsort(h):
            a = min(1.0 - h[idx], diff)
            h[idx] += a; diff -= a
            if diff <= 1e-12: break
    else:
        for idx in np.argsort(h)[::-1]:
            a = min(h[idx], -diff)
            h[idx] -= a; diff += a
            if -diff <= 1e-12: break
    return h

        h[rng.choice(np.where(h == 1)[0])] = 0.0
    while h.sum() < t:
        h[rng.choice(np.where(h == 0)[0])] = 1.0
    return h
