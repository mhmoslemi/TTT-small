import numpy as np
import time

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            return h
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf == 0:
            h = np.clip(h + diff / n, 0, 1)
        else:
            h[free] += diff / nf
            h = np.clip(h, 0, 1)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def gradient_smooth_max(h, tau=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    vals = corr * dx
    m = np.max(vals)
    shifted = tau * (vals - m)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    sw = np.sum(exps)
    if sw < 1e-300:
        weights = np.zeros_like(vals)
        weights[np.argmax(vals)] = 1.0
    else:
        weights = exps / sw
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        c = np.zeros(n)
        if lag >= 0:
            c[lag:] = 1.0
            c[lag:] -= h[:n - lag]
            c[:n - lag] -= h[lag:]
        else:
            l = -lag
            c[:n - l] = 1.0
            c[:n - l] -= h[l:]
            c[l:] -= h[:n - l]
        grad += w * dx * c
    return grad

def grad_descent(h, n, t_limit, max_iter, tau_init, lr_init):
    best_h = h.copy()
    best_c5 = eval_c5(best_h)
    h = h.copy()
    lr = lr_init
    tau = tau_init
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        grad = gradient_smooth_max(h, tau)
        h_new = project_h(h - lr * grad, n)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5
        h = h_new
        if it % 80 == 79:
            lr = max(lr * 0.6, 0.0005)
            tau = min(tau * 1.5, 300.0)
    return best_h, best_c5

def coordinate_descent(h, n, t_limit, max_rounds=50):
    best_h = h.copy()
    best_c5 = eval_c5(best_h)
    for rd in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        for i in range(n):
            if time.time() > t_limit:
                return best_h, best_c5
            old = best_h[i]

                trial = best_h.copy()

                if c5 < best_c5 - 1e-12:

                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return best_h, best_c5
