    starts.append(make_sine(n0, 11))

    for sx in starts:
        if rem_time(t0, budget_s) < 80:
            break
        h, c = slsqp_opt(sx, n0, 250, 200.0)
        if rem_time(t0, budget_s) < 40:
            break
        cb = int(min(6000, rem_time(t0, budget_s) * 25))
        h2, c2 = bsearch(h, n0, cb, seed)
        if c2 < bc:
            bh, bc, _ = h2, c2, n0
        elif c < bc:
            bh, bc = h, c

    if rem_time(t0, budget_s) > 60:
        h, c = slsqp_opt(bh, n0, 300, 500.0)
        if c < bc:
            bh, bc = h, c

    if rem_time(t0, budget_s) > 40:
        nit = int(min(12000, rem_time(t0, budget_s) * 30))
        h, c = bsearch(bh, n0, nit, seed + 200)
        if c < bc:
            bh, bc = h, c

    if rem_time(t0, budget_s) > 60:
        h, c = sa_search(bh, n0, int(min(8000, rem_time(t0, budget_s) * 20)), seed + 300)
        if c < bc:
            bh, bc = h, c
        if rem_time(t0, budget_s) > 40:
            h, c = bsearch(bh, n0, int(min(6000, rem_time(t0, budget_s) * 25)), seed + 400)
            if c < bc:
                bh, bc = h, c

    for nt in [50, 100, 140, 36, 80, 90, 60]:
        if rem_time(t0, budget_s) < 60:
            break
        xn = np.linspace(0.0, 2.0, nt)
        xo = np.linspace(0.0, 2.0, len(bh))
        hi = np.interp(xn, xo, bh, left=bh[0], right=bh[-1])
        hi = project(hi, nt)
        h, c = slsqp_opt(hi, nt, 200, 200.0)
        if rem_time(t0, budget_s) < 25:
            break
        nit = int(min(4000, rem_time(t0, budget_s) * 12))
        h2, c2 = bsearch(h, nt, nit, seed + nt)
        if c2 < bc:
            bh, bc = h2, c2
        elif c < bc:
            bh, bc = h, c

    bh = project(bh, len(bh))
    bc = compute_c5(bh, len(bh))
    return bh, bc, len(bh)
