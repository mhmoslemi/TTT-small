import numpy as np
import time

def repair(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    ar = np.arange(n)
    for i in range(len(corr)):
        if w[i] < 1e-10:
            continue
        shift = i - (n - 1)
        idx1 = ar - shift
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = ar + shift
        v2 = (idx2 >= 0) & (idx2 < n)
        t1 = np.zeros(n)
        t1[v1] = 1.0 - h[idx1[v1]]
        t2 = np.zeros(n)
        t2[v2] = h[idx2[v2]]
        grad += w[i] * (t1 - t2)
    return grad * dx

def grad_desc(h, target_sum, n_iter, temp, lr_init, time_limit):

    best = h.copy()
    best_val = eval_c5(best)

    t0 = time.time()

        g = compute_grad(h, temp)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:

            if lr < 1e-6:

import numpy as np
import time

def repair(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    ar = np.arange(n)
    for i in range(len(corr)):
        if w[i] < 1e-10:
            continue
        shift = i - (n - 1)
        idx1 = ar - shift
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = ar + shift
        v2 = (idx2 >= 0) & (idx2 < n)
        t1 = np.zeros(n)
        t1[v1] = 1.0 - h[idx1[v1]]
        t2 = np.zeros(n)
        t2[v2] = h[idx2[v2]]
        grad += w[i] * (t1 - t2)
    return grad * dx

def grad_desc(h, target_sum, n_iter, temp, lr_init, time_limit):
    h = h.copy()
    best = h.copy()
    best_val = eval_c5(best)
    lr = lr_init
    t0 = time.time()
    for i in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        g = compute_grad(h, temp)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        h_new = repair(h - lr * g / g_norm, target_sum)
        val_new = eval_c5(h_new)
        cur_val = eval_c5(h)
        if val_new < cur_val - 1e-14:
            h = h_new
            if val_new < best_val:
                best = h.copy()
                best_val = val_new
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.8
            if lr < 1e-7:
                break
    return best, best_val

def coord_desc(h, target_sum, n_sweeps, time_limit, rng):
    n = len(h)
    h = h.copy()
    best_val = eval_c5(h)
    t0 = time.time()
    deltas = [0.05, 0.02, 0.01, 0.005]
    for _ in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            improved = False
            for d in deltas:
                for sign in [1, -1]:
                    h[j] = old + sign * d
                    h = repair(h, target_sum)
                    val = eval_c5(h)
                    if val < best_val - 1e-14:
                        best_val = val
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                h[j] = old
                h = repair(h, target_sum)
    return h, best_val

def swap_improve(h, target_sum, n_rounds, time_limit, rng):
    n = len(h)
    h = h.copy()
    best_val = eval_c5(h)
    t0 = time.time()
    for _ in range(n_rounds):
        if time.time() - t0 > time_limit:
            break
        i, j = rng.integers(0, n), rng.integers(0, n)
        if i == j:
            continue
        h2 = h.copy()
        h2[i], h2[j] = h[j], h[i]
        h2 = repair(h2, target_sum)
        val = eval_c5(h2)
        if val < best_val - 1e-14:
            h = h2
            best_val = val
    return h, best_val
