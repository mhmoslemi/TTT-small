def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_vals(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_a = np.exp(alpha * (corr - m))
    s = np.sum(exp_a)
    return (m + np.log(s) / alpha) * dx, exp_a / s, dx

def grad_from_weights(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if abs(wk) < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
