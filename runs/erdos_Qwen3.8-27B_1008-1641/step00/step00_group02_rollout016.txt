from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    n = len(h)
    # Distribute the difference
    tol = 1e-12
    if diff > 0:
        mask = h < 1.0 - tol
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    else:
        mask = h > tol
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-10:
        h += diff2 / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_l2(x, n):
    """Minimize L2 norm of correlations (proxy for minimax)."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    return np.sum(corr**2)

def obj_softmax(x, n, alpha=100.0):
    """Soft approximation of max correlation."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    # softmax approximation of max
    exps = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(exps)) / alpha

        h_new[idx[1]] += delta[1]

        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < compute_c5(best):

    return best, compute_c5(best)

        h_best, c5_best = coord_search(h_best, n, target, 30000, rng, step=0.03)

    # Phase 4: Second round of SLSQP + local
    if elapsed() < budget_s * 0.75:
        for alpha in [200.0, 800.0]:
            if elapsed() > budget_s * 0.74:
                break
            hx = try_slsqp(h_best, n, target, bounds, cons,
                           lambda x, a=alpha: obj_softmax(x, n, a), 300)
            if hx is not None:
                c5_try = compute_c5(hx)
                if c5_try < c5_best:
                    h_best, c5_best = hx, c5_try
        if elapsed() < budget_s * 0.78:
            h_best, c5_best = coord_search(h_best, n, target, 15000, rng, step=0.02)

    # Phase 5: Try other resolutions
    for n_try in [45, 50, 56, 60, 64, 72, 80, 90]:
        if elapsed() > budget_s * 0.92:
            break
        if n_try == n:
            continue
        # Build initial guess by interpolation
        x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_init = np.interp(x_new, x_old, np.concatenate([h_best, [h_best[0]]]))
        h_init = repair(h_init, n_try / 2.0)

        bounds_t = [(0.0, 1.0)] * n_try
        cons_t = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_try / 2.0}]

        hx = try_slsqp(h_init, n_try, n_try / 2.0, bounds_t, cons_t,
                       lambda x: obj_l2(x, n_try), 300)
        if hx is not None:
            c5_try = compute_c5(hx)
            if c5_try < c5_best:
                h_best, c5_best, n = hx, c5_try, n_try
                target = n_try / 2.0
                bounds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]

    # Phase 6: Final polish
    if elapsed() < budget_s * 0.97:
        hx = try_slsqp(h_best, n, target, bounds, cons,
                       lambda x: obj_softmax(x, n, 1000.0), 200)
        if hx is not None:
            c5_try = compute_c5(hx)
            if c5_try < c5_best:
                h_best, c5_best = hx, c5_try
        h_best, c5_best = coord_search(h_best, n, target, 10000, rng, step=0.01)

    # Ensure feasibility
    h_best = repair(h_best, n / 2.0)
    c5_best = compute_c5(h_best)

    return (h_best, c5_best, n)
