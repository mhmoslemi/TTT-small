    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.argsort(-h, kind='stable')
        rem = diff
        for i in idx:
            if rem <= 1e-14: break
            a = min(rem, 1.0 - h[i])
            h[i] += a
            rem -= a
    else:
        idx = np.argsort(h, kind='stable')
        rem = -diff
        for i in idx:
            if rem <= 1e-14: break
            s = min(rem, h[i])
            h[i] -= s
            rem -= s
    return h
