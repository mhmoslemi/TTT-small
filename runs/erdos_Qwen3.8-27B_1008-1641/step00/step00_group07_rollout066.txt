def compute_c5(h, dx):

    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = np.sort(h)[::-1]  # sort descending
    h[:n//2] = 1.0
    h[n//2:] = 0.0
    # But this destroys the original pattern!
    # Better: keep original and just threshold
    h_orig = h.copy()
    # Actually for swap search, we just need exactly n/2 ones
    h_bin = (h_orig > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > n//2:
        # Turn off some ones
        ones = np.where(h_bin == 1)[0]
        # Turn off the ones with lowest original value
        for idx in np.argsort(h_orig[ones])[:n_ones - n//2]:
            h_bin[ones[idx]] = 0
    elif n_ones < n//2:

        for idx in np.argsort(h_orig[zeros])[::-1][:n//2 - n_ones]:

import numpy as np
import time

def compute_c5(h, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def to_binary_fixed(h, n):
    h = np.array(h, dtype=float)
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > target:
        ones = np.where(h_bin == 1.0)[0]
        for idx in ones[:n_ones - target]:
            h_bin[idx] = 0.0
    elif n_ones < target:
        zeros = np.where(h_bin == 0.0)[0]
        for idx in zeros[:target - n_ones]:
            h_bin[idx] = 1.0
    return h_bin

def swap_hill_climb(h, dx, time_limit=30.0):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    start = time.time()
    while time.time() - start < time_limit:
        improved = False
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        for i in zeros:

                h[j] = 0.0

                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
                    ones[ones.index(j)] = i
                    zeros[zeros.index(i)] = j
                else:
                    h[i] = 0.0
                    h[j] = 1.0
            if time.time() - start > time_limit:
                return h, best_c5
        if not improved:
            break
    return h, best_c5
