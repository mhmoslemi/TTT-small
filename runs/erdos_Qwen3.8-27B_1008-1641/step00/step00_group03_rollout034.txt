from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(x, alpha):
    mx = np.max(x)
    return float(mx + np.log(np.sum(np.exp(x - mx))) / alpha)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        step = np.clip(d / np.sum(mask), -1.0, 1.0)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, alpha):
    return smooth_max(get_corr(h) * dx, alpha)

def obj_single(h, dx, idx):
    return float(get_corr(h)[idx] * dx)

def con_eq(h, target):
    return float(np.sum(h) - target)

def con_eq_jac(h, target):
    return np.ones(len(h))

def make_sin_h(n, freq, amp, phase):
    x = np.arange(n, dtype=float)
    return 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)

def gradient_active(h, k, n):
    grad = np.zeros(n)
    lo = max(0, -k)
    hi = min(n, n - k)
    for i in range(lo, hi):
        j1 = i
        j2 = i + k
        if 0 <= j1 < n and 0 <= j2 < n:
            grad[j1] += (1.0 - h[j2])
            grad[j2] += (-h[j1])
    return grad

def projected_subgradient_step(h, grad, target, step_size):
    h_new = h - step_size * grad
    return project_feasible(h_new, target)

    for alpha in [20, 60, 150, 400, 1000, 3000]:

        for restart in range(2):

            try:

                    best_c5 = c5

        h_pert = project_feasible(h_pert, target)
        c5_p = compute_c5(h_pert, dx)
        if c5_p < best_c5 * 1.002:
            try:
                res = minimize(obj_smooth, h_pert, args=(dx, 800),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    # Final SLSQP polish
    if time.time() - t0 < budget_s * 0.99:
        try:
            res = minimize(obj_smooth, best_h, args=(dx, 5000),
                           method='SLSQP', bounds=bounds, constraints=cons,

            c5_new = compute_c5(h_new, dx)

                best_c5 = c5_new
        except Exception:
            pass

    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
