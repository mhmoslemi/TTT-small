            if n > ss:
                g[:n - ss] += 1.0 - h[ss:]
                g[ss:] -= h[:n - ss]
    g *= dx / len(act)
    return g, mv

def subgrad_optimize(h, dx, ts, rng, t_end):
    n = len(h)
    best = h.copy()
    bv = compute_c5(h, dx)
    hc = h.copy()
    ss = 0.04
    ni, pat = 0, 400
    avg = hc.copy()
    cnt = 0
    while time.time() < t_end:
        corr = np.correlate(hc, 1.0 - hc, mode='full') * dx
        mv = float(corr.max())
        act = np.where(corr >= mv - 1e-10)[0]
        if len(act) > 5:
            act = act[np.argsort(corr[act])[::-1][:5]]
        g = np.zeros(n)
        for m in act:
            s = int(m) - (n - 1)
            if s >= 0:
                if n > s:
                    g[s:] += 1.0 - hc[:n - s]
                    g[:n - s] -= hc[s:]
            else:
                ss2 = -s
                if n > ss2:
                    g[:n - ss2] += 1.0 - hc[ss2:]
                    g[ss2:] -= hc[:n - ss2]
        g *= dx / len(act)
        hc = project_feasible(hc - ss * g, ts)
        cnt += 1
        avg = (cnt * avg + hc) / (cnt + 1)
        v = compute_c5(hc, dx)
        va = compute_c5(avg, dx)
        if v < bv - 1e-10:
            best, bv, ni = hc.copy(), v, 0
        elif va < bv - 1e-10:
            best, bv, ni = avg.copy(), va, 0
        else:
            ni += 1
        if ni > pat:
            ss *= 0.7
            if ss < 0.0003:
                ss = 0.04
            hc = project_feasible(best + 0.08 * rng.randn(n), ts)
            ni = 0
            pat = max(80, int(pat * 0.7))
    return best, bv

def coord_polish(h, dx, ts, t_end):
    n = len(h)
    best = h.copy()
    bv = compute_c5(h, dx)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.03, -0.03, 0.015, -0.015]
    while time.time() < t_end:
        improved = False
        for i in range(n):
            for d in deltas:
                t2 = best.copy()
                t2[i] += d
                t2 = project_feasible(t2, ts)
                v = compute_c5(t2, dx)
                if v < bv - 1e-12:
                    best, bv = t2, v
                    improved = True
                    break
        if not improved:
            break
    return best, bv

def make_init(n, rng, variant):
    ts = n / 2.0
    if variant == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = project_feasible(h, ts)
    elif variant == 1:
        x = np.linspace(0, 2, n)
        phase = rng.rand() * 2 * np.pi
        h = 0.5 + 0.25 * np.sin(np.pi * x + phase) + 0.1 * np.sin(2 * np.pi * x + phase * 0.7)
        h = project_feasible(h, ts)
    elif variant == 2:
        h = (rng.rand(n) > 0.5).astype(float)
        h = project_feasible(h, ts)
    elif variant == 3:
        h = rng.rand(n)
        k = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
        h = np.convolve(h, k, mode='same')
        h = project_feasible(h, ts)
    elif variant == 4:
        x = np.linspace(0, 2, n)
        h = np.where(x < 1.0, 1.0 - 0.3 * (x - 0.5)**2, -0.3 * (x - 1.5)**2)
        h = project_feasible(h, ts)
    else:
        h = rng.rand(n)
        h = project_feasible(h, ts)
    return h

def resample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    h_interp = np.interp(np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new, x_old, h_old)
    return project_feasible(h_interp, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    bh, bv, bn = None, 1e9, 0
    parent_n = 0

    try:
        p = np.array(initial_h_values, dtype=float)
        parent_n = len(p)
        dxp = 2.0 / parent_n
        p = project_feasible(p, parent_n / 2.0)
        bh, bv, bn = p, compute_c5(p, dxp), parent_n
    except:
        pass

    if bh is None:
        bn = 45
        bh = make_init(bn, rng, 3)
        bv = compute_c5(bh, 2.0 / bn)

    deadline = t0 + budget_s - 5.0

    # Build list of (n, initializations) to try
    n_candidates = []
    n_candidates.append(bn)
    for cand in [bn - 5, bn + 5, 60, 80, 100]:
        if 30 <= cand <= 200 and cand not in n_candidates:
            n_candidates.append(cand)
    n_candidates = sorted(n_candidates)

    time_per_n = (deadline - t0 - budget_s * 0.30) / max(1, len(n_candidates))
    time_per_n = max(10, min(120, time_per_n))

    for idx, nn in enumerate(n_candidates):
        t_n_start = max(time.time(), t0 + idx * time_per_n)
        t_n_end = t_n_start + time_per_n
        if t_n_end > deadline - budget_s * 0.30 - 5:
            t_n_end = deadline - budget_s * 0.30 - 5
        if time.time() > t_n_end:
            break
        dxn = 2.0 / nn
        tsn = nn / 2.0
        n_inits = max(2, int(time_per_n / 20))
        time_per_init = (t_n_end - t_n_start) / max(1, n_inits)

        for iv in range(n_inits):
            t_i_start = t_n_start + iv * time_per_init
            t_i_end = t_i_start + min(time_per_init, 40)
            if time.time() > t_i_end:
                continue
            if iv == 0 and nn == bn and parent_n == nn:
                h_i = bh.copy()
            elif iv == 1 and parent_n > 0 and nn != parent_n:
                h_i = resample(np.array(initial_h_values, dtype=float), nn)
            else:
                h_i = make_init(nn, rng, iv % 5)
            h_o, v_o = subgrad_optimize(h_i, dxn, tsn, rng, t_i_end)
            if v_o < bv:
                bh, bv, bn = h_o.copy(), v_o, nn

    # Coordinate polish phase
    t_coord_end = deadline
    if time.time() < t_coord_end:
        dx_b = 2.0 / bn
        bh, bv = coord_polish(bh, dx_b, bn / 2.0, t_coord_end)

    # Final projection and evaluation
    bh = project_feasible(bh, bn / 2.0)
    bv = compute_c5(bh, 2.0 / bn)
    return bh, bv, bn
