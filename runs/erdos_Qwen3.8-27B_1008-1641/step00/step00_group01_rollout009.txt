    dx = 2.0 / len(h)
    return float(np.max(corr) * dx)

    """Project h to satisfy sum(h)=n/2, 0<=h<=1."""
    h = np.clip(h.copy(), 0, 1)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, 1.0)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min((-diff) / cnt, 1.0)
            h[mask] = np.maximum(h[mask] - step, 0.0)
    return h


def smooth_max_obj(h, n, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    vals = np.clip((corr - m) / temp, -60.0, 0.0)
    expv = np.exp(vals)
    return m + temp * np.log(np.sum(expv))


def do_optimize(h_init, n, dx, temp, maxiter=800):
    def obj(h):
        return smooth_max_obj(h, n, dx, temp)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h_init, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
    return project_to_constraint(res.x, n)

    best = h.copy()
    bc = compute_c5(best)
    for _ in range(iters):
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
