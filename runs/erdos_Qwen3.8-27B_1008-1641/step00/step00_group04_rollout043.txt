    step = 0.05
    noise = 0.15
    for it in range(200000):
        if time.time() - t0 > time_limit:
            break
        tau = max(0.001, 0.1 * np.exp(-it / 5000.0))
        g = c5_grad(h, tau)
        pert = -step * g + noise * rng.randn(n)
        h_new = project(h + pert, n)
        f_new = c5_value(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.02, 0.15)
        else:
            step *= 0.995
            noise *= 0.999
        if it % 2000 == 0:
            noise = max(noise, 0.02)
    return h, f

def cma_es_refine(x0, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    lam = max(4 + int(3 * np.log(n)), 12)
    mu = lam // 2
    wi = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    wi /= wi.sum()
    mueff = 1.0 / np.sum(wi ** 2)
    x = x0.astype(float).copy()
    sigma = 0.05
    best_x = x.copy()
    best_f = c5_value(project(x, n))
    ps = np.zeros(n)
    pc = np.zeros(n)
    cs = (mueff + 2.0) / (n + mueff + 5.0)
    cc = (4.0 + mueff / n) / (n + 4.0 + mueff / n)
    ds = 1 + 2 * max(0, np.sqrt((mueff - 1) / (n + 1)) - 1) + cs
    chi_n = np.sqrt(n) * (1 - 1.0 / (4 * n) + 1.0 / (21 * n ** 2))
    B = np.eye(n)
    dist = np.ones(n)
    while time.time() - t0 < time_limit:
        y = rng.randn(lam, n)
        x_pop = x[None, :] + sigma * ((y @ np.eye(n)) * dist[None, :]) @ B
        f_pop = np.array([c5_value(project(xi, n)) for xi in x_pop])
        order = np.argsort(f_pop)
        if f_pop[order[0]] < best_f:
            best_f = f_pop[order[0]]
            best_x = x_pop[order[0]].copy()
        m = np.sum(wi[:, None] * x_pop[order[:mu]], axis=0)
        d = m - x
        ps = (1 - cs) * ps + np.sqrt(cs * (2 - cs) * mueff) * d / (sigma + 1e-15)
        nps = np.linalg.norm(ps)
        sigma *= np.exp(cs / ds * (nps / chi_n - 1))
        sigma = min(max(sigma, 1e-10), 0.5)
        hsig = 1.0 if nps / chi_n < 1.4 + 2.0 / (n + 1) else 0.0
        pc = (1 - cc) * pc + cc * hsig * d / (sigma + 1e-15)
        rank_mu = np.zeros((n, n))
        for i in range(mu):
            vi = (x_pop[order[i]] - x) / (sigma + 1e-15)
            rank_mu += wi[i] * np.outer(vi, vi)
        c1 = 2.0 / (n + 2.0) ** 2 + ns
        c1 = 2.0 / (n + 2.0) ** 2 + lam * np.sum(wi ** 2) / (n + 8.0)
        cmu = 1.0 - (1.0 + cs * (2 + cc) - 1.0)
        cmu = 1.0 - c1
        C = (1 - c1 - cmu) * np.eye(n)  # reset to identity for stability
        C = (1 - c1 - cmu) * (B @ (dist ** 2 * (B.T @ np.eye(n) @ B)))  # approx
        # Simpler: just track via dist and B
        # Update dist via C update
        C_old = B @ np.diag(dist ** 2) @ B.T
        C_new = (1 - c1 - cmu) * C_old + c1 * (1 + (1 - hsig) * cc * (2 - cc) - c1) * np.outer(pc, pc) + cmu * rank_mu
        C_new = (C_new + C_new.T) * 0.5
        ev, evec = np.linalg.eigh(C_new)
        ev = np.clip(ev, 1e-12, None)
        ratio = ev.max() / ev.min()
        if ratio > 1e10:
            ev = np.clip(ev, ev.max() * 1e-10, None)
        B, dist = evec, np.sqrt(ev)
        x = m
    return best_x, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = None
    if h_init is None:
        n = 42
        h_init = project(rng.randn(n) * 0.3, n)
    n = len(h_init)
    h_init = project(h_init, n)

    best_h = h_init.copy()
    best_f = c5_value(h_init)

    # Phase 1: Local optimization from initial (50% budget)
    t1 = budget_s * 0.45
    h1, f1 = local_opt(h_init, n, t1, seed)
    if f1 < best_f:
        best_h, best_f = h1, f1

    # Phase 2: CMA-ES refinement (25% budget)
    remaining = budget_s - (time.time() - t_start) - budget_s * 0.15
    if remaining > 30:
        hc, fc = cma_es_refine(best_h, n, remaining, seed + 1)
        hc_p = project(hc, n)
        fc_p = c5_value(hc_p)
        if fc_p < best_f:
            best_h, best_f = hc_p, fc_p
        # Local refine from CMA result
        remaining2 = budget_s - (time.time() - t_start) - 30
        if remaining2 > 20:
            h2, f2 = local_opt(project(hc, n), n, remaining2, seed + 2)
            if f2 < best_f:
                best_h, best_f = h2, f2

    # Phase 3: More local refinement on best (remaining time)
    remaining3 = budget_s - (time.time() - t_start) - 5
    if remaining3 > 15:
        h3, f3 = local_opt(best_h, n, remaining3, seed + 3)
        if f3 < best_f:
            best_h, best_f = h3, f3

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_f = c5_value(best_h)
    if final_f > best_f:
        final_f = best_f

    try:
        evaluated = evaluate_erdos_solution(best_h, n)
        if evaluated < final_f:
            final_f = evaluated
    except Exception:
        pass

    return best_h, final_f, n
