import numpy as np
import scipy.optimize as so
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def soft_max(vals, alpha):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha


def repair_sum(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def objective_fn(hv, n, alpha):
    dx = 2.0 / n
    hc = np.clip(hv, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode="full") * dx
    return soft_max(corr, alpha)


def slsqp_optimize(h0, n, alpha, maxiter=600):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        return objective_fn(hv, n, alpha)

    def eq_fun(hv):
        return np.sum(hv) - target

    def eq_jac(hv):
        return np.ones(n)

    constraints = ({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},)
    try:
        res = so.minimize(obj, h0, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h0.copy()

    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_dst = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(x_dst, x_src, h_src)
