import numpy as np
from scipy.optimize import minimize
import time

_g_n = 47
_g_target = 23.5

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if s > 1e-10:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5_core(params, n, dx, alpha):
    h = np.clip(params, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if s > 1e-10:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    c_shifted = corr - cmax
    return (cmax + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * c_shifted)))) * dx

def obj_func(x, n, dx, alpha):
    return smooth_c5_core(x, n, dx, alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)
