```python
from scipy.optimize import minimize
import time

_CN = [70]

def _cfun(z):
    return np.sum(z) - _CN[0] / 2.0

def _cjac(z):
    return np.ones(len(z))

def _c5(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

def _proj(h, n):
    h = np.array(h, dtype=float).copy()
    tgt = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0, 1)
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h < 1) if d > 0 else (h > 0)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def _smc(h, n, t):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    return (m + t * np.log(np.exp((c - m) / t).sum())) * 2.0 / n

def _ob_lb(z, n, t, p):
    h = np.clip(z, 0, 1)
    return _smc(h, n, t) + p * (h.sum() - n / 2) ** 2

def _ob_sq(z, n, t):
    h = np.clip(z, 0, 1)
    return _smc(h, n, t)

def _ob_nm(z, n, t, p):
    h = np.clip(z, 0, 1)
    return _smc(h, n, t) + p * (h.sum() - n / 2) ** 2

def _rem(t0, bud):
    return max(0.0, bud - 5 - (time.time() - t0))

    bc = _c5(bh, n)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]

        hs = bh + rng.normal(0, 0.15 * min(i+1, 5), n)

        res = minimize(_ob_lb, hs, args=(n, 0.06, 100.0), method='L-BFGS-B',
                       bounds=bd, options={'maxiter': 2000, 'ftol': 1e-14})
        hn = _proj(res.x, n)
        cn = _c5(hn, n)
        if cn < bc:

        hn = _proj(res.x, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_CN = [70]

def _cfun(z):
    return np.sum(z) - _CN[0] / 2.0

def _cjac(z):
    return np.ones(len(z))

def _c5(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

def _proj(h, n):
    h = np.array(h, dtype=float).copy()
    tgt = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0, 1)
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h < 1) if d > 0 else (h > 0)
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
    return h

def _smc(h, n, t):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    return (m + t * np.log(np.exp((c - m) / t).sum())) * 2.0 / n

def _ob_lb(z, n, t, p):
    h = np.clip(z, 0, 1)
    return _smc(h, n, t) + p * (h.sum() - n / 2) ** 2

def _ob_sq(z, n, t):
    h = np.clip(z, 0, 1)
    return _smc(h, n, t)

def _ob_nm(z, n, t, p):
    h = np.clip(z, 0, 1)
    return _smc(h, n, t) + p * (h.sum() - n / 2) ** 2

def _rem(t0, bud):
    return max(0.0, bud - 5 - (time.time() - t0))

def run(seed=42, budget_s=1100, **kwargs):
    global _CN
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    _CN[0] = n
    bh = _proj(h0, n)
    bc = _c5(bh, n)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]

    # Phase 1: L-BFGS-B with penalty, many random restarts
    for i in range(50):
        if _rem(t0, budget_s) < 30:
            break
        sc = 0.15 * min(i + 1, 6)
        hs = _proj(bh + rng.normal(0, sc, n), n)
        res = minimize(_ob_lb, hs, args=(n, 0.06, 100.0), method='L-BFGS-B',
                       bounds=bd, options={'maxiter': 2000, 'ftol': 1e-14})
        hn = _proj(res.x, n)
        cn = _c5(hn, n)
        if cn < bc:
            bh, bc = hn, cn

    # Phase 2: SLSQP with exact equality constraint
    for i in range(20):
        if _rem(t0, budget_s) < 30:
            break
        hs = _proj(bh + rng.normal(0, 0.12, n), n)
        res = minimize(_ob_sq, hs, args=(n, 0.035), method='SLSQP',
                       bounds=bd, constraints=cs,
                       options={'maxiter': 600, 'ftol': 1e-12})
        hn = _proj(res.x, n)
        cn = _c5(hn, n)
        if cn < bc:
            bh, bc = hn, cn

    # Phase 3: Nelder-Mead (derivative-free refinement)
    for i in range(12):
        if _rem(t0, budget_s) < 25:
            break
        hs = _proj(bh + rng.normal(0, 0.05, n), n) if i else bh.copy()
        res = minimize(_ob_nm, hs, args=(n, 0.025, 500.0), method='Nelder-Mead',
                       options={'maxiter': 6000, 'xatol': 1e-8, 'fatol': 1e-10})
        hn = _proj(res.x, n)
        cn = _c5(hn, n)
        if cn < bc:
            bh, bc = hn, cn

    # Phase 4: Coordinate descent with pairwise swaps
    for ss in (0.05, 0.02, 0.01, 0.005, 0.002, 0.001):
        if _rem(t0, budget_s) < 10:
            break
        for it in range(120):
            if _rem(t0, budget_s) < 10:
                break
            ok = False
            for j in range(n):
                if _rem(t0, budget_s) < 10:
                    break
                for k in range(j + 1, n):
                    for sg in (1, -1):
                        vj = bh[j] + sg * ss
                        vk = bh[k] - sg * ss
                        if 0 <= vj <= 1 and 0 <= vk <= 1:

                            vt[k] = vk
                            ct = _c5(vt, n)

                                ok = True
                                break
                    if ok:
                        break
            if not ok:
                break

    bh = _proj(bh, n)
    bc = _c5(bh, n)
    return bh, float(bc), n
```
