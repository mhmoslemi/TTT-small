import numpy as np
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_sum(h, target):
    h = h.copy().astype(float)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(20):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * min(diff / tot, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= h * min(-diff / tot, 1.0)
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def grad_lag(h, lag, N):
    g = np.zeros(N)
    if lag >= 0:
        g[lag:N] += (1.0 - h[:N - lag])
        g[:N - lag] -= h[lag:]
    else:
        la = -lag
        g[:N - la] += (1.0 - h[la:])
        g[la:N] -= h[:N - la]
    return g
