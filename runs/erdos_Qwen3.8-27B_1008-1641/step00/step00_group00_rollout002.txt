import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * corr.max()


def obj_and_grad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = corr.max()
    thresh = val - 1e-10
    active = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)
        j = np.arange(n)
        jm = j - m
        valid_m = (jm >= 0) & (jm < n)
        jp = j + m
        valid_p = (jp >= 0) & (jp < n)
        g = np.zeros(n)
        if np.any(valid_m):
            g[j[valid_m]] += (1.0 - h[jm[valid_m]])
        if np.any(valid_p):
            g[j[valid_p]] -= h[jp[valid_p]]
        grad += g
    if len(active) > 1:
        grad /= len(active)
    return val, grad


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def pgd_optimize(h_init, n, dx, max_iter=20000):

            best_c5 = c5_new
            best_h = h_new.copy()

def slsqp_optimize(h_init, n, dx, maxiter=200):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: compute_c5(x, dx), h_init,
                       method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.fun < compute_c5(h_init, dx):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h_init.copy()
