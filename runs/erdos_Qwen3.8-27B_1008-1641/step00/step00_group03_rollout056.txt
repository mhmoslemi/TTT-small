    return grad

import numpy as np
import time
from scipy.optimize import minimize, Bounds

_N = 70
_DX = 2.0 / 70

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_func(h):
    return compute_c5(h, _DX)

def eq_fun(h):
    return float(np.sum(h) - _N / 2.0)

def eq_jac(h):
    return np.ones(_N)

def project_h(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def run_slsqp(h_start, n, dx, maxiter, ftol):
    global _N, _DX
    _N = n
    _DX = dx
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_func, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-6})
        if res.x is not None and np.all(np.isfinite(res.x)):
            hx = project_h(res.x, n)
            return hx, compute_c5(hx, dx)
    except Exception:
        pass
    return h_start.copy(), compute_c5(h_start, dx)

def corr_grad(h, n, k):
    grad = np.zeros(n)
    if k >= 0:
        grad[:n - k] += (1.0 - h[k:])
        grad[k:] -= h[:n - k]
    else:
        ka = -k
        grad[ka:] += (1.0 - h[:n - ka])
        grad[:n - ka] -= h[ka:]
    return grad

def subgradient_descent(h, n, dx, max_iter, t0=None, budget_s=1100, reserve=10.0):
    h = h.copy()
    cur = compute_c5(h, dx)
    step = 0.04
    stall = 0
    for it in range(max_iter):
        if t0 is not None and time.time() - t0 > budget_s - reserve:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mi = int(np.argmax(corr))
        k = mi - (n - 1)
        g = corr_grad(h, n, k)
        h_new = project_h(h - step * g, n)
        c_new = compute_c5(h_new, dx)
        if c_new <= cur + 1e-13:
            h = h_new
            cur = c_new
            stall = 0
            step = min(step * 1.03, 0.12)
        else:
            stall += 1
            step *= 0.75
            if stall > 30:
                break
    return h, cur

def generate_perturbation(rng, best_h, h0, n, phase):

    if r < 0.35:
        return best_h + rng.normal(0, 0.02 + 0.04 * rng.random(), n)
    elif r < 0.6:
        return best_h + rng.normal(0, 0.08, n)
    elif r < 0.8:
        return h0 + rng.normal(0, 0.05, n)
    elif r < 0.9:

        k = rng.randint(5, n // 2 + 1)
        idx = rng.choice(n, size=k, replace=False)
        hp[idx] += rng.normal(0, 0.12, size=k)
        return hp
    else:
        period = rng.randint(3, n // 3 + 1)
        phase_r = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        amp = rng.uniform(0.1, 0.3)
        return best_h + amp * np.sin(2.0 * np.pi * x / period + phase_r)
