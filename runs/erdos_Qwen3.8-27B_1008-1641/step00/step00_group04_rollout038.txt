def project_to_feasible(x, n):
    target = n / 2
    # Bisection on lambda
    lo, hi = -1, 1
    for _ in range(50):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = target - np.sum(h)
    # Distribute remaining diff among entries not at bounds
    non_fixed = (h > 0) & (h < 1)
    if np.any(non_fixed):
        h[non_fixed] += diff / np.sum(non_fixed)
        h = np.clip(h, 0, 1)
    return h

def smooth_c5_grad(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / np.sum(exp_scaled)  # shape (2n-1,)
    
    grad = np.zeros(n)
    
    for j in range(n):
        g = 0.0
        # k >= 0 contributions (p = n-1+k, k = 0..n-1)
        for k in range(0, n):
            p = n - 1 + k
            # First: I(k <= n-1-j) * (1-h[j+k])
            if k <= n - 1 - j:
                g += w[p] * (1.0 - h[j + k])
            # Second: -I(k <= j) * h[j-k]
            if k <= j:
                g -= w[p] * h[j - k]
        
        # k < 0 contributions (p = n-1-k2, k2 = 1..n-1)
        for k2 in range(1, n):
            p = n - 1 - k2
            # First: I(k2 <= j) * (1-h[j-k2])
            if k2 <= j:
                g += w[p] * (1.0 - h[j - k2])
            # Second: -I(k2 <= n-1-j) * h[j+k2]
            if k2 <= n - 1 - j:
                g -= w[p] * h[j + k2]
        
        grad[j] = dx * g
    
    return grad

    return smooth_grad_fn(x, _T_box[0])

def _eq_con(x):
    return x.sum() - len(x) * 0.5

def _eq_con_jac(x):
    return np.ones(len(x))

def project(x, n):
    target = n * 0.5
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    free = (h > 1e-12) & (h < 1.0 - 1e-12)
    nf = int(free.sum())
    if nf > 0:
        h[free] += diff / nf
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_optimize(h, n, T, max_iter=300):
    _T_box[0] = T
    try:
        res = minimize(
            _obj, h, jac=_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_con, 'jac': _eq_con_jac}],
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )
        return project(res.x, n)
    except Exception:
        return h

def local_search(h, n, rng, n_iter):
    best = h.copy()
    best_f = compute_c5(best)
    step = 0.05
    for i in range(n_iter):
        idx = rng.randint(0, n)
        delta = rng.uniform(-step, step)
        h_try = best.copy()
        h_try[idx] += delta
        h_try = project(h_try, n)
        f_try = compute_c5(h_try)
        if f_try < best_f:
            best = h_try
            best_f = f_try
        if i % 100 == 0:
            step *= 0.9
            step = max(step, 0.005)
    return best, best_f

grad[j] = dx * sum_p w[p] * (∂c[p]/∂h[j])
