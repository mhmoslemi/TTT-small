def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Add/subtract uniformly
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # Fine tune
    # If still off, adjust individual elements
    diff = target - np.sum(h)
    if diff > 0:
        # Need to increase some elements
        indices = np.argsort(-h)  # elements closest to 1 first won't work
        # Better: elements that have room to increase
        room = 1 - h
        order = np.argsort(-room)  # most room first
        for i in order:
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            if abs(diff) < 1e-10:
                break
    elif diff < 0:
        need = -diff
        room = h  # elements that have room to decrease
        order = np.argsort(-room)  # most to decrease first
        for i in order:
            sub = min(need, room[i])
            h[i] -= sub
            need -= sub
            if abs(need) < 1e-10:
                break
    return h

def local_search(h, n, dx, max_iters=5000, temp=0.1):

    best = current.copy()
    best_c5 = current_c5
