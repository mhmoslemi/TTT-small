def proj_to_sum(v, ts):
    v = np.clip(v.copy(), 0.0, 1.0)
    for _ in range(5000):
        d = ts - v.sum()
        if abs(d) < 1e-14:
            return v
        if d > 0:
            m = v < 1.0 - 1e-15
            c = int(np.count_nonzero(m))
            if c == 0:
                return v
            v[m] = np.minimum(v[m] + d / c, 1.0)
        else:
            m = v > 1e-15
            c = int(np.count_nonzero(m))
            if c == 0:
                return v
            v[m] = np.maximum(v[m] + d / c, 0.0)

    cr = np.correlate(h, 1.0 - h, "full")
    return float(np.max(cr) * 2.0 / n)
