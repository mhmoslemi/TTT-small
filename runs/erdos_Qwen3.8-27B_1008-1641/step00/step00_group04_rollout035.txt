def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)

            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not free.any():

import numpy as np
from scipy.optimize import minimize
import time

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not free.any():
            free = np.ones(n, dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def cma_es(mean, sigma, n, tlimit, rng):
    mu = max(4, (n + 3) // 5)
    lam = 2 * mu
    w = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    w /= w.sum()
    mueff = 1.0 / (w @ w)
    cc = (4.0 + mueff / n) / (n + 4.0 + 2.0 * mueff / n)
    cs = (mueff + 2.0) / (n + mueff + 5.0)
    damps = 1.0 + 2.0 * max(0.0, np.sqrt((mueff - 1) / (n + 1) - 1) - 1) + cs
    c1 = (2.0 / ((n + 1.3) ** 2 + mueff)) * (1 - cc) + cs * cc
    cmu = min(1 - c1, 2.0 * (mueff - 2 + 1.0 / mueff) / ((n + 2) ** 2 + mueff))
    C = np.eye(n)
    pc = np.zeros(n)
    ps = np.zeros(n)
    chi_n = np.sqrt(n) * (1 + 1.0 / (4 * n) + 1.0 / (21 * n * n))
    best_h = project_h(mean, n).copy()
    best_val = c5_eval(best_h)
    t_start = time.time()

    while time.time() - t_start < tlimit:
        eigs = np.linalg.eigvalsh(C)
        if eigs[-1] / max(eigs[0], 1e-30) > 1e10:
            C = np.eye(n); sigma *= 0.5; pc[:] = 0; ps[:] = 0
        pops = []
        try:
            L = np.linalg.cholesky(C + 1e-12 * np.eye(n))
        except np.linalg.LinAlgError:
            C = np.eye(n); L = np.eye(n)
        for _ in range(lam):
            z = rng.randn(n)
            child = project_h(mean + sigma * (L @ z), n)
            pops.append(child)
        fits = np.array([c5_eval(p) for p in pops])
        idx = np.argsort(fits)
        if fits[idx[0]] < best_val:
            best_val = fits[idx[0]]
            best_h = pops[idx[0]].copy()
        new_mean = np.zeros(n)
        for i in range(mu):
            new_mean += w[i] * pops[idx[i]]
        new_mean = project_h(new_mean, n)
        diff = (new_mean - mean) / max(sigma, 1e-30)
        pc = (1 - cc) * pc + cc * np.sqrt(mueff) * diff
        ps = (1 - cs) * ps + np.sqrt(cs * (2 - cs) * mueff) * diff
        hsig = np.linalg.norm(ps) / chi_n > 1.4 + cc
        if not hsig:
            C = (1 - c1 - cmu) * C + c1 * np.outer(pc, pc)
            for i in range(mu):
                d = (pops[idx[i]] - mean) / max(sigma, 1e-30)
                C += cmu * w[i] * np.outer(d, d)
        mean = new_mean
        sigma = sigma * np.exp(cs / damps * (np.linalg.norm(ps) / chi_n - 1))
        sigma = np.clip(sigma, 1e-12, 1.0)
    return best_h, best_val

def local_refine(h, n, tlimit, rng):
    best = h.copy()
    best_val = c5_eval(best)
    t_start = time.time()
    lr = 0.03
    while time.time() - t_start < tlimit and lr > 1e-5:
        no_improve = 0
        while no_improve < 200 and time.time() - t_start < tlimit:
            for _ in range(10):
                ii = rng.randint(n)
                dj = rng.uniform(-lr, lr)
                cand = best.copy()
                cand[ii] += dj
                cand = project_h(cand, n)
                v = c5_eval(cand)
                if v < best_val:
                    best = cand; best_val = v; no_improve = 0; break
            else:
                no_improve += 1
        lr *= 0.75
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(42, 0.5)
    n0 = len(h0)
    h_best = project_h(h0.copy(), n0)
    c5_best = c5_eval(h_best)
    rem = lambda: budget_s - (time.time() - t0)

    # Phase 1: CMA-ES at original resolution
    if rem() > 50:
        h1, v1 = cma_es(h_best, 0.15, n0, rem() - 40, rng)
        if v1 < c5_best:
            h_best, c5_best = h1, v1

    # Phase 2: Local refinement
    if rem() > 25:
        h2, v2 = local_refine(h_best, n0, rem() - 20, rng)
        if v2 < c5_best:
            h_best, c5_best = h2, v2

    # Phase 3: Multi-restart CMA-ES with perturbation
    if rem() > 80:
        for restart in range(5):
            if rem() < 80:
                break
            perturb = h_best + 0.1 * rng.randn(n0) * (h_best > 0.05) * (h_best < 0.95)
            h3, v3 = cma_es(project_h(perturb, n0), 0.1, n0, rem() - 60, rng)
            if v3 < c5_best:
                h_best, c5_best = h3, v3

    # Phase 4: Try higher resolution via interpolation + optimization
    for n_try in [84, 168]:
        if rem() < 100:
            break
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = project_h(np.interp(x_new, x_old, h_best), n_try)
        v_interp = c5_eval(h_interp)
        h4, v4 = cma_es(h_interp, 0.06, n_try, rem() - 60, rng)

        elif v_interp < c5_best:

        if v5 < c5_best:

    c5_final = c5_eval(h_best)
    return h_best, c5_final, len(h_best)
