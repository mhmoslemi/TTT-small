        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:

            hi = mid

def eval_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_b = np.exp(beta * (corr - max_c))
    p = exp_b / np.sum(exp_b)
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)

import numpy as np
import time
from scipy.optimize import minimize as scipy_min

def project_feasible(h, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_b = np.exp(beta * (corr - max_c))
    p = exp_b / (np.sum(exp_b) + 1e-30)
    grad = np.zeros(n)
    lags = np.arange(-(n - 1), n)
    for k in range(len(corr)):
        w = p[k]
        if w < 1e-12:
            continue
        lag = lags[k]
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            j_lo1 = max(0, lag)
            j_hi1 = min(n, n + lag)
            if j_lo1 < j_hi1:
                grad[j_lo1:j_hi1] += w * (1.0 - h[j_lo1 - lag:j_hi1 - lag])
            j_lo2 = max(0, -lag)
            j_hi2 = min(n, n - lag)
            if j_lo2 < j_hi2:
                grad[j_lo2:j_hi2] -= w * h[j_lo2 + lag:j_hi2 + lag]
    return grad, corr

def gradient_descent(h, n, beta, lr, n_iter, t0, deadline):
    target = n / 2.0
    best = h.copy()
    best_c5 = eval_c5(h)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        grad, corr = compute_gradient(h, beta)
        c5_curr = float(np.max(corr) * (2.0 / n))
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best = h.copy()
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        if np.max(np.abs(h_new - h)) < 1e-10:
            break
        h = h_new
    return best, best_c5

def polish_scipy(h, n, t0, deadline):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        max_c = np.max(corr)
        shifted = corr - max_c
        return (max_c + np.log(np.sum(np.exp(100.0 * shifted)))) * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    bounds = [(0.0, 1.0)] * n
    res = scipy_min(obj, h.copy(), method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': 200, 'ftol': 1e-10})
    if res.success and res.fun < eval_c5(h):
        return project_feasible(res.x, target)
    return h.copy()
