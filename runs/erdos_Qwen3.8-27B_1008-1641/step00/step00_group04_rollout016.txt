    if len(active) > 1:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx, corr

def project_feasible(h_raw, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h_raw - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h_raw - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-12:
        free = (h_p > 1e-14) & (h_p < 1.0 - 1e-14)
        if free.any():
            h_p[free] += diff / free.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def smooth_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = t * corr * dx
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    grad = np.zeros(n)
    idx = np.arange(n)
    active = np.where(weights > 1e-8)[0]
    for m in active:
        k = m - (n - 1)
        i1 = idx - k
        v1 = (i1 >= 0) & (i1 < n)
        grad[v1] += weights[m] * (1.0 - h[i1[v1]]) * dx
        i2 = idx + k
        v2 = (i2 >= 0) & (i2 < n)
        grad[v2] -= weights[m] * h[i2[v2]] * dx
    return grad

def optimize_sgd(h, n, t, n_steps, momentum, start_time, budget):

    best_c5 = compute_c5(h)[0]
    alpha = 0.02
    vel = np.zeros(n)

        grad = smooth_grad(h, n, t)

            best_c5 = c5
            best_local = h_new.copy()
            h = h_new
        else:
            alpha *= 0.999
            vel *= 0.5
    return best_local, best_c5
