            h[mask] -= delta
        else:
            mask = h < 1.0 - 1e-10
            if not mask.any():
                break
            delta = min((-diff) / mask.sum(), (1.0 - h[mask]).min() * 0.9)
            if delta < 1e-15:
                break
            h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.any():
            h[mask] -= diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(vals, alpha):
    scaled = alpha * vals
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / alpha


def obj_func(h, dx, alpha, target_sum, penalty):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    obj = smooth_max(corr, alpha)
    sv = h.sum() - target_sum
    return obj + penalty * sv * sv


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(80, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = project_feasible(h0, target_sum)
    best_c5 = compute_c5(best_h, dx)

    # Verify with evaluator if available
    if 'evaluate_erdos_solution' in globals():
        try:
            ok, val = evaluate_erdos_solution(best_h)
            if ok:
                best_c5 = val
        except Exception:
            pass

    best_c5_reported = best_c5
    total_rounds = 24

    for rnd in range(total_rounds):
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.82:
            break

        # Increasing alpha for sharper max approximation
        alpha = 20 * (2 ** (rnd // 4))
        alpha = min(alpha, 400)
        penalty = 200.0 * (10 ** (rnd // 5))
        penalty = min(penalty, 50000.0)

        start_h = best_h.copy()

        try:
            res = minimize(
                obj_func, start_h, args=(dx, alpha, target_sum, penalty),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-9, 'maxls': 50}
            )
            candidate = project_feasible(res.x, target_sum)
            c5 = compute_c5(candidate, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate
                best_c5_reported = c5
        except Exception:
            pass

        # Mutation + re-optimize every 3 rounds
        if rnd % 3 == 2 and rnd >= 2:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.82:
                break
            for mut_i in range(4):
                elapsed = time.time() - t0
                if elapsed > budget_s * 0.82:
                    break
                scale = 0.04 + 0.02 * mut_i
                mu = best_h + rng.normal(0, scale, n)
                mu = project_feasible(mu, target_sum)
                try:
                    res2 = minimize(
                        obj_func, mu, args=(dx, alpha, target_sum, penalty),
                        method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': 250, 'ftol': 1e-13}
                    )
                    cand2 = project_feasible(res2.x, target_sum)
                    c5_2 = compute_c5(cand2, dx)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = cand2
                        best_c5_reported = c5_2
                except Exception:
                    pass

        # Also try a "flip" mutation: swap roles of 0-ish and 1-ish coords
        if rnd % 5 == 4:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.82:
                break
            flip_h = 1.0 - best_h
            flip_h = project_feasible(flip_h, target_sum)
            try:
                res3 = minimize(
                    obj_func, flip_h, args=(dx, alpha, target_sum, penalty),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 200, 'ftol': 1e-13}
                )
                cand3 = project_feasible(res3.x, target_sum)
                c5_3 = compute_c5(cand3, dx)
                if c5_3 < best_c5:
                    best_c5 = c5_3
                    best_h = cand3
                    best_c5_reported = c5_3
            except Exception:
                pass

    # Try higher resolutions
    for n_tr in [100, 120, 160, 200]:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.93:
            break
        dx_t = 2.0 / n_tr
        ts_t = n_tr / 2.0
        bounds_t = [(0.0, 1.0)] * n_tr

        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_t)

        for alp in [60, 150]:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.93:
                break
            try:
                res_t = minimize(
                    obj_func, h_interp, args=(dx_t, alp, ts_t, 5000.0),
                    method='L-BFGS-B', bounds=bounds_t,
                    options={'maxiter': 350, 'ftol': 1e-14}
                )
                cand_t = project_feasible(res_t.x, ts_t)
                c5_t = compute_c5(cand_t, dx_t)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = cand_t
                    n = n_tr
                    dx = dx_t
                    target_sum = ts_t
                    bounds = bounds_t
                    best_c5_reported = c5_t
                    h_interp = cand_t
            except Exception:
                pass

    # Final verification
    if 'evaluate_erdos_solution' in globals():
        try:
            ok, val = evaluate_erdos_solution(best_h)
            if ok:
                best_c5_reported = val
        except Exception:
            pass

    return (best_h, best_c5_reported, n)
