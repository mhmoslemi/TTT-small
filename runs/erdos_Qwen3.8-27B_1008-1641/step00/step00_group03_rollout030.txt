        else:
            h = 0.5 * h + 0.5 * h_new
            h = _project(h, target)
            stall += 1
            if stall > 2000:
                lr *= 0.9
                stall = 0

    # Phase 2: 0-1 Simulated Annealing
    if not dl():
        h_bin = _fix_sum_bin(best_h, target)
        cur_c5 = _eval(h_bin, dx)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()

        T = 0.03
        sa_it = 0
        while not dl() and sa_it < 800000:
            sa_it += 1
            i = np.random.randint(n)
            j = np.random.randint(n)
            if h_bin[i] == h_bin[j]:
                continue
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            new_c5 = _eval(h_bin, dx)
            delta = new_c5 - cur_c5
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_bin.copy()
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            if sa_it % 4000 == 0:
                T *= 0.993

    # Phase 3: 0-1 coordinate descent (singleton flips)
    if not dl():
        h_cd = (best_h >= 0.5).astype(float)
        h_cd = _fix_sum_bin(h_cd, target)
        cur_cd = _eval(h_cd, dx)
        for sweep in range(100):
            if dl():
                break
            improved = False
            for i in range(n):
                if dl():
                    break
                if h_cd[i] == 0.0:
                    j = np.where(h_cd == 1.0)[0]
                else:
                    j = np.where(h_cd == 0.0)[0]
                if len(j) == 0:
                    continue
                j = j[0]
                h_cd[i] = 1.0 - h_cd[i]
                h_cd[j] = 1.0 - h_cd[j]
                v = _eval(h_cd, dx)
                if v < cur_cd - 1e-12:
                    cur_cd = v
                    if v < best_c5:
                        best_c5 = v
                        best_h = h_cd.copy()
                    improved = True
                else:
                    h_cd[i] = 1.0 - h_cd[i]
                    h_cd[j] = 1.0 - h_cd[j]
            if not improved:
                break

    # Phase 4: Fractional polish (worst-k gradient)
    if not dl():
        h = best_h.copy()
        lr = 0.01
        for it in range(200000):
            if dl():
                break
            if it % 500 == 0:
                lr = max(lr * 0.997, 0.0005)
            grad = _grad_worst(h, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            h_new = _project(h - lr * grad, target)
            v = _eval(h_new, dx)
            if v < best_c5:
                best_c5 = v
                best_h = h_new.copy()
            h = h_new

    # Final validation
    best_h = _project(best_h, target)
    best_c5 = _eval(best_h, dx)

    return best_h, best_c5, n
