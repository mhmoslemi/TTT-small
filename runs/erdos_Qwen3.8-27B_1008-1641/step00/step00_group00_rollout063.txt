```python
def _c5_and_grad(h, dx):
    n = len(h)
    c = correlate(h, 1.0 - h, mode='full')
    c5 = dx * np.max(c)
    m_star = np.argmax(c)
    lag = m_star - (n - 1)
    # Compute gradient of c[m_star] wrt h
    grad = np.zeros(n)

        if 0 <= jm < n:
```

```python
from scipy.optimize import minimize
from scipy.signal import correlate
import time

_g_ts = 0.0
_g_dx = 1.0

def _c5(h, dx):
    c = correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(c))

def _c5_jac(h, dx):
    n = len(h)
    c = correlate(h, 1.0 - h, mode='full')
    m = int(np.argmax(c))
    k = m - (n - 1)
    grad = np.zeros(n)
    for j in range(max(0, k), n):
        grad[j] += 1.0 - h[j - k]
    for j in range(max(0, -k), min(n, n - k)):
        grad[j] -= h[j + k]
    return dx * grad

def _obj(h):
    return _c5(h, _g_dx)

def _grad(h):
    return _c5_jac(h, _g_dx)

def _eq_fun(h):
    return float(np.sum(h) - _g_ts)

def _eq_jac(h):
    return np.ones(len(h))

def _fix_sum(h, ts):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
```

```python
    c = correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(c))

    for j in range(max(0, -k), min(n, n - k)):
        g[j] -= h[j + k]
    return dx * g
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

_g_ts = 0.0
_g_dx = 1.0

def _c5(h, dx):
    c = correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(c))

def _c5_jac(h, dx):
    n = len(h)
    c = correlate(h, 1.0 - h, mode='full')
    m = int(np.argmax(c))
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += 1.0 - h[-k:]
        g[-k:] -= h[:n + k]
    return dx * g

def _obj(h):
    return _c5(h, _g_dx)

def _grad(h):
    return _c5_jac(h, _g_dx)

def _eqf(h):
    return float(np.sum(h) - _g_ts)

def _eqj(h):
    return np.ones(len(h))

def _fix_sum(h, ts):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(300):
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h > 1e-14) & (h < 1.0 - 1e-14)
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _slsqp(h0, dx, ts, mx=500):
    global _g_ts, _g_dx
    _g_ts, _g_dx = ts, dx
    try:
        r = minimize(_obj, h0.copy(), method='SLSQP', jac=_grad,
                     bounds=[(0.0, 1.0)] * len(h0),
                     constraints=[{'type': 'eq', 'fun': _eqf, 'jac': _eqj}],
                     options={'maxiter': mx, 'ftol': 1e-16})
        h = _fix_sum(r.x, ts)
    except Exception:
        h = _fix_sum(h0, ts)
    return h, _c5(h, dx)

def _rand_pt(n, rng):
    return _fix_sum(rng.random(n), n * 0.5)

def _interp(parent, nn):
    no = len(parent)
    xo = (np.arange(no) + 0.5) / no
    xn = (np.arange(nn) + 0.5) / nn
    return np.interp(xn, xo, parent)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)
    try:

    except Exception:
        par = None
    bh, bc = None, 1e10

    # Phase 1: optimize parent
    if par is not None and len(par) >= 4:
        n = len(par)
        dx, ts = 2.0 / n, n * 0.5
        h = _fix_sum(par, ts)
        bh, bc = h.copy(), _c5(h, dx)
        for r in range(600):
            if time.time() > dl: break
            if r == 0:
                h2 = bh
            else:
                sc = 0.01 + 0.15 * (r / 600.0)
                h2 = bh + rng.standard_normal(n) * sc
            h2, c2 = _slsqp(h2, dx, ts, 500)
            if c2 < bc:
                bh, bc = h2.copy(), c2

    # Phase 2: explore other n
    for nt in [44, 46, 48, 50, 52, 56, 60, 64, 70, 72, 80, 90, 96, 100]:
        if time.time() > dl: break
        dx, ts = 2.0 / nt, nt * 0.5
        for a in range(8):
            if time.time() > dl: break
            if a == 0 and par is not None:
                hs = _fix_sum(_interp(par, nt), ts)
            elif a == 1:
                hs = _fix_sum(np.full(nt, 0.5), ts)
            elif a == 2:
                hs = np.zeros(nt)
                hs[:nt // 2] = 1.0
                hs = _fix_sum(hs, ts)
            elif a == 3:
                p = np.arange(nt, dtype=float)
                hs = _fix_sum(0.5 + 0.3 * np.sin(2.0 * np.pi * p / nt), ts)
            else:
                hs = _rand_pt(nt, rng)
            h2, c2 = _slsqp(hs, dx, ts, 400)
            if c2 < bc:
                bh, bc = h2.copy(), c2

    # Phase 3: fine-tune best
    if bh is not None and time.time() < dl:
        nb = len(bh)
        dxb, tsb = 2.0 / nb, nb * 0.5
        for _ in range(60):
            if time.time() > dl: break
            h2, c2 = _slsqp(bh, dxb, tsb, 3000)
            if c2 >= bc - 1e-13: break
            bh, bc = h2.copy(), c2
        for r in range(800):
            if time.time() > dl: break
            sc = 0.002 + 0.04 * (r / 800.0)
            h2, c2 = _slsqp(bh + rng.standard_normal(nb) * sc, dxb, tsb, 400)
            if c2 < bc:
                bh, bc = h2.copy(), c2

    if bh is None:
        bh = _fix_sum(np.full(50, 0.5), 25.0)
        bc = _c5(bh, 0.04)
    return bh, float(bc), int(len(bh))
```
