from scipy.optimize import minimize
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr), corr * dx

def project_feasible(v, target):
    # Bisection for projection onto {0<=h<=1, sum=h = target}
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def c5_subgrad(h):
    """Return subgradient of C5 at h (gradient of active constraint)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find active lags
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    # Gradient contribution for each active lag
    # corr[k] = sum_i h[i + k - (n-1)] * (1 - h[i]) for valid i
    # Let lag = k - (n-1)
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        # Terms: h[i+lag] * (1-h[i]) for i where both indices valid
        if lag >= 0:
            # i from 0 to n-1-lag
            for i in range(n - lag):
                # d/dh[j] of h[i+lag]*(1-h[i])
                # j = i+lag: contributes (1-h[i])
                # j = i: contributes -h[i+lag]
                grad[i + lag] += (1 - h[i]) * dx
                grad[i] += -h[i + lag] * dx
        else:
            # lag < 0, let l = -lag
            l = -lag
            for i in range(l, n):
                # h[i] * (1-h[i-l])
                grad[i] += (1 - h[i - l]) * dx
                grad[i - l] += -h[i] * dx
    # Average over active constraints
    grad /= len(active)
    return grad

    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()

    lr = lr_start

        val = c5_objective(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        lr = lr_start / (1.0 + it * 0.01)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_feasible(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_val(h, T, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = scaled.max()
    return dx * (m + np.log(np.sum(np.exp(scaled - m)))) / T

def smooth_max_grad(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = scaled.max()
    weights = np.exp(scaled - m)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                grad[i + lag] += w * (1.0 - h[i])
                grad[i] -= w * h[i + lag]
        else:
            l = -lag
            for i in range(l, n):
                grad[i] += w * (1.0 - h[i - l])
                grad[i - l] -= w * h[i]
    return dx * grad

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    active = np.where(np.abs(corr - max_val) < 1e-8)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                grad[i + lag] += (1.0 - h[i])
                grad[i] -= h[i + lag]
        else:
            l = -lag
            for i in range(l, n):
                grad[i] += (1.0 - h[i - l])
                grad[i - l] -= h[i]
    return dx * grad / len(active)

import numpy as np
from scipy.optimize import minimize

_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_feasible(v, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _smooth_obj(x):
    T = _G['T']
    dx = _G['dx']
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr
    m = scaled.max()
    return dx * (m + np.log(np.sum(np.exp(scaled - m)))) / T

def _smooth_grad(x):
    T = _G['T']
    dx = _G['dx']
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr
    m = scaled.max()
    w = np.exp(scaled - m)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                grad[i + lag] += wk * (1.0 - x[i])
                grad[i] -= wk * x[i + lag]
        else:
            l = -lag
            for i in range(l, n):
                grad[i] += wk * (1.0 - x[i - l])
                grad[i - l] -= wk * x[i]
    return dx * grad

    return x.sum() - _G['target']

    mv = corr.max()

    grad = np.zeros(n)

        lag = k - (n - 1)

                grad[i + lag] += (1.0 - h[i])
                grad[i] -= h[i + lag]
        else:
            l = -lag
            for i in range(l, n):
                grad[i] += (1.0 - h[i - l])
                grad[i - l] -= h[i]
    return dx * grad / len(active)

            bv = v
            best = h.copy()
        lr = lr0 / (1.0 + 0.005 * it)
    return best, bv
