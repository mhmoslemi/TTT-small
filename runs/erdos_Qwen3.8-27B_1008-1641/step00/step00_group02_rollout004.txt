        if new_score >= best_score - 1e-13:
            break

    # Phase 2: Stochastic local search with annealing
    step_size = 0.12
    temp = 0.004
    no_improve = 0
    prev_best = best_score

    while time.time() < deadline:
        temp = max(1e-9, temp * 0.9992)

        for _ in range(2000):
            if time.time() > deadline:
                break
            k = int(rng.integers(1, 7))
            idx = rng.integers(0, n, size=k)
            new_h = best_h.copy()
            new_h[idx] += rng.normal(0.0, step_size, size=k)
            new_h = repair_sum(new_h, target_sum)
            score = evaluate_h(new_h, n)

            if score < best_score:
                best_h = new_h
                best_score = score
                no_improve = 0
                step_size = max(0.003, step_size * 0.994)
            else:
                no_improve += 1
                if temp > 1e-10:
                    ratio = np.exp(-(score - best_score) / temp)
                    if rng.random() < ratio:
                        best_h = new_h
                step_size = min(0.35, step_size * 1.004)

        # Periodic jump to escape local optima
        if no_improve > 8000 and time.time() < deadline:
            jitter_scale = 0.25 + 0.1 * rng.random()
            jump = rng.normal(0.0, jitter_scale, n)
            new_h = repair_sum(best_h + jump, target_sum)
            new_score = evaluate_h(new_h, n)
            for _ in range(300):
                if time.time() > deadline:
                    break
                idx2 = int(rng.integers(0, n))
                tmp = new_h.copy()
                tmp[idx2] += rng.normal(0.0, 0.04)
                tmp = repair_sum(tmp, target_sum)
                s2 = evaluate_h(tmp, n)
                if s2 < new_score:
                    new_h = tmp
                    new_score = s2
            if new_score < best_score + 0.02:
                best_h = new_h
                best_score = new_score
            no_improve = 0

        # Reduce temperature if stuck for a while
        if no_improve > 3000:
            temp *= 0.5
            no_improve = 0

    # Phase 3: Final coordinate polish
    if time.time() < deadline:
        for _ in range(3):
            if time.time() > deadline:
                break
            for i in range(n):
                if time.time() > deadline:
                    break
                old_val = best_h[i]
                for delta in [0.001, -0.001, 0.005, -0.005, 0.02, -0.02,
                              0.05, -0.05]:
                    tmp = best_h.copy()
                    tmp[i] = old_val + delta
                    if tmp[i] < 0 or tmp[i] > 1:
                        continue
                    tmp = repair_sum(tmp, target_sum)
                    s = evaluate_h(tmp, n)
                    if s < best_score:
                        best_h = tmp
                        best_score = s

    best_h = repair_sum(best_h, target_sum)
    best_score = evaluate_h(best_h, n)
    return (best_h, best_score, n)
