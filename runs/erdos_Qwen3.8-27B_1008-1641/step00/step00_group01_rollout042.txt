def repair(x, n):
    h = np.clip(x.astype(float), 0.0, 1.0)
    target = n * 0.5
    for _ in range(30):
        s = h.sum()
        if s < 1e-15:
            h[:] = 0.5
            return h
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        m = (h > 1e-8) & (h < 1.0 - 1e-8)
        if m.any():
            h[m] += diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def objective(x):
    n = len(x)
    h = repair(x, n)
    return evaluate_c5(h)


def nm_polish(h, n, max_it=20000):
    result = minimize(
        objective, h,
        method='Nelder-Mead',
        options={'maxiter': max_it, 'xatol': 1e-11, 'fatol': 1e-13, 'adaptive': True}
    )
    h_new = repair(result.x, n)
    return h_new, evaluate_c5(h_new)


def run(seed=42, budget_s=1100, **kwargs):
    import time
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n = len(initial_h_values)
    best_h = np.array(initial_h_values, dtype=float)
    best_h = repair(best_h, n)
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Differential Evolution (multiple rounds)
    for rnd in range(4):
        if tl() < 80:
            break
        sp = 35 if n <= 55 else 25
        result = differential_evolution(
            objective,
            bounds=[(0.0, 1.0)] * n,
            seed=seed + rnd * 7919,
            maxiter=int(max(50, min(500, tl() * 0.12))),
            popsize=sp,
            tol=1e-8,
            mutation=(0.5, 1.2),
            recombination=0.8,
            init='latinhypercube',
            polish=False,
            update_interval=1,
        )
        h_c = repair(result.x, n)
        c = evaluate_c5(h_c)
        if c < best_c5:
            best_c5, best_h = c, h_c

    # Phase 2: Nelder-Mead refinement
    if tl() > 15:
        h_c, c = nm_polish(best_h, n, 30000)
        if c < best_c5:
            best_c5, best_h = c, h_c

    # Phase 3: Perturbation + local search
    for _ in range(15):
        if tl() < 20:
            break
        h_p = best_h + rng.normal(0, 0.15, n)
        h_p = repair(h_p, n)
        h_c, c = nm_polish(h_p, n, 5000)
        if c < best_c5:
            best_c5, best_h = c, h_c

    # Phase 4: Binarize + re-optimize
    if tl() > 20:
        h_b = np.where(best_h > 0.5, 1.0, 0.0)
        h_b = repair(h_b, n)
        h_c, c = nm_polish(h_b, n, 15000)
        if c < best_c5:
            best_c5, best_h = c, h_c

    # Phase 5: Try alternate n values
    for n_try in [30, 35, 50, 60, 80, 100]:
        if tl() < 45:
            break
        if n_try == n:
            continue
        sp = 25 if n_try <= 60 else 20
        result = differential_evolution(
            objective,
            bounds=[(0.0, 1.0)] * n_try,
            seed=seed + n_try * 31,
            maxiter=int(max(30, min(200, tl() * 0.08))),
            popsize=sp,
            tol=0.005,
            mutation=(0.5, 1.5),
            recombination=0.7,
            init='latinhypercube',
            polish=False,
        )
        h_c = repair(result.x, n_try)
        c = evaluate_c5(h_c)
        if c < best_c5:
            best_c5, best_h = c, h_c
            n_try_best = n_try
            if tl() > 15:
                h_c2, c2 = nm_polish(best_h, len(best_h), 10000)
                if c2 < best_c5:
                    best_c5, best_h = c2, h_c2

    # Final polish
    if tl() > 8:
        n_f = len(best_h)
        h_c, c = nm_polish(best_h, n_f, 40000)
        if c < best_c5:
            best_c5, best_h = c, h_c

    # Validate final answer
    best_h = np.clip(best_h, 0.0, 1.0)
    nf = len(best_h)
    target = nf * 0.5
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        m = (best_h > 1e-7) & (best_h < 1.0 - 1e-7)
        if m.any():
            best_h[m] += diff / m.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
