def project_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to increase sum
        # Add to elements not at 1
        mask = h < 1.0
        count = np.sum(mask)
        if count == 0:
            return h  # Can't increase
        increment = diff / count
        # Check if all can absorb the increment
        max_increment = 1.0 - h[mask]
        if np.min(max_increment) >= increment:
            h[mask] += increment
        else:
            # Water filling: sort available capacity
            # Use iterative approach
            remaining = diff
            while remaining > 1e-10:
                mask = h < 1.0
                if not np.any(mask):
                    break
                available = 1.0 - h[mask]
                step = min(remaining / np.sum(mask), np.min(available))
                h[mask] += step
                remaining -= step * np.sum(mask)
    else:
        # Need to decrease sum
        remaining = -diff
        while remaining > 1e-10:
            mask = h > 0.0
            if not np.any(mask):
                break
            available = h[mask]
            step = min(remaining / np.sum(mask), np.min(available))
            h[mask] -= step
            remaining -= step * np.sum(mask)
    return h

So C5 = dx * max(np.correlate(h, 1-h, mode='full')) is correct.
