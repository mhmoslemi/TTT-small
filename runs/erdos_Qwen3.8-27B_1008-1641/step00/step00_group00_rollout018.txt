from scipy.optimize import minimize

def compute_c5(h):

    dx = 2.0 / n

    return np.max(corr) * dx

    S = n / 2.0
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    m_star = int(np.argmax(corr))
    c5_val = float(corr[m_star] * dx)
    lag = m_star - (n - 1)

        grad[:n - L] -= h[L:]

        h_up = upsample(best_h, target_n)
        h_up = project_feasible(h_up, target_n)
        h_pg, c5_pg = pm_subgradient(h_up, target_n, max_iter=15000, time_limit=budget_s, t_start=t0)
        if c5_pg < best_c5:
            best_h, best_c5 = h_pg, c5_pg

    # Final: ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
