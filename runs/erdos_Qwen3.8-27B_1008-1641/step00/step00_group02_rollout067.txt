def c5_value_and_grad(h, top_k=1):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    # Use top_k lags for a smoother subgradient
    top_indices = np.argsort(r)[-top_k:]
    C5 = float(r[top_indices[0]] * dx)  # max value

        valid1 = (0 <= i+s < n)
        # valid2: h[i] appears in (1-h[j]) where j = i, and first factor is h[i-s]
        # h[i-s] must be valid: 0 <= i-s < n, and i must be valid (it is)
        valid2 = (0 <= i-s < n)

        # For lag k, valid j range: max(0, n-1-k) <= j <= min(n-1, n-1+k)
        j_lo = max(0, n-1-k)
        j_hi = min(n-1, n-1+k)

        v1 = (j_lo <= i <= j_hi) and (0 <= i+s < n)
        # valid2: j=i-s must be in [j_lo, j_hi] AND i in [0, n-1] (always true)
        v2 = (j_lo <= i-s <= j_hi) and (0 <= i-s < n)
        
        slope = (1.0 - h[i+s]) * v1 - h[i-s] * v2
        a[k] = slope
        b[k] = r[k] - slope * h[i]
    
    # Now minimize max_k (a[k]*u + b[k]) over u in [-h[i], 1-h[i]]
    u_lo = -h[i]
    u_hi = 1.0 - h[i]
    
    # Grid search
    grid = np.linspace(u_lo, u_hi, 200)
    vals = np.max(a[:, None] * grid[None, :] + b[:, None], axis=0)
    u_star = grid[np.argmin(vals)]
    
    return h[i] + u_star

def solve_lp(n):
    h = cp.Variable(n)
    z = cp.Variable((n, n))  # z[i,j] = h[i]*h[j]
    
    constraints = [h >= 0, h <= 1]
    constraints.append(cp.sum(h) == n/2.0)
    
    # McCormick constraints for all i != j
    for i in range(n):

            if i == j:

            constraints.append(z[i,j] >= 0)
            constraints.append(z[i,j] <= h[i])
            constraints.append(z[i,j] <= h[j])
            constraints.append(z[i,j] >= h[i] + h[j] - 1)

def c5_subgrad(h, top_k=1):
