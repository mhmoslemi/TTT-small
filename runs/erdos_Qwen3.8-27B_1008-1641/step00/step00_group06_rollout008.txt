            if abs(p1 - p2) > L:
                tmp = h_init[p1:p1+L].copy()
                h_init[p1:p1+L] = h_init[p2:p2+L]
                h_init[p2:p2+L] = tmp
            h_init += 0.02 * np.random.randn(n)
        elif r < 0.90:
            # Modulate by random smooth function
            freq = np.random.uniform(1.0, 4.0)
            phases = np.linspace(0, freq * 2 * np.pi, n)
            h_init = best_h + 0.15 * np.sin(phases + np.random.uniform(0, 2*np.pi))
        else:
            h_init += 0.25 * np.random.randn(n)

        h_init = project_feasible(h_init, n, target_sum)
        res = do_optimize(h_init, 200, 1e-13)

        if res is not None and res.fun < best_c5 - 1e-9:
            best_h = project_feasible(res.x, n, target_sum)
            best_c5 = compute_c5(best_h)
            stall = 0
            noise_scale = max(0.01, noise_scale * 0.92)
        else:
            stall += 1
            if stall >= 8:
                noise_scale = min(0.5, noise_scale * 1.4)
                stall = 0

    # Phase 3: Final tight polish
    if elapsed() < budget_s - 10:
        res = do_optimize(best_h, 1000, 1e-15)
        if res is not None and res.fun < best_c5:
            best_h = project_feasible(res.x, n, target_sum)
            best_c5 = compute_c5(best_h)

    # Additional refinement: try moving mass from high-correlation regions
    if elapsed() < budget_s - 5:
        for _ in range(5):
            if elapsed() > budget_s - 2:
                break
            res = do_optimize(best_h + 0.01 * np.random.randn(n), 300, 1e-14)
            if res is not None and res.fun < best_c5:
                best_h = project_feasible(res.x, n, target_sum)
                best_c5 = compute_c5(best_h)

    best_h = project_feasible(best_h, n, target_sum)
    final_c5 = compute_c5(best_h)

    try:
        evaluate_erdos_solution(best_h)
    except Exception:
        pass

    return best_h, final_c5, n
