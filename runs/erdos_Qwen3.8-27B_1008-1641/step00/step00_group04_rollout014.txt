import numpy as np
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(20):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _upsample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 1, n_old, endpoint=False)
    xn = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _coord_optimize(h, n, time_deadline):
    """Coordinate descent to minimize max correlation"""
    h = _project(h.copy(), n)
    best_c5 = _eval(h)
    best_h = h.copy()
    s_range = np.arange(-(n-1), n)  # length 2n-1
    
    sweep = 0
    stuck = 0
    while time.time() < time_deadline and stuck < 100:
        sweep += 1
        sweep_improved = False
        for j in range(n):
            if time.time() > time_deadline:
                break
            # Compute coefficients of h[j] in each lag
            coeffs = np.zeros(2 * n - 1)
            # First factor: s in [j-(n-1), j]
            m1 = (s_range >= j - (n-1)) & (s_range <= j)
            idx1 = j - s_range[m1]
            coeffs[m1] += 1.0 - h[idx1]
            # Second factor: s in [-j, n-1-j]
            m2 = (s_range >= -j) & (s_range <= n - 1 - j)
            idx2 = j + s_range[m2]
            coeffs[m2] -= h[idx2]
            
            # const = corr_full - coeffs * h[j]
            corr_full = np.correlate(h, 1.0 - h, mode="full")
            consts = corr_full - coeffs * h[j]
            
            # Minimize max_p(coeffs[p]*t + consts[p]) over t in [0,1]
            # f is convex, use fine grid
            grid = np.linspace(0.0, 1.0, 64)
            vals = np.max(coeffs[:, None] * grid[None, :] + consts[:, None], axis=0)
            best_idx = np.argmin(vals)
            new_t = grid[best_idx]
            
            if abs(new_t - h[j]) > 1e-10:
                h[j] = new_t
                sweep_improved = True
        
        h = _project(h, n)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5

import numpy as np
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(30):
        diff = h.sum() - target
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _upsample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 1, n_old, endpoint=False)
    xn = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _coord_descent(h_init, n, deadline, grid_size=128):
    h = _project(h_init.copy(), n)
    s_rng = np.arange(-(n - 1), n)
    best_c5 = _eval(h)
    best_h = h.copy()
    stuck = 0
    npz = 2 * n - 1

    while time.time() < deadline and stuck < 500:
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return best_h, best_c5
            coeffs = np.zeros(npz)
            m1 = (s_rng >= j - (n - 1)) & (s_rng <= j) & (s_rng != 0)
            if m1.any():
                coeffs[m1] += 1.0 - h[j - s_rng[m1]]
            m2 = (s_rng >= -j) & (s_rng <= n - 1 - j) & (s_rng != 0)
            if m2.any():
                coeffs[m2] -= h[j + s_rng[m2]]

            corr_full = np.correlate(h, 1.0 - h, mode="full")
            consts = corr_full - coeffs * h[j]
            consts[n - 1] -= h[j] * (1.0 - h[j])

            grid = np.linspace(0.0, 1.0, grid_size)
            vals = coeffs * grid[None, :] + consts[:, None]
            vals[n - 1, :] = consts[n - 1] + grid - grid * grid
            max_vals = np.max(vals, axis=0)
            best_idx = int(np.argmin(max_vals))
            new_t = grid[best_idx]

            if abs(new_t - h[j]) > 1e-10:
                h[j] = new_t
                improved = True

        h = _project(h, n)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

    return best_h, best_c5

def _perturb_restart(h, n, rng):
    h_p = h.copy()
    for _ in range(3):
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            tmp = h_p[i]
            h_p[i] = h_p[j]
            h_p[j] = tmp
    h_p = _project(h_p, n)
    return h_p

def _stochastic_local(h, n, deadline, rng):
    h = h.copy()
    best_c5 = _eval(h)
    best_h = h.copy()
    dx = 2.0 / n
    lr = 0.05

    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode="full")
        top_k = np.argsort(c)[-5:]
        for k in top_k:
            p = k
            s = p - (n - 1)
            i_min = max(0, s)
            i_max = min(n - 1, n - 1 + s)
            if i_min >= i_max:
                continue
            idx = np.arange(i_min, i_max + 1)
            a = h[idx]
            b = h[idx - s]
            pair_score = -a * (1.0 - b)
            worst_local = np.argmin(pair_score)
            ii = idx[worst_local]
            jj = ii - s
            if 0 <= jj < n:
                delta = 0.1 * (1.0 - a[worst_local]) if a[worst_local] > 0.5 else -0.1 * (1.0 - a[worst_local])
                h[ii] = np.clip(h[ii] - delta, 0, 1)
                h[jj] = np.clip(h[jj] + delta, 0, 1)
        h = _project(h, n)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)

    resolutions = [max(n0 * 2, 84), max(n0 * 4, 168), max(n0 * 8, 336)]

                h_start = h_up.copy()

            h_start = _project(h_start, nr)

                h_up = ho.copy()  # Update reference for next restart

    # Phase 3: Final refinement on best solution

    if remaining > 30:
        n_final = len(best_h)
        if n_final < 512:
