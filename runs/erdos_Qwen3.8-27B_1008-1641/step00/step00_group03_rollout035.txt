def soft_max_obj(h, T, dx):
    omh = 1 - h
    corr = np.correlate(h, omh, mode='full')
    vals = corr * dx
    # soft max
    m = np.max(vals)
    e = np.exp((vals - m) / T)
    log_sum = m + T * np.log(np.sum(e))
    # gradient: dx * sum_s exp(vals_s/T) / sum_exp * dC_s/dh[j]
    weights = e / np.sum(e)  # shape (2n-1,)
    grad = np.zeros_like(h)
    for s in range(-n+1, n):  # s is the shift
        w = weights[s + n - 1]
        if w < 1e-10: continue
        # dC_s/dh[j] = (1-h[j+s]) - h[j-s] (with boundary conditions)
        # For shift s: valid i are max(0,-s) to min(n-1, n-1-s)
        # h[j] appears as h[i] when i=j (need 0<=j-s<n): contributes (1-h[j+s])
        # h[j] appears as h[i+s] when i+s=j, i=j-s (need 0<=j-s<n): contributes -h[j-s]
        
        j = np.arange(n)
        valid_j_plus = (j + s >= 0) & (j + s < n)
        valid_j_minus = (j - s >= 0) & (j - s < n)
        
        contrib = np.zeros(n)
        contrib[valid_j_plus] += (1 - h[j[valid_j_plus] + s])
        contrib[valid_j_minus] -= h[j[valid_j_minus] - s]
        
        grad += w * contrib
    
    grad *= dx
    return log_sum, grad

    m = np.max(vals)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def sm_obj(x, T, dx, lam, n):
    omh = 1.0 - x
    corr = np.correlate(x, omh, mode='full')
    vals = corr * dx
    m = np.max(vals)
    e = np.exp((vals - m) / T)
    return m + T * np.log(np.sum(e)) + lam * (np.sum(x) - n / 2.0) ** 2


def sm_grad(x, T, dx, lam, n):
    omh = 1.0 - x
    corr = np.correlate(x, omh, mode='full')
    vals = corr * dx
    m = np.max(vals)
    e = np.exp((vals - m) / T)
    w = e / np.sum(e)
    n2 = 2 * n - 1
    grad = np.zeros(n)
    for idx in range(n2):
        s = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-12:
            continue
        lo1, hi1 = max(0, -s), min(n, n - s)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * omh[lo1 + s:hi1 + s]
        lo2, hi2 = max(0, s), min(n, n + s)
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * x[lo2 - s:hi2 - s]
    grad *= dx
    grad += 2.0 * lam * (np.sum(x) - n / 2.0)
    return grad


def project_sum(h, n):
    h = h + (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)


def anno_optimize(h0, n, deadline, rng, rounds=25):

    h = project_sum(np.clip(h0, 0, 1), n)
    best_h, best_c5 = h.copy(), compute_c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    for r in range(rounds):
        if time.time() > deadline:
            break
        f = r / max(rounds - 1, 1)
        T = max(1.0 * (0.01) ** f, 0.01)
        lam = min(1.0 * (10000.0) ** f, 10000.0)
        args = (T, dx, lam, n)
        res = minimize(sm_obj, h, args=args, jac=sm_grad,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 60, 'ftol': 1e-12, 'gtol': 1e-9})
        h = project_sum(np.clip(res.x, 0, 1), n)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        if r < rounds - 3:
            h = project_sum(h + rng.normal(0, 0.07 * (1.0 - f), n), n)
    return best_h, best_c5
