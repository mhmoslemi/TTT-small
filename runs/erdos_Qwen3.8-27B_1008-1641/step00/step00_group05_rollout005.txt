        s = np.sum(projected)

        else:

    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    vals = corr * dx
    
    max_val = np.max(vals)
    exp_vals = np.exp(tau * (vals - max_val))
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = (1.0/tau) * np.log(total) + max_val  # This is the smooth max

                grad[i-t] += -w * h[i]

def smooth_max_and_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    vals = corr * dx
    
    max_val = np.max(vals)
    exp_vals = np.exp(tau * (vals - max_val))
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = (1.0/tau) * np.log(total) + max_val
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for m in range(2*n-1):

        if w < 1e-12:
            continue

        if s >= 0:
            grad[:n-s] += w * one_minus_h[s:]
            grad[s:] += -w * h[:n-s]
        else:
            t = -s
            grad[t:] += w * one_minus_h[:n-t]
            grad[:n-t] += -w * h[t:]
    
    grad *= dx
    return smooth_val, grad
