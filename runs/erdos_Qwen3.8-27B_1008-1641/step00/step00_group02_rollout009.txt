import numpy as np
import time
from scipy.optimize import minimize

_GLOBAL_N = 0
_GLOBAL_GAMMA = 1.0
_GLOBAL_S = 0.0
_LAST_X = None
_LAST_G = None
_PAIR_STEPS = (1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1, 2e-1, 0.5)
_ACTIVE_STEPS = (0.2, 0.1, 0.05, 0.02, 0.1, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 1e-5)

def project_box_sum(y, S):
    x = np.asarray(y, dtype=float).copy().ravel()
    if x.size == 0:
        return x
    x = np.clip(x, 0.0, 1.0)
    diff = S - float(x.sum())
    if abs(diff) <= 1e-12:
        return x
    a = float(x.min()) - 1.0
    b = float(x.max()) + 1.0
    for _ in range(50):
        m = 0.5 * (a + b)
        s = float(np.clip(x - m, 0.0, 1.0).sum())
        if s > S:
            a = m
        else:
            b = m
    theta = 0.5 * (a + b)
    x = np.clip(x - theta, 0.0, 1.0)
    diff = S - float(x.sum())
    for _ in range(8):
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            mask = x < (1.0 - 1e-12)
        else:
            mask = x > 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
        diff = S - float(x.sum())
    for _ in range(10):
        diff = S - float(x.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        val = x[idx] + diff
        if 0.0 <= val <= 1.0:
            x[idx] = val
        else:
            break
    return x

def score_max(h):
    h = np.asarray(h, dtype=float).ravel()
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def score_c(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return float("inf")
    return score_max(h) * (2.0 / float(h.size))

def smooth_score_and_grad(x):
    x = np.asarray(x, dtype=float).ravel()
    n = int(_GLOBAL_N) if _GLOBAL_N else int(x.size)
    xx = np.clip(x, 0.0, 1.0)
    om = 1.0 - xx
    F = np.correlate(xx, om, mode="full")
    fmax = float(F.max())
    gamma = float(_GLOBAL_GAMMA) if _GLOBAL_GAMMA > 0 else 1.0
    z = gamma * (F - fmax)
    ew = np.exp(z)
    sw = float(ew.sum())
    w = ew / sw
    phi = fmax + float(np.log(sw)) / gamma
    g = np.zeros(n, dtype=float)
    for d in range(n):
        wv = float(w[n - 1 + d])
        if wv > 0.0:
            g[d:] += wv * (1.0 - xx[:n - d])
            g[:n - d] -= wv * xx[d:]
        if d:
            wv = float(w[n - 1 - d])
            if wv > 0.0:
                g[:n - d] += wv * (1.0 - xx[d:])
                g[d:] -= wv * xx[:n - d]
    return phi, g

def slq_obj(x):
    global _LAST_X, _LAST_G
    x = np.asarray(x, dtype=float).ravel()
    phi, g = smooth_score_and_grad(x)
    _LAST_X = x.copy()
    _LAST_G = g
    return phi

def slq_jac(x):
    global _LAST_X, _LAST_G
    x = np.asarray(x, dtype=float).ravel()
    if _LAST_X is not None and _LAST_X.shape == x.shape and np.array_equal(_LAST_X, x):
        return _LAST_G
    _, g = smooth_score_and_grad(x)
    return g

def slq_eq(x):
    return float(np.sum(np.asarray(x, dtype=float).ravel())) - float(_GLOBAL_S)

def slsqp_sequence(x, n, time_stop):
    global _GLOBAL_N, _GLOBAL_GAMMA, _GLOBAL_S
    S = 0.5 * float(n)
    best = project_box_sum(x, S)
    c = score_c(best)
    _GLOBAL_N = int(n)
    _GLOBAL_S = S
    if n <= 60:
        gammas = (20.0, 8.0, 4.0, 2.0, 1.0)
        maxits = (30, 30, 35, 35, 40)
    else:
        gammas = (12.0, 6.0, 3.0, 2.0, 1.0)
        maxits = (18, 18, 22, 22, 25)
    bounds = [(0.0, 1.0)] * int(n)
    cons = [{"type": "eq", "fun": slq_eq}]

        if time.time() >= time_stop - 2.0:

            )
            if res is not None and hasattr(res, "x") and res.x is not None:
                cand = project_box_sum(res.x, S)
                cc = score_c(cand)
                if cc < c - 1e-15:
                    best, c = cand, cc
        except Exception:
            pass
    return best, c

def fourier_candidate(n, rng, t):
    x = np.full(n, 0.5, dtype=float)
    for f in (1, 2, 3, 4):
        a = float(rng.uniform(0.0, 0.25))
        p = float(rng.uniform(0.0, 2.0 * np.pi))
        x += a * np.cos(2.0 * np.pi * f * t + p)
        b = float(rng.uniform(0.0, 0.12))
        q = float(rng.uniform(0.0, 2.0 * np.pi))
        x += b * np.sin(2.0 * np.pi * f * t + q)
    return x

def block_candidate(n, m, frac, phase):
    period = max(1, int(round(float(n) / float(m))))
    L = max(1, int(round(period * frac)))
    L = min(L, period)
    phase = int(phase) % period
    x = np.zeros(n, dtype=float)
    st = -phase
    while st < n:
        a = max(0, st)
        b = min(n, a + L)
        if b > a:
            x[a:b] = 1.0
        st += period
    return x

def try_symmetries(h, n):
    h = np.asarray(h, dtype=float).ravel()
    best = project_box_sum(h, 0.5 * float(n))
    bc = score_c(best)
    rev = h[::-1].copy()

        if c < bc - 1e-15:

                if abs(tv) < tmax - 1e-12:
                    tvals.append(tv)
        for tv in tvals:
            tmp[:] = best + tv * d
            if np.any(tmp < -1e-10) or np.any(tmp > 1.0 + 1e-10):
                continue
            v = score_max(tmp)
            if v < best_v - 1e-13:
                best_v, best_t = v, tv
        if best_v < c - 1e-13:
            best = project_box_sum(best + best_t * d, S)
            c = score_max(best)
            no = 0
        else:
            no += 1
    best = project_box_sum(best, S)
    return best, score_c(best)

def optimize_candidate(x, n, time_stop, seed):
    n = int(n)
    S = 0.5 * float(n)
    best = project_box_sum(x, S)
    bc = score_c(best)
    rng = np.random.default_rng(seed)
    scales = (0.2, 0.08, 0.03, 0.01)
    for phase in range(4):
        if time.time() >= time_stop - 25.0:
            break
        improved = False
        b, c = try_symmetries(best, n)
        if c < bc - 1e-15:
            best, bc = b, c
            improved = True
        if time.time() < time_stop - 30.0:
            t1 = min(time_stop - 25.0, time.time() + (120.0 if phase == 0 else 65.0))
            b, c = slsqp_sequence(best, n, t1)
            b, c = try_symmetries(b, n)
            if c < bc - 1e-15:
                best, bc = b, c
                improved = True
        if time.time() < time_stop - 10.0:
            t2 = min(time_stop - 6.0, time.time() + 45.0)
            b, c = pair_hill(best, t2, seed + phase * 1000 + 11)
            if c < bc - 1e-15:
                best, bc = b, c
                improved = True
        if time.time() < time_stop - 8.0:
            t3 = min(time_stop - 5.0, time.time() + 12.0)
            b, c = active_refine(best, t3, seed + phase * 1000 + 51)
            if c < bc - 1e-15:
                best, bc = b, c
                improved = True
        if time.time() < time_stop - 6.0:
            t4 = min(time_stop - 4.0, time.time() + 12.0)
            b, c = random_direction_search(best, t4, seed + phase * 1000 + 91)
            if c < bc - 1e-15:
                best, bc = b, c
                improved = True
        if not improved and time.time() < time_stop - 30.0:
            sc = scales[phase % 4]
            d = rng.standard_normal(n)
            d = d - d.mean()
            nm = float(np.linalg.norm(d))
            if nm > 1e-9:
                y = project_box_sum(best + sc * (d / nm), S)
                t5 = min(time_stop - 25.0, time.time() + 35.0)
                b, c = slsqp_sequence(y, n, t5)
                b, c = pair_hill(b, min(time_stop - 10.0, time.time() + 15.0), seed + phase * 5000 + 31)
                if c < bc - 1e-15:
                    best, bc = b, c
    best, bc = try_symmetries(best, n)
    best = project_box_sum(best, S)
    bc = score_c(best)
    return best, bc

def resize_h(h0, n):
    h0 = np.asarray(h0, dtype=float).ravel()
    n = int(n)
    if h0.size == n:
        return project_box_sum(h0, 0.5 * float(n))
    x0 = np.linspace(0.0, 1.0, h0.size, endpoint=False)
    x1 = np.linspace(0.0, 1.0, n, endpoint=False)
    y = np.interp(x1, x0, h0)
    return project_box_sum(y, 0.5 * float(n))

def final_fix(h, S):
    x = project_box_sum(h, S)
    x = np.clip(x, 0.0, 1.0)
    for _ in range(20):
        diff = S - float(x.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        val = x[idx] + diff
        if 0.0 <= val <= 1.0:
            x[idx] = val
        else:
            break
    return x

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    max_end = t0 + max(1.0, float(budget_s))
    base = None
    if "initial_h_values" in globals():
        try:
            base = np.array(globals()["initial_h_values"], dtype=float).ravel()
        except Exception:
            base = None
    if base is None or base.size < 10:
        n = 45
        rc = random_candidates(n, seed, 8)
        if rc:
            base, bc0 = rc[0][1], rc[0][0]
        else:
            base = project_box_sum(np.full(n, 0.5), 0.5 * float(n))
            bc0 = score_c(base)
        best = base.copy()
        best_c = float(bc0)
        best_n = n
    else:
        best_n = int(base.size)
        best = project_box_sum(base, 0.5 * float(best_n))
        best_c = score_c(best)

    cands = quick_candidates(best, best_n, seed + 7, 8)
    if not cands:
        cands = [(best_c, best.copy())]
    caps = (260.0, 160.0, 110.0, 85.0, 70.0, 60.0, 50.0, 40.0)
    for idx in range(len(cands)):
        if time.time() > max_end - 160.0:
            break
        rem = max_end - time.time()
        cap = caps[idx] if idx < len(caps) else 35.0
        alloc = min(cap, rem - 150.0)
        if alloc < 15.0:
            alloc = min(cap, rem - 35.0)
        if alloc < 12.0:
            break
        stop = min(max_end - 130.0, time.time() + alloc)
        if stop <= time.time() + 5.0:
            continue
        cx = cands[idx][1]
        if cands[idx][0] > best_c + 0.08 and idx > 0 and rem < 520.0:
            continue
        try:
            nh, nc = optimize_candidate(cx, best_n, stop, seed + idx * 10000 + 1)
        except Exception:
            continue
        if nc < best_c - 1e-15:
            best, best_c, best_n = nh, float(nc), int(len(nh))

    if max_end - time.time() > 260.0 and best.size > 0:
        n2 = 2 * best_n if 2 * best_n <= 200 else 99
        if n2 != best_n and n2 >= 10:
            base2 = resize_h(best, n2)
            cands2 = quick_candidates(base2, n2, seed + 5555, 3)
            if cands2 and max_end - time.time() > 180.0:
                alloc = min(160.0, max_end - time.time() - 130.0)
                if alloc > 30.0:
                    stop = min(max_end - 110.0, time.time() + alloc)
                    try:
                        nh, nc = optimize_candidate(cands2[0][1], n2, stop, seed + 55555)
                        if nc < best_c - 1e-15:
                            best, best_c, best_n = nh, float(nc), int(len(nh))
                    except Exception:
                        pass

    if max_end - time.time() > 40.0 and best.size > 0:
        if max_end - time.time() > 220.0:
            try:
                nh, nc = optimize_candidate(best, best_n, min(max_end - 30.0, time.time() + 190.0), seed + 999901)
                if nc < best_c - 1e-15:
                    best, best_c = nh, float(nc)
            except Exception:
                pass
        stop = min(max_end - 6.0, time.time() + 60.0)
        try:
            b, c = pair_hill(best, stop, seed + 100100)
            if c < best_c - 1e-15:
                best, best_c = b, float(c)
        except Exception:
            pass
        if max_end - time.time() > 25.0:
            stop = min(max_end - 4.0, time.time() + 25.0)
            try:
                b, c = active_refine(best, stop, seed + 100200)
                if c < best_c - 1e-15:
                    best, best_c = b, float(c)
            except Exception:
                pass
        if max_end - time.time() > 20.0:
            stop = min(max_end - 3.0, time.time() + 18.0)
            try:
                b, c = random_direction_search(best, stop, seed + 100300)
                if c < best_c - 1e-15:
                    best, best_c = b, float(c)
            except Exception:
                pass

    b, c = try_symmetries(best, int(best.size))
    best = final_fix(b, 0.5 * float(best.size))
    if best.size < 10:
        best = project_box_sum(np.full(45, 0.5), 22.5)
    final_c = score_c(best)
    return best, float(final_c), int(best.size)
