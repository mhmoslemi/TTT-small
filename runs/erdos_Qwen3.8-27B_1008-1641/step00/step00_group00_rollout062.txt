    scaled = epsilon * corr

    return dx * float(np.max(corr))

def project_feasible(v, target_sum):
    """Project v onto {0<=x<=1, sum(x)=target_sum}."""
    lo = -max(float(np.max(v)), 0.0) - 1.0
    hi = 1.0 - min(float(np.min(v)), 1.0) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v + mid, 0.0, 1.0)))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)

        idx = int(np.argmax(margins))
        x[idx] = float(np.clip(x[idx] + diff, 0.0, 1.0))

The problem says: C₅ = max_k ∫ h(x)(1-h(x+k)) dx

Discretized: C₅ = max_k ∑_i h_i (1-h_{i+k}) dx

Now, `np.correlate(a, v, mode='full')[k] = ∑_n a[n+k] * conj(v[n])` (for valid n)

So `np.correlate(h, 1-h, mode='full')[k]`:
- Here a = h, v = 1-h
- = ∑_n h[n+k] * (1-h[n])

This is ∑_n h[n+k] * (1-h[n]), which is the integral at shift -k (or k, depending on convention).

The integral ∫ h(x)(1-h(x+k)) dx ≈ ∑_i h_i * (1-h_{i+k}) dx

Is this the same as ∑_n h[n+k] * (1-h[n])?

∑_i h_i * (1-h_{i+k}): here i ranges over valid indices where both i and i+k are valid.
Let j = i+k, so i = j-k:
= ∑_j h_{j-k} * (1-h_j)

And np.correlate gives: ∑_n h[n+k] * (1-h[n])

These are different! The first has h shifted forward (i+k), the second has h shifted forward by k.

Wait, let me re-examine. The problem defines:
∫ h(x)(1-h(x+k)) dx

At discretization level:
∑_i h(i·dx) · (1 - h((i+k)·dx)) · dx

= ∑_i h_i · (1 - h_{i+k}) · dx

Now, np.correlate(h, 1-h, 'full')[m] = ∑_i h[i+m] · (1-h[i]) for valid i

Let's substitute: let j = i+m, so i = j-m:
= ∑_j h[j] · (1-h[j-m]) for valid j
= ∑_j h[j] · (1-h[j-m])

Hmm, that's ∑ h[j](1-h[j-m]), not ∑ h[j](1-h[j+m]).

So np.correlate(h, 1-h, 'full') gives us ∑ h[j](1-h[j-m]) at output index m, which corresponds to the integral with k = -m.

But since we take the max over ALL k (both positive and negative), and the full correlation covers all shifts from -(n-1) to (n-1), the max should be the same!

max_m ∑ h[j](1-h[j-m]) = max_k ∑ h[j](1-h[j+k]) (just relabel m → -k)

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = int(np.argmax(corr)) - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        if 0 <= j + k < n:
            g[j] += 1.0 - h[j + k]
        if 0 <= j - k < n:
            g[j] -= h[j - k]
    return project_feasible(h - lr * dx * g, ts)


def sg_search(h0, dx, ts, n_iter, lr0):
    h = h0.copy()
    bh = h0.copy()
    bc = compute_c5(h0, dx)
    for i in range(n_iter):
        lr = lr0 / (1.0 + 0.005 * i)
        h = sg_step(h, lr, dx, ts)
        c = compute_c5(h, dx)
        if c < bc:
            bc = c
            bh = h.copy()
    return bh, bc


def bin_init(n, ts, rng):
    h = np.zeros(n)
    p = rng.permutation(n)
    ni = int(ts)
    if ni < n:
        h[p[:ni]] = 1.0
        h[p[ni]] = ts - ni
    return project_feasible(h, ts)


def try_slsqp(x0, n, dx, ts, B, maxit, ftol):
    try:
        r = minimize(raw_obj, x0, (n, dx, ts), method='SLSQP',
                     bounds=B, options={'maxiter': maxit, 'ftol': ftol})
        return project_feasible(r.x, ts)
    except Exception:
        return None


def try_nm(x0, n, dx, ts, maxit):
    try:
        r = minimize(raw_obj, x0, (n, dx, ts), method='Nelder-Mead',
                     options={'maxiter': maxit, 'xatol': 1e-14,
                              'fatol': 1e-14, 'adaptive': True})
        return project_feasible(r.x, ts)
    except Exception:
        return None


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n
    ts = n / 2.0
    h0 = project_feasible(h0, ts)
    bh = h0.copy()
    bc = compute_c5(h0, dx)
    B = [(0.0, 1.0)] * n

    # Phase 1: SLSQP from initial
    if budget_s - (time.time() - t0) > 10:
        ht = try_slsqp(bh, n, dx, ts, B, 1500, 1e-13)
        if ht is not None and compute_c5(ht, dx) < bc:
            bh, bc = ht, compute_c5(ht, dx)

    # Phase 2: Smooth SLSQP at several temperatures
    for ep in [5, 15, 50, 150, 400]:
        if budget_s - (time.time() - t0) < 10:
            break
        try:
            r = minimize(smooth_obj, bh, (n, dx, ts, ep), method='SLSQP',
                         bounds=B, options={'maxiter': 800, 'ftol': 1e-14})
            ht = project_feasible(r.x, ts)
            ct = compute_c5(ht, dx)
            if ct < bc:
                bh, bc = ht, ct
        except Exception:
            pass

    # Phase 3: Subgradient + SLSQP refinement
    if budget_s - (time.time() - t0) > 15:
        ht, ct = sg_search(bh, dx, ts, 15000, 0.08)
        if ct < bc:
            bh, bc = ht, ct
        ht2 = try_slsqp(bh, n, dx, ts, B, 600, 1e-14)
        if ht2 is not None:
            ct2 = compute_c5(ht2, dx)
            if ct2 < bc:
                bh, bc = ht2, ct2

    # Phase 4: Differential evolution (global search)
    nde = 0
    while budget_s - (time.time() - t0) > 120 and nde < 60:
        try:
            r = differential_evolution(raw_obj, B, args=(n, dx, ts),
                maxiter=500, popsize=25, seed=seed + 13 * nde,
                mutation=(0.4, 1.2), recombination=0.8, tol=1e-10)
            ht = project_feasible(r.x, ts)
            ct = compute_c5(ht, dx)
            if ct < bc:
                bh, bc = ht, ct
                ht2 = try_slsqp(bh, n, dx, ts, B, 400, 1e-14)
                if ht2 is not None:
                    ct2 = compute_c5(ht2, dx)
                    if ct2 < bc:
                        bh, bc = ht2, ct2
        except Exception:
            pass
        nde += 1

    # Phase 5: Binary-like starts + SLSQP
    nb = 0
    while budget_s - (time.time() - t0) > 40 and nb < 40:
        hs = bin_init(n, ts, rng)
        ht = try_slsqp(hs, n, dx, ts, B, 500, 1e-14)
        if ht is not None:
            ct = compute_c5(ht, dx)
            if ct < bc:
                bh, bc = ht, ct
        nb += 1

    # Phase 6: Annealed perturbations + SLSQP
    step = 0.08
    for npert in range(150):
        if budget_s - (time.time() - t0) < 8:
            break
        sig = step * (1.0 + 0.3 * np.sin(0.1 * npert))
        hs = project_feasible(bh + rng.normal(0, max(0.005, sig), n), ts)
        ht = try_slsqp(hs, n, dx, ts, B, 400, 1e-14)
        if ht is not None:
            ct = compute_c5(ht, dx)
            if ct < bc:
                bh, bc = ht, ct
                step *= 0.7
        if npert % 25 == 24:
            step = max(0.005, step * 0.85)

    # Phase 7: Smooth SLSQP again from best
    if budget_s - (time.time() - t0) > 10:
        for ep in [20, 100, 500]:
            try:
                r = minimize(smooth_obj, bh, (n, dx, ts, ep),
                             method='SLSQP', bounds=B,
                             options={'maxiter': 1000, 'ftol': 1e-15})
                ht = project_feasible(r.x, ts)
                ct = compute_c5(ht, dx)
                if ct < bc:
                    bh, bc = ht, ct
            except Exception:
                pass

    # Phase 8: Final Nelder-Mead polish
    if budget_s - (time.time() - t0) > 5:
        ht = try_nm(bh, n, dx, ts, 200000)
        if ht is not None:
            ct = compute_c5(ht, dx)
            if ct < bc:
                bh, bc = ht, ct

    bh = project_feasible(bh, ts)
    bc = compute_c5(bh, dx)
    return bh, bc, n
