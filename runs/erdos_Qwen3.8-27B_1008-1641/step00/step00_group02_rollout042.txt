import time

    global _n, _dx
    _n = n
    _dx = 2.0 / n

_n = 45
_dx = 2.0 / 45.0

def _c5_raw(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * _dx

def _softmax_obj(h, alpha, mu):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * _dx
    m = np.max(vals)
    sm = m + np.log(np.sum(np.exp(vals - m)) / len(vals)) / alpha
    penalty = mu * (np.sum(h) - _n * 0.5) ** 2
    return sm + penalty

def _slsqp_obj(x):
    return x[-1]

def _slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr * _dx

def _slsqp_eq(x):
    return np.sum(x[:-1]) - _n * 0.5

def _project_sum(h):
    """Project h to satisfy sum = n/2 while keeping [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    target = _n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Add constant, clip
    h = h + diff / _n
    h = np.clip(h, 0.0, 1.0)
    # Iterate a few times for better accuracy
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Adjust only free variables
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

    t0 = _c5_raw(h0) + 0.05

        h = _project_sum(result.x[:-1])
        return h, _c5_raw(h)
    return h0, _c5_raw(h0)
