    """Project g onto {h: 0<=h<=1, sum(h)=target_sum}."""
    n = len(g)
    lo, hi = g.min() - 1.0, g.max() + 1.0

def projected_descent(h0, n_iter=5000, step_init=0.01, seed=0):
    """Projected subgradient descent on C5."""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h += (target - h.sum()) / n
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = c5_value(h)
    rng = np.random.RandomState(seed)
    
    step = step_init
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        j = int(np.argmax(corr))
        c5 = float(corr[j] * dx)
        s = j - (n - 1)
        
        # Subgradient
        grad = np.zeros(n)
        for p in range(n):
            if 0 <= p - s < n:
                grad[p] += 1.0 - h[p - s]
            if 0 <= p + s < n:
                grad[p] -= h[p + s]
        grad *= dx
        
        # Update
        step *= 0.9995
        h_new = h - step * grad
        h_new = np.clip(h_new, 0, 1)

        diff = target - h_new.sum()
        if abs(diff) > 0.01:

            if diff > 0:
                mask = h_new < 1.0
            else:

            if mask.any():
                h_new[mask] += diff / mask.sum()
            h_new = np.clip(h_new, 0, 1)
        h = h_new
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        wk = p_k[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            # (1-h[p-s]) for p in [s, n)
            grad[s:] += wk * (1.0 - h[:n-s])
            # -h[p+s] for p in [0, n-s)
            grad[:n-s] += wk * (-h[s:])
        else:

            # (1-h[p+t]) for p in [0, n-t)  [since p-s = p+t, valid when 0<=p+t<n => p<n-t]
            grad[:n-t] += wk * (1.0 - h[t:])
            # -h[p-t] for p in [t, n)  [since p+s = p-t, valid when 0<=p-t<n => p>=t]
            grad[t:] += wk * (-h[:n-t])
    
    grad *= dx
    grad += 2 * penalty * (h.sum() - target)
    
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time


def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_obj_and_grad(h, beta, penalty, target, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    lse = max_c + np.log(exps.sum())
    val = (lse / beta) * dx + penalty * (h.sum() - target) ** 2
    p_k = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = p_k[k]
        if wk < 1e-8:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] += wk * (-h[s:])
        else:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n - t])
    grad *= dx
    grad += 2.0 * penalty * (h.sum() - target)
    return val, grad


def fix_feasibility(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(50):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h


def optimize_smooth(h0, budget_per_call=120):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = fix_feasibility(h0.copy(), target)
    t0 = time.time()
    best_h = h.copy()
    best_c5 = c5_value(h)
    betas = [15.0, 30.0, 60.0, 100.0]
    for beta in betas:
        penalty = 50.0
        h = fix_feasibility(h.copy(), target)
        for _ in range(3):
            time_left = budget_per_call - (time.time() - t0)
            if time_left < 5:
                break

            def obj(g):
                v, _ = smooth_obj_and_grad(g, beta, penalty, target, dx)
                return v

            def gfun(g):
                _, g2 = smooth_obj_and_grad(g, beta, penalty, target, dx)
                return g2

            res = minimize(obj, h, jac=gfun, method="L-BFGS-B",
                           bounds=[(0.0, 1.0)] * n,
                           options={"maxiter": 300, "gtol": 1e-8, "maxfun": 5000})
            h = fix_feasibility(res.x, target)
            cur = c5_value(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            penalty *= 2.0
    return best_h, best_c5


def optimize_subgrad(h0, n_iter=8000, seed=0):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = fix_feasibility(h0.copy(), target)
    best_h = h.copy()
    best_c5 = c5_value(h)
    step = 0.05
    rng = np.random.RandomState(seed)
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idxs = np.argsort(corr)[-3:]
        s_list = [k - (n - 1) for k in idxs]
        grad = np.zeros(n)
        for s in s_list:
            for p in range(n):
                if 0 <= p - s < n:
                    grad[p] += 1.0 - h[p - s]
                if 0 <= p + s < n:
                    grad[p] -= h[p + s]
        grad *= dx / len(s_list)
        step *= 0.9997
        h_new = h - step * grad
        h_new = fix_feasibility(h_new, target)
        h = h_new
        if it % 50 == 0:
            cur = c5_value(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
    cur = c5_value(h)
    if cur < best_c5:
        best_c5 = cur
        best_h = h.copy()
    return best_h, best_c5


def random_init(n, target, rng):
    h = rng.uniform(0.3, 0.7, n)
    return fix_feasibility(h, target)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    best_h = None
    best_c5 = 1.0

    # Use parent if available
    parent = None
    try:
        if 'initial_h_values' in dir() and initial_h_values is not None:
            parent = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if parent is not None and len(parent) > 0:
        parent = fix_feasibility(parent.copy(), len(parent) / 2.0)
        c = c5_value(parent)
        best_h = parent.copy()
        best_c5 = c
    else:
        n0 = 95
        best_h = random_init(n0, n0 / 2.0, rng)
        best_c5 = c5_value(best_h)

    n_parent = len(best_h)

    # Phase 1: Subgradient refinement of parent
    if elapsed() < budget_s - 60:
        h1, c1 = optimize_subgrad(best_h, n_iter=6000, seed=seed)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1

    # Phase 2: Smooth L-BFGS-B refinement
    if elapsed() < budget_s - 40:
        h2, c2 = optimize_smooth(best_h, budget_per_call=150)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 3: More subgradient passes
    if elapsed() < budget_s - 30:
        h3, c3 = optimize_subgrad(best_h, n_iter=10000, seed=seed + 1)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3

    # Phase 4: Try other n values with random starts
    for n_alt in [91, 99, 125, 151]:
        if elapsed() > budget_s - 60:
            break
        target_alt = n_alt / 2.0
        for restart in range(3):
            if elapsed() > budget_s - 60:
                break
            h_r = random_init(n_alt, target_alt, rng)
            # Quick subgradient
            h_s, c_s = optimize_subgrad(h_r, n_iter=3000, seed=seed + n_alt + restart)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s
            # Smooth refine if promising
            if c_s < best_c5 + 0.005 and elapsed() < budget_s - 40:

                if c_sm < best_c5:
                    best_c5 = c_sm

    # Phase 5: Final polish on best
    if elapsed() < budget_s - 15:
        h_f, c_f = optimize_smooth(best_h, budget_per_call=int(budget_s * 0.15))
        if c_f < best_c5:

        if elapsed() < budget_s - 10:

            if c_f2 < best_c5:
                best_c5 = c_f2
                best_h = h_f2

    # Ensure final feasibility
    n_f = len(best_h)
    best_h = fix_feasibility(best_h, n_f / 2.0)
    best_c5 = c5_value(best_h)

    return best_h, best_c5, n_f
