import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project(v, target):
    h = np.clip(v.astype(np.float64), 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
            if not m.any():
                break
            room = 1.0 - h[m]
            total_room = room.sum()
            if total_room <= d:
                h[m] = 1.0
            else:
                h[m] += d / m.sum()
        else:
            m = h > 1e-14
            if not m.any():
                break
            total_val = h[m].sum()
            if total_val <= -d:
                h[m] = 0.0
            else:
                h[m] += d / m.sum()
    d = target - h.sum()
    if abs(d) > 1e-12:
        j = int(np.argmax(h)) if d < 0 else int(np.argmin(h))
        h[j] += d
    return np.clip(h, 0.0, 1.0)

def make_obj(target):
    def obj(x):
        h = project(x, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr)
    return obj

def slsqp_step(v0, n, target, bounds, cons, maxiter=300, ftol=1e-12):

        return np.max(corr)

                 constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project(r.x, target)
    c5 = eval_c5(h_new)

                    best_c5 = c
