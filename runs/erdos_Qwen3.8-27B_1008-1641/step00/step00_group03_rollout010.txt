import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / len(h))


def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    s = alpha * corr * dx
    s -= s.max()
    return (s.max() + np.log(np.sum(np.exp(s)))) / alpha


def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    s = alpha * corr * dx
    s -= s.max()
    p = np.exp(s)
    p /= p.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-12:
            continue
        lo = max(0, m - (n - 1))
        hi = min(n - 1, m)
        for i in range(lo, hi + 1):
            j = i + n - 1 - m
            w = p[m] * dx
            grad[i] += w * (1.0 - h[j])
            grad[j] -= w * h[i]
    return grad


def cnst_fun(h):
    return h.sum() - len(h) / 2.0


def cnst_jac(h):
    return np.ones(len(h))


def lag_obj(h, lag_n):
    lag, n = lag_n
    return np.correlate(h, 1.0 - h, mode="full")[lag] * (2.0 / n)


def project_feasible(h, n):
    target = n / 2.0
    lo_t, hi_t = -3.0, 3.0

        mid = (lo_t + hi_t) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo_t = mid
        else:
            hi_t = mid
    out = np.clip(h - (lo_t + hi_t) / 2.0, 0.0, 1.0)

    if abs(diff) > 1e-12:

        if mask.sum() > 0:
            out[mask] += diff / mask.sum()
            out = np.clip(out, 0.0, 1.0)
    return out
