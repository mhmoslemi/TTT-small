    best_c5 = compute_c5(h, dx)
    while time.time() < deadline:
        improved = False
        for i in range(n):
            if h[i] < 0.5:
                continue
            for j in range(n):
                if i == j or h[j] >= 0.5:
                    continue
                h[i], h[j] = 0.0, 1.0
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, best_c5

def make_binary_init(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = np.unique((np.arange(n_ones) * n / n_ones + rng.rand(n_ones) * 0.3).astype(int) % n)[:n_ones]
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = len(h0)
    else:
        n = 47
        h0 = np.full(n, 0.5)

    target = n / 2.0
    dx = 2.0 / n
    h0 = fix_sum(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Powell from initial
    if deadline - time.time() > 80:
        hn = local_powell(best_h, target, dx, maxiter=10000)
        c5 = compute_c5(hn, dx)
        if c5 < best_c5:
            best_h, best_c5 = hn, c5

    # Phase 2: Coordinate descent
    if deadline - time.time() > 20:
        hn = coord_desc(best_h, target, dx, max_sweeps=120, grid=25, deadline=deadline - 10)
        c5 = compute_c5(hn, dx)
        if c5 < best_c5:
            best_h, best_c5 = hn, c5

    # Phase 3: Random restarts with local opt
    while deadline - time.time() > 70:
        scale = 0.03 + 0.12 * rng.rand()
        h_try = fix_sum(best_h + rng.randn(n) * scale, target)
        hn = local_powell(h_try, target, dx, maxiter=4000)
        c5 = compute_c5(hn, dx)
        if c5 < best_c5:
            best_h, best_c5 = hn, c5
            if deadline - time.time() > 30:
                hn2 = coord_desc(best_h, target, dx, max_sweeps=40, grid=20, deadline=deadline - 10)
                c52 = compute_c5(hn2, dx)
                if c52 < best_c5:
                    best_h, best_c5 = hn2, c52

    # Phase 4: Try different n values
    if deadline - time.time() > 50:
        for n_alt in [45, 49, 50]:
            if deadline - time.time() < 30:
                break
            t_alt = n_alt / 2.0
            dx_alt = 2.0 / n_alt
            h_alt = fix_sum(rng.rand(n_alt) * 0.8 + 0.1, t_alt)
            hn = local_powell(h_alt, t_alt, dx_alt, maxiter=5000)
            c5 = compute_c5(hn, dx_alt)
            if c5 < best_c5:
                best_h, best_c5, n = hn, c5, n_alt
                hn2 = coord_desc(best_h, best_c5 and t_alt, dx_alt, max_sweeps=30, grid=20, deadline=deadline - 10)
                c52 = compute_c5(hn2, dx_alt)
                if c52 < best_c5:
                    best_h, best_c5 = hn2, c52

    # Phase 5: Binary local search for even n
    if deadline - time.time() > 25:
        h_b = make_binary_init(48, rng)
        hb, c5_b = binary_ls(h_b, 48, deadline=deadline - 3)
        if c5_b < best_c5:
            best_h, best_c5, n = hb, c5_b, 48

    # Final refinement on best
    if deadline - time.time() > 15:
        n_f = len(best_h)
        t_f = n_f / 2.0
        dx_f = 2.0 / n_f
        hn = coord_desc(best_h, t_f, dx_f, max_sweeps=50, grid=30, deadline=deadline - 2)
        c5 = compute_c5(hn, dx_f)
        if c5 < best_c5:
            best_h, best_c5 = hn, c5

    return best_h, best_c5, n
