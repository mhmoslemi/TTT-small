def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-9)[0]
    
    grad = np.zeros(n)
    for k_idx in active:
        lag = k_idx - (n - 1)
        j1_start = max(0, -lag)
        j1_end = min(n, n - lag)  # exclusive
        if j1_start < j1_end:
            grad[j1_start:j1_end] += (1.0 - h[j1_start+lag:j1_end+lag])
        j2_start = max(0, lag)
        j2_end = min(n, n + lag)  # exclusive
        if j2_start < j2_end:
            grad[j2_start:j2_end] -= h[j2_start-lag:j2_end-lag]
    
    grad /= len(active)

def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    # Bisection for lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    val, grad = compute_c5_grad(h)
    h_new = h - lr * grad
    h_new = project_feasible(h_new, target_sum)
    return h_new, val

def coordinate_descent(h, target_sum, n_iter=1000):
    n = len(h)
    best_val = compute_c5(h)
    for _ in range(n_iter):
        improved = False
        for i in range(n):

            for direction in [1, -1]:

def smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = dx * np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp((corr - m) / eps)
    sw = np.sum(weights)
    if sw < 1e-30:
        return np.zeros(n)
    weights /= sw
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        lag = k_idx - (n - 1)
        js = max(0, -lag)
        je = min(n, n - lag)
        if js < je:
            grad[js:je] += w * dx * (1.0 - h[js + lag:je + lag])
        js2 = max(0, lag)
        je2 = min(n, n + lag)
        if js2 < je2:
            grad[js2:je2] -= w * dx * h[js2 - lag:je2 - lag]
    return grad
