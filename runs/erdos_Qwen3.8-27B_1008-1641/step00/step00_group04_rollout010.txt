import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def c5_evaluate(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    shifted = alpha * (vals - m)
    shifted = np.clip(shifted, -50, 50)
    log_sum = np.log(np.mean(np.exp(shifted)))
    return m + log_sum / alpha

def make_objective(n, alpha):
    dx = 2.0 / n
    target = n / 2.0
    def obj(h):
        c = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(c * dx, alpha)
    def constraint(h):
        return np.sum(h) - target
    return obj, constraint

def project_box_and_sum(h, n, max_iter=200):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_free = np.sum(free)
        if n_free == 0:
            if diff > 0:
                h[np.argmin(h)] += diff / max(1, n)
            else:
                h[np.argmax(h)] += diff / max(1, n)
            h = np.clip(h, 0, 1)
            continue
        h[free] += diff / n_free
        h = np.clip(h, 0, 1)
    return h

def interp_h(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # Use linear interpolation
    indices = np.searchsorted(x_old, x_new, side='right')
    indices = np.clip(indices, 1, n_old)
    frac = (x_new - x_old[indices - 1]) / (x_old[indices] - x_old[indices - 1]) if n_old > 1 else np.zeros(n_new)
    # Handle edge case
    frac = np.clip(frac, 0, 1)
    i0 = indices - 1
    i1 = np.minimum(indices, n_old - 1)
    h_new = h[i0] * (1 - frac) + h[i1] * frac
    return np.clip(h_new, 0, 1)

    best_c5 = c5_evaluate(best_h)

        dx = 2.0 / n
        obj, constraint = make_objective(n, alpha)
        bounds = [(0, 1)] * n
        cons = [{'type': 'eq', 'fun': constraint}]
        
        h_work = h_start.copy()
        h_best_local = h_work.copy()
        val_best = obj(h_work)
        
        try:
            res = minimize(obj, h_work, method='SLSQP', bounds=bounds,

            if res.x is not None:
                h_cand = np.clip(res.x, 0, 1)
                h_cand = project_box_and_sum(h_cand, n)
                v = obj(h_cand)
                if v < val_best:
                    val_best = v
                    h_best_local = h_cand.copy()
        except Exception:

import numpy as np
from scipy.optimize import minimize
import time

def c5_evaluate(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    shifted = np.clip(alpha * (vals - m), -50.0, 50.0)
    return m + np.log(np.mean(np.exp(shifted))) / alpha

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = np.sum(free)
        if nf == 0:
            idx = np.argmin(h) if diff > 0 else np.argmax(h)
            h[idx] += diff / n
            h = np.clip(h, 0, 1)
            continue
        h[free] += diff / nf
        h = np.clip(h, 0, 1)
    return h

def interp_to_n(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.clip(np.interp(x_new, x_old, h), 0.0, 1.0)

def cma_es_step(pop, sigma, mu, rng):
    n = len(pop[0])
    lam = len(pop)
    mean = np.mean(pop[:mu], axis=0)
    if sigma < 1e-10:
        sigma = 1e-10
    candidates = []
    for i in range(lam):
        z = rng.standard_normal(n)
        candidates.append(mean + sigma * z)
    return candidates, mean

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(42, 0.5)

    n_init = len(h_init)
    best_h = project_feasible(h_init, n_init)
    best_c5 = c5_evaluate(best_h)

    def slsqp_optimize(h0, n, alpha, maxiter, cons_tol):
        dx = 2.0 / n
        target = n / 2.0

        def obj(h):
            c = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(c * dx, alpha)

        def constr(h):
            return np.sum(h) - target

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': constr, 'jac': lambda h: np.ones(n)}]

        h_best_local = h0.copy()
        v_best = obj(h0)
        try:
            res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': cons_tol})
            if res.x is not None:
                hc = np.clip(res.x, 0.0, 1.0)
                hc = project_feasible(hc, n)
                vc = obj(hc)
                if vc < v_best:
                    v_best = vc
                    h_best_local = hc
        except Exception:
            pass
        return h_best_local

    def simple_es(h0, n, alpha, n_it, pop_size, sigma0, rng2):
        dx = 2.0 / n
        target = n / 2.0
        pop = [np.clip(h0 + sigma0 * rng2.standard_normal(n), 0, 1)]

        idx = np.argmin(vals)
