```python
def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def local_search_binary(h, n, dx, max_iter=100000, time_limit=None, start_time=None):
    """Local search on binary h with sum = n/2."""
    h = h.astype(int)
    best_c5 = compute_c5(h.astype(float), dx)
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    
    for it in range(max_iter):
        if time_limit and time.time() - start_time > time_limit:
            break
        # Pick random pair to flip
        if not ones or not zeros:
            break
        i = list(ones)[np.random.randint(len(ones))]
        j = list(zeros)[np.random.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        ones.discard(i)
        ones.add(j)
        zeros.discard(j)
        zeros.add(i)
        
        new_c5 = compute_c5(h.astype(float), dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # Revert
            h[i] = 1
            h[j] = 0
            ones.discard(j)
            ones.add(i)
            zeros.discard(i)
            zeros.add(j)
    
    return h.astype(float), best_c5
```

```python
import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_constraint(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}"""
    # Binary search on lambda
    lo, hi = -1.0, 1.0  # lambda range
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def subgradient_descent(h, dx, n, steps=1000, lr0=0.01, time_limit=None, start_time=None):
    """Minimize max_k f_k(h) subject to sum(h)=n/2, 0<=h<=1"""
    target_sum = n / 2.0
    h = h.copy().astype(float)
    h = project_to_constraint(h, target_sum)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    for step in range(steps):
        if time_limit and time.time() - start_time > time_limit:
            break
        # Compute subgradient
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        active = (corr >= max_val - 1e-6)
        n_active = np.sum(active)
        
        # Gradient of f_k w.r.t. h for active k
        # For simplicity, use numerical gradient of the max
        grad = np.zeros(n)
        eps = 1e-6
        for j in range(n):
            h_p = h.copy()
            h_p[j] = min(1.0, h_p[j] + eps)
            h_m = h.copy()
            h_m[j] = max(0.0, h_m[j] - eps)
            c_p = np.max(np.correlate(h_p, 1.0 - h_p, mode="full"))
            c_m = np.max(np.correlate(h_m, 1.0 - h_m, mode="full"))
            grad[j] = (c_p - c_m) / (h_p[j] - h_m[j] + 1e-12)
        
        # Adaptive learning rate
        lr = lr0 / (1.0 + step * 0.01)
        
        h = h - lr * dx * grad
        h = project_to_constraint(h, target_sum)
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

∂/∂h[j] of second term: -(h[j-k] + h[j+k]) if both j-k and j+k are valid, with appropriate boundary conditions

More precisely:
∂f_k/∂h[j] = [1 if 0≤j-k<n] - [h[j-k] if 0≤j-k<n] - [h[j+k] if 0≤j+k<n]

```python
def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def subgrad_of_max_corr(h, tol=1e-8):
    """Compute subgradient of max_k c[k] where c = correlate(h, 1-h, full)"""
    c = compute_corr(h)
    max_val = np.max(c)
    active = c >= max_val - tol
    n_active = np.sum(active)
    n = len(h)
    grad = np.zeros(n)
    
    for idx in np.where(active)[0]:
        k = idx - (n - 1)  # the lag
        # f_k(h) = sum_{n: valid} h[n+k] * (1-h[n])
        # partial wrt h[j]:
        # = (1 - h[j-k]) * I(0<=j-k<n) - h[j+k] * I(0<=j+k<n)
        # j-k >= 0 and j-k < n means k <= j < n+k, i.e., j >= max(0,k) and j < min(n, n+k)
        # j+k >= 0 and j+k < n means -k <= j < n-k
        js = np.arange(n)
        term1 = (js >= max(0, k)) & (js < min(n, n + k))  # 0<=j-k<n
        term2 = (js >= max(0, -k)) & (js < min(n, n - k))  # 0<=j+k<n

        t1 = np.zeros(n)
        if np.any(term1):

        # For term2: contribution is -h[j+k]
        t2 = np.zeros(n)
        if np.any(term2):
            t2[term2] = -h[js[term2] + k]
        
        grad += (t1 + t2) / n_active
    
    return grad

            best_h = h.copy()
```

```python
def compute_c5(h, dx):
```

```python
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    c = compute_corr(h) * dx
    cm = c.max()
    return np.log(np.sum(np.exp(M * (c - cm)))) / M + cm

def lse_grad(h, dx, M):
    n = len(h)
    c = compute_corr(h) * dx
    cm = c.max()
    w = np.exp(M * (c - cm))
    w /= w.sum()
    js = np.arange(n)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        t1 = (js >= max(0, k)) & (js < min(n, n + k))
        t2 = (js >= max(0, -k)) & (js < min(n, n - k))
        if np.any(t1):
            g[t1] += w[idx] * (1.0 - h[js[t1] - k]) * dx
        if np.any(t2):
            g[t2] += w[idx] * (-h[js[t2] + k]) * dx
    return g
```

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return float(np.max(compute_corr(h))) * dx

def project_sum(h, target):
    lo = float(h.min()) - 1.5
    hi = float(h.max()) + 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_grad(h, dx, M):
    n = len(h)
    c = compute_corr(h) * dx
    cm = c.max()
    w = np.exp(M * (c - cm))
    s = w.sum()
    if s < 1e-30:
        return np.zeros(n)
    w /= s
    js = np.arange(n)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        t1 = (js >= max(0, k)) & (js < min(n, n + k))
        t2 = (js >= max(0, -k)) & (js < min(n, n - k))
        if np.any(t1):
            g[t1] += wi * (1.0 - h[js[t1] - k]) * dx
        if np.any(t2):
            g[t2] += wi * (-h[js[t2] + k]) * dx
    return g

def pgdescent(h0, n, dx, steps, M0, tb, t0):
    h = project_sum(h0.copy(), n * 0.5)
    best = compute_c5(h, dx)
    bh = h.copy()
    for s in range(steps):
        if time.time() - t0 > tb:
            break
        M = M0 * (1.0 + s * 0.001)
        g = lse_grad(h, dx, M)
        lr = 0.5 / (M * np.sqrt(float(s + 1)))
        h = project_sum(h - lr * g, n * 0.5)
        c5 = compute_c5(h, dx)
        if c5 < best:
            best = c5
            bh = h.copy()
    return bh, best

def local_search(h0, n, dx, iters, tb, t0, sa=False):
    h = h0.copy().astype(float)
    best = compute_c5(h, dx)
    target = n * 0.5
    for it in range(iters):
        if time.time() - t0 > tb:
            break
        frac = 1.0 - it / float(iters)
        temp = 0.008 * frac if sa else 0.0
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        base_eps = 0.003 + 0.03 * frac
        eps = base_eps * (0.5 + np.random.random())
        for a, b in ((i, j), (j, i)):
            mv = min(eps, h[a], 1.0 - h[b])
            if mv < 1e-9:
                continue
            oa, ob = h[a], h[b]
            h[a] = oa - mv
            h[b] = ob + mv
            c5 = compute_c5(h, dx)
            if c5 <= best or (sa and np.random.random() < np.exp(-(c5 - best) / (temp + 1e-10))):
                best = min(best, c5)
            else:
                h[a], h[b] = oa, ob
        if it % 5000 == 0:
            h = project_sum(h, target)
    return h, best

def cocktail_h(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i // 2) % 2 == 0 else 0.0
    return project_sum(h, n * 0.5)

def smooth_h(n):
    h = np.zeros(n)
    half = n // 2
    for i in range(half):
        h[i] = 1.0
    return project_sum(h, n * 0.5)
```
