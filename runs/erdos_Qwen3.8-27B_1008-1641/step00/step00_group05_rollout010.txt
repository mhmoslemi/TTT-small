import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def projected_gradient_step(h, grad, step_size, n):
    """Take a projected gradient step onto {sum(h)=n/2, 0<=h<=1}"""
    h_new = h - step_size * grad
    # Project onto sum(h) = n/2, 0 <= h <= 1
    # Use bisection on lambda
    target = n / 2.0
    lo, hi = h_new.min() - 1, h_new.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h_new - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h_new - mid, 0, 1)
    return proj

def gradient_of_max_corr(h, n, top_k=5):
    """Compute gradient of sum of top-k correlation values (approximating max)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    # shifts range from -(n-1) to (n-1), indexed 0 to 2n-2
    # shift for index k is k - (n-1)
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    # Get top k shifts
    top_indices = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for k in top_indices:
        s = shifts[k]  # the shift
        # corr[k] = sum_i h[i+s] * (1-h[i]) for valid i
        # d(corr[k])/d(h[j]) = (1-h[j-s]) - h[j+s] (with bounds checking)
        for i in range(n):
            j = i  # we're computing d/d(h[j]) where j = i
            # Term 1: h[j] appears as h[i+s] when i = j-s
            #   contributes (1-h[i]) = (1-h[j-s]) if 0 <= j-s < n
            # Term 2: h[j] appears in (1-h[i]) when i = j
            #   contributes -h[i+s] = -h[j+s] if 0 <= j+s < n
            term1 = 0.0
            if 0 <= j - s < n:
                term1 = (1 - h[j - s])
            term2 = 0.0
            if 0 <= j + s < n:
                term2 = -h[j + s]
            grad[j] += (term1 + term2)
    
    grad *= dx
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_for_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        m = -s
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def smooth_max_func(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = vals.max()
    shifted = np.clip(alpha * (vals - m), -50, 50)
    exps = np.exp(shifted)
    S = exps.sum()
    obj = m + np.log(S) / alpha + lam * (h.sum() - n / 2.0) ** 2
    weights = exps / S
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = int(shifts[k])
        grad += w * dx * grad_for_shift(h, s)
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return obj, grad

def normalize_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        h[:] = 0.5
        return h
    h = (n / 2.0) * h / s
    return np.clip(h, 0.0, 1.0)

def optimize_from(h0, n, alpha, lam, max_iter):
    h = normalize_h(h0.copy(), n)
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_max_func, h, method='L-BFGS-B', jac=True,
                   args=(alpha, lam), bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-13, 'gtol': 1e-10,
                            'maxls': 40})
    return normalize_h(res.x, n)

def projected_refine(h, n, n_iter=2000, top_k=8):
    h = h.copy()
    dx = 2.0 / n
    shifts = np.arange(2 * n - 1) - (n - 1)
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 0.005
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        vals = corr * dx
        top_idx = np.argsort(vals)[-top_k:]
        w = vals[top_idx] - vals[top_idx].min() + 1e-8
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            s = int(shifts[idx])
            grad += wi * dx * grad_for_shift(h, s)
        # Projected step
        h_new = h - step * grad
        # Project onto sum(h)=n/2, 0<=h<=1 via bisection
        target = n / 2.0
        lo = h_new.min() - 1.0
        hi = h_new.max() + 1.0
        for _ in range(80):
            mid = (lo + hi) / 2.0
            proj = np.clip(h_new - mid, 0.0, 1.0)
            if proj.sum() > target:
                lo = mid
            else:
                hi = mid
        lam_mid = (lo + hi) / 2.0
        h_new = np.clip(h_new - lam_mid, 0.0, 1.0)
        val_new = compute_c5(h_new)
        if val_new < best_val * (1.0 - 1e-12):
            best_val = val_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.7
            if step < 1e-7:
                break
    return best_h, best_val
