            grad[i - s] += (1.0 - h[i])
            grad[i] -= h[i - s]

def project_feasible(h, N):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(x, N):
    return c5_val_grad(x, N)[0]

def jac_fn(x, N):
    return c5_val_grad(x, N)[1]

def constr_fun(x, N):
    return np.sum(x) - N / 2.0

def constr_jac(x, N):
    return np.ones(N)

def do_slsqp(h0, N, maxiter, ftol):
    try:
        res = minimize(
            obj_fn, h0, args=(N,),
            jac=jac_fn,
            method='SLSQP',
            bounds=Bounds(0, 1),
            constraints=[{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac, 'args': (N,)}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        return project_feasible(res.x, N), compute_c5(project_feasible(res.x, N))
    except Exception:
        return h0, compute_c5(h0)

def do_trust(h0, N, maxiter):
    try:
        lc = LinearConstraint(np.ones((1, N)), N/2.0, N/2.0)
        res = minimize(
            obj_fn, h0, args=(N,),
            jac=jac_fn,
            method='trust-constr',
            bounds=Bounds(0, 1),
            constraints=[lc],
            options={'maxiter': maxiter}
        )
        return project_feasible(res.x, N), compute_c5(project_feasible(res.x, N))
    except Exception:
        return h0, compute_c5(h0)

import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_val_grad(h, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (N - 1)
    grad = np.zeros(N)
    if shift >= 0:
        for i in range(N - shift):
            grad[i + shift] += (1.0 - h[i])
            grad[i] -= h[i + shift]
    else:
        s = -shift
        for i in range(s, N):
            grad[i - s] += (1.0 - h[i])
            grad[i] -= h[i - s]
    return float(np.max(corr) * dx), grad * dx

def c5_smooth_val_grad(h, N, tau=30.0):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    corr_shifted = corr - cmax
    exp_vals = np.exp(tau * corr_shifted)
    denom = np.sum(exp_vals)
    weights = exp_vals / denom
    val = (cmax + np.log(denom) / tau) * dx
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        shift = k - (N - 1)
        if shift >= 0:
            grad[0:N - shift] += -w * h[shift:N]
            grad[shift:N] += w * (1.0 - h[0:N - shift])
        else:
            s = -shift
            grad[0:N - s] += w * (1.0 - h[s:N])
            grad[s:N] += -w * h[0:N - s]
    return val, grad * dx

def project_feasible(h, N):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_hd(x, N):
    return c5_val_grad(x, N)[0]

def jac_hd(x, N):
    return c5_val_grad(x, N)[1]

def obj_sm(x, N):
    return c5_smooth_val_grad(x, N)[0]

def jac_sm(x, N):
    return c5_smooth_val_grad(x, N)[1]

def eq_fun(x, N):
    return np.sum(x) - N / 2.0

def eq_jac(x, N):
    return np.ones(N)

def do_slsqp(h0, N, maxiter, ftol, smooth=False):
    try:
        of = obj_sm if smooth else obj_hd
        jf = jac_sm if smooth else jac_hd
        res = minimize(
            of, h0, args=(N,),
            jac=jf,
            method='SLSQP',
            bounds=Bounds(0, 1),
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (N,)}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        return project_feasible(res.x, N), compute_c5(project_feasible(res.x, N))
    except Exception:
        return h0, compute_c5(h0)

def do_trust(h0, N, maxiter):
    try:
        lc = LinearConstraint(np.ones((1, N)), N / 2.0, N / 2.0)
        res = minimize(
            obj_sm, h0, args=(N,),
            jac=jac_sm,
            method='trust-constr',
            bounds=Bounds(0, 1),
            constraints=[lc],
            options={'maxiter': maxiter, 'gtol': 1e-8}
        )
        return project_feasible(res.x, N), compute_c5(project_feasible(res.x, N))
    except Exception:
        return h0, compute_c5(h0)
