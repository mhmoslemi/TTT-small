import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    cmax = np.max(corr)
    ea = T * (corr - cmax)
    ev = np.exp(ea)
    se = np.sum(ev)
    w = ev / se
    obj = (cmax + np.log(se) / T) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        s1, e1 = max(0, -off), min(n - 1, n - 1 - off)
