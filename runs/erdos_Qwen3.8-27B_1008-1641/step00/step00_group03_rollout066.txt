import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    ex = np.exp(shifted)
    s = ex.sum()
    obj = (m + np.log(s) / beta) * dx
    w = ex / s
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    for j in range(n):
        g = 0.0
        for ki in range(n2m1):
            wk = w[ki]
            if wk < 1e-15:
                continue
            k = ki - (n - 1)
            term = 0.0
            jm = j - k
            if 0 <= jm < n:
                term += 1.0 - h[jm]
            jp = j + k
            if 0 <= jp < n:
                term -= h[jp]
            g += wk * term
        grad[j] = g * dx
    return obj, grad
