def project_feasible(h, target_sum):
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        t = 0.5 * (lo + hi)
        if np.sum(np.clip(h - t, 0.0, 1.0)) > target_sum:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    result = np.clip(h - t, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            result[mask] = np.clip(result[mask] + diff / cnt, 0.0, 1.0)
    return result

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)

def neighborhood_search(h, n, rng, max_iter, time_limit):
