g_k(h) = dx * sum_i h[i] - dx * sum_i h[i] * h[i+lag]

The gradient of g_k with respect to h[j]:
∂g_k/∂h[j] = dx * (1 - h[j+lag]) - dx * h[j-lag] (where the second term only applies if j-lag is valid)

Wait, this is getting complicated with the lag indexing. Let me just use numerical gradients or let SLSQP/L-BFGS-B handle it.

For speed, let me use L-BFGS-B with a penalty for the sum constraint:
minimize f(h) + λ * (sum(h) - n/2)^2
subject to 0 ≤ h ≤ 1

This avoids the equality constraint and L-BFGS-B handles bounds natively.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(h)[::-1]
        else:
            order = np.argsort(h)
        for i in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
            else:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
    return h


def resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)


def smooth_max(c, T=40.0):
    m = np.max(c)
    return m + np.log(np.sum(np.exp((c - m) / T))) / T


def _try_slsoqp(h0, n, budget_check):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - n / 2.0}]
    h_c = project_feasible(h0, n)
    c_c = compute_c5(h_c)
    try:
        res = minimize(compute_c5, h_c, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-7})
        if res.x is not None:
            h_r = project_feasible(res.x, n)
            c_r = compute_c5(h_r)
            if c_r < c_c:
                return h_r, c_r
    except Exception:
        pass
    if budget_check() < 20:
        return h_c, c_c
    return h_c, c_c


def _try_lbfgs(h0, n, budget_check):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj_pen(hv, mu=2.0):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        base = smooth_max(corr * dx, T=50.0)
        pen = mu * (np.sum(hv) - n / 2.0) ** 2
        return base + pen

    h_c = project_feasible(h0, n)
    try:
        res = minimize(obj_pen, h_c, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-12})
        if res.x is not None:
            h_r = project_feasible(res.x, n)
            c_r = compute_c5(h_r)
            if c_r < compute_c5(h_c):
                h_c, _ = _try_slsoqp(h_r, n, budget_check)
    except Exception:
        pass
    return h_c, compute_c5(h_c)


def _binary_push(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        to_flip = np.where(h_bin == 1)[0]
        order = np.argsort(h[to_flip])[::-1][:n_ones - target]
        for i in to_flip[order]:
            h_bin[i] = 0.0
    elif n_ones < target:
        to_flip = np.where(h_bin == 0)[0]
        order = np.argsort(h[to_flip])[target - n_ones:]
        for i in to_flip[order]:
            h_bin[i] = 1.0
    c_bin = compute_c5(h_bin)
    return h_bin, c_bin


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.ones(70) / 2.0

    n_best = len(h_best)
    h_best = project_feasible(h_best, n_best)
    c_best = compute_c5(h_best)

    n_list = [n_best, 50, 60, 80, 100, 120]
    n_list = list(dict.fromkeys(n_list))

    for ni in n_list:
        if time_left() < 50:
            break
        h_start = resample(h_best, ni)
        h_start = project_feasible(h_start, ni)

        # L-BFGS-B pass
        if time_left() > 100:
            h_r, c_r = _try_lbfgs(h_start, ni, time_left)
            if c_r < c_best:
                c_best, h_best, n_best = c_r, h_r, ni

        # SLSQP polish
        if time_left() > 60:
            h_r, c_r = _try_slsoqp(h_best if len(h_best) == ni else resample(h_best, ni),
                                   ni, time_left)
            if c_r < c_best:
                c_best, h_best, n_best = c_r, h_r, ni

        # Binary push
        if time_left() > 30:
            h_b, c_b = _binary_push(h_best, ni)
            if c_b < c_best:
                c_best, h_best, n_best = c_b, h_b, ni
                h_r, c_r = _try_slsoqp(h_best, ni, time_left)
                if c_r < c_best:
                    c_best, h_best, n_best = c_r, h_r, ni

    # Multi-start from best solution at various scales
    if time_left() > 80:
        n_use = n_best
        n_trials = max(3, int(min(10, time_left() / 40)))
        for trial in range(n_trials):
            if time_left() < 35:
                break
            sigma = 0.02 + 0.015 * trial
            h_pert = h_best + np.random.normal(0, sigma, n_use)
            h_pert = project_feasible(h_pert, n_use)
            h_r, c_r = _try_slsoqp(h_pert, n_use, time_left)
            if c_r < c_best:
                c_best, h_best, n_best = c_r, h_r, n_use

    # Final polish at best resolution
    if time_left() > 20:
        h_r, c_r = _try_slsoqp(h_best, n_best, time_left)
        if c_r < c_best:
            c_best, h_best, n_best = c_r, h_r, n_best

    h_best = project_feasible(h_best, n_best)
    c_best = compute_c5(h_best)
    return h_best, c_best, len(h_best)
