import numpy as np
import time

def project_box_sum(target, S):
    n = len(target)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo = target.min() - 1.0
    hi = target.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(target - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(target - 0.5 * (lo + hi), 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_val(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def grad_smooth(corr, h, t, dx):
    n = len(h)
    g = 1.0 - h
    c = corr * dx
    mc = c.max()
    if mc < -50:
        return np.zeros(n)
    lw = (c - mc) / t
    lw -= lw.max()
    w = np.exp(lw)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-16:
            continue
        lag = idx - (n - 1)
        js = max(0, -lag)
        je = min(n, n - lag)
        if js < je:
            grad[js:je] += wi * g[js + lag:je + lag]
        js2 = max(0, lag)
        je2 = min(n, n + lag)
        if js2 < je2:
            grad[js2:je2] -= wi * h[js2 - lag:je2 - lag]
    return grad

def smooth_obj(corr, t, dx):
    c = corr * dx
    mc = c.max()
    if mc < -50:
        return mc * len(c)
    return t * (mc + np.log(np.sum(np.exp((c - mc) / t))))

def opt_gradient(h, S, dx, n, rng, t_schedule, max_iter, t0, budget_s):
    best_h = h.copy()
    best_c5 = c5_val(h, dx)
    h = h.copy()
    for t in t_schedule:
        lr = 0.08
        for it in range(max_iter):
            if time.time() - t0 > budget_s - 5:
                break
            corr = get_corr(h)
            c5_now = float(np.max(corr) * dx)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
            grad = grad_smooth(corr, h, t, dx)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            step = lr / (gnorm * n)
            old_s = smooth_obj(corr, t, dx)
            for _ in range(12):
                new_h = project_box_sum(h - step * grad, S)
                new_corr = get_corr(new_h)
                new_s = smooth_obj(new_corr, t, dx)
                if new_s < old_s - 1e-14:
                    h = new_h
                    break
                step *= 0.5
            else:
                lr *= 0.6
                if lr < 1e-13:
                    break
        if time.time() - t0 > budget_s - 5:
            break
    return best_h, best_c5

def make_binary_near(h, S, n):
    """Push towards binary, keeping sum = S"""
    thresh = 0.5
    h_bin = np.round(h)
    h_bin = np.clip(h_bin, 0.0, 1.0)
    diff = S - h_bin.sum()
    if abs(diff) < 1e-12:
        return h_bin
    if diff > 0:
        order = np.argsort(-h + rng_rand(n, 42))
    else:
        order = np.argsort(h + rng_rand(n, 42))
    i = 0
    while abs(diff) > 1e-12 and i < n:
        j = order[i % n]
        if diff > 0 and h_bin[j] < 1.0:
            h_bin[j] = min(1.0, h_bin[j] + diff)
            diff = S - h_bin.sum()
        elif diff < 0 and h_bin[j] > 0.0:
            h_bin[j] = max(0.0, h_bin[j] + diff)
            diff = S - h_bin.sum()
        i += 1
    return project_box_sum(h_bin, S)

    r = np.random.RandomState(seed)
    return r.rand(n) * 0.001
