            h_new = project_feasible(res.x, tgt)
            c5_new = compute_c5_val(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    best_h = project_feasible(best_h, n_best / 2.0)
    best_c5 = compute_c5_val(best_h)
    return best_h, best_c5, n_best
