import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def enforce(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-9:
        order = np.argsort(1.0 - h) if diff2 > 0 else np.argsort(h)
        for i in order:
            if diff2 > 0:
                add = min(diff2, 1.0 - h[i])
            else:
                add = max(diff2, -h[i])
            h[i] += add
            diff2 -= add
            if abs(diff2) < 1e-12:
                break
    return h

def binary_h(n, positions):
    h = np.zeros(n)
    h[positions[:n // 2]] = 1.0
    return h

    phi = (1 + np.sqrt(5)) / 2
    pos = np.zeros(k, dtype=int)

    return sorted(pos.tolist())
