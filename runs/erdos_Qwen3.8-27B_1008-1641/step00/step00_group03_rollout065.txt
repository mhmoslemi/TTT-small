    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0

    best_h, best_c = _optimize(h0, dx, tgt, n, seed, budget_s - 120)

    # Try n=50 if time permits
    if time.time() - t_start < budget_s - 60:
        n2 = 50
        dx2 = 2.0 / n2
        tgt2 = n2 / 2.0
        # Interpolate from best_h to n2 points
        x0 = np.linspace(0, 2, n, endpoint=False) + dx / 2
        x1 = np.linspace(0, 2, n2, endpoint=False) + dx2 / 2
        h_interp = np.interp(x1, x0, best_h)
        h_interp = _proj(h_interp, tgt2)
        # Also try uniform start
        h_uni = np.full(n2, 0.5)
        for h_start in [h_interp, h_uni]:
            if time.time() - t_start > budget_s - 40:
                break
            h_try, c_try = _optimize(h_start.copy(), dx2, tgt2, n2, seed + 1,
                                      min(80, budget_s - 40 - (time.time() - t_start)))
            if c_try < best_c:
                best_c, best_h = c_try, h_try

    # Final verification
    best_h = _proj(best_h, tgt)
    best_c = _c5(best_h, dx)

    return best_h, best_c, best_h.shape[0]
