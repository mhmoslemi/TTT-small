from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_obj(h, beta=40.0):

    max_c = np.max(corr)

    s = np.sum(x) - n / 2.0

    corr = np.correlate(x, 1.0 - x, mode="full")

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)


def _penalty_obj(x):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    c5 = float(np.max(corr) * 2.0 / n)
    s = np.sum(x) - n / 2.0
    return c5 + 50.0 * s * s


def _fix_sum(h, n):
    for _ in range(100):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h


def _to_binary(h, n):
    h_b = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_b))
    if n_ones > n // 2:
        idx = np.where(h_b == 1)[0]
        for i in np.random.choice(idx, n_ones - n // 2, replace=False):
            h_b[i] = 0
    elif n_ones < n // 2:
        idx = np.where(h_b == 0)[0]
        for i in np.random.choice(idx, n // 2 - n_ones, replace=False):
            h_b[i] = 1
    return h_b


def _sa_binary(h, n, t_limit):
    t_start = time.time()
    h = h.copy()
    best = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    T = 0.03
    T_min = 1e-9
    cool = 0.9998
    one_count = n // 2
    while True:
        if time.time() - t_start > t_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        if new_c <= cur_c or np.random.random() < np.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * cool, T_min)
    return best, best_c


def _deterministic_improve(h, n, t_limit):
    t_start = time.time()
    best = h.copy()
    best_c = compute_c5(h)
    improved = True
    while improved and time.time() - t_start < t_limit:
        improved = False
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        best_move_c = best_c - 1e-15
        best_pair = None
        for i in ones:
            if time.time() - t_start > t_limit:
                break
            for j in zeros:
                best[i] = 0.0
                best[j] = 1.0
                c = compute_c5(best)
                best[i] = 1.0
                best[j] = 0.0
                if c < best_move_c:
                    best_move_c = c
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            best[i] = 0.0
            best[j] = 1.0
            best_c = best_move_c
            improved = True
    return best, best_c
