    starts.append(make_binary_start(n_init, rng))
    starts.append(perturb(make_binary_start(n_init, rng), 0.1, rng))

    for hs in starts:
        if not time_ok(5):
            break
        h_t, c5_t = do_slsqp(hs, 300)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
        # Refine best with a small perturbation + polish
        if time_ok(5) and len(h_t) == n_init:
            h_p = perturb(best_h, 0.03, rng)
            h_t2, c5_t2 = do_slsqp(h_p, 200)
            if c5_t2 < best_c5:
                best_h, best_c5 = h_t2, c5_t2

    # Phase 3: LP linearization
    if time_ok(15):
        h_t, c5_t = do_lp_linearize(best_h, 25)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 4: Try larger n values
    for n_tgt in [48, 60, 100, 200]:
        if not time_ok(12):
            break
        h_b = interpolate_to(best_h, n_tgt)
        h_t, c5_t = do_slsqp(h_b, 300)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
        if time_ok(10):
            h_t2, c5_t2 = do_lp_linearize(h_t, 15)
            if c5_t2 < best_c5:
                best_h, best_c5 = h_t2, c5_t2

    # Phase 5: Final polish
    if time_ok(5):
        h_t, c5_t = do_slsqp(best_h, 300)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_final
