def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def make_binary(start_h, n):
    h = (start_h > 0.5).astype(float)
    h = project_feasible(h, n)
    # If still not exactly binary, force it
    s = h.sum()
    if abs(s - n / 2.0) > 1e-10:
        h2 = (start_h > 0.5).astype(float)
        need = n // 2
        cs = h2.sum()
        if cs > need:
            idx = np.where(h2 == 1)[0]
            order = np.argsort(start_h[idx])
            for k in range(int(cs - need)):
                h2[idx[order[k]]] = 0
        else:
            idx = np.where(h2 == 0)[0]
            order = np.argsort(-start_h[idx])
            for k in range(int(need - cs)):
                h2[idx[order[k]]] = 1
        h = h2
    return h


def one_swap_search(h, n, max_sweeps=500):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    for sweep in range(max_sweeps):
        improved = False
        ones = np.where(best >= 0.999)[0]
        zeros = np.where(best <= 0.001)[0]
        for i in ones:
            for j in zeros:
                best[i] = 0.0
                best[j] = 1.0
                c5_new = compute_c5(best, n)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                else:
                    best[i] = 1.0
                    best[j] = 0.0
        if not improved:
            break
    return best, best_c5


def smooth_obj_factory(n, beta):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        if beta < 1e6:
            return dx * (m + np.log(np.mean(np.exp(beta * (corr - m)))) / beta)
        return dx * m
    return obj


def two_swap_search(h, n, max_iters=200, rng=None):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    ones = np.where(best >= 0.999)[0]
    zeros = np.where(best <= 0.001)[0]
    for it in range(max_iters):
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = rng.choice(ones)
        j1 = rng.choice(zeros)
        i2 = rng.choice(ones)
        j2 = rng.choice(zeros)
        if i2 == i1 or j2 == j1:
            continue
        best[i1] = 0.0; best[j1] = 1.0
        best[i2] = 0.0; best[j2] = 1.0
        c5_new = compute_c5(best, n)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            ones = np.where(best >= 0.999)[0]
            zeros = np.where(best <= 0.001)[0]
        else:
            best[i1] = 1.0; best[j1] = 0.0
            best[i2] = 1.0; best[j2] = 0.0
    return best, best_c5
