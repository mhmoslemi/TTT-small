            h_sa = np.zeros(n_sa)
            idx_sa = np.argsort(best_h[:n_sa])[::-1] if n_sa <= len(best_h) else np.random.permutation(n_sa)
            h_sa[idx_sa[:int(ts)]] = 1.0
        c5_sa = compute_c5(h_sa)
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa
        temp = 0.05
        for it in range(5000):
            if not remaining():
                break
            i1 = np.random.randint(n_sa)
            i2 = np.random.randint(n_sa)
            if h_sa[i1] != h_sa[i2]:
                h_sa[i1], h_sa[i2] = h_sa[i2], h_sa[i1]
                c5_new = compute_c5(h_sa)
                delta = c5_new - c5_sa
                if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-6)):
                    c5_sa = c5_new
                    if c5_sa < best_c5:
                        best_c5 = c5_sa
                        best_h = h_sa.copy()
                else:
                    h_sa[i1], h_sa[i2] = h_sa[i2], h_sa[i1]
            temp *= 0.998

    # Final: restore global state
    _global_n = len(best_h)
    _global_target = len(best_h) / 2.0

    return best_h, best_c5, len(best_h)
