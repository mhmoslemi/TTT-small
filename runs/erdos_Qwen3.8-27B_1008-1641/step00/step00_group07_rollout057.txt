def _gradient_smoothed_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full") * dx  # shape (2n-1,)
    
    # Softmax weights
    m = np.max(corr)
    log_weights = np.clip(beta * (corr - m), -500, 500)
    p = np.exp(log_weights)
    p = p / np.sum(p)
    
    # Compute gradient
    grad = np.zeros(n)
    for mk in range(2*n - 1):
        if p[mk] < 1e-10:
            continue
        k = mk - (n - 1)
        # For this lag k, the valid range of i is [max(0,-k), min(n, n-k))
        i_start = max(0, -k)
        i_end = min(n, n - k)
        for i in range(i_start, i_end):
            # Contribution: dx * p[mk] * (1-h[i+k]) to grad[i]
            grad[i] += p[mk] * dx * (1.0 - h[i + k])
            # Contribution: -dx * p[mk] * h[i] to grad[i+k]
            grad[i + k] -= p[mk] * dx * h[i]
    
    return grad

import numpy as np
import time
from scipy.optimize import minimize

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if s > 1e-15:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.where(h < 0.999)[0] if diff > 0 else np.where(h > 0.001)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        i = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[i] = float(np.clip(h[i] + diff, 0.0, 1.0))
    return h

def _smax(v, b):
    m = np.max(v)
    return float(m + np.log(np.sum(np.exp(np.clip(b * (v - m), -500, 500)))) / b)

def _grad_smmax(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    pw = np.exp(np.clip(beta * (corr - m), -500, 500))
    pw /= pw.sum()
    g = np.zeros(n)
    for mk in range(2 * n - 1):
        w = pw[mk]
        if w < 1e-12:
            continue
        k = mk - (n - 1)
        lo = max(0, -k)
        hi = min(n, n - k)
        g[lo:hi] += w * dx * (1.0 - h[lo + k:hi + k])
        g[lo + k:hi + k] -= w * dx * h[lo:hi]
    return g

def _eqcon(x):
    return np.sum(x) - len(x) * 0.5

def _obj_pen(x, args):
    beta, lam = args
    n = len(x)
    dx = 2.0 / n
    xc = np.clip(x, 0.0, 1.0)
    v = _smax(np.correlate(xc, 1.0 - xc, mode="full") * dx, beta)
    v += lam * (np.sum(xc) - n * 0.5) ** 2
    return v

def _grad_pen(x, args):
    beta, lam = args
    n = len(x)
    xc = np.clip(x, 0.0, 1.0)
    g = _grad_smmax(xc, beta)
    g += 2.0 * lam * (np.sum(xc) - n * 0.5)
    return g

def _slsqp_obj(x, args):
    beta = args[0]
    n = len(x)
    dx = 2.0 / n
    xc = np.clip(x, 0.0, 1.0)
    return _smax(np.correlate(xc, 1.0 - xc, mode="full") * dx, beta)

def _do_lbfgsb(h, n, beta, lam, maxiter, gtol):
    try:
        res = minimize(_obj_pen, h, jac=_grad_pen, args=(beta, lam),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-18,
                                'maxfun': maxiter * 50})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _do_slsqp(h, n, beta, maxiter):
