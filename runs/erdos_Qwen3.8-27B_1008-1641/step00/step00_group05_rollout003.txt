def milp_find(n, maxv, timeout):

    try:
        import cvxpy as cp
    except Exception:
        return None
    M = n // 2
    y = cp.Variable(n, boolean=True)
    P = n * (n - 1) // 2
    r = cp.Variable(P, nonneg=True)
    d = cp.Variable(n - 1, nonneg=True)
    t = cp.Variable(nonneg=True)
    cons = [y.sum() == M, r <= 1.0, t <= float(maxv)]
    lag = [[] for _ in range(n)]
    p = 0
    for i in range(n):
        for j in range(i + 1, n):
            lag[j - i].append(p)
            cons += [r[p] >= y[i] - y[j], r[p] >= y[j] - y[i]]
            p += 1

import time, math
import numpy as np
import scipy.optimize as so

_NSLA=70; _MSLA=35.0; _BETA=100.0

def project(x, target):
    y = np.asarray(x, dtype=float)
    if y.size == 0:
        return y
    y = np.clip(y, 0.0, 1.0)
    for _ in range(30):
        d = target - y.sum()
        if abs(d) <= 1e-15:
            break
        f = (y > 1e-12) & (y < 1.0 - 1e-12)
        if f.any():
            y[f] += d / float(f.sum())
        else:
            y += d / y.size
        y = np.clip(y, 0.0, 1.0)
    d = target - y.sum()
    if abs(d) > 1e-16:
        i = int(np.argmax(np.minimum(y, 1.0 - y)))
        y[i] = float(np.clip(y[i] + d, 0.0, 1.0))
    return y

def evalb(h):
    n = h.size
    return float(2.0 * np.max(np.correlate(h, 1.0 - h, mode='full')) / n)

def finish(h, n):
    h = project(h, n / 2.0)
    return h, float(evalb(h)), int(n)

def cc(h):
    n = h.size
    oh = 1.0 - h
    c = np.empty(2 * n - 1)
    m = n - 1
    c[m] = float(np.dot(h, oh))
    for s in range(1, n):
        l = n - s
        c[m + s] = float(np.dot(h[:l], oh[s:]))
        c[m - s] = float(np.dot(h[s:], oh[:l]))
    return c

def smooth(x, n, beta):
    c = cc(x)
    L = 2 * n - 1
    m = n - 1
    maxc = float(c.max())
    if beta <= 1e-8:
        p = np.full(L, 1.0 / L)
        F = maxc
    else:
        e = np.exp(beta * (c - maxc))
        S = float(e.sum())
        if S <= 0.0 or not math.isfinite(S):
            p = np.full(L, 1.0 / L)
            F = maxc
        else:
            p = e / S
            F = maxc + math.log(S) / beta
    g = np.zeros(n)
    g += p[m] * (1.0 - 2.0 * x)
    for s in range(1, n):
        l = n - s
        wp = p[m + s]
        wn = p[m - s]
        if wp:
            g[:l] += wp * (1.0 - x[s:])
            g[l:] -= wp * x[:l]
        if wn:
            g[s:] += wn * (1.0 - x[:l])
            g[:l] -= wn * x[s:]
    return float(F), g

def _obj(x):
    return smooth(x, _NSLA, _BETA)[0]

def _jac(x):
    return smooth(x, _NSLA, _BETA)[1]

def _eq(x):
    return float(x.sum() - _MSLA)

def _eqg(x):
    return np.ones(_NSLA)

def slsqp_polish(h, n, timeout, betas, maxiter):
    global _NSLA, _MSLA, _BETA
    M = n / 2.0
    h = project(h, M)
    best = h.copy()
    bestv = evalb(h)
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eqg}]
    for beta in betas:
        if time.time() - t0 > timeout:
            break
        _NSLA = n
        _MSLA = M
        _BETA = float(beta)
        try:
            res = so.minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        except Exception:
            continue
        if res is not None and getattr(res, 'x', None) is not None:
            x = project(res.x, M)
            v = evalb(x)
            if v < bestv:
                bestv = v
                best = x.copy()
            h = x.copy() if v < bestv + 1e-4 else best.copy()
    return best, bestv

def minc(y, comp=None):
    if comp is None:
        comp = 1 - y
    return int(np.max(np.correlate(y, comp, mode='full')))

def binarize(h, rng):
    n = h.size
    M = n // 2
    y = (h > 0.5).astype(int)
    c = int(y.sum())
    if c > M:
        idx = np.flatnonzero(y)
        if idx.size:
            y[idx[np.argsort(h[idx])[:c - M]]] = 0
    elif c < M:
        idx = np.flatnonzero(1 - y)
        if idx.size:
            y[idx[np.argsort(-h[idx])[:M - c]]] = 1
    return y

def randbin(n, rng):
    M = n // 2
    y = (rng.random(n) < 0.5).astype(int)
    c = int(y.sum())
    if c > M:
        idx = np.flatnonzero(y)
        y[idx[rng.integers(0, idx.size, size=c - M)]] = 0
    elif c < M:
        idx = np.flatnonzero(1 - y)
        y[idx[rng.integers(0, idx.size, size=M - c)]] = 1
    return y

def randswap(y, rng):
    n = y.size
    while True:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if y[i] == 1 and y[j] == 0:
            y[i] = 0
            y[j] = 1
            break

def steepest(y, timeout, target):
    y = y.astype(int).copy()
    comp = 1 - y
    cur = minc(y, comp)
    best = y.copy()
    bestv = cur
    t0 = time.time()
    while time.time() - t0 < timeout and cur > target:
        ones = np.flatnonzero(y)
        zeros = np.flatnonzero(comp)
        bp = None
        bv = cur
        for i in ones:
            y[i] = 0
            comp[i] = 1
            for j in zeros:
                y[j] = 1
                comp[j] = 0
                v = minc(y, comp)
                if v < bv:
                    bv = v
                    bp = (i, j)
                y[j] = 0
                comp[j] = 1
            y[i] = 1
            comp[i] = 0
            if bv <= target:
                break
        if bp is None:
            break
        i, j = bp
        y[i] = 0
        comp[i] = 1
        y[j] = 1
        comp[j] = 0
        cur = bv
        if cur < bestv:
            bestv = cur
            best = y.copy()
    return best, bestv

def sa(y, timeout, target, rng):
    y = y.astype(int).copy()
    comp = 1 - y
    n = y.size
    cur = minc(y, comp)
    best = y.copy()
    bestv = cur
    t0 = time.time()
    while time.time() - t0 < timeout and bestv > target:
        frac = max(0.0, 1.0 - (time.time() - t0) / max(timeout, 1e-9))
        T = 0.05 + 1.95 * frac
        while True:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if y[i] == 1 and y[j] == 0:
                break
        y[i] = 0
        comp[i] = 1
        y[j] = 1
        comp[j] = 0
        v = minc(y, comp)
        if v <= cur or rng.random() < math.exp(-(v - cur) / max(T, 1e-6)):
            cur = v
            if v < bestv:
                bestv = v
                best = y.copy()
        else:
            y[i] = 1
            comp[i] = 0
            y[j] = 0
            comp[j] = 1
    return best, bestv

def binary_search(start, timeout, target, rng):
    n = start.size
    t0 = time.time()
    best, bestv = steepest(start.copy(), min(30.0, max(1.0, timeout * 0.15)), target)
    if bestv <= target:
        return best, bestv
    while time.time() - t0 < timeout - 5.0 and bestv > target:
        rem = timeout - (time.time() - t0)
        if rem < 10.0:
            break
        if rng.random() < 0.6:
            cand = best.copy()
            k = int(rng.integers(3, max(4, min(10, n // 2) + 1)))
            for _ in range(k):
                randswap(cand, rng)
        else:
            cand = randbin(n, rng)
        cand, v = sa(cand, min(20.0, rem * 0.4), target, rng)
        if v < bestv:
            bestv = v
            best = cand.copy()
            if bestv <= target:
                return best, bestv
        cand, v = steepest(cand, min(30.0, rem * 0.5), target)
        if v < bestv:
            bestv = v
            best = cand.copy()
            if bestv <= target:
                return best, bestv
    return best, bestv

def milp_find(n, maxv, timeout):
    if n % 2 or maxv < 1:
        return None
    try:
        import cvxpy as cp
    except Exception:
        return None
    M = n // 2
    y = cp.Variable(n, boolean=True)
    P = n * (n - 1) // 2
    r = cp.Variable(P, nonneg=True)
    d = cp.Variable(n - 1, nonneg=True)
    t = cp.Variable(nonneg=True)
    cons = [y.sum() == M, r <= 1.0, t <= float(maxv)]
    lag = [[] for _ in range(n)]
    p = 0
    for i in range(n):
        for j in range(i + 1, n):
            lag[j - i].append(p)
            cons += [r[p] >= y[i] - y[j], r[p] >= y[j] - y[i]]
            p += 1
    for s in range(1, n):
        l = n - s
        diff = cp.sum(y[:l]) - cp.sum(y[s:])
        cons += [d[s - 1] >= diff, d[s - 1] >= -diff,
                 2.0 * t >= cp.sum([r[p] for p in lag[s]]) + d[s - 1]]
    try:
        cp.Problem(cp.Minimize(0.0), cons).solve(solver=cp.CBC, time_limit=max(1, int(timeout)), verbose=False)
    except Exception:
        return None
    if y.value is None:
        return None
    out = np.round(y.value).astype(int)
    c = int(out.sum())
    if c != M:
        if c > M:
            idx = np.flatnonzero(out)
            order = idx[np.argsort(y.value[idx])]
            out[order[:c - M]] = 0
        else:
            idx = np.flatnonzero(1 - out)
            order = idx[np.argsort(-y.value[idx])]
            out[order[:M - c]] = 1
    if int(out.sum()) != M:
        return None
    if minc(out) <= maxv:
        return out
    return None
