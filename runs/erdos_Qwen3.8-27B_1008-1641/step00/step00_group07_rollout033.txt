def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def project_feasible(h, target_sum):
    """Project onto {0<=h<=1, sum=target_sum} via bisection."""
    h = np.asarray(h, dtype=float)
    lo, hi = -1.5, 1.5

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:

    # Softmax for numerical stability
    scores -= np.max(scores)
    weights = np.exp(scores)
    weights /= np.sum(weights)
    c5_smooth = np.mean(corr) * dx + np.log(np.sum(np.exp(T * (corr * dx - np.min(corr*dx)))) ) / T

import numpy as np
import time

def compute_c5(h):
    """Compute C5 for step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_at_lag(h, k, n):
    """Gradient of correlation at lag k w.r.t. h."""
    m = np.arange(n)
    grad = np.zeros(n)
    idx1 = m - k
    mask1 = (idx1 >= 0) & (idx1 < n)
    grad[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = m + k
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] -= h[idx2[mask2]]
    return grad

def smooth_objective_and_grad(h, T=30.0):
    """Smooth approximation of C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = T * corr * dx
    max_s = np.max(scores)
    exps = np.exp(scores - max_s)
    weights = exps / exps.sum()
    c5_smooth = (max_s + np.log(exps.sum())) / T
    # Gradient: weighted sum

    active = weights > 1e-6
    for idx in np.where(active)[0]:
        k = int(idx) - (n - 1)
        grad += weights[idx] * grad_at_lag(h, k, n)
    grad *= dx
    return c5_smooth, grad

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_at_lag(h, k, n):
    m = np.arange(n)
    grad = np.zeros(n)
    idx1 = m - k
    mask1 = (idx1 >= 0) & (idx1 < n)
    if mask1.any():
        grad[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = m + k
    mask2 = (idx2 >= 0) & (idx2 < n)
    if mask2.any():
        grad[mask2] -= h[idx2[mask2]]
    return grad

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

            continue
        k = idx - (n - 1)
        grad += weights[idx] * grad_at_lag(h, k, n)
    grad *= dx
    return c5_smooth, grad
