    shifted = T * (corr - np.max(corr))
    safe_max = np.max(shifted)
    log_sum = safe_max + np.log(np.sum(np.exp(shifted - safe_max)))

import numpy as np
import time

def project_feasible(v, n, target_sum):
    lo = -v.max() - 0.5
    hi = 1.0 - v.min() + 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def c5_of(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_grad_maxlag(h, n, max_idx):
    grad = np.zeros(n)
    k = max_idx - (n - 1)
    if k >= 0:
        for i in range(n - k):
            grad[i + k] += (1.0 - h[i])
            grad[i] -= h[i + k]
    else:
        j = -k
        for i in range(n - j):
            grad[i] += (1.0 - h[i + j])
            grad[i + j] -= h[i]
    return grad

def local_opt(h, n, dx, target_sum, deadline, T_init=2.0, T_final=200.0):
    best_h = h.copy()
    best_val = c5_of(h, dx)
    momentum = np.zeros(n)
    for it in range(50000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_val = np.max(corr)
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_h = h.copy()
        frac = min(1.0, it / 2000.0)
        T = T_init * (T_final / T_init) ** frac
        shifted = T * (corr - cur_val)
        weights = np.exp(shifted - np.max(shifted))
        wsum = np.sum(weights)
        if wsum < 1e-30:
            break
        weights /= wsum
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if weights[m] < 1e-8:
                continue
            k = m - (n - 1)
            if k >= 0:
                for i in range(n - k):
                    grad[i + k] += weights[m] * (1.0 - h[i])
                    grad[i] -= weights[m] * h[i + k]
            else:
                j = -k
                for i in range(n - j):
                    grad[i] += weights[m] * (1.0 - h[i + j])
                    grad[i + j] -= weights[m] * h[i]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        momentum = 0.7 * momentum + 0.3 * grad / gnorm
        base_step = 0.3 / (1.0 + 0.005 * it)
        step = base_step
        for _ in range(15):

            new_val = c5_of(h_new, dx)
            if new_val < cur_val + 1e-12:
                h = h_new
                break
            step *= 0.5
    return best_h, best_val

    best_val = c5_of(h, dx)
    for r in range(n_rounds):

            if val < best_val - 1e-9:

                best_val = val
                improved = True
