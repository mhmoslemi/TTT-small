import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_feasible(v, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) * 0.5
    result = np.clip(v - mid, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(result < 1.0 - 1e-10)[0]
        else:
            idx = np.where(result > 1e-10)[0]
        if len(idx) > 0:
            result = result.copy()
            result[idx[0]] += diff
    return result

def solve_1d_minmax(base, a, lo, hi):
    if lo >= hi:
        x = min(max(lo, 0.0), 1.0)
        return x, np.max(base + a * x)
    candidates = [lo, hi]
    n_lines = len(base)
    for i in range(n_lines):
        for j in range(i + 1, n_lines):
            da = a[i] - a[j]
            if abs(da) > 1e-14:
                xc = (base[j] - base[i]) / da
                if lo - 1e-9 <= xc <= hi + 1e-9:
                    candidates.append(np.clip(xc, lo, hi))
    best_x = lo
    best_v = np.inf
    for x in candidates:
        v = np.max(base + a * x)
        if v < best_v:
            best_v = v
            best_x = x
    return best_x, best_v

def coordinate_descent(h, target_sum, dx, max_sweeps=2000, tol=1e-14):
    n = len(h)
    n_corr = 2 * n - 1
    h = h.copy()
    prev_c5 = compute_c5(h, dx)
    
    for sweep in range(max_sweeps):
        changed = False
        for i in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            tau_offsets = np.arange(n_corr) - (n - 1)
            a = np.zeros(n_corr)
            for k in range(n_corr):
                tau = tau_offsets[k]
                ak = 0.0
                if 0 <= i + tau < n:
                    ak += 1.0 - h[i + tau]
                if 0 <= i - tau < n:
                    ak -= h[i - tau]
                a[k] = ak
            base = corr - a * h[i]
            lo = max(0.0, target_sum - (n - 1.0))
            hi = min(1.0, target_sum)
            x_opt, _ = solve_1d_minmax(base, a, lo, hi)
            if abs(x_opt - h[i]) > 1e-14:
                h[i] = x_opt
                changed = True
        new_c5 = compute_c5(h, dx)
        if abs(new_c5 - prev_c5) < tol:
            break
        if not changed:
            break
        prev_c5 = new_c5
    return h, compute_c5(h, dx)

    best_c5 = compute_c5(parent, dx)
    best_h = parent.copy()

        return budget_s - (time.time() - t0)

    h_cd, c5_cd = coordinate_descent(parent, target_sum, dx, max_sweeps=5000)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    n_restarts = 500  # Will be limited by time
    for r in range(n_restarts):

        perturbed = best_h + rng.normal(0, scale, n)
        perturbed = project_feasible(perturbed, target_sum)

        if c5_try < best_c5 - 1e-12:

            best_h = h_try.copy()

    if time_left() > 30:

        for beta in [100, 500, 1000, 2000]:
            if time_left() < 10:
                break
            def obj_slsqp(x, _beta=beta):
                h_vec = np.clip(x, 0.0, 1.0)
                corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
                mc = np.max(corr)
                shifted = _beta * (corr - mc)
                return dx * (mc + np.log(np.sum(np.exp(shifted))) / _beta)
            
            def jac_slsqp(x, _beta=beta):
                h_vec = np.clip(x, 0.0, 1.0)
                corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
                mc = np.max(corr)
                shifted = _beta * (corr - mc)
                exp_s = np.exp(shifted)
                weights = exp_s / np.sum(exp_s)
                tau_offsets = np.arange(n_corr) - (n - 1)
                g = np.zeros(n)
                for k in range(n_corr):

                    for j in range(n):
                        val = 0.0
                        if 0 <= j + tau < n:
                            val += 1.0 - h_vec[j + tau]
                        if 0 <= j - tau < n:
                            val -= h_vec[j - tau]
                        g[j] += weights[k] * val
                return dx * g
            
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                     'jac': lambda x: np.ones(n)}]
            bnds = [(0, 1)] * n
            
            res = minimize(obj_slsqp, best_h.copy(), method='SLSQP',
                          jac=jac_slsqp, bounds=bnds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14})
            h_slsqp = project_feasible(res.x, target_sum)
            c5_slsqp = compute_c5(h_slsqp, dx)
            if c5_slsqp < best_c5:
                best_c5 = c5_slsqp
                best_h = h_slsqp.copy()

        if c5_final < best_c5:

    # Ensure feasibility
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    
    return best_h, best_c5, n

def projected_subgradient(h, target_sum, dx, n_steps, alpha0):

    n_corr = 2 * n - 1
    tau_offsets = np.arange(n_corr) - (n - 1)

    alpha = alpha0
    
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        # Active set: k's within small epsilon of max
        active = np.where(corr > max_c - 1e-6)[0]
        # Subgradient: average of gradients over active set
        g = np.zeros(n)
        for k in active:
            tau = tau_offsets[k]
            for j in range(n):
                if 0 <= j + tau < n:
                    g[j] += 1.0 - h[j + tau]
                if 0 <= j - tau < n:
                    g[j] -= h[j - tau]
        g /= len(active)

            h_new = project_feasible(h - alpha * g, target_sum)
            f_new = compute_c5(h_new, dx)
            if f_new < f_old - 1e-10:
                break
            alpha *= 0.5
        else:
            break

        # Decayed learning rate
        alpha *= (1.0 - step / (n_steps * 2.0))
