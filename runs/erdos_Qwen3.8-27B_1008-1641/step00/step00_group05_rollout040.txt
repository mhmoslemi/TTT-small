import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return float(np.max(c * dx))


def _smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(x - m))) / beta


def _obj_grad(h, n, dx, target_sum, beta, pen):
    c = np.correlate(h, 1 - h, mode="full")
    x = c * dx
    m = np.max(x)
    ex = np.exp(x - m)
    sx = np.sum(ex)
    obj = m + np.log(sx) / beta + pen * (np.sum(h) - target_sum) ** 2
    w = ex / (beta * sx)
    grad = np.zeros(n)
    sum_diff = np.sum(h) - target_sum
    for i in range(n):
        g = 0.0
        for m_idx in range(len(c)):
            s = m_idx - n + 1
            t = 0.0
            j1 = i + s
            if 0 <= j1 < n:
                t += (1.0 - h[j1])
            j2 = i - s
            if 0 <= j2 < n:
                t -= h[j2]
            g += w[m_idx] * t
        grad[i] = dx * g + 2.0 * pen * sum_diff
    return obj, grad


def _fix_sum(h, n, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    interior = (h > 0.005) & (h < 0.995)
    if np.sum(interior) > 0:
        h[interior] += diff / np.sum(interior)
    else:
        h += diff / n
    return np.clip(h, 0.0, 1.0)


def _try_optimize(h0, n, dx, target_sum, beta, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        _obj_grad, h0, args=(n, dx, target_sum, beta, pen),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9}
    )
    return _fix_sum(res.x, n, target_sum)
