    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')

        else:
            stall += 1
            accept = np.exp(-(c5_new - best_c5) / max(temp, 1e-10))
            if accept > rng.rand():
                h = h_new
            step = max(0.015, step * 0.995)

        if stall > 2000:
            h = project_h(h + rng.randn(n) * 0.5, target)
            temp = 0.008
            step = 0.15
            stall = 0
        temp = max(temp * 0.9999, 1e-5)

    return best_h, best_c5

def gen_structured(n, period, frac, rng):
    base = np.zeros(period)
    k = max(1, min(period - 1, int(round(period * frac))))
    idx = rng.choice(period, k, replace=False)
    base[idx] = 1.0
    h = np.tile(base, n // period + 1)[:n]
    return project_h(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = project_h(h0, n0 / 2.0)
    best_c5 = compute_c5(h0, dx0)
    best_h = h0.copy()
    best_n = n0

    # Phase 1: Intensive local search at base resolution
    p1 = min(budget_s * 0.35, 300)
    h1, c5_1 = local_search(h0, dx0, p1, time.time(), seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Multi-restart exploration
    p2_dead = t0 + budget_s * 0.65
    rng = np.random.RandomState(seed + 200)
    attempts = 0
    while time.time() < p2_dead:
        h_try = project_h(best_h + rng.randn(best_n) * 0.2, best_n / 2.0)
        st = min(12, p2_dead - time.time() - 3)
        if st < 2:
            break
        h_r, c5_r = local_search(h_try, 2.0 / best_n, st, time.time(), seed + attempts * 7)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
        attempts += 1

    # Phase 3: Structured pattern exploration
    p3_dead = t0 + budget_s * 0.82
    rng2 = np.random.RandomState(seed + 500)
    for period in [3, 5, 7, 9, 15]:
        if time.time() > p3_dead:
            break
        for frac in [0.33, 0.4, 0.44, 0.5]:
            if time.time() > p3_dead - 5:
                break
            h_s = gen_structured(n0, period, frac, rng2)
            st = min(6, p3_dead - time.time() - 2)
            if st < 1.5:
                continue
            h_r, c5_r = local_search(h_s, dx0, st, time.time(), seed + period * 13 + int(frac * 100))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    # Phase 4: Higher resolution attempt
    if time.time() < deadline - 40:
        n2 = n0 * 2
        dx2 = 2.0 / n2
        x_o = np.linspace(0, 2, n0, endpoint=False)
        x_n = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_n, x_o, best_h)
        h2 = project_h(h2, n2 / 2.0)
        st = min(deadline - time.time() - 30, budget_s * 0.12)
        if st > 10:
            h3, c5_3 = local_search(h2, dx2, st, time.time(), seed + 5000)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3
                best_n = n2

    # Phase 5: Final fine polish
    rem = deadline - time.time() - 5
    if rem > 8:
        h_f, c5_f = local_search(best_h, 2.0 / best_n, rem, time.time(), seed + 9999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    best_h = project_h(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, final_c5, best_n
