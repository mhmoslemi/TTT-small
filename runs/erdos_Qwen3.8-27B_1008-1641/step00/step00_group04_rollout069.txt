def c5_eval(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
            if not np.any(m):
                break
            r = 1.0 - h[m]
            t = r.sum()
            if t < 1e-14:
                break
            h[m] += (d / t) * r
        else:
            m = h > 1e-14
            if not np.any(m):
                break
            a = h[m]
            t = a.sum()
            if t < 1e-14:
                break
            h[m] -= (-d / t) * a
    return np.clip(h, 0.0, 1.0)
