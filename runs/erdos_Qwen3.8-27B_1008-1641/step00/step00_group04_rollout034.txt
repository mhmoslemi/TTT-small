        cur_c5 = evaluate_c5(h)
        delta = cur_c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
            if cur_c5 < best_c5:
                best_h = h.copy()
                best_c5 = cur_c5
        else:
            h[i], h[j] = h[j], h[i]
        T = max(T * 0.99998, T_min)
    return best_h, best_c5

def greedy_refine(h_init, rng, deadline):
    h = h_init.copy()
    cur_c5 = evaluate_c5(h)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 0
        best_i, best_j = -1, -1
        best_new_c5 = cur_c5
        n_pairs = min(len(ones) * len(zeros), 800)
        for _ in range(n_pairs):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            new_c5 = evaluate_c5(h)
            delta = new_c5 - cur_c5
            if delta < best_delta - 1e-12:
                best_delta = delta
                best_i, best_j = i, j
                best_new_c5 = new_c5
            h[i], h[j] = h[j], h[i]
        if best_i >= 0:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            cur_c5 = best_new_c5
            improved = True
    return h, cur_c5

def continuous_optimize(h_init, deadline):
    n = len(h_init)
    target_sum = n / 2.0
    tau = 15.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(c)
        return m + np.log(np.sum(np.exp((c - m) * tau))) / tau
    x0 = h_init.astype(float)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, x0, method='SLSQP', constraints=cons, bounds=bnds,
                   options={'maxiter': 300, 'ftol': 1e-10})
    if time.time() < deadline:
        hx = np.clip(res.x, 0, 1)
        sc = np.sum(hx)
        if sc > 0:
            hx = hx / sc * target_sum
            hx = np.clip(hx, 0, 1)
            return hx, evaluate_c5(hx)
    return h_init, evaluate_c5(h_init)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h_parent = None

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Optimize at parent resolution
    if h_parent is not None:
        n = len(h_parent)
        h0 = make_binary(h_parent, n, rng)
        best_h = h0.copy()
        best_c5 = evaluate_c5(h0)

        for rnd in range(8):
            dl = min(deadline - 30, t0 + budget_s * (0.5 + 0.05 * rnd))
            if time.time() >= dl:
                break
            T0_val = 0.04 * (0.7 ** rnd)
            n_it = max(5000, int((dl - time.time()) * 3000))
            h_opt, c5_opt = sa_optimize(best_h, rng, n_it, T0_val, 0.0005, dl)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        # Greedy polish
        if time.time() < deadline - 20:
            h_g, c5_g = greedy_refine(best_h, rng, deadline - 10)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()

    # Phase 2: Try other n values
    n_candidates = [42, 44, 46, 48, 52, 56, 60, 64, 72]
    for n_try in n_candidates:
        if time.time() >= deadline - 50:
            break
        if best_h is not None and n_try == len(best_h):
            continue

        alloc = max(20, min(80, (deadline - time.time()) / max(len(n_candidates), 1)))
        dl = time.time() + alloc

        h_cb = np.zeros(n_try)
        h_cb[::2] = 1.0
        h_cb = make_binary(h_cb, n_try, rng)

        h_blk = np.zeros(n_try)
        h_blk[:n_try // 2] = 1.0

        best_local_h = None
        best_local_c5 = float('inf')

        for h_init in [h_cb, h_blk]:
            if time.time() >= dl - 5:
                break
            c5_init = evaluate_c5(h_init)

            n_it = max(3000, int((dl - time.time() - 5) * 2500))
            h_opt, c5_opt = sa_optimize(h_init, rng, n_it, 0.035, 0.0008, dl - 5)
            if c5_opt < best_local_c5:
                best_local_c5 = c5_opt
                best_local_h = h_opt

            if c5_opt < best_local_c5 or c5_opt < best_c5:
                h_g2, c5_g2 = greedy_refine(h_opt, rng, dl)
                if c5_g2 < best_local_c5:
                    best_local_c5 = c5_g2
                    best_local_h = h_g2

        # Random restart
        if time.time() < dl - 5:
            h_rand = np.zeros(n_try)
            h_rand[rng.choice(n_try, n_try // 2, replace=False)] = 1.0
            n_it2 = max(2000, int((dl - time.time() - 5) * 2500))
            h_opt3, c5_opt3 = sa_optimize(h_rand, rng, n_it2, 0.05, 0.001, dl - 5)
            if c5_opt3 < best_local_c5:
                best_local_c5 = c5_opt3
                best_local_h = h_opt3

        if best_local_h is not None and best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = best_local_h.copy()

    # Phase 3: Final greedy + continuous on overall best
    if best_h is not None:
        if time.time() < deadline - 15:
            h_g3, c5_g3 = greedy_refine(best_h, rng, deadline - 10)
            if c5_g3 < best_c5:
                best_c5 = c5_g3
                best_h = h_g3

        if time.time() < deadline - 15:
            try:
                h_c, c5_c = continuous_optimize(best_h, deadline - 5)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_c
            except Exception:
                pass

    # Fallback
    if best_h is None:
        n = 42
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = evaluate_c5(best_h)

    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, len(best_h)
