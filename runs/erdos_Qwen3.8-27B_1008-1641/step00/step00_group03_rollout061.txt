import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_c5_and_weights(h, T):
    n = len(h)
    dx = 2.0 / n
    h_c = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_c, 1.0 - h_c, mode="full") * dx
    max_c = np.max(corr)
    shifted = (corr - max_c) * T
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    val = max_c + np.log(sum_exp) / T
    weights = exp_s / sum_exp
    return val, weights, h_c

def compute_grad(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    # d = 0
    k0 = n - 1
    grad += weights[k0] * dx * (1.0 - 2.0 * h)
    # d > 0
    for d in range(1, n):
        k = d + n - 1
        w = weights[k] * dx
        if w > 1e-12:
            grad[:n-d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n-d]
    # d < 0
    for d in range(1, n):
        k = n - 1 - d
        w = weights[k] * dx
        if w > 1e-12:
            grad[d:] += w * (1.0 - h[:n-d])
            grad[:n-d] -= w * h[d:]
    return grad

def make_obj(T):
    def _obj(h):
        v, _, _ = soft_c5_and_weights(h, T)
        return v
    return _obj

def make_grad(T):
    def _grad(h):
        _, w, hc = soft_c5_and_weights(h, T)
        n = len(hc)
        dx = 2.0 / n
        return compute_grad(hc, w, dx)
    return _grad

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    target = n / 2.0

    h0 = np.array(initial_h_values, dtype=float)
    h0 = fix_sum(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    bounds = Bounds(0.0, 1.0)
    con = LinearConstraint(np.ones(n), target, target)

    # Phase 1: SLSQP from initial
    best_h = do_opt(h0, 200.0, 300, bounds, con, 'SLSQP')
    c = compute_c5(best_h)
    if c < best_c5:
        best_c5 = c

    elapsed = time.time() - t0
    if elapsed > budget_s * 0.4:
        return best_h, best_c5, n

    # Phase 2: trust-constr
    best_h = do_opt(best_h, 200.0, 500, bounds, con, 'trust-constr')
    c = compute_c5(best_h)
    if c < best_c5:
        best_c5 = c

    elapsed = time.time() - t0
    if elapsed > budget_s * 0.6:
        return best_h, best_c5, n

    # Phase 3: Random restarts at n
    rng = np.random.default_rng(seed)
    n_restarts = 40
    for r in range(n_restarts):
        if time.time() - t0 > budget_s * 0.82:
            break
        scale = 0.03 + 0.12 * (1.0 - r / n_restarts)
        h_p = best_h + rng.uniform(-scale, scale, n)
        h_p = fix_sum(h_p, n)
        h_cand = do_opt(h_p, 200.0, 150, bounds, con, 'SLSQP')
        c = compute_c5(h_cand)
        if c < best_c5:
            best_c5 = c
            best_h = h_cand.copy()

    elapsed = time.time() - t0
    if elapsed > budget_s * 0.88:
        return best_h, best_c5, n

    # Phase 4: Upsample to 2n and optimize
    n2 = n * 2
    h2 = upsample(best_h, n2)
    h2 = fix_sum(h2, n2)
    bounds2 = Bounds(0.0, 1.0)
    con2 = LinearConstraint(np.ones(n2), n2 / 2.0, n2 / 2.0)
    h2 = do_opt(h2, 200.0, 300, bounds2, con2, 'SLSQP')
    c2 = compute_c5(h2)
    if c2 < best_c5:
        best_h = h2.copy()
        best_c5 = c2
        n = n2

    elapsed = time.time() - t0
    if elapsed > budget_s * 0.95:
        best_h = fix_sum(best_h, n)
        return best_h, compute_c5(best_h), n

    # Phase 5: Final polish with high T
    bounds_f = Bounds(0.0, 1.0)
    con_f = LinearConstraint(np.ones(n), n / 2.0, n / 2.0)
    best_h = do_opt(best_h, 500.0, 300, bounds_f, con_f, 'SLSQP')
    c = compute_c5(best_h)
    if c < best_c5:
        best_c5 = c

    best_h = do_opt(best_h, 1000.0, 200, bounds_f, con_f, 'SLSQP')
    c = compute_c5(best_h)
    if c < best_c5:
        best_c5 = c

    best_h = fix_sum(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
