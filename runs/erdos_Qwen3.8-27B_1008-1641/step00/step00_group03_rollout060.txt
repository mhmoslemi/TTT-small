    """Project h to satisfy sum(h) = n/2 and 0 <= h <= 1."""
    h = np.array(h, dtype=float)
    target = n / 2.0

        if np.sum(free) == 0:
            free = np.ones(n, dtype=bool)
        h[free] += diff / np.sum(free)
    return h

    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx

            if i + shift != i:

            grad[i] += (1 - h[i - s])

            hn = project(hg - lr * g, n)
            cn = compute_c5(hn, n)
            if cn < compute_c5(hg, n) - 1e-12:
                if cn < best_c5:
                    best_c5 = cn
                    best_h = hn
                hg = hn
                stuck = 0
                lr = min(lr * 1.05, 0.02)
            else:
                lr *= 0.6
                stuck += 1
                if stuck > 50:
                    break

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
