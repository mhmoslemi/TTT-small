def repair_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    h[~np.isfinite(h)] = 0.0
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(12):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13: break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask): break
            step = diff / float(np.count_nonzero(mask))
            lim = float(np.min(1.0 - h[mask]))
            if step > lim: step = lim
        else:
            mask = h > 1e-15
            if not np.any(mask): break
            step = diff / float(np.count_nonzero(mask))
            lim = -float(np.min(h[mask]))
            if step < lim: step = lim
        h[mask] += step
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - float(h.sum())
    if abs(diff) > 1e-15:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
            for idx in order:
                if diff <= 0: break
                inc = min(diff, 1.0 - float(h[idx]))
                if inc > 0:
                    h[idx] += inc; diff -= inc
        else:
            order = np.argsort(h)
            for idx in order:
                if diff >= 0: break
                dec = max(diff, -float(h[idx]))
                if dec < 0:
                    h[idx] += dec; diff -= dec
    diff = target - float(h.sum())
    if diff > 1e-15:
        idx = int(np.argmax(1.0-h))
        h[idx] = min(1.0, h[idx] + diff)
    elif diff < -1e-15:
        idx = int(np.argmax(h))
        h[idx] = max(0.0, h[idx] + diff)
    return h

def line_search_pair(base, L, E, tlo, thi, scans):
    if thi - tlo <= 1e-14:
        return 0.0, float(np.max(base))
    k = int(max(3, scans))
    ts = np.linspace(tlo, thi, k, dtype=float)
    ts = np.append(ts, 0.0)
    T = ts[:, None]
    vals = base[None, :] + T * L[None, :] - (T * T) * E[None, :]
    f = np.max(vals, axis=1)
    i = int(np.argmin(f))
    bt, bf = float(ts[i]), float(f[i])
    if k >= 6:
        lo = ts[i-1] if i > 0 else ts[i]
        hi = ts[i+1] if i < len(ts)-1 else ts[i]
        if hi - lo > 1e-12:
            ts2 = np.linspace(lo, hi, int(max(5, k//2)), dtype=float)
            T2 = ts2[:, None]
            vals2 = base[None, :] + T2 * L[None, :] - (T2 * T2) * E[None, :]
            f2 = np.max(vals2, axis=1)
            j = int(np.argmin(f2))
            if f2[j] < bf:
                bf = float(f2[j]); bt = float(ts2[j])
    return bt, bf

def pair_local(h, dx, target, scans=21, passes=30, deadline=None, min_gain=1e-10, shuffle=False, seed=0):
    n = int(len(h))
    h = np.array(h, dtype=float, copy=True)
    base = np.correlate(h, 1.0 - h, mode="full")
    best_raw = float(np.max(base)); best_h = h.copy()
    rng = np.random.default_rng(seed) if shuffle else None
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for _ in range(int(max(1, passes))):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        base = np.correlate(h, 1.0 - h, mode="full")
        cur = float(np.max(base))
        if cur < best_raw - min_gain:
            best_raw = cur; best_h = h.copy()
        order = pairs if rng is None else pairs.copy()
        if rng is not None:
            rng.shuffle(order)
        for i, j in order:
            tlo = -float(h[j]); thi = 1.0 - float(h[i])
            if thi <= tlo + 1e-14:
                continue
            u = np.zeros(n, dtype=float); u[i] = 1.0; u[j] = -1.0
            L = np.correlate(u, 1.0 - h, mode="full") - np.correlate(h, u, mode="full")
            E = np.correlate(u, u, mode="full")
            t, rem = line_search_pair(base, L, E, tlo, thi, int(scans))
            if rem < cur - float(min_gain):
                h[i] += t; h[j] -= t

                elif h[i] > 1.0: h[i] = 1.0

```python
import time
import math
import numpy as np
import scipy.optimize

_TARGET = 0.0
_DX = 1.0
_N = 0
_BETA = 60.0
_EYE = None
_GOAL = 0.3808


def exact_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def repair_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    h[~np.isfinite(h)] = 0.0
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(10):
        d = target - float(h.sum())
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            m = h < 1.0 - 1e-15
            if not np.any(m):
                break
            s = d / float(np.count_nonzero(m))
            lim = float(np.min(1.0 - h[m]))
            h[m] += s if s < lim else lim
        else:
            m = h > 1e-15
            if not np.any(m):
                break
            s = d / float(np.count_nonzero(m))
            lim = -float(np.min(h[m]))
            h[m] += s if s > lim else lim
        np.clip(h, 0.0, 1.0, out=h)

    d = target - float(h.sum())
    if d > 1e-15:
        for idx in np.argsort(1.0 - h)[::-1]:
            if d <= 0.0:
                break
            step = min(d, float(1.0 - h[idx]))
            if step > 0.0:
                h[idx] += step
                d -= step
    elif d < -1e-15:
        for idx in np.argsort(h):
            if d >= 0.0:
                break
            step = max(d, -float(h[idx]))
            if step < 0.0:
                h[idx] += step
                d -= step

    d = target - float(h.sum())
    if d > 0.0:
        idx = int(np.argmax(1.0 - h))
        h[idx] = min(1.0, h[idx] + d)
    elif d < 0.0:
        idx = int(np.argmax(h))
        h[idx] = max(0.0, h[idx] + d)
    return h


def quantize_exact(h, target, m=30):
    scale = 1 << m
    q = np.rint(np.clip(h, 0.0, 1.0) * scale).astype(np.int64)
    T = int(round(target * scale))
    d = T - int(q.sum())
    if d > 0:
        for idx in np.argsort(scale - q)[::-1]:
            if d <= 0:
                break
            take = min(d, int(scale - q[idx]))
            if take:
                q[idx] += take
                d -= take
    elif d < 0:
        rem = -d
        for idx in np.argsort(q):
            if rem <= 0:
                break
            take = min(rem, int(q[idx]))
            if take:
                q[idx] -= take
                rem -= take
    return q * float(1.0 / scale)


def smooth_objective(x):
    if not np.all(np.isfinite(x)):
        return 1e100
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    b = float(_BETA)
    if b <= 0.0:
        return m * float(_DX)
    z = np.exp(b * (c - m))
    s = float(np.sum(z))
    if s <= 0.0:
        return m * float(_DX)
    return (m + math.log(s) / b) * float(_DX)


def smooth_gradient(x):
    if not np.all(np.isfinite(x)):
        return np.zeros(len(x), dtype=float)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    b = float(_BETA)
    z = np.exp(b * (c - m))
    s = float(np.sum(z))
    w = z / s if s > 0.0 else np.ones_like(c) / len(c)
    n = len(x)
    y = 1.0 - x
    eye = _EYE
    if eye is None or eye.shape[0] != n:
        eye = np.eye(n, dtype=float)
    g = np.empty(n, dtype=float)
    for i in range(n):
        e = eye[i]
        gi = np.correlate(e, y, mode="full") - np.correlate(x, e, mode="full")
        g[i] = float(np.dot(w, gi))
    return g * float(_DX)


def sum_constraint(x):
    return float(np.sum(x) - _TARGET)


def sum_jacobian(x):
    return np.ones(len(x), dtype=float)


def slsqp_polish(h, dx, target, beta, maxiter):
    global _TARGET, _DX, _N, _BETA, _EYE
    h = np.asarray(h, dtype=float)
    n = len(h)
    _TARGET = float(target)
    _DX = float(dx)
    _N = int(n)
    _BETA = float(beta)
    _EYE = np.eye(n, dtype=float)
    try:
        res = scipy.optimize.minimize(
            smooth_objective,
            h.copy(),
            method="SLSQP",
            jac=smooth_gradient,
            bounds=[(0.0, 1.0)] * n,
            constraints=[{"type": "eq", "fun": sum_constraint, "jac": sum_jacobian}],
            options={"maxiter": int(max(10, maxiter)), "ftol": 1e-12, "disp": False},
        )
        if res is not None and getattr(res, "x", None) is not None:
            return repair_sum(np.asarray(res.x, dtype=float), target)
    except Exception:
        pass
    return repair_sum(h, target)


def line_search_pair(base, L, E, tlo, thi, scans):
    if thi - tlo <= 1e-14:
        return 0.0, float(np.max(base))
    k = int(max(3, scans))
    ts = np.linspace(tlo, thi, k, dtype=float)
    ts = np.append(ts, 0.0)
    T = ts[:, None]
    vals = base[None, :] + T * L[None, :] - (T * T) * E[None, :]
    f = np.max(vals, axis=1)
    i = int(np.argmin(f))
    bt, bf = float(ts[i]), float(f[i])
    if k >= 6:
        lo = ts[i - 1] if i > 0 else ts[i]
        hi = ts[i + 1] if i < len(ts) - 1 else ts[i]
        if hi - lo > 1e-12:
            ts2 = np.linspace(lo, hi, int(max(5, k // 2)), dtype=float)
            T2 = ts2[:, None]
            vals2 = base[None, :] + T2 * L[None, :] - (T2 * T2) * E[None, :]
            f2 = np.max(vals2, axis=1)
            j = int(np.argmin(f2))
            if f2[j] < bf:
                bf, bt = float(f2[j]), float(ts2[j])
    return bt, bf


def pair_local(h, dx, target, scans=21, passes=30, deadline=None, min_gain=1e-10, shuffle=False, seed=0):
    n = int(len(h))
    h = np.array(h, dtype=float, copy=True)
    base = np.correlate(h, 1.0 - h, mode="full")
    best_raw = float(np.max(base))
    best_h = h.copy()
    rng = np.random.default_rng(seed) if shuffle else None
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    cur = best_raw
    for _ in range(int(max(1, passes))):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        base = np.correlate(h, 1.0 - h, mode="full")
        cur = float(np.max(base))
        if cur < best_raw - min_gain:
            best_raw, best_h = cur, h.copy()
        order = pairs if rng is None else pairs.copy()
        if rng is not None:
            rng.shuffle(order)
        for i, j in order:
            tlo = -float(h[j])
            thi = 1.0 - float(h[i])
            if thi <= tlo + 1e-14:
                continue
            u = np.zeros(n, dtype=float)
            u[i] = 1.0
            u[j] = -1.0
            L = np.correlate(u, 1.0 - h, mode="full") - np.correlate(h, u, mode="full")
            E = np.correlate(u, u, mode="full")
            t, rem = line_search_pair(base, L, E, tlo, thi, int(scans))
            if rem < cur - float(min_gain):
                h[i] += t
                h[j] -= t
                if h[i] < 0.0:
                    h[i] = 0.0
                elif h[i] > 1.0:
                    h[i] = 1.0
                if h[j] < 0.0:
                    h[j] = 0.0
                elif h[j] > 1.0:
                    h[j] = 1.0
                base = np.correlate(h, 1.0 - h, mode="full")
                cur = float(np.max(base))
                improved = True
        if improved:
            h = repair_sum(h, target)
            base = np.correlate(h, 1.0 - h, mode="full")
            cur = float(np.max(base))
            if cur < best_raw - min_gain:
                best_raw, best_h = cur, h.copy()
            elif cur > best_raw + 1e-8:
                h = best_h.copy()
                cur = best_raw
        else:
            break
    if cur < best_raw:
        best_raw, best_h = cur, h.copy()
    return best_h, float(best_raw * dx)


def make_pattern_h(n, target, kind, seed):
    rng = np.random.default_rng(seed)
    if kind == 0:
        h = rng.random(n)
    elif kind == 1:
        q = int(math.floor(target))
        r = target - q
        h = np.zeros(n, dtype=float)
        idx = rng.permutation(n)
        if q > 0:
            h[idx[:q]] = 1.0
        if r > 1e-12:
            if q < n:
                h[idx[q]] = r
            else:
                h[n - 1] = target - (n - 1)
    else:
        f = float(rng.integers(1, 4))
        phase = float(rng.uniform(0.0, 2.0 * math.pi))
        x = np.linspace(0.0, 2.0, n, dtype=float)
        h = 0.5 + 0.9 * np.cos(2.0 * math.pi * f * x + phase)
    return repair_sum(h, target)


def interp_h(h, n):
    h = np.asarray(h, dtype=float)
    if len(h) == n:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, len(h), dtype=float)
    x1 = np.linspace(0.0, 2.0, n, dtype=float)
    return np.interp(x1, x0, h)


def perturb_h(h, target, amount, seed):
    rng = np.random.default_rng(seed)
    return repair_sum(np.asarray(h, dtype=float) + amount * (rng.random(len(h)) - 0.5), target)


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    budget = max(60.0, float(budget_s))
    deadline = start + budget
    work = deadline - min(30.0, max(10.0, budget * 0.05))

    best_h = None
    best_c = 1e100
    best_n = 0
    best_dx = 1.0
    best_target = 0.0

    init = None
    gv = globals()
    if "initial_h_values" in gv:
        try:
            init = np.asarray(gv["initial_h_values"], dtype=float).reshape(-1)
        except Exception:
            init = None

    if init is not None and init.size:
        nn = int(init.size)
        tt = 0.5 * float(nn)
        dd = 2.0 / float(nn)
        h0 = repair_sum(init.copy(), tt)
        c0 = exact_c5(h0, dd)
        if np.isfinite(c0):
            best_h, best_c, best_n, best_dx, best_target = h0, c0, nn, dd, tt

    if best_h is None:
        nn = 47
        tt = 0.5 * float(nn)
        dd = 2.0 / float(nn)
        best_h = make_pattern_h(nn, tt, 1, int(seed))
        best_c = exact_c5(best_h, dd)
        best_n, best_dx, best_target = nn, dd, tt

    p1 = start + min(450.0, max(20.0, 0.35 * budget))
    p1 = min(p1, work - max(30.0, budget * 0.15))
    if p1 <= start:
        p1 = min(work, start + 5.0)
    ph, pc = pair_local(best_h, best_dx, best_target, scans=25, passes=120, deadline=p1, min_gain=1e-10)
    if pc < best_c:
        best_h, best_c = ph, pc

    if time.time() < work - max(40.0, budget * 0.10):
        end2 = time.time() + min(250.0, max(30.0, budget * 0.25))
        end2 = min(end2, work - max(20.0, budget * 0.08))
        for beta in (80.0, 180.0, 400.0):
            if time.time() > end2:
                break
            hu = slsqp_polish(best_h, best_dx, best_target, int(beta), 70)
            hs = repair_sum(0.5 * (hu + hu[::-1]), best_target)
            cu, cs = exact_c5(hu, best_dx), exact_c5(hs, best_dx)
            if cs < cu:
                hu, cu = hs, cs
            if cu < best_c - 1e-12:
                best_h, best_c = hu, cu
            if time.time() < work - 30.0:
                ph, pc = pair_local(
                    best_h, best_dx, best_target,
                    scans=15, passes=4,
                    deadline=min(work - 30.0, time.time() + 70.0),
                )
                if pc < best_c:
                    best_h, best_c = ph, pc

    if best_c > _GOAL and time.time() < work - max(80.0, budget * 0.10):
        end3 = min(work - max(50.0, budget * 0.08), time.time() + min(150.0, budget * 0.20))
        for k in range(4):
            if time.time() > end3:
                break
            sh = perturb_h(best_h, best_target, 0.12, int(seed) + 100 * k)
            ph, pc = pair_local(
                sh, best_dx, best_target,
                scans=13, passes=3,
                deadline=min(end3, time.time() + 45.0),
            )
            if pc < best_c:
                best_h, best_c = ph, pc

    if best_c > _GOAL and time.time() < work - max(60.0, budget * 0.06):
        end4 = work - max(10.0, budget * 0.02)
        for nn in (48, 64, 94):
            if time.time() > end4 or best_c <= _GOAL:
                break
            if nn == best_n:
                continue
            tt = 0.5 * float(nn)
            dd = 2.0 / float(nn)
            sh = repair_sum(interp_h(best_h, nn), tt)
            ph, pc = pair_local(
                sh, dd, tt,
                scans=13, passes=3,
                deadline=min(end4, time.time() + 60.0),
            )
            if pc < best_c - 1e-12:
                best_h, best_c, best_n, best_dx, best_target = ph, pc, nn, dd, tt
                continue

            if nn % 2 == 0:
                sh2 = make_pattern_h(nn, tt, 1, int(seed) + 2000)
                ph2, pc2 = pair_local(
                    sh2, dd, tt,
                    scans=13, passes=3,
                    deadline=min(end4, time.time() + 60.0),
                )
                if pc2 < best_c - 1e-12:
                    best_h, best_c, best_n, best_dx, best_target = ph2, pc2, nn, dd, tt
                    continue

            sh3 = make_pattern_h(nn, tt, 2, int(seed) + 3000)
            ph3, pc3 = pair_local(
                sh3, dd, tt,
                scans=13, passes=3,
                deadline=min(end4, time.time() + 60.0),
            )
            if pc3 < best_c - 1e-12:
                best_h, best_c, best_n, best_dx, best_target = ph3, pc3, nn, dd, tt

    hfin = repair_sum(best_h, best_target)
    cfin = exact_c5(hfin, best_dx)
    sh = repair_sum(0.5 * (hfin + hfin[::-1]), best_target)
    cs = exact_c5(sh, best_dx)
    if cs < cfin:
        hfin, cfin = sh, cs

    out_h, out_c = hfin, cfin
    if time.time() < deadline - 3.0:
        for m in (30, 24):
            qh = quantize_exact(hfin, best_target, m)
            if abs(float(qh.sum()) - best_target) > 1e-10:
                qh = repair_sum(qh, best_target)
            qh = np.clip(qh, 0.0, 1.0)
            cq = exact_c5(qh, best_dx)
            if cq <= cfin + 1e-9 and cq < out_c:
                out_h, out_c = qh, cq

    if not np.isfinite(out_c):
        out_h = repair_sum(best_h, best_target)
        out_c = exact_c5(out_h, best_dx)

    return out_h, float(out_c), int(len(out_h))
```
