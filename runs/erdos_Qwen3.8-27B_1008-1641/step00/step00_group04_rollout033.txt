import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(params, target_sum, beta):
    n = len(params)
    dx = 2.0 / n
    h = np.clip(params, 0.0, 1.0)
    corr = correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    logsum = np.log(np.sum(np.exp((corr - m) * beta)))
    sm = m + logsum / beta
    pen = 1e4 * (np.sum(h) - target_sum) ** 2
    return sm + pen

def binarize(h, target):
    h_bin = (h >= 0.5).astype(float)
    s = int(np.sum(h_bin))
    if s != target:
        if s > target:
            idx = np.where(h_bin == 1)[0]
            order = np.argsort(h[idx])
            for i in idx[order[:s - target]]:
                h_bin[i] = 0.0
        else:
            idx = np.where(h_bin == 0)[0]
            order = np.argsort(-h[idx])
            for i in idx[order[:target - s]]:
                h_bin[i] = 1.0
    return h_bin

def optimize_cont(h0, target_sum, timeout):
    t0 = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    n = len(h0)
    betas = [50, 150, 400, 1000, 2500, 5000]
    for beta in betas:
        if time.time() - t0 > timeout:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(target_sum, float(beta)),
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-12,
                                   'maxfun': 5000})
            h_opt = np.clip(res.x, 0.0, 1.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass
    return best_h, best_c5

def local_search_bin(h0, timeout=30.0):
    t0 = time.time()
    h = h0.copy()
    best = compute_c5(h)
    stall = 0
    while time.time() - t0 < timeout:
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        improved = False
        for i in ones:
            for j in zeros:
                if time.time() - t0 > timeout:
                    return h, best
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h)
                if c < best - 1e-14:
                    best = c
                    improved = True
                    break
                h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            stall += 1
            if stall >= 5:
                break
        else:
            stall = 0
    return h, best

def sa_search(h0, timeout=60.0, seed=0):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    temp = 0.005
    while time.time() - t0 < timeout:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c_new = compute_c5(h)
        delta = c_new - cur_c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            cur_c5 = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        temp = max(temp * 0.9995, 1e-6)
    return best_h, best_c5
