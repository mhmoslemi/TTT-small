import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, dx, temperature):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    t = temperature
    cm = np.max(corr)
    shifted = t * (corr - cm)
    es = np.exp(shifted)
    weights = es / es.sum()
    grad = np.zeros(n, dtype=float)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        w = weights[m]
        if w < 1e-14:
            continue
        if k > 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        elif k < 0:
            ka = -k
            grad[ka:] += w * (1.0 - h[:n - ka])
            grad[:n - ka] -= w * h[ka:]
        else:
            grad += w * (1.0 - 2.0 * h)
    return grad * dx

def project_gradient_descent(h0, n, dx, max_iter=5000, seed=42):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    temperature = 40.0
    lr = 0.02
    prev_grad = None
    for it in range(max_iter):
        g = smooth_grad(h, dx, temperature)
        if prev_grad is not None:
            ys = h - prev_h
            gy = -(g - prev_grad).dot(ys)
            yy = ys.dot(ys)
            if yy > 1e-16 and gy > 1e-16:
                lr = float(np.dot(ys, gy) / yy)
                lr = max(1e-6, min(lr, 0.1))
        prev_grad = g.copy()
        prev_h = h.copy()
        h_new = project_feasible(h - lr * g, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= compute_c5(h, dx) * 1.0001:
            h = h_new
        else:
            lr *= 0.5
        if it % 400 == 399:
            temperature *= 0.85
        if lr < 1e-9:
            break
    return best_h, best_c5

def slsqp_refine(h0, n, dx, max_iter=200):
    bounds = Bounds(0.0, 1.0)
    A = np.ones((1, n))
    con = LinearConstraint(A, n / 2.0, n / 2.0)
    temp = 30.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        cm = np.max(corr)
        return (cm + np.log(np.sum(np.exp(temp * (corr - cm)))) / temp) * dx
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=con,
                   tol=1e-12, options={'maxiter': max_iter, 'disp': False})
    h_opt = project_feasible(res.x, n)
    c5_opt = compute_c5(h_opt, dx)
    if c5_opt < compute_c5(h0, dx):
        return h_opt, c5_opt
    return h0, compute_c5(h0, dx)

def coord_descent(h0, n, dx, max_sweeps=50, temp=20.0):
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    for sweep in range(max_sweeps):
        improved = False
        order = np.random.permutation(n)
        for j in order:
            for delta in np.array([-0.1, -0.05, 0.05, 0.1, -0.01, 0.01]):
                h_try = h.copy()
                h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5 - 1e-10:
                    h = h_try.copy()
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

    return best_h, best_c5, n
