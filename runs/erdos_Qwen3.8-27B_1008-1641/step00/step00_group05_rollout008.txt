    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)

def c5_value(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_correlations(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_single_lag(h, dx, k, n):
    """Gradient of c_k with respect to h"""
    g = np.zeros(n)
    j1_lo = max(0, k)
    j1_hi = min(n - 1, n - 1 + k)
    if j1_lo <= j1_hi:
        g[j1_lo:j1_hi+1] += (1.0 - h[j1_lo - k: j1_hi + 1 - k])
    j2_lo = max(0, -k)
    j2_hi = min(n - 1, n - 1 - k)
    if j2_lo <= j2_hi:
        g[j2_lo:j2_hi+1] -= h[j2_lo + k: j2_hi + 1 + k]
    return g * dx

def soft_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_correlations(h)
    m = np.max(corr)
    exp_c = np.exp(corr - m)
    total = exp_c.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_c / total
    grad = np.zeros(n)
    for j in range(len(corr)):
        if w[j] < 1e-10:
            continue
        k = j - (n - 1)
        grad += w[j] * grad_single_lag(h, dx, k, n)
    return grad

def hard_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = get_correlations(h)
    max_v = np.max(corr)
    active = np.where(corr >= max_v - 1e-8)[0]
    grad = np.zeros(n)
    for j in active:
        k = j - (n - 1)
        grad += grad_single_lag(h, dx, k, n)
    grad /= len(active)
    return grad
