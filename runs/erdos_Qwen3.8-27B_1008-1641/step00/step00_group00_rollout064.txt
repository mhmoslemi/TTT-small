    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(x, tau, dx):
    corr = np.correlate(x, 1 - x, mode='full')
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -50, 50)
    return (m + tau * np.log(np.sum(np.exp(shifted)))) * dx


def sum_con(x, target_sum):
    return np.sum(x) - target_sum


def try_optimize(h_start, n, target_sum, dx, bounds, cons, tau_list, maxiter, t0, budget_s):
    best_h = h_start.copy()
    best_c5 = evaluate_c5(h_start)
    for tau in tau_list:
        if time.time() - t0 > budget_s - 20:
            break
        res = minimize(
            smooth_max_obj, best_h, args=(tau, dx),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        candidate = np.clip(res.x, 0.0, 1.0)
        candidate = project_sum(candidate, target_sum)
        c5 = evaluate_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()
    return best_h, best_c5


def projected_gradient(h, target_sum, tau, learning_rate, n_iter, t0, budget_s):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    dt = 0.05
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 10:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.max(corr)
        shifted = (corr - m) / tau
        shifted = np.clip(shifted, -50, 50)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        grad = np.zeros(n)
        eps = 1e-7
        for j in range(n):
            h[j] += eps
            cp = np.correlate(h, 1 - h, mode='full')
            h[j] -= 2 * eps
            cm = np.correlate(h, 1 - h, mode='full')
            h[j] += eps
            grad[j] = np.dot(weights, cp - cm) / (2 * eps)
        grad *= dx
        h_new = h - learning_rate * grad
        h = project_sum(h_new, target_sum)
        dt *= 0.995
        learning_rate *= 0.99
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, Exception):
        h0 = np.ones(45) * 0.5

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (target_sum,)}]

    # Phase 1: SLSQP at initial resolution with decreasing tau
    tau_list1 = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    best_h, best_c5 = try_optimize(h0, n, target_sum, dx, bounds, cons, tau_list1, 800, t0, budget_s)

    # Phase 2: Projected gradient refinement
    if time.time() - t0 < budget_s - 120:
        hg = projected_gradient(best_h, target_sum, 0.01, 0.5, 2000, t0, budget_s - 120)
        c5g = evaluate_c5(hg)
        if c5g < best_c5:
            best_c5 = c5g
            best_h = hg.copy()
        # Continue SLSQP from improved point
        tau_list2 = [0.01, 0.005, 0.002, 0.001]
        best_h, best_c5 = try_optimize(best_h, n, target_sum, dx, bounds, cons, tau_list2, 500, t0, budget_s)

    # Phase 3: Random restarts at same resolution
    n_restarts = 15
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 120:
            break
        scale = 0.15 * (1.0 + 0.05 * r)
        h_pert = best_h + np.random.randn(n) * scale
        h_pert = project_sum(h_pert, target_sum)
        tau_list3 = [0.05, 0.02, 0.01, 0.005]
        hp, c5p = try_optimize(h_pert, n, target_sum, dx, bounds, cons, tau_list3, 400, t0, budget_s)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 4: Binary-like initialization
    if time.time() - t0 < budget_s - 180:
        h_bin = np.zeros(n)
        h_bin[:int(target_sum)] = 1.0
        h_bin = np.random.permutation(h_bin)
        h_bin = project_sum(h_bin, target_sum)
        tau_list4 = [0.1, 0.05, 0.02, 0.01, 0.005]
        hb, c5b = try_optimize(h_bin, n, target_sum, dx, bounds, cons, tau_list4, 600, t0, budget_s)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb.copy()

    # Phase 5: Higher resolution
    for test_n in [90, 180, 360]:
        if time.time() - t0 > budget_s - 150:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, test_n, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, test_n / 2.0)
        dx_new = 2.0 / test_n
        bounds_new = [(0.0, 1.0)] * test_n
        cons_new = [{'type': 'eq', 'fun': sum_con, 'args': (test_n / 2.0,)}]
        tau_list5 = [0.05, 0.02, 0.01, 0.005, 0.002]
        hi2, c5i = try_optimize(h_interp, test_n, test_n / 2.0, dx_new, bounds_new, cons_new, tau_list5, 600, t0, budget_s)
        if c5i < best_c5:
            best_c5 = c5i
            best_h = hi2.copy()

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 30:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        b_f = [(0.0, 1.0)] * n_f
        c_f = [{'type': 'eq', 'fun': sum_con, 'args': (ts_f,)}]
        best_h, best_c5 = try_optimize(best_h, n_f, ts_f, dx_f, b_f, c_f, [0.005, 0.002, 0.001], 800, t0, budget_s)

    return best_h, best_c5, len(best_h)
