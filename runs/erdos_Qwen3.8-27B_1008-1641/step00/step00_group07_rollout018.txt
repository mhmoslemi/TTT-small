def project_box_hyperplane(v, target_sum):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    # Find λ such that sum(clip(v - λ, 0, 1)) = target_sum
    # clip(v-λ, 0, 1) = max(0, min(1, v-λ))
    # f(λ) = sum(clip(v-λ, 0, 1)) is monotonically decreasing in λ
    lo = v.min() - 1
    hi = v.max() + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0, 1)

def smooth_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * corr - beta * max_corr)
    Z = np.sum(exp_corr)
    w = exp_corr / Z  # softmax weights, shape (2n-1,)
    
    obj = (max_corr / beta + np.log(Z) / beta) * dx  # logsumexp / beta * dx
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        # s corresponds to shift s, w index = s + (n-1)
        ws = w[s + n - 1]
        # First part: j such that j+s in [0, n-1]
        j_min = max(0, -s)
        j_max = min(n, n - s)  # exclusive
        if j_min < j_max:
            grad[j_min:j_max] += ws * (1 - h[j_min+s:j_max+s])
        # Second part: j such that j-s in [0, n-1]
        j_min2 = max(0, s)
        j_max2 = min(n, n + s)
        if j_min2 < j_max2:
            grad[j_min2:j_max2] -= ws * h[j_min2-s:j_max2-s]
    
    grad *= dx
    return obj, grad

import numpy as np
import time
from scipy.special import logsumexp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_box_hyperplane(v, target_sum):
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(v - lam, 0.0, 1.0)

def smooth_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = beta * corr
    m = np.max(shifted)
    e = np.exp(shifted - m)
    Z = e.sum()
    w = e / Z
    obj = (m + np.log(Z)) / beta * dx  # = logsumexp / beta * dx
    
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        ws = w[s + n - 1]
        # Part 1: j+s valid → j in [max(0,-s), min(n, n-s))
        j1 = max(0, -s)
        j2 = min(n, n - s)
        if j1 < j2:
            grad[j1:j2] += ws * (1.0 - h[j1 + s:j2 + s])
        # Part 2: j-s valid → j in [max(0,s), min(n, n+s))
        j3 = max(0, s)
        j4 = min(n, n + s)
        if j3 < j4:
            grad[j3:j4] -= ws * h[j3 - s:j4 - s]
    grad *= dx
    return obj, grad

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_hyperplane(v, target_sum):
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(v - lam, 0.0, 1.0)


def smooth_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    bcorr = beta * corr
    m = float(np.max(bcorr))
    e = np.exp(bcorr - m)
    Z = float(e.sum())
    w = e / Z
    obj = (m + np.log(Z)) / beta * dx

    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        ws = w[s + n - 1]
        # (1-h[j-s]) for j-s valid: j in [max(0,s), min(n,s+n))
        j1 = max(0, s)
        j2 = min(n, n + s)
        if j1 < j2:
            grad[j1:j2] += ws * (1.0 - h[j1 - s:j2 - s])
        # -h[j+s] for j+s valid: j in [max(0,-s), min(n,n-s))
        j3 = max(0, -s)
        j4 = min(n, n - s)
        if j3 < j4:
            grad[j3:j4] -= ws * h[j3 + s:j4 + s]
    grad *= dx
    return obj, grad

                stall += 1
