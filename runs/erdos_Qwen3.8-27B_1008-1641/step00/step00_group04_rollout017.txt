    h = np.clip(h, 0.0, 1.0)

    diff = target_sum - s

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    n = len(h)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmin(hp)) if diff > 0 else int(np.argmax(hp))
        hp[idx] = float(np.clip(hp[idx] + diff, 0.0, 1.0))
    return hp

def softmax_weights(corr, alpha):
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    return e / e.sum()

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1 - h, mode="full")
    corr = corr_raw * dx
    w = softmax_weights(corr, alpha)
    grad = np.zeros(n)
    for k_full in range(len(corr)):
        wk = w[k_full]
        if wk < 1e-10:
            continue
        lag = k_full - (n - 1)
        ns = max(0, -lag)
        ne = min(n, n - lag)
        ib = np.arange(ns, ne)
        ia = ib + lag
        grad[ia] += wk * dx * (1.0 - h[ib])
        grad[ib] += wk * dx * (-h[ia])
    return grad

def optimize_pgd(h, target, n_iters, alpha_start, alpha_end, lr0, time_limit=None, t0=0):
    best_h = h.copy()
    best_f = compute_c5(h)
    for i in range(n_iters):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        t = i / max(n_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr0 / (1.0 + 0.05 * i)
        g = grad_smooth_max(h, alpha)
        h_new = project_feasible(h - lr * g, target)
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_f

def coord_descent(h, target, n_sweeps=5, block_size=2, time_limit=None, t0=0):
    n = len(h)
    best_h = h.copy()
    best_f = compute_c5(h)

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target):
    n = len(h)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        idx = int(np.argmin(hp)) if diff > 0 else int(np.argmax(hp))
        hp[idx] = float(np.clip(hp[idx] + diff, 0.0, 1.0))
    return hp


def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for k_full in range(len(corr)):
        wk = w[k_full]
        if wk < 1e-10:
            continue
        lag = k_full - (n - 1)
        ns = max(0, -lag)
        ne = min(n, n - lag)
        ib = np.arange(ns, ne)
        ia = ib + lag
        grad[ia] += wk * dx * (1.0 - h[ib])
        grad[ib] += wk * dx * (-h[ia])
    return grad


def optimize_pgd(h, target, n_iters, alpha_start, alpha_end, lr0, time_limit, t0):
    best_h = h.copy()
    best_f = compute_c5(h)
    cur_f = best_f
    for i in range(n_iters):
        if (time.time() - t0) > time_limit:
            break
        t_frac = i / max(n_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** t_frac
        lr = lr0 / (1.0 + 0.03 * i)
        g = grad_smooth_max(h, alpha)
        h_new = project_feasible(h - lr * g, target)
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        if f_new < cur_f + 1e-14:
            cur_f = f_new
        else:
            lr *= 0.5
            h_new = project_feasible(h - lr * g, target)
            f_new = compute_c5(h_new)
            if f_new < cur_f:
                cur_f = f_new
                if f_new < best_f:
                    best_f = f_new
                    best_h = h_new.copy()
        h = h_new
    return best_h, best_f


def coord_descent(h, target, n_sweeps, block_size, time_limit, t0, seed_val):
    n = len(h)
    best_h = h.copy()
    best_f = compute_c5(h)
    rng = np.random.RandomState(seed_val)
    for sw in range(n_sweeps):
        if (time.time() - t0) > time_limit:
            break
        offset = int(rng.randint(0, n))
        for j in range(offset, n, block_size):
            if (time.time() - t0) > time_limit:
                break
            je = min(j + block_size, n)
            saved = best_h[j:je].copy()
            for trial in range(12):
                delta = rng.uniform(-0.06, 0.06, je - j)
                cand = best_h.copy()
                cand[j:je] = np.clip(saved + delta, 0.0, 1.0)
                cand = project_feasible(cand, target)
                f = compute_c5(cand)
                if f < best_f:
                    best_f = f
                    best_h = cand
            best_h[j:je] = best_h[j:je]  # no-op, keep consistency
    return best_h, best_f
