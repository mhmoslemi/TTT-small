    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")  # length 2n-1, each is sum h[i+k]*(1-h[i])
    # f_k values (unscaled by dx)
    f = corr  # shape (2n-1,)

    T_inv = 1.0 / T
    f_shifted = f * T_inv - np.max(f) * T_inv  # numerical stability
    weights = np.exp(f_shifted)

            break
        delta = rng.uniform(-step, step, n)
        h_new = project_feasible(h + delta, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
            step = min(step * 1.05, 0.4)
        elif c5_new < best_c5:
            h = h_new
        else:
            step = max(step * 0.97, 0.005)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def try_optimize(h_init, t_limit, T_s=50, T_e=500, lr0=0.04, max_it=15000):
        nonlocal best_h, best_c5
        h_opt, c5_opt = optimize_phase(h_init, T_s, T_e, max_it, lr0, t_limit, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 1: Start from initial_h_values
    try:
        h0 = np.array(initial_h_values, dtype=float)
        try_optimize(h0, min(120, budget_s * 0.12))
    except NameError:
        pass

    # Phase 2: Structured and random starts at n=45
    n45 = 45
    tgt45 = 22.5
    h_struct = np.zeros(n45)
    h_struct[:n45 // 2] = 1.0
    h_struct = project_feasible(h_struct, tgt45)
    if time.time() - t0 < budget_s * 0.25:
        try_optimize(h_struct, 60, T_s=30, T_e=300, lr0=0.03, max_it=8000)

    h_rand45 = rng.uniform(0, 1, n45)
    h_rand45 = project_feasible(h_rand45, tgt45)
    if time.time() - t0 < budget_s * 0.35:
        try_optimize(h_rand45, 60, T_s=30, T_e=300, lr0=0.03, max_it=8000)

    # Phase 3: Try n=90
    n90 = 90
    tgt90 = 45.0
    h_s90 = np.zeros(n90)
    h_s90[:n90 // 2] = 1.0
    h_s90 = project_feasible(h_s90, tgt90)
    if time.time() - t0 < budget_s * 0.5:
        try_optimize(h_s90, 80, T_s=30, T_e=400, lr0=0.02, max_it=10000)

    h_r90 = rng.uniform(0, 1, n90)
    h_r90 = project_feasible(h_r90, tgt90)
    if time.time() - t0 < budget_s * 0.6:
        try_optimize(h_r90, 60, T_s=30, T_e=400, lr0=0.02, max_it=8000)

    # Phase 4: Try n=180
    n180 = 180
    tgt180 = 90.0
    h_s180 = np.zeros(n180)
    h_s180[:n180 // 2] = 1.0
    h_s180 = project_feasible(h_s180, tgt180)
    if time.time() - t0 < budget_s * 0.7:
        try_optimize(h_s180, 60, T_s=20, T_e=300, lr0=0.01, max_it=8000)

    # Phase 5: Fine-tune best with high-T smoothing (very sharp)
    if best_h is not None:
        remaining = budget_s - 10 - (time.time() - t0)
        if remaining > 20:
            h_ft, c5_ft = optimize_phase(best_h, 100, 2000, 30000, 0.005, min(remaining * 0.6, 120), rng)
            if c5_ft < best_c5:
                best_c5 = c5_ft
                best_h = h_ft

        # Local search polish
        remaining = budget_s - 5 - (time.time() - t0)
        if remaining > 5:
            best_h, best_c5 = local_search(best_h, remaining, rng)

    # Ultimate fine-tune with very small steps on true C5
    remaining = budget_s - 2 - (time.time() - t0)
    if best_h is not None and remaining > 3:
        n = len(best_h)
        target = n / 2.0
        h = best_h.copy()
        for _ in range(2000):
            if time.time() - t0 > budget_s - 2:
                break
            idx = rng.randint(n)
            for dv in [0.01, -0.01, 0.005, -0.005, 0.001, -0.001]:
                h_try = h.copy()
                h_try[idx] += dv
                h_try = project_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    break

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
