    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return compute_c5(x, dx)

    def obj_jac(x):
        return compute_grad(x, dx)

    rng = np.random.RandomState(seed)

    # Phase 1: SLSQP from initial
    try:
        h_new, c5_new = slsqp_optimize(best_h, dx, n, bounds, cons, obj, obj_jac, 500)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
    except Exception:
        pass

    # Phase 2: Coordinate refinement
    if time.time() - t0 < budget_s * 0.3:
        h_cd, c5_cd = coord_refine(best_h, dx, n, 51, 10)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 3: SLSQP from alternative starts
    alts = make_alternative_starts(n, rng)
    for alt in alts:
        if time.time() - t0 > budget_s * 0.5:
            break
        h_alt = project(alt, n)
        try:
            h_new, c5_new = slsqp_optimize(h_alt, dx, n, bounds, cons, obj, obj_jac, 500)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Phase 4: Multiple perturbation restarts around best
    for r in range(100):
        if time.time() - t0 > budget_s * 0.85:
            break
        scale = 0.02 + 0.05 * rng.random()
        h_p = project(best_h + rng.normal(0, scale, n), n)
        try:
            h_new, c5_new = slsqp_optimize(h_p, dx, n, bounds, cons, obj, obj_jac, 300)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
        if (r + 1) % 15 == 0 and time.time() - t0 < budget_s * 0.8:
            h_cd, c5_cd = coord_refine(best_h, dx, n, 51, 5)
            if c5_cd < best_c5:
                best_h, best_c5 = h_cd, c5_cd

    # Phase 5: Final tight optimization
    if time.time() - t0 < budget_s * 0.92:
        try:
            h_new, c5_new = slsqp_optimize(best_h, dx, n, bounds, cons, obj, obj_jac, 2000, 1e-14)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
        if time.time() - t0 < budget_s * 0.97:
            h_cd, c5_cd = coord_refine(best_h, dx, n, 101, 8)
            if c5_cd < best_c5:
                best_h, best_c5 = h_cd, c5_cd

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
