    diff = len(h_final) / 2.0 - h_final.sum()
    if abs(diff) > 1e-8:
        # Distribute adjustment to non-saturated entries
        mask = (h_final > 1e-6) & (h_final < 1.0 - 1e-6)
        if mask.sum() > 0:
            h_final[mask] += diff / mask.sum()
            h_final = np.clip(h_final, 0.0, 1.0)
        h_final = project(h_final, len(h_final))

    final_c5 = compute_c5(h_final, 2.0 / len(h_final))
    return h_final, final_c5, len(h_final)
