        if cr < best_c5:
            best_c5, best_h = cr, hr

    # Phase 3: SLSQP polish with high beta
    if _time_left() > 25:
        hr, cr = do_slsqp(best_h, maxiter=500, ftol=1e-15, beta=2000)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    # Phase 4: Restart loop with diverse perturbations
    n_restarts = 0
    while _time_left() > 50:
        n_restarts += 1
        strat = n_restarts % 5
        if strat == 0:
            scale = 0.2 + 0.4 * rng.random()
            h_p = project_onto_set(best_h + rng.normal(0, scale, n), ts)
        elif strat == 1:
            h_p = best_h.copy()
            nflip = int(rng.integers(2, 8))
            idx = rng.choice(n, size=nflip, replace=False)
            for i in idx:
                h_p[i] = 1.0 - h_p[i]
            h_p = project_onto_set(h_p, ts)
        elif strat == 2:
            h_p = project_onto_set(rng.uniform(0, 1, n), ts)
        elif strat == 3:
            h_p = best_h.copy()
            nswap = int(rng.integers(2, 6))
            for _ in range(nswap):
                i, j = rng.choice(n, size=2, replace=False)
                h_p[i], h_p[j] = h_p[j], h_p[i]
            h_p = project_onto_set(h_p, ts)
        else:
            h_p = best_h.copy()
            nsmall = int(rng.integers(3, 10))
            idx = rng.choice(n, size=nsmall, replace=False)
            h_p[idx] += rng.normal(0, 0.15, nsmall)
            h_p = project_onto_set(h_p, ts)

        hr, cr = do_pgd(h_p, n_iter=200)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()
            if _time_left() > 30:
                hr2, cr2 = do_slsqp(hr, maxiter=300, beta=1000)
                if cr2 < best_c5:
                    best_c5, best_h = cr2, hr2

    # Final SLSQP polish
    if _time_left() > 20:
        hr, cr = do_slsqp(best_h, maxiter=1000, ftol=1e-15, beta=3000)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    # Final PGD micro-step
    if _time_left() > 5:
        hr, cr = do_pgd(best_h, n_iter=200, lr0=0.005)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    best_h = project_onto_set(best_h, ts)
    best_c5 = c5_value(best_h, dx)
    return best_h, best_c5, n
