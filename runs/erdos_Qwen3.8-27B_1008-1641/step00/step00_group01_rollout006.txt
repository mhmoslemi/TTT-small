import numpy as np
import time
from scipy.optimize import minimize

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def c5_subgrad(h, dx, topk=1):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idxs = np.argsort(corr)[-topk:]
    grad = np.zeros(n)
    for idx in idxs:
        d = n - 1 - idx
        for i in range(n):
            jp = i + d
            jm = i - d
            if 0 <= jp < n:
                grad[i] += (1.0 - h[jp])
            if 0 <= jm < n:
                grad[i] -= h[jm]
    return grad * dx / topk

def project_feasible(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - mid, 0.0, 1.0)

def subgrad_opt(h, dx, target, niter=2000, lr0=0.01, topk=1):
    best = h.copy()
    bv = c5_value(h, dx)
    stall = 0
    for it in range(niter):
        g = c5_subgrad(h, dx, topk)
        lr = lr0 / (1.0 + 0.004 * it)
        h_new = project_feasible(h - lr * g, target)
        v = c5_value(h_new, dx)
        if v < bv:
            bv = v
            best = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                stall = 0
        h = h_new
    return best, bv

def polish_slsqp(h, dx, target, alpha=80, maxiter=200):
    n = len(h)

        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(corr - m))) / alpha
    res = minimize(obj, h, method='SLSQP',

                   options={'maxiter': maxiter, 'ftol': 1e-10})
    hp = project_feasible(res.x, target)
    return hp, c5_value(hp, dx)
