from scipy.optimize import minimize

def compute_c5(h):
    """Compute the C5 objective value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def make_feasible(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum."""
    h = np.asarray(h, dtype=float)
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # Adjust to hit target sum exactly
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Distribute adjustment
    if diff > 0:
        # Need to increase; do it where h < 1
        room = 1.0 - h
        if room.sum() < diff:
            h[:] = 1.0
            # redistribute
            h[h > 0] = 0.5  # rough fix
        else:
            h += diff * room / room.sum()
    else:
        # Need to decrease; do it where h > 0
        room = h
        if room.sum() < -diff:
            h[:] = 0.0
            h[h > 0] = 0.5
        else:
            h += diff * room / room.sum()
    h = np.clip(h, 0.0, 1.0)
    # Fine-tune

    if abs(diff) > 1e-12:

import time

_n_g = 0
_ts_g = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def _make_feasible(h, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
            else:
                h[:] = 1.0
        else:
            mask = h > 0.0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def _c_fun(x):
    return x.sum() - _ts_g

def _c_jac(x):
    return np.ones(_n_g)

def _slsqp(h0, bounds, max_iter, ftol):
    constraints = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    res = minimize(compute_c5, h0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
    if res.x is not None and np.isfinite(res.fun):
        return res.x, float(res.fun)
    return h0, compute_c5(h0)

def _coord_descent(h, n, target_sum, n_sweeps, rng):
    best = h.copy()
    best_val = compute_c5(best)
    for _sw in range(n_sweeps):
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            j = int(rng.randint(n))
            if j == i:
                j = (j + 1) % n
            lo = max(-best[i], best[j] - 1.0)
            hi = min(1.0 - best[i], best[j])
            if hi - lo < 1e-10:
                continue
            best_d = 0.0
            best_f = best_val
            deltas = np.linspace(lo, hi, 15)
            for d in deltas:
                if d == 0.0:
                    continue
                test = best.copy()
                test[i] = best[i] + d
                test[j] = best[j] - d
                if test[i] < -1e-12 or test[i] > 1 + 1e-12 or test[j] < -1e-12 or test[j] > 1 + 1e-12:
                    continue
                f = compute_c5(test)
                if f < best_f:
                    best_f = f
                    best_d = d
            if best_d != 0.0:
                best[i] += best_d
                best[j] -= best_d
                best_val = best_f
                improved = True
        if not improved:
            break
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    global _n_g, _ts_g, initial_h_values
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.full(95, 0.5)

    n = len(best_h)
    _n_g = n
    target_sum = n / 2.0
    _ts_g = target_sum
    best_h = _make_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP from multiple perturbations
    phase1_end = t0 + budget_s * 0.55
    scales = [0.0, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.25]
    trial = 0
    while elapsed() < phase1_end:
        scale = scales[trial % len(scales)]
        if scale == 0.0:
            h0 = best_h.copy()
        else:
            h0 = best_h + rng.randn(n) * scale
        h0 = _make_feasible(h0, target_sum)
        try:
            h_new, c5_new = _slsqp(h0, bounds, 800, 1e-13)
            h_new = _make_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
        trial += 1

    # Phase 2: Coordinate descent refinement
    phase2_end = t0 + budget_s * 0.82
    while elapsed() < phase2_end:
        h_new, c5_new = _coord_descent(best_h, n, target_sum, 30, rng)
        h_new = _make_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new
        else:
            break

    # Phase 3: Final SLSQP polish with very tight tolerance
    while elapsed() < budget_s - 15:
        try:
            h_new, c5_new = _slsqp(best_h, bounds, 3000, 1e-15)
            h_new = _make_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new
            else:
                break
        except Exception:
            break

    # One more round of coord descent if time allows
    if elapsed() < budget_s - 10:
        h_new, c5_new = _coord_descent(best_h, n, target_sum, 50, rng)
        h_new = _make_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
