import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

# Initial construction from initial_h_values (n=45)
initial_h = initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the initial number of points
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the C5 value for a given h
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initial guess
    h_initial = initial_h
    
    # Optimization bounds
    bounds = Bounds(0, 1)
    
    # Optimization constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = LinearConstraint(lambda h: np.sum(h), 0, 0, dtype=np.float64)
    
    # Optimization setup
    result = minimize(
        fun=compute_c5,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={
            'ftol': 1e-8,
            'maxiter': 100,
            'disp': False
        }
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check if we have more than 1000 points, if so, reduce to 1000 by interpolation
    if len(best_h) > 1000:
        # Downsample to 1000 points
        indices = np.linspace(0, len(best_h) - 1, 1000, endpoint=False).astype(int)
        best_h = best_h[indices]
        n_points = 1000
        dx = 2.0 / n_points
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Return the best solution
    return (best_h, final_c5, len(best_h))
