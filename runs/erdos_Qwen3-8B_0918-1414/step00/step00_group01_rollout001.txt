import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with a reasonably sized grid
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Initialize with a simple step function that has equal area
    for i in range(n_points):
        h[i] = 1 if i < n_points / 2 else 0
    # Ensure the sum of h is n_points / 2
    h = h / (np.sum(h) / (n_points / 2))
    # Function to compute the overlap integral
    def compute_C5(h):
        h = np.tile(h, (n_points, 1))
        shifted = np.roll(h, shift=range(n_points))
        overlap = np.sum(h * (1 - shifted), axis=1)
        return np.max(overlap) * dx
    # Optimization function to minimize C5
    def objective(h):
        return compute_C5(h)
    # Define a bounds array for h in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    # Use Nelder-Mead to minimize C5
    result = optimize.minimize(
        objective,
        x0=h,
        bounds=bounds,
        method='Nelder-Mead',
        options={'xatol': 1e-5, 'maxiter': 100}
    )
    best_h = result.x
    best_c5 = compute_C5(best_h)
    return best_h, best_c5, n_points
