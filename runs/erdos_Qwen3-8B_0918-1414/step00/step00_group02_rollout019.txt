import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are pre-imported

def run(seed=42, budget_s=60):
    # Set random seed
    np.random.seed(seed)
    
    # Precompute initial guess
    n_points = 45
    initial_h_values = np.array([0.5 for _ in range(n_points)])
    initial_h_values = np.array(initial_h_values)
    
    # Strategy: Use a combination of local search with gradient descent and convex optimization
    # to refine the step function h to minimize the overlap integral C_5, while satisfying the constraints.
    
    # First, try to refine the initial guess with a local optimization method
    # Define the function to minimize: the maximum overlap integral
    # We will use a numerical approximation of the overlap integral
    
    def compute_C5_values(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        max_overlap = 0.0
        
        # Compute all shifted versions of h
        overlaps = []
        for k in range(1, n // 2 + 1):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps.append(overlap)
        
        return max(overlaps)
    
    # Local optimization function
    def local_optimize(initial_h, max_iter=100, tol=1e-5):
        h = np.array(initial_h)
        for _ in range(max_iter):
            # Compute the overlap integral
            current_C5 = compute_C5_values(h)
            
            # Compute the gradient (approximate) of C5 with respect to h
            # We'll use finite differences
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                grad[i] = (compute_C5_values(h_plus) - compute_C5_values(h_minus)) / (2 * 1e-5)
            
            # Update h using gradient descent
            h -= 0.01 * grad
            
            # Ensure constraints: h ∈ [0, 1], sum(h) ≈ n_points / 2
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            h = h * (n_points / 2) / total if total != 0 else h
        
        return h
    
    # Try refining with local optimization
    refined_h = local_optimize(initial_h_values)
    best_C5 = compute_C5_values(refined_h)
    best_h = refined_h.copy()
    
    # Try to improve further using convex optimization
    # Define the problem: minimize the max overlap
    # We'll use a convex relaxation
    
    def convex_optimize(h_init):
        n = len(h_init)
        dx = 2.0 / n
        h = cp.Variable(n)
        constraints = [
            cp.sum(h) == n / 2,
            0 <= h,
            h <= 1
        ]
        
        # Define the overlap for all shifts
        max_overlap = 0.0
        for k in range(1, n // 2 + 1):
            shifted_h = cp.roll(h, k)
            overlap = cp.sum(h * (1 - shifted_h)) * dx
            max_overlap = cp.Max(max_overlap, overlap)
        
        # Objective: minimize the maximum overlap
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        result = prob.solve(solver=cp.SCS, eps=1e-5)
        
        if result is not None:
            return h.value
        else:
            return h_init
    
    # Try convex optimization
    convex_h = convex_optimize(refined_h)
    convex_C5 = compute_C5_values(convex_h)
    
    if convex_C5 < best_C5:
        best_C5 = convex_C5
        best_h = convex_h
    
    # We are under time budget, so return the best solution found
    return (best_h, best_C5, len(best_h))
