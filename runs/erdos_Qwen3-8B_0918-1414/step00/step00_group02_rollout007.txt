    def objective(h):
        nonlocal n_points, dx
        h = h.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = np.concatenate([h, h])  # To handle circular convolution
        corr = np.correlate(h, 1 - h[:n_points], mode='full')
        c5_candidate = np.max(corr) * dx
        return c5_candidate
    
    # Constraints: sum(h) == n_points / 2
    def constraint(h):
        nonlocal n_points
        return np.sum(h) - n_points / 2
    
    # Optimization using a reduced set of points
    n_points_new = 200
    h_start = np.random.rand(n_points_new) * 2 - 1  # Initial guess
    h_start = np.clip(h_start, 0.0, 1.0)

    def objective_with_h(h):
        nonlocal n_points_new, dx
        h = h.reshape(n_points_new)
        h = np.clip(h, 0.0, 1.0)
        h = np.concatenate([h, h])  # To handle circular convolution
        corr = np.correlate(h, 1 - h[:n_points_new], mode='full')
        c5_candidate = np.max(corr) * dx
        return c5_candidate
    
    def constraint_with_h(h):
        nonlocal n_points_new

    best_h = np.concatenate([best_h, best_h])

    best_c5 = np.max(corr) * (2.0 / n_points_new)
