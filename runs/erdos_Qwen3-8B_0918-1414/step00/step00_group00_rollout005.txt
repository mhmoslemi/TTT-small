import numpy as np
import scipy.optimize as optimize
import time
import itertools
from scipy.optimize import minimize
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    # Initial parameters
    n_points = 100  # Start with a moderate number of points
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = h[:n_points]  # Ensure it's within the n_points limit

    def compute_C5(h):
        # Ensure h is of length n_points
        h = h[:n_points]
        h = np.pad(h, (0, 2), 'constant')  # Pad to handle full convolution
        result = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        return np.max(result * dx)

    # Define the objective function to minimize
    def objective(params):
        h = params.reshape(n_points)
        # Ensure h is within [0,1]
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        integral = np.sum(h) * (2.0 / n_points)
        # Adjust h to meet the constraint
        if integral < 1 - 1e-8:
            # Need to increase the integral
            delta = (1 - integral) / n_points
            h += delta
        elif integral > 1 + 1e-8:
            # Need to decrease the integral
            delta = (integral - 1) / n_points
            h -= delta
        h = np.clip(h, 0, 1)
        return compute_C5(h)

    def constraint(params):
        h = params.reshape(n_points)
        return np.sum(h) - n_points / 2

    # Initial guess
    initial_guess = h.copy()

    # Solve the optimization problem
    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint}]
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h[:n_points]
    best_C5 = compute_C5(best_h)

    return (best_h, best_C5, n_points)
