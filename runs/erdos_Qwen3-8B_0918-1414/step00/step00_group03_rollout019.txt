import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for a balance between speed and accuracy
    h = np.random.uniform(low=0, high=1, size=n_points)
    # Ensure the integral constraint is satisfied
    h /= np.sum(h) * (2.0 / n_points)
    
    # Objective function to minimize max overlap
    def objective(h_values):
        h = h_values
        dx = 2.0 / len(h)
        conv = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        fun=objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'gtol': 1e-5, 'ftol': 1e-5}
    )
    
    # Check elapsed time and stop if needed
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return result.x, result.fun, n_points
    
    # Final check and return
    final_h = result.x
    final_c5 = result.fun
    return final_h, final_c5, n_points

# Dummy start time for budget tracking
start_time = time.time()
