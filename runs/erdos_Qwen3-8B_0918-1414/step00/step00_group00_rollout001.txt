        return np.sum(h) - n_points / 2

    result = minimize(

        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'disp': False, 'maxiter': 1000}
    )

    # Extract the best solution
    h_values = result.x
    c5_bound = objective(h_values)
