import numpy as np

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros_like(h)
    for k in range(1, n_points):
        shifted_h[k:] = h[:n_points - k]
        shifted_h[:k] = 0
        overlap = np.sum(h * (1 - shifted_h)) * dx
    return np.max(overlap)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 256  # Choose a moderate number of points for balance between accuracy and speed
    dx = 2.0 / n_points
    
    # Initial guess
    h = initial_h_values()
    h = h[:n_points]  # Ensure length matches
    h = h / np.sum(h) * (n_points / 2)  # Normalize to satisfy the integral constraint
    
    # Define the optimization problem
    def objective(x):
        h = x.reshape(n_points)
        # Enforce the integral constraint
        integral = np.sum(h) * dx
        # Compute the overlap for all k
        overlaps = []
        for k in range(1, n_points):
            shifted_h = np.zeros_like(h)
            shifted_h[k:] = h[:n_points - k]
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)
    
    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2  # Enforce the integral constraint
    
    # Set up bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    # Perform optimization
    result = opt.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 200, 'ftol': 1e-8})
    
    # Extract best solution
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    return best_h, best_c5, n_points
