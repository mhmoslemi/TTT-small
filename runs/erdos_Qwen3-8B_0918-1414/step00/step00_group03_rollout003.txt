import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import eigs
from time import time

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Placeholder, assuming this is preloaded

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Starting from the initial construction
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral C_5
    def compute_c5(h):
        h = h.reshape(-1)
        n = len(h)
        dx = 2.0 / n
        shift = np.arange(-n + 1, n)
        overlap = np.zeros_like(shift)
        for i in range(n):
            shifted = np.roll(h, i)
            overlap[i] = np.sum(h * (1 - shifted)) * dx
        return np.max(overlap)
    
    # Objective function for optimization
    def objective(x):
        h = x.reshape(-1)
        return compute_c5(h)
    
    # Constraints: sum(h) == n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = h
    
    # Optimization
    result = minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'maxiter': 100, 'ftol': 1e-8})
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Evaluate with a smaller n_points for faster computation
    n_points = 500
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 0.5  # Simple initial guess
    
    # Simple grid search over possible h values to find better C_5
    for i in range(n_points):
        for j in range(n_points):
            h[i] = 0.5
            h[j] = 0.5
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    # Final evaluation
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
