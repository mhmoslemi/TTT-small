import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    max_time = budget_s
    start_time = time.time()

    # Initial guess
    n_points = 100  # small n_points for faster evaluation
    h_initial = initial_h_values[:n_points]  # Use initial h_values as starting point
    h_initial = np.clip(h_initial, 0, 1)  # Ensure values are in [0,1]
    h_initial = h_initial * (n_points / 2)  # Adjust to meet integral constraint

    def objective(h):
        # Compute the overlap integral for all k
        h = np.clip(h, 0, 1)
        h = np.concatenate((h, h))  # Double the array for full convolution
        corr = np.correlate(h, 1 - h, mode="full")
        result = np.max(corr) * (2.0 / n_points)  # dx = 2.0 / n_points
        return result

    def bounds_func(h):
        # Ensure h is in [0, 1]
        return np.clip(h, 0, 1)

    # Optimization setup
    result = optimize.minimize(
        objective,
        x0=h_initial,
        bounds=[(0, 1) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    best_h = result.x
    best_c5 = objective(best_h)

    # Final check and return
    elapsed_time = time.time() - start_time
    if elapsed_time > max_time:
        print("Time limit exceeded, returning best solution found so far")
    return best_h, best_c5, n_points
