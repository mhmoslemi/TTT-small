import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from scipy.integrate import trapz
from scipy.signal import correlate
import time

# Pre-imported functions
def evaluate_erdos_solution(h, n_points):
    """Evaluates the Erdos constant for a given step function h."""
    dx = 2.0 / n_points
    h_array = np.array(h)
    shifted = np.zeros_like(h_array, dtype=float)
    for k in range(1, len(h_array) - 1):
        shifted[k:] = h_array[:len(h_array) - k]
    overlap = correlate(h_array, 1 - shifted, mode='full')
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values(n_points):
    """Initial step function values for n_points samples."""
    # Initial construction with n=47 samples
    # This is a placeholder for the actual initial values
    # For the purpose of this example, we'll return a simple construction
    h = np.zeros(n_points)
    h[:n_points//2] = 1
    return h.tolist()

def run(seed=42, budget_s=60):
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    
    # Initial guess
    h_initial = np.array(initial_h, dtype=float)
    
    # Objective function: minimize the maximum overlap
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        overlap = correlate(h, 1 - h, mode='full')
        return np.max(overlap) * dx

    # Constraint: sum(h) = n_points / 2
    def constraint(h_flat):
        return np.sum(h_flat) - n_points / 2

    # Bounds: each h[i] ∈ [0, 1]
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    # Optimization
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    h_opt = result.x
    c5 = evaluate_erdos_solution(h_opt, n_points)
    return h_opt.tolist(), c5, n_points
