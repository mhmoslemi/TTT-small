import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
import time

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    def initial_h_values(n):
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0.25
        return h
    
    def evaluate_erdos_solution(h):
        n = len(h)
        dx = 2.0 / n
        result = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(result) * dx
        return max_overlap
    
    def objective_func(h, dx, n):
        result = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(result) * dx
        return max_overlap
    
    def constraint_func(h, dx, n):
        return np.sum(h) - n / 2
    
    def finite_diff_derivative(h, dx, n, i):
        h_dup = h.copy()
        h_dup[i] += 1e-6
        return (objective_func(h_dup, dx, n) - objective_func(h, dx, n)) / 1e-6
    
    n_points = 200
    dx = 2.0 / n_points
    h_initial = initial_h_values(n_points)
    
    def optimize_h():
        h = h_initial.copy()
        h = h.astype(float)
        
        max_iter = 500
        lr = 0.1
        best_c5 = float('inf')
        best_h = h.copy()
        
        for it in range(max_iter):
            start_time = time.time()
            if it == 0:
                grad = np.zeros(n_points)
                for i in range(n_points):
                    grad[i] = finite_diff_derivative(h, dx, n_points, i)
            else:
                grad = np.zeros(n_points)
                for i in range(n_points):
                    grad[i] = finite_diff_derivative(h, dx, n_points, i)
            
            h -= lr * grad
            h = np.clip(h, 0, 1)
            
            c5 = objective_func(h, dx, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            if time.time() - start_time > budget_s - 1:
                break
        
        return best_h, best_c5
    
    best_h, best_c5 = optimize_h()
    return best_h, best_c5, n_points
