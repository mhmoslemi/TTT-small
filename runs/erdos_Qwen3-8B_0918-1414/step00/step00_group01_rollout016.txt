```python
import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def compute_overlap_integral(h, dx):
    n = len(h)
    overlap = np.zeros(n - 1)
    for k in range(1, n - 1):
        shifted = np.roll(h, k)
        overlap[k] = np.sum(h * (1 - shifted)) * dx
    return np.max(overlap)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    # Initial sample from initial_h_values
    initial_h_values = np.load("initial_h_values.npy")
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()

    # Function to compute C5
    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        overlap = compute_overlap_integral(h_vals, dx)
        return overlap

    # Optimization with bounds and constraints
    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.minimize(
        objective,
        h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='SLSQPLib',
        options={'maxiter': 1000, 'ftol': 1e-8}
    )

    best_h = result.x
    best_c5 = compute_overlap_integral(best_h, dx)

    return best_h, best_c5, len(best_h)
```
