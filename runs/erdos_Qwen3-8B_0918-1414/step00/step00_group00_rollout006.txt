import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h_array = np.array(h)
        shifted_h = np.concatenate((h_array[-2:], h_array[:-2]))  # shift by 2 (periodic)
        corr = np.correlate(h_array, 1 - shifted_h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def optimize_h(n_points):
        # Define variables
        h = cp.Variable(n_points, nonneg=True, boolean=True)
        obj = cp.Minimize(cp.Max(cp.sum(h * (1 - h[2:])) * dx))
        
        # Constraints
        constraints = [cp.sum(h) == n_points / 2]
        
        # Solve
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CBC, time_limit=budget_s)
            if prob.status == 'optimal':
                return h.value
            else:
                return initial_h
        except:
            return initial_h
    
    # Start with initial guess
    h_values = initial_h
    
    # Try optimizing with different n_points
    for n in [n_points, n_points - 10, n_points - 20, n_points - 30, n_points - 40]:
        if n < 10:
            continue
        try:
            h_opt = optimize_h(n)
            if h_opt is not None:
                h_values = h_opt
                c5_bound = compute_c5(h_values)
                break
        except:
            continue
    
    # Final evaluation
    c5_bound = compute_c5(h_values)
    return h_values, c5_bound, len(h_values)
