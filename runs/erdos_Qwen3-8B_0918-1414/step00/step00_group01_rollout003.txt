def run(seed=42, budget_s=60, **kwargs):
    n_points = 200  # Start with 200 samples, can be adjusted if needed
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h * (n_points / 2)  # Ensure sum(h) == n_points / 2
    h = h / np.sum(h) * (n_points / 2)
    best_c5 = float('inf')
    best_h = h.copy()
    elapsed_time = 0

    # Define the objective function to minimize C5
    def objective(h_values):
        nonlocal best_c5
        h = np.array(h_values)
        max_overlap = 0.0
        for k in range(1, n_points):
            shifted_h = np.zeros_like(h)
            shifted_h[k:] = h[:-k]
            shifted_h[:k] = h[-k:]
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    # Define bounds for h_values
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization settings
    options = {
        'disp': False,
        'ftol': 1e-6,
        'maxiter': 1000,
        'gtol': 1e-6
    }

    # Start the optimization
    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        options=options
    )
    elapsed_time = time.time() - start_time

    # Check if result is successful
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_c5 = float('inf')

    # Evaluate the best solution
    best_c5 = evaluate_erdos_solution(best_h)
