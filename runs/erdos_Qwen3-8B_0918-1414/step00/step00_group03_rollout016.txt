import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
from scipy.optimize import minimize
import time
from functools import partial

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # n=70

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Precompute all possible shifts for the given interval [0, 2]
    n_points = 100  # We'll start with 100 points for faster computation
    dx = 2.0 / n_points
    x_values = np.linspace(0, 2, n_points)
    
    # Initial guess: use the preloaded initial_h_values and resample it to n_points
    def resample_h(initial_h, n_points):
        initial_h = np.array(initial_h)
        initial_h = initial_h[::len(initial_h)//n_points]
        initial_h = np.pad(initial_h, (0, n_points - len(initial_h)))
        return initial_h
    
    initial_h = resample_h(initial_h_values, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_C5(h):
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        shifted_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted_h[i, :] = h[(i + n_points - 1) % n_points]
        overlap = np.dot(h, 1 - shifted_h)
        return np.max(overlap * dx)
    
    # Objective function: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization using L-BFGS-B
    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=cons,
        options={'maxiter': 100, 'disp': False},
        tol=1e-6
    )
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    return best_h, best_C5, n_points
