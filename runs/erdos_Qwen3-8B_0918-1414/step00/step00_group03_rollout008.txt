import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.roll(initial_h, int(n_points / 2))
        initial_h = np.minimum(1.0, np.maximum(0.0, initial_h))
        initial_h = np.roll(initial_h, int(n_points / 4))
        initial_h = (initial_h * n_points) / 2
        initial_h = initial_h[:n_points]
    
    # Ensure the sum is exactly n_points / 2
    initial_h = np.minimum(1.0, np.maximum(0.0, initial_h))
    initial_h = np.roll(initial_h, int(n_points / 4))
    initial_h = (initial_h * n_points) / 2
    initial_h = initial_h[:n_points]
    
    def compute_C5(h):
        h = np.minimum(1.0, np.maximum(0.0, h))
        h = np.roll(h, int(n_points / 4))
        h = (h * n_points) / 2
        h = h[:n_points]
        
        # Compute the overlap integral
        c5_values = np.zeros(n_points - 1)
        for k in range(1, n_points - 1):
            h_k = np.roll(h, k)
            overlap = np.sum(h * (1 - h_k)) * dx
            c5_values[k] = overlap
        return np.max(c5_values)
    
    # Define the objective function
    def objective(h):
        return compute_C5(h)
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    initial_guess = initial_h
    
    # Optimization
    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})
    
    best_h = result.x
    best_c5 = compute_C5(best_h)
    
    return (best_h, best_c5, n_points)
