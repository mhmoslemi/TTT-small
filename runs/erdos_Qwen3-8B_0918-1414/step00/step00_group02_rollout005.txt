import numpy as np
import scipy.optimize
import time
from scipy.optimize import Bounds, minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    return np.loadtxt("initial_h_values.txt")

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Starting with the initial construction
    max_iter = 100
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Load initial solution
    try:
        initial_h = initial_h_values()
        initial_h = initial_h[:n_points]
        n_points = len(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        initial_sum = np.sum(initial_h) * (2.0 / n_points)
        delta = (1.0 - initial_sum) / n_points
        initial_h = np.clip(initial_h + delta, 0, 1)
    except:
        initial_h = np.random.rand(n_points)
        initial_sum = np.sum(initial_h) * (2.0 / n_points)
        delta = (1.0 - initial_sum) / n_points
        initial_h = np.clip(initial_h + delta, 0, 1)
    
    # Function to evaluate a solution
    def evaluate(solution):
        nonlocal best_c5
        h = np.clip(solution, 0, 1)
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Function to perform local search
    def local_search(solution, lr=0.01, max_steps=100):
        for _ in range(max_steps):
            perturb = np.random.normal(0, lr, n_points)
            perturb = np.clip(perturb, -0.1, 0.1)
            new_solution = np.clip(solution + perturb, 0, 1)
            new_sum = np.sum(new_solution) * (2.0 / n_points)
            delta = (1.0 - new_sum) / n_points
            new_solution = np.clip(new_solution - delta, 0, 1)
            if evaluate(new_solution) < best_c5:
                solution = new_solution.copy()
        return solution
    
    # Function to perform gradient descent
    def gradient_descent(solution, lr=0.01, max_iter=100):
        for _ in range(max_iter):
            h = solution
            dx = 2.0 / n_points
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            loss = max_corr * dx
            # Compute derivative of loss with respect to h
            # This is a simplified approximation
            grad = np.zeros(n_points)
            for i in range(n_points):
                for k in range(n_points):
                    if k + i < n_points:
                        grad[i] += (corr[k + i] - corr[k + i + 1]) * dx
                    if k - i >= 0:
                        grad[i] += (corr[k - i] - corr[k - i - 1]) * dx
            grad = np.clip(grad, -1e6, 1e6)
            solution -= lr * grad
            solution = np.clip(solution, 0, 1)
            new_sum = np.sum(solution) * dx
            delta = (1.0 - new_sum) / n_points
            solution = np.clip(solution - delta, 0, 1)
        return solution
    
    # Perform optimization
    solution = initial_h.copy()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        solution = local_search(solution)
        solution = gradient_descent(solution)
        if best_c5 < 0.380876:
            break
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    return best_h, final_c5, n_points
