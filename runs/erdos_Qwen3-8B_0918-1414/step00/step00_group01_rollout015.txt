import numpy as np
import scipy.optimize as optimize
import time
import sys
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial candidate from precomputed values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Compute current C5
    def compute_C5(h):
        h = np.array(h)
        n_points = len(h)
        dx = 2.0 / n_points
        h_shifted = np.roll(h, shift=1)
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Current C5
    current_c5 = compute_C5(initial_h)
    print(f"Initial C5: {current_c5}")
    
    # Define a function to optimize
    def objective(x):
        # x is a vector of h values
        h = x.copy()
        h[h < 0] = 0
        h[h > 1] = 1
        return compute_C5(h)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B optimization with some randomness to escape local optima
    result = optimize.minimize(
        objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,
        options={'gtol': 1e-5, 'maxiter': 1000}
    )
    
    best_h = result.x
    best_c5 = compute_C5(best_h)
    
    # Try a few random perturbations to escape local optima
    for _ in range(5):
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        candidate_h = best_h + perturb
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_c5 = compute_C5(candidate_h)
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = candidate_h
    
    # Final result
    print(f"Optimized C5: {best_c5}")
    return best_h, best_c5, n_points
