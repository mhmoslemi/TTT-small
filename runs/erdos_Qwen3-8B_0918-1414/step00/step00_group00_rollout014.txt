    def compute_C5(h):

        h_array = np.array(h)
        shifted_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted_h[i] = h_array[i:]  # Slide h over the domain
        convolved = np.zeros(n_points)
        for k in range(1, n_points // 2 + 1):
            overlap = np.sum(h_array * (1 - shifted_h[:, k]))
            convolved[k] = overlap
        max_overlap = np.max(convolved)
        return max_overlap * dx
    
    # Objective function to minimize
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i] (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use Sequential Least Squares Programming (SLSQP) to minimize C5
    result = optimize.minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='SLSQP',
        options={'disp': False, 'ftol': 1e-10, 'maxiter': 500}
    )
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    # Ensure all h values are within [0, 1] (due to numerical precision)
