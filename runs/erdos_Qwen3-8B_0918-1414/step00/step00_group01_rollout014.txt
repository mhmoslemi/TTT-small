import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import matplotlib.pyplot as plt

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Assume this contains the initial guess
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    n_points = 100  # Start with 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    total_area = n_points / 2  # Since sum(h) * dx = 1 => sum(h) = n_points / 2

    # Initial guess
    h = initial_h_values[:n_points]
    if len(h) < n_points:
        h = np.concatenate([h, np.zeros(n_points - len(h))])
    h = h[:n_points]
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Ensure area is correct

    # Objective function: minimize the maximum overlap integral
    def objective(x):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx

    # Constraint: sum(h) == n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2

    # Bounds for each sample
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
    h_opt = result.x

    # Final check
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    overlaps = np.correlate(h_opt, 1 - h_opt, mode="full")
    c5_bound = np.max(overlaps) * dx

    return h_opt, c5_bound, n_points
