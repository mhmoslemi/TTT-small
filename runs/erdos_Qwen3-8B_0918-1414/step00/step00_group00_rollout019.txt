import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

# Pre-imported helper functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    
    # Define the objective function to minimize C5
    def objective(x):
        h_new = x
        dx = 2.0 / n_points
        overlap = np.correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Define constraints
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Initial guess
    x0 = h.copy()
    
    # Optimization settings
    options = {'disp': False, 'maxiter': 100, 'gtol': 1e-8}
    
    # Run the optimizer
    result = minimize(objective, x0, method='SLSQP', constraints=cons, options=options)
    h_opt = result.x
    
    # Compute C5 for the optimized solution
    dx = 2.0 / n_points
    overlap = np.correlate(h_opt, 1 - h_opt, mode="full")
    c5_bound = np.max(overlap * dx)
    
    return (h_opt.tolist(), c5_bound, n_points)
