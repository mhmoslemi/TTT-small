import numpy as np
import scipy.optimize as optimize
import time
import sys

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    max_budget = budget_s
    start_time = time.time()
    
    # Number of points to use
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initial guess based on initial_h_values
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        h = np.interp(np.linspace(0, 2, n_points), np.sort(initial_h[:, 0]), initial_h[:, 1])
    else:
        # Default: uniform distribution with sharp transitions
        h = np.zeros(n_points)
        h[:(n_points // 2)] = 1.0
        h[(n_points // 2):] = 0.0
    
    # Ensure the integral condition is satisfied
    integral = np.sum(h) * dx
    h = h * (1.0 / integral) * (n_points / 2.0)
    
    def compute_c5(h_vals):
        h = h_vals
        h = np.clip(h, 0, 1)
        h_extended = np.concatenate([h, h, h])
        result = np.correlate(h_extended, 1 - h, mode='full')
        conv = result[n_points - 1 : 2 * n_points - 1]
        conv = conv * dx
        return np.max(conv)
    
    # Optimization: minimize C5 with constraints
    def objective(x):
        return compute_c5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Bounds: each h[i] is in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Perform optimization
    result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-10, 'maxiter': 50})
    
    h_best = result.x
    c5_best = compute_c5(h_best)
    
    # Ensure we stay within budget
    elapsed = time.time() - start_time
    if elapsed >= max_budget:
        print(f"Timeout exceeded. Returning last best solution.")
        return (h_best, c5_best, n_points)
    
    return (h_best, c5_best, n_points)
