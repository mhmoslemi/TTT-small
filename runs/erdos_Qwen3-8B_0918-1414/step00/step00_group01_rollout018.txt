```python
import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Define initial guess
    h_initial = initial_h_values
    n_points = len(h_initial)
    
    # Define objective function to minimize overlap integral
    def objective(h):
        h = np.array(h)
        conv = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(conv * (2.0 / n_points))
        return overlap
    
    # Define constraints for optimization
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    method = 'L-BFGS-B'
    options = {'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    
    # Run optimization
    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=h_initial,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method=method,
        options=options
    )
    
    # Extract best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure the best solution meets the constraints
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-8:
        best_h = np.array(h_initial)
        best_c5 = objective(best_h)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print(f"Time budget exceeded. Returning initial solution.")
        return (h_initial, objective(h_initial), n_points)
    
    return (best_h, best_c5, n_points)
```
