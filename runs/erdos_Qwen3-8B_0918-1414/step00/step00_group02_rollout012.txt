import numpy as np
import scipy.optimize as optimize
import time
from scipy.sparse import diags
from scipy.optimize import LinearConstraint, Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Initial number of points
    n_points = 200  # Start with 200 points for efficiency and quality balance
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to ensure ∫h(x)dx = 1

    # Smooth the initial guess with a lowpass filter
    h = np.convolve(h, np.ones(5) / 5, mode='same')

    # Define the objective function to minimize the maximum overlap
    def objective(h):
        h = h.reshape(-1)
        max_overlap = -np.inf
        for k in range(1, n_points):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    # Define constraints for the problem:
    # 1. All h_i ∈ [0, 1]
    # 2. ∑h_i * dx = 1 → ∑h_i = n_points / 2

    # Set up the optimization problem
    bounds = Bounds([0.0] * n_points, [1.0] * n_points)
    A_eq = np.eye(n_points)
    c_eq = np.array([n_points / 2.0])

    # Define the objective to minimize the maximum overlap
    def objective_to_minimize(h):
        return objective(h)

    # Use a local optimizer with bounds and constraints
    result = optimize.minimize(
        fun=objective_to_minimize,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=[LinearConstraint(A_eq, c_eq, c_eq)],
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = objective(best_h)

    # Check if we can reduce n_points further while maintaining quality
    if best_c5 < 0.38:
        # Try reducing n_points to 100
        n_points = 100
        dx = 2.0 / n_points
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)

        # Re-evaluate with reduced n_points
        best_c5 = objective(best_h)

    # Check if further reduction is possible
    if best_c5 < 0.38:
        n_points = 50
        dx = 2.0 / n_points
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = objective(best_h)

    # Final evaluation
    final_h = best_h
    final_c5 = best_c5
    return final_h, final_c5, len(final_h)
