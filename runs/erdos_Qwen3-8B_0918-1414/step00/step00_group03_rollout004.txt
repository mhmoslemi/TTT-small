    # Let's try with 200 points for better resolution
    
    n_points = 200
    dx = 2.0 / n_points
    total_sum = n_points / 2
    
    # Start with a uniform initial guess
    h = np.full(n_points, 0.5)
    h[:n_points // 2] = 1.0
    h[n_points // 2:] = 0.0
    h = h / 2.0
    
    def compute_overlap(h):
        h = h.reshape(-1)
        n = len(h)
        overlap = np.zeros(n)
        for k in range(1, n // 2 + 1):
            shifted = np.roll(h, k)
            overlap[k] = np.sum(h * (1 - shifted)) * dx
        return np.max(overlap)
    
    def objective(x):
        x = x.reshape(-1)
        x = np.clip(x, 0, 1)
        return compute_overlap(x)
    
    def constraint(x):
        x = x.reshape(-1)
        return np.sum(x) - total_sum
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B for faster optimization
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'disp': False, 'maxcor': 10})
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    c5_bound = compute_overlap(best_h)
    
    return best_h, c5_bound, n_points
