    # Evaluate the initial C5
    def compute_C5(h):
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        dx = 2.0 / n_points
        shifted = np.roll(h, shift=1)
        overlap = np.correlate(h, 1 - shifted, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def compute_overlap_integral(h):
        h = np.clip(h, 0, 1)
        dx = 2.0 / len(h)
        shifted = np.roll(h, shift=1)
        overlap = np.correlate(h, 1 - shifted, mode="full")
        return np.max(overlap) * dx

    # Optimization strategy:
    # We'll perform a gradient-based optimization with constraints
    # and refine it with a local search.

    # Initial guess
    h_initial = initial_h

    # Define the objective: minimize the maximum overlap
    def objective(x):
        return compute_overlap_integral(x)

    # Define constraints
    # 1. h in [0, 1]
    # 2. Integral of h over [0, 2] is 1 (i.e., sum(h) * dx = 1)
    # So sum(h) = n_points / 2

    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Define constraint: sum(h) = n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Use a local optimization method
    result = optimize.minimize(

        x0=h_initial,
        bounds=bounds,
        constraints=cons,

    best_h = result.x
