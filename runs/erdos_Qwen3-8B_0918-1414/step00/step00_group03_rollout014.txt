import numpy as np
import scipy.optimize as opt
import time
import warnings

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial_h_values
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.concatenate([h[-2:], h])
    conv = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(conv) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial guess (starting from the given initial_h_values)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points

    # Objective function: maximize overlap integral
    def objective(h_values):
        return evaluate_erdos_solution(h_values)

    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization with bounds
    result = opt.minimize(
        objective,
        h,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 500}
    )

    # Check if we have a valid solution
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
        return (best_h, best_c5, n_points)
    else:
        # If optimization fails, use initial guess
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
        return (best_h, best_c5, n_points)
