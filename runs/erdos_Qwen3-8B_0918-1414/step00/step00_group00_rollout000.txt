import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.array(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted[k:] = h[:len(h) - k]
        overlap = np.sum(h * (1 - shifted)) * dx
        return np.max(overlap)
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.all((0 <= h) & (h <= 1))
    
    def minimize_c5(h0):
        result = optimize.minimize(
            objective,
            h0,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': constraint_sum},
                {'type': 'ineq', 'fun': constraint_bounds}
            ],
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        return result.x
    
    start_time = time.time()
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    for _ in range(10):
        if time.time() - start_time > budget_s - 5:
            break
        h = minimize_c5(best_h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
    
    return (best_h, best_c5, len(best_h))
