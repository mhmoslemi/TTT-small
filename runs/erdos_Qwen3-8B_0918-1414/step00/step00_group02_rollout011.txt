import numpy as np

from scipy.optimize import LinearConstraint, Bounds

def run(seed=42, budget_s=60, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    
    # Try several starting points, including the initial one, to avoid local optima
    start_points = [h_initial]
    for _ in range(3):
        perturbed = h_initial + np.random.normal(0, 0.05, n_points_initial)
        perturbed = np.clip(perturbed, 0, 1)
        start_points.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points_initial
    
    for start in start_points:
        start = np.clip(start, 0, 1)
        
        # Define the function to minimize
        def objective(x):
            nonlocal best_c5
            h = x
            h = np.clip(h, 0, 1)
            # Compute the overlap integral for each k
            overlap = []
            for k in range(1, len(h) - 1):
                shifted = np.roll(h, k)
                integral = np.sum(h * (1 - shifted)) * (2.0 / len(h))
                overlap.append(integral)
            c5 = max(overlap)
            return c5
        
        # Constraints: sum(h) = n_points / 2
        n_points = len(start)
        constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
        
        # Bounds for h: [0, 1]
        bounds = Bounds(0, 1)
        
        # Initial guess
        x0 = start
        
        # Optimization
        start_time = time.time()
        result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraint, options={'ftol': 1e-8, 'maxiter': 100})
        
        if time.time() - start_time > budget_s:
            print("Time budget exceeded")
            break
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n_points = n_points
        
    return (best_h, best_c5, best_n_points)
