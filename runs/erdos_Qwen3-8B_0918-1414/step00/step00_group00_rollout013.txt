import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import logging

# Set up logging
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 points for balance between speed and accuracy

    # Initial guess from initial_h_values
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_points = len(initial_h)
    else:
        initial_h = np.random.rand(n_points) * 2 - 1  # Random initial guess scaled to [0, 2]
        initial_h = np.clip(initial_h, 0, 2)
        # Ensure integral constraint is met
        initial_h = initial_h / np.mean(initial_h) * (n_points / 2) / (np.mean(initial_h) * n_points)

    def objective(h):
        nonlocal best_c5, best_h
        h = np.clip(h, 0, 2)
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    def compute_cost(h):
        return objective(h)

    # Objective function for optimization
    def cost_function(x):
        return compute_cost(x)

    # Bounds for each element of h
    bounds = Bounds(0, 2)

    # Constraint to ensure integral is 1
    constraint = LinearConstraint(np.ones(n_points), lb=1, ub=1, name="integral")

    # Optimization using Sequential Least Squares Programming (SLSQP)
    result = opt.minimize(
        cost_function,
        x0=initial_h,
        bounds=bounds,
        constraints=constraint,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Ensure we don't exceed budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # If we have time, perform a local search
        from scipy.optimize import Bounds
        from scipy.optimize import minimize as scipy_minimize

        # Local search around the current best
        local_result = scipy_minimize(
            compute_cost,
            x0=best_h,
            bounds=Bounds(0, 2),
            constraints=LinearConstraint(np.ones(n_points), lb=1, ub=1),
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
        )

        if local_result.fun < best_c5:
            best_c5 = local_result.fun
            best_h = local_result.x

    return best_h, best_c5, n_points
