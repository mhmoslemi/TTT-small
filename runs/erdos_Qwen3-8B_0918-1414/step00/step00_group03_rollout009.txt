import numpy as np
import time
import scipy.optimize
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess based on the provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        shifted_h = np.concatenate([h[-2:], h[:-2]])
        shifted_1_minus_h = 1 - shifted_h
        overlap = np.correlate(h, shifted_1_minus_h, mode="full")
        return np.max(overlap * dx)
    
    def objective(h):
        return compute_overlap(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    # Optimization settings
    max_iter = 100
    tol = 1e-6
    
    # Start optimization
    start_time = time.time()
    result = minimize(
        objective,
        x0=initial_h,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='SLSQP',
        options={'maxiter': max_iter, 'ftol': tol, 'eps': 1e-8}
    )
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Ensure we stay within the time budget
    if time.time() - start_time < budget_s:
        # Try with a coarser grid to explore more space
        n_points_new = max(2 * n_points, 100)
        dx_new = 2.0 / n_points_new
        best_h_new = np.zeros(n_points_new)
        
        # Sample from the best solution and interpolate
        h_interp = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), best_h)
        best_h_new = np.clip(h_interp, 0, 1)
        best_c5_new = compute_overlap(best_h_new)
        
        if best_c5_new < best_c5:
            best_h = best_h_new
            best_c5 = best_c5_new
    
    return best_h, best_c5, len(best_h)
