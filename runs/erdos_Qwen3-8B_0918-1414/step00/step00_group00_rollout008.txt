import numpy as np
import scipy.optimize as optimize
import time
import warnings

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    warnings.simplefilter("ignore")
    
    # Initial setup
    n_points = 100  # Use 100 points for faster evaluation
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Start with the initial construction
    initial_h = initial_h_values
    h = np.interp(np.linspace(0, 2, n_points), np.sort(initial_h), np.arange(n_points))
    h = np.sort(h)
    h = np.interp(np.linspace(0, 2, n_points), np.sort(h), np.sort(np.random.rand(n_points)))
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Start the optimization
    for _ in range(100):
        t0 = time.time()
        if time.time() - t0 > budget_s - 1:
            break
        
        # Evaluate current h
        current_h = h.copy()
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Compute the correlation
        corr = np.correlate(current_h, 1 - current_h, mode="full")
        c5 = np.max(corr) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
        
        # Try a small perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        h = current_h + perturb
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate the best solution
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    # Compute the final C5
    corr = np.correlate(final_h, 1 - final_h, mode="full")
    c5_bound = np.max(corr) * dx
    
    return (final_h, c5_bound, n_points)
