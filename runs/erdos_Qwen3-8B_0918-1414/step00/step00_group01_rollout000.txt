```python
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sparse
import scipy.sparse.linalg as spla
from scipy.optimize import minimize, LinearConstraint, Bounds
from time import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # We'll use 200 points for better precision
    dx = 2.0 / n_points
    max_k = 100  # Maximum shift to consider for correlation
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        initial_h_values[i] = np.sin(i * np.pi / n_points) / 2 + 0.5
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = np.maximum(0, np.minimum(1, initial_h_values))
    initial_h_values /= np.sum(initial_h_values) * dx
    initial_h_values *= n_points / 2

    def evaluate_C5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = np.maximum(0, np.minimum(1, h))
        h = h / np.sum(h) * dx * n_points
        h = h * n_points / 2
        C5_values = []
        for k in range(1, max_k + 1):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h))
            C5_values.append(overlap)
        return np.max(C5_values)

    def objective(x):
        return evaluate_C5(x)

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = LinearConstraint(np.ones(n_points), lb=0, ub=n_points / 2)
    result = minimize(objective, initial_h_values, method='SLSQP', constraints=cons, bounds=bounds, tol=1e-8, options={'maxiter': 1000, 'ftol': 1e-8, 'disp': False})
    best_h = result.x
    best_C5 = evaluate_C5(best_h)
    return best_h, best_C5, n_points
```
