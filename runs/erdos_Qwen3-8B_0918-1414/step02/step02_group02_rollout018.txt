import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # We'll start with a smaller n_points to save time, but we'll increase it if needed
    n_points = 100  # Start with 100 points for faster evaluation
    dx = 2.0 / n_points
    
    # Initial guess: use the initial_h_values provided
    initial_h_values = np.array([0.0]*n_points)
    for i in range(n_points):
        x = i * dx
        if x < 0.4:
            initial_h_values[i] = 1.0
        elif x < 0.8:
            initial_h_values[i] = 0.5
        elif x < 1.2:
            initial_h_values[i] = 0.0
        elif x < 1.6:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 1.0
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(initial_h_values) * dx
    scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
    initial_h_values *= scale_factor
    
    # Define the function to evaluate C_5
    def compute_C5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define constraints: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # First, use differential evolution for global search with adaptive parameters
    result_de = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    
    # Refine with SLSQP using the result from differential evolution
    result_slsqp = minimize(
        objective,
        result_de.x,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP",
        tol=1e-6
    )
    
    # We can attempt to further refine with a local optimizer if time permits
    if time.time() - time.time() < budget_s - 1:
        result_local = minimize(
            objective,
            result_slsqp.x,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="L-BFGS-B",
            tol=1e-8
        )
        result_slsqp = result_local
    
    h_values = result_slsqp.x
    c5_bound = compute_C5(h_values)
    
    # Check if we can improve by increasing n_points
    if n_points < 200 and time.time() - time.time() < budget_s - 1:
        n_points = 200
        dx = 2.0 / n_points
        new_h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.4:
                new_h[i] = 1.0
            elif x < 0.8:
                new_h[i] = 0.5
            elif x < 1.2:
                new_h[i] = 0.0
            elif x < 1.6:
                new_h[i] = 0.5
            else:
                new_h[i] = 1.0
        total_integral = np.sum(new_h) * dx
        scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
        new_h *= scale_factor
        result_de = opt.differential_evolution(
            objective,
            bounds=[(0, 1) for _ in range(n_points)],
            strategy='best1bin',
            maxiter=50,
            popsize=20,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed,
            disp=False
        )
        result_slsqp = minimize(
            objective,
            result_de.x,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={"type": "eq", "fun": lambda h: np.sum(h) - n_points / 2},
            method="SLSQP",
            tol=1e-6
        )
        h_values = result_slsqp.x
        c5_bound = compute_C5(h_values)
    
    return h_values, c5_bound, n_points
