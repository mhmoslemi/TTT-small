import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Starting with a lower number of points for faster optimization
    n_points = 100  # Reduced from 200 for speed
    dx = 2.0 / n_points
    
    # Initial guess based on an optimal construction from literature
    # We use a 5-part step function with strategic placement of 0s and 1s
    # This is a known configuration that's close to optimal
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        # Divide [0, 2] into 5 equal parts for the initial step function
        if x < 0.4:
            initial_h[i] = 1.0
        elif x < 0.8:
            initial_h[i] = 0.0
        elif x < 1.2:
            initial_h[i] = 1.0
        elif x < 1.6:
            initial_h[i] = 0.0
        else:
            initial_h[i] = 1.0
    
    # Ensure integral constraint is satisfied
    total_integral = np.sum(initial_h) * dx
    scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
    initial_h *= scale_factor
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define the constraint: sum(h) * dx == 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start with a global optimization to find the best point
    result_global = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=30,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    
    # Use the global result as a starting point for a local optimization
    result_local = opt.minimize(
        objective,
        result_global.x,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="L-BFGS-B",
        tol=1e-6
    )
    
    h_values = result_local.x
    c5_bound = compute_C5(h_values)
    return h_values, c5_bound, n_points
