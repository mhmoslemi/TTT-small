import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points for better resolution
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Initial guess based on known optimal constructions
    # Use a combination of stepped and smoothed patterns for better distribution
    # Divide the interval [0, 2] into 10 segments for more refined steps
    for i in range(n_points):
        x = i * dx
        if x < 0.2:
            initial_h[i] = 1.0
        elif x < 0.4:
            initial_h[i] = 0.5
        elif x < 0.6:
            initial_h[i] = 0.0
        elif x < 0.8:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 1.0
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(initial_h) * dx
    scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
    initial_h *= scale_factor
    
    # Define the function to evaluate C_5
    def compute_C5(h):
        h = np.array(h)
        # Normalize h to [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define constraints: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a global optimization algorithm with a local refinement
    # First, use differential evolution for global search
    result_de = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=30,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    
    # Refine with SLSQP using the result from differential evolution
    result_slsqp = opt.minimize(
        objective,
        result_de.x,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP",
        tol=1e-6
    )
    
    h_values = result_slsqp.x
    c5_bound = compute_C5(h_values)
    return h_values, c5_bound, n_points
