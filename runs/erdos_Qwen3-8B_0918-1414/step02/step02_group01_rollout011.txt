```python
import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a lower n_points for faster optimization
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    initial_h[0:30] = 1.0
    initial_h[30:60] = 0.5
    initial_h[60:] = 0.0
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_C5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * dx
    
    def objective(x):
        return compute_C5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    best_result = result
    best_h = result['x']
    best_c5 = compute_C5(best_h)
    
    # Try a few perturbations and restarts to avoid local minima
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        new_x = np.clip(best_h + perturb, 0, 1)
        new_x = np.clip(new_x, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=new_x,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if compute_C5(result['x']) < best_c5:
            best_result = result
            best_h = result['x']
            best_c5 = compute_C5(best_h)
    
    # Try with a different n_points if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 150
        dx = 2.0 / n_points
        initial_h = np.zeros(n_points)
        initial_h[0:30] = 1.0
        initial_h[30:60] = 0.5
        initial_h[60:] = 0.0
        initial_h = np.clip(initial_h, 0, 1)
        
        def compute_C5_150(h):
            h = np.array(h)
            correlation = np.correlate(h, 1 - h, mode="full")
            return np.max(correlation) * dx
        
        def objective_150(x):
            return compute_C5_150(x)
        
        def constraint_150(x):
            return np.sum(x) - n_points / 2
        
        bounds_150 = [(0.0, 1.0) for _ in range(n_points)]
        
        x0_150 = initial_h
        
        cons_150 = [{'type': 'eq', 'fun': constraint_150}]
        
        result_150 = optimize.minimize(
            fun=objective_150,
            x0=x0_150,
            bounds=bounds_150,
            constraints=cons_150,
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if compute_C5_150(result_150['x']) < best_c5:
            best_result = result_150
            best_h = result_150['x']
            best_c5 = compute_C5_150(best_h)
    
    return (best_h, best_c5, n_points)
```
