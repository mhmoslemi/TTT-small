import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with n_points=47
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    def compute_C5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * (2.0 / n_points)
    
    def objective(x):
        return compute_C5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    if time.time() - start_time > budget_s - 1:
        return (initial_h, compute_C5(initial_h), n_points)
    
    best_result = result
    best_h = result['x']
    best_c5 = compute_C5(best_h)
    
    # Try a few perturbations and restarts to avoid local minima
    for _ in range(3):
        perturb = np.random.normal(0, 0.05, n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        new_x = np.clip(best_h + perturb, 0, 1)
        new_x = np.clip(new_x, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=new_x,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if compute_C5(result['x']) < best_c5:
            best_result = result
            best_h = result['x']
            best_c5 = compute_C5(best_h)
    
    # Try a different starting point
    new_n_points = 50
    new_h = np.zeros(new_n_points)
    dx = 2.0 / new_n_points
    for i in range(new_n_points):
        x = i * dx
        if x < 0.5:
            new_h[i] = 0.5
        else:
            new_h[i] = 0.5 + 0.5 * np.cos(np.pi * (x - 0.5))
    new_h = np.clip(new_h, 0, 1)
    new_h = new_h * (new_n_points / 2) / np.sum(new_h)
    
    result = optimize.minimize(
        fun=objective,
        x0=new_h,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    if compute_C5(result['x']) < best_c5:
        best_result = result
        best_h = result['x']
        best_c5 = compute_C5(best_h)
    
    return (best_h, best_c5, len(best_h))
