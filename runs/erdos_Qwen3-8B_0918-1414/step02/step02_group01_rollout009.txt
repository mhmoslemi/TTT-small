import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a base solution from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    def compute_C5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * (2.0 / n_points)
    
    # Define the objective function
    def objective(x):
        return compute_C5(x)
    
    # Define the constraint
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization function
    def optimize_solution(x0, bounds, cons, objective, max_iter=100):
        start_time = time.time()
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': max_iter}
        )
        if time.time() - start_time > budget_s - 1:
            return x0
        return result['x']
    
    # Try multiple starting points and optimization strategies
    best_h = initial_h
    best_c5 = compute_C5(initial_h)
    
    # First optimization
    optimized_h = optimize_solution(x0, bounds, cons, objective)
    current_c5 = compute_C5(optimized_h)
    if current_c5 < best_c5:
        best_h = optimized_h
        best_c5 = current_c5
    
    # Second optimization with perturbation
    perturb = np.random.normal(0, 0.05, n_points)
    perturb = np.clip(perturb, -0.1, 0.1)
    new_x = np.clip(best_h + perturb, 0, 1)
    optimized_h = optimize_solution(new_x, bounds, cons, objective)
    current_c5 = compute_C5(optimized_h)
    if current_c5 < best_c5:
        best_h = optimized_h
        best_c5 = current_c5
    
    # Third optimization with random restart
    random_restart = np.random.rand(n_points)
    random_restart = np.clip(random_restart, 0.4, 0.6)
    optimized_h = optimize_solution(random_restart, bounds, cons, objective)
    current_c5 = compute_C5(optimized_h)
    if current_c5 < best_c5:
        best_h = optimized_h
        best_c5 = current_c5
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
