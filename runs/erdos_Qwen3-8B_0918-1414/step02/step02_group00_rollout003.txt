def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Best solution found
    best_h = initial_h_values.copy()
    best_C5 = float('inf')
    best_n_points = len(best_h)
    
    # Try different n_points values
    n_points_list = [48, 64, 96, 128, 192, 256]
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        
        # Initialize h with uniform distribution to satisfy the constraint
        h = np.full(n_points, 0.5)
        
        # Function to compute C_5 for a given h
        def compute_C5(h):
            h = np.array(h)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        # Objective function for optimization
        def objective(h):
            return compute_C5(h)
        
        # Constraint: sum(h) == n_points / 2
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Optimization using differential evolution
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Initial guess: slightly perturb the initial_h_values to avoid local minima
        initial_guess = np.clip(initial_h_values[:n_points] * (n_points / 48), 0, 1)
        initial_guess = np.clip(initial_guess, 0, 1)
        
        # Set up optimization with a time budget
        start_time = time.time()
        result = opt.differential_evolution(objective, bounds, constraints=cons, 
                                            popsize=20, maxiter=100, tol=1e-8, 
                                            mutation=(0.5, 1), recombination=0.9, 
                                            callback=lambda x: None, 
                                            disp=False, polish=False)
        
        # Extract result
        h_opt = result.x
        current_C5 = compute_C5(h_opt)
        
        # Update best solution
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_C5, best_n_points
