def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Best solution found
    best_h = initial_h_values.copy()
    best_C5 = float('inf')
    best_n_points = len(best_h)
    
    n_points = best_n_points
    dx = 2.0 / n_points
    
    # Function to compute C_5 for a given h
    def compute_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Objective function for optimization
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using a more sophisticated approach with Nelder-Mead simplex
    def optimize_h(initial_guess):
        nonlocal best_C5, best_h
        
        # Setup bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Use Nelder-Mead simplex algorithm with bounds and constraints
        result = opt.minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 1000})
        
        # Extract result
        h = result.x
        current_C5 = compute_C5(h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        return h, current_C5
    
    # Initial attempt
    initial_h = initial_h_values.copy()
    h, c5 = optimize_h(initial_h)
    
    # Try several perturbations to avoid local optima
    for _ in range(5):
        # Random perturbation
        perturb = np.random.normal(0, 0.1, n_points)
        perturb = np.clip(perturb, -0.5, 0.5)
        h_perturb = np.clip(initial_h + perturb, 0, 1)
        
        # Ensure sum constraint
        total = np.sum(h_perturb)
        scale = (n_points / 2) / total
        h_perturb = np.clip(h_perturb * scale, 0, 1)
        
        # Optimize perturbed guess
        h, c5 = optimize_h(h_perturb)
    
    return best_h, best_C5, len(best_h)
