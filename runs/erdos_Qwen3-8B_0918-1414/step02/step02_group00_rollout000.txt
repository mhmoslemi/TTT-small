    n_points = 48
    dx = 2.0 / n_points
    
    # Function to compute C_5 for a given h
    def compute_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Optimization using a more sophisticated approach with Nelder-Mead simplex
    def optimize_h(initial_guess):
        nonlocal best_C5, best_h
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
        
        # Use Nelder-Mead simplex algorithm with bounds and constraints
        result = opt.minimize(
            compute_C5,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        # Extract result
        h = result.x
        current_C5 = compute_C5(h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        return h, current_C5
    
    # Initial attempt
    initial_h = initial_h_values.copy()
    best_h = initial_h.copy()

    for perturbation in range(5):

        total = np.sum(h_perturb)

    return best_h, best_C5, len(best_h)
