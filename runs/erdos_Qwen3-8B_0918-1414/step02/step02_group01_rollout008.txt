import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with n_points=47
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    def compute_C5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * (2.0 / n_points)
    
    def objective(x):
        return compute_C5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization with more aggressive settings
    start_time = time.time()
    
    # First optimization with SLSQP
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    best_result = result
    best_h = result['x']
    best_c5 = compute_C5(best_h)
    
    # Try a few perturbations and restarts
    for _ in range(5):
        # Perturb current best solution
        perturb = np.random.normal(0, 0.05, n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        new_x = np.clip(best_h + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=new_x,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if compute_C5(result['x']) < best_c5:
            best_result = result
            best_h = result['x']
            best_c5 = compute_C5(best_h)
    
    # Try a few different starting points
    for _ in range(3):
        # Generate random starting points
        random_h = np.random.uniform(0, 1, n_points)
        random_h = np.clip(random_h, 0, 1)
        random_h = np.clip(random_h, 0, 1)
        # Adjust to satisfy the integral constraint
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        
        result = optimize.minimize(
            fun=objective,
            x0=random_h,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if compute_C5(result['x']) < best_c5:
            best_result = result
            best_h = result['x']
            best_c5 = compute_C5(best_h)
    
    # Check remaining time
    if time.time() - start_time > budget_s - 1:
        return (best_h, best_c5, n_points)
