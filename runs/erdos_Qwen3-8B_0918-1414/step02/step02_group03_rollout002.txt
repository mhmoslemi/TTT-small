import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    n_points = len(initial_h)
    
    def compute_C5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * (2.0 / n_points)
    
    # Define the objective function
    def objective(x):
        return compute_C5(x)
    
    # Define the constraint function
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with multiple restarts
    best_h = initial_h
    best_c5 = compute_C5(initial_h)
    best_n_points = n_points
    
    # Try different starting points
    for restart in range(3):
        # Perturb initial guess
        perturbation = np.random.normal(0, 0.1, n_points)
        x0 = np.clip(initial_h + perturbation, 0.0, 1.0)
        
        # Optimization
        start_time = time.time()
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if time.time() - start_time > budget_s - 1:
            break
        
        current_c5 = result['fun']
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result['x']
            best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
