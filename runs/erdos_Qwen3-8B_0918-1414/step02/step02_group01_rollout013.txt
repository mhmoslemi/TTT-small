import scipy.optimize as optimize
import time
import random

    h = np.array(initial_h)
    
    def compute_C5(h):

        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * (2.0 / n_points)
    
    # Define objective function
    def objective(x):
        return compute_C5(x)
    
    # Constraint function
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization function
    def optimize_with_perturbations(x0, n_perturb=5, n_iter=100):
        best_result = optimize.minimize(
            fun=objective,
            x0=x0,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': n_iter}
        )
        best_h = best_result['x']
        best_c5 = compute_C5(best_h)
        
        for _ in range(n_perturb):
            perturb = np.random.normal(0, 0.05, n_points)
            perturb = np.clip(perturb, -0.1, 0.1)
            new_x = np.clip(best_h + perturb, 0, 1)
            
            result = optimize.minimize(
                fun=objective,
                x0=new_x,
                bounds=bounds,
                constraints=cons,
                method='SLSQP',
                options={'ftol': 1e-9, 'maxiter': n_iter}
            )
            
            current_c5 = compute_C5(result['x'])
            if current_c5 < best_c5:
                best_result = result
                best_h = result['x']
                best_c5 = current_c5
        
        return best_result, best_c5
    
    start_time = time.time()
    
    # Optimizer with perturbations
    result, best_c5 = optimize_with_perturbations(x0)
    
    # If we have time, try even more perturbations
    if time.time() - start_time < budget_s - 5:

            new_x = np.clip(result['x'] + perturb, 0, 1)

            )
