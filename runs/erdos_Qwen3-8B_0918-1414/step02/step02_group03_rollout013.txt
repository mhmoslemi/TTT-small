import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with n_points=47
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    def compute_C5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * (2.0 / n_points)
    
    def objective(x):
        return compute_C5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization with multiple restarts and different solvers
    best_h = x0
    best_c5 = compute_C5(x0)
    best_n_points = n_points
    
    # Try different starting points
    for restart in range(3):
        # Perturb initial guess
        perturbation = np.random.normal(0, 0.1, n_points)
        x0 = np.clip(initial_h + perturbation, 0.0, 1.0)
        
        # Optimization with different solvers
        for method in ['SLSQP', 'L-BFGS-B', 'Nelder-Mead']:
            start_time = time.time()
            try:
                result = optimize.minimize(
                    fun=objective,
                    x0=x0,
                    bounds=bounds,
                    constraints=cons,
                    method=method,
                    options={'ftol': 1e-9, 'maxiter': 100}
                )
                if time.time() - start_time > budget_s - 1:
                    break
                
                current_c5 = result['fun']
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result['x']
                    best_n_points = n_points
            except:
                continue
    
    return (best_h, best_c5, best_n_points)
