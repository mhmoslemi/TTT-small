    def compute_c5(h):

        # Compute the overlap integral for all k
        overlap = np.correlate(h, 1 - h, mode="full")
        # The maximum value of the overlap integral scaled by dx
        max_overlap = np.max(overlap) * dx
        return max_overlap

    # We want to minimize the maximum overlap integral (C₅)
    # Subject to: sum(h) = n_points / 2 (equivalent to ∫ h(x) dx = 1)
    
    def objective(h):
        return compute_c5(h)
    
    # Define bounds for h: each value must be in [0, 1]
    bounds = Bounds(0, 1)

    # Start with a random perturbation of the initial guess
    initial_guess = np.random.uniform(0, 1, n_points)
    initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)
    
    # Perform a global optimization with Nelder-Mead
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        method='Nelder-Mead',
        options={'xatol': 1e-6, 'maxiter': 1000}
    )
