import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with n_points=47
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    def compute_C5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * (2.0 / n_points)
    
    def objective(x):
        return compute_C5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization with multiple restarts and different methods
    best_h = initial_h
    best_c5 = compute_C5(initial_h)
    
    start_time = time.time()
    
    # Try different methods and restarts
    for method in ['SLSQP', 'Nelder-Mead', 'L-BFGS-B']:
        if time.time() - start_time > budget_s - 1:
            break
        
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            bounds=bounds,
            constraints=cons,
            method=method,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if result.success:
            current_h = result['x']
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
