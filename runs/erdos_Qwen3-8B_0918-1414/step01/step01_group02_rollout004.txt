    dx = 2.0 / n_points

    # Function to compute C_5 for a given h
    def compute_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Objective: minimize C_5
    def objective(h):
        return compute_C5(h)

    # Constraints:
    # 1. h[i] ∈ [0, 1]
    # 2. sum(h) = n_points / 2

    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess
    initial_guess = initial_h

    # Optimization using SLSQP with bounds
    def optimize_h(initial_guess):
        result = minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints=[
                {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
            ],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        return result.x, result.fun

    # Run optimization
    start_time = time.time()
    best_h, best_C5 = optimize_h(initial_guess)
    elapsed_time = time.time() - start_time

    # If optimization took too long, use the initial guess
    if elapsed_time > budget_s - 1:
        best_h = initial_guess
        best_C5 = compute_C5(best_h)

    return best_h, best_C5, len(best_h)
