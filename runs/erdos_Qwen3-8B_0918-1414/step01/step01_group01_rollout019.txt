import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 500  # Use 500 points for better resolution
    dx = 2.0 / n_points
    initial_h_values = np.load("initial_h_values.npy")  # Load initial guess from file

    # Ensure the initial_h_values is within [0, 1]
    initial_h_values = np.clip(initial_h_values, 0, 1)
    # Scale to satisfy the integral constraint
    total_integral = np.sum(initial_h_values) * dx
    scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
    initial_h_values *= scale_factor

    # Define the function to evaluate C_5
    def compute_C5(h):
        h = np.array(h)
        # Normalize h to [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define the constraint: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

    # Perform a local optimization using SLSQP with multiple restarts
    results = []
    for _ in range(5):
        start_time = time.time()
        result = opt.minimize(
            objective,
            initial_h_values,
            bounds=bounds,
            constraints=constraint,
            method="SLSQP",
            tol=1e-6,
            options={"maxiter": 100}
        )
        if result.success:
            results.append(result.x)
        # Update initial_h_values with the best result so far
        if not results:
            initial_h_values = result.x
        else:
            # Compute the C5 value for each result and select the best
            c5_values = [compute_C5(r) for r in results]
            best_idx = np.argmin(c5_values)
            initial_h_values = results[best_idx]
        # Check if we've exceeded the budget
        if time.time() - start_time > budget_s:
            break

    # Select the best result from the optimization
    c5_bound = compute_C5(initial_h_values)
    return initial_h_values, c5_bound, n_points
