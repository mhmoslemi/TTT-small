            # Compute current C5
            current_c5 = compute_C5(h)

            grad = np.zeros_like(h)

                h_perturb = h.copy()

                grad[i] = (c5_perturb - current_c5) / 1e-6
            
            # Update h
            h -= learning_rate * grad
            
            # Ensure h is within [0, 1]
            h = np.clip(h, 0, 1)
            
            # Compute new C5
            new_c5 = compute_C5(h)
            
            # If new C5 is lower, keep the update
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            else:
                # Otherwise, revert to previous value
                h = best_h.copy()
        
        return best_h, best_c5
