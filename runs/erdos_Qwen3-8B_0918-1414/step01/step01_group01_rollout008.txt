import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use 200 points for better resolution
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Start with a more refined initial guess based on the current best-known construction
    # The current best-known construction is a piecewise constant function with 5 levels
    # We'll start with a similar structure but with more levels for better optimization
    # Create a piecewise constant function with 5 levels: 1, 0.75, 0.5, 0.25, 0
    # Divide the interval [0, 2] into 5 equal parts of length 0.4
    for i in range(n_points):
        x = i * dx
        if x < 0.4:
            initial_h[i] = 1.0
        elif 0.4 <= x < 0.8:
            initial_h[i] = 0.75
        elif 0.8 <= x < 1.2:
            initial_h[i] = 0.5
        elif 1.2 <= x < 1.6:
            initial_h[i] = 0.25
        else:
            initial_h[i] = 0.0
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(initial_h) * dx
    scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
    initial_h *= scale_factor
    
    # Define the function to evaluate C_5
    def compute_C5(h):
        h = np.array(h)
        # Normalize h to [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define constraints: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup with a local search and global exploration
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for global optimization to avoid local minima
    result = opt.differential_evolution(
        fun=objective,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        popsize=20,
        maxiter=100,
        mutation=(0.5, 1),
        recombination=0.9,
        disp=False,
        polish=True,
        tol=1e-8,
        timebudget=budget_s
    )
    
    h_values = result.x
    c5_bound = compute_C5(h_values)
    return h_values, c5_bound, n_points
