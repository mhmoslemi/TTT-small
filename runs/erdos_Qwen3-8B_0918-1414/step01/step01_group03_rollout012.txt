        h += np.random.normal(0, 0.05, n_points)

        def compute_c5(h):
            h = np.array(h, dtype=np.float64)
            # Compute the overlap integral for all k
            overlap = np.correlate(h, 1 - h, mode="full")
            # The maximum value of the overlap integral scaled by dx
            max_overlap = np.max(overlap) * dx
            return max_overlap
        
        # Define the optimization problem
        # We want to minimize the maximum overlap integral (C₅)
        # Subject to: sum(h) = n_points / 2 (equivalent to ∫ h(x) dx = 1)
        
        def objective(h):
            return compute_c5(h)
        
        # Define bounds for h: each value must be in [0, 1]
        bounds = Bounds(0, 1)
        
        # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
        result = minimize(
            fun=objective,
            x0=h,
            bounds=bounds,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        # Extract the best solution
        current_h = result.x
        current_c5 = compute_c5(current_h)

            best_c5 = current_c5
