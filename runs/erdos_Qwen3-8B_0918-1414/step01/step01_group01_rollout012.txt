import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and quality
    
    # Define the function to evaluate C_5
    def compute_C5(h):
        h = np.array(h)
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define the constraint: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start with an initial guess based on the provided initial_h_values
    # We will use these initial values for the optimization
    # (We assume initial_h_values is available in the context)
    try:
        initial_h_values = initial_h_values
    except NameError:
        # Fallback to a simple initial guess if initial_h_values is not available
        initial_h_values = np.zeros(n_points)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if x < 0.5:
                initial_h_values[i] = 1.0
            elif 0.5 <= x < 1.5:
                initial_h_values[i] = 0.5
            else:
                initial_h_values[i] = 0.0
        # Normalize to ensure the integral constraint is satisfied
        total_integral = np.sum(initial_h_values) * (2.0 / n_points)
        scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
        initial_h_values *= scale_factor
    
    # Perform a local optimization using SLSQP
    start_time = time.time()
    result = opt.minimize(
        objective,
        initial_h_values,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-6, "eps": 1e-6}
    )
    
    # Check if we have enough time to run a local search for improvement
    if time.time() - start_time < budget_s * 0.8:
        # Perform a local search around the current solution
        # Use a small perturbation to explore the neighborhood
        perturb_size = 0.05
        num_perturbations = 10
        best_h = result.x
        best_c5 = compute_C5(best_h)
        
        for _ in range(num_perturbations):
            perturbed_h = best_h + np.random.uniform(-perturb_size, perturb_size, size=n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            # Ensure the integral constraint is satisfied
            total_integral = np.sum(perturbed_h) * (2.0 / n_points)
            scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
            perturbed_h *= scale_factor
            
            # Evaluate the perturbed solution
            current_c5 = compute_C5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        
        result = opt.OptimizeResult(x=best_h, success=True, message="Local search found better solution", fun=best_c5)
    
    h_values = result.x
    c5_bound = compute_C5(h_values)
    
    return h_values, c5_bound, n_points
