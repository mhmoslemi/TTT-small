    bounds = [(0.0, 1.0) for _ in range(n_points)]

    x0 = initial_h
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization
    start_time = time minimize with SLSQP and accept suboptimal solution
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    # Try alternative starting points if time allows
    if time.time() - start_time < budget_s - 10:
        # Alternative starting point: random uniform in [0,1]
        alt_x0 = np.random.uniform(0, 1, n_points)
        alt_result = optimize.minimize(
            fun=objective,
            x0=alt_x0,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        if alt_result.success and alt_result.fun < result.fun:
            result = alt_result
