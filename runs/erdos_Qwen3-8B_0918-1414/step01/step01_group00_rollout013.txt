import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)
    
    # Define the function to compute C5
    def compute_C5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * (2.0 / n_points)
    
    # Objective function to minimize C5
    def objective(x):
        return compute_C5(x)
    
    # Constraint: sum(h) must equal n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization with different strategies
    start_time = time.time()
    
    # First optimization with SLSQP
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints= cons,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    best_h = result['x']
    best_c5 = compute_C5(best_h)
    
    # If time allows, try a second optimization with different settings
    if time.time() - start_time < budget_s - 1:
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            bounds=bounds,
            constraints=cons,
            method='L-BFGS-B',
            options={'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 50}
        )
        best_h = result['x']
        best_c5 = compute_C5(best_h)

    if time.time() - start_time < budget_s - 1:
        for _ in range(10):

            total = np.sum(perturbed_h)
