import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points to allow for optimization
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with a piecewise constant function that alternates between 0 and 1
    # This is a common starting point for minimizing overlap
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 1.0:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(initial_h) * dx
    scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
    initial_h *= scale_factor
    
    # Define the function to evaluate C_5
    def compute_C5(h):
        h = np.array(h)
        # Normalize h to [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define constraints: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform a local optimization using SLSQP
    result = opt.minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP"
    )
    
    h_values = result.x
    c5_bound = compute_C5(h_values)
    return h_values, c5_bound, n_points
