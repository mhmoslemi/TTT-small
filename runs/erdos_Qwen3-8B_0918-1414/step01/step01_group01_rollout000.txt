import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define a range of n_points to try, starting from smaller values for faster evaluation
    n_points_list = [100, 200, 300, 400, 500]
    best_c5 = float('inf')
    best_h = None
    best_n = 100
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        
        # Use the initial_h_values if available (from the problem context)
        # We will create a new initial guess based on a more refined step function
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                initial_h[i] = 1.0
            elif 0.5 <= x < 1.5:
                initial_h[i] = 0.5
            else:
                initial_h[i] = 0.0
        
        # Ensure the integral constraint is satisfied
        total_integral = np.sum(initial_h) * dx
        scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
        initial_h *= scale_factor
        
        # Define the function to evaluate C5
        def compute_C5(h):
            h = np.array(h)
            h = np.clip(h, 0, 1)
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * dx
            return c5
        
        # Define the objective to minimize C5
        def objective(h):
            return compute_C5(h)
        
        # Define constraints: sum(h) * dx == 1 --> sum(h) == n_points / 2
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Optimization setup
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Perform a local optimization using SLSQP
        result = opt.minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP"
        )
        
        h_values = result.x
        c5_bound = compute_C5(h_values)
        
        # Update the best solution
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_values
            best_n = n_points
        
        # Print progress
        print(f"n_points={n_points}, C5={c5_bound:.6f}")
    
    return best_h, best_c5, best_n
