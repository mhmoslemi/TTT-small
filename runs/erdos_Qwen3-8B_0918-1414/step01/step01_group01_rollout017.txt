import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    max_n_points = 200  # Try up to 200 points for better resolution
    n_points = 100  # Start with 100 points for faster initial evaluation

    # Initial guess: step function with customized piecewise regions
    initial_h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(n_points):
        x = i * dx
        if x < 0.4:
            initial_h[i] = 1.0
        elif 0.4 <= x < 1.2:
            initial_h[i] = 0.5
        elif 1.2 <= x < 1.6:
            initial_h[i] = 0.3
        else:
            initial_h[i] = 0.0

    # Ensure the integral constraint is satisfied
    total_integral = np.sum(initial_h) * dx
    scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
    initial_h *= scale_factor

    # Define the function to evaluate C_5
    def compute_C5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)

    # Define constraints: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]

    # Perform a local optimization using SLSQP
    result = opt.minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP"
    )

    # Check if the result is good enough, if not, try with more points
    current_c5 = result.fun
    current_h = result.x

    # If we have time, try with a larger n_points
    elapsed_time = 0
    for n_points in range(100, max_n_points + 1, 20):
        dx = 2.0 / n_points
        new_h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.4:
                new_h[i] = 1.0
            elif 0.4 <= x < 1.2:
                new_h[i] = 0.5
            elif 1.2 <= x < 1.6:
                new_h[i] = 0.3
            else:
                new_h[i] = 0.0

        # Ensure the integral constraint is satisfied
        total_integral = np.sum(new_h) * dx
        scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
        new_h *= scale_factor

        # Check if we have time to perform the optimization
        start_time = time.time()
        result = opt.minimize(
            objective,
            new_h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={"type": "eq", "fun": lambda h: np.sum(h) - n_points / 2},
            method="SLSQP"
        )
        elapsed_time += time.time() - start_time

        if elapsed_time > budget_s:
            break

        new_c5 = result.fun
        if new_c5 < current_c5:
            current_c5 = new_c5
            current_h = result.x
            n_points = n_points

    return current_h, current_c5, n_points
