import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 500  # Start with 500 points for better resolution
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Use the initial_h_values if available (from the problem context)
    # Note: The initial_h_values is not provided, so we manually create an initial guess
    # as a step function that is high in regions where the overlap is likely to be large
    
    # Create an initial step function with multiple segments
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 0.5:
            initial_h[i] = 1.0
        elif 0.5 <= x < 1.0:
            initial_h[i] = 0.75
        elif 1.0 <= x < 1.5:
            initial_h[i] = 0.5
        elif 1.5 <= x < 2.0:
            initial_h[i] = 0.25
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(initial_h) * dx
    scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
    initial_h *= scale_factor
    
    # Define the function to evaluate C_5
    def compute_C5(h):
        h = np.array(h)
        # Normalize h to [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define constraints: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a more robust optimization method with multiple restarts
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    # Perform multiple restarts with different initial guesses
    for _ in range(5):
        # Generate a new initial guess with random perturbations
        perturbation = np.random.uniform(-0.2, 0.2, n_points)
        perturbed_h = np.clip(initial_h + perturbation, 0, 1)
        total_integral = np.sum(perturbed_h) * dx
        scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
        perturbed_h *= scale_factor
        
        # Perform a local optimization using SLSQP
        result = opt.minimize(
            objective,
            perturbed_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP"
        )
        
        h_values = result.x
        c5_bound = compute_C5(h_values)
        
        # Update the best solution
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_values
    
    return best_h, best_c5, n_points
