import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 1000  # Start with 1000 points for more accurate evaluation
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Use the initial_h_values if available (from the problem context)
    # Note: The initial_h_values is not provided, so we manually create an initial guess
    # as a step function that is high in regions where the overlap is likely to be large
    
    # Create an initial step function that is 1 on [0, 0.5], 0.5 on [0.5, 1.5], and 0 on [1.5, 2]
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 0.5:
            initial_h[i] = 1.0
        elif 0.5 <= x < 1.5:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(initial_h) * dx
    scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
    initial_h *= scale_factor
    
    # Define the function to evaluate C_5
    def compute_C5(h):
        h = np.array(h)
        # Normalize h to [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define constraints: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform global optimization using differential evolution
    result = opt.differential_evolution(
        objective,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        strategy='best1bin',
        popsize=20,
        maxiter=100,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        tol=1e-10,
        polish=True
    )
    
    h_values = result.x
    c5_bound = compute_C5(h_values)
    return h_values, c5_bound, n_points
