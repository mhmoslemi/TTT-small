                elif 0.5 <= x < 1.5:

        total_integral = np.sum(initial_h) * dx
        scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
        initial_h *= scale_factor
        
        # Define the function to evaluate C5
        def compute_C5(h):
            h = np.array(h)
            h = np.clip(h, 0, 1)
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * dx
            return c5
        
        # Define the objective to minimize C5
        def objective(h):
            return compute_C5(h)
        
        # Define the constraint: sum(h) * dx == 1 --> sum(h) == n_points / 2
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Optimization setup
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Perform a local optimization using SLSQP, with a lower tolerance for better precision
        result = opt.minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
