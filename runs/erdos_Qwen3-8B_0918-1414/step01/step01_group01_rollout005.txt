import scipy.optimize as opt
import time

    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Use the initial_h_values provided (if available)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(initial_h) * dx
    scale_factor = (1.0 / total_integral) if total_integral != 0 else 1.0
    initial_h *= scale_factor
    
    # Define the function to evaluate C5
    def compute_C5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define constraints: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a more sophisticated optimization method with bounds and constraints
    result = opt.minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="L-BFGS-B",
        options={"ftol": 1e-8, "maxiter": 1000}
    )
    
    # If the optimization is not converged, try a different initial guess
    if result.success and result.fun < 0.38:
        h_values = result.x
        c5_bound = compute_C5(h_values)
        return h_values, c5_bound, n_points
    
    # If not converged, try a different initial guess
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 0.5:
            initial_h[i] = 1.0
        elif 0.5 <= x < 1.5:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(initial_h) * dx

    initial_h *= scale_factor
    
    result = opt.minimize(
        objective,

        bounds=bounds,

        method="L-BFGS-B",

    )
    
    h_values = result.x
