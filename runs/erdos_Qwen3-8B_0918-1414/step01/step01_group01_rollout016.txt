import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use 200 points for better resolution
    dx = 2.0 / n_points
    initial_h_values = np.load("initial_h_values.npy")  # Assuming initial_h_values is provided
    
    # Define the function to evaluate C_5
    def compute_C5(h):
        h = np.array(h)
        # Normalize h to [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Define constraints: sum(h) * dx == 1 --> sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform a local optimization using SLSQP
    result = opt.minimize(
        objective,
        initial_h_values,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP"
    )
    
    h_values = result.x
    c5_bound = compute_C5(h_values)
    return h_values, c5_bound, n_points
