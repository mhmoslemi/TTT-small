        # The maximum value of the overlap integral scaled by dx
        max_overlap = np.max(overlap) * dx
        return max_overlap

    generations = 20

        population = np.random.uniform(0, 1, (pop_size, n_points))

            fitness = np.array([compute_c5(ind) for ind in population])

            parents = population[np.argsort(fitness)][:int(pop_size / 2)]
