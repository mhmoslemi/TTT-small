    n_points = len(initial_h)

    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization
    start_time = time or time.time()
    
    # First optimization
    result1 = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 50}
    )
    
    # Second optimization with perturbation
    perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
    x0_perturb = np.clip(result1['x'] + perturbation, 0, 1)
    result2 = optimize.minimize(
        fun=objective,
        x0=x0_perturb,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 50}
    )
    
    # Third optimization with global search
    result3 = optimize.minimize(
        fun=objective,
        x0=result2['x'],
        bounds=bounds,
        constraints=cons,
        method='L-BFGS-B',
        options={'ftol': 1e-9, 'maxiter': 50}
    )
    
    # Select best result
    results = [result1, result2, result3]
    best_result = min(results, key=lambda r: r['fun'])
    
    # Check time
    if time.time() - start_time > budget_s - 1:
        return (result1['x'], result1['fun'], n_points)

    best_c5 = best_result['fun']
