import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, NonlinearConstraint
import multiprocessing
import random

def run(seed=42, budget_s=60, **kwargs):

    n_points = len(initial_h)
    
    def compute_C5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * (2.0 / n_points)
    
    def objective(x):
        return compute_C5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint}]
    
    def objective_with_gradient(x):
        h = np.array(x)
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * (2.0 / n_points)
        return c5
    
    def gradient(x):
        h = np.array(x)
        correlation = np.correlate(h, 1 - h, mode="full")
        max_val = np.max(correlation)
        max_index = np.argmax(correlation)
        
        # Compute derivative of max_val with respect to h
        dh = np.zeros_like(h)
        for i in range(n_points):
            for j in range(n_points):
                if max_index == i + j - n_points + 1:
                    dh[i] += (1 - h[j]) if i + j < n_points else -h[j]
        return dh * (2.0 / n_points)
    
    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        fun=objective_with_gradient,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        jac=gradient,
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    if time.time() - start_time > budget_s - 1:
        return (initial_h, compute_C5(initial_h), n_points)
    
    best_result = result
    best_h = result['x']
    best_c5 = compute_C5(best_h)
    
    # Perform a grid search over different starting points
    start_time_grid = time.time()
    grid_points = 10
    for _ in range(grid_points):
        perturbed_h = np.array(initial_h)
        for i in range(n_points):
            perturbed_h[i] += np.random.uniform(-0.1, 0.1)
            perturbed_h[i] = np.clip(perturbed_h[i], 0.0, 1.0)
        
        result = optimize.minimize(
            fun=objective_with_gradient,
            x0=perturbed_h,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',

            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if compute_C5(result['x']) < best_c5:
            best_c5 = compute_C5(result['x'])
            best_h = result['x']
    
    if time.time() - start_time_grid > budget_s - 1 - (time.time() - start_time):
        return (best_h, best_c5, n_points)
