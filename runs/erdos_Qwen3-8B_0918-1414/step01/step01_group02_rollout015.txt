import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values  # Predefined initial construction

    # Define a function to compute the overlap integral
    def compute_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / len(h))  # dx = 2.0 / n_points

    # Define the objective function to minimize C_5
    def objective(x):
        return compute_C5(x)

    # Define constraints:
    # 1. x[i] ∈ [0, 1]
    # 2. sum(x) == len(x) / 2

    # Define bounds and constraints
    n_points = len(initial_h)
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})

    # Use the SLSQP solver from scipy.optimize
    result = opt.minimize(
        objective,
        initial_h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    best_h = result.x
    c5_bound = compute_C5(best_h)
    return best_h, c5_bound, len(best_h)
