- Use the provided initial solution as a starting point for higher resolutions
- Employ a combination of gradient descent for local refinement and differential evolution for global exploration

def repair_solution(h, n_points):
    """Repair solution to enforce 0 ≤ h[i] ≤ 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h_clip = np.clip(h, 0, 1)
    s_clip = np.sum(h_clip)
    target = n_points/2
    
    if abs(s_clip - target) < 1e-5:
        return h_clip
    
    # Distribute the difference uniformly
    if s_clip < target:
        deficit = target - s_clip
        h_adjusted = np.minimum(1, h_clip + deficit / n_points)
    else:
        surplus = s_clip - target
        h_adjusted = np.maximum(0, h_clip - surplus / n_points)
    
    # Final repair check
    h_final = np.clip(h_adjusted, 0, 1)
    s_final = np.sum(h_final)
    if abs(s_final - target) > 1e-5:
        # Fallback: scale to meet sum constraint
        scale = target / s_final if s_final > 0 else 1
        h_final = np.clip(h_final * scale, 0, 1)
    
    return h_final

def objective_function(h, n_points, dx):
    """Compute the maximum overlap integral"""
    h_feasible = repair_solution(h, n_points)
    corr = np.correlate(h_feasible, 1 - h_feasible, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def constraint_function(h, n_points):
    """Enforce sum constraint for SLSQP"""
    return np.sum(h) - n_points/2

        bounds = [(0, 1)] * n_points
        result_de = differential_evolution(
            lambda h: objective_function(h, n_points, dx),

        n_eq = 1

            objective_function,

            tol=1e-5

            obj_val = objective_function(repaired, n_points, dx)
