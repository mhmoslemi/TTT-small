import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Triton implementation of the TriMul module from AlphaFold3.
    This kernel computes the outgoing version of the TriMul operation
    over a 4D tensor [B, N, N, C], with support for optional masking.
    Optimized for performance using Triton's advanced features including
    tiling, memory access patterns, and fused operations.

    Parameters:
    - input_tensor: [bs, seq_len, seq_len, dim]
    - mask: [bs, seq_len, seq_len]
    - weights: Dictionary of model weights including norms, projections, gates, etc.
    - config: Dictionary containing model configuration parameters (dim, hidden_dim)

    Returns:
    - output: [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    bs, seqlen, seqlen, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Weights
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()
    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    output = torch.zeros_like(input_tensor)
    
    # LayerNorm (norm) applied to input
    def layer_norm(x):
        x = x * norm_weight + norm_bias
        return x

    # Perform the TriMul computation using Triton kernel
    @triton.jit
    def tritondim_kernel(
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr,
        left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr,
        output_ptr,
        seqlen: tl.constexpr, dim: tl.constexpr, hidden_dim: tl.constexpr,
        B: tl.constexpr, batch_stride: tl.constexpr, 
        block_m: tl.constexpr, block_n: tl.constexpr,
        mask_type: tl.constexpr,
        stride_input: tl.constexpr, stride_mask: tl.constexpr,
        stride_left_proj: tl.constexpr, stride_right_proj: tl.constexpr,
        stride_left_gate: tl.constexpr, stride_right_gate: tl.constexpr,
        stride_out_gate: tl.constexpr, stride_to_out_norm: tl.constexpr,
        stride_to_out: tl.constexpr, stride_output: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        # Define offsets for thread positions
        pid = tl.program_id(axis=0)  # Batch
        pid_m = tl.program_id(axis=1)  # i
        pid_n = tl.program_id(axis=2)  # j
        pid_k = tl.program_id(axis=3)  # k

        # Calculate the thread position within the block
        offs_m = pid_m * BLOCK + tl.arange(0, BLOCK)
        offs_n = pid_n * BLOCK + tl.arange(0, BLOCK)
        offs_k = pid_k * BLOCK + tl.arange(0, BLOCK)

        # Masks and offsets
        m_off = offs_m + tl.arange(0, BLOCK)
        n_off = offs_n + tl.arange(0, BLOCK)
        k_off = offs_k + tl.arange(0, BLOCK)

        # Load inputs
        input_m = tl.load(input_ptr + pid * batch_stride + m_off[:, None] * stride_input + k_off[None, :] * stride_input, 
                          mask=tl.where(k_off[None, :] < seqlen, 1, 0))
        input_n = tl.load(input_ptr + pid * batch_stride + n_off[:, None] * stride_input + k_off[None, :] * stride_input,
                          mask=tl.where(k_off[None, :] < seqlen, 1, 0))
        input_p = tl.load(input_ptr + pid * batch_stride + m_off[:, None] * stride_input + k_off[None, :] * stride_input,
                          mask=tl.where(k_off[None, :] < seqlen, 1, 0))

        # Masking
        mask_m = tl.load(mask_ptr + pid * batch_stride + m_off[:, None] * stride_mask + k_off[None, :] * stride_mask,
                         mask=tl.where(k_off[None, :] < seqlen, 1, 0))
        mask_n = tl.load(mask_ptr + pid * batch_stride + n_off[:, None] * stride_mask + k_off[None, :] * stride_mask,
                         mask=tl.where(k_off[None, :] < seqlen, 1, 0))
        mask_p = tl.load(mask_ptr + pid * batch_stride + m_off[:, None] * stride_mask + k_off[None, :] * stride_mask,
                         mask=tl.where(k_off[None, :] < seqlen, 1, 0))

        input_m = input_m * mask_m
        input_n = input_n * mask_n
        input_p = input_p * mask_p

        # Apply left and right projections
        left_proj = tl.load(left_proj_ptr + m_off[:, None] * stride_left_proj + k_off[None, :] * stride_left_proj, 
                            mask=tl.where(k_off[None, :] < seqlen, 1, 0)) * input_m
        right_proj = tl.load(right_proj_ptr + n_off[:, None] * stride_right_proj + k_off[None, :] * stride_right_proj, 
                             mask=tl.where(k_off[None, :] < seqlen, 1, 0)) * input_n

        # Apply left and right gates
        left_gate = tl.load(left_gate_ptr + m_off[:, None] * stride_left_gate + k_off[None, :] * stride_left_gate,
                           mask=tl.where(k_off[None, :] < seqlen, 1, 0)) 
        left_gate = left_gate.sigmoid()
        left_proj = left_proj * left_gate

        right_gate = tl.load(right_gate_ptr + n_off[:, None] * stride_right_gate + k_off[None, :] * stride_right_gate,
                             mask=tl.where(k_off[None, :] < seqlen, 1, 0))
        right_gate = right_gate.sigmoid()
        right_proj = right_proj * right_gate

        # Compute outer product and sum
        out = tl.dot(left_proj, right_proj, axis=2)

        # Apply output gate and normalization
        out_gate = tl.load(out_gate_ptr + m_off[:, None] * stride_out_gate + k_off[None, :] * stride_out_gate,
                           mask=tl.where(k_off[None, :] < seqlen, 1, 0)) 
        out_gate = out_gate.sigmoid()
        out = out * out_gate

        # Apply to_out normalization
        out_norm = tl.load(to_out_norm_ptr + m_off[:, None] * stride_to_out_norm + k_off[None, :] * stride_to_out_norm,
                           mask=tl.where(k_off[None, :] < seqlen, 1, 0)) 
        out = out * out_norm

        # Apply to_out
        to_out = tl.load(to_out_ptr + m_off[:, None] * stride_to_out + k_off[None, :] * stride_to_out,
                         mask=tl.where(k_off[None, :] < seqlen, 1, 0)) 
        out = out * to_out

        # Write output
        tl.store(output_ptr + pid * batch_stride + m_off[:, None] * stride_output + n_off[None, :] * stride_output, out)
    
    # Parameters for kernel
    # We'll choose a block size that fits in shared memory
    # H100 has 228KB shared memory per SM, which should allow for reasonable blocks
    # Let's use a block size of 64 for now
    BLOCK = 64
    batch_stride_input = seqlen * seqlen * dim
    batch_stride_mask = seqlen * seqlen
    batch_stride_left_proj = seqlen * seqlen * hidden_dim
    batch_stride_right_proj = seqlen * seqlen * hidden_dim
    batch_stride_left_gate = seqlen * seqlen * hidden_dim
    batch_stride_right_gate = seqlen * seqlen * hidden_dim
    batch_stride_out_gate = seqlen * seqlen * hidden_dim
    batch_stride_to_out_norm = seqlen * seqlen * hidden_dim
    batch_stride_to_out = seqlen * seqlen * hidden_dim
    batch_stride_output = seqlen * seqlen * dim

    # Launch kernel
    num_batch = bs
    num_block_i = (seqlen + BLOCK - 1) // BLOCK
    num_block_j = (seqlen + BLOCK - 1) // BLOCK
    num_block_k = (seqlen + BLOCK - 1) // BLOCK

    # Mask type (0 for no mask, 1 for mask)
    mask_type = 0 if mask is None else 1

    # Launch kernel with multiple threads
    grid = (num_batch, num_block_i, num_block_j, num_block_k)
    tritondim_kernel[grid](input_tensor, mask, left_proj_weight, right_proj_weight,
                           left_gate_weight, right_gate_weight, out_gate_weight, 
                           to_out_norm_weight, to_out_weight, 
                           output, 
                           seqlen, dim, hidden_dim,
                           B=bs, batch_stride=batch_stride_input, 
                           block_m=BLOCK, block_n=BLOCK,
                           mask_type=mask_type,
                           stride_input=batch_stride_input, 
                           stride_mask=batch_stride_mask,
                           stride_left_proj=batch_stride_left_proj, 
                           stride_right_proj=batch_stride_right_proj,
                           stride_left_gate=batch_stride_left_gate, 
                           stride_right_gate=batch_stride_right_gate,
                           stride_out_gate=batch_stride_out_gate, 
                           stride_to_out_norm=batch_stride_to_out_norm,
                           stride_to_out=batch_stride_to_out,
                           stride_output=batch_stride_output,
                           BLOCK=BLOCK)

    return output
