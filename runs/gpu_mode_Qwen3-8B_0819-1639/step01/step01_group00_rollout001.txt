import triton
import triton.language as tl
import torch
import math

# Define the custom kernel for the TriMul operation
@triton.jit
def trit_mul_kernel(
    x_ptr, mask_ptr, out_ptr,
    weight_left_proj, weight_left_gate, weight_right_proj, weight_right_gate, weight_out_gate,
    weight_to_out_norm, weight_to_out_norm_bias, weight_to_out,
    norm_weight, norm_bias,
    B, N, C, H, 
    SEQLEN: tl.constexpr,
    BS: tl.constexpr, 
    DIM: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
    BLOCK: tl.constexpr,
):
    # Compute the program ID
    pid = tl.program_id(axis=0)
    # Compute local indices
    local_idx = pid % BLOCK
    # Compute batch index
    b = pid // (N * N * BLOCK)
    # Compute sequence indices
    seq_idx = (pid // BLOCK) % N
    # Compute the offset for the current batch
    offset = b * N * N * C

    # Load the norm weights and compute the layer norm
    norm_weight = tl.load(norm_weight + (local_idx // C) * C + (local_idx % C))
    norm_bias = tl.load(norm_bias + (local_idx // C) * C + (local_idx % C))
    norm_scale = 1.0 / tl.sqrt(tl.math.pow(tl.math.sum((tl.load(x_ptr + offset + (seq_idx * N + k) * C + i) - norm_bias) ** 2 for k in range(N) for i in range(C)), 1.0 / C) + 1e-5)
    
    # Apply LayerNorm
    x = tl.load(x_ptr + offset + (seq_idx * N + k) * C + i)
    x = (x - norm_bias) * norm_scale + norm_bias

    # Apply the projections and gates
    # left_proj
    left_weights = tl.load(weight_left_proj + (local_idx // H) * H + (local_idx % H))
    left = tl.dot(x, left_weights)
    # left_gate
    left_gate_weights = tl.load(weight_left_gate + (local_idx // H) * H + (local_idx % H))
    left_gate = tl.sigmoid(tl.dot(x, left_gate_weights))
    left = left * left_gate

    # right_proj
    right_weights = tl.load(weight_right_proj + (local_idx // H) * H + (local_idx % H))
    right = tl.dot(x, right_weights)
    # right_gate
    right_gate_weights = tl.load(weight_right_gate + (local_idx // H) * H + (local_idx % H))
    right_gate = tl.sigmoid(tl.dot(x, right_gate_weights))
    right = right * right_gate

    # Apply mask
    mask = tl.load(mask_ptr + b * N * N + (seq_idx * N + k) * N + l)
    if mask == 0:
        left = 0
        right = 0

    # Compute the TriMul
    out = tl.einsum('... i k d, ... j k d -> ... i j d', left, right)

    # Apply to_out_norm
    to_out_norm_weights = tl.load(weight_to_out_norm + (local_idx // H) * H + (local_idx % H))
    to_out_norm_bias = tl.load(weight_to_out_norm_bias + (local_idx // H) * H + (local_idx % H))
    out = out * to_out_norm_weights + to_out_norm_bias

    # Apply out_gate
    out_gate_weights = tl.load(weight_out_gate + (local_idx // H) * H + (local_idx % H))
    out_gate = tl.sigmoid(tl.dot(x, out_gate_weights))
    out = out * out_gate

    # Apply to_out
    to_out_weights = tl.load(weight_to_out + (local_idx // H) * H + (local_idx % H))
    out = out * to_out_weights

    # Store the result
    tl.store(out_ptr + offset + (seq_idx * N + i) * N + j, out)

# Main function to implement the custom kernel
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, C = input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[3]
    H = hidden_dim

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Initialize output tensor
    output = torch.zeros_like(input_tensor)

    # Launch the kernel
    grid = (B * N * N + BLOCK - 1) // BLOCK
    trit_mul_kernel[grid, (BLOCK,)](
        input_tensor, mask, output,
        left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias, to_out_weight,
        norm_weight, norm_bias,
        B, N, C, H,
        SEQLEN=N, BS=B, DIM=C, HIDDEN_DIM=H, BLOCK=BLOCK
    )

    return output
