```python
import numpy as np
import time

def compute_convolution(a):
    """Compute linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for efficient FFT
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradient(a, T):
    """
    Compute gradient of soft-max logsumexp of convolution, times 2n/S^2,
    minus the derivative of the 1/S^2 term.
    """
    n = len(a)
    S = np.sum(a)
    c = compute_convolution(a)
    # Soft-max weights
    c_max = np.max(c)
    w = np.exp((c - c_max) / T)
    w_sum = np.sum(w)
    
    # Gradient of logsumexp: (1/T) * w @ (gradient of c w.r.t. a) / w_sum
    # gradient of c_k w.r.t. a_i = 2*a_{k-i} if valid

    #        = (2/T) * sum_k w_k * a_{k-i} / w_sum

    # Specifically: sum_k w_k * a_{k-i} = sum_j w_{j+i} * a_j (substituting k=j+i)
    # This is the cross-correlation (w ⋆ a) at lag i

    # (w ⋆ a)[i] = sum_k w[k] * a[k-i] = (w flipped * a) at appropriate lag
    # Actually, let me use np.fft for correlation

    # corr[i] = sum_k w[k] * a[k-i] (with valid indices)
    # This is the same as convolution of w with reversed a, evaluated at appropriate point

    # IFFT(FFT(w) * conj(FFT(a))) gives the cross-correlation
    corr = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    
    # Gradient of logsumexp part
    grad_L = (2.0 / T) * corr / w_sum

    L_approx = c_max + T * np.log(w_sum)  # soft-max approximation of max
```

```python
import numpy as np
import time

def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]

def fast_cross_corr(w, a):
    """
    Compute corr[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    m = len(w)  # 2n - 1
    N = 1 << (m + n - 2 - 1).bit_length()  # need m + n - 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    # Cross-correlation via FFT
    corr = np.fft.irfft(np.conj(W) * A, N)
    # Extract the linear cross-correlation
    # The result corr[i] for i = 0, ..., n-1 should give sum_k w[k]*a[k-i]
    # But due to the FFT convention, we might need to shift
    return corr[m-1:m-1+n]

def gradient_step(a, T):
    """
    Compute gradient of the objective (using soft-max approximation).
    Returns (gradient, soft_max_value).
    """
    n = len(a)
    S = np.sum(a)
    
    # Compute convolution
    c = fast_convolve(a)
    
    # Soft-max
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / T)
    w_sum = np.sum(exp_c)
    SM = c_max + T * np.log(w_sum)  # soft-max value
    
    # Compute gradient of soft-max log-sum-exp
    # d/d a_i of SM = (1/T) * sum_k exp_c[k] * d(c[k])/d(a_i) / w_sum
    # d(c[k])/d(a_i) = 2*a[k-i] if 0 <= k-i <= n-1
    # So: d/d a_i = (2/T) * sum_k exp_c[k] * a[k-i] / w_sum (for valid k-i)
    
    # sum_k exp_c[k] * a[k-i] is the cross-correlation of exp_c and a at lag i
    corr = fast_cross_corr(exp_c, a)
    
    grad_SM = (2.0 / T) * corr / w_sum
    
    # Full gradient: f = 2n * SM / S^2
    # df/da_i = (2n/S^2) * dSM/da_i - 2n * SM * 2 / S^3
    grad = (2.0 * n) / (S * S) * grad_SM - 4.0 * n * SM / (S * S * S) * np.ones(n)
    
    return grad, SM
```
