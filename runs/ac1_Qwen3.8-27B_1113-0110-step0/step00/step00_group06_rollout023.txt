Hybrid LP-based optimization for minimizing 2*n*max(conv)/sum^2.
Uses a linear programming approach to find directions that reduce
the peak of the auto-convolution, combined with line search and
multiple restarts.

import numpy as np

from scipy import sparse
from scipy.optimize import linprog


def build_conv_constraint_matrix(x):
    """Build sparse matrix A where A @ b = convolve(x, b)."""
    n = len(x)
    # Column j: x[0], x[1], ..., x[n-1] at rows j, j+1, ..., j+n-1
    col_idx = np.repeat(np.arange(n), n)
    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    data = np.tile(x, n)
    A = sparse.csr_array((data, (row_idx, col_idx)), shape=(2*n-1, n))
    return A

    """Solve: max sum(b) s.t. (x*b)_k <= M for all k, b >= 0."""
    n = len(x)

                  options={'presolve': True})
    return res

"""
LP-based minimax optimization for step function convolution peak reduction.
Minimizes 2*n*max(conv(x,x))/sum(x)^2 by iteratively solving LPs to find
directions that reduce the convolution peak, combined with line search,
multiple restarts, and local refinement.

import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def build_sparse_conv_matrix(x):
    """Build sparse matrix A (shape 2n-1 x n) such that A @ b = np.convolve(x, b)."""
    n = len(x)
    # Column j has x[0], x[1], ..., x[n-1] at rows j, j+1, ..., j+n-1
    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    col_idx = np.repeat(np.arange(n), n)
    data = np.tile(x, n)
    A = sparse.csr_array((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A
