def compute_conv_max(p, n):
    """Compute max of full convolution of p with itself."""
    conv = np.convolve(p, p)
    return np.max(conv), conv

def subgradient_at_g(g, p, n):
    """Compute subgradient of max conv at the argmax lag g."""
    subgrad = np.zeros(n)
    # subgrad[j] = 2*p[g-j] if 0 <= g-j < n
    idx = g - np.arange(n)  # g - j for j = 0, ..., n-1
    valid = (idx >= 0) & (idx < n)
    subgrad[valid] = 2.0 * p[idx[valid]]
    return subgrad

def mirror_descent_step(p, lr, n):
    """One mirror descent step with non-negativity (exponential family)."""
    conv = np.convolve(p, p)
    g = int(np.argmax(conv))
    subgrad = subgradient_at_g(g, p, n)
    # Exponential update
    log_p = np.log(p + 1e-30) - lr * subgrad
    # Renormalize (softmax-like)
    log_p -= np.max(log_p)
    p_new = np.exp(log_p)
    p_new = p_new / p_new.sum()
    return p_new, conv, g

        return None

        return None

    q_val = np.maximum(q_val, 0)

    final_score = evaluate_sequence(result)
    print(f"Final: computed_best={best_score:.6f}, evaluate_score={final_score:.6f}")

    if not np.isfinite(final_score) or final_score > 2.0:
        # Fallback to uniform
        result = [1.0] * n
        final_score = evaluate_sequence(result)
        print(f"Fallback uniform: {final_score:.6f}")

    return result
