"""
Hybrid gradient + perturbation search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-K peak gradient descent with adaptive step size, block perturbations
for exploration, and multiple asymmetric initial candidates to break the
symmetry barrier at objective=2.0.
"""
import numpy as np
import time


def _safe_eval(seq):
    try:
        s = evaluate_sequence(seq)
    except Exception:
        return np.inf
    if not np.isfinite(s):
        return np.inf
    return float(s)


def _gradient_step(a, k_peaks, lr):
    """Compute gradient w.r.t. a using specified peak positions, return new a."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-6:
        return a
    b = np.convolve(a, a)
    M = np.max(b)
    base_grad = -(4.0 * n * M) / (S ** 3)
    grad = np.full(n, base_grad, dtype=np.float64)
    coeff = 4.0 * n / (S ** 2)
    for k in k_peaks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        seg = a[k - hi : k - lo + 1][::-1]
        grad[lo:hi + 1] += coeff * seg
    new_a = np.clip(a - lr * grad, 0.0, 1000.0)
    return new_a, b, M, S


def _top_k_indices(b, K):
    """Return indices of top K values in b (the convolution)."""
    n = len(b)
    K = min(K, n)
    if K == n:
        return np.arange(n)
    return np.argpartition(b, -K)[-K::]


def _block_perturb(a, rng, strength=0.15):
    """Apply a random multiplicative block perturbation."""
    n = len(a)
    candidate = a.copy()
    n_blocks = rng.integers(1, 5)
    for _ in range(n_blocks):
        width = max(2, int(n * rng.uniform(0.01, 0.15)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        scale = np.exp(rng.normal(0.0, strength))
        candidate[start:stop] *= scale
    return np.clip(candidate, 0.0, 1000.0)


def _make_half_half(n, c=0.5):
    m = n // 2
    a = np.zeros(n)
    a[:m] = 1.0
    a[m:] = c
    return a


def _make_smooth_asym(n, alpha=1.5):
    x = np.linspace(0, 1, n)
    a = (1.0 - x) ** alpha + 0.1
    return a


def _make_ramp_step(n, frac=0.5):
    m = int(n * frac)
    a = np.zeros(n)
    a[:m] = np.linspace(1.0, 0.3, m)
    a[m:] = 0.3
    return a


def _make_random_smooth(n, rng):
    raw = rng.normal(0, 1, size=n)
    # Smooth it
    kernel = np.ones(max(3, n // 50)) / max(3, n // 50)
    a = np.convolve(raw, kernel, mode='same')
    a = a - a.min() + 0.5
    a = np.clip(a, 0.0, 1000.0)
    return a


def _make_geometric(n, r=0.6):
    i = np.arange(n)
    a = r ** (i.astype(np.float64) * (n / 20.0))
    a = a + 0.1
    return np.clip(a, 0.0, 1000.0)

    for n in lengths:

        candidates.append(_make_half_half(n, 0.4))

        n = len(a)

        local_best_score = sc

                    best_score = sc
                    print(f"  Perturbation improves: {best_score:.6f}")
                stall = 0
                strength = max(0.03, strength * 0.95)

                stall += 1

                    if strength > 0.3:

                        stall = 0

    # Final safety check
    final_sc = _safe_eval(best.tolist())
    if not np.isfinite(final_sc):
        # Fallback
        fallback = np.ones(256)
        best = fallback
        best_score = _safe_eval(best.tolist())

    print(f"FINAL: score={_safe_eval(best.tolist()):.6f}, len={len(best)}")
    return [float(x) for x in best]
