from scipy.optimize import linprog
from scipy import sparse
import time

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """LP-based Frank-Wolfe iteration to minimize max self-convolution peak."""
    rng = np.random.default_rng(seed)

    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial: n={n}, score={best_score}")
    
    for iteration in range(200):
        if time.monotonic() > deadline:
            break
        
        t_start = time.monotonic()
        
        # Normalize f to sum = 1
        f = best / best_sum
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        
        # Build constraint matrix A (sparse)
        # A[k, j] = f[k-j] for 0 <= k-j < n
        # A is (2n-1) x n
        # Using Toeplitz structure
        # Actually, let me construct it using the fact that it's a convolution matrix
        
        # For scipy, I'll build it column by column or use a banded approach.
        # Column j of A: A[k, j] = f[k-j] for k = j, j+1, ..., j+n-1
        # So column j is f[0], f[1], ..., f[n-1] placed at rows j through j+n-1.
        
        # This is a block-Toeplitz matrix. Let me use scipy.sparse.toeplitz or build manually.
        
        # Actually, A is the "convolution matrix" of f. It's lower-Hessenberg with
        # bandwidth n. Let me just build it as a dense matrix if n is small enough,
        # or use sparse.
        
        # For n=6121: 12241 * 6121 * 8 = ~600 MB dense. Too much.
        # Use sparse: each column has n non-zeros, total ~ n*(2n-1)/2 ≈ 18.5 million non-zeros.
        # In CSR format: ~ 18.5M * 12 ≈ 222 MB. Might be tight but should work.
        
        # Alternative: use the matvec to avoid forming A explicitly.
        # A @ g = convolve(f, g) (the first 2n-1 elements of the full convolution)
        # Actually, (A@g)[k] = sum_j A[k,j] g[j] = sum_j f[k-j] g[j] (where valid)
        # = (f * g)[k] = convolve(f, g)[k]
        
        # So the constraint is: convolve(f, g)[k] <= M for all k.
        # And we want to maximize sum(g) subject to g >= 0.
        
        # I can use scipy.optimize.linprog with a custom matvec for A.
        # But scipy's linprog doesn't support custom matvec directly.
        
        # Let me just build the sparse matrix.
        
        print(f"Iter {iteration}: M={M:.6f}, building A...", flush=True)
        
        # Build A as sparse
        # A is (2n-1) x n
        # A[k, j] = f[k-j] if 0 <= k-j <= n-1
        # Column j: entries at rows j, j+1, ..., j+n-1 with values f[0], f[1], ..., f[n-1]
        
        row_indices = []
        col_indices = []
        values = []
        
        for j in range(n):
            rows = j + np.arange(n)
            # Only include rows < 2n-1
            valid_mask = rows < 2*n - 1
            rows = rows[valid_mask]
            vals = f[:len(rows)]
            row_indices.extend(rows)
            col_indices.extend(np.full(len(rows), j))
            values.extend(vals)
        
        A = sparse.csc_matrix((values, (row_indices, col_indices)), shape=(2*n-1, n))
        
        # Objective: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        
        t_solve = time.monotonic() - t_start
        print(f"  LP solved in {t_solve:.1f}s, status={res.message}", flush=True)
        
        if not res.success:
            # Fallback: use subgradient step
            continue
        
        g = res.x
        g_sum = g.sum()
        
        if g_sum < 1e-10:
            continue
        
        # Rescale g to match normalization
        g = g * (best_sum / g_sum)
        
        # Blend: f_new = (1-t)*f + t*g_normalized
        # where g_normalized = g / g.sum() (to have same sum as f)
        g_norm = g / g.sum()
        
        # Try different blending factors
        converged = False
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            candidate = (1-t) * f + t * g_norm
            candidate_scaled = candidate * best_sum
            score = float(evaluate_sequence(candidate_scaled.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate_scaled
                best_sum = best.sum()
                best_score = score
                converged = True
                break
        
        print(f"  Iter {iteration}: best_score={best_score:.6f} (t_solve={t_solve:.1f}s)", flush=True)
        
        if not converged:
            # Try smaller steps
            for t in [0.01, 0.05, 0.2]:
                candidate = (1-t) * f + t * g_norm
                candidate_scaled = candidate * best_sum
                score = float(evaluate_sequence(candidate_scaled.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = candidate_scaled
                    best_sum = best.sum()
                    best_score = score
                    break
    
    return [float(x) for x in best]
