    c = np.convolve(p, p)  # length 2n-1
    c_max = c.max()

    c = auto_convolve(p, N=next_pow2(2*n))
    return 2.0 * n * c.max() / (s*s)

def next_pow2(x):
    n = 1
    while n < x:
        n *= 2
    return n

def get_subgradient(p, k_star):
    """Subgradient of (p*p)[k_star] w.r.t. p."""
    n = len(p)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    j = np.arange(lo, hi + 1)
    grad[j] = p[k_star - j] + p[j]
    return grad

def get_smoothed_gradient(p, tau, N):
    """Gradient of smoothed objective using the W[j] formula."""
    n = len(p)
    c = auto_convolve(p, N)
    c_max = c.max()
    shifted = np.clip(tau * (c - c_max), -500, 500)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2*n-1)
        w[c.argmax()] = 1.0
    w /= w_sum
    
    # T[j] = sum_i p[i] * w[i+j] (cross-correlation)
    p_flip = p[::-1]
    PF = np.fft.rfft(p_flip, N)
    WF = np.fft.rfft(w, N)
    T_full = np.fft.irfft(PF * WF, N)
    T = T_full[n-1:2*n-1]
    
    # W_vec[j] = sum_{k=j}^{min(2n-2, j+n-1)} w[k]
    prefix = np.concatenate([[0.0], np.cumsum(w)])
    j_arr = np.arange(n)
    hi_idx = np.minimum(2*n - 1, j_arr + n)
    W_vec = prefix[hi_idx] - prefix[j_arr]
    
    return T + p * W_vec, w
