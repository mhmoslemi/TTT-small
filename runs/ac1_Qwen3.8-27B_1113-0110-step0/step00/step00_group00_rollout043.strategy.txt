@njit(cache=True)
def _autocorrelation_max(seq):
    """Compute the maximum value of the full linear autocorrelation of a sequence.
    
    Args:
        seq: Array-like of coefficients.
    
    Returns:
        float64: The maximum value of the autocorrelation.
    """
    n = len(seq)
    # Initialize the autocorrelation array with zeros
    autocorr = np.zeros(2*n-1, dtype=float64)
    # Cross-correlation at lag 0 (sum of squares) and other lags
    for k in range(2*n-2):
        for i in range(n):
            if k-i >= 0 and k-i < n:
                autocorr[k] += seq[i] * seq[k-i]
            elif k-i < 0:
                # Second sequence index might be out of bounds
                autocorr[k] += seq[i] * 0.0
    return np.max(autocorr)
