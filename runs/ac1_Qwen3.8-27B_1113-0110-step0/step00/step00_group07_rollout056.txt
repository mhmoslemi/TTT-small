"""
Minimize  score(a) = 2*n*max(conv(a,a)) / (sum(a))**2  for a >= 0.

Key fact: the score is scale invariant.  Writing a = p with sum(p)=1 gives
    score = 2*n*max( conv(p,p) )
so the problem reduces to finding a probability vector p in {0..n-1} whose
self-convolution has the smallest possible peak.  This flattens the
autocorrelation.

Warm start:  p_k proportional to the central-binomial coefficients
    c_0 = 1,  c_k = c_{k-1}*(2k-1)/(2k)
because C(x)^2 = 1/(1-x) makes the (infinite) self-convolution of c exactly
flat (=1 everywhere); truncating to n terms gives a very flat convolution and
an initial score already near pi/2 ~ 1.57.

Refinement:  exponentiated-gradient "peak shaving".  At each step we
    (1) compute conv = p*p (linear) with an FFT,
    (2) weight the peak indices,
    (3) G = w * p  (another FFT convolution) = per-coordinate sensitivity to
        the peaks,
    (4) p <- p * exp(-eta*G) and renormalize to sum 1.
This pushes mass out of the indices that feed the peak and into the valleys,
monotonically flattening the convolution over many iterations.  Several n and
random restarts are tried inside the time budget; the global best p (score
2*n*maxconv) is returned as the sequence (sum 1, all in [0,1]).
"""

import numpy as np
import time


def _fft_size(n):
    m = 2 * n - 1
    p = 1
    while p < m:
        p <<= 1
    return p


def _conv_liner(p, P):
    F = np.fft.rfft(p, P)
    c = np.fft.irfft(F * F, P)
    return c[: 2 * len(p) - 1]


def _central_binomial(n):
    c = np.empty(n, dtype=np.float64)
    c[0] = 1.0
    for k in range(1, n):
        c[k] = c[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    s = c.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return c / s


def _init_p(n, kind, rng):
    if kind == 0:
        return _central_binomial(n)
    if kind == 1:
        # central binomial reversed mix
        cb = _central_binomial(n)
        return (cb + cb[::-1]) / 2.0
    if kind == 2:
        # damped cosine window (smooth bump)
        i = np.arange(n, dtype=np.float64)
        w = 0.5 + 0.5 * np.cos(2.0 * np.pi * (i + 0.5) / n)
        w = np.maximum(w, 1e-12)
        return w / w.sum()
    if kind == 3:
        # random positive vector (geometric-ish)
        base = _central_binomial(n)
        noise = np.exp(rng.normal(0.0, 0.35, size=n))
        v = base * noise
        return v / v.sum()
    return np.full(n, 1.0 / n, dtype=np.float64)


def _peak_shave(p, P, budget_s, rng, eta0, min_eta, freeze):
    """Exponentiated-gradient peak shaving on probability vector p (sum 1).
    Returns best p found (minimizing max self-convolution)."""
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, None)
    s = p.sum()
    if s <= 0:
        p = np.full_like(p, 1.0 / p.size)
        s = 1.0
    p = p / s

    n = p.size
    conv = _conv_liner(p, P)
    best = p.copy()
    bestM = float(conv.max())

    eta = eta0
    t0 = time.monotonic()
    it = 0
    # number of top peaks to overweight
    ntop = max(1, n // 64)

    while time.monotonic() - t0 < budget_s:
        # weight vector concentrated on the peaks
        order = np.argpartition(-conv, ntop)[:ntop]
        w = np.zeros(2 * n - 1, dtype=np.float64)
        # super-linear weight so the true max dominates
        peakval = conv[order]
        w[order] = (peakval / bestM) ** 8.0
        # baseline small weight to keep everything in mind
        w += 1e-6

        # G[i] = sum_k w[k] p[k-i]  (linear convolution, only valid i in [0,n-1])
        Wf = np.fft.rfft(w, P)
        Pf = np.fft.rfft(p, P)
        G = np.fft.irfft(Wf * Pf, P)[:n]
        G = np.maximum(G, 0.0)
        # normalise gradient scale for stable step
        gmax = G.max()
        if gmax > 0:
            G = G / gmax

        # exponentiated update, multiplicative so p stays > 0
        factor = np.exp(-eta * G)
        p = p * factor
        ps = p.sum()
        if not np.isfinite(ps) or ps <= 0:
            break
        p = p / ps

        # occasional small random perturbation to escape plateaus
        if rng.random() < 0.02:
            p = p * np.exp(rng.normal(0.0, 0.05, size=n))
            ss = p.sum()
            p = p / ss

        conv = _conv_liner(p, P)
        M = float(conv.max())
        if M < bestM:
            bestM = M
            best = p.copy()

        # anneal
        eta *= freeze
        if eta < min_eta:
            eta = min_eta
        it += 1
        if it % 200 == 0:
            print("  shave it=%d n=%d M=%.6f score=%.6f eta=%.4g" %
                  (it, n, M, 2.0 * n * M, eta))
    return best, 2.0 * n * bestM
