def _soft_max_and_weights(c, beta):
    """Compute soft max and softmax weights of c."""
    scaled = c / beta
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    p = exp_s / sum_exp
    soft_max = beta * (max_s + np.log(sum_exp))
    return soft_max, p
