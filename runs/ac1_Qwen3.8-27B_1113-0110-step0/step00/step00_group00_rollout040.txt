def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    # Scale to sum = 1 first
    v = v.copy()
    if v.sum() == 0:
        return np.ones(len(v)) / len(v)
    v = v / v.sum()  # Now sum = 1, project onto sum=1 simplex
    # Standard simplex projection
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(v - theta, 0)
    return proj * target_sum
