def next_pow2(m):
    if m <= 1: return 1
    return 1 << (m - 1).bit_length()

def convolve_seq(a):
    n=len(a)
    N=next_pow2(2*n-1)
    fa=np.fft.rfft(a, N)
    return np.fft.irfft(fa*fa, N)[:2*n-1]

def convolve_and_fft(a, N=None):
    n=len(a)
    if N is None: N=next_pow2(2*n-1)
    fa=np.fft.rfft(a,N)
    c=np.fft.irfft(fa*fa,N)[:2*n-1]
    return c, fa

def grad_from_weights(w, fa, N, n):
    # w length 2n-1, fa is rfft of a length N//2+1
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n-1:2*n-1][::-1]

def choose_by_pressure(g, rng, pick_high, temp=0.1):
    g = np.asarray(g, dtype=np.float64)
    m = g.size
    if m == 0:
        return -1
    if m == 1:
        return 0
    if not np.all(np.isfinite(g)):
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    vmin = float(np.min(g)); vmax = float(np.max(g))
    rng_range = vmax - vmin
    if rng_range <= 1e-14:
        return int(rng.integers(0, m))
    denom = max(temp, 1e-12) * rng_range
    if pick_high:
        z = (g - vmax) / denom
    else:
        z = (vmin - g) / denom
    z = np.clip(z, -50.0, 0.0)
    p = np.exp(z)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return int(rng.integers(0, m))
    p = p / s
    return int(rng.choice(m, p=p))

"""
Search for nonnegative step-function heights minimizing

    2*n*max(convolve(a,a)) / (sum(a))^2

by combining:
  - smooth low-dimensional log-parameter optimization,
  - mirror-descent / multiplicative-weight local search in the simplex,
  - binary cost-array simulated annealing,
  - random log-block polishing.

The best sequence found before the deadline is returned.
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_optimize
except Exception:
    _scipy_optimize = None


def next_pow2(m: int) -> int:
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor: float = 0.0) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def softmax_seq(z) -> np.ndarray:
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def convolve_seq(a, N=None) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]


def convolve_and_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if N is None:
        N = next_pow2(3 * n - 2)
    else:
        need = next_pow2(3 * n - 2)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    return c, fa


def grad_from_weights(w, fa, N, n) -> np.ndarray:
    w = np.asarray(w, dtype=np.float64).ravel()
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1 : 2 * n - 1][::-1]


def compute_weights_floor(c, cmax, tau, floor: float = 0.02) -> np.ndarray:
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return c
    if cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = ez / s
    if floor > 0.0:
        w = (1.0 - floor) * w + (floor / float(L))
    return w


def smooth_value_from_c(c, cmax, tau) -> float:
    c = np.asarray(c, dtype=np.float64).ravel()
    cmax = float(cmax)
    tau = float(tau)
    if c.size == 0 or cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    if tau <= 0.0:
        return cmax
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return float(cmax + tau * np.log(s))


def internal_score(a, N=None) -> float:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))


def make_uniform(n: int) -> np.ndarray:
    return np.ones(max(1, int(n)), dtype=np.float64) / float(max(1, int(n)))


def make_sine_sequence(n: int, alphas) -> np.ndarray:
    n = max(2, int(n))
    x = (np.arange(n, dtype=np.float64) - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    base = np.ones(n, dtype=np.float64)
    for m, alpha in enumerate(alphas, start=1):
        base = base + float(alpha) * np.sin(np.pi * float(m) * x)
    return normalize_positive(base, floor=1e-12)


def make_gaussian_sequence(n: int, mu: float, sigma: float) -> np.ndarray:
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    sigma = max(1e-3, float(sigma))
    return normalize_positive(np.exp(-((x - float(mu)) ** 2) / (2.0 * sigma * sigma)), floor=1e-12)


def make_exponential_sequence(n: int, slope: float) -> np.ndarray:
    n = max(2, int(n))
    t = np.linspace(-1.0, 1.0, n, dtype=np.float64)
    return normalize_positive(np.exp(float(slope) * t), floor=1e-12)


def generate_initial_sequences(n: int, seed: int, max_count: int = 12):
    n = max(8, int(n))
    rng = np.random.default_rng(int(seed))
    seqs = [make_uniform(n)]

    alphas_list = [
        (0.35,), (0.55,), (0.75,), (0.95,),
        (-0.35,), (-0.55,), (-0.75,), (-0.95,),
        (0.45, 0.18), (0.45, -0.18),
        (-0.45, 0.18), (-0.45, -0.18),
        (0.35, 0.25, 0.15),
        (0.35, -0.25, 0.15),
        (0.35, 0.25, -0.15),
        (0.30, -0.20, 0.25),
        (0.40, 0.10, -0.20),
    ]
    for al in alphas_list:
        seqs.append(make_sine_sequence(n, al))

    for mu in (0.28, 0.34, 0.42, 0.58, 0.66, 0.72):
        for sigma in (0.12, 0.20, 0.32):
            seqs.append(make_gaussian_sequence(n, mu, sigma))

    for slope in (-3.0, -2.0, -1.2, 1.2, 2.0, 3.0):
        seqs.append(make_exponential_sequence(n, slope))

    for _ in range(5):
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        vals = np.ones(n, dtype=np.float64)
        for m in range(1, 6):
            amp = rng.uniform(0.08, 0.45) / (float(m) ** 0.7)
            vals += amp * np.sin(2.0 * np.pi * m * x + rng.uniform(0.0, 2.0 * np.pi))
            vals += amp * np.cos(2.0 * np.pi * m * x + rng.uniform(0.0, 2.0 * np.pi))
        seqs.append(normalize_positive(vals, floor=1e-12))

    if max_count is not None and len(seqs) > int(max_count):
        keep = list(range(min(10, len(seqs))))
        remaining = list(range(len(keep), len(seqs)))
        rng.shuffle(remaining)
        keep.extend(remaining[: max(0, int(max_count) - len(keep))])
        seqs = [seqs[i] for i in keep]
    return seqs


def resample_seq(a, n: int) -> np.ndarray:
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if old_n == n:
        return a.astype(np.float64, copy=True)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)

def build_logparam_basis(n: int) -> np.ndarray:
    n = max(8, int(n))
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    cols = []

    # Antisymmetric odd modes, useful for reducing the central convolution coefficient.
    for m in range(1, 5):
        cols.append(np.sin(np.pi * float(m) * x))

    # Smooth polynomial modes.
    cols.append(x.copy())
    cols.append(x * x - np.mean(x * x))
    cols.append(x * x * x - np.mean(x * x * x))
    cols.append(x ** 4 - np.mean(x ** 4))

    # Periodic modes.
    for m in range(1, 5):
        t = 2.0 * np.pi * float(m) * i / float(n)
        cols.append(np.sin(t) - np.mean(np.sin(t)))
        cols.append(np.cos(t) - np.mean(np.cos(t)))

    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms <= 1e-14] = 1.0
    B = B / norms
    return B


def theta_value(theta, n, B, tau, N) -> float:
    theta = np.asarray(theta, dtype=np.float64).ravel()
    a = softmax_seq(B @ theta)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    tau_eff = max(float(tau), 1e-12)
    val = smooth_value_from_c(c, cmax, tau_eff)
    return float(2.0 * n * val)


def theta_grad(theta, n, B, tau, N) -> np.ndarray:
    theta = np.asarray(theta, dtype=np.float64).ravel()
    a = softmax_seq(B @ theta)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    cmax = float(np.max(c))
    D = B.shape[1]
    if cmax <= 0.0 or not np.isfinite(cmax):
        return np.zeros(D, dtype=np.float64)

    tau_eff = max(float(tau), 1e-12)
    z = (c - cmax) / tau_eff
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros(D, dtype=np.float64)

    w = ez / s
    ga = grad_from_weights(w, fa, N, n)
    total = float(np.dot(a, ga))
    aB = a @ B
    agaB = (a * ga) @ B
    gtheta = agaB - total * aB
    return (2.0 * n) * gtheta


def optimize_lowdim(n, deadline, seed, initial_a=None, max_starts: int = 4):
    n = max(32, int(n))
    B = build_logparam_basis(n)
    D = B.shape[1]
    N = next_pow2(3 * n - 2)
    rng = np.random.default_rng(int(seed))

    starts = []
    if initial_a is not None:
        try:
            a0 = normalize_positive(np.asarray(initial_a, dtype=np.float64).ravel(), floor=1e-12)
            if a0.size == n:
                z = np.log(np.maximum(a0, 1e-12))
                z -= z.mean()
                theta_proj, _, _, _ = np.linalg.lstsq(B, z, rcond=None)
                starts.append(np.asarray(theta_proj, dtype=np.float64).ravel())
        except Exception:
            pass

    for std in (0.30, 0.60, 1.00, 1.50):
        starts.append(rng.normal(0.0, std, size=D))

    for ci in range(min(D, 5)):
        th = np.zeros(D, dtype=np.float64)
        th[ci] = 1.0
        starts.append(th)

    starts = starts[: max(1, int(max_starts))]

    best_a = None
    best_score = np.inf

    for theta0 in starts:
        if time.monotonic() > deadline:
            break
        theta = np.asarray(theta0, dtype=np.float64).ravel().copy()
        a = softmax_seq(B @ theta)
        sc = internal_score(a, N)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if _scipy_optimize is None:
            break

        max_iter = 20 if n <= 1536 else 12
        factors = (0.40, 0.15, 0.06, 0.025)
        for fac in factors:
            if time.monotonic() > deadline:
                break
            tau = fac / float(n)
            try:
                res = _scipy_optimize.minimize(
                    theta_value,
                    theta,
                    method="L-BFGS-B",
                    jac=theta_grad,
                    args=(n, B, tau, N),
                    options={"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-6, "maxls": 12},
                )
                if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
                    theta = np.asarray(res.x, dtype=np.float64).ravel()
            except Exception:
                pass

            a = softmax_seq(B @ theta)
            sc = internal_score(a, N)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                print("lowdim n={} score={:.8f}".format(n, best_score), flush=True)

    if best_a is None:
        best_a = make_uniform(n)
        best_score = internal_score(best_a, N)
    return best_a, best_score


def optimize_mirror(a, deadline, verbose: bool = False):
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = internal_score(a, N)

    eta = 0.65
    beta = 0.18
    w_floor = 0.045
    stagnant = 0
    it = 0
    last_best = best_score

    while time.monotonic() < deadline:
        c, fa = convolve_and_fft(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()

        if cmax <= 0.0 or not np.isfinite(cmax):
            break

        tau = beta * cmax
        w = compute_weights_floor(c, cmax, tau, floor=w_floor)
        g = grad_from_weights(w, fa, N, n)
        g = g - g.mean()
        scale = float(np.abs(g).max())
        if not np.isfinite(scale) or scale <= 1e-15:
            break
        g = g / scale

        loga = np.log(a + 1e-20)
        J_cur = smooth_value_from_c(c, cmax, tau)
        accepted = False

        for e in (eta, 0.5 * eta, 0.25 * eta, 0.125 * eta):
            if e <= 1e-9:
                continue
            loga_try = loga - e * g
            loga_try = loga_try - float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / float(a_try.sum())
            a_try = normalize_positive(a_try, floor=1e-15)

            c_try, _ = convolve_and_fft(a_try, N)
            cmax_try = float(np.max(c_try))
            score_try = float(2.0 * n * cmax_try)
            J_try = smooth_value_from_c(c_try, cmax_try, tau)

            if np.isfinite(score_try):
                if score_try < score * (1.0 - 1e-10):
                    accepted = True
                    a = a_try
                    score = score_try
                    break
                if np.isfinite(J_try) and J_try < J_cur * (1.0 - 1e-7) and score_try < score * 1.05:
                    accepted = True
                    a = a_try
                    score = score_try
                    break

        if accepted:
            stagnant = 0
            eta = min(1.25, eta * 1.025)
            beta = max(0.004, beta * 0.999)
            w_floor = max(0.0, w_floor * 0.995)
        else:
            stagnant += 1
            eta = max(1e-4, eta * 0.62)
            beta = max(0.003, beta * 0.82)
            if stagnant % 80 == 0:
                beta = min(0.25, beta * 1.7)
                w_floor = min(0.08, w_floor + 0.005)
                stagnant = 0
            if eta < 1e-5 or stagnant > 700:
                break

        it += 1
        if verbose and it % 200 == 0:
            print("mirror n={} iter={} best={:.8f}".format(n, it, best_score), flush=True)

        if it % 500 == 0 and best_score > last_best - 1e-7:
            # Very occasional aggressive escape.
            tmp = a * np.exp(np.clip(np.random.default_rng(it + n).normal(0.0, 0.05, n), -1.0, 1.0))
            tmp = normalize_positive(tmp, floor=1e-15)
            sc_tmp = internal_score(tmp, N)
            if np.isfinite(sc_tmp) and sc_tmp < best_score:
                a = tmp
                best_score = sc_tmp
                best_a = a.copy()
            last_best = best_score

    return best_a, best_score


def binary_from_values(a0, m: int) -> np.ndarray:
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    m = int(max(0, min(n, m)))
    b = np.zeros(n, dtype=np.float64)
    if m <= 0:
        return b
    if m >= n:
        b.fill(1.0)
        return b

    is_binary = np.all((a0 <= 0.5) | (a0 >= 0.5))
    if is_binary:
        vals = (a0 > 0.5).astype(np.float64)
        cur = int(vals.sum())
        if cur == m:
            return vals
        b = vals.copy()
        if cur > m:
            idx = np.nonzero(b > 0.5)[0]
            if idx.size:
                b[idx[-(cur - m):]] = 0.0
        else:
            idx = np.nonzero(b < 0.5)[0]
            if idx.size:
                b[idx[: m - cur]] = 1.0
        return b

    order = np.argsort(a0, kind="stable")
    b[order[-m:]] = 1.0
    return b


def choose_by_pressure(g, rng, pick_high: bool, temp: float = 0.12) -> int:
    g = np.asarray(g, dtype=np.float64).ravel()
    m = g.size
    if m == 0:
        return -1
    if m == 1:
        return 0
    if not np.all(np.isfinite(g)):
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    vmin = float(np.min(g))
    vmax = float(np.max(g))
    range_g = vmax - vmin
    if range_g <= 1e-14:
        return int(rng.integers(0, m))
    denom = max(float(temp), 1e-12) * range_g
    if pick_high:
        z = (g - vmax) / denom
    else:
        z = (vmin - g) / denom
    z = np.clip(z, -50.0, 0.0)
    p = np.exp(z)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return int(rng.integers(0, m))
    p = p / s
    return int(rng.choice(m, p=p))


def optimize_binary(a0, deadline, seed: int = 0):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = max(16, int(a0.size))
    m = max(2, min(n - 2, int(round(0.5 * n))))

    b = binary_from_values(a0, m)
    N = next_pow2(3 * n - 2)
    c = np.rint(convolve_seq(b, next_pow2(2 * n - 1))).astype(np.float64)

    best_b = b.copy()
    best_max = float(np.max(c))
    best_score = float(2.0 * n * best_max / (float(m) * float(m)))

    rng = np.random.default_rng(int(seed))
    beta = 0.75
    temp = 0.045
    stale = 0
    two_b = 2.0 * b

    while time.monotonic() < deadline:
        cmax = float(np.max(c))
        if cmax <= 0.0 or not np.isfinite(cmax):
            break
        score = float(2.0 * n * cmax / (float(m) * float(m)))
        if score < best_score * (1.0 - 1e-12):
            best_score = score
            best_max = cmax
            best_b = b.copy()

        w = compute_weights_floor(c, cmax, beta * cmax, floor=0.06)
        fa = np.fft.rfft(b, N)
        g = grad_from_weights(w, fa, N, n)

        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        si = choose_by_pressure(g[one_idx], rng, True, temp=0.12)
        sj = choose_by_pressure(g[zero_idx], rng, False, temp=0.12)
        if si < 0 or sj < 0:
            break
        i = int(one_idx[si])
        j = int(zero_idx[sj])
        if i == j:
            continue

        # Swap: remove 1 at i, add 1 at j, using old b for linear delta.
        c[i:i + n] -= two_b
        c[2 * i] -= 1.0
        c[j:j + n] += two_b
        c[2 * j] += 1.0
        b[i] = 0.0
        b[j] = 1.0
        two_b = 2.0 * b

        new_max = float(np.max(c))
        new_score = float(2.0 * n * new_max / (float(m) * float(m)))

        accept = False
        if new_score < score * (1.0 - 1e-12):
            accept = True
        else:
            delta = new_score - score
            if delta <= 1e-12:
                if rng.random() < 0.35:
                    accept = True
            elif delta < 1.0:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    accept = True

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_max = new_max
                best_b = b.copy()
            stale = 0
            temp = max(0.0015, temp * 0.998)
            beta = max(0.04, beta * 0.995)
        else:
            # Revert using current b (after swap).
            c[j:j + n] -= two_b
            c[2 * j] -= 1.0
            c[i:i + n] += two_b
            c[2 * i] += 1.0
            b[j] = 0.0
            b[i] = 1.0
            two_b = 2.0 * b
            stale += 1
            temp = max(0.0008, temp * 0.96)
            if stale % 250 == 0:
                beta = min(1.2, beta * 1.6)
                temp = min(0.08, temp * 1.8)
                stale = 0

    return normalize_positive(best_b, floor=0.0), best_score


def local_polish(a, deadline, seed: int = 1):
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = internal_score(a, N)

    loga = np.log(a + 1e-20)
    loga -= float(np.max(loga))
    rng = np.random.default_rng(int(seed))
    sigma = 0.18
    stale = 0

    while time.monotonic() < deadline:
        width_low = max(2, n // 2500)
        width_high = max(3, n // 22)
        width = int(rng.integers(width_low, width_high + 1))
        width = min(width, max(1, n // 2))
        if width < 1:
            width = 1
        start = int(rng.integers(0, n - width + 1))

        t = np.linspace(0.0, 1.0, width, dtype=np.float64)
        env = np.exp(-4.0 * ((t - 0.5) ** 2))
        d = sigma * (0.55 + 0.45 * env) * rng.normal(size=width)
        d -= d.mean()
        d = np.clip(d, -3.0, 3.0)

        old = loga[start:start + width].copy()
        loga[start:start + width] += d
        a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
        a_try = np.maximum(a_try, 1e-300)
        a_try = a_try / float(a_try.sum())
        a_try = normalize_positive(a_try, floor=1e-15)

        score_try = internal_score(a_try, N)
        if np.isfinite(score_try) and score_try < best_score * (1.0 - 1e-10):
            best_a = a_try.copy()
            best_score = score_try
            loga = np.log(a_try + 1e-20)
            loga -= float(np.max(loga))
            stale = 0
            sigma = min(0.38, sigma * 1.04)
        else:
            loga[start:start + width] = old
            stale += 1
            sigma = max(0.03, sigma * 0.96)
            if stale > 250:
                idx = int(rng.integers(0, n))
                old_idx = loga[idx]
                loga[idx] += 1.8 * sigma
                a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
                a_try = np.maximum(a_try, 1e-300)
                a_try = a_try / float(a_try.sum())
                a_try = normalize_positive(a_try, floor=1e-15)
                score_try = internal_score(a_try, N)
                if np.isfinite(score_try) and score_try < best_score:
                    best_a = a_try.copy()
                    best_score = score_try
                    loga = np.log(a_try + 1e-20)
                    loga -= float(np.max(loga))
                else:
                    loga[idx] = old_idx
                stale = 0

        if best_score < 1.05:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()
    total = max(120.0, float(budget_s) - 30.0)
    deadline = start + total

    best_a = make_uniform(1024)
    best_score = internal_score(best_a)
    print("initial fallback score = {:.9f}".format(best_score), flush=True)

    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            print("given sequence score = {:.9f}".format(best_score), flush=True)

    # Fast raw initial shapes.
    for n in (512, 1024, 2048):
        if time.monotonic() > deadline:
            break
        inits = generate_initial_sequences(n, seed + n, max_count=8)
        for a0 in inits:
            sc = internal_score(a0)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a0.copy()
                print("raw init n={} score={:.9f}".format(n, best_score), flush=True)

    # Low-dimensional smooth optimization.
    lowdim_plans = (
        (512, 3, 35),
        (1024, 3, 70),
        (2048, 3, 110),
        (4096, 2, 170),
    )
    for n, starts, allow in lowdim_plans:
        if time.monotonic() > deadline - 420.0:
            break
        allow = min(float(allow), deadline - time.monotonic() - 420.0)
        if allow < 18.0:
            break
        a_init = resample_seq(best_a, n)
        a1, sc1 = optimize_lowdim(
            n,
            time.monotonic() + allow,
            seed=seed + 1000 + n,
            initial_a=a_init,
            max_starts=starts,
        )
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("lowdim selected n={} score={:.9f}".format(n, best_score), flush=True)

    # Quick mirror descent from current best.
    for n in (1024, 2048):
        if time.monotonic() > deadline - 720.0:
            break
        allow = min(55.0, deadline - time.monotonic() - 720.0)
        if allow < 15.0:
            break
        a0 = resample_seq(best_a, n)
        a1, sc1 = optimize_mirror(a0, time.monotonic() + allow, verbose=False)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("quick mirror selected n={} score={:.9f}".format(n, best_score), flush=True)

    # Binary cost-array search. This is often the strongest phase.
    binary_plans = (
        (2048, 220),
        (4096, 420),
        (4498, 300),
        (8192, 300),
    )
    rng_out = np.random.default_rng(seed)
    for n, allow in binary_plans:
        if time.monotonic() > deadline - 260.0:
            break
        allow = min(float(allow), deadline - time.monotonic() - 260.0)
        if allow < 60.0:
            break

        starts_b = []
        starts_b.append(resample_seq(best_a, n))
        starts_b.append(rng_out.random(n))

        sq = np.zeros(n, dtype=np.float64)
        for x in range(n):
            sq[(x * x) % n] = 1.0
        starts_b.append(sq)

        for step in (2, 3, 7, 17):
            perm = (np.arange(n, dtype=np.int64) * int(step)) % n
            bb = np.zeros(n, dtype=np.float64)
            if perm.size:
                nb = max(1, n // 2)
                bb[perm[:nb]] = 1.0
            starts_b.append(bb)

        per = max(18.0, allow / float(max(1, len(starts_b))))
        for a0 in starts_b:
            if time.monotonic() > deadline - 210.0:
                break
            a1, sc1 = optimize_binary(a0, min(deadline, time.monotonic() + per), seed=seed + 5000 + int(n))
            if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                print("binary selected n={} score={:.9f}".format(n, best_score), flush=True)

    # Full mirror descent on longer sequences.
    for n in (2048, 4096, 8192):
        if time.monotonic() > deadline - 130.0:
            break
        allow = min(200.0, deadline - time.monotonic() - 100.0)
        if allow < 35.0:
            break
        a0 = resample_seq(best_a, n)
        a1, sc1 = optimize_mirror(a0, time.monotonic() + allow, verbose=True)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("long mirror selected n={} score={:.9f}".format(n, best_score), flush=True)

    # Final random log-block polish.
    if time.monotonic() < deadline - 15.0:
        a1, sc1 = local_polish(best_a, deadline, seed=1234567)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("polish selected score={:.9f}".format(best_score), flush=True)

    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")

    if not np.isfinite(final_eval):
        best_a = make_uniform(1024)
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print("final internal score = {:.12f}".format(best_score), flush=True)
    print("final evaluated score = {:.12f}, length = {}".format(final_eval, len(out_list)), flush=True)
    return out_list
