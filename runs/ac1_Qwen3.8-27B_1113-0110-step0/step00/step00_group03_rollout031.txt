def water_fill_update(a, k_star, step_size):
    n = len(a)
    # Compute the subgradient: for the active constraint c[k_star] = sum_i a[i]*a[k_star-i]
    # d(c[k_star])/d(a[i]) = a[k_star - i] (if valid)
    pressure = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            pressure[i] = a[j]
    
    # Update: reduce a[i] where pressure is high
    a_new = a - step_size * pressure
    a_new = np.clip(a_new, 0, 1000)
    # Renormalize
    if np.sum(a_new) > 0:
        a_new = a_new / np.sum(a_new) * np.sum(a)
    return a_new

def water_fill_optimize(a, n, eta_start=0.1, eta_end=0.001, max_iter=100000):
    """Water-filling optimization to minimize max of self-convolution."""
    a = a.copy().astype(float)
    a = np.clip(a, 0, 1000)
    if np.sum(a) < 1e-10:
        return a
    a = a / np.sum(a)  # Normalize to sum = 1
    
    best_a = a.copy()
    best_max_c = np.inf
    
    for it in range(max_iter):
        # Adaptive learning rate
        eta = eta_start * (eta_end/eta_start)**(it/max_iter)
        
        # Compute convolution
        c = np.convolve(a, a)
        max_c = np.max(c)
        
        if max_c < best_max_c:
            best_max_c = max_c
            best_a = a.copy()
        
        # Find active constraints (positions where c[k] is close to max)
        threshold = max_c * 0.99  # 99% of max
        active_k = np.where(c >= threshold)[0]
        
        # Compute average pressure
        pressure = np.zeros(n)
        count = np.zeros(n)
        for k in active_k:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[j]
                    count[i] += 1
        pressure[count > 0] /= count[count > 0]
        
        # Multiplicative update
        a = a * np.exp(-eta * pressure)
        a = np.clip(a, 1e-15, None)
        a = a / np.sum(a)
    
    return best_a
