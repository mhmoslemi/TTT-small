import numpy as np
import time

"""
Algorithm summary:
We minimize the scale-invariant objective 2*n*max(conv(a,a))/(sum(a))^2 subject to
a >= 0. Normalizing sum(a)=1, this equals 2*n*max(conv(a,a)). The objective is the
max of quadratic forms q_k(a). Each q_k has gradient 2*a[k-i] (flipped/shifted a),
so a smoothed gradient is the softmax-weighted combination
    grad[i] = 2 * sum_j a[j]*p[i+j]
computed in O(n log n) via FFT cross-correlation. We run projected (nonneg + sum=1)
gradient descent with backtracking line search, ramping the softmax sharpness to focus
on the true peak, and escape local minima with structured restarts (mirror jitter,
symmetric-pair "push-down" that directly attacks the central convolution term).
Multiple diverse starting points are tried over the time budget and the best kept.
"""


def _normalize(a):
    """Clip nonnegative and rescale to unit sum (with safe fallback)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a) / a.size
    return a / s


def _convmax(a):
    return float(np.convolve(a, a).max())


def _soft_weights(conv, beta):
    x = beta * (conv - conv.max())
    e = np.exp(x)
    zz = e.sum()
    if zz < 1e-300:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
        return w
    return e / zz


def _grad_from_weights(p, a, L):
    """grad[i] = 2*sum_j a[j]*p[i+j] via circular cross-correlation (no wrap if L>=3n)."""
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(P * np.conj(A), L)
    n = a.size
    return 2.0 * c[:n]


def _local_search(a0, deadline, rng, base_beta, L):
    n = a0.size
    a = _normalize(a0)
    best_a = a.copy()
    best_max = _convmax(a)
    cur_max = best_max
    beta = max(1.0, base_beta)
    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        cur_max = float(conv.max())
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        p = _soft_weights(conv, beta)
        grad = _grad_from_weights(p, L if L > 0 else 0) if False else _grad_from_weights(p, a, L)
        # backtracking / multiplicative step sizes
        accept = None
        best_trial = cur_max
        for lr in (4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06):
            an = a - lr * grad
            an = _normalize(an)
            m = _convmax(an)
            if m < best_trial:
                best_trial = m
                accept = an
            if m < cur_max - 1e-12:
                break
        if accept is not None and best_trial < cur_max:
            a = accept
            cur_max = best_trial
            no_improve = 0
        else:
            no_improve += 1
        # ramp sharpness slowly so we lock onto the true peak as we converge
        beta = min(80.0, beta + 0.02)
        # escape mechanism when stuck
        if no_improve >= 40:
            no_improve = 0
            r = rng.random()
            if r < 0.34:
                # mirror jitter: blends in an antisymmetric-attack component
                a = 0.45 * a + 0.55 * a[::-1]

                a[j] *= (1.0 - f)

                a = np.maximum(a + rng.normal(0, 0.8 / n, size=n), 0.0)
