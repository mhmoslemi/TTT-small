"""
Minimize  score = 2*n*max(conv(a,a)) / (sum(a))^2   for a non‑negative sequence a.

The objective is scale invariant, so we fix sum(a)=1 and minimize n*max(conv).
Key fact: for SYMMETRIC non‑negative a the centre lag equals sum(a_j^2) >= sum^2/n,
giving score >= 2; but NON‑SYMMETRIC a can push max(conv) below sum^2/n, beating 2.
We therefore run a projected (non‑negativity + fixed sum) sub‑gradient descent on
the active max‑lag constraint, using FFT‑based convolution so each step is O(n log n),
from several smooth non‑symmetric starts and several lengths.  The final best is
checked with the authoritative evaluate_sequence.
"""

import numpy as np
import time


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _fast_conv(a):
    """Linear (full) autocorrelation of real vector a via FFT."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa.conj(), L)[: 2 * n - 1]
    return c


def _proj_sum_nonneg(v, S):
    """Project v onto {x >= 0, sum x = S} (Euler simplex)."""
    v = np.asarray(v, dtype=np.float64)
    lo = float(v.min()) - S - 1.0
    hi = float(v.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.maximum(0.0, v - mid))
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.maximum(0.0, v - 0.5 * (lo + hi))
    x *= S / max(1e-300, float(np.sum(x)))
    return x


def _rush(a, n, lr0, steps):
    """Projected sub‑gradient descent minimizing max(conv) with sum(a) fixed to 1."""
    a = a.copy()
    a = a / max(1e-300, float(np.sum(a)))
    best = float(n * np.max(_fast_conv(a)))
    best_a = a.copy()
    lr = lr0
    for it in range(steps):
        c = _fast_conv(a)
        k = int(np.argmax(c))
        cur = float(c[k])
        if cur < best:
            best = cur
            best_a = a.copy()
        # sub‑gradient of active lag k: g_j = 2 * a[k-j]
        g = np.zeros(n, dtype=np.float64)
        lo_j = max(0, k - (n - 1))
        hi_j = min(n - 1, k)
        js = np.arange(lo_j, hi_j + 1)

        a = _proj_sum_nonneg(a - lr * g, 1.0)
        lr = lr0 * (0.9995 ** it) + 1e-6
    return best, best_a
