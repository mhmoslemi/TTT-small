Hybrid soft-max smoothed projected gradient descent with Adam optimizer

Exploits scale invariance (fix sum=1) and uses FFT for fast autocorrelation.

import numpy as np

def _simplex_project(v, S):
    """Project v onto {x: sum(x) = S, x >= 0} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, S / n)
    rho = idx[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _autocorr(a):
    """Full autocorrelation via FFT. Returns array of length 2*n-1."""
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    return result


def _softmax_grad_autocorr(a, tau):
    """
    Compute gradient of soft_max(conv(a)) w.r.t. a.
    soft_max(conv) = (1/tau)*log(sum exp(tau*conv[k]))
    d(soft_max)/d(a[j]) = sum_k w[k] * d(conv[k])/d(a[j])
    where w[k] = softmax(tau * conv)[k]
    and d(conv[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = a.size
    conv = _autocorr(a)
    # Softmax weights (numerically stable)
    shifted = tau * (conv - conv.max())
    weights = np.exp(shifted)
    weights /= weights.sum()

    # Only process indices with significant weight
    threshold = 1e-10

        if lo <= hi:
            ji = np.arange(lo, hi + 1)
            ai = k - ji
            grad[ji] += 2.0 * w * a[ai]
    return grad, conv


def _exact_score(a, n):
    """Compute the exact objective 2*n*max(conv)/sum^2 (assuming sum=1)."""
    conv = _autocorr(a)
    return 2.0 * n * conv.max()


def _adam_step(a, grad, m, v, t, lr, beta1, beta2, eps):
    """One Adam update step, returning updated a (projected), m, v."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * grad * grad
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = _simplex_project(a_new, 1.0)
    return a_new, m, v

    """Generate diverse initializations (all normalized to sum=1)."""

    inits.append(r / r.sum())

    inits.append(r / r.sum())

    # 7. Power-law decay
    r = 1.0 / (np.arange(1, n + 1) ** 0.5)

    r = np.where(mask, rng.exponential(1.0, n), 0.0)
    if r.sum() > 0:
        inits.append(r / r.sum())

    return inits
