    return np.maximum(v - theta, 0)

def compute_conv(f):
    """Compute linear convolution f*f using FFT."""
    n = len(f)
    size = 2*n - 1
    F = np.fft.fft(f, n=size)
    return np.fft.ifft(F * F).real

def score_of(f):
    """Compute the evaluation score."""
    n = len(f)
    S = f.sum()
    if S < 0.01:
        return float('inf')
    h = compute_conv(f)
    return 2.0 * n * h.max() / (S*S)

def smooth_max_grad(h, tau):
    """Compute softmax weights and gradient of smooth max."""
    h_scaled = h / tau
    h_scaled -= h_scaled.max()
    exp_h = np.exp(h_scaled)
    p = exp_h / exp_h.sum()
    return p

def gradient_step(f, p, lr):
    """Compute gradient and take a step."""
    n = len(f)
    # grad[a] = 2 * sum_j f[j] * p[a+j]
    # This is 2 * np.correlate(p, f, mode='valid')
    grad = 2.0 * np.correlate(p, f, mode='valid')
    return f - lr * grad

def lp_direction(f):
    """Solve LP to find direction g that minimizes linearized peak."""
    n = len(f)
    # LP: min t s.t. (f*g)_k <= t for all k, sum(g)=1, g>=0
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n+1)
    c[n] = 1.0
    
    A_ub = np.zeros((2*n-1, n+1))
    # Build A_ub efficiently
    for i in range(n):
        k_start = i
        k_end = min(2*n-1, i+n)
        A_ub[k_start:k_end, i] = f[i]
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, None)]*n + [(-np.inf, np.inf)], method='highs')
    
    if res.status != 0:
        return None
    return res.x[:n]
