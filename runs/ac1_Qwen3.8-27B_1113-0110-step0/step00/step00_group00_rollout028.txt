import numpy as np
import time

def smooth_gradient(a, e, alpha):
    """Compute gradient of log-sum-exp approximation."""
    n = len(a)
    N = 2 * n
    # weights proportional to exp(alpha * e)
    log_weights = alpha * e - alpha * e.max()  # for numerical stability
    weights = np.exp(log_weights)
    weights /= weights.sum()
    
    # gradient_k = 2 * alpha * sum_j weights[j] * a[j-k]
    # This is 2*alpha * (weights cross-correlation a) at lag k
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    grad = 2 * alpha * np.fft.irfft(W * np.conj(A), N)[:n]
    return grad

    e = np.convolve(a, a)

def _smooth_obj_and_grad(a, alpha):
    """Compute smooth approximation of max(conv(a,a)) and its gradient."""
    n = len(a)
    N = 2 * n
    e = np.convolve(a, a)

    emax = e.max()
    log_sum_exp = emax + np.log(np.exp(alpha * (e - emax)).sum()) / alpha

    log_w = alpha * (e - e.max())
    w = np.exp(log_w)
    w /= w.sum()

    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    grad = 2 * alpha * np.fft.irfft(W * np.conj(A), N)[:n]
