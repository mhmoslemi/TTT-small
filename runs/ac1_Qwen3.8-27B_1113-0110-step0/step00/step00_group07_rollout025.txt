import numpy as np
import time
from scipy import optimize
from scipy import sparse

def compute_conv_peak(a):
    """Compute full convolution of a with itself, return (conv, peak_idx, peak_val)."""
    conv = np.convolve(a, a)
    peak_idx = int(np.argmax(conv))
    peak_val = float(conv[peak_idx])
    return conv, peak_idx, peak_val

def subgrad_step(a, k_star, eta):
    """One subgradient step to reduce peak at k_star."""
    n = len(a)
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if j == i:
                sg[i] = 2.0 * a[i]
            else:
                sg[i] = a[i] + a[j]
    # Update
    a_new = np.maximum(0.0, a - eta * sg)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

def solve_lp_direction(f, M, n, max_constr=200):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for selected constraints, g >= 0.
    Uses active set method: start with tightest constraints, add violated ones.
    """
    # Select initial constraints: those where (f*f)[k] is closest to M
    ff = np.convolve(f, f)
    # Find top max_constr indices
    idx = np.argsort(ff)[::-1][:max_constr]
    
    # Build constraint matrix (sparse)
    def build_conv_matrix(indices):
        """Build sparse matrix A where A[k, i] = f[k-i] if 0<=k-i<n else 0."""
        rows = []
        cols = []
        vals = []
        for k in indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    rows.append(k)
                    cols.append(i)
                    vals.append(f[j])
        return sparse.csr_matrix((vals, (rows, cols)), shape=(len(indices), n))
    
    A = build_conv_matrix(idx)
    b = M * np.ones(len(idx))
    
    # Solve: min -sum(g) s.t. A@g <= b, g >= 0
    c = -np.ones(n)
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if not res.success:
        return None
    
    g = res.x
    
    # Check all constraints
    fg = np.convolve(f, g)
    violated = np.where(fg > M + 1e-8)[0]
    
    # If there are violated constraints, add them and re-solve
    iters = 0
    while len(violated) > 0 and iters < 20:
        iters += 1
        # Add violated constraints (limit total)
        if len(idx) + len(violated) > 2000:
            # Keep only the most violated
            viol_scores = fg[violated]
            top_viol = violated[np.argsort(viol_scores)][::-1][:2000-len(idx)]
        else:
            top_viol = violated
        idx = np.unique(np.concatenate([idx, top_viol]))
        A = build_conv_matrix(idx)
        b = M * np.ones(len(idx))
        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            return None
        g = res.x
        fg = np.convolve(f, g)
        violated = np.where(fg > M + 1e-8)[0]
    
    return g

    idx = int(np.argmax(conv))

def normalize_nonneg(a):
    """Project to non-negative and normalize to sum=1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def solve_lp_direction(f, M, n, max_active=300, time_budget=30.0):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for active constraints, g >= 0.
    Returns the optimal g or None if the LP fails.
    """
    t0 = time.monotonic()
    ff = np.convolve(f, f)
    
    # Start with the most violated (tightest) constraints
    perm = np.argsort(ff)[::-1]
    active = perm[:min(max_active, len(perm))]
    
    g = None
    for iteration in range(30):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Build sparse constraint matrix
        rows = []
        cols = []
        vals = []
        for k in active:
            # Row k: A[k, i] = f[k-i] if 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                rows.append(len(rows))  # row index within active set
                cols.append(i)
                vals.append(f[k - i])
        
        # Actually, I need to be more careful with row indexing
        rows = np.array(rows)
        cols = np.array(cols)
        vals = np.array(vals)
        A = sparse.csr_matrix((vals, (rows, cols)), shape=(len(active), n))
        b = M * np.ones(len(active))
        
        c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        
        res = optimize.linprog(
            c_obj, A_ub=A, b_ub=b,
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': min(10.0, time_budget - (time.monotonic() - t0))}
        )
        
        if not res.success or res.x is None:
            break
        
        g = res.x
        
        # Check all constraints
        fg = np.convolve(f, g)
        violated = np.where(fg > M * (1 + 1e-6))[0]
        
        if len(violated) == 0:
            break
        
        # Add violated constraints (up to a limit)
        if len(active) + len(violated) > 3000:
            # Add only the most violated
            viol_idx = violated[np.argsort(fg[violated])[::-1]][:3000 - len(active)]
        else:
            viol_idx = violated
        active = np.unique(np.concatenate([active, viol_idx]))
    
    return g

    a_eval = best_seq
    final_score = float(evaluate_sequence(a_eval.tolist()))
    print(f"Final score: {final_score:.6f}")
    
    return [float(x) for x in a_eval]

"""
Multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses water-filling subgradient, coordinate descent with vectorized conv updates,
and random perturbation local search.
"""

import numpy as np
import time
from scipy import optimize, sparse


def _normalize(a):
    """Project to non-negative and normalize sum to 1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), 1.0 / len(a))
    return a / s


def _full_conv(a):
    """Full linear convolution of a with itself."""
    return np.convolve(a, a)


def _get_peak(conv):
    """Return (peak_index, peak_value)."""
    idx = int(np.argmax(conv))
    return idx, float(conv[idx])


def _subgradient(a, k_star):
    """
    Compute subgradient of max_k (a*a)[k] evaluated at index k_star.
    Returns array sg where sg[i] = d(a*a)[k_star]/d(a[i]).
    """
    n = len(a)
    sg = np.zeros(n)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    if i_lo > i_hi:
        return sg
    i_range = np.arange(i_lo, i_hi + 1)
    j_range = k_star - i_range
    sg[i_range] = a[i_range] + a[j_range]
    # When k_star is even and mid = k_star//2 is valid, the diagonal term
    # a[mid]*a[mid] has derivative 2*a[mid], but our formula gives a[mid]+a[mid]=2*a[mid]
    # which is already correct. No fix needed.
    return sg


def _conv_update_single(a, conv, i, new_val):
    """
    Efficiently update conv = a*a when a[i] changes to new_val.
    Returns the updated convolution (new array).
    """
    n = len(a)
    old_val = a[i]
    delta = new_val - old_val
    if abs(delta) < 1e-18:
        return conv
    conv_new = conv.copy()
    # Affected positions: k from i to i+n-1
    k_start = i
    k_end = i + n  # exclusive
    # j = k - i ranges from 0 to n-1
    j_range = np.arange(0, n)
    changes = 2.0 * delta * a[j_range]
    # Fix diagonal: k = 2i
    if 2 * i < 2 * n - 1:
        diag_idx_in_range = 2 * i - k_start  # = i
        if 0 <= diag_idx_in_range < n:
            changes[diag_idx_in_range] = 2.0 * delta * old_val + delta * delta
    conv_new[k_start:k_end] += changes
    return conv_new


def _coordinate_descent_sweep(a, conv, k_star, n, num_trials=5, time_limit=None, t0=None):
    """
    One full coordinate descent sweep. For each coordinate, try several values
    and keep the one that minimizes the peak of the convolution.
    Returns updated (a, conv, k_star, peak_val).
    """
    alpha = a.copy()
    c = conv.copy()
    ks = k_star
    best_peak = float(c[ks])

    for i in range(n):
        if time_limit is not None and t0 is not None:
            if time.monotonic() - t0 > time_limit:
                break

        old_val = alpha[i]
        if old_val < 1e-15:
            # Try setting to small positive value
            trials = [0.0, 1.0 / n, 2.0 / n]
        else:
            trials = [
                0.0,
                old_val * 0.3,
                old_val * 0.7,
                old_val,
                old_val * 1.4,
                old_val * 2.0,
                1.0 / n,
            ]

        best_c = c
        best_peak_i = best_peak
        best_trial = old_val

        for tv in trials:
            tv = max(0.0, tv)
            c_trial = _conv_update_single(alpha, c, i, tv)
            pk = float(np.max(c_trial))
            if pk < best_peak_i - 1e-15:
                best_peak_i = pk
                best_c = c_trial
                best_trial = tv

        if best_trial != old_val:
            alpha[i] = best_trial
            c = best_c
            if best_peak_i < best_peak:
                best_peak = best_peak_i
                ks = int(np.argmax(c))

    return alpha, c, ks, best_peak


def _lp_direction(f, M, n, max_time=15.0):
    """
    Solve LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Uses active-set with scipy linprog (HiGHS).
    Returns g array or None.
    """
    t_start = time.monotonic()
    ff = np.convolve(f, f)
    # Start with top constraints
    perm = np.argsort(ff)[::-1]
    initial_count = min(100, 2 * n - 1)
    active = perm[:initial_count]

    c_obj = -np.ones(n)

    for outer_iter in range(50):
        if time.monotonic() - t_start > max_time:
            return None

        # Build sparse A matrix
        active_list = list(active)
        rows = []
        cols = []
        vals = []
        for row_idx, k in enumerate(active_list):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                rows.append(row_idx)
                cols.append(i)
                vals.append(f[k - i])

        if len(rows) == 0:
            return None

        rows = np.array(rows, dtype=np.int32)
        cols = np.array(cols, dtype=np.int32)
        vals = np.array(vals, dtype=np.float64)
        A = sparse.csr_matrix((vals, (rows, cols)), shape=(len(active_list), n))
        b = M * np.ones(len(active_list))

        remaining = max_time - (time.monotonic() - t_start)
        if remaining < 1.0:
            return None

        res = optimize.linprog(
            c_obj, A_ub=A, b_ub=b,
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': min(8.0, remaining - 0.5)}
        )

        if not res.success or res.x is None:
            return None

        g = res.x

        # Check ALL constraints
        fg = np.convolve(f, g)
        violated = np.where(fg > M * (1.0 + 1e-6))[0]

        if len(violated) == 0:
            return g

        # Add violated constraints
        max_total = 3000
        if len(active) + len(violated) > max_total:
            # Add most violated
            viol_sev = fg[violated]
            order = np.argsort(viol_sev)[::-1]
            to_add = violated[order][:max_total - len(active)]
        else:
            to_add = violated
        active = np.unique(np.concatenate([active, to_add]))

    return None
