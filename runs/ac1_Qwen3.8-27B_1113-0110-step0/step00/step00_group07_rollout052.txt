"""
Peak-reduction search for non-negative step-function heights.

Goal: minimize  2*n*max(convolve(a,a)) / (sum(a))**2  over a >= 0.

The objective is scale invariant, so we optimize on the simplex sum(a)=1
and only minimize I(a) = max(convolve(a,a)).  Two complementary,
FFT/BLAS-cheap ideas are combined:

  * soft-peak gradient descent  : grad of (1/beta)*logsumexp(beta*b),
    b = a*a.  The exact derivative d(a*a)[k]/da[m] = 2*a[k-m] (zero-padded)
    is used, giving grad = 2*correlate(p, a) where p = softmax(beta*b).
  * active-set (top-K) gradient : focus the update on the K largest
    convolution entries (the binding constraints of the true max).

A lightweight linear-programming polish (max sum(g) s.t. g>=0,
A@<=M*1, A[k,i]=f[i]) is available for small n and is skipped for the
6733-element instance to stay inside memory limits.  Multiple diverse
starts + periodic perturbed restarts keep the search global.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# small helpers (all top level, no closures / lambdas)
# ----------------------------------------------------------------------
def _normalize(a):
    s = float(a.sum())
    if s <= 1e-12:
        return np.ones_like(a) / a.size
    return a / s


def _feasible(a):
    if a.size == 0:
        return False
    if not np.all(np.isfinite(a)):
        return False
    if float(a.sum()) < 0.01:
        return False
    return True


def _score(a):
    try:
        if not _feasible(a):
            return float("inf")
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float("inf")


def _grad_from_weights(a, p, n):
    """grad[m] = 2 * sum_k p[k] * a[k-m]  (zero-padded)."""
    corr = np.correlate(p, a, mode="full")      # len = 3n-2
    return 2.0 * corr[n - 1:2 * n - 1]


def _soft_weights(b, beta):
    x = beta * b
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s <= 0:
        return np.ones_like(b) / b.size
    return e / s


def _topk_weights(b, K):
    n = b.size
    K = max(1, min(K, n))
    idx = np.argpartition(b, -K)[-K:]
    p = np.zeros(n)
    p[idx] = 1.0 / K
    return p


def _local_search(a, rng, deadline, best_holder):
    """Peak-directed soft/active gradient descent on the simplex."""
    n = a.size
    a = _normalize(np.asarray(a, dtype=np.float64).copy())
    v = np.zeros(n)
    mom = 0.6
    lr = 0.4
    beta = 20.0
    step = 0
    last_improve = 0
    no_improve = 0
    use_active = 0
    active_K = max(8, n // 40)

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        use_active = (use_active + 1) % 5
        if use_active == 0:
            p = _topk_weights(b, active_K)
        else:
            p = _soft_weights(b, beta)

        grad = _grad_from_weights(a, p, n)
        g = grad - grad.mean()            # stay on simplex temp

        scale = float(np.linalg.norm(g) / n) + 1e-15
        step_lr = lr / max(1.0, scale * 4.0)

        v = mom * v + (1.0 - mom) * g
        a_new = np.maximum(a - step_lr * v, 0.0)
        s = float(a_new.sum())
        if s <= 1e-12:
            a_new = a.copy()
        a = _normalize(a_new)
        step += 1

        # anneal
        if step % 40 == 0:
            beta = min(beta * 1.10, 800.0)
            lr = max(lr * 0.985, 0.05)
        if step % 200 == 0:
            beta = max(beta * 0.6, 12.0)   # re-broaden to flatten

        if step % 12 == 0:
            sc = _score(a)
            if sc < best_holder["score"] - 1e-9:
                best_holder["score"] = sc
                best_holder["a"] = a.copy()
                last_improve = step
                no_improve = 0
            elif step - last_improve > 2000:
                # stuck -> perturb current best, reset momentum
                a = best_holder["a"].copy()
                a = a + 0.02 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = _normalize(a)
                v[:] = 0.0
                no_improve += 1
                last_improve = step
    return a


def _lp_polish(f, n, max_iters=8):
    """Small-n LP polish.  max sum(g) s.t. g>=0, (f*g)[k] <= M forall k."""
    if n > 1200:            # memory guard: A has ~n^2 nonzeros
        return f
    f = _normalize(f)
    M = float(np.max(np.convolve(f, f)))
    rows, cols = [], []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            cols.extend(range(lo, hi + 1))
            rows.extend([k] * (hi - lo + 1))
    rows = np.asarray(rows, dtype=np.int32)
    cols = np.asarray(cols, dtype=np.int32)
    try:
        import scipy.sparse as sp
        import scipy.optimize as so
    except Exception:
        return f
    x = f.copy()
    for _ in range(max_iters):
        vals = f[cols]
        A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        res = so.linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                         method="highs", bounds=[(0.0, None)] * n)
        if res.status != 0:
            break
        g = np.maximum(res.x, 0.0)
        gs = float(g.sum())
        if gs <= 0:
            break
        x = 0.4 * x + 0.6 * _normalize(g)
        x = _normalize(x)
        M = float(np.max(np.convolve(x, x)))
    return _normalize(x)


def _make_start(kind, n, rng):
    if kind == 0:
        return np.ones(n)
    if kind == 1:                      # gentle bell
        t = np.linspace(-1.0, 1.0, n)
        return 0.5 + 0.5 * np.cos(np.pi * t)
    if kind == 2:                      # two bumps
        c = 0.5
        t = np.linspace(0.0, 1.0, n)
        return 0.3 + 0.7 * np.exp(-40.0 * ((t - 0.5 * c) ** 2)) \
                        + 0.7 * np.exp(-40.0 * ((t - 1.0 + 0.5 * c) ** 2))
    if kind == 3:                      # smooth random
        return np.abs(rng.standard_normal(n)) + 0.05
    return np.ones(n)
