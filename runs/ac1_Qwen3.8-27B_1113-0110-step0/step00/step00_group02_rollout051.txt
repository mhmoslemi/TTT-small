import numpy as np

def _autocorr_fft(x):
    """Full linear autocorrelation x*x via FFT."""
    n = len(x)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    X = np.fft.rfft(x, L)
    C = np.fft.irfft(X * np.conj(X), L)[:2 * n - 1]
    return np.maximum(C, 0.0)


def _smooth_max_grad(x, gamma):
    """Return (g, grad_g) where g = smooth-max of x*x, grad_g = its gradient (frozen-p approx)."""
    n = len(x)
    C = _autocorr_fft(x)
    Cmax = C.max()
    if Cmax < 1e-30:
        return 0.0, np.zeros(n)
    logits = gamma * (C - Cmax)
    logits = np.clip(logits, -700.0, 0.0)
    el = np.exp(logits)
    ps = el.sum()
    if ps < 1e-30:
        p = np.zeros_like(C)
        p[np.argmax(C)] = 1.0
    else:
        p = el / ps
    g = float(p @ C)
    # grad_g[a] = 2 * sum_j x[j]*p[a+j] for a=0..n-1
    p_rev = p[::-1]
    L2 = 1
    while L2 < 3 * n - 2:
        L2 *= 2
    X2 = np.fft.rfft(x, L2)
    Pr2 = np.fft.rfft(p_rev, L2)
    conv = np.fft.irfft(X2 * Pr2, L2)[:3 * n - 2]
    corr = conv[2 * n - 2: n - 2: -1]
    grad_g = 2.0 * corr
    return g, grad_g

        step = (eta_c / n) * v
        x_new = x - step

        s = x_new.sum()
        if s < 1e-12:

        else:
            x_new /= s
        x = x_new

def _block_perturb(x, rng, n):
    """Random block multiplicative perturbation for diversification."""
    cand = x.copy()
    width = max(2, int(n * rng.uniform(0.01, 0.1)))
    start = int(rng.integers(0, max(1, n - width)))
    stop = min(n, start + width)
    cand[start:stop] *= np.exp(rng.normal(0.0, 0.15))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s > 0:
        cand /= s
    else:
        cand = x.copy()
    return cand
