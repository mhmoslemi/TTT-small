    """One step of projected subgradient descent on max(convolve(a,a))."""
    n = len(a)
    b = np.convolve(a, a)  # length 2n-1
    M = b.max()
    
    # Find active lags (those close to max)
    threshold = M * 0.95  # or M - some tolerance
    active = np.where(b >= threshold)[0]
    
    # Compute subgradient: sum of gradients of active b[k]'s
    # grad[j] = sum_{k in active} 2*a[k-j] (if 0 <= k-j < n)
    grad = np.zeros(n)
    for k in active:
        # For each j, the contribution is 2*a[k-j] if 0 <= k-j < n
        # k-j ranges from k-(n-1) to k, intersected with [0, n-1]
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        # a[k-j] for j in [j_start, j_end]
        # k-j ranges from k-j_end to k-j_start
        idx_start = k - j_end
        idx_end = k - j_start
        grad[j_start:j_end+1] += 2 * a[idx_start:idx_end+1][::-1]
    
    # Project onto sum(a) = constant: subtract mean
    grad -= grad.mean()
    
    return grad, M, len(active)

def soft_gradient(a, tau):
    """Compute the gradient of the soft-max of convolve(a,a)."""
    n = len(a)
    b = np.convolve(a, a)
    
    # Soft-max weights
    b_shifted = b - b.max()  # for numerical stability
    weights = np.exp(b_shifted / tau)
    weights /= weights.sum()
    
    # Weighted sum of gradients
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if weights[k] < 1e-10:
            continue
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        idx = k - np.arange(j_end + 1 - j_start)
        grad[j_start:j_end+1] += weights[k] * 2 * a[idx]
    
    grad -= grad.mean()  # project to maintain sum
    return grad, b.max()

import numpy as np

def peak_aware_perturbation(a, rng, peak_lag, intensity=0.5):
    """Perturb the sequence to reduce the peak at peak_lag."""
    n = len(a)
    candidate = a.copy()
    k = peak_lag
    
    # Find hot positions: j where a[j] * a[k-j] is large
    valid_j = np.arange(max(0, k-n+1), min(n, k+1))
    products = candidate[valid_j] * candidate[k - valid_j]
    
    # Sort by contribution
    sorted_idx = np.argsort(products)[::-1]
    n_hot = max(1, len(valid_j) // 5)  # top 20%
    hot_positions = valid_j[sorted_idx[:n_hot]]
    cold_positions = valid_j[sorted_idx[n_hot:]]
    
    # Reduce hot positions
    for j in hot_positions:
        candidate[j] *= np.exp(rng.normal(0, intensity))
    
    # Increase cold positions
    for j in cold_positions:
        candidate[j] *= np.exp(rng.normal(0, intensity * 0.5))
    
    # Some random perturbation everywhere
    candidate *= np.exp(rng.normal(0, 0.05, n))
    
    return np.clip(candidate, 0, 1000)

"""
Peak-aware local search for minimizing 2n*max(convolve(a,a))/sum(a)^2.
Uses FFT-based soft gradient descent combined with strategic perturbations
targeting the convolution peak. Explores multiple sequence lengths and
initializations to avoid local minima.
"""

import numpy as np
import time


def _fft_convolve(a):
    """Compute linear convolution using FFT for efficiency."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)
    return b[:2 * n - 1]


def _compute_soft_grad(b, a, tau):
    """Compute gradient of soft-max of convolution. Returns zero if tau too large."""
    n = len(a)
    if tau > b.max() * 100:
        return np.zeros(n)
    shifted = b - b.max()
    ex = np.exp(shifted / tau)
    total = ex.sum()
    if total == 0 or not np.isfinite(total):
        return np.zeros(n)
    weights = ex / total
    # Compute weighted sum of gradients using FFT cross-correlation
    # grad[j] = 2 * sum_{k'} weights[j+k'] * a[k']
    flat_weights = np.zeros(4 * n + 2)
    flat_weights[:2 * n - 1] = weights
    T = np.fft.rfft(flat_weights, 4 * n + 2)
    Va = np.fft.rfft(a, 4 * n + 2)
    corr = np.fft.irfft(T * Va, 4 * n + 2)
    grad = 2.0 * corr[:n]
    grad -= grad.mean()
    return grad


def _soft_descent_step(a, tau, rate, max_iter=1500):
    """Perform multiple soft gradient descent steps (working on normalized copy)."""
    n = len(a)
    current = a.copy()
    current_sum = current.sum()
    if current_sum <= 0:
        return a
    current = current / current_sum  # normalize to sum=1

    for _ in range(max_iter):
        b = _fft_convolve(current)
        grad = _compute_soft_grad(b, current, tau)
        if not np.isfinite(grad).any():
            break
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        current = current - rate * grad / gnorm
        current = np.maximum(current, 0.0)
        s = current.sum()
        if s < 1e-12:
            break
        current = current / s
    return current


def _make_random_init(n, rng, style="uniform"):
    """Generate a random initialization with sum=1."""
    if style == "uniform":
        a = rng.random(n)
    elif style == "exponential":
        a = rng.exponential(1.0, n)
    elif style == "pareto":
        a = rng.pareto(1.5, n) + 0.01
    elif style == "bimodal":
        a = np.zeros(n)
        n1 = n // 3
        a[:n1] = rng.random(n1) * 2
        a[n // 2:n // 2 + n1] = rng.random(n1) * 2
    elif style == "decreasing":
        a = np.linspace(1.0, 0.01, n)
        a += rng.normal(0, 0.1, n)
        a = np.maximum(a, 0)
    else:
        a = rng.random(n)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _local_search(a, rng, deadline, label=""):
    """Core local search: alternating soft descent and peak-aware perturbations."""
    n = len(a)
    a = a.copy()
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline:
        iter_count += 1

        # --- Soft gradient descent phase ---
        if no_improve < 30:
            tau = max(0.02, 0.3 * 0.97 ** no_improve)
            steps = max(5, 80 - no_improve)
            a_new = _soft_descent_step(a, tau, 0.4, steps)
            s = a_new.sum()
            if s > 0.005:
                a_new = a_new * (a.sum() / s)
                a_new = np.clip(a_new, 0, 1000)
                score = evaluate_sequence(a_new.tolist())
                if np.isfinite(score) and score < best_score:
                    best = a_new.copy()
                    best_score = score
                    a = a_new
                    no_improve = 0
                    if iter_count % 200 == 1:
                        print(f"  [{label}] iter={iter_count} score={best_score:.6f}")
                else:
                    no_improve += 1
        else:
            no_improve += 1

        # --- Peak-aware perturbation ---
        b = _fft_convolve(a)
        peak_lag = int(np.argmax(b))
        n_active = max(1, n // 8)
        peak_vals = np.sort(b)[-n_active]
        active_thresh = peak_vals[0] if len(peak_vals) > 0 else b.max() * 0.9

        # Find hot positions across all active lags
        active_lags = np.where(b >= active_thresh)[0]
        if len(active_lags) == 0:
            active_lags = np.array([peak_lag])

        hot_count = np.zeros(n)
        for k in active_lags[:min(50, len(active_lags))]:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = k - j_start
            hot_count[j_start:j_end + 1] += idx * a[k - np.arange(j_end - j_start + 1)]

        # Perturb hot positions downward, cold upward
        if np.max(hot_count) > 0:
            perturbation = np.zeros(n)
            hot_mask = hot_count > np.max(hot_count) * 0.3
            cold_mask = ~hot_mask
            scale = 0.15 + 0.1 * min(no_improve / 20.0, 1.0)
            perturbation[hot_mask] = rng.normal(-scale, scale, np.sum(hot_mask))
            perturbation[cold_mask] = rng.normal(0.02, 0.05, np.sum(cold_mask))
            a_new = a * np.exp(perturbation)
            a_new = np.clip(a_new, 0, 1000)
            s_new = a_new.sum()
            s_old = a.sum()
            if s_new > 0.005:
                a_new = a_new * (s_old / s_new)
                a_new = np.clip(a_new, 0, 1000)
                score = evaluate_sequence(a_new.tolist())
                if np.isfinite(score) and score < best_score:
                    best = a_new.copy()
                    best_score = score
                    a = a_new
                    no_improve = 0
                else:
                    if rng.random() < 0.1 + 0.05 * min(no_improve / 10.0, 1.0):
                        a = a_new
                        no_improve += 1
                    else:
                        no_improve += 1
            else:
                no_improve += 1

        # --- Random structure perturbation ---
        if iter_count % 25 == 0:
            style = rng.choice(["block", "random", "mirror"])
            if style == "block":
                width = max(1, n // (50 + int(rng.integers(0, 50))))
                start = int(rng.integers(0, max(1, n - width)))
                a_new = a.copy()
                a_new[start:start + width] *= np.exp(rng.normal(0, 0.3))
                a_new = np.clip(a_new, 0, 1000)
            elif style == "random":
                a_new = a * np.exp(rng.normal(0, 0.2, n))
                a_new = np.clip(a_new, 0, 1000)
            else:
                w = 0.3 * rng.random()
                a_new = (1 - w) * a + w * a[::-1]
            s_new = a_new.sum()
            s_old = a.sum()
            if s_new > 0.005:
                a_new = a_new * (s_old / s_new)
                score = evaluate_sequence(a_new.tolist())
                if np.isfinite(score) and score < best_score:
                    best = a_new.copy()
                    best_score = score
                    a = a_new
                    no_improve = 0

    return best, best_score

    lengths_to_try = [512, 1024, 2048, 4096, 6733]

        sub_deadline = min(phase1_deadline, time.monotonic() + time_per_n)

        for init_i in range(n_inits):

                break

            if init_i == 0 and n_try == (len(height_sequence_1) if 'height_sequence_1' in dir() else -1):
                # Use the given sequence (normalized)
                a = given / given.sum()
            else:

                style = styles[init_i % len(styles)]
                a = _make_random_init(n, rng, style)

            a = np.clip(a, 0, 1000)
            score = evaluate_sequence(a.tolist())
            if not np.isfinite(score):
                continue

            # Local search for this initialization
            deadline_this = min(sub_deadline, time.monotonic() + max(2.0, time_per_n / n_inits - 0.5))
            a_best, score_best = _local_search(a, rng, deadline_this, label=f"n={n},init{init_i}")
            results.append((score_best, a_best, n))
            print(f"  n={n} init={init_i}: score={score_best:.6f}")

    results.sort(key=lambda x: x[0])
    if results:

        best_overall = results[0][1].copy()

    phase2_deadline = deadline
    if best_overall is not None:
        print(f"\nPhase 2: Refining best (score={best_overall_score:.6f}, n={len(best_overall)})")
        best_overall, best_overall_score = _local_search(

    # Phase 3: Try a few more if time remains
    if time.monotonic() < deadline - 30:
        for extra_n in [2048, 4096]:
            if time.monotonic() > deadline - 10:

            for style in ["uniform", "exponential"]:
                if time.monotonic() > deadline - 15:

                a = _make_random_init(extra_n, rng, style)
                a_best, score = _local_search(a, rng, min(deadline - 5, time.monotonic() + 20),

    if not np.isfinite(final_score) or final_score > best_overall_score:
        # fallback to tracked best
        final_score = best_overall_score

    return [float(x) for x in best_overall]
