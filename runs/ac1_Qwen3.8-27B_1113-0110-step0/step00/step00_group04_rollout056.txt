import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fast_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = fft(a, N)
    c = np.real(ifft(A * A)[:2 * n - 1])
    return np.maximum(c, 0.0)


def smooth_max_val(conv, beta):
    m = conv.max()
    if m < 1e-300:
        return 0.0
    e = np.exp(beta * (conv - m))
    return m + np.log(e.sum()) / beta


def compute_h(a, conv, beta):
    """h = cross-correlation(w, a) where w = softmax(beta*conv). h[i] = sum_m w[m+i]*a[m]."""
    n = len(a)
    m = conv.max()
    w = np.exp(beta * (conv - m))
    w /= w.sum()
    N2 = 1
    while N2 < (2 * n - 1) + n - 1:

    A = fft(a, N2)
    cross = np.real(ifft(W * np.conj(A)))

        # Asymmetric multi-lobe
        xg = np.arange(n, dtype=np.float64)
        g = np.exp(-0.5 * ((xg - n * 0.15) / (n * 0.06)) ** 2) + \
            0.5 * np.exp(-0.5 * ((xg - n * 0.5) / (n * 0.1)) ** 2) + \
            np.exp(-0.5 * ((xg - n * 0.85) / (n * 0.06)) ** 2)
        return g / g.sum()
    elif rtype == 10:
        # Power-law decay from center
        xg = np.arange(n, dtype=np.float64)
        d = np.abs(xg - n * 0.5) / (n * 0.5)
        p = (1.0 - d) ** rng.uniform(0.5, 3.0)
        p = np.clip(p, 0.001, None)
        return p / p.sum()
    else:
        return np.ones(n) / n


def local_search(a, rng, deadline, current_score):
    """Targeted local search using gradient-informed directions and random perturbations."""
    n = len(a)
    best = a.copy()
    bs = current_score
    step = 0.05
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < deadline:
        if no_improve > max_no_improve:
            step = 0.15
            no_improve = 0
            best = a.copy()

        if step < 0.003:
            break

        # Strategy A: Gradient descent step (60% of time)
        if rng.random() < 0.6:
            a_pos = np.maximum(best, 1e-10)
            a_pos = a_pos / a_pos.sum()
            x = np.log(a_pos)
            obj, gx = obj_grad_x(x, 200.0)
            # Normalized gradient step
            gnorm = np.linalg.norm(gx)
            if gnorm < 1e-15:
                gnorm = 1.0
            direction = -gx / gnorm
            # Project to stay in softmax space: subtract mean to preserve sum
            direction = direction - a_pos * np.dot(direction, np.ones(n))
            # Actually for softmax params, any direction works
            x_new = x + step * direction * gnorm * 0.5
            a_new = np.exp(x_new - x_new.max())
            a_new = a_new / a_new.sum()
            a_new = np.clip(a_new, 0.0, 1000.0)
        else:
            # Strategy B: Targeted peak reduction
            conv = fast_conv(best)
            peak_lag = np.argmax(conv)
            candidate = best.copy()
            # Identify contributing pairs
            contribs = []
            for j in range(n):
                k2 = peak_lag - j
                if 0 <= k2 < n:
                    contribs.append((best[j] * best[k2], j, k2))
            contribs.sort(reverse=True)
            # Decrease top contributors
            top_k = max(1, min(10, len(contribs)))
            for _, j, k2 in contribs[:top_k]:
                if j != k2:
                    candidate[j] *= (1.0 - step * 0.5)
                    candidate[k2] *= (1.0 - step * 0.5)
                else:
                    candidate[j] *= (1.0 - step * 0.7)
            # Add mass back to low-conv positions
            total_removed = best.sum() - candidate.sum()
            if total_removed > 0:
                conv_cand = fast_conv(candidate)
                min_lag = np.argmin(conv_cand)
                for j in range(n):
                    k2 = min_lag - j
                    if 0 <= k2 < n:
                        candidate[j] += total_removed / (n * 2)
            candidate = np.clip(candidate, 0.0, 1000.0)
            a_new = candidate

        # Strategy C (20%): Random block perturbation
        elif rng.random() < 0.2:
            a_new = best.copy()
            width = int(rng.integers(max(2, n // 50), max(4, n // 5)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            a_new[start:start + width] *= np.exp(rng.normal(0.0, step * 2, width))
            a_new = np.clip(a_new, 0.0, 1000.0)
        else:
            # Strategy D: Single element perturbation
            a_new = best.copy()
            idx = int(rng.integers(0, n))
            a_new[idx] *= np.exp(rng.normal(0.0, step * 3))
            if a_new[idx] < 0:
                a_new[idx] = 0.0
            a_new = np.clip(a_new, 0.0, 1000.0)

        s = a_new.sum()
        if s < 0.01:
            no_improve += 1
            continue

        sc = score_seq(a_new)
        if np.isfinite(sc) and sc < bs:
            best = a_new.copy()
            bs = sc
            step *= 0.97
            no_improve = 0
        else:
            no_improve += 1

        if bs < 1.51:
            break

    return best, bs


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv)/sum^2.
    Uses FFT-based smooth-max optimization with L-BFGS-B and targeted local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 8.0)

    # Collect initializations from provided sequence if available
    provided_seq = None
    try:
        ps = np.asarray(height_sequence_1, dtype=np.float64)
        if ps.ndim == 1 and 0 < len(ps) < 50000 and np.all(np.isfinite(ps)):
            provided_seq = np.clip(ps, 0.0, 1000.0)
            if provided_seq.sum() > 0.01:
                provided_seq = provided_seq / provided_seq.sum()
    except Exception:
        pass

    best_a = None
    best_score = np.inf

    # Phase 1: L-BFGS-B optimization from diverse initializations
    n_values = [512, 1024, 2048, 4096]
    if provided_seq is not None:
        n_values.append(len(provided_seq))

    init_types = list(range(11))

    # Priority: provided sequence first, then smaller n (lower target easier)
    tasks = []
    if provided_seq is not None:
        tasks.append((len(provided_seq), 'provided', None))
    for n in sorted(n_values):
        for rt in init_types:
            tasks.append((n, 'init', rt))

    # Shuffle non-provided tasks a bit for diversity
    np.random.seed(seed)
    provided_tasks = [t for t in tasks if t[1] == 'provided']
    other_tasks = [t for t in tasks if t[1] == 'init']
    rng.shuffle(other_tasks)
    tasks = provided_tasks + other_tasks

    for idx, (n, typ, rt) in enumerate(tasks):
        if time.monotonic() > deadline:
            break

        # Time allocation per task
        time_per_task = max(2.0, min(30.0, (deadline - time.monotonic()) / max(1, len(tasks) - idx)))

        if typ == 'provided' and provided_seq is not None:
            a0 = provided_seq.copy()
        else:
            a0 = make_init(n, rt if rt is not None else 0, rng)

        # Ensure positive for log
        a0 = np.maximum(a0, 1e-8)
        a0 = a0 / a0.sum()
        x0 = np.log(a0)

        # Run L-BFGS-B with time/iteration limits
        beta = 100.0
        t_start = time.monotonic()
        max_iter = max(50, int(time_per_task * 50))
        
        # Wrapper for scipy
        def fn_wrap(x, beta_val=100.0):
            return obj_grad_x(x, beta_val)

        try:
            res = minimize(
                lambda x: fn_wrap(x, beta)[0],
                x0,
                jac=lambda x: fn_wrap(x, beta)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                         'gtol': 1e-5, 'ftol': 1e-8}
            )
            x_opt = res.x
            a_opt = np.exp(x_opt - x_opt.max())
            a_opt = a_opt / a_opt.sum()
            a_opt = np.clip(a_opt, 0.0, 1000.0)
        except Exception:
            a_opt = a0

        sc = score_seq(a_opt)
        t_elapsed = time.monotonic() - t_start

        if np.isfinite(sc) and sc < best_score:
            best_a = a_opt.copy()
            best_score = sc
            if idx % 10 == 0 or sc < 1.8:
                print(f"Phase1 task {idx} (n={n}, type={typ}, rt={rt}): score={sc:.6f}, time={t_elapsed:.1f}s")

        if best_score < 1.51:
            break

    print(f"After Phase 1: best_score={best_score:.6f}, n={len(best_a) if best_a is not None else 0}")

    if best_a is None:
        # Fallback
        best_a = np.ones(1024) / 1024
        best_score = score_seq(best_a)

    # Phase 2: Local search refinement
    ls_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.6)
    best_a, best_score = local_search(best_a, rng, ls_deadline, best_score)
    print(f"After Phase 2: best_score={best_score:.6f}, n={len(best_a)}")

    # Phase 3: Try a few more n values if time remains and score > target
    if best_score > 1.52 and time.monotonic() < deadline:
        for n_try in [256, 768, 3072, 6144]:
            if time.monotonic() > deadline:
                break
            for rt in [0, 4, 5, 7, 8]:
                if time.monotonic() > deadline:
                    break
                a_try = make_init(n_try, rt, rng)
                a_try = np.maximum(a_try, 1e-8)
                a_try = a_try / a_try.sum()
                x_try = np.log(a_try)
                try:
                    res = minimize(
                        lambda x: obj_grad_x(x, 100.0)[0],
                        x_try,
                        jac=lambda x: obj_grad_x(x, 100.0)[1],
                        method='L-BFGS-B',
                        options={'maxiter': 200, 'gtol': 1e-5, 'ftol': 1e-8}
                    )
                    a_res = np.exp(res.x - res.x.max())
                    a_res = a_res / a_res.sum()
                    a_res = np.clip(a_res, 0.0, 1000.0)
                except Exception:
                    a_res = a_try
                sc = score_seq(a_res)
                if np.isfinite(sc) and sc < best_score:
                    best_a = a_res.copy()
                    best_score = sc
                    print(f"Phase3 n={n_try} rt={rt}: score={sc:.6f}")
                    if best_score < 1.51:
                        break
            if best_score < 1.51:
                break

    # Phase 4: Final local search
    if time.monotonic() < deadline:
        best_a, best_score = local_search(best_a, rng, deadline, best_score)
        print(f"After Phase 4: best_score={best_score:.6f}, n={len(best_a)}")

    # Final verification
    final_score = score_seq(best_a)
    if final_score >= best_score + 0.01:
        # Something went wrong, use a safe fallback
        best_a = np.ones(1024) / 1024
        best_score = 2.0

    # Convert to list of floats
    result = [float(max(0.0, min(1000.0, v))) for v in best_a]
    final_check = evaluate_sequence(result)
    print(f"Final: score={final_check:.6f}, n={len(result)}")
    return result
