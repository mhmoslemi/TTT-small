def next_pow2(m):

def fft_convolve_nonneg(a):
    n=a.size
    m=2*n-1

    while N<m: N <<=1

def next_power_of_two(m):

def lin_conv_fft(a, nfft=None):
    n = a.size

    fa = np.fft.rfft(a, nfft)

"""
Search for non-negative step functions with low peak self-convolution.

The evaluator minimizes
    J(a) = 2*n*max(a*a) / (sum(a))^2,
where a*a is the ordinary linear convolution.  The main idea here is that
dense 0/1 sequences behave like random additive Sidon-like sets: their pair-sum
counts are spread fairly evenly, giving J far below the uniform-sequence value
of 2.

Algorithm:
1. Generate many random binary candidates of several lengths and densities.
2. Score them quickly with FFT-based convolution.
3. Refine the best candidate using peak-directed binary swaps and flips,
   re-evaluating moves with FFTs.
4. Return the best sequence found, with an exact evaluator check when feasible.
"""

import math
import time
import numpy as np


def next_power_of_two(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def lin_conv_fft(a, nfft):
    """Linear convolution a*a using zero-padded FFT."""
    fa = np.fft.rfft(a, nfft)
    fa = fa * fa
    out = np.fft.irfft(fa, nfft)
    return out[:2 * a.size - 1]


def score_from_conv(f, S, n):
    """Score from convolution f and sum S for length n."""
    if not np.isfinite(S) or S < 0.01:
        return float("inf")
    if f.size == 0:
        return float("inf")
    M = float(np.max(f))
    if M < 0.0:
        M = 0.0
    return float(2.0 * n * M / (S * S))


def fast_score(a, nfft):
    """Fast FFT-based evaluation."""
    S = float(a.sum())
    f = lin_conv_fft(a, nfft)
    return score_from_conv(f, S, a.size)


def cand_key(item):
    return item[0]


def insert_top(top, item, limit):
    top.append(item)
    top.sort(key=cand_key)
    if len(top) > limit:
        top.pop()
    return top


def make_binary(n, p, rng):
    """Random binary sequence with approximately exactly round(n*p) ones."""
    k = int(round(n * p))
    if k <= 0:
        return np.zeros(n, dtype=np.float64)
    if k >= n:
        return np.ones(n, dtype=np.float64)

    if abs(p - 0.5) < 1e-12:
        a = rng.integers(0, 2, n).astype(np.float64)
    else:
        a = (rng.random(n) < p).astype(np.float64)

    cur = int(a.sum())
    diff = k - cur
    if diff > 0:
        zeros = np.flatnonzero(a == 0.0)
        if zeros.size > 0:
            take = diff if diff < zeros.size else zeros.size
            idx = rng.choice(zeros.size, size=take, replace=False)
            a[zeros[idx]] = 1.0
    elif diff < 0:
        ones = np.flatnonzero(a == 1.0)
        if ones.size > 0:
            take = -diff if -diff < ones.size else ones.size
            idx = rng.choice(ones.size, size=take, replace=False)
            a[ones[idx]] = 0.0
    return a


def get_top_peaks(f, k):
    """Indices of the k largest convolution lags."""
    m = f.size
    if m <= k:
        return np.arange(m, dtype=np.int64)
    return np.argpartition(f, m - k)[m - k:]


def sample_index_where(a, want_one, rng, count_one, count_zero):
    """Sample an index with a[i] > 0 if want_one else a[i] == 0."""
    n = a.size
    if want_one:
        if count_one <= 0:
            return -1
    else:
        if count_zero <= 0:
            return -1

    for _ in range(1000):
        i = int(rng.integers(n))
        if want_one:
            if a[i] > 0.0:
                return i
        else:
            if a[i] == 0.0:
                return i

    if want_one:
        idx = np.flatnonzero(a > 0.0)
    else:
        idx = np.flatnonzero(a == 0.0)
    if idx.size == 0:
        return -1
    return int(idx[int(rng.integers(idx.size))])


def perturb_binary(a, rng, num_flips):
    """Return a copy with num_flips random bits flipped."""
    a = a.copy()
    n = a.size
    for _ in range(num_flips):
        i = int(rng.integers(n))
        if a[i] > 0.5:
            a[i] = 0.0
        else:
            a[i] = 1.0
    return a


def propose_move(a, S, top_peaks, rng):
    """
    Propose a one-bit-change or swap from the current binary sequence.

    Returns a new array, or None if no move is possible.
    """
    n = a.size
    count_one = int(np.rint(S))
    count_zero = n - count_one
    if count_one < 0:
        count_one = 0
    if count_zero < 0:
        count_zero = 0

    u = rng.random()

    # Main move: move a 1 that participates in a high peak to a random 0 slot.
    if u < 0.60 and top_peaks.size > 0:
        k = int(top_peaks[int(rng.integers(top_peaks.size))])
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1

        if lo <= hi:
            for _ in range(40):
                i = int(rng.integers(lo, hi + 1))
                j = k - i
                if 0 <= j < n and a[i] > 0.0 and a[j] > 0.0:
                    pos = i if rng.random() < 0.5 else j
                    z = sample_index_where(a, False, rng, count_one, count_zero)
                    if z >= 0 and z != pos:
                        new_a = a.copy()
                        new_a[pos] = 0.0
                        new_a[z] = 1.0
                        return new_a
                    new_a = a.copy()
                    new_a[pos] = 0.0
                    return new_a

    # Secondary move: random swap of a 1 and a 0.
    if u < 0.85:
        one = sample_index_where(a, True, rng, count_one, count_zero)
        zero = sample_index_where(a, False, rng, count_one, count_zero)
        if one >= 0 and zero >= 0 and one != zero:
            new_a = a.copy()
            new_a[one] = 0.0
            new_a[zero] = 1.0
            return new_a

        if count_one > 0:
            one = sample_index_where(a, True, rng, count_one, count_zero)
            if one >= 0:
                new_a = a.copy()
                new_a[one] = 0.0
                return new_a

        if count_zero > 0:
            zero = sample_index_where(a, False, rng, count_one, count_zero)
            if zero >= 0:
                new_a = a.copy()
                new_a[zero] = 1.0
                return new_a

        return None

    # Tertiary move: density-changing flip.
    density = S / float(n)
    if density > 0.5 or rng.random() < 0.5:
        one = sample_index_where(a, True, rng, count_one, count_zero)
        if one >= 0:
            new_a = a.copy()
            new_a[one] = 0.0
            return new_a

    zero = sample_index_where(a, False, rng, count_one, count_zero)
    if zero >= 0:
        new_a = a.copy()
        new_a[zero] = 1.0
        return new_a

    return None


def accept_score(new_score, old_score, temperature, rng):
    """Strict improvement plus tiny simulated-annealing escape probability."""
    if not np.isfinite(new_score):
        return False
    if not np.isfinite(old_score):
        return True
    if new_score < old_score:
        return True
    if temperature > 0.0:
        delta = (old_score - new_score) / temperature
        if delta > -700.0:
            return rng.random() < math.exp(delta)
    return False


def local_search_binary(a, deadline, rng, nfft):
    """
    Local search over binary sequences using full FFT re-evaluation.

    Uses peak-directed swaps/flips and occasional restarts from the best
    sequence found so far.
    """
    a = (np.asarray(a, dtype=np.float64) > 0.5).astype(np.float64)
    n = a.size
    S = float(a.sum())
    f = lin_conv_fft(a, nfft)
    score = score_from_conv(f, S, n)

    best = a.copy()
    best_score = score
    top_peaks = get_top_peaks(f, 64)

    temperature = 1e-4
    stagnant = 0
    iter_count = 0
    restarts = 0

    while time.monotonic() + 1.0 < deadline:
        new_a = propose_move(a, S, top_peaks, rng)
        if new_a is None:
            stagnant += 1
        else:
            new_S = float(new_a.sum())
            new_f = lin_conv_fft(new_a, nfft)
            new_score = score_from_conv(new_f, new_S, n)
            iter_count += 1

            if accept_score(new_score, score, temperature, rng):
                a = new_a
                S = new_S
                f = new_f
                score = new_score
                top_peaks = get_top_peaks(f, 32)

                if score < best_score - 1e-15:
                    best = a.copy()
                    best_score = score
                    stagnant = 0
                else:
                    stagnant += 1
            else:
                stagnant += 1

        if stagnant > 20000:
            if restarts < 80:
                a = perturb_binary(best, rng, 16)
                S = float(a.sum())
                f = lin_conv_fft(a, nfft)
                score = score_from_conv(f, S, n)
                top_peaks = get_top_peaks(f, 64)
                stagnant = 0
                restarts += 1
            else:
                break

        if temperature > 1e-8:
            temperature *= 0.99999

        if iter_count % 5000 == 0:
            print("local iter", iter_count, "score", score, "best", best_score)

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point.

    Searches random dense binary sequences and refines them with local search.
    Returns a list of non-negative floats.
    """
    start = time.monotonic()
    budget = max(10.0, float(budget_s))

    # Leave a margin before the absolute hard stop.
    hard_stop = start + max(1.0, budget - 5.0)
    deadline = start + max(1.0, budget - 8.0)

    rng = np.random.default_rng(seed)

    # Candidate generation.  Dense binary random sequences are the key.
    lengths = (4096, 8192, 16384)
    ps = (0.40, 0.45, 0.50, 0.55, 0.60)

    tasks = []
    for p in ps:
        for L in lengths:
            tasks.append((L, p))

    nffts = {}
    for L in lengths:
        nffts[L] = next_power_of_two(2 * L - 1)

    top = []
    gen_deadline = min(deadline, start + max(20.0, budget * 0.18))
    ti = 0
    count = 0

    while time.monotonic() < gen_deadline:
        L, p = tasks[ti % len(tasks)]
        ti += 1

        a = make_binary(L, p, rng)
        S = float(a.sum())
        if S < max(50.0, 0.05 * L):
            continue

        f = lin_conv_fft(a, nffts[L])
        sc = score_from_conv(f, S, L)
        if np.isfinite(sc):
            top = insert_top(top, (sc, a), 12)
            count += 1
            if count % 50 == 0:
                print("generated", count, "best internal", top[0][0])

    if top:
        best_a = top[0][1].copy()
        best_score = float(top[0][0])
    else:
        # Very conservative fallback; random generation should make this unreachable.
        best_a = np.ones(1024, dtype=np.float64)
        nfft = next_power_of_two(2 * 1024 - 1)
        best_score = fast_score(best_a, nfft)

    print("initial best internal score", best_score, "length", int(best_a.size))

    # Local refinement, but always leave time for a final safe return.
    local_deadline = min(deadline, start + max(60.0, budget - 50.0))
    if local_deadline > time.monotonic() + 5.0:
        L = int(best_a.size)
        nfft = next_power_of_two(2 * L - 1)
        local_best, local_score = local_search_binary(best_a, local_deadline, rng, nfft)
        if np.isfinite(local_score) and local_score < best_score:
            best_a = local_best
            best_score = local_score
        print("post-local internal score", best_score, "length", int(best_a.size))

    # Optionally verify with the official evaluator for reasonably small final
    # candidates.  This is not required for validity, but guards against
    # implementation-level discrepancies.
    candidates = []
    candidates.append(best_a)
    for item in top[:4]:
        candidates.append(item[1])

    exact_best = None
    exact_best_score = float("inf")

    # Be conservative with exact evaluations because np.convolve is O(n^2).
    for arr in candidates:
        if arr.size > 10000:
            continue
        if time.monotonic() + 12.0 > hard_stop:
            break
        try:
            seq = [float(x) for x in arr]
            sc = float(evaluate_sequence(seq))
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < exact_best_score:
            exact_best_score = sc
            exact_best = seq
            print("exact candidate score", sc)

    if exact_best is not None:
        return exact_best

    # Return internally best sequence.
    return [float(x) for x in best_a]
