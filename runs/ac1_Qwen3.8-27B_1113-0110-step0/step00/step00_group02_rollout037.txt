```python
    n = len(x)
    size = next_power_of_2(2*n - 1)
    X = np.fft.rfft(x, size)
    result = np.fft.irfft(X * X, size)
    return result[:2*n-1]
```

```python
def subgradient_step(x, lr):
    """One step of projected subgradient descent on max(c(k))."""
    n = len(x)
    C = fast_convolve(x)
    # Find top-K lags
    K = min(10, len(C))
    top_lags = np.argsort(C)[::-1][:K]
    # Subgradient: average of gradients at top lags
    sg = np.zeros(n)
    for k in top_lags:
        # d(c[k])/d(x[i]) = 2*x[k-i] for 0 <= k-i < n
        # i.e., for i in [max(0, k-n+1), min(k, n-1)]:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            # x[k-i] for i = i_lo, ..., i_hi
            # = x[k-i_lo], x[k-i_lo-1], ..., x[k-i_hi]
            # = x[k-i_hi : k-i_lo+1][::-1]
            j_lo = k - i_hi
            j_hi = k - i_lo
            sg[i_lo:i_hi+1] += x[j_lo:j_hi+1][::-1]
    sg /= K
    # Gradient step
    x_new = x - 2 * lr * sg  # factor of 2 because d(c)/d(x) = 2*x_shift
    # Project onto simplex {x >= 0, sum x = 1}
    x_new = project_simplex(x_new)
    return x_new, C.max()

def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

"""
Minimize 2*n*max(convolve(x,x))/(sum x)^2 for non-negative x of length n.
Uses projected subgradient descent with FFT-based convolution, multiple 
initializations, and adaptive learning rate.
"""
import numpy as np
import time

def fast_convolve(x):
    """Full linear convolution via FFT."""
    n = len(x)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    X = np.fft.rfft(x, size)
    return np.fft.irfft(X * X, size)[:2*n-1]

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    for k in top_lags:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            j_lo = k - i_hi
            j_hi = k - i_lo
            sg[i_lo:i_hi + 1]
