def compute_autocorrelation(a):
    """Compute the full aperiodic autocorrelation of a."""
    return np.convolve(a, a, mode='full')

def softmax_gradient_descent_step(a, tau, lr):
    """One step of softmax gradient descent to minimize max autocorrelation."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return a.copy()
    
    # Compute autocorrelation
    c = np.convolve(a, a, mode='full')
    
    # Compute softmax weights (focus on peaks)
    c_scaled = tau * c
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / np.sum(exp_c)
    
    # Compute gradient: grad_j = 2 * sum_l a[l] * w[j+l]
    # = 2 * convolve(w, a[::-1], mode='full')[j + n - 1] for j=0..n-1
    conv = np.convolve(w, a[::-1], mode='full')
    grad = 2.0 * conv[n-1 : 2*n-1]
    
    # Update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    
    # Renormalize to maintain sum = n (scale invariance)
    S_new = np.sum(a_new)
    if S_new > 1e-10:
        a_new = a_new * n / S_new
    
    return a_new
