import time
"""
Minimizes 2n*max(auto-convolution)/(sum)^2 using FFT-based gradient descent
on the auto-convolution peak with softmax-weighted subgradients.
Multiple initializations are explored to avoid local minima.
"""

def _auto_conv(a):
    """Compute linear auto-convolution a*a via FFT."""
    n = len(a)
    size = 1 << (2*n-2).bit_length()
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2*n-1]

def _score(a):
    """Compute the evaluation score."""
    n = len(a)
    c = _auto_conv(a)
    s = a.sum()
    if s < 1e-9:
        return 1e18
    return float(2.0 * n * c.max() / (s * s))

def _soft_grad(a, beta):
    """Compute softmax-weighted gradient of auto-convolution peak."""
    n = len(a)
    c = _auto_conv(a)
    max_c = c.max()
    # Softmax weights

    conv = np.convolve(a, w_rev, 'full')
