        M *= 2

    lr = 0.01

    best_score = evaluate_sequence(best_a.tolist())

        A = np.fft.rfft(a, M)
        B = A * A  # Note: this is (A)^2, not |A|^2
        b = np.fft.irfft(B, M)[:2*n-1]
        b = np.maximum(b, 0)  # Numerical stability
        
        # Soft-max weights
        b_shifted = b - b.max()
        weights = np.exp(beta * b_shifted)
        weights /= weights.sum()

        AR = np.fft.rfft(ar, M)
        grad = np.fft.irfft(WR * AR, M)[:n] * 2
        
        # Subtract mean for projection onto sum=const
        grad -= grad.mean()
        
        # Update
        a = a - lr * grad
        a = np.maximum(a, 0)
        a = a / a.sum()  # Renormalize
        
        # Periodically check score
        if step % 1000 == 0:

            if score < best_score:

                best_a = a.copy()
            print(f"Step {step}: score={score:.6f}, best={best_score:.6f}, beta={beta:.1f}, max_b={b.max():.6f}")
        
        # Anneal beta
        beta *= 1.001
        # Decay learning rate
        lr *= 0.9999
    
    # Final evaluation
    final_score = evaluate_sequence(best_a.tolist())

    M = 1
    while M < 2 * n - 1:
        M *= 2

    print(f"Initial score: {best_score:.6f}, n={n}, M={M}")

    lr = 0.005  # Learning rate
    
    for step in range(1000000):
        if time.monotonic() > deadline:
            break
        
        # Compute b = a * a (linear convolution) via FFT
        A = np.fft.fft(a, M)
        B = A * A  # (A)^2, not |A|^2
        b = np.fft.ifft(B).real[:2*n-1]
        b = np.maximum(b, 0)  # Numerical stability
        
        # Softmax weights
        b_max = b.max()
        exp_b = np.exp(beta * (b - b_max))
        w = exp_b / exp_b.sum()
        
        # Gradient: 2 * sum_k w_k * a_{k-j} = 2 * correlate(w, a)[j]
        W = np.fft.fft(w, M)  # Pad w to M
        grad = 2 * np.fft.ifft(W * np.conj(A)).real[:n]
        
        # Project gradient onto sum=0 subspace
        grad = grad - grad.mean()
        
        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_new_sum
        
        a = a_new
        
        # Anneal
        if step < 5000:
            beta = 10.0 + step * 0.1  # Increase from 10 to 600
        else:
            beta *= 1.0005
        
        # Decay learning rate
        lr = 0.005 / (1 + step * 0.0001)
        
        # Evaluate periodically

            score = float(evaluate_sequence(a.tolist()))
            if score < best_score:
