```python
import numpy as np
from scipy.optimize import linprog

def solve_lp(a, M):
    """Solve max sum(g) s.t. g >= 0, (a*g)_k <= M for all k."""
    n = len(a)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (a*g)_k <= M for k = 0, ..., 2n-2
    # (a*g)_k = sum_{i: 0<=i<n, 0<=k-i<n} a_i * g_{k-i}
    # For fixed k, the coefficients of g_j are a_{k-j} for j in [max(0, k-n+1), min(n-1, k)]
    
    # Build the constraint matrix A (shape (2n-1, n))
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    b = np.full(2*n-1, M)
    
    # g >= 0 is handled by bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return result.x
```

```python
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    # Column j: A[k, j] = a[k-j] for k in [j, j+n-1], 0 otherwise
```

```python
from scipy.optimize import linprog

def project_to_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_gradient(a, beta):

    n = len(a)
    c = np.convolve(a, a, mode='full')
    max_c = c.max()

    p = exp_bc / exp_bc.sum()

    g = np.empty(n)
    for i in range(n):
        g[i] = np.dot(a, p[i:i+n])
    g *= 2.0 / beta
    
    return g, c, p
```

```python
def project_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_score_fast(a):

def softmax_gradient(a, beta):
    """Compute gradient of soft-max objective."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    max_c = c.max()
    exp_bc = np.exp(beta * (c - max_c))
    p = exp_bc / exp_bc.sum()
    
    # g_i = (2/beta) * sum_j a_j * p_{i+j}
```
