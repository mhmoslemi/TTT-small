    return f, current_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer:
    1. Try multiple initializations and sequence lengths
    2. Gradient descent with top-K autocorrelation lags
    3. LP-based refinement
    4. Local block-perturbation search
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Phase 1: Try multiple initializations
    best = None
    best_score = np.inf

    # Use provided base if available
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim == 1 and base.size > 0:
            inits = generate_initializations(base.size, rng)
            # Include base itself
            inits.insert(0, base.copy())
            for cand in inits:
                cand = np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)
                if np.sum(cand) < 0.01:
                    cand = cand + 1.0
                cand = cand / np.sum(cand)
                s = compute_score(cand)
                if s < best_score:
                    best_score = s
                    best = cand.copy()
    except Exception:
        pass

    # Try other lengths
    for try_n in [1000, 2048, 4096, 6733, 8192]:
        if time.monotonic() > deadline - 120:
            break
        inits = generate_initializations(try_n, rng)
        for cand in inits:
            if time.monotonic() > deadline - 120:
                break
            cand = np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)
            if np.sum(cand) < 0.01:
                cand = cand + 1.0
            cand = cand / np.sum(cand)
            s = compute_score(cand)
            if s < best_score:
                best_score = s
                best = cand.copy()
                print("  New best init: n={}, score={:.6f}".format(try_n, best_score))

    if best is None:
        best = np.ones(1024, dtype=np.float64)
        best_score = compute_score(best)

    print("Phase 1 done: n={}, score={:.6f}".format(len(best), best_score))

    # Phase 2: Gradient descent
    f = best.copy()
    current_score = best_score
    f, current_score = run_gradient_descent(f, deadline - 180, rng, current_score)
    print("Phase 2 done: score={:.6f}".format(current_score))

    # Phase 3: LP refinement
    if time.monotonic() < deadline - 60:
        f, current_score = run_lp_refinement(f, deadline - 60, rng, current_score, max_steps=20)
        print("Phase 3 done: score={:.6f}".format(current_score))

    # Phase 4: Local search
    if time.monotonic() < deadline - 10:
        f, current_score = run_local_search(f, deadline - 5, rng, current_score)
        print("Phase 4 done: score={:.6f}".format(current_score))

    # Final: short gradient polish
    if time.monotonic() < deadline - 3:
        f, current_score = run_gradient_descent(f, deadline - 2, rng, current_score)
        print("Final polish: score={:.6f}".format(current_score))

    f = np.clip(f, 0.0, 1000.0)
    final_list = [float(x) for x in f]

    try:
        verified = evaluate_sequence(final_list)
        print("Verified score: {:.6f}".format(verified))
    except Exception:
        verified = current_score

    return final_list
