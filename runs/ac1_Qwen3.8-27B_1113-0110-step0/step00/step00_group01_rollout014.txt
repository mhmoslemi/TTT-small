"""
Anti-correlation step design.

Minimizes J(a) = 2n * max_t [(a*a)_t] / (sum a)^2  (full linear
convolution, a >= 0).  J is scale invariant, so the search runs with
sum(a) = 1 and minimizes M = max convolution coefficient.

Method:
  * good initializers: exponentially decaying combs a_i ~ (i/c)^q e^{-alpha i/n},
    two-sided (reflected) versions, and blends with the incumbent.
  * smooth-max descent: c = a*a (FFT), M = max c, w = softmax(beta*(c-M)),
    grad = 2 * conv(w, a) - 2*M*a, backtrack line search a -> clip(a-eta*grad),
    renormalized to sum 1 (reduces M monotonically).
  * exploit hidden symmetry: mirrored average (a + flip(a))/2 never increases M.
  * several branch lengths {4096, 6121, 12242, 16384} run interleaved;
    adaptive restarts using new random initializers / noise / blends escape
    plateaus.
  * at the deadline the top fast candidates are certified with the
    authoritative evaluate_sequence() and the best certified list returned.
"""

import math
import time

import numpy as np

_NP2_CACHE = {}


def _npow2_atleast(L):
    v = _NP2_CACHE.get(L)
    if v is not None:
        return v
    r = 1
    while r < L:
        r <<= 1
    _NP2_CACHE[L] = r
    return r


def _conv_self(a):
    """Full linear auto-convolution c[t] = sum_i a_i a_{t-i}, t=0..2n-2."""
    n = a.size
    N = _npow2_atleast(2 * n - 1)
    F = np.fft.rfft(a, N)
    P = F.real * F.real + F.imag * F.imag
    c = np.fft.irfft(P, N)
    return c[: 2 * n - 1]


def _max_conv(a):
    return float(_conv_self(a).max())


def _conv_wa(w, a):
    """u = conv(w, a) for w of length 2n-1, a of length n; returns first n."""
    n = a.size
    L = (2 * n - 1) + n - 1
    N = _npow2_atleast(max(2 * n - 1, L))
    Fw = np.fft.rfft(w, N)
    Fa = np.fft.rfft(a, N)
    u = np.fft.irfft(Fw * Fa, N)
    return u[:n]


def _norm(a):
    """Clip to non-negative and rescale to sum 1."""
    x = np.asarray(a, dtype=np.float64)
    if x.ndim != 1 or x.size < 8:
        return None
    if not np.all(np.isfinite(x)):
        return None
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s <= 1e-300:
        return None
    return x / s


def _sigmoid(x):
    return 1.0 / (1.0 + math.exp(-x))


def _mk_decay(n, alpha, q, rng=None):
    """One-sided decaying initializer (i*c)^q e^{-alpha i/n}."""
    base = np.arange(n, dtype=np.float64)
    base = base * 0.99999999 / (n - 1) + 1.0e-9
    a = base ** q * np.exp(-alpha * base)
    if rng is not None and rng.random() < 0.10:
        a = a * np.exp(0.10 * rng.standard_normal(n))
    return _norm(a)


def _mk_reflec(n, alpha, q, rng=None):
    """Two-sided initializer peaked near index n*mu."""
    mu = 0.28 + 0.55 * _sigmoid(float(rng.standard_normal()))
    base = np.arange(n, dtype=np.float64) / n
    d = np.maximum(np.abs(base - mu), 1.0e-9)
    a = d ** q * np.exp(-alpha * d)
    return _norm(a)


def _scan_init(n, rng, with_tail3):
    """Quick scan of initializer family; returns (a, M, (alpha,q))."""
    best = (math.inf, None, None)
    for alpha in (0.4, 0.7, 1.1, 1.7, 2.6, 4.0, 6.5):
        for q in (-1.0, 0.0, 1.0, 2.0):
            a = _mk_decay(n, alpha, q, None)
            M = _max_conv(a)
            if M < best[0]:
                best = (M, a, (alpha, q))
    if with_tail3:
        for _ in range(3):
            for alpha in (0.7, 1.3, 2.2):
                for q in (0.0, 1.0):
                    for mu in (0.33, 0.5, 0.66):
                        base = np.arange(n, dtype=np.float64) / n
                        d = np.maximum(np.abs(base - mu), 1.0e-9)
                        a = _norm(d ** q * np.exp(-alpha * d))
                        M = _max_conv(a)
                        if M < best[0]:
                            best = (M, a, (alpha, q))
    return best


def _step(a, Mcur, beta):
    """One smoothed-descent iteration.  Returns (x, Mx) or None."""
    n = a.size
    c = _conv_self(a)
    M = float(c.max())
    if not math.isfinite(M) or M <= 0.0:
        return None
    w = np.exp(beta * (c - M))
    ws = w.sum()
    if ws <= 0.0 or not math.isfinite(ws):
        return None
    w = w / ws
    u = _conv_wa(w, a)
    g = 2.0 * u - 2.0 * M * a
    ng = float(np.linalg.norm(g))
    if not math.isfinite(ng) or ng < 1e-300:
        return None
    eta = 0.5 / ng
    for _ in range(12):
        x = a - eta * g
        x = np.clip(x, 0.0, None)
        s = x.sum()
        if s <= 1e-300:
            eta *= 0.5
            continue
        x = x / s
        Mx = _max_conv(x)
        if math.isfinite(Mx) and Mx <= M * (1.0 - 1e-12) - 1e-22:
            return x, Mx
        eta *= 0.5
    return None


def _try_mirror(b):
    """(a + flip(a))/2 can never increase the max coefficient."""
    a = b['a']
    m = np.maximum(0.5 * (a + a[::-1]) - 1e-18, 0.0)
    s = m.sum()
    if s <= 1e-300:
        return
    m = m / s
    Mm = _max_conv(m)
    if math.isfinite(Mm) and Mm < b['M'] * (1.0 - 5e-7):
        b['a'], b['M'], b['J'] = m, Mm, 2.0 * a.size * Mm


def _adapt_beta(b):
    v = 8.0 / b['M'] if b['M'] > 1e-300 else 1e4
    b['beta'] = float(min(1.0e5, max(40.0, 0.65 * b['beta'] + 0.35 * v)))


def _restart(b, gbest, rng):
    """Escape plateau: fresh random initializer / noise / blend with incumbent."""
    n = b['n']
    a0 = b['a']
    M0 = b['M']
    best_a, best_m = a0, M0

    trials = []
    if gbest is not None:
        c = _norm(0.6 * a0 + 0.4 * gbest)
        if c is not None:
            trials.append(c)
    for _ in range(2):
        c = _norm(a0 * np.exp(0.12 * rng.standard_normal(n)))
        if c is not None:
            trials.append(c)
    base_al = (0.4, 0.7, 1.1, 1.7, 2.6, 4.0)
    for q in (-1.0, 0.0, 1.0, 2.0):
        for alpha in base_al:
            trials.append(_mk_decay(n, alpha * (0.85 + 0.3 * rng.random()), q, rng))
    for q in (0.0, 1.0):
        for alpha in base_al[:3]:
            trials.append(_mk_reflec(n, alpha, q, rng))

    for t in trials:
        if t is None:
            continue
        M = _max_conv(t)
        if math.isfinite(M) and M < best_m:
            best_m, best_a = M, t
    b['a'] = best_a
    b['M'] = best_m
    b['J'] = 2.0 * n * best_m
    b['fails'] = 0
    b['noimp'] = 0
    _adapt_beta(b)


def propose_candidate(seed=7, budget_s=1800.0, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s) if budget_s is not None else 1800.0
    rng = np.random.default_rng(seed)
    t_end = t0 + max(90.0, budget) - 45.0

    pads = []

    def _init_pad(n):

    _init_pad(4096)
    _init_pad(6121)

    best_a = min(pads, key=lambda b: b['J'])['a'].copy()

    parent = None

    print("init best fast J = %.6f (n=%d)" % (best_J, best_a.size), flush=True)
