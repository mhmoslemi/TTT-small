def greedy_update(f, M):
    """Compute max sum g s.t. (f*g)[k] <= M, g >= 0 using greedy."""
    n = len(f)
    g = np.zeros(n)
    # Process in some order (e.g., random or reverse)
    for i in range(n):
        # Max value of g[i] given current g
        # (f*g)[k] for k = i to i+n-1 includes f[k-i]*g[i]
        # Current (f*g)[k] without g[i]:
        curr = np.convolve(f, g, mode='full')
        # g[i] contributes f[k-i]*g[i] to position k (for valid k)
        # We need: curr[k] + f[k-i]*g[i] <= M for all k where f[k-i] > 0
        # k-i ranges from 0 to n-1, so k ranges from i to i+n-1
        # f[k-i] for k-i in [0, n-1] is just f[0], f[1], ..., f[n-1]
        # So constraint: g[i] <= (M - curr[k]) / f[k-i] for all k in [i, i+n-1] with f[k-i] > 0
        valid = np.arange(i, min(i+n, 2*n-1))
        f_vals = f[valid - i]  # f[0], f[1], ... for the valid range
        mask = f_vals > 0
        if mask.any():
            max_g = np.min((M - curr[valid[mask]]) / f_vals[mask])
            g[i] = max(0, max_g)
        else:
            g[i] = 0  # no constraint binds
    return g
