"""
LP-based iterative optimization for minimizing the autocorrelation peak ratio.
Uses the Frank-Wolfe style approach: fix one side of the bilinear form, solve an LP
for the other side, then blend. Multiple diverse starting points are tried, and
periodic random kicks help escape local minima.
"""

import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time


def compute_score_fast(a):
    """Compute 2*n*max(a*a) / (sum(a))^2 using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (S * S)


def build_lp_matrix(f):
    """Build sparse constraint matrix A where A[k,i]=f[k-i] if 0<=k-i<n."""
    n = len(f)
    rows_list = []
    cols_list = []
    vals_list = []
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        cnt = k_end - i
        if cnt <= 0:
            continue
        rows_list.append(np.arange(k_start, k_end))
        cols_list.append(np.full(cnt, i, dtype=np.int32))
        vals_list.append(f[:cnt])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_for_g(f, M):
    """Solve max sum(g) s.t. g>=0, (f*g)_k <= M for all k. Returns g or None."""
    n = len(f)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-6,
                               'primal_feasibility_tolerance': 1e-6})
    if res.status != 0:
        return None
    return res.x


def make_sinusoidal(n, p, alpha, offset=0):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * (x + offset) / p)
    return np.clip(a, 0.0, 1000.0)


def make_parabolic(n, alpha):
    x = np.linspace(-1.0, 1.0, n)
    a = 1.0 + alpha * x * x
    return np.clip(a, 0.0, 1000.0)


def make_random_exp(n, rng, sigma):
    return np.exp(rng.normal(0.0, sigma, n))


def make_block_pattern(n, rng, n_blocks, alpha):
    a = np.ones(n)
    block_size = n // n_blocks
    for i in range(n_blocks):
        sign = 1.0 if rng.random() < 0.5 else -1.0
        a[i * block_size:(i + 1) * block_size] *= (1.0 + sign * alpha)
    return np.clip(a, 0.0, 1000.0)
