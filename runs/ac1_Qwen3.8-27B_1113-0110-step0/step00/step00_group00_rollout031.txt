import time

def _fft_convolve_full(a, b):

    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:n]

    return _fft_convolve_full(a, a)

def _fft_conv(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    sz = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:n]


def _score(a):
    """Fast score computation for non-negative a."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = _fft_conv(a, a)
    return 2.0 * n * c.max() / (S * S)


def _smooth_max_grad(a, T):
    """
    Compute smoothed max and its gradient using softmax.
    Returns (smooth_max, gradient_array_of_length_n).
    """
    n = len(a)
    c = _fft_conv(a, a)
    # Softmax for numerical stability
    c_shifted = T * (c - c.max())
    weights = np.exp(c_shifted)
    weights /= weights.sum()
    smooth_max_val = c.max() + np.log(weights.sum()) / T  # This is the log-sum-exp / T
    # Actually: smooth_max = (1/T) * log(sum(exp(T*c)))
    # = c.max() + (1/T) * log(sum(exp(T*(c - c.max()))))
    # = c.max() + (1/T) * log(sum(exp(c_shifted)))
    # Since weights = exp(c_shifted) / sum(exp(c_shifted)):
    # sum(exp(c_shifted)) = ... let me just compute directly
    log_sum_exp = c.max() * T + np.log(np.sum(np.exp(T * (c - c.max()))))
    smooth_max_val = log_sum_exp / T
    
    # Gradient: d(smooth_max)/da[p] = 2 * sum_k weights[k] * a[k-p] * 1(0<=k-p<n)
    # = 2 * sum_j a[j] * weights[j+p] (where j+p < 2n-1)
    # This is: 2 * cross_correlation(a, weights)[0:n]

    if n_act < 1:

        n_act = 1.0

    conv = _fft_conv(a, w_flipped)
    h = conv[n-1:2*n-1][::-1]
    dM = 2.0 * h / n_act
    return dM, M, c

A "block" sequence (a = [1, 1, ..., 1, 0, 0, ..., 0] with n/2 ones):

c_max = n/2 (at the center)
score = 2n * (n/2) / (n/2)^2 = 2n * n/2 / (n^2/4) = 2n * 2/n = 4

A "spread out" sequence with some structure... 

I think the key insight is: for non-negative sequences, the autocorrelation peak is hard to reduce below the "trivial" value because all contributions are positive. The best we can do is to make the sequence as "irregular" as possible to avoid constructive interference at any single lag.

"""
Minimizes 2n*max(autocorrelation)/sum^2 using FFT-accelerated smoothed gradient
descent (log-sum-exp), exact subgradient descent, and local search polishing.
Multiple restarts from different initializations (uniform perturbation, random,
structured) with adaptive time allocation.
"""
import numpy as np
import time


def _fft_conv(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    sz = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:n]


def _score(a):
    """Fast score for non-negative array a (assumed sum > 0)."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    c = _fft_conv(a, a)
    return 2.0 * n * float(c.max()) / (S * S)


def _smooth_grad(a, T):
    """
    Smoothed max(c) and its gradient via softmax weights.
    Returns (smoothed_max_value, gradient_array_length_n).
    """
    n = len(a)
    c = _fft_conv(a, a)
    cmax = c.max()
    # Numerically stable log-sum-exp
    shifted = T * (c - cmax)
    max_shifted = shifted.max()
    log_sum_exp = cmax * T + max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    smooth_val = log_sum_exp / T
    # Softmax weights
    weights = np.exp(shifted - max_shifted)
    weights /= weights.sum()
    # Gradient: d(smooth_max)/da[p] = 2 * sum_k weights[k] * a[k-p] * 1(0<=k-p<n)
    # = 2 * h[p] where h[p] = sum_j a[j]*weights[j+p]
    # h = (a * weights_flipped)[n-1 : 2n-1][::-1]
    wf = weights[::-1]
    conv = _fft_conv(a, wf)
    h = conv[n - 1:2 * n - 1][::-1]
    grad = 2.0 * h
    return smooth_val, grad


def _exact_subgrad(a):
    """Exact subgradient of max(c). Returns (dM_array, M, c)."""
    n = len(a)
    c = _fft_conv(a, a)
    M = float(c.max())
    tol = 1e-10 * max(M, 1e-30)
    active = (c >= M - tol).astype(np.float64)
    n_act = float(active.sum())
    if n_act < 1.0:
        active[int(np.argmax(c))] = 1.0
        n_act = 1.0
    wf = active[::-1]
    conv = _fft_conv(a, wf)
    h = conv[n - 1:2 * n - 1][::-1]
    dM = 2.0 * h / n_act
    return dM, M, c


def _normalize(a):
    """Project to simplex {a>=0, sum=1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _optimize_smoothed(a0, n_iters, T_init, T_final, lr_init, lr_final, time_budget):
    """Smoothed gradient descent (multiplicative) on the simplex."""
    a = _normalize(a0.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_conv(a, a).max())
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break
        frac = it / max(1, n_iters - 1)
        T = T_init * (T_final / T_init) ** frac
        lr = lr_init * (lr_final / lr_init) ** frac

        sm, grad = _smooth_grad(a, T)
        sc = 2.0 * n * sm
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Project gradient onto tangent space of simplex
        grad -= grad.mean()
        # Multiplicative update for positivity
        scale = lr * grad / (n * max(sm, 1e-30))
        a = a * np.exp(-np.clip(scale, -2.0, 2.0))
        a = _normalize(a)

    return best_a, best_score


def _optimize_subgrad(a0, n_iters, lr_init, time_budget):
    """Exact subgradient descent (multiplicative) on the simplex."""
    a = _normalize(a0.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_conv(a, a).max())
    lr = lr_init
    stuck = 0
    cooldown = 200
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break
        dM, M, c = _exact_subgrad(a)
        sc = 2.0 * n * M
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck >= cooldown:
            lr *= 0.6
            stuck = 0
            if lr < 1e-8:
                break

        grad = 2.0 * n * dM
        grad -= grad.mean()
        scale = lr * grad / (n * max(M, 1e-30))
        a = a * np.exp(-np.clip(scale, -3.0, 3.0))
        a = _normalize(a)

    return best_a, best_score


def _local_search(a0, budget_s, seed=42):
    """Block perturbation local search with FFT scoring."""
    rng = np.random.default_rng(seed)
    a = _normalize(a0.astype(np.float64))
    best = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    while time.monotonic() - t0 < budget_s:
        candidate = best.copy()
        n_blocks = int(rng.integers(1, 4))
        for _ in range(n_blocks):
            w = max(2, n // int(rng.integers(50, 500)))
            p = int(rng.integers(0, max(1, n - w)))
            candidate[p:p + w] *= np.exp(rng.normal(0, 0.12, size=w))
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.maximum(candidate, 0.0)
        sc = _score(candidate)
        if np.isfinite(sc) and sc < best_sc:
            best = candidate
            best_sc = sc

    return best, best_sc
