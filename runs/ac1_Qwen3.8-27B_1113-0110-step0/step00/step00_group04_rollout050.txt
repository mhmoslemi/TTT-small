Minimize  f(a) = 2*n*max(conv(a,a)) / (sum a)^2   over a >= 0.

The objective is scale invariant, so internally we fix sum(a)=1 and minimize
F(a) = max_k (a*a)_k.  We run many short restarts of a smoothed (log-sum-exp)
projected sub-gradient method with annealing: at each step we spread probability
mass p over the convolution lags, compute the sub-gradient of the active lags
(grad_j = 2 * sum_i p[j+i] a[i]), step against it, clip to the non-negative
orthant and renormalize.  A backtracking line-search accepts only improving
steps.  Diverse structured initializers plus random and the supplied seed
sequence give good coverage; the global best (by the authoritative evaluator)
is retained and returned.  Because f = 2*C with C a pure shape constant, we
also try a smaller length to afford more iterations at the same wall time.

import time
import numpy as np


def _normalize(a):
    s = a.sum()
    if s <= 0:
        return np.ones_like(a)
    return a / s


def _objective_core(a):
    """F = max(convolve(a,a)) for non-negative a (assumed normalized to sum 1)."""
    b = np.convolve(a, a, mode="full")
    return float(b.max())


def _softmax(a, tau):
    a = a - a.max()
    z = np.exp(a / tau)
    s = z.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(a, 1.0 / len(a))
    return z / s


def _subgrad(p, a):
    """grad_j = 2 * sum_i p[j+i] * a[i]  for j = 0..n-1  (subgrad of smoothed max)."""
    # np.correlate(p, a, 'full')[j] = sum_n p[n+j] * a[n]
    return 2.0 * np.correlate(p, a, mode="full")[: a.size]


def _one_restart(n, init, deadline, rng, tag=""):

    a = _normalize(a)
    F_local = _objective_core(a)

    best_a = a.copy()
    best_F = F_local

    tau0 = 0.6
    tau_min = 0.02
    lr = 0.12

    it = 0
    while time.monotonic() < deadline:
        b = np.convolve(a, a, mode="full")
        p = _softmax(b, tau0)
        g = _subgrad(p, a)
        gn = np.linalg.norm(g)
        if not np.isfinite(gn) or gn == 0.0:
            break

        # backtracking
        accepted = False
        for _ in range(8):
            a_new = np.clip(a - lr * g, 0.0, None)
            a_new = _normalize(a_new)
            F_new = _objective_core(a_new)
            if F_new < F_local:
                a = a_new
                F_local = F_new
                if F_new < best_F:
                    best_F = F_new
                    best_a = a.copy()
                lr = min(lr * 1.15, 2.0)
                accepted = True

            lr *= 0.5

            F_local = _objective_core(a)
            lr = 0.05

        it += 1
        # anneal tau over the restart
        tau0 = max(tau_min, tau0 * 0.9995)

    return best_a, best_F
