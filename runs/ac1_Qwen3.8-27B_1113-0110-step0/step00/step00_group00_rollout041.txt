import numpy as np
import time

def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    fsize = 1
    while fsize < 2 * n - 1:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    conv = np.fft.irfft(fa * fa, fsize)[:2 * n - 1]
    return np.maximum(conv, 0)  # clip tiny negatives from FFT roundoff

def simplex_project(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

def compute_grad_at_k(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a. Result is 2*a[k-i] for valid i."""
    n = len(a)
    grad = np.zeros(n)
    # grad[i] = 2 * a[k-i] if 0 <= k-i < n, else 0
    # i ranges from max(0, k-n+1) to min(k, n-1)
    i_lo = max(0, k - n + 1)
    i_hi = min(k, n - 1)
    if i_lo <= i_hi:
        grad[i_lo:i_hi+1] = 2.0 * a[k - i_lo : k - i_hi + 1][::-1]
    return grad

    """Compute the score (scale-invariant, assuming sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def optimize(a_init, budget_s=100, seed=0):
    """Subgradient descent with line search on the simplex."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    else:
        a = a / s  # normalize to sum = 1
    
    best_a = a.copy()
    best_score = objective(a)
    deadline = time.monotonic() + budget_s
    
    lr = 0.1  # initial learning rate
    
    while time.monotonic() < deadline:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        grad = compute_grad_at_k(a, k_star)
        
        # Line search: try a few step sizes
        best_trial = a.copy()
        best_trial_score = best_score
        
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 2.0]:
            alpha = lr * factor
            a_trial = a - alpha * grad
            a_trial = simplex_project(a_trial)
            score = objective(a_trial)
            if score < best_trial_score:
                best_trial_score = score
                best_trial = a_trial.copy()
        
        if best_trial_score < best_score:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            # Decrease learning rate slightly
            lr *= 0.99
        else:
            # No improvement, decrease learning rate and add noise
            lr *= 0.5
            if lr < 1e-6:

                noise = rng.normal(0, 0.01, size=a.shape)
                a_noisy = simplex_project(a + noise)
                score_noisy = objective(a_noisy)
                if score_noisy < best_score:
                    a = a_noisy
                    best_score = score_noisy
                    best_a = a.copy()
                lr = 0.05  # reset
    
    return best_a, best_score

    n = len(a)

    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    result = np.fft.irfft(fa * fa, fsize)[:size]

    np.maximum(result, 0, out=result)
    return result

def _simplex_project(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (float(r + 1))
    return np.maximum(v - theta, 0.0)

    """Score for a sequence normalized to sum=1: 2n * max(conv)."""
    n = len(a)

    """
    Gradient of (a*a)[k] w.r.t. a.
    Result[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

    n = len(a)

    i_hi = min(k, n - 1)

    result = np.zeros(n)
    result[i_lo:i_hi+1] = 2.0 * a[k - i_lo : k - i_hi + 1][::-1]
    return result

def _smoothed_grad(a, K=10, tau=0.01):
    """Compute smoothed subgradient using top-K convolution values."""
    n = len(a)
    conv = _fast_convolve(a)
    
    # Get top K indices
    # conv has 2n-1 elements
    top_k = np.argpartition(conv, -K)[-K:]
    weights = np.exp(conv[top_k] / tau)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx in range(K):
        k = int(top_k[idx])
        grad += weights[idx] * _grad_at_k(a, k)
    
    return grad, conv[top_k].max()

```python
import numpy as np
import time

"""
Minimizes 2n*max(auto_conv)/sum^2 via subgradient descent on the simplex.
Uses FFT convolution, smoothed subgradients (top-K), multiple starts,
and a polishing phase.
"""

def _convolve_full(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    out = np.fft.irfft(fa * fa, fsize)[:size]
    np.maximum(out, 0.0, out=out)
    return out

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > cssv - 1.0)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score assuming sum(a) = 1: 2n * max(conv)."""
    n = len(a)
    c = _convolve_full(a)
    return 2.0 * n * float(c.max())

def _grad_k(a, k):
    """d/da of (a*a)[k]."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    g = np.zeros(n)
    if lo <= hi:
        g[lo:hi+1] = 2.0 * a[k - lo : k - hi + 1][::-1]
    return g

def _smoothed_subgrad(a, K=20, tau=0.005):
    """Smoothed subgradient: weighted avg of top-K active gradients."""
    c = _convolve_full(a)
    if len(c) <= K:
        idxs = np.arange(len(c))
    else:
        idxs = np.argpartition(c, -K)[-K:]
    w = np.exp(c[idxs] / tau)
    w /= w.sum()
    g = np.zeros(len(a))
    for i, idx in enumerate(idxs):
        g += w[i] * _grad_k(a, int(idx))
    return g, c.max()

def _optimize(a0, budget, seed=0):
    """Subgradient descent with adaptive step size on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.ones(len(a)) / len(a)
    
    best = a.copy()
    best_sc = _score(a)
    deadline = time.monotonic() + budget
    
    lr = 0.5 / len(a)
    stagnant = 0
    
    while time.monotonic() < deadline:
        g, _ = _smoothed_subgrad(a)
        # Line search
        improved = False
        for f in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            alpha = lr * f
            trial = _project_simplex(a - alpha * g)
            sc = _score(trial)
            if sc < best_sc:
                best_sc = sc
                best = trial.copy()
                a = trial
                improved = True
                break
        if improved:
            stagnant = 0
            lr *= 0.9995
        else:
            stagnant += 1
            if stagnant > 100:
                # Escape: add noise
                size = len(a)
                noise = rng.standard_normal(size) * (0.1 / np.sqrt(size))
                a = _project_simplex(a + noise)
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best = a.copy()
                stagnant = 0
                lr = 0.5 / len(a)
            else:
                lr *= 0.9
    
    return best, best_sc

def _parametric_init(n, seed=0):
    """Try a few parametric families and return the best one."""
    best = None
    best_sc = np.inf
    x = np.arange(n, dtype=np.float64) / n  # in [0, 1)
    
    # Gaussian: exp(-w*(x-0.5)^2)
    for w in [2, 4, 8, 16, 32]:
        a = np.exp(-w * (x - 0.5)**2)
        a /= a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
    
    # Cosine: 0.5*(1+cos(2*pi*x)) ** p
    for p in [0.5, 1, 2, 4]:
        a = (0.5 * (1 + np.cos(2 * np.pi * x))) ** p
        a = np.maximum(a, 1e-12)
        a /= a.sum()
        sc = _score(a)
        if sc < best_sc:
