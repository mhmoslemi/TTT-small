import numpy as np
import time

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}. O(n log n)."""
    n = len(v)
    if np.sum(v) > s and np.all(v > 0):
        # Just scale
        return v * s / np.sum(v)
    # Full projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_step(a, eta):
    """One subgradient step to minimize max autocorrelation."""
    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()
    # Find active constraints (within tolerance)
    tol = 1e-10
    active = np.where(conv >= M - tol)[0]
    
    # Gradient of max = average of gradients of active constraints
    # grad_j = 2 * a[k-j] for each active k
    # Efficient: for each active k, add 2*a shifted by k
    grad = np.zeros(n)
    for k in active:
        # grad[j] += 2 * a[k-j] if 0 <= k-j < n
        lo = max(0, k - n + 1)  # j such that k-j >= 0 => j <= k, and k-j < n => j > k-n
        hi = min(n-1, k)         # j such that k-j >= 0 => j <= k
        j_idx = np.arange(lo, hi + 1)
        a_idx = k - j_idx
        grad[j_idx] += 2 * a[a_idx]
    
    grad /= len(active)
    
    # Project gradient to tangent space of simplex
    grad -= grad.mean()
    
    # Step
    a_new = a - eta * grad
    
    # Project to simplex
    a_new = project_simplex(a_new, 1.0)
    
    return a_new, M, conv

def optimize_one(a_init, time_budget, seed):
    """Run subgradient optimization from a_init."""
    rng = np.random.default_rng(seed)
    a = a_init / a_init.sum()  # normalize
    n = len(a)
    
    best_a = a.copy()
    conv = fast_autocorr(a)
    best_M = conv.max()
    best_score = 2 * n * best_M  # since sum=1, score = 2n*M
    
    eta0 = 0.01  # initial step size
    eta = eta0
    stall_count = 0
    max_stall = 500
    
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_budget:
        a_new, M, _ = subgradient_step(a, eta)
        it += 1
        
        if M < best_M:
            best_M = M
            best_a = a_new.copy()
            best_score = 2 * n * best_M
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        # Adapt step size
        eta = eta0 / (1 + it * 0.001)
        
        if stall_count > max_stall:
            # Perturb to escape
            a += rng.normal(0, 0.005, n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0
            max_stall = int(max_stall * 1.5)
    
    return best_a, best_score

import numpy as np
import time

def fast_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)

def fast_conv(a):

def proj_simplex(v, s=1.0):

    n = len(v)
