Key insight: For any non-negative sequence a of length n, the maximum of the

import time


def try_uniform(n):
    """Evaluate a uniform sequence of length n."""
    seq = [1.0] * n
    return evaluate_sequence(seq)


def try_perturb(base, rng, n_trials=200):
    """Try random perturbations of base to see if anything beats 2.0."""
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = len(base)
    for _ in range(n_trials):
        # Various perturbation strategies
        strategy = rng.integers(0, 4)
        candidate = base.copy()
        if strategy == 0:
            # Small random noise
            candidate += rng.normal(0, 0.1, size=n)
        elif strategy == 1:
            # Bipolar perturbation
            k = rng.integers(1, n // 2 + 1)
            candidate[:k] += 0.5
            candidate[k:] -= 0.5

            # Taper toward edges
            t = np.linspace(0, 1, n)
            taper = 1.0 - 0.5 * ((2 * t - 1) ** 2)
            candidate = base * (1.0 + 0.3 * rng.random() * (taper - 1.0))
        else:
            # Random block scaling
            width = rng.integers(max(1, n // 50), max(2, n // 5))
            start = rng.integers(0, n - width)
            candidate[start:start + width] *= rng.uniform(0.5, 1.5)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
