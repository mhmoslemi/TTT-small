```python
import time
import numpy as np


"""
Search algorithm summary:
- The problem is scale-invariant.  We work with a length-n array f whose sum is n,
  so f/n is a feasible normalized sequence and the score is 2 * max(conv(f))/n.
- The first optimizer searches a strong structural manifold:
      f[i] + f[n-1-i] = 2,  0 <= f[i] <= 2.
  This is parameterized by theta: f[i] = 1 + sin(theta[i]) for the first half,
  and the second half is its forced anti-complement.
- The max-convolution objective is smoothed with temperature log-sum-exp.
  Gradients are computed exactly from the FFT convolution and the weighted
  anti-convolution of the softmax weights with f.
- Multiple anti-reversal initializations are used, including a cosine Anti-
  complement start, the provided height_sequence_1 if available, and smooth
  random anti starts.  The length is gradually increased (256, 512, 1024).
- A final phase removes the anti-reversal symmetry by optimizing in softmax
  coordinates a = softmax(y), still using FFT-based smooth-max gradients.
"""

def next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def clean_sequence(arr):
    if arr is None:
        return None
    try:
        a = np.asarray(arr, dtype=np.float64)
    except Exception:
        return None
    a = a.reshape(-1)
    if a.size == 0:
        return None
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    return a


def load_height_array():
    try:
        hs = height_sequence_1
    except NameError:
        return None
    return clean_sequence(hs)


def softmax_weights(t, tau):
    """Return (max(t), normalized softmax weights of (t - max)/tau)."""
    t = np.asarray(t, dtype=np.float64)
    m = float(np.max(t))
    if not np.isfinite(m):
        w = np.zeros_like(t, dtype=np.float64)
        if t.size:
            w[int(np.argmax(t))] = 1.0
        return m, w
    shifted = (t - m) / float(tau)
    shifted = np.minimum(shifted, 0.0)
    e = np.exp(shifted)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 1e-300:
        w = np.zeros_like(t, dtype=np.float64)
        idx = int(np.argmax(t))
        w[idx] = 1.0
        return m, w
    return m, e / s


def build_f_from_theta(theta, n):
    m = n // 2
    f_first = 1.0 + np.sin(theta)
    f = np.empty(n, dtype=np.float64)
    f[:m] = f_first
    f[m:] = 2.0 - f_first[::-1]
    return f


def make_cosine_theta(n):
    m = n // 2
    x = np.arange(m, dtype=np.float64) / float(n - 1)
    f_first = 1.0 + np.cos(np.pi * x)
    s = np.clip(f_first - 1.0, -1.0, 1.0)
    return np.arcsin(s)


def make_random_theta(n, rng, amp=0.9):
    m = n // 2
    raw = rng.normal(0.0, 1.0, size=m)
    k = np.ones(9, dtype=np.float64)
    k /= float(k.sum())
    raw = np.convolve(raw, k, mode="same")
    # mild taper to avoid too energetic endpoints
    taper = np.linspace(0.25, 1.0, m)
    raw *= taper
    mx = float(np.max(np.abs(raw)))
    if mx > 1e-12:
        raw *= float(amp) / mx
    return np.clip(raw, -1.55, 1.55)


def project_seq_to_theta(seq, n):
    arr = clean_sequence(seq)
    if arr is None:
        return make_cosine_theta(n)
    if arr.size != n:
        x_old = np.linspace(0.0, 1.0, arr.size, dtype=np.float64)
        x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
        f = np.interp(x_new, x_old, arr)
    else:
        f = arr.astype(np.float64, copy=True)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 1e-12:
        f = np.ones(n, dtype=np.float64)
        s = float(n)
    f = f * (float(n) / s)
    # Anti-complement projection: want x + x_rev = 2.
    x = 0.5 * (f + (2.0 - f[::-1]))
    x = np.clip(x, 0.0, 2.0)
    m = n // 2
    s = np.clip(x[:m] - 1.0, -1.0, 1.0)
    return np.arcsin(s)


def make_theta_inits(n, state, height_arr, rng):
    inits = []
    # Best known anti-reversal sequence from an earlier phase, if good.
    if state is not None and state.get("f") is not None and state.get("score", np.inf) < 1.98:
        inits.append(project_seq_to_theta(state["seq"], n))
    # Provided known sequence, if any.
    if height_arr is not None:
        inits.append(project_seq_to_theta(height_arr, n))
    # Robust analytic start.
    inits.append(make_cosine_theta(n))
    # Smooth random start mixed with the analytic start.
    r = make_random_theta(n, rng, 0.95)
    c = make_cosine_theta(n)
    inits.append(np.clip(0.55 * c + 0.45 * r, -1.55, 1.55))
    # Optionally a pure smoother random start.
    r2 = make_random_theta(n, rng, 1.1)
    inits.append(np.clip(0.35 * c + 0.65 * r2, -1.55, 1.55))
    return inits


def anti_eval_grad(theta, n, tau, L, rev_idx):
    m = n // 2
    f_first = 1.0 + np.sin(theta)
    f = np.empty(n, dtype=np.float64)
    f[:m] = f_first
    f[m:] = 2.0 - f_first[::-1]

    F = np.fft.rfft(f, L)
    h = np.fft.irfft(F * F, L).real[:2 * n - 1]
    t = h / float(n)
    max_t, w = softmax_weights(t, tau)

    # Gradient of weighted sum of t_k with respect to f:
    # g_p = 2/n * sum_q f_q * w_{p+q}.
    W = np.fft.rfft(w[::-1], L)
    g_full = np.fft.irfft(F * W, L).real
    gf = (2.0 / float(n)) * g_full[n - 1:2 * n - 1][::-1]

    cos_theta = np.cos(theta)
    gx = (gf[:m] - gf[rev_idx]) * cos_theta
    return max_t, gx, f


def update_state_with_f(state, f, n, score):
    f = np.clip(np.asarray(f, dtype=np.float64), 0.0, 2.0)
    a = f / float(n)
    sa = float(np.sum(a))
    if not np.isfinite(sa) or sa <= 1e-12:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a / sa
    state["seq"] = [float(x) for x in a]
    state["score"] = float(score)
    state["f"] = f
    state["n"] = int(n)


def run_theta_init(n, theta_init, budget_s, state, rng, tau_list, base_lr, deadline):
    if budget_s <= 1.0:
        return
    local_deadline = min(deadline, time.monotonic() + float(budget_s))
    if time.monotonic() >= local_deadline:
        return

    m = n // 2
    rev_idx = n - 1 - np.arange(m, dtype=np.int64)
    L = next_pow2(3 * n - 2)

    theta = np.clip(np.asarray(theta_init, dtype=np.float64).copy(), -1.55, 1.55)
    if theta.size != m:
        theta = make_cosine_theta(n)

    mom = np.zeros(m, dtype=np.float64)
    var = np.zeros(m, dtype=np.float64)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    step = 0
    lr = float(base_lr)

    local_best_score = np.inf
    local_best_theta = theta.copy()
    local_best_f = build_f_from_theta(theta, n)
    stall = 0

    phase_budget = float(budget_s) / max(1, len(tau_list))

    for tau in tau_list:
        phase_end = min(local_deadline, time.monotonic() + phase_budget)
        while time.monotonic() < phase_end:
            if time.monotonic() >= local_deadline:
                break
            try:
                max_t, gx, f = anti_eval_grad(theta, n, float(tau), L, rev_idx)
            except Exception:
                break
            score = 2.0 * max_t
            if not np.isfinite(score):
                theta = local_best_theta.copy()
                mom.fill(0.0)
                var.fill(0.0)
                continue

            if score < local_best_score - 1e-12:
                local_best_score = score
                local_best_theta = theta.copy()
                local_best_f = f.copy()
                stall = 0
                if score < state.get("score", np.inf):
                    update_state_with_f(state, f, n, score)
                    print(f"theta n={n} tau={tau:.4f} improved score={score:.8f}", flush=True)
            else:
                stall += 1

            if stall > 5000 or (not np.all(np.isfinite(gx))):
                theta = local_best_theta.copy()
                theta = theta + 0.08 * rng.standard_normal(m)
                theta = np.clip(theta, -1.55, 1.55)
                mom.fill(0.0)
                var.fill(0.0)
                lr = max(0.02, lr * 0.75)
                stall = 0
                continue

            gx = np.nan_to_num(gx, nan=0.0, posinf=0.0, neginf=0.0)
            step += 1
            mom = b1 * mom + (1.0 - b1) * gx
            var = b2 * var + (1.0 - b2) * (gx * gx)
            m_hat = mom / (1.0 - b1 ** step)
            v_hat = var / (1.0 - b2 ** step)
            delta = lr * m_hat / (np.sqrt(v_hat) + eps)

            theta_new = theta - delta
            if not np.all(np.isfinite(theta_new)):
                theta = local_best_theta.copy()
                mom.fill(0.0)
                var.fill(0.0)
                continue
            theta = np.clip(theta_new, -1.55, 1.55)

        if time.monotonic() >= local_deadline:
            break

    if np.isfinite(local_best_score) and local_best_score < state.get("score", np.inf):
        update_state_with_f(state, local_best_f, n, local_best_score)


def run_theta_phase(n, budget_s, state, rng, height_arr, tau_list, base_lr, deadline):
    if budget_s <= 10.0 or time.monotonic() >= deadline:
        return
    inits = make_theta_inits(n, state, height_arr, rng)
    if not inits:
        return
    k = len(inits)
    # Give first (normally the best projected start) most time.
    fracs = []
    if k == 1:
        fracs.append(1.0)
    else:
        fracs.append(0.45)
        rest = (1.0 - 0.45) / float(k - 1)
        for _ in range(k - 1):
            fracs.append(rest)
    print(f"theta phase n={n} budget={budget_s:.1f}s inits={k} current_best={state.get('score'):.6f}", flush=True)
    for i, th in enumerate(inits):
        chunk = float(budget_s) * fracs[i]
        if chunk <= 5.0 or time.monotonic() >= deadline:
            continue
        run_theta_init(n, th, chunk, state, rng, tau_list, base_lr, deadline)
    print(f"theta phase n={n} done best={state.get('score'):.6f}", flush=True)


def softmax_eval_grad(a, n, tau, L):
    F = np.fft.rfft(a, L)
    h = np.fft.irfft(F * F, L).real[:2 * n - 1]
    max_h, w = softmax_weights(h, tau)
    W = np.fft.rfft(w[::-1], L)
    g_full = np.fft.irfft(F * W, L).real
    ga = 2.0 * g_full[n - 1:2 * n - 1][::-1]
    return max_h, ga


def update_state_with_a(state, a, score):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones_like(a) / max(1, a.size)
    else:
        a = a / s
    state["seq"] = [float(x) for x in a]
    state["score"] = float(score)
    state["f"] = None
    state["n"] = int(a.size)


def run_softmax_phase(state, budget_s, rng, deadline):

    b2 = 0.999

    step = 0
    lr = 0.22

    local_best_y = y.copy()
    stall = 0

    phase_budget = float(budget_s) / max(1, len(taus))
    print(f"softmax phase n={n} budget={budget_s:.1f}s start_best={local_best_score:.6f}", flush=True)

    for tau in taus:
        phase_end = min(deadline, time.monotonic() + phase_budget)
        while time.monotonic() < phase_end:
            if time.monotonic() >= deadline:
                break
            try:
                max_h, ga = softmax_eval_grad(a, n, float(tau), L)
            except Exception:
                break
            score = 2.0 * float(n) * max_h
            if not np.isfinite(score):
                y = local_best_y.copy()
                a = softmax_from_y(y)
                mom.fill(0.0)
                var.fill(0.0)
                continue

            if score < local_best_score - 1e-12:
                local_best_score = score
                local_best_a = a.copy()
                local_best_y = y.copy()
                stall = 0
                if score < state.get("score", np.inf):
                    update_state_with_a(state, a, score)
                    print(f"softmax n={n} tau={tau:.3e} improved score={score:.8f}", flush=True)
            else:
                stall += 1

            if stall > 6000 or (not np.all(np.isfinite(ga))):
                y = local_best_y.copy()
                a = softmax_from_y(y)
                y = y + 0.05 * rng.standard_normal(n)
                y = y - float(np.mean(y))
                mom.fill(0.0)
                var.fill(0.0)
                lr = max(0.02, lr * 0.75)
                stall = 0
                continue

            ga = np.nan_to_num(ga, nan=0.0, posinf=0.0, neginf=0.0)
            step += 1
            mom = b1 * mom + (1.0 - b1) * ga
            var = b2 * var + (1.0 - b2) * (ga * ga)
            m_hat = mom / (1.0 - b1 ** step)
            v_hat = var / (1.0 - b2 ** step)

            # d/dy softmax(y): a * (grad_a - sum(grad_a*a))
            dot = float(np.dot(ga, a))
            gy = a * (ga - dot)
            delta = lr * m_hat / (np.sqrt(v_hat) + eps)
            # A simple, conservative stabilization: keep the softmax direction,
            # but do not let y wander far.
            step_norm = float(np.linalg.norm(delta))
            if step_norm > 1.2:
                delta *= (1.2 / step_norm)

            y_new = y - delta
            if not np.all(np.isfinite(y_new)):
                y = local_best_y.copy()
                a = softmax_from_y(y)
                mom.fill(0.0)
                var.fill(0.0)
                continue
            y = np.clip(y_new, -40.0, 40.0)
            y = y - float(np.mean(y))
            a = softmax_from_y(y)

        if time.monotonic() >= deadline:
            break

    if local_best_score < state.get("score", np.inf):
        update_state_with_a(state, local_best_a, local_best_score)
    print(f"softmax phase done best={state.get('score'):.6f}", flush=True)


def softmax_from_y(y):
    y = np.asarray(y, dtype=np.float64)
    y = y - float(np.max(y))
    e = np.exp(y)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones_like(y) / float(y.size)
    return e / s


def final_fallback_sequence():
    n = 256
    return [1.0 / float(n) for _ in range(n)]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Return the best sequence found for the given time budget.
    The returned sequence is non-negative and finite.
    """
    start = time.monotonic()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1800.0
    if not np.isfinite(budget_s) or budget_s <= 30.0:
        return final_fallback_sequence()

    deadline = start + max(1.0, budget_s - 6.0)
    rng = np.random.default_rng(int(seed) if seed is not None else 12345)
    height_arr = load_height_array()

    state = {
        "seq": final_fallback_sequence(),
        "score": 2.0,
        "f": None,
        "n": 0,
    }

    available = max(0.0, deadline - time.monotonic())
    if available <= 20.0:
        return state["seq"]

    # Fractional schedule.  Larger n gets most of the work.
    frac = [0.08, 0.30, 0.47, 0.15]
    budgets = [available * x for x in frac]

    # If the budget is tiny, collapse to one anti-theta phase.
    if available < 300.0:
        budgets = [available - 10.0]
        n_phases = [512]
    else:
        n_phases = [256, 512, 1024]

    tau_list = [0.35, 0.18, 0.085, 0.040, 0.018, 0.008]
    base_lr = 0.18

    anti_phase_count = 3 if available >= 300.0 else len(n_phases)
    for pi in range(anti_phase_count):
        n = n_phases[pi]
        bud = budgets[pi]
        if time.monotonic() >= deadline:
            break
        run_theta_phase(n, bud, state, rng, height_arr, tau_list, base_lr, deadline)

    if time.monotonic() < deadline - 15.0:
        # Use the largest length we already used; if no anti phase ran, use 256.
        soft_n = 1024 if available >= 300.0 else 256
        # If the current best has a different length, softmax can still handle it.
        # We force the larger length by resampling if that is reasonable.
        cur_seq = state.get("seq", None)
        if cur_seq is not None and available >= 300.0:
            try:
                tmp = clean_sequence(cur_seq)
                if tmp is not None and tmp.size != soft_n:
                    x_old = np.linspace(0.0, 1.0, tmp.size, dtype=np.float64)
                    x_new = np.linspace(0.0, 1.0, soft_n, dtype=np.float64)
                    a = np.interp(x_new, x_old, tmp)
                    s = float(np.sum(a))
                    if np.isfinite(s) and s > 1e-12:
                        state["seq"] = [float(v) / s for v in np.clip(a, 0.0, 1000.0)]
            except Exception:
                pass
        run_softmax_phase(state, max(0.0, budget_s - 10.0 - (time.monotonic() - start)), rng, deadline)

    seq = state.get("seq", None)
    if seq is None:
        seq = final_fallback_sequence()

    # Final sanitation and validation.
    try:
        arr = np.asarray(seq, dtype=np.float64)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(np.sum(arr))
        if not np.isfinite(s) or s <= 1e-6:
            seq = final_fallback_sequence()
        else:
            arr = arr / s
            seq = [float(v) for v in arr]
        # Confirm with the actual evaluator when available.
        try:
            exact = evaluate_sequence(seq)
            if not np.isfinite(exact):
                seq = final_fallback_sequence()
        except Exception:
            pass
    except Exception:
        seq = final_fallback_sequence()

    return seq
```
