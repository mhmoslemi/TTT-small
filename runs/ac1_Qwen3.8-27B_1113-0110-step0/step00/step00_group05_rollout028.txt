import numpy as np
import time

def compute_conv(x):
    """Compute the auto-convolution of x."""
    return np.convolve(x, x, mode='full')

def compute_score(x):
    """Compute the objective value (without calling evaluate_sequence)."""
    n = len(x)
    S = np.sum(x)
    if S < 0.01:
        return np.inf
    C = np.convolve(x, x, mode='full')
    return 2 * n * np.max(C) / (S**2)

def project_simplex(x, S=1.0):
    """Project x onto {y >= 0, sum(y) = S}."""
    # Water-filling algorithm
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)

def compute_score_fast(x):
    n = len(x)
    S = np.sum(x)
    if S < 0.01:
        return np.inf
    C = np.convolve(x, x, mode='full')
    return 2.0 * n * np.max(C) / (S * S)

def project_simplex(x, s=1.0):
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0]

    return np.maximum(x - theta, 0.0)
