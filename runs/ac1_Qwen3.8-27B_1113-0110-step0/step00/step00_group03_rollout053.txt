"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using FFT-accelerated subgradient descent
with momentum, greedy mass transfer, multiple initializations, and random restarts.
Exploits the insight that asymmetry (one end large, other small) reduces center-lag
autocorrelation while smoothness keeps other lags bounded.
"""
import numpy as np
import time


def fast_self_convolve(a):
    """Compute linear self-convolution using FFT (power of 2 size)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def normalize_nonneg(y, n):
    """Clip to non-negative and normalize to sum=1."""
    y = np.maximum(y, 0.0)
    s = y.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return y / s


def compute_weighted_shift(y, top_idx, w, n):
    """Compute h[m] = sum_k w[i] * y[top_idx[i] - m] with zero padding."""
    h = np.zeros(n)
    for idx in range(len(top_idx)):
        k = int(top_idx[idx])
        wv = w[idx]
        if wv < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            h[lo:hi + 1] += wv * y[k - lo::-1][:hi - lo + 1]
    return h


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Optimizes non-negative sequence to minimize 2n*max(autocorr)/sum^2.
    Uses subgradient descent, greedy transfer, multiple starts, and restarts.
    """
    rng = np.random.default_rng(seed)

    # Get initial sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except (NameError, AttributeError):

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all() or parent.sum() < 0.01:
        parent = np.ones(1024, dtype=np.float64)

    n = len(parent)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    inits = []
    inits.append(best / best.sum())  # Parent normalized

    # Ramp (linearly increasing)
    t = np.linspace(0, 1, n)
    inits.append(normalize_nonneg(1.0 + 3.0 * t, n))

    # Exponential decay
    inits.append(normalize_nonneg(np.exp(-3.0 * t), n))

    # Exponential decay reversed
    inits.append(normalize_nonneg(np.exp(-3.0 * (1.0 - t)), n))

    # Cosine bump (high at ends, low in middle)
    inits.append(normalize_nonneg(0.5 * (1 + np.cos(2 * np.pi * t)), n))

    inits.append(normalize_nonneg(rng.random(n), n))

    inits.append(normalize_nonneg(1.0 + 5.0 * t ** 2, n))

    def run_subgradient(y_init, max_iters, lr0, K, T, mu):
        """Subgradient descent with momentum."""
        y = normalize_nonneg(y_init, n).copy()
        momentum = np.zeros(n)
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            b = fast_self_convolve(y)
            if K >= len(b):
                top_idx = np.arange(len(b))
                top_vals = b.copy()
            else:
                top_idx = np.argpartition(b, -K)[-K:].astype(np.int64)
                top_vals = b[top_idx]
            shifted = top_vals - top_vals.max()
            w = np.exp(shifted / T)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(len(top_vals)) / len(top_vals)
            else:
                w /= w_sum
            h = compute_weighted_shift(y, top_idx, w, n)
            lr = lr0 / (1.0 + 0.002 * it)
            grad = h

            y = y - lr * momentum
            y = normalize_nonneg(y, n)
        return y

        else:

        score = evaluate_candidate(y)
        if np.isfinite(score) and score < best_score:

            best_y = y.copy()
            stuck = 0

            best_y = normalize_nonneg(y_p, n)

    print("Phase 3: Final polish...", flush=True)
