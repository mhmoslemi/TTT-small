    if typ == 0:
        # Gaussian
        a = np.exp(-((x - n/2) / (n/4))**2)
    elif typ == 1:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / (n - 1))
    elif typ == 2:

        a = rng.random(n)

    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a

def compute_gradient(a, k_star):
    """Compute subgradient of c[k_star] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = k_star - np.arange(lo, hi + 1)
        g[lo:hi+1] = 2.0 * a[idx]
    return g

def project_simplex(a):
    """Project a onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def fw_iteration(a, c, k_star, t):
    """One Frank-Wolfe step."""
    n = len(a)
    g = compute_gradient(a, k_star)
    # Vertex: s = e_j where j = argmin(g)
    j = np.argmin(g)
    s = np.zeros(n)
    s[j] = 1.0
    a_new = (1 - t) * a + t * s
    return project_simplex(a_new)

def distributed_fw_iteration(a, c, k_star, t):

    n = len(a)
    g = compute_gradient(a, k_star)
    # Soft vertex: s proportional to (g_max - g)^+
    g_max = g.max()
    s = np.maximum(0, g_max - g)
    s_sum = s.sum()
    if s_sum < 1e-15:
        s = np.ones(n) / n
    else:
        s /= s_sum
    a_new = (1 - t) * a + t * s
    return project_simplex(a_new)

def lp_step(a, M, max_time=10.0):
    """Solve the LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    from scipy.optimize import linprog
    from scipy import sparse
    n = len(a)
    
    # Build constraint matrix A: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # A is (2n-1) x n
    # Use csr format
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # c vector (maximize sum(g)) → minimize -sum(g)
    c_obj = -np.ones(n)
    
    # b vector (A @ g <= M)
    b = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    t0 = time.monotonic()
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    elapsed = time.monotonic() - t0
    
    if res.status != 0:
        return None, elapsed
    
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None, elapsed
    
    return g / g_sum, elapsed

For each column j, the nonzeros are at rows k = j, j+1, ..., j+n-1 (clipped to [0, 2n-2]), with values a[0], a[1], ..., a[n-1].

Wait: A[k, j] = a[k-j] for k-j in [0, n-1], so k in [j, j+n-1]. And k must be in [0, 2n-2].

So for column j:
- Rows: max(0, j) to min(2n-2, j+n-1) = j to min(2n-2, j+n-1)
- Values: a[0], a[1], ..., a[min(2n-2-j, n-1)]

This is a banded structure! Each column has n nonzeros (or fewer at the edges). The matrix is (2n-1) x n with n nonzeros per column.

    # Column j has nonzeros at rows j to j+n-1, values a[0] to a[n-1]
    cols = []
    rows = []
    data = []

        r_end = min(2*n-1, j+n)

        cols.extend([j] * length)
        rows.extend(range(r_start, r_end))
        data.extend(a[:length])

This can give O(1/t^2) convergence for smooth objectives.

Let me also consider a completely different approach: **SDP relaxation / SOCP**.

The problem: minimize M subject to a >= 0, sum a = 1, and for all k, sum_i a[i]*a[k-i] <= M.

The quadratic constraints sum_i a[i]*a[k-i] <= M can be rewritten using the "Schur complement" trick, but that requires the quadratic form to be positive semidefinite. The matrix for c[k*] is a permutation-like matrix, which is NOT PSD. So SDP relaxation doesn't directly apply.

However, I can use a different formulation. Let u = a (unchanged). The constraint is u^T P_k u <= M where P_k is a fixed symmetric matrix (the "autocorrelation matrix" at lag k-n+1). Since P_k is not PSD, this is a non-convex constraint.

One approach: replace P_k with its PSD part. But this weakens the constraint.

I think the practical approach is best: use iterative methods (FW, subgradient) with good practical accelerations.

Let me also think about the "away-step" FW more carefully:
- At each step, compute g = subgradient.
- Find s = e_{argmin(g)} (toward step).
- Find i = argmax(a * g) (away step: the coordinate that's large AND has high gradient).
- Update: a_new = (1-t1-t2)*a + t1*s - t2*a[i]*e_i.
- Project onto simplex.

This is the "Greedy FW" or "Away-step FW" algorithm.

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses long sequences (n~2000-5000) with distributed Frank-Wolfe iterations,
LP-based refinement, and random polish. Target: < 1.5027.
"""
import numpy as np
import time
from scipy.signal import fftconvolve

def make_init(n, typ, rng):
    x = np.arange(n, dtype=np.float64)
    if typ == 0:
        a = np.exp(-2.0 * ((x - 0.5*(n-1)) / (0.35*n))**2)
    elif typ == 1:
        a = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1))
    elif typ == 2:
        a = (1.0 + np.sin(2.0 * np.pi * 3.0 * x / n)) 
        a = np.maximum(a, 0.01)
    elif typ == 3:
        # Smooth random
        a = rng.random(n)
        kn = np.fft.rfftfreq(n)
        mask = np.exp(-10.0 * kn**2 * n * n)
        a = np.fft.irfft(np.fft.rfft(a) * mask, n=n)
        a = np.abs(a) + 0.01
    elif typ == 4:
        # Two-peak
        a = np.exp(-((x - 0.3*n)/(0.1*n))**2) + np.exp(-((x - 0.7*n)/(0.1*n))**2)
    elif typ == 5:
        a = 1.0 / (1.0 + np.exp(-5.0 * (x / n - 0.5)))  # Sigmoid
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    return a

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s

def dist_fw_step(a, k_star, t):
    """Distributed Frank-Wolfe step targeting peak at k_star."""
    n = len(a)
    # Gradient of c[k_star] w.r.t. a
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    g = np.zeros(n)
    if lo <= hi:
        idx = k_star - np.arange(lo, hi + 1)
        g[lo:hi+1] = 2.0 * a[idx]
    
    # Soft vertex: s ~ (g_max - g)^+
    g_max = g.max()
    if g_max < 1e-15:
        s = np.ones(n) / n
    else:
        s = np.maximum(0.0, g_max - g)
        s_sum = s.sum()
        if s_sum < 1e-15:
            s = np.ones(n) / n
        else:
            s /= s_sum
    a_new = (1.0 - t) * a + t * s
    return project_simplex(a_new)

def multi_peak_fw_step(a, c, t, n_peaks=3):
    """FW step using multiple peaks."""
    n = len(a)
    # Find top peaks
    sorted_idx = np.argsort(c)[::-1]
    top_k = sorted_idx[:n_peaks]
    # Weights proportional to peak values
    peak_vals = c[top_k]
    weights = peak_vals / peak_vals.sum()
    
    # Combined soft vertex
    s_accum = np.zeros(n)
    for w, k_star in zip(weights, top_k):
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        g = np.zeros(n)
        if lo <= hi:
            idx = k_star - np.arange(lo, hi + 1)
            g[lo:hi+1] = 2.0 * a[idx]
        g_max = g.max()
        if g_max > 1e-15:
            s = np.maximum(0.0, g_max - g)
            s_sum = s.sum()
            if s_sum > 1e-15:
                s_accum += w * (s / s_sum)
    
    s_sum = s_accum.sum()
    if s_sum > 1e-15:
        s = s_accum / s_sum
    else:
        s = np.ones(n) / n
    
    a_new = (1.0 - t) * a + t * s
    return project_simplex(a_new)

def optimize_seq(a, deadline, verbose=True):
    """Main optimization: distributed FW with periodic re-evaluation."""
    n = len(a)
    a = project_simplex(a.copy())
    
    # Compute initial
    c = fftconvolve(a, a, mode='full')
    best_score = 2.0 * n * c.max()
    best_a = a.copy()
    
    if verbose:
        print(f"    Init: {best_score:.6f}")
    
    patience = 0
    max_patience = 2000
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        if patience > max_patience:
            max_patience = int(max_patience * 1.2)
            if max_patience > 20000:
                break
            # Restart with perturbation
            a = best_a + np.random.default_rng(it).normal(0, 0.02, n)
            a = project_simplex(a)
            patience = 0
            continue
        
        # Choose step size
        t = min(0.3, 2.0 / (it + 2.0))
        
        # Determine number of peaks to use
        n_peaks = 1 if it < 1000 else (2 if it < 5000 else 3)
        
        # FW step
        if n_peaks == 1:
            k_star = c.argmax()
            a = dist_fw_step(a, k_star, t)
        else:
            a = multi_peak_fw_step(a, c, t, n_peaks)
        
        # Re-evaluate periodically
        if it % 10 == 0:
            c = fftconvolve(a, a, mode='full')
            score = 2.0 * n * c.max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
    
    # Final evaluation
    c = fftconvolve(best_a, best_a, mode='full')
    best_score = 2.0 * n * c.max()
    
    if verbose:
        print(f"    Done: {best_score:.6f} (it={it})")
    
    return best_a, best_score

def lp_refine(a, deadline, max_iters=50):
    """LP-based refinement (SOTA approach)."""
    from scipy.optimize import linprog
    from scipy import sparse
    n = len(a)
    
    a = a.copy()
    c = fftconvolve(a, a, mode='full')
    best_score = 2.0 * n * c.max()
    best_a = a.copy()
    
    for lp_it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = c.max()
        
        # Build sparse constraint matrix A (2n-1) x n
        # A[k,j] = a[k-j] for valid (k,j)
        # Build column by column
        nnz = 0
        for j in range(n):
            r_lo = j
            r_hi = min(2*n-1, j+n)
            nnz += r_hi - r_lo
        
        rows = np.empty(nnz, dtype=np.int32)
        cols = np.empty(nnz, dtype=np.int32)
        data = np.empty(nnz, dtype=np.float64)
        pos = 0
        for j in range(n):
            r_lo = j
            r_hi = min(2*n-1, j+n)
            length = r_hi - r_lo
            rows[pos:pos+length] = np.arange(r_lo, r_hi)
            cols[pos:pos+length] = j
            data[pos:pos+length] = a[:length]
            pos += length
        
        A = sparse.csc_matrix((data, rows, cols), shape=(2*n-1, n))
        
        # LP: max sum(g) s.t. A@g <= M, g >= 0
        # → min -sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        t0 = time.monotonic()
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 10.0})
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        g = g / g_sum
        
        # Try several mixing fractions
        for frac in [0.5, 0.3, 0.7, 0.2, 0.4]:
            a_try = (1.0 - frac) * best_a + frac * g
            a_try = project_simplex(a_try)
            c_try = fftconvolve(a_try, a_try, mode='full')
            score_try = 2.0 * n * c_try.max()
            if score_try < best_score:
                best_score = score_try
                best_a = a_try.copy()
                a = a_try.copy()
                c = c_try.copy()
                break
    
    return
