def solve_lp_with_active_constraints(a, M, active_set, solver='highs'):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for k in active_set, g >= 0"""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    n = len(a)
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix for active constraints only
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_set):
        # (a*g)[k] = sum_{j: 0<=k-j<n} a[k-j] * g[j]
        # Coefficient of g[j] is a[k-j] if 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            m = k - j  # 0 <= m < n
            vals.append(a[m])
            rows.append(idx)
            cols.append(j)
    
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(len(active_set), n))
    b = np.full(len(active_set), M)
    
    res = linprog(c_obj, A_ub=A_sparse, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
