from scipy.optimize import linprog
from scipy.sparse import csc_matrix

def compute_convolution(a, b):
    """Compute full convolution using FFT for efficiency."""
    n = len(a) + len(b) - 1
    # Use FFT for O(n log n) convolution
    fa = np.fft.fft(a, 2 * n)
    fb = np.fft.fft(b, 2 * n)
    result = np.fft.ifft(fa * fb)[:n]
    return np.real(result)

def evaluate_locally(sequence):
    """Compute the objective value directly (faster than evaluate_sequence)."""
    a = np.clip(np.asarray(sequence, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    b = compute_convolution(a, a)
    max_b = np.max(b)
    return float(2 * n * max_b / (s ** 2))

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for all k."""
    n = len(f)
    # Build constraint matrix A where A[k, i] = f[k-i] if 0 <= k-i < n
    # A is (2n-1) x n
    # For efficiency, build it as a sparse matrix
    
    # Actually, for the convolution constraint:
    # conv(f, g)[k] = sum_i f[i] * g[k-i] (with out-of-bounds = 0)
    # = sum_i f[k-i] * g[i] (with out-of-bounds = 0)
    # So A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    # Build rows
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            val = f[k - i]
            if val > 0:  # Skip zeros for sparsity
                rows.append(k)
                cols.append(i)
                data.append(val)

    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')

def conv_full(a, b):
    """Full convolution using FFT."""
    n = len(a) + len(b) - 1
    N = next_fast_len(n)
    result = ifft(fft(a, N) * fft(b, N))[:n]
    return result.real

def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_objective(a):
    """Compute the objective value for a sequence (assumes sum(a)=1)."""
    n = len(a)
    c = conv_full(a, a)
    return 2.0 * n * np.max(c)  # Since sum(a)=1, (sum(a))^2 = 1

        if obj < best_obj:

        a = a_new

    """
    Objective and gradient for softmax parameterization.
    u: (n,) array of log-values
    Returns: (objective_value, gradient_wrt_u)
    """
    # Softmax
    u_max = u.max()
    exp_u = np.exp(u - u_max)
    a = exp_u / exp_u.sum()
    
    # Compute smooth max of conv(a,a)
    c = conv_full(a, a)
    val, p = smooth_max_and_grad(c, beta)
    
    # Gradient of val w.r.t. a: dval/da[j] = sum_k p[k] * dc[k]/da[j]
    # dc[k]/da[j] = 2*a[k-j]*1{0<=k-j<n}
    # dval/da[j] = 2*sum_m a[m]*p[j+m] = 2*(a_rev*p)[n-1+j]
    a_rev = a[::-1]
    N = next_fast_len(3*n)
    conv_ap = ifft(fft(a_rev, N) * fft(p, N))[:3*n-1].real
    grad_a = 2.0 * conv_ap[n-1:2*n-1]
    
    # Chain rule: dval/du[j] = sum_i dval/da[i] * da[i]/du[j]
    # da[i]/du[j] = a[j]*(delta_ij - a[i])
    # dval/du[j] = a[j]*grad_a[j] - a[j]*sum_i grad_a[i]*a[i]
    dot = np.dot(grad_a, a)
    grad_u = a * (grad_a - dot)
    
    # Scale invariant: the actual objective is 2n*smooth_max/(sum_a)^2
    # Since sum(a)=1 (from softmax), it's just 2n*smooth_max
    obj = 2.0 * n * val
    
    return obj, grad_u
