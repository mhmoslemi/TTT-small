import numpy as np
from scipy.optimize import linprog
import time

def compute_score(p, n):
    """Compute the score for a probability vector p of length n."""
    conv = np.convolve(p, p)
    return 2 * n * np.max(conv)

def mirror_descent_step(p, eta):
    """One step of mirror descent for minimizing max(p*p)."""
    n = len(p)
    conv = np.convolve(p, p)
    k_star = np.argmax(conv)
    M = conv[k_star]
    
    # Subgradient: g_i = 2 * p[k_star - i] (with p[m] = 0 for m outside [0,n-1])
    # Build the subgradient vector
    g = np.zeros(n)
    # p[k_star - i] for i in [0, n-1], valid when 0 <= k_star - i <= n-1
    # i.e., k_star - (n-1) <= i <= k_star
    i_start = max(0, k_star - (n-1))
    i_end = min(n-1, k_star)
    if i_start <= i_end:
        indices = np.arange(i_start, i_end + 1)
        g[indices] = 2 * p[k_star - indices]
    
    # Mirror descent update: p_i <- p_i * exp(-eta * g_i) / Z
    log_weights = -eta * g
    log_weights -= np.max(log_weights)  # for numerical stability
    weights = np.exp(log_weights)
    p_new = p * weights
    p_new /= p_new.sum()
    return p_new

def solve_lp_for_direction(p, M):
    """Solve the LP to find the best direction g."""
    n = len(p)
    
    # Build constraint matrix A: A[k, j] = p[k-j] if 0 <= k-j <= n-1

        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)

            idx = np.arange(j_start, j_end + 1)
            A[k, idx] = p[k - idx]

The problem is: minimize max_k (p*p)[k] subject to sum p = 1, p >= 0.

By symmetry, the optimal solution should be symmetric: p_i = p_{n-1-i}.

If p is symmetric, then (p*p)[k] = (p*p)[2n-2-k], so we only need to consider k in [0, n-1].

For the case where the max is achieved at a unique k*, the KKT conditions give:
- ∂(p*p)[k*]/∂p_i = λ * (1 if i is not at the boundary, 0 if p_i = 0)
- 2*p[k*-i] = λ for all i with p_i > 0

This means: for all i with p_i > 0, p[k*-i] = λ/2 = constant.
So p is constant on the set {k*-i : p_i > 0}.

If the support of p is {i_1, i_2, ..., i_m}, then p[k*-i_1] = p[k*-i_2] = ... = p[k*-i_m] = λ/2.

This is a strong constraint. It means that the "mirror" of the support (reflected around k*/2) must be a subset of the support, and p must be constant on that mirror set.

    n = len(p)
