Signal optimization via projected gradient descent on log-sum-exp approximation
of the max self-convolution. Multiple initializations with adaptive annealing.
"""

import numpy as np
import time
import sys

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return float(2.0 * n * c.max() / (s * s))

def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def single_gradient_step(a, T, lr):
    """One projected gradient step on log-sum-exp of self-convolution."""
    n = len(a)
    a = normalize_nonneg(a)
    
    # Self-convolution (using FFT for speed)
    c = np.convolve(a, a, mode='full')  # length 2n-1
    # For n=6121, np.convolve might use direct or FFT. Let me force FFT for large n.
    # Actually, numpy handles this internally. For n > ~200, it uses FFT.
    
    # Numerically stable log-sum-exp weights
    c_max = c.max()
    w = T * (c - c_max)
    # w values are <= 0
    exp_w = np.exp(w)
    p = exp_w / exp_w.sum()  # length 2n-1, sums to 1
    
    # Gradient: 2 * correlate(p, a, 'valid')
    # For p of length 2n-1 and a of length n:
    # correlate(p, a, 'valid')[m] = sum_j p[j+m] * a[j] for j=0..n-1
    # Output length = (2n-1) - n + 1 = n
    grad = 2.0 * np.correlate(p, a, mode='valid')[:n]
    
    # Gradient step + projection
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a  # fallback, no change
    a_new = a_new / s_new
    return a_new

    base_lr = 0.03
    no_improve_count = 0

    step_count = 0
    
    while time.monotonic() - start < time_budget:

        progress = min(1.0, elapsed / time_budget)
        
        # Anneal: increase T, decrease lr
        T = 5.0 + 200.0 * progress * progress  # quadratic ramp
        lr = base_lr * (1.0 - 0.7 * progress)
        
        a = single_gradient_step(a, T, lr)
        step_count += 1
        
        if step_count % check_interval == 0:
            score = compute_score(a)
            if score < best_score - 1e-6:
                best_score = score
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
            
            # Restart if stuck

"""
Minimizes 2n*max(self-conv)/(sum)^2 via projected gradient descent on
log-sum-exp approximation, with FFT-accelerated gradients and multi-init search.
"""

import numpy as np
import time

def compute_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return float(2.0 * n * c.max() / (s * s))

def normalize_nonneg(a):
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s

def fft_gradient(p, a):
    """Compute grad = 2 * [sum_j p[j+m]*a[j]]_{m=0}^{n-1} using FFT."""
    n = len(a)
    p_len = len(p)
    pr = p[::-1]
    conv_len = p_len + n - 1
    N = 1
    while N < conv_len + 1:
        N <<= 1
    Pr = np.fft.rfft(pr, N)
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(Pr * A, N)
    start = p_len - 1
    indices = np.arange(start, start - n, -1)
    return 2.0 * conv[indices]

def gradient_step(a, T, lr):
    n = len(a)
    a = normalize_nonneg(a)
    c = np.convolve(a, a, mode='full')
    cmax = c.max()
    w = T * (c - cmax)
    ew = np.exp(w)
    p = ew / ew.sum()
    grad = fft_gradient(p, a)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

def optimize(a_init, budget, seed, T0=5.0, T1=300.0, lr0=0.05, lr1=0.005,
             restart_patience=60, perturb_scale=0.05):

    step = 0

        prog = min(1.0, (time.monotonic() - t0) / budget)
        T = T0 + (T1 - T0) * prog
        lr = lr0 * (lr1 / lr0) ** prog
        a = gradient_step(a, T, lr)
        step += 1
        if step % check_every == 0:
            sc = compute_score(a)
            if sc < best_score - 1e-7:
                best_score = sc
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
            if no_imp >= restart_patience:
                a = best_a + perturb_scale * rng.standard_normal(n)
                a = normalize_nonneg(a)
                no_imp = 0
    
    sc = compute_score(best_a)
    return best_a, sc
