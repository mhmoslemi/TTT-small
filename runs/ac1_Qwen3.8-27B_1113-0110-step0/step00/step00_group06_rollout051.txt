def _compute_gradient(a, alpha):
    """Compute smooth-max subgradient of peak self-convolution.
    Returns (c_max, g) where g[i] = sum_k p[k]*a[k-i]."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    c_shifted = alpha * (c - c_max)
    p = np.exp(c_shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = p / p_sum
    g = np.correlate(p, a, 'full')[n - 1:2 * n - 1]
    return c_max, g
