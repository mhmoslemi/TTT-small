import numpy as np
import time

def _compute_subgrad(f):
    """Compute subgradient of 2n*max(conv(f,f))/sum(f)^2 w.r.t. f.
    
    Returns (grad, score) where grad[i] is d(score)/df[i] (subgradient).
    """
    n = f.size
    S = f.sum()
    if S < 1e-10:
        return np.zeros(n), np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    score = 2.0 * n * M / (S * S)
    
    # Find max indices
    tol = 1e-8 * M
    max_idx = np.where(conv >= M - tol)[0]
    
    # Subgradient of M
    grad_M = np.zeros(n)
    for ki in max_idx:
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad_M[idx] += f[ki - idx] + f[idx]
    grad_M /= len(max_idx)
    
    # Full subgradient of score
    # d(score)/df[i] = 2n/S^2 * grad_M[i] - 4nM/S^3
    grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S * S * S))
    
    return grad, score
