"""
Multiplicative subgradient descent for minimizing max self-convolution.
Uses FFT-based pressure computation, adaptive step sizes, and multiple
initializations with random restarts to escape local minima.
"""

import numpy as np
import time


def compute_pressure(a, active):
    """Compute p[i] = sum_j a[j] * active[i+j] using FFT cross-correlation."""
    n = len(a)
    N = next_power_of_2(n + len(active) - 1)
    fa = np.fft.rfft(a, N)
    fA = np.fft.rfft(active, N)
    p = np.fft.irfft(fA * np.conj(fa), N)
    return np.maximum(p[:n], 0)


def next_power_of_2(x):
    return 1 << (int(np.ceil(np.log2(x))))


def multiplicative_subgradient_step(a, pressure, eta):
    """Apply multiplicative (mirror) descent step."""
    p_max = pressure.max()
    if p_max < 1e-15:
        return a
    pressure_norm = pressure / p_max
    a_new = a * np.exp(-eta * pressure_norm)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def fast_score(a):
    """Compute the objective without calling evaluate_sequence."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def local_search(a_init, deadline, eta_init=0.1, tol_active=0.99, seed=42):
    """Run multiplicative subgradient local search until deadline."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = a_init.copy()

"""
Multiplicative subgradient descent for minimizing max self-convolution ratio.
Uses FFT-based pressure computation, top-K active sets, adaptive step sizes,
multiple initializations, and random multiplicative restarts to escape local minima.
"""

import numpy as np
import time


def _next_pow2(x):
    """Return the next power of 2 >= x."""
    if x <= 1:
        return 1
    return 1 << int(np.ceil(np.log2(x)))


def _compute_pressure(a, active):
    """
    Compute p[i] = sum_j a[j] * active[i+j] via FFT cross-correlation.
    a: length n, active: length 2n-1.
    Returns array of length n.
    """
    n = len(a)
    m = len(active)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fA = np.fft.rfft(active, N)
    p = np.fft.irfft(fA * np.conj(fa), N)
    return np.maximum(p[:n], 0.0)


def _mult_step(a, pressure, eta):
    """Multiplicative (mirror) descent step with normalization."""
    pmax = pressure.max()
    if pmax < 1e-15:
        return a
    pn = pressure / pmax
    a_new = a * np.exp(-eta * pn)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / (sum(a))^2 directly."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s), c

    """
    Multiplicative subgradient local search.
    Returns (best_array, best_score).
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    a = a / a.sum()

    best = a.copy()
    best_score = _score(best)[0]
    eta = eta_init
    no_improve = 0
    max_stall = 400
    topk = max(3, n // 200)  # adaptive top-K
    it = 0

    while time.monotonic() < deadline:
        # Compute convolution and find top-K active positions
        c = np.convolve(a, a)
        M = c.max()

        # Top-K active set
        if M > 0:
            thresh = np.sort(c)[-topk]
            active = (c >= thresh).astype(np.float64)
        else:
            break

        # Compute pressure and take step
        p = _compute_pressure(a, active)
        a_new = _mult_step(a, p, eta)

        # Score the new candidate
        score_new, _ = _score(a_new)
        it += 1

        if np.isfinite(score_new) and score_new < best_score:
            best = a_new.copy()
            best_score = score_new

        else:

            no_improve = 0
