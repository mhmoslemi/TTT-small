def _fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]

def _cross_correlate(p, x):
    """Compute (p ⋆ x)[k] = sum_j p[j+k]*x[j] for k=0..len(x)-1."""
    rev_x = x[::-1]
    conv = _fft_convolve(p, rev_x)
    n = len(x)
    return conv[n-1:2*n-1]

def _soft_max_objective(x, beta):
    """Compute the soft approximation of the objective and gradient."""
    n = len(x)
    c = _fft_convolve(x, x)
    # Softmax for max approximation
    c_shifted = beta * c - beta * np.max(c)
    exp_c = np.exp(c_shifted)
    p = exp_c / np.sum(exp_c)
    # Cross-correlation
    r = _cross_correlate(p, x)
    # Gradient of sum_i p[i] * c[i] with respect to x[k]

    g[valid] -= p[diag[valid]] * x[valid]

    soft_max = (np.max(c) + np.log(np.sum(exp_c))) / beta
    return soft_max, g

import numpy as np
import time

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]

def _cross_correlate(p, x):
    n = len(x)
    rev_x = x[::-1]
    conv = _fft_convolve(p, rev_x)
    return conv[n-1:2*n-1]

def _compute_gradient(x, soft_beta):
    """Compute gradient of soft-max objective w.r.t. x."""
    n = len(x)
    c = _fft_convolve(x, x)
    c_shifted = soft_beta * c - soft_beta * np.max(c)
    exp_c = np.exp(c_shifted)
    p = exp_c / np.sum(exp_c)
    r = _cross_correlate(p, x)
    g = 2.0 * r
    diag = 2 * np.arange(n)
    valid = diag < len(p)
    g[valid] -= p[diag[valid]] * x[valid]
    return g, c
