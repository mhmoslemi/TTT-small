def solve_lp(p, n):
    # Variables: q_0, ..., q_{n-1}, M
    # Minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: sum_i p_i * q_{k-i} - M <= 0
    # Valid i for each k: max(0, k-(n-1)) <= i <= min(n-1, k)
    # q_{k-i} is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
    # So i ranges from max(0, k-n+1) to min(n-1, k)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def solve_best_response(p, n):
    """Solve LP: min M s.t. (p*q)[k] <= M for all k, q >= 0, sum(q) = 1."""
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    row_idx = []
    col_idx = []
    vals = []
    
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            row_idx.append(k)
            col_idx.append(k - i)
            vals.append(p[i])
        row_idx.append(k)
        col_idx.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    if result.success:
        return result.x[:n], result.x[n]
    return p.copy(), np.inf

def compute_objective(p):
    n = len(p)
    conv = np.convolve(p, p)
    return 2.0 * n * np.max(conv)

    and we iterate p -> (1-t)*p + t*q until convergence.
    """

    best_score = np.inf
    best_p = None
    best_n = 0
    
    # Try several lengths
    lengths = [4096, 3072, 5120, 2048, 6144]
    
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        
        p = np.ones(n) / n
        t = 1.0
        prev_obj = compute_objective(p)
        print(f"Starting n={n}, initial obj={prev_obj:.10f}")
        
        stagnant = 0
        max_iter = 300
        
        for it in range(max_iter):
            if time.monotonic() >= deadline - 60:  # leave time for other lengths
                break
            
            t0 = time.monotonic()
            q, lp_val = solve_best_response(p, n)
            lp_time = time.monotonic() - t0
            
            p_new = (1 - t) * p + t * q
            p_new = np.maximum(p_new, 0.0)
            s = p_new.sum()
            if s < 1e-10:
                p_new = np.ones(n) / n
            else:
                p_new /= s
            
            obj = compute_objective(p_new)
            
            if it % 10 == 0:
                print(f"  n={n}, it={it}, lp_val={lp_val:.10f}, obj={obj:.10f}, "

def build_lp_matrices(p, n):
    """Build sparse LP matrices for the best response problem."""
    n_ineq = 2 * n - 1
    n_var = n + 1
    
    row_idx = []
    col_idx = []
    vals = []
    
    for k in range(n_ineq):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            row_idx.append(k)
            col_idx.append(k - i)
            vals.append(p[i])
        row_idx.append(k)
        col_idx.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(n_ineq, n_var))
    b_ub = np.zeros(n_ineq)
    
    A_eq = csr_matrix(np.ones((1, n_var)))
    A_eq[0, n] = 0.0  # M doesn't appear in equality
    b_eq = np.array([1.0])

def solve_best_response(p, n, A_eq, b_eq):
    """Solve LP: min M s.t. (p*q)[k] <= M for all k, q >= 0, sum(q) = 1."""
    n_ineq = 2 * n - 1
    n_var = n + 1
    
    c = np.zeros(n_var)
    c[n] = 1.0

    col_idx = []
