def _fft_convolve(a, L):
    """Compute linear convolution a*a using FFT."""
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:2*len(a)-1]

def _compute_gradient(a, w, L):
    """Compute gradient of soft-max objective.
    g[j] = sum_k w[k] * a[k-j] (cross-correlation of w and a)

    Wf = np.fft.rfft(w, L)
    Af = np.fft.rfft(a, L)
    cross = np.fft.irfft(np.conj(Wf) * Af, L)
    return cross[:len(a)]
