"""
Subgradient optimization with FFT-based gradient computation for minimizing
the peak-to-sum ratio of self-convolution of non-negative sequences.
Uses diverse initializations and Nesterov momentum for acceleration.
"""

import numpy as np
import time

def compute_objective(f):
    """Compute the objective value (without calling evaluate_sequence)."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return float('inf')
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)

def compute_conv_and_gradient(f, active_indices=None):
    """Compute full convolution and subgradient of max convolution.
    
    Returns (conv, max_val, max_idx, grad)
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = conv.max()
    
    # Find active (near-max) indices
    if active_indices is None:
        threshold = max_val * (1.0 - 1e-8)
        active = np.where(conv >= threshold)[0]
    else:
        active = active_indices
    
    # Compute gradient using FFT
    # grad[j] = 2 * sum_{m in active} f[m-j] (when 0 <= m-j <= n-1)
    # This is 2 * cross_correlation(b, f)[j] where b is the active indicator
    b = np.zeros(2 * n - 1)
    b[active] = 1.0
    
    # Cross-correlation via FFT
    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2
    ft_b = np.fft.fft(b, n=fft_size)
    ft_f = np.fft.fft(f, n=fft_size)
    corr = np.fft.ifft(ft_b * np.conj(ft_f)).real
    grad = 2.0 * corr[:n]
    
    return conv, max_val, active, grad

def subgradient_optimize(f0, max_time=60.0, verbose=False):
    """Run subgradient optimization with Nesterov momentum."""
    f0 = np.asarray(f0, dtype=np.float64).copy()
    n = len(f0)
    if n == 0 or f0.sum() < 1e-10:
        return f0
    
    # Normalize
    f = f0 / f0.sum()
    
    # Nesterov momentum
    y = f.copy()  # momentum point
    beta = 0.9  # momentum parameter
    
    # Initial step size
    _, max_val0, _, grad0 = compute_conv_and_gradient(f)
    grad_norm0 = np.linalg.norm(grad0)
    if grad_norm0 == 0:
        return f
    alpha = 2.0 / (1.0 + grad_norm0)  # initial step size
    
    best_f = f.copy()
    best_conv_max = max_val0
    
    start_time = time.monotonic()
    t = 0
    stuck_count = 0
    improvement_threshold = 1e-10
    
    while time.monotonic() - start_time < max_time:
        t += 1
        # Evaluate gradient at momentum point y
        _, _, _, grad_y = compute_conv_and_gradient(y)
        
        # Update: f_new = y - alpha * grad_y
        f_new = y - alpha * grad_y
        f_new = np.maximum(f_new, 0.0)  # project to non-negative
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / f_new_sum
        
        # Compute objective at new point
        conv_new = np.convolve(f_new, f_new)
        max_new = conv_new.max()
        
        if max_new < best_conv_max - improvement_threshold:
            best_f = f_new.copy()
            best_conv_max = max_new
            stuck_count = 0
            if verbose and t % 1000 == 0:
                score = 2.0 * n * max_new  # (sum is 1)
                print(f"  iter {t}: conv_max={max_new:.8f}, obj={score:.6f}")
        else:
            stuck_count += 1
        
        # Nesterov update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = f_new + (beta_new / (beta + 1.0)) * (f_new - f)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-10:
            y = f_new
        else:
            y = y / y_sum
        
        f = f_new
        beta = beta_new
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad_y)
        if grad_norm > 0:
            # Decay step size
            alpha *= 0.999
            # Or use: alpha = 2.0 / (t + grad_norm0)
        
        # Early stop if stuck too long
        if stuck_count > 5000:
            break
    
    return best_f

def block_perturbation(f, rng, num_blocks=5, block_size=50):
    """Apply random block perturbations to f."""
    f = f.copy()
    n = len(f)
    for _ in range(num_blocks):
        start = rng.integers(0, max(1, n - block_size))
        end = min(n, start + block_size)
        f[start:end] *= np.exp(rng.normal(0, 0.1))
    f = np.maximum(f, 0)
    f_sum = f.sum()
    if f_sum > 1e-10:
        f = f / f_sum
    return f

    where p has length 2n-1 and f has length n.
    """
    n = len(f)
    fft_size = 1 << (3 * n - 1).bit_length()
    Ft_p = np.fft.fft(p, n=fft_size)
    Ft_f = np.fft.fft(f, n=fft_size)
    corr = np.fft.ifft(Ft_p * np.conj(Ft_f)).real[:n]
    return corr

def _smooth_grad(f, tau=0.005):
    """Compute smooth gradient of the objective (with sum f = 1 fixed)."""
    n = len(f)
    conv = _fft_convolve(f)
    x = conv / tau
    x -= x.max()
    e = np.exp(x)
    p = e / e.sum()
    corr = _grad_via_fft(p, f)
    return 2.0 * corr

def _evaluate_internal(f):
    """Compute objective value directly."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return float('inf')
    f_norm = f / s
    conv = _fft_convolve(f_norm)
    return 2.0 * n * conv.max()

def _optimize(f0, max_time=20.0, verbose=False):
    """Adam-based optimization."""
    f = np.asarray(f0, dtype=np.float64).copy()
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Learning rate
    lr = 0.01
    
    best_f = f.copy()
    best_obj = _evaluate_internal(f)
    
    tau = 0.01  # smoothing parameter
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        grad = _smooth_grad(f, tau=tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / s_new
        
        # Evaluate
        obj = _evaluate_internal(f_new)
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
            if verbose and t % 1000 == 0:
                print(f"      iter {t}: obj={best_obj:.8f}")
        
        f = f_new
        
        # Anneal tau
        if t % 2000 == 0:
            tau = max(0.001, tau * 0.8)
        
        # Decay learning rate
        if t % 5000 == 0:
            lr *= 0.5

def _fft_convolve(f):
    """Full convolution via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_n = 1 << (size - 1).bit_length()
    ft = np.fft.rfft(f, fft_n)
    return np.fft.irfft(ft * ft, fft_n)[:size]

def _cross_corr_fft(p, f):
    """
    Compute (corr)[j] = sum_m p[m] f[m-j] for j=0..n-1
