```
'''
Search for nonnegative sequence minimizing 2n max conv(a,a)/(sum a)^2.
Uses binomial (central square flat) construction as baseline, then log-domain
smoothed gradient descent in the simplex to escape local finite-n minima,
with FFT-based fast convolution and periodic exact checks.
'''
import numpy as np
import time

def _fast_len(n):
    n = int(n)
    p=1
    while p<n: p<<=1
    # maybe use mixed radix? Use power of two good for rfft.
    return p

def _conv_square_fft(a):
    n=len(a)
    L=2*n-1
    N=_fast_len(L)
    fa=np.fft.rfft(a, N)
    out=np.fft.irfft(fa*fa.conj(), N)[:L]
    out[out<0]=0.0
    return out

def _hankel_qa_fft(q,a):
    # y_i = sum_j q_{i+j} a_j for i=0..n-1, valid i+j<2n-1
    n=len(a)
    a_rev=a[::-1]
    L=n+len(q)-1 # n+2n-1-1=3n-2
    N=_fast_len(L)
    fa=np.fft.rfft(a_rev, N)
    fq=np.fft.rfft(q, N)
    conv=np.fft.irfft(fa*fq, N)
    y=conv[n-1:2*n-1] # length n? n-1 to 2n-2 inclusive
    # Actually indices n-1 through n-1+n-1 =2n-2; slice end 2*n-1
    return y
```

```
    if n <=1: return 1
    p=1
    while p < n: p <<=1
    return p

def _square_fast(a):
    a=np.asarray(a,float)
    n=a.size
    L=2*n-1
    N=_next_pow2(L)
    fa=np.fft.rfft(a,N)
    out=np.fft.irfft(fa*np.conjugate(fa), N)[:L]
    np.maximum(out,0.0,out)
    return out

def _hankel_qa_fast(q,a):
    n=a.size
    a_rev=a[::-1]
    L=n + q.size -1
    N=_next_pow2(L)
    fa=np.fft.rfft(a_rev,N)
    fq=np.fft.rfft(q.N)
    conv=np.fft.irfft(fa*fq,N)
    y=conv[n-1:2*n-1]
    if y.size != n: # fallback direct
       y=np.empty(n)
       for i in range(n): y[i]=np.dot(q[i:i+n], a)
    np.maximum(y,0.0,y)
    return y
```

```python
"""
Search for nonnegative sequences minimizing
    2*n*max(conv(a,a)) / (sum(a))**2 .

The deterministic baseline is the binomial "flat-square" construction
A(x)^2 with very low initial coefficients, whose asymptotic cost is pi/2.
The program then applies a low-dimensional log-domain Adam deformation of
that baseline, using FFT-based smoothed max-convolution gradients.  The
best sequence verified by the official evaluator is returned.
"""

import numpy as np
import time


_FEATURE_CACHE = {}


def _next_pow2(n):
    n = int(max(1, n))
    p = 1
    while p < n:
        p <<= 1
    return p


def _normalize_to_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _make_list(a):
    a = _normalize_to_sum1(a)
    return [float(x) for x in a]


def _binomial_raw(n):
    n = int(n)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    c = np.empty(n, dtype=np.float64)
    c[0] = 1.0
    for k in range(1, n):
        # c_k = C(2k,k)/4^k, recurrence ratio = 1 - 1/(2k)
        c[k] = c[k - 1] * (1.0 - 0.5 / float(k))
    return c


def _official_score(lst):
    try:
        return float(evaluate_sequence(lst))
    except Exception:
        return float("inf")


def _square_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * np.conjugate(fa), N)[:L]
    np.maximum(out, 0.0, out)
    return out


def _fast_score(a):
    a = _normalize_to_sum1(a)
    d = _square_fast(a)
    m = float(d.max())
    if not np.isfinite(m) or m < 0.0:
        return float("inf")
    return 2.0 * float(a.size) * m


def _hankel_qa_fast(q, a):
    """
    y_i = sum_j q_{i+j} a_j for i=0..n-1.
    """
    n = a.size
    a_rev = a[::-1]
    L = n + q.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a_rev, N)
    fq = np.fft.rfft(q, N)
    conv = np.fft.irfft(fa * fq, N)
    y = conv[n - 1:2 * n - 1]
    if y.size != n:
        y = np.empty(n, dtype=np.float64)
        for i in range(n):
            y[i] = float(np.dot(q[i:i + n], a))
    np.maximum(y, 0.0, y)
    return y


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, -35.0)
    x = np.clip(x, -35.0, 35.0)
    m = float(np.max(x))
    b = np.exp(x - m)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(x, dtype=np.float64) / float(x.size)
    return b / s


def _center_unit(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return v
    v -= float(np.mean(v))
    norm = float(np.linalg.norm(v))
    if not np.isfinite(norm) or norm < 1e-12:
        return np.zeros_like(v, dtype=np.float64)
    return v / norm


def _feature_matrix(n, p, seed=0):
    key = (int(n), int(p), int(seed))
    if key in _FEATURE_CACHE:
        return _FEATURE_CACHE[key]

    n = int(n)
    p = int(p)
    if n <= 0 or p <= 0:
        phi = np.empty((0, n), dtype=np.float64)
        _FEATURE_CACHE[key] = phi
        return phi

    x = np.linspace(0.0, 1.0, n)
    feats = []

    feats.append(_center_unit(2.0 * x - 1.0))
    feats.append(_center_unit(x * x - 1.0 / 3.0))
    feats.append(_center_unit((1.0 - x) * (1.0 - x) - 1.0 / 3.0))
    feats.append(_center_unit(np.sqrt(x) - 2.0 / 3.0))
    feats.append(_center_unit(np.sqrt(1.0 - x) - 2.0 / 3.0))

    eps = 1e-6
    feats.append(_center_unit(np.log(x + eps)))
    feats.append(_center_unit(np.log(1.0 - x + eps)))

    m = 1
    while len(feats) < p and m <= 5000:
        c = np.cos(2.0 * np.pi * m * x)
        s = np.sin(2.0 * np.pi * m * x)
        fc = _center_unit(c)
        fs = _center_unit(s)
        if float(np.linalg.norm(fc)) > 1e-12:
            feats.append(fc)
        if len(feats) < p and float(np.linalg.norm(fs)) > 1e-12:
            feats.append(fs)
        m += 1

    rng = np.random.default_rng(seed)
    while len(feats) < p:
        feats.append(_center_unit(rng.standard_normal(n)))

    if len(feats) > p:
        feats = feats[:p]

    phi = np.vstack(feats).astype(np.float64, copy=False)
    _FEATURE_CACHE[key] = phi
    return phi


def _lowdim_search(n, p, max_time, theta_scale, seed, base_array=None,
                   mu_start=0.35, mu_end=0.05, lr=0.10, reg=1e-4):
    """
    Low-dimensional Adam search in a smooth deformation space of a base
    nonnegative sequence normalized to sum 1.
    """
    n = int(n)
    p = int(max(1, p))
    max_time = float(max(0.0, max_time))
    if n <= 1:
        if base_array is None:
            base_array = np.ones(1)
        a = _normalize_to_sum1(base_array)
        return a.copy(), _fast_score(a)

    rng = np.random.default_rng(seed)

    if base_array is None:
        base = _normalize_to_sum1(_binomial_raw(n))
    else:
        base = _normalize_to_sum1(base_array)

    base_x = np.log(np.maximum(base, 1e-300))
    base_x -= float(np.mean(base_x))

    phi = _feature_matrix(n, p, 0)
    p_act = max(1, int(phi.shape[0]))

    ts = float(theta_scale)
    if ts == 0.0:
        theta = np.zeros(p_act, dtype=np.float64)
    else:
        theta = rng.standard_normal(p_act).astype(np.float64) * ts
    theta = np.clip(theta, -6.0, 6.0)

    X = base_x + phi.dot(theta)
    a = _softmax(X)
    best_a = a.copy()
    best_fast = _fast_score(a)
    last_improve = time.monotonic()

    Adam_m = np.zeros(p_act, dtype=np.float64)
    Adam_v = np.zeros(p_act, dtype=np.float64)
    t = 0
    start = time.monotonic()
    end = start + max_time

    while time.monotonic() < end:
        t += 1

        X = base_x + phi.dot(theta)
        a = _softmax(X)

        d = _square_fast(a)
        M = float(d.max())
        if not np.isfinite(M) or M <= 0.0:
            break

        fast = 2.0 * float(n) * M
        if fast < best_fast - 1e-9:
            best_fast = fast
            best_a = a.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_improve > 45.0 and t > 500:
            break

        frac = min(1.0, float(t) / 8000.0)
        mu = mu_start + (mu_end - mu_start) * frac

        z = (d - M) / (M * mu)
        z = np.clip(z, -700.0, 0.0)
        w = np.exp(z)
        sw = float(w.sum())
        if not np.isfinite(sw) or sw <= 0.0:
            break
        w /= sw

        y = _hankel_qa_fast(w, a)
        gs = float(y.sum())
        if not np.isfinite(gs):
            break

        g_a = 2.0 * y / M
        D = a * g_a
        A = float(np.dot(a, D))
        gx = D - a * A

        grad = phi.T.dot(gx)
        grad += reg * theta

        if not np.all(np.isfinite(grad)):
            break

        beta1 = 0.90
        beta2 = 0.999
        Adam_m = beta1 * Adam_m + (1.0 - beta1) * grad
        Adam_v = beta2 * Adam_v + (1.0 - beta2) * (grad * grad)

        m_hat = Adam_m / (1.0 - (beta1 ** t))
        v_hat = Adam_v / (1.0 - (beta2 ** t))

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        theta = np.clip(theta, -6.0, 6.0)

        if not np.all(np.isfinite(theta)):
            break

    return best_a, best_fast


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Return a nonnegative list of heights.  The official evaluator is used
    to certify the best sequence found under the supplied time budget.
    """
    seed = int(seed)
    deadline = time.monotonic() + max(60.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float("inf")

    # Optional known global construction.  If it is short enough and valid,
    # evaluate it and use it as a possible warm start.
    extra_arr = None
    extra_len = 0
    extra_score = float("inf")
    try:
        raw = height_sequence_1
        L = int(len(raw))
        if 16 <= L <= 10000:
            tmp = np.asarray(raw, dtype=np.float64).ravel()
            if tmp.size == L and np.all(np.isfinite(tmp)):
                extra_arr = _normalize_to_sum1(tmp)
                extra_len = L
                extra_score = _official_score(_make_list(extra_arr))
                if np.isfinite(extra_score) and extra_score < best_score:
                    best_score = extra_score
                    best_seq = _make_list(extra_arr)
                print("previous construction length=%d score=%.12f" %
                      (extra_len, extra_score))
    except Exception:
        pass

    # Deterministic flat-square baseline lengths.
    base_lengths = [1024, 2048, 4096, 8192]
    if extra_len and 256 <= extra_len <= 8192 and extra_len not in base_lengths:
        base_lengths.insert(0, int(extra_len))

    for n in base_lengths:
        if deadline - time.monotonic() < 30.0:
            break
        n = int(n)
        a = _normalize_to_sum1(_binomial_raw(n))
        lst = _make_list(a)
        sc = _official_score(lst)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = lst
        print("baseline n=%d score=%.12f" % (n, sc))

    # Search lengths: binomial/deformation attempts on a few lengths.
    search_lengths = []
    for n in [4096, 8192]:
        if n not in search_lengths:
            search_lengths.append(n)
    if extra_len and 512 <= extra_len <= 8192 and extra_len not in search_lengths:
        search_lengths.insert(0, int(extra_len))

    if len(search_lengths) > 3:
        search_lengths = search_lengths[:3]

    scales = [0.0, 0.5, 1.0, 0.3]
    p_features = 96

    for ni, n in enumerate(search_lengths):
        n = int(n)
        if deadline - time.monotonic() < 90.0:
            break

        base_arr = _normalize_to_sum1(_binomial_raw(n))
        base_list = _make_list(base_arr)
        base_score = _official_score(base_list)
        if np.isfinite(base_score) and base_score < best_score:
            best_score = base_score
            best_seq = base_list

        if (extra_arr is not None and extra_len == n and
                np.isfinite(extra_score) and extra_score < base_score):
            base_arr = extra_arr.copy()
            base_score = extra_score
            print("using previous construction as search base n=%d score=%.12f" %
                  (n, base_score))
        elif (best_seq is not None and len(best_seq) == n and
              np.isfinite(best_score) and best_score < base_score):
            base_arr = np.asarray(best_seq, dtype=np.float64).ravel()
            base_score = best_score

        for si, scale in enumerate(scales):
            if deadline - time.monotonic() < 90.0:
                break

            remaining = (len(search_lengths) - ni) * len(scales) - si
            if remaining < 1:
                remaining = 1
            per_time = (deadline - time.monotonic() - 30.0) / float(remaining)
            per_time = float(min(900.0, max(60.0, per_time)))
            if per_time < 30.0:
                break

            seed_val = seed + 1000 * ni + 10 * si
            try:
                a, fast = _lowdim_search(
                    n, p_features, per_time, scale, seed_val,
                    base_array=base_arr
                )
            except Exception:
                a, fast = base_arr, float("inf")

            if a is None or not np.all(np.isfinite(a)):
                a = base_arr

            lst = _make_list(a)
            sc = _official_score(lst)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = lst
            if np.isfinite(sc) and sc < base_score - 1e-12:
                base_arr = np.asarray(a, dtype=np.float64).ravel().copy()
                base_score = sc

            print("lowdim n=%d scale=%.3f fast=%.12f exact=%.12f best=%.12f" %
                  (n, scale, fast, sc, best_score))

    # Final safe return.
    if best_seq is None:
        best_seq = _make_list(_normalize_to_sum1(_binomial_raw(2048)))
    else:
        best_seq = _make_list(best_seq)

    if len(best_seq) == 0:
        best_seq = [1.0]

    print("final length=%d score=%.12f" % (len(best_seq), best_score))
    return best_seq
```
