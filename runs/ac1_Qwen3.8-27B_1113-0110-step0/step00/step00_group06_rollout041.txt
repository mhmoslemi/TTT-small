    # Softmax weights
    max_b = b.max()
    weights = np.exp(beta * (b - max_b))
    weights /= weights.sum()
    # Gradient: weighted sum of a[k-j] for each k
    # grad[j] = sum_k weights[k] * a[k-j] (for valid indices)
    grad = np.zeros_like(a)

import time


def _fft_cross_correlation(x, y):
    """Compute linear cross-correlation (x ⋆ y)[k] = sum_n x[n+k]*y[n] using FFT."""
    nx = len(x)
    ny = len(y)
    N = 1 << (nx + ny - 1).bit_length()
    ft_x = np.fft.fft(x, n=N)
    ft_y = np.fft.fft(y, n=N)
    result = np.fft.ifft(ft_x * np.conj(ft_y), n=N)
    return result.real[:nx + ny - 1]


def _optimize_sequence(a_init, max_time, beta_start=5.0, lr_start=0.5, verbose=True):
    """
    Optimize a non-negative sequence to minimize max of its self-convolution
    (equivalently, minimize the score given the sum constraint).
    
    Uses iterative soft-max gradient descent with multiplicative updates.
    """
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()
    n = len(a)
    lr = lr_start
    beta = beta_start
    best_max_b = np.inf
    best_a = a.copy()
    no_improve = 0
    max_no_improve = 5000
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < max_time:
        # Compute convolution
        b = np.convolve(a, a)
        max_b = b.max()
        
        # Track best
        if max_b < best_max_b * (1 - 1e-8):
            best_max_b = max_b
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Softmax weights (temperature-scaled)
        shifted = beta * (b - max_b)
        # Guard against underflow
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.zeros_like(b)
            weights[np.argmax(b)] = 1.0
        else:
            weights /= wsum
        
        # Gradient via FFT cross-correlation
        # grad[j] = sum_k weights[k] * a[k-j] = (weights ⋆ a)[j]
        grad = _fft_cross_correlation(weights, a)[:n]
        
        # Ensure gradient is non-negative (it should be since both weights and a are non-negative)
        grad = np.maximum(grad, 0)
        
        # Multiplicative update (mirror descent / exponentiated gradient)
        step = lr * grad
        # Clip step to prevent numerical issues
        step = np.minimum(step, 2.0)
        a = a * np.exp(-step)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Adaptive learning rate
        b_new = np.convolve(a, a)
        max_b_new = b_new.max()
        if max_b_new > max_b * 1.05:
            lr *= 0.85
        elif max_b_new < max_b * 0.99:
            lr = min(lr * 1.01, lr_start * 2)
        
        # Gradually increase beta (focus more on peak)
        beta = min(beta * 1.005, 50.0)
        
        # Periodic rebalancing to avoid degeneracy
        if it % 500 == 499:
            # Mix with a random perturbation of the uniform
            noise = 0.05 * np.ones(n)
            a = 0.95 * a + 0.05 * noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
        
        it += 1
        
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  iter {it}, t={elapsed:.1f}s, max_b={max_b:.8f}, score={2*n*max_b:.6f}, lr={lr:.4f}, beta={beta:.2f}")
    
    score = 2 * n * best_max_b
    return best_a, score
