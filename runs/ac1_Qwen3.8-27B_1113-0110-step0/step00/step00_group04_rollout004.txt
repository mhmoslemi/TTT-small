import numpy as np
import time
from scipy import fft as spfft

def fast_full_convolve(a, b):
    """Full convolution using FFT."""
    n = len(a) + len(b) - 1
    fa = spfft.rfft(a, 2 * n)
    fb = spfft.rfft(b, 2 * n)
    return spfft.irfft(fa * fb, 2 * n)[:n]

def fast_self_convolve(a):
    """Self-convolution using FFT (exploits symmetry)."""
    n = len(a)
    size = 2 * n - 1
    fa = spfft.rfft(a, 2 * n)
    c = spfft.irfft(fa * fa, 2 * n)[:size]
    return c

def obj_fast(a):
    """Fast computation of 2n*max(conv(a,a))/sum(a)^2."""
    c = fast_self_convolve(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * len(a) * np.max(c) / (s * s)
