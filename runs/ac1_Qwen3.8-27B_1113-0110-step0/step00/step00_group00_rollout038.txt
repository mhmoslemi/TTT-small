    We use length 1024 (power of 2) for exact floating-point arithmetic.
    """
    # Theoretical minimum is 2.0, achieved by uniform sequences.
    # Use a power-of-2 length for exact float arithmetic.
    n = 1024
    value = 1.0
    
    # Return uniform sequence
    sequence = [value] * n
    
    # Verify the score
    score = evaluate_sequence(sequence)
    print(f"Uniform sequence length={n}, value={value}, score={score}")
    
    return sequence
