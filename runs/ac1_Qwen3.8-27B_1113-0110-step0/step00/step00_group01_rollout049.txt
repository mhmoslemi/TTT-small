        for v in [0, 1, 2, 3, 5, 7, 8, 9]:
            inits_list.append(_make_initial(n, v, rng))
            labels_list.append(f"v{v}")
        # deduplicate: skip if label says from_best and we already have enough
        for label, a_init in zip(labels_list, inits_list):
            if time.monotonic() > phase_deadline:
                break
            print(f"    init={label}", end=" ")
            a_opt, score = _alternating_lp(a_init, 300, phase_deadline, initial_blend=0.8)
            print(f" score={score:.6f}")
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    ** New global best: {best_score:.6f} (n={n})")

    print(f"\nAfter Phase 1: best_score={best_score:.6f}, n={len(best_a) if best_a is not None else 0}")

    # Phase 2: Subgradient for larger n (more degrees of freedom)
    print("\n=== Phase 2: Subgradient (n=2048, 4096, 8192) ===")
    for n in [2048, 4096, 8192]:
        if time.monotonic() > total_deadline - 100:
            break
        phase_deadline = min(total_deadline, time.monotonic() + 300.0)
        print(f"\n  --- Subgradient n={n} ---")
        if best_a is not None:
            a_init = _interpolate_sequence(best_a, n)
        else:
            a_init = _make_initial(n, 0, rng)
        a_opt, score = _subgradient(a_init, 30000, 0.2, 0.9999, 10, phase_deadline)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New global best: {best_score:.6f} (n={n})")

    # Phase 3: LP polish on the best solution (downsample if needed)
    print("\n=== Phase 3: LP Polish ===")
    if best_a is not None and time.monotonic() < total_deadline - 50:
        n_best = len(best_a)
        n_lp = min(n_best, 500)
        polish_deadline = total_deadline - 10
        if n_best <= 500:
            a_ds = best_a.copy()
        else:
            a_ds = _interpolate_sequence(best_a, n_lp)
        print(f"  Polishing with LP at n={n_lp}")
        a_lp, score_lp = _alternating_lp(a_ds, 200, polish_deadline, initial_blend=0.5)
        if n_best != n_lp:
            a_up = _interpolate_sequence(a_lp, n_best)
            score_lp = _score(a_up)
        else:
            a_up = a_lp
        if score_lp < best_score:
            best_score = score_lp
            best_a = a_up.copy()
            print(f"  ** Polished best: {best_score:.6f}")

    # Phase 4: Try a few more LP runs with different blends/restarts if time remains
    if best_a is not None and time.monotonic() < total_deadline - 30:
        print("\n=== Phase 4: Extra LP refinement ===")
        n_best = len(best_a)
        n_lp = min(n_best, 500)
        a_ds = _interpolate_sequence(best_a, n_lp) if n_best != n_lp else best_a.copy()
        for blend_try in [0.3, 0.5, 0.9]:
            if time.monotonic() > total_deadline - 15:
                break
            a_opt, score = _alternating_lp(a_ds, 100, total_deadline - 15, initial_blend=blend_try)
            if n_best != n_lp:
                a_opt = _interpolate_sequence(a_opt, n_best)
                score = _score(a_opt)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  blend={blend_try}: new best {best_score:.6f}")

    # Final verification
    print(f"\n=== Final: score={best_score:.6f}, n={len(best_a) if best_a is not None else 0} ===")
    if best_a is not None:
        result = [float(x) for x in best_a]
        verified = float(evaluate_sequence(result))
        print(f"Verified: {verified:.6f}")
        return result

    # Absolute fallback
    return [1.0] * 100
