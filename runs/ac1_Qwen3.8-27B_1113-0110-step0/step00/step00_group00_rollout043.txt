"""
LP-based iterative optimization for minimizing the auto-convolution peak-to-mean ratio.
Uses scipy.optimize.linprog with HiGHS solver for the LP subproblems.
Combined with local search refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.signal import fftconvolve
import time

def build_conv_matrix(f):
    """Build the convolution constraint matrix A where (A @ g)[k] = conv(f,g)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        length = k_end - k_start + 1
        A[k_start:k_end+1, j] = f[:length]
    return A

def solve_lp(f):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0, where M = max(conv(f,f))."""
    n = len(f)
    conv_f = fftconvolve(f, f)
    M = conv_f.max()
    if M <= 0:
        return None, 0.0
    
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x, M
    return None, M

def local_search(f, deadline, rng, max_iters=10000):
