```python
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def solve_lp_for_g(f, M):
    """Solve max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    # Build sparse constraint matrix A, shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: max sum(g) => min -sum(g)
    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    # Constraints: A @ g <= M
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if not res.success:
        return None
    return res.x
```

```python
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def build_constraint_matrix(f):
    """Build sparse matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    n_rows = 2 * n - 1
    row_list = []
    col_list = []
    val_list = []
    for k in range(n_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            row_list += [k] * len(js)
            col_list += js.tolist()
            val_list += f[k - js].tolist()
    return sp.csr_matrix((val_list, (row_list, col_list)), shape=(n_rows, n))

def lp_iteration(f, M, time_limit=20):
```
