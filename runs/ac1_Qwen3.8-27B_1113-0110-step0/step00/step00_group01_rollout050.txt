Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed mirror descent (multiplicative updates) with adaptive beta,
periodic LP-based best-response jumps, and multiple asymmetric starting points.

import numpy as np

from scipy import sparse

def _normalize_nonneg(x):
    """Clip to non-negative and normalize to sum=1."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-30:
        return x
    return x / s

def _soft_max_weights(conv, beta):
    """Compute soft-max weights p[k] = exp(beta*conv[k]) / Z."""
    cmax = conv.max()
    exps = np.exp(beta * (conv - cmax))
    Z = exps.sum()
    return exps / Z, cmax

def _gradient_via_fft(a, p):
    """Compute gradient: g[j] = 2 * sum_m a[m] * p[m+j] = 2 * (a_rev * p)[j+n-1]."""
    n = len(a)
    conv_full = np.convolve(a[::-1], p, mode='full')
    g = 2.0 * conv_full[n-1 : 2*n-1]
    return g

def _mirror_update(a, g, eta):
    """Multiplicative update: a *= exp(-eta*g), clip, normalize."""
    a_new = a * np.exp(-eta * g)
    return _normalize_nonneg(a_new)

def _lp_response(a, M, max_constraints=300):
    """Solve reduced LP: max sum(b) s.t. conv(a,b)[k] <= M, b>=0."""
    n = len(a)
    conv_aa = np.convolve(a, a)
    # Select top constraints
    threshold = np.percentile(conv_aa, 100.0 * (1.0 - max_constraints / len(conv_aa)))
    active = np.where(conv_aa >= threshold)[0]
    
    rows, cols, vals = [], [], []
    rhs_list = []
    for idx, k in enumerate(active):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            if a[i] > 1e-12:
                rows.append(idx)
                cols.append(k - i)
                vals.append(a[i])
        rhs_list.append(M)
    
    if len(rows) == 0:
        return None
    
    A_ub = sparse.csc_matrix((vals, (rows, cols)), shape=(len(active), n))
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b=np.array(rhs_list, dtype=np.float64),
                      bounds=[(0, np.inf)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0 and res.x is not None:
            return res.x
    except:
        pass
    return None

    return float(2.0 * n * c.max() / (s * s))

"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed mirror descent (multiplicative updates via FFT gradient),
periodic LP-based best-response jumps, and multiple asymmetric starting points.
"""
import numpy as np

from scipy.optimize import linprog
from scipy import sparse

def _normalize_nonneg(x):
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-30:
        return x
    return x / s

def _soft_max_weights(conv, beta):
    cmax = conv.max()
    exps = np.exp(beta * (conv - cmax))
    Z = exps.sum()
    return exps / Z

def _gradient_via_fft(a, p):
    n = len(a)
    conv_full = np.convolve(a[::-1], p, mode='full')
    g = 2.0 * conv_full[n - 1: 2 * n - 1]
    return g

def _mirror_update(a, g, eta):
    log_scale = -eta * g
    log_scale = np.clip(log_scale, -40.0, 40.0)
    a_new = a * np.exp(log_scale)
    return _normalize_nonneg(a_new)

def _lp_response(a, M, max_constraints=200, time_limit=45):
    n = len(a)
    conv_aa = np.convolve(a, a)
    if max_constraints < len(conv_aa):
        thresh = np.percentile(conv_aa, 100.0 * (1.0 - max_constraints / len(conv_aa)))
        active = np.where(conv_aa >= thresh)[0]
    else:
        active = np.arange(len(conv_aa))

    rows, cols, vals = [], [], []
    rhs_list = []
    for idx, k in enumerate(active):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            if a[i] > 1e-14:
                rows.append(idx)
                cols.append(k - i)
                vals.append(a[i])
        rhs_list.append(M)

    if len(rows) == 0:
        return None

    A_ub = sparse.csc_matrix((vals, (rows, cols)), shape=(len(active), n))
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b=np.array(rhs_list, dtype=np.float64),

                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0 and res.x is not None and np.all(np.isfinite(res.x)):

    except Exception:
        pass
    return None

    return cur, cur_score, best_score, best
