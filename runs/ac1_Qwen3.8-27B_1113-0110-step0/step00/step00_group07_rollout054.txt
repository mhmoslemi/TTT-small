def soft_max_grad(a, beta):
    """Compute the soft-max value and gradient of convolve(a,a)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    shifted = np.maximum(shifted, -500)  # for numerical stability
    exps = np.exp(shifted)
    total = exps.sum()
    p = exps / total
    # Gradient: 2 * correlate(p, a, 'valid') - diagonal correction
    grad = 2.0 * np.correlate(p, a, mode='valid')
    # Diagonal correction: subtract a[j]*p[2j] for valid 2j
    diag_idx = np.arange(n) * 2
    valid_diag = diag_idx < (2*n - 1)
    grad[valid_diag] -= a[valid_diag] * p[diag_idx[valid_diag]]
    return max_c, grad

    total = exps.sum()

    grad = 2.0 * np.correlate(p, a, mode='valid')

    diag = 2 * np.arange(n)

    grad[valid] -= a[valid] * p[diag[valid]]
