        S = a.sum()

        threshold = M - 0.02 * M  # peaks within 2% of max
        peak_indices = np.where(c >= threshold)[0]

            i_end = min(n, k + 1)

        grad = grad - 2 * M / S

        a = a / S_new

                best_a = a.copy()

    # Phase 3: Local perturbations
    a = best_a.copy()
    for it in range(10000):
        if time.monotonic() > deadline:
            break

        width = rng.integers(5, 50)
        start = rng.integers(0, n - width)
        a_pert = a.copy()
        a_pert[start:start+width] *= np.exp(rng.normal(0, 0.2, width))

                best_a = a_pert.copy()
                a = a_pert.copy()
                print(f"Perturb iter {it}: score={best_score:.6f}")
    
    print(f"Final: score={best_score:.6f}")
    return [float(x) for x in best_a]

import time

    Minimizes 2n*max(conv(a,a))/(sum(a))^2 using:

    2. Projected subgradient descent with multi-peak handling
    3. Periodic LP-based jumps (inspired by SOTA)
    4. Random perturbations for diversity

        S = n
