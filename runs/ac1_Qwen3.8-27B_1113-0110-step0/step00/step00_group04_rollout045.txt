def optimize_sequence(a, max_iters=100, lr=0.5):
    """LP-based optimization of the sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    
    for it in range(max_iters):
        # Normalize
        a_sum = a.sum()
        if a_sum < 1e-10:
            break
        a_norm = a / a_sum  # sum = 1
        
        # Compute current max
        conv = np.convolve(a_norm, a_norm)
        M = conv.max()
        
        # Build constraint matrix
        A = build_constraint_matrix(n, a_norm)
        
        # Solve LP: max sum(b) s.t. A@b <= M, b >= 0
        c_obj = -np.ones(n)  # minimize -sum(b)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), 
                      bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0:
            break
        
        b = res.x
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        
        # Rescale b to match a's sum
        b = b * (a_sum / b_sum)
        
        # Update
        a = (1 - lr) * a + lr * b
    
    return a
