    """Build the matrix T such that T @ g = convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n))

def project_simplex(v, S):
    """Euclidean projection onto {x : x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho: largest j such that u[j-1] > (cssv[j-1] - S) / j
    cond = u * np.arange(1, n+1) > (cssv - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.max(np.where(cond))
    tau = (cssv[rho-1] - S) / rho
    return np.maximum(v - tau, 0)
