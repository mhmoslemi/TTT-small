```python
import numpy as np
import time

def soft_max_conv(a, p):
    """Compute (sum conv^p)^(1/p) and its gradient w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    M_p = np.sum(conv**p) ** (1.0/p)
    
    # Gradient of M_p w.r.t. a[i]
    # = M_p^(1-p) * 2 * sum_{j=0}^{n-1} conv[i+j]^(p-1) * a[j]
    w = conv**(p-1)  # length 2n-1
    # G[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # Using convolution: G[i]/2 = (w * a_flipped)[i + n - 1]
    a_flipped = a[::-1]
    corr = np.convolve(w, a_flipped, mode='full')
    G = 2.0 * corr[n-1:n-1+n]
    
    dM_p = M_p**(1-p) * G
    return M_p, dM_p

def gradient_step(a, p, lr):
    """One gradient descent step."""
    n = len(a)
    conv = np.convolve(a, a)
    S = np.sum(a)
    
    M_p, dM_p = soft_max_conv(a, p)
    
    # dJ/da[i] = 2n/S² * dM_p[i] - 4n*M_p/S³
    grad = 2.0*n/S**2 * dM_p - 4.0*n*M_p/S**3
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    
    return a_new, np.max(conv)/S**2

def gradient_descent(a_init, time_limit=200, p=8, lr_init=0.1, seed=42):
    """Run gradient descent to minimize J."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()  # normalize
    
    start = time.monotonic()
    deadline = start + time_limit
    
    lr = lr_init
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    
    iteration = 0
    while time.monotonic() < deadline:
        a_new, _ = gradient_step(a, p, lr)
        
        # Check if we made progress
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Adaptive learning rate
        if not np.isfinite(score) or score > best_score * 1.01:
            lr *= 0.5
        elif score < best_score:
            lr = min(lr * 1.05, 1.0)
        
        a = a_new
        iteration += 1
        
        if iteration % 1000 == 0:
            print(f"  iter {iteration}, score {best_score:.6f}, lr {lr:.6f}")
    
    return best_a
```
