"""
LP-based optimization for low-peak autocorrelation sequences.
Uses the approach: at each step, solve an LP to find the best direction
that doesn't increase the peak convolution, then blend in that direction.
Combined with gradient-based polishing and local search.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix

try:
    import cvxpy as cp
    HAS_CVXPY = True
except:
    HAS_CVXPY = False

try:
    from scipy.optimize import minimize as _minimize
    HAS_SCIPY_OPT = True
except:
    HAS_SCIPY_OPT = False


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _score(a, n=None):
    """Compute 2*n*max(conv(a,a)) for normalized a."""
    if n is None:
        n = len(a)
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2*n-1]
    return 2 * n * np.max(c)


def _convolve(a, b, n=None):
    """Linear convolution using FFT."""
    if n is None:
        n = len(a)
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    return np.fft.irfft(A * B, n=L)[:2*n-1]


def _build_conv_matrix(a):
    """Build the sparse matrix G such that G @ g = convolve(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        if a[i] == 0:
            continue
        r_start = i
        r_end = min(i + n - 1, 2 * n - 2)
        count = r_end - r_start + 1
        cols.extend([i] * count)
        rows.extend(range(r_start, r_end + 1))
        vals.extend([a[i]] * count)
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def _lp_step(a, M, solver=None):
    """
    Solve the LP: max sum(g) s.t. g >= 0, convolve(a, g) <= M.
    Returns the optimal g (normalized to same sum as a).
    """
    n = len(a)
    if not HAS_CVXPY:
        return None
    
    G = _build_conv_matrix(a)
    g_var = cp.Variable(n)
    
    constraints = [g_var >= 0, G @ g_var <= M]
    obj = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(obj, constraints)
    
    if solver is None:
        solver = cp.CLARABEL
    
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                return None
    
    if g_var.value is None:
        return None
    
    g = np.asarray(g_var.value, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0)
    g = np.maximum(g, 0)
    
    if g.sum() <= 0:
        return None
    
    return g


def _line_search(a, g, M, max_iter=50):
    """
    Find the largest t in [0,1] such that score((1-t)*a + t*g) <= M * (1+tol).
    """
    n = len(a)
    tol = 1e-10
    lo, hi = 0.0, 1.0
    
    # First check if t=1 works
    h = g / g.sum() * a.sum()  # normalize g to same sum as a
    # Actually, let's just work with (1-t)*a + t*g directly
    # and normalize at the end
    
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        h = (1 - mid) * a + mid * g
        h = np.maximum(h, 0)
        s = h.sum()
        if s <= 0:
            hi = mid
            continue
        h = h / s  # normalize to sum 1
        sc = _score(h, n)
        if sc <= M * (1 + tol):
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-10:
            break
    
    return lo


def _subgradient_step(a, n, lr):
    """
    One subgradient step for the smooth max objective.
    Finds the peak lag and takes a step to reduce it.
    """
    c = _convolve(a, a, n)
    k_star = int(np.argmax(c))
    M = c[k_star]
    
    # Gradient of c[k_star] = sum_i a[i]*a[k_star-i] with respect to a
    # dc/da[j] = a[k_star-j] + a[j-k_star] (with a[x]=0 for x outside [0,n-1])
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = k_star - j
        if 0 <= i1 < n:
            val += a[i1]
        i2 = j - k_star
        if 0 <= i2 < n:
            val += a[i2]
        grad[j] = val
    
    # Projected gradient step onto simplex {a >= 0, sum a = 1}
    a_new = a - lr * grad
    # Project to simplex
    a_new = _project_simplex(a_new)
    return a_new, k_star


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.where(np.isfinite(v), v, 0)
    n = v.size
    if n == 0:
        return v
    
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    
    r = np.nonzero(cond)[0][-1]
    tau = (css[r] - 1.0) / (r + 1)
    out = np.maximum(v - tau, 0)
    s = out.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return out / s
