```python
import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _cross_correlate(a, p, n):
    """Compute cross-correlation: (a ⊛ p)[j] = sum_m a[m] * p[m+j] for j=0..n-1.
    a has length n, p has length 2n-1."""
    size = n + len(p) - 1
    fft_size = _next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    # Circular cross-correlation: IFFT(conj(A) * P)
    cc = np.fft.irfft(np.conj(A) * P, fft_size)
    # We want cc[j] for j=0..n-1, which corresponds to sum_m a[m] * p[m+j]
    # In circular correlation with padding: cc[j] = sum_m a[m] * p[(m+j) mod fft_size]
    # With sufficient padding, this equals the linear cross-correlation.
    return cc[:n]

def _smooth_grad(a, tau):
    """Compute gradient of smooth approximation f_tau(a) = (1/tau)*log(sum(exp(tau*c)))
    where c = conv(a,a). Returns gradient vector of length n."""
    n = len(a)
    c = _fft_convolve(a)
    # Softmax for numerical stability
    c_max = np.max(c)
    exp_tc = np.exp(tau * (c - c_max))
    p = exp_tc / np.sum(exp_tc)
    # Gradient: 2 * cross_correlate(a, p)
    grad = 2.0 * _cross_correlate(a, p, n)
    return grad, c, p

def _optimize_smooth(a0, time_budget, rng, tau_init=50.0, tau_min=1.0):
    """Optimize using mirror descent on the smooth approximation."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    best_score = 2.0 * n * float(np.max(_fft_convolve(a)))
    best_a = a.copy()
    
    eta = 0.1
    tau = tau_init
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        grad, c, p = _smooth_grad(a, tau)
        
        score = 2.0 * n * float(np.max(c))
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 10.0:
                print(f"    [smooth] it={it}, score={score:.6f}, tau={tau:.1f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        if no_improve > 500:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            eta = 0.1
            continue
        
        # Project gradient onto the tangent space of the simplex
        grad_centered = grad - np.mean(grad)
        
        # Multiplicative update
        a = a * np.exp(-eta * grad_centered)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Adaptive step size
        if no_improve > 300:
            eta *= 0.97
        elif no_improve < 50:
            eta = min(eta * 1.01, 0.5)
        
        # Decrease tau over time (make the approximation sharper)
        # Actually, we want tau to increase (sharper max approximation)
        # But for optimization, moderate tau is better
        # Let's keep tau fixed or slowly increase
        tau = min(tau * 1.001, 200.0)
        
        it += 1
    
    return best_a, best_score
```

```python
    c = _fft_convolve(a)  # length 2n-1
    # Softmax
```

```python
import math


"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses smooth gradient (log-sum-exp) with FFT-accelerated cross-correlation,
mirror descent with adaptive step size, KKT projection to enforce optimality
structure, and diverse parametric initializations including multi-harmonic
Fourier modulations.

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _compute_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum(exp(tau*c))) where c=conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a)
    c_max = np.max(c)
    exp_tc = np.exp(tau * (c - c_max))
    total = np.sum(exp_tc)
    if total < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = exp_tc / total
    # Cross-correlation: grad[j] = 2 * sum_m a[m] * p[j+m]
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    cc = np.fft.irfft(np.conj(A) * P, N)
    grad = 2.0 * cc[:n]
    return grad, c, p

    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)

    eta = 0.08
    tau = tau_init
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    kkt_counter = 0

    while time.monotonic() < deadline:
        grad, c, p = _compute_grad(a, tau)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 8.0:
                print(f"    [smooth] it={it}, score={score:.6f}, tau={tau:.1f}, eta={eta:.4f}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 600:
            # Restart with perturbation
            a = best_a * np.exp(rng.normal(0, 0.06, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            eta = 0.08
            tau = max(tau * 0.5, tau_init * 0.5)
            it += 1
            continue

        # Center the gradient (tangent to simplex)
        grad_centered = grad - np.mean(grad)
        g_norm = np.linalg.norm(grad_centered)
        if g_norm < 1e-15:
            break

        # Multiplicative update
        scale = eta * np.max(np.abs(grad_centered))
        a = a * np.exp(-scale * grad_centered / max(np.max(np.abs(grad_centered)), 1e-15))
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a = a / s

        # KKT projection every 50 iterations
        kkt_counter += 1
        if kkt_counter % 50 == 0:
            kkt_alpha = min(0.5, 0.1 + 0.01 * (kkt_counter // 500))
            a = _kkt_project(a, k_max, kkt_alpha)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

        # Adaptive step size
        if no_improve > 400:
            eta *= 0.96
        elif no_improve > 200:
            eta *= 0.98
        elif no_improve < 50:
            eta = min(eta * 1.005, 0.3)

        # Gradually increase tau (sharper max approximation)
        tau = min(tau * 1.0005, tau_max)

        # Diversity injection
        if it % 300 == 150 and rng.random() < 0.15:
            block = max(1, n // 40)
            start = int(rng.integers(0, n - block))
            a[start:start + block] *= np.exp(rng.normal(0, 0.12))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

        it += 1

    return best_a, best_score


def _optimize_subgrad_v2(a0, time_budget, rng):
    """Enhanced subgradient method with multiple peak tracking."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()

    c0 = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c0))
    best_a = a.copy()

    eta = 0.12
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.7
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 8.0:
                print(f"    [subgrad2] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 800:
            a = best_a * np.exp(rng.normal(0, 0.07, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            eta = 0.12
            momentum = np.zeros(n)
            it += 1
            continue

        # Subgradient at k_max
        g = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            g[lo:hi + 1] = 2.0 * a[k_max - j_range]

        # Also add contribution from near-max peaks (temporal smoothing)
        c_sorted_idx = np.argsort(c)[::-1]
        for idx in c_sorted_idx[1:4]:
            if c[idx] > 0.95 * c[k_max]:
                lo2 = max(0, int(idx) - n + 1)
                hi2 = min(n - 1, int(idx))
                if lo2 <= hi2:
                    j_range2 = np.arange(lo2, hi2 + 1)
                    g[lo2:hi2 + 1] += 0.3 * a[int(idx) - j_range2]

        g_max = np.max(g)
        if g_max < 1e-15:
            break
        g_norm = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g_norm

        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a = a / s

        if no_improve > 500:
            eta *= 0.95
        elif no_improve < 30:
            eta = min(eta * 1.003, 0.35)

        it += 1

    return best_a, best_score


def _optimize_fista(a0, time_budget, rng, tau=50.0):
    """FISTA (accelerated gradient) on the smooth approximation."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    y = a.copy()
    t_k = 1.0

    c0 = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c0))
    best_a = a.copy()

    L = 2.0 * tau * n  # Lipschitz constant estimate (upper bound)
    step = 1.0 / L
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        grad, c, p = _compute_grad(y, tau)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max] / (np.sum(y) ** 2)

        if score < best_score - 1e-12:
            best_score = score
            normed_y = y / y.sum()
            best_a = normed_y.copy()
            no_improve = 0
            if time.monotonic() - last_print > 8.0:
                print(f"    [fista] it={it}, score={score:.6f}, tau={tau:.1f}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 500:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            y = a.copy()
            t_k = 1.0
            no_improve = 0
            it += 1
            continue

        # Gradient step with projection onto {a >= 0, sum(a) = 1}
        # Use multiplicative projection (Bregman projection for KL)
        a_new = a * np.exp(-step * (grad - np.mean(grad)))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Y update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + (t_k - 1.0) / t_new * (a_new - a)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        a = a_new
        t_k = t_new

        if no_improve > 300:
            tau = min(tau * 1.01, 200.0)
            L = 2.0 * tau * n
            step = 1.0 / L
        elif no_improve < 50 and tau > 10:
            tau = max(tau * 0.995, 10.0)
            L = 2.0 * tau * n
            step = 1.0 / L

        it += 1

    return best_a, best_score


def _make_initializations(seed, n_list):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    candidates = []

    # Load previous best if available
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            candidates.append(("prev_best", hs.astype(np.float64).copy()))
    except (NameError, Exception):
        pass

    for n in n_list:
        x = np.arange(n, dtype=np.float64)

        # Single harmonic sinusoidal (various phases and amplitudes)
        for amp in [0.3, 0.4, 0.5]:
            for phase in [0.0, np.pi / 4, np.pi / 3, np.pi / 2]:
                a = 1.0 + amp * np.sin(2.0 * np.pi * x / n + phase)
                a = np.maximum(a, 0.01)
                candidates.append((f"sin1_{n}_{amp}_{phase:.2f}", a))

        # Two-harmonic Fourier
        for a1, a2, p2 in [(0.3, 0.2, 0.0), (0.3, 0.2, np.pi / 2),
                            (0.4, 0.15, 0.0), (0.2, 0.3, np.pi / 3)]:
            a = 1.0 + a1 * np.cos(2.0 * np.pi * x / n) + a2 * np.cos(4.0 * np.pi * x / n + p2)
            a = np.maximum(a, 0.01)
            candidates.append((f"sin2_{n}_{a1}_{a2}", a))

        # U-shape (low in middle, high at edges)
        for c in [0.3, 0.5, 0.7]:
            a = 1.0 - c * np.cos(2.0 * np.pi * x / n)
            a = np.maximum(a, 0.01)
            candidates.append((f"ushape_{n}_{c}", a))

        # Inverted U (high in middle)
        for c in [0.3, 0.5]:
            a = 1.0 + c * np.cos(2.0 * np.pi * x / n)
            a = np.maximum(a, 0.01)
            candidates.append((f"hill_{n}_{c}", a))

        # Cosine-squared bump
        a = np.cos(np.pi * (x + 0.5) / (2.0 * n)) ** 2
        a = a + 0.1
        candidates.append((f"cos2_{n}", a))

        # Exponential decay
        a = np.exp(-1.0 * x / n) + 0.2
        candidates.append((f"decay_{n}", a))

        # Two-lobe (mass at 1/4 and 3/4)
        for sigma_frac in [0.1, 0.15]:
            sigma = n * sigma_frac
            a = np.exp(-0.5 * ((x - n * 0.25) / sigma) ** 2) + np.exp(-0.5 * ((x - n * 0.75) / sigma) ** 2)
            a = a + 0.05
            candidates.append((f"twolobe_{n}_{sigma_frac}", a))

        # Three-lobe
        a = (np.exp(-0.5 * ((x - n * 0.15) / (n * 0.1)) ** 2) +
             np.exp(-0.5 * ((x - n * 0.5) / (n * 0.12)) ** 2) +
             np.exp(-0.5 * ((x - n * 0.85) / (n * 0.1)) ** 2))
        a = a + 0.05
        candidates.append((f"threelobe_{n}", a))

        # Step-like (high first half, low second half) with smooth transition
        mid = n // 2
        width = n // 20
        a = np.zeros(n)
        a[:mid] = 1.5
        a[mid:] = 0.5
        # Smooth the transition
        for i in range(max(0, mid - width), min(n, mid + width)):
            t = (i - (mid - width)) / (2.0 * width)
            a[i] = 1.5 * (1 - t) + 0.5 * t
        candidates.append((f"step_{n}", a))

        # Random (for diversity)
        a = rng.random(n) + 0.1
        candidates.append((f"rand_{n}", a))

    return candidates


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    
    Strategy: Smooth gradient (log-sum-exp) with FFT cross-correlation,
    mirror descent + FISTA, KKT projection, diverse parametric
    initializations (Fourier modulations, U-shapes, multi-lobe),
    and multi-scale tau adaptation.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20.0  # safety margin

    best_overall = None
    best_overall_score = np.inf

    n_list = [1500, 2000, 2500, 3000]
    print(f"Generating initializations for n in {n_list}...")
    candidates = _make_initializations(seed, n_list)
    print(f"Total candidates: {len(candidates)}")

    # Quick initial scoring
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        c0 = _fft_convolve(a0)
        init_score = 2.0 * len(a0) * float(np.max(c0))
        scored.append((init_score, name, a0))

    scored.sort(key=lambda x: x[0])
    print("Top 12 initial scores:")
    for i in range(min(12, len(scored))):
        sc, name, arr = scored[i]
        print(f"  {i+1}. {name}: {sc:.6f} (n={len(arr)})")

    # Allocate time: top 10 candidates get time, geometric allocation
    n_top = min(len(scored), 12)
    remaining_time = budget_s - 30.0
    weights = []
    for i in range(n_top):
        w = max(1.0, (n_top + 1 - i) ** 1.5)
        weights.append(w)
    total_w = sum(weights)
    times = [remaining_time * w / total_w for w in weights]

    # Optimize top candidates
    for i in range(n_top):
        if time.monotonic() > deadline:
            break

        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(10.0, deadline - time.monotonic()))
        n = len(a0)

        print(f"\nOptimizing '{name}' (n={n}, init={init_score:.6f}, time={t_alloc:.0f}s)")

        best_local = a0.copy()
        best_local_score = init_score

        # Method 1: Smooth gradient (tau adaptive)
        t1 = t_alloc * 0.4
        if t1 > 5:
            a1, s1 = _optimize_smooth(a0, t1, rng, tau_init=20.0, tau_max=200.0)
            eval1 = _score(a1)
            print(f"  smooth(20): {eval1:.6f}")
            if eval1 < best_local_score:
                best_local_score = eval1
                best_local = a1.copy()

        # Method 2: Smooth gradient (higher tau)
        t2 = t_alloc * 0.25
        if t2 > 5:
            a2, s2 = _optimize_smooth(best_local, t2, rng, tau_init=80.0, tau_max=500.0)
            eval2 = _score(a2)
            print(f"  smooth(80): {eval2:.6f}")
            if eval2 < best_local_score:
                best_local_score = eval2
                best_local = a2.copy()

        # Method 3: Subgradient v2 (multi-peak)
        t3 = t_alloc * 0.2
        if t3 > 5:
            a3, s3 = _optimize_subgrad_v2(best_local, t3, rng)
            eval3 = _score(a3)
            print(f"  subgrad2: {eval3:.6f}")
            if eval3 < best_local_score:
                best_local_score = eval3
                best_local = a3.copy()

        # Method 4: FISTA
        t4 = max(5.0, t_alloc - t1 - t2 - t3)
        if t4 > 5:
            a4, s4 = _optimize_fista(best_local, t4, rng, tau=50.0)
            eval4 = _score(a4)
            print(f"  fista: {eval4:.6f}")
            if eval4 < best_local_score:
                best_local_score = eval4
                best_local = a4.copy()

        # Update global best
        eval_best = _score(best_local)
        if eval_best < best_overall_score:
            best_overall_score = eval_best
            best_overall = best_local.copy()
            print(f"  *** NEW GLOBAL BEST: {eval_best:.6f} (n={len(best_local)}) ***")

    # Final polishing phase
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\n=== Final polishing ({remaining:.0f}s) ===")

        # Multiple polishing rounds with different methods
        polish_results = []

        # Round 1: Smooth with low tau (explore)
        t_p1 = remaining * 0.25
        if t_p1 > 10:
            a_p1, s_p1 = _optimize_smooth(best_overall, t_p1, rng, tau_init=10.0, tau_max=50.0)
            e_p1 = _score(a_p1)
            print(f"  polish1 (tau=10-50): {e_p1:.6f}")
            if e_p1 < best_overall_score:
                best_overall_score = e_p1
                best_overall = a_p1.copy()

        # Round 2: Smooth with high tau (exploit)
        t_p2 = remaining * 0.25
        if t_p2 > 10:
            remaining = deadline - time.monotonic()
            a_p2, s_p2 = _optimize_smooth(best_overall, t_p2, rng, tau_init=100.0, tau_max=1000.0)
            e_p2 = _score(a_p2)
            print(f"  polish2 (tau=100-1000): {e_p2:.6f}")
            if e_p2 < best_overall_score:
                best_overall_score = e_p2
                best_overall = a_p2.copy()

        # Round 3: Subgradient refinement
        t_p3 = remaining * 0.2
        if t_p3 > 10:
            remaining = deadline - time.monotonic()
            a_p3, s_p3 = _optimize_subgrad_v2(best_overall, t_p3, rng)
            e_p3 = _score(a_p3)
            print(f"  polish3 (subgrad): {e_p3:.6f}")
            if e_p3 < best_overall_score:
                best_overall_score = e_p3
                best_overall = a_p3.copy()

        # Round 4: Try perturbed restarts from best
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n = len(best_overall)
            for restart in range(3):
                if time.monotonic() > deadline - 10:
                    break
                t_r = (deadline - 10) - time.monotonic()
                if t_r < 10:
                    break
                a_r = best_overall * np.exp(rng.normal(0, 0.04, n))
                a_r = np.maximum(a_r, 1e-15)
                a_r = a_r / a_r.sum()
                a_r, s_r = _optimize_smooth(a_r, t_r * 0.7, rng, tau_init=50.0, tau_max=300.0)
                e_r = _score(a_r)
                print(f"  restart{restart+1}: {e_r:.6f}")
                if e_r < best_overall_score:
                    best_overall_score = e_r
                    best_overall = a_r.copy()

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s")

        # One last check: try truncating or extending if it helps
        # (the score has 2n in numerator, so shorter sequences might help)
        if final_score < np.inf:
            for trim in [0.9, 0.8, 0.7]:
                n_trim = int(len(best_overall) * trim)
                if n_trim < 500:
                    break
                a_trim = best_overall[:n_trim]
                s = a_trim.sum()
                if s > 0.01:
                    a_trim = a_trim / s
                    s_trim = _score(a_trim)
                    if s_trim < final_score:
                        print(f"  Trim to {n_trim}: {s_trim:.6f}")
                        final_score = s_trim
                        best_overall = a_trim

        return [float(x) for x in best_overall]
    else:
        a = np.ones(1000, dtype=np.float64)
        print("Falling back to uniform sequence")
        return a.tolist()
```
