def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = next_pow2(2 * n - 1)
    if N is None:
        N = need
    elif int(N) < int(need):
        N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))


def resample_seq(a, n_target):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n_target = max(2, int(n_target))
    if old_n == 1:
        return np.full(n_target, 1.0 / float(n_target), dtype=np.float64)
    if old_n == n_target:
        return a.astype(np.float64, copy=True)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n_target, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)

def make_exp_power(n, shape_param, decay_rate):
    """a[i] = exp(-decay_rate * (i/(n-1))^shape_param), normalized."""
    n = max(4, int(n))
    t = np.arange(n, dtype=np.float64) / float(n - 1)

    decay_rate = max(0.1, float(decay_rate))
    a = np.exp(-decay_rate * t ** shape_param)

def make_sigmoid(n, steepness, center):
    """a[i] = 1/(1+exp(steepness*(i/(n-1)-center))), normalized."""

    center = max(0.0, min(1.0, float(center)))
    a = 1.0 / (1.0 + np.exp(steepness * (t - center)))
    return normalize_positive(a, floor=1e-12)

    """a[i] = (1 - i/(n-1))^power, normalized."""
    n = max(4, int(n))
    t = np.arange(n, dtype=np.float64) / float(n - 1)
    power = max(0.1, float(power))
    a = (1.0 - t) ** power
    return normalize_positive(a, floor=1e-12)


def optimize_parametric(n, deadline, seed):
    """Optimize over parametric families using L-BFGS-B."""
    n = max(64, int(n))
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(int(seed))
    best_a = None
    best_score = np.inf

    def score_exp_power(params):
        sp, dr = float(params[0]), float(params[1])
        sp = np.clip(sp, 0.1, 10.0)
        dr = np.clip(dr, 0.1, 20.0)
        a = make_exp_power(n, sp, dr)
        c = convolve_seq(a, N)
        return 2.0 * n * float(np.max(c))

    def score_sigmoid(params):
        st, ce = float(params[0]), float(params[1])
        st = np.clip(st, 0.1, 50.0)
        ce = np.clip(ce, 0.0, 1.0)
        a = make_sigmoid(n, st, ce)
        c = convolve_seq(a, N)
        return 2.0 * n * float(np.max(c))

    def score_poly(params):
        pw = float(params[0])
        pw = np.clip(pw, 0.1, 20.0)
        a = make_poly_decay(n, pw)
        c = convolve_seq(a, N)
        return 2.0 * n * float(np.max(c))

    if _scipy_opt is not None:
        # Exponential power
        for p0 in [(1.0, 2.0), (1.5, 4.0), (2.0, 6.0), (0.8, 3.0), (2.5, 8.0), (1.2, 5.0)]:
            if time.monotonic() > deadline:
                break
            try:
                res = _scipy_opt.minimize(score_exp_power, np.array(p0, dtype=np.float64),
                                          method='L-BFGS-B', options={'maxiter': 80, 'ftol': 1e-14})
                if res.fun < best_score and np.isfinite(res.fun):
                    best_score = float(res.fun)
                    best_a = make_exp_power(n, float(np.clip(res.x[0], 0.1, 10.0)),
                                            float(np.clip(res.x[1], 0.1, 20.0)))
            except Exception:
                pass

        # Sigmoid
        for p0 in [(5.0, 0.4), (10.0, 0.5), (20.0, 0.5), (8.0, 0.3), (15.0, 0.4)]:
            if time.monotonic() > deadline:
                break
            try:
                res = _scipy_opt.minimize(score_sigmoid, np.array(p0, dtype=np.float64),
                                          method='L-BFGS-B', options={'maxiter': 80, 'ftol': 1e-14})
                if res.fun < best_score and np.isfinite(res.fun):
                    best_score = float(res.fun)
                    best_a = make_sigmoid(n, float(np.clip(res.x[0], 0.1, 50.0)),
                                          float(np.clip(res.x[1], 0.0, 1.0)))
            except Exception:
                pass

        # Polynomial decay
        for p0 in [[1.0], [2.0], [3.0], [0.5], [1.5]]:
            if time.monotonic() > deadline:
                break
            try:
                res = _scipy_opt.minimize(score_poly, np.array(p0, dtype=np.float64),
                                          method='L-BFGS-B', options={'maxiter': 50, 'ftol': 1e-14})
                if res.fun < best_score and np.isfinite(res.fun):
                    best_score = float(res.fun)
                    best_a = make_poly_decay(n, float(np.clip(res.x[0], 0.1, 20.0)))
            except Exception:
                pass

    # Also try random parametric points
    for _ in range(30):
        if time.monotonic() > deadline:
            break
        sp = rng.uniform(0.3, 5.0)
        dr = rng.uniform(0.5, 15.0)
        a = make_exp_power(n, sp, dr)
        sc = internal_score(a, N)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    if best_a is None:
        best_a = make_exp_power(n, 1.5, 4.0)
        best_score = internal_score(best_a, N)
    return best_a, best_score


def mirror_descent(a, deadline, N=None, verbose=False):
    """Adaptive mirror descent with log-sum-exp smoothing."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    if N is None:
        N = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = internal_score(a, N)
    loga = np.log(a + 1e-20)
    loga -= float(np.max(loga))

    eta = 0.5
    beta = 0.12
    tau_init = beta * float(np.max(convolve_seq(a, N)))
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        c = convolve_seq(a, N)
        cmax = float(np.max(c))
        score = 2.0 * n * cmax
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

        if cmax <= 0 or not np.isfinite(cmax):
            break

        tau = max(0.005 * cmax, beta * cmax)
        z = (c - cmax) / tau
        z = np.clip(z, -50.0, 0.0)
        ez = np.exp(z)
        s_ez = float(ez.sum())
        if not np.isfinite(s_ez) or s_ez <= 0:
            break
        w = ez / s_ez

        # Gradient: 2 * (w * a) circ convolution, reversed appropriately
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w[::-1], N)
        conv_w = np.fft.irfft(fa * fw, N)
        g = 2.0 * conv_w[n - 1:2 * n - 1][::-1]
        g = g - g.mean()
        gscale = float(np.abs(g).max())
        if not np.isfinite(gscale) or gscale <= 1e-15:
            break
        g = g / gscale

        J_cur = cmax + tau * np.log(s_ez)
        accepted = False

        for e_mult in (1.0, 0.5, 0.25, 0.125):
            e = eta * e_mult
            if e < 1e-10:
                continue
            loga_try = loga - e * g
            loga_try -= float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / float(a_try.sum())
            a_try = normalize_positive(a_try, floor=1e-15)

            c_try = convolve_seq(a_try, N)
            cmax_try = float(np.max(c_try))
            score_try = 2.0 * n * cmax_try
            J_try = cmax_try + tau * np.log(max(float(np.exp(np.clip((c_try - cmax_try) / tau, -50, 0)).sum()), 1e-300))

            if np.isfinite(score_try) and score_try < score * (1.0 - 1e-10):
                accepted = True
                a = a_try
                loga = np.log(a + 1e-20)
                loga -= float(np.max(loga))
                break
            if np.isfinite(J_try) and J_try < J_cur * (1.0 - 1e-6) and np.isfinite(score_try) and score_try < score * 1.03:
                accepted = True
                a = a_try
                loga = np.log(a + 1e-20)
                loga -= float(np.max(loga))
                break

        if accepted:
            stagnant = 0
            eta = min(1.5, eta * 1.02)
            beta = max(0.003, beta * 0.997)
        else:
            stagnant += 1
            eta = max(1e-5, eta * 0.6)
            if stagnant % 100 == 0:
                beta = min(0.3, beta * 1.5)
            if eta < 1e-6 or stagnant > 800:
                break

        it += 1
        if verbose and it % 500 == 0:
            print("  mirror iter={} best={:.9f}".format(it, best_score), flush=True)

        # Periodic escape
        if it % 1000 == 0 and stagnant > 200:
            rng_esc = np.random.default_rng(it + n)
            noise = rng_esc.normal(0, 0.08, n)
            a_esc = a * np.exp(np.clip(noise, -1, 1))
            a_esc = normalize_positive(a_esc, floor=1e-15)
            sc_esc = internal_score(a_esc, N)
            if np.isfinite(sc_esc) and sc_esc < best_score:
                best_a = a_esc.copy()
                best_score = sc_esc
                a = a_esc.copy()
                loga = np.log(a + 1e-20)
                loga -= float(np.max(loga))
                stagnant = 0
                eta = 0.3

    return best_a, best_score


def topk_score(c, K):
    """Sum of top K values in convolution (convex-ish surrogate for max)."""
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size < K:
        return float(c.sum())
    idx = np.argpartition(c, -K)[-K:]
    return float(c[idx].sum())


def binary_local_search(a_init, deadline, seed, K=8):
    """Binary local search with top-K objective and multiple restarts."""
    n = max(64, int(a_init.size))
    m = max(4, n // 2)
    N = next_pow2(2 * n - 1)

    best_b = None
    best_score = np.inf
    best_cmax = np.inf

    def init_binary(a0, m, rng):
        """Initialize binary from a continuous sequence."""
        b = np.zeros(n, dtype=np.float64)
        order = np.argsort(a0, kind='stable')
        b[order[-m:]] = 1.0
        return b

    def eval_binary(b):
        c = convolve_seq(b, N)
        cmax = float(np.max(c))
        score = 2.0 * n * cmax / (float(m) * float(m))
        return c, cmax, score

    rng = np.random.default_rng(int(seed))
    beta = 0.08
    temp = 0.05
    max_restarts = 3

    for restart in range(max_restarts):
        if time.monotonic() > deadline:
            break

        if restart == 0:
            b = init_binary(a_init, m, rng)
        else:
            # Perturb the best found
            if best_b is not None:
                b = best_b.copy()
                # Flip 5-10 bits
                n_flip = rng.integers(5, min(11, n // 4))
                ones = np.nonzero(b > 0.5)[0]
                zeros = np.nonzero(b < 0.5)[0]
                if ones.size >= n_flip // 2 and zeros.size >= n_flip // 2:
                    flip_out = rng.choice(ones, size=min(n_flip, ones.size), replace=False)
                    flip_in = rng.choice(zeros, size=min(n_flip, zeros.size), replace=False)
                    b[flip_out] = 0.0
                    b[flip_in] = 1.0
            else:
                b = init_binary(rng.random(n), m, rng)

        c, cmax, score = eval_binary(b)
        if score < best_score:
            best_score = score
            best_cmax = cmax
            best_b = b.copy()

        temp = 0.05 + 0.03 * restart
        beta = 0.08
        stale = 0

        while time.monotonic() < deadline:
            cmax = float(np.max(c))
            score = 2.0 * n * cmax / (float(m) * float(m))
            if score < best_score:
                best_score = score
                best_cmax = cmax
                best_b = b.copy()

            # Compute gradient using top-K weighting
            # Weight positions by how close they are to the top K
            sorted_idx = np.argsort(c)[::-1]
            topk_val = float(c[sorted_idx[K - 1]]) if K < c.size else float(c.min())
            margin = cmax - topk_val
            if margin < 1e-8:
                w = (c >= cmax * 0.99).astype(np.float64)
            else:
                w = np.clip((c - topk_val) / margin, 0, 1) + 0.1 * (c >= cmax * 0.95).astype(np.float64)
            wsum = float(w.sum())
            if wsum < 1e-10:
                w = np.ones_like(c) / c.size
            else:
                w = w / wsum

            # Gradient via FFT
            fa = np.fft.rfft(b, N)
            fw = np.fft.rfft(w[::-1], N)
            conv_w = np.fft.irfft(fa * fw, N)
            g = 2.0 * conv_w[n - 1:2 * n - 1][::-1]

            # Choose a 1 to remove (high gradient) and a 0 to add (low gradient)
            one_idx = np.nonzero(b > 0.5)[0]
            zero_idx = np.nonzero(b < 0.5)[0]
            if one_idx.size == 0 or zero_idx.size == 0:
                break

            # Softmax selection with temperature
            g_ones = g[one_idx]
            g_zeros = g[zero_idx]

            # Pick from ones: prefer high gradient (remove it)
            g_ones_shift = g_ones - float(g_ones.max())
            p_ones = np.exp(np.clip(g_ones_shift / max(temp, 0.01), -50, 0))
            p_ones = p_ones / max(float(p_ones.sum()), 1e-300)
            si = int(rng.choice(len(one_idx), p=p_ones))

            # Pick from zeros: prefer low gradient (add here)
            g_zeros_shift = float(g_zeros.min()) - g_zeros
            p_zeros = np.exp(np.clip(g_zeros_shift / max(temp, 0.01), -50, 0))
            p_zeros = p_zeros / max(float(p_zeros.sum()), 1e-300)
            sj = int(rng.choice(len(zero_idx), p=p_zeros))

            i = int(one_idx[si])
            j = int(zero_idx[sj])

            if i == j:
                continue

            # Apply swap
            two_b = 2.0 * b
            c[i:i + n] -= two_b
            c[2 * i] -= 1.0
            c[j:j + n] += two_b
            c[2 * j] += 1.0
            b[i] = 0.0
            b[j] = 1.0

            new_cmax = float(np.max(c))
            new_score = 2.0 * n * new_cmax / (float(m) * float(m))

            accept = False
            if new_score < score * (1.0 - 1e-12):
                accept = True
            else:
                delta = new_score - score
                if delta < 0.5:
                    if rng.random() < np.exp(-delta / max(temp, 0.005)):
                        accept = True

            if accept:
                stale = 0
                temp = max(0.005, temp * 0.9985)
            else:
                # Revert
                two_b_new = 2.0 * b
                c[j:j + n] -= two_b_new
                c[2 * j] -= 1.0
                c[i:i + n] += two_b_new
                c[2 * i] += 1.0
                b[j] = 0.0
                b[i] = 1.0
                stale += 1
                temp = max(0.003, temp * 0.97)
                if stale % 500 == 0:
                    temp = min(0.15, temp * 2.0)
                    stale = 0
                if stale > 3000:
                    break

            # Periodic evaluation with actual score
            if (it % 1000 if 'it' in dir() else 1) == 0:
                pass

    if best_b is None:
        best_b = init_binary(a_init, m, np.random.default_rng(0))
        _, _, best_score = eval_binary(best_b)

    return normalize_positive(best_b, floor=0.0), best_score


def lowdim_logspace(a_init, n, deadline, seed):
    """Optimize in a low-dimensional log-space parameterization."""
    n = max(64, int(n))
    N = next_pow2(2 * n - 1)
    B = _build_basis(n)
    D = B.shape[1]
    rng = np.random.default_rng(int(seed))

    a = resample_seq(a_init, n)
    z0 = np.log(np.maximum(a, 1e-12))
    z0 -= z0.mean()
    theta0, _, _, _ = np.linalg.lstsq(B, z0, rcond=None)
    theta = theta0.astype(np.float64).ravel().copy()

    best_a = a.copy()
    best_score = internal_score(a, N)

    if _scipy_opt is None:
        return best_a, best_score

    def obj_jac(theta):
        theta = np.asarray(theta, dtype=np.float64).ravel()
        z = B @ theta
        z = z - float(np.max(z))
        a_loc = np.exp(np.clip(z, -50, 0))
        a_loc = a_loc / float(a_loc.sum())
        c = convolve_seq(a_loc, N)
        cmax = float(np.max(c))
        val = 2.0 * n * cmax
        # Subgradient for max: gradient of active components
        active = (c >= cmax * 0.999)
        w = active.astype(np.float64)
        wsum = float(w.sum())
        if wsum < 1e-10:
            return val, np.zeros(D)
        w = w / wsum
        fa = np.fft.rfft(a_loc, N)
        fw = np.fft.rfft(w[::-1], N)
        conv_w = np.fft.irfft(fa * fw, N)
        ga = 2.0 * conv_w[n - 1:2 * n - 1][::-1]
        total = float(np.dot(a_loc, ga))
        aB = a_loc @ B
        agaB = (a_loc * ga) @ B
        gtheta = agaB - total * aB
        return val, (2.0 * n) * gtheta

    for fac in (0.3, 0.1, 0.04, 0.015):
        if time.monotonic() > deadline:
            break
        try:
            res = _scipy_opt.minimize(
                lambda t: obj_jac(t)[0],
                theta,
                jac=lambda t: obj_jac(t)[1],
                method='L-BFGS-B',
                options={'maxiter': 30, 'ftol': 1e-14, 'gtol': 1e-6}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                theta = res.x.astype(np.float64).ravel()
        except Exception:
            pass

        z = B @ theta
        z = z - float(np.max(z))
        a_try = np.exp(np.clip(z, -50, 0))
        a_try = a_try / float(a_try.sum())
        a_try = normalize_positive(a_try, floor=1e-12)
        sc = internal_score(a_try, N)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_try.copy()

    return best_a, best_score


def _build_basis(n):
    n = max(8, int(n))
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    cols = []
    # Monomial basis (centered)
    for p in range(1, 7):
        col = x ** p
        col = col - col.mean()
        cols.append(col)
    # Sine modes
    for m in range(1, 8):
        cols.append(np.sin(np.pi * m * x))
        cols.append(np.cos(np.pi * m * x))
    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms <= 1e-14] = 1.0
    B = B / norms
    return B


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()
    total = max(120.0, float(budget_s) - 30.0)
    deadline = start + total

    best_a = np.ones(1024, dtype=np.float64) / 1024.0
    best_score = 2.0
    print("initial score = {:.9f}".format(best_score), flush=True)

    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            print("given score = {:.9f}".format(best_score), flush=True)

    # Phase 1: Parametric optimization (120s)
    p1_deadline = start + min(120.0, total * 0.08)
    for n in (1024, 2048, 4096):
        if time.monotonic() > p1_deadline:
            break
        allow = min(30.0, p1_deadline - time.monotonic())
        if allow < 5.0:
            break
        a_init = resample_seq(best_a, n)
        a1, sc1 = optimize_parametric(n, time.monotonic() + allow, seed + n)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("parametric n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 2: Low-dimensional log-space optimization (180s)
    p2_deadline = start + min(300.0, total * 0.15)
    for n in (1024, 2048, 4096, 8192):
        if time.monotonic() > p2_deadline:
            break
        allow = min(40.0, p2_deadline - time.monotonic())
        if allow < 8.0:
            break
        a_init = resample_seq(best_a, n)
        a1, sc1 = lowdim_logspace(a_init, n, time.monotonic() + allow, seed + 2000 + n)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("lowdim n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 3: Mirror descent (600s)
    p3_deadline = start + min(900.0, total * 0.4)
    for n in (2048, 4096, 8192):
        if time.monotonic() > p3_deadline:
            break
        allow = min(200.0, p3_deadline - time.monotonic())
        if allow < 20.0:
            break
        a_init = resample_seq(best_a, n)
        a1, sc1 = mirror_descent(a_init, time.monotonic() + allow, verbose=True)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("mirror n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 4: Binary local search (remaining - 60s)
    p4_deadline = deadline - 60.0
    for n in (4096, 8192, 16384):
        if time.monotonic() > p4_deadline:
            break
        allow = min(400.0, p4_deadline - time.monotonic())
        if allow < 30.0:
            break
        a_init = resample_seq(best_a, n)
        a1, sc1 = binary_local_search(a_init, time.monotonic() + allow, seed=seed + 9000 + n, K=6)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("binary n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 5: Final mirror polish on best
    if time.monotonic() < deadline - 20.0:
        n_final = best_a.size
        a1, sc1 = mirror_descent(best_a, time.monotonic() + min(120.0, deadline - time.monotonic() - 15.0))
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("final polish score={:.9f}".format(best_score), flush=True)

    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]

    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float('inf')

    if not np.isfinite(final_eval):
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print("FINAL: internal={:.12f} evaluated={:.12f} len={}".format(
        best_score, final_eval, len(out_list)), flush=True)
    return out_list
