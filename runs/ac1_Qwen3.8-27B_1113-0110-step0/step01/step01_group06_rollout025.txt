Strategy: Full-vector multiplicative-weight optimization with:

- FFT-based gradients
- Momentum for gradient smoothing

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    """Compute gradient of sum_k w_k * b_k w.r.t. a, where b = a*a (full conv).
    grad_i = 2 * sum_k w_k * (a * w)_i (circular-ish, but we use linear)
    Actually: d/db_k (a_i) = a_{k-i}, so d/d a_i [sum_k w_k b_k] = 2 * sum_k w_k * a_{k-i}
    which is 2 * (w * a)_i (linear convolution, but we only need the part that overlaps with a's support)
    """
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _make_weights_hybrid(b, focus, topk_frac=0.1, eps=0.01):
    """Hybrid weighting: exponential on (max-b) + uniform on top-k."""
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    
    # Exponential weights
    with np.errstate(over="ignore", under="ignore"):
        z = (maxb - b) / maxb
        p_exp = np.exp(-z * float(focus))
    s_exp = float(p_exp.sum())
    if s_exp > 1e-12:
        p_exp = p_exp / s_exp
    else:
        p_exp = np.zeros(L, dtype=np.float64)
        p_exp[int(np.argmax(b))] = 1.0
    
    # Top-k uniform weights
    k = max(1, int(L * topk_frac))
    topk_indices = np.argpartition(b, -k)[-k:]
    p_topk = np.zeros(L, dtype=np.float64)
    p_topk[topk_indices] = 1.0 / k
    
    # Hybrid
    alpha = min(1.0, focus / 50.0)  # Blend between top-k (low focus) and exponential (high focus)
    p = (1 - alpha) * p_topk + alpha * p_exp
    p = (1.0 - eps) * p + eps / L
    
    return p, maxb


def _make_weights_topk(b, topk_frac=0.05, eps=0.01):
    """Pure top-k weighting."""
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    
    k = max(1, int(L * topk_frac))
    topk_indices = np.argpartition(b, -k)[-k:]
    p = np.zeros(L, dtype=np.float64)
    p[topk_indices] = 1.0 / k
    p = (1.0 - eps) * p + eps / L
    
    return p, maxb

def _anti_log_init(n, rng, amp=3.0, smooth=False):

        z[m] = 0.0

    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    return a


def _tapered_init(n, rng):
    """Initialize with a tapered (bell-shaped) profile modified by noise."""
    n = int(n)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    base = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    a = base * np.exp(rng.normal(0.0, 0.5, size=n))
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a

def _greedy_x(n, rng, shuffle=True):

    sel = _greedy_support(n, rng, shuffle=shuffle)

def _greedy_support(n, rng, shuffle=True):
    n = int(n)
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m, dtype=np.int64)
    if shuffle:
        rng.shuffle(order)

    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            affected_l = int(np.max(counts[s + l])) if t > 0 else 0
            self_l = int(counts[2 * l])
            metric_l = max(affected_l + 2, self_l + 1)

            affected_r = int(np.max(counts[s + r])) if t > 0 else 0
            self_r = int(counts[2 * r])
            metric_r = max(affected_r + 2, self_r + 1)

            y = l if metric_l <= metric_r else r

        if t > 0:
            counts[selected[:t] + y] += 2
        counts[2 * y] += 1
        selected[t] = y
        t += 1

    return selected

"""
Optimize a non-negative step function sequence to minimize
    2 * n * max(convolve(a, a)) / (sum(a))**2.

Uses full-vector multiplicative-weight optimization with:
- Hybrid top-k + exponential weighting on convolution coefficients
- FFT-based gradients with momentum
- Diverse initializations (anti-palindromic, random, binary, tapered)
- Adaptive learning rate and focus scheduling
- Multi-scale search over sequence lengths
- Extended final refinement
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _make_weights_hybrid(b, focus, topk_frac=0.05, eps=0.01):
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    with np.errstate(over="ignore", under="ignore"):
        z = (maxb - b) / maxb
        p_exp = np.exp(-z * float(focus))
    s_exp = float(p_exp.sum())
    if s_exp > 1e-12:
        p_exp = p_exp / s_exp
    else:
        p_exp = np.zeros(L, dtype=np.float64)
        p_exp[int(np.argmax(b))] = 1.0
    k = max(1, int(L * topk_frac))
    topk_indices = np.argpartition(b, -k)[-k:]
    p_topk = np.zeros(L, dtype=np.float64)
    p_topk[topk_indices] = 1.0 / k
    alpha = min(1.0, focus / 30.0)
    p = (1.0 - alpha) * p_topk + alpha * p_exp
    p = (1.0 - eps) * p + eps / L
    return p, maxb


def _random_positive(n, rng, amp=3.0):
    n = int(n)
    z = rng.normal(0.0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _anti_log_init(n, rng, amp=3.0, smooth=False):
    n = int(n)
    m = n // 2
    if m < 1:
        return np.ones(n, dtype=np.float64) / max(1, n)
    x = np.arange(m, dtype=np.float64)
    if smooth:
        v = np.zeros(m, dtype=np.float64)
        for f in (1, 2, 3, 5, 7, 11):
            v += (rng.normal(0.0, 1.0) / max(1, f)) * np.sin(
                2.0 * np.pi * f * x / m + rng.uniform(0.0, 2.0 * np.pi)
            )
        v += 0.1 * rng.normal(0.0, 1.0, m)
    else:
        v = rng.normal(0.0, 1.0, m)
    v = v - float(v.mean())
    std = float(v.std())
    if std > 1e-12:
        v = (float(amp) * v) / std
    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0
    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _binary_anti_init(n, rng):
    n = int(n)
    m = n // 2
    a = np.zeros(n, dtype=np.float64)
    c = 2.0 / n
    for i in range(m):
        if rng.random() < 0.5:
            a[i] = c
            a[n - 1 - i] = 0.0
        else:
            a[i] = 0.0
            a[n - 1 - i] = c
    if n % 2 == 1:
        a[m] = c / 2.0
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _tapered_init(n, rng):
    n = int(n)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    base = np.exp(-((x - center) ** 2) / (2.0 * sigma ** 2))
    a = base * np.exp(rng.normal(0.0, 0.5, size=n))
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _a_from_x(x, n=None):
    x = np.asarray(x, dtype=np.float64).ravel()
    m = x.size
    if n is None:
        n = 2 * m
    t = 1.0 / m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a


def _optimize_full(a0, time_limit, rng, verbose=False, focus_start=5.0,
                   max_focus=3000.0, label=""):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / max(1, n)
    else:
        a /= s
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = None
    score = float("inf")
    best_score = float("inf")
    best_a = a.copy()

    focus = float(focus_start)
    focus_inc = 1.012
    eta = 0.5
    eta_min = 0.005
    eta_max = 15.0
    momentum = np.zeros(n, dtype=np.float64)
    momentum_decay = 0.95
    no_improve = 0
    max_no = 600
    resets = 0
    max_resets = 60
    t0 = time.monotonic()
    it = 0
    topk_frac = 0.05

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 300 == 0):
            print(
                f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                f"score={score:.6f} best={best_score:.6f} "
                f"focus={focus:.1f} eta={eta:.3f} resets={resets}"
            )

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.15)
            r = rng.random()
            if r < 0.25:
                a = 0.7 * best_a + 0.3 * _random_positive(n, rng, amp=3.0)
            elif r < 0.45:
                a = 0.8 * best_a + 0.2 * _anti_log_init(n, rng, amp=3.0, smooth=(r < 0.35))
            elif r < 0.65:
                a = best_a * np.exp(rng.normal(0.0, 0.6, size=n))
            elif r < 0.80:
                a = 0.5 * best_a + 0.5 * _tapered_init(n, rng)
            elif r < 0.90:
                a = 0.9 * best_a + 0.1 * _binary_anti_init(n, rng)
            else:
                a = _random_positive(n, rng, amp=2.0)
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            momentum = np.zeros(n, dtype=np.float64)
            b = None
            score = float("inf")
            focus = float(rng.choice([1.0, 2.0, 5.0, 10.0, 50.0]))
            eta = 0.5
            topk_frac = float(rng.choice([0.02, 0.05, 0.1, 0.15]))
            continue

        p, _ = _make_weights_hybrid(b, focus, topk_frac=topk_frac)
        grad = _fft_grad(p, a)
        gd = grad - float(np.dot(a, grad))
        momentum = momentum_decay * momentum + (1.0 - momentum_decay) * gd
        maxg = float(np.max(np.abs(momentum)))
        if maxg < 1e-18:
            break
        gd = momentum / maxg
        a_new = a * np.exp(-eta * gd)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss <= 1e-300:
            break
        a_new /= ss
        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)
        wt_old = float(np.dot(p, b))
        wt_new = float(np.dot(p, b_new))

        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        elif score_new < score * 1.003 and wt_new < wt_old - 1e-10:
            accept = True

        if accept:
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.03, eta_max)
        else:
            eta = max(eta * 0.8, eta_min)
            if eta < 0.03:
                a = a * np.exp(rng.normal(0.0, 0.15, size=n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                momentum = np.zeros(n, dtype=np.float64)
                b = None
                score = float("inf")
                eta = 0.3

        if it % 1500 == 1499:
            focus = max(1.0, focus * 0.4)
        else:
            focus = min(focus * focus_inc, float(max_focus))
        topk_frac = 0.05 + 0.03 * np.sin(it / 400.0)
        topk_frac = max(0.01, min(0.15, topk_frac))
        it += 1

    return best_a, float(best_score)


def _optimize_pair(x0, time_limit, rng, verbose=False, focus_start=5.0,
                   max_focus=3000.0, label=""):
    x = np.asarray(x0, dtype=np.float64).copy()
    m = int(x.size)
    n = 2 * m
    x_eps = 1e-8
    x = np.clip(x, x_eps, 1.0 - x_eps)

    a = _a_from_x(x, n)
    b = None
    score = float("inf")
    best_score = float("inf")
    best_x = x.copy()

    focus = float(focus_start)
    focus_inc = 1.012
    eta = 0.5
    eta_min = 0.005
    eta_max = 15.0
    momentum = np.zeros(m, dtype=np.float64)
    momentum_decay = 0.95
    no_improve = 0
    max_no = 800
    resets = 0
    max_resets = 40
    t0 = time.monotonic()
    it = 0
    topk_frac = 0.05

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-14:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 500 == 0):
            print(
                f"[{label}] pair it={it} t={time.monotonic()-t0:.1f}s "
                f"score={score:.6f} best={best_score:.6f} "
                f"focus={focus:.1f} eta={eta:.3f} resets={resets}"
            )

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.15)
            x = 0.7 * best_x + 0.3 * np.clip(rng.normal(0.5, 0.3, size=m), x_eps, 1.0 - x_eps)
            momentum = np.zeros(m, dtype=np.float64)
            a = _a_from_x(x, n)
            b = None
            score = float("inf")
            focus = float(rng.choice([2.0, 5.0, 10.0, 50.0]))
            eta = 0.5
            continue

        p, _ = _make_weights_hybrid(b, focus, topk_frac=topk_frac)
        ga = _fft_grad(p, a)
        gx = ga[:m] - ga[m:][::-1]
        momentum = momentum_decay * momentum + (1.0 - momentum_decay) * gx
        maxg = float(np.max(np.abs(momentum)))
        if maxg < 1e-18:
            break
        gx = momentum / maxg
        x_new = x - eta * gx
        x_new = np.clip(x_new, x_eps, 1.0 - x_eps)
        if not np.any(x_new != x):
            x_new = np.clip(x + rng.normal(0.0, 0.05, size=m), x_eps, 1.0 - x_eps)
        a_new = _a_from_x(x_new, n)
        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)
        wt_old = float(np.dot(p, b))
        wt_new = float(np.dot(p, b_new))

        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        elif score_new < score * 1.003 and wt_new < wt_old - 1e-10:
            accept = True

        if accept:
            x = x_new
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.03, eta_max)
        else:
            eta = max(eta * 0.8, eta_min)
            if eta < 0.03:
                x = np.clip(x + rng.normal(0.0, 0.08, size=m), x_eps, 1.0 - x_eps)
                a = _a_from_x(x, n)
                momentum = np.zeros(m, dtype=np.float64)
                b = None
                score = float("inf")
                eta = 0.3

        if it % 1500 == 1499:
            focus = max(1.0, focus * 0.4)
        else:
            focus = min(focus * focus_inc, float(max_focus))
        topk_frac = 0.05 + 0.03 * np.sin(it / 300.0)
        topk_frac = max(0.01, min(0.12, topk_frac))
        it += 1

    best_a = _a_from_x(best_x, n)
    return best_a, float(best_score)


def _greedy_support(n, rng, shuffle=True):
    n = int(n)
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m, dtype=np.int64)
    if shuffle:
        rng.shuffle(order)
    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            affected_l = int(np.max(counts[s + l]))
            self_l = int(counts[2 * l])
            metric_l = max(affected_l + 2, self_l + 1)
            affected_r = int(np.max(counts[s + r]))
            self_r = int(counts[2 * r])
            metric_r = max(affected_r + 2, self_r + 1)
            y = l if metric_l <= metric_r else r
        if t > 0:
            counts[selected[:t] + y] += 2
        counts[2 * y] += 1
        selected[t] = y
        t += 1
    return selected


def _greedy_x(n, rng, shuffle=True):
    sel = _greedy_support(n, rng, shuffle=shuffle)
    m = n // 2
    x = np.full(m, 1e-6, dtype=np.float64)
    left = sel < m
    if np.any(left):
        x[sel[left]] = 1.0 - 1e-6
    return x


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
    Uses multiplicative-weight optimization with momentum, hybrid weighting,
    diverse restarts, and multi-length exploration.
    """
    seed = int(seed)
    total = max(60.0, float(budget_s))
    now0 = time.monotonic()
    final_deadline = now0 + total - 15.0

    print("=== Step-function convolution optimizer v3 ===")
    best_a = None
    best_score = float("inf")

    # Load given sequence
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                sc, _, _ = _internal_score(gn)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = gn.copy()
                given_a = gn
                print(f"given: n={gn.size}, score={sc:.6f}")
    except Exception:
        pass

    # Build task list: prioritize the most promising configurations
    tasks = []

    if given_a is not None:
        # Long optimization on the given sequence (best performer last time)
        tasks.append(("full", given_a.size, 0, 350.0, "given"))
        tasks.append(("full", given_a.size, 1, 300.0, "given2"))

    # Full-vector at 8192
    tasks.append(("full", 8192, 0, 250.0, "full8192"))
    # Larger n
    tasks.append(("full", 12000, 0, 250.0, "full12000"))
    # Anti-palindromic
    tasks.append(("pair", 8192, 0, 150.0, "pair8192"))
    # Binary init full
    tasks.append(("full", 8192, 2, 150.0, "bin8192"))
    # Even larger
    tasks.append(("full", 16000, 0, 200.0, "full16000"))

    idx = 0
    for kind, n, task_seed, max_alloc, tag in tasks:
        now = time.monotonic()
        remaining = final_deadline - now
        # Reserve time for final refinement
        reserve = 250.0 if best_a is not None else 50.0
        if remaining < reserve + 30.0:
            break
        alloc = min(max_alloc, remaining - reserve)
        if alloc < 30.0:
            break

        ng_seed = (seed * 1000003 + idx * 7919 + task_seed * 131) % (2**32)
        task_rng = np.random.default_rng(ng_seed)

        print(f"\n--- task {idx} [{tag}] kind={kind} n={n} alloc={alloc:.0f}s best={best_score:.6f} ---")
        t0 = time.monotonic()
        cand_a = None
        cand_score = float("inf")

        try:
            if kind == "full":
                if tag == "given" and given_a is not None:
                    a0 = given_a.copy()
                elif tag == "given2" and given_a is not None:
                    a0 = given_a * np.exp(task_rng.normal(0.0, 0.4, size=given_a.size))
                    a0 = np.maximum(a0, 1e-15)
                    a0 /= a0.sum()
                elif task_seed == 0:
                    a0 = _anti_log_init(n, task_rng, amp=3.0, smooth=True)
                elif task_seed == 1:
                    a0 = _random_positive(n, task_rng, amp=2.5)
                elif task_seed == 2:
                    a0 = _binary_anti_init(n, task_rng)
                else:
                    a0 = _anti_log_init(n, task_rng, amp=4.0, smooth=False)

                cand_a, cand_score = _optimize_full(
                    a0, alloc, task_rng,
                    verbose=(alloc > 200),
                    focus_start=3.0 + task_seed * 2.0,
                    label=tag,
                )
            elif kind == "pair":
                x0 = _greedy_x(n, task_rng, shuffle=(task_seed != 0))
                cand_a, cand_score = _optimize_pair(
                    x0, alloc, task_rng,
                    verbose=False,
                    focus_start=5.0,
                    label=tag,
                )
            else:
                a0 = _anti_log_init(n, task_rng, amp=3.0)
                cand_a, cand_score = _optimize_full(
                    a0, alloc, task_rng, focus_start=5.0, label=tag
                )

            if cand_a is not None:
                cand_a = np.nan_to_num(cand_a, nan=0.0, posinf=0.0, neginf=0.0)
                cand_a = np.clip(cand_a, 0.0, 1000.0)
                cc = float(cand_a.sum())
                if cc > 0.01:
                    cand_a /= cc
                    internal, _, _ = _internal_score(cand_a)
                    if np.isfinite(internal) and internal < cand_score:
                        cand_score = internal
                    if np.isfinite(cand_score) and cand_score < best_score:
                        best_score = cand_score
                        best_a = cand_a.copy()
                        print(f"  *** NEW BEST: score={best_score:.6f} n={cand_a.size} ***")
        except Exception as exc:
            print(f"  task {idx} failed: {repr(exc)}")

        print(f"--- task {idx} done: used={time.monotonic()-t0:.1f}s cand={cand_score:.6f} best={best_score:.6f} ---")
        idx += 1

    # Final refinement: optimize the best found so far
    if best_a is not None:
        remaining = final_deadline - time.monotonic()
        if remaining > 100.0:
            refine_time = min(remaining - 20.0, 600.0)
            if refine_time > 30.0:
                task_rng = np.random.default_rng((seed + 777777) % (2**32))
                print(f"\n=== Final refinement: time={refine_time:.0f}s, starting score={best_score:.6f} ===")
                refined_a, refined_score = _optimize_full(
                    best_a, refine_time, task_rng,
                    verbose=True, focus_start=8.0,
                    label="FINAL",
                )
                if refined_a is not None and np.isfinite(refined_score):
                    if refined_score < best_score:
                        best_a = refined_a
                        best_score = refined_score
                    print(f"After refinement: score={best_score:.6f}")

    # Secondary refinement with perturbation
    if best_a is not None and time.monotonic() < final_deadline - 30.0:
        remaining = final_deadline - time.monotonic()
        if remaining > 60.0:
            refine_time = min(remaining - 15.0, 200.0)
            if refine_time > 20.0:
                task_rng = np.random.default_rng((seed + 888888) % (2**32))
                perturbed = best_a * np.exp(task_rng.normal(0.0, 0.2, size=best_a.size))
                perturbed = np.maximum(perturbed, 1e-15)
                perturbed /= perturbed.sum()
                print(f"\n=== Secondary refinement: time={refine_time:.0f}s ===")
                r2_a, r2_score = _optimize_full(
                    perturbed, refine_time, task_rng,
                    verbose=False, focus_start=5.0,
                    label="SECOND",
                )
                if r2_a is not None and np.isfinite(r2_score) and r2_score < best_score:
                    best_a = r2_a
                    best_score = r2_score
                    print(f"After secondary: score={best_score:.6f}")

    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]

    if best_a.size <= 25000 and time.monotonic() < final_deadline - 3.0:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"\nFINAL: internal={best_score:.6f} exact={exact:.6f} n={best_a.size}")
        else:
            print(f"\nFINAL: internal={best_score:.6f} exact=n/a n={best_a.size}")
    else:
        print(f"\nFINAL: internal={best_score:.6f} n={best_a.size}")

    return final_seq
