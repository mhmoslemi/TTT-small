```python
import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _lp_solve(f, M, max_iter=50000):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0."""
    n = len(f)
    # Build C_f efficiently using Toeplitz structure
    # C_f[k, j] = f[k-j] if 0 <= k-j < n, else 0
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = f
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.status == 0:
        return result.x
    return None

def _alternating_lp_pure(a_init, max_iters, deadline):
    """Pure alternating LP (no blending)."""
    a = a_init.copy()
    n = len(a)
    s = float(np.sum(a))
    a = a / s  # normalize to sum 1
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g  # normalize
        # Check if g is significantly different from a
        diff = np.max(np.abs(g - a))
        if diff < 1e-8:
            break  # converged
        a = g
        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score
```

import time
import math
from scipy.optimize import linprog

def _fft_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2n * max(conv)."""
    n = len(a)
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c))

def _build_C(f):
    n = len(f)
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = f
    return C

def _solve_lp(C, M):
    n = C.shape[1]
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.status == 0 and result.fun is not None:
        return result.x
    return None

def _lp_iteration(a, M):
    """One LP iteration: solve for g given a and M."""
    n = len(a)
    C = _build_C(a)
    g = _solve_lp(C, M)
    if g is None:
        return None
    return g

def _alternating_lp(a_init, max_iters, deadline, blend_schedule='decay'):
    """Alternating LP with adaptive blending."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    
    blend = 1.0  # Start with pure LP solution
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(np.max(c))
        
        g = _lp_iteration(a, M)
        if g is None:
            break
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        
        # Check convergence
        diff = float(np.max(np.abs(g - a)))
        if diff < 1e-10:
            break
        
        # Blend
        if blend_schedule == 'decay':
            a_new = (1.0 - blend) * a + blend * g
        else:
            a_new = g
        
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        blend = max(0.3, blend * 0.995)
    
    return best_a, best_score

def _project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _sensitivity_at_lag(a, k):
    """d(conv[k])/da: sensitivity vector at lag k."""
    n = len(a)
    s = np.zeros(n)
    # Term 1: a[k-j] for j where 0<=k-j<n => j in [k-n+1, k] intersect [0, n-1]
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        s[j_lo:j_hi+1] += a[k - j_lo:k - j_hi + 1][::-1]
    # Term 2: a[j-k] for j where 0<=j-k<n => j in [k, k+n-1] intersect [0, n-1]
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2:
        s[j_lo2:j_hi2+1] += a[j_lo2-k:j_hi2-k+1]
    return s

def _smoothed_gradient(a, tau, top_k=50):
    """Compute gradient of smoothed max using top-k lags."""
    n = len(a)
    c = _fft_convolve(a)
    # Get top-k lags
    if top_k >= len(c):
        top_k = len(c)
    idx = np.argpartition(c, -top_k)[-top_k:]
    # Softmax weights over top-k
    c_top = c[idx]
    max_ct = c_top.max()
    w = np.exp(tau * (c_top - max_ct))
    w /= w.sum()
    # Weighted sum of sensitivity vectors
    grad = np.zeros(n)
    for i, k in enumerate(idx):
        if w[i] > 1e-10:
            grad += w[i] * _sensitivity_at_lag(a, int(k))
    return grad

def _projected_gradient(a_init, lr_init, tau, max_iters, deadline, lr_decay=0.999):
    """Projected gradient descent on smoothed max(conv) with simplex constraint."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad = _smoothed_gradient(a, tau)
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)
        
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        lr *= lr_decay
    
    return best_a, best_score

def _interpolate_sequence(src, n_target):
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out
