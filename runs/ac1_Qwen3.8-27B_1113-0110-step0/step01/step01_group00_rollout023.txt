def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _sensitivity(a, k):
    """Compute sensitivity vector (row of C_k) for peak lag k."""
    n = len(a)
    j = np.arange(n)
    s = np.zeros(n)
    idx1 = k - j
    m1 = (idx1 >= 0) & (idx1 < n)
    s[m1] += a[idx1[m1]]
    idx2 = j - k
    m2 = (idx2 >= 0) & (idx2 < n)
    s[m2] += a[idx2[m2]]
    return s


def _softplus_objective_and_grad(a_flat, T, top_k=30):
    """Smoothed objective and gradient using softmax approximation of max."""

    c = _fft_convolve(a_flat)

    if S < 1e-10:
        return 1e10, np.zeros(n)

    # Numerically stable log-sum-exp
    c_max = float(np.max(c))
    logits = (c - c_max) / T
    exp_logits = np.exp(logits)
    log_sum = c_max / T + np.log(np.sum(exp_logits))
    r_smooth = T * log_sum  # = log(sum exp(c/T)) * T = softmax-approx of max

    # Softmax weights
    p = exp_logits / np.sum(exp_logits)

    # Gradient: sum_k p[k] * 2 * sensitivity(a, k)

def _build_conv_cross_matrix(f, n_g):
    """Build matrix C where (C @ g)[k] = conv(f, g)[k]."""
    n_f = len(f)
    C = np.zeros((n_f + n_g - 1, n_g), dtype=np.float64)
    for j in range(n_g):
        lo = j
        hi = min(n_f + n_g - 1, j + n_f)
        C[lo:hi, j] = f[:hi - lo]
    return C


def _lp_solve_cross(f, M, top_k_lags=None):
    """Solve max sum(g) s.t. conv(f,g)[k] <= M for selected lags, g >= 0."""
    n = len(f)
    if top_k_lags is not None:
        c_ff = _fft_convolve(f)
        idx = np.argsort(c_ff)[::-1][:top_k_lags]
    else:
        idx = np.arange(2 * n - 1)

    C = _build_conv_cross_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(len(idx))
    C_sub = C[idx, :]
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C_sub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None

def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

        print(f"  L-BFGS T: {t_start} -> {t_end}")
        a_opt, sc = _lbfgs_smooth(best_a, t_start, t_end, max_iter=max_it,
                                   deadline=phase1_deadline, top_k=30)
        print(f"  Result: score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")

    # ================================================================
    # Phase 2: Improved Iterative LP at the best length
    # ================================================================
    print(f"\n=== Phase 2: Iterative LP (n={n_best}) ===")
    phase2_deadline = min(total_deadline, time.monotonic() + 700.0)

    # Try with different top_k_lags and blends
    for top_k in [None, 20, 10, 5]:
        if time.monotonic() > phase2_deadline:
            break
        label = f"top_k={top_k}" if top_k else "all lags"
        print(f"  LP {label}")
        a_opt, sc = _iterative_lp(best_a, max_iters=300, deadline=phase2_deadline,
                                   initial_blend=0.7, top_k_lags=top_k, line_search=True)
        print(f"  {label} result: score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")

    # Second LP pass with more conservative blend
    if time.monotonic() < phase2_deadline - 5:
        print("  LP conservative blend=0.4")
        a_opt, sc = _iterative_lp(best_a, max_iters=200, deadline=phase2_deadline,
                                   initial_blend=0.4, top_k_lags=15, line_search=True)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")

    # ================================================================
    # Phase 3: Multi-n search
    # ================================================================
    print(f"\n=== Phase 3: Multi-n Search ===")
    phase3_deadline = min(total_deadline, time.monotonic() + 500.0)
    n_candidates = [300, 400, 600, 700, 800]

    for n_try in n_candidates:
        if time.monotonic() > phase3_deadline:
            break
        if n_try == n_best:
            continue
        print(f"\n  --- n={n_try} ---")
        a_init = _interpolate_sequence(best_a, n_try)

        # L-BFGS on this size
        if time.monotonic() < phase3_deadline - 30:
            a_opt, sc = _lbfgs_smooth(a_init, T_start=0.3, T_end=0.02,
                                       max_iter=300, deadline=phase3_deadline, top_k=20)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  ** New best at n={n_try}: {best_score:.6f}")

        # LP on this size (only if n is manageable)
        if n_try <= 800 and time.monotonic() < phase3_deadline - 30:
            a_opt, sc = _iterative_lp(a_init, max_iters=150, deadline=phase3_deadline,
                                       initial_blend=0.6, top_k_lags=10, line_search=True)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  ** New best at n={n_try}: {best_score:.6f}")

    # ================================================================
    # Phase 4: Diverse initializations at best n
    # ================================================================
    print(f"\n=== Phase 4: Diverse Inits (n around {len(best_a)}) ===")
    phase4_deadline = min(total_deadline, time.monotonic() + 300.0)
    n_cur = len(best_a)

    for v in [5, 10, 11, 12, 13, 6]:
        if time.monotonic() > phase4_deadline:
            break
        a_init = _make_initial(n_cur, v, rng)
        a_opt, sc = _lbfgs_smooth(a_init, T_start=0.5, T_end=0.02,
                                   max_iter=200, deadline=phase4_deadline, top_k=20)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  v{v}: ** New best: {best_score:.6f}")
        else:
            print(f"  v{v}: score={sc:.6f} (no improvement)")

    # ================================================================
    # Phase 5: Random perturbations + re-optimize
    # ================================================================
    print(f"\n=== Phase 5: Perturbations ===")
    phase5_deadline = min(total_deadline, time.monotonic() + 150.0)

    for trial in range(8):
        if time.monotonic() > phase5_deadline:
            break
        a_pert, sc = _perturb_and_reoptimize(best_a, rng, n_perturb=3,
                                             deadline=phase5_deadline, max_lbfgs_iter=150)
        if sc < best_score:
            best_score = sc
            best_a = a_pert.copy()
            print(f"  Perturb {trial}: ** New best: {best_score:.6f}")
        else:
            print(f"  Perturb {trial}: score={sc:.6f}")

    # ================================================================
    # Phase 6: Final LP polish
    # ================================================================
    print(f"\n=== Phase 6: Final Polish ===")
    if time.monotonic() < total_deadline - 30:
        n_cur = len(best_a)
        # If n is too large for LP, downsample
        n_lp = min(n_cur, 600)
        a_ds = _interpolate_sequence(best_a, n_lp) if n_cur != n_lp else best_a.copy()

        a_opt, sc = _iterative_lp(a_ds, max_iters=100, deadline=total_deadline - 10,
                                   initial_blend=0.5, top_k_lags=10, line_search=True)
        if n_cur != n_lp:
            a_opt = _interpolate_sequence(a_opt, n_cur)
            sc = _score(a_opt)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Polish: ** New best: {best_score:.6f}")

    # ================================================================
    # Final verification
    # ================================================================
    print(f"\n=== Final: score={best_score:.6f}, n={len(best_a)} ===")
    result = [float(x) for x in best_a]
    verified = float(evaluate_sequence(result))
    print(f"Verified via evaluate_sequence: {verified:.6f}")

    return result
