import numpy as np

from scipy.optimize import linprog

def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    """Build convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C

def _lp_minM(f, deadline=None):
    """Solve min M s.t. C_f @ g <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(f)
    C = _build_conv_matrix(f)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': max(1.0, deadline - time.monotonic()) if deadline else None})
        if result.status == 0:
            return result.x[:n].copy(), float(result.x[-1])
    except Exception:
        pass
    return None, None

        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    best_a = a.copy()

    best_M = float(np.max(_fft_convolve(a)))
    blend = blend_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g, M_lp = _lp_minM(a)
        if g is None:
            break
        
        a_new = np.maximum((1.0 - blend) * a + blend * g, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            break
        a_new /= s_new
        
        c_new = _fft_convolve(a_new)
        M_new = float(np.max(c_new))
        score = 2.0 * n * M_new

        a = a_new

    return best_a, best_score

def _subgradient(a_init, deadline, lr_init=0.1, lr_decay=0.99995, n_peaks=20):
    """Subgradient descent for flat autocorrelation."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(_fft_convolve(a)))
    lr = lr_init
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Top K peaks
        if n_peaks >= len(c):
            top_k = np.arange(len(c))
        else:
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # Sensitivity
        s = np.zeros(n)
        for k in top_k:
            k = int(k)
            j = np.arange(n)
            idx1 = k - j
            idx2 = j - k
            m1 = (idx1 >= 0) & (idx1 < n)
            m2 = (idx2 >= 0) & (idx2 < n)
            s[m1] += a[idx1[m1]]
            s[m2] += a[idx2[m2]]
        s /= len(top_k)
        
        max_s = float(np.max(s))
        if max_s > 1e-15:
            s = s / max_s
            a = a - lr * s
            a = np.maximum(a, 0.0)
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a = a * S / S_new
        
        lr *= lr_decay
    
    return best_a, best_score

    return out

    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(_fft_convolve(a)))

    T = 0.01  # temperature
    
    for trial in range(n_trials):

        n_perturb = max(1, n // 50)
        idx = rng.integers(0, n, size=n_perturb)

        a_try[idx] += rng.normal(0, scale, size=n_perturb)

            a = a_try
        elif score_try < best_score + T:
            if rng.random() < math.exp(-(score_try - best_score) / T):
                a = a_try

Multi-phase optimizer for minimizing J(a) = 2n * max(autocorr(a)) / (sum(a))^2.
Uses alternating LP (min-M), subgradient descent, and stochastic polish.

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C

def _lp_minM(f):
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.concatenate([C, -np.ones((2 * n - 1, 1))], axis=1)
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    try:
        r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs', options={'presolve': True})
        if r.status == 0:
            return r.x[:n].copy(), float(r.x[-1])
    except Exception:
        pass
    return None, None

def _line_search_blend(a, g, current_score):
    """Find best blend t in (0,1) of a and g."""
    best_t = 0.0
    best_val = current_score
    n = len(a)
    for t in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_try = np.maximum((1 - t) * a + t * g, 0.0)
        s = float(np.sum(a_try))
        if s < 1e-12:
            continue
        a_try = a_try / s
        val = 2.0 * n * float(np.max(_fft_convolve(a_try)))
        if val < best_val:
            best_val = val
            best_t = t
    return best_t, best_val

def _alternating_lp(a_init, max_iters, deadline, blend_init=0.5):
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(_fft_convolve(a)))

        if time.monotonic() > deadline:
            break
        g, M_lp = _lp_minM(a)
        if g is None:
            break
        # Line search
        t_best, score_try = _line_search_blend(a, g, best_score)
        if t_best == 0.0:
            # Try decreasing blend
            t_best = max(0.01, blend_init * (0.97 ** it))
            a_new = np.maximum((1 - t_best) * a + t_best * g, 0.0)

            if s_new < 1e-12:
                break
            a_new /= s_new
            score_try = 2.0 * n * float(np.max(_fft_convolve(a_new)))
        
        a_new = np.maximum((1 - t_best) * a + t_best * g, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            break
        a_new /= s_new

            if it % 10 == 0:
                print(f"    LP {it}: {score_try:.6f} (t={t_best:.3f}, M={M_lp:.6e})")

def _subgradient(a_init, deadline, lr_init=0.1, n_peaks=10):
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(_fft_convolve(a)))
    
    # Adaptive step
    step = lr_init
    patience = 0
    max_patience = 500
    
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                step *= 0.5
                patience = 0
                if step < 1e-6:
                    break
        
        if n_peaks >= len(c):
            top_k = np.arange(len(c))
        else:
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        s = np.zeros(n)
        for k in top_k:
            ki = int(k)
            j = np.arange(n)
            i1 = ki - j
            i2 = j - ki
            m1 = (i1 >= 0) & (i1 < n)
            m2 = (i2 >= 0) & (i2 < n)
            s[m1] += a[i1[m1]]
            s[m2] += a[i2[m2]]
        s /= len(top_k)
        
        ns = float(np.max(s))
        if ns > 1e-15:
            s /= ns
            a = np.maximum(a - step * s, 0.0)
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a = a * S / S_new
    
    return best_a, best_score

        s_try = 2.0 * n * float(np.max(_fft_convolve(a_try)))

            a = a_try

This is O(n) per peak, O(n * n_peaks) total. For n=4096, n_peaks=15: ~61k per iter. Fast in numpy.

Alternatively, I can compute the sensitivity more efficiently:
s[i] = sum_{k in peaks} [a[k-i] (if valid) + a[i-k] (if valid)]
     = sum_{k in peaks} a|max(k-i, i-k)| (with zero extension)

This is like a "sum of shifted a" over the peak set. Can be computed with convolution:
s = convolve(a, indicator_peaks) + convolve(a, indicator_peaks_reversed)

Where indicator_peaks has 1s at the peak positions.

"""
Multi-phase optimizer minimizing J(a) = 2n*max(autocorr(a))/sum(a)^2
for non-negative sequences. Uses alternating LP (min-M), subgradient
descent, and stochastic polish.
"""
import numpy as np
import time
import math
from scipy.optimize import linprog


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    c = np.fft.irfft(A * A, fs)
    return c[:size]


def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    mc = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * mc / (s * s)


def _conv_mat(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_minM(f):
    n = len(f)
    C = _conv_mat(f)
    cv = np.zeros(n + 1)
    cv[-1] = 1.0
    A = np.empty((2 * n - 1, n + 1))
    A[:, :n] = C
    A[:, n] = -1.0
    b = np.zeros(2 * n - 1)
    Ae = np.zeros((1, n + 1))
    Ae[0, :n] = 1.0
    be = np.array([1.0])
    bd = [(0, None)] * (n + 1)
    try:
        r = linprog(cv, A_ub=A, b_ub=b, A_eq=Ae, b_eq=be,
                    bounds=bd, method='highs')
        if r.status == 0:
            return r.x[:n].copy(), float(r.x[-1])
    except Exception:
        pass
    return None, None


def _alt_lp(ai, max_it, dl):
    a = ai.astype(np.float64).copy()
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a /= S
    ba = a.copy()
    bs = 2.0 * n * float(np.max(_fft_convolve(a)))
    no_improve = 0
    for it in range(max_it):
        if time.monotonic() > dl:
            break
        g, ml = _lp_minM(a)
        if g is None:
            break
        best_t = 0.0
        bv = bs
        ba_try = a.copy()
        for t in [0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            at = np.maximum((1 - t) * a + t * g, 0.0)
            st = float(np.sum(at))
            if st < 1e-12:
                continue
            at /= st
            v = 2.0 * n * float(np.max(_fft_convolve(at)))
            if v < bv:
                bv = v
                best_t = t
                ba_try = at.copy()
        if bv < bs - 1e-10:
            bs = bv
            ba = ba_try
            a = ba_try
            no_improve = 0
            if it % 15 == 0:
                print(f"    LP {it}: {bs:.6f} t={best_t:.2f} M={ml:.5e}")
        else:
            no_improve += 1
            if no_improve > 25:
                break
            a = ba_try  # still move, even if score didn't improve
    return ba, bs


def _subgrad(ai, dl, lr0=0.1, npk=10):
    a = ai.astype(np.float64).copy()
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a /= S
    ba = a.copy()
    bs = 2.0 * n * float(np.max(_fft_convolve(a)))
    lr = lr0
    pat = 0
    j = np.arange(n)
    for it in range(500000):
        if time.monotonic() > dl:
            break
        c = _fft_convolve(a)
        mc = float(np.max(c))
        sc = 2.0 * n * mc
        if sc < bs - 1e-12:
            bs = sc
            ba = a.copy()
            pat = 0
        else:
            pat += 1
            if pat > 400:
                lr *= 0.5
                pat = 0
                if lr < 1e-8:
                    break
        if npk >= len(c):
            tk = np.arange(len(c))
        else:
            tk = np.argpartition(c, -npk)[-npk:]
        s = np.zeros(n)
        for k in tk:
            ki = int(k)
            i1 = ki - j
            m1 = (i1 >= 0) & (i1 < n)
            if m1.any():
                s[m1] += a[i1[m1]]
            i2 = j - ki
            m2 = (i2 >= 0) & (i2 < n)
            if m2.any():
                s[m2] += a[i2[m2]]
        s /= len(tk)
        ms = float(np.max(s))
        if ms > 1e-15:
            s /= ms
            a = np.maximum(a - lr * s, 0.0)
            sn = float(np.sum(a))
            if sn > 1e-10:
                a *= S / sn
    return ba, bs


def _stoch(ai, dl, nt=5000):
    a = ai.astype(np.float64).copy()
    n = len(a)
    S = float(np.sum(a))
    a /= S
    ba = a.copy()
    bs = 2.0 * n * float(np.max(_fft_convolve(a)))
    rng = np.random.default_rng(77)
    T = 0.03
    for tr in range(nt):
        if time.monotonic() > dl:
            break
        at = a.copy()
        ne = max(1, n // 40)
        ix = rng.integers(0, n, ne)
        sc = 0.04 * float(np.std(at))
        at[ix] += rng.normal(0, sc, ne)
        at = np.maximum(at, 0)
        st = float(np.sum(at))
        if st < 1e-10:
            continue
        at /= st
        sv = 2.0 * n * float(np.max(_fft_convolve(at)))
        if sv < bs:
            bs = sv
            ba = at.copy()
            a = at
        elif sv < bs + T:
            if rng.random() < math.exp(-(sv - bs) / max(T, 1e-9)):
                a = at
        T = max(0.001, T * 0.9995)
    return ba, bs

def _interp(src, nt):
    if len(src) == nt:
        return src.copy()
    out = np.interp(np.linspace(0, 1, nt), np.linspace(0, 1, len(src)), src)
    out = np.maximum(out, 0)
    s = float(np.sum(out))
    return out / s if s > 1e-10 else np.ones(nt) / nt
