- Coordinate descent (1D min over each element)
- Simulated annealing + periodic restarts
- Multi-length exploration (parent, 2048, 4096, 8192)
"""


def fft_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n <= 1:
        return np.array([a[0] * a[0]] if n == 1 else [0.0])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_score(a):
    """Fast score via FFT (approximate)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * c.max() / (S * S)


def normalize(a):
    """Clamp to nonneg, scale to sum = len(a)."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a), dtype=np.float64)
    return a * (len(a) / S)


def subgrad_max(a, c, peaks):
    """Subgradient of max(c) w.r.t. a at the given peak indices."""
    n = len(a)
    g = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            g[lo:hi + 1] += a[k - np.arange(lo, hi + 1)]
    return 2.0 * g


def grad_l2conv(a, c):
    """Gradient of sum(c[k]^2) w.r.t. a. Computed via cross-correlation."""
    h = np.correlate(c, a, mode='valid')
    return 4.0 * h


def spectral_equipower(a, n_steps=3, gamma=0.1):
    """Iteratively flatten |FFT(a)|^2 to reduce convolution variance."""
    n = len(a)
    a = a.copy()
    N = 1
    while N < 2 * n:
        N *= 2
    for _ in range(n_steps):
        fa = np.fft.rfft(a, N)
        pw = fa.real ** 2 + fa.imag ** 2
        mp = pw.mean()
        if mp < 1e-30:
            break
        new_pw = pw * np.power(mp / (pw + 1e-30), gamma)
        phase = np.angle(fa)
        fa_new = np.sqrt(new_pw) * (np.cos(phase) + 1j * np.sin(phase))
        a_new = np.fft.irfft(fa_new, N)[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a * (n / S)
    return a


def one_dim_descent(a, c, j):
    """Find optimal a[j] given all other elements fixed, via 1D minimization."""
    n = len(a)
    a_other_sum = a.sum() - a[j]
    c_other = np.zeros(2 * n - 1)
    for i in range(n):
        if i == j:
            continue
        c_other[i:i + n] += a[i]
        c_other[n - 1 - (i):n] += a[i]
    # c_total[x] = c_other[x] + a[j] * (a[j-x] + a[j+x] contributions)
    # Actually: c_k = sum_i a[i]*a[k-i]. With a[j] variable:
    # c_k = c_other_k + a[j]*(contrib from a[j] in both slots)
    # If j != k-j: c_k gets a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] (if k!=2j)
    # If j == k-j (k=2j): c_k gets a[j]^2
    # So c_k = c_other_k + 2*a[j]*a[k-j]*[k!=2j] + a[j]^2*[k==2j]
    # For the optimization, for each k: c_k(x) = c_other_k + 2*x*a[k-j] + x^2*[k==2j]
    # But c_other already has the contributions from a[j] removed.
    # Wait, I need to be more careful.
    # c_k = sum_i a[i]*a[k-i]. The terms involving a[j] are:
    #   i=j: a[j]*a[k-j]
    #   k-i=j => i=k-j: a[k-j]*a[j]
    # If j != k-j: two terms, each a[j]*a[k-j], total 2*a[j]*a[k-j]
    # If j == k-j (k=2j): one term a[j]^2
    # So c_k = (c_k without a[j] terms) + 2*a[j]*a[k-j] (k != 2j) or a[j]^2 (k=2j)
    # "c_k without" = sum_{i != j, k-i != j} a[i]*a[k-i]
    
    # For 1D minimization of 2n*max_k c_k(x)/(a_other_sum + x)^2:
    # The active peak is k* = argmax c_k(a). We optimize for that peak.
    k_star = int(np.argmax(c))
    
    # Determine the quadratic/linear form for c_{k*}(x)
    if k_star == 2 * j and 0 <= k_star <= 2 * n - 2:
        # c_{k*}(x) = c_other_{k*} + x^2
        # where c_other_{k*} = sum_{i != j} a[i]*a[k*-i] (excluding the j term)
        c_ok = c[k_star] - a[j] ** 2
        # c_{k*}(x) = c_ok + x^2
        alpha = 1.0
        beta_coeff = 0.0
        gamma_coeff = c_ok
    else:
        # c_{k*}(x) = c_ok + 2*x*a[k*-j] (if 0<=k*-j<n) or c_ok + 2*x*a[j-k*] (if 0<=j-k*<n)
        coeff = 0.0
        if 0 <= k_star - j < n:
            coeff += a[k_star - j]
        if 0 <= j - k_star < n:
            coeff += a[j - k_star]
        c_ok = c[k_star] - 2.0 * a[j] * (coeff / 2.0) if coeff != 0 else c[k_star]
        # Actually c_k = c without j terms + 2*a[j]*a[k-j]
        # So c without = c_k - 2*a[j]*a[k-j] (if k!=2j)
        c_ok = c[k_star] - 2.0 * a[j] * (coeff / 2.0) if coeff != 0 else c[k_star]
        # Simpler: c_ok = c[k_star] - a[j]*coeff (since the j contribution is a[j]*coeff)
        c_ok = c[k_star] - a[j] * coeff
        alpha = 0.0
        beta_coeff = coeff  # c_k(x) = c_ok + beta_coeff * x
        gamma_coeff = c_ok
    
    S0 = a_other_sum
    
    def obj(x):
        if x < 0:
            x = 0.0
        Sx = S0 + x
        if Sx < 0.01:
            return 1e10
        ck = gamma_coeff
        if alpha > 0:
            ck += alpha * x * x
        ck += beta_coeff * x
        # Also need to check if another peak becomes max (approximation: just use this peak)
        return 2.0 * n * max(ck, 0) / (Sx * Sx)
    
    res = minimize_scalar(obj, bounds=(0, 10 * (a.sum() / n)), method='bounded',
                          options={'xatol': 1e-10})
    a_new = a.copy()
    x_opt = max(0.0, res.x)
    a_new[j] = x_opt
    return a_new, res.fun


def propose_candidate(seed=42, budget_s=2280, **kwargs):
    """
    Hybrid optimizer: subgradient + spectral equalization + coordinate descent + annealing.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(1024, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(1024, dtype=np.float64)

    best_a = parent.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(best_score):
        best_a = np.ones(1024)
        best_score = 2.0
    n_best = len(best_a)
    print(f"Start: n={n_best}, score={best_score:.8f}")

    # === Phase 0: Diversified initialization ===
    lengths_to_try = sorted(set([n_best, 2048, 4096, 8192]))
    lengths_to_try = [l for l in lengths_to_try if 64 <= l <= 16384]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 2000:
            break
        # Random phase init
        for rep in range(2):
            Np = 1
            while Np < 2 * n_try:
                Np *= 2
            ph = rng.uniform(0, 2 * np.pi, Np // 2 + 1)
            fa = np.exp(1j * ph)
            a_try = np.fft.irfft(fa, Np)[:n_try]
            a_try = np.maximum(a_try, 0)
            S = a_try.sum()
            if S < 0.01:
                continue
            a_try = a_try * (n_try / S)
            sc = fast_score(a_try)
            if sc < best_score * 1.05:
                exact = float(evaluate_sequence(a_try.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_a = a_try.copy()
                    n_best = n_try
                    print(f"  Init n={n_try}: {best_score:.8f}")

        # Spectrally-flat init (equi-magnitude FFT)
        Np = 1
        while Np < 2 * n_try:
            Np *= 2
        for attempts in range(3):
            a_try = np.ones(n_try) / np.sqrt(n_try)
            for _ in range(10 + attempts * 5):
                fa = np.fft.rfft(a_try, Np)
                pw = np.abs(fa) ** 2
                mp = pw.mean()
                target = np.full_like(pw, mp)
                blend = 0.5
                new_pw = (1 - blend) * pw + blend * target
                phase = np.angle(fa)
                fa_new = np.sqrt(new_pw) * (np.cos(phase) + 1j * np.sin(phase))
                a_new = np.fft.irfft(fa_new, Np)[:n_try]
                a_try = np.maximum(a_new, 0)
                S = a_try.sum()
                if S < 0.01:
                    break
                a_try = a_try * (n_try / S)
            sc = fast_score(a_try)
            if sc < best_score * 1.05:
                exact = float(evaluate_sequence(a_try.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_a = a_try.copy()
                    n_best = n_try
                    print(f"  Flat-spectrum n={n_try}: {best_score:.8f}")

    # === Phase 1: Main subgradient + spectral optimization ===
    a = best_a.copy()
    n = len(a)
    lr = 3.0 / np.sqrt(n)
    beta = 0.85
    mom = np.zeros(n)
    temp = 0.08
    lam = 0.15
    it = 0
    no_imp = 0
    coord_cd_interval = 50  # do coordinate descent every this many iterations

    while time.monotonic() < deadline - 150:
        a = normalize(a)
        c = fft_conv(a)
        M = c.max()

        # Active peaks (within 3% of max)
        thresh = M * 0.97
        pk = np.where(c >= thresh)[0]
        if len(pk) == 0:
            pk = np.array([np.argmax(c)])
        if len(pk) > 10:
            pk = pk[np.argsort(-c[pk])][:10]

        # Subgradient of max
        g_max = subgrad_max(a, c, pk.tolist())

        # Spectral L2 gradient
        g_spec = grad_l2conv(a, c)

        # Normalize for blending
        gn_m = np.linalg.norm(g_max)
        gn_s = np.linalg.norm(g_spec)
        if gn_m > 1e-12:
            g_max = g_max / gn_m
        if gn_s > 1e-12:
            g_spec = g_spec / gn_s

        g = (1.0 - lam) * g_max + lam * g_spec
        g_norm = np.linalg.norm(g)

        if g_norm > 1e-14:
            mom = beta * mom - lr * g
            a = a + mom
            a = np.maximum(a, 0.0)
            a = normalize(a)

        it += 1

        # Coordinate descent step
        if it % coord_cd_interval == 0:
            # Pick top 8 random indices + the most important ones
            a = normalize(a)
            c = fft_conv(a)
            # Do CD on a few coordinates
            for _ in range(6):
                j = int(rng.integers(0, n))
                a, _ = one_dim_descent(a, c, j)
            a = normalize(a)

        # Spectral equalization
        if it % 800 == 0:
            a = spectral_equipower(a, n_steps=3, gamma=0.08)

        # Annealing
        if temp > 0.002:
            a += temp * rng.standard_normal(n) / np.sqrt(n)
            a = np.maximum(a, 0.0)
            a = normalize(a)
            temp *= 0.9992

        # Exact evaluation
        if it % 30 == 0:
            sc_ff = fast_score(a)
            exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_a = a.copy()
                n_best = n
                no_imp = 0
                if it % 300 == 0:
                    print(f"  it={it}, n={n}: {best_score:.8f}")
            else:
                no_imp += 1

            if no_imp >= 20:
                lr *= 0.55
                temp *= 0.5
                lam = min(lam * 1.4, 0.6)
                no_imp = 0
                mom *= 0.1
                if lr < 1e-6:
                    a = best_a * np.exp(rng.normal(0, 0.12, n))
                    a = normalize(a)
                    a = spectral_equipower(a, n_steps=5, gamma=0.12)
                    lr = 3.0 / np.sqrt(n)
                    temp = 0.04
                    lam = 0.1
                    mom = np.zeros(n)

    print(f"Phase 1: score={best_score:.8f}, n={n_best}")

    # === Phase 2: Fine-tuning (single peak, small steps) ===
    a = best_a.copy()
    n = len(a)
    lr = 1.0 / np.sqrt(n)
    mom = np.zeros(n)
    it2 = 0
    no_imp2 = 0

    while time.monotonic() < deadline - 60:
        a = normalize(a)
        c = fft_conv(a)
        k_star = int(np.argmax(c))
        g = subgrad_max(a, c, [k_star])
        gn = np.linalg.norm(g)
        if gn > 1e-14:
            g /= gn
            mom = 0.88 * mom - lr * g
            a = a + mom
            a = np.maximum(a, 0.0)
            a = normalize(a)
        it2 += 1

        # Coordinate descent every 20 iters
        if it2 % 20 == 0:
            c = fft_conv(a)
            for _ in range(4):
                j = int(rng.integers(0, n))
                a, _ = one_dim_descent(a, c, j)
            a = normalize(a)

        if it2 % 25 == 0:
            exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_a = a.copy()
                no_imp2 = 0
                if it2 % 250 == 0:
                    print(f"  Fine it={it2}: {best_score:.8f}")
            else:
                no_imp2 += 1
                if no_imp2 >= 40:
                    lr *= 0.6
                    mom *= 0.2
                    no_imp2 = 0

    print(f"Phase 2: score={best_score:.8f}")

    # === Phase 3: Coordinate descent polish ===
    a = best_a.copy()
    n = len(a)
    it3 = 0
    no_imp3 = 0
    while time.monotonic() < deadline - 20:
        a = normalize(a)
        c = fft_conv(a)
        # CD on random + important coordinates
        for _ in range(10):
            if it3 % 3 == 0:
                # Pick coordinate with largest gradient contribution
                k_star = int(np.argmax(c))
                lo = max(0, k_star - n + 1)
                hi = min(n - 1, k_star)
                j = int(rng.integers(lo, hi + 1))
            else:
                j = int(rng.integers(0, n))
            a, _ = one_dim_descent(a, c, j)
            c = fft_conv(a)
        a = normalize(a)
        it3 += 1

        if it3 % 10 == 0:
            exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_a = a.copy()
                no_imp3 = 0
            else:
                no_imp3 += 1
                if no_imp3 >= 30:
                    break

    print(f"Phase 3: score={best_score:.8f}")

    # === Phase 4: Final spectral equalization + re-optimize ===
    if time.monotonic() < deadline - 10:
        a = best_a.copy()
        a = spectral_equipower(a, n_steps=8, gamma=0.06)
        exact = float(evaluate_sequence(a.tolist()))
        if np.isfinite(exact) and exact < best_score:
            best_score = exact
            best_a = a.copy()
            # Quick re-optimize
            n = len(a)
            lr = 2.0 / np.sqrt(n)
            mom = np.zeros(n)
            for _ in range(300):
                if time.monotonic() > deadline - 5:
                    break
                a = normalize(a)
                c = fft_conv(a)
                pk = np.where(c >= c.max() * 0.98)[0][:6]
                g = subgrad_max(a, c, pk.tolist())
                gn = np.linalg.norm(g)
                if gn > 1e-14:
                    g /= gn
                    mom = 0.85 * mom - lr * g
                    a = np.maximum(a + mom, 0.0)
                    a = normalize(a)
            a = normalize(a)
            exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_a = a.copy()

    # Final validation
    best_a = np.clip(best_a, 0.0, 1000.0)
    final = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final):
        best_score = final
    print(f"FINAL: score={best_score:.8f}, n={len(best_a)}")
    return [float(x) for x in best_a]
