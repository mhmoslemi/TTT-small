import numpy as np

def _fft_convolve(a):
    """Fast full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _sensitivity(a, j):
    """Compute d(conv(a,a))[k]/da[j] for all k."""
    n = len(a)
    m = 2 * n - 1
    s = np.zeros(m)
    k_start = j
    k_end = min(m, j + n)
    s[k_start:k_end] = 2.0 * a[:k_end - j]
    return s

def _optimize_coord(a, j, c):
    """Optimize a[j] to minimize max(c). Returns new value of a[j]."""
    current = a[j]
    s = _sensitivity(a, j)
    
    if current < 1e-15:
        lower = 0.0
    else:
        lower = -current

    if S < 1e-10:
        a[:] = 1.0 / n
        S = 1.0
    a /= S
    
    c = _fft_convolve(a)
    best_max = float(np.max(c))
    best_a = a.copy()
    no_improve = 0
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            new_val = _optimize_coord(a, j, c)
            delta = new_val - a[j]
            if abs(delta) > 1e-14:
                a[j] = new_val
                s = _sensitivity(a, j)  # Use new a for sensitivity
                # Actually, we should use the OLD sensitivity for the update
                # Let me recompute: the update is c += s_old * delta
                # But s_old = 2 * a_old[k-j], and we've already updated a[j]
                # So s_old[k] = 2 * a_old[k-j] = 2 * a_new[k-j] (since k-j ≠ j for k ≠ 2j)
                # Only at k=2j is there a difference: s_old[2j] = 2*a[j]_old, s_new[2j] = 2*a[j]_new
                # The error is 2*delta at k=2j, plus we need the delta^2 term.
                # For simplicity, let's just use s_old (computed before the update).
                pass
        
        # Recompute c from scratch every pass for accuracy
        c = _fft_convolve(a)
        
        # Renormalize
        S = float(np.sum(a))
        if S > 1e-10:
            a /= S
            c = _fft_convolve(a)
        
        current_max = float(np.max(c))
        if current_max < best_max - 1e-12:
            best_max = current_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and p % 100 == 0:
            print(f"      Pass {p}: max={current_max:.8f}, best={best_max:.8f}")
