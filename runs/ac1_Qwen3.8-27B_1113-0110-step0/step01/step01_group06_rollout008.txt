"""
Optimize non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam optimizer with FFT-based gradients, top-K peak focusing, diverse
initializations across multiple sequence lengths, and intensive refinement.

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_conv_self(a):
    n = a.size
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _score_from_b(a, b):
    s = float(a.sum())
    if s <= 1e-300:
        return float('inf'), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _grad_fft(weights, a):
    """Compute grad_j = 2 * sum_k w_k * a_{k-j} approximately."""
    n = a.size
    L = 2 * n - 1
    N = _pow2(3 * n)
    a_flip = a[::-1]
    w_pad = np.zeros(N)
    w_pad[:L] = weights
    V = np.fft.rfft(a_flip, n=N)
    W = np.fft.rfft(w_pad, n=N)
    h = np.fft.irfft(V * W, n=N).real
    g = h[n - 1: n - 1 + n]
    return 2.0 * g


def _init_anti_palin(n, rng, amp=3.0):
    m = n // 2
    v = rng.normal(0, 1, m)
    v -= v.mean()
    std = v.std()
    if std > 1e-12:
        v = amp * v / std
    z = np.zeros(n)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0
    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _init_cosine_rem(n, rng, amp=2.0):
    x = np.arange(n, dtype=np.float64)
    base = 0.5 * (1 + np.cos(2.0 * np.pi * (x - n / 2) / n))
    noise = rng.normal(0, 1, n) * amp * 0.3
    z = np.log(base + 1e-6) + noise
    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _init_smooth_random(n, rng):
    a = rng.uniform(0.3, 1.7, n)
    for _ in range(3):
        a = 0.5 * a + 0.25 * np.roll(a, 1) + 0.25 * np.roll(a, -1)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _adam_optimize(a0, time_limit, rng, top_k=8, lr_init=0.1, verbose=False, label=""):
    """Adam optimizer with top-K peak focusing for minimizing 2n*max_b (sum=1)."""
    n = a0.size
    a = a0.copy()
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = _fft_conv_self(a)
    score = 2.0 * n * float(b.max())

    best_score = score
    best_a = a.copy()

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0

    beta1, beta2, eps = 0.85, 0.999, 1e-8

    no_improve = 0
    max_no = 300
    resets = 0
    max_resets = 30
    t0 = time.monotonic()

    # Adaptive top-K
    cur_k = top_k
    k_min = 3
    k_max = min(50, n // 2)

    while time.monotonic() - t0 < time_limit:
        t += 1
        if b is None:
            b = _fft_conv_self(a)
            score = 2.0 * n * float(b.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.4)
            # Perturb best
            scale = rng.uniform(0.1, 0.5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = None
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            t = 0
            cur_k = rng.integers(k_min, min(k_max, 20))
            continue

        # Compute top-K weights
        k = min(cur_k, b.size)
        L = b.size
        # Use softmax on the top peaks
        cutoff = np.sort(b)[-k]
        mask = b >= cutoff * 0.5  # include peaks near the top
        if mask.sum() < k:
            mask = b >= np.sort(b)[-(k + int(mask.sum()) + 1)]
        top_max = b.max()
        with np.errstate(over='ignore'):
            beta_f = 100.0
            w_full = np.zeros(L)
            w_full[mask] = np.exp(-beta_f * (top_max - b[mask]) / max(top_max, 1e-30))
        sw = float(w_full.sum())
        if sw > 1e-12:
            w_full /= sw
        else:
            w_full = np.zeros(L)
            w_full[int(np.argmax(b))] = 1.0

        # Compute gradient
        grad_raw = _grad_fft(w_full, a)
        # Remove mean to preserve sum constraint
        grad = grad_raw - float(np.mean(grad_raw))

        # Adaptive learning rate schedule
        progress = min(1.0, t / max(1, int(time_limit / lr_init)))
        lr = lr_init * (0.1 + 0.9 * (1.0 - progress)) * (0.5 * (1 + np.cos(np.pi * min(1.0, t / max(1, int(time_limit * 0.8 / lr_init))))))
        lr = max(lr, lr_init * 0.005)

        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad * grad
        bc1 = 1 - beta1 ** t
        bc2 = 1 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2

        step = lr * m_c / (np.sqrt(v_c) + eps)
        a_new = a + step
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        b_new = _fft_conv_self(a_new)
        score_new = 2.0 * n * float(b_new.max())

        # Acceptance
        w_peak_old = float(np.dot(w_full, b))
        w_peak_new = float(np.dot(w_full, b_new))

        if score_new < best_score - 1e-15:
            a, b, score = a_new, b_new, score_new
        elif score_new < score - 1e-15:
            a, b, score = a_new, b_new, score_new
        elif w_peak_new < w_peak_old - 1e-10:
            a, b, score = a_new, b_new, score_new
        else:
            # Try half step
            a_half = a + 0.5 * step
            a_half = np.maximum(a_half, 1e-15)
            a_half /= a_half.sum()
            b_half = _fft_conv_self(a_half)
            score_half = 2.0 * n * float(b_half.max())
            if score_half < score:
                a, b, score = a_half, b_half, score_half

        # Adapt top-K
        if score < best_score:
            cur_k = max(k_min, int(cur_k * 0.9))
        else:
            cur_k = min(k_max, int(cur_k * 1.1) + 1)

        if verbose and t % 200 == 0:
            print(f"[{label}] t={t} score={score:.6f} best={best_score:.6f} k={cur_k} resets={resets}")

    return best_a, float(best_score)


def _targeted_refine(a0, time_limit, rng, verbose=False, label=""):
    """Intensive local refinement: focus on reducing individual peaks."""
    n = a0.size
    a = a0.copy()
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = _fft_conv_self(a)
    score = 2.0 * n * float(b.max())
    best_score = score
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if b is None:
            b = _fft_conv_self(a)
            score = 2.0 * n * float(b.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Find top peaks
        top_idx = np.argsort(b)[-5:]
        top_max = b[top_idx[0]]

        # For each top peak, find which a_i contribute most
        # (a*a)_k = sum_i a_i * a_{k-i}
        # For peak at index k: d(a*a)_k / da_j = 2*a_{k-j} (approx)
        # So a_j contributes proportional to a_{k-j}

        # Try reducing the a_j that contribute most to the top peak
        # and increasing a_j that contribute least

        for pk in top_idx[:3]:
            pk = int(pk)
            # Contribution of a_j to b_pk: proportional to a_{pk-j}
            contrib = np.zeros(n)
            valid_j = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            for j in valid_j:
                k_val = pk - int(j)
                if 0 <= k_val < n:
                    contrib[int(j)] = a[k_val]

            if contrib.max() < 1e-20:
                continue

            # Find a_j with highest contribution and lowest contribution
            # Reduce high-contrib CI (indices where a_i is high AND contributes to peak)
            # Increase low-contrib CI

            # Simple: move mass from high contribution to low contribution
            thresh_high = np.percentile(contrib, 90)
            thresh_low = np.percentile(contrib, 10)

            high_mask = contrib > thresh_high
            low_mask = contrib < thresh_low and contrib > 0

            if high_mask.sum() < 1 or low_mask.sum() < 1:
                continue

            # Move a small amount
            amt = 0.005 * a[high_mask].sum()
            a_old = a.copy()
            a[high_mask] -= amt * a[high_mask] / a[high_mask].sum()
            a[low_mask] += amt * (a[low_mask] + 1e-15) / (a[low_mask] + 1e-15).sum()
            a = np.maximum(a, 1e-15)
            a /= a.sum()

            b_check = _fft_conv_self(a)
            s_check = 2.0 * n * float(b_check.max())

            if s_check < score:
                score = s_check
                b = b_check
            else:
                a = a_old

        # Also try pure gradient step every few iterations
        if it % 5 == 0:
            top_max2 = float(b.max())
            with np.errstate(over='ignore'):
                w2 = np.exp(-50.0 * (top_max2 - b) / max(top_max2, 1e-30))
            sw2 = float(w2.sum())
            if sw2 > 1e-12:
                w2 /= sw2
            grad2 = _grad_fft(w2, a)
            grad2 -= grad2.mean()
            a_try = a - 0.02 * grad2 / max(1e-20, np.abs(grad2).max())
            a_try = np.maximum(a_try, 1e-15)
            a_try /= a_try.sum()
            b_try = _fft_conv_self(a_try)
            s_try = 2.0 * n * float(b_try.max())
            if s_try < score - 1e-15:
                a, b, score = a_try, b_try, s_try

        if verbose and it % 50 == 0:
            print(f"[{label}] refine it={it} score={score:.6f} best={best_score:.6f}")

    return best_a, float(best_score)


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float('inf')
    try:
        return float(ev(seq))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: optimize non-negative sequence to minimize 2n*max_b/s^2.
    Strategy: diverse initializations + Adam optimization + targeted refinement.
    """
    seed = int(seed)
    total = max(60.0, float(budget_s))
    now0 = time.monotonic()
    final_deadline = now0 + total - 20.0

    print("Starting convolution optimizer v2")
    best_a = None
    best_score = float('inf')

    # Load given sequence
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                b_gn = _fft_conv_self(gn)
                sc_gn = 2.0 * gn.size * float(b_gn.max())
                if np.isfinite(sc_gn) and sc_gn < best_score:
                    best_score = sc_gn
                    best_a = gn.copy()
                given_a = gn
                print(f"Given: n={gn.size}, score={sc_gn:.6f}")
    except Exception as e:
        print(f"Error loading given: {e}")
        given_a = None

    # Define tasks
    main_rng = np.random.default_rng(seed)

    tasks = []

    if given_a is not None and given_a.size <= 12000:
        tasks.append((given_a, "given", 360.0))

    # Various initializations at different sizes
    for n in [8192, 16384]:
        for init_type in ['anti', 'cos', 'rand']:
            rng_i = np.random.default_rng((seed * 7919 + n * 31 + init_type) % (2**32))
            if init_type == 'anti':
                a_init = _init_anti_palin(n, rng_i, amp=3.0)
            elif init_type == 'cos':
                a_init = _init_cosine_rem(n, rng_i, amp=2.0)
            else:
                a_init = _init_smooth_random(n, rng_i)
            tasks.append((a_init, f"{init_type}_{n}", 180.0))

    # Run tasks
    task_deadline = now0 + max(60.0, total - 200.0)

    for idx, (a_init, label, alloc) in enumerate(tasks):
        now = time.monotonic()
        if now >= task_deadline:
            break
        alloc = min(alloc, task_deadline - now)
        if alloc < 30.0:
            break

        n = a_init.size
        print(f"Task {idx}: {label} n={n} alloc={alloc:.0f}s best_so_far={best_score:.6f}")
        t0 = time.monotonic()

        try:
            task_rng = np.random.default_rng((seed * 1000033 + idx * 7919) % (2**32))

            if label == "given":
                # Two phases: Adam then targeted
                adam_time = alloc * 0.6
                refine_time = alloc * 0.4
                cand_a, cand_score = _adam_optimize(
                    a_init, adam_time, task_rng, top_k=8, lr_init=0.15,
                    verbose=False, label=label
                )
                if cand_score < best_score and refine_time > 30:
                    cand_a2, cand_score2 = _targeted_refine(
                        cand_a, refine_time, task_rng,
                        verbose=(idx == 0), label=label
                    )
                    if cand_score2 < cand_score:
                        cand_a, cand_score = cand_a2, cand_score2
            else:
                cand_a, cand_score = _adam_optimize(
                    a_init, alloc, task_rng, top_k=8, lr_init=0.1,
                    verbose=False, label=label
                )

            if cand_a is not None:
                cand_a = np.nan_to_num(cand_a, nan=0.0, posinf=0.0, neginf=0.0)
                cand_a = np.clip(cand_a, 0.0, 1000.0)
                cc = float(cand_a.sum())
                if cc > 0.01:
                    cand_a /= cc
                    # Verify with exact score
                    b_exact = _fft_conv_self(cand_a)
                    internal = 2.0 * cand_a.size * float(b_exact.max())
                    if np.isfinite(internal) and internal < cand_score:
                        cand_score = internal
                    if np.isfinite(cand_score) and cand_score < best_score:
                        best_score = cand_score
                        best_a = cand_a.copy()
                        print(f"  New best: {cand_score:.6f} (n={cand_a.size})")

        except Exception as exc:
            print(f"  Task failed: {repr(exc)}")

        print(f"  Done {label}: {time.monotonic()-t0:.1f}s, score={cand_score:.6f}")

    # Final intensive refinement
    if best_a is not None:
        remaining = final_deadline - time.monotonic()
        if remaining > 60:
            refine_time = min(remaining - 20, 150.0)
            if refine_time > 20:
                print(f"Final refinement: {refine_time:.0f}s on n={best_a.size}")
                task_rng = np.random.default_rng((seed + 12345) % (2**32))
                ra, rs = _targeted_refine(
                    best_a, refine_time, task_rng,
                    verbose=False, label="final"
                )
                if rs < best_score:
                    best_a, best_score = ra, rs
                    print(f"  After refine: {best_score:.6f}")

                # One more Adam pass with small lr
                remaining2 = final_deadline - time.monotonic()
                if remaining2 > 40:
                    rt2 = min(remaining2 - 15, 80.0)
                    if rt2 > 15:
                        task_rng2 = np.random.default_rng((seed + 99999) % (2**32))
                        ra2, rs2 = _adam_optimize(
                            best_a, rt2, task_rng2, top_k=5, lr_init=0.03,
                            verbose=False, label="final2"
                        )
                        if rs2 < best_score:
                            best_a, best_score = ra2, rs2
                            print(f"  After final Adam: {best_score:.6f}")

    # Fallback
    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        best_score = 2.0

    # Final cleanup
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]
    print(f"FINAL: n={best_a.size}, internal_score={best_score:.6f}")

    # Verify with exact evaluator if possible
    if best_a.size <= 12000:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"Exact evaluation: {exact:.6f}")
        else:
            print("Exact evaluation: unavailable/inf")
    else:
        print(f"Sequence too large for exact evaluation (n={best_a.size})")

    return final_seq
