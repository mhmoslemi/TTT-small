def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _nesterov_subgrad(a_init, max_iters, deadline, n_lags_schedule=None, lr_init=0.1):
    """Nesterov accelerated subgradient method."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)  # velocity
    
    lr = lr_init
    mu = 1.0  # momentum
    
    if n_lags_schedule is None:
        n_lags_schedule = lambda it: min(1 + it // 500, 20)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov step
        t_k = (1.0 + math.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        y = a + (mu / t_k) * (a - y)
        mu = t_k
        
        # Compute subgradient at y
        c = _fft_convolve(y)
        max_c = float(np.max(c))
        n_act = n_lags_schedule(it)
        
        if n_act >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_act)[-n_act:]
        
        grad = np.zeros(n)
        for k in top_idx:
            idx = int(k) - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            grad[valid] += 2.0 * y[idx[valid]]
        grad /= len(top_idx)
        
        # Step size: use a decreasing schedule
        # lr_k = lr_init / (1 + it * 0.001)
        lr_k = lr_init / (1.0 + 0.01 * it)
        
        a_new = y - lr_k * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new
        
        # Track best
        if it % 10 == 0:
            score = _score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    # Final score
    score = _score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def _simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""

    s = float(np.sum(v))

    else:

    return v

        max_c = float(np.max(c))

        threshold = max_c * 0.95
        active = np.where(c >= threshold)[0]
        if len(active) > n_lags:

        grad = np.zeros(n)

            idx = int(k) - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            grad[valid] += 2.0 * a[idx[valid]]
        grad /= float(len(active))
        
        # Scale gradient
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm > 1e-15:

def _fft_convolve(a, b=None):
    """Fast full linear convolution using FFT."""

        b = a

    m = len(b)
    size = n + m - 1

    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    c = np.fft.irfft(A * B, fft_size)
    return c[:size]

def _score(a):

    c = _fft_convolve(a)
    max_c = float(np.max(c))

import numpy as np
import time
import math
from scipy.optimize import linprog


def _fft_convolve(a, b=None):
    """Fast full linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    c = np.fft.irfft(A * B, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _smooth_grad(a, tau):
    """Compute smooth log-sum-exp objective and its gradient."""
    c = _fft_convolve(a)
    c_max = float(np.max(c))
    scaled = tau * (c - c_max)
    # Clip to avoid underflow (values very negative just contribute ~0)
    scaled = np.maximum(scaled, -700.0)
    log_sum_exp = c_max + np.log(np.sum(np.exp(scaled)))
    f = log_sum_exp / tau

    # Softmax weights p[k]
    p = np.exp(scaled)
    p_sum = float(np.sum(p))
    if p_sum < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p /= p_sum

    # Gradient: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    # = (a_rev * p)[j + n - 1]
    n = len(a)
    a_rev = a[::-1]
    cross = _fft_convolve(a_rev, p)
    grad = 2.0 * cross[n - 1:2 * n - 1].copy()

    return f, grad, c, c_max


def _simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        print("No parent available")

    # === Phase 1: Smooth optimization at n=500 (primary) ===
    print("\n=== Phase 1: Smooth gradient n=500 ===")
    n1 = 500
    phase_dl = min(total_deadline, time.monotonic() + 600.0)

    if best_a is not None:
        a_start = _interpolate(best_a, n1)
    else:
        a_start = _diverse_init(n1, 0, rng)

    a_opt, score = _smooth_optimize(a_start, 30000, phase_dl, tau_init=30.0, tau_growth=1.004)
    print(f"  Smooth (tau=30): {score:.6f}")
    if score < best_score:
        best_score = score
        best_a = a_opt.copy()
        print(f"  ** New best: {best_score:.6f}")

    # === Phase 2: LP refinement at n=500 ===
    print("\n=== Phase 2: LP refine n=500 ===")
    phase_dl = min(total_deadline, time.monotonic() + 400.0)
    a_opt, score = _lp_refine(best_a, 60, phase_dl, blend=0.5)
    if score < best_score:
        best_score = score
        best_a = a_opt.copy()
        print(f"  LP: {best_score:.6f}")

    # === Phase 3: Continue smooth with higher tau ===
    print("\n=== Phase 3: Smooth (high tau) n=500 ===")
    phase_dl = min(total_deadline, time.monotonic() + 400.0)
    a_opt, score = _smooth_optimize(a_opt, 20000, phase_dl, tau_init=100.0, tau_growth=1.005)
    if score < best_score:
        best_score = score
        best_a = a_opt.copy()
        print(f"  Smooth (tau=100): {best_score:.6f}")

    # === Phase 4: Try diverse inits at n=500 ===
    print("\n=== Phase 4: Diverse inits n=500 ===")
    remaining = total_deadline - time.monotonic()
    if remaining > 200:
        time_per = min(80.0, remaining / 8.0)
        for v in [0, 1, 3, 5, 7]:
            if time.monotonic() > total_deadline - 200:
                break
            a_v = _diverse_init(n1, v, rng)
            dl = time.monotonic() + time_per
            a_opt, score = _smooth_optimize(a_v, 15000, dl, tau_init=40.0, tau_growth=1.004)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  init v{v}: {best_score:.6f} **")
            else:
                print(f"  init v{v}: {score:.6f}")

    # === Phase 5: Try n=1000 ===
    print("\n=== Phase 5: n=1000 ===")
    if time.monotonic() < total_deadline - 250:
        n2 = 1000
        a_start2 = _interpolate(best_a, n2)
        phase_dl = min(total_deadline, time.monotonic() + 250.0)
        a_opt, score = _smooth_optimize(a_start2, 15000, phase_dl, tau_init=40.0, tau_growth=1.004)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  n=1000: {best_score:.6f}")
        else:
            print(f"  n=1000: {score:.6f}")

    # === Phase 6: Try n=300 with LP (smaller n = better LP conditioning) ===
    print("\n=== Phase 6: n=300 LP ===")
    if time.monotonic() < total_deadline - 200:
        n3 = 300
        a_start3 = _interpolate(best_a, n3)
        phase_dl = min(total_deadline, time.monotonic() + 200.0)
        a_opt, score = _lp_refine(a_start3, 100, phase_dl, blend=0.4)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  LP n=300: {best_score:.6f}")
        else:
            print(f"  LP n=300: {score:.6f}")

    # === Phase 7: Final combined polish ===
    print("\n=== Phase 7: Final polish ===")
    if best_a is not None and time.monotonic() < total_deadline - 50:
        n_best = len(best_a)
        # Smooth at current n
        phase_dl = min(total_deadline, time.monotonic() + 150.0)
        a_opt, score = _smooth_optimize(best_a, 15000, phase_dl, tau_init=150.0, tau_growth=1.006)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  Smooth polish: {best_score:.6f}")

        # LP at reduced n
        if time.monotonic() < total_deadline - 40:
            n_lp = min(n_best, 350)
            a_ds = _interpolate(best_a, n_lp) if n_best != n_lp else best_a.copy()
            a_opt2, score2 = _lp_refine(a_ds, 40, total_deadline - 20, blend=0.3)
            if n_best != n_lp:
                a_opt2 = _interpolate(a_opt2, n_best)
                score2 = _score(a_opt2)
            if score2 < best_score:
                best_score = score2
                best_a = a_opt2.copy()
                print(f"  LP polish: {best_score:.6f}")

    # Verify and return
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a)} ===")
    result = [float(x) for x in best_a]
    verified = float(evaluate_sequence(result))
    print(f"Verified: {verified:.6f}")
    return result
