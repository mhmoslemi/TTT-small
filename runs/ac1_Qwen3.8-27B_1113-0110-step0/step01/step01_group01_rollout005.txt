import time

    import scipy.optimize as _so
except Exception:
    _so = None


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size) / float(a.size)
    return a / s


def softmax_seq(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size))
    return e / s


def convolve_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_fft(a, N)
    return float(2.0 * n * float(np.max(c)))


def make_uniform(n):
    return np.ones(max(1, int(n))) / float(max(1, int(n)))


def make_sine(n, alphas):
    n = max(2, int(n))
    x = (np.arange(n) - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    base = np.ones(n)
    for m, al in enumerate(alphas, 1):
        base += float(al) * np.sin(np.pi * m * x)
    return normalize_positive(base, 1e-12)


def resample_seq(a, n):
    a = normalize_positive(a, 0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == n:
        return a.astype(np.float64, copy=True)
    if old_n == 1:
        return np.full(n, 1.0 / n)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, 1e-12)

def build_basis(n):
    n = max(16, int(n))
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    cols = []
    for m in range(1, 6):
        cols.append(np.sin(np.pi * m * x))
    cols.append(x.copy())
    cols.append(x * x - np.mean(x * x))
    cols.append(x * x * x - np.mean(x * x * x))
    for m in range(1, 5):
        t = 2.0 * np.pi * m * i / n
        cols.append(np.sin(t) - np.mean(np.sin(t)))
        cols.append(np.cos(t) - np.mean(np.cos(t)))
    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms <= 1e-14] = 1.0
    B = B / norms
    return B


def theta_value_obj(theta, n, B, tau, N):
    a = softmax_seq(B @ theta)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    cmax = float(np.max(c))
    if cmax <= 0 or not np.isfinite(cmax):
        return 0.0
    z = np.clip((c - cmax) / max(tau, 1e-12), -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0:
        return cmax
    return 2.0 * n * (cmax + tau * np.log(s))


def grad_from_w(w, fa, N, n):
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1: 2 * n - 1][::-1]

    D = B.shape[1]
    N = next_pow2(3 * n - 2)
    rng = np.random.default_rng(seed)
    starts = []

        z = np.log(np.maximum(initial_a, 1e-12))
        z -= z.mean()
        theta_proj = np.linalg.lstsq(B, z, rcond=None)[0]
        starts.append(theta_proj)
    for std in (0.3, 0.6, 1.0, 1.6):
        starts.append(rng.normal(0, std, D))
    starts = starts[:5]

        if _so is None:

def binary_init_from_smooth(a_smooth, n, m):
    a = np.asarray(a_smooth, dtype=np.float64).ravel()

        a = resample_seq(a, n)
    order = np.argsort(a, kind="stable")
    b = np.zeros(n)
    b[order[-m:]] = 1.0
    return b

        perm2 = rng.permutation(chosen.size)
        b[:] = 0
        b[perm_sorted := np.sort(chosen[perm2][:m])] = 1.0
    return b


def compute_weights(c, cmax, tau, floor=0.02):
    if cmax <= 0 or not np.isfinite(cmax) or tau <= 0:
        return np.full(c.size, 1.0 / c.size)
    z = np.clip((c - cmax) / tau, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0:
        return np.full(c.size, 1.0 / c.size)
    w = ez / s
    if floor > 0:
        w = (1 - floor) * w + floor / c.size
    return w


def choose_idx(g, rng, pick_high, temp=0.1):
    g = np.asarray(g, dtype=np.float64)
    m = g.size
    if m == 0:
        return -1
    if m == 1:
        return 0
    if not np.all(np.isfinite(g)):
        g = np.nan_to_num(g)
    vmin, vmax = float(g.min()), float(g.max())
    rng_g = vmax - vmin
    if rng_g <= 1e-14:
        return int(rng.integers(0, m))
    denom = max(temp, 1e-12) * rng_g
    if pick_high:
        z = (g - vmax) / denom
    else:
        z = (vmin - g) / denom
    z = np.clip(z, -50.0, 0.0)
    p = np.exp(z)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        return int(rng.integers(0, m))
    p /= s
    return int(rng.choice(m, p=p))


def optimize_binary(a0, deadline, seed=0, target_m=None):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    if a0.size == 0:
        return None, np.inf

    m = int(round(float(a0.sum()))) if target_m is None else int(target_m)
    m = max(2, min(n - 2, m))

    # Threshold to binary
    if m >= n:
        b = np.ones(n)
    elif m <= 0:
        b = np.zeros(n)
    else:
        order = np.argsort(a0, kind="stable")
        b = np.zeros(n)
        b[order[-m:]] = 1.0

    N = next_pow2(3 * n - 2)
    Nc = next_pow2(2 * n - 1)
    c = convolve_fft(b, Nc).copy()

    best_b = b.copy()
    best_max = float(c.max())
    best_score = 2.0 * n * best_max / (m * m)

    rng = np.random.default_rng(seed)
    beta = 0.6
    temp = 0.05
    stale = 0
    two_b = 2.0 * b
    total_iters = 0

    while time.monotonic() < deadline:
        cmax = float(c.max())
        if cmax <= 0 or not np.isfinite(cmax):
            break
        score = 2.0 * n * cmax / (m * m)
        if score < best_score * (1.0 - 1e-13):
            best_score = score
            best_max = cmax
            best_b = b.copy()

        w = compute_weights(c, cmax, beta * cmax, floor=0.03)
        fa = np.fft.rfft(b, N)
        g = grad_from_w(w, fa, N, n)
        g -= g.mean()
        gscale = float(np.abs(g).max())
        if gscale <= 1e-15 or not np.isfinite(gscale):
            break
        g = g / gscale

        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        si = choose_idx(g[one_idx], rng, True, temp=0.1)
        sj = choose_idx(g[zero_idx], rng, False, temp=0.1)
        if si < 0 or sj < 0:
            break
        i = int(one_idx[si])
        j = int(zero_idx[sj])
        if i == j:
            continue

        # Apply swap
        c[i:i + n] -= two_b
        c[2 * i] -= 1.0
        c[j:j + n] += two_b
        c[2 * j] += 1.0
        b[i] = 0.0
        b[j] = 1.0
        two_b = 2.0 * b

        new_max = float(c.max())
        new_score = 2.0 * n * new_max / (m * m)

        accept = False
        if new_score < score * (1.0 - 1e-13):
            accept = True
        else:
            delta = new_score - score
            if delta < 1.0:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    accept = True

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_max = new_max
                best_b = b.copy()
            stale = 0
            temp = max(0.0008, temp * 0.9985)
            beta = max(0.03, beta * 0.997)
        else:
            c[j:j + n] -= two_b
            c[2 * j] -= 1.0
            c[i:i + n] += two_b
            c[2 * i] += 1.0
            b[j] = 0.0
            b[i] = 1.0
            two_b = 2.0 * b
            stale += 1
            temp = max(0.0005, temp * 0.96)
            if stale % 200 == 0:
                beta = min(1.5, beta * 1.5)
                temp = min(0.06, temp * 1.6)
                stale = 0
            if temp < 0.0003 and stale > 600:
                # Random restart with perturbation
                for _ in range(20):
                    i2 = int(rng.integers(0, n))
                    if b[i2] > 0.5:
                        c[i2:i2 + n] -= two_b
                        c[2 * i2] -= 1.0
                        b[i2] = 0.0
                        two_b = 2.0 * b
                    j2 = int(rng.integers(0, n))
                    if b[j2] < 0.5:
                        c[j2:j2 + n] += two_b
                        c[2 * j2] += 1.0
                        b[j2] = 1.0
                        two_b = 2.0 * b
                temp = 0.04
                stale = 0

        total_iters += 1
        if total_iters % 5000 == 0:
            print("  binary n={} m={} iter={} best={:.9f} temp={:.5f}".format(
                n, m, total_iters, best_score, temp), flush=True)

    return normalize_positive(best_b, 0.0), best_score


def polish_binary_continuous(a_bin, deadline, seed=1):
    """Start from binary, allow small continuous perturbations."""
    a = np.asarray(a_bin, dtype=np.float64).ravel().copy()
    n = a.size
    N = next_pow2(3 * n - 2)
    best_a = a.copy()
    best_score = internal_score(a, N)
    if not np.isfinite(best_score):
        return a, np.inf

    rng = np.random.default_rng(seed)
    sigma = 0.08
    eps = 0.15  # max deviation from binary
    stale = 0

    while time.monotonic() < deadline:
        width = int(rng.integers(max(2, n // 200), max(3, n // 12) + 1))
        width = min(width, n // 2)
        start = int(rng.integers(0, n - width + 1))

        t = np.linspace(0, 1, width)
        env = np.exp(-3.0 * (t - 0.5) ** 2)
        d = sigma * env * rng.normal(size=width)
        d -= d.mean()

        old = a[start:start + width].copy()
        a_try = a.copy()
        a_try[start:start + width] = old + d
        # Clip to stay near binary: [0,0]*0.5 or [0,1]*1.5 approximately
        a_try = np.clip(a_try, 0.0, None)
        s = float(a_try.sum())
        if s < 1e-10:
            a_try = a.copy()
        else:
            a_try = a_try / s * (s / float(a.sum()))  # keep same sum roughly
            a_try = a_try / a_try.sum()  # normalize

        scoring = internal_score(a_try, N)
        if np.isfinite(scoring) and scoring < best_score * (1.0 - 1e-10):
            best_a = a_try.copy()
            best_score = scoring
            a = a_try.copy()
            stale = 0
            sigma = min(0.2, sigma * 1.03)
        else:
            a[start:start + width] = old
            stale += 1
            sigma = max(0.01, sigma * 0.95)
            if stale > 300:
                # Jump
                idx = rng.integers(0, n, size=min(50, n // 100))
                a[idx] = rng.random(size=idx.size)
                a = normalize_positive(a, 0.0)
                sc = internal_score(a, N)
                if np.isfinite(sc) and sc < best_score:
                    best_a = a.copy()
                    best_score = sc
                else:
                    a = best_a.copy()
                stale = 0
                sigma = 0.1

        if best_score < 1.45:
            break

    return best_a, best_score


def local_search_continuous(a, deadline, seed=2):
    """Random log-block perturbations on a continuous (non-binary) sequence."""
    a = normalize_positive(a, 1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)
    best_a = a.copy()
    best_score = internal_score(a, N)

    loga = np.log(a + 1e-20)
    loga -= loga.max()
    rng = np.random.default_rng(seed)
    sigma = 0.15
    stale = 0

    while time.monotonic() < deadline:
        width_low = max(2, n // 3000)
        width_high = max(3, n // 20)
        width = int(rng.integers(width_low, min(width_high, n // 2) + 1))
        start = int(rng.integers(0, n - width + 1))

        old = loga[start:start + width].copy()
        d = sigma * rng.normal(size=width)
        d -= d.mean()
        d = np.clip(d, -2.5, 2.5)
        loga[start:start + width] += d

        a_try = np.exp(np.clip(loga - loga.max(), -60.0, 0.0))
        a_try = np.maximum(a_try, 1e-300)
        a_try = a_try / a_try.sum()
        a_try = normalize_positive(a_try, 1e-15)

        sc = internal_score(a_try, N)
        if np.isfinite(sc) and sc < best_score * (1.0 - 1e-10):
            best_a = a_try.copy()
            best_score = sc
            loga = np.log(best_a + 1e-20)
            loga -= loga.max()
            a = best_a.copy()
            stale = 0
            sigma = min(0.3, sigma * 1.03)
        else:
            loga[start:start + width] = old
            stale += 1
            sigma = max(0.02, sigma * 0.96)
            if stale > 400:
                idx = rng.integers(0, n, size=min(100, n // 80))
                loga[idx] += 1.5 * sigma * rng.normal(size=idx.size)
                a_try = np.exp(np.clip(loga - loga.max(), -60.0, 0.0))
                a_try = np.maximum(a_try, 1e-300)
                a_try = a_try / a_try.sum()
                sc = internal_score(a_try, N)
                if np.isfinite(sc) and sc < best_score:
                    best_a = a_try.copy()
                    best_score = sc
                    a = best_a.copy()
                    loga = np.log(best_a + 1e-20)
                    loga -= loga.max()
                else:
                    loga[idx] -= 1.5 * sigma * rng.normal(size=idx.size)
                stale = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()
    total = max(120.0, float(budget_s) - 45.0)
    deadline = start + total

    best_a = make_uniform(1024)
    best_score = internal_score(best_a)
    print("Initial uniform score = {:.9f}".format(best_score), flush=True)

    # Load previous best if available
    prev = get_prev_sequence()
    if prev is not None:
        sc = internal_score(prev)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = prev.copy()
            print("Previous sequence score = {:.9f}, n={}".format(best_score, best_a.size), flush=True)

    # --- Phase 1: Quick smooth optimization at a few sizes ---
    for n in (1024, 2048):
        if time.monotonic() > deadline - 1800:
            break
        a0 = resample_seq(best_a, n)
        a1, sc1 = optimize_smooth(n, min(deadline - 1700, time.monotonic() + 40), seed + n, a0)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("Smooth n={} score={:.9f}".format(n, best_score), flush=True)

    # --- Phase 2: Binary search at multiple sizes ---
    # Key: try various n and m/n ratios
    binary_configs = []
    # Sort by priority: larger n first (better asymptotic), then various ratios
    for n in (4096, 8192, 2048, 1024):
        for ratio in (0.5, 0.45, 0.55, 0.40, 0.60):
            m = int(n * ratio)
            if 4 <= m <= n - 4:
                binary_configs.append((n, m))

    # Time allocation per binary run
    time_per_binary = min(
        max(20.0, (deadline - time.monotonic() - 300.0) / max(1, len(binary_configs))),
        250.0
    )

    rng_out = np.random.default_rng(seed + 99999)

    for n, m in binary_configs:
        remaining = deadline - time.monotonic() - 250.0
        if remaining < 30.0:
            break

        a_smooth = resample_seq(best_a, n)
        inits = []
        inits.append(binary_init_from_smooth(a_smooth, n, m))
        inits.append(binary_init_random(n, m, rng_out))

        # Stride-based init
        for step in (2, 3, 7):
            inits.append(binary_init_stride(n, m, step, rng_out))

        # QR-like init for prime n
        if n % 2 == 1:
            p = n
            # Quadratic residues mod p
            if p >= 7:
                bqr = np.zeros(n)
                bqr[0] = 1.0
                for i in range(1, n):
                    if pow(i, (p - 1) // 2, p) == 1:
                        bqr[i] = 1.0
                mqr = int(bqr.sum())
                if abs(mqr - m) <= n // 10:
                    inits.append(bqr)

        # Limit inits to save time
        max_inits = min(4, max(2, int(remaining / max(30, time_per_binary))))
        inits = inits[:max_inits]

        t_per = min(time_per_binary, remaining / max(1, len(inits)))
        if t_per < 15.0:
            t_per = max(15.0, remaining / max(1, len(inits)))

        for k, init in enumerate(inits):
            if time.monotonic() > deadline - 230.0:
                break
            tt = min(deadline - 230.0, time.monotonic() + t_per)
            a1, sc1 = optimize_binary(init, tt, seed=seed + 50000 + n + k * 7, target_m=m)
            if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                print("Binary n={} m={} score={:.9f}".format(n, m, best_score), flush=True)

    # --- Phase 3: Post-binary continuous polish ---
    if time.monotonic() < deadline - 80.0:
        n_pol = best_a.size
        a1, sc1 = polish_binary_continuous(best_a, min(deadline - 60, time.monotonic() + 200), seed=777)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("Polish score={:.9f}".format(best_score), flush=True)

    # --- Phase 4: Continuous local search ---
    if time.monotonic() < deadline - 40.0:
        a1, sc1 = local_search_continuous(best_a, min(deadline - 20, time.monotonic() + 200), seed=888)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("Local search score={:.9f}".format(best_score), flush=True)

    # --- Phase 5: Try a different size with what's left ---
    if time.monotonic() < deadline - 60.0:
        remaining = deadline - time.monotonic() - 40.0
        if remaining > 60:
            # Try one more binary run at a good size
            n2 = 4096
            m2 = n2 // 2
            a_smooth2 = resample_seq(best_a, n2)
            init2 = binary_init_from_smooth(a_smooth2, n2, m2)
            a1, sc1 = optimize_binary(init2, deadline - 35.0, seed=seed + 31415, target_m=m2)
            if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                print("Final binary n={} score={:.9f}".format(n2, best_score), flush=True)

            # Quick polish
            if time.monotonic() < deadline - 20.0:
                a1, sc1 = polish_binary_continuous(best_a, deadline - 15.0, seed=555)
                if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                    best_score = sc1
                    best_a = a1.copy()
                    print("Final polish score={:.9f}".format(best_score), flush=True)

    # Final validation
    best_a = normalize_positive(best_a, 0.0)
    out_list = [float(x) for x in best_a]
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")

    if not np.isfinite(final_eval) or final_eval > 2.0:
        best_a = make_uniform(1024)
        out_list = [float(x) for x in best_a]
        try:
            final_eval = float(evaluate_sequence(out_list))
        except Exception:
            final_eval = 2.0

    elapsed = time.monotonic() - start
    print("DONE in {:.1f}s. Final score={:.12f}, n={}".format(elapsed, final_eval, len(out_list)), flush=True)
    return out_list
