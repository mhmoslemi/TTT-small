```
def compute_gradient(p, w, L, mode='mult', Fp=None, scale=True):
    p=np.asarray(p,dtype=np.float64); w=np.asarray(w,dtype=np.float64)
    n=p.size
    if Fp is None or Fp.size != L//2+1:
        Fp=np.fft.rfft(p,L)
    Fw=np.fft.rfft(w[::-1], L)
    H=np.fft.irfft(Fp*Fw, L)
    r=H[n-1:2*n-1][::-1]
    r=np.where(np.isfinite(r), r, 0.0)
    r=np.maximum(r,0.0)
    mean=float(np.dot(p,r))
    if mode == 'correct':
        g=p*(r-mean)
    else:
        g=r-mean
    g=np.where(np.isfinite(g), g, 0.0)
    if scale:
        rms=np.sqrt(float(np.mean(g*g)))
        if np.isfinite(rms) and rms > 1e-14:
            g *= 1.0/rms
        maxabs=float(np.max(np.abs(g))) if g.size else 0.0
        if np.isfinite(maxabs) and maxabs > 10.0:
            g *= 10.0/maxabs
    return g
```

```
def make_seed(n, kind, rng):
    n=int(n)
    x=(np.arange(n)+0.5)/n
    vals=None
    if kind=='uniform': vals=np.ones(n)
    elif kind=='linear': a=float(rng.uniform(-0.7,0.7)); vals=1+a*(2*x-1)
    elif kind in ('cos','smooth'):
       D=int(rng.integers(2,6))
       vals=np.ones(n)
       for m in range(1,D+1):
           amp=float(rng.uniform(0.05, 0.9/m))
           phase=float(rng.uniform(-np.pi,np.pi))
           if rng.random()<0.5:
               vals += amp*np.cos(np.pi*m*x+phase)
           else:
               vals += amp*np.cos(2*np.pi*m*x+phase)
    elif kind=='gauss':
       K=int(rng.integers(1,4))
       weights=rng.dirichlet(np.ones(K))
       means=rng.uniform(0.05,0.95,K)
       sigmas=rng.uniform(0.05,0.4,K)
       vals=0.05*np.ones(n)
       for w,mu,sig in zip(weights,means,sigmas):
           vals += (w/sig)*np.exp(-0.5*((x-mu)/sig)**2)
    elif kind=='asym_cos':
       a1=float(rng.uniform(-0.9,0.9)); p1=float(rng.uniform(-np.pi,np.pi))
       a2=float(rng.uniform(-0.5,0.5)); p2=float(rng.uniform(-np.pi,np.pi))
       a3=float(rng.uniform(-0.3,0.3)); p3=float(rng.uniform(-np.pi,np.pi))
       vals=1 + a1*np.cos(np.pi*x+p1) + a2*np.cos(2*np.pi*x+p2) + a3*np.cos(3*np.pi*x+p3)
    else:
       vals=np.ones(n)
    vals=np.where(np.isfinite(vals), vals, 0.0)
    vals=np.maximum(vals, 1e-8)
    return normalize_positive(vals)
```

```python
"""
Search for low-peak self-convolution step functions.

The objective is scale invariant.  We work with nonnegative probability
distributions p (sum p = 1) and minimize

    J(p) = 2 * n * max_k (p * p)[k].

The optimizer combines:
  - diverse nonnegative seeds (Fourier/Gaussian/linear shapes),
  - coarse-to-fine Adam-style multiplicative updates in log-space,
  - adaptive smoothed-max / active-set weights for the convolution peaks,
  - periodic restarts from alternative seeds,
  - final small power/low-pass/length refinements.
"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    if a is None:
        return None
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return arr / s


def compare_update(p, s, best_p, best_s):
    if p is None:
        return best_p, best_s
    arr = normalize_positive(p)
    if arr is None:
        return best_p, best_s
    s = float(s)
    if not np.isfinite(s):
        return best_p, best_s
    if best_p is None or s < best_s - 1e-14:
        return arr.copy(), s
    return best_p, best_s


def consider_candidate(p, m, best_p, best_s, best_n):
    if p is None:
        return best_p, best_s, best_n
    a = resize_sequence(p, int(m))
    if a is None:
        return best_p, best_s, best_n
    s = fast_score(a)
    if np.isfinite(s) and (best_p is None or s < best_s - 1e-12):
        return a, float(s), int(m)
    return best_p, best_s, best_n


def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -80.0, 80.0)
    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(z)
        s = float(e.size)
    return e / s


def resize_sequence(seq, m):
    a = normalize_positive(seq)
    m = int(m)
    if a is None or m <= 0:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)

    x_old = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    x_new = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    loga = np.log(np.maximum(a, 1e-300))
    loga = np.where(np.isfinite(loga), loga, -80.0)
    lognew = np.interp(x_new, x_old, loga)
    lognew = np.clip(lognew, -80.0, 80.0)
    q = np.exp(lognew)
    return normalize_positive(q)


def conv_c(p, L=None):
    a = normalize_positive(p)
    if a is None:
        return np.zeros(0, dtype=np.float64)
    p = a
    n = p.size
    if L is None:
        L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)
    need = 2 * n - 1
    if c.size < need:
        c = np.pad(c[:need], (0, need - c.size), mode="constant")
    else:
        c = c[:need]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def fast_score(p):
    a = normalize_positive(p)
    if a is None:
        return np.inf
    n = a.size
    c = conv_c(a)
    if c.size == 0:
        return np.inf
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def compute_gradient(p, w, L, mode="mult", Fp=None, scale=True):
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    if Fp is None or Fp.size != L // 2 + 1:
        Fp = np.fft.rfft(p, L)
    Fw = np.fft.rfft(w[::-1], L)
    H = np.fft.irfft(Fp * Fw, L)
    r = H[n - 1:2 * n - 1][::-1]
    r = np.where(np.isfinite(r), r, 0.0)
    r = np.maximum(r, 0.0)
    mean = float(np.dot(p, r))

    if mode == "correct":
        g = p * (r - mean)
    else:
        g = r - mean

    g = np.where(np.isfinite(g), g, 0.0)
    if scale:
        rms = np.sqrt(float(np.mean(g * g)))
        if np.isfinite(rms) and rms > 1e-14:
            g = g / rms
        if g.size:
            maxabs = float(np.max(np.abs(g)))
            if np.isfinite(maxabs) and maxabs > 10.0:
                g = g * (10.0 / maxabs)
    return g


def make_seed(n, kind, rng):
    n = int(n)
    if n <= 0:
        return None
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    vals = None

    if kind == "uniform":
        vals = np.ones(n, dtype=np.float64)
    elif kind == "linear":
        a = float(rng.uniform(-0.75, 0.75))
        vals = 1.0 + a * (2.0 * x - 1.0)
    elif kind in ("cos", "smooth"):
        D = int(rng.integers(2, 6))
        vals = np.ones(n, dtype=np.float64)
        for m in range(1, D + 1):
            amp = float(rng.uniform(0.05, 0.9 / m))
            phase = float(rng.uniform(-np.pi, np.pi))
            if rng.random() < 0.5:
                vals += amp * np.cos(np.pi * m * x + phase)
            else:
                vals += amp * np.cos(2.0 * np.pi * m * x + phase)
    elif kind == "asym_cos":
        a1 = float(rng.uniform(-0.9, 0.9))
        p1 = float(rng.uniform(-np.pi, np.pi))
        a2 = float(rng.uniform(-0.55, 0.55))
        p2 = float(rng.uniform(-np.pi, np.pi))
        a3 = float(rng.uniform(-0.35, 0.35))
        p3 = float(rng.uniform(-np.pi, np.pi))
        vals = (
            1.0
            + a1 * np.cos(np.pi * x + p1)
            + a2 * np.cos(2.0 * np.pi * x + p2)
            + a3 * np.cos(3.0 * np.pi * x + p3)
        )
    elif kind == "gauss":
        K = int(rng.integers(1, 4))
        weights = rng.dirichlet(np.ones(K, dtype=np.float64))
        means = rng.uniform(0.05, 0.95, K)
        sigmas = rng.uniform(0.05, 0.40, K)
        vals = 0.05 * np.ones(n, dtype=np.float64)
        for wgt, mu, sig in zip(weights, means, sigmas):
            vals += (float(wgt) / float(sig)) * np.exp(-0.5 * ((x - float(mu)) / float(sig)) ** 2)
    else:
        vals = np.ones(n, dtype=np.float64)

    vals = np.where(np.isfinite(vals), vals, 0.0)
    vals = np.maximum(vals, 1e-8)
    return normalize_positive(vals)


def lowpass_filter(p, fraction=0.9):
    a = normalize_positive(p)
    if a is None:
        return None
    n = a.size
    L = next_pow2(n)
    F = np.fft.rfft(a, L)
    length = F.size
    fraction = float(np.clip(fraction, 0.0, 1.0))
    cut = int(np.floor(length * fraction))
    if cut <= 0:
        cut = 1
    if cut < length:
        F[cut:] = 0.0
    q = np.fft.irfft(F, L)[:n]
    return normalize_positive(q)


def power_transform(p, gamma=1.0):
    a = normalize_positive(p)
    if a is None:
        return None
    gamma = float(gamma)
    q = np.power(np.maximum(a, 0.0), gamma)
    q = np.where(np.isfinite(q), q, 0.0)
    return normalize_positive(q)


def adam_run(parent, duration_s, rng, tau_ratio=0.10, lr0=0.02, mode="mult",
             top_k=9, top_blend=0.25, max_no_improve=600, noise_std=0.08,
             alt_seeds=None, lr_min=1e-4, lr_max=0.08, tau_min=5e-4, verbose=False):
    p = normalize_positive(parent)
    if p is None:
        return None, np.inf
    n = p.size
    if n <= 0:
        return None, np.inf
    duration_s = max(0.0, float(duration_s))
    if duration_s <= 0.0:
        return p, fast_score(p)

    L = next_pow2(3 * n - 2)
    c0 = conv_c(p, L)
    if c0.size == 0:
        return p, np.inf
    best_p = p.copy()
    best_s = float(2.0 * n * max(0.0, float(c0.max())))

    deadline = time.monotonic() + duration_s
    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, 0.0)
    z -= float(z.mean())
    z = np.clip(z, -80.0, 80.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    bias = 0
    step = 0
    no_improve = 0
    tau_r = float(max(tau_min, tau_ratio))
    lr = float(np.clip(lr0, lr_min, lr_max))

    while time.monotonic() < deadline:
        Fp = np.fft.rfft(p, L)
        c = np.fft.irfft(Fp * Fp, L)
        need = 2 * n - 1
        if c.size < need:
            c = np.pad(c[:need], (0, need - c.size), mode="constant")
        else:
            c = c[:need]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        cm = float(c.max()) if c.size else 0.0
        if not np.isfinite(cm) or cm <= 0.0:
            break

        true_s = float(2.0 * n * cm)
        if true_s < best_s - 1e-12:
            best_s = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        tau = max(1e-18, cm * tau_r)
        zc = (c - cm) / tau
        zc = np.clip(zc, -50.0, 0.0)
        expz = np.exp(zc)
        expsum = float(expz.sum())
        if not np.isfinite(expsum) or expsum <= 0.0:
            w = np.ones(c.size, dtype=np.float64) / float(c.size)
        else:
            w = expz / expsum

        if top_k is not None and int(top_k) > 0:
            K = int(min(int(top_k), c.size))
            if K >= 1:
                idx = np.argpartition(c, c.size - K)[-K:]
                w_top = np.zeros_like(w)
                w_top[idx] = 1.0 / float(K)
                bl = float(top_blend)
                if bl > 0.0:
                    w = (1.0 - bl) * w + bl * w_top
                else:
                    w = w_top

        g = compute_gradient(p, w, L, mode=mode, Fp=Fp, scale=True)

        bias += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        b1c = 1.0 - beta1 ** bias
        b2c = 1.0 - beta2 ** bias
        delta = lr * (m / b1c) / (np.sqrt(v / b2c) + eps)
        z = z - delta
        z = np.where(np.isfinite(z), z, 0.0)
        z -= float(z.mean())
        z = np.clip(z, -80.0, 80.0)
        p = softmax(z)
        step += 1

        if step % 25 == 0:
            if no_improve > 90 and lr > lr_min:
                lr = max(lr_min, lr * 0.86)
            elif no_improve < 5 and lr < lr_max:
                lr = min(lr_max, lr * 1.10)

        if step % 200 == 0:
            tau_r = max(tau_min, tau_r * 0.99)

        if no_improve >= max_no_improve:
            base = None
            if alt_seeds is not None and len(alt_seeds) > 0 and rng.random() < 0.30:
                idx = int(rng.integers(0, len(alt_seeds)))
                alt = resize_sequence(alt_seeds[idx], n)
                if alt is not None:
                    ratio = float(rng.uniform(0.25, 0.65))
                    base = ratio * best_p + (1.0 - ratio) * alt
            if base is None:
                base = best_p.copy()
            base = normalize_positive(base)
            if base is None:
                base = best_p.copy()

            z = np.log(np.maximum(base, 1e-300))
            z = np.where(np.isfinite(z), z, 0.0)
            z = z + rng.normal(0.0, float(noise_std), n)
            z -= float(z.mean())
            z = np.clip(z, -80.0, 80.0)
            p = softmax(z)
            m *= 0.25
            v *= 0.25
            no_improve = 0
            tau_r = float(np.clip(tau_r * 1.6, 0.03, 0.50))
            lr = max(lr_min, lr * 0.85)

        if verbose and step % 5000 == 0:
            print(f"adam n={n} step={step} best={best_s:.6f} tau={tau_r:.5f} lr={lr:.5f}")

    return best_p, best_s


def propose_candidate(seed=42, budget_s=2300.0, **kwargs):
    start = time.monotonic()
    raw_budget = max(60.0, float(budget_s))
    deadline = start + raw_budget - 30.0

    rng = np.random.default_rng(int(seed))

    if raw_budget > 1200.0:
        N = 8192
    elif raw_budget > 500.0:
        N = 4096
    else:
        N = 2048

    best_p = None
    best_s = np.inf

    # Use previous state-of-the-art if available.
    try:
        hs = globals().get("height_sequence_1", None)
        if hs is not None:
            p = resize_sequence(hs, N)
            if p is not None:
                s = fast_score(p)
                best_p, best_s = compare_update(p, s, best_p, best_s)
                if np.isfinite(best_s):
                    print(f"previous sequence score: {best_s:.6f}")
    except Exception:
        pass

    # Build a diverse set of initial candidates.
    seeds = [np.ones(N, dtype=np.float64) / float(N)]
    for kind in ("asym_cos", "cos", "smooth", "gauss", "linear"):
        for _ in range(3):
            p = make_seed(N, kind, rng)
            if p is not None:
                seeds.append(p)

    scored = []
    for p in seeds:
        s = fast_score(p)
        if np.isfinite(s):
            scored.append((s, p.copy()))
    scored.sort()

    for s, p in scored:
        best_p, best_s = compare_update(p, s, best_p, best_s)

    if best_p is None:
        best_p = np.ones(N, dtype=np.float64) / float(N)
        best_s = 2.0

    best_n = N
    pool = [best_p.copy()]
    for _, p in scored[:8]:
        if len(pool) >= 12:
            break
        pool.append(p.copy())

    print(f"initial best n={best_n} score={best_s:.6f}")

    # Coarse exploration from a few of the best seeds.
    now = time.monotonic()
    if deadline - now > 60.0:
        exp_total = min(180.0, max(30.0, (deadline - now) * 0.10))
        N_coarse = max(256, N // 8)
        per = exp_total / float(max(1, min(6, len(pool))))
        for pseed in pool[:6]:
            if time.monotonic() > deadline - 60.0:
                break
            p0 = resize_sequence(pseed, N_coarse)
            if p0 is None:
                continue
            p1, _ = adam_run(
                p0,
                per,
                rng,
                tau_ratio=0.25,
                lr0=0.045,
                mode="mult",
                top_k=21,
                top_blend=0.0,
                max_no_improve=150,
                noise_std=0.12,
            )
            pf = resize_sequence(p1, N)
            if pf is not None:
                s = fast_score(pf)
                best_p, best_s = compare_update(pf, s, best_p, best_s)
                if len(pool) < 16:
                    pool.append(pf)
        print(f"after coarse exploration score={best_s:.6f}")

    now = time.monotonic()
    total_main = max(0.0, (deadline - now) * 0.90)
    configs = [
        (0.20,
         {"tau_ratio": 0.30, "lr0": 0.045, "mode": "mult",
          "top_k": 51, "top_blend": 0.0, "max_no_improve": 450, "noise_std": 0.12}),
        (0.25,
         {"tau_ratio": 0.10, "lr0": 0.030, "mode": "mult",
          "top_k": 21, "top_blend": 0.50, "max_no_improve": 450, "noise_std": 0.10}),
        (0.25,
         {"tau_ratio": 0.05, "lr0": 0.020, "mode": "mult",
          "top_k": 9, "top_blend": 1.0, "max_no_improve": 400, "noise_std": 0.08}),
        (0.15,
         {"tau_ratio": 0.02, "lr0": 0.012, "mode": "mult",
          "top_k": 5, "top_blend": 1.0, "max_no_improve": 320, "noise_std": 0.06}),
        (0.15,
         {"tau_ratio": 0.008, "lr0": 0.007, "mode": "correct",
          "top_k": 3, "top_blend": 1.0, "max_no_improve": 300, "noise_std": 0.05}),
    ]

    for frac, cfg in configs:
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break
        dur = min(total_main * float(frac), rem * 0.95)
        if dur < 15.0:
            dur = min(15.0, rem * 0.90)
        if dur <= 0.0:
            break

        parent = best_p
        p1, s1 = adam_run(parent, dur, rng, alt_seeds=pool, verbose=True, **cfg)
        best_p, best_s = compare_update(p1, s1, best_p, best_s)
        if p1 is not None and len(pool) < 20:
            pool.append(p1.copy())
        print(f"config frac={frac:.2f} done score={best_s:.6f}")

    # Final cheap refinements: length changes, power maps, low-pass filters.
    candidate_lengths = []
    if N // 2 >= 512:
        candidate_lengths.append(N // 2)
    candidate_lengths.append(N)
    if N * 2 <= 8192:
        candidate_lengths.append(N * 2)
    if N != 8192:
        candidate_lengths.append(8192)

    for m in candidate_lengths:
        best_p, best_s, best_n = consider_candidate(best_p, m, best_p, best_s, best_n)

    work_p = best_p
    for gamma in (0.5, 0.7, 1.0, 1.4, 2.0):
        q = power_transform(work_p, gamma)
        s = fast_score(q)
        best_p, best_s = compare_update(q, s, best_p, best_s)

    for frac in (1.0, 0.90, 0.75, 0.55):
        q = lowpass_filter(best_p, frac)
        s = fast_score(q)
        best_p, best_s = compare_update(q, s, best_p, best_s)

    # Scale the returned sequence to a comfortable range.
    out = np.asarray(best_p, dtype=np.float64)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)

    scale = 10.0
    scaled = out * scale
    ssum = float(scaled.sum())
    if not np.isfinite(ssum) or ssum < 0.01:
        scale = 100.0
        scaled = out * scale
        ssum = float(scaled.sum())

    scaled = np.where(np.isfinite(scaled), scaled, 0.0)
    scaled = np.maximum(scaled, 0.0)
    scaled = np.minimum(scaled, 1000.0)

    final_list = [float(x) for x in scaled]

    # Safety fallback and final validation.
    valid = True
    try:
        if not final_list:
            valid = False
        else:
            sc = evaluate_sequence(final_list)
            if not np.isfinite(sc):
                valid = False
    except Exception:
        valid = False

    if not valid:
        fallback_n = 1024
        fallback_list = [float(10.0 / fallback_n) for _ in range(fallback_n)]
        try:
            if not np.isfinite(evaluate_sequence(fallback_list)):
                fallback_list = [1.0 / 1024.0] * 1024
        except Exception:
            pass
        return fallback_list

    return final_list
```
