import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a, b=None):
    """Linear convolution of a with b (or a with itself if b is None)."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(fa * fb, fft_size)
    return result[:size]

def _softmax_subgrad(a, beta):
    """Compute softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a, a)  # length 2n-1
    c_max = c.max()
    w = np.exp(beta * (c - c_max))
    w_sum = w.sum()
    w = w / w_sum
    
    # grad[j] = sum_{i=0}^{n-1} a[i] * (w[j+i] + w[j-i])
    # = conv(flip(a), w)[j+n-1] + conv(a, w)[j]
    
    flip_a = a[::-1]
    # conv(flip_a, w) has length n + (2n-1) - 1 = 3n-2
    cw1 = _fft_convolve(flip_a, w)
    cw2 = _fft_convolve(a, w)
    
    grad = np.zeros(n)
    # cw1[j+n-1] for j=0,...,n-1
    grad[:] = cw1[n-1:n-1+n]  # indices n-1 to 2n-2
    # cw2[j] for j=0,...,n-1
    grad[:] += cw2[0:n]
    
    return grad * 2.0  # factor of 2 from the derivative

def optimize_smooth_max(a0, time_budget, rng, beta_start=5.0, beta_end=500.0):
    """Optimize using smooth max (softmax) approximation with multiplicative updates."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 1e-15)
    a = a / a.sum()
    
    c = _fft_convolve(a, a)
    best_score = 2.0 * n * c.max()
    best_a = a.copy()
    
    beta = beta_start
    eta = 0.3
    momentum = np.zeros(n)
    mu = 0.5
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = 0.0
    
    while time.monotonic() < deadline:
        # Adaptive beta: increase over time
        frac = min(1.0, it / max(1, int(time_budget * 8)))  # ~8 iterations per second target
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, beta_end)
        
        grad = _softmax_subgrad(a, beta)
        
        if time.monotonic() - last_print > 15:
            c_now = _fft_convolve(a, a)
            score_now = 2.0 * n * c_now.max()
            print(f"  it={it}, beta={beta:.1f}, score={score_now:.6f}, best={best_score:.6f}")
            last_print = time.monotonic()
        
        # Normalize gradient
        g_max = grad.max()
        if g_max < 1e-15:
            break
        grad = grad / g_max
        
        # Momentum
        momentum = mu * momentum + (1.0 - mu) * grad
        
        # Multiplicative update
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = best_a.copy()
            eta *= 0.5
        
        # Check score
        if it % 10 == 0:
            c = _fft_convolve(a, a)
            score = 2.0 * n * c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adaptive eta
        if no_improve > 100:
            eta *= 0.98
        elif no_improve == 0:
            eta = min(eta * 1.01, 0.8)
        
        # Restart
        if no_improve > 500:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.3
            momentum = np.zeros(n)
        
        it += 1
    
    # Final check
    c = _fft_convolve(best_a, best_a)
    best_score = 2.0 * n * c.max()
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using softmax-smoothed subgradient descent
with adaptive beta (progressive sharpening), multiplicative updates, and diverse
multi-frequency sinusoidal initializations. Key improvement over prior subgradient:
uses exponential-weighted (softmax) subgradient for a smooth, differentiable
approximation of the max, enabling faster and more stable convergence.
"""

import numpy as np
import time


def _next_pow2(n):
    """Return the next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a, b=None):
    """Fast linear convolution of a with b (or a with itself)."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(fa * fb, fft_size)
    return result[:size]


def _score_fast(a):
    """Compute the objective quickly (no evaluate_sequence call)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())


def _softmax_gradient(a, beta):
    """
    Compute the gradient of the smooth max: (1/beta)*log(sum_k exp(beta*conv[k])).
    Returns gradient array of length n.
    """
    n = len(a)
    c = _fft_convolve(a, a)  # length 2n-1
    c_max = c.max()
    # Numerically stable softmax
    w = np.exp(beta * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum

    # grad[j] = sum_i a[i] * (w[j+i] + w[j-i]) for valid indices
    # = conv(flip(a), w)[j+n-1] + conv(a, w)[j]
    flip_a = a[::-1]
    cw1 = _fft_convolve(flip_a, w)  # length n + (2n-1) - 1 = 3n-2
    cw2 = _fft_convolve(a, w)  # length n + (2n-1) - 1 = 3n-2

    grad = np.zeros(n)
    grad[:] = cw1[n - 1:n - 1 + n]  # indices n-1 to 2n-2
    grad[:] += cw2[0:n]
    return grad


def _optimize_softmax(a0, time_budget, rng, beta_start=10.0, beta_max=800.0):
    """
    Optimizes using softmax-smoothed subgradient with multiplicative updates.
    beta increases over time (annealing from smooth to sharp).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 1e-15)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    best_score = _score_fast(a)
    best_a = a.copy()

    eta = 0.4
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.55
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = 0.0
    total_iters_planned = max(100, int(time_budget * 12))  # ~12 iters/sec estimate

    while time.monotonic() < deadline:
        # Progressive beta: ramp from beta_start to beta_max
        frac = min(1.0, it / total_iters_planned)
        beta = beta_start * (beta_max / beta_start) ** frac
        beta = min(beta, beta_max)

        # Compute softmax gradient
        grad = _softmax_gradient(a, beta)
        g_max = grad.max()
        if g_max < 1e-15:
            break
        grad = grad / g_max

        # Momentum
        momentum = mu * momentum + (1.0 - mu) * grad

        # Multiplicative update (reduce where gradient is high)
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = best_a.copy()
            eta *= 0.5

        # Periodic score check (every 15 iters for speed)
        if it % 15 == 0:
            score = _score_fast(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if time.monotonic() - last_print > 20.0:
                print(f"    [sm] it={it}, beta={beta:.0f}, eta={eta:.3f}, "
                      f"score={best_score:.6f}")
                last_print = time.monotonic()

        # Adaptive eta
        if no_improve > 200:
            eta *= 0.97
        elif no_improve > 80:
            eta *= 0.99
        elif no_improve == 0:
            eta = min(eta * 1.005, 0.7)

        # Restart from best with perturbation
        if no_improve > 600:
            perturb_scale = 0.06 * (1.0 + 0.5 * np.sin(it * 0.01))
            a = best_a * np.exp(rng.normal(0, perturb_scale, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.4
            momentum = np.zeros(n)
            mu = 0.55

        # Occasional structural perturbation
        if it % 500 == 250 and rng.random() < 0.3:
            block_size = max(1, n // 30)
            n_blocks = int(n // block_size)
            for _ in range(3):
                start = int(rng.integers(0, n - block_size))
                a[start:start + block_size] *= np.exp(rng.normal(0, 0.2))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()

        it += 1

    # Final evaluation
    final_score = _score_fast(best_a)
    return best_a, final_score


def _optimize_target_equalize(a0, time_budget, rng):
    """
    Alternative: target-matching approach. Pick a target T and push conv toward T.
    Uses a penalty on (conv[k] - T)^2 weighted by softmax for the max part.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 1e-15)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    best_score = _score_fast(a)
    best_a = a.copy()

    beta = 30.0  # moderate softmax for the "excess" penalty
    eta = 0.15
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.6
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = 0.0

    while time.monotonic() < deadline:
        c = _fft_convolve(a, a)
        c_max = c.max()
        # Target: pull everything toward the mean, push down the peaks
        c_mean = c.mean()
        target = (c_max + c_mean) / 2.0  # between mean and max

        # "Excess" gradient: weight conv[k] that exceed target
        excess = np.maximum(c - target, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            excess_w = np.zeros(len(c))
            excess_w[int(np.argmax(c))] = 1.0
        else:
            excess_w = excess / total_excess

        # Gradient from excess
        flip_a = a[::-1]
        cw1 = _fft_convolve(flip_a, excess_w)
        cw2 = _fft_convolve(a, excess_w)
        grad = np.zeros(n)
        grad[:] = cw1[n - 1:n - 1 + n]
        grad[:] += cw2[0:n]

        g_max = grad.max()
        if g_max < 1e-15:
            break
        grad = grad / g_max

        momentum = mu * momentum + (1.0 - mu) * grad
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = best_a.copy()

        if it % 20 == 0:
            score = _score_fast(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if time.monotonic() - last_print > 20.0:
                print(f"    [eq] it={it}, eta={eta:.3f}, score={best_score:.6f}")
                last_print = time.monotonic()

        if no_improve > 300:
            eta *= 0.95
        if no_improve > 800:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.15
            momentum = np.zeros(n)

        it += 1

    final_score = _score_fast(best_a)
    return best_a, final_score


def _make_init_halfsin(n, A=0.8):
    """Half-period sine, squared to ensure non-negativity."""
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * x / (n - 1))
    a = 1.0 + A * a ** 2
    return a


def _make_init_fullsin(n, A=0.4, phase=0.0):
    """Full-period sinusoidal modulation."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + A * np.sin(2.0 * np.pi * x / n + phase)
    a = np.maximum(a, 0.05)
    return a


def _make_init_multisin(n, freqs=None, amps=None):
    """Multi-frequency sinusoidal."""
    if freqs is None:
        freqs = [1, 2, 3, 5]
    if amps is None:
        amps = [0.3, 0.15, 0.08, 0.04]
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f, amp in zip(freqs, amps):
        a += amp * np.sin(2.0 * np.pi * f * x / n)
    a = np.maximum(a, 0.05)
    return a


def _make_init_decay(n, rate=0.8):
    """Exponential decay from one end."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * x / n) + 0.1
    return a


def _make_init_blocks(n, ratio=0.6, amp_high=2.0, amp_low=0.3):
    """Block structure: high first portion, low second."""
    a = np.full(n, amp_low)
    cut = int(n * ratio)
    a[:cut] = amp_high
    # Smooth the transition
    trans = min(50, n // 20)
    if trans > 1:
        for i in range(trans):
            frac = i / trans
            a[cut - trans + i] = amp_low + (amp_high - amp_low) * frac
    return a


def _make_init_chirp(n):
    """Linear chirp (frequency increases linearly)."""
    x = np.arange(n, dtype=np.float64)
    phase = np.pi * x * x / n
    a = 1.0 + 0.3 * np.sin(phase)
    a = np.maximum(a, 0.05)
    return a
