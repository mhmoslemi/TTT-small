def _score(a, c=None):
    n = a.size
    if c is None:
        c = np.convolve(a, a)
    return 2.0 * n * float(np.max(c))


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (css - 1))[0][-1]
    theta = (css[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_step(a, c, score, eta, active_frac=0.1):
    """One subgradient step on the active lags of the convolution."""
    n = a.size
    mx = float(c.max())
    if mx <= 0:
        return a
    # Find active lags (within active_frac of the max)
    threshold = mx * (1.0 - active_frac)
    active_mask = c >= threshold
    active_indices = np.nonzero(active_mask)[0]
    if active_indices.size == 0:
        active_indices = np.array([int(np.argmax(c))])
    
    # Compute subgradient
    # d(conv[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i <= n-1
    # Subgradient of max = average over active lags
    grad = np.zeros(n, dtype=np.float64)
    for k in active_indices:
        # k-i is valid when 0 <= k-i <= n-1, i.e., max(0, k-n+1) <= i <= min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - lo:k - lo + (hi - lo + 1)]
    grad /= float(active_indices.size)
    
    # Full gradient of score = 2n*max(c)/(sum a)^2
    # d(score)/d(a[i]) = 2n * (d(max(c))/d(a[i]))/(sum a)^2 - 2*score*a[i]/(sum a)
    # With sum a = 1: = 2n * grad[i] - 2*score*a[i]
    sa = float(a.sum())
    full_grad = (2.0 * n * grad) / (sa * sa) - 2.0 * score * a / (sa * sa)
    
    # Take step and project
    a_new = a - eta * full_grad
    a_new = _project_simplex(a_new)
    c_new = np.convolve(a_new, a_new)
    score_new = _score(a_new, c_new)
    return a_new, c_new, score_new

            c = best_c.copy()
            score = best_score
            patience = 0
            if eta < 1e-6:
                break
    
    return best_a, best_c, best_score


def _coord_descent_v2(a, c, score, deadline, rng, n_grid=28, refine=8):
    """Fast coordinate descent with vectorized grid evaluation."""
    n = a.size
    best_a = a.copy()
    best_c = c.copy()
    best_score = score
    
    grid = np.concatenate(([0.0], np.logspace(-7.0, 1.5, n_grid)))
    
    stagnant = 0
    perturb_count = 0
    max_perturb = 6
    noise_scales = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02]
    sweeps = 0
    
    while time.monotonic() < deadline - 1.0:
        sweeps += 1
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline - 0.5:
                break
            i = int(i)
            old = float(a[i])
            D = 1.0 - old
            
            # Affected conv range: [i, i+n-1]
            lo = i
            hi = i + n - 1
            
            # Max of unaffected region
            max_unaff = 0.0
            if lo > 0:
                m1 = float(c[:lo].max())
                if m1 > max_unaff:
                    max_unaff = m1
            if hi + 1 < c.size:
                m2 = float(c[hi + 1:].max())
                if m2 > max_unaff:
                    max_unaff = m2
            
            # For candidate x: new_a[i] = x, new_sum = D + x
            # Affected conv: c[lo:hi+1]
            # new_c[lo+j] = c[lo+j] + 2*(x-old)*a[j] for j != i
            # new_c[2i] = c[2i] + x^2 - old^2
            aff_c = c[lo:hi + 1]
            a_part = a  # a[0:n] = a
            
            # Vectorize over grid
            valid = grid >= 0.0
            S = D + grid
            valid = valid & (S > 1e-15)
            
            if not np.any(valid):
                continue
            
            # new_aff = aff_c[None,:] + 2*(x-old)*a_part[None,:]
            # But fix the diagonal (index i within the affected range, which is at position i-lo = i-i = 0... wait)
            # lo = i, so position k in affected range corresponds to conv index i+k
            # The diagonal (2i) is at position 2i - lo = 2i - i = i
            diag_pos = i  # position within aff_c where 2i is
            
            xg = grid[valid]
            # Shape: (len(xg), n)
            delta_new = aff_c[None, :] + 2.0 * (xg[:, None] - old) * a_part[None, :]
            # Fix diagonal
            delta_new[:, diag_pos] = c[2 * i] + xg * xg - old * old
            
            max_aff = np.max(delta_new, axis=1)
            max_total = np.maximum(max_unaff, max_aff)
            vals = 2.0 * n * max_total / (S[valid] * S[valid])
            
            idx = int(np.argmin(vals))
            if vals[idx] >= score - 1e-12:
                continue
            
            best_x = float(xg[idx])
            best_val = float(vals[idx])
            
            # Local refinement
            if refine > 0 and len(xg) > 2:
                left = float(xg[idx - 1]) if idx > 0 else 0.0
                right = float(xg[idx + 1]) if idx < len(xg) - 1 else best_x * 3.0
                if best_x > 0:
                    local = best_x * np.logspace(-1.0, 1.0, refine)
                else:
                    local = np.linspace(0.0, right, refine)
                local = np.unique(np.concatenate(([best_x], local)))
                local = local[(local >= 0) & (local <= 10000)]
                
                if local.size > 1:
                    Sloc = D + local
                    valid2 = Sloc > 1e-15
                    delta_loc = aff_c[None, :] + 2.0 * (local[valid2, None] - old) * a_part[None, :]
                    delta_loc[:, diag_pos] = c[2 * i] + local[valid2] ** 2 - old * old
                    max_aff2 = np.max(delta_loc, axis=1)
                    max_total2 = np.maximum(max_unaff, max_aff2)
                    vals2 = 2.0 * n * max_total2 / (Sloc[valid2] * Sloc[valid2])
                    idx2 = int(np.argmin(vals2))
                    if vals2[idx2] < best_val:
                        best_x = float(local[valid2][idx2])
                        best_val = float(vals2[idx2])
            
            if best_val >= score - 1e-12:
                continue
            
            # Accept
            a[i] = best_x
            S = D + best_x
            if S <= 1e-15:
                continue
            a /= S
            # Update conv incrementally
            c[lo:hi + 1] = (aff_c + 2.0 * (best_x - old) * a_part) / (S * S)
            c[lo:hi + 1][diag_pos] = (c[2 * i] + best_x * best_x - old * old)  # wait this is wrong
            # Actually let me just recompute the affected part properly
            # a has been updated and normalized. Let me recompute affected conv.
            c[lo:hi + 1] = 0.0
            for k in range(lo, hi + 1):
                # c[k] = sum_j a[j]*a[k-j]
                lo_j = max(0, k - n + 1)
                hi_j = min(n - 1, k)
                c[k] = np.dot(a[lo_j:hi_j + 1], a[k - hi_j:k - lo_j + 1][::-1])
            
            score = _score(a, c)
            improved = True
        
        # Full recompute to avoid drift
        c = np.convolve(a, a)
        np.clip(c, 0.0, None, out=c)
        s = float(a.sum())
        if s > 0:
            a = a / s
            c = c / (s * s)
        score = _score(a, c)
        
        if score < best_score - 1e-11:
            best_score = score
            best_a = a.copy()
            best_c = c.copy()
        
        if improved:
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant >= 3 and perturb_count < max_perturb:
            scale = noise_scales[min(perturb_count, len(noise_scales) - 1)]
            a = best_a * np.exp(rng.normal(0, scale, n))
            np.clip(a, 0, None, out=a)
            s = float(a.sum())
            if s > 0:
                a = a / s
                c = np.convolve(a, a)
                score = _score(a, c)
            perturb_count += 1
            stagnant = 0
        elif stagnant >= 10:
            a = best_a.copy()
            c = best_c.copy()
            score = best_score
            if stagnant >= 20:
                break
        
        if sweeps % 5 == 0 and time.monotonic() < deadline:
            print(f"  Sweep {sweeps}, score={score:.6f}, best={best_score:.6f}")
    
    return best_a, best_c, best_score


def _paired_init(n, rng, alpha=1.0, beta_mean=0.7, beta_std=0.15):
    """Create anti-correlated paired initialization."""
    m = (n + 1) // 2
    if alpha <= 0:
        alpha = 0.1
    weights = rng.dirichlet(np.full(m, alpha))
    a = np.zeros(n, dtype=np.float64)
    betas = np.clip(rng.normal(beta_mean, beta_std, m), 0.05, 0.95)
    for j in range(m):
        k = n - 1 - j
        if j == k:
            a[j] = weights[j]
        else:
            a[j] = weights[j] * (1.0 + betas[j]) * 0.5
            a[k] = weights[j] * (1.0 - betas[j]) * 0.5
    return _norm(a)


def _smooth_paired_init(n, rng, shape='bell'):
    """Smooth paired initialization with envelope shape."""
    x = np.linspace(0, 1, n)
    if shape == 'bell':
        envelope = np.exp(-3.0 * (x - 0.5) ** 2)
    elif shape == 'edge':
        envelope = np.sin(np.pi * x) ** 0.5
    elif shape == 'flat':
        envelope = np.ones(n)
    else:
        envelope = np.exp(-2.0 * np.abs(x - 0.5))
    envelope += 0.05 * rng.random(n)
    m = (n + 1) // 2
    a = np.zeros(n, dtype=np.float64)
    for j in range(m):
        k = n - 1 - j
        w = envelope[j] + envelope[k]
        if w <= 0:
            continue
        beta = np.clip(rng.normal(0.65, 0.2), 0.1, 0.9)
        a[j] = w * (1.0 + beta) * 0.5
        a[k] = w * (1.0 - beta) * 0.5
    return _norm(a)


def _expand(proto, target_len):
    """Expand a short sequence to target length by block replication."""
    R = proto.size
    if R == target_len:
        return proto.copy()
    if R > target_len:
        # Downsample
        out = np.zeros(target_len)
        for j in range(target_len):
            lo = int(j * R / target_len)
            hi = max(lo + 1, int((j + 1) * R / target_len))
            out[j] = proto[lo:hi].mean()
        return _norm(out)
    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=int)
    if rem > 0:
        lengths[:rem] += 1
    out = np.zeros(target_len)
    idx = 0
    for i in range(R):
        L = int(lengths[i])
        if L > 0:
            out[idx:idx + L] = proto[i] / L
            idx += L
    return _norm(out)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return the best non-negative coefficient sequence found within budget_s."""
    budget = max(10.0, float(budget_s))
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget - 10.0)
    rng = np.random.default_rng(int(seed))
    
    final_len = 3809
    best_seq = None
    best_score = float('inf')
    
    def _try_update(a, c=None):
        nonlocal best_seq, best_score
        if c is None:
            c = np.convolve(a, a)
        sc = _score(a, c)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = _norm(a).copy()
            print(f"  New best: {sc:.6f} (len={a.size})")
    
    # Try to use provided starting point
    try:
        hseq = _clean(height_sequence_1)
        if hseq.size > 0:
            hc = np.convolve(hseq, hseq)
            _try_update(hseq, hc)
    except:
        pass
    
    # Baseline: two-level
    base = np.zeros(final_len)
    half = final_len // 2
    base[:half] = 2.0 / 3.0 / half
    base[half:] = 1.0 / 3.0 / (final_len - half)
    base = _norm(base)
    _try_update(base)
    
    print(f"Initial best: {best_score:.6f}")
    
    # Phase 1: Short sequence exploration (for diversity and expansion)
    short_deadline = start_time + min(300.0, 0.15 * budget)
    short_results = []
    
    plan_short = [
        (16, 5, 3.0),
        (32, 5, 4.0),
        (64, 4, 5.0),
        (128, 3, 6.0),
        (256, 2, 8.0),
        (512, 2, 10.0),
    ]
    
    for n_s, trials, t_slice in plan_short:
        if time.monotonic() >= short_deadline:
            break
        for t in range(trials):
            if time.monotonic() >= short_deadline:
                break
            td = min(short_deadline, time.monotonic() + t_slice + 1.0)
            # Paired init
            a_s = _paired_init(n_s, rng, alpha=rng.uniform(0.3, 2.0),
                               beta_mean=rng.uniform(0.5, 0.85),
                               beta_std=rng.uniform(0.1, 0.25))
            c_s = np.convolve(a_s, a_s)
            sc_s = _score(a_s, c_s)
            _try_update(a_s, c_s)
            short_results.append((sc_s, a_s.copy()))
            
            if time.monotonic() < td - 0.5:
                a_s, c_s, sc_s = _coord_descent_v2(a_s, c_s, sc_s, td, rng,
                                                    n_grid=min(40, max(20, 50 - n_s // 8)),
                                                    refine=6)
                _try_update(a_s, c_s)
                short_results.append((sc_s, a_s.copy()))
    
    # Expand good short sequences
    short_results.sort(key=lambda x: x[0])
    for sc, a_s in short_results[:8]:
        if time.monotonic() >= deadline - 300:
            break
        a_exp = _expand(a_s, final_len)
        c_exp = np.convolve(a_exp, a_exp)
        sc_exp = _score(a_exp, c_exp)
        _try_update(a_exp, c_exp)
    
    # Phase 2: Direct optimization at full length with multiple starts
    print(f"\nPhase 2: Full-length optimization, best={best_score:.6f}")
    
    n_starts = 4
    time_per_start = (deadline - time.monotonic() - 200.0) / n_starts
    time_per_start = max(60.0, min(600.0, time_per_start))
    
    for s in range(n_starts):
        if time.monotonic() >= deadline - 180:
            break
        start_t = time.monotonic()
        end_t = min(deadline - 180.0, start_t + time_per_start)
        print(f"\n  Start {s+1}/{n_starts}, time budget={time_per_start:.0f}s, deadline={end_t - start_time:.0f}s")
        
        # Different initialization for each start
        if s == 0:
            # Best from phase 1 or provided
            if best_seq is not None and best_seq.size == final_len:
                a_full = best_seq.copy()
            else:
                a_full = _smooth_paired_init(final_len, rng, shape='bell')
        elif s == 1:
            a_full = _paired_init(final_len, rng, alpha=0.5, beta_mean=0.75, beta_std=0.1)
        elif s == 2:
            a_full = _smooth_paired_init(final_len, rng, shape='edge')
        else:
            # Mix of best and random
            if best_seq is not None and best_seq.size == final_len:
                a_full = 0.6 * best_seq + 0.4 * _paired_init(final_len, rng, alpha=1.0,
                                                              beta_mean=0.7, beta_std=0.2)
            else:
                a_full = _paired_init(final_len, rng, alpha=1.5, beta_mean=0.8, beta_std=0.1)
        
        a_full = _norm(a_full)
        c_full = np.convolve(a_full, a_full)
        sc_full = _score(a_full, c_full)
        print(f"  Initial score: {sc_full:.6f}")
        _try_update(a_full, c_full)
        
        # Coordinate descent
        a_full, c_full, sc_full = _coord_descent_v2(a_full, c_full, sc_full,
                                                     end_t - 60.0, rng,
                                                     n_grid=28, refine=8)
        _try_update(a_full, c_full)
        print(f"  After coord descent: {sc_full:.6f}")
        
        # Subgradient polish
        if time.monotonic() < end_t - 10:
            a_full, c_full, sc_full = _subgradient_polish(a_full, c_full, sc_full,
                                                           end_t, max_iters=3000,
                                                           active_frac=rng.uniform(0.05, 0.15))
            _try_update(a_full, c_full)
            print(f"  After subgradient: {sc_full:.6f}")
        
        # Final coord descent on polished result
        if time.monotonic() < end_t:
            a_full, c_full, sc_full = _coord_descent_v2(a_full, c_full, sc_full,
                                                         end_t, rng, n_grid=24, refine=6)
            _try_update(a_full, c_full)
            print(f"  Final: {sc_full:.6f}")
    
    # Phase 3: If time remains, do more subgradient + coord cycles on best
    if best_seq is not None and best_seq.size == final_len and time.monotonic() < deadline - 30:
        print(f"\nPhase 3: Final polishing, best={best_score:.6f}")
        a_pol = best_seq.copy()
        c_pol = np.convolve(a_pol, a_pol)
        sc_pol = _score(a_pol, c_pol)
        
        for _ in range(5):
            if time.monotonic() >= deadline - 20:
                break
            half_time = (deadline - 20.0 - time.monotonic()) / 2.0
            t1 = time.monotonic() + max(10, half_time * 0.6)
            t2 = deadline - 20.0
            
            a_pol, c_pol, sc_pol = _coord_descent_v2(a_pol, c_pol, sc_pol, t1, rng,
                                                      n_grid=30, refine=10)
            _try_update(a_pol, c_pol)
            
            if time.monotonic() < t2:
                a_pol, c_pol, sc_pol = _subgradient_polish(a_pol, c_pol, sc_pol,
                                                            t2, max_iters=2000,
                                                            active_frac=0.08)
                _try_update(a_pol, c_pol)
    
    # Final evaluation
    if best_seq is not None:
        final_eval = _eval_ext(best_seq)
        print(f"\nFinal best (internal): {best_score:.6f}")
        print(f"Final best (external eval): {final_eval:.6f}")
        out = _norm(best_seq)
        return [float(x) for x in out]
    
    # Fallback
    out = _norm(base)
    return [float(x) for x in out]


def _eval_ext(seq):
    try:
        return float(evaluate_sequence([float(x) for x in seq]))
    except:
        a = np.asarray(seq, dtype=np.float64)
        if a.size == 0:
            return float('inf')
        c = np.convolve(a, a)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        return 2.0 * a.size * float(c.max()) / (s * s)
