    n = a.size
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.clip(c, 0, None)
    c = np.where(np.isfinite(c), c, 0.0)
    
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf, np.inf, np.zeros(n)
    
    # Smooth max weights
    z = (c - cm) / tau
    z = np.clip(z, -50, 0)
    e = np.exp(z)
    s = e.sum()
    w = e / s  # length 2n-1

    # Using FFT: r[i] = conv(a, w_reversed)[2n-2-i]

    r = conv[n-1:2*n-1][::-1].copy()
    r = np.where(np.isfinite(r), r, 0.0)

"""
Improved optimization for minimum autocorrelation peaks in non-negative sequences.
Uses adaptive smooth-max Adam with multi-scale tau, diverse initialization,
top-K peak tracking, and L-BFGS refinement.
"""

import time
import numpy as np

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_pos(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return None
    return a / s


def fast_score(a):
    p = normalize_pos(a)
    if p is None:
        return np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf
    return float(2.0 * n * cm)


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -100.0)
    z = z - z.max()
    z = np.clip(z, -100.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(z) / z.size
    return e / s


def compute_obj_grad(a, tau, L, L2):
    """Compute (true_score, smooth_score, grad_log_domain) for a given tau."""
    n = a.size
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.clip(c, 0, None)
    c = np.where(np.isfinite(c), c, 0.0)

    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf, np.inf, np.zeros(n)

    z = (c - cm) / tau
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        e = np.zeros_like(z)
        e[int(np.argmax(c))] = 1.0
        s = 1.0
    w = e / s

    w_rev = w[::-1].copy()
    A_f = np.fft.rfft(a, L2)
    W_f = np.fft.rfft(w_rev, L2)
    conv = np.fft.irfft(A_f * W_f, L2)
    r = conv[n - 1:2 * n - 1][::-1].copy()
    r = np.where(np.isfinite(r), r, 0.0)

    grad_c = 2.0 * r
    mean_gc = float(np.dot(a, grad_c))
    grad_w = a * (grad_c - mean_gc)

    mg = float(np.max(np.abs(grad_w)))
    if mg > 1e-3:
        grad_w *= 1e-3 / mg

    true_score = 2.0 * n * cm
    smooth_val = cm + tau * float(np.log(s))
    smooth_score = 2.0 * n * smooth_val
    return true_score, smooth_score, grad_w


def resize_seq(seq, m):
    p = normalize_pos(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def make_diverse_candidates(N, rng, deadline):
    """Generate diverse initial candidates."""
    candidates = []
    x = np.linspace(0.0, 1.0, N)

    # Sinusoidal families
    for freq in [1, 2, 3, 4, 6, 8, 12, 16, 24]:
        for amp in [0.1, 0.3, 0.5, 0.8]:
            h = 1.0 + amp * np.cos(2.0 * np.pi * freq * x)
            p = normalize_pos(h)
            if p is not None:
                candidates.append(p)

    # Polynomial families
    for d in [2, 4, 6, 8, 12]:
        t = 2.0 * x - 1.0
        h = 1.0 + 0.3 * np.polynomial.chebyshev.chebval(t, np.array([1.0] + [0.5 * ((-1) ** k) for k in range(1, d + 1)]))
        p = normalize_pos(h)
        if p is not None:
            candidates.append(p)

    # U-shaped / V-shaped
    for p_exp in [0.5, 1.0, 2.0, 4.0]:
        h = np.abs(x - 0.5) ** p_exp + 0.1
        p = normalize_pos(h)
        if p is not None:
            candidates.append(p)

    # Multi-bump
    for nb in [3, 4, 5, 7]:
        h = np.zeros(N)
        for k in range(nb):
            center = (k + 0.5) / nb
            sigma = 0.15 / nb
            h += np.exp(-0.5 * ((x - center) / sigma) ** 2)
        h += 0.05
        p = normalize_pos(h)
        if p is not None:
            candidates.append(p)

    # Random smooth
    for _ in range(20):
        coeffs = rng.normal(0, 0.3, 12)
        t = 2.0 * x - 1.0
        h = 1.0 + np.polynomial.chebyshev.chebval(t, coeffs)
        p = normalize_pos(h)
        if p is not None:
            candidates.append(p)

    # Score and sort
    scored = []
    for p in candidates:
        if time.monotonic() > deadline:
            break
        s = fast_score(p)
        scored.append((s, p))

    scored.sort(key=lambda x: x[0])
    return scored


def adam_optimize(a_init, duration, rng, L, L2, tau_start=0.15, tau_min=0.002,
                  lr_start=0.03, lr_min=0.001, verbose=True):
    """Adam with adaptive tau and learning rate."""
    n = a_init.size
    p = normalize_pos(a_init)
    if p is None:
        return None, np.inf

    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0
    bias_step = 0

    tau = float(tau_start)
    lr = float(lr_start)

    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    smooth_prev = None

    deadline = time.monotonic() + max(0.0, float(duration))

    while time.monotonic() < deadline:
        ts, ss, grad = compute_obj_grad(p, tau, L, L2)
        if not np.isfinite(ts):
            break

        if ts < best_score - 1e-12:
            best_score = ts
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive tau: decrease slowly, increase when stuck
        if step > 0 and step % 2000 == 0:
            tau = max(tau_min, tau * 0.95)

        if no_improve > 5000:
            # Stuck: increase tau for global view, add noise
            tau = min(0.5, tau * 2.0)
            lr = max(lr_min, lr * 0.8)
            z = z + rng.normal(0.0, 0.1, n)
            z -= z.mean()
            z = np.clip(z, -50.0, 50.0)
            m *= 0.3
            v *= 0.3
            no_improve = 0
            if verbose:
                print(f"  restart at step {step}: tau={tau:.4f} lr={lr:.5f} best={best_score:.6f}")

        # Periodic learning rate decay
        if step > 0 and step % 10000 == 0:
            lr = max(lr_min, lr * 0.9)

        # Adam update
        bias_step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        b1 = 1.0 - beta1 ** bias_step
        b2 = 1.0 - beta2 ** bias_step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax_fn(z)
        step += 1

        if verbose and step % 20000 == 0:
            print(f"  adam step {step}: true={ts:.6f} best={best_score:.6f} tau={tau:.4f} lr={lr:.5f}")

    return best_p, best_score


def lbfgs_phase(a_init, tau, L, L2, time_limit=120, max_iter=2000):
    """L-BFGS refinement on the smooth objective."""
    n = a_init.size
    p = normalize_pos(a_init)
    if p is None:
        return None, np.inf

    z0 = np.log(np.maximum(p, 1e-300))
    z0 = np.where(np.isfinite(z0), z0, -50.0)
    z0 -= z0.mean()
    z0 = np.clip(z0, -50.0, 50.0)

    start = time.monotonic()
    best_z = z0.copy()
    best_ss = np.inf

    def func_grad(z):
        nonlocal best_z, best_ss
        if time.monotonic() - start > time_limit:
            raise ValueError("timeout")
        p_ff = softmax_fn(z)
        ts, ss, g = compute_obj_grad(p_ff, tau, L, L2)
        if np.isfinite(ss) and ss < best_ss:
            best_ss = ss
            best_z = z.copy()
        return ss, g

    try:
        res = minimize(func_grad, z0, method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter, 'maxfun': 100000,
                                'gtol': 1e-10, 'maxls': 100, 'ftol': 1e-12})
    except (ValueError, Exception):
        pass

    p_best = softmax_fn(best_z)
    s = fast_score(p_best)
    return p_best, s


def top_k_gradient(a, L, L2, K=20):
    """Compute gradient using top-K peak selection (exact subgradient)."""
    n = a.size
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.clip(c, 0, None)
    c = np.where(np.isfinite(c), c, 0.0)

    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf, np.zeros(n)

    # Hard top-K weights
    K = min(K, c.size)
    idx = np.argpartition(c, -K)[-K:]
    # Only include peaks close to the max (within 5% of max)
    threshold = cm * 0.95
    idx = idx[c[idx] >= threshold]
    if len(idx) == 0:
        idx = np.array([np.argmax(c)])

    w = np.zeros(c.size)
    w[idx] = 1.0 / len(idx)

    w_rev = w[::-1].copy()
    L2_local = next_pow2(3 * n)
    A_f = np.fft.rfft(a, L2_local)
    W_f = np.fft.rfft(w_rev, L2_local)
    conv = np.fft.irfft(A_f * W_f, L2_local)
    r = conv[n - 1:2 * n - 1][::-1].copy()
    r = np.where(np.isfinite(r), r, 0.0)

    grad_c = 2.0 * r
    mean_gc = float(np.dot(a, grad_c))
    grad_w = a * (grad_c - mean_gc)

    mg = float(np.max(np.abs(grad_w)))
    if mg > 1e-3:
        grad_w *= 1e-3 / mg

    true_score = 2.0 * n * cm
    return true_score, grad_w


def adam_topk(a_init, duration, rng, L, L2, K=30, lr=0.015):
    """Adam using top-K subgradient."""
    n = a_init.size
    p = normalize_pos(a_init)
    if p is None:
        return None, np.inf

    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    m = np.zeros(n, dtype=np	float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0
    bias_step = 0

    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0

    deadline = time.monotonic() + max(0.0, float(duration))

    while time.monotonic() < deadline:
        ts, grad = top_k_gradient(p, L, L2, K)
        if not np.isfinite(ts):
            break

        if ts < best_score - 1e-12:
            best_score = ts
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            z = z + rng.normal(0.0, 0.08, n)
            z -= z.mean()
            z = np.clip(z, -50.0, 50.0)
            m *= 0.3
            v *= 0.3
            no_improve = 0
            K = max(5, K // 2)
            if step % 20000 == 0:
                print(f"  topk restart: K={K} best={best_score:.6f}")

        bias_step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        b1 = 1.0 - beta1 ** bias_step
        b2 = 1.0 - beta2 ** bias_step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax_fn(z)
        step += 1

        if step % 20000 == 0:
            print(f"  topk step {step}: score={ts:.6f} best={best_score:.6f}")

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    raw_budget = max(50.0, float(budget_s))
    deadline = start_time + raw_budget - 40.0

    N = 8192
    L = next_pow2(2 * N - 1)
    L2 = next_pow2(3 * N)

    best_p = None
    best_s = np.inf

    def try_update(p, s, label=""):
        nonlocal best_p, best_s
        if p is None or not np.isfinite(s):
            return
        if best_p is None or s < best_s:
            best_p = p.copy()
            best_s = float(s)
            print(f"  NEW BEST ({label}): {s:.6f}")

    # Phase 0: Get starting candidates
    print(f"Phase 0: Initialization (budget={raw_budget:.0f}s)")

    # From height_sequence_1 if available
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and float(hs.sum()) > 1e-9:
            pN = resize_seq(hs, N)
            sN = fast_score(pN)
            try_update(pN, sN, "height_sequence_1 resized")
    except Exception:
        pass

    # Also try the original (non-resized)
    try:
        hs2 = np.asarray(height_sequence_1, dtype=np.float64)
        if hs2.size > 0 and float(hs2.sum()) > 1e-9:
            s_orig = fast_score(hs2)
            try_update(hs2, s_orig, "height_sequence_1 orig")
    except Exception:
        pass

    # Uniform baseline
    uni = np.ones(N, dtype=np.float64) / N
    try_update(uni, 2.0, "uniform")

    # Diverse candidates
    cand_deadline = min(deadline, time.monotonic() + 30.0)
    scored = make_diverse_candidates(N, rng, cand_deadline)
    for s, p in scored[:10]:
        try_update(p, s, "diverse")

    if best_p is None:
        best_p = np.ones(N, dtype=np.float64) / N
        best_s = 2.0

    print(f"  After init: best={best_s:.6f}")

    # Phase 1: Main Adam optimization with adaptive tau
    rem = deadline - time.monotonic()
    if rem > 200:
        print(f"Phase 1: Adam optimization ({rem:.0f}s available)")
        dur = min(rem * 0.55, 900.0)
        p1, s1 = adam_optimize(best_p, dur, rng, L, L2,
                               tau_start=0.2, tau_min=0.003,
                               lr_start=0.035, lr_min=0.002, verbose=True)
        if p1 is not None:
            try_update(p1, s1, "adam1")

    # Phase 2: Second Adam run from best with different params
    rem = deadline - time.monotonic()
    if rem > 200:
        print(f"Phase 2: Adam refinement ({rem:.0f}s available)")
        # Perturb and restart
        noise = np.exp(rng.normal(0.0, 0.08, N))
        perturbed = normalize_pos(best_p * noise)
        if perturbed is None:
            perturbed = best_p.copy()
        dur = min(rem * 0.35, 500.0)
        p2, s2 = adam_optimize(perturbed, dur, rng, L, L2,
                               tau_start=0.08, tau_min=0.002,
                               lr_start=0.02, lr_min=0.001, verbose=True)
        if p2 is not None:
            try_update(p2, s2, "adam2")

    # Phase 3: L-BFGS refinement
    rem = deadline - time.monotonic()
    if HAVE_SCIPY and rem > 200:
        print(f"Phase 3: L-BFGS refinement ({rem:.0f}s available)")
        for tau_lb in [0.05, 0.02, 0.01, 0.005]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_limit = min(rem * 0.3, 150.0)
            p_lb, s_lb = lbfgs_phase(best_p, tau_lb, L, L2, time_limit=t_limit, max_iter=1500)
            if p_lb is not None:
                try_update(p_lb, s_lb, f"lbfgs tau={tau_lb}")
            rem = deadline - time.monotonic()
            if rem < 60:
                break

    # Phase 4: Top-K gradient Adam (different method)
    rem = deadline - time.monotonic()
    if rem > 150:
        print(f"Phase 4: Top-K Adam ({rem:.0f}s available)")
        dur = min(rem * 0.6, 400.0)
        p4, s4 = adam_topk(best_p, dur, rng, L, L2, K=30, lr=0.012)
        if p4 is not None:
            try_update(p4, s4, "topk")

    # Phase 5: Final Adam polish with very small tau
    rem = deadline - time.monotonic()
    if rem > 80:
        print(f"Phase 5: Final polish ({rem:.0f}s available)")
        dur = max(30.0, rem * 0.9)
        p5, s5 = adam_optimize(best_p, dur, rng, L, L2,
                               tau_start=0.01, tau_min=0.001,
                               lr_start=0.005, lr_min=0.0005, verbose=True)
        if p5 is not None:
            try_update(p5, s5, "polish")

    # Phase 6: Final validation
    print(f"Phase 6: Validation. Best fast_score={best_s:.6f}")

    final_candidates = []
    if best_p is not None:
        final_candidates.append(best_p)

    # Also try a few fallbacks
    try:
        hs3 = np.asarray(height_sequence_1, dtype=np.float64)
        if hs3.size > 0:
            final_candidates.append(resize_seq(hs3, N))
    except Exception:
        pass

    final_seq = None
    final_score = np.inf

    for cand in final_candidates:
        if cand is None:
            continue
        lst = [float(x) for x in np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst
            print(f"  validated candidate: score={sc:.6f} len={len(lst)}")

    # If nothing validated, use a simple fallback
    if final_seq is None:
        n_fb = 1024
        final_seq = [1.0 / n_fb] * n_fb
        try:
            sc = evaluate_sequence(final_seq)
            print(f"  fallback uniform: score={sc:.6f}")
        except Exception:
            pass

    elapsed = time.monotonic() - start_time
    print(f"Done in {elapsed:.1f}s. Final score: {final_score:.6f}")
    return final_seq
