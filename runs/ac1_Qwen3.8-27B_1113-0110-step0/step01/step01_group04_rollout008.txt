import time

def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)

def _normalize(v):
    v = _clean(v)
    s = v.sum()
    if s <= 0:
        return np.full(v.size, 1.0/v.size)
    return v/s

def _score(a):
    """Compute 2*n*max(convolve(a,a)) with sum(a)=1."""
    a = _clean(a)
    n = a.size
    if n < 1:
        return np.inf
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a/s
    # FFT-based convolution
    L = 1
    while L < 2*n-1:
        L *= 2
    A = np.fft.rfft(a, n=L)
    C = A*A
    c = np.fft.irfft(C, n=L)[:2*n-1]
    mx = max(c.max(), 0.0)
    return 2.0*n*mx

def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n+1)
    cond = x > (css - 1.0)/idx
    if not np.any(cond):
        return np.full(n, 1.0/n)
    r = np.nonzero(cond)[0][-1]
    tau = (css[r+1] - 1.0)/(r+1)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s <= 0:
        return np.full(n, 1.0/n)
    return out/s

    e = np.exp(np.clip(x - m, -700, 700))
    s = e.sum() + 1.0  # the "+1" is for a_last (exp(0)=1)
    a[:-1] = e / s
    a[-1] = 1.0 / s
    return a

def _smoothed_max_grad(x, tau):
    """
    Objective: tau * log(sum exp(c_k/tau)) where c = a*a
    Returns (value, gradient w.r.t. x)
    """
    a = _from_log_ratios(x)
    n = a.size
    L = 1
    while L < 2*n-1:
        L *= 2
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A*A, n=L)[:2*n-1]
    c = np.maximum(c, 0.0)
    
    cmax = c.max()
    z = (c - cmax) / tau
    z = np.clip(z, -700, 0)
    e = np.exp(z)
    se = e.sum()
    
    val = cmax + tau * np.log(se)
    
    w = e / se  # weights on each lag
    W = np.fft.rfft(w, n=L)
    # grad of val w.r.t. a: 2 * (w * a) circular... actually linear conv
    # d(c_k)/d(a_j) = a_{k-j} (with appropriate bounds)
    # grad_val_wrt_a_j = sum_k w_k * d(c_k)/d(a_j) = 2 * sum_k w_k * a_{k-j} (for valid k-j)
    # This is 2 * (w convolved with a), but only the part that affects a
    # Actually: c = a*a (linear), so dc_k/da_j = a_{k-j} if 0<=k-j<n, and also a_{k-j} if k-j represents the "other" term
    # For linear convolution: c_k = sum_{j=0}^{k} a_j * a_{k-j} (with a_i=0 for i outside [0,n-1])
    # dc_k/da_j = a_{k-j} + a_{k-j} (symmetry) = 2*a_{k-j} for j in [0, min(k, n-1)]
    # Wait, that's not right. c_k = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a_i * a_{k-i}
    # dc_k/da_j = a_{k-j} (if 0 <= k-j < n) + a_{k-j} (if 0 <= k-j < n, same term counted twice?)
    # No: c_k = sum_i a_i * a_{k-i}
    # dc_k/da_j: a_j appears as a_i when i=j: gives a_{k-j}
    #           a_j appears as a_{k-i} when k-i=j i.e. i=k-j: gives a_j (but that's the same term)
    # Wait, for j != k-j: dc_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j}
    # For j = k-j (i.e., k = 2j): dc_k/da_j = 2*a_j
    # So dc_k/da_j = 2*a_{k-j} for all valid j (with a_i=0 outside [0,n-1])
    
    # grad_val_wrt_a_j = sum_k w_k * dc_k/da_j = 2 * sum_k w_k * a_{k-j}
    # This is 2 * (full_convolution(w, a))[j] ... actually it's 2 * (w * a) at position j
    # where (w * a)_j = sum_k w_k * a_{k-j}
    # This is the cross-correlation of w and a, evaluated at lag j
    # = (w convolved with a)[j] ... hmm, let me think more carefully
    
    # (w * a)_k = sum_i w_i * a_{k-i} (linear convolution)
    # We want sum_k w_k * a_{k-j}
    # Let l = k-j: sum_l w_{l+j} * a_l
    # This is the cross-correlation <w_shifted, a> where w_shifted[l] = w[l+j]
    # = (w reversed, convolved with a)[n-1-j] ... this is getting complicated
    
    # Simpler: use FFT
    # grad_a = 2 * ifft(fft(w) * conj(fft(a)))[:n]
    # because ifft(fft(w)*conj(fft(a))) gives the circular cross-correlation
    # For linear convolution, we need zero-padding
    
    grad_a = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    
    # Project gradient to the simplex tangent space
    # The parameterization is x_i = log(a_i/a_{n-1}), so:
    # a_i = exp(x_i) * a_{n-1}, and a_{n-1} = 1/(sum(exp(x_i)) + 1)
    # da_i/dx_j = a_i * (delta_{ij} - a_j) for i,j < n-1
    # da_{n-1}/dx_j = -a_{n-1} * a_j
    
    # grad_x_j = sum_i grad_a_i * da_i/dx_j = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
    #           = a_j * (grad_a_j - dot(a, grad_a))
    
    dot = a @ grad_a
    grad_x = a[:-1] * (grad_a[:-1] - dot)
    
    if not np.all(np.isfinite(grad_x)):
        grad_x = np.zeros_like(x)
    
    return val, grad_x

def _subgrad_step(a, eta, top_m=5):
    """
    Subgradient step targeting the top-m peaks of the convolution.
    Returns a new a on the simplex (sum=1, a>=0).
    """
    c = _conv(a)
    n = a.size
    # Find top-m peaks
    idx = np.argsort(c)[::-1][:top_m]
    vals = c[idx]
    # Weights: softmax over top peaks
    w = np.exp((vals - vals.max()) / (vals.max() + 1e-30))
    w = w / w.sum()
    
    # Subgradient: s_j = sum_m w_m * a_{k_m - j} (for valid indices)
    s = np.zeros(n)
    for k in idx:
        k = int(k)
        wm = w[list(idx).index(k)]
        # a_{k-j} for j where 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            s[j_start:j_end+1] += wm * a[k - j_end:k - j_start + 1][::-1]
    
    # Update
    a_new = _project_simplex(a - eta * s)
    return a_new, c.max()

"""
Minimize 2*n*max(conv)(sum)^2 for non-negative step functions.
Combines multi-temperature L-BFGS on smoothed max, subgradient steps
targeting top convolution peaks, and convolution-aware local search.
"""

import numpy as np
import time
from scipy.optimize import minimize as _scipy_minimize


def _cln(v):
    v = np.asarray(v, dtype=np.float64)
    return np.maximum(np.where(np.isfinite(v), v, 0.0), 0.0)


def _nrm(v):
    v = _cln(v)
    s = v.sum()
    if s < 1e-15:
        return np.full(v.size, 1.0 / max(v.size, 1))
    return v / s


def _psx(v):
    """Project onto probability simplex {x>=0, sum=1}."""
    v = _cln(v)
    n = v.size
    if n == 0:
        return v
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    r = np.nonzero(cond)[0][-1]
    tau = (css[r + 1] - 1.0) / (r + 1)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    return out / s


def _fft_len(n):
    L = 1
    target = 2 * n - 1
    while L < target:
        L <<= 1
    return L


def _conv_full(a):
    """Linear convolution a*a, with a normalized (sum=1)."""
    a = _cln(a)
    s = a.sum()
    if s < 1e-15:
        return np.zeros(2 * a.size - 1)
    a = a / s
    n = a.size
    L = _fft_len(n)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    a = _cln(a)
    n = a.size
    if n < 1:
        return np.inf
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = _conv_full(a)
    return 2.0 * n * c.max()


def _lr_params(a):
    """Log-ratio params: x_i = log(a_i / a_{n-1}), i=0..n-2."""
    a = _nrm(a)
    a_safe = np.maximum(a, 1e-25)
    a_safe = a_safe / a_safe.sum()
    ref = a_safe[-1]
    x = np.log(a_safe[:-1] / ref)
    return np.clip(x, -50.0, 50.0)


def _lr_unparam(x):
    """Reconstruct a from log-ratio params."""
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    m = x.max() if x.size else 0.0
    e = np.exp(np.clip(x - m, -700.0, 700.0))
    s = e.sum() + 1.0
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = 1.0 / s
    return a


def _smmax_fun(x, tau):
    """Smoothed max: tau*log(exp(c/tau)) and gradient w.r.t. x."""
    a = _lr_unparam(x)
    n = a.size
    L = _fft_len(n)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cm = c.max()
    z = np.clip((c - cm) / tau, -700.0, 0.0)
    e = np.exp(z)
    se = e.sum()
    val = cm + tau * np.log(se)
    w = e / se
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = a @ ga
    gx = a[:-1] * (ga[:-1] - dot)
    if not np.all(np.isfinite(gx)):
        gx = np.zeros_like(x)
    return val, gx


def _smmax_lbfgs(x, tau, bounds, maxiter=60):
    """Run L-BFGS on smoothed max for maxiter iterations."""
    if _scipy_minimize is None:
        return x, 99
    try:

                     "ftol": 1e-22, "gtol": 1e-15,

def _subgrad(a, eta, topm=5):
    """Subgradient step targeting top-m peaks. Returns (a_new, peak)."""
    c = _conv_full(a)
    n = a.size
    order = np.argsort(c)[::-1][:topm]
    vals = c[order]
    w = np.exp((vals - vals.max()) / (vals.max() + 1e-30))
    w /= w.sum()
    s = np.zeros(n)
    for i in range(topm):
        k = int(order[i])
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        if j0 <= j1:
            s[j0:j1 + 1] += w[i] * a[k - j1:k - j0 + 1][::-1]
    a_new = _psx(a - eta * s)
    return a_new, c.max()

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative step functions.
Multi-phase optimizer: diverse init -> L-BFGS smoothed-max refinement with
geometric cooling -> top-peak subgradient & convolution-penalty steps ->
convolution-aware multiplicative local search.
"""

import numpy as np
import time
from scipy.optimize import minimize as _scipy_minimize


def _cln(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _psx(v):
    v = _cln(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r + 1] - 1.0) / (r + 1)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _fft_len(n):
    L = 1
    t = 2 * n - 1
    while L < t:
        L <<= 1
    return L


def _conv_full(a):
    a = _cln(a)
    s = a.sum()
    if s < 1e-15:
        return np.zeros(2 * a.size - 1)
    a = a / s
    n = a.size
    L = _fft_len(n)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    a = _cln(a)
    n = a.size
    if n < 1:
        return np.inf
    if a.sum() < 1e-12:
        return np.inf
    c = _conv_full(a)
    return 2.0 * n * c.max()


def _lr_params(a):
    a = _cln(a)
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.zeros(n - 1)
    a = a / s
    a_safe = np.maximum(a, 1e-25)
    a_safe = a_safe / a_safe.sum()
    ref = a_safe[-1]
    x = np.log(a_safe[:-1] / ref)
    return np.clip(x, -50.0, 50.0)


def _lr_unparam(x):
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    m = float(x.max()) if x.size else 0.0
    e = np.exp(np.clip(x - m, -700.0, 700.0))
    s = float(e.sum()) + 1.0
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = 1.0 / s
    return a


def _smmax_fun(x, tau):
    a = _lr_unparam(x)
    n = a.size
    L = _fft_len(n)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    z = np.clip((c - cm) / tau, -700.0, 0.0)
    e = np.exp(z)
    se = float(e.sum())
    val = cm + tau * np.log(se)
    w = e / se
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(a @ ga)
    gx = a[:-1] * (ga[:-1] - dot)
    if not np.all(np.isfinite(gx)):
        gx = np.zeros_like(x, dtype=np.float64)
    return val, gx


def _lbfgs(x, tau, bounds, mi=60):
    if _scipy_minimize is None:
        return x, 99
    try:
        res = _scipy_minimize(
            _smmax_fun, x, args=(tau,), method="L-BFGS-B",
            jac=True, bounds=bounds,
            options={"maxiter": mi, "disp": 0, "ftol": 1e-23,
                     "gtol": 1e-15, "maxfun": max(300, 18 * mi)})
        xn = np.where(np.isfinite(res.x), res.x, 0.0)
        return np.clip(xn, -50.0, 50.0), int(res.status)
    except Exception:
        return x, 99


def _subgrad_step(a, eta, topm=5):
    c = _conv_full(a)
    n = a.size
    order = np.argsort(c)[::-1][:topm]
    vals = c[order]
    w = np.exp((vals - vals.max()) / (vals.max() + 1e-30))
    w /= w.sum()
    s = np.zeros(n, dtype=np.float64)
    for i in range(topm):
        k = int(order[i])
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        if j0 <= j1:
            s[j0:j1 + 1] += 2.0 * w[i] * a[k - j1:k - j0 + 1][::-1]
    return _psx(a - eta * s), c.max()


def _conv_penalty_grad(a, M_target):
    c = _conv_full(a)
    n = a.size
    excess = np.maximum(c - M_target, 0.0)
    idx = np.where(excess > 0.0)[0]
    ga = np.zeros(n, dtype=np.float64)
    for k in idx:
        k = int(k)
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        if j0 <= j1:
            ga[j0:j1 + 1] += 2.0 * excess[k] * a[k - j1:k - j0 + 1][::-1]
    return ga, c.max()

        if idx < len(pool) and end_t - cur > 5.0:
            print(f"\n  === Refine candidate {idx} (sc={pool[idx][0]:.8f}) "
                  f"t={time.monotonic()-t0:.0f}s -> {end_t-t0:.0f}s ===")
            a, sc = _refine_start(pool[idx][2], end_t, int(seed) + 301 * (idx + 1))
            if sc < gb_sc:
                gb_sc = sc
                gb_a = np.asarray(a, dtype=np.float64).copy()
                print(f"  ** new global best: {gb_sc:.8f}")
        used += frac

    # Final local search polish with any remaining time
    remain = deadline - time.monotonic()
    if remain > 10.0:
        print(f"\n  Final local search: {remain:.0f}s")
        a, sc = _local_search(gb_a, deadline, int(seed) + 99999)
        if sc < gb_sc:
            gb_sc = sc
            gb_a = a.copy()
            print(f"  ** local search improved: {gb_sc:.8f}")

    gb_a = _psx(gb_a)
    out = [float(max(0.0, x)) for x in gb_a]

    # Validate
    try:
        val = evaluate_sequence(out)
        print(f"  evaluate_sequence: {val:.8f}")
        if not np.isfinite(val):
            print("  INVALID - fallback")
            j = np.arange(n, dtype=np.float64) + 1.0
            v = j ** (-0.5)
            out = [float(x) for x in v / v.sum()]
    except Exception as e:
        print(f"  eval error: {e}")

    print(f"  FINAL: n={len(out)}  score={gb_sc:.8f}  "
          f"time={time.monotonic()-t0:.1f}s")
    return out
