```python
"""
Search for a non-negative sequence minimizing
J(a) = 2*n*max(conv(a,a)) / sum(a)^2.

Strategy:
- Normalize to sum(a)=1, since the objective is homogeneous.
- Use a tangent-plane linear program (CCP/Frank-Wolfe-like step):
  for current f, minimize t over g>=0, sum(g)=1,
      2*(f*g)_k - (f*f)_k <= t
  for selected convolution lags k.
- Polish with projected/multiplicative smooth subgradient/mirror steps using
  the correct gradient of the convolution peak.
- Explore nearby lengths and templates, keeping the verified best sequence.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def _next_fast(size):
    size = int(size)
    if size <= 1:
        return 1
    return 1 << (size - 1).bit_length()


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    N = _next_fast(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.ascontiguousarray(c, dtype=np.float64)
    c[c < 0.0] = 0.0
    return c


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.zeros(0, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    c = _fft_convolve(a)
    if c.size == 0:
        return float("inf")
    mc = float(c.max())
    if mc < 0.0:
        mc = 0.0
    return float(2.0 * n * mc / (s * s))


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (u + target / rho - css) > 0.0
    idxs = np.where(cond)[0]
    if idxs.size == 0:
        return np.full(n, target / n, dtype=np.float64)
    i = int(idxs[-1])
    tau = (css[i] - target) / (i + 1)
    x = v - tau
    x[x < 0.0] = 0.0
    s = float(x.sum())
    if s > 1e-300:
        return x * (target / s)
    return np.full(n, target / n, dtype=np.float64)


def _resample(a, n):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    a = _normalize(a)
    m = a.size
    if m == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if m == n:
        return a.copy()
    xs = np.linspace(0.0, 1.0, m)
    xt = np.linspace(0.0, 1.0, n)
    out = np.interp(xt, xs, a, left=0.0, right=0.0)
    return _normalize(out)


def _make_init(n, variant, rng):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    v = int(variant) % 14
    if v == 0:
        a = 1.0 - t
    elif v == 1:
        a = np.exp(-3.0 * t)
    elif v == 2:
        a = (1.0 - t) * (1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t))
        a = np.maximum(a, 0.0)
    elif v == 3:
        a = (1.0 - t) * (1.0 + 0.7 * np.sin(2.0 * np.pi * 5.0 * t))
        a = np.maximum(a, 0.0)
    elif v == 4:
        a = np.abs(rng.normal(0.0, 1.0, n))
    elif v == 5:
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05)
    elif v == 6:
        a = t * t * (3.0 - 2.0 * t)
    elif v == 7:
        a = (1.0 - t) ** 1.5 * (1.0 + 0.4 * np.cos(2.0 * np.pi * 2.0 * t + 1.0))
        a = np.maximum(a, 0.0)
    elif v == 8:
        a = (1.0 - t) * (1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t))
        a = np.maximum(a, 0.0)
    elif v == 9:
        a = np.exp(-2.0 * t) * (1.0 + 0.5 * np.sin(2.0 * np.pi * 4.0 * t))
        a = np.maximum(a, 0.0)
    elif v == 10:
        c1 = float(rng.random())
        c2 = float(rng.random())
        w1 = 0.02 + 0.10 * float(rng.random())
        w2 = 0.02 + 0.10 * float(rng.random())
        a = np.exp(-((t - c1) ** 2) / w1) + 0.7 * np.exp(-((t - c2) ** 2) / w2)
    elif v == 11:
        f = int(rng.integers(1, 11))
        ph = float(rng.random()) * 2.0 * np.pi
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * f * t + ph)
        a = np.maximum(a, 0.0)
    elif v == 12:
        a = 0.3 + 0.5 / np.sqrt(t * (1.0 - t) + 1e-6)
    else:
        a = np.ones(n, dtype=np.float64)
    return _normalize(a)


def _select_peaks(c, q):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return np.empty(0, dtype=int)
    if q is None:
        return np.arange(L, dtype=int)
    q = int(q)
    if q <= 0:
        return np.empty(0, dtype=int)
    if q >= L:
        return np.arange(L, dtype=int)
    idx = np.argpartition(c, -q)[-q:].astype(int)
    mir = (L - 1) - idx
    allidx = np.concatenate((idx, mir))
    return np.unique(allidx).astype(int)


def _grad_all(a, p):
    """
    Gradient D_k(a) for full convolution index k:
      D_k(a)_j = 2 * a[k-j] if 0 <= k-j < n, else 0.
    Given weights p_k, returns sum_k p_k D_k(a).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    p = np.asarray(p, dtype=np.float64).ravel()
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    if p.size != L:
        tmp = np.zeros(L, dtype=np.float64)
        m = min(p.size, L)
        if m > 0:
            tmp[:m] = p[:m]
        p = tmp
    size = n + L - 1
    N = _next_fast(size)
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * P, N)[:size]
    seg = conv[L - n:L][::-1]
    return 2.0 * np.asarray(seg, dtype=np.float64)


def _tangent_lp(f, max_constraints=None):
    """
    Tangent-plane LP:
      min t
      s.t. 2*(f*g)_k <= t + (f*f)_k   for selected k
           sum(g)=1, g>=0, t>=0.
    """
    f = _normalize(f)
    n = f.size
    if n == 0:
        return None
    c = _fft_convolve(f)
    L = c.size
    idx = _select_peaks(c, max_constraints)
    if idx.size == 0:
        return None
    m = idx.size

    A = np.zeros((m, n + 1), dtype=np.float64)
    b = np.empty(m, dtype=np.float64)
    for r, k in enumerate(idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[r, lo:hi + 1] = 2.0 * f[k - hi:k - lo + 1][::-1]
        A[r, n] = -1.0
        b[r] = c[k]

    A_eq = np.ones(n + 1, dtype=np.float64)
    A_eq[n] = 0.0
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b,
            A_eq=A_eq,
            b_eq=np.array([1.0], dtype=np.float64),
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if res.status == 0:
            g = np.asarray(res.x[:n], dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = float(g.sum())
            if s > 1e-12 and np.isfinite(s):
                return g / s, float(res.x[n]), c
    except Exception:
        pass

    # Fallback: simpler LP min max_k (f*g)_k, s.t. sum(g)=1, g>=0.
    A = np.zeros((m, n + 1), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)
    for r, k in enumerate(idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[r, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
        A[r, n] = -1.0

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b,
            A_eq=A_eq,
            b_eq=np.array([1.0], dtype=np.float64),
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if res.status == 0:
            g = np.asarray(res.x[:n], dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = float(g.sum())
            if s > 1e-12 and np.isfinite(s):
                return g / s, float(res.x[n]), c
    except Exception:
        pass

    return None


def _lp_max_constraints(n):
    n = int(n)
    if n <= 600:
        return None
    if n <= 1200:
        return 300
    if n <= 2500:
        return 150
    return 80


def _tangent_optimize(a0, deadline, rng, max_constraints=None):
    a_best = _normalize(a0)
    s_best = _score(a_best)
    a = a_best.copy()
    s_a = s_best
    fails = 0
    variant = 0

    if a_best.size > 1000:
        alphas = (1.0, 0.5, 0.25)
    else:
        alphas = (1.0, 0.75, 0.5, 0.25, 0.1)

    while time.monotonic() < deadline:
        lp_out = _tangent_lp(a, max_constraints)
        if lp_out is None:
            break
        g, t, _ = lp_out

        new_a = a
        new_s = s_a
        for alpha in alphas:
            cand = (1.0 - alpha) * a + alpha * g
            cand = np.maximum(cand, 0.0)
            ss = float(cand.sum())
            if ss <= 1e-12:
                continue
            cand = cand / ss
            sc = _score(cand)
            if sc < new_s:
                new_a = cand
                new_s = sc

        if new_s < s_a - 1e-12:
            a = new_a
            s_a = new_s
            if new_s < s_best:
                s_best = new_s
                a_best = a.copy()
            fails = 0
        else:
            fails += 1
            if s_a > s_best + 0.02 and fails > 4:
                a = a_best.copy()
                s_a = s_best
            elif fails <= 2:
                a = a + rng.normal(0.0, 0.1 / max(1, a.size), a.size)
                a = _normalize(a)
                s_a = _score(a)
            elif fails <= 5:
                tmpl = _make_init(a.size, variant % 14, rng)
                a = _normalize(0.8 * a + 0.2 * tmpl)
                s_a = _score(a)
            else:
                tmpl = _make_init(a.size, variant % 14, rng)
                a = _normalize(0.6 * a + 0.4 * tmpl)
                s_a = _score(a)
            variant += 1
            if fails > 30:
                break

    return a_best, s_best


def _smooth_round(a0, deadline, rng, mode="add", beta_factor=0.0, top=0, max_iters=50000):
    a_best = _normalize(a0)
    s_best = _score(a_best)
    a = a_best.copy()
    s_a = s_best
    n = a.size
    if n == 0:
        return a_best, s_best

    step_scale = 1.0
    fails = 0
    base_update = 1.0
    n_steps = 3 if n > 1500 else 5

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        c = _fft_convolve(a)
        if c.size == 0:
            break
        maxc = float(c.max())
        if maxc <= 1e-300:
            break
        cur_s = float(2.0 * n * maxc)

        if cur_s < s_a:
            s_a = cur_s
        if cur_s < s_best - 1e-12:
            s_best = cur_s
            a_best = a.copy()

        L = c.size
        if top is not None and int(top) > 0:
            idx = _select_peaks(c, int(top))
            p = np.zeros(L, dtype=np.float64)
            if idx.size > 0:
                p[idx] = 1.0
                p /= p.sum()
            else:
                amax = int(np.argmax(c))
                p[amax] = 1.0
        elif beta_factor > 0.0:
            beta = float(beta_factor) / maxc
            z = beta * (c - maxc)
            z = np.where(z > -50.0, z, -50.0)
            w = np.exp(z)
            sw = float(w.sum())
            if sw <= 0.0 or not np.isfinite(sw):
                p = np.zeros(L, dtype=np.float64)
                amax = int(np.argmax(c))
                p[amax] = 1.0
            else:
                p = w / sw
        else:
            idx = _select_peaks(c, 1)
            p = np.zeros(L, dtype=np.float64)
            if idx.size > 0:
                p[idx] = 1.0
                p /= p.sum()
            else:
                p[int(np.argmax(c))] = 1.0

        G = _grad_all(a, p)
        maxG = float(np.max(np.abs(G))) if G.size else 0.0
        if maxG <= 1e-15:
            break
        dirv = G / maxG

        improved = False
        s_step = step_scale * base_update
        for bt in range(n_steps):
            s_use = s_step * (0.5 ** bt)
            if s_use <= 0.0:
                continue

            if mode == "mul":
                adj = np.exp(-np.minimum(s_use * dirv, 50.0))
                cand = a * adj
                cand = np.maximum(cand, 0.0)
                ss = float(cand.sum())
                if ss > 1e-300:
                    cand = cand / ss
                else:
                    cand = a.copy()
            else:
                cand = a - (s_use / float(n)) * dirv
                cand = _project_simplex(cand, 1.0)

            sc = _score(cand)
            if sc < s_a - 1e-13:
                a = cand
                s_a = sc
                if sc < s_best - 1e-13:
                    s_best = sc
                    a_best = a.copy()
                step_scale = min(2.0, step_scale * 1.15)
                improved = True
                break

        if improved:
            fails = 0
        else:
            step_scale = max(0.03, step_scale * 0.65)
            fails += 1
            if fails > 40:
                if s_a > s_best + 0.02:
                    a = a_best.copy()
                    s_a = s_best
                else:
                    tmpl = _make_init(n, (it // 5) % 14, rng)
                    a = a + rng.normal(0.0, 0.05 / max(1, n), n)
                    a = _normalize(0.85 * a + 0.15 * tmpl)
                    s_a = _score(a)
                fails = 0
                step_scale = 1.0

    return a_best, s_best


def _gradient_polish(a0, deadline, rng):
    a_best = _normalize(a0)
    s_best = _score(a_best)
    if a_best.size == 0:
        return a_best, s_best

    remaining = max(0.0, deadline - time.monotonic())
    configs = [
        ("add", 0.0, 3, 0.18),
        ("mul", 0.0, 3, 0.18),
        ("add", 0.0, 1, 0.14),
        ("mul", 0.0, 1, 0.14),
        ("add", 0.0, 5, 0.15),
        ("mul", 50.0, 0, 0.21),
    ]

    for cfg in configs:
        if time.monotonic() >= deadline - 5.0:
            break
        mode, bf, top, frac = cfg
        cfg_deadline = min(deadline, time.monotonic() + max(5.0, remaining * frac))
        if cfg_deadline <= time.monotonic():
            continue
        a1, s1 = _smooth_round(
            a_best,
            cfg_deadline,
            rng,
            mode=mode,
            beta_factor=bf,
            top=top,
        )
        if s1 < s_best:
            a_best = a1
            s_best = s1

    return a_best, s_best


def _refine_sequence(a0, deadline, rng):
    a = _normalize(a0)
    if a.size == 0:
        return np.zeros(0, dtype=np.float64), float("inf")

    best_a = a.copy()
    best_s = _score(a)
    start = time.monotonic()
    budget = max(0.0, deadline - start)

    if budget > 60.0:
        lp_deadline = min(deadline, start + max(20.0, 0.5 * budget))
        a1, s1 = _tangent_optimize(
            a,
            lp_deadline,
            rng,
            max_constraints=_lp_max_constraints(a.size),
        )
        if s1 < best_s:
            best_a = a1
            best_s = s1

    if time.monotonic() < deadline - 8.0:
        a2, s2 = _gradient_polish(best_a, deadline, rng)
        if s2 < best_s:
            best_a = a2
            best_s = s2

    return best_a, best_s


def _to_output_list(a):
    a = _normalize(a)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0:
        return [1.0]
    return [float(x) for x in a]
```
