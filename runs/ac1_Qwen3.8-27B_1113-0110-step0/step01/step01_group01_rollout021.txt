def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def symmetrize(a):
    """Make sequence symmetric: a[j] = a[n-1-j]."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    return (a + a[::-1]) / 2.0


def convolve_fft(a, N=None):
    n = a.size
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = a.size
    s = float(a.sum())
    if s <= 1e-15:
        return np.inf
    c = convolve_fft(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))

def make_initial_symmetric(n, seed=42):
    """Create a symmetric initial sequence."""
    rng = np.random.default_rng(seed)
    n = max(16, int(n))
    half = (n + 1) // 2
    
    # Start with a symmetric base
    x = np.linspace(0, 1, half)
    # Edge-concentrated: high at edges, low in middle
    base = np.ones(half)
    # Add some structure
    for trial in range(3):
        alpha = rng.uniform(0.1, 0.8)
        mode = rng.integers(1, 7)
        base += alpha * np.cos(np.pi * mode * x / 2)  # symmetric modes
        base = np.maximum(base, 0.01)
    
    # Make full symmetric
    if n % 2 == 0:
        a = np.concatenate([base[half-1::-1], base])
    else:
        a = np.concatenate([base[half-1:0:-1], base])
    
    return normalize_positive(a, floor=1e-10)

def pressure_local_search(a, deadline, seed=0, verbose=False):

def fourier_init(n, seed=42):
    """
    Create an initial sequence using Fourier domain reasoning.
    We want |P̂(m)|^2 to be as flat as possible.
    Start with uniform and add corrections.
    """
    rng = np.random.default_rng(seed)
    N = next_pow2(2 * n - 1)
    
    # Start with uniform
    a = np.ones(n) / n
    
    # Add symmetric corrections to flatten the convolution
    # The uniform convolution is triangular; we want to reduce the center.
    # Adding a cosine mode cos(2πj/n) reduces the center peak.
    for _ in range(20):
        # Try different modes and amplitudes
        mode = rng.integers(1, n // 2)
        alpha = rng.uniform(-0.5, 0.5)
        correction = alpha * np.cos(2 * np.pi * mode * np.arange(n) / n)
        trial = a + correction
        trial = symmetrize(trial)
        trial = np.maximum(trial, 0)
        s = trial.sum()
        if s < 0.5:
            continue
        trial = trial / s
        sc = compute_score_normalized(trial)
        if sc < compute_score_normalized(a):
            a = trial
    
    return normalize_positive(a, floor=1e-10)
