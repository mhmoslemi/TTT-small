def _gradient_step(p, c, k_star, lr):
    """Projected gradient step. p is normalized (sum=1). Returns new p, new c, new score."""
    n = p.size
    M = float(c[k_star])
    # Gradient of score w.r.t. p[i] with sum(p)=1, score = 2n*M
    # d(score)/d(p[i]) = 2n * dc[k_star]/d(p[i]) (since we're in normalized space)
    # dc[k]/d(p[i]) = 2*p[k-i] if 0<=k-i<n, else 0
    # But in normalized space, we need to account for the constraint sum=1.
    # Using Lagrange: minimize 2n*M subject to sum(p)=1, p>=0
    # grad_i = 2n * 2*p[k*-i] - lambda, where lambda = (1/n) * sum_j 2n*2*p[k*-j] ... 
    # Actually simpler: work with unnormalized, then renormalize.
    # In unnormalized: score = 2n*max_b / S^2
    # d(score)/d(a[i]) = 2n*(dc[k*]/da[i]*S - 2*M) / S^3
    # With S=1: = 2n*(dc[k*]/da[i] - 2*M)
    # dc[k*]/da[i] = 2*a[k*-i] (if valid)
    # So grad[i] = 2n*(2*a[k*-i] - 2*M) = 4n*(a[k*-i] - M)
    
    rev = np.zeros(n)
    # a[k*-i] for i in [0, n-1]: k*-i in [k*-n+1, k*]
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        j = k_star - i
        if 0 <= j < n:
            rev[i] = p[j]
    
    grad = 4.0 * n * (rev - M)
    # Negative gradient direction (we want to decrease score)
    direction = -grad
    
    a_new = p + lr * direction
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return p, c, _score_unit(p)
    a_new = a_new / s
    c_new = np.convolve(a_new, a_new)
    sc = _score_unit(a_new)
    return a_new, c_new, sc


def _subgradient_step(p, c, lr, eps_rel=0.01):
    """Subgradient step using all near-max positions."""
    n = p.size
    M = float(c.max())
    if M <= 0:
        return p, c, _score_unit(p)
    # Find active positions
    active = np.where(c >= M * (1.0 - eps_rel))[0]
    # Average gradient over active positions
    grad = np.zeros(n)
    for k in active:
        rev = np.zeros(n)
        lo = max(0, int(k) - (n - 1))
        hi = min(n - 1, int(k))
        for i in range(lo, hi + 1):
            j = int(k) - i
            if 0 <= j < n:
                rev[i] = p[j]
        grad += 4.0 * n * (rev - c[k])
    grad /= len(active)
    direction = -grad
    a_new = p + lr * direction
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return p, c, _score_unit(p)
    a_new = a_new / s
    c_new = np.convolve(a_new, a_new)
    return a_new, c_new, _score_unit(a_new)

def _lp_polish(p, c, sc, deadline, max_iters=3):
    """LP-based direction search."""
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except ImportError:
        return p, c, sc
    
    n = p.size
    if n < 2 or n > 128:
        return p, c, sc
    
    M = float(c.max())
    if M <= 0:
        return p, c, sc
    
    best_p = p.copy()
    best_c = c.copy()
    best_sc = float(sc)
    
    for _ in range(max_iters):
        if time.monotonic() > deadline - 10:
            break
        
        # Build LP: max sum(b) s.t. convolve(p, b)_k <= M, b >= 0
        # convolve(p, b)[k] = sum_i p[i]*b[k-i]
        # For variable b[j], it appears in c[k] for k = j, j+1, ..., j+n-1 with coefficient p[k-j]
        # c[k] = sum_j p[k-j]*b[j] (where 0<=k-j<n)
        
        i_idx = []
        j_idx = []
        vals = []
        for k in range(2 * n - 1):
            for j in range(max(0, k - n + 1), min(n, k + 1)):

                if 0 <= i_val < n:
                    i_idx.append(k)
                    j_idx.append(j)
                    vals.append(p[i_val])
        
        K = sparse.csr_matrix((vals, (i_idx, j_idx)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)
        obj = -np.ones(n)
        
        try:
            res = linprog(c=obj, A_ub=K, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True})

            break
        
        if res is None or not res.success:
            break
        
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs < 1:
            break
        g = g / gs
        
        # Line search: minimize score of (1-t)*p + t*g
        cross = np.convolve(p, g)
        cg = np.convolve(g, g)
        
        best_t = 0.0
        best_val = float(sc)
        for t in np.linspace(0, 1, 21):
            ct = (1-t)**2 * c + 2*(1-t)*t * cross + t**2 * cg
            val = float(2.0 * n * ct.max())

                best_t = float(t)
        
        # Refine

            hi = min(1, best_t + 0.2)
            for t in np.linspace(lo, hi, 21):

                val = float(2.0 * n * ct.max())
                if val < best_val:
                    best_val = val
                    best_t = float(t)
        
        if best_val >= sc - 1e-10:
            break
        
        p = (1 - best_t) * p + best_t * g
        p = _norm(p)
        c = np.convolve(p, p)
        sc = _score_unit(p)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_p = p.copy()
            best_c = c.copy()
    
    return best_p, best_c, best_sc


def _expand(proto, target_len, method='block'):
    """Expand prototype to target length."""
    target_len = int(target_len)
    p = _norm(proto)
    R = p.size
    if R == target_len:
        return p
    if R <= 1:
        return np.ones(target_len, dtype=np.float64) / target_len
    
    if method == 'block':
        base = target_len // R
        rem = target_len % R
        lengths = np.full(R, base)
        lengths[:rem] += 1
        out = np.zeros(target_len)
        idx = 0
        for i in range(R):
            L = int(lengths[i])
            if L > 0:
                out[idx:idx+L] = p[i] / L
                idx += L
        return _norm(out)
    elif method == 'interp':
        x_old = np.linspace(0, 1, R)
        x_new = np.linspace(0, 1, target_len)
        out = np.interp(x_new, x_old, p)
        return _norm(out)
    else:
        # Repeat
        reps = (target_len + R - 1) // R
        out = np.tile(p, reps)[:target_len]
        return _norm(out)
