def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not cond.any():
        return np.full(n, 1.0/n)
    r = np.nonzero(cond)[0][-1]
    tau = (css[r] - 1.0) / (r+1)
    out = np.maximum(v - tau, 0.0)
    out /= out.sum()
    return out
