import numpy as np

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _optimize_gradient(x0, tau, n_steps, time_budget, rng):
    """Gradient-based optimization in log-space."""
    n = len(x0)
    x = x0.copy().astype(np.float64)
    # Normalize: shift x so that sum of softmax is well-scaled
    # Actually, softmax is shift-invariant, so we can shift for numerical stability
    
    best_score = np.inf
    best_x = x.copy()
    
    # Adam optimizer
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    t = 0
    
    deadline = time.monotonic() + time_budget
    no_improve = 0
    
    while time.monotonic() < deadline:
        t += 1
        # Softmax
        x_shift = x - x.max()  # for numerical stability
        a = np.exp(x_shift)
        a /= a.sum()
        
        # Convolution
        c = _fft_convolve(a)
        sum_c = c.sum()  # should be 1
        c_max = c.max()
        
        # Soft probabilities
        c_norm = (c - c_max) / tau
        p = np.exp(c_norm)
        p /= p.sum()
        
        # Score
        score = 2 * n * (c_max + tau * np.log(p.sum()))  # = 2n * LSE
        # Wait, LSE = c_max + tau * log(sum exp((c-c_max)/tau))
        # But p = exp((c-c_max)/tau) / sum(exp((c-c_max)/tau))
        # So LSE = c_max + tau * log(sum p) ... no.
        # LSE = c_max + tau * log(sum_k exp((c_k - c_max)/tau))
        # p_k = exp((c_k - c_max)/tau) / sum_m exp((c_m - c_max)/tau)
        # sum exp((c-c_max)/tau) = 1/p_k * exp((c_k-c_max)/tau) ... this is circular.
        # Let me just compute LSE directly:
        LSE = c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))
        score = 2 * n * LSE
        
        if score < best_score:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Gradient
        E_pc = np.sum(p * c)
        G = _cross_correlate(p, a)
        grad = 4.0 * n * a * (G - E_pc)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Adaptive temperature
        if no_improve > 200:
            tau *= 0.8  # reduce tau (sharper max)
        elif no_improve < 50:
            tau = min(tau * 1.1, 1.0 / (2*n) * 100)  # increase tau
        
        # Periodic restart
        if no_improve > 300:
            # Perturb from best
            x = best_x + rng.normal(0, 0.01, n)
            no_improve = 0
            t = 0
            m = np.zeros(n)
            v = np.zeros(n)
    
    return best_x, best_score

import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n-1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa*fa, fft_size)
    return ca[:size]

def _cross_corr(p, a):
    """G[j] = sum_k p[k] * a[k-j] for j=0..n-1. p has length 2n-1, a has length n."""
    n = len(a)
    fft_size = _next_pow2(3*n - 1)
    fa_r = np.fft.rfft(np.reverse(a), fft_size)
    fp = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(fa_r * fp, fft_size)
    return conv[n-1 : 2*n-1]

def _compute_grad(a, c, p, n):
    """Compute gradient of 2n*LSE(c,tau) w.r.t. x (log-space)."""
    E_pc = np.dot(p, c)
    G = _cross_corr(p, a)
    # grad_x = 4*n*a*(G - E_pc)  (for softmax parameterization)
    # But we also need to account for the constraint sum(a)=1.
    # In log-space with softmax, the gradient is:
    # grad_x_j = 2n * sum_i (dc/da_i) * (da_i/dx_j)
    # = 2n * sum_i 2*a_{k-i} * a_i * (delta_ij - a_j)  [summing over k with weights p_k]
    # = 2n * [2*a_j * sum_k p_k * a_{k-j} - a_j * 2 * sum_k p_k * c_k]
    # = 4n * a_j * (G_j - E_pc)
    grad = 4.0 * n * a * (G - E_pc)
    return grad

def _optimize_grad(a0, tau_init, time_budget, rng):

    n = len(a0)
    a0 = np.maximum(a0, 1e-10)

    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_step = 0
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0

    while time.monotonic() < deadline:

        a = np.exp(x - x.max())
        a /= a.sum()
        
        c = _fft_convolve(a)
        c_max = c.max()

            best_score = score
            best_a = a.copy()
            no_improve = 0
            if it - last_print > 2000 and time.monotonic() - last_print > 5:
                print(f"    [grad] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        if no_improve > 1000:
            break
        
        # Soft probabilities
        c_shift = (c - c_max) / tau
        exp_c = np.exp(np.minimum(c_shift, 0))  # cap to avoid overflow
        p = exp_c / exp_c.sum()
        
        # Gradient
        G = _cross_corr(p, a)
        E_pc = float(np.dot(p, c))
        grad = 4.0 * n * a * (G - E_pc)

        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*grad*grad
        m_hat = m / (1 - beta1**t_step)

            m = np.zeros(n)

            t_step = 0
