def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(v.size, 1.0 / float(v.size))
    return v / s


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n))
    r = int(np.nonzero(cond)[0][-1])

    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n))

    v = v / s
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0:
        mx = max(0.0, mx) if np.isfinite(mx) else 0.0
    return 2.0 * float(n) * mx


def _conv(a):
    n = a.size
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _ratios(a):
    a = _clean(a).copy()
    n = a.size
    if n == 0:
        return a
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    if n == 1:
        return a
    ref = float(a[-1])
    if ref <= 1e-200:
        a[-1] = 1e-12
        a = a / float(a.sum())
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -100.0, 100.0)


def _from_ratios(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1)
    m = float(np.max(x)) if x.size else 0.0
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0:
        return np.full(n, 1.0 / n)
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _q_obj_grad(x, q):
    a = _from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    L = _pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    if q == 2.0:
        f = float(np.dot(c, c))
        w = c
    else:
        f = float(np.sum(np.power(c, q)))
        w = np.power(c, q - 1.0)
    W = np.fft.rfft(w, n=L)
    ga = (2.0 * q) * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    if not np.isfinite(f):
        f = 1e300
    return f, gy


def _soft_obj_grad(x, eps):
    a = _from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    L = _pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    z = np.clip((c - cmax) / eps, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0:
        return 1e300, np.zeros_like(x)
    f = cmax + eps * float(np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _top_peak_grad_obj(x, top_m):
    """Objective: sum of top `top_m` peaks of convolution. Gradient via softmax."""
    a = _from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    L = _pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    # Use softmax over top peaks with tight temperature
    eps = cmax * 0.02
    z = np.clip(c / eps, -500.0, 500.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    w = e / float(e.sum())
    f = float(np.dot(w, c))
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _lbfgs(fun, x, args, maxiter, bounds):
    if _minimize is None:
        return x, 99
    try:
        res = _minimize(fun, x, args=args, method="L-BFGS-B", jac=True,
                        bounds=bounds, options={"maxiter": maxiter, "disp": 0,
                                                 "maxfun": max(200, 12 * maxiter),
                                                 "ftol": 1e-18, "gtol": 1e-13})
        xn = np.asarray(res.x, dtype=np.float64)
        xn = np.where(np.isfinite(xn), xn, 0.0)
        xn = np.clip(xn, -100.0, 100.0)
        return xn, int(res.status)
    except Exception:
        return x, 99


def _power_seq(n, p, off):
    j = np.arange(n, dtype=np.float64)
    v = np.power(np.maximum(j + off, 1e-8), -p)
    return _project_simplex(v)


def _bump_seq(n, center, width):
    j = np.arange(n, dtype=np.float64)
    v = np.exp(-0.5 * ((j - center) / width) ** 2)
    return _project_simplex(v)


def _two_sided(n, p, off, w):
    j = np.arange(n, dtype=np.float64)
    left = np.power(np.maximum(j + off, 1e-8), -p)
    right = np.power(np.maximum(n - 1.0 - j + off, 1e-8), -p)
    v = left + w * right
    return _project_simplex(v)


def _cosine_taper(n, inner_p, inner_off, taper_frac):
    """Cosine-tapered power law."""
    j = np.arange(n, dtype=np.float64)
    v = np.power(np.maximum(j + inner_off, 1e-8), -inner_p)
    t = np.linspace(0, 1, n)
    # Apply cosine taper at both ends
    taper_start = int(taper_frac * n)
    taper = np.ones(n)
    if taper_start > 0:
        taper[:taper_start] = 0.5 * (1 + np.cos(np.pi * np.arange(taper_start) / taper_start))
        taper[-taper_start:] = 0.5 * (1 + np.cos(np.pi * np.arange(taper_start) / taper_start))
    v = v * taper
    return _project_simplex(v)


def _generate_cands(n):
    cands = []
    try:
        h = height_sequence_1
        arr = np.asarray(h, dtype=np.float64)
        arr = _clean(arr)
        if arr.size == n and arr.sum() > 0:
            cands.append(_project_simplex(arr))
            # Scaled variants
            for f in (0.8, 0.9, 1.1, 1.2):
                cands.append(_project_simplex(arr * f))
    except Exception:
        pass

    cands.append(np.full(n, 1.0 / n))
    for p in (0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8):
        for off in (0.3, 0.5, 1.0, 1.5, 2.0):
            cands.append(_power_seq(n, p, off))
    for tw in (0.01, 0.05, 0.1, 0.2, 0.5):
        for p in (0.4, 0.5, 0.6):
            cands.append(_two_sided(n, p, 1.0, tw))
    for fr in (0.05, 0.1, 0.2, 0.3):
        cands.append(_cosine_taper(n, 0.5, 1.0, fr))
        cands.append(_cosine_taper(n, 0.6, 1.0, fr))
    for center in (n * 0.2, n * 0.3, n * 0.4, n * 0.5):
        for width in (n * 0.1, n * 0.15, n * 0.2):
            cands.append(_bump_seq(n, center, width))
    # Random structured
    for seed in (7, 13, 29, 47, 67, 89):
        rng = np.random.default_rng(seed)
        v = np.exp(rng.normal(0, 0.8, n)) * np.power(np.arange(n) + 1, -0.5)
        cands.append(_project_simplex(v))
    return cands


def _peak_equalize(a, end_time, seed, top_m=8):
    """
    Iteratively reduce the spread of top peaks in the convolution.
    Uses projected gradient on variance of top-m peaks.
    """
    a = _project_simplex(a.copy())
    n = a.size
    Lscore = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score(best)
    rng = np.random.default_rng(seed)

    lr = 0.002
    patience = 0
    last_improve = time.monotonic()

    while time.monotonic() < end_time - 5.0:
        c = _conv(a)
        # Identify top m peaks
        if c.size > top_m:
            top_idx = np.argsort(c)[::-1][:top_m]
        else:
            top_idx = np.arange(c.size)
        top_vals = c[top_idx]
        mean_top = float(np.mean(top_vals))
        # Weight: emphasize peaks that are above mean
        weights = np.maximum(top_vals - mean_top * 0.5, 0.0)
        wsum = float(weights.sum())
        if wsum < 1e-15:
            # All peaks are close; switch to reducing max directly
            top_idx = np.argmax(c)
            weights = np.zeros(c.size)
            weights[top_idx] = 1.0
            wsum = 1.0

        # Build weight vector over all lags
        w_full = np.zeros(c.size)
        w_full[top_idx] = weights / wsum

        # Gradient of sum(w_k * c_k) wrt a
        L = _pow2(3 * n - 2)
        A = np.fft.rfft(a, n=L)
        W = np.fft.rfft(w_full, n=L)
        g = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]

        # Project onto tangent of simplex
        g = g - float(np.mean(g))
        # Also ensure we don't push into negative territory too hard
        rms = float(np.sqrt(np.mean(g * g))) + 1e-12
        g = g / rms

        a_new = _project_simplex(a - lr * g)
        sc = _score(a_new)

        if np.isfinite(sc) and sc < best_score - 1e-14:
            best = a_new.copy()
            best_score = sc
            a = a_new
            patience = 0
            lr = min(0.01, lr * 1.03)
            last_improve = time.monotonic()
        else:
            patience += 1
            if patience % 10 == 0:
                lr = max(1e-5, lr * 0.85)
            if time.monotonic() - last_improve > 15.0:
                # Add small noise to escape
                a = _project_simplex(best * np.exp(rng.normal(0, 0.01, n)))
                lr = 0.003
                patience = 0
                last_improve = time.monotonic()

    return best, best_score


def _local_search(a, end_time, seed):
    """Adaptive block local search with simulated annealing."""
    a = _project_simplex(a.copy())
    n = a.size
    best = a.copy()
    best_score = _score(best)
    rng = np.random.default_rng(seed)
    sigma = 0.08
    T = 0.001  # annealing temperature
    last_improve = time.monotonic()
    start = last_improve

    while time.monotonic() < end_time - 2.0:
        elapsed = time.monotonic() - start
        T = max(1e-6, T * 0.9995)  # slow cooling

        cand = best.copy()
        nblocks = 1 + int(rng.integers(0, 2))
        for _ in range(nblocks):
            maxw = max(3, n // 10)
            w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            mult = np.exp(rng.normal(0.0, sigma, w))
            cand[s:s + w] *= mult

        cand = _project_simplex(cand)
        sc = _score(cand)

        if np.isfinite(sc):
            delta = sc - best_score
            accept = delta < 0 or (T > 1e-7 and rng.random() < np.exp(-delta / T))
            if accept:
                if sc < best_score:
                    best = cand
                    best_score = sc
                    last_improve = time.monotonic()
                    sigma = min(0.3, sigma * 1.04)
                a = cand
            else:
                if time.monotonic() - last_improve > 8.0:
                    sigma = max(0.01, sigma * 0.8)
                    if time.monotonic() - last_improve > 30.0:
                        a = _project_simplex(best * np.exp(rng.normal(0, 0.02, n)))
                        last_improve = time.monotonic()

    return best, best_score


def _refine(a0, end_time, seed):
    """Multi-stage refinement of one starting point."""
    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score(best)
    start = time.monotonic()
    total = max(1.0, end_time - start)
    if total < 5.0:
        return best, best_score

    x = _ratios(a)
    bounds = [(-100.0, 100.0)] * max(0, x.size)

    if _minimize is not None:
        # Stage 1: L2
        t1 = start + 0.12 * total
        while time.monotonic() < t1:
            x, st = _lbfgs(_q_obj_grad, x, (2.0,), 100, bounds)
            if st == 1:
                break
        a = _from_ratios(x)
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc

        # Stage 2: L4
        t2 = start + 0.25 * total
        if time.monotonic() < t2:
            while time.monotonic() < t2:
                x, st = _lbfgs(_q_obj_grad, x, (4.0,), 100, bounds)
                if st == 1:
                    break
            a = _from_ratios(x)
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best = a.copy()
                best_score = sc

        # Stage 3: L8
        t3 = start + 0.40 * total
        if time.monotonic() < t3:
            while time.monotonic() < t3:
                x, st = _lbfgs(_q_obj_grad, x, (8.0,), 100, bounds)
                if st == 1:
                    break
            a = _from_ratios(x)
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best = a.copy()
                best_score = sc

        # Stage 4: Soft-max with decreasing temp
        t4 = start + 0.70 * total
        if time.monotonic() < t4:
            c = _conv(best)
            cmax = float(c.max())
            eps = max(1e-6, cmax * 0.05)
            eps_min = max(1e-8, cmax * 0.001)
            stall = 0
            while time.monotonic() < t4:
                x, st = _lbfgs(_soft_obj_grad, x, (eps,), 80, bounds)
                if st == 1:
                    break
                a = _from_ratios(x)
                sc = _score(a)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best = a.copy()
                    best_score = sc
                    stall = 0
                    eps = max(eps_min, eps * 0.7)
                else:
                    stall += 1
                    if stall >= 3:
                        if eps <= eps_min * 1.01:
                            break
                        eps = max(eps_min, eps * 0.5)
                        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                        x = np.clip(x + rng.normal(0, 0.03, x.size), -100, 100)
                        stall = 0

        # Stage 5: Top-peak focused
        t5 = end_time - max(50.0, 0.15 * total)
        if time.monotonic() < t5:
            while time.monotonic() < t5:
                x, st = _lbfgs(_top_peak_grad_obj, x, (10,), 60, bounds)
                if st == 1:
                    break
                a = _from_ratios(x)
                sc = _score(a)
                if np.isfinite(sc) and sc < best_score:
                    best = a.copy()
                    best_score = sc

    # Peak equalization phase
    t_pe = end_time - max(60.0, 0.12 * total)
    if time.monotonic() < t_pe:
        a, sc = _peak_equalize(best, t_pe, seed + 313)
        if np.isfinite(sc) and sc < best_score:
            best = a
            best_score = sc

    # Final local search
    if time.monotonic() < end_time - 1.0:
        a, sc = _local_search(best, end_time, seed + 619)
        if np.isfinite(sc) and sc < best_score:
            best = a
            best_score = sc

    return best, best_score


def _try_length(n, a_template, end_time, seed):
    """Try to adapt a template to a different length n."""
    if a_template is None:
        a = _power_seq(n, 0.5, 1.0)
    else:
        # Resample template to length n
        m = a_template.size
        if m == n:
            a = a_template.copy()
        else:
            old_idx = np.linspace(0, 1, m)
            new_idx = np.linspace(0, 1, n)
            a = np.interp(new_idx, old_idx, a_template)
        a = _project_simplex(a)
    sc = _score(a)
    return a, sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Return the best sequence found before the time budget.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s) - 10.0)
    deadline = t0 + budget

    # Decide length: try current best length and a few others
    n_base = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n_base = int(arr.size)
    except Exception:
        pass

    # Try a few lengths; the score formula has 2n in numerator,
    # so larger n can help if we can keep max(a*a)/n small enough.
    candidate_ns = [n_base]
    # Only try other lengths if they're in a reasonable range
    for n_try in (n_base // 2, n_base * 2, 5000, 7600):
        if 200 <= n_try <= 20000 and n_try not in candidate_ns:
            candidate_ns.append(n_try)

    # Quick score for each length with a default initialization
    length_scores = []
    for n in candidate_ns:
        a = _power_seq(n, 0.5, 1.0)
        sc = _score(a)
        length_scores.append((sc, n))
        print(f"n={n}: quick score = {sc:.6f}")

    # Sort by score (lower is better) - pick the best length
    length_scores.sort()
    best_n = length_scores[0][1]
    print(f"Selected n = {best_n} (score {length_scores[0][0]:.6f})")

    # Generate candidates for the best length
    cands = _generate_cands(best_n)
    scored = []
    for i, a in enumerate(cands):
        sc = _score(a)
        if np.isfinite(sc):
            scored.append((sc, i, a))
            if i < 5 or sc < 1.55:
                print(f"  cand {i}: score = {sc:.8f}")
    scored.sort()

    if not scored:
        a = _power_seq(best_n, 0.5, 1.0)
        return [float(x) for x in a]

    global_best_a = np.asarray(scored[0][2], dtype=np.float64).copy()
    global_best_score = float(scored[0][0])
    print(f"Best initial candidate score: {global_best_score:.8f}")

    # Allocate time among top candidates
    top_k = min(4, len(scored))
    top = scored[:top_k]

    # Time allocation: give more time to better candidates
    fracs = [0.50, 0.22, 0.14, 0.14]
    used_time = 0.0

    for i in range(top_k):
        if time.monotonic() >= deadline - 30.0:
            break
        frac = fracs[i] if i < len(fracs) else 0.1
        alloc_end = min(deadline, t0 + (used_time + frac) * budget)
        if alloc_end > time.monotonic() + 10.0:
            a_init = np.asarray(top[i][2], dtype=np.float64).copy()
            print(f"\nRefining candidate {i} (init score {top[i][0]:.8f}), "
                  f"time budget {alloc_end - time.monotonic():.1f}s")
            a, sc = _refine(a_init, alloc_end, int(seed) + 100 * (i + 1))
            if np.isfinite(sc) and sc < global_best_score:
                global_best_score = sc
                global_best_a = np.asarray(a, dtype=np.float64).copy()
                print(f"  New global best: {sc:.8f}")
        used_time += frac

    # Final polish
    if time.monotonic() < deadline - 5.0:
        print(f"\nFinal local search, time remaining: {deadline - time.monotonic():.1f}s")
        a, sc = _local_search(global_best_a, deadline - 2.0, int(seed) + 9999)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best_a = np.asarray(a, dtype=np.float64).copy()
            print(f"  After local search: {sc:.8f}")

    global_best_a = _project_simplex(global_best_a)
    final_score = _score(global_best_a)
    print(f"\nFinal score: {final_score:.8f}")

    # Validate
    out = [float(x) for x in global_best_a]
    try:
        val = evaluate_sequence(out)
        if not np.isfinite(val):
            print("Warning: evaluate_sequence returned non-finite, using fallback")
            out = [float(x) for x in _power_seq(len(out), 0.5, 1.0)]
        else:
            print(f"Official evaluate_sequence: {val:.8f}")
    except Exception as e:
        print(f"Warning: evaluate_sequence failed: {e}")

    return out
