    ca = np.fft.irfft(fa * fa, fft_size)

def _softmax_gradient(a, alpha):

    Compute gradient of the log-sum-exp smooth approximation of max(conv(a,a)).
    Returns (gradient, soft_value, true_max_conv).

    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    exp_sum = exp_c.sum()
    if exp_sum < 1e-300:

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses smooth mirror descent (log-sum-exp approximation of max) with adaptive
temperature, Nesterov momentum, and backtracking line search. Combines
multiple diverse initializations with perturbation-based local refinement
and basin-hopping restarts to escape local minima.
"""

import numpy as np
import time


def _next_pow2(n):
    """Return the next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    """Fast linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _fft_convolve2(a, b):
    """Linear convolution of a and b using FFT."""
    total_len = len(a) + len(b) - 1
    fft_size = _next_pow2(total_len)
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    c = np.fft.irfft(fa * fb, fft_size)
    return c[:total_len]


def _softmax_gradient(a, alpha):
    """
    Compute gradient of (1/alpha)*log(sum_k exp(alpha*c_k)) where c = conv(a,a).
    Returns (gradient, soft_value, true_max_c).
    """
    n = len(a)
    c = _fft_convolve(a)

    # Numerically stable softmax
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    exp_sum = exp_c.sum()
    if exp_sum < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_c / exp_sum

    # grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    # = 2 * (p * flip(a))[j + n - 1]  (linear convolution indexing)
    flip_a = a[::-1].copy()
    conv = _fft_convolve2(p, flip_a)
    grad = 2.0 * conv[n - 1: 2 * n - 1]

    soft_val = (c_max + np.log(max(exp_sum, 1e-300))) / alpha
    return grad, soft_val, float(c_max)


def _mirror_descent_smooth(a0, time_budget, rng, alpha_start=5.0, alpha_end=300.0, name=""):
    """
    Smooth mirror descent with adaptive alpha, momentum, and backtracking.
    Normalizes a to sum 1, minimizes 2n*max(conv(a,a)).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()

    c0 = _fft_convolve(a)
    best_score = 2.0 * n * float(c0.max())
    best_a = a.copy()

    deadline = time.monotonic() + time_budget
    eta = 1.0
    mu = 0.85
    v = np.zeros(n)
    it = 0
    no_improve = 0
    last_print = time.monotonic()
    stagnation_count = 0

    while time.monotonic() < deadline:
        # Adaptive alpha (exponential ramp over iterations)
        progress = min(1.0, it / 3000.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        # Compute soft gradient
        grad, soft_val, true_max = _softmax_gradient(a, alpha)
        score = 2.0 * n * true_max

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            stagnation_count = 0
            if time.monotonic() - last_print > 5.0:
                print(f"    [{name}] it={it}, score={score:.7f}, alpha={alpha:.1f}, eta={eta:.4f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
            if no_improve > 100:
                stagnation_count += 1

        # If severely stuck, restart from best with perturbation
        if stagnation_count > 5:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)

            no_improve = 0
            stagnation_count = 0
            it += 1
            continue

        if no_improve > 400:
            eta = max(eta * 0.96, 0.05)
        elif no_improve > 150:
            eta = max(eta * 0.99, 0.1)
        elif no_improve < 30:
            eta = min(eta * 1.005, 3.0)

        # Gradient normalization and momentum
        g_max = grad.max()
        if g_max < 1e-15:
            g_max = 1.0
        g_norm = grad / g_max
        v = mu * v + (1.0 - mu) * g_norm

        # Backtracking line search
        accepted = False
        for bt in range(10):
            a_new = a * np.exp(-eta * v)
            a_new = np.maximum(a_new, 1e-15)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            c_new = _fft_convolve(a_new)
            new_score = 2.0 * n * float(c_new.max())
            if new_score <= score + 1e-10:
                a = a_new
                accepted = True
                break
            eta *= 0.5

        if not accepted:
            # Too stuck, reset
            a = best_a.copy()
            v = np.zeros(n)
            eta = max(eta, 0.05)

        it += 1

    return best_a, best_score


def _hard_subgradient_descent(a0, time_budget, rng, name=""):
    """
    Hard subgradient descent (focus on single max lag) with momentum.
    Complements the smooth approach.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()

    c0 = _fft_convolve(a)
    best_score = 2.0 * n * float(c0.max())
    best_a = a.copy()

    deadline = time.monotonic() + time_budget
    eta = 0.3
    mu = 0.7
    v = np.zeros(n)
    it = 0
    no_improve = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 5.0:
                print(f"    [{name}] it={it}, score={score:.7f}, eta={eta:.4f}, k_max={k_max}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 600:
            a = best_a * np.exp(rng.normal(0, 0.06, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)
            eta = 0.3
            no_improve = 0
            it += 1
            continue

        # Subgradient of c[k_max]: g[j] = 2*a[k_max-j] if 0 <= k_max-j < n
        g = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            g[lo:hi + 1] = 2.0 * a[k_max - j_range]

        g_max = g.max()
        if g_max < 1e-15:
            break
        g_norm = g / g_max
        v = mu * v + (1.0 - mu) * g_norm

        a = a * np.exp(-eta * v)
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
        else:
            a = a / s

        if no_improve > 300:
            eta *= 0.97
        elif no_improve < 50:
            eta = min(eta * 1.002, 1.0)

        it += 1

    return best_a, best_score


def _perturb_refine(a0, time_budget, rng, name=""):
    """
    Local refinement using random perturbations with accept/reject (simulated annealing).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    c = _fft_convolve(a)
    best_score = 2.0 * n * float(c.max())
    best_a = a.copy()

    deadline = time.monotonic() + time_budget
    temp = 0.03
    it = 0
    last_print = time.monotonic()
    eval_count = 0

    while time.monotonic() < deadline:
        pert_type = rng.integers(0, 5)
        a_trial = a.copy()

        if pert_type == 0:
            # Gaussian on random subset
            k = int(rng.integers(5, max(6, n // 10)))
            idx = rng.choice(n, size=k, replace=False)
            scale = temp * max(a.mean(), 0.01)
            a_trial[idx] += rng.normal(0, scale, k)
        elif pert_type == 1:
            # Block multiplicative
            block = int(rng.integers(1, max(2, n // 30)))
            start = int(rng.integers(0, n - block))
            a_trial[start:start + block] *= np.exp(rng.normal(0, temp * 2, block))
        elif pert_type == 2:
            # Pairwise swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = min(a[i], a.mean() * 0.5) * rng.uniform(0.2, 0.8)
                a_trial[i] -= amt
                a_trial[j] += amt
        elif pert_type == 3:
            # Global smooth perturbation (low-frequency)
            freq = rng.uniform(0.5, 3.0)
            x = np.arange(n)
            delta = temp * 0.5 * np.sin(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
            a_trial *= np.exp(delta)
        else:
            # Single coordinate nudge
            j = int(rng.integers(0, n))
            a_trial[j] *= np.exp(rng.normal(0, temp * 3))

        a_trial = np.maximum(a_trial, 0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            continue
        a_trial = a_trial / s_trial

        c_trial = _fft_convolve(a_trial)
        trial_score = 2.0 * n * float(c_trial.max())
        eval_count += 1

        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            a = a_trial.copy()
            temp = max(temp * 0.9995, 0.002)
        else:
            # Simulated annealing acceptance
            delta = trial_score - best_score
            if delta < 0.01 and rng.random() < np.exp(-delta / (temp * 0.05)):
                a = a_trial.copy()

        it += 1
        if it % 5000 == 0 and time.monotonic() - last_print > 10.0:
            print(f"    [{name}] pert it={it}, best={best_score:.7f}, temp={temp:.5f}")
            last_print = time.monotonic()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    
    Strategy: Smooth mirror descent (log-sum-exp approximation) with adaptive
    temperature and momentum, combined with hard subgradient descent, diverse
    initializations, and perturbation-based local refinement with simulated
    annealing. Basin-hopping restarts help escape local minima.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 15.0

    best_overall = None
    best_overall_score = np.inf

    # Build diverse initializations
    candidates = []

    # 1. Previous best solution
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 50 and hs.sum() > 0.01:
            candidates.append(("prev_best", hs.copy()))
    except (NameError, Exception):
        pass

    n_main = 2000

    # 2. Sinusoidal variants (different frequencies, amplitudes, phases)
    for n in [n_main, 2500, 3000]:
        x = np.arange(n, dtype=np.float64)
        for freq, amp, phase in [
            (1, 0.4, np.pi / 4),
            (1, 0.6, 0.0),
            (1, 0.5, np.pi / 2),
            (2, 0.3, np.pi / 3),
            (2, 0.4, 0.0),
            (3, 0.25, 1.0),
        ]:
            a = 1.0 + amp * np.sin(2 * np.pi * freq * x / n + phase)
            a = np.maximum(a, 0.05)
            candidates.append((f"sin_n{n}_f{freq}_a{amp:.1f}_p{phase:.1f}", a))

    # 3. Cosine shapes (W and M)
    for n in [n_main, 2500]:
        x = np.arange(n, dtype=np.float64)
        t = x / n
        a = 1.0 - 0.6 * np.cos(np.pi * t)
        a = np.maximum(a, 0.05)
        candidates.append((f"wshape_n{n}", a))

        a = 1.0 + 0.6 * np.cos(np.pi * t)
        candidates.append((f"mshape_n{n}", a))

    # 4. Multi-peak (two sine peaks)
    for n in [n_main]:
        x = np.arange(n, dtype=np.float64)
        t = x / n
        a = 1.0 + 0.5 * np.sin(2 * np.pi * t) + 0.3 * np.sin(4 * np.pi * t + 0.8)
        a = np.maximum(a, 0.05)
        candidates.append((f"multipeak_n{n}", a))

    # 5. Exponential / asymmetric
    for n in [n_main, 2500]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-2.0 * x / n) + 0.15
        candidates.append((f"expdecay_n{n}", a))

    # 6. Quadratic U-shape
    for n in [n_main]:
        x = np.arange(n, dtype=np.float64) / n
        a = 1.0 + 1.0 * (2 * x - 1) ** 2
        candidates.append((f"uquad_n{n}", a))

    # 7. Cubic S-shape
    for n in [n_main]:
        x = np.arange(n, dtype=np.float64) / n
        a = 1.0 + 0.7 * ((2 * x - 1) ** 3)
        a = np.maximum(a, 0.05)
        candidates.append((f"cubic_n{n}", a))

    print(f"Total candidates: {len(candidates)}")

    # Quick initial evaluation to rank
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        c0 = _fft_convolve(a0)
        init_score = 2.0 * len(a0) * float(c0.max())
        scored.append((init_score, name, a0))

    scored.sort(key=lambda x: x[0])
    print("Initial scores (sorted, top 12):")
    for i, (sc, name, a) in enumerate(scored[:12]):
        print(f"  {i + 1}. {name} (n={len(a)}): {sc:.6f}")

    # Allocate time: top candidates get more time
    n_top = min(len(scored), 10)
    remaining_time = budget_s - 40.0

    # Geometric weighting: best gets most time
    weights = [max(1.0, 1.4 ** (n_top - 1 - i)) for i in range(n_top)]
    total_w = sum(weights)
    times = [remaining_time * w / total_w for w in weights]

    for i in range(n_top):
        if time.monotonic() > deadline:
            break

        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(15.0, deadline - time.monotonic()))

        print(f"\n--- Optimizing '{name}' (n={len(a0)}, init={init_score:.6f}, time={t_alloc:.0f}s) ---")

        # Phase 1: Smooth mirror descent (50% of time)
        t1 = t_alloc * 0.45
        a1, s1 = _mirror_descent_smooth(a0, t1, rng,
                                         alpha_start=5.0, alpha_end=300.0, name=name + "_smooth")
        eval1 = evaluate_sequence(a1.tolist())
        print(f"  Smooth MD: score={s1:.7f}, eval={eval1:.7f}")

        if np.isfinite(eval1) and eval1 < best_overall_score:
            best_overall_score = eval1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {eval1:.7f} ***")

        best_phase1 = a1
        best_phase1_score = s1

        # Phase 2: Hard subgradient from phase 1 result (25% of time)
        t2 = t_alloc * 0.25
        a2, s2 = _hard_subgradient_descent(a1, t2, rng, name=name + "_hard")
        eval2 = evaluate_sequence(a2.tolist())
        print(f"  Hard SG: score={s2:.7f}, eval={eval2:.7f}")

        if np.isfinite(eval2) and eval2 < best_overall_score:
            best_overall_score = eval2
            best_overall = a2.copy()
            print(f"  *** NEW BEST: {eval2:.7f} ***")

        best_phase2 = a2 if s2 < best_phase1_score else best_phase1

        # Phase 3: Perturbation refinement (30% of time)
        t3 = t_alloc * 0.30
        a3, s3 = _perturb_refine(best_phase2, t3, rng, name=name + "_pert")
        eval3 = evaluate_sequence(a3.tolist())
        print(f"  Perturb: score={s3:.7f}, eval={eval3:.7f}")

        if np.isfinite(eval3) and eval3 < best_overall_score:
            best_overall_score = eval3
            best_overall = a3.copy()
            print(f"  *** NEW BEST: {eval3:.7f} ***")

    # Final polishing phase
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\n=== Final polishing ({remaining:.0f}s remaining) ===")

        # Polishing 1: Smooth MD with high alpha (focus on peak)
        t_p1 = remaining * 0.35
        a_p1, s_p1 = _mirror_descent_smooth(best_overall, t_p1, rng,
                                             alpha_start=50.0, alpha_end=1000.0,
                                             name="polish_smooth_high")
        eval_p1 = evaluate_sequence(a_p1.tolist())
        print(f"  Polish smooth (high alpha): {eval_p1:.7f}")
        if np.isfinite(eval_p1) and eval_p1 < best_overall_score:
            best_overall_score = eval_p1
            best_overall = a_p1.copy()
            print(f"  *** NEW BEST: {eval_p1:.7f} ***")

        # Polishing 2: Hard subgradient
        t_p2 = remaining * 0.25
        a_p2, s_p2 = _hard_subgradient_descent(best_overall, t_p2, rng, name="polish_hard")
        eval_p2 = evaluate_sequence(a_p2.tolist())
        print(f"  Polish hard: {eval_p2:.7f}")
        if np.isfinite(eval_p2) and eval_p2 < best_overall_score:
            best_overall_score = eval_p2
            best_overall = a_p2.copy()
            print(f"  *** NEW BEST: {eval_p2:.7f} ***")

        # Polishing 3: Fine perturbation
        t_p3 = remaining * 0.40
        a_p3, s_p3 = _perturb_refine(best_overall, t_p3, rng, name="polish_pert")
        eval_p3 = evaluate_sequence(a_p3.tolist())
        print(f"  Polish perturb: {eval_p3:.7f}")
        if np.isfinite(eval_p3) and eval_p3 < best_overall_score:
            best_overall_score = eval_p3
            best_overall = a_p3.copy()
            print(f"  *** NEW BEST: {eval_p3:.7f} ***")

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\n=== FINAL: score={final_score:.7f}, n={len(best_overall)}, time={elapsed:.1f}s ===")
        return [float(x) for x in best_overall]
    else:
        a = np.ones(1000, dtype=np.float64)
        print("FALLBACK: uniform sequence")
        return a.tolist()
