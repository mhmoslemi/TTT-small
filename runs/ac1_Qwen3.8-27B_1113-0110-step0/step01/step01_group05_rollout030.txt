```python
"""
Minimizer for 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse structured initializations (skewed bumps, geometric, level sets)
2. Subgradient descent with Nesterov momentum in multiplicative (exponential) form
3. Peak-targeted coordinate descent for local refinement
4. Basin hopping with structured perturbations to escape local minima
5. Final coordinate descent polish
"""

import numpy as np
import time

def _clean(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    if a.size == 0:
        a = np.ones(1)
    return a

def _norm1(a):
    a = _clean(a)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(a) / a.size
    return a / s

def _score(a):
    """Score assuming sum(a) = 1: 2n * max(conv(a,a))"""
    n = a.size
    if n == 0:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * n * float(np.max(c))

def _score_general(a):
    """Score for general a: 2n * max(conv(a,a)) / (sum(a))^2"""
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return float('inf')
    n = a.size
    c = np.convolve(a, a)
    return 2.0 * n * float(np.max(c)) / (s * s)

def _subgradient(a, peak_frac=0.95):
    """Compute subgradient of 2n*max(conv(a,a)) w.r.t. a (normalized)."""
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return np.zeros(n)
    # Active set: positions where c[k] >= peak_frac * M
    A = np.where(c >= peak_frac * M)[0]
    # grad[i] = sum_{k in A} a[k-i] (a[j]=0 outside [0,n-1])
    # = full_conv[2n-2-i] where full_conv = convolve(a, ind_A[::-1])
    ind = np.zeros(2 * n - 1)
    ind[A] = 1.0
    full = np.convolve(a, ind[::-1])
    # full has length 3n-2
    start = 2 * n - 2
    # grad[i] = full[start - i] for i = 0, ..., n-1
    grad = full[start - np.arange(n)]
    # Multiply by 4n/|A| for the full subgradient of 2n*max(c)
    grad *= 4.0 * n / len(A)
    return grad, M, A

def _subgradient_step(a, eta, peak_frac=0.95):
    """One multiplicative subgradient step."""
    grad, M, A = _subgradient(a, peak_frac)
    a_new = a * np.exp(-eta * grad)
    s = a_new.sum()
    if s <= 0 or not np.isfinite(s):
        return a, M
    a_new /= s
    c_new = np.convolve(a_new, a_new)
    M_new = float(np.max(c_new))
    return a_new, M_new

def _subgradient_optimize(a, deadline, peak_frac=0.95, eta0=0.5, momentum=0.9):
    """Subgradient descent with momentum (Nesterov-like)."""
    a = _norm1(a.copy())
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    
    v = np.zeros(n)  # momentum
    eta = eta0
    no_improve = 0
    
    while time.monotonic() < deadline - 0.5:
        grad, M_curr, A = _subgradient(a, peak_frac)
        
        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad
        
        # Adaptive step size
        a_new = a * np.exp(-eta * v)
        s = a_new.sum()
        if s <= 0:
            eta *= 0.5
            continue
        a_new /= s
        M_new = float(np.max(np.convolve(a_new, a_new)))
        
        if M_new < M - 1e-10:
            a = a_new
            M = M_new
            eta = min(eta * 1.05, 2.0)  # increase slightly
            no_improve = 0
            if M < best_M:
                best_M = M
                best_a = a.copy()
        else:
            # Rejection: try smaller step
            eta *= 0.7
            no_improve += 1
            if no_improve > 50:
                # Reset and reduce peak_frac to include more coordinates
                peak_frac = max(0.5, peak_frac - 0.05)
                eta = eta0 * 0.5
                no_improve = 0
                v = np.zeros(n)
                a = best_a.copy()
                M = best_M
        
        # Periodic coordinate descent polish
        if no_improve > 30 and time.monotonic() < deadline - 2.0:
            a = _coord_polish(a, deadline - 1.0, n)
            c = np.convolve(a, a)
            M = float(np.max(c))
            if M < best_M:
                best_M = M
                best_a = a.copy()
            no_improve = 0
    
    return best_a, best_M

    c = np.convolve(a, a)
    M = float(np.max(c))
    
    grid = np.concatenate(([0.0], np.logspace(-6, 1, grid_size)))
    
    improved = True
    while improved and time.monotonic() < deadline:
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old = a[i]
            D = 1.0 - old
            # New c after changing a[i] to x:
            # c_new[k] = c[k] - 2*old*a[k-i] + x*(a[k-i] + a[i-(k-i)]) + x^2 if k-i==i
            # Simplified: c_new[k] = c[k] + (x-old)*(2*a[k-i] - old*delta(k,2i)) ... 
            # Actually let me just recompute the affected part
            # c[k] = sum_j a[j]*a[k-j]
            # Change a[i] from old to x:
            # delta_c[k] = (x - old) * (a[k-i] + a[k-i]) - old*(x-old)*delta[k, 2i]
```

"""
Minimizer for 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse structured initializations (skewed polynomial, binomial-like, geometric)
2. Multiplicative subgradient descent with momentum (new core optimizer)

4. Basin hopping with structured perturbations
5. Smooth interpolation to expand prototypes

import numpy as np
import time


def _clean(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    if a.size == 0:
        a = np.ones(1)
    return a


def _norm1(a):
    a = _clean(a)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(a) / max(a.size, 1)
    return a / s


def _score_norm(a):
    """Score for sum(a)=1: returns 2n*max(conv(a,a))."""
    n = a.size
    if n == 0:
        return float('inf')
    c = np.convolve(a, a)
    mx = float(np.max(c))
    return 2.0 * n * mx


def _score_general(a):
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return float('inf')
    n = a.size
    c = np.convolve(a, a)
    return 2.0 * n * float(np.max(c)) / (s * s)


def _subgradient(a, peak_frac=0.98):
    """
    Compute subgradient of 2n*max(conv(a,a)) w.r.t. a, assuming sum(a)=1.
    Returns (grad, peak_value, active_set_size).
    """
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return np.zeros(n), M, 1
    thresh = peak_frac * M
    A = np.where(c >= thresh)[0]
    n_A = len(A)
    # grad[i] = sum_{k in A} a[k-i] (a[j]=0 outside [0,n-1])
    # = full[2n-2-i] where full = convolve(a, ind_A[::-1])
    ind = np.zeros(2 * n - 1)
    ind[A] = 1.0
    full = np.convolve(a, ind[::-1])
    start = 2 * n - 2
    grad = full[start - np.arange(n)]
    # Full subgradient of 2n*max(c) is 2n * (1/|A|) * sum_{k in A} 2*a[k-i]
    # = 4n/|A| * grad
    grad *= 4.0 * n / n_A
    return grad, M, n_A


def _subgrad_optimize(a, deadline, peak_frac=0.98, eta0=0.3, momentum=0.85,
                      coord_polish_interval=50):
    """
    Multiplicative subgradient descent with Nesterov momentum.
    Works on normalized sequence (sum=1).
    Returns (best_a, best_score).
    """
    a = _norm1(a.copy())
    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    score = 2.0 * n * M
    best_a = a.copy()
    best_score = score

    v = np.zeros(n)
    eta = eta0
    no_improve = 0
    step_count = 0

    while time.monotonic() < deadline - 0.5:
        grad, M_curr, n_A = _subgradient(a, peak_frac)
        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad

        # Multiplicative update
        scale = eta * v
        # Clip to prevent underflow/overflow
        scale = np.clip(scale, -50.0, 50.0)
        a_new = a * np.exp(-scale)
        np.maximum(a_new, 1e-18, out=a_new)
        s = a_new.sum()
        if s <= 0 or not np.isfinite(s):
            eta *= 0.5
            no_improve += 1
            continue
        a_new /= s

        c_new = np.convolve(a_new, a_new)
        M_new = float(np.max(c_new))
        score_new = 2.0 * n * M_new

        if score_new < score - 1e-12:
            a = a_new
            score = score_new
            eta = min(eta * 1.02, 3.0)
            no_improve = 0
            if score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            eta *= 0.8
            no_improve += 1
            if no_improve > 80:

                peak_frac = min(0.999, peak_frac + 0.01)
                if peak_frac > 0.999:
                    peak_frac = 0.95
                    no_improve = 0
                # Revert to best
                a = best_a.copy()
                score = best_score
                no_improve = 0

        step_count += 1
        # Periodic coordinate polish

            sc2 = _score_norm(a2)

            k_hi = min(k_hi, i + n - 1)

            # a[k-i] for k in [k_lo, k_hi]
            # k-i ranges from k_lo-i to k_hi-i, clamped to [0, n-1]
            j_lo = max(0, k_lo - i)
            j_hi = min(n - 1, k_hi - i)
            a_seg = a[j_lo:j_hi + 1]

            c_seg = c[k_lo:k_hi + 1].copy()

            # Unaffected max
            max_unaff = 0.0
            if k_lo > 0:
                m1 = float(np.max(c[:k_lo]))
                if m1 > max_unaff:
                    max_unaff = m1
            if k_hi < 2 * n - 2:
                m2 = float(np.max(c[k_hi + 1:]))
                if m2 > max_unaff:
                    max_unaff = m2

            # For each candidate x (new value of a[i]):
            # c_new[k] = c[k] + 2*(x-old)*a[k-i] for non-diagonal
            #            c[k] + (x^2 - old^2) for diagonal (k = 2i)
            diag_pos = 2 * i
            has_diag = (k_lo <= diag_pos <= k_hi)
            diag_idx = diag_pos - k_lo if has_diag else -1

            # Vectorized: compute max of affected c for each x
            # aff[j] for x: c_seg[j] + 2*(x-old)*a_seg[j] (non-diag)
            #               c_seg[diag_idx] + x^2 - old^2 (diag)
            base = c_seg - 2.0 * old * a_seg  # constant part
            # For non-diag: base[j] + 2*x*a_seg[j]
            # For diag: c_seg[diag_idx] + x^2 - old^2

            if has_diag:
                # Handle diagonal separately
                base_non_diag = base.copy()
                base_non_diag[diag_idx] = 0  # will overwrite
                a_non_diag = a_seg.copy()
                a_non_diag[diag_idx] = 0

                # Max of non-diag: max_j (base_non_diag[j] + 2*x*a_non_diag[j])
                # This is max of affine functions in x
                # For efficiency, just evaluate on grid
                aff = base[None, :] + 2.0 * grid[:, None] * a_seg[None, :]
                if has_diag:
                    aff[:, diag_idx] = c_seg[diag_idx] + grid * grid - old * old
                max_aff = np.max(aff, axis=1)
            else:
                aff = base[None, :] + 2.0 * grid[:, None] * a_seg[None, :]
                max_aff = np.max(aff, axis=1)

            max_total = np.maximum(max_aff, max_unaff)
            S_new = D + grid
            vals = np.full(grid.size, float('inf'))
            valid = S_new > 1e-15
            vals[valid] = 2.0 * n * max_total[valid] / (S_new[valid] * S_new[valid])

            idx = int(np.argmin(vals))
            if vals[idx] < score - 1e-13:
                x = float(grid[idx])
                a[i] = x
                c[k_lo:k_hi + 1] = c_seg + 2.0 * (x - old) * a_seg
                if has_diag:
                    c[k_lo + diag_idx] = c_seg[diag_idx] + x * x - old * old
                np.maximum(c, 0.0, out=c)
                s_new = D + x
                if s_new > 1e-15:
                    a /= s_new
                    c /= (s_new * s_new)
                    score = 2.0 * n * float(np.max(c))
                    improved_any = True

    return a


def _make_initial(n, strategy, rng):
    """Generate diverse initial sequences."""
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n))
    st = int(strategy) % 20
    if st == 0:
        p = np.ones(n)
    elif st == 1:
        p = rng.dirichlet(np.full(n, 1.0))
    elif st == 2:
        p = rng.dirichlet(np.full(n, 0.3))
    elif st == 3:
        # Right-skewed polynomial
        x = np.arange(n, dtype=np.float64) / (n - 1)
        p = x * (1 - x) * (0.5 + x)  # skewed right
        p = np.maximum(p, 1e-6)
    elif st == 4:
        # Left-skewed polynomial
        x = np.arange(n, dtype=np.float64) / (n - 1)
        p = x * (1 - x) * (0.5 + (1 - x))  # skewed left
        p = np.maximum(p, 1e-6)
    elif st == 5:
        # Strongly right-skewed
        x = np.arange(n, dtype=np.float64) / (n - 1)
        p = x**2 * (1 - x)
        p = np.maximum(p, 1e-8)
    elif st == 6:
        # Strongly left-skewed
        x = np.arange(n, dtype=np.float64) / (n - 1)
        p = x * (1 - x)**2
        p = np.maximum(p, 1e-8)
    elif st == 7:
        # Geometric decreasing
        r = rng.uniform(0.95, 0.995)
        p = r ** np.arange(n)
    elif st == 8:
        # Geometric increasing
        r = rng.uniform(0.95, 0.995)
        p = r ** (n - 1 - np.arange(n))
    elif st == 9:
        # Skewed Gaussian (right)
        x = np.arange(n, dtype=np.float64) / (n - 1)
        mu = rng.uniform(0.5, 0.7)
        sigma = rng.uniform(0.1, 0.2)
        p = np.exp(-((x - mu) / sigma) ** 2)
        p = np.maximum(p, 1e-10)
    elif st == 10:
        # Skewed Gaussian (left)
        x = np.arange(n, dtype=np.float64) / (n - 1)
        mu = rng.uniform(0.3, 0.5)
        sigma = rng.uniform(0.1, 0.2)
        p = np.exp(-((x - mu) / sigma) ** 2)
        p = np.maximum(p, 1e-10)
    elif st == 11:
        # Two-level: first 60% high, last 40% low
        p = np.zeros(n)
        m = int(0.6 * n)
        p[:m] = 1.0
        p[m:] = 0.3
    elif st == 12:
        # Two-level: first 70% high, last 30% low
        p = np.zeros(n)
        m = int(0.7 * n)
        p[:m] = 1.0
        p[m:] = 0.2
    elif st == 13:
        # Sine bump shifted right
        x = np.arange(n, dtype=np.float64) / (n - 1)
        p = np.sin(np.pi * x ** 0.7)
        p = np.maximum(p, 1e-8)
    elif st == 14:
        # Sine bump shifted left
        x = np.arange(n, dtype=np.float64) / (n - 1)
        p = np.sin(np.pi * (1 - x) ** 0.7)
        p = np.maximum(p, 1e-8)
    elif st == 15:
        # Power-law right: a[i] ~ (i+1)^(-alpha)
        alpha = rng.uniform(0.3, 0.8)
        p = (np.arange(n) + 1) ** (-alpha)
    elif st == 16:
        # Power-law left: a[i] ~ (n-i)^(-alpha)
        alpha = rng.uniform(0.3, 0.8)
        p = (n - np.arange(n)) ** (-alpha)
    elif st == 17:
        # Beta-like distribution (skewed)
        a_beta = rng.uniform(1.5, 4.0)
        b_beta = rng.uniform(1.5, 4.0)
        x = (np.arange(n) + 0.5) / n
        p = x ** (a_beta - 1) * (1 - x) ** (b_beta - 1)
        p = np.maximum(p, 1e-10)
    elif st == 18:
        # Exponential with reflected ends
        x = np.arange(n, dtype=np.float64) / (n - 1)
        p = np.exp(-3.0 * (x - 0.6)) * np.exp(-10.0 * x * (1 - x) ** 3)
        p = np.maximum(p, 1e-8)
    else:
        # Random with structure
        x = np.arange(n, dtype=np.float64) / (n - 1)
        p = np.exp(rng.normal(0, 0.5, n)) * (0.5 + 0.5 * x)
        p = np.maximum(p, 1e-8)
    return _norm1(p)


def _interpolate_smooth(a, target_len):
    """Smoothly interpolate a sequence to target length using cubic-like blending."""
    n = a.size
    if n == target_len:
        return a.copy()
    if n == 1:
        return np.full(target_len, a[0] / target_len)
    # Map target indices to source positions
    xs = np.arange(target_len, dtype=np.float64)
    src_pos = xs * (n - 1) / (target_len - 1) if target_len > 1 else np.array([0.0])
    idx0 = np.floor(src_pos).astype(int)
    idx1 = np.minimum(idx0 + 1, n - 1)
    frac = src_pos - idx0
    # Linear interpolation (safe, no overshoot for non-negative data)
    out = a[idx0] * (1 - frac) + a[idx1] * frac
    np.maximum(out, 0.0, out=out)
    return _norm1(out)


def _perturb(a, rng, strength=0.3):
    """Apply a structured perturbation to escape local minima."""
    a = a.copy()
    n = a.size
    mode = int(rng.integers(0, 5))
    if mode == 0:
        # Multiplicative noise
        a = a * np.exp(rng.normal(0, strength, n))
    elif mode == 1:
        # Block swap
        block_len = rng.integers(max(1, n // 10), max(2, n // 3))
        i = rng.integers(0, n - block_len)
        j = rng.integers(0, n - block_len)
        if i != j:
            tmp = a[i:i + block_len].copy()
            a[i:i + block_len] = a[j:j + block_len]
            a[j:j + block_len] = tmp
    elif mode == 2:
        # Add sinusoidal perturbation
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n
        a = a * (1 + strength * np.sin(2 * np.pi * freq * x + phase))
    elif mode == 3:
        # Global shift of center of mass
        shift = rng.uniform(-0.2, 0.2)
        x = np.arange(n, dtype=np.float64) / (n - 1)
        x_new = np.clip(x + shift, 0, 1)
        # Bilinear resampling
        idx0 = (x_new * (n - 1)).astype(int)
        idx1 = np.minimum(idx0 + 1, n - 1)
        frac = x_new * (n - 1) - idx0
        a = a[idx0] * (1 - frac) + a[idx1] * frac
    else:
        # Reverse with noise
        a = a[::-1] * np.exp(rng.normal(0, strength * 0.5, n))
    np.maximum(a, 0.0, out=a)
    return _norm1(a)


def _combine_optimizations(a, deadline, rng, tag=""):
    """
    Run subgradient + coordinate descent in an alternating fashion.
    """
    a = _norm1(a.copy())
    n = a.size
    sc = _score_norm(a)
    best_a = a.copy()
    best_sc = sc
    print(f"  [{tag}] Start: score={sc:.6f}")

    phase_time = 20.0
    phase = "subgrad"
    phase_deadline = time.monotonic() + phase_time
    iter_count = 0

    while time.monotonic() < deadline - 2.0:
        if time.monotonic() >= phase_deadline:
            # Switch phase
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"  [{tag}] Phase {iter_count} done: score={best_sc:.6f}")
            if phase == "subgrad":
                phase = "coord"
                phase_deadline = time.monotonic() + min(phase_time, max(3.0, (deadline - time.monotonic()) * 0.3))
            else:
                phase = "subgrad"
                phase_deadline = time.monotonic() + min(phase_time * 1.5, max(5.0, (deadline - time.monotonic()) * 0.4))
            # Small perturbation between phases
            if iter_count > 2 and rng.random() < 0.3:
                a = _perturb(a, rng, strength=0.15)
            iter_count += 1
            continue

        if phase == "subgrad":
            a, sc = _subgrad_optimize(
                a, phase_deadline,
                peak_frac=0.98,
                eta0=0.3 + 0.1 * rng.random(),
                momentum=0.8 + 0.1 * rng.random(),
                coord_polish_interval=40
            )
        else:
            a = _coord_polish(a, phase_deadline)
            sc = _score_norm(a)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Return the best nonnegative coefficient sequence found within budget_s.
    Uses subgradient descent with momentum, coordinate descent, and basin hopping.
    """
    budget = max(10.0, float(budget_s))
    deadline = time.monotonic() + max(1.0, budget - 5.0)
    rng = np.random.default_rng(int(seed))

    final_len = 3809
    best_seq = None
    best_score = float('inf')

    def _try_update(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_score = float(sc)
            best_seq = _norm1(seq).copy()

    # --- Seed with existing best if available ---
    try:
        hseq = _clean(height_sequence_1)
        if hseq.size > 0:
            sc = _score_general(hseq)
            if np.isfinite(sc):
                _try_update(hseq, sc)
                print(f"Seeded with existing: score={sc:.6f}, len={hseq.size}")
    except Exception:
        pass

    # --- Time budget allocation ---
    t_init = 30.0
    t_main = budget * 0.65
    t_polish = budget * 0.2
    t_reserve = max(30.0, budget * 0.05)

    init_end = time.monotonic() + t_init
    main_end = init_end + t_main
    polish_end = main_end + t_polish

    print(f"Budget: {budget:.0f}s, init_end={t_init:.0f}s, main_end={t_main + t_init:.0f}s, polish_end={polish_end + t_init:.0f}s")

    # --- Phase 1: Generate and evaluate diverse initializations ---
    # Try on a medium length first for speed, then expand
    proto_len = 100
    candidates = []  # list of (score, array)

    strategies = list(range(20))
    for st in strategies:
        if time.monotonic() > init_end:
            break
        try:
            p = _make_initial(proto_len, st, rng)
            sc = _score_norm(p)
            if np.isfinite(sc):
                candidates.append((sc, p.copy()))
                _try_update(p, sc)
        except Exception:
            pass

    # Also try on the full length directly with a few strategies
    for st in [3, 5, 9, 13, 17]:
        if time.monotonic() > init_end:
            break
        try:
            p = _make_initial(final_len, st, rng)
            sc = _score_norm(p)
            if np.isfinite(sc):
                _try_update(p, sc)
        except Exception:
            pass

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: {len(candidates)} candidates, best={candidates[0][0] if candidates else 'inf':.6f}")

    # --- Phase 2: Main optimization loop ---
    # Pick top diverse candidates and optimize each
    max_trials = 12
    trial_time = t_main / max_trials

    # Ensure diversity: don't pick too many similar initializations
    selected = []
    for sc, p in candidates:
        if len(selected) >= max_trials:
            break
        # Check diversity (simple L2 distance in normalized space)
        diverse = True
        for _, p2 in selected:
            if p.size == p2.size and np.linalg.norm(p - p2) < 0.3:
                diverse = False
                break
        if diverse:
            selected.append((sc, p))

    # If not enough diverse, fill with next best
    for sc, p in candidates:
        if len(selected) >= max_trials:
            break
        if p not in [x[1] for x in selected]:
            selected.append((sc, p))

    # Also add the existing best if we have it
    try:
        hseq_p = _norm1(height_sequence_1) if 'height_sequence_1' in dir() or True else None
    except Exception:
        hseq_p = None
    if hseq_p is not None and hseq_p.size > 0:
        selected.append((_score_general(hseq_p), hseq_p))

    print(f"Phase 2: optimizing {len(selected)} candidates")

    for trial_idx, (sc_init, p_init) in enumerate(selected):
        remaining = main_end - time.monotonic()
        if remaining < 5.0:
            break
        trial_deadline = time.monotonic() + min(trial_time, remaining)

        # Work on a reasonable length (use proto or full)
        if p_init.size <= 200:
            # Optimize on proto length, then expand
            opt_deadline = trial_deadline - max(2.0, (trial_deadline - time.monotonic()) * 0.2)
            p_opt, sc_opt = _combine_optimizations(
                p_init, opt_deadline, rng, tag=f"T{trial_idx}/P"
            )
            # Expand to full length
            if time.monotonic() < trial_deadline:
                p_full = _interpolate_smooth(p_opt, final_len)
                sc_full = _score_norm(p_full)
                _try_update(p_full, sc_full)
                # Quick polish on full
                if time.monotonic() < trial_deadline - 2.0:
                    p_full = _coord_polish(p_full, trial_deadline)
                    sc_full = _score_norm(p_full)
                    _try_update(p_full, sc_full)
        else:
            # Already full length, optimize directly
            p_opt, sc_opt = _combine_optimizations(
                p_init, trial_deadline, rng, tag=f"T{trial_idx}/F"
            )
            _try_update(p_opt, sc_opt)

        print(f"  Trial {trial_idx}: best so far = {best_score:.6f}")

    # --- Phase 3: Basin hopping on the best sequence ---
    if best_seq is not None and time.monotonic() < polish_end - 10:
        print(f"Phase 3: Basin hopping from score={best_score:.6f}")
        hop_deadline = min(polish_end, time.monotonic() + t_polish * 0.6)
        hop_time = max(3.0, (hop_deadline - time.monotonic()) / 8.0)

        for hop in range(20):
            if time.monotonic() > hop_deadline:
                break
            # Perturb
            p_pert = _perturb(best_seq, rng, strength=0.2 + 0.1 * rng.random())
            # Optimize
            hd = min(hop_deadline, time.monotonic() + hop_time)
            if hd - time.monotonic() < 2.0:
                break
            p_opt, sc_opt = _combine_optimizations(p_pert, hd, rng, tag=f"BH{hop}")
            if sc_opt < best_score:
                _try_update(p_opt, sc_opt)
                print(f"  Hop {hop}: improved to {best_score:.6f}")

    # --- Phase 4: Final coordinate descent polish ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"Phase 4: Final polish from score={best_score:.6f}")
        p_final = best_seq.copy()
        # If not full length, expand first
        if p_final.size != final_len:
            p_final = _interpolate_smooth(p_final, final_len)
        # Multiple passes of coordinate descent with increasing grid
        for pass_num in range(3):
            if time.monotonic() > deadline - 5:
                break
            sub_deadline = min(deadline - 3.0, time.monotonic() + max(5.0, (deadline - time.monotonic()) * 0.3))
            p_final = _coord_polish(p_final, sub_deadline)
            sc = _score_norm(
