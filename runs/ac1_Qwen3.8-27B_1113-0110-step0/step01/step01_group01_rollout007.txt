Let me think more carefully about the LP step. The problem says:
- Given f (our sequence), M = ||f*f||_inf
- Solve: maximize sum(b) subject to b ≥ 0, ||f*b||_inf ≤ M
- Rescale b to match normalization
- Update: f ← (1-t)f + t*b

This is interesting because:
- If f is uniform, the LP gives b = uniform (by symmetry), so no progress.
- If f has concentrated peaks in its autocorrelation, the LP will find a b that avoids those peaks, potentially creating a more "spread out" sequence.

The LP has n variables and 2n-1 constraints (one for each convolution coefficient ≤ M), plus n non-negativity constraints. For n = 2048, this is manageable with scipy's linprog or a dedicated solver.

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size) / a.size
    return a / s

def convolve_fft(a, N=None):
    n = a.size
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c, fa

def internal_score(a):
    n = a.size
    s = float(a.sum())
    if s <= 1e-15:
        return np.inf
    a = a / s
    c, _ = convolve_fft(a)
    return float(2.0 * n * np.max(c))

def compute_conv_matrix(f, N):
    """Compute the convolution matrix C where C @ b = conv(f, b)[:2n-1]"""
    n = f.size
    m = 2 * n - 1
    from scipy.sparse import diags, eye
    # Actually, let's just use dense for n <= 4096
    # C[k, i] = f[k-i] if 0 <= k-i < n, else 0
    C = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            C[k, i] = f[k - i]
    return C

    # c has length 2n-1, r has length n.
    # A[i,j] = c[i-j] if i>=j, or r[j-i] if j>i
    # For j > i: A[i,j] = r[j-i] = a[0] if j-i=0... 
    # Hmm, this doesn't seem right. Let me check.
    # Actually, for a Toeplitz matrix, A[i,j] depends on i-j.
    # A[i,j] = c[i-j] if i>=j (using first column)
    # A[i,j] = r[j-i] if j>i (using first row)
    # First column: c[0]=a[0], c[1]=a[1], ..., c[n-1]=a[n-1], c[n]=0, ...
    # So A[i,j] = c[i-j] = a[i-j] if 0 <= i-j < n (and i>=j)
    # First row: r[0]=a[0], r[1]=?, ...
    # A[0,j] for j>0: A[0,j] = r[j] 
    # But we want A[0,j] = a[0-j] = 0 for j>0 (since a is only defined for indices 0 to n-1)
    # So r = [a[0], 0, 0, ..., 0] (length n)
    
    r_vec = np.zeros(n)
    r_vec[0] = a[0]
    A_mat = toeplitz(col, r_vec)
    
    # Sanity check: A_mat[0,:] should be [a[0], 0, 0, ..., 0]
    # A_mat[1,:] should be [a[1], a[0], 0, ..., 0]
    # A_mat[n-1,:] should be [a[n-1], a[n-2], ..., a[0]]
    # A_mat[n,:] should be [0, a[n-1], ..., a[1], a[0]]... wait
    # A_mat[n, 0] = c[n] = 0
    # A_mat[n, 1] = c[n-1] = a[n-1]
    # A_mat[n, 2] = c[n-2] = a[n-2]
    # ...
    # A_mat[n, n-1] = c[1] = a[1]
    # A_mat[n, n] = r[n]... but r only has length n, so r[n] doesn't exist.
    # Actually A has shape (2n-1, n), so max column index is n-1.
    # A_mat[n, n-1] = c[n-(n-1)] = c[1] = a[1] ✓
    
    # The LP: maximize 1^T b subject to A b <= M, b >= 0
    # In scipy linprog form: minimize -1^T b subject to A b <= M, 0 <= b
    c_obj = -np.ones(n)
    # Constraints: A b <= M (M scalar applies to all)
    # Actually the constraint is A b <= M * ones(2n-1)
    b_ub_below = np.full(2*n-1, M)
    
    # scipy.optimize.linprog(c_obj, A_ub=A_mat, b_ub=b_ub_below, bounds=[(0,None)]*n, method='highs')
    # For n=2048, this is a 2048-variable LP with 4095 constraints. Should be fast.
    
    try:
        # Use HiGHS for speed

        if result.status == 0 and result.x is not None:

            b_sum = float(b_opt.sum())
            if b_sum > 1e-10:
                b_opt = b_opt / b_sum
                return b_opt
    except Exception as e:
        print(f"  LP step failed: {e}", flush=True)
    
    return a

"""
Multi-phase optimizer for min 2*n*max(conv(a,a))/(sum a)^2 over a >= 0.
Combines: Adam low-dim optimization, LP-inspired reweighting,
peak-focused gradient descent, binary refinement, and multi-n search.
"""

import time
import numpy as np
try:
    import scipy.optimize as so
    from scipy.linalg import toeplitz
except Exception:
    so = None
    toeplitz = None


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.ones(1)
    a = np.maximum(a, float(floor))
    s = float(a.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.ones(n) / n
    return a / s


def internal_score(a):
    n = a.size
    s = float(a.sum())
    if s <= 1e-15:
        return np.inf
    a = a / s
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return float(2.0 * n * c.max())


def softmax_safe(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    z = np.clip(z, -60, 0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / z.size
    return e / s


def build_basis(n, n_modes=None):
    x = np.arange(n, dtype=np.float64)
    t = x / max(1, n - 1)
    cols = []
    if n_modes is None:
        n_modes = min(20, n // 3)
    for m in range(1, n_modes + 1):
        cols.append(np.sin(2 * np.pi * m * t))
        cols.append(np.cos(2 * np.pi * m * t))
    xc = (x - (n - 1) / 2.0)
    for m in range(1, min(8, n_modes + 1)):
        cols.append(np.sin(np.pi * m * xc / max(1, n - 1)))
    for d in range(1, 7):
        p = t ** d
        p = p - p.mean()
        cols.append(p)
    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    Q, R = np.linalg.qr(B, mode='reduced')
    diag = np.abs(np.diag(R))
    valid = diag > 1e-10
    Q = Q[:, valid]
    return Q


def get_grad_conv(a, w, N=None):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a. Returns 2*x where x[m]=sum_k w[k]*a[k-m]."""
    n = a.size
    w_rev = w[::-1]
    a_rev = a[::-1]
    Nw = next_pow2((2 * n - 1) + n - 1)
    if N is not None:
        Nw = max(Nw, N)
    fw = np.fft.rfft(w_rev, Nw)
    fa = np.fft.rfft(a, Nw)
    conv_wa = np.fft.irfft(fw * fa, Nw)
    x = conv_wa[2 * n - 2: n - 2: -1]
    return 2.0 * x


def adam_lowdim(n, deadline, seed, a_init=None):
    B = build_basis(n)
    D = B.shape[1]
    N = next_pow2(3 * n - 2)
    rng = np.random.default_rng(seed)

    best_a = a_init.copy() if a_init is not None else np.ones(n) / n
    best_score = internal_score(best_a)

    starts = []
    if a_init is not None:
        z = np.log(np.maximum(a_init, 1e-15))
        z = z - z.mean()
        theta0, _, _, _ = np.linalg.lstsq(B, z, rcond=None)
        starts.append(np.asarray(theta0, dtype=np.float64))
    for std in (0.3, 0.6, 1.0, 1.5, 2.0):
        starts.append(rng.normal(0, std, D))

    for theta0 in starts:
        if time.monotonic() > deadline:
            break
        theta = np.asarray(theta0, dtype=np.float64).copy()
        m_t = np.zeros(D)
        v_t = np.zeros(D)
        b1, b2, eps = 0.85, 0.999, 1e-8
        lr = 0.08
        t_step = 0

        while time.monotonic() < deadline:
            a = softmax_safe(B @ theta)
            fa = np.fft.rfft(a, N)
            c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
            cmax = float(c.max())

            frac = min(1.0, t_step / 2000.0)
            tau = (0.5 / n) * (1 - frac) + (0.005 / n) * frac

            z_w = (c - cmax) / max(tau, 1e-12)
            z_w = np.clip(z_w, -50, 0)
            ez = np.exp(z_w)
            w = ez / ez.sum()

            grad_a = get_grad_conv(a, w, N)
            ga = grad_a * a
            gB = ga @ B
            aB = a @ B
            ga_s = float(ga.sum())
            grad_th = gB - ga_s * aB
            grad_th = np.nan_to_num(grad_th, 0, 0, 0)

            t_step += 1
            m_t = b1 * m_t + (1 - b1) * grad_th
            v_t = b2 * v_t + (1 - b2) * grad_th ** 2
            m_hat = m_t / (1 - b1 ** t_step)
            v_hat = v_t / (1 - b2 ** t_step)
            theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

            if t_step % 100 == 0:
                lr = max(0.003, lr * 0.97)

            if t_step % 50 == 0:
                sc = internal_score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    if t_step % 500 == 0:
                        print(f"  adam n={n} step={t_step} score={best_score:.8f}", flush=True)

            if t_step > 8000:
                break

    return best_a, best_score


def lp_step(a, max_time=None):
    """Apply one LP-inspired reweighting step."""
    if so is None or toeplitz is None:
        return a
    n = a.size
    a = normalize_positive(a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    M = float(c.max())
    if M <= 0 or not np.isfinite(M):
        return a

    col = np.zeros(2 * n - 1)
    col[:n] = a
    r_vec = np.zeros(n)
    r_vec[0] = a[0]
    A_mat = toeplitz(col, r_vec)

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        t0 = time.monotonic()
        result = so.linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                            bounds=[(0, None)] * n, method='highs',
                            options={'time_limit': 30 if max_time is None else max_time})
        if result.status == 0 and result.x is not None:
            b_opt = np.maximum(result.x, 0)
            b_sum = float(b_opt.sum())
            if b_sum > 1e-10:
                b_opt = b_opt / b_sum
                sc_old = internal_score(a)
                sc_new = internal_score(b_opt)
                if sc_new < sc_old:
                    print(f"  LP step: {sc_old:.8f} -> {sc_new:.8f}", flush=True)
                    return b_opt
        if time.monotonic() - t0 > 5:
            pass
    except Exception:
        pass
    return a


def lp_multi(a, n_steps=3):
    best_a = a.copy()
    best_score = internal_score(a)
    for s in range(n_steps):
        a_try = lp_step(a)
        sc_try = internal_score(a_try)
        if np.isfinite(sc_try) and sc_try < best_score:
            best_score = sc_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try mixing
            for t in (0.3, 0.5, 0.7):
                a_mix = (1 - t) * a + t * a_try
                a_mix = normalize_positive(a_mix)
                sc_mix = internal_score(a_mix)
                if np.isfinite(sc_mix) and sc_mix < best_score:
                    best_score = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    break
    return best_a, best_score


def peak_focused_descent(a, deadline, seed=0):
    """Gradient descent focusing on top-k peaks."""
    n = a.size
    N = next_pow2(3 * n - 2)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = internal_score(a)

    loga = np.log(a + 1e-20)
    loga -= loga.max()

    lr = 0.4
    momentum = 0.0
    v = np.zeros(n)
    top_k = 8
    stale = 0

    while time.monotonic() < deadline:
        a_cur = softmax_safe(loga * n)  # scale for diversity
        a_cur = normalize_positive(a_cur)
        fa = np.fft.rfft(a_cur, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        cmax = float(c.max())
        score = 2.0 * n * cmax

        if np.isfinite(score) and score < best_score * (1 - 1e-10):
            best_score = score
            best_a = a_cur.copy()
            stale = 0
        else:
            stale += 1

        # Focus on top-k peaks
        top_idx = np.argsort(c)[-top_k:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / top_k

        grad_a = get_grad_conv(a_cur, w, N)
        grad_a = grad_a - grad_a.mean()

        # Adam-like update on loga
        lr_cur = lr * (0.999 ** stale)
        v = 0.9 * v + 0.1 * grad_a ** 2
        step = lr_cur * grad_a / (np.sqrt(v) + 1e-8)
        loga = loga - step

        if stale > 100:
            # Random perturbation
            idx = rng.integers(0, n, size=max(1, n // 50))
            loga[idx] += rng.normal(0, 0.1, size=idx.size)
            stale = 0
            lr = max(0.05, lr * 0.8)

        if stale > 500:
            break

    return best_a, best_score


def resample(a, n_new):
    a = normalize_positive(a)
    n_old = a.size
    n_new = max(2, int(n_new))
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    b = np.interp(x_new, x_old, a)
    return normalize_positive(b, 1e-12)


def binary_optimize(a0, deadline, seed=0):
    """Optimize a binary (0/1) sequence via swaps."""
    n = a0.size
    m = max(2, n // 2)
    order = np.argsort(a0, kind='stable')
    b = np.zeros(n)
    b[order[-m:]] = 1.0

    N = next_pow2(2 * n - 1)
    c = np.convolve(b, b)
    m_ones = m

    best_b = b.copy()
    best_score = float(2 * n * c.max() / (m_ones ** 2))
    rng = np.random.default_rng(seed)
    temp = 0.05
    stale = 0

    two_b = 2.0 * b

    while time.monotonic() < deadline:
        cmax = float(c.max())
        score = float(2 * n * cmax / (m_ones ** 2))
        if score < best_score * (1 - 1e-12):
            best_score = score
            best_b = b.copy()
            stale = 0
        else:
            stale += 1

        # Pick swap: remove 1 at high-pressure position, add at low-pressure
        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        # Pressure based on local convolution contribution
        g_one = c[one_idx:one_idx + n].sum(axis=1) if one_idx.size < n else c[:one_idx.size + n - 1].sum(axis=1)
        # Simpler: just use c values at random positions
        si = int(rng.integers(0, one_idx.size))
        sj = int(rng.integers(0, zero_idx.size))
        i = one_idx[si]
        j = zero_idx[sj]
        if i == j:
            continue

        old_score = score
        # Apply swap
        c[i:i + n] -= two_b
        if 2 * i < 2 * n - 1:
            c[2 * i] -= 1.0
        c[j:j + n] += two_b
        if 2 * j < 2 * n - 1:
            c[2 * j] += 1.0
        b[i] = 0.0
        b[j] = 1.0
        two_b = 2.0 * b

        new_cmax = float(c.max())
        new_score = float(2 * n * new_cmax / (m_ones ** 2))

        accept = False
        if new_score < old_score:
            accept = True
        else:
            delta = new_score - old_score
            if delta < 0.1 and rng.random() < np.exp(-delta / max(temp, 0.001)):
                accept = True

        if not accept:
            # Revert
            c[j:j + n] -= two_b
            if 2 * j < 2 * n - 1:
                c[2 * j] -= 1.0
            c[i:i + n] += two_b
            if 2 * i < 2 * n - 1:
                c[2 * i] += 1.0
            b[j] = 0.0
            b[i] = 1.0
            two_b = 2.0 * b
            temp = max(0.001, temp * 0.98)
            if stale > 2000:
                break
        else:
            temp = max(0.001, temp * 0.999)

    return normalize_positive(best_b), best_score


def binary_to_continuous(b, n_cont, alpha=0.3):
    """Convert binary to continuous via smoothing."""
    n_bin = b.size
    if n_bin == n_cont:
        return b * (1 - alpha) + alpha * np.ones(n_cont) / n_cont
    b_smooth = resample(b, n_cont)
    return normalize_positive(b_smooth * (1 - alpha) + alpha * np.ones(n_cont) / n_cont)

        n_final = best_a.size
        t7 = min(150.0, deadline - time.monotonic() - 180)
        a7, s7 = adam_lowdim(n_final, time.monotonic() + t7, seed=seed + 500, a_init=best_a)
        if a7 is not None and np.isfinite(s7) and s7 < best_score:
            best_score = s7
            best_a = a7.copy()
            print(f"Phase 7 (final adam): {best_score:.8f}", flush=True)

    # Phase 8: LP final
    if time.monotonic() < deadline - 100:
        a8, s8 = lp_multi(best_a, n_steps=2)
        if np.isfinite(s8) and s8 < best_score:
            best_score = s8
            best_a = a8.copy()
            print(f"Phase 8 (LP final): {best_score:.8f}", flush=True)

    # Phase 9: Tiny perturbations
    if time.monotonic() < deadline - 30:
        n = best_a.size
        rng = np.random.default_rng(seed + 999)
        N = next_pow2(2 * n - 1)
        for _ in range(200):
            if time.monotonic() > deadline - 30:
                break
            a_try = best_a.copy()
            idx = rng.integers(0, n, size=max(1, n // 200))
            a_try[idx] *= np.exp(rng.normal(0, 0.05, size=idx.size))
            a_try = normalize_positive(a_try)
            sc_try = internal_score(a_try)
            if np.isfinite(sc_try) and sc_try < best_score:
                best_score = sc_try
                best_a = a_try.copy()

    best_a = normalize_positive(best_a, 0.0)
    out_list = [float(x) for x in best_a]

    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = np.inf

    if not np.isfinite(final_eval):
        best_a = np.ones(1024) / 1024
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print(f"FINAL: internal={best_score:.10f} eval={final_eval:.10f} n={len(out_list)}", flush=True)
    return out_list
