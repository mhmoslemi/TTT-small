        return a.copy()
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def compute_v_fft(a, w):
    """Compute v[j] = sum_{m=0}^{n-1} a[m] * w[j+m] for j=0,...,n-1."""
    n = len(a)
    w_rev = w[::-1]
    total_len = n + len(w_rev) - 1
    size = 1
    while size < total_len:
        size *= 2
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:total_len]
    # v[j] = conv[2n-2-j]
    v = conv[2 * n - 2: n - 2: -1]
    return v


def softmax_to_a(b, n):
    """Convert b to a = n * softmax(b)."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    Z = exp_b.sum()
    if Z < 1e-300:
        return np.ones(n)
    return n * exp_b / Z


def soft_max_score_and_grad_b(b, tau, n):
    """Compute soft-max score and gradient w.r.t. b."""
    a = softmax_to_a(b, n)
    c = fft_self_conv(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    Z = exp_c.sum()
    if Z < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = exp_c / Z
    M = tau * (c_max + np.log(Z))
    score = 2.0 * M / n
    v = compute_v_fft(a, w)
    mean_v = v.mean()
    grad_b = (4.0 * a / n) * (v - mean_v)
    return score, grad_b, a, c, w

def coordinate_ascent(a, max_rounds=3, deadline=None):
    """Cyclic coordinate ascent to refine the solution."""
    n = len(a)
    a = a.copy()
    best_local = a.copy()
    best_local_score = exact_score_from_a(a)

    for round_idx in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                return best_local, best_local_score
            # Save current value
            aj = a[j]
            if aj < 1e-15:
                continue

            # Compute the "base" convolution (with a[j] = 0)
            # c[k] = c_base[k] + 2*a[j]*a[k-j] for k != 2j
            # c[2j] = c_base[2j] + a[j]^2

            # Find the current max lag
            c = fft_self_conv(a)
            k_star = int(np.argmax(c))
            M = c[k_star]

            # Determine direction: should we increase or decrease a[j]?
            # Increasing a[j] increases c[k] for all k where a[k-j] > 0
            # So if a[j] contributes to the max, we should decrease it
            contribution = 0.0
            if 0 <= k_star - j < n:
                if k_star == 2 * j:
                    contribution = a[j]  # derivative of a[j]^2 at a[j]
                else:
                    contribution = 2.0 * a[k_star - j]

            if contribution <= 0:
                # Can't reduce by changing a[j], try small increase
                # (might reduce another peak)
                continue

            # Binary search for optimal a[j]
            # We want to minimize max(c_base[k] + a[j]*a[k-j] + a[j]*a[k-j] or a[j]^2)
            # This is a convex function of a[j] (max of linear/quadratic functions)
            lo = 0.0
            hi = aj * 2.0  # don't go much higher

            for _ in range(20):
                mid = (lo + hi) / 2.0
                a[j] = mid
                c_mid = fft_self_conv(a)
                M_mid = c_mid.max()
                a[j] = mid  # keep for now

                # Compare with decreasing a[j]
                test_lo = lo
                a[j] = test_lo
                c_lo = fft_self_conv(a)
                M_lo = c_lo.max()

                if M_mid < M_lo:
                    lo = mid
                else:
                    hi = mid
                a[j] = mid

            # Set to best found
            a[j] = lo
            c_final = fft_self_conv(a)
            M_final = c_final.max()

            # Check if this improved things
            if M_final < M - 1e-12:
                improved = True

            # Restore if worse
            a[j] = lo  # keep the improved value

        # Check overall improvement
        sc = fast_score(a)
        if sc < best_local_score - 1e-8:
            best_local = a.copy()
            best_local_score = sc

    # Final exact check
    final_sc = exact_score_from_a(a)
    if np.isfinite(final_sc) and final_sc < best_local_score:
        best_local = a.copy()
        best_local_score = final_sc

    return best_local, best_local_score
