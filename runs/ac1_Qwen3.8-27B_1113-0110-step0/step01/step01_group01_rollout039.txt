```python
def gs_step(a, n, N, alpha=1.0):
    # a is normalized to sum=1
    A = np.fft.rfft(a, N)
    # Modify |A| to be flatter
    # Option 1: set all |A| to their geometric mean (except DC)
    # Option 2: interpolate between current |A| and flat
    mag = np.abs(A)
    # Keep DC fixed
    flat_mag = np.full(N//2+1, mag[0] / np.sqrt(N))  # target: |A[k]| = S/sqrt(N)
    # But this might not be achievable with a >= 0
    # Blending
    new_mag = (1-alpha) * mag + alpha * flat_mag
    new_mag[0] = mag[0]  # preserve DC
    # Reconstruct with target magnitude and current phase
    phase = np.angle(A)
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new
```

```python
def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a_norm = a / s
    c = convolve_seq(a_norm, N)
    return float(2.0 * n * float(np.max(c)))


def make_uniform(n):
    return np.ones(max(1, int(n)), dtype=np.float64) / float(max(1, int(n)))


def make_sine_sequence(n, alphas):
    n = max(2, int(n))
    x = (np.arange(n, dtype=np.float64) - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    base = np.ones(n, dtype=np.float64)
    for m, alpha in enumerate(alphas, start=1):
        base = base + float(alpha) * np.sin(np.pi * float(m) * x)
    return normalize_positive(base, floor=1e-12)


def make_gaussian_sequence(n, mu, sigma):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    sigma = max(1e-3, float(sigma))
    return normalize_positive(np.exp(-((x - float(mu)) ** 2) / (2.0 * sigma * sigma)), floor=1e-12)

def gs_spectral_shaping(a, n, N, alpha, n_iters, deadline, verbose=False):
    """GS algorithm: alternate between time (non-negativity) and freq (flatten spectrum)."""
    a = normalize_positive(a, floor=1e-15).copy()
    n = a.size
    best_a = a.copy()
    best_score = internal_score(a, N)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # FFT
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat spectrum |A[k]| = S/sqrt(N)
        S = float(a.sum())
        target_mag = S / np.sqrt(float(N))
        
        # Blend current magnitude with target
        if alpha <= 0:
            new_mag = mag.copy()
        elif alpha >= 1:
            new_mag = np.full_like(mag, target_mag)
        else:
            # Geometric blend: |A| = mag^(1-alpha) * target^alpha
            log_mag = np.log(mag + 1e-300)
            log_target = np.log(target_mag + 1e-300)
            new_mag = np.exp((1 - alpha) * log_mag + alpha * log_target)
        
        # Preserve DC
        new_mag[0] = mag[0]
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Enforce non-negativity
        a_new = np.clip(a_new, 0.0, None)
        s = float(a_new.sum())
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = internal_score(a_new, N)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Relaxation: blend with previous to avoid oscillation
        a = 0.7 * a_new + 0.3 * a
        a = normalize_positive(a, floor=0.0)
        
        if verbose and it % 50 == 0:
            print("  GS iter={} alpha={:.3f} best={:.9f}".format(it, alpha, best_score), flush=True)
    
    return best_a, best_score

def compute_conv_and_grad_weighted(a, N, n, weights):
    """Compute convolution and gradient from weighted lags."""
    c = convolve_seq(a, N)
    cmax = float(np.max(c))
    if cmax <= 0:
        return c, np.zeros(n), 0.0
    
    # weights should be same length as c
    w = weights / weights.sum()
    w_rev = w[::-1]
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    grad = 2.0 * conv[n - 1: 2 * n - 1][::-1]
    return c, grad, cmax

            best_a = a.copy()

            break

        z = (top_vals - float(top_vals.max())) / max(tau, 1e-15)

        g = g / scale

            stagnant += 1

        for e in (eta, 0.5 * eta, 0.25 * eta):
            if e < 1e-8:
                continue
            loga_try = loga - e * g
            loga_try = loga_try - float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / a_try.sum()
            a_try = normalize_positive(a_try, floor=1e-15)
            
            sc_try = internal_score(a_try, N)
            if np.isfinite(sc_try) and sc_try < score * (1 - 1e-10):
                a = a_try
                score = sc_try
                accepted = True
                break
        
        if accepted:
            stagnant = 0
            eta = min(1.3, eta * 1.01)
        else:
            stagnant += 1
            eta = max(1e-4, eta * 0.7)
            if stagnant > 300:
                # Restart with perturbation
                rng = np.random.default_rng(it * 31 + n)
                a = a * np.exp(rng.normal(0, 0.03, n))
                a = normalize_positive(a, floor=1e-15)
                stagnant = 0
            if eta < 1e-5:
                break
        
        it += 1
        if verbose and it % 300 == 0:
            print("    mirror v2 iter={} score={:.9f} K={}".format(it, best_score, K_eff), flush=True)
    
    return best_a, best_score


def binary_cost_array(a0, n, m, deadline, seed=0):
    """Optimize a binary cost array (positions of ones) to minimize max autocorrelation."""
    n = max(16, int(n))
    m = max(2, min(n - 2, int(m)))
    
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    if a0.size != n:
        a0 = resample_seq(a0, n)
    
    # Initialize: pick top m positions
    order = np.argsort(a0, kind="stable")
    b = np.zeros(n, dtype=np.float64)
    b[order[-m:]] = 1.0
    
    N = next_pow2(2 * n - 1)
    c = convolve_seq(b, N)
    best_b = b.copy()
    best_cmax = float(np.max(c))
    best_score = float(2.0 * n * best_cmax / (m * m))
    
    rng = np.random.default_rng(seed)
    temp = 0.05
    stale_count = 0
    
    two_b = 2.0 * b
    
    while time.monotonic() < deadline:
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax / (m * m))
        
        if score < best_score * (1 - 1e-12):
            best_score = score
            best_cmax = cmax
            best_b = b.copy()
        
        # Weighted gradient from autocorrelation
        tau = 0.1 * cmax
        z = (c - cmax) / max(tau, 1e-15)
        z = np.clip(z, -50.0, 0.0)
        w = np.exp(z)
        w = w / w.sum()
        
        w_rev = w[::-1]
        fa = np.fft.rfft(b, N)
        fw = np.fft.rfft(w_rev, N)
        conv_g = np.fft.irfft(fa * fw, N)
        g = 2.0 * conv_g[n - 1: 2 * n - 1][::-1]
        
        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        
        # Remove from high gradient, add to low gradient
        g_one = g[one_idx]
        g_zero = g[zero_idx]
        
        # Probabilistic selection
        temp_g = 0.15
        if g_one.max() - g_one.min() > 1e-14:
            p_remove = np.exp(-(g_one - g_one.max()) / (temp_g * (g_one.max() - g_one.min()) + 1e-15))
            p_remove = p_remove / p_remove.sum()
        else:
            p_remove = np.full(len(one_idx), 1.0 / len(one_idx))
        
        if g_zero.max() - g_zero.min() > 1e-14:
            p_add = np.exp(-(g_zero.min() - g_zero) / (temp_g * (g_zero.max() - g_zero.min()) + 1e-15))
            p_add = p_add / p_add.sum()
        else:
            p_add = np.full(len(zero_idx), 1.0 / len(zero_idx))
        
        i = int(one_idx[rng.choice(len(one_idx), p=p_remove)])
        j = int(zero_idx[rng.choice(len(zero_idx), p=p_add)])
        
        if i == j:
            continue
        
        # Swap
        c[i:i + n] -= two_b
        c[2 * i] -= 1.0
        c[j:j + n] += two_b
        c[2 * j] += 1.0
        b[i] = 0.0
        b[j] = 1.0
        two_b = 2.0 * b
        
        new_cmax = float(np.max(c))
        new_score = float(2.0 * n * new_cmax / (m * m))
        
        # Accept/reject
        if new_score < score:
            stale_count = 0
            temp = max(0.003, temp * 0.997)
        else:
            delta = new_score - score
            if delta < 0.5 and rng.random() < np.exp(-delta / max(temp, 1e-12)):
                stale_count = 0
            else:
                # Revert
                c[j:j + n] -= two_b
                c[2 * j] -= 1.0
                c[i:i + n] += two_b
                c[2 * i] += 1.0
                b[j] = 0.0
                b[i] = 1.0
                two_b = 2.0 * b
                stale_count += 1
                temp = max(0.001, temp * 0.95)
        
        if stale_count > 400:
            temp = min(0.1, temp * 2.0)
            stale_count = 0
    
    return normalize_positive(best_b, floor=0.0), best_score


def local_polish(a, deadline, seed=1):
    """Random log-block perturbation polish."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = internal_score(a, N)
    
    loga = np.log(a + 1e-20)
    loga -= float(np.max(loga))
    rng = np.random.default_rng(seed)
    sigma = 0.15
    stale = 0
    it = 0
    
    while time.monotonic() < deadline:
        width_low = max(2, n // 3000)
        width_high = max(3, n // 15)
        width = int(rng.integers(width_low, min(width_high, n // 3) + 1))
        start = int(rng.integers(0, n - width + 1))
        
        t = np.linspace(0.0, 1.0, width, dtype=np.float64)
        env = np.exp(-4.0 * ((t - 0.5) ** 2))
        d = sigma * (0.5 + 0.5 * env) * rng.normal(size=width)
        d -= d.mean()
        d = np.clip(d, -3.0, 3.0)
        
        old = loga[start:start + width].copy()
        loga[start:start + width] += d
        a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
        a_try = np.maximum(a_try, 1e-300)
        a_try = a_try / a_try.sum()
        a_try = normalize_positive(a_try, floor=1e-15)
        
        sc_try = internal_score(a_try, N)
        if np.isfinite(sc_try) and sc_try < best_score * (1 - 1e-10):
            best_a = a_try.copy()
            best_score = sc_try
            loga = np.log(a_try + 1e-20)
            loga -= float(np.max(loga))
            stale = 0
            sigma = min(0.3, sigma * 1.03)
        else:
            loga[start:start + width] = old
            stale += 1
            sigma = max(0.02, sigma * 0.97)
            if stale > 300:
                # Large perturbation
                idx = int(rng.integers(0, n))
                old_idx = loga[idx]
                loga[idx] += 2.0 * sigma
                a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
                a_try = np.maximum(a_try, 1e-300)
                a_try = a_try / a_try.sum()
                a_try = normalize_positive(a_try, floor=1e-15)
                sc_try = internal_score(a_try, N)
                if np.isfinite(sc_try) and sc_try < best_score:
                    best_a = a_try.copy()
                    best_score = sc_try
                    loga = np.log(a_try + 1e-20)
                    loga -= float(np.max(loga))
                else:
                    loga[idx] = old_idx
                stale = 0
        
        it += 1
    
    return best_a, best_score


def lowdim_smooth_optimize(a, n, deadline, seed=0):
    """Low-dimensional smooth optimization using a Fourier basis in log-space."""
    n = max(32, int(n))
    N = next_pow2(2 * n - 1)
    
    # Build basis: low-frequency Fourier modes
    n_modes = min(12, n // 4)
    i = np.arange(n, dtype=np.float64)
    cols = []
    for m in range(1, n_modes + 1):
        t = 2.0 * np.pi * float(m) * i / float(n)
        cols.append(np.sin(t))
        cols.append(np.cos(t))
    
    # Add polynomial modes
    x = (i - 0.5 * (n - 1)) / (0.5 * n)
    cols.append(x.copy())
    cols.append(x * x - np.mean(x * x))
    
    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-14] = 1.0
    B = B / norms
    D = B.shape[1]
    
    # Initialize theta from current a
    a_init = normalize_positive(a, floor=1e-12)
    if a_init.size != n:
        a_init = resample_seq(a_init, n)
    z0 = np.log(a_init + 1e-12)
    z0 -= z0.mean()
    theta0, _, _, _ = np.linalg.lstsq(B, z0, rcond=None)
    theta0 = theta0.astype(np.float64)
    
    best_a = a_init.copy()
    best_score = internal_score(a_init, N)
    
    rng = np.random.default_rng(seed)
    
    # Multiple restarts with different scales
    for restart in range(4):
        if time.monotonic() > deadline:
            break
        
        if restart == 0:
            theta = theta0.copy()
        else:
            theta = theta0 + rng.normal(0, 0.5 * restart, D)
        
        # Simple L-BFGS with smooth approximation of max
        def obj_hess(theta_t):
            a_t = np.exp(B @ theta_t)
            a_t = a_t / a_t.sum()
            c = convolve_seq(a_t, N)
            cmax = float(np.max(c))
            # Smooth max approximation
            tau = 0.02 * cmax
            z = (c - cmax) / max(tau, 1e-15)
            z = np.clip(z, -50.0, 0.0)
            s = float(np.exp(z).sum())
            smooth = cmax + tau * np.log(s)
            return 2.0 * n * smooth
        
        def grad(theta_t):
            a_t = np.exp(B @ theta_t)
            a_t = a_t / a_t.sum()
            c = convolve_seq(a_t, N)
            cmax = float(np.max(c))
            tau = 0.02 * cmax
            z = (c - cmax) / max(tau, 1e-15)
            z = np.clip(z, -50.0, 0.0)
            ez = np.exp(z)
            w = ez / ez.sum()
            
            # Gradient of weighted sum of c
            w_rev = w[::-1]
            fa = np.fft.rfft(a_t, N)
            fw = np.fft.rfft(w_rev, N)
            conv_g = np.fft.irfft(fa * fw, N)
            ga = 2.0 * conv_g[n - 1: 2 * n - 1][::-1]
            
            # Chain rule through softmax-like normalization
            # a = exp(B@theta) / sum(exp(B@theta))
            # da/dtheta = a * (B - B^T @ a) (approximately, for the log-sum-exp)
            total = float(np.dot(a_t, ga))
            aB = a_t @ B
            gtheta = (a_t * ga) @ B - total * aB
            return 2.0 * n * gtheta
        
        try:
            res = _scipy_optimize.minimize(
                obj_hess, theta, method="L-BFGS-B", jac=grad,
                options={"maxiter": 30, "ftol": 1e-14, "gtol": 1e-8, "maxls": 20}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_t = np.exp(B @ res.x)
                a_t = a_t / a_t.sum()
                a_t = normalize_positive(a_t, floor=1e-15)
                sc = internal_score(a_t, N)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_t.copy()
                    print("    lowdim restart={} score={:.9f}".format(restart, best_score), flush=True)
        except Exception:
            pass
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()
    total = max(120.0, float(budget_s) - 30.0)
    deadline = start + total
    
    best_a = make_uniform(1024)
    best_score = internal_score(best_a)
    print("initial uniform score = {:.9f}".format(best_score), flush=True)
    
    # Check given sequence
    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            print("given sequence score = {:.9f}".format(best_score), flush=True)
    
    # === Phase 1: GS Spectral Shaping (new approach) ===
    print("\n=== Phase 1: GS Spectral Shaping ===", flush=True)
    gs_deadline = start + min(400.0, total * 0.25)
    
    for n in (1024, 2048, 4096):
        if time.monotonic() > gs_deadline:
            break
        a_start = resample_seq(best_a, n)
        N = next_pow2(2 * n - 1) * 2
        
        # Multiple GS runs with different alphas
        for alpha in (0.2, 0.35, 0.5):
            if time.monotonic() > gs_deadline:
                break
            a1, sc1 = gs_spectral_shaping(a_start, n, N, alpha, 300, gs_deadline, verbose=False)
            if np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                print("  GS n={} alpha={:.2f} score={:.9f}".format(n, alpha, best_score), flush=True)
        
        # Follow-up with different alpha
        if time.monotonic() < gs_deadline - 30:
            a_start2 = resample_seq(best_a, n)
            a1, sc1 = gs_spectral_shaping(a_start2, n, N, 0.15, 400, gs_deadline, verbose=False)
            if np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                print("  GS refine n={} score={:.9f}".format(n, best_score), flush=True)
    
    # === Phase 2: Hot-lag focused optimization ===
    print("\n=== Phase 2: Hot-lag Optimization ===", flush=True)
    hl_deadline = start + min(900.0, total * 0.5)
    
    for n in (1024, 2048, 4096):
        if time.monotonic() > hl_deadline:
            break
        allow = min(180.0, hl_deadline - time.monotonic())
        if allow < 20:
            break
        a_start = resample_seq(best_a, n)
        a1, sc1 = hot_lag_optimize(a_start, n, time.monotonic() + allow, K=80, verbose=True)
        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("  hotlag n={} score={:.9f}".format(n, best_score), flush=True)
    
    # === Phase 3: Mirror descent refinement ===
    print("\n=== Phase 3: Mirror Descent V2 ===", flush=True)
    md_deadline = start + min(1400.0, total * 0.75)
    
    for n in (2048, 4096):
        if time.monotonic() > md_deadline:
            break
        allow = min(200.0, md_deadline - time.monotonic())
        if allow < 30:
            break
        a_start = resample_seq(best_a, n)
        a1, sc1 = mirror_descent_v2(a_start, time.monotonic() + allow, K_lags=200, verbose=True)
        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("  mirror v2 n={} score={:.9f}".format(n, best_score), flush=True)
    
    # === Phase 4: Low-dim smooth optimization ===
    print("\n=== Phase 4: Low-dim Optimization ===", flush=True)
    if time.monotonic() < deadline - 200:
        for n in (1024, 2048, 4096):
            if time.monotonic() > deadline - 200:
                break
            allow = min(100.0, deadline - time.monotonic() - 200)
            if allow < 15:
                break
            a1, sc1 = lowdim_smooth_optimize(best_a, n, time.monotonic() + allow, seed=seed + n)
            if np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                print("  lowdim n={} score={:.9f}".format(n, best_score), flush=True)
    
    # === Phase 5: Binary cost array search ===
    print("\n=== Phase 5: Binary Search ===", flush=True)
    if time.monotonic() < deadline - 100:
        for n in (2048, 4096):
            if time.monotonic() > deadline - 100:
                break
            allow = min(80.0, deadline - time.monotonic() - 100)
            if allow < 15:
                break
            m = n // 2
            a1, sc1 = binary_cost_array(best_a, n, m, time.monotonic() + allow, seed=seed + n)
            if np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                print("  binary n={} score={:.9f}".format(n, best_score), flush=True)
    
    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final Polish ===", flush=True)
    if time.monotonic() < deadline - 10:
        a1, sc1 = local_polish(best_a, deadline, seed=99999)
        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("  polish score={:.9f}".format(best_score), flush=True)
    
    # Final evaluation
    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]
    
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")
    
    if not np.isfinite(final_eval):
        best_a = make_uniform(1024)
        out_list = [float(x) for x in best_a]
        final_eval = 2.0
    
    elapsed = time.monotonic() - start
    print("\n=== FINAL ===", flush=True)
    print("internal score: {:.12f}".format(best_score), flush=True)
    print("evaluated score: {:.12f}".format(final_eval), flush=True)
    print("length: {}".format(len(out_list)), flush=True)
    print("time: {:.1f}s".format(elapsed), flush=True)
    
    return out_list
```
