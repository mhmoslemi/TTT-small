def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    n = len(f)
    # C[k, j] such that (C @ g)[k] = sum_j f[|k-j| or appropriate] * g[j]
    # (f*g)[k] = sum_j f[j] * g[k-j] = sum_j g[j] * f[k-j]
    # So C[k, j] = f[k-j] if 0 <= k-j < n, else 0
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # (f*g)[k] for k = j, j+1, ..., j+n-1 comes from g[j] * f[0], f[1], ..., f[n-1]
        # Wait: (f*g)[k] = sum_j' f[j'] * g[k-j']
        # For a fixed j (index in g): g[j] * f[k-j]
        # So C[k, j] = f[k-j] if 0 <= k-j < n
        # That is, for row k and column j: C[k,j] = f[k-j] if 0<=k-j<n
        pass
    # Alternative: C[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This means for column j, the non-zero entries are in rows j to j+n-1
    # C[j+r, j] = f[r] for r = 0, ..., n-1
    for j in range(n):
        C[j:j+n, j] = f
    return C

def _lp_solve(f, M, max_iter=200):

def _local_search(a, deadline, n_trials=20000):
    """Local search: try small random perturbations."""
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(123)
    
    # Try coordinate-wise perturbations
    step_size = 0.01
    count = 0
    while count < n_trials and time.monotonic() < deadline:
        # Pick a random coordinate
        j = rng.integers(0, n)
        # Try increasing and decreasing
        for direction in [1, -1]:
            a_try = a.copy()
            a_try[j] += direction * step_size * a[j] * 0.1
            a_try = np.maximum(a_try, 0.0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
                a = a_try.copy()
        count += 1
    
    if count % 5000 == 0:
        print(f"    Local search: count={count}, score={best_score:.6f}")
    
    return best_a, best_score

        else:
            top_k = np.argpartition(c, -n_peaks)[-min(n_peaks, len(c)):]
        
        # Compute subgradient (average of sensitivities at top peaks)
        s = np.zeros(n)
        for k in top_k:
            k = int(k)
            for j in range(n):
                r1 = k - j
                r2 = j - k
                if 0 <= r1 < n:
                    s[j] += a[r1]
                if 0 <= r2 < n:
                    s[j] += a[r2]
        s /= len(top_k)
        
        # Normalize subgradient
        s_norm = float(np.linalg.norm(s))
        if s_norm < 1e-15:
            break
        s = s / s_norm
        
        # Step
        a_new = a - alpha * s
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new * S / s_new
        
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Adapt step size (simple: reduce if score increased)
        if sc > _score(a):
            alpha *= 0.9
        else:
            alpha = min(alpha * 1.05, 0.1)
        
        a = a_new
    
    print(f"    Subgrad polish final: score={best_score:.6f}")
    return best_a, best_score

def _interpolate(seq, n_target):
    n_src = len(seq)
    if n_src == n_target:
        return seq.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, seq)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

"""
Minimizes J(a) = 2n*max(conv(a,a))/sum(a)^2 using a multi-phase approach:
1. Parametric initialization with various window functions
2. Alternating LP with adaptive blending and M-pushing at multiple scales
3. Random perturbation search (simulated annealing style)
4. Subgradient polish with top-peak focus
5. Final LP refinement
"""

import numpy as np
import time
from scipy.optimize import linprog
import math

def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score_fast(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2 without full safety checks."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    """Build matrix C where (C @ g)[k] = (f*g)[k] for full linear convolution."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C

def _lp_solve(f, M, time_limit=20):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0. Returns g or None."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None

                    break  # infeasible, don't try tighter

        # Adaptive blend: decrease slowly, but reset if making progress
        blend *= 0.99
        blend = max(0.15, blend)
        if no_improve == 0:
            blend = min(0.8, blend + 0.02)

        if it % 60 == 0:
            print(f"      LP iter {it}: score={best_score:.6f}, M={M:.6e}, no_improve={no_improve}")

    return best_a, best_score

def _simulated_annealing(a_init, deadline, max_iters=200000):
    """Random perturbation search with simulated annealing acceptance."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    a = a / S

    best_a = a.copy()
    best_score = _score_fast(a)
    rng = np.random.default_rng(777)

    T = 0.005  # initial temperature
    T_min = 1e-7
    cool_rate = 0.9997

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Random perturbation: pick a random subset of elements and nudge them
        n_perturb = rng.integers(1, max(2, n // 20) + 1)
        idx = rng.integers(0, n, size=n_perturb)
        scale = rng.uniform(0.02, 0.15)
        deltas = rng.normal(0, scale, size=n_perturb)

        a_try = a.copy()
        a_try[idx] += deltas * a[idx]
        a_try = np.maximum(a_try, 0.0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = a.copy()
            s_try = S
        a_try = a_try * S / s_try

        sc = _score_fast(a_try)

        # Acceptance criterion
        if sc < best_score:
            best_score = sc
            best_a = a_try.copy()
            a = a_try
        elif sc <= _score_fast(a) or rng.random() < math.exp((_score_fast(a) - sc) / T):
            a = a_try

        T = max(T_min, T * cool_rate)

        if it % 50000 == 0:
            print(f"    SA iter {it}: best={best_score:.6f}, current={_score_fast(a):.6f}, T={T:.2e}")

    return best_a, best_score

def _subgradient_polish(a_init, deadline, max_iters=100000, n_peaks=8):
    """Subgradient method focusing on top peaks with adaptive step size."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    a = a / S

    best_a = a.copy()
    best_score = _score_fast(a)

    alpha = 0.02
    alpha_min = 1e-6
    prev_score = best_score

    # Precompute index arrays for sensitivity
    j_arr = np.arange(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        max_c = float(np.max(c))
        current_score = 2.0 * n * max_c  # since sum=1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        # Find top peaks
        k = min(n_peaks, len(c))
        top_k = np.argpartition(c, -k)[-k:]

        # Compute averaged sensitivity
        s = np.zeros(n)
        for ki in top_k:
            ki = int(ki)
            # s[j] += a[ki-j] + a[j-ki] (where valid)
            r1 = ki - j_arr
            r2 = j_arr - ki
            m1 = (r1 >= 0) & (r1 < n)
            m2 = (r2 >= 0) & (r2 < n)
            s[m1] += a[r1[m1]]
            s[m2] += a[r2[m2]]
        s /= k

        # Normalize and step
        s_norm = float(np.linalg.norm(s))
        if s_norm < 1e-15:
            break
        s = s / s_norm

        a_new = a - alpha * s
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new * S / s_new

        sc_new = _score_fast(a_new)

        # Adaptive step size
        if sc_new <= current_score:
            alpha = min(0.05, alpha * 1.02)
        else:
            alpha *= 0.92
            a_new = a  # revert
            sc_new = current_score

        a = a_new

        if it % 20000 == 0:
            print(f"    SG iter {it}: best={best_score:.6f}, current={sc_new:.6f}, alpha={alpha:.2e}")

    return best_a, best_score

def _parametric_init(n, seed=0):
    """Try various parametric families and return the best scoring one."""
    t = np.linspace(0, 1, n)
    candidates = []

    # Various window functions
    candidates.append(np.ones(n))
    candidates.append(0.5 * (1 - np.cos(2 * np.pi * t)))  # Hann
    candidates.append(0.54 - 0.46 * np.cos(2 * np.pi * t))  # Hamming
    candidates.append(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t))  # Blackman
    candidates.append(np.cos(np.pi * t / 2) ** 2)  # Raised cosine
    candidates.append(1 - np.abs(2 * t - 1))  # Triangular
    candidates.append(np.exp(-2 * t))
    candidates.append(np.exp(-5 * t))
    candidates.append((1 - t) * (1 + 0.5 * np.sin(2 * np.pi * 3 * t)))
    candidates.append((1 - t) * (1 + 0.3 * np.cos(2 * np.pi * 7 * t)))
    candidates.append(np.exp(-((t - 0.3)**2) / 0.02) + 0.5 * np.exp(-((t - 0.7)**2) / 0.05))
    candidates.append(t**2 * (3 - 2*t))
    candidates.append((1-t)**1.5)
    candidates.append(np.exp(-3*t) * (1 + 0.4*np.sin(2*np.pi*5*t)))

    rng = np.random.default_rng(seed)
    # Random initializations
    for _ in range(5):
        candidates.append(np.abs(rng.normal(0, 1, n)))
    for _ in range(3):
        candidates.append(np.exp(rng.uniform(0.5, 3) * t) / np.exp(rng.uniform(0.5, 3) * t)[0])

    best_a = None
    best_sc = float('inf')
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = float(np.sum(c))
        if s < 1e-10:
            continue
        c = c / s
        sc = _score_fast(c)
        if sc < best_sc:
            best_sc = sc
            best_a = c.copy()

    return best_a, best_sc

def _interpolate(seq, n_target):
    """Interpolate a sequence to a new length, preserving non-negativity and normalizing."""
    n_src = len(seq)
    if n_src == n_target:
        return seq.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, seq)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

    Multi-phase optimizer minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses parametric init, alternating LP at multiple scales, simulated annealing,

    # Phase 1: LP at the best n and nearby values

    if best_a is not None:
        n_main = min(max(len(best_a), 200), 1000)
    else:
        n_main = 500

    n_list = sorted(set([n_main, max(200, n_main - 100), min(1000, n_main + 100), 500]))
    # Sort by proximity to n_main
    n_list = sorted(n_list, key=lambda x: abs(x - n_main))

    for n in n_list:

        # Get initial sequence
