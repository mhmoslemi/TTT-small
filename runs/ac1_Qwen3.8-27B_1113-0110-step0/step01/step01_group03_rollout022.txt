def solve_lp_direction(a, M):

    Solve the LP: maximize sum(g) subject to g >= 0 and (a * g)[k] <= M for all k.
    Returns the optimal g.
    """

    # This is a linear function of g.
    # For each k (0 to 2n-2), the coefficient of g[j] in (a*g)[k] is a[k-j] (when 0 <= k-j <= n-1).

    # and g >= 0 (handled by bounds).

    # This is a "Toeplitz-like" matrix.

    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []

    if res.status == 0:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n)
