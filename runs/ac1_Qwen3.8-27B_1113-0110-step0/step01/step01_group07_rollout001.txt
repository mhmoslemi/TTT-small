```python
import time
import numpy as np

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)

def compute_conv(a, L=None):
    n = a.size
    if L is None:
        L = next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    return np.where(np.isfinite(c), c, 0.0)

def conv_grad(a, c, L=None):
    """Gradient of max(c) w.r.t. a, using soft attention on c."""
    n = a.size
    cm = c.max()
```

def resize_seq(a, m):
    a = normalize_positive(a)
    if a is None:
        return np.ones(m) / m
    n = a.size
    if n == m:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0)
    s = out.sum()
    if s <= 0:
        return np.ones(m) / m
    return out / s


def cheb_parents(n, D=6):

    # Various Chebyshev combinations

        for mode in [0, 1]:

            h = np.maximum(h, 0)
            s = h.sum()
            if s > 0:
