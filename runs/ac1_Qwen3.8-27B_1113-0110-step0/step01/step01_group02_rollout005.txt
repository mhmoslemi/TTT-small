from scipy.optimize import linprog

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _lp_solve(f, M, n):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0"""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi+1)
        A_ub[k, idx] = f[k - idx]
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.success:
        return res.x
    return None

def _lp_optimize(f0, time_budget, rng):

    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    if s < 1e-15:
        return f0, np.inf
    f = f / s
    
    c = _fft_convolve(f)
    best_score = 2.0 * n * float(c.max())
    best_f = f.copy()
    
    t = 0.25  # initial blend parameter
    t_decay = 0.998
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(f)
        M = float(c.max())
        score = 2.0 * n * M
        
        if score < best_score - 1e-10:
            best_score = score
            best_f = f.copy()
            no_improve = 0
            if time.monotonic() - last_print > 5.0:
                print(f"    [LP] it={it}, score={score:.6f}, t={t:.4f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart with perturbation
            f = best_f * np.exp(rng.normal(0, 0.05, n))
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            no_improve = 0
            t = 0.25
            it += 1
            continue
        
        # Solve LP
        f_safe = np.maximum(f, 1e-10)
        g = _lp_solve(f_safe, M, n)
        
        if g is None:
            # LP failed, fall back to subgradient
            a = f.copy()
            k_max = int(np.argmax(c))
            grad = np.zeros(n)
            lo = max(0, k_max - n + 1)
            hi = min(n - 1, k_max)
            j_range = np.arange(lo, hi+1)
            grad[lo:hi+1] = 2.0 * f[k_max - j_range]
            a = a * np.exp(-0.1 * grad / max(grad.max(), 1e-15))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            f = a
            t *= t_decay
            it += 1
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
            continue
        f = f / s
        
        t *= t_decay
        it += 1

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score(a):
    c = _fft_convolve(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    return 2.0 * len(a) * float(np.max(c)) / (s * s)

def _lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None."""
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            A_ub[k, lo:hi+1] = f[k - idx]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum > 1e-15:
                return g / g_sum
    except Exception as e:
        print(f"    LP error: {e}")
    
    return None
