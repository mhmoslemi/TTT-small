    k_peaks = np.where(mask)[0]
    g = np.zeros(n)
    for k in k_peaks:
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_max >= j_min:
            s_start = k - j_max
            s_end = k - j_min
            g[j_min:j_max + 1] += a[s_start:s_end + 1][::-1]
    g /= len(k_peaks)
    return g, len(k_peaks)


def build_lp_matrix(a):
    """Build the (2n-1) x n constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_max >= j_min:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_direction(a, M, time_limit=25):
    """Solve max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'time_limit': time_limit, 'presolve': True}
        )
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Phase 1: LP-based global direction with convex combination steps.
    Phase 2: Smoothed subgradient (log-sum-exp) local refinement.
    Phase 3: Hard subgradient fine-tuning.
    """
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(512, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(512, dtype=np.float64)

    n = len(parent)
    S_parent = parent.sum()
    if S_parent < 0.01:
        parent = np.ones(n)
        S_parent = n

    a = normalize_sum(parent, n)
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    print(f"Initial: score={best_score:.8f}, n={n}")

    deadline = time.monotonic() + budget_s - 40.0

    # === Phase 1: LP-based iteration ===
    lp_count = 0
    lp_fail_count = 0
    step_frac = 0.35

    while time.monotonic() < deadline:
        a = normalize_sum(a, n)
        c = fft_self_conv(a)
        M = c.max()

        # Solve LP
        t0 = time.monotonic()
        b_dir = solve_lp_direction(a, M * 1.0001, time_limit=20)
        lp_time = time.monotonic() - t0

        if b_dir is not None and b_dir.sum() > 0:
            lp_count += 1
            b_sum = b_dir.sum()
            b_norm = b_dir / b_sum * n

            # Take convex combination
            t = step_frac
            # Try a few step sizes and pick the best
            best_t = t
            best_M_try = np.inf
            for t_try in [t, t * 0.7, t * 1.3, 0.2, 0.5]:
                a_try = (1 - t_try) * a + t_try * b_norm
                a_try = np.maximum(a_try, 0.0)
                a_try = normalize_sum(a_try, n)
                c_try = fft_self_conv(a_try)
                M_try = c_try.max()
                if M_try < best_M_try:
                    best_M_try = M_try
                    best_t = t_try

            a = (1 - best_t) * a + best_t * b_norm
            a = np.maximum(a, 0.0)
            a = normalize_sum(a, n)

            if lp_count <= 5 or lp_count % 5 == 0:
                score = float(evaluate_sequence(a.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_a = a.copy()
                    best_score = score
                print(f"  LP#{lp_count} ({lp_time:.1f}s): t={best_t:.2f}, "
                      f"b_sum_ratio={b_sum/n:.4f}, score={best_score:.8f}")
        else:
            lp_fail_count += 1
            print(f"  LP#{lp_count + lp_fail_count} failed ({lp_time:.1f}s)")

        # Evaluate and track
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_a = a.copy()
            best_score = score

        # If LP is taking too long, reduce frequency
        if lp_time > 40 and lp_count > 10:
            break

        # Safety: if no improvement after several LPs, perturb
        if lp_count > 0 and lp_count % 8 == 0:
            a_check = normalize_sum(a, n)
            c_check = fft_self_conv(a_check)
            a_best_check = normalize_sum(best_a, n)
            c_best = fft_self_conv(a_best_check)
            if c_check.max() > c_best.max():
                a = a_best_check.copy()
                # Small perturbation
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = normalize_sum(a, n)

    # === Phase 2: Smoothed subgradient refinement ===
    a = best_a.copy()
    tau = 0.3
    lr = 0.5
    it = 0
    no_improve = 0

    while time.monotonic() < deadline - 30:
        a = normalize_sum(a, n)
        c = fft_self_conv(a)
        M = c.max()

        g = smoothed_gradient(a, c, tau)
        g = g - np.mean(g)

        # Line search
        step = lr
        accepted = False
        for _ in range(10):
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize_sum(a_try, n)
            c_try = fft_self_conv(a_try)
            if c_try.max() < M * (1.0 - 1e-7):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            a = a - step * g
            a = np.maximum(a, 0.0)
            a = normalize_sum(a, n)

        it += 1
        no_improve += 0 if accepted else 1

        if it % 10 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_a = a.copy()
                best_score = score
                no_improve = 0
                if it % 100 == 0 or best_score < 1.51:
                    print(f"  P2 it={it}: score={best_score:.8f}, tau={tau:.3f}")

            if no_improve >= 40:
                no_improve = 0
                tau *= 0.7
                lr *= 0.75
                if tau < 0.005:
                    break

    # === Phase 3: Hard subgradient fine-tuning ===
    a = best_a.copy()
    lr = 0.02
    it3 = 0
    no_improve3 = 0

    while time.monotonic() < deadline - 5:
        a = normalize_sum(a, n)
        c = fft_self_conv(a)
        M = c.max()

        g, n_peaks = hard_subgradient(a, c, rel_eps=1e-5)
        g = g - np.mean(g)

        step = lr
        accepted = False
        for _ in range(12):
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize_sum(a_try, n)
            c_try = fft_self_conv(a_try)
            if c_try.max() < M * (1.0 - 1e-8):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            if no_improve3 > 30:
                # Micro-perturbation
                a = a + rng.normal(0, 0.0005 * max(a.mean(), 0.1), n)
                a = np.maximum(a, 0.0)
                a = normalize_sum(a, n)
                no_improve3 = 0
            else:
                a = a - step * g
                a = np.maximum(a, 0.0)
                a = normalize_sum(a, n)

        it3 += 1
        no_improve3 += 0 if accepted else 1

        if it3 % 20 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_a = a.copy()
                best_score = score
                no_improve3 = 0
                if it3 % 200 == 0 or best_score < 1.505:
                    print(f"  P3 it={it3}: score={best_score:.8f}")

            if no_improve3 >= 40:
                lr *= 0.7
                no_improve3 = 0
                if lr < 1e-5:
                    break

    # Final validation
    a = normalize_sum(best_a, n)
    final_score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        best_a = a.copy()

    print(f"Final: score={best_score:.8f}, n={n}, LPs={lp_count}")
    return [float(x) for x in best_a]
