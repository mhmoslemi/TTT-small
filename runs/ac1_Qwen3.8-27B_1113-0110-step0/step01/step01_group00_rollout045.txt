import numpy as np

def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _project_to_simplex(a, target_sum=1.0):
    """Project a onto {x: sum(x) = target_sum, x >= 0} using Duchi et al."""
    n = len(a)
    v = np.sort(a)[::-1]
    cssv = np.cumsum(v)
    rho_candidates = np.nonzero(v * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        theta = (cssv[-1] - target_sum) / n
    else:
        rho = rho_candidates[-1]
        theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)


def _cross_corr_fft(a, w):
    """Compute cross-correlation: out[i] = sum_j a[j] * w[j+i] for i=0..len(a)-1.
    a has length n, w has length 2n-1."""
    n = len(a)
    L = 1 << (n + len(w) - 1).bit_length()
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    corr = np.fft.irftp(np.conj(A) * W, L)
    return corr[:n]


def _gradient_softobj(a, p):
    """Gradient of soft-max(conv(a,a)): grad[i] = 2 * sum_j a[j]*p[i+j]."""
    corr = _cross_corr_fft(a, p)
    return 2.0 * corr

    n = len(a)

    if s < 1e-10:
        a = np.ones(n) / n

    a = a / s

    best_score = _score(a)

        if deadline is not None and time.monotonic() > deadline:
            break

        M = float(np.max(c))

        # = C[k] + 2 * (convolve where one side is a, other is da)

        # For efficiency, build G as a convolution matrix
        # G[k, i] = 2 * a[k-i] if 0 <= k-i < n, else 0
        # For variable da[i], the linearized constraint is:
        # C[k] + sum_i G[k,i] * da[i] <= M_new

        # Objective: minimize M_new
        # Constraints:
        #   sum_i G[k,i] * da[i] - M_new <= -C[k]  for k = 0,...,2n-2
        #   sum_i da[i] = 0  (preserve normalization)
        #   da[i] >= -a[i]  (preserve non-negativity: a[i] + da[i] >= 0)

        # Build A_ub matrix

        n_convs = 2 * n - 1

        b_ub = np.zeros(n_convs + n)

            valid_start = max(0, k - (n - 1))

                A_ub[k, i] = 2.0 * a[k - i]
            A_ub[k, n] = -1.0
            b_ub[k] = -c[k]

        for i in range(n):
            A_ub[n_convs + i, i] = -1.0
            b_ub[n_convs + i] = a[i]

        # Equality: sum(da) = 0
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        # Objective: minimize M_new
        c_obj = np.zeros(n_vars)
        c_obj[n] = 1.0

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=[(None, None)] * n + [(None, None)],
                             method='highs', options={'presolve': True,
                                                       'time_limit': 5})
            if result.status == 0:
                da = result.x[:n]
                M_new = result.x[n]

                # Take a fraction of the step (line search)
                # Try t = 1.0, 0.5, 0.25, ...
                found = False
                for frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
                    a_trial = a + frac * da
                    a_trial = np.maximum(a_trial, 0.0)
                    s_trial = float(np.sum(a_trial))
                    if s_trial < 1e-10:
                        continue
                    a_trial = a_trial / s_trial
                    score_trial = _score(a_trial)
                    M_trial = float(np.max(_fft_convolve(a_trial)))
                    if M_trial < M:
                        a = a_trial
                        M = M_trial
                        found = True
                        break

                if not found:
                    # Try smaller fractions or skip
                    pass

                score = _score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if it % 5 == 0:
                        print(f"    LP iter {it}: score={best_score:.8f}, M={M:.6e}")

                if abs(M - current_M) < 1e-12 * M:
                    break
                current_M = M

        except Exception as e:
            break

    return best_a, best_score


def _direct_lp(a_init, M, deadline):
    """Direct LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a_init)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a_init
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 10})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _alternating_refine(a_init, max_iters=100, deadline=None, blend=0.6):
    """Alternating LP with adaptive blending."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _direct_lp(a, M, deadline)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g

        # Adaptive blend: use more of g if it's very different
        diff = float(np.mean(np.abs(g - a)))
        blend_local = min(1.0, blend + diff)
        a_new = (1.0 - blend_local) * a + blend_local * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            break
        a_new = a_new / s_new

        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if it % 20 == 0:
                print(f"    AltLP iter {it}: score={best_score:.8f}")
        a = a_new

    return best_a, best_score


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def _make_interesting_initial(n, variant, rng):
    """Create diverse initial sequences."""
    t = np.linspace(0, 1, n)
    if variant == 0:
        a = 1.0 - t
    elif variant == 1:
        a = np.exp(-3.0 * t)
    elif variant == 2:
        a = (1.0 - t) * (1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 3:
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05)
    elif variant == 4:
        a = t ** 1.5 * (1.0 - t) ** 0.5 + 1e-6
    elif variant == 5:
        a = np.abs(np.cos(np.pi * t)) ** 2 * (1.0 + 0.3 * np.cos(2 * np.pi * 4 * t))
        a = np.maximum(a, 0.0)
    elif variant == 6:
        a = (1.0 - t) * (1.0 + 0.4 * np.sin(2.0 * np.pi * 6.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 7:
        a = np.exp(-2.0 * t) * (1.0 + 0.5 * np.sin(2.0 * np.pi * 3.0 * t + 1.0))
        a = np.maximum(a, 0.0)
    elif variant == 8:
        # Gaussian bump off-center
        a = np.exp(-((t - 0.4) ** 2) / 0.03)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    s = float(np.sum(a))
    a = a / s
    return a


def propose_candidate():
    """
    Multi-phase optimizer:
    Phase 1: FISTA with soft-max at multiple n values
    Phase 2: Alternating LP refinement
    Phase 3: Linearized LP polish
    Phase 4: Random restarts at best n
    """
    total_deadline = time.monotonic() + 2350.0
    rng = np.random.default_rng(123)

    best_a = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = _score(parent)
            print(f"Parent: n={parent.size}, score={parent_score:.8f}")
            best_a = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    # Phase 1: FISTA at various n
    print("\n=== Phase 1: FISTA optimization ===")
    n_list = [300, 400, 500, 600, 800]
    for n in n_list:
        if time.monotonic() > total_deadline - 600:
            break
        phase_deadline = min(total_deadline - 600, time.monotonic() + 250.0)
        print(f"\n  --- FISTA n={n} ---")

        # Multiple initializations
        inits = []
        if best_a is not None:
            inits.append(_interpolate_sequence(best_a, n))
        inits.append(_make_interesting_initial(n, 0, rng))
        inits.append(_make_interesting_initial(n, 3, rng))
        inits.append(_make_interesting_initial(n, 4, rng))

        for idx, a_init in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            print(f"    init {idx}", end=" ")
            a_opt, score = _fista_optimize(a_init, phase_deadline,
                                            tau_start=20, tau_end=150,
                                            topk_frac=0.05)
            print(f" score={score:.8f}")
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    ** New best: {best_score:.8f} (n={n})")

    print(f"\nAfter Phase 1: best={best_score:.8f}, n={len(best_a) if best_a is not None else 0}")

    # Phase 2: Alternating LP refinement on the best
    print("\n=== Phase 2: Alternating LP refinement ===")
    if best_a is not None:
        n_best = len(best_a)
        # Try at the best n and a slightly smaller one
        for n_try in [min(n_best, 500), min(n_best, 400), n_best]:
            if time.monotonic() > total_deadline - 300:
                break
            a_ds = _interpolate_sequence(best_a, n_try) if n_try != n_best else best_a.copy()
            phase_deadline = min(total_deadline - 300, time.monotonic() + 200.0)
            print(f"\n  --- AltLP n={n_try} ---")
            a_opt, score = _alternating_refine(a_ds, max_iters=200,
                                                deadline=phase_deadline, blend=0.5)
            if n_try != n_best:
                a_opt = _interpolate_sequence(a_opt, n_best)
                score = _score(a_opt)
            print(f"  AltLP result: score={score:.8f}")
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  ** New best: {best_score:.8f} (n={n_best})")

    # Phase 3: Linearized LP polish
    print("\n=== Phase 3: Linearized LP polish ===")
    if best_a is not None and time.monotonic() < total_deadline - 200:
        n_best = len(best_a)
        # For LP polish, use a manageable size
        n_lp = min(n_best, 400)
        a_ds = _interpolate_sequence(best_a, n_lp) if n_best != n_lp else best_a.copy()
        polish_deadline = min(total_deadline - 200, time.monotonic() + 150.0)
        print(f"  Polishing at n={n_lp}")
        a_opt, score = _linearized_lp_polish(a_ds, max_iters=30, deadline=polish_deadline)
        if n_best != n_lp:
            a_opt_full = _interpolate_sequence(a_opt, n_best)
            score = _score(a_opt_full)
        else:
            a_opt_full = a_opt
        print(f"  LP polish result: score={score:.8f}")
        if score < best_score:
            best_score = score
            best_a = a_opt_full.copy()
            print(f"  ** New best: {best_score:.8f}")

    # Phase 4: Random restarts at best n
    print("\n=== Phase 4: Random restarts ===")
    if best_a is not None and time.monotonic() < total_deadline - 100:
        n_best = len(best_a)
        n_restart = min(n_best, 500)
        for restart in range(5):
            if time.monotonic() > total_deadline - 60:
                break
            # Perturb the best solution
            noise_scale = 0.05 / n_restart
            a_perturbed = best_a.copy()
            if n_best != n_restart:
                a_perturbed = _interpolate_sequence(best_a, n_restart)
            a_perturbed += rng.normal(0, noise_scale, n_restart)
            a_perturbed = np.maximum(a_perturbed, 0)
            s = float(np.sum(a_perturbed))
            if s < 1e-10:
                continue
            a_perturbed = a_perturbed / s

            phase_deadline = min(total_deadline - 60, time.monotonic() + 100.0)
            a_opt, score = _fista_optimize(a_perturbed, phase_deadline,
                                            tau_start=50, tau_end=300,
                                            topk_frac=0.03)
            if n_best != n_restart:
                a_opt = _interpolate_sequence(a_opt, n_best)
                score = _score(a_opt)
            print(f"  Restart {restart}: score={score:.8f}")
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  ** New best: {best_score:.8f}")

    # Phase 5: Final FISTA with high tau for precision
    print("\n=== Phase 5: Final precision FISTA ===")
    if best_a is not None and time.monotonic() < total_deadline - 30:
        n_best = len(best_a)
        n_final = min(n_best, 500)
        a_final = _interpolate_sequence(best_a, n_final) if n_best != n_final else best_a.copy()
        final_deadline = total_deadline - 10
        a_opt, score = _fista_optimize(a_final, final_deadline,
                                       tau_start=100, tau_end=500,
                                       topk_frac=0.02)
        if n_best != n_final:
            a_opt = _interpolate_sequence(a_opt, n_best)
            score = _score(a_opt)
        print(f"  Final FISTA: score={score:.8f}")
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.8f}")

    # Final verification
    print(f"\n=== Final: score={best_score:.8f}, n={len(best_a) if best_a is not None else 0} ===")
    if best_a is not None:
        result = [float(x) for x in best_a]
        verified = float(evaluate_sequence(result))
        print(f"Verified: {verified:.8f}")
        return result

    # Fallback
    return [1.0] * 100
