"""
Multi-stage optimizer for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
Uses alternating LP, subgradient descent with adaptive step sizing,
coordinate descent, and simulated annealing for local refinement.
"""
import numpy as np
import time
import math
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build the convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0. Returns g or None."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.8):
    """Run alternating LP: fix f, solve LP for g, blend, repeat."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        blend = max(0.05, blend * 0.96)

    return best_a, best_score


def _subgradient_opt(a_init, deadline, n_peaks=10, lr_init=0.05, n_iter_cap=50000):
    """Subgradient optimization with adaptive step size for flat autocorrelation."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    lr = lr_init
    no_improve = 0

    for it in range(n_iter_cap):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c = np.maximum(c, 0.0)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c  # since sum(a)=1

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-8:
                    break

        # Top K peaks
        k = min(n_peaks, len(c))
        top_k = np.argpartition(c, -k)[-k:]

        # Subgradient
        s_vec = np.zeros(n)
        for idx in top_k:
            ki = int(idx)
            j = np.arange(n)
            target = ki - j
            mask = (target >= 0) & (target < n)
            s_vec[mask] += a[target[mask]]

        s_vec /= k
        # Normalize direction
        s_vec = s_vec - max_c / 1.0  # subtract mean contribution

        # Update
        a = a - lr * s_vec
        a = np.maximum(a, 0.0)
        S_new = float(np.sum(a))
        if S_new < 1e-10:
            a = np.ones(n) / n
            S_new = 1.0
        a = a / S_new

        # Decay learning rate slowly
        lr *= 0.99995

    return best_a, best_score / 1.0  # score is already 2n*max_c since sum=1


def _coordinate_descent(a_init, deadline, max_sweeps=200):
    """Coordinate descent: optimize one element at a time."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Optimize a[i] given all others fixed
            a_save = a[i]
            # Compute base convolution (with a[i]=0)
            a_zero = a.copy()
            a_zero[i] = 0.0
            c_base = _fft_convolve(a_zero)
            S_base = float(np.sum(a_zero))

            # c(k; x) = c_base[k] + 2*x*a[k-i] (for k != 2i) or c_base[k] + x^2 (for k=2i)
            # score(x) = 2n * max_k c(k;x) / (S_base + x)^2

            # Golden section search for x in [0, a_save * 3] or [0, 1]
            lo = 0.0
            hi = max(a_save * 3.0, 0.1)
            if a_save < 1e-15:
                hi = 1.0

            for _ in range(30):
                m1 = lo + 0.382 * (hi - lo)
                m2 = lo + 0.618 * (hi - lo)

                # Evaluate score at m1
                a_test = a_zero.copy()
                a_test[i] = m1
                sc1 = _score(a_test) if float(np.sum(a_test)) > 0.01 else float('inf')

                # Evaluate score at m2
                a_test[i] = m2
                sc2 = _score(a_test) if float(np.sum(a_test)) > 0.01 else float('inf')

                if sc1 < sc2:
                    hi = m2
                else:
                    lo = m1

            optimal_x = (lo + hi) / 2.0
            a[i] = optimal_x
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_score


def _simulated_annealing(a_init, deadline, n_moves=5000):
    """Simulated annealing with structured moves."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    T_init = 0.02
    T_final = 0.0001
    rng = np.random.default_rng(123)

    for move_i in range(n_moves):
        if time.monotonic() > deadline:
            break
        T = T_init * (T_final / T_init) ** (move_i / n_moves)

        # Generate move
        move_type = rng.integers(0, 4)
        a_new = a.copy()

        if move_type == 0:
            # Perturb random subset
            n_perturb = max(5, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_new[idx] += rng.normal(0, 0.1, n_perturb)
        elif move_type == 1:
            # Scale a contiguous block
            block_len = max(5, n // 10)
            start = rng.integers(0, max(1, n - block_len))
            factor = 1.0 + rng.normal(0, 0.3)
            factor = max(0.1, min(3.0, factor))
            a_new[start:start + block_len] *= factor
        elif move_type == 2:
            # Add sinusoidal perturbation
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.linspace(0, 1, n)
            perturbation = 0.1 * np.sin(2 * np.pi * freq * t + phase)
            a_new += perturbation
        else:
            # Swap two random blocks
            b1 = rng.integers(0, n // 2)
            b2 = rng.integers(n // 2, n)
            swap_len = max(3, n // 30)
            b1 = min(b1, n - swap_len)
            b2 = min(b2, n - swap_len)
            temp = a_new[b1:b1 + swap_len].copy()
            a_new[b1:b1 + swap_len] = a_new[b2:b2 + swap_len]
            a_new[b2:b2 + swap_len] = temp

        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        new_score = _score(a_new)

        # Accept/reject
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


def _make_initial(n, variant, rng):
    """Create diverse initial non-negative sequences."""
    t = np.linspace(0, 1, n)
    if variant == 0:
        a = 1.0 - t
    elif variant == 1:
        a = np.exp(-3.0 * t)
    elif variant == 2:
        a = (1.0 - t) * (1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 3:
        a = (1.0 - t) * (1.0 + 0.7 * np.sin(2.0 * np.pi * 5.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 4:
        a = np.abs(rng.normal(0, 1, n))
    elif variant == 5:
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05)
    elif variant == 6:
        a = t ** 2 * (3.0 - 2.0 * t)
    elif variant == 7:
        a = (1.0 - t) ** 1.5 * (1.0 + 0.4 * np.cos(2.0 * np.pi * 2.0 * t + 1.0))
        a = np.maximum(a, 0.0)
    elif variant == 8:
        a = (1.0 - t) * (1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 9:
        a = np.exp(-2.0 * t) * (1.0 + 0.5 * np.sin(2.0 * np.pi * 4.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 10:
        # Parabolic with edge emphasis
        a = 1.0 - 4.0 * t * (1.0 - t)
        a = np.maximum(a, 0.0)
    elif variant == 11:
        # Two-lobe
        a = np.exp(-((t - 0.25) ** 2) / 0.01) + np.exp(-((t - 0.75) ** 2) / 0.01)
    elif variant == 12:
        # Cosine window
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif variant == 13:
        # Raised cosine with notch
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) * (1.0 - 0.3 * np.cos(4.0 * np.pi * t))
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def propose_candidate():
    """
    Multi-phase optimizer minimizing J(a)=2n*max(conv(a,a))/sum(a)^2.
    Uses alternating LP, subgradient descent, coordinate descent,
    and simulated annealing across multiple sequence lengths.
    """
    total_deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(42)

    best_a = None
    best_score = float('inf')

    # Try parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = _score(parent)
            print(f"Parent sequence: n={parent.size}, score={parent_score:.6f}")
            best_a = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    # === Phase 1: Alternating LP ===
    print("\n=== Phase 1: Alternating LP ===")
    n_list = [200, 300, 400, 500, 600, 800]
    for n in n_list:
        if time.monotonic() > total_deadline - 300:
            break
        phase_deadline = min(total_deadline - 300, time.monotonic() + 150.0)
        print(f"\n  --- LP n={n} ---")

        # Build initialization list
        inits_list = []
        if best_a is not None:
            inits_list.append(_interpolate_sequence(best_a, n))
        for v in [2, 5, 7, 10, 11, 13]:
            inits_list.append(_make_initial(n, v, rng))

        for a_init in inits_list:
            if time.monotonic() > phase_deadline:
                break
            # Try two different blends
            for blend in [0.7, 0.9]:
                if time.monotonic() > phase_deadline:
                    break
                a_opt, score = _alternating_lp(a_init, 200, phase_deadline, initial_blend=blend)
                if score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"    ** New best: {best_score:.6f} (n={n}, blend={blend})")
                print(f"    LP n={n} blend={blend}: {score:.6f}")

    print(f"\nAfter Phase 1: best={best_score:.6f}, n={len(best_a) if best_a is not None else 0}")

    # === Phase 2: Subgradient refinement ===
    print("\n=== Phase 2: Subgradient refinement ===")
    if best_a is not None:
        n_best = len(best_a)
        # Refine at current n
        sub_deadline = min(total_deadline - 200, time.monotonic() + 250.0)
        for n_peak in [5, 10, 20]:
            if time.monotonic() > sub_deadline:
                break
            a_opt, score = _subgradient_opt(best_a, sub_deadline, n_peaks=n_peak, lr_init=0.03, n_iter_cap=30000)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    ** Subgrad n_peaks={n_peak}: new best {best_score:.6f}")
            else:
                print(f"    Subgrad n_peaks={n_peak}: {score:.6f}")

        # Also try at nearby n values
        for n_try in [n_best - 100, n_best + 100]:
            if time.monotonic() > sub_deadline or n_try < 50:
                continue
            a_interp = _interpolate_sequence(best_a, n_try)
            sub_deadline2 = min(total_deadline - 200, time.monotonic() + 100.0)
            a_opt, score = _subgradient_opt(a_interp, sub_deadline2, n_peaks=10, lr_init=0.03, n_iter_cap=20000)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    ** Subgrad n={n_try}: new best {best_score:.6f}")
            else:
                print(f"    Subgrad n={n_try}: {score:.6f}")

    print(f"\nAfter Phase 2: best={best_score:.6f}")

    # === Phase 3: Coordinate descent ===
    print("\n=== Phase 3: Coordinate Descent ===")
    if best_a is not None:
        cd_deadline = min(total_deadline - 100, time.monotonic() + 300.0)
        a_opt, score = _coordinate_descent(best_a, cd_deadline, max_sweeps=50)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"    ** Coord descent: new best {best_score:.6f}")
        else:
            print(f"    Coord descent: {score:.6f}")

    # === Phase 4: Simulated annealing ===
    print("\n=== Phase 4: Simulated Annealing ===")
    if best_a is not None:
        sa_deadline = min(total_deadline - 50, time.monotonic() + 200.0)
        a_opt, score = _simulated_annealing(best_a, sa_deadline, n_moves=5000)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"    ** SA: new best {best_score:.6f}")
        else:
            print(f"    SA: {score:.6f}")

    # === Phase 5: Final LP polish ===
    print("\n=== Phase 5: Final LP Polish ===")
    if best_a is not None and time.monotonic() < total_deadline - 30:
        n_best = len(best_a)
        n_lp = min(n_best, 500)
        polish_deadline = total_deadline - 10
        if n_best <= 500:
            a_ds = best_a.copy()
        else:
            a_ds = _interpolate_sequence(best_a, n_lp)
        print(f"  Polishing with LP at n={n_lp}")
        a_lp, score_lp = _alternating_lp(a_ds, 50, polish_deadline, initial_blend=0.5)
        if n_best != n_lp:
            a_up = _interpolate_sequence(a_lp, n_best)
            score_lp = _score(a_up)
        else:
            a_up = a_lp
        if score_lp < best_score:
            best_score = score_lp
            best_a = a_up.copy()
            print(f"  ** Polished: {best_score:.6f}")
        else:
            print(f"  Polish: {score_lp:.6f}")

    # === Phase 6: Quick subgradient on final best ===
    if best_a is not None and time.monotonic() < total_deadline - 20:
        final_deadline = total_deadline - 5
        a_opt, score = _subgradient_opt(best_a, final_deadline, n_peaks=8, lr_init=0.01, n_iter_cap=10000)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** Final subgrad: {best_score:.6f}")

    # Final verification
    print(f"\n=== Final: score={best_score:.6f}, n={len(best_a) if best_a is not None else 0} ===")
    if best_a is not None:
        result = [float(x) for x in best_a]
        verified = float(evaluate_sequence(result))
        print(f"Verified: {verified:.6f}")
        return result

    # Absolute fallback
    return [1.0] * 100
