```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences a.
Combines LP-based structured iteration with exponential gradient refinement
and population diversity for global exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_convolve(x, y):
    """Full convolution via FFT."""
    nx, ny = len(x), len(y)
    L = nx + ny - 1
    N = 1 << (L - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(x, N) * np.fft.rfft(y, N), N)[:L].real


def _auto_convolve(a):
    """Full auto-convolution via FFT."""
    return _fft_convolve(a, a)


def _score_of(a):
    """Compute 2*n*max_b / sum_a^2 and the convolution."""
    n = len(a)
    b = _auto_convolve(a)
    s = float(a.sum())
    if s < 1e-300:
        return float('inf'), b, 0.0
    max_b = float(b.max())
    return 2.0 * n * max_b / (s * s), b, max_b


def _normalize(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def _anti_pal_init(n, rng, temperature=1.0):
    """Anti-palindromic initialization: a_i + a_{n-1-i} = 1/m."""
    m = n // 2
    if m < 1:
        return np.ones(n) / max(1, n)
    
    # Random perturbation
    v = rng.normal(0, 1, size=m)
    v -= v.mean()
    std = v.std()
    if std > 1e-12:
        v *= temperature / std
    
    z = np.zeros(n)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m+1:] = -v[::-1]
        z[m] = 0.0
    
    # Softmax to make positive
    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _build_conv_matrix(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid, else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        if length <= 0:
            continue
        r = j + np.arange(length, dtype=np.int32)
        c = np.full(length, j, dtype=np.int32)
        v = a[:length].astype(np.float64)
        mask = v > 1e-15
        r = r[mask]
        c = c[mask]
        v = v[mask]
        if len(v) > 0:
            rows.append(r)
            cols.append(c)
            vals.append(v)
    
    if not rows:
        return csr_matrix((2 * n - 1, n))
    
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_iteration(a, inflate=1.0, shaped_delta=0.0):
    """One LP iteration: find g maximizing sum(g) subject to conv(a,g) <= M*constraints."""
    n = len(a)
    b = _auto_convolve(a)
    M = float(b.max())
    if M < 1e-300:
        return None
    
    # Build constraint matrix
    A = _build_conv_matrix(a)
    
    # RHS: M * inflate, optionally shaped
    if shaped_delta > 0:
        b_ub = M * inflate - shaped_delta * (b - M)  # more relaxed where b is smaller
        b_ub = np.maximum(b_ub, M * 0.01)  # ensure positive
    else:
        b_ub = np.full(2 * n - 1, M * inflate)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
    except Exception:
        return None
    
    if res.status != 0 or not hasattr(res, 'x'):
        return None
    
    g = np.array(res.x, dtype=np.float64)
    g = np.clip(g, 0, None)
    total = float(g.sum())
    if total < 1e-10:
        return None
    
    g /= total
    return g


def _gradient_step(a, b, focus=5.0, eta=0.3, top_k=None):
    """Exponential gradient step focusing on large convolution coefficients."""
    n = len(a)
    max_b = float(b.max())
    if max_b < 1e-300:
        return a.copy()
    
    # Weight function
    if top_k is not None and top_k < len(b):
        # Focus on top-k peaks
        p = np.zeros(len(b))
        idx = np.argpartition(b, -top_k)[-top_k:]
        p[idx] = 1.0 / top_k
    else:
        z = (max_b - b) / max_b
        with np.errstate(over='ignore', under='ignore'):
            p = np.exp(-z * focus)
        s = float(p.sum())
        if s < 1e-12:
            p = np.zeros(len(b))
            p[int(np.argmax(b))] = 1.0
        else:
            p /= s
    
    # Gradient via FFT: grad = 2 * (p * a) truncated to n
    L = len(p)
    N = 1 << (L + n - 2).bit_length()
    grad = np.fft.irfft(np.fft.rfft(p, N) * np.fft.rfft(a, N), N)[:n].real
    grad *= 2.0
    
    # Center the gradient (for projection onto sum=1)
    grad -= float(np.dot(a, grad))
    
    # Update
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-300)
    a_new /= float(a_new.sum())
    return a_new

    n = len(a)
    a = _normalize(a.copy())
    b = _auto_convolve(a)
    score, _, _ = _score_of(a)
    best_score = score
    best_a = a.copy()
    
    focus = float(focus_start)
    eta = 0.3
    no_improve = 0
    max_no_improve = 500
    resets = 0
    max_resets = 20
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        if b is None:
            b = _auto_convolve(a)
            score, _, _ = _score_of(a)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.5)
            # Perturb
            a = a * np.exp(rng.normal(0, 0.3, size=n))
            a = _normalize(a)
            b = None
            score = float('inf')
            focus = float(rng.choice([3.0, 5.0, 10.0, 20.0, 50.0]))
            eta = 0.3
            continue
        
        p_max_b = float(b.max())
        a_new = _gradient_step(a, b, focus=focus, eta=eta)
        b_new = _auto_convolve(a_new)
        score_new, _, _ = _score_of(a_new)
        
        if score_new < score * (1 + 1e-5):
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.03, 5.0)
        else:
            eta = max(eta * 0.7, 0.05)
            if eta < 0.1:
                a = a * np.exp(rng.normal(0, 0.1, size=n))
                a = _normalize(a)
                b = None
                score = float('inf')
        
        focus = min(focus * 1.01, focus_max)
    
    return best_a, best_score


def _population_search(initial_a, time_limit, rng, pop_size=5):
    """Genetic-algorithm-like population search."""
    n = len(initial_a)
    population = [initial_a.copy()]
    for _ in range(pop_size - 1):
        perturbation = rng.normal(0, 0.5, size=n)
        candidate = initial_a * np.exp(perturbation)
        population.append(_normalize(np.maximum(candidate, 1e-15)))
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
New approach: LP-based structured iteration combined with adaptive
exponential gradient refinement, population diversity, and multi-scale search.
The LP step finds the maximum-sum vector in the 'convolution ball' of the
current sequence, enabling large jumps to new basins of attraction.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_convolve(x, y):
    nx, ny = len(x), len(y)
    L = nx + ny - 1
    N = 1 << (L - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(x, N) * np.fft.rfft(y, N), N)[:L].real


def _auto_conv(a):
    return _fft_convolve(a, a)


def _score(a):
    n = len(a)
    b = _auto_conv(a)
    s = float(a.sum())
    if s < 1e-300:
        return float('inf'), b, 0.0
    mb = float(b.max())
    return 2.0 * n * mb / (s * s), b, mb


def _norm(a):
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def _anti_init(n, rng, temp=1.0):
    m = n // 2
    if m < 1:
        return np.ones(n, dtype=np.float64) / max(1, n)
    v = rng.normal(0, 1, m)
    v -= v.mean()
    std = float(v.std())
    if std > 1e-12:
        v *= temp / std
    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0
    sh = float(z.max())
    if sh > 0:
        z -= sh
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _build_A(a):
    n = len(a)
    rows_parts = []
    cols_parts = []
    vals_parts = []
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        if length <= 0:
            continue
        seg = a[:length]
        mask = seg > 1e-14
        if not np.any(mask):
            continue
        r = (j + np.arange(length))[mask].astype(np.int32)
        c = np.full(int(mask.sum()), j, dtype=np.int32)
        v = seg[mask]
        rows_parts.append(r)
        cols_parts.append(c)
        vals_parts.append(v)
    if not rows_parts:
        return csr_matrix((2 * n - 1, n))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    vals = np.concatenate(vals_parts)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_step(a, inflate=1.05, shaped=0.2, timeout=40):
    n = len(a)
    b = _auto_conv(a)
    M = float(b.max())
    if M < 1e-300:
        return None
    A = _build_A(a)
    if shaped > 0:
        b_ub = M * inflate - shaped * (b - M)
        b_ub = np.maximum(b_ub, M * 0.05)
    else:
        b_ub = np.full(2 * n - 1, M * inflate, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True,
                               'mips_rel_gap': 0.0})
    except Exception:
        return None
    if res.status != 0:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    tot = float(g.sum())
    if tot < 1e-10:
        return None
    g /= tot
    return g


def _grad_weights(b, focus, top_k=None):
    L = len(b)
    mb = float(b.max())
    if mb < 1e-300:
        return np.full(L, 1.0 / L)
    if top_k is not None and top_k < L:
        p = np.zeros(L)
        idx = np.argpartition(b, -top_k)[-top_k:]
        p[idx] = 1.0 / top_k
        return p
    z = (mb - b) / mb
    with np.errstate(over='ignore', under='ignore'):
        p = np.exp(-z * focus)
    s = float(p.sum())
    if s < 1e-12:
        p = np.zeros(L)
        p[int(np.argmax(b))] = 1.0
    else:
        p /= s
    return p


def _grad_direction(p, a):
    n = len(a)
    L = len(p)
    N = 1 << (L + n - 2).bit_length()
    g = np.fft.irfft(np.fft.rfft(p, N) * np.fft.rfft(a, N), N)[:n].real
    g *= 2.0
    g -= float(np.dot(a, g))
    return g


def _clip_eta_line(a, b, p, eta_lo, eta_hi, n_trials=4):
    score0, _, _ = _score(a)
    grad = _grad_direction(p, a)
    mg = float(np.max(np.abs(grad)))
    if mg < 1e-18:
        return a.copy(), score0
    grad /= mg
    best_eta = eta_lo
    best_a = a.copy()
    best_sc = score0
    for i in range(n_trials):
        eta = eta_lo * (eta_hi / eta_lo) ** (i / max(1, n_trials - 1))
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()
        sc, _, _ = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new
            best_eta = eta
    return best_a, best_sc

    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a = best_a / ss

    final_seq = [float(x) for x in best_a]

    # Validate with evaluate_sequence if available and time permits
    if time.monotonic() < deadline:
        exact = _safe_eval(final_seq)
        int_sc, _, _ = _score(best_a)
        print(f"\n=== FINAL: n={len(final_seq)} internal={int_sc:.6f} "
              f"exact={exact:.6f} time={time.monotonic()-t_start:.0f}s ===")
    else:
        print(f"\n=== FINAL: n={len(final_seq)} "
              f"time={time.monotonic()-t_start:.0f}s (no eval time) ===")

    return final_seq
```
