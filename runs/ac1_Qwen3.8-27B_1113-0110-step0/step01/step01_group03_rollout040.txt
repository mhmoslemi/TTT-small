Minimizes 2n*max(convolve(a,a))/sum(a)^2 using:
1. LP-based iteration (maximize sum(b) s.t. (f*b)[k] <= M, b >= 0)
2. Multiple starting points from parametric families
3. Gradient descent fine-tuning
4. Random restarts to escape local minima

def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]

def normalize_sum(a, target):
    """Normalize array to have sum=target, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return np.full_like(a, target / len(a))
    return a / S * target

def build_conv_matrix(f):
    """Build sparse matrix T such that T @ b = cross-convolution of f and b."""
    n = len(f)
    T = lil_matrix((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = f
    return T.tocsr()

def lp_step(f, M, T=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0. Returns b or None."""
    n = len(f)
    if T is None:
        T = build_conv_matrix(f)
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        return np.asarray(res.x).flatten()
    return None

    c = fft_self_conv(a)

        families.append(np.maximum(1.0 + a * (2 * t - 1)**2, 0.1))
        families.append(np.maximum(1.0 - a * (2 * t - 1)**2, 0.1))
    
    # Sine

        for amp in [0.3, 0.5, 0.7]:
            families.append(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.1))
    
    # Cosine
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            families.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.1))
    
    # Bump
    for width in [0.2, 0.3, 0.5]:
        for amp in [0.5, 1.0, 2.0]:
            center = 0.5
            families.append(np.maximum(1.0 + amp * np.exp(-((t - center)**2) / (2 * width**2)), 0.1))
    
    # Random (a few)
    for _ in range(5):
        families.append(np.maximum(rng.random(n), 0.1))
    
    return families

def gradient_step(a, lr, momentum, beta):
    """One gradient descent step."""
    n = len(a)
    c = fft_self_conv(a)
    M = c.max()
    k_star = int(np.argmax(c))
    
    # Compute gradient of score = 2n*M/S^2 w.r.t. a
    # d(score)/d(a_j) = 2n * a[k_star - j] / S^2 - 2n * M * 2 / S^3
    S = a.sum()
    j_min = max(0, k_star - (n - 1))
    j_max = min(n - 1, k_star)
    
    grad = np.zeros(n)
    # a[k_star - j] for j in [j_min, j_max]
    indices = k_star - np.arange(j_min, j_max + 1)
    grad[j_min:j_max + 1] = 2.0 * n * a[indices] / (S * S) - 4.0 * n * M / (S ** 3)
    
    momentum = beta * momentum - lr * grad
    a = np.maximum(a + momentum, 0.0)
    return a, momentum

    print(f"After LP phase: score={best_score:.8f}, iters={lp_count}")

    # === Phase 2: Gradient descent with momentum ===
    f = best_f.copy()
    momentum = np.zeros(n)
    lr = 0.03
    beta = 0.8
    gd_iter = 0
    no_improve_gd = 0
    phase2_deadline = deadline - 100

    while time.monotonic() < phase2_deadline:
        f = normalize_sum(f, n)
        grad = compute_gradient(f)
        momentum = beta * momentum - lr * grad
        f = np.maximum(f + momentum, 0.0)
        f = normalize_sum(f, n)
        gd_iter += 1

        if gd_iter % 25 == 0:
            score = float(evaluate_sequence(f.tolist()))
            if np.isfinite(score) and score < best_score:
                best_f = f.copy()
                best_score = score
                no_improve_gd = 0
                if gd_iter % 500 == 0:
                    print(f"  GD[{gd_iter}]: score={score:.8f}")
            else:
                no_improve_gd += 1

            if no_improve_gd >= 15:
                lr *= 0.6
                momentum *= 0.3
                no_improve_gd = 0
                if lr < 1e-9:
                    f = best_f * np.exp(rng.normal(0, 0.02, n))
                    f = normalize_sum(f, n)
                    lr = 0.005
                    momentum = np.zeros(n)

    print(f"After GD phase: score={best_score:.8f}")

    # === Phase 3: Ultra-fine tuning ===
    f = best_f.copy()
    momentum = np.zeros(n)
    lr = 0.0005
    beta = 0.4
    fine_iter = 0

    while time.monotonic() < deadline - 10:
        f = normalize_sum(f, n)
        grad = compute_gradient(f)
        momentum = beta * momentum - lr * grad
        f = np.maximum(f + momentum, 0.0)
        f = normalize_sum(f, n)
        fine_iter += 1

        if fine_iter % 15 == 0:
            score = float(evaluate_sequence(f.tolist()))
            if np.isfinite(score) and score < best_score:
                best_f = f.copy()
                best_score = score

    # Final validation
    best_f = normalize_sum(best_f, n)
    final_score = float(evaluate_sequence(best_f.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"Final score: {best_score:.8f}")
    return [float(x) for x in best_f]
