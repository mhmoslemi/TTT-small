```python
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    c = irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _smooth_max(c, eps):
    """Log-sum-exp approximation of max."""
    c_max = np.max(c)
    if eps == 0:
        return c_max
    shifted = eps * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted))) / eps + c_max
    return log_sum

def _smooth_max_gradient(a, eps):
    """Gradient of smoothed max convolution."""
    n = len(a)
    c = _fft_convolve(a)
    # Compute weights
    c_max = np.max(c)
    shifted = eps * (c - c_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    # Gradient: sum_k w_k * 2*a_{k-i}
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-10:
            continue
        # c_k = sum_i a_i * a_{k-i}
        # d(c_k)/d(a_i) = 2*a_{k-i} (if k-i in [0,n-1])
        k_int = int(k)
        # For each i, the contribution is 2*a_{k-i} if 0 <= k-i < n
        i_min = max(0, k_int - (n - 1))
        i_max = min(n - 1, k_int)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            a_vals = a[k_int - indices]
            grad[i_min:i_max+1] += w[k] * 2.0 * a_vals
    return grad, c

        return np.ones(n) / n

    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.max(np.where(v_sorted > (cumsum - 1) / np.arange(1, n+1))[0])
    tau = (cumsum[rho] - 1) / (rho + 1)
    result = np.maximum(v - tau, 0)
    # Ensure sum is exactly 1
    s = np.sum(result)
    if s < 1e-15:
        return np.ones(n) / n
    return result / s

def _optimize_smoothed(a_init, eps_start, eps_end, max_iters, deadline, lr_init=0.05, lr_decay=0.999):
    """Optimize using smoothed max with projected gradient."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    eps = eps_start
    lr = lr_init
    
    eps_decay = (eps_end / eps_start) ** (1.0 / max_iters) if max_iters > 0 else 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, c = _smooth_max_gradient(a, eps)
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project onto simplex
        a_new = _project_onto_simplex(a_new)

        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        eps *= eps_decay
        lr *= lr_decay
    
    return best_a, best_score

def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.8):
    a = np.maximum(a_init, 0).astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:

        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if it % 50 == 0:
                print(f"      LP iter {it}: score={best_score:.6f}")
        a = a_new
        blend = max(0.1, blend * 0.98)

    return best_a, best_score

def _interpolate_sequence(src, n_target):
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

def _make_initial(n, variant, rng):
```

- Or use FFT: grad = 2 * IFFT(FFT(a) * FFT(w_flipped_shifted))

```python
# w has length 2n-1, a has length n
# We want: grad[i] = 2 * dot(a, w[i:i+n]) for i = 0, ..., n-1
# This is a "sliding dot product"
# Can be computed as:
# Let W_rev[j] = w[n-1-j] (reverse of first n elements of w)
# Then grad[i] = 2 * conv(a, W_rev)[n-1-i]... 
```

```python
# Efficient computation using matrix multiply

W_matrix[idx[:, None], idx[None, :]] = w[np.arange(n)[:, None] + np.arange(n)[None, :]]
```

```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

"""
Multi-phase optimizer minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
Strategies: (1) Alternating LP, (2) Smoothed gradient with momentum,
(3) Active-set LP polishing, (4) Multi-n exploration.
"""


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    c = irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build matrix C where (C @ g)[k] = (f*g)[k] for full convolution."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M, time_limit=20):
    """Solve max sum(g) s.t. conv(f,g)_k <= M, g >= 0."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _smooth_max_and_grad(a, eps):
    """Compute smoothed max of conv(a,a) and its gradient.
    Returns (smoothed_value, gradient_vector).
    """
    n = len(a)
    c = _fft_convolve(a)
    c_max = np.max(c)
    shifted = eps * (c - c_max)
    # Clamp for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    if total < 1e-300:
        w = np.zeros_like(c)
        w[c_max == c] = 1.0 / np.sum(c_max == c)
    else:
        w = exp_shifted / total
    log_sum = np.log(total) / eps + c_max

    # Gradient: grad[i] = 2 * sum_j a[j] * w[i+j]
    indices = np.arange(n)[:, None] + np.arange(n)[None, :]
    W = w[indices]
    grad = 2.0 * (W @ a)

    return log_sum, grad, c, w


def _project_simplex(v):
    """Project v onto {a: a >= 0, sum(a) = 1}."""
    n = len(v)
    v = v.copy()
    if np.sum(v) <= 1e-15:
        return np.ones(n) / n
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted > (css - 1.0) / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (css[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    s = np.sum(result)
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def _smoothed_gradient_descent(a_init, eps_start, eps_end, max_iters, deadline,
                                lr_init=0.3, momentum=0.85):
    """Optimize using smoothed max with momentum gradient descent."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    eps = eps_start
    lr = lr_init
    velocity = np.zeros(n)

    if max_iters <= 1:
        eps_ratio = 1.0
    else:
        eps_ratio = (eps_end / eps_start) ** (1.0 / max_iters)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        val, grad, c, w = _smooth_max_and_grad(a, eps)

        # Momentum update
        velocity = momentum * velocity - lr * grad
        a_new = a + velocity
        a_new = _project_simplex(a_new)

        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if it % 200 == 0:
                print(f"      SGD iter {it}: score={best_score:.6f}, eps={eps:.1f}")

        a = a_new
        eps *= eps_ratio
        lr *= 0.9995

    return best_a, best_score


def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.7):
    """Run alternating LP with adaptive blending."""
    a = np.maximum(a_init, 0.0).astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _lp_solve(a, M, time_limit=15)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if it % 100 == 0:
                print(f"      LP iter {it}: score={best_score:.6f}")
        a = a_new
        blend = max(0.15, blend * 0.985)

    return best_a, best_score


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

def _active_set_refine(a_init, deadline, n_top=10):
    """Refine by iteratively tightening the active set of peak constraints."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

    for it in range(200):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        # Find top peaks
        top_k = np.argpartition(c, -n_top)[-n_top:]
        # Solve LP with only top constraints
        C = _build_conv_matrix(a)
        C_sub = C[top_k, :]
        c_obj = -np.ones(n)
        b_ub = M * np.ones(n_top)
        bounds = [(0.0, None)] * n
        try:
            result = linprog(c_obj, A_ub=C_sub, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'presolve': True, 'time_limit': 10})
            if result.status != 0:
                break
            g = result.x
        except Exception:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        blend = 0.5
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            break
        a_new = a_new / s_new
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _scipy_minimize(a_init, deadline):
    """Use scipy L-BFGS-B with log-sum-exp objective."""
    n = len(a_init)

    def objective_and_grad(b):
        a = np.maximum(b, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            return 1e6, np.zeros(n)
        a = a / s
        eps = 100.0
        val, grad, c, w = _smooth_max_and_grad(a, eps)
        # Scale back to original objective
        obj = 2.0 * n * val
        # Gradient w.r.t. b (before normalization)
        # a_i = b_i / s, da_i/db_j = (delta_ij * s - b_i) / s^2 = (delta_ij - a_i) * a_i / a_i... 
        # Simpler: work in normalized space
        # grad_b = grad * s - graddot * a (where graddot = dot(grad, a))
        graddot = np.dot(grad, a)
        grad_b = s * grad - graddot * a
        return obj, grad_b

    best_a = np.maximum(a_init, 0.0).astype(np.float64)
    s = np.sum(best_a)
    if s < 1e-10:
        return best_a, _score(best_a)
    best_a = best_a / s
    best_score = _score(best_a)

    b0 = best_a.copy()
    bounds = [(0.0, None)] * n

    it_counter = [0]

    def callback(bk):
        it_counter[0] += 1
        a_k = np.maximum(bk, 0.0)
        s_k = np.sum(a_k)
        if s_k < 1e-10:
            return False
        a_k = a_k / s_k
        sc = _score(a_k)
        if sc < best_score:
            nonlocal best_a, best_score
            best_score = sc
            best_a = a_k.copy()
        if time.monotonic() > deadline:
            return True
        return False

    try:
        result = minimize(objective_and_grad, b0, method='L-BFGS-B',
                          bounds=bounds, callback=callback,
                          options={'maxiter': 3000, 'maxfun': 5000,
                                   'ftol': 1e-12, 'gtol': 1e-8})
    except Exception:
        pass

    return best_a, best_score
```
