"""
Search for non-negative step-function heights minimizing
    2 * n * max(convolve(a, a)) / (sum(a)**2).

Method used here:
  1. Work mainly on the anti-palindromic ansatz
         a_i + a_{n-1-i} = 1/(n/2),  n even.
     This keeps the center convolution coefficient low and halves the
     effective dimension.
  2. Refine with smoothed L-BFGS on the anti-palindromic variables, using
     tau * log(sum(exp(b/tau))) as a smooth surrogate for max(b).
  3. Use a full-vector L-BFGS refinement to escape the anti-palindromic
     subspace if it helps.
  4. Use a mirror-descent / exponentiated-gradient optimizer for robust
     long runs and restarts.
  5. Use a coordinate-refinement pass on the most active anti-palindromic
     coordinates.
  6. Occasionally call a sparse active-set LP "best-response" direction
     max {sum(g) : convolve(f, g) <= M, g >= 0, g <= ub}, and line-search
     along the resulting direction.  This is the LP direction from the
     previous state-of-the-art description.
"""

import numpy as np
import time

try:
    from scipy import optimize as _sp_opt
    from scipy import sparse as _sp_sparse
    _HAS_SCIPY = True
except Exception:

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _corr_weights(w, a):
    """
    Return c_i = sum_k w_k a_{k-i}, zero-padded outside [0, n-1].
    This is the derivative shape for b = convolve(a,a) with weights w.
    """
    w = np.asarray(w, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    L = int(w.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(w, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)


def _smooth_logsum(b, tau):
    b = np.asarray(b, dtype=np.float64).ravel()
    L = int(b.size)
    maxb = float(b.max())
    if L == 0 or maxb <= 1e-300:
        return 0.0, np.full(L, 1.0 / max(1, L), dtype=np.float64)

    if tau <= 0.0 or not np.isfinite(tau):
        p = np.zeros(L, dtype=np.float64)
        idx = np.flatnonzero(b == maxb)
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))], dtype=np.int64)
        p[idx] = 1.0 / idx.size
        return maxb, p

    z = (b - maxb) / tau
    z = np.clip(z, -745.0, 709.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
        return maxb, p
    h = maxb + tau * float(np.log(s))
    p = e / s
    return float(h), p


def _internal_score(a, b=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return float("inf"), np.array([0.0], dtype=np.float64), 0.0
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    return float(2.0 * n * maxb / (s * s)), b, maxb


def _ensure_valid(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 0.01 or a.size < 8:
        nmax = max(1024, int(a.size) if a.size > 0 else 1024)
        a = np.ones(nmax, dtype=np.float64) / float(nmax)
        s = float(a.sum())
    a = a / s
    return a


def _a_from_x(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    m = int(x.size)
    if m <= 0:
        return np.ones(2, dtype=np.float64) / 2.0
    t = 1.0 / float(m)
    n = 2 * m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return np.maximum(a, 0.0)


def _project_anti(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n % 2 != 0 or n < 4:
        return None
    m = n // 2
    t = 1.0 / float(m)
    left = a[:m]
    right = a[m:][::-1]
    x = (left - right + t) / (2.0 * t)
    return np.clip(x, 0.0, 1.0)


def _anti_x_random(n, rng, amp=2.0, smooth=True):
    n = int(n)
    m = n // 2
    if m <= 0:
        return np.array([0.5], dtype=np.float64)
    idx = np.arange(m, dtype=np.float64)
    v = np.zeros(m, dtype=np.float64)
    if smooth:
        for f in (1, 2, 3, 5, 7, 11, 17):
            ph = rng.uniform(0.0, 2.0 * np.pi)
            v += (rng.normal(0.0, 1.0) / float(f)) * np.sin(2.0 * np.pi * f * idx / float(m) + ph)
        v += 0.15 * rng.normal(0.0, 1.0, size=m)
    else:
        v = rng.normal(0.0, 1.0, size=m)
    std = float(v.std())
    if std > 1e-12:
        v = (float(amp) * (v - float(v.mean()))) / std
    x = 0.5 + 0.45 * np.tanh(v)
    return np.clip(x, 1e-9, 1.0 - 1e-9)


def _anti_x_binary(n, rng):
    m = n // 2
    if m <= 0:
        return np.array([0.5], dtype=np.float64)
    x = (rng.random(m) < 0.5).astype(np.float64)
    return np.clip(x, 1e-9, 1.0 - 1e-9)


def _make_focus_weights(b, focus, eps=0.01):
    b = np.asarray(b, dtype=np.float64).ravel()
    maxb = float(b.max())
    L = int(b.size)
    if maxb <= 1e-300:
        return np.full(L, 1.0 / max(1, L), dtype=np.float64), maxb
    tau = maxb / max(1e-12, float(focus))
    _, p = _smooth_logsum(b, tau)
    p = (1.0 - eps) * p + (eps / max(1, L))
    return p, maxb


def _mirror_optimize_anti(x0, time_limit, rng):
    x = np.clip(np.asarray(x0, dtype=np.float64).ravel(), 1e-10, 1.0 - 1e-10)
    m = int(x.size)
    if m <= 0:
        x = np.array([0.5], dtype=np.float64)
        m = 1
    n = 2 * m
    t = 1.0 / float(m)

    a = _a_from_x(x)
    best_a = a.copy()
    best_x = x.copy()
    best_score = float("inf")

    b = None
    score = float("inf")
    focus = 20.0
    eta = 0.30
    eta_min = 0.01
    eta_max = 2.0
    no_improve = 0
    max_no = 700
    resets = 0
    max_resets = 25
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.3)
            base = best_x
            fresh = _anti_x_random(n, rng, amp=float(rng.choice([1.5, 2.0, 3.0, 4.0])),
                                   smooth=bool(rng.random() < 0.7))
            x = np.clip(0.82 * base + 0.18 * fresh, 1e-10, 1.0 - 1e-10)
            a = _a_from_x(x)
            b = None
            score = float("inf")
            focus = float(rng.choice([5.0, 10.0, 20.0, 50.0, 100.0, 250.0]))
            eta = 0.3
            continue

        _, maxb = _smooth_logsum(b, 1.0)
        p, _ = _make_focus_weights(b, focus)
        c = _corr_weights(p, a)
        dda = 2.0 * c
        grad = t * (dda[:m] - dda[m:][::-1])
        gmax = float(np.max(np.abs(grad)))
        if gmax <= 1e-300:
            break
        grad = grad / gmax

        x_new = np.clip(x - eta * grad, 1e-10, 1.0 - 1e-10)
        a_new = _a_from_x(x_new)
        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        wt_old = float(np.dot(p, b))
        wt_new = float(np.dot(p, b_new))

        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        elif score_new < score * 1.001 and wt_new < wt_old - 1e-14:
            accept = True

        if accept:
            x = x_new
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.05, eta_max)
        else:
            eta = max(eta * 0.78, eta_min)
            if eta < 0.06:
                x = np.clip(x + rng.normal(0.0, 0.05, size=m), 1e-10, 1.0 - 1e-10)
                a = _a_from_x(x)
                b = None
                score = float("inf")

        focus = min(focus * 1.008, 3000.0)
        it += 1

    return best_a, float(best_score), best_x

def _lbfgs_eval_anti(x, st):
    m = int(st["m"])
    n = int(st["n"])
    tau = float(st["tau"])
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.clip(x, 0.0, 1.0)
    a = _a_from_x(x)
    b = _fft_square(a)
    h, p = _smooth_logsum(b, tau)
    c = _corr_weights(p, a)
    dda = 2.0 * c
    t = 1.0 / float(m)
    grad = t * (dda[:m] - dda[m:][::-1])
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    if not np.isfinite(h):
        return 1e100, np.zeros(m, dtype=np.float64)
    return float(h), grad


def _lbfgs_eval_full(y, st):
    n = int(st["n"])
    tau = float(st["tau"])
    y = np.asarray(y, dtype=np.float64).ravel()
    y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
    z = y - float(np.mean(y))
    z = z - float(np.max(z))
    a = np.exp(z)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a / s

    b = _fft_square(a)
    h, p = _smooth_logsum(b, tau)
    c = _corr_weights(p, a)
    dda = 2.0 * c
    grad = a * (dda - float(np.dot(a, dda)))
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    if not np.isfinite(h):
        return 1e100, np.zeros(n, dtype=np.float64)
    return float(h), grad


def _lbfgs_eval(y, st):
    mode = st.get("mode", "anti")
    if mode == "anti":
        return _lbfgs_eval_anti(y, st)
    return _lbfgs_eval_full(y, st)


def _lbfgs_compute(y, st):
    y_arr = np.nan_to_num(np.asarray(y, dtype=np.float64), nan=0.0, posinf=0.0, neginf=0.0)
    ly = st.get("last_y")
    if ly is not None and ly.shape == y_arr.shape:
        if np.max(np.abs(y_arr - ly)) == 0.0:
            return st["last_val"], st["last_grad"]
    val, grad = _lbfgs_eval(y_arr, st)
    st["last_y"] = y_arr.copy()
    st["last_val"] = float(val)
    st["last_grad"] = np.asarray(grad, dtype=np.float64)
    return float(val), st["last_grad"]


def _lbfgs_fun(y, st):
    val, _ = _lbfgs_compute(y, st)
    return float(val)


def _lbfgs_jac(y, st):
    _, grad = _lbfgs_compute(y, st)
    return np.asarray(grad, dtype=np.float64)


def _lbfgs_anti(x0, time_limit, rng, verbose=False):
    x = np.clip(np.asarray(x0, dtype=np.float64).ravel(), 0.0, 1.0)
    m = int(x.size)

        m = 1
    n = 2 * m

    best_x = x.copy()

    if not _HAS_SCIPY:
        return _mirror_optimize_anti(x, time_limit, rng)

    st = {"mode": "anti", "n": n, "m": m}
    bounds = [(0.0, 1.0)] * m

    if m <= 1024:
        maxiter = 260
    elif m <= 4096:
        maxiter = 170
    elif m <= 8192:
        maxiter = 110
    else:
        maxiter = 60

    phases = [0.30, 0.16, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002]

            break
        _, maxb_phase = _internal_score(best_a)

        x_next = best_x.copy()
        try:
            res = _sp_opt.minimize(
                _lbfgs_fun,
                x_next,
                args=(st,),
                jac=_lbfgs_jac,

                options={
                    "maxiter": maxiter,
                    "maxcor": 20,
                    "maxls": 10,
                    "gtol": 1e-12,
                    "ftol": 1e-12,
                },
            )
            x_new = getattr(res, "x", None)
            if x_new is not None:
                x_new = np.asarray(x_new, dtype=np.float64)
                if x_new.size == m and np.all(np.isfinite(x_new)):

        except Exception:

        a_next = _a_from_x(x_next)

            best_a = a_next
            if verbose:

        else:
            # keep the best point as the next L-BFGS start
            x_next = best_x.copy()

    return best_a, float(best_score), best_x

            y = y_next.copy()
            if verbose:
                print("full lbfgs improve phase={:.4f} score={:.8f}".format(phase, best_score))
        else:
            y = np.log(np.maximum(best_a, 1e-300))
            y = y - float(np.mean(y))

    return best_a, float(best_score), y


def _build_active_matrix(a, rows):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    K = int(rows.size)
    indptr = np.empty(K + 1, dtype=np.int64)
    data_parts = []
    cols_parts = []
    ptr = 0

    for r in range(K):
        k = int(rows[r])
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        length = je - js + 1
        if length <= 0:
            indptr[r + 1] = ptr
            continue

        i_start = k - js
        i_end = k - je
        vals = a[i_end:i_start + 1][::-1].astype(np.float64, copy=True)
        cols = np.arange(js, je + 1, dtype=np.int32)

        data_parts.append(vals)
        cols_parts.append(cols)
        ptr += length
        indptr[r + 1] = ptr

    if data_parts:
        data = np.concatenate(data_parts)
        cols = np.concatenate(cols_parts)
    else:
        data = np.empty(0, dtype=np.float64)
        cols = np.empty(0, dtype=np.int32)

    A = _sp_sparse.csr_matrix((data, cols, indptr), shape=(K, n))
    A = A.astype(np.float64)
    return A


def _try_lp_update(f):
    if not _HAS_SCIPY:
        return None

    f = _ensure_valid(f)
    n = int(f.size)
    if n < 16 or n > 30000:
        return None

    b, score, maxb = _internal_score(f)
    L = 2 * n - 1

    if n <= 4096:
        max_nnz = 14_000_000
        max_k = 1400
    elif n <= 8192:
        max_nnz = 10_000_000
        max_k = 1100
    else:
        max_nnz = 7_000_000
        max_k = 700

    K_probe = max(64, int(max_nnz // max(1, n)))
    K = min(L, max_k, max(96, K_probe))

    top = np.argsort(b, kind="mergesort")[-K:]
    nbound = min(64, max(16, K // 4))
    bound = np.linspace(0, L - 1, nbound).astype(np.int64)
    rows_arr = np.unique(np.concatenate((top, bound)))

    keep = min(rows_arr.size, K + nbound)
    if rows_arr.size > keep:
        validx = np.argsort(b[rows_arr])[-keep:]
        rows_arr = rows_arr[validx]

    A = _build_active_matrix(f, rows_arr)
    nnz = int(A.nnz)
    if nnz <= 0:
        return None

    c = -np.ones(n, dtype=np.float64)
    b_ub = maxb * np.ones(rows_arr.size, dtype=np.float64)
    ub0 = max(20.0 / float(n), 20.0 * float(f.max()))
    ub0 = min(max(ub0, 1.0 / float(n)), 10.0)

    res = None
    try:
        res = _sp_opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=(0.0, ub0),
            method="highs",
            options={"disp": False},
        )
    except Exception:
        return None

    g = None
    if res is not None and getattr(res, "success", False):
        g = getattr(res, "x", None)

    if g is None and res is not None and getattr(res, "status", None) in (2, 3):
        like getattr(res, "x", None) is not None:
            pass

    if g is None:
        try:
            ub1 = min(ub0 * 3.0, 100.0 / float(n), 10.0)
            res = _sp_opt.linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=(0.0, ub1),
                method="highs",
                options={"disp": False},
            )
            if res is not None and getattr(res, "success", False):
                g = getattr(res, "x", None)
        except Exception:
            g = None

    if g is None:
        return None

    g = np.nan_to_num(np.asarray(g, dtype=np.float64), nan=0.0, posinf=0.0, neginf=0.0)
    g = np.maximum(g, 0.0)
    sg = float(g.sum())
    if sg <= 1e-12:
        return None

    g0 = g / sg

    best_h = f.copy()
    best_sc = float(score)

    ts = [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.35, 0.55, 0.8, 1.0]
    for tv in ts:
        h = (1.0 - tv) * f + tv * g0
        hs = float(h.sum())
        if hs <= 1e-300:
            continue
        h = h / hs
        sc, _, _ = _internal_score(h)
        if sc < best_sc:
            best_sc = float(sc)
            best_h = h.copy()

    px = _project_anti(g0)
    if px is not None:
        ga = _a_from_x(np.clip(px, 0.0, 1.0))
        for tv in ts:
            h = (1.0 - tv) * f + tv * ga
            hs = float(h.sum())
            if hs <= 1e-300:
                continue
            h = h / hs
            sc, _, _ = _internal_score(h)
            if sc < best_sc:
                best_sc = float(sc)
                best_h = h.copy()

    if best_sc < score - 1e-12:
        return best_h, best_sc
    return None


def _coordinate_refine_x(x0, time_limit, rng):
    x = np.clip(np.asarray(x0, dtype=np.float64).ravel(), 1e-10, 1.0 - 1e-10)
    m = int(x.size)
    if m <= 0:
        return x, float("inf"), _a_from_x(x)

    a = _a_from_x(x)
    b, score, maxb = _internal_score(a)
    best_a = a.copy()
    best_x = x.copy()
    best_score = float(score)

    t0 = time.monotonic()
    no_improve = 0
    base_k = min(70, max(12, m // 120))
    delta_cycle = 0

    while time.monotonic() - t0 < float(time_limit) and no_improve < 35:
        p, _ = _make_focus_weights(b, 25.0, eps=0.02)
        dda = 2.0 * _corr_weights(p, a)
        t = 1.0 / float(m)
        grad = t * (dda[:m] - dda[m:][::-1])
        gag = np.abs(grad)
        if float(np.max(gag)) <= 1e-300:
            break

        k = base_k
        idx = np.argsort(gag)[-k:]
        idx = idx.copy()
        rng.shuffle(idx)

        improved = False
        current_score = float(score)
        current_b = b

        for j in idx:
            j = int(j)
            if time.monotonic() - t0 >= float(time_limit):
                break
            cur = float(x[j])

            if delta_cycle % 3 == 0:
                deltas = [0.04, 0.12, 0.30]
            elif delta_cycle % 3 == 1:
                deltas = [0.015, 0.06, 0.18]
            else:
                deltas = [0.09, 0.25, 0.60]

            candidates = [cur, 1.0 - cur]
            for d in deltas:
                candidates.append(cur - d)
                candidates.append(cur + d)

            best_val = current_score
            best_b = current_b
            best_c = cur

            for val in candidates:
                val = float(np.clip(val, 1e-10, 1.0 - 1e-10))
                if val == best_c:
                    continue
                x2 = x.copy()
                x2[j] = val
                a2 = _a_from_x(x2)
                b2 = _fft_square(a2)
                s2, _, _ = _internal_score(a2, b2)
                if s2 < best_val - 1e-15:
                    best_val = float(s2)
                    best_b = b2
                    best_c = val

            if best_c != cur:
                x[j] = best_c
                a = _a_from_x(x)
                b = best_b
                current_score = best_val
                improved = True

        if improved:
            _a, _b, sc, _maxb = _internal_score(a, b)
            current_score = float(sc)
            if current_score < best_score - 1e-12:
                best_score = current_score
                best_a = a.copy()
                best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
            mask = gag > max(1e-300, float(np.percentile(gag, 90)))
            x = np.clip(x + rng.normal(0.0, 0.045, size=m) * mask, 1e-10, 1.0 - 1e-10)
            a = _a_from_x(x)
            b = _fft_square(a)
            score, _, maxb = _internal_score(a, b)
            current_score = float(score)
            if no_improve >= 18:
                x = np.clip(0.7 * best_x + 0.3 * _anti_x_random(2 * m, rng, amp=2.5, smooth=True),
                            1e-10, 1.0 - 1e-10)
                a = _a_from_x(x)
                b = _fft_square(a)
                score, _, maxb = _internal_score(a, b)
                current_score = float(score)

        delta_cycle += 1

    return best_x, float(best_score), best_a


def _safe_evaluate(seq):
    fn = globals().get("evaluate_sequence")
    if fn is None or not callable(fn):
        return float("inf")
    try:
        return float(fn(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=2400):
    seed = int(seed)
    total = max(120.0, float(budget_s))
    now0 = time.monotonic()
    final_deadline = now0 + total - 130.0

    rng = np.random.default_rng(seed)

    best_a = None
    best_score = float("inf")
    best_x = None

    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            if g.size >= 16 and float(g.sum()) > 0.01:
                g = g / float(g.sum())
                sc, _, _ = _internal_score(g)
                if np.isfinite(sc):
                    best_score = float(sc)
                    best_a = g.copy()
                    best_x = _project_anti(best_a)
                    print("given internal score={:.8f} n={}".format(best_score, best_a.size))
    except Exception:
        pass

    if best_a is None:
        n_init = 8192
        x_init = _anti_x_random(n_init, rng, amp=2.5, smooth=True)
        best_a = _a_from_x(x_init)
        best_score, _, _ = _internal_score(best_a)
        best_x = x_init.copy()
        print("initialized random anti n={}".format(best_a.size))

    best_a = _ensure_valid(best_a)
    print("start best_score={:.8f} n={}".format(best_score, best_a.size))

    def _remaining():
        return float(final_deadline - time.monotonic())

    # Phase 1: anti-palindromic L-BFGS refinement of the current best.
    if best_x is not None and _remaining() > 500.0:
        alloc = min(900.0, _remaining() - 500.0)
        if alloc > 80.0:
            try:
                a1, s1, x1 = _lbfgs_anti(best_x, alloc, rng, verbose=False)
                if np.isfinite(s1) and s1 < best_score + 1e-12:
                    if s1 < best_score:
                        print("anti lbfgs improved to {:.8f}".format(s1))
                    best_a = _ensure_valid(a1)
                    best_score = float(s1)
                    best_x = np.clip(x1, 0.0, 1.0)
                    best_score, _, _ = _internal_score(best_a)
            except Exception as exc:
                print("anti lbfgs failed:", repr(exc))

    # Phase 2: coordinate polish.
    if best_x is not None and _remaining() > 420.0:
        alloc = min(160.0, _remaining() - 420.0)
        if alloc > 30.0:
            try:
                xc, sc, ac = _coordinate_refine_x(best_x, alloc, rng)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    print("coordinate improved to {:.8f}".format(sc))
                    best_a = _ensure_valid(ac)
                    best_score = float(sc)
                    best_x = np.clip(xc, 0.0, 1.0)
            except Exception as exc:
                print("coordinate failed:", repr(exc))

    # Phase 3: full-vector L-BFGS if it may still help.
    if _remaining() > 520.0:
        alloc = min(350.0, _remaining() - 520.0)
        if alloc > 60.0:
            try:
                a2, s2, _y2 = _lbfgs_full(best_a, alloc, rng, verbose=False)
                if np.isfinite(s2) and s2 < best_score - 1e-12:
                    print("full lbfgs improved to {:.8f}".format(s2))
                    best_a = _ensure_valid(a2)
                    best_score = float(s2)
                    best_x = _project_anti(best_a)
            except Exception as exc:
                print("full lbfgs failed:", repr(exc))

    # If anti-palindromic projection was bad, run a full mirror optimizer.
    if best_x is None and _remaining() > 450.0:
        alloc = min(260.0, _remaining() - 450.0)
        if alloc > 50.0:
            try:
                a3, s3 = _mirror_optimize_full(best_a, alloc, rng)
                if np.isfinite(s3) and s3 < best_score - 1e-12:
                    print("full mirror improved to {:.8f}".format(s3))
                    best_a = _ensure_valid(a3)
                    best_score = float(s3)
                    best_x = _project_anti(best_a)
            except Exception as exc:
                print("full mirror failed:", repr(exc))

    # Phase 4: LP best-response direction, then recovery refinement.
    lp_tries = 0
    while _HAS_SCIPY and best_a.size <= 30000 and _remaining() > 360.0 and lp_tries < 2:
        lp_tries += 1
        try:
            lp = _try_lp_update(best_a)
        except Exception as exc:
            print("lp failed:", repr(exc))
            lp = None

        if lp is None:
            break
        la, ls = lp
        if np.isfinite(ls) and ls < best_score - 1e-12:
            print("lp direction improved to {:.8f}".format(ls))
            best_a = _ensure_valid(la)
            best_score = float(ls)
            best_x = _project_anti(best_a)

            if best_x is not None and _remaining() > 360.0:
                alloc = min(180.0, _remaining() - 360.0)
                if alloc > 40.0:
                    try:
                        ar, sr, xr = _lbfgs_anti(best_x, alloc, rng, verbose=False)
                        if np.isfinite(sr) and sr < best_score - 1e-12:
                            print("post-lbfgs after lp improved to {:.8f}".format(sr))
                            best_a = _ensure_valid(ar)
                            best_score = float(sr)
                            best_x = np.clip(xr, 0.0, 1.0)
                    except Exception:
                        pass
        else:
            break

    # Phase 5: multi-start diversity.
    multi_allowed = _remaining() > 900.0
    if multi_allowed:
        cur_n = int(best_a.size)

        starts = []

        if best_x is not None:
            starts.append(("noise1", np.clip(best_x + rng.normal(0.0, 0.12, size=best_x.size), 1e-9, 1.0 - 1e-9)))
            starts.append(("noise2", np.clip(best_x + rng.normal(0.0, 0.28, size=best_x.size), 1e-9, 1.0 - 1e-9)))

        starts.append(("rand_smooth", _anti_x_random(cur_n, rng, amp=3.0, smooth=True)))
        starts.append(("rand_jitter", _anti_x_random(cur_n, rng, amp=1.6, smooth=False)))
        starts.append(("binary", _anti_x_binary(cur_n, rng)))

        if cur_n > 4096 and _remaining() > 1300.0:
            starts.append(("n4096", _anti_x_random(4096, rng, amp=3.0, smooth=True)))
        if cur_n < 16384 and _remaining() > 1500.0:
            starts.append(("n16384", _anti_x_random(16384, rng, amp=3.0, smooth=True)))

        for label, xs in starts:
            if _remaining() < 650.0:
                break
            if xs is None:
                continue
            xs = np.asarray(xs, dtype=np.float64)
            if xs.size < 4:
                continue

            alloc = min(150.0, _remaining() - 700.0)
            if alloc < 45.0:
                break

            try:
                am, sm, xm = _lbfgs_anti(xs, alloc, rng, verbose=False)
                if np.isfinite(sm) and sm < best_score - 1e-12:
                    print("multi start {} improved to {:.8f}".format(label, sm))
                    best_a = _ensure_valid(am)
                    best_score = float(sm)
                    best_x = np.clip(xm, 0.0, 1.0)

                    if _remaining() > 650.0:
                        ca, cs, cx = _coordinate_refine_x(best_x, min(70.0, _remaining() - 650.0), rng)
                        if np.isfinite(cs) and cs < best_score - 1e-12:
                            print("multi start {} coordinate improved to {:.8f}".format(label, cs))
                            best_a = _ensure_valid(ca)
                            best_score = float(cs)
                            best_x = np.clip(cx, 0.0, 1.0)
            except Exception as exc:
                print("multi start {} failed:".format(label), repr(exc))

    # Phase 6: final long anti-palindromic refinement.
    if best_x is not None and _remaining() > 260.0:
        alloc = max(80.0, min(1100.0, _remaining() - 180.0))
        try:
            af, sf, xf = _lbfgs_anti(best_x, alloc, rng, verbose=True)
            if np.isfinite(sf) and sf < best_score + 1e-12:
                if sf < best_score:
                    print("final anti lbfgs improved to {:.8f}".format(sf))
                best_a = _ensure_valid(af)
                best_score = float(sf)
                best_x = np.clip(xf, 0.0, 1.0)
                best_score, _, _ = _internal_score(best_a)
        except Exception as exc:
            print("final anti lbfgs failed:", repr(exc))

        if _remaining() > 180.0:
            ca, cs, cx = _coordinate_refine_x(best_x, min(120.0, _remaining() - 150.0), rng)
            if np.isfinite(cs) and cs < best_score - 1e-12:
                print("final coordinate improved to {:.8f}".format(cs))
                best_a = _ensure_valid(ca)
                best_score = float(cs)
                best_x = np.clip(cx, 0.0, 1.0)

    # If for any reason we still have a decent odd/full sequence, do full mirror polish.
    if best_x is None and _remaining() > 180.0:
        alloc = min(120.0, _remaining() - 150.0)
        if alloc > 30.0:
            try:
                af, sf = _mirror_optimize_full(best_a, alloc, rng)
                if np.isfinite(sf) and sf < best_score - 1e-12:
                    best_a = _ensure_valid(af)
                    best_score = float(sf)
            except Exception:
                pass

    best_a = _ensure_valid(best_a)
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a = best_a / ss

    final_seq = [float(v) for v in best_a]

    if best_a.size <= 20000 and time.monotonic() < final_deadline - 5.0:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print("final internal score={:.8f} exact score={:.8f} n={}".format(best_score, exact, best_a.size))
        else:
            print("final internal score={:.8f} exact unavailable n={}".format(best_score, best_a.size))
    else:
        print("final internal score={:.8f} n={}".format(best_score, best_a.size))

    return final_seq
