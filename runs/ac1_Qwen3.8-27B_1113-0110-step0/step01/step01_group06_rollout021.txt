"""
Hybrid subgradient and smooth-gradient optimizer for step-function convolution minimization.
Uses multi-phase strategy: refine given solution, anti-palindromic exploration at larger n,
cross-pollination, and final polishing with diverse restarts.
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _make_weights(b, focus, eps=0.05):
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    with np.errstate(over="ignore", under="ignore"):
        z = (maxb - b) / maxb
        p = np.exp(-z * float(focus))
    s = float(p.sum())
    if s <= 1e-12:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - eps) * (p / s) + eps / L
    return p, maxb


def _make_topk_weights(b, k, eps=0.03):
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    k = min(k, L)
    topk = np.argpartition(b, -k)[-k:]
    p = np.zeros(L, dtype=np.float64)
    p[topk] = 1.0
    p = (1.0 - eps) * p + eps / L
    p /= p.sum()
    return p, maxb


def _random_positive(n, rng, amp=3.0):
    z = rng.normal(0.0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _anti_log_init(n, rng, amp=3.0, smooth=False):
    n = int(n)
    m = n // 2
    if m < 1:
        return np.ones(n, dtype=np.float64) / max(1, n)
    x = np.arange(m, dtype=np.float64)
    if smooth:
        v = np.zeros(m, dtype=np.float64)
        for f in (1, 2, 3, 5, 7, 11, 13):
            v += (rng.normal(0.0, 1.0) / max(1, f)) * np.sin(
                2.0 * np.pi * f * x / m + rng.uniform(0.0, 2.0 * np.pi)
            )
        v += 0.05 * rng.normal(0.0, 1.0, m)
    else:
        v = rng.normal(0.0, 1.0, m)
    v = v - float(v.mean())
    std = float(v.std())
    if std > 1e-12:
        v = (float(amp) * v) / std
    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0
    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _greedy_support(n, rng, shuffle=True):
    n = int(n)
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m, dtype=np.int64)
    if shuffle:
        rng.shuffle(order)
    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            affected_l = int(np.max(counts[s + l]))
            self_l = int(counts[2 * l])
            metric_l = max(affected_l + 2, self_l + 1)
            affected_r = int(np.max(counts[s + r]))
            self_r = int(counts[2 * r])
            metric_r = max(affected_r + 2, self_r + 1)
            y = l if metric_l <= metric_r else r
        if t > 0:
            counts[selected[:t] + y] += 2
        counts[2 * y] += 1
        selected[t] = y
        t += 1
    return selected


def _greedy_x(n, rng, shuffle=True):
    sel = _greedy_support(n, rng, shuffle=shuffle)
    m = n // 2
    x = np.full(m, 1e-6, dtype=np.float64)
    left = sel < m
    if np.any(left):
        x[sel[left]] = 1.0 - 1e-6
    return x


def _a_from_x(x, n=None):
    x = np.asarray(x, dtype=np.float64).ravel()
    m = x.size
    if n is None:
        n = 2 * m
    t = 1.0 / m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a


def _optimize_full(a0, time_limit, rng, focus_start=8.0, max_focus=200.0,
                   label="", verbose=False, use_topk=False, topk_size=20):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = None
    score = float("inf")
    best_score = float("inf")
    best_a = a.copy()

    focus = float(focus_start)
    focus_inc = 1.015
    eta = 0.5
    eta_min = 0.02
    eta_max = 10.0

    no_improve = 0
    max_no = 800
    resets = 0
    max_resets = 40
    t0 = time.monotonic()
    it = 0

    # Momentum buffer
    grad_momentum = np.zeros(n, dtype=np.float64)
    beta_mom = 0.7

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 2000 == 0):
            print(f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} "
                  f"focus={focus:.1f} eta={eta:.3f} resets={resets}")

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.4)
            base = best_a.copy()
            if resets % 3 == 0:
                fresh = _anti_log_init(n, rng, amp=3.0, smooth=False)
                a = 0.85 * base + 0.15 * fresh
            elif resets % 3 == 1:
                a = base * np.exp(rng.normal(0.0, 0.4, size=n))
            else:
                fresh = _random_positive(n, rng, amp=4.0)
                a = 0.7 * base + 0.3 * fresh
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = None
            score = float("inf")
            focus = float(rng.choice([5.0, 10.0, 20.0, 40.0, 80.0]))
            eta = 0.5
            grad_momentum = np.zeros(n, dtype=np.float64)
            continue

        # Compute weights
        if use_topk and it > 100:
            p, _ = _make_topk_weights(b, topk_size)
        else:
            p, _ = _make_weights(b, focus)

        grad = _fft_grad(p, a)
        # Remove mean to preserve sum (approximately)
        gd = grad - float(np.dot(a, grad))

        # Momentum
        grad_momentum = beta_mom * grad_momentum + (1.0 - beta_mom) * gd
        maxg = float(np.max(np.abs(grad_momentum)))
        if maxg < 1e-18:
            break

        gd_norm = grad_momentum / maxg
        a_new = a * np.exp(-eta * gd_norm)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss <= 1e-300:
            break
        a_new /= ss

        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        wt_old = float(np.dot(p, b))
        wt_new = float(np.dot(p, b_new))

        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        elif score_new < score * 1.0005 and wt_new < wt_old - 1e-12:
            accept = True

        if accept:
            x_old = a.copy()
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.03, eta_max)
        else:
            eta = max(eta * 0.7, eta_min)
            if eta < 0.06:
                a = a * np.exp(rng.normal(0.0, 0.08, size=n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                b = None
                score = float("inf")
                grad_momentum = np.zeros(n, dtype=np.float64)

        focus = min(focus * focus_inc, float(max_focus))
        it += 1

    return best_a, float(best_score)


def _optimize_subgrad(a0, time_limit, rng, label="", verbose=False):
    """Subgradient-based optimizer: always step to reduce the current argmax."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = None
    score = float("inf")
    best_score = float("inf")
    best_a = a.copy()

    eta = 1.0
    eta_min = 0.05
    eta_max = 20.0
    no_improve = 0
    max_no = 500
    resets = 0
    max_resets = 30
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 2000 == 0):
            print(f"[{label}-sub] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} eta={eta:.3f}")

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.5)
            a = best_a * np.exp(rng.normal(0.0, 0.3, size=n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = None
            score = float("inf")
            eta = 1.0
            continue

        # Subgradient: find argmax, compute gradient direction
        k_star = int(np.argmax(b))
        # The subgradient of max_k (a*a)[k] wrt a is:
        # g[j] = a[(k_star - j) mod ... ] + a[(j - k_star) ... ]
        # More precisely: g[j] = a[k_star - j] (if valid) + a[j - k_star] (if valid)
        g = np.zeros(n, dtype=np.float64)
        # Term 1: a[k_star - j] for j = 0, ..., min(k_star, n-1)
        j1 = np.arange(min(k_star + 1, n))
        g[j1] += a[k_star - j1]
        # Term 2: a[j - k_star] for j = k_star, ..., n-1
        if k_star < n:
            j2 = np.arange(k_star, n)
            g[j2] += a[j2 - k_star]

        gd = g - float(np.dot(a, g))
        maxg = float(np.max(np.abs(gd)))
        if maxg < 1e-18:
            break
        gd = gd / maxg

        a_new = a * np.exp(-eta * gd)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss <= 1e-300:
            break
        a_new /= ss

        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        if score_new < score - 1e-12 or score_new < best_score - 1e-12:
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.04, eta_max)
        else:
            eta = max(eta * 0.72, eta_min)
            if eta < 0.08:
                a = a * np.exp(rng.normal(0.0, 0.1, size=n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                b = None
                score = float("inf")

        it += 1

    return best_a, float(best_score)


def _optimize_pair(x0, time_limit, rng, focus_start=8.0, max_focus=300.0,
                   label="", verbose=False):
    x = np.asarray(x0, dtype=np.float64).copy()
    m = int(x.size)
    n = 2 * m
    x_eps = 1e-8
    x = np.clip(x, x_eps, 1.0 - x_eps)

    a = _a_from_x(x, n)
    b = None
    score = float("inf")
    best_score = float("inf")
    best_x = x.copy()

    focus = float(focus_start)
    focus_inc = 1.012
    eta = 0.5
    eta_min = 0.02
    eta_max = 12.0
    no_improve = 0
    max_no = 1000
    resets = 0
    max_resets = 35
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 3000 == 0):
            print(f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} "
                  f"focus={focus:.1f} eta={eta:.3f} resets={resets}")

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.3)
            x = 0.75 * best_x + 0.25 * rng.normal(0.5, 0.3, size=m)
            x = np.clip(x, x_eps, 1.0 - x_eps)
            a = _a_from_x(x, n)
            b = None
            score = float("inf")
            focus = float(rng.choice([3.0, 8.0, 20.0, 50.0, 100.0]))
            eta = 0.5
            continue

        p, _ = _make_weights(b, focus)
        ga = _fft_grad(p, a)
        gx = ga[:m] - ga[m:][::-1]
        maxg = float(np.max(np.abs(gx)))
        if maxg < 1e-18:
            break
        gx = gx / maxg

        x_new = x - eta * gx
        x_new = np.clip(x_new, x_eps, 1.0 - x_eps)
        if not np.any(x_new != x):
            x_new = np.clip(x + rng.normal(0.0, 0.06, size=m), x_eps, 1.0 - x_eps)

        a_new = _a_from_x(x_new, n)
        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        wt_old = float(np.dot(p, b))
        wt_new = float(np.dot(p, b_new))

        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        elif score_new < score * 1.0005 and wt_new < wt_old - 1e-12:
            accept = True

        if accept:
            x = x_new
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.03, eta_max)
        else:
            eta = max(eta * 0.72, eta_min)
            if eta < 0.06:
                x = np.clip(x + rng.normal(0.0, 0.06, size=m), x_eps, 1.0 - x_eps)
                a = _a_from_x(x, n)
                b = None
                score = float("inf")

        focus = min(focus * focus_inc, float(max_focus))
        it += 1

    return _a_from_x(best_x, n), float(best_score)


def _upsample(a, n_target):
    """Upsample sequence to target length via zero-stuffing in frequency domain."""
    n = int(a.size)
    if n == n_target:
        return a.copy()
    N_old = _pow2(2 * n - 1)
    N_new = _pow2(2 * n_target - 1)
    fa = np.fft.rfft(a, n=N_old)
    fa_padded = np.zeros(N_new // 2 + 1, dtype=np.complex128)
    copy_len = min(len(fa), len(fa_padded))
    fa_padded[:copy_len] = fa[:copy_len]
    a_new = np.fft.irfft(fa_padded, n=N_new).real[:n_target]
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(n_target, dtype=np.float64) / n_target
    return a_new


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Refine given solution with full optimizer (smooth gradient)
    2. Refine with subgradient optimizer
    3. Anti-palindromic optimization at n=8192
    4. Full optimization from anti-pal seeds
    5. Multi-start with diverse initializations
    6. Final refinement of best candidate
    """
    seed = int(seed)
    total_budget = 2400.0
    now0 = time.monotonic()
    final_deadline = now0 + total_budget - 45.0

    print("starting hybrid optimizer, budget={}s".format(total_budget))
    best_a = None
    best_score = float("inf")

    # Load given solution
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                gn = np.maximum(gn, 1e-15)
                gn /= gn.sum()
                sc, _, _ = _internal_score(gn)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = gn.copy()
                given_a = gn
                print(f"given: n={gn.size}, score={sc:.6f}")
    except Exception:
        pass

    main_rng = np.random.default_rng(seed)

    # === Phase 1: Refine given solution with smooth gradient (400s) ===
    if given_a is not None and time.monotonic() < final_deadline - 300:
        alloc = min(400.0, final_deadline - time.monotonic() - 300.0)
        if alloc > 60:
            t0 = time.monotonic()
            print(f"phase1: refine given, n={given_a.size}, alloc={alloc:.0f}s")
            rng1 = np.random.default_rng((seed * 7919 + 1) % (2**32))
            try:
                cand, sc = _optimize_full(
                    given_a.copy(), alloc, rng1,
                    focus_start=8.0, max_focus=150.0,
                    label="p1", verbose=True
                )
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    print(f"phase1 improved: {sc:.6f}")
            except Exception as e:
                print("phase1 error:", repr(e))
            print(f"phase1 done, used={time.monotonic()-t0:.0f}s, best={best_score:.6f}")

    # === Phase 2: Subgradient refinement (200s) ===
    if best_a is not None and time.monotonic() < final_deadline - 200:
        alloc = min(200.0, final_deadline - time.monotonic() - 200.0)
        if alloc > 50:
            t0 = time.monotonic()
            print(f"phase2: subgradient, n={best_a.size}, alloc={alloc:.0f}s")
            rng2 = np.random.default_rng((seed * 104729 + 2) % (2**32))
            try:
                cand, sc = _optimize_subgrad(
                    best_a.copy(), alloc, rng2,
                    label="p2", verbose=True
                )
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    print(f"phase2 improved: {sc:.6f}")
            except Exception as e:
                print("phase2 error:", repr(e))
            print(f"phase2 done, used={time.monotonic()-t0:.0f}s, best={best_score:.6f}")

    # === Phase 3: Anti-palindromic at n=8192 (300s) ===
    if time.monotonic() < final_deadline - 200:
        n_anti = 8192
        alloc = min(300.0, final_deadline - time.monotonic() - 200.0)
        if alloc > 60:
            t0 = time.monotonic()
            print(f"phase3: anti-pal n={n_anti}, alloc={alloc:.0f}s")
            rng3 = np.random.default_rng((seed * 15485863 + 3) % (2**32))
            for attempt in range(2):
                if time.monotonic() >= final_deadline - 200:
                    break
                remaining = final_deadline - time.monotonic() - 200.0
                sub_alloc = min(alloc / 2.0, remaining)
                if sub_alloc < 40:
                    break
                x0 = _greedy_x(n_anti, rng3, shuffle=(attempt == 1))
                try:
                    cand, sc = _optimize_pair(
                        x0, sub_alloc, rng3,
                        focus_start=6.0, max_focus=200.0,
                        label=f"p3-{attempt}", verbose=(attempt == 0)
                    )
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = cand.copy()
                        print(f"phase3-{attempt} improved: {sc:.6f}")
                except Exception as e:
                    print(f"phase3-{attempt} error:", repr(e))
            print(f"phase3 done, used={time.monotonic()-t0:.0f}s, best={best_score:.6f}")

    # === Phase 4: Full optimization from best (smooth, topk variant) (300s) ===
    if best_a is not None and time.monotonic() < final_deadline - 150:
        alloc = min(300.0, final_deadline - time.monotonic() - 150.0)
        if alloc > 50:
            t0 = time.monotonic()
            # Try with different n if beneficial
            n_target = best_a.size
            print(f"phase4: full topk, n={n_target}, alloc={alloc:.0f}s")
            rng4 = np.random.default_rng((seed * 1000000007 + 4) % (2**32))
            try:
                cand, sc = _optimize_full(
                    best_a.copy(), alloc, rng4,
                    focus_start=5.0, max_focus=100.0,
                    label="p4", verbose=True,
                    use_topk=True, topk_size=30
                )
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    print(f"phase4 improved: {sc:.6f}")
            except Exception as e:
                print("phase4 error:", repr(e))
            print(f"phase4 done, used={time.monotonic()-t0:.0f}s, best={best_score:.6f}")

    # === Phase 5: Multi-start at various n (250s) ===
    if time.monotonic() < final_deadline - 100:
        alloc_remaining = final_deadline - time.monotonic() - 100.0
        n_starts = [5582, 8192, 4096]
        if best_a is not None and best_a.size not in n_starts:
            n_starts.insert(0, best_a.size)
        n_starts = list(set(n_starts))[:3]

        per_start = max(40.0, alloc_remaining / len(n_starts))
        t0 = time.monotonic()
        print(f"phase5: multi-start n={n_starts}, alloc={alloc_remaining:.0f}s")

        for si, n_s in enumerate(n_starts):
            if time.monotonic() >= final_deadline - 100:
                break
            remaining = final_deadline - time.monotonic() - 100.0
            if remaining < 40:
                break
            alloc_s = min(per_start, remaining)
            rng5 = np.random.default_rng((seed * 31 + si * 9973) % (2**32))

            if best_a is not None and best_a.size == n_s:
                a0 = best_a.copy()
            else:
                if n_s % 2 == 0:
                    a0 = _anti_log_init(n_s, rng5, amp=3.0, smooth=True)
                else:
                    a0 = _random_positive(n_s, rng5, amp=3.0)

            print(f"  phase5-{si}: n={n_s}, alloc={alloc_s:.0f}s")
            try:
                cand, sc = _optimize_full(
                    a0, alloc_s, rng5,
                    focus_start=10.0, max_focus=120.0,
                    label=f"p5-{si}", verbose=False
                )
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    print(f"  phase5-{si} improved: {sc:.6f}")
            except Exception as e:
                print(f"  phase5-{si} error:", repr(e))

        print(f"phase5 done, used={time.monotonic()-t0:.0f}s, best={best_score:.6f}")

    # === Phase 6: Final refinement (remaining time) ===
    if best_a is not None:
        remaining = final_deadline - time.monotonic()
        if remaining > 50:
            refine_time = min(remaining - 30.0, remaining * 0.6)
            if refine_time > 30:
                t0 = time.monotonic()
                print(f"phase6: final refine, n={best_a.size}, alloc={refine_time:.0f}s")
                rng6 = np.random.default_rng((seed + 7777777) % (2**32))
                try:
                    # Try both smooth and subgradient
                    half_time = refine_time / 2.0
                    cand, sc = _optimize_full(
                        best_a.copy(), half_time, rng6,
                        focus_start=5.0, max_focus=80.0,
                        label="p6a", verbose=True
                    )
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = cand.copy()

                    cand2, sc2 = _optimize_subgrad(
                        best_a.copy(), refine_time - half_time, rng6,
                        label="p6b", verbose=False
                    )
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_score = sc2
                        best_a = cand2.copy()
                except Exception as e:
                    print("phase6 error:", repr(e))
                print(f"phase6 done, used={time.monotonic()-t0:.0f}s, best={best_score:.6f}")

    # Ensure valid output
    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]
    print(f"FINAL: n={best_a.size}, internal_score={best_score:.6f}")

    if time.monotonic() < final_deadline - 5.0 and best_a.size <= 20000:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"FINAL exact score: {exact:.6f}")

    return final_seq
