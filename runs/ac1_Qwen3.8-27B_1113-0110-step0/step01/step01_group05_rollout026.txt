import numpy as np

def _clean(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _normalize(a):
    a = _clean(a)
    s = a.sum()
    if s <= 0:
        return np.ones_like(a) / len(a)
    return a / s


def _score(a):
    a = _normalize(a)
    c = np.convolve(a, a)
    return 2.0 * len(a) * c.max()


def _score_p(a, p):
    a = _normalize(a)
    c = np.convolve(a, a)
    return 2.0 * len(a) * (np.sum(c**p))**(1.0/p)

def _proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.sum(v) <= 0:
        return np.full(n, 1.0/n)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.argmax(v_sorted > (cssv - 1.0) / np.arange(1, n+1))
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)


def _pgd_optimize(a0, deadline, p_schedule, verbose=True):

    a0: initial point (will be normalized).

    for p in p_schedule:

        c = np.convolve(a, a)

                sc = _score(a)

                    if verbose:
                        print(f"  p={p}, it={it}, score={best_score:.6f}")

    return best_a, best_score


def _make_initial_candidates(n, rng):
    """Generate diverse initial candidates."""
    cands = []
    
    # Two-level half block
    a = np.zeros(n)
    a[:n//2] = 1.0/(n//2) * 0.7
    a[n//2:] = 1.0/(n-n//2) * 0.3
    cands.append(_normalize(a))
    
    # Various two-level splits
    for frac in [0.3, 0.4, 0.6, 0.2, 0.8]:
        a = np.zeros(n)
        split = int(frac * n)
        a[:split] = 1.0/split
        a[split:] = 0.4/(n-split)
        cands.append(_normalize(a))
    
    # Linear ramp
    a = np.linspace(0.01, 1.0, n)
    cands.append(_normalize(a))
    
    # Quadratic
    x = np.linspace(0, 1, n)
    a = x**2 + 0.01
    cands.append(_normalize(a))
    a = (1-x)**2 + 0.01
    cands.append(_normalize(a))
    
    # Gaussian centered
    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.2, 0.3, 0.35, 0.4]:
        a = np.exp(-0.5*((x - n/2)/(n*sigma_frac))**2)
        cands.append(_normalize(a))
    
    # Raised cosine
    a = 0.5*(1 + np.cos(2*np.pi*np.arange(n)/n))
    a = np.maximum(a, 0.01)
    cands.append(_normalize(a))
    
    # Sine-squared
    a = np.sin(np.pi*np.arange(n)/n)**2 + 0.01
    cands.append(_normalize(a))
    
    # Anti-symmetric (high at one end)
    a = np.linspace(0.05, 1.0, n)**1.5
    cands.append(_normalize(a))
    a = np.linspace(1.0, 0.05, n)**1.5
    cands.append(_normalize(a))
    
    # U-shaped (high at edges)
    x = np.linspace(0, 1, n)
    a = (x - 0.5)**2 * 4 + 0.1
    cands.append(_normalize(a))
    
    # Random
    for _ in range(3):
        a = rng.random(n) ** 2
        cands.append(_normalize(a))
    
    # Paired anti-correlated
    m = (n+1)//2
    for _ in range(2):
        a = np.zeros(n)
        for r in range(m):
            i = r
            j = n - 1 - r
            if i == j:
                a[i] = rng.random()
            else:
                a[i] = rng.random()
                a[j] = max(0, 1 - a[i]) * rng.random()
        cands.append(_normalize(a))
    
    return cands


def _expand_proto(proto, target_len):
    """Expand a short prototype to target length by block replication."""
    proto = _normalize(proto)
    R = len(proto)
    if R == target_len:
        return proto.copy()
    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=int)
    if rem > 0:
        lengths[:rem] += 1
    out = np.zeros(target_len)
    idx = 0
    for i in range(R):
        L = lengths[i]
        if L > 0:
            out[idx:idx+L] = proto[i] / L
            idx += L
    return _normalize(out)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Find the best non-negative coefficient sequence minimizing the score."""
    budget = max(10.0, float(budget_s))
    deadline = time.monotonic() + budget - 10.0
    rng = np.random.default_rng(int(seed))
    n = 3809
    
    print(f"Budget: {budget}s, deadline in {deadline - time.monotonic():.1f}s")
    
    # Collect candidates
    all_candidates = []  # list of (score, array)
    
    # Try to use provided sequence
    try:
        hseq = _clean(height_sequence_1)
        if hseq.size > 0:
            sc = _score(hseq)
            all_candidates.append((sc, hseq.copy()))
            print(f"Provided sequence score: {sc:.6f}")
    except:
        pass
    
    # Two-level baseline
    a_base = np.zeros(n)
    a_base[:n//2] = 0.7/(n//2)
    a_base[n//2:] = 0.3/(n-n//2)
    a_base = _normalize(a_base)
    all_candidates.append((_score(a_base), a_base.copy()))
    
    # Phase 1: Optimize short prototypes (fast)
    print("\n=== Phase 1: Short prototype optimization ===")
    proto_deadline = time.monotonic() + min(200.0, 0.25 * budget)
    
    proto_results = []
    for pn in [16, 32, 48, 64, 96, 128, 192, 256]:
        if time.monotonic() > proto_deadline:
            break
        for trial in range(3):
            if time.monotonic() > proto_deadline:
                break
            strat = int(rng.integers(0, 8))
            if strat == 0:
                p = np.ones(pn)
            elif strat == 1:
                p = rng.dirichlet(np.ones(pn) * 0.5)
            elif strat == 2:
                p = np.linspace(0.01, 1.0, pn)
            elif strat == 3:
                x = np.arange(pn, dtype=float)
                p = np.exp(-0.5*((x-pn/2)/(pn*0.3))**2)
            elif strat == 4:
                p = np.zeros(pn)
                p[:pn//2] = 1.0/(pn//2)
                p[pn//2:] = 0.4/(pn-pn//2)
            elif strat == 5:
                p = np.linspace(1.0, 0.01, pn)**2
            elif strat == 6:
                p = (np.sin(np.pi*np.arange(pn)/pn)**2 + 0.1)
            else:
                p = rng.random(pn)**2
            p = _normalize(p)
            
            sc = _score_p(p, 8)
            p_opt, sc_opt = _pgd_optimize(p, time.monotonic() + max(5, (proto_deadline - time.monotonic())/15), [4, 8, 16])
            true_sc = _score(p_opt)
            proto_results.append((true_sc, p_opt))
            if true_sc < min(s for s, _ in all_candidates):
                all_candidates.append((true_sc, p_opt.copy()))
            
            if time.monotonic() % 10 < 5:
                print(f"  proto n={pn}, trial={trial}, score={true_sc:.6f}")
    
    # Sort proto results
    proto_results.sort(key=lambda x: x[0])
    print(f"  Best proto score: {proto_results[0][0]:.6f}")
    
    # Phase 2: Expand best prototypes and optimize at full length
    print("\n=== Phase 2: Full-length optimization ===")
    
    # Generate initial candidates at full length
    full_cands = _make_initial_candidates(n, rng)
    
    # Add expanded prototypes
    for sc, p in proto_results[:6]:
        exp = _expand_proto(p, n)
        full_cands.append(exp)
    
    # Use provided sequence if available
    try:
        hseq = _clean(height_sequence_1)
        if hseq.size == n:
            full_cands.append(_normalize(hseq))
    except:
        pass
    
    # Optimize each full-length candidate
    best_full_a = None
    best_full_score = float('inf')
    
    full_deadline = deadline - 60.0  # leave time for polish
    n_opt = len(full_cands)
    time_per_cand = max(5.0, (full_deadline - time.monotonic()) / max(1, n_opt))
    
    # Sort candidates by initial score (optimize promising ones first)
    scored_cands = [( _score(c), c) for c in full_cands]
    scored_cands.sort(key=lambda x: x[0])
    
    print(f"  {n_opt} candidates to optimize, ~{time_per_cand:.0f}s each")
    
    for idx, (sc0, cand) in enumerate(scored_cands):
        if time.monotonic() > full_deadline:
            break
        cand_deadline = time.monotonic() + time_per_cand
        if idx == 0:
            cand_deadline = min(deadline - 60, time.monotonic() + time_per_cand * 1.5)
        
        sc_opt, a_opt = _pgd_optimize(cand, cand_deadline, [4, 8, 16], verbose=False)
        true_sc = _score(a_opt)
        
        if true_sc < best_full_score:
            best_full_score = true_sc
            best_full_a = a_opt.copy()
        
        all_candidates.append((true_sc, a_opt.copy()))
        print(f"  cand {idx}: init={sc0:.6f}, opt={true_sc:.6f}, time_left={deadline-time.monotonic():.0f}s")
    
    # Phase 3: Coordinate descent polish on best
    print(f"\n=== Phase 3: Coordinate polish ===")
    print(f"  Current best: {best_full_score:.6f}")
    
    if best_full_a is not None:
        polish_deadline = deadline - 20.0
        a_polished, sc_polished = _coordinate_polish(best_full_a, polish_deadline)
        if sc_polished < best_full_score:
            best_full_score = sc_polished
            best_full_a = a_polished.copy()
            print(f"  After polish: {best_full_score:.6f}")
    
    # Also try PGD from the polished point with higher p
    if best_full_a is not None and time.monotonic() < deadline - 30:
        a_hp, sc_hp = _pgd_optimize(best_full_a, deadline - 25, [32, 64, 128], verbose=False)
        true_hp = _score(a_hp)
        if true_hp < best_full_score:
            best_full_score = true_hp
            best_full_a = a_hp.copy()
            print(f"  After high-p PGD: {best_full_score:.6f}")
    
    # Final: verify with external evaluator
    print(f"\nFinal best score (internal): {best_full_score:.6f}")
    
    if best_full_a is not None:
        try:
            ext_sc = evaluate_sequence([float(x) for x in best_full_a])
            print(f"Final score (external): {ext_sc:.6f}")
        except:
            pass
    
    # Return best
    if best_full_a is not None:
        out = _normalize(best_full_a)
        return [float(x) for x in out]
    else:
        return [float(x) for x in _normalize(a_base)]
