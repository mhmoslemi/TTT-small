Optimized step-function search using accelerated mirror descent with 
structured escape mechanisms and multi-resolution refinement.
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_optimize
except Exception:
    _scipy_optimize = None


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))


def compute_weights(c, cmax, tau, floor=0.0):
    """Compute softmax weights over convolution coefficients."""
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return c
    if cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)

    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = ez / s
    if floor > 0.0:
        w = (1.0 - floor) * w + (floor / float(L))
    return w

    """Compute gradient of weighted convolution sum."""
    w = np.asarray(w, dtype=np.float64).ravel()

    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1: 2 * n - 1][::-1]

Actually, for L-BFGS, I just need the gradient, and L-BFGS approximates the Hessian internally. So I just need:

grad f_smooth(a) = 2n * (1/τ) * sum_k p_k * ∂c_k/∂a = 2n * (1/τ) * (p ⋆ a) * 2

where p = softmax(c/τ).

This is the same as what we're already computing! So L-BFGS with the smoothed objective is essentially the same as the mirror descent with a fixed temperature.

The difference is:
- Mirror descent uses a multiplicative update (guided by the mirror map)
- L-BFGS uses an additive update with a curved step (guided by the approximated Hessian)

L-BFGS might be better near the optimum (where the Hessian information is more useful), while mirror descent might be better in the early stages.

**Strategy: Use L-BFGS for the final polish, after mirror descent has gotten close.**

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))


def compute_weights(c, cmax, tau, floor=0.0):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return c
    if cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = ez / s
    if floor > 0.0:
        w = (1.0 - floor) * w + (floor / float(L))
    return w


def grad_from_conv_weights(w, fa, N, n):
    w = np.asarray(w, dtype=np.float64).ravel()
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1: 2 * n - 1][::-1]


def make_uniform(n):
    return np.ones(max(1, int(n)), dtype=np.float64) / float(max(1, int(n)))


def resample_seq(a, n):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == n:
        return a.astype(np.float64, copy=True)
    if old_n == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)

def smooth_objective(a, N, tau):
    """Smoothed version of 2n*max(c)."""
    c = convolve_seq(a, N)
    cmax = float(np.max(c))
    n = a.size
    if cmax <= 0 or not np.isfinite(cmax) or tau <= 0:
        return float(2 * n * cmax)
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0:
        return float(2 * n * cmax)
    return float(2 * n * (cmax + tau * np.log(s)))


def smooth_grad(a, N, tau):
    """Gradient of smoothed objective."""
    n = a.size
    c = convolve_seq(a, N)
    cmax = float(np.max(c))
    if cmax <= 0 or not np.isfinite(cmax) or tau <= 0:
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    else:
        z = (c - cmax) / tau
        z = np.clip(z, -50.0, 0.0)
        ez = np.exp(z)
        s = float(ez.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w = ez / s
    fa = np.fft.rfft(a, N)
    g = grad_from_conv_weights(w, fa, N, n)
    return float(2 * n) * g


def mirror_descent_momentum(a_init, deadline, seed=0, verbose=False, max_iter=50000):
    """Mirror descent with momentum, adaptive temperature, and structured escapes."""
    a = normalize_positive(a_init, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = internal_score(a, N)

    # Parameters
    eta = 0.4
    tau_factor = 0.12
    momentum = 0.8
    floor_frac = 0.03

    # State
    mom = np.zeros(n, dtype=np.float64)
    stagnant = 0
    it = 0
    last_best = best_score
    escape_count = 0

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    cmax = float(np.max(c))
    score = float(2.0 * n * cmax)

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        tau = tau_factor * max(cmax, 1e-10)
        w = compute_weights(c, cmax, tau, floor=floor_frac)
        g = grad_from_conv_weights(w, fa, N, n)
        g = g - g.mean()

        # Momentum
        mom = momentum * mom + (1.0 - momentum) * g
        g_eff = g + momentum * mom

        g_scale = float(np.abs(g_eff).max())
        if not np.isfinite(g_scale) or g_scale < 1e-15:
            break
        g_eff = g_eff / g_scale

        # Try multiple step sizes
        accepted = False
        for e in (eta, 0.6 * eta, 0.4 * eta, 0.25 * eta):
            if e < 1e-12:
                continue
            loga = np.log(a + 1e-20)
            loga_try = loga - e * g_eff
            loga_try = loga_try - float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -50.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try

            fa_try = np.fft.rfft(a_try, N)
            c_try = np.fft.irfft(fa_try * fa_try, N)[: 2 * n - 1]
            cmax_try = float(np.max(c_try))
            score_try = float(2.0 * n * cmax_try)

            if np.isfinite(score_try) and score_try < score * (1.0 - 1e-10):
                a = a_try
                fa = fa_try
                c = c_try
                cmax = cmax_try
                score = score_try
                accepted = True
                if score < best_score * (1.0 - 1e-12):
                    best_score = score
                    best_a = a.copy()
                break

        if accepted:
            stagnant = 0
            eta = min(1.2, eta * 1.015)
            tau_factor = max(0.02, tau_factor * 0.9985)
        else:
            stagnant += 1
            eta = max(0.02, eta * 0.75)

            # Structured escape
            if stagnant > 80 and stagnant % 40 == 0:
                escape_count += 1
                esc_type = escape_count % 4

                if esc_type == 1:
                    # Sinusoidal perturbation
                    freq = rng.uniform(1.0, float(n) // 3)
                    phase = rng.uniform(0, 2 * np.pi)
                    amp = rng.uniform(0.03, 0.12)
                    pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                elif esc_type == 2:
                    # Block perturbation
                    block_size = int(rng.integers(max(4, n // 32), max(5, n // 8)))
                    start = int(rng.integers(0, n - block_size))
                    pert = np.zeros(n)
                    pert[start:start + block_size] = rng.uniform(-0.08, 0.08, block_size)
                elif esc_type == 3:
                    # Fourier-domain perturbation
                    fa_pert = fa * np.exp(1j * rng.normal(0, 0.3, fa.size))
                    a_pert = np.fft.irfft(fa_pert, N)[:n]
                    a_pert = normalize_positive(a_pert, floor=1e-15)
                    pert = 0.3 * np.log(a_pert + 1e-20) - 0.3 * np.log(a + 1e-20)
                else:
                    # Random smooth perturbation
                    sigma = 0.06
                    pert = rng.normal(0, sigma, n)
                    # Smooth it
                    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 3.0) ** 2)
                    kernel /= kernel.sum()
                    pert = np.convolve(pert, kernel, mode='same')

                a_try = a * np.exp(np.clip(pert, -1.0, 1.0))
                a_try = normalize_positive(a_try, floor=1e-15)
                sc_try = internal_score(a_try, N)

                if np.isfinite(sc_try):
                    a = a_try
                    fa = np.fft.rfft(a, N)
                    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
                    cmax = float(np.max(c))
                    score = float(2.0 * n * cmax)
                    mom = np.zeros(n)
                    eta = 0.3
                    tau_factor = max(0.05, tau_factor * 1.3)
                    stagnant = 0

            if stagnant > 400:
                # Try a completely different random restart blended with best
                blend = rng.uniform(0.3, 0.7)
                a_try = blend * best_a + (1.0 - blend) * normalize_positive(
                    rng.dirichlet(np.ones(n)), floor=1e-15)
                a_try = normalize_positive(a_try, floor=1e-15)
                sc_try = internal_score(a_try, N)
                if np.isfinite(sc_try) and sc_try < score:
                    a = a_try
                    fa = np.fft.rfft(a, N)
                    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
                    cmax = float(np.max(c))
                    score = float(2.0 * n * cmax)
                    mom = np.zeros(n)
                    eta = 0.4
                    stagnant = 0
                elif stagnant > 600:
                    break

        if verbose and it % 1000 == 0:
            print("  MD it={} score={:.10f} best={:.10f} eta={:.3f} tau_f={:.4f}".format(
                it, score, best_score, eta, tau_factor), flush=True)

    return best_a, best_score

def block_coordinate_descent(a_init, deadline, seed=0, block_size=32):
    """Cyclic block coordinate descent: optimize blocks of coordinates."""
    a = normalize_positive(a_init, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = internal_score(a, N)

    bs = min(block_size, n)
    num_blocks = int(np.ceil(n / bs))

    # Precompute the "fixed" part contribution
    # For block i at positions [s, e), the objective depends on c = conv(a,a)
    # We optimize by doing a few gradient steps on the block coordinates

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Shuffle block order
        block_order = list(range(num_blocks))
        rng.shuffle(block_order)

        for bi in block_order:
            if time.monotonic() > deadline:
                break
            s = bi * bs
            e = min(s + bs, n)
            if e <= s:
                continue

            # Compute current convolution
            fa = np.fft.rfft(a, N)
            c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
            cmax = float(np.max(c))
            tau = 0.08 * max(cmax, 1e-10)
            w = compute_weights(c, cmax, tau, floor=0.02)
            g = grad_from_conv_weights(w, fa, N, n)

            # Optimize block [s:e] with a few gradient steps
            for inner_it in range(3):
                g_block = g[s:e] - g[s:e].mean()
                if np.abs(g_block).max() < 1e-14:
                    break
                # Multiplicative update on the block
                step = 0.3
                loga_block = np.log(a[s:e] + 1e-20) - step * g_block / (np.abs(g_block).max() + 1e-15)
                a_new_block = np.exp(np.clip(loga_block - loga_block.max(), -50, 0))
                a_new_block = np.maximum(a_new_block, 1e-200)

                a[s:e] = a_new_block
                a = normalize_positive(a, floor=1e-15)

            # Check improvement
            fa = np.fft.rfft(a, N)
            c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
            cmax = float(np.max(c))
            score = float(2.0 * n * cmax)
            if np.isfinite(score) and score < best_score * (1 - 1e-12):
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def generate_diverse_starts(n, seed, num_starts=12):
    """Generate a diverse set of initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []

    # Uniform
    starts.append(make_uniform(n))

    # Sine modulated

            x = np.linspace(0, 1, n)
            a = 1.0 + amp * np.sin(2 * np.pi * freq * x)
            starts.append(normalize_positive(a, floor=1e-12))

    # Cosine modulated
    for freq in [2, 3, 5]:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x)
        starts.append(normalize_positive(a, floor=1e-12))

    # Gaussian centered
    for mu in [0.25, 0.4, 0.5, 0.6, 0.75]:
        for sig in [0.08, 0.15, 0.25, 0.35]:
            x = np.linspace(0, 1, n)
            a = np.exp(-((x - mu) ** 2) / (2 * sig * sig))
            starts.append(normalize_positive(a, floor=1e-12))

    # Smooth random
    for _ in range(5):
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for m in range(1, 7):
            a += rng.uniform(0.05, 0.3) / (m ** 0.5) * np.sin(np.pi * m * x + rng.uniform(0, 2 * np.pi))
            a += rng.uniform(0.05, 0
</br>.2) / (m ** 0.5) * np.cos(2 * np.pi * m * x + rng.uniform(0, 2 * np.pi))
        starts.append(normalize_positive(a, floor=1e-12))

    # Power law
    for alpha in [0.3, 0.7, 1.3, 2.0]:
        x = np.linspace(0.01, 1, n)
        a = x ** alpha
        starts.append(normalize_positive(a, floor=1e-12))

    # Quadratic
    x = np.linspace(0, 1, n)
    for peak in [0.3, 0.5, 0.7]:
        a = 1.0 - 0.5 * ((x - peak) / 0.3) ** 2
        starts.append(normalize_positive(a, floor=1e-12))

    # Truncate to num_starts, keeping diverse ones
    N_eval = next_pow2(2 * n - 1)
    scored = []
    for s in starts:
        sc = internal_score(s, N_eval)
        scored.append((sc, s))
    scored.sort(key=lambda x: x[0])

    # Take top few and some random ones for diversity
    result = [s for _, s in scored[:min(6, len(scored))]]
    remaining = [s for _, s in scored[6:]]
    if remaining and len(result) < num_starts:
        idx = rng.choice(len(remaining), size=min(len(remaining), num_starts - len(result)), replace=False)
        result.extend([remaining[i] for i in idx])

    return result[:num_starts]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start_time = time.monotonic()
    total = max(120.0, float(budget_s) - 30.0)
    deadline = start_time + total

    # Initial best
    best_a = make_uniform(2048)
    best_score = internal_score(best_a)
    print("initial uniform n=2048 score = {:.10f}".format(best_score), flush=True)

    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            print("given sequence score = {:.10f}, n={}".format(best_score, len(best_a)), flush=True)

    # Phase 1: Quick exploration across lengths (15% of time)
    phase1_deadline = start_time + 0.15 * total
    for n in [1024, 2048, 4096]:
        if time.monotonic() > phase1_deadline:
            break
        starts = generate_diverse_starts(n, seed + n, num_starts=8)
        N_eval = next_pow2(2 * n - 1)
        for s in starts:
            sc = internal_score(s, N_eval)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = s.copy()
                print("exploration n={} score={:.10f}".format(n, best_score), flush=True)

    # Phase 2: Mirror descent at multiple resolutions (40% of time)
    phase2_deadline = start_time + 0.55 * total
    for n in [2048, 4096]:
        if time.monotonic() > phase2_deadline:
            break
        a_init = resample_seq(best_a, n)
        time_per_run = min(60.0, (phase2_deadline - time.monotonic()) / 2.0)
        if time_per_run < 15:
            continue

        # Run from best start
        a_opt, sc_opt = mirror_descent_momentum(
            a_init, time.monotonic() + time_per_run, seed=seed + 100 + n
        )
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print("MD n={} score={:.10f}".format(n, best_score), flush=True)

        # Run from a different start for diversity
        if time.monotonic() < phase2_deadline - 15:
            starts = generate_diverse_starts(n, seed + 500 + n, num_starts=3)
            for si, s in enumerate(starts):
                if time.monotonic() > phase2_deadline:
                    break
                t_per = min(30.0, phase2_deadline - time.monotonic() - 10.0)
                if t_per < 10:
                    break
                a_opt2, sc2 = mirror_descent_momentum(
                    s, time.monotonic() + t_per, seed=seed + 200 + n + si
                )
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_a = a_opt2.copy()
                    print("MD alt n={} start={} score={:.10f}".format(n, si, best_score), flush=True)

    # Phase 3: L-BFGS polish with annealing (25% of time)
    phase3_deadline = start_time + 0.80 * total
    for n in [best_a.size]:
        if time.monotonic() > phase3_deadline:
            break
        a_init = resample_seq(best_a, n)
        t_allow = min(120.0, phase3_deadline - time.monotonic())
        if t_allow < 20:
            continue
        a_opt, sc_opt = lbfqs_pollish(a_init, time.monotonic() + t_allow, seed=seed + 5000)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print("L-BFGS n={} score={:.10f}".format(n, best_score), flush=True)

    # Phase 4: Long mirror descent on best sequence (remaining time minus buffer)
    phase4_deadline = deadline - 30.0
    if time.monotonic() < phase4_deadline:
        a_init = best_a.copy()
        a_opt, sc_opt = mirror_descent_momentum(
            a_init, phase4_deadline, seed=seed + 9000, verbose=True
        )
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print("final MD score={:.10f}".format(best_score), flush=True)

    # Phase 5: Try longer sequence if time remains
    if time.monotonic() < deadline - 60:
        for n_try in [4096, 8192]:
            if time.monotonic() > deadline - 40:
                break
            a_init = resample_seq(best_a, n_try)
            t_allow = min(60.0, deadline - time.monotonic() - 20)
            if t_allow < 20:
                break
            a_opt, sc_opt = mirror_descent_momentum(a_init, time.monotonic() + t_allow, seed=seed + 99000 + n_try)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print("longer n={} score={:.10f}".format(n_try, best_score), flush=True)

    # Final L-BFGS if time permits
    if time.monotonic() < deadline - 15 and _scipy_optimize is not None:
        t_allow = deadline - time.monotonic() - 5
        if t_allow > 10:
            a_opt, sc_opt = lbfqs_pollish(best_a, time.monotonic() + t_allow, seed=seed + 77777)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print("final L-BFGS score={:.10f}".format(best_score), flush=True)

    # Output
    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")

    if not np.isfinite(final_eval):
        best_a = make_uniform(2048)
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print("FINAL: internal={:.12f}, evaluated={:.12f}, len={}".format(
        best_score, final_eval, len(out_list)), flush=True)
    return out_list
