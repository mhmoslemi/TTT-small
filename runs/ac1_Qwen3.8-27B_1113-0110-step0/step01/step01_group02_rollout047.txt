This is related to the concept of a "difference set" in combinatorics. A $(v, k, \lambda)$ difference set in a group of order $v$ has the property that the difference multiset is uniform (each non-identity element appears exactly $\lambda$ times). If we could find a "soft" version of a difference set (a probability distribution whose "convolution" is as flat as possible), that would be optimal.

For the cyclic group $\mathbb{Z}_n$, a $(v, k, \lambda)$ difference set has $k(k-1) = \lambda(v-1)$. The "convolution" of the indicator function with its reverse would be $\lambda$ for all non-zero shifts and $k$ for shift 0.

For a "soft" version (probability distribution), we want $p * \tilde{p}$ (where $\tilde{p}_j = p_{-j}$) to be as constant as possible. This is exactly what we want!

In the Fourier domain: $\widehat{p * \tilde{p}} = |\hat{p}|^2$. So we want $|\hat{p}(\omega)|^2$ to be as constant as possible (except at $\omega = 0$ where it's $S^2$).

This means we want $|\hat{p}(\omega)|$ to be approximately constant for $\omega \neq 0$. This is the "flat spectrum" or "equiangular" condition.

For a sequence of length $n$ with $\sum p_j = 1$, we have $\hat{p}(0) = 1$. For the other frequencies, we want $|\hat{p}(2\pi m/n)| \approx c$ for some constant $c$.

        return np.maximum(a, 0.005)

    # Build candidate list: (name, n, array)
    candidates = []

    # Previous best
    if prev_best is not None:
        candidates.append(("prev_best", len(prev_best), prev_best.copy()))

    # Key: decaying uniform (theoretical optimal shape for large n)
    for n in [2000, 4000, 8000]:
        candidates.append((f"decay_{n}", n, gen_decay_uniform(n)))

    # Sine modulations (previous best was from this family)
    for n in [2000, 4000]:
        candidates.append((f"sin_{n}", n, gen_sine_mod(n, 0.4, np.pi / 4)))
        candidates.append((f"sin_{n}_b", n, gen_sine_mod(n, 0.35, 0.3, 1.0)))

    # Two-harmonic
    for n in [2000, 4000]:
        candidates.append((f"2har_{n}", n, gen_two_harmonic(n)))
        candidates.append((f"2har_{n}_b", n, gen_two_harmonic(n, 0.4, 0.15, 0.8, 1.5)))

    # Bell shapes
    candidates.append(("bell_2000", 2000, gen_bell(2000, 0.5, 0.35)))
    candidates.append(("bell_3000", 3000, gen_bell(3000, 0.45, 0.3)))

    # Decay + sine
    candidates.append(("dsin_2000", 2000, gen_decay_sin(2000, 1.2, 0.35)))
    candidates.append(("dsin_3000", 3000, gen_decay_sin(3000, 1.5,

0.3)))

    # Random perturbations of previous best
    if prev_best is not None:
        for trial in range(3):
            noise = rng.normal(0, 0.05, len(prev_best))
            a_pert = np.maximum(prev_best * np.exp(noise), 0)
            candidates.append((f"prev_noisy_{trial}", len(prev_best), a_pert))

    print(f"Total candidates: {len(candidates)}")

    # Quick score all candidates
    scored = []
    for name, n, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s0 = a0.sum()
        if s0 < 1e-10:
            continue
        a0 = a0 / s0
        c0 = _fft_convolve(a0)
        init_score = 2.0 * n * float(np.max(c0))
        scored.append((init_score, name, a0))

    scored.sort(key=lambda x: x[0])
    print("Initial scores (top 10):")
    for i, (sc, name, _) in enumerate(scored[:10]):
        print(f"  {i+1}. {name}: {sc:.6f}")

    # Allocate time to candidates (geometric weighting)
    n_top = min(len(scored), 10)
    remaining = deadline - time.monotonic()
    weights = [max(1.0, n_top - i) ** 1.5 for i in range(n_top)]
    total_w = sum(weights)
    times = [remaining * w / total_w for w in weights]

    # Optimize top candidates
    for i in range(n_top):
        time_left = deadline - time.monotonic()
        if time_left < 30:
            break

        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], time_left)
        n = len(a0)

        print(f"\n[{i+1}/{n_top}] Optimizing '{name}' (n={n}, init={init_score:.6f}, time={t_alloc:.0f}s)")

        # Full refinement pipeline
        a_opt, score = _refine_sequence(a0, t_alloc, rng)
        eval_score = evaluate_sequence(a_opt.tolist())
        print(f"  Result: {eval_score:.6f}")

        if np.isfinite(eval_score) and eval_score < best_overall_score:
            best_overall_score = eval_score
            best_overall = a_opt.copy()
            print(f"  *** NEW BEST: {eval_score:.6f} ***")

        # Also try coordinate descent directly (sometimes better than subgrad first)
        time_left2 = deadline - time.monotonic()
        if time_left2 > 20 and t_alloc > 60:
            t_coord_only = min(time_left2 * 0.4, 120.0)
            a_c, s_c = _coordinate_descent_cycle(a0, t_coord_only, rng)
            eval_c = evaluate_sequence(a_c.tolist())
            if np.isfinite(eval_c) and eval_c < best_overall_score:
                best_overall_score = eval_c
                best_overall = a_c.copy()
                print(f"  *** NEW BEST (coord only): {eval_c:.6f} ***")

    # Final polishing from the best solution found
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\nFinal polishing with {remaining:.0f}s remaining...")
        n = len(best_overall)

        # Coordinate descent polish
        t_polish = remaining * 0.5
        a_p, s_p = _coordinate_descent_cycle(best_overall, t_polish, rng)
        eval_p = evaluate_sequence(a_p.tolist())
        print(f"  Polish (coord): {eval_p:.6f}")
        if np.isfinite(eval_p) and eval_p < best_overall_score:
            best_overall_score = eval_p
            best_overall = a_p.copy()
            print(f"  *** NEW BEST: {eval_p:.6f} ***")

        # Subgradient polish
        t_polish2 = deadline - time.monotonic()
        if t_polish2 > 15:
            a_p2, s_p2 = _optimize_subgrad(best_overall, t_polish2, rng)
            eval_p2 = evaluate_sequence(a_p2.tolist())
            print(f"  Polish (subgrad): {eval_p2:.6f}")
            if np.isfinite(eval_p2) and eval_p2 < best_overall_score:
                best_overall_score = eval_p2
                best_overall = a_p2.copy()
                print(f"  *** NEW BEST: {eval_p2:.6f} ***")

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s")
        return [float(x) for x in best_overall]
    else:
        # Fallback: decaying uniform
        n = 2000
        eta = 1.0 + 4.0 / np.sqrt(2.0 * n)
        a = np.exp(-eta * np.arange(n, dtype=np.float64) / n)
        print("Fallback to decaying uniform")
        return a.tolist()
