"""
Optimize nonnegative step-function heights to minimize
    2*n*max(convolve(a,a)) / sum(a)^2
using improved mirror descent with momentum, adaptive temperature,
peak-targeted reduction, and multi-scale exploration.
"""

import time
import numpy as np
import scipy.optimize

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s

def softmax_seq(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    return e / s

def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]

def convolve_and_fft(a, N):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    return c, fa

def grad_from_weights(w, fa, N, n):
    w = np.asarray(w, dtype=np.float64).ravel()
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1: 2 * n - 1][::-1]

def smooth_max_val(c, cmax, tau):
    if c.size == 0 or cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    if tau <= 0.0:
        return cmax
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return float(cmax + tau * np.log(s))

def compute_weights(c, cmax, tau, floor=0.0):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return c
    if cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = ez / s
    if floor > 0.0:
        w = (1.0 - floor) * w + (floor / float(L))
    return w

def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))

def get_given_sequence():
    try:
        seq = globals().get("height_sequence_1", None)
        if seq is None:
            return None
        a = np.asarray(seq, dtype=np.float64).ravel()
        if a.size == 0:
            return None
        a = normalize_positive(a, floor=0.0)
        if not np.all(np.isfinite(a)):
            return None
        return a
    except Exception:
        return None

def resample_seq(a, n):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if old_n == n:
        return a.astype(np.float64, copy=True)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)

def make_sine(n, alpha, mode=1, phase=0.0):
    n = max(4, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    base = 1.0 + float(alpha) * np.sin(2.0 * np.pi * float(mode) * x + float(phase))
    return normalize_positive(base, floor=1e-12)

def build_basis(n):
    n = max(8, int(n))
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    cols = []
    for m in range(1, 8):
        cols.append(np.sin(np.pi * float(m) * x))
    cols.append(x.copy())
    cols.append(x * x - np.mean(x * x))
    cols.append(x * x * x - np.mean(x * x * x))
    t = 2.0 * np.pi * i / float(n)
    for m in range(1, 6):
        cols.append(np.sin(float(m) * t) - np.mean(np.sin(float(m) * t)))
        cols.append(np.cos(float(m) * t) - np.mean(np.cos(float(m) * t)))
    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms <= 1e-14] = 1.0
    B = B / norms
    return B

def theta_value(theta, n, B, tau, N):
    a = softmax_seq(B @ theta)
    c = convolve_seq(a, N)
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    tau_eff = max(float(tau), 1e-12)
    return float(2.0 * n * smooth_max_val(c, cmax, tau_eff))

def theta_grad(theta, n, B, tau, N):
    a = softmax_seq(B @ theta)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    cmax = float(np.max(c))
    D = B.shape[1]
    if cmax <= 0.0 or not np.isfinite(cmax):
        return np.zeros(D, dtype=np.float64)
    tau_eff = max(float(tau), 1e-12)
    z = (c - cmax) / tau_eff
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros(D, dtype=np.float64)
    w = ez / s
    ga = grad_from_weights(w, fa, N, n)
    total = float(np.dot(a, ga))
    aB = a @ B
    agaB = (a * ga) @ B
    gtheta = agaB - total * aB
    return (2.0 * n) * gtheta

def peak_reduction(a, deadline, seed=0):
    """Targeted peak reduction: find the lag with max autocorrelation and reduce it."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = internal_score(a, N)
    rng = np.random.default_rng(int(seed))

    it = 0
    stale = 0
    while time.monotonic() < deadline and stale < 100:
        c = convolve_seq(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        if score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()
            stale = 0
        else:
            stale += 1

        # Find the peak lag
        k_star = int(np.argmax(c))
        if k_star == 0:
            # Can't reduce c_0 without changing the sequence length
            break

        # The peak is c_{k_star} = sum_i a_i * a_{i+k_star}
        # Reduce by decreasing a_i where a_i * a_{i+k_star} is largest
        if k_star < n:
            products = a[:n - k_star] * a[k_star:n]
            # Sort by product (largest first)
            order = np.argsort(products)[::-1]
            n_reduce = max(1, n // 20)
            for idx in order[:n_reduce]:
                i = int(idx)
                # Reduce a[i] and a[i+k_star] proportionally
                reduction = 0.02 * min(a[i], a[i + k_star])
                a[i] = max(0.0, a[i] - reduction)
                a[i + k_star] = max(0.0, a[i + k_star] - reduction)

        a = normalize_positive(a, floor=1e-15)
        it += 1

    return best_a, best_score

    candidates.append(normalize_positive(np.exp(-2.0 * ((x - 0.5) ** 2)), floor=1e-12))

    candidates.append(normalize_positive(1.0 + 0.5 * np.cos(4.0 * np.pi * x), floor=1e-12))

    # Evaluate and pick the best initial
    best_a = None
    best_score = np.inf
    for a0 in candidates:
        sc = internal_score(a0, N)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a0.copy()

    # Phase 1: Low-dim optimization
    t_allow = min(60.0, (deadline - time.monotonic()) * 0.15)
    if t_allow > 10.0:
        a1, sc1 = optimize_lowdim(n, time.monotonic() + t_allow, seed=seed, initial_a=best_a, max_starts=3)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()

    # Phase 2: Mirror descent (multiple rounds with different parameters)
    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        # Round 1: standard mirror descent
        t_md = min(remaining * 0.4, 180.0)
        a2, sc2 = mirror_descent_improved(best_a, time.monotonic() + t_md, seed=seed + 1)
        if a2 is not None and np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
            print("  mirror1 n={} score={:.9f}".format(n, best_score), flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        # Round 2: peak reduction
        t_pr = min(remaining * 0.15, 60.0)
        a3, sc3 = peak_reduction(best_a, time.monotonic() + t_pr, seed=seed + 2)
        if a3 is not None and np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
            print("  peak_red n={} score={:.9f}".format(n, best_score), flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        # Round 3: mirror descent again from improved point
        t_md2 = min(remaining * 0.35, 150.0)
        a4, sc4 = mirror_descent_improved(best_a, time.monotonic() + t_md2, seed=seed + 3)
        if a4 is not None and np.isfinite(sc4) and sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()
            print("  mirror2 n={} score={:.9f}".format(n, best_score), flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        # Final polish
        t_fp = min(remaining * 0.3, 60.0)
        a5, sc5 = local_polish(best_a, time.monotonic() + t_fp, seed=seed + 4)
        if a5 is not None and np.isfinite(sc5) and sc5 < best_score:
            best_score = sc5
            best_a = a5.copy()
            print("  polish n={} score={:.9f}".format(n, best_score), flush=True)

    return best_a, best_score

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()
    total_time = max(120.0, float(budget_s) - 30.0)
    grand_deadline = start + total_time

    print("Starting optimization with budget {:.0f}s".format(total_time), flush=True)

    # Get the previous best sequence
    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        print("Given sequence: n={}, score={:.9f}".format(given.size, sc), flush=True)

    best_a = None
    best_score = np.inf

    # Try multiple n values, prioritizing the one that worked before
    n_plans = [2048, 4096, 1024, 3072, 5120]

    # Allocate time: give more time to n=2048 (where we got the best result)
    time_alloc = {2048: 0.35, 4096: 0.25, 1024: 0.15, 3072: 0.12, 5120: 0.13}

    for n in n_plans:
        if time.monotonic() > grand_deadline - 30.0:
            break
        remain = grand_deadline - time.monotonic() - 30.0
        if remain < 60.0:
            break
        alloc = time_alloc.get(n, 0.1) * total_time
        alloc = min(alloc, remain)
        if alloc < 40.0:
            continue

        print("Optimizing n={} with {:.0f}s".format(n, alloc), flush=True)
        n_deadline = time.monotonic() + alloc
        a, sc = optimize_for_n(n, n_deadline, seed=seed + n, initial_a=given)
        if a is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print("  NEW BEST n={} score={:.9f}".format(n, best_score), flush=True)

    # If we still have time, do additional mirror descent rounds on the best
    remain = grand_deadline - time.monotonic() - 20.0
    if best_a is not None and remain > 60.0:
        print("Extra optimization, {:.0f}s remaining".format(remain), flush=True)
        n_best = best_a.size
        alloc = min(remain * 0.4, 200.0)
        a, sc = mirror_descent_improved(best_a, time.monotonic() + alloc, seed=seed + 9999)
        if a is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print("  extra mirror score={:.9f}".format(best_score), flush=True)

    # Final polish
    remain = grand_deadline - time.monotonic() - 10.0
    if best_a is not None and remain > 20.0:
        a, sc = local_polish(best_a, time.monotonic() + remain, seed=seed + 77777)
        if a is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print("  final polish score={:.9f}".format(best_score), flush=True)

    if best_a is None:
        best_a = make_sine(2048, 0.5)
        best_score = internal_score(best_a)

    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]

    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")

    if not np.isfinite(final_eval):
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print("FINAL: score={:.12f}, n={}".format(final_eval, len(out_list)), flush=True)
    return out_list
