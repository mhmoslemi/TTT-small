    c = np.convolve(a, a)  # length 2n-1
    # Compute softmax weights
    eps_c = eps * c
    eps_c -= np.max(eps_c)  # for numerical stability
    p = np.exp(eps_c)
    p /= np.sum(p)
    
    # Compute gradient
    grad = np.zeros(n)
    # For each i, grad[i] = sum_j p[j] * (a[j-i] + a[i-j]) (with bounds)

    # corr2[i] = sum_j p[j] * a[i-j] = cross-correlation of p and reversed_a at lag i

def _objective_and_grad(a, eps):
    """Objective: 2*n*logsumexp(eps*conv(a))/(sum(a))^2 and its gradient."""
    n = len(a)
    c = _fft_convolve(a)
    eps_c = eps * c
    eps_c -= np.max(eps_c)
    exp_c = np.exp(eps_c)
    sum_exp = np.sum(exp_c)
    f = (1.0 / eps) * np.log(sum_exp)
    S = np.sum(a)
    obj = 2.0 * n * f / (S * S)
    
    p = exp_c / sum_exp
    # grad of f w.r.t. a
    grad_f = np.zeros(n)
    for i in range(n):
        j_start = i
        j_end = min(i + n - 1, 2 * n - 2)
        if j_start <= j_end:

        if 2 * i < 2 * n - 1:
            grad_f[i] -= a[i] * p[2 * i]
    
    # grad of obj = 2n/S * (grad_f - 2f/S)
    grad_obj = (2.0 * n / S) * (grad_f - 2.0 * f / S)
    return obj, grad_obj

    n = len(f)
    # Maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for all k
    # (f*g)[k] = sum_j f[j] * g[k-j] where k-j in [0, n-1]
    # For constraint k: sum_j f[j] * g[k-j] <= M
    # The coefficient of g[i] in constraint k is f[k-i] (if 0 <= k-i < n)
    
    A_ub = np.zeros((2*n-1, n))

        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = f[j]
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    c = -np.ones(n)
    
    # A_ub[k, i] = f[k-i] if 0 <= k-i < n else 0
    # This is a Toeplitz-like matrix
    ks = np.arange(2*n-1)[:, None]
    is_ = np.arange(n)[None, :]
    diff = ks - is_  # shape (2n-1, n), diff[k,i] = k-i
    mask = (diff >= 0) & (diff < n)
    A_ub = np.where(mask, f[diff.clip(0, n-1)], 0.0)
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

from scipy.optimize import minimize, linprog

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B with log-sum-exp smoothing (annealed eps continuation method)
2. LP-based conditional gradient (Frank-Wolfe style) direction finding

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    ca = np.fft.irfft(fa * fa, size)
    return ca[:2 * n - 1]


def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def _lse_obj_grad(a, eps):
    """Compute smoothed objective 2n*logsumexp(eps*conv(a))/(sum(a))^2 and its gradient."""
    n = len(a)
    c = _fft_convolve(a)
    eps_c = eps * c

    exp_c = np.exp(eps_c - mx)
    sum_exp = np.sum(exp_c)
    f_val = (1.0 / eps) * (mx + np.log(sum_exp))
    S = np.sum(a)
    obj = 2.0 * n * f_val / (S * S)
    p = exp_c / sum_exp
    # grad_f[i] = 2*sum_j p[j]*a[j-i] - a[i]*p[2i]

        if i <= j_end:

        if 2 * i < 2 * n - 1:
            grad_f[i] -= a[i] * p[2 * i]
    grad_obj = (2.0 * n / S) * (grad_f - 2.0 * f_val / S)
    return obj, grad_obj


def _lbfgs_optimize(a0, eps, maxiter=400):
    """Run L-BFGS-B on the smoothed objective with given eps."""
    a0 = np.maximum(a0, 0.0)
    S = np.sum(a0)
    if S < 1e-15:
        return a0, np.inf
    a0 = a0 / S

    n = len(a0)
    bounds = [(0.0, None)] * n

    result = minimize(_lse_obj_grad, a0, args=(eps,), method='L-BFGS-B',
                      jac=True, bounds=bounds,
                      options={'maxiter': maxiter, 'maxfun': maxiter * 8,
                               'maxls': 50, 'ftol': 1e-14, 'gtol': 1e-10})

    a_opt = np.maximum(result.x, 0.0)
    S = np.sum(a_opt)
    if S > 1e-15:
        a_opt = a_opt / S
    else:
        a_opt = a0.copy()
    sc = _score(a_opt)
    return a_opt, sc


def _lbfgs_anneal(a0, eps_list, maxiter_per=300, time_limit=None):
    """Run L-BFGS-B with annealed eps (continuation method)."""
    a_cur = a0.copy()
    sc_cur = _score(a_cur)
    start = time.monotonic()
    for eps in eps_list:
        if time_limit is not None and time.monotonic() - start > time_limit:
            break
        a_cur, sc = _lbfgs_optimize(a_cur, eps, maxiter=maxiter_per)
        if np.isfinite(sc) and sc < sc_cur:
            sc_cur = sc
    return a_cur, sc_cur


def _lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    n = len(f)
    c_obj = np.ones(n)  # minimize -sum(g) => c = -1, but linprog minimizes c'x
    c_obj = -np.ones(n)
    # Build A_ub: A_ub[k,i] = f[k-i] if 0<=k-i<n else 0
    ks = np.arange(2 * n - 1)[:, None].astype(np.int64)
    is_ = np.arange(n)[None, :].astype(np.int64)
    diff = ks - is_
    mask = (diff >= 0) & (diff < n)
    A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
    A_ub[mask] = f[diff[mask]]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def _lp_optimize(a0, time_budget, max_iters=300):
    """Frank-Wolfe / LP-based conditional gradient optimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return a0, np.inf
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    deadline = time.monotonic() + time_budget
    it = 0
    no_improve = 0
    t_step = 2.0 / (2.0 + it + 1)  # FW step size
    while time.monotonic() < deadline and it < max_iters:
        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _lp_direction(a, M)
        if g is None:
            break
        Sg = np.sum(g)
        if Sg < 1e-15:
            break
        g = g / Sg
        # FW step: new point = (1-t)*a + t*g
        t = 2.0 / (it + 2.0)
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        St = np.sum(a_try)
        if St < 1e-15:
            break
        a_try = a_try / St
        sc_try = _score(a_try)
        if sc_try < best_score:
            best_score = sc_try
            best_a = a_try.copy()
            no_improve = 0
            if it % 20 == 0:
                print(f"    [lp] it={it}, t={t:.4f}, score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 50:
                break
        a = a_try
        it += 1
    return best_a, best_score


def _subgrad_optimize(a0, time_budget, rng):
    """Multiplicative subgradient descent with momentum and restarts."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return a0, np.inf
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    eta = 0.12
    mu = 0.7
    momentum = np.zeros(n)
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 600:
            a = best_a * np.exp(rng.normal(0, 0.06, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.12
            momentum = np.zeros(n)
            it += 1
            continue
        g = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            indices = k_max - j_range
            g[lo:hi + 1] = 2.0 * a[indices]
        g_max = g.max()
        if g_max < 1e-15:
            break
        g = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        a = a / a.sum()
        if no_improve > 300:
            eta *= 0.97
        elif no_improve < 30:
            eta = min(eta * 1.005, 0.35)
        it += 1
    return best_a, best_score


def propose_candidate(seed=42, budget_s=2350, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
    Uses L-BFGS-B with log-sum-exp smoothing, LP-based conditional gradient,
    and subgradient polishing. Multiple multi-harmonic sinusoidal initializations.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30.0

    best_overall = None
    best_score = np.inf

    # Get provided sequence
    provided = None
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            provided = hs / hs.sum()
    except Exception:
        pass

    # Generate initial candidates
    candidates = []
    n_base = 2000

    if provided is not None:
        candidates.append(("provided", provided.copy()))

    x = np.arange(n_base, dtype=np.float64)
    # Single harmonic variations
    for amp in [0.3, 0.35, 0.4, 0.45, 0.5]:
        for phase in [np.pi / 6, np.pi / 4, np.pi / 3, np.pi / 2]:
            a = 1.0 + amp * np.sin(2 * np.pi * x / n_base + phase)
            a = np.maximum(a, 0.01)
            a = a / a.sum()
            candidates.append((f"sin_{amp:.2f}_{phase:.2f}", a.copy()))

    # Multi-harmonic
    a = 1.0 + 0.3 * np.sin(2 * np.pi * x / n_base + np.pi / 4) + 0.15 * np.sin(4 * np.pi * x / n_base + 0.5)
    a = np.maximum(a, 0.01)
    candidates.append(("multi_a", (a / a.sum()).copy()))

    a = 1.0 + 0.35 * np.sin(2 * np.pi * x / n_base + np.pi / 4) + 0.1 * np.cos(4 * np.pi * x / n_base) + 0.05 * np.sin(6 * np.pi * x / n_base + 1.0)
    a = np.maximum(a, 0.01)
    candidates.append(("multi_b", (a / a.sum()).copy()))

    a = 1.0 + 0.25 * np.sin(2 * np.pi * x / n_base) + 0.2 * np.sin(2 * np.pi * 2 * x / n_base + np.pi / 3) + 0.1 * np.sin(2 * np.pi * 3 * x / n_base + 2.0)
    a = np.maximum(a, 0.01)
    candidates.append(("multi_c", (a / a.sum()).copy()))

    # Decay
    a = np.exp(-0.8 * x / n_base) + 0.1
    candidates.append(("decay", (a / a.sum()).copy()))

    # Quadratic modulation
    a = 1.0 + 0.3 * np.cos(2 * np.pi * (x / n_base) ** 2)
    a = np.maximum(a, 0.01)
    candidates.append(("chirp", (a / a.sum()).copy()))

    # n=3000 variants
    x3 = np.arange(3000, dtype=np.float64)
    a = 1.0 + 0.4 * np.sin(2 * np.pi * x3 / 3000 + np.pi / 4)
    a = np.maximum(a, 0.01)
    candidates.append(("sin_3000", (a / a.sum()).copy()))

    # Score and sort
    scored = []
    for name, a in candidates:
        sc = _score(a)
        if np.isfinite(sc):
            scored.append((sc, name, a))
    scored.sort(key=lambda x: x[0])

    print(f"Candidates: {len(scored)}")
    for i, (sc, name, _) in enumerate(scored[:8]):
        print(f"  {i + 1}. {name}: {sc:.6f}")

    # Ensure provided is in top candidates
    # Phase allocation
    t_lenc = budget_s * 0.50
    t_lp = budget_s * 0.25
    t_sub = budget_s * 0.15
    t_final = budget_s * 0.10

    # Phase 1: L-BFGS-B annealing on top candidates
    n_top = min(5, len(scored))
    t_per_lenc = t_lenc / n_top
    eps_schedule = [2.0, 1.0, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015]

    print(f"\n=== Phase 1: L-BFGS-B annealing ({t_lencs:.0f}s total) ===" if False else "\n=== Phase 1: L-BFGS-B ===")

    for idx in range(n_top):
        if time.monotonic() > deadline - t_lp - t_sub:
            break
        _, name, a0 = scored[idx]
        t_alloc = min(t_per_lenc, max(20, deadline - time.monotonic() - t_lp - t_sub))
        print(f"\n  L-BFGS-B on '{name}' (n={len(a0)}, init={scored[idx][0]:.6f}, budget={t_alloc:.0f}s)")
        a_opt, sc_opt = _lbfgs_anneal(a0, eps_schedule, maxiter_per=300, time_limit=t_alloc)
        print(f"  Result: {sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_overall = a_opt.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 2: LP-based optimization
    if best_overall is not None and time.monotonic() < deadline - t_sub:
        t_lp_actual = min(t_lp, deadline - time.monotonic() - t_sub)
        print(f"\n=== Phase 2: LP-based ({t_lp_actual:.0f}s) ===")
        a_lp, sc_lp = _lp_optimize(best_overall, t_lp_actual, max_iters=500)
        print(f"  LP result: {sc_lp:.6f}")
        if sc_lp < best_score:
            best_score = sc_lp
            best_overall = a_lp.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 3: More L-BFGS-B on LP result if improved
    if best_overall is not None and time.monotonic() < deadline - t_sub:
        t_extra = min(120, deadline - time.monotonic() - t_sub)
        if t_extra > 30:
            print(f"\n=== Phase 3: L-BFGS-B refinement ({t_extra:.0f}s) ===")
            eps_fine = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005]
            a_fin, sc_fin = _lbfgs_anneal(best_overall, eps_fine, maxiter_per=400, time_limit=t_extra)
            print(f"  Refined: {sc_fin:.6f}")
            if sc_fin < best_score:
                best_score = sc_fin
                best_overall = a_fin.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 4: Subgradient polish
    if best_overall is not None and time.monotonic() < deadline - 10:
        t_sub_actual = min(t_sub, deadline - time.monotonic() - 10)
        print(f"\n=== Phase 4: Subgradient polish ({t_sub_actual:.0f}s) ===")
        a_sub, sc_sub = _subgrad_optimize(best_overall, t_sub_actual, rng)
        print(f"  Subgrad result: {sc_sub:.6f}")
        if sc_sub < best_score:
            best_score = sc_sub
            best_overall = a_sub.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 5: Final L-BFGS-B polish with very small eps
    if best_overall is not None and time.monotonic() < deadline - 10:
        t_final_actual = min(t_final, deadline - time.monotonic() - 10)
        if t_final_actual > 20:
            print(f"\n=== Phase 5: Final polish ({t_final_actual:.0f}s) ===")
            eps_very_fine = [0.02, 0.01, 0.007, 0.005, 0.003]
            a_pp, sc_pp = _lbfgs_anneal(best_overall, eps_very_fine, maxiter_per=500, time_limit=t_final_actual)
            print(f"  Final polished: {sc_pp:.6f}")
            if sc_pp < best_score:
                best_score = sc_pp
                best_overall = a_pp.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s ===")
        if not np.isfinite(final_score) or final_score >= 2.0:
            # Fallback to a known-good sequence
            a = 1.0 + 0.4 * np.sin(2 * np.pi * np.arange(2000) / 2000 + np.pi / 4)
            a = np.maximum(a, 0.01)
            return a.tolist()
        return [float(x) for x in best_overall]
    else:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * np.arange(2000) / 2000 + np.pi / 4)
        a = np.maximum(a, 0.01)
        print("Fallback to sinusoidal sequence")
        return a.tolist()
