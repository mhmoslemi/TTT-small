def _priority_order(a, b, n, rng):
    """Order coordinates by their contribution to top lags (high first)."""
    top_k = np.argsort(b)[-10:][::-1]  # top 10 lags

    for k in top_k:

        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                contrib[i] += a[i] * a[j]
    # Mix priority with randomness
    order = np.arange(n)
    rng.shuffle(order)
    # Sort by -contribution (high first) but with some randomness
    ranks = np.argsort(-contrib)
    # Blend: 70% contribution order, 30% random

                    a, b, sc, _ = _coord_update(a, b, sc, i, g_small, 5, N)
                s = a.sum()
                if s > 1e-15:
                    a /= s
                    b = np.convolve(a, a)
                    sc = _score1(a, b)
                if sc < best_sc - 1e-10:
                    best_sc = sc
                    best_a = a.copy()
                    best_b = b.copy()

        elif stagnant >= 8:
            # Revert to best and try larger perturbation
            a = best_a.copy()
            b = best_b.copy()
            sc = best_sc
            a = _perturb_gaussian(a, rng, 0.05)
            b = np.convolve(a, a)
            sc = _score1(a, b)
            pert_count += 1
            stagnant = 0

        if pass_count % 20 == 0 and pass_count > 10:
            print(f"  Phase2 pass {pass_count}: sc={sc:.6f}, best={best_sc:.6f}")
        pass_count += 1

    print(f"Phase 2 done. Best: {best_sc:.6f}")

    # Phase 3: Fine polishing (remaining time, ~300s+)
    p3_deadline = deadline - 10.0
    a = best_a.copy()
    b = best_b.copy()
    sc = best_sc
    stagnant = 0

    # Focus on top-contributing coordinates
    top_lags = np.argsort(b)[-15:]
    top_coords = set()
    for k in top_lags:
        lo = max(0, k - N + 1)
        hi = min(N - 1, k)
        top_coords.update(range(lo, hi + 1))
    top_coords = list(top_coords)[:1000]  # cap at 1000

    while time.monotonic() < p3_deadline:
        # Cycle through top coordinates with fine grid
        if len(top_coords) > 0:
            order = np.array(top_coords)
            rng.shuffle(order)
        else:
            order = rng.permutation(N)

        improved = False
        for i in order:
            if time.monotonic() > p3_deadline:
                break
            cv = float(a[i])
            # Very fine grid around current value
            scale = max(cv, 1.0 / N)
            lo = max(0.0, cv - 2.0 * scale)
            hi = cv + 3.0 * scale
            g_fine = np.concatenate(([0.0], np.linspace(lo, hi, 40),
                                     np.logspace(np.log10(max(lo, 1e-12)), np.log10(max(hi, lo + 1e-12)), 20)))
            g_fine = np.unique(g_fine[g_fine >= 0.0])
            if len(g_fine) < 5:
                g_fine = _make_grid(cv, N, coarse_pts=30)
            a, b, sc, imp = _coord_update(a, b, sc, int(i), g_fine, 15, N)
            if imp:
                improved = True

        s = a.sum()
        if s > 1e-15:
            a /= s
            b = np.convolve(a, a)
            sc = _score1(a, b)

        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a.copy()
            best_b = b.copy()
            stagnant = 0
        else:
            stagnant += 1
        if not improved:
            stagnant += 1
            if stagnant >= 3:
                a = _perturb_gaussian(best_a, rng, 0.005)
                b = np.convolve(a, a)
                sc = _score1(a, b)
                stagnant = 0
            elif stagnant >= 6:
                break

        pass_count += 1
        if pass_count % 10 == 0:
            print(f"  Phase3 pass {pass_count}: sc={sc:.6f}, best={best_sc:.6f}")

    print(f"Phase 3 done. Best: {best_sc:.6f}")

    # Try a few random restarts from best with small mutations
    if time.monotonic() < deadline - 60.0:
        for attempt in range(5):
            if time.monotonic() > deadline - 60.0:
                break
            a_try = _perturb_gaussian(best_a, rng, 0.01)
            b_try = np.convolve(a_try, a_try)
            sc_try = _score1(a_try, b_try)
            if sc_try < best_sc:
                best_a = a_try.copy()
                best_b = b_try.copy()
                best_sc = sc_try
                print(f"  Restart {attempt}: improved to {best_sc:.6f}")
            # Quick coordinate descent from here
            sub_dl = min(deadline - 30.0, time.monotonic() + 30.0)
            a_cur = a_try.copy()
            b_cur = np.convolve(a_cur, a_cur)
            sc_cur = sc_try
            for _ in range(3):
                if time.monotonic() > sub_dl:
                    break
                for i in rng.permutation(min(N, 300)):
                    if time.monotonic() > sub_dl:
                        break
                    cv = float(a_cur[i])
                    g = _make_grid(cv, N, coarse_pts=20)
                    a_cur, b_cur, sc_cur, _ = _coord_update(a_cur, b_cur, sc_cur, int(i), g, 8, N)
                s = a_cur.sum()
                if s > 1e-15:
                    a_cur /= s
                    b_cur = np.convolve(a_cur, a_cur)
                    sc_cur = _score1(a_cur, b_cur)
                if sc_cur < best_sc - 1e-10:
                    best_sc = sc_cur
                    best_a = a_cur.copy()
                    best_b = b_cur.copy()
                    print(f"  Post-restart best: {best_sc:.6f}")

    # Also try expanding a short optimized prototype
    if time.monotonic() < deadline - 30.0:
        for proto_len in [16, 32, 64]:
            if time.monotonic() > deadline - 30.0:
                break
            for trial in range(3):
                if time.monotonic() > deadline - 30.0:
                    break
                p = rng.dirichlet(np.full(proto_len, 0.5))
                pc = np.convolve(p, p)
                ps = _score1(p, pc)
                # Quick coordinate descent on prototype
                for _ in range(5):
                    for i in range(proto_len):
                        cv = float(p[i])
                        g = _make_grid(cv, proto_len, coarse_pts=20)
                        p, pc, ps, _ = _coord_update(p, pc, ps, i, g, 6, proto_len)
                # Expand to N
                base = N // proto_len
                rem = N % proto_len
                lengths = np.full(proto_len, base)
                lengths[:rem] += 1
                full = np.zeros(N)
                idx = 0
                for i in range(proto_len):
                    L = int(lengths[i])
                    full[idx:idx + L] = p[i] / L
                    idx += L
                full = _norm(full)
                fb = np.convolve(full, full)
                fs = _score1(full, fb)
                if fs < best_sc - 1e-6:
                    best_sc = fs
                    best_a = full.copy()
                    best_b = fb.copy()
                    print(f"  Proto {proto_len} trial {trial}: {best_sc:.6f}")

    print(f"\nFinal best score: {best_sc:.6f} (target: {TARGET})")

    # Validate with external evaluator
    final_sc = _eval_ext(best_a)
    if np.isfinite(final_sc):
        print(f"External eval: {final_sc:.6f}")
        # Also check a few alternatives
        alt_candidates = [best_a]
        try:
            alt_candidates.append(_clean(np.asarray(height_sequence_1)))
        except Exception:
            pass
        best_final = best_a
        best_final_sc = final_sc
        for cand in alt_candidates:
            cand = _norm(_clean(cand))
            csc = _eval_ext(cand)
            if np.isfinite(csc) and csc < best_final_sc:
                best_final = cand
                best_final_sc = csc

        out = _norm(best_final)
        return [float(x) for x in out]
    else:
        return [float(x) for x in best_a]
