    # Difference: x^2-old^2 - 2*(x-old)*a[mid] = (x-old)(x+old - 2*a[mid]) = (x-old)(x-old) = (x-old)^2
    # Since a[mid] = a[i] = old!
    # So: new_c_slice[mid] = c_slice[mid] + 2*(x-old)*old + (x-old)^2 = c_slice[mid] + x^2 - old^2. ✓
    
    affected = c_slice[None, :] + 2.0 * (xs[:, None] - old) * a[None, :]
    affected[:, mid] += (xs - old) ** 2
    peak_aff = np.max(affected, axis=1)
    peak = np.maximum(c_out, peak_aff)
    S = D + xs
    valid = S > 1e-12
    scores = np.full(len(xs), np.inf)
    scores[valid] = 2.0 * n * peak[valid] / (S[valid] ** 2)
    idx = int(np.argmin(scores))
    best_x = float(xs[idx])
    best_sc = float(scores[idx])
    
    # Current score at x=old
    cur_sc = 2.0 * n * float(np.max(c))  # since sum=1
    
    if best_x < old - 1e-12 or best_x > old + 1e-12:
        # Update
        a_new = a.copy()
        a_new[i] = best_x
        S = D + best_x
        if S < 1e-12:
            return a, c, False
        a_new /= S
        c_new = np.convolve(a_new, a_new)
        return a_new, c_new, True
    return a, c, False


def _coord_descent_round(a, c, score, deadline, rng, grid, max_passes=200):
    """Full coordinate descent until deadline or convergence."""
    n = len(a)
    a = a.copy()
    c = c.copy()
    score = float(score)
    best_a = a.copy()
    best_score = score
    stagnant = 0
    
    for p in range(int(max_passes)):
        if time.monotonic() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            a, c, imp = _coord_update(a, c, int(i), grid)
            if imp:
                improved = True
                sc = _score_from_conv(a, c)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        if not improved:
            stagnant += 1
            if stagnant >= 3:
                break
        else:
            stagnant = 0
    
    # Residual
    s = float(np.sum(a))
    if s > 1e-12:
        a = a / s
    c = np.convolve(a, a)
    score = _score_from_conv(a, c)
    if score < best_score:
        best_a = a.copy()
        best_score = score
    
    return best_a, np.convolve(best_a, best_a), best_score


def _lp_direction(a, M, deadline, max_constr=None):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns the direction g (normalized) or None.
    """
    n = len(a)
    if n < 4:
        return None
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except ImportError:
        return None
    
    # Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n, else 0
    # Shape: (2n-1) x n
    if max_constr is not None and 2 * n - 1 > max_constr:
        # Subsample constraints: use every other, or top peaks
        conv = np.convolve(a, a)
        k_order = np.argsort(conv)[::-1]
        k_order = k_order[:max_constr]
        k_sel = np.sort(k_order)
    else:
        k_sel = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for k in k_sel:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(int(k))
            cols.append(int(j))
            vals.append(float(a[k - j]))
    
    # Remap rows to 0..len(k_sel)-1
    k_map = {int(k): ii for ii, k in enumerate(k_sel)}
    rows = [k_map[r] for r in rows]
    
    A_sp = sparse.csr_matrix((vals, (rows, cols)), shape=(len(k_sel), n))
    
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(len(k_sel), M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c=c_obj, A_ub=A_sp, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(1.0, deadline - time.monotonic() - 5.0)})
    except Exception:
        return None
    
    if res is None or not getattr(res, 'success', False):
        return None
    g = np.asarray(res.x, dtype=np.float64)
    np.clip(g, 0.0, None, out=g)
    gs = float(np.sum(g))
    if gs < 1e-10:
        return None
    return g / gs


def _lp_polish(a, c, score, deadline, max_lp_iters=10):
    """Iterative LP polish: solve LP, line search, repeat."""
    n = len(a)
    a = a.copy()
    c = c.copy()
    score = float(score)
    best_a = a.copy()
    best_score = score
    
    for it in range(int(max_lp_iters)):
        if time.monotonic() > deadline - 10.0:
            break
        M = float(np.max(c))
        if M < 1e-15:
            break
        g = _lp_direction(a, M, deadline)
        if g is None:
            break
        # Line search: a_new = (1-t)*a + t*g, find best t
        cross = np.convolve(a, g)
        cg = np.convolve(g, g)
        best_t = 0.0
        best_val = score
        for t in np.linspace(0.0, 1.0, 41):
            t = float(t)
            ct = (1-t)**2 * c + 2*t*(1-t)*cross + t*t*cg
            mx = float(np.max(ct))
            val = 2.0 * n * mx
            if val < best_val:
                best_val = val
                best_t = t
        if best_t < 1e-6 or best_val >= score - 1e-12:
            break
        # Refine
        lo = max(0.0, best_t - 0.2)
        hi = min(1.0, best_t + 0.2)
        for t in np.linspace(lo, hi, 33):
            t = float(t)
            ct = (1-t)**2 * c + 2*t*(1-t)*cross + t*t*cg
            mx = float(np.max(ct))
            val = 2.0 * n * mx
            if val < best_val:
                best_val = val
                best_t = t
        if best_val >= score - 1e-12:
            break
        a = (1.0 - best_t) * a + best_t * g
        a = _norm(a)
        c = np.convolve(a, a)
        score = _score_from_conv(a, c)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, np.convolve(best_a, best_a), best_score


def _block_move(a, rng, block_size=32, strength=0.3):
    """Redistribute mass within a random block."""
    n = len(a)
    if n < block_size:
        block_size = max(4, n // 2)
    start = int(rng.integers(0, n - block_size + 1))
    end = start + block_size
    block = a[start:end].copy()
    # Add noise and re-normalize the block
    noise = np.exp(rng.normal(0.0, strength, block_size))
    block_new = block * noise
    total = float(np.sum(block_new))
    orig_total = float(np.sum(block))
    if total > 1e-15 and orig_total > 1e-15:
        block_new *= orig_total / total
    a[start:end] = block_new
    return _norm(a)


def _make_initial(n, strategy, rng):
    n = int(n)
    if strategy == 0:
        return np.ones(n, dtype=np.float64) / n
    elif strategy == 1:
        return _norm(rng.dirichlet(np.ones(n)))
    elif strategy == 2:
        return _norm(rng.dirichlet(np.full(n, 0.3)))
    elif strategy == 3:
        return _norm(np.exp(rng.normal(0, 1.0, n)))
    elif strategy == 4:
        # Two-level: first half high, second half low
        p = np.zeros(n)
        m = n // 2
        p[:m] = 1.0
        p[m:] = 0.1
        return _norm(p)
    elif strategy == 5:
        # Alternating
        p = np.zeros(n)
        p[0::2] = 1.0
        p[1::2] = 0.4
        return _norm(p)
    elif strategy == 6:
        # Cosine bell
        x = np.linspace(0, np.pi, n)
        return _norm(np.abs(np.cos(x / 2)))
    elif strategy == 7:
        # Gaussian
        x = np.linspace(-2, 2, n)
        return _norm(np.exp(-x**2))
    elif strategy == 8:
        # Random binary smoothed
        p = (rng.random(n) > 0.5).astype(np.float64)
        # Smooth
        kernel = np.ones(5) / 5
        p = np.convolve(p, kernel, mode='same')
        return _norm(p)
    elif strategy == 9:
        # Multi-scale: combine several frequencies
        x = np.arange(n)
        p = 1.0 + 0.3 * np.sin(2 * np.pi * x / n * 3) + 0.2 * np.sin(2 * np.pi * x / n * 7)
        p = np.abs(p)
        return _norm(p)
    elif strategy == 10:
        # Anti-peak: zeros at regular intervals
        p = np.ones(n)
        p[::3] = 0.2
        return _norm(p + 0.01 * rng.random(n))
    else:
        return _norm(rng.random(n))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Return the best non-negative coefficient sequence found within budget_s seconds."""
    budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + max(0.0, budget - 5.0)
    rng = np.random.default_rng(int(seed))
    
    N = 3809  # target length
    
    # Load existing best if available
    best_a = None
    best_score = np.inf
    try:
        hseq = np.asarray(height_sequence_1, dtype=np.float64)
        hseq = np.clip(hseq, 0, None)
        if hseq.size > 0:
            best_a = _norm(hseq)
            best_score = _score(best_a)
            print(f"Loaded height_sequence_1: len={best_a.size}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_a is None or best_a.size != N:
        # Create initial
        a0 = _make_initial(N, 4, rng)
        c0 = np.convolve(a0, a0)
        s0 = _score_from_conv(a0, c0)
        print(f"Initial two-level: score={s0:.6f}")
        best_a = a0
        best_score = s0
    
    # Also try a few other initializations at full length
    t0 = time.monotonic()
    for strat in [0, 1, 5, 7, 9]:
        if time.monotonic() > deadline - 600:
            break
        a_init = _make_initial(N, strat, rng)
        c_init = np.convolve(a_init, a_init)
        s_init = _score_from_conv(a_init, c_init)
        print(f"  Strategy {strat}: score={s_init:.6f}")
        if s_init < best_score:
            best_score = s_init
            best_a = a_init.copy()
    
    print(f"After initialization: best_score={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")
    
    # Main optimization loop
    grid = np.concatenate(([0.0], np.logspace(-6, 1.5, 25)))
    
    phases = [
        # (mirror_lr, cd_passes, lp_iters, block_prob, label)
        (0.02, 5, 2, 0.1, "coarse"),
        (0.01, 10, 3, 0.05, "medium"),
        (0.005, 20, 2, 0.02, "fine"),
        (0.002, 30, 3, 0.01, "ultra-fine"),
    ]
    
    current_a = best_a.copy()
    current_c = np.convolve(current_a, current_a)
    current_score = best_score
    
    for phase_idx, (mlr, cd_passes, lp_iters, block_prob, label) in enumerate(phases):
        if time.monotonic() > deadline - 120:
            break
        print(f"\nPhase '{label}' (mirror_lr={mlr}, cd_passes={cd_passes}, lp_iters={lp_iters})")
        phase_start = time.monotonic()
        
        iterations = 0
        while time.monotonic() < deadline - 120:
            it_deadline = min(deadline - 120, time.monotonic() + 60.0)
            iterations += 1
            
            # 1. Mirror descent step
            n_mirror = 3
            for _ in range(n_mirror):
                current_a = _mirror_step(current_a, current_c, mlr, tau=0.005)
                current_c = np.convolve(current_a, current_a)
            
            sc_after_mirror = _score_from_conv(current_a, current_c)
            
            # 2. Coordinate descent
            cd_deadline = min(it_deadline, time.monotonic() + max(5.0, 40.0 / (1 + phase_idx)))
            current_a, current_c, sc_cd = _coord_descent_round(
                current_a, current_c, sc_after_mirror, cd_deadline, rng, grid,
                max_passes=cd_passes
            )
            
            # 3. Block move (probabilistic)
            if rng.random() < block_prob:
                a_block = _block_move(current_a, rng, block_size=64, strength=0.4)
                c_block = np.convolve(a_block, a_block)
                sc_block = _score_from_conv(a_block, c_block)
                if sc_block < sc_cd:
                    current_a = a_block
                    current_c = c_block
                    sc_cd = sc_block
            
            # 4. LP polish (less frequent)
            if iterations % 3 == 0 and time.monotonic() < deadline - 120:
                lp_deadline = min(it_deadline, time.monotonic() + 45.0)
                current_a, current_c, sc_lp = _lp_polish(
                    current_a, current_c, sc_cd, lp_deadline, max_lp_iters=lp_iters
                )
                sc_cd = sc_lp
            
            sc_final = _score_from_conv(current_a, current_c)
            if sc_final < best_score:
                best_score = sc_final
                best_a = current_a.copy()
                print(f"  [{label}] iter {iterations}: NEW BEST = {best_score:.6f} (elapsed={time.monotonic()-t0:.0f}s)")
            elif iterations % 5 == 1:
                print(f"  [{label}] iter {iterations}: score={sc_final:.6f} (elapsed={time.monotonic()-t0:.0f}s)")
            
            if sc_final > best_score * 1.05:
                # Reset to best and perturb
                current_a = best_a * np.exp(rng.normal(0, 0.1, best_a.size))
                current_a = _norm(current_a)
                current_c = np.convolve(current_a, current_a)
                current_score = _score_from_conv(current_a, current_c)
        
        print(f"Phase '{label}' done in {time.monotonic()-phase_start:.1f}s, best={best_score:.6f}")
    
    # Final: verify with evaluate_sequence
    final_seq = [float(x) for x in best_a]
    try:
        final_sc = evaluate_sequence(final_seq)
        print(f"\nFinal evaluate_sequence score: {final_sc:.6f}")
    except Exception as e:
        print(f"evaluate_sequence failed: {e}")
        final_sc = best_score
    
    # If the full-length sequence doesn't evaluate well, try the best_a directly
    if not np.isfinite(final_sc):
        final_seq = [float(max(0, min(1000, x))) for x in best_a]
    
    print(f"Returning sequence of length {len(final_seq)}, expected score ~{best_score:.6f}")
    return final_seq
