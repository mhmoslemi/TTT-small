def _next_pow2(n):
    if n <= 1:

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

        g[lo:hi + 1] = 2.0 * a[k_max - j_range]

    cols = []

    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

            best_score = score
            best_a = a.copy()
            if time.monotonic() - last_print > 15.0:
                print(f"    [LP] it={it}, score={score:.6f}, M={M:.8f}")
                last_print = time.monotonic()

        g = _lp_step(a, M)
        if g is None or g.sum() < 1e-12:
            break

        g = g / g.sum()
        a_new = (1.0 - t_cur) * a + t_cur * g
        a = np.maximum(a_new, 0.0)
        a = a / a.sum()

        # Decrease learning rate slowly
        t_cur = max(0.1, t_cur * 0.999)

        it += 1

    return best_a, best_score


def _perturb_and_check(a, rng, n_perturb):
    """Randomly perturb a subset of elements, return perturbed if score improves."""
    n = len(a)
    a2 = a.copy()
    idx = rng.choice(n, n_perturb, replace=False)
    a2[idx] *= np.exp(rng.normal(0, 0.08, n_perturb))
    a2 = np.maximum(a2, 1e-20)
    a2 = a2 / a2.sum()
    return a2


def _make_inits():
    """Generate diverse initializations."""
    inits = []
    n = 2000

    # Multi-frequency sinusoids (key insight: combine freq 1 and freq 2)
    x = np.arange(n, dtype=np.float64)
    for A1, A2, ph1, ph2 in [
        (0.4, 0.15, np.pi / 4, 0.0),
        (0.35, 0.2, np.pi / 3, np.pi / 6),
        (0.4, 0.1, 0.0, np.pi / 2),
        (0.3, 0.25, np.pi / 6, np.pi / 4),
        (0.45, 0.1, np.pi / 5, np.pi / 3),
    ]:
        a = 1.0 + A1 * np.sin(2 * np.pi * x / n + ph1) + A2 * np.sin(4 * np.pi * x / n + ph2)
        a = np.maximum(a, 0.01)
        inits.append((f"mf_A1{A1}_A2{A2}", a))

    # Single frequency with various phases
    for ph in [0, np.pi / 4, np.pi / 3, np.pi / 2, 2 * np.pi / 3]:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * x / n + ph)
        a = np.maximum(a, 0.05)
        inits.append((f"sin_ph{ph:.2f}", a))

    # Exponential decay
    for rate in [0.5, 0.8, 1.2]:
        a = np.exp(-rate * x / n) + 0.1
        inits.append((f"decay_{rate}", a))

    # Cosine (bell-shaped)
    for width in [0.3, 0.5, 0.7]:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / (n * width))
        a = np.maximum(a, 0.02)
        inits.append((f"cos_w{width}", a))

    # Two-hump
    a = np.exp(-((x - n * 0.25) / (n * 0.15)) ** 2) + np.exp(-((x - n * 0.75) / (n * 0.15)) ** 2) + 0.05
    inits.append(("two_hump", a))

    # S-curve (sigmoid)
    for ctr in [0.3, 0.4, 0.5]:
        a = 1.0 / (1.0 + np.exp(-6 * (x / n - ctr)))
        a = a - a.min() + 0.1
        inits.append((f"sigmoid_{ctr}", a))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Hybrid optimizer: multi-peak subgradient + LP direction + multi-freq init.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25.0  # safety margin

    best_overall = None
    best_overall_score = np.inf

    def _update_best(a, label):
        nonlocal best_overall, best_overall_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_sequence(a.tolist())
        if np.isfinite(sc) and sc < best_overall_score:
            best_overall_score = sc
            best_overall = a.copy()
            print(f"  *** NEW BEST ({label}): {sc:.6f}, n={len(a)} ***")

    # --- Get provided sequence ---
    provided = None
    try:
        p = np.array(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, None)
        if p.size > 50 and p.sum() > 0.01:
            provided = p
            print(f"Using provided sequence: n={p.size}")
    except (NameError, Exception):
        pass

    # --- Phase 1: Quick subgradient from diverse inits (200s) ---
    print("\n=== Phase 1: Quick subgradient from diverse inits ===")
    inits = _make_inits()
    if provided is not None:
        inits.append(("provided", provided.copy()))

    # Score initializations
    scored = []
    for name, a0 in inits:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-10:
            continue
        a0 = a0 / s
        c0 = _fft_convolve(a0)
        sc0 = 2.0 * len(a0) * float(np.max(c0))
        scored.append((sc0, name, a0))

    scored.sort(key=lambda x: x[0])
    print("Top inits:")
    for i, (sc, name, _) in enumerate(scored[:6]):
        print(f"  {i+1}. {name}: {sc:.6f}")

    # Run quick subgradient on top 5
    t_phase1 = min(200.0, deadline - time.monotonic())
    t_per_init = t_phase1 / 5.0
    subgrad_results = []

    for i in range(min(5, len(scored))):
        if time.monotonic() > deadline:
            break
        init_sc, name, a0 = scored[i]
        t_alloc = min(t_per_init, deadline - time.monotonic())
        if t_alloc < 5:
            break
        print(f"  Subgrad '{name}' (n={len(a0)}, t={t_alloc:.0f}s)")
        a_res, sc_res = _optimize_subgrad(a0, t_alloc, rng, eta_init=0.12, top_k=5)
        _update_best(a_res, f"sg_{name}")
        subgrad_results.append((sc_res, name, a_res))

    subgrad_results.sort(key=lambda x: x[0])

    # --- Phase 2: LP-based optimization (1200s) ---
    print("\n=== Phase 2: LP-based optimization ===")
    t_phase2 = min(1200.0, deadline - time.monotonic())
    if best_overall is not None and t_phase2 > 60:
        print(f"  Starting LP from best (score={best_overall_score:.6f}, n={len(best_overall)})")
        a_lp, sc_lp = _optimize_lp(best_overall, t_phase2 * 0.7, rng, t_lr=0.35)
        _update_best(a_lp, "LP_main")

        # Also try LP from second-best subgradient
        remaining2 = deadline - time.monotonic()
        if remaining2 > 60:
            for j, (s2, nm2, a2) in enumerate(subgrad_results[:2]):
                if a2 is best_overall:
                    continue
                t_alloc = min(remaining2 * 0.5, deadline - time.monotonic())
                if t_alloc < 30:
                    break
                print(f"  LP from '{nm2}' (score={s2:.6f}, t={t_alloc:.0f}s)")
                a_lp2, sc_lp2 = _optimize_lp(a2, t_alloc, rng, t_lr=0.4)
                _update_best(a_lp2, f"LP_{nm2}")

    # --- Phase 3: Fine subgradient polish (500s) ---
    print("\n=== Phase 3: Fine subgradient polish ===")
    t_phase3 = min(500.0, deadline - time.monotonic())
    if best_overall is not None and t_phase3 > 30:
        # Multiple polish passes with different parameters
        polish_configs = [
            (0.06, 0.8, 7),
            (0.04, 0.85, 5),
            (0.08, 0.7, 3),
            (0.05, 0.75, 9),
        ]
        t_per_polish = t_phase3 / len(polish_configs)
        for cfg in polish_configs:
            t_alloc = min(t_per_polish, deadline - time.monotonic())
            if t_alloc < 10:
                break
            eta_p, mu_p, kp = cfg
            print(f"  Polish (eta={eta_p}, mu={mu_p}, k={kp}, t={t_alloc:.0f}s)")
            a_pol, sc_pol = _optimize_subgrad(best_overall, t_alloc, rng,
                                              eta_init=eta_p, mu=mu_p, top_k=kp)
            _update_best(a_pol, f"polish_{eta_p}_{kp}")

    # --- Phase 4: Exploration with different n and perturbations (300s) ---
    print("\n=== Phase 4: Exploration ===")
    t_phase4 = min(300.0, deadline - time.monotonic())
    if t_phase4 > 20:
        # Try different n values
        for n_try in [3000, 2500, 1500]:
            t_alloc = min(60.0, deadline - time.monotonic())
            if t_alloc < 10:
                break
            x = np.arange(n_try, dtype=np.float64)
            a_try = 1.0 + 0.4 * np.sin(2 * np.pi * x / n_try + np.pi / 4) \
                     + 0.15 * np.sin(4 * np.pi * x / n_try)
            a_try = np.maximum(a_try, 0.01)
            a_try = a_try / a_try.sum()
            print(f"  Exploring n={n_try} (t={t_alloc:.0f}s)")
            a_res, sc_res = _optimize_subgrad(a_try, t_alloc, rng, eta_init=0.1, top_k=5)
            _update_best(a_res, f"explore_n{n_try}")

        # Perturbation search on best
        remaining4 = deadline - time.monotonic()
        if best_overall is not None and remaining4 > 20:
            print(f"  Perturbation search (t={remaining4:.0f}s)")
            n = len(best_overall)
            for _ in range(50):
                t_rem = deadline - time.monotonic()
                if t_rem < 2:
                    break
                a_pert = _perturb_and_check(best_overall, rng, max(5, n // 100))
                sc_pert = 2.0 * n * float(np.max(_fft_convolve(a_pert)))
                if sc_pert < best_overall_score:
                    _update_best(a_pert, "perturb")
                    # Short re-optimization from perturbed point
                    a_re, sc_re = _optimize_subgrad(a_pert, min(20.0, t_rem - 2), rng,
                                                    eta_init=0.05, top_k=5)
                    _update_best(a_re, "perturb+opt")

    # --- Phase 5: Final LP + polish (remaining time) ---
    print("\n=== Phase 5: Final refinement ===")
    remaining5 = deadline - time.monotonic()
    if best_overall is not None and remaining5 > 30:
        # Final LP pass
        t_lp = min(remaining5 * 0.4, deadline - time.monotonic())
        if t_lp > 20:
            print(f"  Final LP (t={t_lp:.0f}s)")
            a_lp, sc_lp = _optimize_lp(best_overall, t_lp, rng, t_lr=0.3)
            _update_best(a_lp, "final_LP")

        # Final fine subgrad
        t_sg = min(remaining5 * 0.5, deadline - time.monotonic())
        if t_sg > 10:
            print(f"  Final subgrad (t={t_sg:.0f}s)")
            a_f, sc_f = _optimize_subgrad(best_overall, t_sg, rng,
                                          eta_init=0.03, mu=0.85, top_k=7)
            _update_best(a_f, "final_sg")

    # --- Final verification ---
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s ===")
        return [float(max(0.0, x)) for x in best_overall]
    else:
        # Fallback
        a = np.ones(1000)
        print("Fallback to uniform")
        return a.tolist()
