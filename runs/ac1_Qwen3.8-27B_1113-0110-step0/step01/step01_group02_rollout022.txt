"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses smoothed multiplicative gradient descent (softmax-weighted subgradient) with beta annealing,
combined with parametric sinusoidal search and Nesterov-accelerated updates.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _smoothed_grad(a, p, n):
    """Compute g[m] = sum_j a[j] * p[j+m] for m=0..n-1 using FFT."""
    a_rev = a[::-1].copy()
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a_rev, N)
    P = np.fft.rfft(p, N)
    conv = np.fft.irfft(A * P, N)
    return conv[n-1:2*n-1]

def _proj_simplex(v):
    """Project v onto the probability simplex {a>=0, sum(a)=1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
   .theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _score(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a."""
    n = len(a)
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c))

def _optimize_smoothed_mult(a0, time_budget, rng, beta_schedule=None):
    """Smoothed multiplicative gradient descent with beta annealing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-20)
    a = a / a.sum()
    
    if beta_schedule is None:
        beta_schedule = [1.0, 2.0, 5.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 5000.0]
    
    best_score = _score(a)
    best_a = a.copy()
    deadline = time.monotonic() + time_budget
    momentum = np.zeros(n)
    mu = 0.6
    eta = 0.3
    it = 0
    last_print = time.monotonic()
    
    total_betas = len(beta_schedule)
    time_per_beta = time_budget / total_betas
    
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        beta_deadline = time.monotonic() + time_per_beta
        eta = min(0.3, 0.1 + 0.2 / beta)
        local_it = 0
        no_improve = 0
        
        while time.monotonic() < beta_deadline:
            c = _fft_convolve(a)
            c_max = float(np.max(c))
            score = 2.0 * n * c_max
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
                if time.monotonic() - last_print > 15.0:
                    print(f"      [smooth] beta={beta:.0f}, it={local_it}, score={score:.6f}")
                    last_print = time.monotonic()
            else:
                no_improve += 1
            
            if no_improve > 300:
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-20)
                a = a / a.sum()
                no_improve = 0
                momentum = np.zeros(n)
                local_it += 1
                continue
            
            # Compute softmax weights
            c_shifted = c - c_max
            if beta * c_shifted[-1] < -700:
                log_weights = beta * c_shifted
                log_weights[log_weights < -700] = -700
            else:
                log_weights = beta * c_shifted
            log_weights -= np.max(log_weights)
            weights = np.exp(log_weights)
            p = weights / weights.sum()
            
            # Compute smoothed gradient
            g = _smoothed_grad(a, p, n)
            g_max = g.max()
            if g_max < 1e-30:
                break
            g = g / g_max
            
            # Nesterov-style momentum
            momentum = mu * momentum + (1.0 - mu) * g
            
            # Multiplicative update
            a = a * np.exp(-eta * momentum)
            a = np.maximum(a, 1e-20)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                eta *= 0.5
                local_it += 1
                continue
            a = a / s
            
            # Adaptive eta
            if no_improve > 200:
                eta *= 0.95
            elif no_improve < 50:
                eta = min(eta * 1.01, 0.5)
            
            local_it += 1
            it += 1
    
    return best_a, best_score

def _optimize_subgrad_topk(a0, time_budget, rng, K=5):
    """Subgradient descent using top-K peaks of the convolution."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-20)
    a = a / a.sum()
    
    best_score = _score(a)
    best_a = a.copy()
    deadline = time.monotonic() + time_budget
    momentum = np.zeros(n)
    mu = 0.7
    eta = 0.15
    it = 0
    no_improve = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        c_max = float(np.max(c))
        score = 2.0 * n * c_max
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 15.0:
                print(f"      [topk] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        if no_improve > 500:
            a = best_a * np.exp(rng.normal(0, 0.1, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.15
            momentum = np.zeros(n)
            it += 1
            continue
        
        # Find top-K peaks
        top_k_indices = np.argsort(c)[::-1][:K]
        # Weight by proximity to max
        weights = np.exp(-0.5 * (c[top_k_indices] - c[top_k_indices[0]]) * 50)
        weights = weights / weights.sum()
        
        # Combined subgradient
        g = np.zeros(n)
        for wi, ki in zip(weights, top_k_indices):
            ki = int(ki)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo <= hi:
                j_range = np.arange(lo, hi + 1)
                indices = ki - j_range
                g[lo:hi+1] += wi * 2.0 * a[indices]
        
        g_max = g.max()
        if g_max < 1e-15:
            break
        g = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g
        
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a = a / s
        
        if no_improve > 300:
            eta *= 0.97
        elif no_improve < 30:
            eta = min(eta * 1.005, 0.35)
        
        it += 1
    
    return best_a, best_score

def _optimize_additive_smooth(a0, time_budget, rng):
    """Additive smoothed gradient with Euclidean projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-20)
    a = a / a.sum()
    
    best_score = _score(a)
    best_a = a.copy()
    deadline = time.monotonic() + time_budget
    momentum = np.zeros(n)
    mu = 0.75
    eta = 0.02
    it = 0
    no_improve = 0
    beta = 50.0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        c_max = float(np.max(c))
        score = 2.0 * n * c_max
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 15.0:
                print(f"      [add_smooth] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        if no_improve > 400:
            a = best_a + rng.normal(0, 0.02, n)
            a = np.maximum(a, 0)
            a = a / max(a.sum(), 1e-15)
            no_improve = 0
            momentum = np.zeros(n)
            it += 1
            continue
        
        c_shifted = c - c_max
        log_w = beta * c_shifted
        log_w -= np.max(log_w)
        p = np.exp(log_w)
        p = p / p.sum()
        
        g = _smoothed_grad(a, p, n)
        g_max = g.max()
        if g_max < 1e-30:
            break
        g = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g
        
        a = a - eta * momentum
        a = _proj_simplex(a)
        a = np.maximum(a, 1e-20)
        a = a / a.sum()
        
        if no_improve > 200:
            eta *= 0.96
        elif no_improve < 30:
            eta = min(eta * 1.02, 0.1)
        
        it += 1
    
    return best_a, best_score

def _parametric_search(n, time_budget, rng):
    """Search over sum-of-sinusoids parametrization."""
    def make_a(params):
        x = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        n_sinusoids = (len(params) - 1) // 3
        for i in range(n_sinusoids):
            A = params[1 + 3*i]
            f = params[2 + 3*i]
            phi = params[3 + 3*i]
            a = a + A * np.sin(2 * np.pi * f * x / n + phi)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    
    best_score = np.inf
    best_params = None
    
    # Try different numbers of sinusoids
    for n_sin in [1, 2, 3]:
        if time.monotonic() > time.monotonic() + time_budget - 5:
            break
        n_params = 1 + 3 * n_sin
        # Base amplitude
        x0 = np.zeros(n_params)
        x0[0] = 0.3
        for i in range(n_sin):
            x0[1 + 3*i] = 0.3  # amplitude
            x0[2 + 3*i] = i + 1  # frequency
            x0[3 + 3*i] = rng.uniform(0, 2*np.pi)  # phase
        
        # Bounds
        lbs = np.array([0.01] + [0.01, 0.5, -2*np.pi] * n_sin)
        ubs = np.array([1.0] + [1.0, 5.0, 2*np.pi] * n_sin)
        
        def objective(params):
            a = make_a(params)
            c = _fft_convolve(a)
            return 2.0 * n * float(np.max(c))
        
        result = scipy_minimize(objective, x0, method='L-BFGS-B',
                                bounds=list(zip(lbs, ubs)),
                                options={'maxiter': 200, 'ftol': 1e-10})
        
        if result.fun < best_score:
            best_score = result.fun
            best_params = result.x.copy()
    
    if best_params is not None:
        a = make_a(best_params)
        return a, best_score
    return None, np.inf

def _try_truncation(a, max_tries=20):
    """Try removing elements from the ends or middle to see if shorter is better."""
    n = len(a)
    best_a = a
    best_score = _score(a)
    
    for trim in range(0, min(n // 2, 1000), max(1, n // max_tries)):
        if trim == 0:
            continue
        # Trim from end
        a_trim = a[:n - trim]
        if a_trim.sum() > 0.01:
            s = _score(a_trim)
            if s < best_score:
                best_score = s
                best_a = a_trim
        # Trim from beginning
        a_trim = a[trim:]
        if a_trim.sum() > 0.01:
            s = _score(a_trim)
            if s < best_score:
                best_score = s
                best_a = a_trim
    return best_a, best_score

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    Uses smoothed multiplicative gradient descent with beta annealing,
    top-K subgradient, parametric sinusoidal search, and Nesterov acceleration.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20.0
    
    best_overall = None
    best_overall_score = np.inf
    
    print(f"Starting optimization with budget {budget_s}s")
    
    # Load previous best if available
    prev_best = None
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            prev_best = hs / hs.sum()
            print(f"Loaded previous best: n={len(prev_best)}, score={_score(prev_best):.6f}")
    except (NameError, Exception):
        pass
    
    # Build candidates
    candidates = []
    
    if prev_best is not None:
        candidates.append(("prev_best", prev_best))
    
    # Sinusoidal initializations with various parameters
    for n in [1800, 2000, 2200, 2500, 3000]:
        x = np.arange(n, dtype=np.float64)
        for amp in [0.3, 0.4, 0.5]:
            for phase in [np.pi/4, np.pi/3, np.pi/2]:
                a = 1.0 + amp * np.sin(2 * np.pi * x / n + phase)
                a = np.maximum(a, 0.05)
                candidates.append((f"sin_{n}_{amp}_{phase:.2f}", a / a.sum()))
    
    # Double sinusoid
    for n in [2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x / n) + 0.2 * np.sin(4 * np.pi * x / n + np.pi/3)
        a = np.maximum(a, 0.05)
        candidates.append((f"dsin_{n}", a / a.sum()))
    
    # Exponential decay
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.8 * x / n) + 0.1
        candidates.append((f"decay_{n}", a / a.sum()))
    
    # V-shaped (low in middle, high at edges)
    for n in [2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        center = n / 2
        a = 1.0 + 0.5 * np.abs(x - center) / center
        candidates.append((f"vshape_{n}", a / a.sum()))
    
    # Cosine bell (high in middle)
    for n in [2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.4 * np.cos(np.pi * x / n)
        a = np.maximum(a, 0.05)
        candidates.append((f"cosbell_{n}", a / a.sum()))
    
    # Sawtooth-like
    for n in [2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        t = x / n
        a = 1.0 + 0.3 * (2 * (t - np.floor(t + 0.5)))
        a = np.maximum(a, 0.05)
        candidates.append((f"saw_{n}", a / a.sum()))
    
    print(f"Total candidates: {len(candidates)}")
    
    # Quick scoring
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        sc = _score(a0)
        scored.append((sc, name, a0))
    
    scored.sort(key=lambda x: x[0])
    print("Top initial scores:")
    for i, (sc, name, _) in enumerate(scored[:10]):
        print(f"  {i+1}. {name}: {sc:.6f}")
    
    # Phase 1: Parametric search (fast, explores the landscape)
    print("\n=== Phase 1: Parametric Search ===")
    p1_deadline = total_start + 300
    for n in [1800, 2000, 2200, 2500, 3000]:
        if time.monotonic() > p1_deadline:
            break
        t_remaining = max(20, p1_deadline - time.monotonic())
        t_alloc = min(t_remaining, 50.0)
        print(f"  Parametric search n={n}, budget={t_alloc:.0f}s")
        a_p, s_p = _parametric_search(n, t_alloc, rng)
        if a_p is not None and s_p < best_overall_score:
            best_overall_score = s_p
            best_overall = a_p.copy()
            print(f"  *** Parametric best: {s_p:.6f} (n={n}) ***")
    
    # Phase 2: Optimize top candidates with multiple methods
    print("\n=== Phase 2: Full Optimization ===")
    p2_deadline = total_start + 1800
    n_top = min(len(scored), 6)
    
    # Allocate time
    remaining_for_p2 = p2_deadline - time.monotonic()
    weights = [max(1.0, n_top - i) ** 1.5 for i in range(n_top)]
    total_w = sum(weights)
    times = [remaining_for_p2 * w / total_w for w in weights]
    
    for i in range(n_top):
        if time.monotonic() > p2_deadline:
            break
        
        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(10.0, p2_deadline - time.monotonic()))
        
        print(f"\n  Optimizing '{name}' (n={len(a0)}, init={init_score:.6f}, time={t_alloc:.0f}s)")
        
        # Method 1: Smoothed multiplicative
        t1 = t_alloc * 0.4
        a1, s1 = _optimize_smoothed_mult(a0, t1, rng)
        ev1 = evaluate_sequence(a1.tolist())
        print(f"    Smoothed mult: score={ev1:.6f}")
        if np.isfinite(ev1) and ev1 < best_overall_score:
            best_overall_score = ev1
            best_overall = a1.copy()
            print(f"    *** NEW BEST: {ev1:.6f} ***")
        
        # Method 2: Top-K subgradient from best so far
        t2 = t_alloc * 0.3
        start2 = a1 if np.isfinite(s1) and s1 < init_score else a0
        a2, s2 = _optimize_subgrad_topk(start2, t2, rng, K=5)
        ev2 = evaluate_sequence(a2.tolist())
        print(f"    Top-K subgrad: score={ev2:.6f}")
        if np.isfinite(ev2) and ev2 < best_overall_score:
            best_overall_score = ev2
            best_overall = a2.copy()
            print(f"    *** NEW BEST: {ev2:.6f} ***")
        
        # Method 3: Additive smooth
        t3 = t_alloc * 0.3
        start3 = best_overall if best_overall is not None and len(best_overall) == len(a0) else start2
        if start3 is None or len(start3) != len(a0):
            start3 = start2
        a3, s3 = _optimize_additive_smooth(start3, t3, rng)
        ev3 = evaluate_sequence(a3.tolist())
        print(f"    Additive smooth: score={ev3:.6f}")
        if np.isfinite(ev3) and ev3 < best_overall_score:
            best_overall_score = ev3
            best_overall = a3.copy()
            print(f"    *** NEW BEST: {ev3:.6f} ***")
    
    # Phase 3: Cross-pollination - optimize from the global best with different n
    print("\n=== Phase 3: Cross-pollination ===")
    p3_deadline = total_start + 2200
    
    if best_overall is not None:
        n_best = len(best_overall)
        print(f"  Current best: score={best_overall_score:.6f}, n={n_best}")
        
        # Try different n by resampling the best
        for n_try in [n_best - 200, n_best + 200, n_best - 500, n_best + 500]:
            if time.monotonic() > p3_deadline or n_try < 500:
                continue
            t_alloc = min(60.0, max(10.0, p3_deadline - time.monotonic()))
            
            # Resample to new length
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_resampled = np.interp(x_new, x_old, best_overall)
            a_resampled = np.maximum(a_resampled, 0)
            a_resampled = a_resampled / a_resampled.sum()
            
            print(f"  Resampled to n={n_try}: init_score={_score(a_resampled):.6f}")
            a_r, s_r = _optimize_smoothed_mult(a_resampled, t_alloc, rng)
            ev_r = evaluate_sequence(a_r.tolist())
            print(f"    Score: {ev_r:.6f}")
            if np.isfinite(ev_r) and ev_r < best_overall_score:
                best_overall_score = ev_r
                best_overall = a_r.copy()
                print(f"    *** NEW BEST: {ev_r:.6f} ***")
    
    # Phase 4: Final polishing
    print("\n=== Phase 4: Final Polishing ===")
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 20:
        print(f"  Polishing with {remaining:.0f}s remaining")
        
        # Multiple polish rounds with different parameters
        for polish_i in range(3):
            if time.monotonic() > deadline:
                break
            t_polish = max(10.0, (deadline - time.monotonic()) / (3 - polish_i))
            
            # Vary the beta schedule
            if polish_i == 0:
                bs = [10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0]
            elif polish_i == 1:
                bs = [5.0, 20.0, 100.0, 500.0, 2000.0, 10000.0]
            else:
                bs = [1.0, 5.0, 20.0, 100.0, 500.0, 2000.0]
            
            a_pol, s_pol = _optimize_smoothed_mult(best_overall, t_polish, rng, beta_schedule=bs)
            ev_pol = evaluate_sequence(a_pol.tolist())
            print(f"    Polish {polish_i+1}: {ev_pol:.6f}")
            if np.isfinite(ev_pol) and ev_pol < best_overall_score:
                best_overall_score = ev_pol
                best_overall = a_pol.copy()
                print(f"    *** NEW BEST: {ev_pol:.6f} ***")
        
        # Try top-K from the polished result
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_pk, s_pk = _optimize_subgrad_topk(best_overall, remaining2, rng, K=10)
            ev_pk = evaluate_sequence(a_pk.tolist())
            print(f"    Final top-K: {ev_pk:.6f}")
            if np.isfinite(ev_pk) and ev_pk < best_overall_score:
                best_overall_score = ev_pk
                best_overall = a_pk.copy()
                print(f"    *** NEW BEST: {ev_pk:.6f} ***")
    
    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s")
        
        # Convert to proper list
        result = [float(x) for x in best_overall]
        # Verify all positive
        assert all(x >= 0 for x in result)
        assert sum(result) > 0.01
        return result
    else:
        # Fallback
        a = np.ones(2000, dtype=np.float64)
        print("Falling back to uniform sequence")
        return a.tolist()
