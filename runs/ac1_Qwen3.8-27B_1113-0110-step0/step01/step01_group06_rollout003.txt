import numpy as np

from scipy import optimize, sparse

def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _convolve_fft(a):
    n = len(a)
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:L]
    b[b < 0] = 0  # numerical artifact
    return b

def _score(a, b=None):
    if b is None:
        b = _convolve_fft(a)
    s = a.sum()
    if s < 1e-300:
        return np.inf, b
    return 2.0 * len(a) * b.max() / (s * s), b

def _subgrad(a, b, topk=50, focus=10.0):
    """Compute subgradient of max(b) or smoothed version."""
    n = len(a)
    L = len(b)
    maxb = b.max()
    
    if topk is not None and topk < L:
        # Hard top-K focus
        vals = np.partition(b, -topk)[-topk:]
        min_top = vals[0]
        mask = b >= min_top
        w = np.zeros(L)
        w[mask] = 1.0
        w /= w.sum()
    else:
        # Soft focus
        w = np.exp(focus * (b - maxb) / maxb if maxb > 1e-300 else 0)
        w /= w.sum()
    
    # Gradient: 2 * ifft(FFT(w) * conj(FFT(a)))[:n]
    N = _next_pow2(L + n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    g = 2.0 * np.fft.irfft(fw * fa.conj(), N)[:n]
    
    return g, b, w

def _project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    v = v.copy()
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n+1) > cssv - s)[0]) if n > 0 else 0
    theta = (cssv[int(rho)] - s) / (rho + 1)
    return np.maximum(v - theta, 0.0)

Wait, the simplex projection formula: to project v onto {x >= 0, sum(x) = s}:
x_i = max(v_i - theta, 0) where theta is chosen so sum(x) = s.

The standard algorithm:
1. Sort v in descending order: u_1 >= u_2 >= ... >= u_n
2. Find rho = max{j : u_j - (1/j) * sum(u_1,...,u_j) > 1 - s/j}... 

Actually, for projection onto {x >= 0, sum(x) = 1} (the standard simplex):
theta = (sum of top rho elements - 1) / rho
where rho is the number of elements above theta.

For general sum s:
x_i = max(v_i - theta, 0) where sum(x) = s
theta = (sum of elements > theta - s) / (count of elements > theta)

The standard O(n log n) algorithm:
1. Sort v descending: u
2. Compute cumulative sums
3. Find rho such that u_rho > (cumsum(rho) - s) / rho >= u_{rho+1}
4. theta = (cumsum(rho) - s) / rho
5. x = max(v - theta, 0)

    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho: largest j such that u[j-1] > (cssv[j-1] - s) / j
    # i.e., j * u[j-1] > cssv[j-1] - s
    # i.e., j * u[j-1] + s > cssv[j-1]
    idx = np.arange(1, n + 1)
    cond = idx * u + s > cssv
    rho = np.max(cond.nonzero()[0]) + 1  # 1-indexed
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0.0)
