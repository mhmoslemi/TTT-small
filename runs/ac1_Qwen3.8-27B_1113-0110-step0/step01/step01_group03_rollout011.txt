```python
def solve_lp(a, M):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0"""
    n = len(a)
    # Build constraint matrix A_csp such that A_csp @ b <= M * ones(2n-1)
    # A_csp[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # Using sparse matrix
    from scipy import sparse
    row_idx = []
    col_idx = []
    vals = []
    for k in range(2*n-1):
        j_min = max(0, k-(n-1))
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            row_idx.append(k)
            col_idx.append(j)
            vals.append(a[k-j])
    
    A_csp = sparse.csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    b_csp = np.full(2*n-1, M)
    
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    
    res = scipy.optimize.linprog(c, A_ub=A_csp, b_ub=b_csp, bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
"""
LP-based alternating minimization + gradient fine-tuning to minimize
2*n*max(convolve(a,a))/sum(a)^2. Uses sparse HiGHS LP for inner
optimization, multiple initializations, and spectral-aware refinement.
"""
import numpy as np
import time
from scipy import sparse
from scipy import optimize

def fft_self_conv(a):
    n = len(a)
    if n <= 1:
        return a * a.astype(np.float64)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]

def fast_score(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_self_conv(a)
    M = c.max()
    return 2.0 * n * M / (S * S)

def build_conv_matrix(a):
    n = len(a)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        length = j_hi - j_lo + 1
        rows[idx:idx+length] = k
        cols[idx:idx+length] = np.arange(j_lo, j_hi + 1)
        vals[idx:idx+length] = a[k - np.arange(j_lo, j_hi + 1)]
        idx += length
    return sparse.csr_matrix((vals, (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))

def solve_inner_lp(a, M, timeout=20):
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = optimize.linprog(c_obj, A_ub=C, b_ub=b_ub,
                               bounds=[(0, None)] * n, method='highs',
                               options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception as e:
        print(f"  LP error: {e}")
    return None

    for it in range(max_iters):

            break

        M = float(c.max())

        b = solve_inner_lp(a, M, timeout=20)
        lp_ct += 1
        if b is None:

            b = np.maximum(b, 0.0)

        S_b = b.sum()
        if S_b < 1e-10:

        b = b / S_b * n

        a = (1.0 - t) * a + t * b
        a = np.maximum(a, 0.0)
```
