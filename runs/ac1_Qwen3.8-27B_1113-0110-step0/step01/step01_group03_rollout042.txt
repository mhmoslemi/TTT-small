```python
def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]

def fast_score(a):
    """Fast score using FFT."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_self_conv(a)
    M = c.max()
    return float(2.0 * n * M / (S * S))

def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Solve: sum(max(v - tau, 0)) = target_sum
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Find tau
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv[-1] - target_sum))[0][-1]
```

```python
def compute_grad_fft(y, P):
    """
    Compute gradient using FFT.
    y: length n
    P: length 2n-1 (softmax weights over convolution lags)
    Returns: gradient of length n
    """
    n = len(y)
    # Pad to avoid circular effects
    size = 1
    while size < 3 * n:
        size *= 2
    
    Y = np.fft.rfft(y, size)
    Pf = np.fft.rfft(P, size)
    
    # S1[j] = sum_k P[k] * Y[j-k] = convolution
    S1 = np.fft.irfft(Pf * Y, size)[:n]
    
    # S2[j] = sum_k P[k] * Y[j+k] = cross-correlation
    S2 = np.fft.irfft(Pf * np.conj(Y), size)[:n]
    
    grad = S1 + S2
    return grad
```

Now, for the Nesterov acceleration scheme:

The algorithm for minimizing f(x) + g(x) where f is smooth and g is indicator of the constraint set:
1. x_0, y_0 = x_0
2. For t = 1, 2, ...:
   a. t_new = (1 + sqrt(1 + 4*t²)) / 2
   b. y_new = x_t + (t/t_new - 1) * (x_t - x_{t-1})
   c. x_{t+1} = prox_g(1/L)(y_new - (1/L) * ∇f(y_new))
   d. t = t_new

Where L is the Lipschitz constant of ∇f and prox_g is the projection onto the constraint set.

- f(y) = eps * log(sum exp(c_k/eps)) (smooth)
- g(x) = 0 if sum(x) = n and x ≥ 0, else +∞

||∇²f|| ≤ (1/eps) * ||C|| where C is some matrix related to the convolution...

```python
import numpy as np
import time

def fft_self_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]

def fast_score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_self_conv(a)
    M = c.max()
    return float(2.0 * n * M / (S * S))

def project_simplex(v, target_sum):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cs = cssv - target_sum
    rho = np.nonzero(u * np.arange(1, n + 1) > cs)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    tau = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)

def compute_grad_fft(y, P):
    n = len(y)
    size = 1
    while size < 3 * n:
        size *= 2
    Y = np.fft.rfft(y, size)
    Pf = np.fft.rfft(P, size)
    S1 = np.fft.irfft(Pf * Y, size)[:n]
    S2 = np.fft.irfft(Pf * np.conj(Y), size)[:n]
    return S1 + S2

def nesterov_optimize(a_init, n, deadline, eps_init=1.0, lr_init=0.5, seed=42):
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = project_simplex(a, n)
    a_prev = a.copy()
    y = a.copy()
    t = 1.0
    eps = eps_init
    lr = lr_init
    best_local = a.copy()
    best_score_local = fast_score(a)
    it = 0
    no_improve = 0
    
    while time.time() < deadline - 60:
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + (t / t_new - 1.0) * (a - a_prev)
        y = np.maximum(y, 0.0)
        S_y = y.sum()
        if S_y < 1e-10:
            y = a.copy()
            S_y = a.sum()
        y = y / S_y * n  # Renormalize
        
        # Compute convolution and softmax weights
        c = fft_self_conv(y)
        c_max = c.max()
        P = np.exp((c - c_max) / eps)
        P = P / P.sum()
        
        # Compute gradient
        grad = compute_grad_fft(y, P)
        
        # Gradient step + projection
        a_new = y - lr * grad
        a_new = project_simplex(a_new, n)
        
        a_prev = a
        a = a_new
        t = t_new
        it += 1
        
        # Periodic evaluation
        if it % 50 == 0:
            score = fast_score(a)
            if score < best_score_local - 1e-8:
                best_local = a.copy()
                best_score_local = score
                no_improve = 0
            else:
                no_improve += 1
            
            # Adapt hyperparameters
            if no_improve >= 10:
                no_improve = 0
                if eps > 0.01:
                    eps *= 0.7  # Reduce smoothing
                    t = 1.0  # Reset momentum
                    a_prev = a.copy()
                    y = a.copy()
                elif lr > 0.001:
                    lr *= 0.7
                    t = 1.0
                    a_prev = a.copy()
                    y = a.copy()
                else:
                    # Random restart
                    perturb = np.exp(rng.normal(0, 0.1, n))
                    a = best_local * perturb
                    a = project_simplex(a, n)
                    a_prev = a.copy()
                    y = a.copy()
                    t = 1.0
                    lr = 0.1
                    eps = max(eps, 0.1)
            
            # Exact evaluation every 500 iterations
            if it % 500 == 0:
                exact = float(evaluate_sequence(a.tolist()))
                if np.isfinite(exact) and exact < best_score_local:
                    best_local = a.copy()
                    best_score_local = exact
                    print(f"  [{it}] exact score: {exact:.8f}")
        
        # Time check
        if time.time() > deadline - 90:
            break
    
    return best_local, best_score_local
```
