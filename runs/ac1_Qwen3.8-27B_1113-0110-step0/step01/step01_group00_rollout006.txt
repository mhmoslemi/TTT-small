```python
def _smooth_gradient(a, T):

    c = _fft_convolve(a)
    # Compute softmax weights
    Tc = T * c - T * np.max(c)  # numerical stability
    weights = np.exp(Tc)
```

```python
"""
Minimize J(a) = 2n*max(conv(a,a))/sum(a)^2 for non-negative sequence a.
Strategy: Alternating LP + smooth gradient refinement + perturbation restarts
across multiple sequence lengths. Key insight: optimal sequences have peak
autocorrelation at non-zero lags, achievable via structured zero patterns.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        C[i:i + n, i] = f
    return C


def _lp_solve(f, M):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _lp_solve_topk(f, M, topk):
    """LP with only top-k binding constraints (faster for large n)."""
    n = len(f)
    c = _fft_convolve(f)
    idx = np.argpartition(c, -topk)[-topk:]
    C = np.zeros((topk, n), dtype=np.float64)
    for i in range(n):
        C[:, i] = f * 0
    for ii, k in enumerate(idx):
        for i in range(n):
            val = int(k) - i
            if 0 <= val < n:
                C[ii, i] = f[val]
    c_obj = -np.ones(n)
    b_ub = M * np.ones(topk)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _smooth_gradient(a, T):
    """Gradient of smooth max approximation of autocorrelation peak."""
    n = len(a)
    c = _fft_convolve(a)
    Tc = T * c - T * np.max(c)
    weights = np.exp(Tc)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum
    grad = np.zeros(n)
    # Only use significant weights for speed
    thresh = 1e-8 / len(c)
    significant = weights > thresh
    if not np.any(significant):
        return np.zeros(n)
    for k in np.where(significant)[0]:
        w = weights[k]
        k = int(k)
        if k == 0:
            grad += 2.0 * w * a
        else:
            grad[:n - k] += w * a[k:]
            grad[k:] += w * a[:n - k]
    return grad


def _make_initials(n, rng):
    """Generate diverse initial sequences of length n."""
    t = np.linspace(0, 1, n)
    inits = []
    # Linear decay
    inits.append(np.maximum(1.0 - t, 0.0))
    # Exponential decay
    inits.append(np.exp(-3.0 * t))
    # Two bumps
    inits.append(np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05))
    # Cosine-modulated
    a = (1.0 - t) * (1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t))
    inits.append(np.maximum(a, 0.0))
    # Smooth step
    inits.append(t ** 2 * (3.0 - 2.0 * t))
    # Random
    inits.append(np.abs(rng.normal(0, 1, n)))
    # Binary-like (structured zeros)
    a_bin = np.zeros(n)
    a_bin[::2] = 1.0
    inits.append(a_bin)
    # Periodic with defects
    a_per = np.tile([1.0, 1.0, 0.3], n // 3 + 1)[:n]
    inits.append(a_per)
    # Ramped with oscillation
    a_osc = (1.0 - t) * (1.0 + 0.7 * np.sin(2.0 * np.pi * 5.0 * t))
    inits.append(np.maximum(a_osc, 0.0))
    # Bell at center
    inits.append(np.exp(-((t - 0.5) ** 2) / 0.05))
    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2*np.pi*2*t) + 0.2 * np.cos(2*np.pi*7*t), 0.0))

    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        result.append(a)
    return result


def _interpolate_seq(src, n_target):
    """Interpolate sequence to new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.7, use_topk=False, topk_frac=0.5):
    """Alternating LP optimization."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        if use_topk and n > 600:
            topk = max(50, int(topk_frac * (2 * n - 1)))
            g = _lp_solve_topk(a, M, topk)
        else:
            g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        a = a_new
        blend = max(0.15, blend * 0.96)
        if stall_count > 80:
            # Small perturbation to escape
            pert = np.random.default_rng(42 + it).normal(0, 0.01, n)
            a = a + pert * a
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stall_count = 0
            blend = min(1.0, blend * 1.5)

    return best_a, best_score


def _smooth_grad_phase(a_init, n_iters, T_start, T_end, lr_init, deadline):
    """Smooth gradient descent with Adam."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = _score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_iters - 1))
        S = float(np.sum(a))
        c = _fft_convolve(a)
        score = 2.0 * n * float(np.max(c)) / (S * S)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        grad = _smooth_gradient(a, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        t = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / math.sqrt(t + 1)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S_new = float(np.sum(a))
        if S_new > 1e-10:
            a = a * S / S_new
        else:
            a = best_a.copy()
            break

    return best_a, best_score


def _perturb_and_optimize(a_best, n, rng, deadline):
    """Create perturbed version and do quick LP."""
    a = a_best.copy()
    if len(a) != n:
        a = _interpolate_seq(a, n)
    # Random perturbation
    noise_scale = rng.choice([0.02, 0.05, 0.1, 0.2])
    a = a + noise_scale * rng.normal(0, 1, n) * a
    # Random zeroing
    n_zero = rng.integers(1, n // 5 + 1)
    zero_idx = rng.choice(n, n_zero, replace=False)
    a[zero_idx] = 0.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a_best, _score(a_best)
    a = a / s
    a_opt, score = _alternating_lp(a, 40, deadline, initial_blend=0.5)
    return a_opt, score


def propose_candidate():
    """
    Multi-phase optimizer for minimizing J(a)=2n*max(conv(a,a))/sum(a)^2.
    Phase 1: Quick LP sweep across n values and initializations.
    Phase 2: Focused LP refinement on best candidate.
    Phase 3: Smooth gradient + LP interleaved.
    Phase 4: Perturbation restarts for exploration.
    Phase 5: Final polish.
    """
    total_deadline = time.monotonic() + 2350.0
    rng = np.random.default_rng(12345)

    best_a = None
    best_score = float('inf')
    best_n = 0

    # Try to use parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            ps = float(np.sum(parent))
            if ps > 0.01:
                parent_score = _score(parent)
                print(f"Parent: n={parent.size}, score={parent_score:.6f}")
                best_a = parent.copy() / ps
                best_score = parent_score
                best_n = parent.size
    except Exception:
        pass

    # Phase 1: Quick LP sweep
    print("\n=== Phase 1: Quick LP sweep ===")
    n_candidates = [150, 200, 256, 300, 400, 500, 600]
    phase1_deadline = total_deadline - 1800.0
    for n in n_candidates:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}:", end=" ")
        inits = _make_initials(n, rng)
        if best_a is not None:
            inits.append(_interpolate_seq(best_a, n))
        phase_best = None
        phase_best_score = float('inf')
        for idx, a_init in enumerate(inits):
            sub_deadline = min(phase1_deadline, time.monotonic() + 25.0)
            a_opt, sc = _alternating_lp(a_init, 60, sub_deadline, initial_blend=0.7)
            if sc < phase_best_score:
                phase_best_score = sc
                phase_best = a_opt
        if phase_best is not None:
            print(f"best={phase_best_score:.6f}", end="")
            if phase_best_score < best_score:
                best_score = phase_best_score
                best_a = phase_best.copy()
                best_n = n
                print(" ** NEW BEST")
            else:
            print()
        print()
    print(f"Phase 1 done: best={best_score:.6f} (n={best_n})")

    # Phase 2: Focused LP on best
    print("\n=== Phase 2: Focused LP refinement ===")
    phase2_deadline = total_deadline - 1000.0
    if best_a is not None:
        n2 = best_n
        # Also try interpolating to a few nearby n values
        n_try_list = [n2]
        if n2 < 300:
            n_try_list.append(min(n2 + 100, 500))
        elif n2 < 600:
            n_try_list.append(min(n2 + 100, 800))

        for n_try in n_try_list:
            if time.monotonic() > phase2_deadline:
                break
            print(f"  LP at n={n_try} (deadline={3600 - (time.monotonic() - (total_deadline - 2350)):0.0f}s left)")
            a_init = _interpolate_seq(best_a, n_try) if len(best_a) != n_try else best_a.copy()
            a_opt, sc = _alternating_lp(a_init, 300, phase2_deadline, initial_blend=0.6)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                best_n = n_try
                print(f"    ** New best: {best_score:.6f}")
            else:
                print(f"    score={sc:.6f} (no improvement)")

    # Phase 3: Smooth gradient + LP interleaved
    print(f"\n=== Phase 3: Gradient + LP interleaved (n={best_n}) ===")
    phase3_deadline = total_deadline - 500.0
    if best_a is not None and time.monotonic() < phase3_deadline:
        a_cur = best_a.copy()
        n3 = best_n
        for cycle in range(5):
            if time.monotonic() > phase3_deadline:
                break
            # Gradient step
            sub_dl = min(phase3_deadline, time.monotonic() + 120.0)
            a_grad, sc_grad = _smooth_grad_phase(a_cur, 2000, 5.0, 50.0, 0.05, sub_dl)
            if sc_grad < best_score:
                best_score = sc_grad
                best_a = a_grad.copy()
            a_cur = a_grad.copy()
            # LP step
            sub_dl = min(phase3_deadline, time.monotonic() + 120.0)
            a_lp, sc_lp = _alternating_lp(a_cur, 50, sub_dl, initial_blend=0.4)
            if sc_lp < best_score:
                best_score = sc_lp
                best_a = a_lp.copy()
            a_cur = a_lp.copy()
            print(f"  cycle {cycle}: grad={sc_grad:.6f}, lp={sc_lp:.6f}, best={best_score:.6f}")

    # Phase 4: Perturbation restarts
    print(f"\n=== Phase 4: Perturbation restarts (best={best_score:.6f}) ===")
    phase4_deadline = total_deadline - 200.0
    if best_a is not None:
        n4 = best_n
        n_perturb = 20
        for i in range(n_perturb):
            if time.monotonic() > phase4_deadline:
                break
            sub_dl = min(phase4_deadline, time.monotonic() + 60.0)
            a_p, sc_p = _perturb_and_optimize(best_a, n4, rng, sub_dl)
            if sc_p < best_score:
                best_score = sc_p
                best_a = a_p.copy()
                best_n = n4
                print(f"  pert {i}: ** new best {best_score:.6f}")
            elif i % 5 == 0:
                print(f"  pert {i}: {sc_p:.6f}")
        # Also try different n values with perturbation
        for n_alt in [n4 - 50, n4 + 50, n4 + 100]:
            if n_alt < 50 or time.monotonic() > phase4_deadline:
                continue
            sub_dl = min(phase4_deadline, time.monotonic() + 60.0)
            a_init = _interpolate_seq(best_a, n_alt)
            noise = 0.05 * rng.normal(0, 1, n_alt) * a_init
            a_init = np.maximum(a_init + noise, 0.0)
            a_init = a_init / a_init.sum()
            a_p, sc_p = _alternating_lp(a_init, 80, sub_dl, initial_blend=0.5)
            if sc_p < best_score:
                best_score = sc_p
                best_a = a_p.copy()
                best_n = n_alt
                print(f"  n={n_alt}: ** new best {best_score:.6f}")

    # Phase 5: Final polish
    print(f"\n=== Phase 5: Final polish ===")
    phase5_deadline = total_deadline - 5.0
    if best_a is not None and time.monotonic() < phase5_deadline:
        n5 = best_n
        # Short LP with small blend
        a_polish, sc_pol = _alternating_lp(best_a.copy(), 100, phase5_deadline, initial_blend=0.3)
        if sc_pol < best_score:
            best_score = sc_pol
            best_a = a_polish.copy()
        # Try geometric mean update
        c = _fft_convolve(best_a)
        M = float(np.max(c))
        g = _lp_solve(best_a, M)
        if g is not None:
            sg = float(np.sum(g))
            if sg > 1e-10:
                g = g / sg
                # Geometric mean
                eps_g = 1e-12
                a_geo = np.sqrt(np.maximum(best_a, eps_g) * np.maximum(g, eps_g))
                a_geo = np.maximum(a_geo, 0.0)
                s_geo = a_geo.sum()
                if s_geo > 1e-10:
                    a_geo = a_geo / s_geo
                    sc_geo = _score(a_geo)
                    if sc_geo < best_score:
                        best_score = sc_geo
                        best_a = a_geo.copy()
                        print(f"  Geometric mean: {best_score:.6f}")

    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a) if best_a is not None else 0} ===")

    if best_a is not None:
        result = [float(x) for x in best_a]
        # Verify
        try:
            verified = float(evaluate_sequence(result))
            print(f"Verified score: {verified:.6f}")
        except Exception as e:
            print(f"Verification error: {e}")
        return result

    return [1.0] * 200
```
