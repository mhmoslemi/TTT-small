def fft_conv(u, v):
    nu, nv = len(u), len(v)
    size=1; while size < nu+nv-1: size*=2
    U=np.fft.rfft(u,size); V=np.fft.rfft(v,size)
    out=np.fft.irfft(U*V,size)
    return out[:nu+nv-1]

    size=1; while size<2*n-1: size*=2
    F=np.fft.rfft(a,size)
    out=np.fft.irfft(F*F,size)

import numpy as np
import time


"""
Minimize  2*n*max(conv(a,a))  /  (sum a)^2   over non-negative a.

Key ideas (different from the previous hard-max multi-peak momentum method):
  * Normalize sum(a)=n so the objective == 2*M/n  where M = max of the full
    linear auto-correlation c = conv(a,a).  So we simply minimize the peak of
    the *entire* auto-correlation (all 2n-1 lags), which is a minimax /
    equioscillation style problem.
  * Use a differentiable SOFT-MAX surrogate  L = (1/beta) log sum_k exp(beta*(c_k-mean))
    over ALL lags.  Its gradient is a soft-weighted sum of the per-lag
    gradients and can be computed in O(n log n) as a single convolution
    (cross-correlation) a_rev * softmax(beta*(c-mean)).  This naturally pulls
    every "hot" lag down together and flattens the top of the auto-correlation
    far more effectively than only using the exact top-1..3 peaks.
  * Homotopy on beta (broad -> sharp): first flatten broadly, then shave the
    dominant peak(s).
  * Multi-length racing: the asymptotic optimum improves with length, so we
    race a few candidate lengths (transferring the parent shape via
    interpolation), then commit the remaining budget to the best length.
  * All bookkeeping of the BEST answer is done with the exact evaluate_sequence.
"""


# ----------------------------------------------------------------------
# basic helpers
# ----------------------------------------------------------------------
def _next_pow2(x):
    s = 1
    while s < x:
        s *= 2
    return s


def _fft_self_conv(a):
    n = len(a)
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(a, size)
    out = np.fft.irfft(f * f, size)
    return out[:2 * n - 1]


def _fft_conv(u, v):
    nu, nv = len(u), len(v)
    size = _next_pow2(nu + nv - 1)
    U = np.fft.rfft(u, size)
    V = np.fft.rfft(v, size)
    out = np.fft.irfft(U * V, size)
    return out[:nu + nv - 1]


def _normalize_sum(a, target):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.full_like(a, target / len(a), dtype=np.float64)
    return a / s * target


def _fast_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _fft_self_conv(a)
    m = float(c.max())
    if m < 0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def _softmax_weights(c, mean, beta):
    z = beta * (c - mean)
    z = z - z.max()
    w = np.exp(z)
    ws = w.sum()
    if ws <= 0.0 or not np.isfinite(ws):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return w / ws


# ----------------------------------------------------------------------
# analytic starting shapes (all returned with sum == n)
# ----------------------------------------------------------------------
def _make_analytic(n, kind, rng):
    t = np.linspace(0.0, 1.0, n)
    if kind == "uniform":
        base = np.ones(n)
    elif kind == "ramp":
        base = np.arange(n, dtype=np.float64)
    elif kind == "ramp2":
        base = np.arange(n, dtype=np.float64) ** 2
    elif kind == "u_shape":           # high at the ends
        base = (2.0 * t - 1.0) ** 2
    elif kind == "n_shape":           # high in the center
        base = 1.0 - np.abs(2.0 * t - 1.0)
    elif kind == "two_peak":
        base = np.sin(np.pi * 3.0 * t) ** 2
    elif kind == "four_peak":
        base = np.sin(np.pi * 6.0 * t) ** 2
    elif kind == "highfreq":
        base = 1.5 + np.sin(2.0 * np.pi * 20.0 * t)
    elif kind == "bump":
        base = np.exp(-((t - 0.5) / 0.12) ** 2)
    elif kind == "blocks":
        base = (np.sin(2.0 * np.pi * 5.0 * t) > 0.0).astype(np.float64) + 0.1
    elif kind == "random":
        base = rng.random(n)
    else:
        base = np.ones(n)
    base = np.maximum(base, 0.0)
    return _normalize_sum(base, n)


def _resample_to(parent, n):
    if len(parent) == n:
        return parent.copy()
    x_old = np.linspace(0.0, 1.0, len(parent))
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, parent)
    return np.maximum(out, 0.0)


# ----------------------------------------------------------------------
# soft-max auto-correlation peak minimisation (gradient descent)
# ----------------------------------------------------------------------
def _softmax_descent(a0, n, deadline, rng, tag=""):
    a = _normalize_sum(a0.astype(np.float64).copy(), n)
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(best_score):
        best_score = np.inf

    eval_interval = max(30, int(n // 60))

    momentum = np.zeros(n, dtype=np.float64)
    mu = 0.90
    lr = 0.30
    beta_target = 10.0
    it = 0
    no_improve = 0
    guard = _fast_score(a)
    rev = a[::-1].copy()

    while time.monotonic() < deadline:
        c = _fft_self_conv(a)
        s = float(a.sum())
        mean = (s * s) / len(c)
        cmax = float(c.max())
        spread = cmax - mean
        if spread <= 1e-9:
            break

        beta = beta_target / spread
        w = _softmax_weights(c, mean, beta)
        # G_i  =  sum_j a_j w_{j+i}   (== cross correlation, up to a const)
        rev = a[::-1]
        conv = _fft_conv(rev, w)
        G = conv[(n - 1):(2 * n - 1)]
        G = np.clip(G, 0.0, 50.0)

        momentum = mu * momentum - lr * G
        a = a + momentum
        a = np.maximum(a, 0.0)
        a = _normalize_sum(a, n)
        it += 1

        # divergence guard (cheap FFT score)
        if it % 5 == 0:
            fs = _fast_score(a)
            if (not np.isfinite(fs)) or fs > 2.5 * (guard if np.isfinite(guard) else fs):
                a = best_a.copy()
                momentum *= 0.2
                lr *= 0.5
                fs = _fast_score(a)
            if np.isfinite(fs):
                guard = fs

        if it % eval_interval == 0:
            exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact) and exact < best_score - 1e-9:
                best_score = exact
                best_a = a.copy()
                no_improve = 0
                guard = _fast_score(a)
                beta_target = min(140.0, beta_target * 1.05)
                if tag and (it % (eval_interval * 10) == 0 or no_improve < 2):
                    print("  [%s] n=%d it=%d score=%.8f beta=%.1f lr=%.4f"
                          % (tag, n, it, best_score, beta_target, lr))
            else:
                no_improve += 1

            if no_improve >= 40:
                lr *= 0.6
                momentum *= 0.3
                no_improve = 0
                if lr < 1e-4:
                    a = best_a * np.exp(rng.normal(0.0, 0.07, n))
                    a = _normalize_sum(a, n)
                    lr = 0.10
                    momentum = np.zeros(n, dtype=np.float64)
                    beta_target = 8.0
                    no_improve = 0

            # homotopy: focus tighter on the dominant peak as we progress
            if it % 300 == 0:
                beta_target = min(140.0, beta_target * 1.18)

        if time.monotonic() > deadline - 2:
            break

    # final exact validation of the local best
    final = _normalize_sum(best_a, n)
    fs_exact = float(evaluate_sequence(final.tolist()))
    if np.isfinite(fs_exact) and fs_exact < best_score:
        best_score = fs_exact
        best_a = final
    return best_a, best_score
