def _normalize(a):
    a = np.clip(a, 0.0, 1e6)
    s = float(a.sum())

def _logsumexp_approx(c, T):
    """Approximate max using logsumexp for smoothness."""
    c_shifted = c - c.max()
    return c.max() + T * np.log(np.sum(np.exp(c_shifted / T)))


def _make_weights(b, focus, top_k=0):
    """Create weights concentrated on large b values."""
    L = b.size
    maxb = float(b.max())
    if maxb < 1e-30:
        return np.full(L, 1.0 / L)
    z = (maxb - b) / maxb
    with np.errstate(over='ignore', under='ignore'):
        p = np.exp(-z * focus)
    s = p.sum()
    if s < 1e-12:
        p = np.zeros(L)
        p[np.argmax(b)] = 1.0
    else:
        eps = 0.01
        p = (1 - eps) * (p / s) + eps / L
    if top_k > 0:
        # Boost top-k entries
        idx = np.argsort(b)[-top_k:]
        p[idx] *= 3.0
        p /= p.sum()
    return p

    a = _normalize(a)
    n = a.size
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = _score(a)[0]
    
    b = _fft_square(a)
    score = best_score
    s = float(a.sum())  # normalized to 1
    
    focus = 3.0
    focus_inc = 1.005
    focus_max = 200.0
    eta = 0.5
    eta_min = 0.05
    eta_max = 5.0
    T = T_start
    
    no_improve = 0
    max_no_improve = 500
    resets = 0
    max_resets = 30
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if b is None:
            b = _fft_square(a)
            score, _, _ = _score(a, b)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.4)
            # Reset with perturbation
            amp = 0.3 + 0.1 * resets
            a = best_a * np.exp(rng.normal(0, amp, size=n))
            a = _normalize(a)
            b = None
            score = float('inf')
            focus = float(rng.choice([3.0, 5.0, 10.0, 20.0, 50.0]))
            eta = 0.5
            T = T_start
            continue
        
        # Make weights
        p = _make_weights(b, focus)
        
        # Gradient via FFT
        grad = _fft_grad_conv(p, a)
        # Subtract mean to preserve sum
        grad = grad - float(np.dot(a, grad))
        
        maxg = float(np.max(np.abs(grad)))
        if maxg < 1e-16:
            break
        grad = grad / maxg
        
        # Exponentiated update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-300)
        a_new = _normalize(a_new)
        
        b_new = _fft_square(a_new)
        score_new, _, _ = _score(a_new, b_new)
        
        # Also compute smoothed score for acceptance
        smooth_old = _logsumexp_approx(b, T)
        smooth_new = _logsumexp_approx(b_new, T)
        
        accept = False
        if score_new < best_score - 1e-14:
            accept = True
        elif score_new < score - 1e-14:
            accept = True
        elif smooth_new < smooth_old - 1e-10:
            accept = True
        elif score_new < score * 1.005 and smooth_new < smooth_old:
            accept = True
        
        if accept:
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.03, eta_max)
        else:
            eta = max(eta * 0.8, eta_min)
            if eta < 0.1:
                a = a * np.exp(rng.normal(0, 0.1, size=n))
                a = _normalize(a)
                b = None
                score = float('inf')
        
        focus = min(focus * focus_inc, focus_max)
        T = max(T * 0.999, T_end)
        it += 1
        
        if (it % 5000 == 0):
            print(f"  [{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} focus={focus:.1f} eta={eta:.3f} resets={resets}")
    
    return best_a, best_score


def _optimize_anti_pal(x0, time_limit, rng, label=""):
    """Optimize in anti-palindromic parameterization: a[i]+a[n-1-i] = 2/n."""
    x = np.array(x0, dtype=np.float64).copy()
    m = x.size
    n = 2 * m
    x = np.clip(x, 1e-8, 1.0 - 1e-8)
    
    def _a_from_x(xx):
        t = 1.0 / m
        aa = np.empty(n)
        aa[:m] = t * xx
        aa[m:] = t * (1.0 - xx[::-1])
        return aa
    
    a = _a_from_x(x)
    best_x = x.copy()
    best_score = _score(a)[0]
    b = _fft_square(a)
    score = best_score
    
    t0 = time.monotonic()
    focus = 5.0
    focus_inc = 1.008
    focus_max = 300.0
    eta = 0.3
    eta_min = 0.03
    no_improve = 0
    max_no = 800
    resets = 0
    max_resets = 20
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if b is None:
            b = _fft_square(a)
            score, _, _ = _score(a, b)
        
        if score < best_score - 1e-14:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.3)
            x = 0.6 * best_x + 0.4 * rng.uniform(0.2, 0.8, size=m)
            x = np.clip(x, 1e-8, 1.0 - 1e-8)
            a = _a_from_x(x)
            b = None
            score = float('inf')
            focus = float(rng.choice([3.0, 8.0, 20.0, 50.0]))
            eta = 0.3
            continue
        
        p = _make_weights(b, focus)
        grad_full = _fft_grad_conv(p, a)
        # Project gradient to x-space: dx_i = grad[i] - grad[n-1-i] (for anti-pal)
        gx = grad_full[:m] - grad_full[n-1:n-1+m][::-1]
        maxg = float(np.max(np.abs(gx)))
        if maxg < 1e-16:
            break
        gx = gx / maxg
        
        x_new = x - eta * gx
        x_new = np.clip(x_new, 1e-8, 1.0 - 1e-8)
        
        if not np.any(x_new != x):
            x_new = np.clip(x + rng.normal(0, 0.05, size=m), 1e-8, 1.0 - 1e-8)
        
        a_new = _a_from_x(x_new)
        b_new = _fft_square(a_new)
        score_new, _, _ = _score(a_new, b_new)
        
        accept = (score_new < best_score - 1e-14) or (score_new < score - 1e-14)
        
        if accept:
            x = x_new
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.04, 3.0)
        else:
            eta = max(eta * 0.78, eta_min)
            if eta < 0.08:
                x = np.clip(x + rng.normal(0, 0.08, size=m), 1e-8, 1.0 - 1e-8)
                a = _a_from_x(x)
                b = None
                score = float('inf')
        
        focus = min(focus * focus_inc, focus_max)
        it += 1
    
    return _a_from_x(best_x), best_score


def _basin_hopping(a0, time_limit, rng, inner_time=5.0, label=""):
    """Basin hopping: perturb then do short local optimization."""
    a = np.array(a0, dtype=np.float64).copy()
    a = _normalize(a)
    n = a.size
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = _score(a)[0]
    T = 0.5  # SA temperature
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Generate perturbation
        perturb_scale = 0.1 + 0.3 * T
        a_pert = a * np.exp(rng.normal(0, perturb_scale, size=n))
        a_pert = _normalize(a_pert)
        
        # Short local optimization
        a_local, s_local = _optimize_full(a_pert, inner_time, rng, label=f"{label}-inner")
        
        current_score = _score(a)[0]
        
        # Accept: always accept improvements, sometimes accept worse (SA)
        if s_local < best_score - 1e-14:
            accept = True
        elif s_local < current_score - 1e-10:
            accept = True
        elif T > 0.01:
            delta = s_local - current_score
            if delta < 0:
                accept = True
            else:
                accept = (rng.random() < math.exp(-delta / (T * 0.01 + 1e-10)))
        else:
            accept = (s_local < current_score - 1e-10)
        
        if accept:
            a = a_local
            if s_local < best_score:
                best_score = s_local
                best_a = a.copy()
        
        # Cool
        T *= 0.995
        it += 1
        
        if it % 50 == 0:
            print(f"  [{label}] bounce={it} t={time.monotonic()-t0:.1f}s "
                  f"score={current_score:.6f} best={best_score:.6f} T={T:.4f}")
    
    return best_a, best_score


def _topk_descent(a0, time_limit, rng, label=""):
    """Targeted descent: repeatedly identify top convolution peaks and do local adjustments."""
    a = np.array(a0, dtype=np.float64).copy()
    a = _normalize(a)
    n = a.size
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = _score(a)[0]
    b = _fft_square(a)
    score = best_score
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if b is None:
            b = _fft_square(a)
            score, _, _ = _score(a, b)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Find top-5 lags
        top_lags = np.argsort(b)[-5:]
        # For each top lag, identify contributing indices
        # Try small perturbations at those indices
        improved = False
        for lag in top_lags:
            lag = int(lag)
            # c[lag] = sum_i a[i]*a[lag-i]
            # Indices: i from max(0, lag-(n-1)) to min(n-1, lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(n - 1, lag)
            if i_max < i_min:
                continue
            # Try reducing a[i] and a[lag-i] slightly for the largest contributing pair
            contrib = a[i_min:i_max+1] * a[lag-i_max:lag-i_min+1][::-1]
            if contrib.size > 0:
                j_best = i_min + int(np.argmax(contrib))
                k_best = lag - j_best
                if 0 <= k_best < n:
                    # Try reducing a[j_best]
                    for factor in [0.9, 0.8, 0.7]:
                        a_try = a.copy()
                        a_try[j_best] *= factor
                        a_try[k_best] *= (1.0 + (1.0 - factor) * a[j_best] / max(a[k_best], 1e-300))
                        a_try = _normalize(a_try)
                        b_try = _fft_square(a_try)
                        s_try = _score(a_try, b_try)[0]
                        if s_try < score - 1e-12:
                            a = a_try
                            b = b_try
                            score = s_try
                            improved = True
                            break
        
        if not improved:
            # Random small perturbation
            idx = rng.choice(n, size=max(1, n // 100), replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.2, size=len(idx)))
            a = _normalize(a)
            b = None
            score = float('inf')
            # If too many failures, do bigger jump
            if it % 200 == 0 and it > 0:
                a = best_a * np.exp(rng.normal(0, 0.3, size=n))
                a = _normalize(a)
        
        it += 1
        if it % 2000 == 0:
            print(f"  [{label}] it={it} t={time.monotonic()-t0:.1f}s score={score:.6f} best={best_score:.6f}")
    
    return best_a, best_score


def _rnd_init(n, rng, amp=2.0):
    z = rng.normal(0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    return _normalize(a)


def _anti_pal_init(n, rng, amp=2.0):
    """Initialize anti-palindromic: a[i] + a[n-1-i] = 2/n."""
    m = n // 2
    x = rng.uniform(0.3, 0.7, size=m)
    # Add some structure
    freqs = rng.choice([1, 2, 3, 5, 7, 11, 13], size=3)
    for f in freqs:
        x += 0.1 * np.sin(2 * np.pi * f * np.arange(m) / m + rng.uniform(0, 2 * np.pi))
    x = np.clip(x, 0.05, 0.95)
    
    t = 1.0 / m
    a = np.empty(n)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return _normalize(a), x


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns a list of non-negative floats minimizing the score."""
    seed = int(seed)
    total_budget = max(120.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 30.0
    
    rng = np.random.default_rng(seed)
    
    print(f"Starting optimizer: budget={total_budget:.0f}s, seed={seed}")
    
    best_a = None
    best_score = float('inf')
    
    # === Load given sequence ===
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None and len(given) > 50:
            g = np.array(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s_g = float(g.sum())
            if s_g > 0.01:
                gn = g / s_g
                sc = _score(gn)[0]
                print(f"Given sequence: n={gn.size}, internal_score={sc:.6f}")
                if sc < best_score:
                    best_score = sc
                    best_a = gn.copy()
                if gn.size <= 20000:
                    given_a = gn
    except Exception as e:
        print(f"Failed to load given sequence: {e}")
    
    if best_a is None:
        best_a = np.ones(4096) / 4096.0
        best_score = 2.0
        print("Using uniform fallback, score=2.0")
    
    # === Phase 1: Optimize given/full sequence (if we have one) ===
    if given_a is not None and time.monotonic() < deadline - 300:
        alloc = min(120.0, deadline - time.monotonic() - 300)
        if alloc > 30:
            print(f"\n--- Phase 1: Full optimization of given (n={given_a.size}), alloc={alloc:.0f}s ---")
            t0 = time.monotonic()
            try:
                a1, s1 = _optimize_full(given_a, alloc, rng, label=f"full-n{given_a.size}")
                print(f"  Result: score={s1:.6f} (took {time.monotonic()-t0:.1f}s)")
                if s1 < best_score:
                    best_score = s1
                    best_a = a1
            except Exception as e:
                print(f"  Phase 1 failed: {e}")
    
    # === Phase 2: Basin hopping on best ===
    if time.monotonic() < deadline - 200:
        alloc = min(300.0, deadline - time.monotonic() - 200)
        if alloc > 60:
            print(f"\n--- Phase 2: Basin hopping (n={best_a.size}), alloc={alloc:.0f}s ---")
            t0 = time.monotonic()
            try:
                a2, s2 = _basin_hopping(best_a, alloc, rng, inner_time=4.0, label="basin")
                print(f"  Result: score={s2:.6f} (took {time.monotonic()-t0:.1f}s)")
                if s2 < best_score:
                    best_score = s2
                    best_a = a2
            except Exception as e:
                print(f"  Phase 2 failed: {e}")
    
    # === Phase 3: Anti-palindromic search at various lengths ===
    for n_trial in [4096, 8192]:
        if time.monotonic() < deadline - 150:
            alloc = min(200.0, deadline - time.monotonic() - 150)
            if alloc < 60:
                break
            print(f"\n--- Phase 3: Anti-pal n={n_trial}, alloc={alloc:.0f}s ---")
            t0 = time.monotonic()
            try:
                _, x0 = _anti_pal_init(n_trial, rng, amp=2.5)
                a3, s3 = _optimize_anti_pal(x0, alloc, rng, label=f"anti{n_trial}")
                print(f"  Result: score={s3:.6f} (took {time.monotonic()-t0:.1f}s)")
                if s3 < best_score:
                    best_score = s3
                    best_a = a3
            except Exception as e:
                print(f"  Phase 3 (n={n_trial}) failed: {e}")
    
    # === Phase 4: Second round of full optimization on best ===
    if time.monotonic() < deadline - 100:
        alloc = min(200.0, deadline - time.monotonic() - 100)
        if alloc > 40:
            print(f"\n--- Phase 4: Full re-optimization (n={best_a.size}), alloc={alloc:.0f}s ---")
            t0 = time.monotonic()
            try:
                a4, s4 = _optimize_full(best_a, alloc, rng, label=f"refine-n{best_a.size}",
                                        T_start=0.005, T_end=0.0005)
                print(f"  Result: score={s4:.6f} (took {time.monotonic()-t0:.1f}s)")
                if s4 < best_score:
                    best_score = s4
                    best_a = a4
            except Exception as e:
                print(f"  Phase 4 failed: {e}")
    
    # === Phase 5: Top-k descent for final polish ===
    if time.monotonic() < deadline - 60:
        alloc = min(120.0, deadline - time.monotonic() - 60)
        if alloc > 30:
            print(f"\n--- Phase 5: Top-k descent (n={best_a.size}), alloc={alloc:.0f}s ---")
            t0 = time.monotonic()
            try:
                a5, s5 = _topk_descent(best_a, alloc, rng, label="topk")
                print(f"  Result: score={s5:.6f} (took {time.monotonic()-t0:.1f}s)")
                if s5 < best_score:
                    best_score = s5
                    best_a = a5
            except Exception as e:
                print(f"  Phase 5 failed: {e}")
    
    # === Phase 6: One more basin hop with smaller perturbations ===
    if time.monotonic() < deadline - 50:
        alloc = min(100.0, deadline - time.monotonic() - 50)
        if alloc > 20:
            print(f"\n--- Phase 6: Fine basin hopping, alloc={alloc:.0f}s ---")
            t0 = time.monotonic()
            try:
                a6, s6 = _basin_hopping(best_a, alloc, rng, inner_time=3.0, label="fine-basin")
                print(f"  Result: score={s6:.6f} (took {time.monotonic()-t0:.1f}s)")
                if s6 < best_score:
                    best_score = s6
                    best_a = a6
            except Exception as e:
                print(f"  Phase 6 failed: {e}")
    
    # === Final cleanup ===
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s_final = float(best_a.sum())
    if s_final < 0.01:
        best_a = np.ones(4096) / 4096.0
    else:
        best_a = best_a / s_final
    
    final_score_internal = _score(best_a)[0]
    final_seq = [float(x) for x in best_a]
    
    # Verify with actual evaluator if possible
    exact_score = _safe_evaluate(final_seq)
    print(f"\n=== FINAL ===")
    print(f"n={best_a.size}, internal_score={final_score_internal:.6f}, exact_score={exact_score:.6f}")
    
    if not np.isfinite(exact_score) and exact_score != float('inf'):
        print("WARNING: exact evaluation returned non-finite, using fallback")
        fallback = np.ones(1024) / 1024.0
        final_seq = [float(x) for x in fallback]
    
    return final_seq
