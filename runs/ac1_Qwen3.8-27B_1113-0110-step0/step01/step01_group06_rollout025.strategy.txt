def compute_gradients(a):
    """
    computes the gradient of the objective function for the sequence;
    gradient for a normalized vector to maintain scale and bounds.
    """
    if not isinstance(a, (list, np.ndarray)) or len(a) == 0:
        return np.zeros(1, dtype=np.float64), float('inf')

    candidates_length = len(a)
    # Convolution using provided method
    conv = np.convolve(a, a, mode='full')
    max_conv = conv.max()

    # Score for this convolution
    sum_a = sum(a)
    score = 2.0 * candidates_length * max_conv / (sum_a**2)

    # Gradient is the derivative of max_conv with respect to a.
    # This is not trivial, but we can use numerical methods here for the sake of the implementation.
    # Simple numerical derivative using finite differences:
    gradient = np.zeros(candidates_length, dtype=np.float64)
    for i in range(candidates_length):

        a_perturbed[i] += 0.001
        conv_perturbed = np.convolve(a_perturbed, a_perturbed, mode='full')
        max_conv_perturbed = conv_perturbed.max()
        score_perturbed = 2.0 * candidates_length * max_conv_perturbed / ((sum_a + 0.001)**2)
        gradient[i] = (score_perturbed - score) / 0.001

        # Additionally, we can use the convolution output to inform
        # the gradient's focus
    return gradient, score
