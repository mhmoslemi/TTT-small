    inits.append(np.exp(-((t_arr / n - 0.3) ** 2) / 0.01) +
                 np.exp(-((t_arr / n - 0.7) ** 2) / 0.01) + 0.1)

    for cand in inits:
        a_try = normalize_to_sum(cand, n)
        sc = fast_score(a_try)
        if sc < score_best - 1e-5:
            exact = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(exact) and exact < score_best:
                score_best = exact
                a_best = a_try.copy()
                print(f"  Init: {score_best:.8f}")

    print(f"Phase 1: score={score_best:.8f}")

    # === Phase 2: Subgradient descent with Polyak step ===
    phase2_budget = total_budget * 0.38
    phase2_end = time.monotonic() + phase2_budget

    a_work = a_best.copy()
    g_ema = np.zeros(n)
    ema_alpha = 0.78
    peak_count = 3
    min_sep = max(5, n // 300)
    polyak_c = 1.0
    f_lb = 1.0

    it = 0
    stall = 0

    while time.monotonic() < phase2_end:
        a_work = normalize_to_sum(a_work, n)
        c = fft_self_conv(a_work)
        M = c.max()
        S = a_work.sum()
        cur_score = 2.0 * n * M / (S * S)

        k_stars = find_top_peaks(c, peak_count, min_sep)
        g = compute_subgradient(a_work, c, k_stars)
        g_ema = ema_alpha * g_ema + (1.0 - ema_alpha) * g

        g_norm = np.sqrt(np.dot(g_ema, g_ema))
        if g_norm < 1e-15:
            g_norm = 1e-15
        gap = max(cur_score - f_lb, 0.003)
        eta = polyak_c * gap / (g_norm * np.sqrt(float(it + 1)))
        eta = min(eta, 0.2)

        a_work = a_work - eta * g_ema
        a_work = normalize_to_sum(a_work, n)
        it += 1

        if it % 20 == 0:
            exact = float(evaluate_sequence(a_work.tolist()))
            if np.isfinite(exact) and exact < score_best:
                score_best = exact
                a_best = a_work.copy()
                stall = 0
            else:
                stall += 1

            if stall == 30:
                peak_count = min(peak_count + 1, 8)
                ema_alpha = min(ema_alpha + 0.04, 0.93)
            elif stall == 60:
                polyak_c *= 0.75
            elif stall >= 90:
                noise_scale = 0.025 + 0.035 * rng.random()
                a_work = a_best * np.exp(rng.normal(0, noise_scale, n))
                a_work = normalize_to_sum(a_work, n)
                g_ema = np.zeros(n)
                stall = 0
                peak_count = int(rng.integers(2, 6))
                polyak_c = 1.0
                ema_alpha = 0.80
                if it % 400 < 20:
                    print(f"  P2 it={it}: best={score_best:.8f}")

    print(f"Phase 2: score={score_best:.8f}")

    # === Phase 3: Inner-peak flattening ===
    # Focus on reducing interior peaks (excluding the main lag and edges)
    phase3_budget = total_budget * 0.18
    phase3_end = time.monotonic() + phase3_budget

    a_work = a_best.copy()
    a_work = normalize_to_sum(a_work, n)
    it3 = 0
    stall3 = 0
    eta3 = 0.08

    while time.monotonic() < phase3_end:
        a_work = normalize_to_sum(a_work, n)
        c = fft_self_conv(a_work)
        M = c.max()
        S = a_work.sum()
        cur_score = 2.0 * n * M / (S * S)

        # Find the main peak lag
        main_k = int(np.argmax(c))
        # Find inner peaks to flatten
        penalty, inner_ks = inner_peaks_penalty(c, n, [main_k])

        if len(inner_ks) > 0:
            # Gradient to reduce inner peaks
            g_inner = compute_subgradient(a_work, c, inner_ks)
            a_work = a_work - eta3 * g_inner
            a_work = normalize_to_sum(a_work, n)
        it3 += 1

        if it3 % 25 == 0:
            exact = float(evaluate_sequence(a_work.tolist()))
            if np.isfinite(exact) and exact < score_best:
                score_best = exact
                a_best = a_work.copy()
                stall3 = 0
            else:
                stall3 += 1
            if stall3 >= 40:
                a_work = a_best * np.exp(rng.normal(0, 0.03, n))
                a_work = normalize_to_sum(a_work, n)
                stall3 = 0
                eta3 *= 0.8

    print(f"Phase 3: score={score_best:.8f}")

    # === Phase 4: Multiplicative mirror descent ===
    phase4_budget = total_budget * 0.22
    phase4_end = time.monotonic() + phase4_budget

    a_work = a_best.copy()
    a_work = normalize_to_sum(a_work, n)
    eta_m = 0.05
    stall_m = 0
    it_m = 0

    while time.monotonic() < phase4_end:
        a_work = normalize_to_sum(a_work, n)
        c = fft_self_conv(a_work)

        k_stars = find_top_peaks(c, 4, min_sep)

        g = np.zeros(n)
        for k in k_stars:
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            if j_max >= j_min:
                g[j_min:j_max + 1] += a_work[k - j_max: k - j_min + 1][::-1]
        g /= len(k_stars)
        g_mean = g.mean()
        if g_mean > 1e-10:
            g = g / g_mean

        # Multiplicative: decrease where g is high, increase where low
        ratio = np.maximum(g, 1e-10)
        a_work = a_work * np.power(ratio, -eta_m)
        a_work = normalize_to_sum(a_work, n)
        it_m += 1

        if it_m % 20 == 0:
            exact = float(evaluate_sequence(a_work.tolist()))
            if np.isfinite(exact) and exact < score_best:
                score_best = exact
                a_best = a_work.copy()
                stall_m = 0
                eta_m = max(eta_m * 0.96, 0.003)
            else:
                stall_m += 1
                if stall_m >= 30:
                    eta_m *= 0.6
                    stall_m = 0
                if stall_m >= 60:
                    a_work = a_best * np.exp(rng.normal(0, 0.03, n))
                    a_work = normalize_to_sum(a_work, n)
                    stall_m = 0
                    eta_m = 0.05

    print(f"Phase 4: score={score_best:.8f}")

    # === Phase 5: Second subgradient pass (refinement) ===
    phase5_budget = total_budget * 0.10
    phase5_end = time.monotonic() + phase5_budget

    a_work = a_best.copy()
    g_ema = np.zeros(n)
    ema_alpha = 0.85
    peak_count = 4
    polyak_c = 0.5
    it5 = 0
    stall5 = 0

    while time.monotonic() < phase5_end:
        a_work = normalize_to_sum(a_work, n)
        c = fft_self_conv(a_work)
        M = c.max()
        S = a_work.sum()
        cur_score = 2.0 * n * M / (S * S)

        k_stars = find_top_peaks(c, peak_count, min_sep)
        g = compute_subgradient(a_work, c, k_stars)
        g_ema = ema_alpha * g_ema + (1.0 - ema_alpha) * g

        g_norm = np.sqrt(np.dot(g_ema, g_ema))
        if g_norm < 1e-15:
            g_norm = 1e-15
        gap = max(cur_score - f_lb, 0.002)
        eta = polyak_c * gap / (g_norm * np.sqrt(float(it5 + 1)))
        eta = min(eta, 0.1)

        a_work = a_work - eta * g_ema
        a_work = normalize_to_sum(a_work, n)
        it5 += 1

        if it5 % 25 == 0:
            exact = float(evaluate_sequence(a_work.tolist()))
            if np.isfinite(exact) and exact < score_best:
                score_best = exact
                a_best = a_work.copy()
                stall5 = 0
            else:
                stall5 += 1
            if stall5 >= 50:
                a_work = a_best * np.exp(rng.normal(0, 0.02, n))
                a_work = normalize_to_sum(a_work, n)
                g_ema = np.zeros(n)
                stall5 = 0

    print(f"Phase 5: score={score_best:.8f}")

    # === Phase 6: Fine-tuning with tiny steps ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_work = a_best.copy()
        a_work = normalize_to_sum(a_work, n)
        eta_f = 0.001
        it_f = 0
        stall_f = 0

        phase6_end = time.monotonic() + min(remaining - 5, total_budget * 0.07)

        while time.monotonic() < phase6_end:
            a_work = normalize_to_sum(a_work, n)
            c = fft_self_conv(a_work)
            k_stars = [int(np.argmax(c))]
            g = compute_subgradient(a_work, c, k_stars)
            a_work = a_work - eta_f * g
            a_work = normalize_to_sum(a_work, n)
            it_f += 1

            if it_f % 30 == 0:
                exact = float(evaluate_sequence(a_work.tolist()))
                if np.isfinite(exact) and exact < score_best:
                    score_best = exact
                    a_best = a_work.copy()
                    stall_f = 0
                    if it_f % 300 < 30:
                        print(f"  Fine it={it_f}: {score_best:.8f}")
                else:
                    stall_f += 1
                if stall_f >= 80:
                    a_work = a_best * np.exp(rng.normal(0, 0.01, n))
                    a_work = normalize_to_sum(a_work, n)
                    stall_f = 0
                    eta_f *= 0.7

    # Final validation
    final_a = normalize_to_sum(a_best, n)
    final_score = float(evaluate_sequence(final_a.tolist()))
    if np.isfinite(final_score) and final_score < score_best:
        score_best = final_score
        a_best = final_a

    print(f"Final score: {score_best:.8f}")
    return [float(x) for x in a_best]
