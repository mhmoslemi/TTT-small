    """One Newton-like step to flatten the convolution."""
    n = len(a)

    c_mean = np.mean(c)

    # DFT(da) * DFT(a) ≈ -DFT(e)/2

def _next_pow2(n):

def _fft_convolve(a):

            it += 1
            continue

        g = _subgradient_vec(a, k_max)
        g_max = g.max()
        if g_max < 1e-15:
            break
        g_norm = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g_norm

        a = a - eta * momentum
        a = np.maximum(a, 1e-20)
        a = a / a.sum()

        if no_improve > 300:
            eta = max(eta * 0.97, 0.005)
        elif no_improve < 30:
            eta = min(eta * 1.005, 0.15)

        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    Uses hybrid soft-subgradient + Newton spectral flattening with diverse
    multi-harmonic and tapered initializations.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25.0

    best_overall = None
    best_overall_score = np.inf

    candidates = []

    # 1. Provided sequence
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            candidates.append(("provided", hs.copy()))
    except (NameError, Exception):
        pass

    # 2. Multi-harmonic sinusoidal (key initialization family)
    for n in [1500, 2000, 2500, 3000, 4000]:
        x = np.arange(n, dtype=np.float64)
        # 2 harmonics
        a = np.ones(n)
        a += 0.35 * np.sin(2 * np.pi * x / n + np.pi / 4)
        a += 0.15 * np.sin(4 * np.pi * x / n + np.pi / 3)
        a = np.maximum(a, 0.01)
        candidates.append((f"mh2_{n}", a))
        # 3 harmonics
        a = np.ones(n)
        a += 0.3 * np.sin(2 * np.pi * x / n + np.pi / 5)
        a += 0.15 * np.sin(4 * np.pi * x / n + np.pi / 2)
        a += 0.1 * np.sin(6 * np.pi * x / n + np.pi / 6)
        a = np.maximum(a, 0.01)
        candidates.append((f"mh3_{n}", a))
        # Single harmonic, various params
        for amp, ph in [(0.4, np.pi / 4), (0.5, np.pi / 3), (0.45, 0.7)]:
            a = 1.0 + amp * np.sin(2 * np.pi * x / n + ph)
            a = np.maximum(a, 0.01)
            candidates.append((f"sin1_{n}_{amp}", a))

    # 3. Tapered windows (anti-symmetric energy distribution)
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64) / (n - 1)
        for p in [0.7, 1.0, 1.5]:
            a = np.sin(np.pi * x) ** p + 0.08
            candidates.append((f"taper_{n}_{p}", a))
        # Asymmetric
        for p, q in [(0.5, 2.0), (2.0, 0.5), (1.0, 2.0)]:
            a = (x + 0.01) ** p * (1.0 - x + 0.01) ** q
            a = a + 0.05
            candidates.append((f"asym_{n}_{p}_{q}", a))

    # 4. Exponential decay (worked well before)
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        for rate in [0.8, 1.5, 2.5]:
            a = np.exp(-rate * x / n) + 0.1
            candidates.append((f"dec_{n}_{rate}", a))

    # 5. Structured random (low-pass)
    for n in [2000, 3000]:
        a = rng.standard_normal(n)
        A = np.fft.rfft(a)
        cutoff = 20
        A[cutoff:] = 0
        a = np.fft.irfft(A, n=n)
        a = np.abs(a) + 0.5
        candidates.append((f"lpf_{n}", a))

    # Score and rank
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        sc = _compute_score(a0)
        scored.append((sc, name, a0))

    scored.sort(key=lambda x: x[0])
    print(f"Total candidates: {len(scored)}")
    print("Top 10 initial scores:")
    for i, (sc, name, _) in enumerate(scored[:10]):
        print(f"  {i+1}. {name}: {sc:.6f} (n={len(_) if False else len(scored[i][2])})")

    # Allocate time: top candidates get more
    n_top = min(len(scored), 8)
    remaining = budget_s - 40.0
    w = np.array([(n_top - i + 1) ** 1.3 for i in range(n_top)], dtype=np.float64)
    w = w / w.sum()
    times = w * remaining

    for i in range(n_top):
        if time.monotonic() > deadline:
            break

        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(15.0, deadline - time.monotonic()))

        print(f"\n[{name}] n={len(a0)}, init={init_score:.6f}, t={t_alloc:.0f}s")

        # Split time between hybrid and additive
        t1 = t_alloc * 0.6
        t2 = t_alloc * 0.4

        a1, s1 = _optimize_hybrid(a0, t1, rng)
        ev1 = evaluate_sequence(a1.tolist())
        print(f"  hybrid: {s1:.6f} / eval={ev1:.6f}")
        if np.isfinite(ev1) and ev1 < best_overall_score:
            best_overall_score = ev1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {ev1:.6f} ***")

        # Continue from best of (a1, a0) with additive
        start2 = a1 if s1 < init_score else a0
        a2, s2 = _optimize_additive(start2, t2, rng)
        ev2 = evaluate_sequence(a2.tolist())
        print(f"  additive: {s2:.6f} / eval={ev2:.6f}")
        if np.isfinite(ev2) and ev2 < best_overall_score:
            best_overall_score = ev2
            best_overall = a2.copy()
            print(f"  *** NEW BEST: {ev2:.6f} ***")

    # Final polishing
    if best_overall is not None:
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            print(f"\nFinal polish: {remaining2:.0f}s")
            a_p, s_p = _optimize_hybrid(best_overall, remaining2 * 0.7, rng)
            ev_p = evaluate_sequence(a_p.tolist())
            print(f"  polish hybrid: {ev_p:.6f}")
            if np.isfinite(ev_p) and ev_p < best_overall_score:
                best_overall_score = ev_p
                best_overall = a_p.copy()

            rem3 = deadline - time.monotonic()
            if rem3 > 15:
                a_p2, s_p2 = _optimize_additive(best_overall, rem3, rng)
                ev_p2 = evaluate_sequence(a_p2.tolist())
                print(f"  polish add: {ev_p2:.6f}")
                if np.isfinite(ev_p2) and ev_p2 < best_overall_score:
                    best_overall_score = ev_p2
                    best_overall = a_p2.copy()

    if best_overall is not None:
        final = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\n=== FINAL: score={final:.6f}, n={len(best_overall)}, time={elapsed:.1f}s ===")
        return [float(x) for x in best_overall]
    else:
        print("FALLBACK: uniform")
        return [1.0] * 1000
