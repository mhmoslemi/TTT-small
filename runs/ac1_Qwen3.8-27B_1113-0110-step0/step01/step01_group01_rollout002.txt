"""
Search for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2
using LP-based alternating minimization, binary search, and local polish.
"""

import time
import numpy as np
import scipy.optimize as opt

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s

def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]

def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))

def lp_alternating(f, deadline, t_lr=0.1, max_iters=200):
    """
    LP-based alternating minimization.
    f normalized to sum=1. Returns (best_f, best_score).
    """
    n = len(f)
    f = np.array(f, dtype=np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s  # normalize to sum=1
    
    N = next_pow2(2 * n - 1)
    c = convolve_seq(f, N)
    best_score = float(2.0 * n * np.max(c))
    best_f = f.copy()
    
    # In the normalized-to-sum-1 setting:
    # score = 2n * max(convolve(f,f))
    # We minimize this.
    # The LP: max sum(b) s.t. (f*b)[k] <= max_c for all k, b >= 0
    # Then rescale b to sum=1 and update.
    
    A = build_A(f)  # Will need to rebuild when f changes
    
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        c = convolve_seq(f, N)
        M = float(np.max(c))
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_f = f.copy()
            print("LP iter {} score={:.9f}".format(it, best_score), flush=True)
        
        if M <= 0 or not np.isfinite(M):
            break
        
        # Build A for current f
        A = build_A(f)
        
        # Solve LP: min -sum(b) s.t. A*b <= M*ones, b >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        
        try:
            res = sco.linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs')
            if not res.success:
                break
            g0 = np.maximum(res.x, 0.0)
        except Exception:
            break
        
        s_g0 = float(g0.sum())
        if s_g0 < 1e-15:
            break
        
        # Rescale to sum=1
        g_rescaled = g0 / s_g0
        
        # Update
        f = (1.0 - t_lr) * f + t_lr * g_rescaled
        f = np.maximum(f, 0.0)
        s_f = f.sum()
        if s_f < 1e-15:
            break
        f = f / s_f
        
        it += 1
    
    return best_f, best_score
