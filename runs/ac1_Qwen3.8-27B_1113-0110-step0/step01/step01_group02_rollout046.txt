import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import math

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _build_conv_matrix_sparse(a):
    """Build the sparse matrix A such that (A @ b)[k] = (a*b)[k] for linear conv."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    return 2.0 * n * c.max() / (a.sum() ** 2)

def _lp_step(a, M, max_time=10.0):
    """Solve the LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    n = len(a)
    # Build constraint matrix
    conv_mat = _build_conv_matrix_sparse(a)
    # This is (2n-1, n)
    
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: conv_mat @ b <= M * ones(2n-1)
    b_ub = np.full(2 * n - 1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    # Solve
    t0 = time.monotonic()
    result = linprog(c_obj, A_ub=conv_mat, b_ub=b_ub, bounds=bounds, method='highs')
    t1 = time.monotonic()
    
    if result.success:
        b = np.maximum(result.x, 0)
        s = b.sum()
        if s > 1e-10:
            return b / s * n  # rescale so sum = n
    return None

    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:

    cumsum = np.cumsum(sorted_v)
    # Find rho
    rho_candidates = np.arange(1, n + 1)
    # Condition: sorted_v[rho-1] > (cumsum[rho-1] - target_sum) / rho

            #   c[k] + 2*delta*a[k-j]
            # The new c at position 2j is:
            #   c[2j] + 2*x*delta + delta^2

            #   max over k (c[k] + 2*delta*a[k-j]) for valid k, k != 2j
            #   c[2j] + 2*x*delta + delta^2
            #   max over k where k-j is NOT valid (these don't change)

            # c[k] + 2*delta*a[k-j] for k in [j, min(j+n-1, 2n-2)]
            
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            
            if k_lo <= k_hi:
                k_range = np.arange(k_lo, k_hi + 1)
                a_idx = k_range - j
                changed_vals = c[k_range] + 2.0 * delta * a[a_idx]
                # Handle k = 2j
                if 2 * j >= k_lo and 2 * j <= k_hi:
                    changed_vals[2 * j - k_lo] = c[2 * j] + 2.0 * x * delta + delta * delta
                max_changed = changed_vals.max()
            else:
                max_changed = c[2 * j] + 2.0 * x * delta + delta * delta if 2*j < 2*n-1 else 0
            
            # For k outside [k_lo, k_hi], c doesn't change
            # The overall max is max(max_changed, max of c outside the changed range)
            if k_lo > 0:
                max_before = c[:k_lo].max()
            else:
                max_before = 0
            if k_hi < 2 * n - 2:
                max_after = c[k_hi + 1:].max()
            else:
                max_after = 0
            
            new_max = max(max_changed, max_before, max_after)
            new_sum = 1.0 + delta
            new_score = 2.0 * n * new_max / (new_sum * new_sum)
            
            if new_score < best_local:
                best_local = new_score
                best_delta = delta
        
        if abs(best_delta) > 1e-12:
            a[j] = x + best_delta
            c = _fft_convolve(a)  # recomputes the full convolution (correct but slower)
            score = 2.0 * n * c.max() / (a.sum() ** 2)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        it += 1
        if it % 50000 == 0 and time.monotonic() - last_print > 10.0:
            print(f"    [coord] it={it}, score={best_score:.8f}")
            last_print = time.monotonic()

            a = np.maximum(a, 0)
            c = _fft_convolve(a)
            no_improve = 0
            it = 0
    
    # Normalize
    best_a = best_a / best_a.sum() * n  # scale so sum = n
    return best_a, best_score
