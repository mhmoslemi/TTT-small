import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_max_sum(f, M):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0. Returns g or None."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(C.shape[0])
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True})
        if result.status == 0:

    except Exception:
        pass
    return None

def _qp_flatten(f, deadline_iter=100):

    Uses scipy minimize on the QP."""

    C = _build_conv_matrix(f)
    
    def obj(g):
        fg = C @ g
        return np.sum(fg * fg) - np.sum(fg)**2 / len(fg)
    
    def obj_grad(g):
        fg = C @ g
        mean_fg = np.sum(fg) / len(fg)
        return 2.0 * C.T @ (fg - mean_fg)

    constraints = [{'type': 'eq', 'fun': lambda g: np.sum(g) - 1.0,
                    'jac': lambda g: np.ones(n)}]

    try:

                          bounds=bounds, constraints=constraints,

        pass
    return None


def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.5, tighten=1.0):
    """Alternating LP with optional M tightening."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        M_target = M * tighten
        g = _lp_max_sum(a, M_target)
        if g is None:
            g = _lp_max_sum(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        blend = max(0.05, blend * 0.995)
        if no_improve > 200 and blend <= 0.051:
            break
    
    return best_a, best_score


def _directed_peak_reduction(a, max_iters=200, deadline=None):
    """ iteratively target the single peak lag using LP. """
    a = a.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        peak_idx = int(np.argmax(c))
        peak_val = float(c[peak_idx])
        
        # Build constraint matrix excluding the peak constraint
        # We want: (a*g)[k] <= peak_val for all k != peak_idx
        # and maximize sum(g) or find feasible g
        C = _build_conv_matrix(a)
        M_arr = peak_val * np.ones(C.shape[0])
        # Relax the peak constraint slightly (allow it to stay)
        M_arr[peak_idx] = peak_val * 1.01
        
        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n
        try:
            result = linprog(c_obj, A_ub=C, b_ub=M_arr, bounds=bounds,
                             method='highs', options={'presolve': True})
            if result.status != 0:
                break
            g = result.x
        except Exception:
            break
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        
        # Blend
        a_new = 0.5 * a + 0.5 * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        score = _score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def _subgradient_v2(a_init, max_iters, deadline, n_peaks=5):
    """Vectorized subgradient with Nesterov acceleration."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.9
    lr = 0.5 / n
    lr_decay = 0.99995
    patience = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(y)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c  # S = 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 1000:
            lr *= 0.5
            patience = 0
            momentum = min(momentum + 0.01, 0.99)
        
        # Find top peak lags
        if n_peaks < len(c):
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]
        else:
            top_k = np.arange(len(c))
        
        # Compute subgradient vectorized
        s = np.zeros(n)
        j = np.arange(n)
        for k in top_k:
            k = int(k)
            idx1 = k - j
            idx2 = j - k
            m1 = (idx1 >= 0) & (idx1 < n)
            m2 = (idx2 >= 0) & (idx2 < n)
            s[m1] += y[idx1[m1]]
            s[m2] += y[idx2[m2]]
        
        s /= len(top_k)
        s_norm = float(np.linalg.norm(s))
        if s_norm > 1e-15:
            s = s / s_norm
            # Nesterov step
            a_new = a - lr * s
            a_new = np.maximum(a_new, 0.0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            # Update y
            t = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
            y = a_new + momentum * (a_new - a)
            y = np.maximum(y, 0.0)
            y = y / np.sum(y)
            a = a_new
        
        lr *= lr_decay
    
    return best_a, best_score


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def _make_initial(n, variant, rng):
    """Create diverse initial sequences."""
    t = np.linspace(0, 1, n)
    if variant == 0:
        a = 1.0 - t
    elif variant == 1:
        a = np.exp(-3.0 * t)
    elif variant == 2:
        a = (1.0 - t) * (1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 3:
        a = (1.0 - t) * (1.0 + 0.7 * np.sin(2.0 * np.pi * 5.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 4:
        a = np.abs(rng.normal(0, 1, n))
    elif variant == 5:
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05)
    elif variant == 6:
        a = t ** 2 * (3.0 - 2.0 * t)
    elif variant == 7:
        a = (1.0 - t) ** 1.5 * (1.0 + 0.4 * np.cos(2.0 * np.pi * 2.0 * t + 1.0))
        a = np.maximum(a, 0.0)
    elif variant == 8:
        a = (1.0 - t) ** 0.7
    elif variant == 9:
        a = np.exp(-2.0 * t) * (1.0 + 0.5 * np.sin(2.0 * np.pi * 4.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 10:
        a = (1.0 - t) * np.exp(-np.pi * t)
    elif variant == 11:
        a = (1.0 - t) ** 2 * (1.0 + 0.3 * np.sin(2.0 * np.pi * 7.0 * t))
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    """

    Primary: alternating LP at multiple n values.

            parent_score = _score(parent)

    except Exception:

    # === Phase 1: Alternating LP at key n values ===
    print("\n=== Phase 1: Alternating LP ===")
    n_list = [400, 500, 600, 750]

        t_budget = min(time_alloc.get(n, 300), max(100, total_deadline - time.monotonic() - 400))
