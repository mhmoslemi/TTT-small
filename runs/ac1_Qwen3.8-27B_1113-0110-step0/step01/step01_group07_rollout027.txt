"""
Improved hybrid optimizer for minimum-peak autocorrelation sequences.
Uses active-set subgradient descent with mirror steps, multi-scale refinement,
and adaptive smoothing.
"""

import time
import numpy as np

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

def fast_conv(p, L):
    """Compute linear convolution p*p using FFT of length L."""
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2*len(p)-1]
    return np.where(np.isfinite(c), c, 0.0)

def fast_score(p):
    p = normalize_positive(p)
    if p is None:
        return np.inf
    n = p.size
    L = next_pow2(2*n - 1)
    c = fast_conv(p, L)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf
    return float(2.0 * n * cm)

def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(z) / len(z)
    return e / s

def active_set_subgrad(p, L, top_m=5):
    """
    Compute subgradient of max_k (p*p)(k) using top-m active lags.
    Returns (score, subgradient, active_lags).
    """
    n = p.size
    c = fast_conv(p, L)
    cm = float(c.max())
    
    # Find top-m lags
    if top_m >= len(c):
        K = set(range(len(c)))
        weights = np.ones(len(K)) / len(K)
        active_idx = list(K)
    else:
        top_idx = np.argpartition(c, -top_m)[-top_m:]
        # Filter: only include lags that are close to max (within 50% of the gap to the next non-top)
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[top_m-1]  # the m-th largest value
        # Include all lags within threshold * 1.01 (to capture near-active lags)
        K = np.where(c >= threshold * 0.999)[0]
        if len(K) == 0:
            K = top_idx
        active_idx = list(K)
    
    # Compute subgradient: g[i] = 2 * sum_{k in K} (1/|K|) * p[k-i] (with range)
    g = np.zeros(n, dtype=np.float64)
    for k in active_idx:
        # p[k-i] is valid when 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # p[k-i] for i in [i_lo, i_hi] is p[k-i_lo] down to p[k-i_hi]
            vals = p[i_lo:i_hi+1][::-1]  # p[k-i_lo], p[k-i_lo-1], ..., p[k-i_hi]
            g[i_lo:i_hi+1] += 2.0 * vals / len(active_idx)
    
    g = np.where(np.isfinite(g), g, 0.0)
    return float(2.0 * n * cm), g, active_idx

def mirror_descent_step(p, g, eta, z):
    """
    Perform a mirror descent step in the log domain.
    z is the log-parameter, p = softmax(z).
    """
    n = p.size
    # In the log-domain (exponential family), the mirror descent update is:
    # z_new = z - eta * g (then renormalize via softmax)
    z_new = z - eta * g
    z_new = np.where(np.isfinite(z_new), z_new, -50.0)
    z_new -= z_new.mean()
    z_new = np.clip(z_new, -50.0, 50.0)
    p_new = softmax(z_new)
    return p_new, z_new

def mirror_descent_optimize(p0, duration_s, rng, eta0=0.1, top_m=5, patience=500):

    Mirror descent with adaptive step size for the active set subgradient.
    """
    p = normalize_positive(p0)
    if p is None:
        return None, np.inf
    
    n = p.size
    L = next_pow2(2*n - 1)
    deadline = time.monotonic() + max(0.0, float(duration_s))
    
    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)
    
    # Adam state
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    beta1, beta2 = 0.85, 0.999
    eps = 1e-8
    step = 0
    
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    eta = eta0
    
    while time.monotonic() < deadline:
        score, g, K = active_set_subgrad(p, L, top_m=top_m)
        
        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g
        v_buf = beta2 * v_buf + (1 - beta2) * (g * g)
        b1 = 1 - beta1 ** step
        b2 = 1 - beta2 ** step
        grad_adj = (m_buf / b1) / (np.sqrt(v_buf / b2) + eps)
        
        z -= eta * grad_adj
        z = np.where(np.isfinite(z), z, -50.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax(z)
        
        # Adaptive step size
        if no_improve > patience:
            eta = max(0.005, eta * 0.7)
            # Perturb to escape
            if no_improve > patience * 3:
                z += rng.normal(0, 0.1, n)
                z -= z.mean()
                z = np.clip(z, -50.0, 50.0)
                p = softmax(z)
                no_improve = patience  # reset counter
                m_buf *= 0.1
                v_buf *= 0.1
        
        if step % 5000 == 0:
            print(f"  MD step {step} score {score:.6f} best {best_score:.6f} eta {eta:.5f} |K|={len(K)}")
    
    return best_p, best_score

def smooth_max_score_grad(p, tau, L):
    """Compute smooth approximation of the score and its gradient."""
    n = p.size
    c = fast_conv(p, L)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    
    if not np.isfinite(cm) or cm <= 0:
        return np.inf, np.inf, np.zeros(n)
    
    z = (c - cm) / tau
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    
    value_coeff = cm + tau * np.log(expsum)
    
    # Gradient of smooth max w.r.t. p
    Wtau = tau * expz / expsum
    Wref = Wtau[::-1]
    H = np.fft.irfft(np.fft.rfft(p, L) * np.fft.rfft(Wref, L), L)
    g_p = 2.0 * H[n-1:2*n-1][::-1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)
    
    # Adjust for the constraint sum(p) = 1 (project gradient)
    mean_g = float(np.dot(p, g_p))
    grad = g_p - p * mean_g
    
    # Clip gradient
    max_g = float(np.max(np.abs(grad)))
    if max_g > 1e-2:
        grad *= (1e-2 / max_g)
    
    return float(2.0 * n * value_coeff), float(2.0 * n * cm), grad

def adam_smooth_optimize(p0, duration_s, rng, tau0=0.03, lr0=0.015, n_top=10):
    """Adam optimizer on the smooth-max objective in log domain."""
    p = normalize_positive(p0)

def cheb_basis(n, D):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1.0, 1.0)
    ang = np.arccos(t)
    out = np.empty((n, D), dtype=np.float64)
    for m in range(1, D + 1):
        out[:, m - 1] = np.cos(m * ang)
    return out

def poly_parent(theta, B, mode=0, eps=0.12):
    theta = np.asarray(theta, dtype=np.float64)
    val = B.dot(theta)
    if mode == 0:
        h = 1.0 + eps + val
    else:
        h = np.exp(np.clip(val, -8.0, 8.0))
    return normalize_positive(h)

def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m) / m
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0/m)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones(m) / m
    return out / s

"""
Optimized search for minimum-peak linear autocorrelation sequences.
Key insight: the convolution peak is NOT bounded below by 1/n (unlike sum(p^2)),
so scores well below 2 are achievable. Uses active-set subgradient descent
with Adam, multi-phase optimization, and diverse initialization.

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def compute_conv(p, L):
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * len(p) - 1]
    return np.where(np.isfinite(c), c, 0.0)


def fast_score(p):
    p = normalize_positive(p)
    if p is None:
        return np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    c = compute_conv(p, L)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf
    return float(2.0 * n * cm)


def softmax_stable(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(z) / z.size
    return e / s


def active_subgradient(p, c, top_m=10):
    """
    Compute subgradient of max_k c[k] where c = convolve(p,p).
    Uses top_m active lags. Returns (score, gradient, num_active).
    """
    n = p.size
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf, np.zeros(n), 0

    # Find active lags (those close to the max)
    sorted_c = np.sort(c)[::-1]
    # Include lags that are within 1% of the max OR the top_m
    threshold = sorted_c[min(top_m - 1, len(c) - 1)]
    K = np.where(c >= threshold - 1e-15)[0]
    if len(K) > 2 * top_m:
        K = np.argpartition(c, -top_m)[-top_m:]

    m = len(K)
    if m == 0:
        return cm, np.zeros(n), 0

    # Subgradient: g[i] = (2/m) * sum_{k in K} p[k-i] (valid range only)
    g = np.zeros(n, dtype=np.float64)
    for k in K:
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # p[k-i] for i = i_lo..i_hi gives p[k-i_lo] down to p[k-i_hi]
            idx = np.arange(i_hi, i_lo - 1, -1)
            g[i_lo:i_hi + 1] += 2.0 * p[idx] / m

    g = np.where(np.isfinite(g), g, 0.0)
    return float(2.0 * n * cm), g, m


def smooth_max_grad(p, c, tau):
    """Compute smooth max approximation gradient."""
    n = p.size
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf, np.zeros(n)

    z = (c - cm) / tau
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0

    value_coeff = cm + tau * np.log(expsum)

    Wtau = tau * expz / expsum
    Wref = Wtau[::-1]
    L = next_pow2(2 * n - 1)
    H = np.fft.irfft(np.fft.rfft(p, L) * np.fft.rfft(Wref, L), L)
    g_p = 2.0 * H[n - 1:2 * n - 1][::-1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)

    mean_g = float(np.dot(p, g_p))
    grad = g_p - p * mean_g

    max_g = float(np.max(np.abs(grad)))
    if max_g > 1e-2:
        grad *= (1e-2 / max_g)

    return float(2.0 * n * value_coeff), grad


def adam_active_set(p0, duration_s, rng, eta0=0.05, top_m=10, patience=400):
    """Adam optimizer using active-set subgradient in log domain."""
    p = normalize_positive(p0)
    if p is None:
        return None, np.inf

    n = p.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, float(duration_s))

    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    eta = eta0

    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0

    while time.monotonic() < deadline:
        c = compute_conv(p, L)
        score, g, m_act = active_subgradient(p, c, top_m=top_m)

        if not np.isfinite(score):
            break

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g
        v_buf = beta2 * v_buf + (1 - beta2) * (g * g)
        b1 = 1 - beta1 ** step
        b2 = 1 - beta2 ** step

        z -= eta * (m_buf / b1) / (np.sqrt(v_buf / b2) + eps)
        z = np.where(np.isfinite(z), z, -50.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax_stable(z)

        if no_improve > patience:
            eta = max(0.003, eta * 0.75)
            if no_improve > patience * 4:
                z += rng.normal(0, 0.08, n)
                z -= z.mean()
                z = np.clip(z, -50.0, 50.0)
                p = softmax_stable(z)
                no_improve = patience
                m_buf *= 0.1
                v_buf *= 0.1
                eta = max(0.01, eta)

        if step % 8000 == 0:
            print(f"    AS step {step} score {score:.6f} best {best_score:.6f} eta {eta:.5f} |K|={m_act}")

    return best_p, best_score


def adam_smooth(p0, duration_s, rng, tau0=0.03, lr0=0.012):
    """Adam optimizer using smooth-max objective in log domain."""
    p = normalize_positive(p0)
    if p is None:
        return None, np.inf

    n = p.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, float(duration_s))

    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    tau = float(tau0)
    lr = float(lr0)

    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0

    while time.monotonic() < deadline:
        c = compute_conv(p, L)
        smooth_s, grad = smooth_max_grad(p, c, tau)
        true_s = fast_score(p)

        if true_s < best_score - 1e-12:
            best_score = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * (grad * grad)
        b1 = 1 - beta1 ** step
        b2 = 1 - beta2 ** step

        z -= lr * (m_buf / b1) / (np.sqrt(v_buf / b2) + eps)
        z = np.where(np.isfinite(z), z, -50.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax_stable(z)

        if step % 4000 == 0:
            tau = max(0.003, tau * 0.93)
            lr = max(0.002, lr * 0.97)

        if no_improve > 2500:
            zz = np.log(np.maximum(best_p, 1e-300))
            zz = np.where(np.isfinite(zz), zz, -50.0)
            zz += rng.normal(0, 0.1, n)
            zz -= zz.mean()
            z = np.clip(zz, -50.0, 50.0)
            m_buf *= 0.1
            v_buf *= 0.1
            no_improve = 0
            tau = max(0.015, tau * 0.6)
            lr = max(0.005, lr * 0.7)

        if step % 8000 == 0:
            print(f"    SM step {step} true {true_s:.6f} best {best_score:.6f} tau {tau:.5f}")

    return best_p, best_score


def cheb_basis(n, D):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1.0, 1.0)
    ang = np.arccos(t)
    out = np.empty((n, D), dtype=np.float64)
    for m in range(1, D + 1):
        out[:, m - 1] = np.cos(m * ang)
    return out


def poly_parent(theta, B, mode=0, eps=0.12):
    theta = np.asarray(theta, dtype=np.float64)
    val = B.dot(theta)
    if mode == 0:
        h = 1.0 + eps + val
    else:
        h = np.exp(np.clip(val, -8.0, 8.0))
    return normalize_positive(h)


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m) / m
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / m)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones(m) / m
    return out / s


def random_scan(n, count, deadline, rng, B, D=10):
    best_p = None
    best_s = np.inf
    for _ in range(count):

        theta = rng.normal(0, 0.45, D)
        theta[0] = (1.0 if rng.random() < 0.5 else -1.0) * rng.uniform(0.3, 1.5)
        if D > 4:
            theta[2:] *= 0.5
        if D > 6:
            theta[4:] *= 0.3
        mode = 0 if rng.random() < 0.6 else 1
        p = poly_parent(theta, B, mode, 0.12)
        if p is None:
            continue
        s = fast_score(p)
        if np.isfinite(s) and s < best_s:
            best_s = s
            best_p = p

    # Also try modulated uniforms
    for f in [1, 2, 3, 5, 7, 11, 13]:
        for a in [0.2, 0.4, 0.6, 0.8]:
            if time.monotonic() > deadline:
                break
            mod = 1.0 + a * np.cos(2 * np.pi * f * np.arange(n) / n)
            p = normalize_positive(mod)
            if p is not None:
                s = fast_score(p)
                if np.isfinite(s) and s < best_s:
                    best_s = s
                    best_p = p

    return best_p, best_s


def local_perturb_refine(p0, duration_s, rng, n_perturb=20):
    """Try random perturbations of a good sequence and keep the best."""
    p_best = normalize_positive(p0)
    s_best = fast_score(p_best)
    n = p_best.size
    deadline = time.monotonic() + max(0.0, float(duration_s))

    for _ in range(n_perturb):
        if time.monotonic() > deadline:
            break
        # Multiplicative perturbation
        log_noise = rng.normal(0, 0.3, n)
        p_try = normalize_positive(p_best * np.exp(log_noise))
        if p_try is None:
            continue
        s_try = fast_score(p_try)
        if np.isfinite(s_try) and s_try < s_best:
            s_best = s_try
            p_best = p_try

        # Additive perturbation at random positions
        p_try2 = p_best.copy()
        idx = rng.choice(n, size=max(2, n // 50), replace=False)
        p_try2[idx] += rng.uniform(0, 0.05, size=len(idx))
        p_try2 = normalize_positive(p_try2)
        if p_try2 is not None:
            s_try2 = fast_score(p_try2)
            if np.isfinite(s_try2) and s_try2 < s_best:
                s_best = s_try2
                p_best = p_try2

    return p_best, s_best

        print(f"Phase 5: Final polish ({dur:.0f}s)")
        p5, s5 = adam_smooth(best_p, dur, rng, tau0=0.01, lr0=0.004)
        if p5 is not None:
            try_update(resize_sequence(p5, N), s5)
        print(f"  After P5: {best_s:.6f}")

    # Final validation
    candidates = []
    if best_p is not None:
        candidates.append(best_p)
    candidates.append(np.ones(4096) / 4096)

    final_seq = None
    final_score = np.inf
    for cand in candidates:
        lst = [float(x) for x in np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst

    if final_seq is None:
        final_seq = [1.0 / 4096.0] * 4096

    print(f"Final validated score: {final_score:.6f}")
    return final_seq
