```python
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = int(np.max(np.where(cond)[0])) + 1
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0)
```

```python
2. Block coordinate refinement

def get_given_sequence():
    try:
        seq = globals().get("height_sequence_1", None)
        if seq is None:
            return None
        a = np.asarray(seq, dtype=np.float64).ravel()
        if a.size == 0:

        s = float(a.sum())
        if s < 1e-15:

        return a / s

    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / a.size
    return a / s

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def convolve_self(a, N=None):
    n = a.size
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]

def score(c, n):
    return 2.0 * float(n) * float(np.max(c))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = int(np.max(np.where(cond)[0])) + 1 if np.any(cond) else n
    theta = float((cssv[rho - 1] - 1) / rho)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, c, n, N_fft, tau):
    """Compute weighted subgradient g = 2 * cross_corr(w, a) where w = softmax(c[:n]/tau)."""
    c_half = c[:n]
    z = c_half / tau
    z -= float(np.max(z))
    w = np.exp(z)
    ws = float(w.sum())
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    
    # g[i] = 2 * sum_{k=i}^{n-1} w[k] * a[k-i]
    # = 2 * (w_rev * a)[n-1-i]
    w_rev = w[::-1]
    # Linear convolution via FFT
    Aw = np.fft.rfft(a, N_fft)
    Ww = np.fft.rfft(w_rev, N_fft)
    conv = np.fft.irfft(Aw * Ww, N_fft)[:2 * n - 1]
    h = conv[n - 1::-1]  # h[i] = conv[n-1-i]
    return 2.0 * h

def nesterov_subgradient(a0, n, deadline, tau_init=0.02, step_init=0.3, 
                          decay=0.9999, restart_interval=2000, seed=0):
    """Nesterov-accelerated projected subgradient method."""
    a = a0.copy()
    N_fft = next_pow2(2 * n - 1)
    
    best_a = a.copy()
    c = convolve_self(a, N_fft)
    best_score = score(c, n)
    
    t = 1
    step = step_init
    tau = tau_init
    x_prev = a.copy()
    momentum = (t - 1) / (t + 2)
    
    rng = np.random.default_rng(seed)
    
    while time.monotonic() < deadline:
        # Nesterov extrapolation
        if t > 1:
            y = a + momentum * (a - x_prev)
            y = project_simplex(y)
        else:
            y = a
        
        x_prev = a.copy()
        
        # Compute gradient at y
        c_y = convolve_self(y, N_fft)
        g = compute_gradient(y, c_y, n, N_fft, tau)
        
        # Gradient step
        a_new = y - step * g
        a_new = project_simplex(a_new)
        
        # Evaluate
        c_new = convolve_self(a_new, N_fft)
        sc_new = score(c_new, n)
        
        if sc_new < best_score * (1 - 1e-10):
            best_score = sc_new
            best_a = a_new.copy()
        
        # Adaptive step size
        sc_y = score(c_y, n)
        if sc_new < sc_y:
            step = min(2.0, step * 1.02)
        else:
            step = max(0.005, step * 0.7)
        
        # Update momentum
        t += 1
        momentum = (t - 1) / (t + 2)
        
        # Decay tau
        tau = max(0.001, tau * decay)
        
        # Periodic restart to escape local minima
        if t % restart_interval == 0:
            # Add small random perturbation
            a_new = a_new + 0.01 * rng.standard_normal(n)
            a_new = project_simplex(a_new)
            c_restart = convolve_self(a_new, N_fft)
            if score(c_restart, n) < best_score:
                best_score = score(c_restart, n)
                best_a = a_new.copy()
            a = a_new
            step = step_init * 0.5
            # Occasionally increase tau (annealing restart)
            if rng.random() < 0.3:
                tau = tau_init
            t = 1
            momentum = 0.0
            x_prev = a.copy()
        
        a = a_new
    
    return best_a, best_score, t
```

```python
"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated projected subgradient descent with FFT-based softmax-weighted gradients
2. Block coordinate refinement with local subgradient polishing
3. Multi-start, multi-length search with adaptive budget and annealing restarts
"""

import time
import numpy as np

def get_given_sequence():
    try:
        seq = globals().get("height_sequence_1", None)
        if seq is None:
            return None
        a = np.asarray(seq, dtype=np.float64).ravel()
        if a.size == 0:
            return None
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s < 1e-15:
            return None
        return a / s
    except Exception:
        return None

def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / a.size
    return a / s

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def convolve_self(a, N):
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * a.size - 1]

def project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.where(cond)[0])) + 1
    theta = float((cssv[rho - 1] - 1.0) / rho)
    return np.maximum(v - theta, 0.0)

def compute_grad_fft(a, c, n, N, tau):
    c_half = c[:n]
    z = c_half / tau
    z -= float(np.max(z))
    w = np.exp(z)
    ws = float(w.sum())
    if ws < 1e-30:
        return np.zeros(n, dtype=np.float64)
    w = w / ws
    w_rev = w[::-1]
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)[:2 * n - 1]
    h = conv[n - 1::-1]
    return 2.0 * h

def run_nesterov_subgrad(a0, n, deadline, tau_init=0.03, step_init=0.5, seed=0):
    a = a0.copy()
    N = next_pow2(2 * n - 1)
    best_a = a.copy()
    c = convolve_self(a, N)
    best_score = 2.0 * n * float(np.max(c))
    t = 1
    step = step_init
    tau = tau_init
    x_prev = a.copy()
    rng = np.random.default_rng(seed)
    stall_count = 0
    stall_limit = 8000
    restart_count = 0

    while time.monotonic() < deadline:
        if t > 1:
            mu = (t - 1) / (t + 2.0)
            y = a + mu * (a - x_prev)
            y = project_simplex(y)
        else:
            y = a
        x_prev = a.copy()

        c_y = convolve_self(y, N)
        gy = compute_grad_fft(y, c_y, n, N, tau)
        a_new = y - step * gy
        a_new = project_simplex(a_new)

        c_new = convolve_self(a_new, N)
        sc_new = 2.0 * n * float(np.max(c_new))
        sc_y = 2.0 * n * float(np.max(c_y))

        if sc_new < best_score * (1.0 - 1e-10):
            best_score = sc_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        if sc_new <= sc_y * 1.001:
            step = min(3.0, step * 1.015)
        else:
            step = max(0.01, step * 0.65)

        t += 1
        tau = max(0.005, tau * 0.99995)

        if stall_count >= stall_limit:
            restart_count += 1
            pert = 0.02 * restart_count
            a_new = a_new + pert * rng.standard_normal(n)
            a_new = project_simplex(a_new)
            c_r = convolve_self(a_new, N)
            sc_r = 2.0 * n * float(np.max(c_r))
            if sc_r < best_score:
                best_score = sc_r
                best_a = a_new.copy()
            a = a_new
            step = step_init
            tau = max(tau * 2.0, tau_init * 0.5)
            t = 1
            x_prev = a.copy()
            stall_count = 0
            if restart_count > 30:
                break

        a = a_new

    return best_a, best_score

def block_refine(a, n, deadline, block_size=64, alpha=0.15, seed=0):
    a = a.copy()
    N = next_pow2(2 * n - 1)
    c = convolve_self(a, N)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    tau = 0.015
    step = 0.3

    while time.monotonic() < deadline:
        c = convolve_self(a, N)
        sc = 2.0 * n * float(np.max(c))
        if sc < best_score * (1.0 - 1e-10):
            best_score = sc
            best_a = a.copy()

        s = int(rng.integers(0, max(1, n - block_size + 1)))
        e = min(n, s + block_size)
        m = e - s
        if m < 2:
            continue
        c_half = c[:n]
        z = c_half / tau
        z -= float(np.max(z))
        w = np.exp(z)
        ws = float(w.sum())
        if ws < 1e-30:
            continue
        w = w / ws

        g_block = np.zeros(m, dtype=np.float64)
        for i in range(m):
            gi = 0.0
            ai = a[s + i]
            for k in range(n):
                wi = float(w[k])
                if wi < 1e-12:
                    continue
                j = k - (s + i)
                if 0 <= j < n:
                    gi += 2.0 * wi * float(a[j])
            g_block[i] = gi

        g_block = g_block - g_block.mean()
        old_block = a[s:e].copy()
        new_block = old_block - alpha * step * g_block
        new_block = np.maximum(new_block, 0.0)
        rem_sum = 1.0 - float(new_block.sum())
        rem_idx = np.concatenate([np.arange(0, s), np.arange(e, n)])
        if rem_idx.size > 0 and rem_sum > 0:
            rem = a[rem_idx]
            rem_s = float(rem.sum())
            if rem_s > 1e-15:
                new_rem = rem * (rem_sum / rem_s)
                new_rem = np.maximum(new_rem, 0.0)
                new_rem_s = float(new_rem.sum())
                if new_rem_s > 1e-15:
                    new_rem = new_rem * (rem_sum / new_rem_s)
                    a[rem_idx] = new_rem
        a[s:e] = new_block
        a = project_simplex(a)

        c_new = convolve_self(a, N)
        sc_new = 2.0 * n * float(np.max(c_new))
        if sc_new > sc * 1.02:
            a[s:e] = old_block
            a = project_simplex(a)
            step = max(0.01, step * 0.8)
        else:
            step = min(1.5, step * 1.03)
            tau = max(0.004, tau * 0.999)

    return best_a, best_score

def resample_seq(a, n_target):
    a = np.asarray(a, dtype=np.float64).ravel()
    old_n = a.size
    n_target = max(2, int(n_target))
    if old_n == 1:
        return np.full(n_target, 1.0 / float(n_target), dtype=np.float64)
    if old_n == n_target:
        return normalize_sum1(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, n_target)
    b = np.interp(new_x, old_x, a)
    return normalize_sum1(b)

def make_structured_init(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    candidates = []
    for m in range(1, 8):
        for amp in (0.2, 0.4, 0.6, 0.8):
            for phase in (0.0, np.pi / 4, np.pi / 2):
                v = 1.0 + amp * np.sin(2.0 * np.pi * m * x + phase)
                candidates.append(normalize_sum1(np.maximum(v, 1e-10)))
    for mu in (0.3, 0.4, 0.5, 0.6, 0.7):
        for sig in (0.1, 0.15, 0.25, 0.35):
            candidates.append(normalize_sum1(np.exp(-((x - mu) ** 2) / (2 * sig * sig)) + 0.01))
    for sl in (-2, -1, -0.5, 0.5, 1, 2):
        candidates.append(normalize_sum1(np.exp(sl * (x - 0.5) * 4) + 0.01))
    for _ in range(6):
        v = np.ones(n)
        for m in range(1, 7):
            v += rng.uniform(0.05, 0.35) / (m ** 0.6) * np.sin(2 * np.pi * m * x + rng.uniform(0, 2 * np.pi))
        candidates.append(normalize_sum1(np.maximum(v, 1e-10)))
    return candidates
```
