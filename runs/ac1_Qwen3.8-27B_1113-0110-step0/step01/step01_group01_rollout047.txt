def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]

def recursive_init(n, target_score=1.5):
    """Use recursive formula to build a sequence with approximately flat convolution."""
    n = max(8, int(n))
    s = np.sqrt(2.0 * n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    a[0] = 1.0 / s
    target = 1.0 / (2.0 * n - 1.0)
    for k in range(1, n):
        val = target
        # Subtract contributions from pairs (i, k-i) with both i < k
        for i in range(1, k):
            if i < n and (k - i) < n:
                val -= 2.0 * a[i] * a[k - i] if i != k - i else a[i] * a[k - i]
            elif i < n:
                val -= 2.0 * a[i] * 0.0

def peak_gradient(a, c, top_k=None, tau_frac=0.005):
    """
    Compute gradient focused on the top-k peaks of the convolution.
    Uses a very sharp softmax so weights concentrate on the actual peaks.
    Returns gradient w.r.t. a (for mirror descent in log-space).
    """
    n = a.size
    cmax = float(np.max(c))
    if cmax <= 0 or not np.isfinite(cmax):
        return np.zeros(n, dtype=np.float64)
    if top_k is not None and top_k < len(c):
        sorted_idx = np.argsort(c)[::-1][:top_k]
        mask = np.zeros(len(c), dtype=np.float64)
        mask[sorted_idx] = 1.0
        c_masked = c * mask
        tau = max(tau_frac * cmax, 1e-15)
    else:
        c_masked = c
        tau = max(tau_frac * cmax, 1e-15)
    z = (c_masked - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(len(c), dtype=np.float64)
        w[np.argmax(c)] = 1.0
    else:
        w = ez / s
    # Gradient: dC(k)/da_j = a_{k-j} (for valid indices)
    # In FFT: grad_j = 2 * (w_convolved_with_a)_j
    N = next_pow2(3 * n - 2)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w[::-1], N)
    grad = np.fft.irfft(fa * fw, N)
    return 2.0 * grad[n - 1:2 * n - 1][::-1]

            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / float(a_try.sum())
            a_try = normalize_positive(a_try, floor=1e-15)
            score_try = internal_score(a_try, N)
            if np.isfinite(score_try) and score_try < score * (1.0 - 1e-10):
                accepted = True
                a = a_try
                break

        if accepted:
            eta = min(1.5, eta * 1.03)
            tau_frac = max(0.001, tau_frac * 0.997)
        else:
            stagnant += 1
            eta = max(1e-5, eta * 0.55)
            tau_frac = min(0.08, tau_frac * 1.3)
            if stagnant >= 150:
                # Random perturbation to escape
                rng = np.random.default_rng(it * 7 + n)
                noise = rng.normal(0, 0.08, n)
                a_try = a * np.exp(noise)
                a_try = normalize_positive(a_try, floor=1e-15)
                sc_try = internal_score(a_try, N)
                if np.isfinite(sc_try) and sc_try < best_score * 1.1:
                    a = a_try
                stagnant = 0
                eta = 0.5
                tau_frac = 0.02
            if eta < 1e-6:
                break

        it += 1
        if verbose and it % 100 == 0:
            print("  mirror n={} iter={} best={:.9f} score={:.9f}".format(n, it, best_score, score), flush=True)
        if time.monotonic() - last_improve > 60.0:
            # No improvement for 60 seconds, break to save time for other methods
            break

    return best_a, best_score


def peak_local_search(a, deadline, verbose=False):
    """
    Local search that directly targets the peak convolution coefficients.
    At each step, identify the peak, find the contributing pairs, and
    redistribute mass to reduce the peak.
    """
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)
    best_a = a.copy()
    best_score = internal_score(a, N)
    c = convolve_seq(a, N)
    rng = np.random.default_rng(42)
    stagnant = 0

    while time.monotonic() < deadline:
        c = convolve_seq(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Find the peak position
        k_peak = int(np.argmax(c))

        # Find contributing pairs (i, k_peak - i)
        i_min = max(0, k_peak - (n - 1))
        i_max = min(n - 1, k_peak)
        if i_max < i_min:
            break
        indices = np.arange(i_min, i_max + 1)
        partners = k_peak - indices

        # Find the pair with the largest product
        products = a[indices] * a[partners]
        if products.size == 0:
            break
        # Try reducing the largest contributor
        order = np.argsort(products)[::-1]
        tried = 0
        accepted = False
        for idx in order[:min(5, order.size)]:
            i = int(indices[idx])
            j = int(partners[idx])
            if i == j:
                continue
            # Try moving mass from the larger element to a random small element
            if a[i] <= a[j]:
                src, dst_candidate = j, i
            else:
                src, dst_candidate = i, j

            # Find a "cold" position to move mass to
            # Prefer positions that don't contribute much to the peak
            all_pos = np.arange(n)
            # Score positions by their contribution to the peak (lower = colder)
            contrib = np.zeros(n, dtype=np.float64)
            for pos in range(n):
                # pos contributes to peak if (pos + k_peak) is valid or (k_peak - pos) is valid
                p1 = k_peak - pos
                p2 = pos  # when k_peak - pos < 0, check pos + ... 
                if 0 <= p1 < n:
                    contrib[pos] += a[pos] * a[p1]
                p3 = pos - k_peak
                if 0 <= p3 < n:
                    contrib[pos] += a[pos] * a[p3]

            # Pick a cold position
            cold_candidates = all_pos[contrib < 0.3 * np.median(contrib + 1e-20)]
            if cold_candidates.size == 0:
                cold_candidates = all_pos[np.argsort(contrib)[:n // 4]]
            dst = int(rng.choice(cold_candidates))
            if dst == src or dst == dst_candidate:
                continue

            # Move a small fraction of mass from src to dst
            frac = 0.05 + 0.05 * rng.random()
            delta = frac * a[src]
            a_try = a.copy()
            a_try[src] -= delta
            a_try[dst] += delta
            a_try = normalize_positive(a_try, floor=1e-15)
            sc_try = internal_score(a_try, N)
            if np.isfinite(sc_try) and sc_try < score * (1.0 - 1e-12):
                a = a_try
                accepted = True
                tried += 1
                break

        if not accepted and tried == 0:
            # Try a random perturbation
            mask = np.zeros(n, dtype=np.float64)
            width = max(2, n // 50)
            start = int(rng.integers(0, n - width))
            mask[start:start + width] = 1.0
            noise = rng.normal(0, 0.1, n) * mask
            a_try = a * np.exp(noise)
            a_try = normalize_positive(a_try, floor=1e-15)
            sc_try = internal_score(a_try, N)
            if np.isfinite(sc_try) and sc_try < score * 1.01:
                a = a_try

        if stagnant > 300:
            break

    return best_a, best_score


def optimize_lowdim_smooth(n, deadline, seed, initial_a=None):
    """Optimize using a low-dimensional smooth parameterization."""
    n = max(32, int(n))
    rng = np.random.default_rng(int(seed))

    # Basis: power-law decay + oscillations
    x = np.linspace(0, 1, n, dtype=np.float64)
    B_list = [
        np.ones(n, dtype=np.float64),
        (1 - x),
        (1 - x) ** 2,
        (1 - x) ** 3,
        np.sin(2 * np.pi * x),
        np.sin(4 * np.pi * x),
        np.cos(2 * np.pi * x),
        x * (1 - x),
    ]
    # Add a Gaussian
    for mu, sig in [(0.2, 0.1), (0.5, 0.2), (0.8, 0.1)]:
        B_list.append(np.exp(-((x - mu) ** 2) / (2 * sig ** 2)))
    B = np.column_stack(B_list)
    D = B.shape[1]

    N = next_pow2(3 * n - 2)

    def objective(theta):
        a = B @ theta
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 1e-15:
            return 1e6
        a = a / s
        c = convolve_seq(a, N)
        return float(2.0 * n * np.max(c))

    best_theta = None
    best_score = np.inf

    starts = []
    if initial_a is not None:
        a0 = normalize_positive(initial_a, floor=1e-12)
        if a0.size == n:
            # Project onto basis
            theta0, _, _, _ = np.linalg.lstsq(B, a0, rcond=None)
            starts.append(np.maximum(theta0, 0).copy())

    for _ in range(4):
        starts.append(rng.uniform(0, 1, D))

    for theta0 in starts:
        if time.monotonic() > deadline:
            break
        theta = theta0.copy()
        sc = objective(theta)
        if sc < best_score:
            best_score = sc
            best_theta = theta.copy()

        if _scipy_optimize is not None:
            try:
                res = _scipy_optimize.minimize(
                    objective, theta, method='Nelder-Mead',
                    options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-8}
                )
                if res.x is not None:
                    sc2 = objective(res.x)
                    if sc2 < best_score:
                        best_score = sc2
                        best_theta = res.x.copy()
            except Exception:
                pass

    if best_theta is None:
        return make_uniform(n), internal_score(make_uniform(n))
    a_out = B @ best_theta
    a_out = np.maximum(a_out, 0)
    return normalize_positive(a_out, floor=0.0), best_score


def binary_search(a0, deadline, seed=0):
    """Binary (0/1) search with pressure-based swaps."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = max(16, int(a0.size))
    m = max(2, min(n - 2, int(round(0.5 * n))))

    # Initialize from a0: take top m values
    order = np.argsort(a0, kind='stable')
    b = np.zeros(n, dtype=np.float64)
    b[order[-m:]] = 1.0
    N = next_pow2(3 * n - 2)
    c = convolve_seq(b, N)
    c = np.rint(c)  # Should be integer for 0/1 sequence

    best_b = b.copy()
    best_score = float(2.0 * n * np.max(c) / (m * m))

    rng = np.random.default_rng(int(seed))
    temp = 0.03
    stale = 0

    while time.monotonic() < deadline:
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax / (m * m))
        if score < best_score * (1.0 - 1e-12):
            best_score = score
            best_b = b.copy()
            stale = 0

        # Compute pressure: which positions to flip
        # For a 1 at position i, removing it changes C by -2*b at positions i..i+n-1 (plus diagonal)
        # For a 0 at position j, adding it changes C by +2*b at positions j..j+n-1
        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        # Score: for each 1, how much does removing it reduce the max?
        # For each 0, how much does adding it reduce the max?
        # Approximate: use the local contribution
        # Remove 1 at i: delta C at position k is -2*b_{k-i} (for valid k-i)
        # The "cost" of having a 1 at i is the max of 2*b over the affected region

        # Simplified: pick a 1 that is in a "hot" region and a 0 in a "cold" region
        # Hot region for 1s: positions where the convolution through them is high
        # For a 1 at position i, it contributes to C[k] for k in [i, i+n-1]
        # The contribution to C[k] is 2*b_{k-i} (plus b_i^2 = 1 for k=2i)

        # Use a simpler heuristic: pick the 1 with highest "pressure" (contributes to most peaks)
        # and the 0 with lowest pressure
        pressures_1 = np.zeros(one_idx.size, dtype=np.float64)
        for xi in range(one_idx.size):
            i = one_idx[xi]
            # This 1 contributes to C[i:i+n] (roughly)
            lo = max(0, i)
            hi = min(2 * n - 1, i + n)
            pressures_1[xi] = np.max(c[lo:hi])

        pressures_0 = np.zeros(zero_idx.size, dtype=np.float64)
        for xj in range(zero_idx.size):
            j = zero_idx[xj]
            lo = max(0, j)
            hi = min(2 * n - 1, j + n)
            pressures_0[xj] = np.max(c[lo:hi])

        # Pick from high-pressure 1s and low-pressure 0s with some randomness
        top_p = int(max(1, 0.3 * pressures_1.size))
        bottom_p = int(max(1, 0.3 * pressures_0.size))
        hot_ones = one_idx[np.argsort(pressures_1)[::-1][:top_p]]
        cold_zeros = zero_idx[np.argsort(pressures_0)[:bottom_p]]

        i = int(rng.choice(hot_ones))
        j = int(rng.choice(cold_zeros))
        if i == j:
            continue

        # Try the swap
        two_b = 2.0 * b
        c[i:i + n] -= two_b
        c[2 * i] -= 1.0
        c[j:j + n] += two_b
        c[2 * j] += 1.0
        b[i] = 0.0
        b[j] = 1.0
        two_b = 2.0 * b

        new_max = float(np.max(c))
        new_score = float(2.0 * n * new_max / (m * m))

        accept = False
        if new_score < score * (1.0 - 1e-12):
            accept = True
        else:
            delta = new_score - score
            if delta < 0.5 and rng.random() < np.exp(-delta / max(temp, 1e-6)):
                accept = True

        if accept:
            temp = max(0.005, temp * 0.997)
        else:
            # Revert
            c[j:j + n] -= two_b
            c[2 * j] -= 1.0
            c[i:i + n] += two_b
            c[2 * i] += 1.0
            b[j] = 0.0
            b[i] = 1.0
            two_b = 2.0 * b
            stale += 1
            temp = max(0.002, temp * 0.95)
            if stale % 200 == 0:
                temp = min(0.08, temp * 2.0)
                stale = 0

        if stale > 500:
            break

    return normalize_positive(best_b, floor=0.0), best_score


def frequency_init(n, seed=0):
    """Initialize by trying to make |FFT(a)|^2 as flat as possible."""
    rng = np.random.default_rng(int(seed))
    best_a = make_uniform(n)
    best_var = np.inf

    for trial in range(20):
        # Random phase in frequency domain
        N = next_pow2(n)
        freq = np.zeros(N // 2 + 1, dtype=np.complex128)
        # Target: flat magnitude
        mag_target = np.ones(N // 2 + 1, dtype=np.float64)
        mag_target[0] = 0  # Zero the DC for now
        phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        freq = mag_target * np.exp(1j * phase)
        a = np.fft.irfft(freq, N)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            c = convolve_seq(a)
            var_c = float(np.var(c))
            if var_c < best_var:
                best_var = var_c
                best_a = a.copy()

    return best_a
