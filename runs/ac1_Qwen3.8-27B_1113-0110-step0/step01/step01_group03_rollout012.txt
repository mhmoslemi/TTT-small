def smooth_noise(n, rng, bands=5):
   if n <=1: return np.zeros(n)
   z=rng.standard_normal(n)
   F=np.fft.rfft(z)
   k=min(len(F), max(1, bands))
   if k < len(F): F[k:] = 0.0
   h=np.fft.irfft(F, n)
   h = h - float(np.mean(h))
   return h

import numpy as np
import time


def fft_self_conv(a):
    """Fast full linear self-convolution via zero-padded FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if n == 1:
        return (a * a).astype(np.float64, copy=True)
    size = 1
    L = 2 * n - 1
    while size < L:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:L].astype(np.float64, copy=False)


def exact_self_conv(a):
    """Exact direct self-convolution, used for small diagnostic examples."""
    a = np.asarray(a, dtype=np.float64)
    return np.asarray(np.convolve(a, a), dtype=np.float64)


def normalize_search_array(a, target):
    """Keep a nonnegative, force sum=target when possible, clamp to 1000."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return a
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a.fill(float(target) / float(a.size))
        S = float(a.sum())
    else:
        a *= float(target) / S
        S = float(target)
    mx = float(np.max(a))
    if mx > 1000.0:
        a = np.minimum(a, 1000.0)
        S = float(a.sum())
        if S > 1e-12:
            a *= float(target) / S
    return a


def ranges_for_model(n, model_id):
    """Return disjoint inclusive [start, end] lag ranges for a model."""
    if n <= 0:
        return [(0, 0)]
    L = 2 * n - 1
    if model_id == "full":
        raw = [(0, L - 1)]
    elif model_id == "no0":
        raw = [(1, L - 1)] if L > 1 else [(0, L - 1)]
    elif model_id == "pos":
        raw = [(1, n - 1)] if n > 1 else [(0, L - 1)]
    elif model_id == "same":
        s = (n - 1) // 2
        e = s + n - 1
        raw = [(s, e)]
    elif model_id == "same_no_center":
        s = (n - 1) // 2
        e = s + n - 1
        c = n - 1
        raw = []
        if s <= c - 1:
            raw.append((s, c - 1))
        if c + 1 <= e:
            raw.append((c + 1, e))
        if not raw:
            raw = [(s, e)]
    elif model_id == "interior":
        raw = [(1, L - 2)] if L > 2 else [(0, L - 1)]
    else:
        raw = [(0, L - 1)]

    out = []
    for s, e in raw:
        s = int(s)
        e = int(e)
        if s < 0:
            s = 0
        if e >= L:
            e = L - 1
        if e >= s:
            out.append((s, e))
    if not out:
        out = [(0, L - 1)]
    return out


def max_over_ranges(c, ranges):
    """Maximum of c over a list of inclusive ranges."""
    c = np.asarray(c)
    mx = -np.inf
    for s, e in ranges:
        if s < 0:
            s = 0
        if e >= c.size:
            e = c.size - 1
        if e >= s:
            val = np.max(c[s:e + 1])
            if val > mx:
                mx = val
    if not np.isfinite(mx):
        return 0.0
    return float(mx)


def score_from_c(c, S, n, ranges):
    """Score from a stored convolution."""
    if n <= 0 or S < 1e-12:
        return float("inf")
    M = max_over_ranges(c, ranges)
    return float(2.0 * n * M / (S * S))


def surrogate_score_array(a, ranges):

    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 0:

    S = float(a.sum())

    c = fft_self_conv(a)
    M = max_over_ranges(c, ranges)
    return float(2.0 * n * M / (S * S))

def candidate_xs(y, mean, rng, max_val=1000.0, k_max=8):

def smooth_noise(n, rng, bands=5):
    """Smooth centered random noise using FFT low-pass."""
    if n <= 1:
        return np.zeros(n, dtype=np.float64)
    z = rng.standard_normal(n)
    F = np.fft.rfft(z)
    k = min(len(F), max(1, int(bands)))
    if k < len(F):
        F[k:] = 0.0
    h = np.fft.irfft(F, n)
    return h - float(np.mean(h))


def make_family(n, mode, rng):

    if n <= 0:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, endpoint=False)

        a = np.ones(n, dtype=np.float64)
    elif mode == 1:
        a = 1.0 + 0.6 * (2.0 * x - 1.0)
    elif mode == 2:
        a = 1.0 - 0.6 * (2.0 * x - 1.0) ** 2
    elif mode == 3:
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * x)
    elif mode == 4:
        a = 1.0 + 0.5 * np.sin(4.0 * np.pi * x + 0.3)

        h = smooth_noise(n, rng, 5)
        sd = float(np.std(h))
        if sd > 1e-12:

        else:
            h = np.zeros(n, dtype=np.float64)
        a = 1.0 + 0.6 * h
    elif mode == 6:
        a = np.ones(n, dtype=np.float64)
        a[: n // 2] += 0.8
        a[n // 2:] -= 0.4
    elif mode == 7:
        a = 1.0 + 0.8 * np.sin(np.pi * x) ** 2
    else:
        a = 1.0 + 0.5 * np.cos(6.0 * np.pi * x)

    a = np.maximum(a, 1e-9)
    return normalize_search_array(a, float(n))


def resample_array(a0, n):
    """Piecewise-linear resample to a new length."""
    a0 = np.asarray(a0, dtype=np.float64)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    m = a0.size
    if m == 0:
        return np.ones(n, dtype=np.float64)
    if m == n:
        return normalize_search_array(a0.copy(), float(n))
    if m == 1:
        return np.full(n, float(a0[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    a = np.interp(x_new, x_old, a0)
    return normalize_search_array(a, float(n))


def prepare_lengths(base):
    """Choose a small set of lengths to explore."""
    desired = [int(base), 512, 1024, 2048, 3072, 4096, 6000]
    out = []
    for L in desired:
        if 64 <= L <= 6000:
            dup = False
            for x in out:
                if abs(L - x) <= 64:
                    dup = True
                    break
            if not dup:
                out.append(L)
    out.sort()
    return out


def sort_candidates(cands):
    """Sort list of (score, array) without lambda."""
    out = []
    while cands:
        best_i = 0
        best_s = float("inf")
        for i in range(len(cands)):
            s = cands[i][0]
            if np.isfinite(s) and s < best_s:
                best_s = s
                best_i = i
        out.append(cands.pop(best_i))
    return out


def choose_mask_models(examples, default="no0"):
    """Infer which lag-range model best matches observed evaluator scores."""
    model_ids = ["full", "no0", "pos", "same", "same_no_center", "interior"]
    errors = [0.0] * len(model_ids)
    any_ex = False
    first_actual = None

    for c, S, n, actual in examples:
        if not (np.isfinite(actual) and S > 0.0 and n > 0):
            continue
        any_ex = True
        if first_actual is None:
            first_actual = float(actual)
        for i, mid in enumerate(model_ids):
            ranges = ranges_for_model(int(n), mid)
            val = score_from_c(c, S, int(n), ranges)
            if np.isfinite(val):
                errors[i] += abs(val - float(actual)) / max(1.0, abs(float(actual)))

    if not any_ex:
        return default

    best_i = int(np.argmin(errors))
    no0_i = model_ids.index("no0")
    full_i = model_ids.index("full")

    if first_actual is not None:
        if first_actual < 1.999:
            if errors[no0_i] <= 1.15 * errors[best_i]:
                best_i = no0_i
        else:
            if errors[full_i] <= 1.15 * errors[best_i]:
                best_i = full_i
    return model_ids[best_i]


def select_active(c, ranges, max_peaks=16, rel_tol=1e-3):
    """Choose active peak lags and soft weights for gradient-like moves."""
    M = -np.inf
    best_k = 0
    for s, e in ranges:
        if s >= c.size:
            continue
        e = min(e, c.size - 1)
        if e >= s:
            seg = c[s:e + 1]
            val = np.max(seg)
            if val > M:
                M = float(val)
                best_k = int(s + int(np.argmax(seg)))
    if M <= 0.0 or not np.isfinite(M):
        return [], []

    gap = max(M * rel_tol, 1e-12)
    found = []
    seen = set()
    for s, e in ranges:
        e = min(e, c.size - 1)
        if e < s:
            continue
        seg = c[s:e + 1]
        loc = np.nonzero(seg >= M - gap)[0]
        for v in loc:
            ii = int(v + s)
            if ii not in seen:
                seen.add(ii)
                found.append(ii)

    if not found:
        found = [best_k]

    if len(found) > max_peaks:
        vals = np.array([float(c[i]) for i in found])
        order = np.argsort(-vals)
        found = [int(found[j]) for j in order[:max_peaks]]

    vals = np.array([float(c[i]) for i in found])
    w = np.exp((vals - M) / gap)
    sw = float(w.sum())
    if sw <= 0.0 or not np.isfinite(sw):
        w = np.full(len(found), 1.0 / max(1, len(found)))
    else:
        w = w / sw
    return [int(i) for i in found], [float(float(wi)) for wi in w]


def gradient_polish(a0, deadline, time_budget, rng, ranges):
    """Short momentum subgradient polish on a max-lag surrogate."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    if n <= 0:
        return a, float("inf")

    a = normalize_search_array(a, float(n))
    s_cur = surrogate_score_array(a, ranges)
    best_a = a.copy()
    best_score = s_cur

    v = np.zeros(n, dtype=np.float64)
    beta = 0.85
    step = 0.12
    stop = time.monotonic() + max(1.0, float(time_budget))
    no_improve = 0

    while time.monotonic() < stop and time.monotonic() < deadline:
        c = fft_self_conv(a)
        if c.size:
            np.maximum(c, 0.0, out=c)
        M = max_over_ranges(c, ranges)
        if M <= 0.0 or not np.isfinite(M):
            break

        idxs, ws = select_active(c, ranges, 16, 1e-3)
        Ba = np.zeros(n, dtype=np.float64)
        for k, ww in zip(idxs, ws):
            jmin = max(0, k - (n - 1))
            jmax = min(n - 1, k)
            if jmax >= jmin:
                st = k - jmax
                en = k - jmin + 1
                Ba[jmin:jmax + 1] += ww * a[st:en][::-1]

        direction = -4.0 * Ba + (4.0 * M / float(n)) * 1.0
        norm = float(np.max(np.abs(direction)))
        if norm > 1.0:
            direction /= norm
        elif norm < 1e-12:
            break

        v = beta * v + direction
        vn = float(np.max(np.abs(v)))
        if vn > 1.0:
            v /= vn

        a_try = a + step * v
        a_try = np.maximum(a_try, 0.0)
        S = float(a_try.sum())
        if S < 1e-12:
            a_try = a * np.exp(rng.normal(0.0, 0.05, n))
            a_try = normalize_search_array(a_try, float(n))
        else:
            a_try = a_try * (float(n) / S)

        s_try = surrogate_score_array(a_try, ranges)
        if np.isfinite(s_try) and s_try < s_cur - 1e-12:
            a = a_try
            s_cur = s_try
            if s_cur < best_score:
                best_score = s_cur
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                step *= 0.75
                v *= 0.4
                no_improve = 0
                if step < 1e-4:
                    a = a * np.exp(rng.normal(0.0, 0.08, n))
                    a = normalize_search_array(a, float(n))
                    v = np.zeros(n, dtype=np.float64)
                    step = 0.08
                    s_cur = surrogate_score_array(a, ranges)
                    if np.isfinite(s_cur) and s_cur < best_score:
                        best_score = s_cur
                        best_a = a.copy()
    return best_a, best_score


def coordinate_search(a0, deadline, time_budget, rng, ranges):
    """
    Fast local search exploiting incremental O(n) updates to the full
    convolution after a single coordinate change.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    if n <= 0:
        return a0, float("inf")

    target = float(n)
    a = normalize_search_array(a, target)
    S = float(a.sum())
    c = fft_self_conv(a)
    if c.size:
        np.maximum(c, 0.0, out=c)
    cur_score = score_from_c(c, S, n, ranges)

    best_a = a.copy()
    best_c = c.copy()
    best_S = S
    best_score = cur_score

    stop = time.monotonic() + max(1.0, float(time_budget))
    it = 0
    no_improve = 0
    sweep_pos = 0
    recompute_count = 0
    improvements = 0
    phase_random = max(10, 3 * n)
    T = 1e-5

    while time.monotonic() < stop and time.monotonic() < deadline:
        if it < phase_random:
            p = int(rng.integers(n))
        else:
            p = sweep_pos
            sweep_pos += 1
            if sweep_pos >= n:
                sweep_pos = 0
        it += 1

        y = float(a[p])
        mean = S / float(n)
        xs = candidate_xs(y, mean, rng, 1000.0, 8 if n <= 3000 else 6)
        if not xs:
            xs = [y]
        deltas = np.asarray(xs, dtype=np.float64) - y
        if deltas.size == 0:
            no_improve += 1
            continue

        old_aff = c[p:p + n].copy()
        unaff_m = 0.0
        for rs, re in ranges:
            # left outside affected interval
            le = min(re, p - 1)
            if le >= rs:
                val = float(np.max(c[rs:le + 1]))
                if val > unaff_m:
                    unaff_m = val
            # right outside affected interval
            st = max(rs, p + n)
            if re >= st:
                val = float(np.max(c[st:re + 1]))
                if val > unaff_m:
                    unaff_m = val

        m_aff = np.zeros(len(deltas), dtype=np.float64)
        diag_global = 2 * p
        for rs, re in ranges:
            lo = max(0, rs - p)
            hi = min(n - 1, re - p)
            if hi < lo:
                continue
            seg = old_aff[lo:hi + 1]
            seg_a = a[lo:hi + 1]
            vals = seg[None, :] + 2.0 * deltas[:, None] * seg_a[None, :]
            if rs <= diag_global <= re and lo <= p <= hi:
                vals[:, p - lo] += deltas * deltas
            seg_max = vals.max(axis=1)
            if seg_max.size:
                np.maximum(seg_max, 0.0, out=seg_max)
                m_aff = np.maximum(m_aff, seg_max)

        M_new = np.maximum(unaff_m, m_aff)
        S_new = S + deltas
        scores = np.full(len(deltas), float("inf"), dtype=np.float64)
        valid = S_new > 1e-8
        if np.any(valid):
            scores[valid] = 2.0 * n * M_new[valid] / (S_new[valid] * S_new[valid])

        if not np.any(np.isfinite(scores)):
            no_improve += 1
            continue

        bi = int(np.argmin(scores))
        best_x = float(xs[bi])
        best_sc = float(scores[bi])

        accepted = False
        if abs(best_x - y) > 1e-12 and np.isfinite(best_sc):
            if best_sc < cur_score - 1e-12:
                accepted = True
            elif no_improve > n and best_sc < cur_score + T:
                dlt = max(0.0, best_sc - cur_score)
                if rng.random() < float(np.exp(-dlt / max(T, 1e-12))):
                    accepted = True

        if accepted:
            saved_a = a.copy()
            saved_c = c.copy()
            saved_S = S
            saved_cur = cur_score

            delta = best_x - y
            c[p:p + n] += 2.0 * delta * a
            c[2 * p] += delta * delta
            a[p] = best_x
            S += delta

            if S < 1e-8:
                a = saved_a
                c = saved_c
                S = saved_S
                cur_score = saved_cur
                no_improve += 1
                continue

            M = max_over_ranges(c, ranges)
            new_cur = 2.0 * n * M / (S * S)
            if not np.isfinite(new_cur):
                a = saved_a
                c = saved_c
                S = saved_S
                cur_score = saved_cur
                no_improve += 1
                continue

            a = np.maximum(a, 0.0)
            cur_score = float(new_cur)
            recompute_count += 1

            if cur_score < best_score - 1e-12:
                best_a = a.copy()
                best_c = c.copy()
                best_S = S
                best_score = cur_score
                improvements += 1
                no_improve = 0
            else:
                no_improve += 1

            if S < 0.5 * target or S > 2.0 * target:
                factor = target / S if S > 1e-12 else 1.0
                a *= factor
                c *= factor * factor
                S = target
                a = np.maximum(a, 0.0)
                if a.size and np.max(a) > 1000.0:
                    a = np.minimum(a, 1000.0)
                    c = fft_self_conv(a)
                    np.maximum(c, 0.0, out=c)
                    S = float(a.sum())
                    if S > 1e-12:
                        a *= target / S
                        c *= 1.0
                    S = target
                M = max_over_ranges(c, ranges)
                if S > 1e-12:
                    cur_score = float(2.0 * n * M / (S * S))

        else:
            no_improve += 1

        # Periodic global multiplicative perturbation to escape local minima.
        if no_improve > max(1000, 2 * n):
            saved_a = a.copy()
            saved_c = c.copy()
            saved_S = S
            saved_cur = cur_score

            sigma = 0.05 + 0.20 * ((no_improve % (2 * n)) / float(max(1, 2 * n)))
            a = a * np.exp(rng.normal(0.0, sigma, n))
            a = normalize_search_array(a, target)
            S = float(a.sum())
            c = fft_self_conv(a)
            if c.size:
                np.maximum(c, 0.0, out=c)
            M = max_over_ranges(c, ranges)
            new_cur = float(2.0 * n * M / (S * S)) if S > 1e-8 else float("inf")

            if np.isfinite(new_cur) and new_cur < best_score - 1e-12:
                best_a = a.copy()
                best_c = c.copy()
                best_S = S
                best_score = new_cur
            if np.isfinite(new_cur) and new_cur <= cur_score + 1e-9:
                cur_score = new_cur
            else:
                a = saved_a
                c = saved_c
                S = saved_S
                cur_score = saved_cur
            no_improve = 0

        # Correct accumulated floating point drift.
        recompute_count += 1
        if recompute_count >= 4000:
            recompute_count = 0
            c = fft_self_conv(a)
            if c.size:
                np.maximum(c, 0.0, out=c)
            S = float(a.sum())
            if S > 1e-12:
                a = a * (target / S)
                c = c * (target / S) ** 2
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            M = max_over_ranges(c, ranges)
            if S > 1e-8:
                cur_score = float(2.0 * n * M / (S * S))

    best_a = normalize_search_array(best_a, float(len(best_a)))
    if len(best_a) and np.max(best_a) > 1000.0:
        best_a = np.minimum(best_a, 1000.0)
        S = float(best_a.sum())
        if S > 1e-12:
            best_a *= float(len(best_a)) / S
    best_a = np.maximum(best_a, 0.0)
    return best_a, float(best_score)


def propose_candidate():
    """
    Hybrid search for a low-score nonnegative sequence:
    1) infer the effective lag-model from the evaluator on diagnostics,
    2) generate diverse smooth initial candidates in several lengths,
    3) short FFT subgradient polish,
    4) intensive incremental coordinate local search with annealing,
    5) final exact validation and returns the best sequence found.
    """
    t0 = time.monotonic()
    deadline = t0 + 2380.0
    rng = np.random.default_rng(202405)

    # Parent sequence, if available.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(1024, dtype=np.float64)

    if parent.size:
        parent = np.clip(parent, 0.0, 1000.0)
        parent = parent[np.isfinite(parent)]
    if parent.size == 0 or float(parent.sum()) < 0.01:
        parent = np.ones(1024, dtype=np.float64)

    parent_work = parent.copy()
    best_a = parent_work.copy()

    # Avoid a very expensive first exact evaluation if parent is unusually large.
    if len(best_a) <= 8000:
        best_actual = exact_score_array(best_a)
    else:
        best_actual = float("inf")
        best_a = normalize_search_array(parent_work, 4096.0)

    if not np.isfinite(best_actual):
        fallback = np.ones(1024, dtype=np.float64)
        f_val = exact_score_array(fallback)
        if np.isfinite(f_val):
            best_a = fallback
            best_actual = f_val
        else:
            best_a = fallback
            best_actual = float("inf")

    print(f"Initial exact score: {best_actual:.10f}, length={len(best_a)}")

    # Infer which lag-window model is being optimized.
    examples = []
    n_par = int(parent_work.size)
    if n_par > 0:
        if n_par <= 3000:
            c_par = exact_self_conv(parent_work)
        else:
            c_par = fft_self_conv(parent_work)
        S_par = float(parent_work.sum())
        if np.isfinite(best_actual) and S_par > 0.01:
            examples.append((c_par, S_par, n_par, best_actual))

    # Small probe examples to disambiguate models.
    if time.monotonic() < deadline - 90.0:
        probe_n = 512 if n_par > 1024 else 256
        for mode in range(5):
            if time.monotonic() > deadline - 90.0:
                break
            a_probe = make_family(probe_n, mode, rng)
            es = exact_score_array(a_probe)
            if np.isfinite(es):
                if a_probe.size <= 1000:
                    c_probe = exact_self_conv(a_probe)
                else:
                    c_probe = fft_self_conv(a_probe)
                examples.append((c_probe, float(a_probe.sum()), int(a_probe.size), es))

        if np.isfinite(best_actual) and len(best_a) > 0:
            rn = min(768, max(128, len(best_a) // 4))
            a_probe = normalize_search_array(best_a, float(rn)) if len(a_probe) == rn else resample_array(best_a, rn)
            es = exact_score_array(a_probe)
            if np.isfinite(es):
                c_probe = exact_self_conv(a_probe) if rn <= 1000 else fft_self_conv(a_probe)
                examples.append((c_probe, float(a_probe.sum()), int(a_probe.size), es))

    model_id = choose_mask_models(examples, default="no0")
    print(f"Chosen lag model: {model_id}")

    # Build candidate pool.
    lengths = prepare_lengths(parent_work.size)
    if parent_work.size > 6000:
        extras = [1024, 2048, 4096]
        for L in extras:
            if L not in lengths:
                lengths.append(L)
        lengths.sort()

    candidate_list = []
    for nC in lengths:
        if time.monotonic() > deadline - 180.0:
            break

        arrays = []
        if parent_work.size == nC:
            arrays.append(parent_work.copy())
        elif parent_work.size > 0:
            arrays.append(resample_array(parent_work, nC))

        if nC <= 2048:
            nmodes = 8
        elif nC <= 4096:
            nmodes = 6
        else:
            nmodes = 4

        for mode in range(nmodes):
            arrays.append(make_family(nC, mode, rng))

        # Symmetrized parent resample often helps centered-lag objectives.
        if parent_work.size > 0 and nC <= 4096:
            rp = resample_array(parent_work, nC)
            rp = np.maximum(rp, 0.0)
            rsym = 0.5 * (rp + rp[::-1])
            arrays.append(normalize_search_array(rsym, float(nC)))

        rangesC = ranges_for_model(nC, model_id)
        scored = []
        for a in arrays:
            if time.monotonic() > deadline - 180.0:
                break
            a = normalize_search_array(a, float(nC))
            if a.size != nC or float(a.sum()) < 0.01:
                continue
            s = surrogate_score_array(a, rangesC)
            if np.isfinite(s):
                scored.append((s, a))

        if not scored:
            continue
        scored = sort_candidates(scored)

        # Exact-validate a few top surrogate candidates.
        exact_count = 2 if nC <= 3000 else 1
        for i in range(min(exact_count, len(scored))):
            if time.monotonic() > deadline - 180.0:
                break
            a_try = scored[i][1]
            es = exact_score_array(a_try)
            if np.isfinite(es) and es < best_actual - 1e-12:
                best_actual = es
                best_a = a_try.copy()
                print(f"New exact best from candidates: {best_actual:.10f}, length={nC}")

        if nC <= 6000:
            for i in range(min(2, len(scored))):
                candidate_list.append((scored[i][0], scored[i][1]))

    candidate_list = sort_candidates(candidate_list)

    # Ensure the current exact best is available for polishing if feasible.
    if best_a is not None and 1 <= len(best_a) <= 6000:
        rangesB = ranges_for_model(len(best_a), model_id)
        sb = surrogate_score_array(best_a, rangesB)
        if np.isfinite(sb):
            present = False
            for _, a in candidate_list:
                if len(a) == len(best_a):
                    present = True
                    break
            if not present:
                candidate_list.insert(0, (sb, best_a.copy()))

    print(f"Candidates for local polishing: {len(candidate_list)}")

    polished_count = {}
    rank = 0
    for item in candidate_list:
        if time.monotonic() > deadline - 60.0:
            break
        s_item, a_item = item
        nIt = int(a_item.size)
        if nIt < 1 or nIt > 6000:
            continue
        if polished_count.get(nIt, 0) >= 2:
            continue
        polished_count[nIt] = polished_count.get(nIt, 0) + 1

        remaining = deadline - time.monotonic()
        if remaining < 80.0:
            break

        if rank == 0:
            budget = min(1200.0, max(120.0, remaining * 0.45))
        elif rank == 1:
            budget = min(450.0, max(60.0, remaining * 0.20))
        elif rank == 2:
            budget = min(250.0, max(60.0, remaining * 0.12))
        else:
            budget = min(160.0, max(45.0, remaining * 0.06))
        rank += 1

        rangesI = ranges_for_model(nIt, model_id)
        a_work = a_item.copy()

        # Short gradient polish before coordinate search.
        if budget > 200.0 and time.monotonic() + 45.0 < deadline - 120.0:
            gb = min(60.0, budget * 0.20)
            if gb > 10.0:
                a_g, _ = gradient_polish(a_work, deadline, gb, rng, rangesI)
                if a_g is not None and a_g.size == nIt:
                    a_work = normalize_search_array(a_g, float(nIt))
                es = exact_score_array(a_work)
                if np.isfinite(es) and es < best_actual - 1e-12:
                    best_actual = es
                    best_a = a_work.copy()
                    print(f"Exact best after gradient: {best_actual:.10f}, length={nIt}")
                budget = max(20.0, budget - gb)

        if time.monotonic() + 25.0 < deadline:
            a_c, _ = coordinate_search(a_work, deadline, max(20.0, budget), rng, rangesI)
            if a_c is not None and a_c.size == nIt:
                es = exact_score_array(a_c)
                if np.isfinite(es) and es < best_actual - 1e-12:
                    best_actual = es
                    best_a = a_c.copy()
                    print(f"Exact best after coordinate search: {best_actual:.10f}, length={nIt}")

        print(
            f"Finished rank {rank - 1}, length={nIt}, "
            f"global exact best={best_actual:.10f}, "
            f"time_left={max(0.0, deadline - time.monotonic()):.1f}s"
        )

    # Final polish of the globally best sequence if time remains.
    if (best_a is not None and 1 <= len(best_a) <= 6000 and
            time.monotonic() < deadline - 200.0):
        rangesB = ranges_for_model(len(best_a), model_id)
        rem = deadline - time.monotonic() - 40.0
        if rem > 40.0:
            bud = min(800.0, rem)
            a_f, _ = coordinate_search(best_a, deadline, bud, rng, rangesB)
            if a_f is not None and a_f.size == len(best_a):
                es = exact_score_array(a_f)
                if np.isfinite(es) and es < best_actual - 1e-12:
                    best_actual = es
                    best_a = a_f.copy()
                    print(f"Final polish exact best: {best_actual:.10f}")

    # A few quick random perturbations from the best if time remains.
    if best_a is not None and time.monotonic() < deadline - 30.0:
        for _ in range(5):
            rem = deadline - time.monotonic() - 30.0
            if rem < 20.0:
                break
            nB = len(best_a)
            pert = best_a * np.exp(rng.normal(0.0, 0.05, nB))
            pert = normalize_search_array(pert, float(nB))
            rangesB = ranges_for_model(nB, model_id)
            if rem > 60.0:
                a_r, _ = coordinate_search(pert, deadline, min(60.0, rem), rng, rangesB)
            else:
                a_r = pert
            es = exact_score_array(a_r)
            if np.isfinite(es) and es < best_actual - 1e-12:
                best_actual = es
                best_a = a_r.copy()
                print(f"Random restart exact best: {best_actual:.10f}")

    # Final validation and safe fallback.
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(1024, dtype=np.float64)
    best_a = normalize_search_array(best_a, float(len(best_a)))
    final_val = exact_score_array(best_a)
    if np.isfinite(final_val):
        best_actual = final_val
    else:
        fallback = np.ones(1024, dtype=np.float64)
        fv = exact_score_array(fallback)
        if np.isfinite(fv):
            best_actual = fv
            best_a = fallback
        else:
            best_a = np.ones(1024, dtype=np.float64)
            best_actual = float("inf")

    final_val = exact_score_array(best_a)
    if np.isfinite(final_val):
        best_actual = final_val

    print(f"Final returned score: {best_actual:.10f}, length={len(best_a)}")
    return [float(x) for x in best_a.tolist()]
