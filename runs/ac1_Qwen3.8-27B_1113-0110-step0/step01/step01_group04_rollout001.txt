def _fft_convolve(a):
    """Compute full linear convolution via FFT."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    A = np.fft.rfft(a, n=L)
    C = A * A
    return np.fft.irfft(C, n=L)[:2 * n - 1]


def _score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _fft_convolve(a)
    mx = c.max()
    return 2.0 * len(a) * mx


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full(len(v), 1.0 / len(v))
    return v / s


def _ratios(a):
    """Log-ratio parameterization relative to last element."""
    a = np.maximum(a, 1e-300).copy()
    a = a / a.sum()
    ref = a[-1]
    return np.clip(np.log(a[:-1] / ref), -100, 100)


def _from_ratios(x):
    """Reconstruct simplex from log-ratios to last element."""
    x = np.where(np.isfinite(x), x, 0.0)
    m = max(0.0, x.max()) if x.size else 0.0
    e = np.exp(np.clip(x - m, -700, 700))
    e_last = np.exp(-m)
    s = e.sum() + e_last
    if s < 1e-300:
        return np.full(x.size + 1, 1.0 / (x.size + 1))
    out = np.empty(x.size + 1)
    out[:-1] = e / s
    out[-1] = e_last / s
    return out
