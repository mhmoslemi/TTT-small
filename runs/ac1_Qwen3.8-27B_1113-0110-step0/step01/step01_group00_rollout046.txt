import numpy as np

def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)

    s = float(np.sum(a))

def _spectral_shaping(a_init, max_iters, deadline, verbose=True):

    Spectral method: iteratively flatten |A(f)|^2 to reduce autocorrelation peaks.
    Uses constraint projection: project onto non-negativity and sum=1.
    """

    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = _score(a)

        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, fft_size)
        spec = A * np.conj(A)  # |A|^2

        # Compute autocorrelation
        c = np.fft.irfft(spec, fft_size)[:2 * n - 1]

        # Identify peaks (top 5% of lags)
        n_peaks = max(3, (2 * n - 1) // 20)
        peak_threshold = np.sort(c)[::-1][n_peaks - 1]
        peak_mask = c > peak_threshold

        # Shape the spectrum: boost flat, suppress peaks in time domain
        # Equivalent: in frequency domain, make |A|^2 flatter
        # Use "amplitude adjustment": reduce |A| where it contributes to peaks
        # Simple approach: modify c to flatten, then recover a
        c_target = np.ones(2 * n - 1) / (2 * n - 1)
        # Blend: reduce peaks, keep valleys
        c_new = c.copy()
        c_new[peak_mask] = c_new[peak_mask] * 0.5 + c_target[peak_mask] * 0.5
        c_new[~peak_mask] = c_new[~peak_mask] * 0.9 + c_target[~peak_mask] * 0.1

        # Ensure c_new is a valid autocorrelation (positive definite spectrum)
        spec_new = np.fft.rfft(c_new, fft_size)
        spec_new = np.maximum(spec_new, 0)  # clip negative (numerical)
        # Ensure DC component is correct
        spec_new[0] = 1.0  # (sum a)^2 = 1

        # Recover a from spectrum (take square root - this gives the "flat" version)
        A_new = np.sqrt(spec_new[0] + 0j) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new * np.conj(A_new) / np.abs(A_new)[0:1] + A * 0, fft_size)[:n]
        
        # Actually, simpler: just project
        a_new = np.fft.irfft(spec_new ** 0.5 * np.exp(1j * np.angle(A[:len(spec_new)])), fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Blend with previous
        t = 0.3 if it < 20 else 0.1
        a = (1 - t) * a + t * a_new
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            break
        a = a / s

        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose and it % 50 == 0:
                print(f"    Spectral iter {it}: score={best_score:.8f}")

    return best_a, best_score


def _param_bump(n, n_bumps, rng, verbose=False):
    """Create sequence as sum of Gaussian bumps."""
    t = np.linspace(0, 1, n)
    a = np.zeros(n)
    for _ in range(n_bumps):
        mu = rng.uniform(0.1, 0.9)
        sigma = rng.uniform(0.03, 0.15)
        w = rng.uniform(0.3, 1.0)
        a += w * np.exp(-((t - mu) ** 2) / (2 * sigma ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _ramp_modulated(n, freq, amp, rng):
    """Ramp with sinusoidal modulation."""
    t = np.linspace(0, 1, n)
    a = (1 - t) * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def _perturb_and_refine(a, n_perturbs, deadline, rng):
    """Try random perturbations and refine each with CD."""
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    for p in range(n_perturbs):
        if time.monotonic() > deadline:
            break
        # Random perturbation
        noise = rng.normal(0, 0.05, n)
        a_pert = a + noise
        a_pert = np.maximum(a_pert, 0)
        s = float(np.sum(a_pert))
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        # Quick CD (few sweeps)
        a_opt, score = _coord_descent(a_pert, 30, min(deadline, time.monotonic() + 10),
                                       recompute_every=0, verbose=False)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()

    return best_a, best_score


def propose_candidate():
    """
    Multi-strategy optimizer for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses coordinate descent (exploiting per-coordinate linearity), spectral shaping,
    and parameterized initializations across multiple sequence lengths.
    """
    total_deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)

    best_a = None
    best_score = float('inf')

    # Load parent if available
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = _score(parent)
            print(f"Parent: n={parent.size}, score={parent_score:.8f}")
            best_a = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    # Phase 1: Coordinate descent on parent and various initializations
    print("\n=== Phase 1: Coordinate Descent ===")
    n_targets = [400, 500, 600, 800, 1000]

    for n in n_targets:
        if time.monotonic() > total_deadline - 100:
            break

        time_budget = min(250.0, total_deadline - time.monotonic() - 50)
        deadline_n = time.monotonic() + time_budget
        print(f"\n  --- n={n} (budget {time_budget:.0f}s) ---")

        # Build list of initializations
        inits = []
        labels = []

        if best_a is not None:
            inits.append(_interpolate_sequence(best_a, n))
            labels.append("parent")

        # Diverse initializations
        inits.append(_param_bump(n, 2, rng))
        labels.append("2bump")
        inits.append(_param_bump(n, 3, rng))
        labels.append("3bump")
        inits.append(_param_bump(n, 4, rng))
        labels.append("4bump")
        inits.append(_ramp_modulated(n, 3, 0.5, rng))
        labels.append("ramp3")
        inits.append(_ramp_modulated(n, 5, 0.6, rng))
        labels.append("ramp5")
        inits.append(_ramp_modulated(n, 7, 0.4, rng))
        labels.append("ramp7")

        # Double bump (specific good structure)
        t = np.linspace(0, 1, n)
        a_db = np.exp(-((t - 0.3) ** 2) / 0.015) + 0.6 * np.exp(-((t - 0.7) ** 2) / 0.03)
        a_db = np.maximum(a_db, 0)
        a_db = a_db / np.sum(a_db)
        inits.append(a_db)
        labels.append("dbl_bump")

        # Asymmetric ramp
        a_ramp = (1 - t) ** 1.5
        a_ramp = a_ramp / np.sum(a_ramp)
        inits.append(a_ramp)
        labels.append("ramp15")

        for label, a_init in zip(labels, inits):
            if time.monotonic() > deadline_n:
                break

            print(f"    {label}:", end=" ", flush=True)
            a_opt, score = _coord_descent(a_init, 200, deadline_n,
                                           recompute_every=25, verbose=False)
            print(f"score={score:.8f}")

            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    ** New best: {best_score:.8f} (n={n}, init={label})")

    print(f"\nAfter Phase 1: best={best_score:.8f}, n={len(best_a) if best_a is not None else 0}")

    # Phase 2: Further refinement of the best solution
    print("\n=== Phase 2: Refinement ===")
    if best_a is not None:
        n_best = len(best_a)
        refine_deadline = min(total_deadline - time.monotonic(), 300.0)

        # More aggressive CD with sequential updates
        print(f"  Sequential CD on best (n={n_best}):")
        a_opt, score = _coord_descent(best_a, 300, time.monotonic() + refine_deadline,
                                       recompute_every=0, verbose=True)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** Improved: {best_score:.8f}")

    # Phase 3: Perturbation and restart
    print("\n=== Phase 3: Perturbation & Restart ===")
    if best_a is not None and time.monotonic() < total_deadline - 50:
        pert_deadline = min(total_deadline - time.monotonic(), 200.0)
        n_best = len(best_a)
        a_opt, score = _perturb_and_refine(best_a, 20, time.monotonic() + pert_deadline, rng)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** Improved after perturbation: {best_score:.8f}")

    # Phase 4: Try spectral shaping on the best
    print("\n=== Phase 4: Spectral Shaping ===")
    if best_a is not None and time.monotonic() < total_deadline - 50:
        spec_deadline = min(total_deadline - time.monotonic(), 100.0)
        a_opt, score = _spectral_shaping(best_a, 500, time.monotonic() + spec_deadline,
                                          verbose=True)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** Improved after spectral: {best_score:.8f}")
        # Follow up with CD
        if time.monotonic() < total_deadline - 20:
            a_opt, score = _coord_descent(a_opt, 100, time.monotonic() + 50,
                                           recompute_every=10, verbose=False)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  ** CD after spectral: {best_score:.8f}")

    # Phase 5: Final polish - multiple short CD runs
    print("\n=== Phase 5: Final Polish ===")
    if best_a is not None and time.monotonic() < total_deadline - 10:
        polish_deadline = total_deadline - 5
        for restart in range(5):
            if time.monotonic() > polish_deadline:
                break
            a_try = best_a.copy()
            if restart > 0:
                noise = rng.normal(0, 0.02, len(a_try))
                a_try = np.maximum(a_try + noise, 0)
                a_try = a_try / np.sum(a_try)
            a_opt, score = _coord_descent(a_try, 100, polish_deadline,
                                           recompute_every=0, verbose=False)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  Polish {restart}: {best_score:.8f}")

    # Final
    print(f"\n=== FINAL: score={best_score:.8f}, n={len(best_a) if best_a is not None else 0} ===")
    if best_a is not None:
        result = [float(x) for x in best_a]
        verified = float(evaluate_sequence(result))
        print(f"Verified: {verified:.8f}")
        return result

    return [1.0] * 100
