def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)

def _cross_corr(w, a):
    """Compute cross-correlation (w ⋆ a)[i] = sum_j w[i+j]*a[j] for i=0..n-1."""
    n = len(a)
    m = len(w)
    # Need linear cross-correlation, zero-pad to >= m + n - 1
    size = m + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    return corr[:n]

    n = len(a)
    conv = _fft_convolve(a)
    return 2.0 * n * conv.max()

    conv = _fft_convolve(a)
    shift = conv - conv.max()
    return eps * np.log(np.sum(np.exp(shift / eps)))

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new
        
        # Check improvement on smoothed objective
        score_new = _smoothed_score(a_new, eps)
        score_old = _smoothed_score(a, eps)

                best_a = a.copy()
