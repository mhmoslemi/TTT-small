```python
"""
Hybrid optimization for non-negative coefficient sequences minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses FFT-based fast scoring, Adam optimizer
with smooth-max gradient in log-parameterization, multi-scale coarse-to-fine
approach, and diverse asymmetric initializations.
"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(z)
        s = float(e.size)
    return e / s


def compute_smooth_grad(p, tau, L):
    """Compute true score and gradient of smooth max w.r.t. z (log-parameterization).
    Returns (true_score, grad_z) where grad_z = p * (g_p - dot(p, g_p)).
    """
    n = p.size
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.zeros(n, dtype=np.float64)

    z = (c - cm) / tau
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0

    W = expz / expsum
    Wref = W[::-1]
    H = np.fft.irfft(P * np.fft.rfft(Wref, L), L)
    g_raw = 2.0 * H[n - 1:2 * n - 1][::-1]
    g_raw = np.where(np.isfinite(g_raw), g_raw, 0.0)

    true_score = 2.0 * n * cm

    # Gradient w.r.t. p: g_p = g_raw + p (the +p term from dc[k]/dp[i])
    g_p = g_raw + p
    # Project to simplex tangent: grad_z = p * (g_p - dot(p, g_p))
    mean_g = float(np.dot(p, g_p))
    grad_z = p * (g_p - mean_g)
    grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)

    # Clip gradient for stability
    max_g = float(np.max(np.abs(grad_z)))
    if max_g > 0.5:
        grad_z *= (0.5 / max_g)

    return true_score, grad_z

    """Compute true score and exact subgradient (one-hot at argmax) w.r.t. z."""
    n = p.size
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.zeros(n, dtype=np.float64)

    true_score = 2.0 * n * cm

    # Find all indices where c is within tolerance of max
    tol = 1e-10 * cm + 1e-15
    mask = c >= (cm - tol)
    W = np.zeros(2 * n - 1, dtype=np.float64)
    W[mask] = 1.0 / float(mask.sum())

    Wref = W[::-1]
    H = np.fft.irfft(P * np.fft.rfft(Wref, L), L)
    g_raw = 2.0 * H[n - 1:2 * n - 1][::-1]
    g_raw = np.where(np.isfinite(g_raw), g_raw, 0.0)

    g_p = g_raw + p
    mean_g = float(np.dot(p, g_p))
    grad_z = p * (g_p - mean_g)
    grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)

    max_g = float(np.max(np.abs(grad_z)))
    if max_g > 0.5:
        grad_z *= (0.5 / max_g)

    return true_score, grad_z


def make_asymmetric_parent(n, mode, rng=None):
    """Create diverse asymmetric non-negative parents."""
    if mode == 0:
        # Linear ramp (increasing)
        a = np.linspace(0.5, 2.0, n)
    elif mode == 1:
        # Quadratic ramp
        x = np.linspace(0, 1, n)
        a = 0.5 + 1.5 * x * x
    elif mode == 2:
        # Exponential ramp
        x = np.linspace(0, 1, n)
        a = np.exp(x)
    elif mode == 3:
        # Sine (asymmetric)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * np.sin(np.pi * x)
    elif mode == 4:
        # Cosine bump (shifted left)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.7 * np.cos(np.pi * (x - 0.3))
    elif mode == 5:
        # Gaussian at left
        x = np.linspace(0, 1, n)
        a = 0.3 + np.exp(-((x - 0.3) / 0.2) ** 2)
    elif mode == 6:
        # Gaussian at right
        x = np.linspace(0, 1, n)
        a = 0.3 + np.exp(-((x - 0.7) / 0.2) ** 2)
    elif mode == 7:
        # Two bumps (asymmetric)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.25) / 0.15) ** 2) + 0.6 * np.exp(-((x - 0.75) / 0.1) ** 2)
    elif mode == 8:
        # Power law
        x = np.linspace(1e-6, 1, n)
        a = x ** 0.5 + 0.5
    elif mode == 9:
        # Triangle (peak at 1/3)
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(x - 1.0 / 3.0) * 1.5
        a = np.maximum(a, 0.1)
    elif mode == 10:
        # Chebyshev-based
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.5 * np.cos(np.arccos(x)) + 0.3 * np.cos(2 * np.arccos(x))
        a = np.maximum(a, 0.1)
    elif mode == 11:
        # Random smooth (Gaussian process-like)
        if rng is None:
            rng = np.random.default_rng(0)
        f = np.fft.rfft(rng.normal(0, 1, n))
        f = f * np.exp(-np.linspace(0, 3, n // 2 + 1))
        a = np.fft.irfft(f, n)
        a = 1.0 + 0.5 * a / max(1e-10, abs(a).max())
        a = np.maximum(a, 0.1)
    else:
        a = np.ones(n)

    return normalize_positive(a)


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def adam_run(parent, duration_s, rng, tau_ratio=0.05, lr0=0.02, use_exact=False):
    """Adam optimizer with smooth-max or exact subgradient in log-domain."""
    p = normalize_positive(parent)
    if p is None:
        return None, np.inf

    n = p.size
    L = next_pow2(3 * n - 2)
    deadline = time.monotonic() + max(0.0, float(duration_s))

    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    bias_step = 0

    current_tau_ratio = float(tau_ratio)
    lr = float(lr0)
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    step = 0
    use_exact_phase = False

    while time.monotonic() < deadline:
        if use_exact_phase or (current_tau_ratio < 0.003):
            tau = 1e-12
            true_s, grad = compute_exact_subgrad(p, L)
        else:
            tau = max(1e-12, current_tau_ratio / float(n))
            true_s, grad = compute_smooth_grad(p, tau, L)

        if not np.isfinite(true_s):
            break

        if true_s < best_score - 1e-12:
            best_score = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Schedule: switch to exact when tau is small
        if current_tau_ratio < 0.003 and not use_exact_phase:
            use_exact_phase = True

        # Learning rate and tau decay
        if step > 0 and step % 3000 == 0:
            lr = max(0.0005, lr * 0.92)
            if not use_exact_phase:
                current_tau_ratio = max(0.002, current_tau_ratio * 0.85)

        # Restart if stuck
        if no_improve > 5000:
            zz = np.log(np.maximum(best_p, 1e-300))
            zz = np.where(np.isfinite(zz), zz, -50.0)
            noise_scale = 0.1 + 0.1 * (no_improve / 10000.0)
            zz = zz + rng.normal(0.0, noise_scale, n)
            zz -= zz.mean()
            z = np.clip(zz, -50.0, 50.0)
            m *= 0.05
            v *= 0.05
            no_improve = 0
            current_tau_ratio = max(0.005, current_tau_ratio * 0.6)
            lr = max(0.003, lr * 0.7)

        bias_step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        b1 = 1.0 - beta1 ** bias_step
        b2 = 1.0 - beta2 ** bias_step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax_fn(z)
        step += 1

        if step % 5000 == 0:
            print(f"  adam step {step} score {true_s:.6f} best {best_score:.6f} tau_r {current_tau_ratio:.4f} lr {lr:.5f}")

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    """Main search function. Returns best sequence found."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    raw_budget = max(0.0, float(budget_s))
    deadline = start_time + max(0.0, raw_budget - 25.0)

    # Determine target length based on available time
    rem_total = deadline - time.monotonic()
    if rem_total < 400.0:
        N = 4096
    elif rem_total < 1200.0:
        N = 8192
    else:
        N = 16384

    best_p = None
    best_s = np.inf

    def try_update(p, s):
        nonlocal best_p, best_s
        if p is None or not np.isfinite(s):
            return
        if best_p is None or s < best_s:
            best_p = p
            best_s = float(s)

    # Evaluate provided sequence
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and float(hs.sum()) > 1e-9:
            s0 = fast_score(hs)
            try_update(hs, s0)
            pN = resize_sequence(hs, N)
            sN = fast_score(pN)
            try_update(pN, sN)
            print(f"Provided sequence: score={s0:.6f}, resized({N})={sN:.6f}")
    except Exception:
        pass

    # Try diverse asymmetric initializations at target length
    init_scores = []
    for mode in range(12):
        p = make_asymmetric_parent(N, mode, rng)
        if p is not None:
            s = fast_score(p)
            init_scores.append((s, mode))
            try_update(p, s)
    init_scores.sort()
    if init_scores:
        print(f"Best init: mode={init_scores[0][1]} score={init_scores[0][0]:.6f}")
    print(f"Current best after inits: {best_s:.6f}")

    # Coarse-to-fine: optimize at smaller length first to find good shape
    if rem_total > 600.0:
        N_coarse = N // 4
        if N_coarse < 1024:
            N_coarse = 1024
        coarse_deadline = time.monotonic() + min(180.0, rem_total * 0.12)

        # Best parent for coarse optimization
        coarse_parent = resize_sequence(best_p, N_coarse)
        s_coarse = fast_score(coarse_parent)
        print(f"Coarse phase (N={N_coarse}): starting score {s_coarse:.6f}")

        p_c, s_c = adam_run(coarse_parent, coarse_deadline - time.monotonic(),
                            rng, tau_ratio=0.08, lr0=0.03)
        if p_c is not None:
            s_c_resized = fast_score(resize_sequence(p_c, N))
            try_update(resize_sequence(p_c, N), s_c_resized)
            print(f"Coarse phase done: coarse={s_c:.6f}, resized={s_c_resized:.6f}")

    # Fine optimization at target length
    rem = deadline - time.monotonic()
    if rem > 60.0:
        parent = resize_sequence(best_p, N)
        s_parent = fast_score(parent)
        print(f"\nFine phase (N={N}): starting from score {s_parent:.6f}, time remaining {rem:.0f}s")

        # Phase 1: medium tau
        dur1 = min(rem * 0.45, 900.0)
        if dur1 > 30:
            p1, s1 = adam_run(parent, dur1, rng, tau_ratio=0.04, lr0=0.02)
            if p1 is not None:
                try_update(p1, s1)
            print(f"Phase 1 done: best={best_s:.6f}")

        # Phase 2: smaller tau, from best with slight perturbation
        rem = deadline - time.monotonic()
        if rem > 30:
            noisy = best_p.copy()
            log_noisy = np.log(np.maximum(noisy, 1e-300))
            log_noisy += rng.normal(0, 0.05, noisy.size)
            log_noisy -= log_noisy.mean()
            noisy = softmax_fn(np.clip(log_noisy, -50, 50))
            dur2 = min(rem * 0.45, 800.0)
            if dur2 > 30:
                p2, s2 = adam_run(noisy, dur2, rng, tau_ratio=0.015, lr0=0.01)
                if p2 is not None:
                    try_update(p2, s2)
            print(f"Phase 2 done: best={best_s:.6f}")

        # Phase 3: exact subgradient, very fine
        rem = deadline - time.monotonic()
        if rem > 20:
            p3parent = resize_sequence(best_p, N)
            dur3 = rem * 0.95
            if dur3 > 20:
                p3, s3 = adam_run(p3parent, dur3, rng, tau_ratio=0.005, lr0=0.005, use_exact=True)
                if p3 is not None:
                    try_update(p3, s3)
            print(f"Phase 3 done: best={best_s:.6f}")

    # Final: try a few more quick optimizations from different angles
    rem = deadline - time.monotonic()
    if rem > 10:
        # Try from a completely different shape
        alt = make_asymmetric_parent(N, rng.integers(0, 12), rng)
        if alt is not None:
            p_alt, s_alt = adam_run(alt, min(rem * 0.5, 60.0), rng, tau_ratio=0.02, lr0=0.015)
            if p_alt is not None:
                try_update(p_alt, s_alt)

    # Final validation using actual evaluator
    print(f"\nFinal best fast_score: {best_s:.6f}")
    candidates = []
    if best_p is not None:
        candidates.append(best_p)
    # Safety fallbacks
    candidates.append(np.ones(1024, dtype=np.float64) / 1024.0)

    final_seq = None
    final_score = np.inf

    for cand in candidates:
        arr = np.asarray(cand, dtype=np.float64)
        arr = np.clip(arr, 0.0, 1000.0)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        lst = [float(x) for x in arr]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst
            print(f"  candidate len={len(lst)} eval_score={sc:.6f}")

    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024
        print("  WARNING: using uniform fallback!")

    elapsed = time.monotonic() - start_time
    print(f"\nDone in {elapsed:.1f}s. Final score: {final_score:.6f}")
    return final_seq
```
