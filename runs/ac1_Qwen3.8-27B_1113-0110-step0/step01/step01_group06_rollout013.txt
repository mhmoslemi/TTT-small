def _resize_sequence(a, n_target):
    n=a.size
    if n==n_target: return a.copy()
    if n==1: return np.full(n_target, a.sum()/n_target)
    x_old=np.linspace(0,1,n)
    x_new=np.linspace(0,1,n_target)
    z=np.log(np.maximum(a,1e-300))
    z = z - z.max()
    zi=np.interp(x_new,x_old,z)
    out=np.exp(zi)
    out=np.maximum(out,1e-300)
    out/=out.sum()
    return out

```python
"""
Search for non-negative step-function heights minimizing

    2*n*max(convolve(a, a)) / (sum(a))**2.

The optimizer combines:
  - a smooth / active-lag multiplicative gradient method,
  - a randomized one-mass-transfer local search using exact 1-D evaluation,
  - symmetric pair-split refinement (a special case of the transfer method),
  - periodic random restarts and optional upscaling smoothed from the best
    incumbent.

The objective is scale-invariant, so all internal sequences are normalized
to sum 1.  FFTs are used for convolutions and weighted-convolution
gradients.  The final sequence is returned as a Python list.
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _fft_grad(weights, a):
    """
    Gradient of F(a) = sum_k weights[k] * (a*a)[k] with respect to a.
    Returns 2 * correlation(weights, a), length len(a).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    weights = np.asarray(weights, dtype=np.float64).ravel()
    n = int(a.size)
    L = int(weights.size)
    if n == 0 or L == 0:
        return np.zeros(0, dtype=np.float64)
    N = _pow2(2 * n)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N).real[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if b is None:
        b = _fft_square(a)
    s = float(np.sum(a))
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(np.max(b)) if b.size else 0.0
    return float(2.0 * n * maxb / (s * s)), b, maxb


def _normalize_positive(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        n = max(1, int(a.size))
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _random_positive(n, rng, amp=3.0):
    n = max(2, int(n))
    z = rng.normal(0.0, float(amp), size=n)
    z -= float(np.max(z))
    a = np.exp(z)
    a = np.maximum(a, 1e-300)
    a /= float(np.sum(a))
    return a


def _anti_init(n, rng, amp=3.0):
    n = max(2, int(n))
    m = n // 2
    if m < 1:
        return np.ones(n, dtype=np.float64) / float(n)

    c = 1.0 / float(m)
    v = rng.normal(0.0, 1.0, size=m)
    v = v - float(np.mean(v))
    std = float(np.std(v))
    if std > 1e-14:
        v = (0.35 * float(amp)) * (v / std)
    else:
        v = (0.35 * float(amp)) * v

    x = np.clip(0.5 + v, 1e-6, 1.0 - 1e-6)
    a = np.zeros(n, dtype=np.float64)
    a[:m] = c * x
    if n % 2 == 0:
        a[m:] = c * (1.0 - x[::-1])
    else:
        a[m + 1:] = c * (1.0 - x[::-1])

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _resize_sequence(a, target_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    target_n = max(2, int(target_n))
    a = _normalize_positive(a)
    n = int(a.size)
    if n == target_n:
        return a.copy()
    if n <= 1:
        return np.full(target_n, 1.0 / float(target_n), dtype=np.float64)

    xold = np.linspace(0.0, 1.0, n)
    xnew = np.linspace(0.0, 1.0, target_n)
    z = np.log(np.maximum(a, 1e-300))
    z = z - float(np.max(z))
    zi = np.interp(xnew, xold, z)
    out = np.exp(zi)
    out = np.maximum(out, 1e-300)
    out /= float(np.sum(out))
    return out


def _make_weights(b, focus, eps=0.02):
    b = np.asarray(b, dtype=np.float64).ravel()
    L = int(b.size)
    maxb = float(np.max(b)) if L else 0.0
    if maxb <= 1e-300:
        return np.full(L, 1.0 / max(1, L), dtype=np.float64), maxb
    z = (maxb - b) / maxb
    with np.errstate(over="ignore", under="ignore"):
        p = np.exp(-float(focus) * z)
    s = float(np.sum(p))
    if s <= 1e-12:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - float(eps)) * (p / s) + float(eps) / float(L)
    return p, maxb


def _consider_candidate(best_a, best_score, candidate_a):
    if candidate_a is None:
        return best_a, best_score
    c = _normalize_positive(candidate_a)
    if c.size < 2:
        return best_a, best_score
    s, _, _ = _internal_score(c)
    if np.isfinite(s) and s < best_score:
        return c, s
    return best_a, best_score


def _gradient_refine(a0, limit_s, rng, verbose=False,
                     focus0=20.0, step0=0.5, label="grad"):
    a = _normalize_positive(a0)
    n = int(a.size)
    if n < 2:
        return a, float("inf")
    limit_s = float(limit_s)
    if limit_s <= 0.0:
        return a, _internal_score(a)[0]

    b = _fft_square(a)
    score = 2.0 * float(n) * float(np.max(b))
    best_a = a.copy()
    best_score = score

    focus = float(focus0)
    step = float(step0)
    no_improve = 0
    max_no = 320
    resets = 0
    max_resets = 20
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < limit_s:
        maxb = float(np.max(b))
        score = 2.0 * float(n) * maxb

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 2000 == 0):
            print(
                f"[{label}] t={time.monotonic() - t0:.1f}s "
                f"score={score:.6f} best={best_score:.6f} "
                f"focus={focus:.1f} step={step:.3f} resets={resets}"
            )

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.35)

            kind = int(rng.integers(0, 3))
            if kind == 0:
                fresh = _anti_init(n, rng, amp=3.0)
            elif kind == 1:
                fresh = _random_positive(n, rng, amp=4.0)
            else:
                base = best_a.copy()
                base = np.maximum(base, 1e-12)
                fresh = base * np.exp(rng.normal(0.0, 0.25, size=n))
                fresh = _normalize_positive(fresh)

            base = best_a.copy()
            base = np.maximum(base, 1e-12)
            base = base / float(np.sum(base))
            a = 0.75 * base + 0.25 * fresh
            a = _normalize_positive(a)
            b = _fft_square(a)
            focus = float(rng.choice([5.0, 10.0, 20.0, 50.0, 100.0,
                                      200.0, 500.0, 1000.0, 2000.0]))
            step = float(rng.uniform(0.1, 1.0))
            it += 1
            continue

        p, _ = _make_weights(b, focus)
        g = _fft_grad(p, a)
        gd = g - float(np.dot(a, g))
        if gd.size == 0:
            break
        maxg = float(np.max(np.abs(gd)))
        if maxg < 1e-18:
            break
        gd = gd / maxg

        a_new = a * np.exp(-step * gd)
        a_new = np.maximum(a_new, 1e-300)
        snf = float(np.sum(a_new))
        if snf <= 1e-300:
            step = max(step * 0.5, 0.01)
            it += 1
            continue
        a_new = a_new / snf

        b_new = _fft_square(a_new)
        score_new = 2.0 * float(n) * float(np.max(b_new))
        wt_old = float(np.dot(p, b))
        wt_new = float(np.dot(p, b_new))

        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        elif score_new < score * 1.0005 and wt_new < wt_old - 1e-12:
            accept = True

        if accept:
            a = a_new
            b = b_new
            score = score_new
            step = min(step * 1.08, 4.0)
        else:
            step = max(step * 0.85, 0.01)
            if step < 0.03:
                base = best_a.copy()
                base = np.maximum(base, 1e-12)
                base = base / float(np.sum(base))
                a = base * np.exp(rng.normal(0.0, 0.08, size=n))
                a = _normalize_positive(a)
                b = _fft_square(a)
                step = 0.3
                focus = float(rng.choice([5.0, 10.0, 20.0, 50.0, 100.0,
                                          200.0, 500.0]))

        if focus < 20000.0:
            focus = min(focus * 1.02, 20000.0)

        it += 1

    return best_a, best_score


def _transfer_candidate_max(b, W, temp, t, p1, p2, p3):
    if t == 0.0:
        temp[:] = b
    else:
        np.multiply(W, t, out=temp)
        np.add(b, temp, out=temp)
    t2 = t * t
    if t2 != 0.0:
        temp[p1] += t2
        temp[p2] += t2
        temp[p3] -= 2.0 * t2
    return float(np.max(temp))


def _transfer_step(a, twoa, b, i, j, n, W, temp, cur_maxb):
    """
    Move mass t from j to i: a_i <- a_i + t, a_j <- a_j - t.
    Return the new max convolution coefficient if the move improves it.
    """
    if i == j:
        return cur_maxb

    oi = float(a[i])
    oj = float(a[j])
    lo = -oj
    hi = oi
    if hi - lo <= 1e-300:
        return cur_maxb

    W.fill(0.0)
    W[i:i + n] += twoa
    W[j:j + n] -= twoa

    p1 = 2 * i
    p2 = 2 * j
    p3 = i + j

    gr = 0.6180339887498949
    a0 = lo
    b0 = hi

    t1 = a0 + gr * (b0 - a0)
    t2 = b0 - gr * (b0 - a0)

    f1 = _transfer_candidate_max(b, W, temp, t1, p1, p2, p3)
    f2 = _transfer_candidate_max(b, W, temp, t2, p1, p2, p3)

    fl = _transfer_candidate_max(b, W, temp, lo, p1, p2, p3)
    fh = _transfer_candidate_max(b, W, temp, hi, p1, p2, p3)

    best_t = 0.0
    best_f = float(cur_maxb)

    if f1 < best_f:
        best_f = f1
        best_t = t1
    if f2 < best_f:
        best_f = f2
        best_t = t2
    if fl < best_f:
        best_f = fl
        best_t = lo
    if fh < best_f:
        best_f = fh
        best_t = hi

    for _ in range(12):
        if f1 <= f2:
            b0 = t2
            t2 = t1
            f2 = f1
            t1 = b0 - gr * (b0 - a0)
            f1 = _transfer_candidate_max(b, W, temp, t1, p1, p2, p3)
        else:
            a0 = t1
            t1 = t2
            f1 = f2
            t2 = a0 + gr * (b0 - a0)
            f2 = _transfer_candidate_max(b, W, temp, t2, p1, p2, p3)

        if f1 < best_f:
            best_f = f1
            best_t = t1
        if f2 < best_f:
            best_f = f2
            best_t = t2

    if best_f < float(cur_maxb) - 1e-12:
        t = float(best_t)
        if t < a0:
            t = a0
        elif t > b0:
            t = b0

        if abs(t) > 1e-15:
            np.multiply(W, t, out=temp)
            np.add(b, temp, out=b)

            t2t = t * t
            b[p1] += t2t
            b[p2] += t2t
            b[p3] -= 2.0 * t2t

            ni = oi + t
            nj = oj - t
            if ni < 0.0:
                ni = 0.0
            if nj < 0.0:
                nj = 0.0

            a[i] = ni
            a[j] = nj
            twoa[i] = 2.0 * ni
            twoa[j] = 2.0 * nj
            return float(np.max(b))

    return cur_maxb


def _transfer_cd_refine(a0, limit_s, rng, symmetric=False, verbose=False):
    """
    Randomized coordinate-wise mass-transfer local search.

    symmetric=True optimizes pairs (i, n-1-i), preserving pair sums.
    symmetric=False chooses arbitrary pairs, including all coordinates.
    """
    a = _normalize_positive(a0)
    n = int(a.size)

    if symmetric and (n % 2 == 1):
        a = a[:-1]
        n = int(a.size)

    if n < 2:
        return a, float("inf")

    limit_s = float(limit_s)
    if limit_s <= 0.0:
        return a, _internal_score(a)[0]

    L = 2 * n - 1
    W = np.empty(L, dtype=np.float64)
    temp = np.empty(L, dtype=np.float64)

    b = _fft_square(a)
    cur_maxb = float(np.max(b))
    best_score = 2.0 * float(n) * cur_maxb
    best_a = a.copy()
    twoa = 2.0 * a.astype(np.float64)

    t0 = time.monotonic()
    no_improve = 0
    sweep = 0

    while time.monotonic() - t0 < limit_s:
        improved = 0

        if symmetric:
            m = n // 2
            order = rng.permutation(m)
            for p in order:
                if time.monotonic() >= limit_s:
                    break
                i = int(p)
                j = n - 1 - i
                new_maxb = _transfer_step(a, twoa, b, i, j, n, W, temp, cur_maxb)
                if new_maxb < cur_maxb - 1e-12:
                    cur_maxb = new_maxb
                    improved += 1
                    sc = 2.0 * float(n) * cur_maxb
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a.copy()
        else:
            target_steps = max(64, n // 2)
            steps = 0
            while time.monotonic() < limit_s and steps < target_steps:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    new_maxb = _transfer_step(a, twoa, b, i, j, n, W, temp, cur_maxb)
                    if new_maxb < cur_maxb - 1e-12:
                        cur_maxb = new_maxb
                        improved += 1
                        sc = 2.0 * float(n) * cur_maxb
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best_a = a.copy()
                steps += 1

        # Remove FFT drift.
        b = _fft_square(a)
        cur_maxb = float(np.max(b))
        twoa = 2.0 * a.astype(np.float64)
        sc = 2.0 * float(n) * cur_maxb
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

        if verbose:
            print(
                f"[transfer-cd:{'sym' if symmetric else 'rand'}] "
                f"sweep={sweep} t={time.monotonic() - t0:.1f}s "
                f"cur={sc:.6f} best={best_score:.6f}"
            )

        sweep += 1
        if improved > 0:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 2:
            k = max(1, min(n, n // 10))
            idx = rng.choice(n, size=k, replace=False)
            factors = np.exp(rng.normal(0.0, 0.12, size=k))
            a[idx] = np.maximum(a[idx], 1e-12) * factors
            a = _normalize_positive(a)
            b = _fft_square(a)
            cur_maxb = float(np.max(b))
            twoa = 2.0 * a.astype(np.float64)
            sc = 2.0 * float(n) * cur_maxb
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
            no_improve = 0

        if best_score <= 1.42:
            break

    return best_a, best_score


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a good non-negative sequence and return it as a list."""
    seed_val = int(seed) if seed is not None else 42
    budget = float(budget_s) if budget_s is not None else 2400.0
    total = max(60.0, budget)

    start = time.monotonic()
    final_deadline = start + total - 30.0

    rng = np.random.default_rng((seed_val * 1000003 + 987654321) % (2**32))

    best_a = None
    best_score = float("inf")

    # Load previous best if present.
    try:
        gv = globals().get("height_sequence_1")
        if gv is not None:
            g = np.asarray(gv, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            if g.size >= 2:
                gsum = float(np.sum(g))
                if gsum > 0.01:
                    gn = g / gsum
                    if gn.size > 16384:
                        gn = _resize_sequence(gn, 16384)
                    sc, _, _ = _internal_score(gn)
                    if np.isfinite(sc):
                        best_a = gn.copy()
                        best_score = sc
    except Exception as exc:
        print("failed to load given sequence:", repr(exc))

    if best_a is None:
        best_a = _anti_init(4096, rng, amp=3.0)
        best_score, _, _ = _internal_score(best_a)

    print(f"initial n={int(best_a.size)} score={best_score:.6f}")

    # Stage 1: gradient refinement of the incumbent.
    now = time.monotonic()
    if now < final_deadline - 60.0:
        alloc = min(0.15 * total, final_deadline - now - 60.0)
        if alloc > 25.0:
            t0 = time.monotonic()
            try:
                a1, s1 = _gradient_refine(
                    best_a, alloc, rng,
                    verbose=False,
                    focus0=10.0, step0=0.5, label="g1"
                )
                best_a, best_score = _consider_candidate(best_a, best_score, a1)
            except Exception as exc:
                print("stage1 failed:", repr(exc))
            print(f"stage1 grad used={time.monotonic() - t0:.1f}s best={best_score:.6f}")

    # Stage 2: symmetric pair-split transfer refinement.
    now = time.monotonic()
    if now < final_deadline - 60.0:
        alloc = min(0.20 * total, final_deadline - now - 60.0)
        if alloc > 30.0:
            t0 = time.monotonic()
            try:
                a2, s2 = _transfer_cd_refine(
                    best_a, alloc, rng,
                    symmetric=True, verbose=False
                )
                best_a, best_score = _consider_candidate(best_a, best_score, a2)
            except Exception as exc:
                print("stage2 sym transfer failed:", repr(exc))
            print(f"stage2 sym used={time.monotonic() - t0:.1f}s best={best_score:.6f}")

    # Stage 3: arbitrary transfer refinement.
    now = time.monotonic()
    if now < final_deadline - 60.0:
        alloc = min(0.15 * total, final_deadline - now - 60.0)
        if alloc > 30.0:
            t0 = time.monotonic()
            try:
                a3, s3 = _transfer_cd_refine(
                    best_a, alloc, rng,
                    symmetric=False, verbose=False
                )
                best_a, best_score = _consider_candidate(best_a, best_score, a3)
            except Exception as exc:
                print("stage3 rand transfer failed:", repr(exc))
            print(f"stage3 rand used={time.monotonic() - t0:.1f}s best={best_score:.6f}")

    # Stage 4: smooth upscale to a larger length if it may help.
    target = None
    now = time.monotonic()
    if best_a.size < 8192 and now < final_deadline - 400.0:
        target = 8192
    elif best_score > 1.51 and best_a.size < 16384 and now < final_deadline - 500.0:
        target = 16384

    if target is not None:
        now = time.monotonic()
        alloc = min(0.15 * total, final_deadline - now - 200.0)
        if alloc > 40.0:
            t0 = time.monotonic()
            try:
                ar = _resize_sequence(best_a, int(target))
                a4, s4 = _gradient_refine(
                    ar, alloc, rng,
                    verbose=False,
                    focus0=20.0, step0=0.4, label=f"upscale{int(target)}"
                )
                best_a, best_score = _consider_candidate(best_a, best_score, a4)
            except Exception as exc:
                print("stage4 upscale failed:", repr(exc))
            print(f"stage4 upscale{target} used={time.monotonic() - t0:.1f}s best={best_score:.6f}")

    # Final: devote remaining time to polishing the current best.
    now = time.monotonic()
    if now < final_deadline - 5.0:
        alloc = min(total, final_deadline - now - 3.0)
        if alloc > 15.0:
            t0 = time.monotonic()
            try:
                a5, s5 = _gradient_refine(
                    best_a, alloc, rng,
                    verbose=False,
                    focus0=50.0, step0=0.35, label="final"
                )
                best_a, best_score = _consider_candidate(best_a, best_score, a5)
            except Exception as exc:
                print("final refinement failed:", repr(exc))
            print(f"final grad used={time.monotonic() - t0:.1f}s best={best_score:.6f}")

    # Avoid returning excessively large sequences.
    if best_a is None:
        best_a = _anti_init(8192, rng, amp=3.0)
        best_score, _, _ = _internal_score(best_a)

    if best_a.size > 16384:
        best_a = _resize_sequence(best_a, 16384)
        best_score, _, _ = _internal_score(best_a)

    # Final cleanup.
    best_a = _normalize_positive(best_a)
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.size < 2:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a[best_a < 1e-15] = 0.0
    s = float(np.sum(best_a))
    if s <= 0.0:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        s = 1.0
    else:
        best_a = best_a / s

    final_seq = [float(x) for x in best_a.tolist()]

    exact = float("inf")
    if best_a.size <= 12000 and time.monotonic() < final_deadline - 5.0:
        exact = _safe_evaluate(final_seq)

    print(
        f"final internal score={best_score:.6f}, exact score={exact:.6f}, "
        f"n={int(best_a.size)}"
    )

    return final_seq
```
