"""
Optimizes a non-negative step function sequence to minimize
    2 * n * max(convolve(a, a)) / (sum(a))^2.

Strategy:
1. Use annealed log-sum-exp (smoothed max) surrogate for the max convolution coefficient.
2. Mirror descent (exponentiated gradient) with adaptive step size and line search.
3. Anti-palindromic parameterization to eliminate the center convolution peak.
4. Coordinate descent refinement for fine-tuning.
5. Multiple restarts, varying sequence lengths, and diverse initializations.
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _fft_grad_smoothed(p, a):
    """Gradient of log-sum-exp(beta * conv(a,a)) wrt a, using weights p."""
    n = int(a.size)
    L = int(p.size)
    N = _pow2(L + n - 1)
    fp = np.fft.rfft(p, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fp * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s < 1e-300:
        return float("inf"), b
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b


def _smoothed_score(a, b, beta):
    """Compute log-sum-exp(beta * b) / beta as a smoothed surrogate for max(b)."""
    if beta < 1e-6:
        return float(b.max())
    bm = float(b.max())
    if bm < 1e-300:
        return 0.0
    shifted = beta * (b - bm)
    log_sum = float(np.log(np.sum(np.exp(shifted)))) + beta * bm
    return log_sum / beta


def _softmax_weights(b, beta, temperature=1.0):
    """Compute softmax weights over convolution coefficients."""
    betat = max(beta * temperature, 1e-6)
    bm = float(b.max())
    shifted = betat * (b - bm)
    exp_vals = np.exp(shifted)
    s = float(exp_vals.sum())
    if s < 1e-300:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = exp_vals / s
    return p


def _anti_palin_init(n, rng, mode="bimodal"):
    """Create anti-palindromic initialization: a[i] + a[n-1-i] = 2/n."""
    m = n // 2
    a = np.zeros(n, dtype=np.float64)
    if mode == "bimodal":
        # Concentrate mass at endpoints of pairs
        for i in range(m):
            r = rng.random()
            # Biased: put more mass at one end
            a[i] = (2.0 / n) * r
            a[n - 1 - i] = (2.0 / n) * (1.0 - r)
    elif mode == "spread":
        # More spread out within each pair
        for i in range(m):
            r = rng.uniform(0.2, 0.8)
            a[i] = (2.0 / n) * r
            a[n - 1 - i] = (2.0 / n) * (1.0 - r)
    elif mode == "gaussian":
        # Gaussian-like profile within pairs
        x = np.arange(m, dtype=np.float64)
        center = m / 2.0
        width = m / 4.0
        v = np.exp(-0.5 * ((x - center) / width) ** 2)
        v = v / v.sum() * m
        for i in range(m):
            t = v[i] / m
            t = np.clip(t, 0.0, 1.0)
            a[i] = (2.0 / n) * t
            a[n - 1 - i] = (2.0 / n) * (1.0 - t)
    else:
        # Random
        for i in range(m):
            r = rng.random()
            a[i] = (2.0 / n) * r
            a[n - 1 - i] = (2.0 / n) * (1.0 - r)
    
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def _optimize_mirror_descent(a0, time_limit, rng, beta_start=50.0, beta_end=5000.0,

                             label=""):

    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    
    b = _fft_square(a)
    score, _ = _score(a, b)
    best_score = score
    best_a = a.copy()
    
    beta = beta_start
    eta = eta_start
    no_improve = 0
    max_no_improve = 500
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute smoothed weights
        p = _softmax_weights(b, beta)
        
        # Compute gradient
        grad = _fft_grad_smoothed(p, a)
        
        # Center gradient (for mirror descent with simplex constraint)
        avg_grad = float(np.dot(a, grad))
        grad_centered = grad - avg_grad
        
        max_g = float(np.max(np.abs(grad_centered)))
        if max_g < 1e-20:
            break
        grad_centered /= max_g
        
        # Exponentiated gradient update (mirror descent on simplex)
        a_new = a * np.exp(-eta * grad_centered)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()
        
        # Evaluate
        b_new = _fft_square(a_new)
        score_new, _ = _score(a_new, b_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        elif score_new < score - 1e-12:
            no_improve = min(no_improve, 100)
        else:
            no_improve += 1
        
        # Adaptive step size
        if score_new < score:
            eta = min(eta * 1.03, 5.0)
        else:
            eta = max(eta * 0.7, eta_min)
        
        # Accept or reject
        if score_new < score * 1.002:
            a = a_new
            b = b_new
            score = score_new
        else:
            # Add noise to escape
            a = a * np.exp(rng.normal(0.0, 0.1, size=n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = _fft_square(a)
            score = score_new if score_new < score else score
        
        # Anneal beta
        beta = min(beta * beta_growth, beta_end)
        
        # Reset if stuck
        if no_improve > max_no_improve:
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.2)
            # Mix with random
            fresh = np.exp(rng.normal(0.0, 2.0, size=n))
            fresh = np.maximum(fresh, 1e-15)
            fresh /= fresh.sum()
            a = 0.7 * best_a + 0.3 * fresh
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = _fft_square(a)
            score, _ = _score(a, b)
            eta = eta_start
            beta = max(beta_start, beta * 0.5)
        
        it += 1
    
    return best_a, best_score

        def f_of_x(x):
            # Compute max of new_b
            new_b_vals = np.empty(2 * n - 1, dtype=np.float64)
            new_b_vals[:] = bg
            for k in range(2 * n - 1):
                kj = k - i
                if 0 <= kj < n:
                    if kj == i:
                        new_b_vals[k] = bg[k] + x * x
                    else:
                        new_b_vals[k] = bg[k] + 2.0 * x * a[kj]
            return float(new_b_vals.max())
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        x_lo = 0.0
        x_hi = x_max
        x1 = x_hi - gr * (x_hi - x_lo)
        x2 = x_lo + gr * (x_hi - x_lo)
        f1 = f_of_x(x1)
        f2 = f_of_x(x2)
        
        for _ in range(50):
            if f1 < f2:
                x_hi = x2
                x2 = x1
                f2 = f1
                x1 = x_hi - gr * (x_hi - x_lo)
                f1 = f_of_x(x1)
            else:
                x_lo = x1
                x1 = x2
                f1 = f2
                x2 = x_lo + gr * (x_hi - x_lo)
                f2 = f_of_x(x2)
        
        x_opt = (x_lo + x_hi) / 2.0
        x_opt = max(x_opt, 1e-15)
        
        # Apply update
        old_val = a[i]
        a[i] = x_opt
        a_new_sum = float(a.sum())
        
        # Check improvement (use actual score with normalization)
        b_new = _fft_square(a)
        score_new, _ = _score(a, b_new)
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a.copy()
        elif score_new > score * 1.05:
            # Bad, revert
            a[i] = old_val
            b = _fft_square(a)
            score = float(2.0 * n * float(b.max()) / (float(a.sum()) ** 2))
        else:
            b = b_new
            score = score_new
        
        it += 1
    
    return best_a, best_score


def _optimize_pair_soft(x0, time_limit, rng, beta_start=100.0, beta_end=10000.0,
                        label=""):
    """Optimize anti-palindromic parameterization with smoothed max."""
    x = np.asarray(x0, dtype=np.float64).copy()
    m = int(x.size)
    n = 2 * m
    x = np.clip(x, 1e-10, 1.0 - 1e-10)
    
    def x_to_a(xv):
        t = 1.0 / m
        av = np.empty(2 * m, dtype=np.float64)
        av[:m] = t * xv
        av[m:] = t * (1.0 - xv[::-1])
        return av
    
    a = x_to_a(x)
    b = _fft_square(a)
    score, _ = _score(a, b)
    best_score = score
    best_x = x.copy()
    
    beta = beta_start
    eta = 0.3
    t0 = time.monotonic()
    no_improve = 0
    max_no = 800
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Compute smoothed weights
        p = _softmax_weights(b, beta)
        
        # Gradient w.r.t. a
        ga = _fft_grad_smoothed(p, a)
        
        # Project to x-space: a_x = [a_0,...,a_{m-1}], a_{m+i} = 1/m - a_{m-1-i}
        # da_i/dx_i = 1/m (for i < m)
        # da_{m+i}/dx_i = -1/m (for the paired element)
        # ga_x_i = ga_i/m - ga_{m + (m-1-i)}/m... 
        # Actually: x_to_a: a[i] = x[i]/m for i < m
        #            a[m+j] = (1 - x[m-1-j])/m for j < m
        # So da[i]/dx[i] = 1/m, for i < m
        #    da[m+j]/dx[k] = -1/m if k = m-1-j, else 0
        # ga_x[k] = sum_i ga[i] * da[i]/dx[k]
        #          = ga[k]/m (from a[k])  [for k < m]
        #          + ga[m + (m-1-k)] * (-1/m) (from a[m+m-1-k] = a[2m-1-k])
        
        gx = np.empty(m, dtype=np.float64)
        for k in range(m):
            partner = 2 * m - 1 - k
            gx[k] = (ga[k] - ga[partner]) / m
        
        max_g = float(np.max(np.abs(gx)))
        if max_g < 1e-20:
            break
        gx /= max_g
        
        x_new = x - eta * gx
        x_new = np.clip(x_new, 1e-10, 1.0 - 1e-10)
        
        a_new = x_to_a(x_new)
        b_new = _fft_square(a_new)
        score_new, _ = _score(a_new, b_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_x = x_new.copy()
            no_improve = 0
        elif score_new < score - 1e-12:
            no_improve = min(no_improve, 100)
        else:
            no_improve += 1
        
        if score_new < score * 1.002:
            x = x_new
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.02, 3.0)
        else:
            eta = max(eta * 0.75, 0.01)
            if eta < 0.05:
                x = np.clip(x + rng.normal(0.0, 0.05, size=m), 1e-10, 1.0 - 1e-10)
                a = x_to_a(x)
                b = _fft_square(a)
                score, _ = _score(a, b)
        
        beta = min(beta * 1.003, beta_end)
        
        if no_improve > max_no:
            no_improve = 0
            max_no = int(max_no * 1.3)
            x = 0.6 * best_x + 0.4 * rng.random(m)
            x = np.clip(x, 1e-10, 1.0 - 1e-10)
            a = x_to_a(x)
            b = _fft_square(a)
            score, _ = _score(a, b)
            eta = 0.3
            beta = max(beta_start, beta * 0.3)
    
    return x_to_a(best_x), best_score


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function: finds the best non-negative sequence minimizing the evaluation score."""
    seed = int(seed)
    total_budget = min(float(budget_s), 2350.0)
    
    now0 = time.monotonic()
    final_deadline = now0 + total_budget - 20.0
    task_deadline = now0 + total_budget - 120.0
    
    print("=== Step Function Convolution Optimizer ===")
    best_a = None
    best_score = float("inf")
    
    # Try to get the given sequence
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                sc, _ = _score(gn)
                if np.isfinite(sc):
                    print(f"Given sequence: n={gn.size}, score={sc:.6f}")
                    if sc < best_score:
                        best_score = sc
                        best_a = gn.copy()
                    given_a = gn
    except Exception as e:
        print(f"Error reading given sequence: {e}")
    
    # Define tasks: (name, n, method, time_alloc)
    tasks = []
    
    # Task 1: Optimize given sequence directly
    if given_a is not None and given_a.size <= 12000:
        tasks.append(("given_md", given_a, "mirror", 400.0))
    
    # Task 2: Anti-palindromic on n=8192 (bimodal init)
    tasks.append(("palin8192a", None, "pair_bimodal", 400.0))
    
    # Task 3: Anti-palindromic on n=8192 (spread init)  
    tasks.append(("palin8192b", None, "pair_spread", 300.0))
    
    # Task 4: Full optimization on n=8192 (random init)
    tasks.append(("full8192", None, "full_random", 350.0))
    
    # Task 5: Anti-palindromic on n=16384
    tasks.append(("palin16384", None, "pair_bimodal", 350.0))
    
    # Task 6: Full optimization on n=4096 (faster iterations)
    tasks.append(("full4096", None, "full_random", 200.0))
    
    for idx, (name, init_a, method, alloc) in enumerate(tasks):
        now = time.monotonic()
        if now >= task_deadline:
            print("Task deadline reached, skipping remaining tasks")
            break
        alloc = min(alloc, task_deadline - now)
        if alloc < 30.0:
            break
        
        task_seed = (seed * 7919 + idx * 104729) % (2**32)
        rng = np.random.default_rng(task_seed)
        
        print(f"\n--- Task {idx}: {name}, alloc={alloc:.0f}s, current best={best_score:.6f} ---")
        t0 = time.monotonic()
        cand_a = None
        cand_score = float("inf")
        
        try:
            if method == "mirror":
                a0 = init_a.copy() if init_a is not None else np.ones(4096) / 4096
                cand_a, cand_score = _optimize_mirror_descent(
                    a0, alloc, rng,
                    beta_start=200.0, beta_end=8000.0,
                    beta_growth=1.008, eta_start=0.4,
                    label=name
                )
            elif method == "pair_bimodal":
                n = 8192 if "8192" in name else 16384
                m = n // 2
                # Bimodal anti-palindromic init
                x0 = np.full(m, 0.5, dtype=np.float64)
                # Perturb: make some pairs very asymmetric
                for i in range(m):
                    if rng.random() < 0.3:
                        x0[i] = rng.uniform(0.01, 0.99)
                cand_a, cand_score = _optimize_pair_soft(
                    x0, alloc, rng,
                    beta_start=100.0, beta_end=10000.0,
                    label=name
                )
            elif method == "pair_spread":
                n = 8192
                m = n // 2
                # Spread init: x[i] distributed more evenly
                x0 = rng.uniform(0.15, 0.85, size=m)
                # Add some structure
                x0 += 0.1 * np.sin(2 * np.pi * rng.integers(1, 10) * np.arange(m) / m)
                x0 = np.clip(x0, 0.01, 0.99)
                cand_a, cand_score = _optimize_pair_soft(
                    x0, alloc, rng,
                    beta_start=200.0, beta_end=15000.0,
                    label=name
                )
            elif method == "full_random":
                n = 8192 if "8192" in name else 4096
                a0 = np.exp(rng.normal(0.0, 1.5, size=n))
                a0 = np.maximum(a0, 1e-15)
                a0 /= a0.sum()
                cand_a, cand_score = _optimize_mirror_descent(
                    a0, alloc, rng,
                    beta_start=100.0, beta_end=6000.0,
                    beta_growth=1.006, eta_start=0.3,
                    label=name
                )
            
            if cand_a is not None:
                cand_a = np.nan_to_num(cand_a, nan=0.0, posinf=0.0, neginf=0.0)
                cand_a = np.clip(cand_a, 0.0, 1000.0)
                cs = float(cand_a.sum())
                if cs > 0.01:
                    cand_a /= cs
                    internal, _ = _score(cand_a)
                    if np.isfinite(internal):
                        cand_score = min(cand_score, internal)
                    if np.isfinite(cand_score) and cand_score < best_score:
                        best_score = cand_score
                        best_a = cand_a.copy()
                        print(f"  NEW BEST: {cand_score:.6f} (n={cand_a.size})")
            else:
                print(f"  Task returned None")
                
        except Exception as e:
            print(f"  Task {name} failed: {repr(e)}")
        
        print(f"--- Task {name} done: score={cand_score:.6f}, best={best_score:.6f}, "
              f"time={time.monotonic()-t0:.1f}s ---")
    
    # Final refinement phase
    if best_a is not None:
        remaining = task_deadline - time.monotonic()
        if remaining > 60.0:
            refine_time = min(100.0, remaining - 10.0)
            if refine_time > 30.0:
                print(f"\n=== Final refinement: {refine_time:.0f}s ===")
                rng = np.random.default_rng((seed + 12345) % (2**32))
                
                # Try mirror descent refinement
                refined_a, refined_score = _optimize_mirror_descent(
                    best_a, refine_time * 0.6, rng,
                    beta_start=500.0, beta_end=20000.0,
                    beta_growth=1.01, eta_start=0.1,
                    label="refine_md"
                )
                if np.isfinite(refined_score) and refined_score < best_score:
                    best_score = refined_score
                    best_a = refined_a.copy()
                    print(f"  Refinement improved: {best_score:.6f}")
                
                # Try coordinate descent on the best
                remaining2 = task_deadline - time.monotonic()
                if remaining2 > 30.0:
                    ref2_time = min(50.0, remaining2 - 5.0)
                    cd_a, cd_score = _optimize_coord_descent(
                        best_a, ref2_time, rng,
                        label="refine_cd"
                    )
                    if np.isfinite(cd_score) and cd_score < best_score:
                        best_score = cd_score
                        best_a = cd_a.copy()
                        print(f"  Coord descent improved: {best_score:.6f}")
    
    if best_a is None:
        # Fallback
        best_a = np.ones(2048, dtype=np.float64) / 2048.0
        best_score, _ = _score(best_a)
        print(f"Using fallback: score={best_score:.6f}")
    
    # Final validation
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss
    
    final_seq = [float(x) for x in best_a]
    final_score, _ = _score(best_a)
    
    # Exact evaluation if time permits
    if time.monotonic() < final_deadline:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"\n=== FINAL: internal={final_score:.6f}, exact={exact:.6f}, n={best_a.size} ===")
        else:
            print(f"\n=== FINAL: internal={final_score:.6f}, exact=unavailable, n={best_a.size} ===")
    else:
        print(f"\n=== FINAL: internal={final_score:.6f}, n={best_a.size} ===")
    
    return final_seq
