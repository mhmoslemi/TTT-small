def _fft_convolve(a):
    n = len(a)

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    n = len(f)
    # C has shape (2n-1, n)
    # C[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # (C @ g)[k] = sum_j C[k,j] * g[j] = sum_j f[k-j] * g[j] = (f*g)[k]
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # For row k, C[k, j] = f[k-j] if 0 <= k-j < n
        # k ranges from j to j+n-1
        rows = slice(j, j + n)
        C[rows, j] = f
    return C

            return result.x

    return None, None if get_dual else None

        # For a[i] to appear in c[k], we need 0 <= k-i < n

        # (The factor of 2 comes from a[i] appearing in both the "forward" and "backward" terms)
        # But when i = k-i (i.e., k = 2i), it's a[i]*a[i], so derivative is 2*a[i].
        # In both cases, the formula is 2*a[k-i] (since when i=k-i, a[k-i] = a[i]).

        i_max = min(n - 1, k)

        # But wait, if k-i == i (i.e., i = k/2), the term is a[i]^2, derivative is 2*a[i] = 2*a[k-i]. Same formula!

def _bmat(f):
    """Build convolution matrix: (C@g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = f
    return C

def _lp(f, M, tl=15):
    """Solve: max sum(g) s.t. C_f @ g <= M, g >= 0."""
    n = len(f)
    C = _bmat(f)
    try:
        r = linprog(-np.ones(n), A_ub=C, b_ub=M*np.ones(2*n-1),
                    bounds=[(0, None)]*n, method='highs',
                    options={'presolve': True, 'time_limit': tl})
        if r.status == 0:
            return r.x

        pass
    return None

    s = float(np.sum(out))

    return out
