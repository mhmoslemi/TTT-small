- Lag 0 of the convolution is sum a_j² (the "energy")
- For a binary sequence, this equals the number of 1s ≈ n/2
- For a uniform sequence, this equals n*(1/n)² = 1/n (much smaller!)
- So we need to smooth enough that the energy is small, but not so much that the autocorrelation peaks at other lags.

The optimal smoothness is a trade-off. Let me parametrize it.

Actually, let me reconsider. For a sequence with values in [0,1]:
- Lag 0: sum a_j². Minimized by uniform (all 1/n), maximized by binary.
- Center lag: sum a_j * a_{n-1-j}. Minimized by anti-symmetric (high-left, low-right).
- Other lags: various trade-offs.

The optimal shape balances all of these. It's not simply "smoothed binary" or "sinusoid" - it's something in between.

Let me try a different approach: **use the exponential family parametrization with more flexibility**.

a_j = exp(θ_0 + θ_1*(j/n) + θ_2*(j/n)² + ... + θ_m*(j/n)^m) / normalizer

This is a polynomial in j/n, which can represent smooth shapes with up to m+1 "features". With m=5-10, we should be able to capture the optimal shape.

Or even better: use a **spline** parametrization. Place m control points and use cubic spline interpolation. This gives a smooth, flexible curve.

```python
"""
Optimizes non-negative sequences to minimize 2*n*max(self_convolution)/sum^2.
Uses a multi-pronged approach:
1. Low-dimensional parametrization (Fourier, spline) with L-BFGS for global search
2. Element-wise Adam optimization with log-sum-exp smooth max for local refinement
3. Diverse initializations including smoothed binary sequences
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import CubicSpline


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _smooth_max(c, beta):
    """Smooth approximation to max using log-sum-exp."""
    mx = c.max()
    if beta * (c.max() - c.min()) > 700:
        return mx  # prevent overflow
    return (mx + np.log(np.sum(np.exp(beta * (c - mx))))) / beta


def _seq_from_fourier(w, n, n_harm):
    """Generate sequence from Fourier coefficients using softmax."""
    x = np.arange(n, dtype=np.float64)
    f = np.zeros(n)
    for k in range(n_harm):
        f += w[2*k] * np.cos(2*np.pi*(k+1)*x/n) + w[2*k+1] * np.sin(2*np.pi*(k+1)*x/n)
    f = f - f.max()
    a = np.exp(f)
    s = a.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return a / s


def _fourier_obj(w, n, n_harm, beta):
    a = _seq_from_fourier(w, n, n_harm)
    c = _fft_convolve(a)
    return _smooth_max(c, beta)


def _seq_from_spline(ctrl, n, offset=0.01):
    """Generate sequence from spline control points."""
    m = len(ctrl)
    x_ctrl = np.linspace(0, 1, m)
    x = np.linspace(0, 1, n)
    try:
        cs = CubicSpline(x_ctrl, ctrl)
        a = cs(x)
    except Exception:
        a = np.interp(x, x_ctrl, ctrl)
    a = np.maximum(a, 0.0) + offset
    s = a.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return a / s


def _spline_obj(params, n, n_ctrl, offset, beta):
    """Objective for spline optimization. params = [ctrl_values..., ]"""
    ctrl = params
    a = _seq_from_spline(ctrl, n, offset)
    c = _fft_convolve(a)
    return _smooth_max(c, beta)


def _adam_optimize(a0, time_budget, rng, beta_start=10.0, lr=0.01):
    """Adam optimizer in log-space with log-sum-exp objective."""
    n = len(a0)
    log_a = np.log(np.maximum(a0, 1e-30))
    # Normalize: log_a - mean(log_a) to handle scale invariance
    log_a = log_a - np.mean(log_a)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a0.copy()
    c = _fft_convolve(a0)
    best_score = 2.0 * n * c.max()
    
    deadline = time.monotonic() + time_budget
    lr_current = lr
    beta = beta_start
    last_print = time.monotonic()
    no_improve = 0
    
    while time.monotonic() < deadline:
        t += 1
        
        # Get current sequence
        a = np.exp(log_a)
        a = a / a.sum()
        
        # Compute objective and gradient
        c = _fft_convolve(a)
        beta_c = beta * c
        max_bc = beta_c.max()
        if max_bc > 700:
            beta *= 0.5
            beta_c = beta * c
            max_bc = beta_c.max()
        weights = np.exp(beta_c - max_bc)
        weights /= weights.sum()
        
        score = 2.0 * n * c.max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 15:
                print(f"    [adam] t={t}, score={score:.6f}, lr={lr_current:.6f}, beta={beta:.1f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        # Gradient of smooth_max w.r.t. a:
        # d/d(a_j) [sum_k w_k * conv_k] where w_k = softmax(beta * c)
        # = sum_k w_k * d(conv_k)/d(a_j)
        # d(conv_k)/d(a_j) = 2 * a_{k-j} (if 0 <= k-j < n)
        # So grad[j] = 2 * sum_k w_k * a_{k-j}
        
        # This is 2 * cross-correlation of a and weights
        # grad[j] = 2 * sum_k weights[k] * a[k-j] (for valid k-j)
        # Let l = k-j: grad[j] = 2 * sum_l weights[l+j] * a[l]
        # This is the cross-correlation of a and (reversed weights) at specific lags
        
        # Use FFT for efficiency
        N = _next_pow2(n + 2*n - 2)
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(weights, N)
        # cross_corr[j] = sum_l a[l] * weights[l+j]
        # = ifft(conj(fft(a)) * fft(weights))[j]  (circular, but with zero-padding = linear)
        xcorr = np.fft.irfft(np.conj(fa) * fw, N)
        grad_a = 2.0 * xcorr[:n]
        
        # Gradient w.r.t. log_a:
        # a_j = exp(log_a_j) * a_j / sum(exp(log_a)) ... 
        # Actually, a = exp(log_a) / sum(exp(log_a))
        # d(a_j)/d(log_a_i) = a_j * (delta_{ji} - a_i)
        # So grad_log_a[i] = sum_j grad_a[j] * a_j * (delta_{ji} - a_i)
        # = grad_a[i] * a_i - a_i * sum_j grad_a[j] * a_j
        
        grad_log_a = grad_a * a - a * np.dot(grad_a, a)
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_log_a
        v = b2 * v + (1 - b2) * grad_log_a ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        log_a = log_a - lr_current * m_hat / (np.sqrt(v_hat) + eps)
        
        # Keep log_a bounded
        log_a = np.clip(log_a, -20, 20)
        
        # Adaptive learning rate
        if no_improve > 500:
            lr_current *= 0.95
        elif no_improve > 200:
            lr_current *= 0.99
        elif no_improve < 20:
            lr_current = min(lr_current * 1.01, lr * 2)
        
        # Gradually increase beta (sharpen the max approximation)
        if t % 100 == 0:
            beta = min(beta * 1.02, 50.0)
        
        # Periodic perturbation
        if no_improve > 800:
            log_a = log_a + rng.normal(0, 0.1, n)
            log_a = log_a - np.mean(log_a)
            no_improve = 0
            lr_current = lr
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
    
    return best_a, best_score


def _subgrad_optimize(a0, time_budget, rng):
    """Multiplicative subgradient descent (from previous code, improved)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s
    
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    
    eta = 0.12
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.75
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 15:
                print(f"    [subgrad] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        if no_improve > 600:
            a = best_a * np.exp(rng.normal(0, 0.1, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.12
            momentum = np.zeros(n)
            it += 1
            continue
        
        g = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            indices = k_max - j_range
            g[lo:hi+1] = 2.0 * a[indices]
        
        g_max = g.max()
        if g_max < 1e-15:
            break
        g = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a = a / s
        
        if no_improve > 400:
            eta *= 0.97
        elif no_improve < 30:
            eta = min(eta * 1.005, 0.3)
        
        it += 1
    
    return best_a, best_score
