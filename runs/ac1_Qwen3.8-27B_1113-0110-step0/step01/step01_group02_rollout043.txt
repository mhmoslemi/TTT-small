import numpy as np
import time
import scipy.sparse

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

                a = a * np.exp(-0.04 * g_sub)
                a = np.maximum(a, 1e-20)
                a = a / a.sum()

        it += 1

    return best_a, best_score


def _optimize_subgrad(a0, time_budget, rng):
    """Multiplicative subgradient with momentum (for polishing)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()

    eta = 0.06
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.7
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 10.0:
                print(f"    [SG] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 600:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.06
            momentum = np.zeros(n)
            it += 1
            continue

        g = _subgradient_at_max(a, k_max)
        g_max = g.max()
        if g_max < 1e-15:
            break
        g = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g

        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        a = a / a.sum()

        if no_improve > 350:
            eta *= 0.98
        elif no_improve < 60:
            eta = min(eta * 1.004, 0.15)

        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
    with multiplicative subgradient polishing and multiple initializations.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 15.0

    best_overall = None
    best_overall_score = np.inf

    # Build initializations
    initializations = []

    # Previous best (highest priority)
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            a = hs / hs.sum()
            initializations.append(("prev_best", a))
            sc = _score(a)
            best_overall = a.copy()
            best_overall_score = sc
            print(f"Previous best loaded: n={len(a)}, score={sc:.6f}")
    except (NameError, Exception):
        pass

    # Sinusoidal variations
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        for amp, phase in [(0.4, np.pi/4), (0.5, 0.0), (0.35, np.pi/3)]:
            a = 1.0 + amp * np.sin(2.0 * np.pi * x / n + phase)
            a = np.maximum(a, 0.01)
            a = a / a.sum()
            initializations.append((f"sin_{n}_a{amp}_p{phase:.2f}", a))

    # Multi-frequency
    for n in [2000]:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.25 * np.sin(2.0 * np.pi * x / n) + 0.15 * np.sin(4.0 * np.pi * x / n + 0.7)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
        initializations.append((f"multi_{n}", a))

    # Exponential decay
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        for rate in [0.6, 1.0, 1.5]:
            a = np.exp(-rate * x / n) + 0.05
            a = a / a.sum()
            initializations.append((f"decay_{n}_r{rate}", a))

    # Triangular (double exponential)
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        for center in [0.35, 0.5]:
            cp_ = n * center
            a = np.exp(-2.0 * np.abs(x - cp_) / (n * 0.4))
            a = a / a.sum()
            initializations.append((f"tri_{n}_c{center}", a))

    # Uniform
    for n in [2000, 3000]:
        a = np.ones(n, dtype=np.float64) / n
        initializations.append((f"unif_{n}", a))

    # Ramp
    for n in [2000]:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
        initializations.append((f"ramp_{n}", a))

    print(f"Total initializations: {len(initializations)}")

    # Score and sort
    scored = []
    for name, a0 in initializations:
        sc = _score(a0)
        scored.append((sc, name, a0))

    scored.sort(key=lambda x: x[0])
    print("Initial scores (top 10):")
    for i, (sc, name, _) in enumerate(scored[:10]):
        print(f"  {i+1}. {name}: {sc:.6f}")

    # Allocate time: give most to best candidates
    n_top = min(len(scored), 7)
    remaining_time = budget_s - 30.0
    weights = [max(1.0, float(n_top - i) ** 1.5) for i in range(n_top)]
    total_w = sum(weights)
    times = [remaining_time * w / total_w for w in weights]

    for i in range(n_top):
        if time.monotonic() > deadline:
            break

        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(15.0, deadline - time.monotonic()))

        print(f"\nOptimizing '{name}' (n={len(a0)}, init={init_score:.6f}, time={t_alloc:.0f}s)")

        # Phase 1: LP optimization (65% of time)
        t_lp = t_alloc * 0.65
        a1, s1 = _optimize_lp(a0, t_lp, rng, eps=0.88)
        eval1 = evaluate_sequence(a1.tolist())
        print(f"  LP: internal={s1:.6f}, eval={eval1:.6f}")

        if np.isfinite(eval1) and eval1 < best_overall_score:
            best_overall_score = eval1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {eval1:.6f} (from {name}) ***")

        # Phase 2: Subgradient polishing (35% of time)
        t_sg = t_alloc * 0.35
        a2, s2 = _optimize_subgrad(a1, t_sg, rng)
        eval2 = evaluate_sequence(a2.tolist())
        print(f"  SG: internal={s2:.6f}, eval={eval2:.6f}")

        if np.isfinite(eval2) and eval2 < best_overall_score:
            best_overall_score = eval2
            best_overall = a2.copy()
            print(f"  *** NEW BEST: {eval2:.6f} (from {name}) ***")

    # Final polishing from global best
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 20:
        print(f"\nFinal polishing with {remaining:.0f}s remaining...")

        # Try LP with tighter eps
        t_final_lp = remaining * 0.5
        a_p, sp = _optimize_lp(best_overall, t_final_lp, rng, eps=0.85)
        eval_p = evaluate_sequence(a_p.tolist())
        print(f"  Final LP (eps=0.85): {eval_p:.6f}")
        if np.isfinite(eval_p) and eval_p < best_overall_score:
            best_overall_score = eval_p
            best_overall = a_p.copy()
            print(f"  *** NEW BEST: {eval_p:.6f} ***")

        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_p2, sp2 = _optimize_subgrad(best_overall, remaining2, rng)
            eval_p2 = evaluate_sequence(a_p2.tolist())
            print(f"  Final SG: {eval_p2:.6f}")
            if np.isfinite(eval_p2) and eval_p2 < best_overall_score:
                best_overall_score = eval_p2
                best_overall = a_p2.copy()
                print(f"  *** NEW BEST: {eval_p2:.6f} ***")

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s")
        return [float(x) for x in best_overall]
    else:
        a = np.ones(1000, dtype=np.float64)
        print("Fallback to uniform")
        return a.tolist()
