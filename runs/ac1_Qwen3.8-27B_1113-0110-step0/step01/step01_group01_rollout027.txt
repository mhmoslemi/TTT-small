"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 using:
  - Nesterov-accelerated mirror descent with geometrically annealed smoothing
  - AdaGrad per-coordinate adaptive step sizes
  - Structured initializations (quadratic residues, Fourier mixes, keystone)
  - Multi-scale length sweep
  - Flattening heuristic for plateau escape

import time
import numpy as np


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def make_uniform(n):
    return np.ones(max(1, int(n)), dtype=np.float64) / float(max(1, int(n)))


def _is_prime(k):
    if k < 2:
        return False
    if k < 4:
        return True
    if k % 2 == 0 or k % 3 == 0:
        return False
    i = 5
    while i * i <= k:
        if k % i == 0 or k % (i + 2) == 0:
            return False
        i += 6
    return True


def make_quadratic_residue_smooth(n, seed=0):
    n = max(16, int(n))
    p = max(n, 7)
    while not _is_prime(p):
        p += 1
    qr = np.zeros(p, dtype=np.float64)
    for i in range(1, p):
        if pow(i, (p - 1) // 2, p) == 1:
            qr[i] = 1.0
    if p >= n:
        seq = qr[:n].copy()
    else:
        seq = np.tile(qr, (n // p) + 1)[:n].copy()
    klen = max(3, n // 80)
    if klen % 2 == 0:
        klen += 1
    half = klen // 2
    kernel = np.exp(-0.5 * (np.arange(klen) - half) ** 2 / max(1.0, half * half / 4.0))
    kernel /= kernel.sum()
    seq = np.convolve(seq, kernel, mode='same')
    return normalize_positive(seq, floor=1e-10)


def make_fourier_mix(n, seed=42):
    n = max(16, int(n))
    rng = np.random.default_rng(seed)
    n_freqs = max(4, min(n // 20, 30))
    freqs = rng.choice(range(1, n // 2 + 1), size=n_freqs, replace=False)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    amps = rng.uniform(0.15, 0.5, size=n_freqs) / (1.0 + 0.4 * np.sort(freqs))
    x = np.arange(n, dtype=np.float64)
    seq = np.ones(n, dtype=np.float64)
    for f, ph, a in zip(freqs, phases, amps):
        seq += a * np.sin(2.0 * np.pi * f * x / n + ph)
    return normalize_positive(seq, floor=1e-10)


def make_keystone(n, alpha=0.4):
    n = max(4, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    peak = 1.0 + alpha
    base = 1.0 - alpha * 0.3
    seq = np.where(x <= 0.5, base + (peak - base) * (1.0 - 2.0 * x),
                   base + (peak - base) * (2.0 * x - 1.0))
    return normalize_positive(np.maximum(seq, 0.01), floor=1e-10)


def make_triangular_envelope(n, center=0.5, width=0.6):
    n = max(4, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    seq = np.maximum(0.0, 1.0 - np.abs(x - center) / width)
    seq = seq + 0.05
    return normalize_positive(seq, floor=1e-10)


def resample_to_length(a, n):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(4, int(n))
    if old_n == n:
        return a.astype(np.float64, copy=True)
    if old_n == 1:
        return make_uniform(n)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    return normalize_positive(np.interp(new_x, old_x, a), floor=1e-12)


def compute_score_and_c(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf, np.array([])
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.inf, np.array([])
    p = a / s
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(p, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    cmax = float(np.max(c))
    score = 2.0 * n * cmax
    return score, c


def compute_grad_weights(c, cmax, tau, n):
    L = 2 * n - 1
    if cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return ez / s


def grad_from_weights(w, fa, N, n):
    w_rev = w[::-1].copy()
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return conv[n - 1:2 * n - 1][::-1].copy()


def smooth_obj_value(c, cmax, tau):
    if cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return cmax
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return cmax + tau * np.log(s)


def mirror_step(loga, g, lr, N, n):
    new_loga = loga - lr * g
    new_loga = new_loga - float(np.max(new_loga))
    a_new = np.exp(np.clip(new_loga, -60.0, 0.0))
    a_new = np.maximum(a_new, 1e-300)
    s = float(a_new.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return a_new, new_loga
    a_new = a_new / s
    a_new = np.maximum(a_new, 1e-300)
    new_loga = np.log(a_new)
    new_loga = new_loga - float(np.max(new_loga))
    return a_new, new_loga


def nesterov_mirror_descent(p_init, tau_start, tau_end, max_iter, deadline,
                            N, n, momentum_base=0.85, lr_base=0.45,
                            ada_decay=0.995, flatten_every=600, verbose=False):
    p = normalize_positive(p_init, floor=1e-15)
    loga = np.log(p + 1e-20)
    loga = loga - float(np.max(loga))

    best_a = p.copy()
    score, c = compute_score_and_c(p, N)
    best_score = score
    cmax = float(np.max(c))

    g = np.zeros(n, dtype=np.float64)
    y_loga = loga.copy()
    mu = momentum_base
    t_count = 0.0
    ada = np.ones(n, dtype=np.float64) * 0.1

    current_tau = tau_start
    total_tau_range = float(np.log(max(tau_start, 1e-15) / max(tau_end, 1e-15)))
    if total_tau_range < 0.01:
        total_tau_range = 6.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_count += 1.0
        t_now = min(1.0, t_count / float(max(max_iter, 1)))
        current_tau = max(tau_end, tau_start * np.exp(-total_tau_range * t_now))
        current_mu = min(0.98, momentum_base + (0.98 - momentum_base) * t_now * 0.5)

        score, c = compute_score_and_c(p, N)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = p.copy()
        cmax = float(np.max(c))
        if cmax <= 0.0 or not np.isfinite(cmax):
            break

        w = compute_grad_weights(c, cmax, current_tau, n)
        fa = np.fft.rfft(p, N)
        g_new = grad_from_weights(w, fa, N, n)
        g_new -= g_new.mean()
        g = (1.0 - current_mu) * g + current_mu * g_new
        g_norm = float(np.sqrt(np.dot(g, g)))
        if g_norm < 1e-15:
            break

        g = g / (g_norm + 1e-15)
        ada = ada_decay * ada + (1.0 - ada_decay) * (g * g)
        lr = lr_base * (1.0 + 0.3 * (1.0 - t_now)) / (np.sqrt(ada) + 1e-8)
        lr = np.minimum(lr, 2.5)

        y_loga = (1.0 - current_mu) * loga + current_mu * y_loga
        a_try, new_loga = mirror_step(y_loga, g, 1.0, N, n)

        c_try = convolve_seq(a_try, N)
        score_try = 2.0 * n * float(np.max(c_try))

        best_local = best_score
        for scale in (1.0, 0.6, 0.35, 0.2):
            a_try, new_loga = mirror_step(y_loga, g, scale, N, n)
            c_try = convolve_seq(a_try, N)
            score_try = 2.0 * n * float(np.max(c_try))
            if np.isfinite(score_try):
                if score_try < score * (1.0 - 1e-12):
                    p = a_try
                    loga = new_loga
                    score = score_try
                    if score < best_score * (1.0 - 1e-12):
                        best_score = score
                        best_a = p.copy()
                    break

        if it > 0 and it % flatten_every == 0 and best_score - score < 0.005:
            p_flatten = p.copy()
            threshold = float(np.percentile(p_flatten, 75))
            high_mask = p_flatten > threshold
            low_mask = p_flatten < float(np.percentile(p_flatten, 25))
            if high_mask.any() and low_mask.any():
                excess = float(p_flatten[high_mask].sum() - high_mask.sum() * float(np.mean(p_flatten)))
                deficit = float(low_mask.sum() * float(np.mean(p_flatten)) - p_flatten[low_mask].sum())
                transfer = 0.15 * min(excess, deficit)
                if transfer > 0:
                    p_flatten[high_mask] -= transfer / int(high_mask.sum())
                    p_flatten[low_mask] += transfer / int(low_mask.sum())
                    p_flatten = normalize_positive(p_flatten, floor=1e-15)
                    sc_f = 2.0 * n * float(np.max(convolve_seq(p_flatten, N)))
                    if np.isfinite(sc_f) and sc_f < best_score * 1.01:
                        p = p_flatten
                        loga = np.log(p + 1e-20)

                        score, c = compute_score_and_c(p, N)

                            best_a = p.copy()
                    else:
                        p = best_a.copy()
                        loga = np.log(p + 1e-20)
                        loga = loga - float(np.max(loga))
                        score, c = compute_score_and_c(p, N)

        if verbose and it % 50 == 0:
            print("  NMD n={} it={} tau={:.4f} score={:.9f} best={:.9f}".format(
                n, it, current_tau, score, best_score), flush=True)

    return best_a, best_score


def fast_gradient_descent(p_init, max_iter, deadline, N, n):

    best_score = score
    base_lr = 0.8 / np.sqrt(float(n))
    it = 0
    while it < max_iter and time.monotonic() < deadline:
        score, c = compute_score_and_c(p, N)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = p.copy()
        cmax = float(np.max(c))
        if cmax <= 0 or not np.isfinite(cmax):
            break
        w = compute_grad_weights(c, cmax, 0.03 * cmax, n)
        fa = np.fft.rfft(p, N)
        g = grad_from_weights(w, fa, N, n)
        g -= g.mean()
        g_norm = float(np.sqrt(np.dot(g, g)))
        if g_norm < 1e-15:
            break
        g = g / g_norm
        p_new = p - base_lr * g
        p_new = normalize_positive(p_new, floor=1e-15)
        sc_new = 2.0 * n * float(np.max(convolve_seq(p_new, N)))
        if np.isfinite(sc_new) and sc_new < score * (1.0 + 1e-6):
            p = p_new
            if sc_new < best_score * (1.0 - 1e-12):
                best_score = sc_new
                best_a = p.copy()
        else:
            base_lr *= 0.7
        it += 1
    return best_a, best_score

    final_eval = np.inf
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        pass
    if not np.isfinite(final_eval):
        best_a_overall = make_uniform(1024)
        out_list = [float(x) for x in best_a_overall]
        try:
            final_eval = float(evaluate_sequence(out_list))
        except Exception:
            final_eval = 2.0

    elapsed = time.monotonic() - start_time
    print("=== FINAL: score={:.12f} eval={:.12f} n={} time={:.1f}s ===".format(
        best_score_overall, final_eval, len(out_list), elapsed), flush=True)
    return out_list
