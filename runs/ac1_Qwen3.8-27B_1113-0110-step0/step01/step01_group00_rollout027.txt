import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C

def _lp_solve(f, M):
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-10, 'primal_feasibility_tolerance': 1e-10})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None

def _alternating_lp_aggressive(a_init, max_iters, deadline, tight_factor=0.995, initial_blend=0.6):
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M_current = float(np.max(c))
        # Try with slightly tighter M
        M_tight = M_current * tight_factor
        g = _lp_solve(a, M_tight)
        if g is None:
            # Fall back to current M
            g = _lp_solve(a, M_current)
            if g is None:
                break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        blend = max(0.15, blend * 0.995)
    
    return best_a, best_score

import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):

    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    """Build convolution matrix C where (C @ g)[k] = sum_j f[k-j] * g[j] for valid indices."""
    n = len(f)
    m = 2 * n - 1
    C = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        # c[k] = sum_i a[i] * g[k-i], fixed a, variable g
        # c[k] = sum_j g[j] * a[k-j] (where a[k-j] is valid)
        # For row k, column j: coefficient is a[k-j] if 0 <= k-j < n, else 0
        for k in range(j, j + n):
            C[k, j] = f[k - j]

def _lp_minimize_peak(f, eps):
    """Solve min M s.t. C@f'*g <= M, sum(g)=1, g in [max(0,f-eps), f+eps]"""
    n = len(f)
    C = _build_conv_matrix(f)
    m = 2 * n - 1
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M (coefficient vector: [0,...,0, 1])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints:
    # 1) C @ g - M <= 0  (i.e., C @ g <= M for each row)
    #    [C, -1] @ [g, M] <= 0
    A1 = np.hstack([C, -np.ones((m, 1))])
    b1 = np.zeros(m)
    
    # 2) -g_j <= -lower_j (i.e., g_j >= lower_j)
    lower = np.maximum(0.0, f - eps)
    A2 = -np.eye(n)
    b2 = -lower
    
    # 3) g_j <= upper_j (i.e., g_j <= upper_j)
    upper = f + eps
    A3 = np.eye(n)
    b3 = upper
    
    # Combine inequality constraints
    A_ub = np.vstack([A1, A2, A3])
    b_ub = np.concatenate([b1, b2, b3])
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: M >= 0
    bounds = [(None, None)] * n + [(0.0, None)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-12,
                                  'primal_feasibility_tolerance': 1e-12})
        if result.status == 0:
            return result.x[:n], result.x[-1]
    except Exception as e:
        pass
    return None, None

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C

def _lp_minimize_peak(f, eps, deadline=None):
    """Minimize M s.t. (f*g)[k] <= M, sum(g)=1, |g-f|_inf <= eps, g>=0."""
    n = len(f)
    m = 2 * n - 1
    C = _build_conv_matrix(f)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    b1 = np.zeros(m)

    lower = np.maximum(0.0, f - eps)
    # g <= f + eps
    upper = f + eps

    b_ub = np.concatenate([b1, -lower, upper])

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def _sensitivity(a, k):
    """Gradient of c[k] = sum_j a[j]*a[k-j] with respect to a."""
    n = len(a)
    j = np.arange(n)
    idx1 = k - j  # a[k-j] coefficient for a[j]
    idx2 = j - k  # a[j-k] coefficient for a[j]
    s = np.zeros(n)
    m1 = (idx1 >= 0) & (idx1 < n)
    s[m1] += a[idx1[m1]]
    m2 = (idx2 >= 0) & (idx2 < n)
    s[m2] += a[idx2[m2]]
    # c[k] = sum_j a[j]*a[k-j]
    # d/d a[j] of c[k] = a[k-j] + a[j-k] (the two terms where a[j] appears)
    # But this double-counts when k-j = j, i.e., k = 2j
    # Actually no: c[k] = sum_j a[j] * a[k-j]
    # The derivative w.r.t. a[j0] is: a[k-j0] (from the term where j=j0) + a[j0-k] (from the term where k-j=j0, i.e., j=k-j0)
    # But these are the same term when j0 = k-j0, i.e., k = 2*j0
    # In that case, the derivative is 2*a[j0]
    # My computation above handles this correctly:
    # m1: when k-j0 is valid, contributes a[k-j0]
    # m2: when j0-k is valid, contributes a[j0-k]
    # When k = 2*j0, both m1 and m2 are true for j=j0, so we get a[j0] + a[j0] = 2*a[j0]. Correct!
    return s

def _nesterov_subgradient(a_init, max_iters, deadline, n_top_lags=5, lr_init=0.01, momentum=0.9):
    """Nesterov-accelerated subgradient method targeting top lags."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    # Nesterov state
    y = a.copy()  # extrapolated point
    best_a = a.copy()
    best_score = _score(a)
    t_k = 1.0
    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        c = _fft_convolve(y)
        top_k = np.argpartition(c, -n_top_lags)[-n_top_lags:]
        
        grad = np.zeros(n)
        for k in top_k:
            grad += _sensitivity(y, int(k))
        grad /= len(top_k)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        
        # Step
        y_new = y - lr * grad
        # Project onto {a >= 0, sum(a) = S}
        y_new = np.maximum(y_new, 0.0)
        s_new = float(np.sum(y_new))
        if s_new > 1e-10:
            y_new = y_new * S / s_new
        
        # Nesterov extrapolation
        a_new = y_new + beta * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = float(np.sum(a_new))
        if s_a > 1e-10:
            a_new = a_new * S / s_a
        
        y = y_new
        a = a_new
        t_k = t_new
        lr *= 0.9999  # Slow decay
        
        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

    x_src = np.linspace(0, 1, n_src)

    out = np.maximum(out, 0.0)

from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_minimize_peak(f, eps, time_limit=10.0):
    """
    Solve: min M s.t. (f*g)[k] <= M for all k,
           sum(g) = 1, max(0, f-eps) <= g <= f+eps, g >= 0.
    Returns g or None.
    """
    n = len(f)
    m = 2 * n - 1
    C = _build_conv_matrix(f)

    # Variables: [g_0, ..., g_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # C @ g - M <= 0  =>  [C | -1] @ [g; M] <= 0
    A1 = np.hstack([C, -np.ones((m, 1))])
    b1 = np.zeros(m)

    lower = np.maximum(0.0, f - eps)
    upper = f + eps

    # g >= lower => -g <= -lower
    A_ub = np.vstack([A1, -np.eye(n), np.eye(n)])
    b_ub = np.concatenate([b1, -lower, upper])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(lower[j], upper[j]) for j in range(n)] + [(0.0, None)]

    try:
        opts = {'presolve': True, 'time_limit': time_limit}
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options=opts)
        if result.status == 0:
            g = result.x[:n]
            if np.all(g >= -1e-10):
                return np.maximum(g, 0.0)
    except Exception:
        pass
    return None


def _lp_alternating(f, M, time_limit=10.0):
    """Solve max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns g or None."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        opts = {'presolve': True, 'time_limit': time_limit}
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options=opts)
        if result.status == 0:
            g = result.x
            if np.all(g >= -1e-10):
                return np.maximum(g, 0.0)
    except Exception:
        pass
    return None


def _sensitivity(a, k):
    """Compute d/d a of c[k] where c = conv(a, a)."""
    n = len(a)
    j = np.arange(n)
    idx1 = k - j
    idx2 = j - k
    s = np.zeros(n)
    m1 = (idx1 >= 0) & (idx1 < n)
    s[m1] += a[idx1[m1]]
    m2 = (idx2 >= 0) & (idx2 < n)
    s[m2] += a[idx2[m2]]
    return s


def _nesterov_subgradient(a_init, max_iters, deadline, n_top_lags=5, lr_init=0.005):
    """Nesterov-accelerated subgradient targeting top lags."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    y = a.copy()
    best_a = a.copy()
    best_score = _score(a)
    t_k = 1.0
    lr = lr_init
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(y)
        top_k = np.argpartition(c, -n_top_lags)[-n_top_lags:]

        grad = np.zeros(n)
        for k in top_k:
            grad += _sensitivity(y, int(k))
        grad /= len(top_k)

        # Adaptive step: reduce if gradient is large
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-8:
            step = lr / (1.0 + 0.1 * g_norm)
        else:
            step = 0.0

        y_new = y - step * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = float(np.sum(y_new))
        if s_y > 1e-10:
            y_new = y_new * S / s_y

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new if t_new > 0 else 0.0

        a_new = y_new + beta * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = float(np.sum(a_new))
        if s_a > 1e-10:
            a_new = a_new * S / s_a

        y = y_new
        a = a_new
        t_k = t_new

        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 500:
                lr *= 0.9
                stagnant = 0

        if it % 10000 == 0 and it > 0:
            print(f"      Nesterov iter {it}: score={best_score:.6f}")

    return best_a, best_score


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def _bell_initial(n, center=0.5, width=0.18):
    """Gaussian bell-shaped initial sequence."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - center) ** 2) / (2.0 * width ** 2))
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_diverse_initial(n, idx, rng):
    """Create diverse initial sequences."""
    t = np.linspace(0, 1, n)
    if idx == 0:
        a = _bell_initial(n, 0.5, 0.15)
    elif idx == 1:
        a = _bell_initial(n, 0.5, 0.25)
    elif idx == 2:
        a = _bell_initial(n, 0.4, 0.2)
    elif idx == 3:
        a = _bell_initial(n, 0.6, 0.2)
    elif idx == 4:
        a = t * (1 - t) * 4.0
    elif idx == 5:
        a = np.exp(-2.0 * t) * (1.0 + 0.3 * np.sin(2.0 * np.pi * 3.0 * t))
        a = np.maximum(a, 0.0)
    elif idx == 6:
        a = (1.0 - t) ** 1.5
    elif idx == 7:
        a = np.abs(rng.normal(0, 1, n))
    elif idx == 8:
        a = np.exp(-((t - 0.3) ** 2) / 0.03) + 0.7 * np.exp(-((t - 0.7) ** 2) / 0.05)
    else:
        a = _bell_initial(n, 0.5, 0.18)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _run_direct_lp(a_init, deadline, n_top=5):
    """Run direct LP peak minimization with adaptive eps."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_score = _score(a)
    eps = 0.05
    prev_score = best_score
    stagnant = 0

    for it in range(5000):
        if time.monotonic() > deadline:
            break
        if eps < 0.0005:
            break

        g = _lp_minimize_peak(a, eps, time_limit=5.0)
        if g is None:
            eps = min(0.5, eps * 2.0)
            if eps > 0.3:
                break
            continue

        s_g = float(np.sum(g))
        if s_g < 0.5:
            break
        g = g / s_g

        # Blend: use 1.0 initially (take LP solution directly), reduce later
        blend = 1.0 if stagnant == 0 else 0.7
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score = _score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            stagnant = 0
            # Decrease eps when making progress
            eps *= 0.97
        else:
            stagnant += 1
            if stagnant > 50:
                eps *= 0.9
                stagnant = 0

        a = a_new
        prev_score = score

        if it % 100 == 0 and it > 0:
            print(f"      LP iter {it}: score={best_score:.6f}, eps={eps:.5f}")

    return best_a, best_score


def _run_alternating_lp(a_init, deadline, tight_factor=0.99):
    """Run alternating LP with tight constraints."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_score = _score(a)
    blend = 0.6

    for it in range(3000):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M_current = float(np.max(c))
        M_tight = M_current * tight_factor

        g = _lp_alternating(a, M_tight, time_limit=5.0)
        if g is None:
            g = _lp_alternating(a, M_current, time_limit=5.0)
            if g is None:
                break

        s_g = float(np.sum(g))
        if s_g < 1e-10:
            break
        g = g / s_g

        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        blend = max(0.2, blend * 0.998)

        if it % 200 == 0 and it > 0:
            print(f"      AltLP iter {it}: score={best_score:.6f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimizer minimizing J(a)=2n*max(conv(a,a))/sum(a)^2.
    Uses direct LP peak minimization, alternating LP, and Nesterov subgradient.
    """
    total_deadline = time.monotonic() + 2340.0
    rng = np.random.default_rng(42)

    best_a = None
    best_score = float('inf')

    # Load parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            # Check if parent is all zeros (score would be inf)
            if float(np.sum(parent)) > 0.01:
                parent_score = _score(parent)
                parent_score_real = evaluate_sequence([float(x) for x in parent])
                print(f"Parent: n={parent.size}, score={parent_score:.6f}, verified={parent_score_real:.6f}")
                best_a = parent.copy()
                best_score = parent_score
            else:
                print("Parent is degenerate (sum too small), ignoring.")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Phase 1: Direct LP at multiple n values
    print("\n=== Phase 1: Direct LP Peak Minimization ===")
    n_list = [300, 400, 500]

    for n in n_list:
        if time.monotonic() > total_deadline - 400:
            break
        phase_deadline = min(total_deadline - 200, time.monotonic() + 650.0)
        print(f"\n  --- n={n} ---")

        # Build list of initializations
        inits = []
        if best_a is not None:
            inits.append(_interpolate_sequence(best_a, n))
            inits.append(_bell_initial(n, 0.5, 0.15))
            inits.append(_bell_initial(n, 0.5, 0.22))
            inits.append(_make_diverse_initial(n, 6, rng))
        else:
            for idx in range(9):
                inits.append(_make_diverse_initial(n, idx, rng))

        n_inits = len(inits)
        time_per_init = 600.0 / n_inits

        for init_idx, a_init in enumerate(inits):
            init_deadline = min(phase_deadline, time.monotonic() + time_per_init)
            if time.monotonic() > phase_deadline:
                break

            a_opt, score = _run_direct_lp(a_init, init_deadline)
            print(f"    init {init_idx}: score={score:.6f}")

            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    ** New global best: {best_score:.6f} (n={n})")

    print(f"\nAfter Phase 1: best={best_score:.6f}, n={len(best_a) if best_a is not None else 0}")

    # Phase 2: Alternating LP on best solution
    print("\n=== Phase 2: Alternating LP Refinement ===")
    if best_a is not None and time.monotonic() < total_deadline - 300:
        n_best = len(best_a)
        # Try at n_best and a smaller n for LP efficiency
        for n_try in [min(n_best, 400), min(n_best, 300)]:
            if time.monotonic() > total_deadline - 100:
                break
            a_work = _interpolate_sequence(best_a, n_try) if n_best != n_try else best_a.copy()
            a_opt, score = _run_alternating_lp(a_work, total_deadline - 100, tight_factor=0.99)

            if n_best != n_try:
                a_opt = _interpolate_sequence(a_opt, n_best)
                score = _score(a_opt)

            print(f"    AltLP n={n_try}: score={score:.6f}")
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    ** New global best: {best_score:.6f}")

    # Phase 3: Nesterov subgradient
    print("\n=== Phase 3: Nesterov Subgradient ===")
    if best_a is not None and time.monotonic() < total_deadline - 100:
        for n_try in [len(best_a), len(best_a) + 50]:
            if time.monotonic() > total_deadline - 30:
                break
            if n_try != len(best_a):
                a_init = _interpolate_sequence(best_a, n_try)
            else:
                a_init = best_a.copy()

            sub_deadline = min(total_deadline - 20, time.monotonic() + 300.0)
            a_opt, score = _nesterov_subgradient(a_init, 1000000, sub_deadline,
                                                  n_top_lags=5, lr_init=0.003)
            print(f"  Nesterov n={n_try}: score={score:.6f}")

            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  ** New global best: {best_score:.6f}")

    # Phase 4: Final LP polish with very small eps
    print("\n=== Phase 4: Final LP Polish ===")
    if best_a is not None and time.monotonic() < total_deadline - 20:
        n_best = len(best_a)
        n_lp = min(n_best, 350)
        a_work = _interpolate_sequence(best_a, n_lp) if n_best != n_lp else best_a.copy()
        a_work = a_work / float(np.sum(a_work))

        polish_deadline = total_deadline - 5
        eps = 0.005
        for it in range(2000):
            if time.monotonic() > polish_deadline:
                break
            if eps < 0.0003:
                break
            g = _lp_minimize_peak(a_work, eps, time_limit=3.0)
            if g is None:
                eps *= 1.5
                continue
            s_g = float(np.sum(g))
            if s_g < 0.5:
                break
            g = g / s_g
            a_work = 0.6 * a_work + 0.4 * g
            a_work = np.maximum(a_work, 0.0)
            s_w = float(np.sum(a_work))
            if s_w < 1e-10:
                break
            a_work = a_work / s_w
            eps *= 0.96

        if n_best != n_lp:
            a_final = _interpolate_sequence(a_work, n_best)
        else:
            a_final = a_work
        score = _score(a_final)
        print(f"  Polish final: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a_final.copy()
            print(f"  ** New global best: {best_score:.6f}")

    # Final verification
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a) if best_a is not None else 0} ===")
    if best_a is not None and len(best_a) > 0:
        result = [float(x) for x in best_a]
        verified = float(evaluate_sequence(result))
        print(f"Verified score: {verified:.6f}")
        return result

    # Absolute fallback
    return [1.0] * 100
