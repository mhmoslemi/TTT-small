Optimize non-negative step function to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses peak-targeting gradient descent with FFT, simulated annealing,
momentum, coordinate descent polish, and diverse multi-scale initializations.
Focuses on top-fraction convolution peaks with adaptive step sizes.

import numpy as np
import time


def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_convolve(a, b=None):
    if b is None:
        b = a
    na = int(a.size)
    nb = int(b.size)
    L = na + nb - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a.astype(np.float64), n=N)
    fb = np.fft.rfft(b.astype(np.float64), n=N)
    return np.fft.irfft(fa * fb, n=N).real[:L]


def _fft_grad_peak(a, top_frac=0.05, sharpness=30.0):
    """Compute gradient of weighted peak sum w.r.t. a using FFT."""
    n = int(a.size)
    b = _fft_convolve(a, a)
    L = len(b)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.zeros(n, dtype=np.float64), b, maxb
    # Soft top-fraction weighting
    thresh = (1.0 - top_frac) ** (1.0 / sharpness)
    # w[k] = 1 if b[k] > thresh*maxb, tapering below
    ratio = b / maxb
    w = np.where(ratio > thresh, 1.0, np.maximum(0.0, (ratio - thresh * 0.5) / (thresh * 0.5)))
    ws = float(w.sum())
    if ws <= 0:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        ws = 1.0
    else:
        w = w / ws
    # Gradient: ∂(sum_k w[k]*b[k])/∂a[j]
    # = cross-correlation of w with a (approximately)
    N = _next_pow2(L + n - 1)
    fw = np.fft.rfft(w, n=N)
    fa = np.fft.rfft(a, n=N)
    # The gradient of sum_k w[k] * sum_i a[i]*a[k-i] w.r.t. a[j]
    # = 2 * (w * a)[j] in the right window
    c = np.fft.irfft(fw * fa, n=N)[:n]
    # More precisely: grad[j] = sum_k w[k] * (a[k-j] + (1 if k-j!=j else 0)*a[k-j])
    # For large n and smooth w, approximate as 2 * correlation
    grad = 2.0 * c
    return grad, b, maxb

    b = _fft_convolve(a, a)

    return float(2.0 * n * maxb / (s * s))

    if m < 1:

        v = rng.normal(0.0, 1.0, m)
    v -= float(v.mean())
    std = float(v.std())
    if std > 1e-12:
        v = amp * v / std
    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0
    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    return a


def _random_init(n, rng, smoothness=0.0):
    """Random positive initialization with optional smoothing."""
    v = rng.normal(0.0, 1.0, size=n)
    if smoothness > 0:
        # Simple moving average smoothing
        k = max(1, int(smoothness))
        kernel = np.ones(k) / k
        v = np.convolve(v, kernel, mode='same')
    a = np.exp(v)
    a = np.maximum(a, 1e-15)
    return a


def _optimize_gradient(a0, time_limit, rng, label=""):
    """Main gradient-based optimizer with momentum, SA, and adaptive focus."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1e6)
    a = np.maximum(a, 1e-15)
    a = _normalize(a)

    best_score = float('inf')
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    eta = 0.5
    eta_min = 0.005
    eta_max = 10.0
    momentum = 0.7
    v_prev = np.zeros(n, dtype=np.float64)
    top_frac = 0.5
    top_frac_min = 0.002
    top_frac_decay = 0.998
    sharpness = 10.0
    temperature = 5.0
    temp_decay = 0.9995
    no_improve = 0
    max_no_improve = 2000
    resets = 0
    max_resets = 40

    while time.monotonic() - t0 < time_limit:
        b = _fft_convolve(a, a)
        maxb = float(b.max())
        s = float(a.sum())
        score = float(2.0 * n * maxb / (s * s)) if s > 0.01 else float('inf')

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.4)
            # Perturb from best
            scale = 0.05 + 0.1 * (resets % 5)
            a = best_a * np.exp(rng.normal(0.0, scale, size=n))
            a = np.maximum(a, 1e-15)
            a = _normalize(a)
            eta = 0.5
            top_frac = max(top_frac, 0.2)
            temperature = max(temperature, 1.0)
            v_prev = np.zeros(n, dtype=np.float64)
            continue

        # Compute gradient
        grad, b_cur, maxb_cur = _fft_grad_peak(a, top_frac=top_frac, sharpness=sharpness)
        maxg = float(np.max(np.abs(grad)))
        if maxg < 1e-20:
            # Try larger perturbation
            a = a * np.exp(rng.normal(0.0, 0.1, size=n))
            a = np.maximum(a, 1e-15)
            a = _normalize(a)
            it += 1
            continue
        grad = grad / maxg

        # Momentum update
        v = momentum * v_prev + (1.0 - momentum) * grad
        v_prev = v

        # Multiplicative update
        a_new = a * np.exp(-eta * v)
        a_new = np.maximum(a_new, 1e-300)
        a_new = _normalize(a_new)

        # Evaluate
        b_new = _fft_convolve(a_new, a_new)
        maxb_new = float(b_new.max())
        s_new = float(a_new.sum())
        score_new = float(2.0 * n * maxb_new / (s_new * s_new)) if s_new > 0.01 else float('inf')

        # Accept/reject with simulated annealing
        accept = False
        if score_new < best_score - 1e-14:
            accept = True
        elif score_new < score - 1e-14:
            accept = True
        elif score_new < score * 1.01:
            delta = score_new - score
            if np.log(rng.uniform(1e-30, 1.0)) > delta / max(temperature, 1e-10):
                accept = True

        if accept:
            a = a_new
            score = score_new
            eta = min(eta * 1.03, eta_max)
        else:
            eta = max(eta * 0.8, eta_min)
            if eta < 0.02:
                a = a * np.exp(rng.normal(0.0, 0.05, size=n))
                a = np.maximum(a, 1e-15)
                a = _normalize(a)
                eta = 0.3
                v_prev = np.zeros(n, dtype=np.float64)

        # Adaptive parameters
        top_frac = max(top_frac * top_frac_decay, top_frac_min)
        sharpness = min(sharpness * 1.005, 200.0)
        temperature = max(temperature * temp_decay, 0.01)
        it += 1

        if verbose_check(it, 5000):
            elapsed = time.monotonic() - t0
            print(f"  [{label}] it={it} t={elapsed:.1f}s score={score:.6f} best={best_score:.6f} "
                  f"eta={eta:.3f} top={top_frac:.4f} temp={temperature:.3f} resets={resets}")

    return best_a, best_score


def verbose_check(it, interval):
    return (it % interval == 0)


def _coordinate_descent(a0, time_limit, rng, label=""):
    """Coordinate descent: try adjusting one element at a time."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.maximum(a, 0.0)
    best_score = _score_internal(a)
    if not np.isfinite(best_score):
        return a, best_score
    best_a = a.copy()

    t0 = time.monotonic()
    sweeps = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random subset of indices for this sweep
        order = rng.permutation(n)
        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(idx)
            old_val = a[i]
            if old_val < 1e-15:
                # Try small positive values
                for trial in [1e-4, 1e-3, 1e-2, 0.01 / n]:
                    a[i] = trial
                    sc = _score_internal(a)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                    a[i] = old_val
            else:
                # Try multiplicative perturbations
                for factor in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 0.0]:
                    a[i] = old_val * factor
                    a[i] = max(0.0, a[i])
                    sc = _score_internal(a)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                    a[i] = old_val
        a = best_a.copy()
        sweeps += 1
        if not improved and sweeps > 2:
            # Perturb and restart
            a = best_a * np.exp(rng.normal(0.0, 0.05, size=n))
            a = np.maximum(a, 0.0)
            sc = _score_internal(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            else:
                a = best_a.copy()
            if sweeps > 10:
                break

    return best_a, best_score


def _block_optimize(a0, time_limit, rng, n_blocks=8, label=""):
    """Block-wise optimization: fix all but one block, optimize that block."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    block_size = n // n_blocks
    if block_size < 4:
        return a, _score_internal(a)

    best_score = _score_internal(a)
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved_any = False
        for blk in range(n_blocks):
            if time.monotonic() - t0 > time_limit:
                break
            start = blk * block_size
            end = start + block_size if blk < n_blocks - 1 else n
            # Optimize this block with a few iterations
            block_a = a[start:end].copy()
            for _tr in range(20):
                b = _fft_convolve(a, a)
                maxb = float(b.max())
                s = float(a.sum())
                sc = float(2.0 * n * maxb / (s * s))
                # Gradient for this block
                w = np.zeros(len(b), dtype=np.float64)
                maxb = float(b.max())
                if maxb <= 0:
                    break
                ratio = b / maxb
                top_k = max(1, int(0.05 * len(b)))
                top_idx = np.argsort(ratio)[-top_k:]
                w[top_idx] = 1.0 / top_k
                N = _next_pow2(len(w) + n - 1)
                fw = np.fft.rfft(w, n=N)
                fa = np.fft.rfft(a, n=N)
                c = np.fft.irfft(fw * fa, n=N)[:n]
                grad_block = 2.0 * c[start:end]
                mg = float(np.max(np.abs(grad_block)))
                if mg < 1e-20:
                    break
                block_a = block_a * np.exp(-0.3 * grad_block / mg)
                block_a = np.maximum(block_a, 1e-15)
                a[start:end] = block_a
                bc = float(a.sum())
                if bc > 0.01:
                    a = _normalize(a)
                    block_a = a[start:end].copy()
            sc_new = _score_internal(a)
            if sc_new < best_score - 1e-12:
                best_score = sc_new
                best_a = a.copy()
                improved_any = True

        if not improved_any:
            # Perturb
            a = best_a * np.exp(rng.normal(0.0, 0.03, size=n))
            a = np.maximum(a, 1e-15)
            a = _normalize(a)
            sc = _score_internal(a)
            if sc >= best_score:
                a = best_a.copy()
                if time.monotonic() - t0 > time_limit * 0.5:
                    break

    return best_a, best_score


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float('inf')
    try:
        return float(ev(seq))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400):
    """Main search function. Returns best sequence found."""
    seed = int(seed)
    total_budget = max(120.0, float(budget_s))
    now0 = time.monotonic()
    hard_deadline = now0 + total_budget - 10.0

    print(f"=== Step function optimizer started, budget={total_budget:.0f}s ===")
    best_a = None
    best_score = float('inf')

    # Try to use previously found sequence
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                sc = _score_internal(gn)
                print(f"Given sequence: n={gn.size}, score={sc:.6f}")
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = gn.copy()
                if g.size <= 32768:
                    given_a = gn
    except Exception as e:
        print(f"Could not load given: {e}")

    # Task list: (n, strategy, duration, params)
    # We'll explore multiple sizes and strategies
    tasks = []
    if given_a is not None:
        tasks.append((given_a.size, "gradient", 180, {"top_frac": 0.3}))

    tasks.append((8192, "gradient", 300, {"top_frac": 0.3}))
    tasks.append((16384, "gradient", 400, {"top_frac": 0.2}))
    tasks.append((8192, "gradient", 250, {"top_frac": 0.1}))
    tasks.append((16384, "gradient", 350, {"top_frac": 0.15}))
    tasks.append((8192, "block", 200, {"n_blocks": 8}))
    tasks.append((8192, "gradient", 200, {"top_frac": 0.5}))

    task_idx = 0
    for n, strategy, duration, params in tasks:
        now = time.monotonic()
        if now >= hard_deadline - 120:
            break
        alloc = min(duration, hard_deadline - now - 80)
        if alloc < 30:
            break

        task_seed = (seed * 7919 + task_idx * 104729) % (2**31)
        rng = np.random.default_rng(task_seed)
        print(f"\nTask {task_idx}: n={n} strategy={strategy} alloc={alloc:.0f}s params={params}")

        # Generate initial
        if task_idx == 0 and given_a is not None:
            a0 = given_a.copy()
            if a0.size != n:
                # Interpolate or truncate
                a0 = _anti_init(n, rng, smooth=True)
        elif strategy == "gradient":
            tf = params.get("top_frac", 0.3)
            if tf < 0.15:
                a0 = _anti_init(n, rng, freq_scale=2.0, amp=5.0, smooth=False)
            elif tf < 0.25:
                a0 = _anti_init(n, rng, freq_scale=1.0, amp=4.0, smooth=True)
            else:
                a0 = _anti_init(n, rng, freq_scale=0.5, amp=3.0, smooth=True)
            # Add some randomness
            a0 = a0 * (1.0 + 0.2 * rng.normal(size=n))
            a0 = np.maximum(a0, 1e-15)
        elif strategy == "block":
            a0 = _anti_init(n, rng, freq_scale=1.5, amp=4.0, smooth=False)
        else:
            a0 = _random_init(n, rng, smoothness=2)

        a0 = _normalize(a0)
        init_score = _score_internal(a0)
        print(f"  Initial score: {init_score:.6f}")

        if strategy == "gradient":
            cand_a, cand_score = _optimize_gradient(a0, alloc, rng, label=f"{strategy}{n}_{task_idx}")
        elif strategy == "block":
            # First do some gradient, then block
            grad_time = alloc * 0.4
            block_time = alloc * 0.6 - 5
            cand_a, cand_score = _optimize_gradient(a0, grad_time, rng, label=f"pre{strategy}")
            cand_a, cand_score2 = _block_optimize(cand_a, block_time, rng, n_blocks=params.get("n_blocks", 8), label=f"block{task_idx}")
            cand_score = min(cand_score, cand_score2)
        else:
            cand_a, cand_score = _optimize_gradient(a0, alloc, rng, label=f"{strategy}{task_idx}")

        if cand_a is not None:
            cand_a = np.nan_to_num(cand_a, nan=0.0, posinf=0.0, neginf=0.0)
            cand_a = np.clip(cand_a, 0.0, 1e6)
            cs = float(cand_a.sum())
            if cs > 0.01:
                cand_a = _normalize(cand_a)
                internal = _score_internal(cand_a)
                if np.isfinite(internal) and internal < cand_score:
                    cand_score = internal
                if np.isfinite(cand_score) and cand_score < best_score:
                    best_score = cand_score
                    best_a = cand_a.copy()
                    print(f"  ** New best: {best_score:.6f} (n={n}) **")

        print(f"  Task {task_idx} done: best={best_score:.6f}, elapsed={time.monotonic()-now0:.0f}s")
        task_idx += 1

    # Final refinement phase
    if best_a is not None:
        remaining = hard_deadline - time.monotonic()
        print(f"\n=== Final refinement phase, remaining={remaining:.0f}s ===")
        if remaining > 60:
            # Gradient refinement
            grad_time = min(120.0, remaining * 0.5)
            if grad_time > 20:
                rng_ref = np.random.default_rng((seed + 55555) % (2**31))
                print(f"  Gradient refinement ({grad_time:.0f}s)...")
                ref_a, ref_score = _optimize_gradient(best_a, grad_time, rng_ref, label="final_grad")
                if np.isfinite(ref_score) and ref_score < best_score:
                    best_score = ref_score
                    best_a = ref_a
                    print(f"  After gradient: {best_score:.6f}")

            # Coordinate descent polish
            remaining2 = hard_deadline - time.monotonic()
            if remaining2 > 20 and best_a.size <= 20000:
                cd_time = min(60.0, remaining2 * 0.6)
                if cd_time > 10:
                    rng_cd = np.random.default_rng((seed + 77777) % (2**31))
                    print(f"  Coordinate descent ({cd_time:.0f}s)...")
                    cd_a, cd_score = _coordinate_descent(best_a, cd_time, rng_cd, label="cd")
                    if np.isfinite(cd_score) and cd_score < best_score:
                        best_score = cd_score
                        best_a = cd_a
                        print(f"  After CD: {best_score:.6f}")

            # One more gradient pass
            remaining3 = hard_deadline - time.monotonic()
            if remaining3 > 15:
                rng_final = np.random.default_rng((seed + 99999) % (2**31))
                final_time = min(40.0, remaining3 - 5)
                if final_time > 10:
                    print(f"  Final gradient pass ({final_time:.0f}s)...")
                    fa2, fs2 = _optimize_gradient(best_a, final_time, rng_final, label="final2")
                    if np.isfinite(fs2) and fs2 < best_score:
                        best_score = fs2
                        best_a = fa2
                        print(f"  Final: {best_score:.6f}")

    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        best_score = 2.0

    # Final cleanup
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a = best_a / ss

    final_seq = [float(x) for x in best_a]

    # Verify with exact evaluator if available
    if best_a.size <= 50000 and time.monotonic() < hard_deadline - 3.0:
        exact = _safe_evaluate(final_seq)
        print(f"\n=== FINAL: n={best_a.size}, internal_score={_score_internal(best_a):.6f}, exact={exact:.6f} ===")
    else:
        print(f"\n=== FINAL: n={best_a.size}, internal_score={_score_internal(best_a):.6f} ===")

    return final_seq
