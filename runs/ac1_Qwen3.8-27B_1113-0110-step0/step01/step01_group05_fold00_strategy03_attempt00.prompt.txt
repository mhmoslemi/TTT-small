def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _paired_initial(n, rng, alpha_w=1.0, alpha_split=0.3):
    n = int(n)

        return np.ones(max(1, n), dtype=np.float64)
    m = (n + 1) // 2
    weights = rng.dirichlet(np.full(m, float(alpha_w), dtype=np.float64))
    p = np.zeros(n, dtype=np.float64)
    split_alpha = max(0.01, float(alpha_split))
    for r in range(m):
        a = r
        b = n - 1 - r

            p[a] = weights[r]

def _expand_to_length(proto, target_len):
    target_len = int(target_len)
    p = _normalize(proto)
    R = int(p.size)
    if R <= 0 or target_len <= 0:
        return np.ones(max(1, target_len), dtype=np.float64)
    if R == target_len:
        return p.copy()
    if R > target_len:
        out = np.ones(target_len, dtype=np.float64)
        for j in range(target_len):
            lo = (j * R) // target_len
            hi = max(lo + 1, ((j + 1) * R) // target_len)
            chunk = p[lo:hi]
            out[j] = float(np.mean(chunk)) if chunk.size > 0 else 1.0
        return _normalize(out)
    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=np.int64)
    if rem > 0:
        lengths[:rem] += 1
    out = np.zeros(target_len, dtype=np.float64)
    idx = 0
    for i in range(R):
        L = int(lengths[i])
        if L > 0:
            out[idx:idx + L] = p[i] / float(L)
            idx += L
    return _normalize(out)

def _lp_polish(p, c, score, deadline, max_iters=2, max_n=96):
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except Exception:
        return p, c, score

    n = int(p.size)
    if n < 2 or n > int(max_n):
        return p, c, score

    p = _normalize(p)
    c = np.convolve(p, p)
    score = _score_normalized(p, c)

    best_p = p.copy()
    best_c = c.copy()
    best_score = score

    for _ in range(int(max_iters)):
        if time.monotonic() > deadline - 30.0:
            break
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            break

        i_idx = np.repeat(np.arange(n), n)
        j_idx = np.tile(np.arange(n), n)
        k_idx = i_idx + j_idx
        vals = np.tile(p, n)
        K = sparse.csr_matrix((vals, (k_idx, i_idx)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        obj = -np.ones(n, dtype=np.float64)

        try:
            res = linprog(
                c=obj,
                A_ub=K,
                b_ub=b_ub,
                bounds=[(0.0, None)] * n,
                method="highs",
                options={"presolve": True}
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False):
            break
        g = np.asarray(res.x, dtype=np.float64)
        np.clip(g, 0.0, None, out=g)
        gs = float(np.sum(g))
        if not np.isfinite(gs) or gs <= 1.0:
            break
        g = g / gs

        cg = np.convolve(g, g)
        cross = np.convolve(p, g)

        best_t = 0.0
        best_val = float(score)
        ts = np.linspace(0.0, 1.0, 33)
        for t in ts:
            a = float(1.0 - t)
            b = float(t)
            ct = a * a * c + 2.0 * a * b * cross + b * b * cg
            val = _score_normalized(np.ones(n, dtype=np.float64), ct)
            if val < best_val:
                best_val = val
                best_t = float(t)

        if best_val >= float(score) - 1.0e-10:
            break

        lo = max(0.0, best_t - 0.25)
        hi = min(1.0, best_t + 0.25)
        if hi > lo:
            ts2 = np.linspace(lo, hi, 49)
            for t in ts2:
                a = float(1.0 - t)
                b = float(t)
                ct = a * a * c + 2.0 * a * b * cross + b * b * cg
                val = _score_normalized(np.ones(n, dtype=np.float64), ct)
                if val < best_val:
                    best_val = val
                    best_t = float(t)

        if best_val >= float(score) - 1.0e-10:
            break

        p = (1.0 - best_t) * p + best_t * g
        p = _normalize(p)
        c = np.convolve(p, p)
        np.clip(c, 0.0, None, out=c)
        score = _score_normalized(p, c)
        if score < best_score - 1.0e-12:
            best_score = score
            best_p = p.copy()
            best_c = c.copy()

    return best_p, best_c, best_score
