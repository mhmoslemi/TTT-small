Let me implement the LP-based iterative method:

```python
def lp_step(f, M):
    """Given f (normalized, sum=1) and M = max(f*f), find g that maximizes sum(g)
    subject to (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    # Build constraint matrix A where A[k,j] = f[j] if 0 <= k-j < n
    # Number of constraints: 2n-1
    # Number of variables: n
    
    # Use scipy linprog
    from scipy.optimize import linprog
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    # Build A_ub and b_ub
    # A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # b_ub[k] = M for all k
    
    # The matrix is sparse, so use sparse format
    from scipy.sparse import coo_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A_sparse = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return f  # fallback
    
```

import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c))

def _build_sparse_conv_matrix(f, n, active_lags=None):
    """Build sparse matrix A where (A @ g)[k] = (f*g)[k] for k in active_lags."""
    if active_lags is None:
        active_lags = range(2*n - 1)
    
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_lags):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        rows.extend([idx] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[js].tolist())
    
    return coo_matrix((vals, (rows, cols)), shape=(len(list(active_lags)), n))

def _lp_step(f, M, n_active=200, time_limit=60):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active lags, g >= 0."""
    n = len(f)
    
    # Select active lags (where f*f is closest to M)
    ff = _fft_convolve(f)
    if n_active < 2*n - 1:
        # Get top n_active lags
        k_all = np.arange(2*n - 1)
        # We want lags where ff[k] is high (close to M)
        threshold = M * 0.8  # use lags where ff >= 0.8*M
        active = k_all[ff >= threshold]
        if len(active) > n_active:
            # Keep the top n_active
            top_idx = np.argsort(ff[active])[::-1][:n_active]
            active = active[top_idx]
    else:
        active = np.arange(2*n - 1)
    
    if len(active) == 0:
        active = np.argmax(ff).reshape(1)
    
    A = _build_sparse_conv_matrix(f, n, active)
    
    c_obj = -np.ones(n)
    b_ub = np.full(len(active), M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-15:
            return g / s
    return f.copy()

def _subgrad_optimize(a0, time_budget, rng, eta=0.15, mu=0.7):
    """Multiplicative subgradient optimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s
    
    c = _fft_convolve(a)
    best_score = float(np.max(c)) * 2 * n
    best_a = a.copy()
    
    no_improve = 0
    momentum = np.zeros(n)
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 5.0:
                print(f"    [sg] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        if no_improve > 500:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.15
            momentum = np.zeros(n)
            it += 1
            continue
        
        # Subgradient
        g = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            indices = k_max - j_range
            g[lo:hi + 1] = 2.0 * a[indices]
        
        g_max = g.max()
        if g_max < 1e-15:
            break
        g_norm = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g_norm
        
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a = a / s
        
        if no_improve > 300:
            eta *= 0.96
        elif no_improve < 20:
            eta = min(eta * 1.005, 0.4)
        
        it += 1
    
    return best_a, best_score
