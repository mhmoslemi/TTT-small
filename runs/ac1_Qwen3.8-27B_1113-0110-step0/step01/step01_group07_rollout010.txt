def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:

    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)

    s = float(e.sum())

def compute_gradient(p, tau, L):
    """Compute the z-space gradient of the smooth max objective."""
    n = p.size
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.zeros(n, dtype=np.float64), inf
    
    # Smooth max weights
    z = (c - cm) / tau
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:

        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    
    w = expz / expsum  # weights for smooth max
    
    # p-space gradient: g[j] = 2 * sum_m w[m] * p[m-j]
    # = 2 * (w_rev * p)[2n-2-j] where w_rev[m] = w[2n-2-m]
    w_rev = w[::-1]

    mean_g = float(np.dot(p, g_p))
    grad_z = p * (g_p - mean_g)
    
    return grad_z, 2.0 * n * cm

def cheb_basis(n, D):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1, 1)
    ang = np.arccos(t)
    out = np.empty((n, D), dtype=np.float64)
    for m in range(1, D + 1):
        out[:, m - 1] = np.cos(m * ang)
    return out

def poly_parent(B, coeffs, eps=0.12):
    val = B.dot(coeffs)
    h = 1.0 + eps + val
    return normalize_positive(h)
