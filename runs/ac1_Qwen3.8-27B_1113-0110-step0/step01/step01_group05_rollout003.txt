def _coord_update(p, c, score, i, grid, n):
    """Update coordinate i. Returns (p_new, c_new, score_new, improved)."""
    old = p[i]

    if D < 0:

    # Convolution slice for position i
    ks_start = max(0, i)

    if hi < 2*n - 1:

def _coord_pass(p, c, score, deadline, rng, grid, max_inner=1):

    improved_total = False
    for _ in range(max_inner):
        if time.monotonic() > deadline - 0.5:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline - 0.3:
                break
            p, c, score, imp = _coord_update(p, c, score, int(i), grid, n)
            if imp:
                improved = True
        # Renormalize
        s = float(np.sum(p))
        if s > 1e-15:
            p = p / s
            c = np.convolve(p, p)
            np.clip(c, 0.0, None, out=c)
            score = _score(p, c)
        if not improved:
            break
        improved_total = True
    return p, c, score, improved_total

def _lp_direction(a, deadline, max_lp_iters=8, frac_active=0.15):
    """Find LP direction using cutting plane method."""
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except Exception:
        return None

    n = a.size
    c = np.convolve(a, a)
    M = float(np.max(c))
    if not np.isfinite(M) or M < 1e-12:
        return None

    # Start with most binding constraints
    k_all = np.arange(2*n - 1)
    threshold = M * (1.0 - frac_active)
    active_ks = k_all[c >= threshold]
    if active_ks.size == 0:
        active_ks = k_all[c >= 0.9 * M]
    if active_ks.size < 3:
        active_ks = k_all[np.argsort(c)[-max(3, n//10):]]

    g = None
    for lp_iter in range(max_lp_iters):
        if time.monotonic() > deadline - 10:
            break

        # Build sparse constraint matrix for active constraints
        # Row for constraint k: K[k, j] = a[k-j] if 0 <= k-j <= n-1
        rows = active_ks
        nnz = 0
        for k in rows:
            lo_j = max(0, k - (n - 1))
            hi_j = min(n - 1, k)
            nnz += (hi_j - lo_j + 1)

        if nnz > 50_000_000:
            break  # Too large

        row_idx = []
        col_idx = []
        data = []
        for k in rows:
            lo_j = max(0, k - (n - 1))
            hi_j = min(n - 1, k)
            for j in range(lo_j, hi_j + 1):
                row_idx.append(int(k - active_ks[0]) if False else int(np.searchsorted(active_ks, k)))
                col_idx.append(j)
                data.append(a[k - j])

        # Actually let me be more careful with row indexing
        row_idx = []
        col_idx = []
        data = []
        for ii, k in enumerate(rows):
            lo_j = max(0, k - (n - 1))
            hi_j = min(n - 1, k)
            for j in range(lo_j, hi_j + 1):
                row_idx.append(ii)
                col_idx.append(j)
                data.append(a[k - j])

        if len(row_idx) == 0:
            break

        K = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(len(rows), n))
        b_ub = np.full(len(rows), M, dtype=np.float64)
        obj = -np.ones(n, dtype=np.float64)

        try:
            res = linprog(
                c=obj, A_ub=K, b_ub=b_ub,
                bounds=[(0.0, None)] * n,
                method='highs',
                options={'time_limit': min(60.0, max(1.0, deadline - time.monotonic() - 5.0))}
            )
        except Exception:
            break

        if res is None or not getattr(res, 'success', False):
            break

        g = np.asarray(res.x, dtype=np.float64)
        np.clip(g, 0.0, None, out=g)
        gs = float(np.sum(g))
        if not np.isfinite(gs) or gs < 1e-6:
            break

        # Check feasibility against ALL constraints
        if g is not None and gs > 1e-6:

            viol = ag > M * (1 + 1e-6)

                active_ks = np.unique(np.concatenate([active_ks, viol_ks]))
                if active_ks.size > 2*n - 1:
                    active_ks = k_all
                continue
            else:
                g = g / gs
                return g
        else:
            break

    # Fallback: return last g normalized
    if g is not None and np.sum(g) > 1e-6:
        return g / np.sum(g)
    return None

def _line_search_blend(a, g, score_a, deadline, n_steps=20):
    """Find best t in [0,1] for a_new = (1-t)*a + t*g."""
    n = a.size
    cg = np.convolve(g, g)
    cross = np.convolve(a, g)
    ca = np.convolve(a, a)

    best_t = 0.0
    best_val = score_a

    ts = np.linspace(0.0, 1.0, n_steps)
    for t in ts:
        at = (1.0 - t) * a + t * g
        ct = (1-t)**2 * ca + 2*(1-t)*t * cross + t**2 * cg
        np.clip(ct, 0.0, None, out=ct)
        st = float(np.sum(at))
        if st < 1e-15:
            continue
        val = 2.0 * n * float(np.max(ct)) / (st * st)
        if np.isfinite(val) and val < best_val:
            best_val = val
            best_t = float(t)

    return best_t, best_val

                best_t, best_val = _line_search_blend(a, g, score, min(deadline - 60, time.monotonic() + 30.0))
                if best_t > 0.001 and best_val < score - 1e-6:
                    a = (1.0 - best_t) * a + best_t * g
                    a = _normalize(a)
                    c = np.convolve(a, a)
                    np.clip(c, 0.0, None, out=c)
                    score = _score(a, c)
                    try_update(a, c)
                    print(f"  LP step t={best_t:.3f} -> {score:.6f}")

        # Track stagnation
        if score < best_ever_score - 1e-5:
            best_ever_score = score
            stagnant = 0
        else:
            stagnant += 1

        # Perturbation to escape local minima
        if stagnant >= 5 and perturbs < max_perturbs:
            n = a.size
            scale = 0.3 * (1.0 - 0.8 * perturbs / max_perturbs)
            a_pert = a * np.exp(rng.normal(0, scale, n))
            a_pert = np.clip(a_pert, 0, None)
            sp = float(np.sum(a_pert))
            if sp > 1e-10:
                a_pert = a_pert / sp
                cp = np.convolve(a_pert, a_pert)
                np.clip(cp, 0, None, out=cp)
                scp = _score(a_pert, cp)
                if scp < score + 0.1:  # Allow some worsening
                    a = a_pert
                    c = cp
                    score = scp
            perturbs += 1
            stagnant = 0
            print(f"  Perturbed #{perturbs} -> {score:.6f}")

        # Progress report
        if int(time.monotonic() - start_time) % 60 < 120:
            pass

    print(f"Phase 1 done. Score: {best_score:.6f} at t={time.monotonic()-start_time:.0f}s")

    # Phase 2: Final intensive coordinate descent on best sequence
    if best_a is not None and time.monotonic() < deadline - 10:
        a = best_a.copy()
        c = np.convolve(a, a)
        np.clip(c, 0.0, None, out=c)
        score = _score(a, c)
        # Finer grid for final polish
        grid_fine = np.concatenate(([0.0], np.logspace(-8.0, 3.5, 40)))
        a, c, score, _ = _coord_pass(a, c, score, deadline - 10, rng, grid_fine, max_inner=4)
        try_update(a, c)

    # Phase 3: Try a few fresh starts with LP
    if time.monotonic() < deadline - 30 and best_score > 1.51:
        for trial in range(5):
            if time.monotonic() > deadline - 15:
                break
            a2 = _make_initial(target_len, rng, trial % 10)
            c2 = np.convolve(a2, a2)
            sc2 = _score(a2, c2)
            if sc2 < best_score:
                a = a2
                c = c2
                score = sc2
            # Quick coord pass
            a, c, score, _ = _coord_pass(a, c, score, min(deadline - 15, time.monotonic() + 60), rng, grid, max_inner=1)
            try_update(a, c)

    # Final evaluation
    out = best_a if best_a is not None else _two_level(target_len)
    out = _normalize(out)
    print(f"Final score (internal): {_score(out):.6f}")
    try:
        ext_score = evaluate_sequence([float(x) for x in out])
        print(f"Final score (external): {ext_score:.6f}")
    except Exception as e:
        print(f"Evaluation error: {e}")

    return [float(x) for x in out]
