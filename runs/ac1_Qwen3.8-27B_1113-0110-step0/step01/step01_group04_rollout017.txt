import numpy as np
from scipy.optimize import linprog
import time

def water_filling_lp(a, M):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k.
    a is the current sequence, M is the current peak level.
    """
    n = len(a)
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (a*b)_k <= M for k = 0, ..., 2n-2
    # (a*b)_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a_i * b_{k-i}
    
    A_ub = np.zeros((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i  # index of b
            A_ub[k, j] = a[i]
    
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        b = np.maximum(res.x, 0)
        if b.sum() > 1e-15:
            b = b / b.sum()
        return b
    return a.copy()

"""
Search for non-negative step-function heights minimizing 2n*max(a*a)/(sum a)^2.
Uses LP-based water-filling iterations + gradient polishing + local search.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csc_matrix

def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)

def _normalize(v):
    v = _clean(v)
    s = v.sum()
    if s < 1e-15:
        return np.full(v.size, 1.0/v.size) if v.size else v
    return v / s

def _score(a):
    """Compute 2n*max(a*a)/(sum a)^2 using FFT."""
    v = _clean(a)
    n = v.size
    s = v.sum()
    if s < 1e-15:
        return np.inf
    v = v / s
    # FFT-based convolution
    L = 1
    while L < 2*n-1:
        L *= 2
    fa = np.fft.rfft(v, L)
    c = np.fft.irfft(fa*fa, L)[:2*n-1]
    mx = c.max()
    return 2.0*n*mx

def _convolve_fft(a, b=None):
    if b is None:
        b = a
    n = a.size
    L = 1
    while L < 2*n-1:
        L *= 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa*fb, L)[:2*n-1]

def _lp_water_filling(a, M):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k.
    Returns the solution b normalized to sum=1.
    """
    n = len(a)
    
    c_obj = -np.ones(n)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    
    A_ub = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    
    return a.copy()

    rows = []
    cols = []
    vals = []
    
    for k in range(2*n-1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        i_range = np.arange(i_lo, i_hi + 1)
        j_range = k - i_range
        rows.extend(k for _ in i_range)
        cols.extend(j_range.tolist())
        vals.extend(a[i_range].tolist())

"""
Minimize f(a) = 2n*max((a*a)_k)/(sum a)^2 via LP water-filling + gradient polish + local search.
Uses FFT for convolution, HiGHS for LP, and adaptive step sizes.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csc_matrix


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    n = v.size
    if n == 0:
        return v
    if s < 1e-15 or not np.isfinite(s):
        return np.full(n, 1.0/n)
    return v / s


def _pow2(m):
    m = max(1, int(m))
    r = 1
    while r < m:
        r <<= 1
    return r


def _conv_fft(a, b=None):
    if b is None:
        b = a
    n = a.size
    L = _pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:2 * n - 1]


def _score(a):
    v = _normalize(a)
    n = v.size
    c = _conv_fft(v)
    mx = float(c.max()) if c.size else 0.0
    if not np.isfinite(mx) or mx < 0:
        mx = 0.0
    return 2.0 * n * mx


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / (r + 1)
    out = np.maximum(v - tau, 0.0)
    s = float(out.sum())
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    return out / s


def _build_lp_structure(n):
    """Build row/col indices for the LP constraint matrix."""
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    pos = 0
    for k in range(2 * n - 1):
        ilo = max(0, k - n + 1)
        ihi = min(n - 1, k)
        cnt = ihi - ilo + 1
        rows[pos:pos + cnt] = k
        cols[pos:pos + cnt] = k - np.arange(ilo, ihi + 1, dtype=np.int32)
        pos += cnt
    assert pos == total

For $Q_p(a) = \sum_k c_k^p$ where $c = a * a$:
$\frac{\partial Q_p}{\partial a_j} = 2p \sum_k c_k^{p-1} \cdot a_{k-j}$ (for valid $k-j$)

In convolution form: $= 2p \cdot (c^{p-1} \star a)_j$ where $\star$ is cross-correlation.

In FFT (with padding to length $L \geq 2n-1$):
$g = 2p \cdot \text{ifft}(\text{fft}(c^{p-1}, L) \cdot \overline{\text{fft}(a, L)})[:n]$

Let me verify: $\text{ifft}(\text{fft}(w, L) \cdot \overline{\text{fft}(a, L)})_j = \sum_i w_{(i+j) \mod L} a_i$.

For $j \in [0, n-1]$ and $L \geq 2n-1$: this equals $\sum_{i=0}^{n-1} w_{i+j} a_i$ (no wrap-around).

And $\sum_k c_k^{p-1} a_{k-j}$ with $k$ from 0 to $2n-2$: let $i = k - j$, then $\sum_i c_{i+j}^{p-1} a_i$. This matches the FFT cross-correlation at lag $j$.

So the gradient is:
$g_j = 2p \sum_i c_{i+j}^{p-1} a_i = 2p \cdot h_j$

where $h = \text{ifft}(\text{fft}(c^{p-1}, L) \cdot \overline{\text{fft}(a, L)})[:n]$.

On the simplex, the projected gradient is $g - \bar{g}$ (subtract mean to maintain the sum constraint).

    tl=time.monotonic()
    cands=_cands(n)
    scored=sorted([(_sc(c),c) for c in cands])
    scored=[(s,c) for s,c in scored if np.isfinite(s)]
    print(f"Phase 1: {len(scored)} cands, best={scored[0][0]:.6f} ({time.monotonic()-tl:.1f}s)")
    
    ba=scored[0][1].copy(); bs=scored[0][0]
    
    # Phase 2: LP
    tl=time.monotonic()
    print("Building LP structure...")
    rows,cols=_lpsi(n)
    ax=rows-cols
    print(f"LP structure: {time.monotonic()-tl:.1f}s")
    
    lp_end=t0+0.62*(dl-t0)
    for rank in range(min(3,len(scored))):
        if time.monotonic()>lp_end-20: break
        a,s=scored[rank]
        print(f"LP refinement #{rank}: initial={s:.6f}")
        a,sc=_lpr(a,lp_end,rows,cols,ax,f"LP{rank}")
        if sc<bs:
            ba,bs=a.copy(),sc
    
    print(f"After LP: {bs:.6f}")
