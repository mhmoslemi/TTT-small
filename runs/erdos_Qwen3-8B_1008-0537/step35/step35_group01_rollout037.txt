import scipy.optimize as optimize
import time

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]

    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_rough_waves(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def gen_harmonic(f, d):

    guesses = [h]
    for freq in range(2, 15):
        guesses.append(gen_harmonic(freq, 0.5))

        guesses.append(gen_saw(freq, 0.5))

    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):

        return res.x if res.success else g

    for g in guesses:
