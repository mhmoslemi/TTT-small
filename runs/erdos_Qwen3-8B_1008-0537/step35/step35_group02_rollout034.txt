    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def obj(h_vals):
        max_overlap = 0.0
        for k in np.linspace(0, 2, 1000):
            shifted = np.roll(h_vals, int((k / dx + 0.5) % n_points))
            overlap = np.sum(h_vals * (1 - shifted) * dx)
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=30, maxiter=50, mutation=(0.8, 1.2), recombination=0.9)
        return res.x

    for _ in range(4):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = optimize.minimize(obj, h_opt, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': cons}],
                                 options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-10}).x
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-10})
        return res.x if res.success else r

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
