import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_sine(freq, phase):
        x = np.linspace(0, 2, n_points)
        return 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for f in [1, 2, 3, 4]:
        guesses.append(gen_sine(f, 0.1))
        guesses.append(gen_sine(f, 0.5))
        guesses.append(gen_sine(f, 0.9))
    for d in [0.25, 0.5, 0.75]:
        guesses.append(gen_sine(2, d))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 250, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:

    def de_guess():

        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)

    def refine(r):
