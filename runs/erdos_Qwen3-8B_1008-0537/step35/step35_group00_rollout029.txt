import scipy.optimize as optimize
import time

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_segments(freq, duty, n=8):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        pulses = (wave + 1) * duty / 2
        return np.clip(pulses, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in [1, 2, 3, 4]:
        guesses.append(gen_segments(freq, 0.5))
        for duty in [0.4, 0.5, 0.6]:
            guesses.append(gen_segments(freq, duty))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:

        h_opt = de_guess()
