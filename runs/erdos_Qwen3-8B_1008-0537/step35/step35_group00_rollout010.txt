import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_binary():
        pos = np.random.uniform(0, 2, size=3)
        pos.sort()
        lengths = np.random.uniform(0.1, 0.3, size=3)
        mask = np.zeros(n_points)
        for i in range(3):
            start = np.searchsorted(np.linspace(0, 2, n_points), pos[i])
            end = np.searchsorted(np.linspace(0, 2, n_points), pos[i] + lengths[i])
            mask[start:end] = 1
        mask = mask / np.sum(mask) * target_sum
        return mask

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 10):
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.4, 0.6, 0.8]:
        guesses.append(gen_saw(3, duty))
        guesses.append(gen_triangle(3, duty))
    for _ in range(5):
        guesses.append(gen_random())
    for _ in range(5):
        guesses.append(gen_binary())

    def slsqp_guess(g):

        return res.x if res.success else g
