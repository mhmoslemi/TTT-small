import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < target_sum:
            h[i] = 1.0
    h = h.astype(float)

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    guesses = []
    for freq in range(3, 9):
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(5, duty))

    def slsqp_guess(g):
        try:
            res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 150, 'ftol': 1e-12})
            return res.x if res.success else g
        except:
            return g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        try:
            res = optimize.differential_evolution(obj, bounds, popsize=120, maxiter=70,
                                                 mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
            return res.x
        except:
            return np.random.rand(n_points)

    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        try:
            res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 200, 'ftol': 1e-12})
            return res.x if res.success else r

        best_h = refine(best_h)

    return best_h, best_c5, n_points
