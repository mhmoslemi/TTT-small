import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def gen_basic():
        h = np.zeros(n_points)
        h[::4] = 1
        h[::3] = 0
        h = h / np.sum(h) * target_sum
        return h

    def gen_spaced(density):
        h = np.zeros(n_points)
        step = n_points // density
        for i in range(density):
            h[i::step] = 1
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 11):
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_triangle(3, duty))
    for _ in range(5):
        guesses.append(gen_random())
    guesses.append(gen_basic())
    for d in [5, 6, 7, 8]:
        guesses.append(gen_spaced(d))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-12})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():

        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)

        if c < best_c5:

    def refine(r):
