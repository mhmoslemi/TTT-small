import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    phase = np.linspace(0, 1, n_points)

    def gen_pulse(freq, duty):
        wave = np.sin(2 * np.pi * freq * phase)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        wave = 1 - np.abs(2 * freq * phase - 1) * duty
        return np.clip(wave, 0, 1)

    def gen_tri(freq, duty):
        wave = 1 - np.abs(2 * freq * phase - 1) * duty
        return np.clip(wave, 0, 1)

    def gen_rand():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    freqs = np.linspace(2, 15, 10)
    for freq in freqs:
        guesses.append(gen_pulse(freq, 0.5))
        guesses.append(gen_saw(freq, 0.4))
        guesses.append(gen_tri(freq, 0.3))

    for _ in range(5):
        guesses.append(gen_rand())

    def de_optimize():
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=50,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(3):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_optimize()
        h_opt = optimize.minimize(obj, h_opt, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': cons}],
                                 options={'maxiter': 200, 'ftol': 1e-9}).x
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-9})
        return res.x if res.success else r

    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
