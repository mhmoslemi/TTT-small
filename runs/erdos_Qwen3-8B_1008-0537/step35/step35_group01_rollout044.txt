import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_binary():
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, size=n_points//2, replace=False)
        h[idx] = 1.0
        h = h / np.sum(h) * target_sum
        return h

    def gen_ternary(freq, phase):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((wave + 1) * 0.5, 0, 1)

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    guesses = [h, gen_binary()]

    for freq in [2, 3, 4, 5, 6]:
        guesses.append(gen_ternary(freq, 0.0))
        guesses.append(gen_ternary(freq, np.pi/2))
        guesses.append(gen_ternary(freq, np.pi))
        guesses.append(gen_ternary(freq, 3*np.pi/2))
    for freq in [2, 3, 4]:
        guesses.append(gen_pulses(freq, 0.5))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=60,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 40:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-9})
        return res.x if res.success else r

    for _ in range(15):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
