        return np.max(corr) * dx

    def cons(h_vals):

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

        for center in centers:

            end = min(2.0, center + 0.5 * w)
            idx_start = int(start * n_points / 2)

            h[idx_start:idx_end] = 1.0

    def gen_random():

    guesses = [h]

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,

    for g in guesses:
