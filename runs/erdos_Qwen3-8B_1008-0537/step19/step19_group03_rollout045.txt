import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess, method):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_binary_guess(num_ones):
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess]
    for num_ones in [15, 20, 25, 30]:
        for _ in range(3):
            initial_guesses.append(generate_binary_guess(num_ones))
    
    for guess in initial_guesses:
        result = optimize_guess(guess, 'SLSQP')
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=250,

            tol=1e-10,
            vectorized=True,

        )
        if result_de.success:
            result_cobyla = optimize.minimize(
                fun=objective,

                options={'maxiter': 200, 'disp': False}
            )
            if result_cobyla.success and objective(result_cobyla.x) < best_c5:
                best_c5 = objective(result_cobyla.x)
                best_h = result_cobyla.x
    except:
        pass
