import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic_pattern(n_points, period, start=0):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if (i - start) % period < period / 2:
                pattern[i] = 1.0
            else:
                pattern[i] = 0.0
        return pattern
    
    def generate_sine_pattern(n_points, freq):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            val = 0.5 + 0.5 * np.sin(2 * np.pi * freq * i / n_points)
            pattern[i] = np.clip(val, 0, 1)
        return pattern
    
    structured_patterns = []
    periods = [2, 3, 4, 5, 6]
    for period in periods:
        pattern = generate_periodic_pattern(n_points, period, 1)
        structured_patterns.append(pattern)
    
    for freq in [0.5, 0.6, 0.7]:
        pattern = generate_sine_pattern(n_points, freq)
        structured_patterns.append(pattern)

            rand[idx] += 0.5
