import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def normalize_h(h_values):
        current_sum = np.sum(h_values)
        target_sum = n_points / 2
        if current_sum == 0:
            return np.zeros_like(h_values) + 0.5
        scale = target_sum / current_sum
        scaled = h_values * scale
        scaled = np.clip(scaled, 0.0, 1.0)
        new_sum = np.sum(scaled)
        delta = target_sum - new_sum
        if delta != 0:
            scaled += delta / n_points
            scaled = np.clip(scaled, 0.0, 1.0)
        return scaled
    
    def objective(h_values):
        h = normalize_h(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_binary_guess():
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_regular_guess(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum < target_sum:
            remaining = target_sum - current_sum
            indices = np.where(h == 0)[0]
            for i in indices:
                if remaining <= 0:
                    break
                h[i] = min(1.0, h[i] + remaining / len(indices))
                remaining -= max(0.0, (1.0 - h[i]))
        return h
    
    initial_guesses = [initial_guess]
    for _ in range(5):
        initial_guesses.append(generate_binary_guess())
    initial_guesses.append(generate_regular_guess(3))
    initial_guesses.append(generate_regular_guess(5))
    
    for guess in initial_guesses:
        result = optimize_guess(guess)

            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=120,
            maxiter=200,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,

                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}

    except:
        pass

                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}

            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
