import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def make_symmetric(h):
        n = len(h)
        mirrored = np.zeros(n)
        for i in range(n):
            mirrored[i] = h[n - 1 - i]
        sum_mirrored = np.sum(mirrored)
        factor = (target_sum) / sum_mirrored if sum_mirrored != 0 else 1.0
        return mirrored * factor
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    structured_guesses = []
    # Symmetric initial guesses
    symmetric_guess = make_symmetric(initial_guess)
    structured_guesses.append(symmetric_guess)
    
    # Two-interval symmetric guess
    two_interval_sym = np.zeros(n_points)
    length = 0.5
    start = 0.25
    for i in range(n_points):
        x = i * dx
        if start <= x < start + length:
            two_interval_sym[i] = 1.0
        elif 2 - start - length <= x < 2 - start:
            two_interval_sym[i] = 1.0
    structured_guesses.append(two_interval_sym)
    
    # Three-interval symmetric guess
    three_interval_sym = np.zeros(n_points)
    length = 1.0 / 3.0
    start = 0.5
    for i in range(n_points):
        x = i * dx
        for offset in [0, 1, 2]:
            pos = start + offset * length
            if pos - length / 2 <= x < pos + length / 2:
                three_interval_sym[i] = 1.0
                break
    structured_guesses.append(three_interval_sym)
    
    initial_guesses = [initial_guess, symmetric_guess]
    initial_guesses.extend(structured_guesses)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=200,
            mutation=(0.6, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
