import scipy.optimize as optimize
import time

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def generate_block_patterns(n_points, block_counts):
        patterns = []

            pattern = np.zeros(n_points)
            for i in range(n_points):
                if i % m == 0:
                    pattern[i] = 1.0
            pattern[-1] = 0.5 if np.sum(pattern) < target_sum else 0.0
            patterns.append(pattern)
        return patterns
    
    structured_patterns = generate_block_patterns(n_points, [3, 4, 5])

            result = optimize.minimize(

                method='SLSQP',
                bounds=bounds,

                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}

            return result.x if result.success else None
