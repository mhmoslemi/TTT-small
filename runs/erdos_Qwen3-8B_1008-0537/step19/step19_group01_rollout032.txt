    for i in range(n_points):
        if i % 3 == 0:
            structured_h[i] = 1.0
        elif i % 3 == 1:
            structured_h[i] = 0.5
        else:
            structured_h[i] = 0.0
    structured_h[-1] = 0.5
    initial_guesses = [initial_guess, structured_h]
    
    # Two-interval guess
    two_interval = np.zeros(n_points)
    for i in range(len(initial_guesses)):
        start = 0.0 if i == 0 else 1.0
        end = start + 0.5
        for j in range(n_points):
            x = j * dx
            if start <= x < end:
                two_interval[j] = 1.0
    initial_guesses.append(two_interval)
    
    # Three-interval guess
    three_interval = np.zeros(n_points)
    for i in range(len(initial_guesses)):
        start = 0.0 if i == 0 else 0.8333333333 if i == 1 else 1.3333333333
        end = start + 0.3333333333
        for j in range(n_points):
            x = j * dx
            if start <= x < end:
                three_interval[j] = 1.0
    initial_guesses.append(three_interval)
    
    # Random initial guess
    random_guess = np.random.rand(n_points)
    total_sum = np.sum(random_guess)
    scaling_factor = (n_points / 2.0) / total_sum
    random_guess *= scaling_factor
    initial_guesses.append(random_guess)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=200,
            maxiter=300,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
