import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_symmetric_binary():
        half = n_points // 2
        h = np.zeros(n_points)
        for i in range(half):
            h[i] = np.random.choice([0, 1])
        h = np.concatenate([h, h[::-1][:-1]])
        diff = target_sum - np.sum(h)
        if diff > 0:
            idx = np.random.choice(np.where(h == 0)[0], int(diff * 2), replace=False)
            h[idx] = 0.5
        elif diff < 0:
            idx = np.random.choice(np.where(h == 1)[0], -int(diff * 2), replace=False)
            h[idx] = 0.5
        return h

    def generate_binary_varied(num_ones):
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        diff = target_sum - np.sum(h)
        if diff > 0:
            idx = np.random.choice(np.where(h == 0)[0], int(diff * 2), replace=False)
            h[idx] = 0.5
        elif diff < 0:
            idx = np.random.choice(np.where(h == 1)[0], -int(diff * 2), replace=False)
            h[idx] = 0.5
        return h

    initial_guesses = [initial_guess]
    for _ in range(5):
        initial_guesses.append(generate_symmetric_binary())
    for _ in range(5):
        initial_guesses.append(generate_binary_varied(np.random.randint(15, 30)))

    def optimize_guess(guess, method='SLSQP'):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None

    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result

    def differential_evolution_step():
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=120,
                maxiter=150,
                mutation=(0.7, 0.9),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                    best_c5 = objective(result_slsqp.x)
                    best_h = result_slsqp.x
        except:
            pass

    differential_evolution_step()

    while time.time() - start_time < budget_s - 100:
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x

    return best_h, best_c5, n_points
