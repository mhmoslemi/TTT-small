                pattern[i] = 0.0
        current_sum = np.sum(pattern)
        if current_sum < target_sum:
            idx = np.argmin(pattern)
            pattern[idx] += 0.5
        elif current_sum > target_sum:
            idx = np.argmax(pattern)
            pattern[idx] -= 0.5
        structured_patterns.append(pattern)
    
    # Add random patterns
    random_patterns = []
    for _ in range(10):
        m = int(target_sum)
        indices = np.random.choice(n_points, size=m, replace=False)
        rand = np.zeros(n_points)
        rand[indices] = 1.0
        current_sum = np.sum(rand)
        if current_sum < target_sum:
            idx = np.argmin(rand)
            rand[idx] += target_sum - current_sum
        elif current_sum > target_sum:
            idx = np.argmax(rand)
            rand[idx] -= current_sum - target_sum
        random_patterns.append(rand)
    
    initial_guesses = [initial_guess] + structured_patterns + random_patterns
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=150,
            mutation=(0.5, 0.9),
            recombination=0.85,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 20:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
