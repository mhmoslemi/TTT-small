import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    # Generate uniform pattern
    def generate_uniform_pattern(n_points, m):

        return pattern
    
    # Generate structured patterns
    structured_patterns = []

                pattern[i] = 1.0
            elif i % period == 1:
                pattern[i] = 0.5

                pattern[i] = 0.0
        pattern[-1] = 0.5  # Ensure sum is correct
        structured_patterns.append(pattern)
    
    # Generate random structured patterns
    m = int(n_points // 2)
    random_patterns = []

    initial_guesses = [initial_guess] + structured_patterns + random_patterns

    try:
        result_de = optimize.differential_evolution(
            fun=objective,

            tol=1e-10,
            vectorized=True,

        )
        if result_de.success:
            result_slsqp = optimize.minimize(

            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 20:
        try:
            result = optimize.minimize(
