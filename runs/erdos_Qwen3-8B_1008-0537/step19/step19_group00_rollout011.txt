import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_pattern(m):
        indices = np.random.choice(n_points, size=m, replace=False)
        pattern = np.zeros(n_points)
        pattern[indices] = 1.0
        # Adjust for sum
        if np.sum(pattern) != m:
            idx = np.random.choice(n_points, size=1, replace=False)
            pattern[idx] += 0.5
        return pattern

    initial_guesses = [initial_guess]

        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
