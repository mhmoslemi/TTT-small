        if i % 2 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0
    sum_val = np.sum(structured_h) * dx
    adjustment = (target_sum - sum_val) / dx
    structured_h[-1] += adjustment
    
    structured_h_2 = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h_2[i] = 1.0
        elif i % 3 == 1:
            structured_h_2[i] = 0.5
        else:
            structured_h_2[i] = 0.0
    structured_h_2[-1] = 0.5
    
    structured_h_3 = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            structured_h_3[i] = 1.0
        elif i % 4 == 1:
            structured_h_3[i] = 0.5
        else:
            structured_h_3[i] = 0.0
    structured_h_3[-1] = 0.5
    
    initial_guesses = [initial_guess, structured_h, structured_h_2, structured_h_3]
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=200,
            mutation=(0.5, 0.8),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
