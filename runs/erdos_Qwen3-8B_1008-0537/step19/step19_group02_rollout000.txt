import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess, method='SLSQP'):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_perturbed_guess(base, sigma=0.1):
        return np.clip(base + sigma * np.random.randn(*base.shape), 0, 1)
    
    def generate_structured_guess(modes=3, base=0.5):
        n = n_points
        structured = np.zeros(n)
        for i in range(n):
            for m in range(modes):
                if i % (n // modes) == m:
                    structured[i] = base
        return structured
    
    initial_guesses = [h_initial]
    for _ in range(3):
        initial_guesses.append(generate_perturbed_guess(h_initial))
        initial_guesses.append(generate_structured_guess(modes=5, base=0.5))
    
    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None and objective(res) < best_c5:
            best_c5 = objective(res)
            best_h = res
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=300,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            res = optimize_guess(result_de.x, method='SLSQP')
            if res is not None and objective(res) < best_c5:

    except:
