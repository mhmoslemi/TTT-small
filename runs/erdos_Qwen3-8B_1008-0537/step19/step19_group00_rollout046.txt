                pattern[i] = 0.5

                pattern[i] = 0.0
        pattern[-1] = 0.5
        return pattern
    
    structured_patterns = [
        generate_periodic_pattern(n_points, p) for p in [2, 3, 4, 5, 6, 7, 8]
    ]
    
    # Generate random structured patterns
    def generate_random_pattern(n_points, m):
        indices = np.random.choice(n_points, size=m, replace=False)
        rand = np.zeros(n_points)
        rand[indices] = 1.0
        if np.sum(rand) != m:
            idx = np.random.choice(n_points, size=1, replace=False)
            rand[idx] += 0.5
        return rand
    
    random_patterns = [generate_random_pattern(n_points, 22) for _ in range(5)]
    
    initial_guesses = [initial_guess] + structured_patterns + random_patterns
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_h = result
            best_c5 = objective(result)
    
    # Differential evolution
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=120,
            maxiter=200,
            mutation=(0.6, 0.9),
            recombination=0.85,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_h = result_slsqp.x
                best_c5 = objective(result_slsqp.x)
    except:
        pass
    
    # Local refinement
    while time.time() - start_time < budget_s - 40:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_h = result.x
                best_c5 = objective(result.x)
        except:
            break
    
    return best_h, best_c5, n_points
