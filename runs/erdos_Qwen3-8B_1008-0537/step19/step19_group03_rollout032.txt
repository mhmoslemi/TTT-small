import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_symmetric_guess(num_ones=22):
        indices = []
        for i in range(n_points//2):
            if np.random.rand() < 0.6:
                indices.append(i)
                indices.append(n_points-1-i)
        while len(indices) < num_ones:
            i = np.random.randint(n_points//2)
            if i not in indices and n_points-1-i not in indices:
                indices.append(i)
                indices.append(n_points-1-i)
        indices = np.random.choice(indices, size=num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess, generate_symmetric_guess()]
    for _ in range(3):
        initial_guesses.append(generate_symmetric_guess())
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:

            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(

            vectorized=True,

        )
        if result_de.success:
            result_slsqp = optimize.minimize(
