                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h[i] = 1.0
        elif i % 3 == 1:
            structured_h[i] = 0.5
        else:
            structured_h[i] = 0.0
    structured_h[-1] = 0.5
    initial_guesses = [initial_guess, structured_h]
    
    # Two-interval guess
    interval_starts = [0.0, 1.0]
    interval_len = 0.5
    two_interval = np.zeros(n_points)
    for i in range(len(interval_starts)):
        start = interval_starts[i]
        end = start + interval_len
        for j in range(n_points):
            x = j * dx
            if start <= x < end:
                two_interval[j] = 1.0
    initial_guesses.append(two_interval)
    
    # Three-interval guess
    interval_starts = [0.0, 0.8333333333, 1.3333333333]
    interval_len = 0.3333333333
    three_interval = np.zeros(n_points)
    for i in range(len(interval_starts)):
        start = interval_starts[i]
        end = start + interval_len
        for j in range(n_points):
            x = j * dx
            if start <= x < end:
                three_interval[j] = 1.0
    initial_guesses.append(three_interval)
    
    # Binary pattern guess
    binary_guess = np.zeros(n_points)
    count_ones = 0
    for i in range(n_points):
        if count_ones < n_points // 2:
            binary_guess[i] = 1.0
            count_ones += 1
        else:
            binary_guess[i] = 0.0
    if n_points % 2 == 1:
        idx = n_points // 2
        binary_guess[idx] = 0.5
    np.random.shuffle(binary_guess)
    initial_guesses.append(binary_guess)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=300,
            mutation=(0.5, 0.9),
            recombination=0.7,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
