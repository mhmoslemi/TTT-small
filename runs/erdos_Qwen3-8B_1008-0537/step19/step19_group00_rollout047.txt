        current_sum = np.sum(h)
        target_sum = n_points / 2
        excess = current_sum - target_sum
        if abs(excess) > 1e-6:
            idx = np.argmin(np.abs(x - positions[0]))
            h[idx] += excess
        return h
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    structured_patterns = []
    m_even = int(n_points / 2)
    m_even_plus = m_even + 1
    # Generate even spacing patterns
    for m in [m_even, m_even_plus]:
        pattern = generate_even_spacing(n_points, m)
        structured_patterns.append(pattern)
    
    # Add periodic patterns
    for period in [3, 4, 5, 6]:
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                pattern[i] = 1.0
            elif i % period == 1:
                pattern[i] = 0.5
            else:
                pattern[i] = 0.0
        pattern[-1] = 0.5
        structured_patterns.append(pattern)
    
    # Add random structured patterns
    for _ in range(3):
        indices = np.random.choice(n_points, size=m_even, replace=False)
        rand = np.zeros(n_points)
        rand[indices] = 1.0
        if np.sum(rand) != m_even:
            idx = np.random.choice(n_points, size=1, replace=False)
            rand[idx] += 0.5
        structured_patterns.append(rand)
    
    initial_guesses = [initial_guess] + structured_patterns
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=120,
            maxiter=200,
            mutation=(0.6, 0.9),
            recombination=0.85,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 20:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
