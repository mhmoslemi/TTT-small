import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_periodic_pattern(period):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                pattern[i] = 1.0
            elif i % period == 1:
                pattern[i] = 0.5
            else:
                pattern[i] = 0.0
        pattern[-1] = 0.5
        return pattern
    
    def generate_greedy_pattern():
        positions = np.linspace(0, 2, n_points)
        ones = np.zeros(n_points)
        used = set()
        for i in range(int(n_points / 2)):
            pos = np.random.rand() * 2
            while pos in used:
                pos += np.random.rand() * 0.5
            pos = np.clip(pos, 0, 2)
            idx = np.argmin(np.abs(positions - pos))
            ones[idx] = 1.0
            used.add(pos)
        return ones
    
    structured_patterns = [
        generate_periodic_pattern(3),
        generate_periodic_pattern(4),
        generate_periodic_pattern(5),
        generate_periodic_pattern(6),
        generate_periodic_pattern(7),

    ]
    
    random_patterns = [generate_greedy_pattern() for _ in range(3)]
    
    initial_guesses = [initial_guess] + structured_patterns + random_patterns

            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=200,

            tol=1e-10,
            vectorized=True,

        )
        if result_de.success:
            result_slsqp = optimize.minimize(

                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}

            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)

    except:
