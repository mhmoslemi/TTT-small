        pattern = np.zeros(n_points)
        m = 4
        for i in range(n_points):
            segment = i // (n_points // m)
            if segment % 2 == 0:
                pattern[i] = 1.0
            else:
                pattern[i] = 0.0
        current_sum = np.sum(pattern)
        diff = target_sum - current_sum
        if diff > 0:
            mask = pattern == 0
            idx = np.where(mask)[0]
            if idx.size > 0:
                idx = idx[:int(diff)]
                pattern[idx] = 0.5
        else:
            mask = pattern == 1
            idx = np.where(mask)[0]
            if idx.size > 0:
                idx = idx[:int(-diff)]
                pattern[idx] = 0.5
        return pattern
    
    structured_h = create_structured_pattern(n_points)
    initial_guesses = [initial_guess, structured_h]
    
    for _ in range(4):
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        rand_h = np.zeros(n_points)
        rand_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            rand_h[idx] += remaining
        initial_guesses.append(rand_h)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=200,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
