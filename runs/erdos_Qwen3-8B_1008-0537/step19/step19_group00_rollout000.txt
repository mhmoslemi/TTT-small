import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    # Generate symmetric initial guesses
    def generate_symmetric_pattern(n_points, m):
        indices = np.linspace(0, n_points // 2, m, endpoint=False).astype(int)
        pattern = np.zeros(n_points)
        pattern[indices] = 1.0
        pattern[n_points - 1 - indices] = 1.0
        return pattern
    
    structured_patterns = []
    m = int(n_points // 2)
    sym_pattern = generate_symmetric_pattern(n_points, m)
    structured_patterns.append(sym_pattern)
    
    for period in [3, 4, 5]:
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                pattern[i] = 1.0
            elif i % period == 1:
                pattern[i] = 0.5
            else:
                pattern[i] = 0.0
        pattern[-1] = 0.5
        structured_patterns.append(pattern)
    
    # Generate random symmetric patterns
    random_patterns = []
    for _ in range(3):
        indices = np.random.choice(n_points // 2, size=m, replace=False)
        rand = np.zeros(n_points)
        rand[indices] = 1.0
        rand[n_points - 1 - indices] = 1.0
        # Adjust for sum
        if np.sum(rand) != 2 * m:
            idx = np.random.choice(n_points // 2, size=1, replace=False)
            rand[idx] += 0.5
            rand[n_points - 1 - idx] += 0.5
        random_patterns.append(rand)
    
    initial_guesses = [initial_guess] + structured_patterns + random_patterns
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=120,
            maxiter=200,
            mutation=(0.65, 0.95),
            recombination=0.85,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)

        pass
