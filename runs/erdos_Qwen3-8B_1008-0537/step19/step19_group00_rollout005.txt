import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    structured_patterns = []
    m = int(n_points // 1.5)
    
    def make_periodic(period, shift):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            p = (i + shift) % period
            if p < m:
                pattern[i] = 1.0
            elif p < 2*m:
                pattern[i] = 0.5
        return pattern
    
    for period in [3, 4, 5]:
        for shift in range(3):
            pattern = make_periodic(period, shift)
            pattern[-1] = 0.5
            structured_patterns.append(pattern)
    
    initial_guesses = [initial_guess] + structured_patterns
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=120,
            maxiter=150,

            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            res = optimize.minimize(
                fun=objective,

                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)

    except:
