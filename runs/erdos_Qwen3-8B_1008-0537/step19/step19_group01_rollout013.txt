import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h[i] = 1.0
        elif i % 3 == 1:
            structured_h[i] = 0.5
        else:
            structured_h[i] = 0.0
    structured_h[-1] = 0.5
    initial_guesses = [initial_guess, structured_h]
    
    interval_starts = [0.0, 1.0]
    interval_len = 0.5
    two_interval = np.zeros(n_points)
    for i in range(len(interval_starts)):
        start = interval_starts[i]
        end = start + interval_len
        for j in range(n_points):
            x = j * dx
            if start <= x < end:
                two_interval[j] = 1.0
    initial_guesses.append(two_interval)
    
    interval_starts = [0.0, 0.8333333333, 1.3333333333]
    interval_len = 0.3333333333

    for i in range(len(interval_starts)):

        end = start + interval_len
        for j in range(n_points):
            x = j * dx
            if start <= x < end:
                three_interval[j] = 1.0
    initial_guesses.append(three_interval)
    
    interval_len = 0.25
    interval_starts = [0.0, 0.5, 1.0, 1.5]
    four_interval = np.zeros(n_points)
