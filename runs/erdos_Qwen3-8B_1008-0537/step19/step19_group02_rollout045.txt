    structured[-1] = 0.5
    structured2 = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured2[i] = 1.0
        else:
            structured2[i] = 0.0
    # Adjust for sum
    current_sum = np.sum(structured2)
    if current_sum > target_sum:
        idx = np.random.choice(n_points, replace=False)
        structured2[idx] = 0.5
    elif current_sum < target_sum:
        idx = np.random.choice(n_points, replace=False)
        structured2[idx] = 1.5
    initial_guesses = [initial_guess, structured, structured2]
    
    # Add random guesses
    for _ in range(3):
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        rand_h = np.zeros(n_points)
        rand_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            rand_h[idx] += remaining
        initial_guesses.append(rand_h)
    
    # Evaluate initial guesses
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential Evolution with tuned hyperparameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=300,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Final refinement with L-BFGS-B
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
