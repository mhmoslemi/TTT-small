import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guesses = []

    # Generate structured initial guesses for m=2, 3, 4
    for m in [2, 3, 4]:
        L = 1.0 / m
        D = 1.0 / (m - 1)
        current_start = 0.0
        h = np.zeros(n_points)
        for _ in range(m):
            start = current_start
            end = start + L
            for j in range(n_points):
                if start <= j * dx < end:
                    h[j] = 1.0
            current_start = end + D
        initial_guesses.append(h)

    # Include initial_h_values if available
    if hasattr(initial_h_values, '__len__'):
        initial_guesses.append(np.array(initial_h_values, dtype=float))

    # Add a random initial guess
    random_h = np.random.rand(n_points)
    random_h /= np.sum(random_h)
    initial_guesses.append(random_h)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)

    start_time = time.time()

    # Run differential evolution with increased parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=20,
            maxiter=150,
            mutation=(0.6, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass

    # Local optimization around best_h
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break

    return best_h, best_c5, n_points
