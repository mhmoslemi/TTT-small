import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_random_guess(n):
        m = int(target_sum)
        r = target_sum - m
        pos = np.zeros(n)
        idxs = np.random.choice(n, size=m, replace=False)
        pos[idxs] = 1.0
        if r > 0:
            pos[np.random.choice(n)] += r
        return pos
    
    def generate_structured_guess(n):
        m = n // 2
        pos = np.zeros(n)

            idx = int(i * (n - 1) / (m - 1))

        if abs(delta) > 1e-5:
            pos[-1] += delta
        return pos
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
