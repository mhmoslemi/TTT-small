                h[remove] -= 0.5
        return h

    initial_guess1 = np.array(initial_h_values, dtype=float)
    initial_guess2 = generate_evenly_spaced(n_points)
    initial_guesses = [initial_guess1, initial_guess2]

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess1.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e3 * constraint_violation
        return c5 + penalty

    for guess in initial_guesses:
        try:
            result_de = optimize.differential_evolution(
                fun=objective_de,
                bounds=bounds,
                popsize=50,
                maxiter=300,
                mutation=(0.6, 0.9),
                recombination=0.8,
                tol=1e-10,
                vectorized=True
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass

    for _ in range(2):
        random_guess = generate_evenly_spaced(n_points)
        result_slsqp_random = optimize.minimize(
            fun=objective,
            x0=random_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp_random.success:
            current_c5 = objective(result_slsqp_random.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp_random.x

    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.003, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x

    return best_h, best_c5, n_points
