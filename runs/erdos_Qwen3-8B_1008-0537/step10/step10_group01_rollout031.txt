        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_initial.success:
        current_c5 = objective(result_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_initial.x
    
    # Run SLSQP on random guess
    random_guess = generate_random_guess(n_points)
    result_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_random.success:
        current_c5 = objective(result_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_random.x
    
    # Perturb initial guess and run SLSQP
    for _ in range(4):
        perturbed = best_h + np.random.normal(0, 0.005, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result_perturbed = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_perturbed.success:
            current_c5 = objective(result_perturbed.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_perturbed.x
    
    # Ensure we don't exceed the time budget
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result_perturbed = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_perturbed.success:
            current_c5 = objective(result_perturbed.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_perturbed.x
    
    return best_h, best_c5, n_points
