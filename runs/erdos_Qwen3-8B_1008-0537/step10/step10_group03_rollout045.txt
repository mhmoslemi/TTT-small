import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    # Generate a sine-based initial guess that satisfies the integral constraint

    target_sum = n_points / 2

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e5 * constraint_violation  # Stronger penalty
        return c5 + penalty

            fun=objective_de,
            bounds=bounds,
            popsize=50,
            maxiter=250,

            tol=1e-10,
            vectorized=True,
            seed=seed

        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
