import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        # Basin-hopping with SLSQP local optimization
        result_bh = optimize.basinhopping(
            fun=objective,
            x0=initial_guess,
            T=1.0,
            niter=20,
            minimizer_kwargs={
                'method': 'SLSQP',
                'bounds': bounds,
                'constraints': [{'type': 'eq', 'fun': constraint_sum}],
                'options': {'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            },
            take_step=None,
            callback=None
        )
        if result_bh.success:
            current_c5 = result_bh.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_bh.x
    except:
        pass
    
    # Local refinement with SLSQP
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_c5 = result_slsqp.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
    
    # Perturbation loop
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = new_guess.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
