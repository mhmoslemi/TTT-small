import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def generate_even_spaced(n):
        m = n // 2
        h = np.zeros(n)
        h[::2] = 1.0
        if n % 2:
            idx = np.random.randint(n)
            h[idx] += 0.5
        return h

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e-2 * constraint_violation
        return c5 + penalty

    initial_guesses = [np.array(initial_h_values, dtype=float), generate_even_spaced(n_points)]
    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)

    start_time = time.time()

    for guess in initial_guesses:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            popsize=50,
            maxiter=200,
            mutation=(0.3, 0.7),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x

    random_guess = generate_even_spaced(n_points)
    result_slsqp_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_random.success and result_slsqp_random.fun < best_c5:
        best_c5 = result_slsqp_random.fun
        best_h = result_slsqp_random.x

    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.003, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success and new_guess.fun < best_c5:
            best_c5 = new_guess.fun
            best_h = new_guess.x

    return best_h, best_c5, n_points
