        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess

    # Initial guess
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):

        corr = np.correlate(h, 1 - h, mode='full')

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    # Bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Start with initial guess
    result_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,

    perturbed_guess = initial_guess + np.random.normal(0, 0.02, n_points)
