        positions = np.linspace(0, 2, n)
        indices = np.linspace(0, n-1, ones_needed+1, dtype=int)
        indices = np.sort(indices[1:-1])
        h = np.zeros(n)
        h[indices] = 1.0
        if fractional_value > 0:
            h[indices[-1]] += fractional_value
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e4 * constraint_violation  # Adjusted penalty
        return c5 + penalty
    
    # Generate initial guesses
    guesses = [initial_guess.copy(), generate_even_guess(n_points), generate_random_guess(n_points)]
    
    for guess in guesses:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=30,
            maxiter=200,
            mutation=(0.5, 0.8),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    
    # Local refinement
    for _ in range(3):
        if time.time() - start_time >= budget_s - 50:
            break
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points

def generate_random_guess(n):
    target_sum = n / 2
    num_integers = int(target_sum)
    remaining = target_sum - num_integers
    positions = np.random.choice(n, size=num_integers, replace=False)
    guess = np.zeros(n)
    guess[positions] = 1.0
    if remaining > 0:
        pos = np.random.choice(n, size=1, replace=False)
        guess[pos] += remaining
    return guess
