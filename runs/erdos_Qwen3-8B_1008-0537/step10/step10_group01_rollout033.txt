                    break
        return h

    initial_guesses = [initial_guess, generate_random_guess(n_points), generate_spaced_guess(n_points)]
    best_h = initial_guess.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in initial_guesses:
        try:
            result_de = optimize.differential_evolution(
                fun=lambda x: objective(x) + 1e4 * abs(np.sum(x) - target_sum),
                bounds=bounds,
                popsize=50,
                maxiter=300,
                mutation=(0.8, 0.9),
                recombination=0.8,
                tol=1e-10,
                vectorized=True
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = result_slsqp.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass

    def perturb_and_optimize(h_start):
        for _ in range(3):
            perturbed = h_start + np.random.normal(0, 0.003, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    h_start = result.x
        return h_start

    best_h = perturb_and_optimize(best_h)
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.003, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
