import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def generate_spaced_guess(n):
        m = n // 2
        spacing = n // m if m > 0 else 1
        guess = np.zeros(n)
        for i in range(m):
            idx = i * spacing
            if idx < n:
                guess[idx] = 1.0
        required_sum = n / 2
        current_sum = np.sum(guess)
        delta = required_sum - current_sum
        if delta > 0:
            idx = np.where(guess == 0)[0][0]
            guess[idx] += delta
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e-3 * constraint_violation  # Reduced penalty
        return c5 + penalty
    
    initial_guesses = [initial_guess, generate_random_guess(n_points), generate_spaced_guess(n_points)]
    results = []
    
    for guess in initial_guesses:

                mutation=(0.3, 0.7),
                recombination=0.8,
                tol=1e-10,
                vectorized=True
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}

    for res in results:
        result_slsqp = optimize.minimize(

            x0=res,

            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}

        if result_slsqp.success:

            if current_c5 < best_c5:

                best_h = result_slsqp.x
    
    while time.time() - start_time < budget_s - 50:
