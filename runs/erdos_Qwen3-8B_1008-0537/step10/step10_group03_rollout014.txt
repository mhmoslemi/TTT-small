import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        h = np.array(h_values)

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

        guesses.append(guess1)

        guess2[:n_points//2] = 1.0
        guesses.append(guess2)
        guess3 = np.zeros_like(initial_guess)

        guesses.append(guess3)
        return guesses
    
    def objective_de(h_values):

        c5 = np.max(corr) * dx

        )

            result_slsqp = optimize.minimize(
