import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_structured_guess(n):
        target_sum = n / 2
        grid = np.linspace(0, 2, n)
        h = np.zeros(n)
        # Create a structured pattern with alternating intervals
        num_intervals = 5
        interval_len = 2 / num_intervals
        for i in range(num_intervals):
            start_idx = int(grid.searchsorted(i * interval_len))
            end_idx = int(grid.searchsorted((i + 1) * interval_len))
            if i % 2 == 0:
                h[start_idx:end_idx] = 1.0
            else:
                h[start_idx:end_idx] = 0.0
        # Adjust to meet sum constraint
        h = h / np.sum(h) * target_sum
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=30,
            maxiter=200,

            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
