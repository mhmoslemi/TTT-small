    def objective(h_values):

        for m in range(-n_points + 1, n_points):
            integral = 0.0
            for i in range(n_points):
                j = i + m
                if 0 <= j < n_points:
                    integral += h[i] * (1 - h[j])
            integral *= dx
            if integral > max_c5:
                max_c5 = integral
        return max_c5
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_padded_guess(h):

        return h_padded
    
    def generate_log_spaced_guess(h):
