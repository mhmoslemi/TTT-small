            best_c5 = c5
            best_h = result_slsqp_initial.x
    
    random_guess = generate_random_guess(n_points)
    result_slsqp_random = optimize.minimize(
        fun=objective_slsqp,
        x0=random_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-9}
    )
    if result_slsqp_random.success:
        c5 = objective_slsqp(result_slsqp_random.x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = result_slsqp_random.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.003, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective_slsqp,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if new_guess.success:
            c5 = objective_slsqp(new_guess.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
