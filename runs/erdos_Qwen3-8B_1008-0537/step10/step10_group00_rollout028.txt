    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Generate symmetric initial guess
    symmetric_initial = (initial_h_values + np.flipud(initial_h_values)) / 2

    # Generate random initial guess
    def generate_random_guess(n):

        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess

    initial_guesses = [initial_h_values, symmetric_initial, generate_random_guess(n_points)]

    best_h = initial_guesses[0]
