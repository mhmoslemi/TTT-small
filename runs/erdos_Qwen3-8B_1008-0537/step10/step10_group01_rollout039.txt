import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_smooth_guess(n):
        target_sum = n / 2
        # Start with a uniform distribution of 1's
        h = np.zeros(n)
        h[:int(target_sum)] = 1.0
        # Spread out the 1's to minimize overlap
        h = np.sort(h)
        # Introduce gradual transitions
        h = np.concatenate((h, np.zeros(n - len(h))))
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e5 * constraint_violation  # Increased penalty
        return c5 + penalty
    
    try:
        # Run DE with multiple initial guesses
        guesses = [initial_guess.copy()]
        guesses.append(generate_smooth_guess(n_points))
        guesses.append(np.random.rand(n_points))
        
        for guess in guesses:
            result_de = optimize.differential_evolution(
                fun=objective_de,
                bounds=bounds,
                popsize=30,
                maxiter=300,
                mutation=(0.7, 0.9),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                x0=guess
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
