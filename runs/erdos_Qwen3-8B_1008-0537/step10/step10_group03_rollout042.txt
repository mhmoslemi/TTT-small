def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Generate structured initial guess
    idx_ones = np.random.choice(n_points, 22, replace=False)
    idx_zero = np.random.choice(n_points, 1, replace=False)
    idx_half = np.random.choice(n_points, 1, replace=False)

    initial_guess = np.zeros(n_points)
    initial_guess[idx_ones] = 1.0
    initial_guess[idx_zero] = 0.0
    initial_guess[idx_half] = 0.5

    # Function to calculate the objective
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint function
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    start_time = time.time()

    # Run SLSQP on the initial guess
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result.success:
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    # Generate perturbed versions
    for _ in range(10):
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result_perturbed = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_perturbed.success:
            current_c5 = objective(result_perturbed.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_perturbed.x

    # Stop if time is up
    if time.time() - start_time > budget_s - 50:
