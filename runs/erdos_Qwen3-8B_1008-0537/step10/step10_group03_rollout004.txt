import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_perturbations(h, noise_scale=0.05, num_perts=5):
        perts = []
        for _ in range(num_perts):
            p = noise_scale * np.random.randn(len(h))
            hp = np.clip(h + p, 0, 1)
            hp = hp * (target_sum / np.sum(hp))
            perts.append(hp)
        return perts

    def generate_log_guess(h, steps=3):
        idx = np.argsort(np.log(np.abs(h)))
        h_sorted = h[idx].copy()
        h_sorted[::steps] = 1.0
        h_sorted[1::steps] = 0.5
        h_sorted[2::steps] = 0.0
        return h_sorted

    def objective_de(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e3 * abs(np.sum(h) - target_sum)
        return c5 + penalty

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial perturbations
    perts = generate_perturbations(initial_h, noise_scale=0.02, num_perts=3)
    all_guesses = [initial_h] + perts + [generate_log_guess(initial_h, steps=3)]

    for guess in all_guesses:
        try:
            result_de = optimize.differential_evolution(

                bounds=bounds,

                recombination=0.9,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}

                    if current_c5 < best_c5:

        except:

    # SLSQP on best so far
    result_slsqp = optimize.minimize(

    if result_slsqp.success:

            best_h = result_slsqp.x

    # Final perturbation refinement
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
