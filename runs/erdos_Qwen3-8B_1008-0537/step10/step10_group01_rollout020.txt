        guess = np.zeros(n)
        for i in range(n_ints):
            start = intervals[i]
            end = intervals[i + 1] if i + 1 < n_ints else 2
            start_idx = int(start / dx)
            end_idx = int(end / dx)
            guess[start_idx:end_idx] = 1.0
        if remaining > 0:
            last = intervals[-1] if n_ints > 0 else 0
            start = last + 0.5 * remaining
            start_idx = int(start / dx)
            guess[start_idx] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
