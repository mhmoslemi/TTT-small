        num_integers = int(target_sum)

        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def structured_guess(n):
        guess = np.zeros(n)
        for i in range(n):
            if i % 2 == 0:
                guess[i] = 1.0
        sum_val = np.sum(guess)
        if sum_val > target_sum:
            idx = np.where(guess == 1)[0][0]
            guess[idx] -= (sum_val - target_sum)

            idx = np.where(guess == 0)[0][0]
            guess[idx] += (target_sum - sum_val)

        return np.max(corr) * dx
