import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_regular_guess(n):
        positions = np.linspace(0, 2, n, endpoint=False)
        m = int(target_sum)
        remaining = target_sum - m
        indices = np.argsort(np.abs(positions - 1.0))
        guess = np.zeros(n)
        guess[indices[:m]] = 1.0
        if remaining > 0:
            guess[indices[m]] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e2 * constraint_violation
        return c5 + penalty
    
    def optimize_guess(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else x0
    
    # Initial guesses
    guesses = [initial_guess, generate_regular_guess(n_points)]
    
    for guess in guesses:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=15,
            maxiter=150,
            mutation=(0.5, 0.9),
            recombination=0.9,
            tol=1e-9,
            vectorized=True,
            polish=True
        )
        if result_de.success:
            refined = optimize_guess(result_de.x)
            current_c5 = objective(refined)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = refined
    
    random_guess = np.random.uniform(0, 1, size=n_points)

    result = optimize_guess(random_guess)
