import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_padded_guess(h):
        h_padded = np.zeros_like(h)
        h_padded[::2] = h[::2]
        h_padded[1::2] = h[1::2]
        return h_padded
    
    def generate_symmetric_guess(h):
        h_sym = h.copy()
        h_sym = (h_sym + h_sym[::-1]) / 2
        return h_sym
    
    def optimize_with_slsqp(x0, maxiter=300):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result
    
    def evaluate_guess(x, name):
        nonlocal best_c5, best_h
        if not x.success:
            return
        current_c5 = objective(x.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = x.x
        print(f"{name} C5: {best_c5:.6f}")
    
    try:
        # Initial SLSQP turbo
        result = optimize_with_slsqp(initial_guess, maxiter=300)
        evaluate_guess(result, "Initial")
    except:
        pass
    
    try:
        # DE with enhanced parameters
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=250,
            mutation=(0.7, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        result_slsqp = optimize_with_slsqp(result_de.x, maxiter=250)
        evaluate_guess(result_slsqp, "DE+SLSP")
    except:
        pass
    
    try:
        # Symmetric guess
        sym_guess = generate_symmetric_guess(initial_guess)
        result = optimize_with_slsqp(sym_guess, maxiter=200)
        evaluate_guess(result, "Symmetric")
    except:
        pass
    
    try:
        # Padded guess
        padded = generate_padded_guess(initial_guess)
        result = optimize_with_slsqp(padded, maxiter=200)
        evaluate_guess(result, "Padded")
    except:
        pass
    
    # Final iterative refinement
    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        perturbed = best_h + np.random.normal(0, 0.005, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize_with_slsqp(perturbed, maxiter=150)
        evaluate_guess(result, "Iterative")
    
    return best_h, best_c5, n_points
