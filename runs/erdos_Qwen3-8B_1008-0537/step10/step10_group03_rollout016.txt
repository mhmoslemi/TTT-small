        for _ in range(2):
            perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)
            h_opt, c5_opt = optimize_guess(perturbed)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
    
    return best_h, best_c5, n_points
