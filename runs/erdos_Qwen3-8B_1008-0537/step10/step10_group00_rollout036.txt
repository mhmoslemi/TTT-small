import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    def generate_uniform_guess(n):
        m = int(target_sum)
        remaining = target_sum - m
        indices = np.linspace(0, n - 1, m + 1, endpoint=False).astype(int)
        guess = np.zeros(n)
        guess[indices] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=40,
        maxiter=150,
        mutation=(0.7, 1.0),
        recombination=0.9,
        tol=1e-10,
        vectorized=True
    )
    
    if result_de.success:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
