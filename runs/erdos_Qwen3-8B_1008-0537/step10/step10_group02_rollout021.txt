import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def generate_regular_guess(n):

        remaining = target_sum - num_ones
        positions = np.linspace(0, n-1, num_ones, endpoint=False).astype(int)
        h = np.zeros(n)
        h[positions] = 1.0
        if remaining > 0:
            mid_index = n // 2
            h[mid_index] += remaining
        return h

    def generate_random_guess(n):
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
