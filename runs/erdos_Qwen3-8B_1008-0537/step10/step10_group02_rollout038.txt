import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initial guess from provided data
    initial_guess = np.array(initial_h_values, dtype=float)
    
    # Structured initial guess - even spacing with 22.5 units of 1s
    structured_guess = np.zeros(n_points)
    structured_guess[:22] = 1.0
    structured_guess[22] = 0.5
    structured_guess = np.roll(structured_guess, 2)  # Spread out the pattern

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = structured_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e0 * constraint_violation  # Reduced penalty
        return c5 + penalty

    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=50,
            maxiter=300,
            mutation=(0.5, 0.9),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass

    result_slsqp_initial = optimize.minimize(

        x0=initial_guess,
        method='SLSQP',

    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x

    result_slsqp_structured = optimize.minimize(

    while time.time() - start_time < budget_s - 50:
