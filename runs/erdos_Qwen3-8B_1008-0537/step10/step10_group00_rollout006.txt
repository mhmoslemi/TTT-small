import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guesses = [np.array(initial_h_values, dtype=float)]
    
    def generate_structured_guess(n):
        dx = 2.0 / n
        x = np.linspace(0, 2, n, endpoint=False)
        h = np.zeros(n)
        for i in range(n):
            xi = x[i]
            # Two intervals: [0.25,0.75) and [1.25,1.75)
            if (0.25 <= xi < 0.75) or (1.25 <= xi < 1.75):
                h[i] = 1.0
        return h
    
    structured_guess = generate_structured_guess(n_points)
    initial_guesses.append(structured_guess)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_with_guess(x0, maxiter=300):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x, objective(result.x)
        return x0, best_c5
    
    # Initial optimizations
    for guess in initial_guesses:
        h_opt, c5_opt = optimize_with_guess(guess)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Local search with perturbation
    for _ in range(5):
        if time.time() - start_time > budget_s - 50:
            break
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        h_opt, c5_opt = optimize_with_guess(perturbed, maxiter=150)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    return best_h, best_c5, n_points
