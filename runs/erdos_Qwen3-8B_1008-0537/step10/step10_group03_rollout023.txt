import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbed_guess(h):
        noise = np.random.normal(0, 0.001, size=n_points)
        return np.clip(h + noise, 0, 1)
    
    def generate_log_guess(h):
        idx = np.argsort(np.log(1 + h))
        h_sorted = h[idx].copy()
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted
    
    def generate_spaced_guess(h):
        idx = np.argsort(np.sin(np.linspace(0, 2*np.pi, n_points)))
        h_sorted = h[idx].copy()
        h_sorted[:n_points//2] = 1.0
        h_sorted[n_points//2:] = 0.0
        return h_sorted
    
    def optimize_with_slsqp(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result
    
    def optimize_with_de(x0):
        result = optimize.differential_evolution(
            fun=lambda x: objective(x) + 1e5*abs(constraint_sum(x)),
            bounds=bounds,
            popsize=20,
            maxiter=100,
            mutation=(0.5, 0.8),
            recombination=0.9,
            tol=1e-10,
            seed=seed
        )
        return result
    
    def evaluate_candidate(x):
        nonlocal best_c5, best_h
        res = optimize_with_slsqp(x)
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
    
    initial_candidates = [
        initial_guess,
        generate_perturbed_guess(initial_guess),
        generate_log_guess(initial_guess),
        generate_spaced_guess(initial_guess)
    ]
    
    for cand in initial_candidates:
        evaluate_candidate(cand)
    
    while time.time() - start_time < budget_s - 50:
        new_guess = generate_perturbed_guess(best_h)
        evaluate_candidate(new_guess)
    
    return best_h, best_c5, n_points
