import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def perturb(h, scale=0.003):
        return np.clip(h + np.random.normal(0, scale, size=h.shape), 0, 1)
    
    def optimize_with_slsqp(h0):
        result = optimize.minimize(
            fun=objective,
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result
        return None
    
    # Initial optimization
    result = optimize_with_slsqp(h_initial)
    if result is not None and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Random guess
    random_guess = np.zeros(n_points)
    random_guess[np.random.choice(n_points, size=int(target_sum), replace=False)] = 1.0
    if np.sum(random_guess) < target_sum:
        random_guess[np.random.choice(n_points, size=1)] += 1.0 - (target_sum - np.sum(random_guess))
    result = optimize_with_slsqp(random_guess)
    if result is not None and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Multi-start perturbation
    for _ in range(3):
        if time.time() - start_time > budget_s - 50:
            break
        h_perturbed = perturb(best_h)
        result = optimize_with_slsqp(h_perturbed)
        if result is not None and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Local refinement
    for _ in range(2):
        if time.time() - start_time > budget_s - 50:
            break
        h_perturbed = perturb(best_h, scale=0.001)
        result = optimize_with_slsqp(h_perturbed)
        if result is not None and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
