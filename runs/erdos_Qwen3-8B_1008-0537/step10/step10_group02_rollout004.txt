import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e3 * constraint_violation  # Increased penalty
        return c5 + penalty
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=50,
            maxiter=150,

            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
