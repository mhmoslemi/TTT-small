import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_pattern(n):
        m = n // 2
        if n % 2:
            m += 1
        h = np.zeros(n)
        for i in range(m):
            pos = int(i * n / m)
            h[pos] = 1.0
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=50,
            maxiter=200,
            mutation=(0.5, 1.5),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_cobyla = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'disp': False}
            )
            if result_cobyla.success:
                current_c5 = objective(result_cobyla.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
    except:
        pass
    
    result_cobyla_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'disp': False}
    )
    if result_cobyla_initial.success:
        current_c5 = objective(result_cobyla_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_cobyla_initial.x
    
    pattern_guess = generate_pattern(n_points)
    result_cobyla_pattern = optimize.minimize(

        x0=pattern_guess,
