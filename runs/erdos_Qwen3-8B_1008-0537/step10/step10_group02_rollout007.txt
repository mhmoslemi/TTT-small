import scipy.optimize as optimize

    def generate_random_guess(n):
        target_sum = n / 2
        if target_sum.is_integer():
            num_integers = int(target_sum)
            positions = np.random.choice(n, size=num_integers, replace=False)
            guess = np.zeros(n)
            guess[positions] = 1.0
        else:
            num_integers = int(target_sum)
            positions = np.random.choice(n, size=num_integers, replace=False)
            guess = np.zeros(n)
            guess[positions] = 1.0
            remaining = target_sum - num_integers
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

        penalty = 1e-5 * constraint_violation  # Reduced penalty
        return c5 + penalty
    
    try:
        result_de = optimize.differential_evolution(

        if result_de.success:
