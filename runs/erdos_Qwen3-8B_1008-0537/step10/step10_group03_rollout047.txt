import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_padded_guess(h):
        h_padded = np.zeros_like(h)
        h_padded[::2] = h[::2]
        h_padded[1::2] = h[1::2]
        return h_padded
    
    def generate_log_spaced_guess(h):
        h_sorted = np.sort(h)
        h_sorted = np.linalg.norm(h_sorted) * np.log(np.abs(h_sorted[:len(h_sorted)//2]))
        h_sorted = h_sorted / np.linalg.norm(h_sorted)
        h_sorted[:len(h_sorted)//3] = 1.0
        h_sorted[len(h_sorted)//3:2*len(h_sorted)//3] = 0.5
        h_sorted[2*len(h_sorted)//3:] = 0.0
        return h_sorted

    def generate_binary_guess(n, m):
        """Generate a binary guess with m blocks of 1s and 0s."""
        block_size = n // m
        rest = n % m
        h = np.zeros(n)
        for i in range(m):
            if i < rest:
                block_size += 1
            h[i*block_size:(i+1)*block_size] = 1.0
        target = (n / 2) * dx
        total = np.sum(h) * dx
        h = h / (total / target)
        return h

    def generate_wave_guess(n, m):
        """Generate a wave-like guess with m peaks."""
        h = np.zeros(n)
        for i in range(m):
            h[int(n/2 - m + i):int(n/2 + m - i)] = 1.0 / m
        target = (n / 2) * dx
        total = np.sum(h) * dx
        h = h / (total / target)
        return h

    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e3 * constraint_violation  # Reduced penalty
        return c5 + penalty
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=25,
            maxiter=150,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
