import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_structured_guess(m):
        h = np.zeros(n_points)
        for region_id in range(m):
            start = region_id * 2.0 / m
            end = start + 1.0 / m
            i_low = int(np.ceil(start / dx))
            i_high = int(np.floor(end / dx))
            if i_low <= i_high:
                h[i_low:i_high+1] = 1.0
        current_sum = np.sum(h)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        return h * scale
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate structured guesses
    structured_guesses = []
    for m in [2, 3, 4, 5, 6]:
        structured_guesses.append(generate_structured_guess(m))
    
    # Generate random guess
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    random_guess = generate_random_guess(n_points)
    structured_guesses.append(random_guess)
    
    # Run SLSQP on all initial guesses
    for h in [initial_guess] + structured_guesses:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=h,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
        except:
            pass
    
    # Perturb the best solution
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        try:
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
        except:
            pass
    
    return best_h, best_c5, n_points
