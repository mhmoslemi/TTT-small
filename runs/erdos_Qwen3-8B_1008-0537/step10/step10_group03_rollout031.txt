import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_random_initial(n_points):
        h = np.random.uniform(0, 1, n_points)
        scale = (target_sum) / np.sum(h)
        h *= scale
        return h
    
    def generate_alternating_initial(n_points):
        h = np.zeros(n_points)
        alt = True
        for i in range(n_points):
            if alt:
                h[i] = 1.0
            else:
                h[i] = 0.0
            alt = not alt
        scale = target_sum / (np.sum(h) * dx)
        h *= scale
        return h
    
    initial_guesses = [initial_guess, generate_random_initial(n_points), generate_alternating_initial(n_points)]
    
    for guess in initial_guesses:
        try:
            result_de = optimize.differential_evolution(
                fun=lambda h: objective(h) + 1e5 * abs(np.sum(h) - target_sum),
                bounds=bounds,
                popsize=20,
                maxiter=100,
                mutation=(0.5, 1.0),
                recombination=0.9,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass
    
    perturbations = [best_h + np.random.normal(0, 0.001, n_points) for _ in range(3)]
    perturbations = [np.clip(p, 0, 1) for p in perturbations]
    for perturbed in perturbations:
        try:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
    
    if time.time() - start_time < budget_s - 10:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
