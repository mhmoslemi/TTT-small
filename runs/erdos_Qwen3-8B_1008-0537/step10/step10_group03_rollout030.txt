import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = 1.0
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    def generate_even_spaced_guess(h):
        idx = np.argsort(np.abs(h - 0.5))
        h_sorted = h[idx].copy()
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted
    
    def generate_sparse_guess(h):
        h_dup = np.zeros_like(h)
        h_dup[h > 0.7] = 1.0
        h_dup[h < 0.3] = 0.0
        return h_dup
    
    def generate_periodic_guess(h):
        period = n_points // 3
        h_dup = np.zeros_like(h)
        h_dup[::period] = 1.0
        h_dup[period::2*period] = 0.5
        h_dup[2*period::3*period] = 0.0
        return h_dup
    
    def generate_uniform_guess():
        return np.full(n_points, 0.5)
    
    def evaluate_guess(guess):
        nonlocal best_c5
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 700, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    
    initial_guesses = [best_h, generate_even_spaced_guess(best_h), generate_sparse_guess(best_h), generate_periodic_guess(best_h), generate_uniform_guess()]
    
    for guess in initial_guesses:
        evaluate_guess(guess)
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e3 * constraint_violation  # Moderate penalty
        return c5 + penalty
    
    result_de = optimize.differential_evolution(
        fun=objective_de,
        bounds=bounds,
        popsize=30,
        maxiter=400,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-10,
        vectorized=True,
        seed=seed
    )
    
    if result_de.success:
        evaluate_guess(result_de.x)
    
    while time.time() - start_time < budget_s - 20:
        perturbed = best_h + np.random.normal(0, 0.003, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result_perturb = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_perturb.success:
            current_c5 = objective(result_perturb.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_perturb.x.copy()
    
    return best_h, best_c5, n_points
