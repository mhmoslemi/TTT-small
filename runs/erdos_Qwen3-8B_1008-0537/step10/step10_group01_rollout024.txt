            indices = np.where(h == 0.0)[0]
            num_to_increase = int(target_sum - current_sum)
            h[indices[:num_to_increase]] = 1.0
        return h
    
    structured_guess = generate_structured_guess(n_points)
    result_slsqp_structured = optimize.minimize(
        fun=objective,
        x0=structured_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_structured.success:
        current_c5 = objective(result_slsqp_structured.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_structured.x
    
    remaining_time = budget_s - (time.time() - start_time)
    while remaining_time > 50:
        perturbed = best_h + np.random.normal(0, 0.01, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        remaining_time = budget_s - (time.time() - start_time)
    
    return best_h, best_c5, n_points
