    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e5 * constraint_violation  # Increased penalty
        return c5 + penalty
    
    def optimize_from_guess(guess):
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=40,
            maxiter=200,
            mutation=(0.5, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            workers=1
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                return result_slsqp.x
        return result_de.x
    
    candidates = [initial_guess]
    if n_points % 2 == 0:
        candidates.append(generate_uniform_guess(n_points))
    candidates.append(generate_random_guess(n_points))
    
    for guess in candidates:
        try:
            refined = optimize_from_guess(guess)
            current_c5 = objective(refined)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = refined
        except:
            pass
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.003, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
