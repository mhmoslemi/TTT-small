import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_block_initial():
        total_length = 1.0
        block1_len = total_length / 2
        block2_len = total_length / 2
        block1_start = 0.0
        block2_start = block1_start + block1_len
        block2_end = block2_start + block2_len
        h = np.zeros(n_points)
        for i in range(n_points):
            x_i = i * dx
            if block1_start <= x_i < block1_start + block1_len:
                h[i] = 1.0
            elif block2_start <= x_i < block2_end:
                h[i] = 1.0
        return h
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_from_guess(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            return result.x
        return x0
    
    try:
        for _ in range(2):
            guess = optimize_from_guess(best_h)
            if objective(guess) < best_c5:
                best_c5 = objective(guess)
                best_h = guess
        
        for _ in range(2):
            guess = generate_block_initial()
            guess = optimize_from_guess(guess)
            if objective(guess) < best_c5:
                best_c5 = objective(guess)
                best_h = guess
        
        for _ in range(2):
            perturbed = best_h + np.random.normal(0, 0.005, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)

            if objective(guess) < best_c5:
