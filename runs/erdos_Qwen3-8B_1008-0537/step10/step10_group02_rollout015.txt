import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def sinusoidal_guess(n):
        x = np.linspace(0, 2, n)
        a = 1 / (2 + 4/np.pi)
        h = a * (1 + np.sin(np.pi * x / 2))
        h = np.clip(h, 0, 1)
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_from_guess(x0):
        res = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success:
            return res.x, objective(res.x)
        return x0, objective(x0)
    
    # Initial guesses
    guesses = [
        initial_guess,
        generate_random_guess(n_points),
        sinusoidal_guess(n_points)
    ]
    
    for guess in guesses:
        h_perturbed = guess.copy()
        h_perturbed = optimize_from_guess(h_perturbed)[0]
        current_c5 = objective(h_perturbed)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed
    
    # Perturb and refine
    for _ in range(5):
        if time.time() - start_time >= budget_s - 10:
            break
        perturbed = best_h + np.random.normal(0, 0.003, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize_from_guess(perturbed)
        if new_guess[1] < best_c5:
            best_c5 = new_guess[1]
            best_h = new_guess[0]
    
    return best_h, best_c5, n_points
