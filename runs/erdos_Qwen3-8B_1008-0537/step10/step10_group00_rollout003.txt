import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_random_guess():
        idxs = np.random.choice(n_points, size=int(target_sum), replace=False)
        guess = np.zeros(n_points)
        guess[idxs] = 1.0
        if target_sum - int(target_sum) > 1e-6:
            guess[np.random.choice(n_points)] += target_sum - int(target_sum)
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_from_guess(x0):
        result = optimize.minimize(fun=objective, x0=x0, method='SLSQP',
                                 bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                 options={'maxiter': 400, 'ftol': 1e-10, 'eps': 1e-9})
        if result.success:
            return result.x, objective(result.x)
        return x0, best_c5
    
    result_de = optimize.differential_evolution(
        fun=objective, bounds=bounds, popsize=50, maxiter=200,
        mutation=(0.6, 1.0), recombination=0.8, tol=1e-10, vectorized=True)
    if result_de.success:
        best_h, best_c5 = optimize_from_guess(result_de.x)
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 50:
            break
        perturbed = best_h + np.random.normal(0, 0.005, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        best_h, best_c5 = optimize_from_guess(perturbed)
    
    random_guess = generate_random_guess()
    best_h, best_c5 = optimize_from_guess(random_guess)
    
    return best_h, best_c5, n_points
