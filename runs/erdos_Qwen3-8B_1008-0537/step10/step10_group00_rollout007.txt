        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess

        h = np.array(h_values)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
