import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_dense_guess(h):
        h_dense = np.zeros_like(h)
        h_dense[:len(h)//2] = 1.0
        h_dense[len(h)//2:] = 0.0
        return h_dense
    
    def generate_sparse_guess(h):
        idx = np.argsort(np.abs(h - 0.5))
        h_sorted = h[idx].copy()
        h_sorted[::2] = 1.0
        h_sorted[1::2] = 0.0
        return h_sorted
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e5 * constraint_violation
        return c5 + penalty
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=15,

            vectorized=True,

                    best_h = result_slsqp.x
    except:
        pass
