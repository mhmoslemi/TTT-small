        h = np.array(h_values)

        return np.sum(h_values) - target_sum

        guess = base + perturb

        diff = np.sum(guess) - target_sum
        if abs(diff) > 1e-4:
            idx = np.random.choice(np.arange(n_points), size=1)
            guess[idx] += diff
            guess = np.clip(guess, 0, 1)
