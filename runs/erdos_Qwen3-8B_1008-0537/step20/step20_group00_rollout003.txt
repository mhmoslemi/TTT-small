            positions = np.round(positions).astype(int)
            positions = np.unique(positions)
            num_ones = len(positions)
            ones = np.zeros(n_points)
            ones[positions] = 1.0
            if np.sum(ones) < target_sum:
                remaining = target_sum - np.sum(ones)
                candidates = np.where(ones == 0)[0]
                if len(candidates) > 0:
                    idx = np.random.choice(candidates, size=int(remaining), replace=False)
                    ones[idx] = 0.5
            guesses.append(ones)
        return guesses
    
    def evaluate_and_update(best_h, current_h, objective, candidate):
