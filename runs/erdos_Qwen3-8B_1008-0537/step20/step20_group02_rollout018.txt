import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_evenly_spaced_guess():
        num_ones = 22
        spacing = 2.0 / num_ones
        positions = np.linspace(0, 2 - spacing, num_ones)
        indices = np.round(positions / dx).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_random_guess():
        num_ones = 22
        indices = np.argsort(np.random.rand(n_points))
        h = np.zeros(n_points)
        h[indices[:num_ones]] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_clustered_guess():
        num_ones = 22
        cluster_size = 4
        clusters = np.random.choice(n_points, size=num_ones // cluster_size, replace=False)
        h = np.zeros(n_points)
        for center in clusters:
            for i in range(max(0, center - cluster_size), min(n_points, center + cluster_size)):
                h[i] = 1.0
        if np.sum(h) > target_sum:
            idx = np.random.choice(np.where(h == 1)[0], size=1)
            h[idx] = 0.5
        elif np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [
        initial_guess,
        generate_evenly_spaced_guess(),
        generate_random_guess(),
        generate_clustered_guess()
    ]
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None:
            val = objective(result)
            if val < best_c5:
                best_c5 = val
                best_h = result
    
    def differential_evolution_wrapper():
        try:
            result = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=150,
                maxiter=200,
                mutation=(0.5, 1.0),
                recombination=0.9,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    val = objective(result_slsqp.x)
                    if val < best_c5:
                        best_c5 = val
                        best_h = result_slsqp.x
        except:
            pass
    
    differential_evolution_wrapper()

                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                val = objective(result.x)
                if val < best_c5:
                    best_c5 = val
