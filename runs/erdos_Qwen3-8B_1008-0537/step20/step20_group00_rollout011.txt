import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_initial_guesses(num_trials=12):

            indices = np.random.choice(n_points, size=num_ones, replace=False)
            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:

                h[idx] = 0.5

                h = np.random.rand(n_points)
                h = (h < 0.5).astype(float)

                guesses.append(h.copy())
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
