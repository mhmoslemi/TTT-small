import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    def generate_initial_guesses(num_trials=8):
        guesses = [initial_guess.copy()]
        for _ in range(num_trials):
            num_ones = np.random.randint(12, 18)
            indices = np.random.choice(n_points, size=num_ones, replace=False)
            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:
                idx = np.random.choice(np.where(h == 0)[0], size=1)
                h[idx] = 0.5
            guesses.append(h.copy())
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                return result.x
            return None
        except:
            return None
    
    for guess in initial_guesses:

        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)

    start_time = time.time()
    
    def differential_evolution():
        try:
            result = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=40,
                maxiter=150,
                mutation=(0.6, 0.9),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result.success:
                result_slsqp = optimize.minimize(

            return best_c5
        except:
