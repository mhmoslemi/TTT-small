        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_evenly_spaced_guess():
        num_ones = 22
        step = n_points / (num_ones + 1)
        indices = np.round(np.arange(0, n_points, step)).astype(int)
        indices = np.unique(indices)
        h = np.zeros(n_points)
        h[indices[:num_ones]] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_structured_guess():
        pos = np.sort(np.random.choice(n_points, size=22, replace=False))
        h = np.zeros(n_points)
        h[pos] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_binary_guess():
        num_ones = 22
        indices = np.argsort(np.random.rand(n_points))
        h = np.zeros(n_points)
        h[indices[:num_ones]] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess, generate_evenly_spaced_guess(), generate_structured_guess(), generate_binary_guess()]
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=150,
            mutation=(0.5, 0.8),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
