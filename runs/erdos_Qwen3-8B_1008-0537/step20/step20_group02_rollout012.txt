import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_even_guess():
        ones = int(target_sum)
        if target_sum - ones > 0.5:
            ones += 1
        if ones == 0:
            return np.zeros(n_points)
        step = 2.0 / (ones + 1)
        positions = np.arange(0, 2 + 1e-10, step)
        h = np.zeros(n_points)
        spacing = 2.0 / n_points
        for pos in positions:
            idx = int(np.round(pos / spacing))
            if idx < n_points:
                h[idx] = 1.0
        while np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        if np.sum(h) > target_sum:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.5
        return h
    
    def generate_random_guess():
        num_ones = int(target_sum)
        if target_sum - num_ones > 0.5:
            num_ones += 1
        indices = np.argsort(np.random.rand(n_points))
        h = np.zeros(n_points)
        h[indices[:num_ones]] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess, generate_even_guess(), generate_random_guess()]
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
