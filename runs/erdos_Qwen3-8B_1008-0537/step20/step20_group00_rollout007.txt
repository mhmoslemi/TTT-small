    def generate_initial_guesses(num_trials=12):

            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:
                idx = np.random.choice(np.where(h == 0)[0], size=1)
                h[idx] = 0.5
