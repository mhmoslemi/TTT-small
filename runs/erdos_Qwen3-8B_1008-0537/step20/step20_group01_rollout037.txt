        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_perturbed_guess(base, std=0.1):
        perturb = np.random.normal(0, std, n_points)
        guess = np.clip(base + perturb, 0, 1)
        if np.sum(guess) < target_sum - 1e-4:
            idx = np.random.choice(np.where(guess < 0.5)[0], size=1)
            guess[idx] = 0.5
        return guess
    
    initial_guesses = [initial_guess]
    # Generate a structured initial guess: evenly spaced 1s
    m = int(n_points / 2)
    structured_guess = np.zeros(n_points)
    for i in range(m):
        structured_guess[i * 2] = 1.0
    structured_guess = np.clip(structured_guess, 0, 1)
    if np.sum(structured_guess) < target_sum - 1e-4:
        idx = np.random.choice(np.where(structured_guess < 0.5)[0], size=1)
        structured_guess[idx] = 1.0
    initial_guesses.append(structured_guess)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=300,
            mutation=(0.9, 1.1),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success and objective(result_de.x) < best_c5:
            best_c5 = objective(result_de.x)
            best_h = result_de.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 80:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
