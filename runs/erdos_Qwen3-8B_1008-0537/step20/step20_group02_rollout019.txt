import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_structured_guess():
        pos = np.sort(np.random.choice(n_points, size=20, replace=False))
        h = np.zeros(n_points)
        h[pos] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_binary_guess():
        indices = np.argsort(np.random.rand(n_points))
        h = np.zeros(n_points)

            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_even_guess(m):
        positions = np.linspace(0, 2, m + 1)[:-1]

        h = np.zeros(n_points)

                h[i] = 1.0

                fun=objective,

                best_c5 = objective(result_slsqp.x)

    except:
