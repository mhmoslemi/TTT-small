import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_structured_guess(m):
        dx = 2.0 / n_points
        length_intervals = 1.0 / m
        spacing = 1.0 / (m - 1)
        starts = [0.0]
        for i in range(1, m):
            starts.append(starts[-1] + length_intervals + spacing)
        h = np.zeros(n_points)
        for i in range(n_points):
            x_i = i * dx
            for j in range(m):
                interval_start = starts[j]
                interval_end = interval_start + length_intervals
                if interval_start <= x_i < interval_end:
                    h[i] = 1.0
                    break
        return h
    
    initial_guesses = [initial_guess]
    for m in [20, 22, 24]:
        initial_guesses.append(generate_structured_guess(m))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:

            best_c5 = objective(result)
            best_h = result

        result_de = optimize.differential_evolution(

            vectorized=True,
