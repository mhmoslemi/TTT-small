                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                candidate_c5 = objective(result.x)
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
