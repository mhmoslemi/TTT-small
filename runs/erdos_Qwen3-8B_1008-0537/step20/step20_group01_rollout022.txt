import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbed(base, std=0.1):
        perturb = np.random.normal(0, std, n_points)
        guess = np.clip(base + perturb, 0, 1)
        if np.sum(guess) < target_sum - 1e-4:
            idx = np.random.choice(np.where(guess < 0.5)[0], size=1)
            guess[idx] = 0.5
        return guess
    
    def generate_random():
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
        return np.clip(h, 0, 1)
    
    initial_guesses = [initial_guess]
    for _ in range(2):
        initial_guesses.append(generate_perturbed(initial_guess))
    for _ in range(2):
        initial_guesses.append(generate_random())
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=200,
            maxiter=300,
            mutation=(0.9, 1.1),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 80:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
