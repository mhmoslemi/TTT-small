import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def uniform_guess():
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        uniform_guess = np.zeros(n_points)
        for i in range(m):
            uniform_guess[idxs[i]] = 1.0
        if remaining > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1)))
            uniform_guess[idx % n_points] += remaining
        return uniform_guess
    
    def staggered_guess():
        base = np.zeros(n_points)
        m = int(target_sum)
        step = 2.0 / n_points
        for i in range(m):
            pos = i * step + (step / 2)
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos))
            base[idx] = 1.0
        rem = target_sum - m
        if rem > 0:
            pos = (m / (m + 1)) * step

            base[idx % n_points] += rem
        return base

        result_de = optimize.differential_evolution(
            fun=objective,

            popsize=60,
            maxiter=150,

            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',

                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
