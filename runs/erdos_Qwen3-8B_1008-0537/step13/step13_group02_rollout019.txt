import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2.0

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def generate_structured_guess():

        h = np.zeros(n_points)

            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - x))
            h[idx] = 1
        h *= (target_sum / np.sum(h))
        return h

    structured_guess = generate_structured_guess()

    def optimize_with_method(guess, opt_method, maxiter=500):

            method=opt_method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:

        return None, None

    for guess in guesses:
