import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def make_symmetric(h):
        center = n_points // 2
        half = (n_points + 1) // 2
        for i in range(half):
            if i != center:
                h[i] = h[n_points - 1 - i]
        return h
    
    def generate_evenly_spaced_guess():
        m = int(target_sum)
        positions = np.linspace(0, 2, m)
        h = np.zeros(n_points)
        pos = np.linspace(0, 2, n_points)
        for i in range(n_points):
            x = pos[i]
            idx = np.argmin(np.abs(positions - x))
            if idx < m:
                h[i] = 1.0
        return h
    
    initial_guesses = [initial_guess, make_symmetric(initial_guess), generate_evenly_spaced_guess()]
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=200,
            mutation=(0.7, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,

                method='SLSQP',

                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )

                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)

        new_guess = optimize.minimize(

        )
        if new_guess.success and objective(new_guess.x) < best_c5:
            best_c5 = objective(new_guess.x)
            best_h = new_guess.x
