    start_time = time.time()
    
    # Generate structured initial guesses
    def generate_even_guess():
        m = n_points // 2
        positions = np.linspace(0, 2, m + 1)
        h = np.zeros(n_points)
        for x in positions[1:-1]:
            idx = int((x / 2) * n_points)
            h[idx] = 1.0
        h = np.clip(h, 0, 1)
        return h
    
    def generate_uniform_guess():
        m = n_points // 2
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        uniform_guess = np.zeros(n_points)
        for i in range(m):
            uniform_guess[idxs[i]] = 1.0
        if n_points % 2:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (x[-1] - x[0]) * m / (m + 1)))
            uniform_guess[idx % n_points] = 1.0
        return np.clip(uniform_guess, 0, 1)
    
    def generate_random_guess():
        h = np.random.rand(n_points)
        h.sort()
        m = n_points // 2
        h[:m] = 1.0
        h[m:] = 0.0
        while np.abs(np.sum(h) - target_sum) > 1e-5:
            h += np.random.normal(0, 1e-3, n_points)
            h = np.clip(h, 0, 1)
        h.sort()
        h[:m] = 1.0
        h[m:] = 0.0
        return h
    
    # Multi-start optimization
    initial_guesses = [initial_guess, generate_even_guess(), generate_uniform_guess(), generate_random_guess()]
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Perturbation refinement
    while time.time() - start_time < budget_s - 30:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
