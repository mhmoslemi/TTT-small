            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}

        return result_slsqp.x, result_slsqp.success
    
    # Evaluate initial guess
    init_h, init_success = optimize_from_guess(initial_guess)
    if init_success:
        current_c5 = objective(init_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = init_h
    
    # Evaluate uniform guess
    uni_h, uni_success = optimize_from_guess(uniform_guess())
    if uni_success:
        current_c5 = objective(uni_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = uni_h
    
    # Evaluate random guess
    rand_h, rand_success = optimize_from_guess(random_guess())
    if rand_success:
        current_c5 = objective(rand_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = rand_h
    
    # Differential evolution with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=300,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            de_h = result_de.x
            current_c5 = objective(de_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = de_h
    except:
        pass
    
    # Local refinement with perturbations
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess, success = optimize_from_guess(perturbed)
        if success:
            current_c5 = objective(new_guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess
    
    return best_h, best_c5, n_points
