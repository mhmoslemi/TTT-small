            h0[i] = 1.0
        h0 = h0.tolist()
    
    n_points = len(h0)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(h0, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_with_slsqp(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success:
            return result.x
        return x0
    
    # Initial optimization from given h
    result = optimize_with_slsqp(h)
    current_c5 = objective(result)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = result
    
    # Generate better initial guess: evenly spaced
    m = int(target_sum)
    remaining = target_sum - m
    positions = np.linspace(0, 2, m + 1)
    idxs = np.searchsorted(np.linspace(0, 2, n_points), positions)
    uniform_guess = np.zeros(n_points)
    for i in range(m):
        uniform_guess[idxs[i]] = 1.0
    if remaining > 0:
        # Spread the remaining to minimize clustering
        spread_pos = np.linspace(0, 2, n_points)
        best_idx = np.argmin(np.abs(spread_pos - (positions[-1] - positions[0]) * m / (m + 1)))
        uniform_guess[best_idx % n_points] += remaining
    result = optimize_with_slsqp(uniform_guess)
    current_c5 = objective(result)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = result
    
    # Differential evolution for global search
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=200,
            mutation=(0.8, 1.2),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize_with_slsqp(result_de.x)
            current_c5 = objective(result_slsqp)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp
    except:
        pass
    
    # Local search with perturbations
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize_with_slsqp(perturbed)
        current_c5 = objective(new_guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_guess
    
    return best_h, best_c5, n_points
