    dx = 2.0 / n_points

    def generate_regular_guess(n_points):
        h = np.zeros(n_points)
        h[:22] = 1.0
        h[22] = 0.5
        return h

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)
    start_time = time.time()

    def optimize_start(x0, method, **kwargs):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result

    def run_from_guess(x0):
        result = optimize_start(x0, 'SLSQP')
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        return best_c5

    try:
        # Initial guess
        run_from_guess(best_h)
    except:
        pass

    try:
        # Regular guess
        regular_guess = generate_regular_guess(n_points)
        run_from_guess(regular_guess)
    except:
        pass

    try:
        # Uniform guess
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        uniform_guess = np.zeros(n_points)
        for i in range(m):
            uniform_guess[idxs[i]] = 1.0
        if remaining > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1)))
            uniform_guess[idx % n_points] += remaining
        run_from_guess(uniform_guess)
    except:
        pass

    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        try:
            result = optimize_start(perturbed, 'SLSQP', options={'maxiter': 200})
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    return best_h, best_c5, n_points
