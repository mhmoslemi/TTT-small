import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # explore a larger configuration
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = np.zeros(n_points)
    best_c5 = 1.0
    
    # Random initial guess
    def random_guess():
        h = np.random.rand(n_points)
        total = np.sum(h)
        scale = target_sum / total
        h *= scale
        h = np.clip(h, 0, 1)
        return h
    
    # Periodic alternating pattern
    def alt_pattern():
        h = np.zeros(n_points)
        for i in range(n_points):
            if (i % 4 == 0) or (i % 4 == 1):
                h[i] = 1.0
        return h
    
    # Gradient descent with momentum for local refinement
    def optimize_with_momentum(start, iterations=200, lr=0.01, decay=0.9):
        h = start.copy()
        v = np.zeros_like(h)
        for _ in range(iterations):
            grad = np.zeros_like(h)
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            if max_idx >= len(h):
                max_idx = len(h) - 1
            h_shifted = h[max_idx - len(h) + 1 : max_idx + 1]
            for i in range(len(h)):
                if i < len(h_shifted):
                    grad[i] = - (h[i] - h_shifted[i])
                else:
                    grad[i] = 0.0
            v = decay * v + lr * grad
            h = np.clip(h + v, 0, 1)
            if np.sum(h) < target_sum - 1e-5:
                h = h * (target_sum / np.sum(h))
            if objective(h) < best_c5:
                nonlocal best_c5
                best_c5 = objective(h)
                nonlocal best_h
                best_h = h.copy()
        return h
    
    # Try multiple initial strategies
    initial_guesses = [random_guess(), alt_pattern()]
    
    start_time = time.time()
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
