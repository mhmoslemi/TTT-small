    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate additional initial guesses
    rng = np.random.Generator(np.random.PCG64(seed))
    additional_guesses = []
    
    # Random binary guess
    m = int(target_sum)
    random_indices = rng.choice(n_points, size=m, replace=False)
    random_guess = np.zeros(n_points)
    random_guess[random_indices] = 1.0
    additional_guesses.append(random_guess)
    
    # Two-cluster guess
    half = n_points // 2
    m1 = int(target_sum // 2)
    m2 = m - m1
    two_cluster_guess = np.zeros(n_points)
    two_cluster_guess[:m1] = 1.0
    start_index = half
    end_index = start_index + m2
    if end_index > n_points:
        end_index = n_points
    two_cluster_guess[start_index:end_index] = 1.0
    additional_guesses.append(two_cluster_guess)
    
    # Optimize each initial guess
    for guess in [initial_guess] + additional_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Perturbation steps
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + rng.normal(0, 0.0005, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
