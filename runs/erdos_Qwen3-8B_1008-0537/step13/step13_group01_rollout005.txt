        return np.max(corr) * dx

        return np.sum(h_values) - target_sum

    def optimize_guess(guess, method='SLSQP'):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    def make_symmetric(h):
        center = n_points // 2

                h[i] = h[n_points - 1 - i]
        return h
    
    initial_symmetric = initial_guess.copy()

    initial_guesses = [initial_guess, initial_symmetric]
    
    for guess in initial_guesses:
