import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)

    def objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def uniform_guess():
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        g = np.zeros(n_points)
        for i in range(m):
            g[idxs[i]] = 1.0
        if remaining:
            pos = np.argmin(np.abs(np.linspace(0, 2, n_points) - (2 * m) / (m + 1)))
            g[pos % n_points] += remaining
        return g

    for init in [best_h, uniform_guess(), np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))]:
        result = optimize.minimize(
            objective, init, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_h, best_c5 = result.x, objective(result.x)

    start = time.time()
    while time.time() - start < budget_s - 50:
        perturb = best_h + np.random.normal(0, 0.001, n_points)
        perturb = np.clip(perturb, 0, 1)
        result = optimize.minimize(
            objective, perturb, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_h, best_c5 = result.x, objective(result.x)

    return best_h, best_c5, n_points
