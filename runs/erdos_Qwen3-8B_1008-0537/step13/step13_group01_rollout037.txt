    remaining = target_sum - m
    ideal_positions = np.linspace(0, 2, m)
    grid = np.linspace(0, 2, n_points)
    idxs = np.searchsorted(grid, ideal_positions)
    idxs = np.clip(idxs, 0, n_points - 1)
    uniform_guess = np.zeros(n_points)
    for i in range(m):
        idx = idxs[i]
        if i < len(idxs) and idx < n_points:
            uniform_guess[idx] = 1.0
    if remaining > 0:
        idx_add = np.argmin(np.abs(grid - (ideal_positions[-1] + ideal_positions[0])/2))
        uniform_guess[idx_add] += remaining
    result = optimize_guess(uniform_guess)
    if result is not None and objective(result) < best_c5:
        best_c5 = objective(result)
        best_h = result
    
    # Differential evolution with improved parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=100,
            maxiter=300,
            mutation=(0.8, 1.2),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with aggressive perturbation
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.01, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success and objective(new_guess.x) < best_c5:
            best_c5 = objective(new_guess.x)
            best_h = new_guess.x
    
    return best_h, best_c5, n_points
