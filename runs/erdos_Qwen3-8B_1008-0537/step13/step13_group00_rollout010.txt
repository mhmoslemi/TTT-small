        return np.sum(h_values) - target_sum

        m = int(target_sum)

        x = np.linspace(0, 2, m + 1)

        guess = np.zeros(n_points)
        for i in range(m):
            guess[idxs[i]] = 1.0
        if remain > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (2 * m / (m + 1))))

        return guess
