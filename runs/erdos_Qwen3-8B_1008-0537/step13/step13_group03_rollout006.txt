import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_uniform_guess():
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        guess = np.zeros(n_points)
        for i in range(m):
            guess[idxs[i]] = 1.0
        if remaining > 0:
            pos = np.mean(x)
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos))
            guess[idx] += remaining
        return guess
    
    def generate_periodic_guess():
        res = np.zeros(n_points)
        step = 2.0 / (n_points // 2 + 1)
        for i in range(n_points // 2):
            idx = int((i * step) * n_points / 2)
            res[idx] = 1.0
        return res
    
    def optimize_guess(guess, max_iter=100):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': max_iter, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x
        return guess
    
    def perturb_guess(guess, scale=0.001):
