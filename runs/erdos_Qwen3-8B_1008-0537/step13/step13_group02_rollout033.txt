import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_uniform_initial():
        m = int(target_sum)
        remaining = target_sum - m
        positions = np.linspace(0, 2, m + 1)[1:]  # exclude first point for even spacing
        idxs = np.searchsorted(np.linspace(0, 2, n_points), positions)
        uniform_guess = np.zeros(n_points)
        for i in range(m):
            uniform_guess[idxs[i]] = 1.0
        if remaining > 0:
            pos = 2.0 - (2.0 / (m + 1))
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos))
            uniform_guess[idx] += remaining
        return uniform_guess
    
    uniform_guess = generate_uniform_initial()
    
    results = []
    for guess in [initial_guess, uniform_guess]:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-7}

            best_c5 = c
