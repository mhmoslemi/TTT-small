import scipy.optimize as optimize

def run(seed=42, budget_s=100, **kwargs):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def uniform_guess():
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        uniform_guess = np.zeros(n_points)
        for i in range(m):
            uniform_guess[idxs[i]] = 1.0
        if remaining > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1)))
            uniform_guess[idx % n_points] += remaining
        return uniform_guess
    
    def symmetric_guess():
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 1, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        sym_guess = np.zeros(n_points)
        for i in range(m):
            sym_guess[idxs[i]] = 1.0
            sym_guess[n_points - 1 - idxs[i]] = 1.0
        if remaining > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1)))
            sym_guess[idx % n_points] += remaining
            sym_guess[n_points - 1 - idx % n_points] += remaining
        return sym_guess

    result_slsqp_initial = optimize.minimize(
        fun=objective,
