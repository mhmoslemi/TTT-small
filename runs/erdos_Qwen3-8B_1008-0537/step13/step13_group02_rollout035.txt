        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = objective(best_h)

    def generate_structured_guess():
        m = int(target_sum)

            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - p))
            guess[idx] += 1.0
        rem = target_sum - m
        if rem > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos[-1]))

    guesses = [initial_guess.copy(), generate_structured_guess(), np.random.rand(n_points), np.zeros(n_points)]

        result_slsqp = optimize.minimize(
