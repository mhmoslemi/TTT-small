            best_h = result

    m = int(target_sum)
    remaining = target_sum - m
    x = np.linspace(0, 2, m + 1)
    idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
    uniform_guess = np.zeros(n_points)
    for i in range(m):
        uniform_guess[idxs[i]] = 1.0
    if remaining > 0:
        uniform_guess[np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1))) % n_points] += remaining
    result = optimize_guess(uniform_guess)
    if result is not None and objective(result) < best_c5:
        best_c5 = objective(result)
        best_h = result
    
    # Differential evolution
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=100,
            maxiter=300,
            mutation=(0.6, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.01, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success and objective(new_guess.x) < best_c5:
            best_c5 = objective(new_guess.x)
            best_h = new_guess.x
    
    return best_h, best_c5, n_points
