import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess, method='SLSQP'):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    def make_symmetric(h):
        center = n_points // 2
        for i in range(center):
            h[i] = h[n_points - 1 - i]
        return h
    
    def gen_uniform():
        m = int(target_sum)
        rem = target_sum - m
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        guess = np.zeros(n_points)
        for i in range(m):
            guess[idxs[i]] = 1.0
        if rem > 0:
            guess[np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1))) % n_points] += rem
        return guess
    
    def gen_random():
        m = int(target_sum)
        rem = target_sum - m
        guess = np.zeros(n_points)
        indices = np.random.choice(n_points, size=m, replace=False)
        guess[indices] = 1.0
        if rem > 0:
            idx = np.random.choice(n_points)
            guess[idx] += rem
        return guess
    
    initial_guesses = [initial_guess.copy(), make_symmetric(initial_guess.copy()), gen_uniform(), gen_random()]
    
    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None and objective(res) < best_c5:
            best_c5 = objective(res)
            best_h = res
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=40,
            maxiter=150,
            mutation=(0.7, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            res = optimize_guess(result_de.x, method='SLSQP')
            if res is not None and objective(res) < best_c5:
                best_c5 = objective(res)
                best_h = res
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        res = optimize_guess(perturbed)
        if res is not None and objective(res) < best_c5:
            best_c5 = objective(res)
            best_h = res
    
    return best_h, best_c5, n_points
