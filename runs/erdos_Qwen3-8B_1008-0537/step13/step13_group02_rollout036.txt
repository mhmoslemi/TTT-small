        idx = np.random.choice(n_points)
        rand[idx] += rem
    res = optimize.minimize(objective, rand, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], options={'maxiter': 200, 'ftol': 1e-12})
    if res.success:
        curr = objective(res.x)
        if curr < best_c5:
            best_h, best_c5 = res.x, curr

    # Local refinement
    for _ in range(5):
        pert = best_h + np.random.normal(0, 0.01, n_points)
        pert = np.clip(pert, 0, 1)
        res = optimize.minimize(objective, pert, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], options={'maxiter': 50, 'ftol': 1e-12})
        if res.success:
            curr = objective(res.x)
            if curr < best_c5:
                best_h, best_c5 = res.x, curr

    return best_h, best_c5, n_points
