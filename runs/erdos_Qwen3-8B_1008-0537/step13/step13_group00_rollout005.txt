        return np.sum(h) - target_sum

    try:

            fun=objective,

            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,

                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}

        return uniform_guess
    
    result_slsqp_uniform = optimize.minimize(
