import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def refine_guess(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else x0
    
    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=100,
        maxiter=300,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-10,
        vectorized=True
    )
    if result_de.success:
        best_h = refine_guess(result_de.x)
        best_c5 = objective(best_h)
    
    def generate_uniform_guess():
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        guess = np.zeros(n_points)
        for i in range(m):
            guess[idxs[i]] = 1.0
        if remaining > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1)))
            guess[idx % n_points] += remaining
        return guess
    
    guess = generate_uniform_guess()
    best_h = refine_guess(guess)
    best_c5 = objective(best_h)
    
    def generate_sparse_guess():
        m = int(target_sum)
        idxs = np.random.choice(n_points, m, replace=False)
        guess = np.zeros(n_points)
        guess[idxs] = 1.0
        return guess
    
    guess = generate_sparse_guess()
    best_h = refine_guess(guess)
    best_c5 = objective(best_h)
    
    while time.time() - start_time < budget_s - 30:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        best_h = refine_guess(perturbed)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
