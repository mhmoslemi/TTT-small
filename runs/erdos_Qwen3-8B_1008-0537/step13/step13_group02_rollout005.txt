import scipy.optimize as optimize
import time

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def create_even_guess():
        m = int(target_sum)
        remaining = target_sum - m
        positions = np.linspace(0, 2, m + 1)
        positions = positions[:-1]
        grid = np.linspace(0, 2, n_points)
        guess = np.zeros(n_points)
        for pos in positions:
            idx = np.argmin(np.abs(grid - pos))
            guess[idx] += 1.0
        if remaining > 0:
            pos_last = positions[-1]
            idx_last = np.argmin(np.abs(grid - pos_last))
            guess[idx_last] += remaining
        return guess
    
    guesses = [initial_guess, np.random.rand(n_points), np.zeros(n_points)]
