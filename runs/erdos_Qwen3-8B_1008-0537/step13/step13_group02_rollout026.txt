import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def create_uniform_guess(n_points):
        m = int(n_points / 2)
        remaining = n_points / 2 - m
        x = np.linspace(0, 2, n_points)
        idxs = np.searchsorted(x, np.linspace(0, 2, m + 1)[:-1])
        guess = np.zeros(n_points)
        for i in range(m):
            guess[idxs[i]] = 1.0
        if remaining > 0:
            guess[idxs[-1]] += remaining
        return guess

    uniform_guess = create_uniform_guess(n_points)

    start_time = time.time()

    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=300,
            mutation=(0.7, 1.0),
            recombination=0.8,
            tol=1e-9
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,

                options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9}

            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
