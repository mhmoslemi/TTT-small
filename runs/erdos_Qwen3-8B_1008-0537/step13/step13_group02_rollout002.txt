import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_slsqp(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    result_slsqp = optimize.minimize(
        fun=objective_slsqp,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_c5 = objective_slsqp(result_slsqp.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
    
    result_de = optimize.differential_evolution(
        fun=objective_slsqp,
        bounds=bounds,
        popsize=50,
        maxiter=200,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-10,
        vectorized=True,
        polish=True
    )
    if result_de.success:
        result_slsqp_polish = optimize.minimize(
            fun=objective_slsqp,
            x0=result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp_polish.success:
            current_c5 = objective_slsqp(result_slsqp_polish.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp_polish.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
