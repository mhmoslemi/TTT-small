import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def make_symmetric(h):
        n = n_points
        for i in range(n // 2 + 1):
            h[i] = h[n - 1 - i]
        return h
    
    def make_spaced(h):
        m = int(target_sum)
        remaining = target_sum - m
        spacing = n_points / (m + 1)
        positions = np.arange(1, m+1) * spacing
        indices = np.searchsorted(np.linspace(0, 2, n_points), positions)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if remaining > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (2 - 0.5 * spacing)))
            h[idx] += remaining
        return h
    
    initial_guesses = [
        initial_guess,
        make_symmetric(initial_guess.copy()),
        np.zeros(n_points),
        make_spaced(np.zeros(n_points))
    ]
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    def random_guess():
        h = np.zeros(n_points)

        )

            best_c5 = objective(result_slsqp.x)
