import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def uniform_guess():
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        uniform_guess = np.zeros(n_points)
        for i in range(m):
            uniform_guess[idxs[i]] = 1.0
        if remaining > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1)))
            uniform_guess[idx % n_points] += remaining
        return uniform_guess
    
    def optimize_once(x0, method='SLSQP', **kwargs):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x, objective(result.x)
        return x0, objective(x0)
    
    def optimize_with_perturbations(x0, budget):
        nonlocal best_h, best_c5
        x, c5 = optimize_once(x0)
        if c5 < best_c5:
            best_h, best_c5 = x, c5
        while time.time() - start_time < budget - 10:
            perturbed = x + np.random.normal(0, 0.001, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)
            x_pert, c5_pert = optimize_once(perturbed, method='SLSQP', options={'maxiter': 100})
            if c5_pert < best_c5:
                best_h, best_c5 = x_pert, c5_pert
            x, c5 = x_pert, c5_pert
    
    try:
        optimize_with_perturbations(initial_guess, budget_s - 20)
