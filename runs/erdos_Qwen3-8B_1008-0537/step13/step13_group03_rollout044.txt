import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Generate initial guesses
    initial_guesses = [np.array(initial_h_values, dtype=float)]
    m = n_points // 2
    uniform_guess = np.zeros(n_points)
    uniform_guess[:m] = 1.0
    if n_points % 2 == 1:
        uniform_guess[m] = 0.5
    initial_guesses.append(uniform_guess)
    random_guess = np.random.rand(n_points)
    total = np.sum(random_guess)
    scale = (n_points / 2) / total
    random_guess = np.clip(random_guess * scale, 0, 1)
    initial_guesses.append(random_guess)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for h0 in initial_guesses:
        # Run differential evolution
        try:
            result_de = optimize.differential_evolution(
                fun=lambda x: np.max(np.correlate(x, 1 - np.array(x), mode='full')) * dx,
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                popsize=50,
                maxiter=150,
                mutation=(0.7, 1.0),
                recombination=0.8,
                tol=1e-10,
                vectorized=True
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=lambda x: np.max(np.correlate(x, 1 - np.array(x), mode='full')) * dx,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=[(0.0, 1.0) for _ in range(n_points)],
                    constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                    options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = result_slsqp.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x.copy()
        except:

        # Run SLSQP directly with initial guess

            result_slsqp = optimize.minimize(

                x0=h0,
