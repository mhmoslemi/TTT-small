    m = int(target_sum)
    positions = np.linspace(0, 2, m + 1)
    grid = np.linspace(0, 2, n_points)
    regular_guess = np.zeros(n_points)
    for pos in positions[:-1]:
        idx = np.argmin(np.abs(grid - pos))
        regular_guess[idx] = 1.0
    if m != target_sum:
        remaining = target_sum - m
        pos = positions[-1] - (positions[-1] - positions[-2]) / 2
        idx = np.argmin(np.abs(grid - pos))
        regular_guess[idx] += remaining
    initial_guesses.append(regular_guess)

    # Random initial guess
    random_guess = np.zeros(n_points)
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    random_guess[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        random_guess[idx] += remaining
    initial_guesses.append(random_guess)

    # Uniform initial guess
    m = int(target_sum)
    remaining = target_sum - m
    x = np.linspace(0, 2, m + 1)
    idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
    uniform_guess = np.zeros(n_points)
    for i in range(m):
        uniform_guess[idxs[i]] = 1.0
    if remaining > 0:
        uniform_guess[np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1))) % n_points] += remaining
    initial_guesses.append(uniform_guess)

    best_h = initial_guesses[0]
    best_c5 = float('inf')
    start_time = time.time()

    for guess in initial_guesses:
        result = optimize.minimize(
            fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
            x0=guess,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    # Perturbation step
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize.minimize(
            fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
            x0=perturbed,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
