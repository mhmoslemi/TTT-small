        for i in range(m):
            uniform_guess[idxs[i]] = 1.0
        if remaining > 0:
            pos = np.linspace(0, 2, n_points)
            idx = np.argmin(np.abs(pos - (np.max(x) - np.min(x)) * m / (m + 1)))
            uniform_guess[idx % n_points] += remaining
        return uniform_guess
    
    uniform_guess = generate_uniform_initial()
    
    def optimize_with_guess(guess):
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=200,
            mutation=(0.5, 1.0),
            recombination=0.4,
            tol=1e-10,
            vectorized=True,
            strategy='best1bin'
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                return result_slsqp.x
        return guess
    
    best_h = optimize_with_guess(initial_guess)
    best_c5 = objective(best_h)
    best_h = optimize_with_guess(uniform_guess)
    best_c5 = objective(best_h)
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0005, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
