            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x, objective(result.x)
        return guess, objective(guess)

    guesses = [initial_guess, binary_guess, uniform_guess]
    results = [(g, objective(g)) for g in guesses]

    for g, c in results:
        _, c_new = run_optimize(g)
        if c_new < best_c5:
            best_h, best_c5 = run_optimize(g)

    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_h, best_c5 = new_guess.x, current_c5

    return best_h, best_c5, n_points
