    initial_guess_opt, initial_c5 = optimize_guess(initial_guess)

    best_h = initial_guess_opt
    
    guessing = [initial_guess, generate_evenly_spread_binary(), generate_uniform_guess()]
    
    for guess in guessing:
        if np.sum(guess) * dx != 1.0:
            guess = guess.copy()
            guess[np.argmin(guess)] += (1.0 - np.sum(guess) * dx)
            guess = np.clip(guess, 0, 1)
        guess_opt, c5 = optimize_guess(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = guess_opt
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
