def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    regular_guess = np.zeros(n_points)
    m = 5
    L = 1.0 / m
    S = 1.0 / (m - 1)
    current_start = 0.0
    for i in range(m):
        current_end = current_start + L
        i_start = int(np.floor(current_start / dx))
        i_end = int(np.ceil(current_end / dx))
        i_start = np.clip(i_start, 0, n_points - 1)
        i_end = np.clip(i_end, 0, n_points - 1)
        if i_end > i_start:
            regular_guess[i_start:i_end] = 1.0
        current_start = current_end + S
    guesses = [initial_guess, regular_guess]
    random_guess = np.zeros(n_points)
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    random_guess[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        random_guess[idx] += remaining
    guesses.append(random_guess)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for guess in guesses:
        try:
            result_de = optimize.differential_evolution(
                fun=objective_de,

                popsize=100,
                maxiter=300,

                tol=1e-10,

                x0=guess
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
