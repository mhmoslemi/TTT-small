import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.inf
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_uniform_guess():
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 2, n_points)
        positions = 2 * np.arange(1, m + 1) / (m + 1)
        indices = np.searchsorted(x, positions, side='right') - 1
        guess = np.zeros(n_points)
        for i in range(m):
            idx = indices[i]
            if 0 <= idx < n_points:
                guess[idx] = 1.0
        if remaining > 0:
            pos = (np.max(x) - np.min(x)) * m / (m + 1)
            idx = np.argmin(np.abs(x - pos))
            guess[idx] += remaining
        return guess
    
    def generate_random_guess():
        indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        guess = np.zeros(n_points)
        guess[indices] = 1.0
        return guess
    
    def run_optimization(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x, result.fun
        return x0, np.inf
    
    start_time = time.time()
    
    # Initial guess
    x0 = initial_h_values
    h, c5 = run_optimization(x0)
    if c5 < best_c5:

        best_h = h
    
    # Uniform guess
    x0 = generate_uniform_guess()

    if c5 < best_c5:

        best_h = h
    
    # Random guess
    x0 = generate_random_guess()

    # Differential evolution
    try:
        result_de = optimize.differential_evolution(

    except:
