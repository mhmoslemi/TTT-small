import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp, phase=0.0):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * (freq * t + phase))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_tri(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.abs(2 * (t - 0.5) * freq - 1) * 2
        return np.clip(0.5 - wave * 0.5, 0, 1)
    
    def generate_random_perturb(h_base):
        noise = np.random.normal(0, 0.1, n_points)
        h_perturb = h_base + noise
        h_perturb = np.clip(h_perturb, 0, 1)
        scale = 1.0 - (np.sum(h_perturb) - target_sum) / target_sum
        h_perturb *= scale / np.sum(h_perturb)
        return h_perturb
    
    guesses = [h]
    for f, a in [(3, 0.6), (4, 0.5), (5, 0.4), (7, 0.3), (2, 0.7)]:
        guesses.append(generate_sine(f, a))
        guesses.append(generate_sine(f, a, phase=1.0))
    guesses.append(generate_tri(3, 0.6))
    guesses.append(generate_tri(4, 0.5))
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        for guess in guesses + [generate_random_perturb(h)]:
            res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:

        res = optimize.differential_evolution(objective, bounds, popsize=25, maxiter=30,

                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_bfgs(h_init):
