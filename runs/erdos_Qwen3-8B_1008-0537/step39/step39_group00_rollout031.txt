import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_regular(m):
        positions = np.linspace(0, 2, m, endpoint=False)
        h = np.zeros(n_points)
        for x in positions:
            idx = int(x / dx)
            if idx < n_points:
                h[idx] = 1.0
        return h
    
    def generate_clustered(total_ones, split_ratio=0.5):
        split = int(total_ones * split_ratio)
        cluster1 = np.linspace(0, 1, split + 1, endpoint=False)
        cluster2 = np.linspace(1, 2, total_ones - split + 1, endpoint=False)
        h = np.zeros(n_points)
        for x in cluster1:
            idx = int(x / dx)
            if idx < n_points:
                h[idx] = 1.0
        for x in cluster2:
            idx = int(x / dx)
            if idx < n_points:
                h[idx] = 1.0
        return h
    
    def generate_random(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        return h
    
    guesses = [h]
    for m in (16, 20, 24, 28, 32):
        guesses.append(generate_regular(m))
    guesses.append(generate_clustered(22))
    guesses.append(generate_clustered(23))
    guesses.append(generate_random(22))
    guesses.append(generate_random(23))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):

        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=20,

            if res_slsqp.success and objective(res_slsqp.x) < best_c5:

                best_h = res_slsqp.x
    
    def refine_with_neldermead(h_init):
