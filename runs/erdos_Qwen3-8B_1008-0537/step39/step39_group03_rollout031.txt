import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * amp * np.sin(2 * np.pi * freq * t), 0, 1)
    
    def generate_square(freq):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sign(np.sin(2 * np.pi * freq * t)), 0, 1)
    
    def generate_triangular(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.abs(2 * t - 1) - 0.5
        return np.clip(0.5 + 0.5 * wave * np.sin(2 * np.pi * freq * t), 0, 1)
    
    def generate_random_perturbation(base):
        noise = 0.1 * np.random.randn(n_points)
        return np.clip(base + noise, 0, 1)
    
    guesses = [h]
    guesses += [generate_sine(f, a) for f, a in [(3, 0.6), (4, 0.5), (5, 0.4)]]
    guesses += [generate_square(f) for f in [3, 4, 5]]
    guesses += [generate_triangular(f) for f in [3, 4, 5]]
    guesses += [generate_random_perturbation(h)]
    
    for guess in guesses:
        try:
            res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9})
        except:
            res = optimize.minimize(objective, guess, method='L-BFGS-B', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'gtol': 1e-9, 'maxiter': 70})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=80,

        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,

                                         options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:

                best_h = res_slsqp.x
    
    def refine_with_bfgs(h_init):
        try:
            res = optimize.minimize(objective, h_init, method='L-BFGS-B', bounds=bounds,
