        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = objective(best_h)

        positions = np.linspace(0, 2, m, endpoint=False)
        h = np.zeros(n_points)
        for x in positions:
            idx = int(x / dx)
            if idx < n_points:
                h[idx] = 1.0
        return h
    
    def generate_random(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        return h
    
    guesses = [

    ]
