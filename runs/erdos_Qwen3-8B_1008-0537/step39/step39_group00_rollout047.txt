        return np.sum(h_vals) - target_sum

        positions = np.linspace(0, 2, m, endpoint=False)
        h = np.zeros(n_points)

            idx = int(x / dx)
            if idx < n_points:
                h[idx] = 1.0
        # Adjust sum to target_sum
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-12:
            idx = np.argmin(h) if diff > 0 else np.argmax(h)
            if idx < n_points:
                h[idx] += diff
        return h
    
    def generate_random(ones):
