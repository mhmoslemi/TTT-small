import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(freq, amp, type='sine'):
        t = np.linspace(0, 1, n_points)
        if type == 'sine':
            return np.clip(0.5 + 0.5 * amp * np.sin(2 * np.pi * freq * t), 0, 1)
        elif type == 'square':
            return np.clip(0.5 + 0.5 * np.sign(np.sin(2 * np.pi * freq * t)), 0, 1)
        elif type == 'triangle':
            return np.clip(0.5 + 0.5 * np.abs(2 * np.pi * freq * t - np.pi) / np.pi, 0, 1)
        else:
            return h
    
    # Generate diverse initial guesses
    guesses = [h]
    frequencies = [3, 4, 5, 6, 7]
    amps = [0.9, 0.75, 0.6, 0.5, 0.4]
    for f, a in zip(frequencies, amps):
        for wtype in ['sine', 'square', 'triangle']:
            guesses.append(generate_wave(f, a, wtype))
    guesses.append(np.clip(h + 0.1 * np.random.randn(n_points), 0, 1))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=50,
                                             mutation=(0.9, 1.1), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_bfgs(h_init):
        res = optimize.minimize(objective, h_init, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'gtol': 1e-9, 'maxiter': 70})
        return res.x if res.success else h_init
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_bfgs(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
