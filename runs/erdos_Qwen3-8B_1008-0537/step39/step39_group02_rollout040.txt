import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)

        return np.clip(h_base + 0.1 * np.random.randn(n_points), 0, 1)
    
    def generate_checkerboard(pattern_size):
        pattern = np.zeros(n_points)
        for i in range(pattern_size):
            start = int(i * n_points / pattern_size)
            pattern[start:start+1] = 1
        return np.concatenate([pattern, 1 - pattern])
    
    guesses = [h]
    guesses += [generate_random_perturbation(h) for _ in range(2)]

        if res.success and objective(res.x) < best_c5:
