import scipy.optimize as optimize
import time

        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = objective(best_h)

        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * amp * np.sin(2 * np.pi * freq * t), 0, 1)
    
    def generate_square(freq):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sign(np.sin(2 * np.pi * freq * t)), 0, 1)
    
    def generate_triangle(freq):

        return np.clip(0.5 + 0.5 * np.abs(2 * np.pi * freq * t % 2 - 1), 0, 1)
    
    def generate_binary():
        h = np.zeros(n_points)
        needed = target_sum - np.sum(h)
        idxs = np.random.choice(n_points, size=int(round(needed)), replace=False)
        h[idxs] = 1
        return h
