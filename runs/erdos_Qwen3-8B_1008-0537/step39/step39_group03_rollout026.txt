import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    h = np.array(initial_h_values, dtype=float)

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_sine(freq, amp):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * amp * np.sin(2 * np.pi * freq * t), 0, 1)

    def generate_square(freq):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sign(np.sin(2 * np.pi * freq * t)), 0, 1)

    guesses = [h]
    for freq in range(2, 6):
        for amp in np.linspace(0.3, 0.7, 5):
            guesses.append(generate_sine(freq, amp))
        for amp in np.linspace(0.3, 0.7, 5):
            guesses.append(generate_square(freq))

    guesses += [np.clip(h + 0.1 * np.random.randn(n_points), 0, 1)]

    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:

        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=50,

        if res.success:
