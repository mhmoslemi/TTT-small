import scipy.optimize as optimize
import time

    def objective(h_vals):

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)

        t = np.linspace(0, 2, n_points)
        return np.clip(0.5 + 0.5 * amp * np.sin(2 * np.pi * freq * t + phase), 0, 1)
    
    def generate_rectangular(freq, duty):
        t = np.linspace(0, 2, n_points)
        return np.clip(0.5 + 0.5 * duty * np.sign(np.sin(2 * np.pi * freq * t)), 0, 1)
    
    guesses = [h]

    guesses += [generate_rectangular(f, d) for f, d in [(3, 0.3), (4, 0.35), (5, 0.3)]]
    guesses += [np.clip(h + 0.1 * np.random.randn(n_points), 0, 1)]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
