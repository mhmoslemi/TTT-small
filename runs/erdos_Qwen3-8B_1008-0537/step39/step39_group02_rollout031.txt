import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_blocks(num_blocks):
        block_size = 2 / num_blocks
        pattern = np.zeros(n_points)
        for i in range(num_blocks):
            start = int(i * block_size * n_points / 2)
            end = int((i + 1) * block_size * n_points / 2)
            pattern[start:end] = 1
        return np.clip(pattern, 0, 1)
    
    def generate_triangle(freq):
        t = np.linspace(0, 2, n_points)
        wave = np.abs(np.sin(np.pi * freq * t))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_sawtooth(freq):
        t = np.linspace(0, 2, n_points)
        wave = (np.sin(np.pi * freq * t) + 1) / 2
        return np.clip(wave, 0, 1)
    
    def generate_random_perturbation(h_base):
        return np.clip(h_base + 0.03 * np.random.randn(n_points), 0, 1)
    
    guesses = [h]
    guesses += [generate_binary_blocks(5), generate_binary_blocks(7), generate_binary_blocks(10)]
    guesses += [generate_triangle(2), generate_triangle(3), generate_triangle(5)]
    guesses += [generate_sawtooth(2), generate_sawtooth(3), generate_sawtooth(5)]
    guesses += [generate_random_perturbation(h)]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        bounds_ = [(0.0, 1.0) for _ in range(n_points)]
        res = optimize.differential_evolution(objective, bounds_, popsize=50, maxiter=40,

        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds_,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_bfgs(h_init):
        res = optimize.minimize(objective, h_init, method='L-BFGS-B', bounds=bounds,
