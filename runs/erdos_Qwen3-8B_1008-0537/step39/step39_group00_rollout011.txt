import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Use provided initial_h_values if available
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Generate diverse initial guesses
    def generate_regular(m):
        pos = np.linspace(0, 2, m, endpoint=False)
        h = np.zeros(n_points)
        for x in pos:
            idx = int(x / dx)
            if idx < n_points:
                h[idx] = 1.0
        return h

    def generate_random(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        return h

    guesses = [h]
    for m in [15, 20, 25, 30, 35, 40]:
        guesses.append(generate_regular(m))
    for ones in [15, 20, 25, 30, 35, 40]:
        guesses.append(generate_random(ones))

    # Optimize each guess
    for guess in guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    # Differential evolution for global search
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        res_de = optimize.differential_evolution(
            objective, bounds, popsize=40, maxiter=40,
            mutation=(0.6, 0.9), recombination=0.8, tol=1e-9
        )
        if res_de.success:
            res_slsqp = optimize.minimize(
                objective, res_de.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x

    # Refinement with Nelder-Mead
    for _ in range(2):
        if time.time() - start_time >= budget_s - 5:
            break
        res_nm = optimize.minimize(
            objective, best_h, method='Nelder-Mead', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'xatol': 1e-9, 'fatol': 1e-9, 'maxiter': 30}
        )
        if res_nm.success and objective(res_nm.x) < best_c5:
            best_c5 = objective(res_nm.x)
            best_h = res_nm.x

    return best_h, best_c5, n_points
