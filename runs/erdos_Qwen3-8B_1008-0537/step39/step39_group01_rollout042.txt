def generate_step(m):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    interval_length = 1.0 / m
    gap_length = 1.0 / (m - 1)
    current_start = 0.0
    for i in range(m):
        # Interval i starts at current_start, ends at current_start + interval_length
        # For all x in [current_start, current_start + interval_length), h(x)=1
        start_x = current_start
        end_x = current_start + interval_length
        for j in range(n_points):
            x_j = j * dx
            if start_x <= x_j < end_x:
                h[j] = 1.0
        current_start += interval_length + gap_length
    return h

import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp, phase=0.0):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * (freq * t + phase))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_tanh(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.tanh(10 * (freq * t - 0.5))
        return np.clip(0.1 + 0.8 * wave, 0, 1)
    
    def generate_step(m):
        h_step = np.zeros(n_points)
        interval_length = 1.0 / m
        gap_length = 1.0 / (m - 1)
        current_start = 0.0
        for i in range(m):
            start_x = current_start
            end_x = current_start + interval_length
            for j in range(n_points):
                x_j = j * dx
                if start_x <= x_j < end_x:
                    h_step[j] = 1.0
            current_start += interval_length + gap_length
        return h_step
    
    guesses = [h]
    for f, a in [(3, 0.6), (4, 0.5), (5, 0.4), (7, 0.3), (2, 0.7)]:
        guesses.append(generate_sine(f, a))
        guesses.append(generate_sine(f, a, phase=1.0))
    guesses.append(generate_tanh(3, 0.6))
    guesses.append(generate_tanh(4, 0.5))
    guesses.append(generate_step(3))
    guesses.append(generate_step(5))
    guesses.append(generate_step(7))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=25, maxiter=30,
                                             mutation=(0.8, 1.2), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_bfgs(h_init):
        res = optimize.minimize(objective, h_init, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'gtol': 1e-9, 'maxiter': 60})
        return res.x if res.success else h_init
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_bfgs(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
