import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp, phase):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * amp * np.sin(2 * np.pi * freq * t + phase), 0, 1)
    
    def generate_square(freq, phase):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sign(np.sin(2 * np.pi * freq * t + phase)), 0, 1)
    
    guesses = [h]
    # More diverse sine and square waves
    for freq in [2, 3, 4, 5, 6, 7]:
        for amp in [0.6, 0.5, 0.4]:
            guesses.append(generate_sine(freq, amp, np.random.uniform(0, 2*np.pi)))
    for freq in [3, 4, 5, 6, 7]:
        for phase in [np.random.uniform(0, 2*np.pi) for _ in range(3)]:
            guesses.append(generate_square(freq, phase))
    guesses.append(np.clip(h + 0.1 * np.random.randn(n_points), 0, 1))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(4):

        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=50,
