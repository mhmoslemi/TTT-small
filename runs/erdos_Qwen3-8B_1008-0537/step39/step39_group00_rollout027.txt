import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_symmetric(h_init):
        h = h_init.copy()
        for i in range(n_points // 2):
            h[n_points - 1 - i] = h[i]
        h = h[:n_points // 2 + (n_points % 2)]
        h = np.pad(h, (0, n_points - len(h)), 'constant')
        return h
    
    def generate_regular(m):
        h = np.zeros(n_points)
        for i in range(m):
            pos = np.random.uniform(0, 2)
            idx = int(pos / dx)
            if idx < n_points:
                h[idx] = 1.0
        return h
    
    def generate_random(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        return h

    for guess in guesses:
