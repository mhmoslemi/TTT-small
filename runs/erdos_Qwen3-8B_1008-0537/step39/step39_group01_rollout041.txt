import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_pattern(n_points):
        pattern = np.zeros(n_points, dtype=float)
        pattern[::2] = 1.0
        sum_val = np.sum(pattern)
        if sum_val != target_sum:
            diff = sum_val - target_sum
            if diff > 0:
                for i in range(n_points):
                    if pattern[i] == 1.0:
                        pattern[i] = 0.0
                        break
            else:
                for i in range(n_points):
                    if pattern[i] == 0.0:
                        pattern[i] = 1.0
                        break
        return pattern
    
    def generate_sine(freq, amp, phase=0.0):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * (freq * t + phase))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_tanh(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.tanh(10 * (freq * t - 0.5))
        return np.clip(0.1 + 0.8 * wave, 0, 1)
    
    guesses = [h]

    guesses.append(generate_tanh(3, 0.6))
    guesses.append(generate_tanh(4, 0.5))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
