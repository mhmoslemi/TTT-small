import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_blocks(num_blocks):
        block_size = 2 / num_blocks
        pattern = np.zeros(n_points)
        for i in range(num_blocks):
            idx = int(i * block_size * n_points / 2)
            pattern[idx:idx+1] = 1
        return np.clip(pattern, 0, 1)
    
    def generate_triangle(freq):
        t = np.linspace(0, 2, n_points)
        wave = np.abs(np.sin(np.pi * freq * t))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_random_perturbation(h_base):
        noise = np.random.normal(0, 0.05, n_points)
        return np.clip(h_base + noise, 0, 1)
    
    guesses = [h]
    for blocks in [3, 4, 5, 6, 7, 8, 10]:
        guesses.append(generate_binary_blocks(blocks))
    for freq in [1, 2, 3, 4, 5]:
        guesses.append(generate_triangle(freq))
    guesses.append(generate_random_perturbation(h))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):

        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=40,

            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
