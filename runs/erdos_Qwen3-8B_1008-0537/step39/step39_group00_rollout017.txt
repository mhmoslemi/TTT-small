import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

        positions = np.linspace(0, 2, m, endpoint=False)
        weights = np.zeros(n_points)
        for x in positions:
            idx = int(round(x / dx))
            if 0 <= idx < n_points:
                weights[idx] += 1.0
        return weights
    
    def generate_random(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        weights = np.zeros(n_points)
        weights[idx] = 1.0
        return weights
    
    def refine_with_neldermead(h_init):
        res = optimize.minimize(objective, h_init, method='Nelder-Mead', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xatol': 1e-9, 'fatol': 1e-9, 'maxiter': 30})
        return res.x if res.success else h_init
