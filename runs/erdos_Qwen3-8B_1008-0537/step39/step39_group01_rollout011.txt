import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp, phase=0.0):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * (freq * t + phase))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_tanh(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.tanh(10 * (freq * t - 0.5))
        return np.clip(0.1 + 0.8 * wave, 0, 1)
    
    def generate_square(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sign(np.sin(2 * np.pi * freq * t))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    guesses = [h]
    for f, a in [(3, 0.6), (4, 0.5), (5, 0.4), (7, 0.3), (2, 0.7), (1, 0.8)]:
        guesses.append(generate_sine(f, a))
        guesses.append(generate_sine(f, a, phase=1.0))
        guesses.append(generate_tanh(f, a))
        guesses.append(generate_square(f, a))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=40,

                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_l_bfgs(h_init):
