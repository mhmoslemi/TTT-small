        return np.max(corr) * dx

        positions = np.linspace(0, 2, m, endpoint=False)
        h = np.zeros(n_points)

            idx = int(x / dx)
            if idx < n_points:
                h[idx] = 1.0
        return h

        h[idx] = 1.0
