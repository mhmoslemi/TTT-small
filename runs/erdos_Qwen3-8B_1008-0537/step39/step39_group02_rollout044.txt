import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_binary_blocks(num_blocks):
        block_size = 2 / num_blocks
        pattern = np.zeros(n_points)
        for i in range(num_blocks):
            start = int(i * block_size * n_points / 2)
            end = int((i + 1) * block_size * n_points / 2)
            pattern[start:end] = 1
        return np.clip(pattern, 0, 1)

    def generate_triangle(freq):
        t = np.linspace(0, 2, n_points)
        wave = np.abs(np.sin(np.pi * freq * t))
        return np.clip(0.5 + 0.5 * wave, 0, 1)

    def generate_sine(freq, phase):
        t = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * t + phase)
        return np.clip((wave + 1)/2, 0, 1)

    def generate_random_binary():
        return np.clip(np.random.rand(n_points) < 0.5, 0, 1)

    def generate_perturbed(h_base, std_dev):
        return np.clip(h_base + np.random.normal(0, std_dev, n_points), 0, 1)

    guesses = [h]
    for freq in [2, 3, 5]:
        guesses += [generate_triangle(freq), generate_sine(freq, 0), generate_sine(freq, np.pi/2)]
    guesses += [generate_binary_blocks(5), generate_binary_blocks(7), generate_binary_blocks(10)]
    guesses += [generate_random_binary(), generate_perturbed(h, 0.05), generate_perturbed(h, 0.1)]

    best_guesses = []
    for guess in guesses:
        if np.allclose(np.sum(guess), target_sum, atol=1e-6):
            best_guesses.append(guess)
        else:
            guess = generate_perturbed(guess, 0.1)
            best_guesses.append(guess)

    for guess in best_guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success:

            if current_c5 < best_c5:

    for _ in range(2):
