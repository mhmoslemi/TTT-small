import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_fibonacci(freq, amp, phase=0.0):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * (freq * t + phase) + np.pi * np.sin(4 * np.pi * (freq * t + phase)))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_three_peaks():
        x = np.linspace(0, 2, n_points)
        peaks = np.zeros_like(x)
        for i in range(3):
            peaks += np.exp(-((x - 0.5 + i * 0.5) / 0.25) ** 2)
        return np.clip(peaks / np.max(peaks), 0, 1)
    
    def generate_sparse_rects():
        x = np.linspace(0, 2, n_points)
        rects = np.zeros_like(x)
        for i in range(4):
            rect = np.zeros_like(x)
            rect[20 + i*10:20 + i*10 + 5] = 1
            rects += rect
        return np.clip(rects / np.max(rects), 0, 1)
    
    guesses = [h]
    for f, a in [(3, 0.6), (4, 0.5), (5, 0.4), (7, 0.3), (2, 0.7), (8, 0.45), (10, 0.35)]:
        guesses.append(generate_fibonacci(f, a))
        guesses.append(generate_fibonacci(f, a, phase=1.0))
    guesses.append(generate_three_peaks())
    guesses.append(generate_sparse_rects())
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:

            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:

        res = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=20,

    def refine_with_bfgs(h_init):
