            break

                                                mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Nelder-Mead refinement
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        res_nm = optimize.minimize(objective, best_h, method='Nelder-Mead', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'xatol': 1e-9, 'fatol': 1e-9, 'maxiter': 30})
        if res_nm.success and objective(res_nm.x) < best_c5:
            best_c5 = objective(res_nm.x)
            best_h = res_nm.x
    
    return best_h, best_c5, n_points
