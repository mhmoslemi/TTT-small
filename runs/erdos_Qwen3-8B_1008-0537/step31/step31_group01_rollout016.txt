import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    h = np.array(initial_h, dtype=float)  # Start with a sine-based estimate
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)

    def generate_sine(freq, amp):

        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_step(n_intervals):
        starts = np.sort(np.random.rand(n_intervals))
        lengths = np.random.rand(n_intervals) * 0.25
        intervals = []
        for start, length in zip(starts, lengths):
            end = start + length
            if end > 1:
                end = 1
            intervals.append((start, end))
        h = np.zeros(n_points)
        for start, end in intervals:
            idx = np.searchsorted(np.linspace(0, 1, n_points), [start, end])
            h[idx[0]:idx[1]] = 1
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_points), h)
        h = np.clip(h, 0, 1)
        return h
    
    guesses = [h]
    guesses += [generate_sine(f, a) for f, a in [(2, 0.6), (3, 0.5), (4, 0.4), (5, 0.3), (6, 0.2)]]
