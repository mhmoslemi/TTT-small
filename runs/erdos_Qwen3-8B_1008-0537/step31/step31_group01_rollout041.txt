import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(freq, amp, wave_type):
        t = np.linspace(0, 1, n_points)
        if wave_type == 'sin':
            wave = amp * np.sin(2 * np.pi * freq * t)
        elif wave_type == 'square':
            wave = amp * np.sign(np.sin(2 * np.pi * freq * t))
        elif wave_type == 'triangle':
            wave = amp * 2 * (np.abs(2 * np.pi * freq * t % 2 - 1) - 0.5)
        wave = 0.5 + 0.5 * wave
        return np.clip(wave, 0, 1)
    
    def generate_random(h_base):
        return np.clip(h_base + 0.05 * np.random.randn(n_points), 0, 1)
    
    guesses = [h]
    guesses += [generate_wave(f, a, 'sin') for f, a in [(3, 0.6), (4, 0.5), (5, 0.4)]]
    guesses += [generate_wave(f, a, 'square') for f, a in [(2, 0.6), (3, 0.5)]]
    guesses += [generate_wave(f, a, 'triangle') for f, a in [(2, 0.5), (3, 0.5)]]
    guesses += [generate_random(h)]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=60,
                                             mutation=(0.9, 1.1), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 80, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_bfgs(h_init):
        res = optimize.minimize(objective, h_init, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'gtol': 1e-9, 'maxiter': 100})
        return res.x if res.success else h_init
    
    for _ in range(4):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_bfgs(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
