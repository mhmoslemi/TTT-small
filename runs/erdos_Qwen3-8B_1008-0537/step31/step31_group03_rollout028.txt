import scipy.optimize as optimize
import time

    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulse(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * (x - phase))
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * (x - phase) - 1) * duty, 0, 1)
    
    guesses = [h]
    for freq in [2, 3, 4, 5]:
        for duty in [0.3, 0.4, 0.5]:
            for phase in [0.0, 0.25, 0.5, 0.75]:
                guesses.append(gen_pulse(freq, duty, phase))
    for freq in [2, 3, 4]:
        for duty in [0.2, 0.3, 0.4, 0.5]:
            for phase in [0.0, 0.25, 0.5, 0.75]:
                guesses.append(gen_saw(freq, duty, phase))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 220, 'ftol': 1e-11})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=90, maxiter=40,

        h_opt = slsqp_guess(res.x)

    def refine(r):
