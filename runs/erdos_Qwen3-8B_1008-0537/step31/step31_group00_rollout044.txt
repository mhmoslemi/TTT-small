import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty
    
    def generate_triangular(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * (2 * t - 1) * np.abs(1 - 2 * np.abs(2 * freq * t - 1))
        return np.clip(wave, 0, 1)
    
    def generate_sawtooth(freq, duty):
        t = np.linspace(0, 1, n_points)
        wave = 2 * duty * (1 - np.abs(2 * freq * t - 1))
        return np.clip(wave, 0, 1)
    
    def generate_cosine(freq):
        return np.clip(0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n_points)), 0, 1)
    
    def generate_evenly_spaced_binary(intervals_count):
        a = 1.0 / intervals_count
        t = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(intervals_count):
            start = i * a
            end = (i + 1) * a
            indices = np.where((t >= start) & (t < end))[0]
            if indices.size > 0:
                h[indices] = 1
        return h
    
    def generate_single_interval():
        start = 0.5
        end = 1.5
        t = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        indices = np.where((t >= start) & (t < end))[0]
        if indices.size > 0:
            h[indices] = 1
        return h
    
    guesses = [generate_periodic(p) for p in [3, 4, 5, 6, 7, 8]]
    guesses += [generate_binary(f, d) for f, d in [(3, 0.2), (4, 0.4), (5, 0.3), (6, 0.2)]]
    guesses += [generate_triangular(f, a) for f, a in [(3, 0.6), (4, 0.5), (5, 0.4)]]
    guesses += [generate_sawtooth(f, d) for f, d in [(3, 0.3), (4, 0.25), (5, 0.2)]]
    guesses += [generate_cosine(f) for f in [3, 4, 5]]
    guesses.append(np.random.rand(n_points))
    guesses.append(h)
    guesses.append(generate_evenly_spaced_binary(4))
    guesses.append(generate_single_interval())
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:

    return best_h, best_c5, n_points
