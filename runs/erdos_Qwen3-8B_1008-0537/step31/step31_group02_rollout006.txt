import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_regular(m):
        positions = np.linspace(0, 2, m, endpoint=False)
        h = np.zeros(n_points)
        for x in positions:
            idx = int(x / dx)
            if idx < n_points:
                h[idx] = 1.0
        return h
    
    def generate_random(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        return h
    
    guesses = [h, generate_regular(22), generate_regular(23), generate_random(22), generate_random(23)]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=30,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_neldermead(h_init):
        res = optimize.minimize(objective, h_init, method='Nelder-Mead', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xatol': 1e-9, 'fatol': 1e-9, 'maxiter': 30})
        return res.x if res.success else h_init
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 5:
            break
        best_h = refine_with_neldermead(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
