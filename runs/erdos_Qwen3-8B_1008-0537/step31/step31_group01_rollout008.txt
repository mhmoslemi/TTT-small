import scipy.optimize as optimize
import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

        ones = np.random.binomial(1, 0.5, n_points)
        ones = (ones * 2 - 1).astype(float)
        ones *= 0.5
        ones += 0.5
        ones = np.clip(ones, 0, 1)
        total = np.sum(ones)
        diff = target_sum - total
        if diff != 0:
            idx = np.argsort(ones)
            ones[idx[-int(np.abs(diff)):] += np.sign(diff) * 0.5
            ones = np.clip(ones, 0, 1)
        return ones
    
    def generate_sine(freq, amp):

        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
