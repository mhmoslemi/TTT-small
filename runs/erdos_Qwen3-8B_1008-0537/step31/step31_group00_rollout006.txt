import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary(f, d):
        return np.clip(np.sin(2 * np.pi * f * np.arange(n_points)) > 0, 0, 1) * d
    
    def generate_triangle(f, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * (2 * t - 1) * np.abs(1 - 2 * np.abs(2 * f * t - 1))
        return np.clip(wave, 0, 1)
    
    def generate_sawtooth(f, d):
        t = np.linspace(0, 1, n_points)
        wave = 2 * d * (1 - np.abs(2 * f * t - 1))
        return np.clip(wave, 0, 1)
    
    guesses = [generate_binary(f, d) for f, d in [(3, 0.4), (4, 0.3), (5, 0.25)]]
    guesses += [generate_triangle(f, amp) for f, amp in [(2, 0.6), (3, 0.5), (4, 0.4)]]
    guesses += [generate_sawtooth(f, d) for f, d in [(2, 0.3), (3, 0.25), (4, 0.2)]]
    guesses.append(np.random.rand(n_points))
    guesses.append(h)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=70, maxiter=35,
