import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp, phase=0):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * freq * t + phase)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_step_pattern():
        h_step = np.zeros(n_points)
        h_step[:22] = 1.0
        h_step[22] = 0.5
        h_step[23:] = 0.0
        return h_step
    
    def generate_symmetric_pattern():
        t = np.linspace(0, 2, n_points)
        h_sym = np.zeros(n_points)
        for i in range(n_points):
            if t[i] <= 1:
                h_sym[i] = 1.0
            else:
                h_sym[i] = 0.0
        h_sym = (h_sym - np.mean(h_sym)) / np.std(h_sym) * 0.6 + 0.5
        return np.clip(h_sym, 0, 1)
    
    def generate_alternating_pattern():
        h_alt = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h_alt[i] = 1.0
            else:
                h_alt[i] = 0.0
        h_alt = (h_alt - np.mean(h_alt)) / np.std(h_alt) * 0.6 + 0.5
        return np.clip(h_alt, 0, 1)
    
    guesses = [h]
    guesses += [generate_sine(f, a) for f, a in [(3, 0.6), (4, 0.5), (5, 0.4), (2, 0.7)]]

    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,

        if res.success:

        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=40,

        if res.success:

            if res_slsqp.success:
