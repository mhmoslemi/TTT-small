    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_even_spaced(n_points, m):
        dx = 2.0 / n_points
        ideal_pos = np.linspace(0, 2, m)
        indices = np.round(ideal_pos / dx).astype(int)
        indices = np.unique(np.clip(indices, 0, n_points - 1))
        h = np.zeros(n_points)
        if len(indices) < m:
            extra = np.random.choice(n_points, size=m - len(indices), replace=False)
            indices = np.concatenate((indices, extra))
        h[indices] = 1
        return h
    
    guesses = [
        generate_even_spaced(n_points, 22),
        generate_even_spaced(n_points, 23),
        generate_binary_spaced(3), generate_binary_spaced(4),
        generate_binary_spaced(5), generate_binary_spaced(6),
        generate_binary_rand(22), generate_binary_rand(23),
        np.random.rand(n_points), h
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=25,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_neldermead(h_init):
        res = optimize.minimize(objective, h_init, method='Nelder-Mead', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xatol': 1e-8, 'fatol': 1e-8, 'maxiter': 30})
        return res.x if res.success else h_init
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_neldermead(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
