import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_even_spaced_guess(m):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % m == 0 or i % m == m//2:
                h[i] = 1.0
        return h
    
    def generate_clustered_guess(m):
        h = np.zeros(n_points)
        for i in range(n_points):
            if (i // (n_points // m)) % 2 == 0:
                h[i] = 1.0
        return h
    
    def generate_staggered_guess(m):
        h = np.zeros(n_points)
        spacing = n_points // m
        for i in range(m):
            h[i::spacing] = 1.0
        return h
    
    guesses = [
        generate_even_spaced_guess(5),
        generate_clustered_guess(3),
        generate_staggered_guess(5),
        generate_binary_rand(22),

        h
    ]
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 20:

        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
