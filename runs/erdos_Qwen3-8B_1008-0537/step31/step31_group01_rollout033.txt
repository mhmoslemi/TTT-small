        for i in range(n_points):
            if i % 2 == 0:
                pattern[i] = 1.0
            else:
                pattern[i] = 0.0
        sum_pattern = np.sum(pattern)
        adjustment = target_sum - sum_pattern
        if adjustment != 0:

        return pattern
    
    def generate_sine(freq, amp):

        if res.success and objective(res.x) < best_c5:

    for _ in range(3):

    def refine_with_bfgs(h_init):
