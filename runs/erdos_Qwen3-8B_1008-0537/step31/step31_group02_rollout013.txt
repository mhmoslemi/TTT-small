import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Improved initial guess generators
    def gen_spaced(spacing, max_ones):
        h = np.zeros(n_points)
        cnt = 0
        for i in range(n_points):
            if i % spacing == 0 or i % (spacing + 1) == 0:
                h[i] = 1
                cnt += 1
                if cnt > max_ones:
                    break
        return h
    
    def gen_rand(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h
    
    initial_guesses = [
        gen_spaced(3, n_points//2), gen_spaced(4, n_points//2),
        gen_spaced(5, n_points//2), gen_spaced(6, n_points//2),
        gen_rand(n_points//2 - 5), gen_rand(n_points//2),
        gen_rand(n_points//2 + 5), h
    ]
    
    # Initial optimization with SLSQP
    for guess in initial_guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential Evolution with enhanced parameters
    if time.time() - start_time < budget_s - 30:
        res_de = optimize.differential_evolution(
            objective, bounds, popsize=50, maxiter=80,
            mutation=(0.5, 0.9), recombination=0.8, tol=1e-9
        )
        if res_de.success:
            res_slsqp = optimize.minimize(
                objective, res_de.x, method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Refinement with hybrid methods
    def hybrid_refine(h_init):
        res = optimize.minimize(
            objective, h_init, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8}
        )
        return res.x if res.success else h_init
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = hybrid_refine(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
