import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def compute_c5(h_vals):
        max_c5 = 0.0
        for m in range(-n_points + 1, n_points):
            total = 0.0
            for i in range(n_points):
                j = i + m
                if 0 <= j < n_points:
                    total += h_vals[i] * (1.0 - h_vals[j])
            current_c5 = total * dx
            if current_c5 > max_c5:
                max_c5 = current_c5
        return max_c5
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_random_perturbation(h_base):
        return np.clip(h_base + 0.1 * np.random.randn(n_points), 0, 1)
    
    guesses = [h]
    guesses += [generate_sine(f, a) for f, a in [(3, 0.6), (4, 0.5), (5, 0.4)]]
    guesses += [generate_random_perturbation(h)]
    
    for guess in guesses:
        res = optimize.minimize(compute_c5, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and compute_c5(res.x) < best_c5:
            best_c5 = compute_c5(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(lambda x: compute_c5(x), bounds, popsize=50, maxiter=40,
                                             mutation=(0.8, 1.2), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(compute_c5, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and compute_c5(res_slsqp.x) < best_c5:
                best_c5 = compute_c5(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_bfgs(h_init):
