                               options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential evolution for global search
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=30,
                                             mutation=(0.8, 1.0), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Refinement with Nelder-Mead
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='Nelder-Mead', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xatol': 1e-9, 'fatol': 1e-9, 'maxiter': 30})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
