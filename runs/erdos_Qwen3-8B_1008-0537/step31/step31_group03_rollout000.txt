import scipy.optimize as optimize
import time

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_wave(freq, shape, duty=0.5):
        x = np.linspace(0, 1, n_points)
        if shape == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
            return np.clip((wave + 1) * duty / 2, 0, 1)
        elif shape == 'saw':
            return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif shape == 'tri':
            wave = 2 * np.abs(2 * freq * x - 1) - 1
            return np.clip(wave * duty + 0.5, 0, 1)
        else:
            return np.zeros(n_points)
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6]:
        guesses.append(gen_wave(freq, 'pulse', 0.5))
        guesses.append(gen_wave(freq, 'pulse', 0.3))
        guesses.append(gen_wave(freq, 'saw', 0.3))
    for shape in ['pulse', 'saw', 'tri']:
        guesses.append(gen_wave(3, shape))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else g
    
    for g in guesses:
