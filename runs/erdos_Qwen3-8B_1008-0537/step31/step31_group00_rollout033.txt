import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate diverse initial guesses
    def generate_binary(freq, duty):
        t = np.linspace(0, 1, n_points)
        return np.clip(np.sin(2 * np.pi * freq * t) > 0, 0, 1) * duty
    
    def generate_periodic(freq):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * t), 0, 1)
    
    def generate_triangular(freq):
        t = np.linspace(0, 1, n_points)
        wave = 2 * freq * t - 1
        wave = np.abs(1 - 2 * np.abs(wave))
        return np.clip(0.5 * wave, 0, 1)
    
    def generate_sawtooth(freq):
        t = np.linspace(0, 1, n_points)
        t = (t * freq) % 1
        return np.clip(1 - t, 0, 1)
    
    # Initial guesses
    guesses = [h.copy()]
    for f in [3, 4, 5]:
        guesses.append(generate_binary(f, 0.3))
        guesses.append(generate_periodic(f))
        guesses.append(generate_triangular(f))
        guesses.append(generate_sawtooth(f))
    
    # Evaluate initial guesses
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    # Differential evolution
    if time.time() - start_time < budget_s - 30:
        res_de = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=30,
                                              mutation=(0.7, 0.95), recombination=0.85, tol=1e-9)
        if res_de.success:

                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and res_slsqp.fun < best_c5:
