import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_spaced(spacing, n_ones):
        h = np.zeros(n_points)
        cnt = 0
        for i in range(n_points):
            if i % spacing == 0:
                h[i] = 1
                cnt += 1
                if cnt >= n_ones:
                    break
        return h
    
    def generate_binary_rand(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h
    
    initial_guesses = []
    # Generate spaced patterns with different spacings and counts
    for spacing in [2, 3, 4, 5, 6, 7, 8, 10]:
        n_ones = min(n_points // 2, (n_points + spacing - 1) // spacing)
        if n_ones > 0:
            initial_guesses.append(generate_binary_spaced(spacing, n_ones))
    # Add random patterns
    for ones in [n_points // 2, n_points // 2 - 1, n_points // 2 + 1]:
        initial_guesses.append(generate_binary_rand(ones))
    # Add the initial guess
    initial_guesses.append(h)
    # Add some histograms
    for i in range(5):
        initial_guesses.append(np.random.rand(n_points))
    
    for guess in initial_guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=30,

        if res.success:

                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:

                best_h = res_slsqp.x

        res = optimize.minimize(objective, best_h, method='Nelder-Mead', bounds=bounds,
