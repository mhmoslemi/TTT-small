        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbed_guess(base):
        perturb = np.random.normal(0, 0.1, n_points)
        perturb = np.clip(perturb, -0.5, 0.5)
        return np.clip(base + perturb, 0.0, 1.0)
    
    def objective_h(h):
        return objective(h)
    
    def constraint_sum_h(h):
        return constraint_sum(h)
    
    def generate_initial_guesses():
        guesses = [initial_guess]
        for _ in range(5):
            guess = np.random.uniform(0.0, 1.0, n_points)
            s = np.sum(guess)
            guess *= (target_sum / s)
            guess = np.clip(guess, 0.0, 1.0)
            s = np.sum(guess)
            guess += (target_sum - s) / n_points
            guesses.append(guess)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective_h,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum_h}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_h,
            bounds=bounds,
            popsize=20,
            maxiter=150,
            mutation=(0.5, 1.0),
            recombination=0.7,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective_h,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum_h}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective_h,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum_h}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
