import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_structured_guess():
        m = int(np.round(target_sum))
        positions = np.linspace(0, 2, m + 1)
        centers = (positions[:-1] + positions[1:]) / 2
        h = np.zeros(n_points)
        for i, center in enumerate(centers):
            idx = int(np.round(center / dx))
            if 0 <= idx < n_points:
                h[idx] = 1.0
        current_sum = np.sum(h)
        adjust = current_sum - target_sum
        if adjust > 1e-6:
            h[-1] -= adjust
        elif adjust < -1e-6:
            h[-1] += -adjust
        return h
    
    initial_guesses = [initial_guess]
    structured_guess = generate_structured_guess()
    initial_guesses.append(structured_guess)
    
    for guess in initial_guesses:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=30,
            maxiter=150,
            mutation=(0.5, 0.9),
            recombination=0.7,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
