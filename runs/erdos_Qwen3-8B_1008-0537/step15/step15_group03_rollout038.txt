import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess]
    for _ in range(10):
        random_guess = np.zeros(n_points)
        num_ones = int(round(target_sum))
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        random_guess[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            random_guess[idx] += remaining
        initial_guesses.append(random_guess)
    
    # Add even-spaced initial guess
    num_ones_even = int(target_sum)
    remaining_even = target_sum - num_ones_even
    positions = np.linspace(0, 2, num_ones_even + 1)[:-1]
    positions = positions / dx
    positions = np.floor(positions).astype(int)
    positions = np.unique(np.clip(positions, 0, n_points - 1))
    if len(positions) < num_ones_even:
        additional = num_ones_even - len(positions)
        new_pos = np.random.choice(n_points, size=additional, replace=False)
        positions = np.concatenate([positions, new_pos])
    even_guess = np.zeros(n_points)
    even_guess[positions[:num_ones_even]] = 1.0
    if remaining_even > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        even_guess[idx] += remaining_even
    initial_guesses.append(even_guess)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=200,
            mutation=(0.5, 0.8),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with COBYLA
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(

                x0=best_h,

                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
