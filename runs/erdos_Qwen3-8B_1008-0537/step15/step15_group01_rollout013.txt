import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    # Generate diverse initial guesses with structured patterns
    initial_guesses = [initial_guess]

        sym_pattern[indices] = 1.0
        sym_pattern[n_points - indices - 1] = 1.0
        sym_pattern /= 2.0  # Ensure sum is target_sum
        initial_guesses.append(sym_pattern)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None:
            c5 = objective(result)
            if c5 < best_c5:
                best_c5 = c5

    # Global optimization with DE
    try:

            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success and objective(result_de.x) < best_c5:
            best_c5 = objective(result_de.x)

    except:
