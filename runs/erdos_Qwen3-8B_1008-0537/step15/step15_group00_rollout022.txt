import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guesses(num_guesses=10):
        guesses = [initial_guess.copy()]
        # Add sinusoidal guess
        x = np.linspace(0, 2, n_points)
        sin_guess = 0.5 + 0.5 * np.sin(np.pi * x)
        guesses.append(sin_guess)
        # Add binary guess
        binary_guess = np.zeros(n_points)
        binary_guess[::2] = 1.0
        binary_guess = np.clip(binary_guess, 0, 1)
        guesses.append(binary_guess)
        # Add random guesses
        for _ in range(num_guesses - 3):
            h = np.zeros(n_points)
            num_ones = int(target_sum)
            idxs = np.argsort(np.random.rand(n_points))
            h[idxs[:num_ones]] = 1.0
            if target_sum - num_ones > 0:
                idx = np.random.choice(n_points, size=1)
                h[idx] += 1.0
            guesses.append(h)
        return guesses
    
    initial_guesses = generate_guesses(10)
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(

            tol=1e-9,

        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:

    while time.time() - start_time < budget_s - 40:

            result = optimize.minimize(
