import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbations(base, n_perturb=10):
        perturbations = []
        for _ in range(n_perturb):
            noise = np.random.normal(0, 0.05, len(base))
            perturbed = base + noise
            perturbed = np.clip(perturbed, 0.0, 1.0)
            if np.sum(perturbed) > 1e-8:
                scaling = target_sum / np.sum(perturbed)
                perturbed *= scaling
            perturbations.append(perturbed)
        return perturbations
    
    initial_guesses = [initial_guess] + generate_perturbations(initial_guess, 10)
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=500,
            mutation=(0.8, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 10:
