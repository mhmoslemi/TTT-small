import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        # Adjust for relevant shifts [-2, 2] by slicing the correlation
        relevant_indices = np.arange(len(corr))
        # Find indices where the shift is approximately within [-2, 2]
        # For n_points=45, dx=2/45, a shift of 2 corresponds to 2/(2/45) = 45 samples
        # So shifts in the range [-45, +45] would cover k in [-2, 2]
        # But we only need a subset. For simplicity, take a window around the center
        center = n_points - 1
        window_size = int(45)  # Adjust based on n_points
        start = max(0, center - window_size)
        end = min(len(corr), center + window_size)
        relevant_corr = corr[start:end]
        return np.max(relevant_corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess, maxiter=300, ftol=1e-8):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    # Generate structured initial guesses with even spacing
    def generate_guesses(num_guesses=10):
        guesses = [initial_guess.copy()]
        for _ in range(num_guesses-1):
            # Evenly spaced points with some variation
            h = np.zeros(n_points)
            total_ones = int(target_sum + 0.5)  # Round to nearest integer
            positions = np.linspace(0, 2, n_points)
            idxs = np.argsort(np.random.rand(n_points))
            selected = idxs[:total_ones]
            h[selected] = 1.0
            # Apply a small perturbation to avoid symmetry
            h += np.random.uniform(-0.1, 0.1, size=n_points)
            h = np.clip(h, 0, 1)
            guesses.append(h)
        return guesses
    
    initial_guesses = generate_guesses(10)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with aggressive early stopping
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=40,
            maxiter=200,
            mutation=(0.5, 0.8),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with BFGS and tighter tolerances
    while time.time() - start_time < budget_s - 40:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
