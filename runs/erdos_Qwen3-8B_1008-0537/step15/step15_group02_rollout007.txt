        return np.sum(h_values) - target_sum

        random_guess[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            random_guess[idx] += remaining
        return random_guess

    for _ in range(8):
        initial_guesses.append(generate_random_guess())

            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],

        )
