import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess, maxiter=300, ftol=1e-10):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    def generate_guesses(num_guesses=15):
        guesses = [initial_guess.copy()]
        # Structured guess: spaced 1s
        structured_h = np.zeros(n_points)
        step = int(n_points / (int(target_sum) + 1))
        for i in range(int(target_sum)):
            structured_h[i * step] = 1.0
        remaining = target_sum - int(target_sum)
        if remaining > 0:
            structured_h[-1] += remaining
        guesses.append(structured_h)
        # Random guesses
        for _ in range(num_guesses - 2):
            h = np.zeros(n_points)
            num_ones = int(target_sum)
            idxs = np.argsort(np.random.rand(n_points))
            h[idxs[:num_ones]] = 1.0
            if target_sum - num_ones > 0:
                idx = np.random.choice(n_points, size=1)
                h[idx] += target_sum - num_ones
            guesses.append(h)
        return guesses
    
    initial_guesses = generate_guesses(15)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=300,
            mutation=(0.6, 0.95),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement
    while time.time() - start_time < budget_s - 30:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
