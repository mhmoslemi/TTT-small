        result_bh = basinhopping(
            fun=objective,
            x0=best_h,
            niter=50,
            T=1.0,
            stepsize=0.5,
            minimizer_kwargs={
                'method': 'SLSQP',
                'bounds': bounds,
                'constraints': [{'type': 'eq', 'fun': constraint_sum}],
                'options': {'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            }
        )
        if result_bh.success and objective(result_bh.x) < best_c5:
            best_c5 = objective(result_bh.x)
            best_h = result_bh.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
