import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def obj(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def cons(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_initial_guesses(num=10):
        guesses = []
        # Add random guesses
        for _ in range(num//2):
            vals = np.zeros(n_points)
            num_ones = int(target_sum) + np.random.randint(0, 2)
            idx = np.random.choice(n_points, size=num_ones, replace=False)
            vals[idx] = 1.0
            remaining = target_sum - num_ones
            if remaining > 0:
                idx2 = np.random.choice(n_points, size=1, replace=False)
                vals[idx2] += remaining
            guesses.append(vals)
        # Add evenly spaced ones
        for _ in range(num//2):
            vals = np.zeros(n_points)
            cnt = int(np.ceil(target_sum))
            spacing = n_points / cnt
            idxs = np.round(np.arange(0, cnt) * spacing).astype(int)
            idxs = np.unique(idxs)
            vals[idxs] = 1.0
            remaining = target_sum - len(idxs)
            if remaining > 0:
                idx = np.random.choice(n_points, size=1, replace=False)
                vals[idx] += remaining
            guesses.append(vals)
        return guesses
    
    initial_guesses = gen_initial_guesses(10)
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=obj,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': cons}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            if obj(result.x) < best_c5:
                best_c5 = obj(result.x)
                best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=obj,
            bounds=bounds,
            popsize=30,

            mutation=(0.5, 1.0),

            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=obj,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,

                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}

            if result_slsqp.success and obj(result_slsqp.x) < best_c5:
                best_c5 = obj(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        result = optimize.minimize(

            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
