import scipy.optimize as optimize
import time

    dx = 2.0 / n_points
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_structured_guess():
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        positions = np.linspace(0, 2, n_points)
        mid_idx = len(positions) // 2
        spread = np.sort(np.random.choice(n_points, size=num_ones, replace=False))
        guess = np.zeros(n_points)
        for i in spread:
            guess[i] = 1.0
        if remaining > 0:
            idx = np.argsort(abs(positions - 1.0))
            guess[idx[0]] += remaining
        return guess
    
    guess = generate_structured_guess()
    result = optimize_guess(guess)
    if result is not None and objective(result) < best_c5:
        best_c5 = objective(result)
        best_h = result
    
    # Differential evolution with reduced iterations
    try:
        result_de = optimize.differential_evolution(
            fun=objective,

        )
