        structured_guess[positions] = 1.0
        sum_structured = np.sum(structured_guess)
        if abs(sum_structured - target_sum) > 1e-6:
            idx = np.random.choice(n_points, size=1, replace=False)
            structured_guess[idx] += target_sum - sum_structured
        initial_guesses.append(structured_guess)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Enhanced differential evolution with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=500,
            mutation=(0.7, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with BFGS and adjusted options
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
