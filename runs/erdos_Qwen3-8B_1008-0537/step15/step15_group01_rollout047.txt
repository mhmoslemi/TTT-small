import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]
    
    # Symmetric initial guess
    symmetric_guess = np.zeros(n_points)
    symmetric_guess[22] = 0.5  # Midpoint
    for i in range(11):
        symmetric_guess[i] = 1.0
        symmetric_guess[n_points - 1 - i] = 1.0
    for i in range(11, 22):
        symmetric_guess[i] = 0.0
        symmetric_guess[n_points - 1 - i] = 0.0
    initial_guesses.append(symmetric_guess)
    
    # Random initial guesses with proper fractional handling
    for _ in range(10):
        rand_h = np.zeros(n_points)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        rand_h[indices] = 1.0
        # Choose an index not in indices
        available = [i for i in range(n_points) if i not in indices]
        if available:
            idx = np.random.choice(available, size=1)
            rand_h[idx] += remaining
        initial_guesses.append(rand_h)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Enhanced differential evolution with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=200,
            mutation=(0.5, 0.7),
            recombination=0.7,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with BFGS and adjusted parameters
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
