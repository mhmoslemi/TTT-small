import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_regular_guess(m):
        guess = np.zeros(n_points)
        step = n_points / m
        indices = np.arange(0, n_points, step).astype(int)
        guess[indices] = 1.0
        remaining = target_sum - m
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            guess[idx] += remaining
        return guess
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8}
            )
            return result.x if result.success else None
        except:
            return None
    
    # Generate initial guesses
    initial_guesses = [initial_guess]

    for _ in range(5):
        guess = np.zeros(n_points)

        remaining = target_sum - m
        indices = np.random.choice(n_points, size=m, replace=False)
        guess[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            guess[idx] += remaining
        initial_guesses.append(guess)
    
    for guess in initial_guesses:

    # Differential evolution with reduced population

            tol=1e-9,
            vectorized=True
        )
        if result_de.success:

    # Local refinement with COBYLA
