import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    # Generate more diverse initial guesses
    initial_guesses = [initial_guess]
    
    # Add structured evenly spaced guess
    def create_structured_guess():
        m = int(np.floor(target_sum))
        remaining = target_sum - m
        positions = np.linspace(0, 2, m + 1)[:-1]
        indices = np.round(positions * (n_points / 2)).astype(int)
        indices = np.unique(np.clip(indices, 0, n_points - 1))
        guess = np.zeros(n_points)
        for i in indices:
            if i < n_points:
                guess[i] = 1.0
        if remaining > 0:
            available = np.where(guess == 0)[0]
            if available.size > 0:
                idx = available[0]
                guess[idx] += remaining
        return guess
    
    initial_guesses.append(create_structured_guess())
    
    # Generate additional random guesses
    for _ in range(8):
        random_guess = np.zeros(n_points)
        num_ones = int(np.floor(target_sum))
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        random_guess[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            random_guess[idx] += remaining
        initial_guesses.append(random_guess)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=300,
            mutation=(0.9, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with BFGS
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
