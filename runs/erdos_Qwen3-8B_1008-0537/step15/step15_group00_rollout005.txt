import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guesses(num_guesses=10):
        guesses = [initial_guess.copy()]
        for _ in range(num_guesses-1):
            h = np.zeros(n_points)
            num_ones = int(target_sum)
            idxs = np.argsort(np.random.rand(n_points))
            h[idxs[:num_ones]] = 1.0
            if target_sum - num_ones > 0:
                idx = np.random.choice(n_points, size=1)
                h[idx] += 1.0
            guesses.append(h)
            # Add structured guess: evenly spaced
            if len(guesses) < num_guesses:
                m = min(int(target_sum), n_points // 2)
                idxs = np.sort(np.random.choice(n_points, size=m, replace=False))
                h_struct = np.zeros(n_points)
                h_struct[idxs] = 1.0
                guesses.append(h_struct)
        return guesses
    
    initial_guesses = generate_guesses(15)
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    # Differential evolution with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=40,
            maxiter=200,
            mutation=(0.5, 0.7),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
