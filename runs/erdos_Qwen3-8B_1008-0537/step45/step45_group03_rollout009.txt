import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

        x = np.linspace(0, 1, n_points)
        return np.clip(amp * np.sin(2 * np.pi * freq * x) + 0.5, 0, 1)
    
    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    initial_guesses = [h, gen_sine(2, 0.5), gen_sine(3, 0.5), gen_sine(4, 0.5), gen_pulse(3, 0.4)]
    
    def slsqp_guess(g):

                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else g
    
    for g in initial_guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    if time.time() - start_time > budget_s - 10:
        return best_h, best_c5, n_points
    
    res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=100,
                                         mutation=(0.6, 0.95), recombination=0.9, tol=1e-9)
    h_opt = slsqp_guess(res.x)
    c = obj(h_opt)
    if c < best_c5:
        best_h, best_c5 = h_opt, c
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
