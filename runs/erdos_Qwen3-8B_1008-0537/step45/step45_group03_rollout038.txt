import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    def gen_rectangles(n_rect, spacing):
        x = np.linspace(0, 1, n_points)
        h_rect = np.zeros_like(x)
        idx = np.argsort(np.abs(x - 0.5))
        rect_len = 1.0 / n_rect
        for i in range(n_rect):
            start = 0.5 - (rect_len / 2) + i * (rect_len + spacing)
            end = start + rect_len
            if start < 0 or end > 1:
                break
            h_rect[idx[:int(n_points * (end - start))]] = 1
        return h_rect
    
    guesses = [h]
    for freq in [3, 4, 5, 6, 7]:
        guesses.append(gen_pulses(freq, 0.4))
    for duty in [0.3, 0.4, 0.5]:
        guesses.append(gen_saw(5, duty))
    for duty in [0.3, 0.4, 0.5]:
        guesses.append(gen_triangle(5, duty))
    for n_rect in [5, 6, 7]:
        guesses.append(gen_rectangles(n_rect, 0.3))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],

        c = obj(h_opt)
