import time

        return np.clip(pulse, 0, 1)
    
    def gen_saw(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        sawtooth = 1 - np.abs(2 * freq * x - 1 + phase) * duty
        return np.clip(sawtooth, 0, 1)
    
    def gen_triangle(freq, duty, phase):
