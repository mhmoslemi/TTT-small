def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Perturb the initial guess slightly
    h += np.random.normal(0, 0.05, size=h.shape)
    h = np.clip(h, 0., 1.)
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def refine(r):
        # Use L-BFGS-B for refinement
        res = optimize.minimize(obj, r, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-12})
        return res.x if res.success else r
    
    # Run differential evolution with different parameters
    for _ in range(3):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=30, maxiter=50,
                                             mutation=(0.7, 0.95), recombination=0.9, tol=1e-9)
        h_opt = refine(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    # Run more refinement steps
    for _ in range(5):

    return best_h, best_c5, n_points

import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Perturb initial guess
    noise = np.random.normal(0, 0.05, size=h.shape)
    h += noise
    h = np.clip(h, 0., 1.)

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def refine(r):
        res = optimize.minimize(obj, r, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-12})
        return res.x if res.success else r

    # Run differential evolution with tuned parameters
    for _ in range(3):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=30, maxiter=50,
                                             mutation=(0.7, 0.95), recombination=0.9, tol=1e-9)
        h_opt = refine(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Intensive refinement steps
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
