import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def refine(r, maxiter=200, ftol=1e-9, method='SLSQP'):
        res = optimize.minimize(obj, r, method=method, bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': maxiter, 'ftol': ftol})
        return res.x if res.success else r

    for _ in range(3):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=30, maxiter=40,
                                             mutation=(0.8, 1.0), recombination=0.8, tol=1e-9)
        h_opt = refine(res.x, maxiter=200, ftol=1e-9, method='trust-constr')
        c = obj(h_opt)
