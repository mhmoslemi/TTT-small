import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_sine(freq, amp, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(amp * np.sin(2 * np.pi * freq * x + phase) + 0.5, 0, 1)

    guesses = [h]
    for freq in range(2, 14):
        for phase in [0, np.pi/2, np.pi, 3*np.pi/2]:
            guesses.append(gen_pulse(freq, 0.5, phase))
    for freq in range(2, 8):
        for phase in [0, np.pi/2, np.pi, 3*np.pi/2]:
            guesses.append(gen_sine(freq, 0.5, phase))
    for freq in range(2, 7):
        for duty in [0.3, 0.5, 0.7]:
            guesses.append(gen_pulse(freq, duty, 0))
    for duty in [0.2, 0.4, 0.6]:
        guesses.append(gen_pulse(4, duty, 0))
    for duty in [0.2, 0.4, 0.6]:
        guesses.append(gen_pulse(5, duty, 0))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 30, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(5):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=40,
                                             mutation=(0.5, 0.8), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

        best_h = slsqp_guess(best_h)
