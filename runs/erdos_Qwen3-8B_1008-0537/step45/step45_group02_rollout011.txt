import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def refine(r, max_iter=200, ftol=1e-12):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': max_iter, 'ftol': ftol, 'disp': False})
        return res.x if res.success else r
    
    # First, refine with a high-precision local optimizer
    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        ref = refine(best_h, max_iter=500, ftol=1e-12)
        c = obj(ref)
        if c < best_c5:
            best_h, best_c5 = ref, c
    
    # Second, perform a more aggressive refinement with SLSQP
    for _ in range(3):
