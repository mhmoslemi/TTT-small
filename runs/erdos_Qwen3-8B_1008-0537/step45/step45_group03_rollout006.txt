import time

    def obj(h_vals):

        return np.max(corr) * dx
    
    def cons(h_vals):

    best_c5 = obj(best_h)
    start_time = time.time()
    
    def slsqp_guess(g, maxiter=300, ftol=1e-9):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': maxiter, 'ftol': ftol})
        return res.x if res.success else g
    
    best_h = slsqp_guess(best_h, maxiter=500, ftol=1e-10)
    best_c5 = obj(best_h)
