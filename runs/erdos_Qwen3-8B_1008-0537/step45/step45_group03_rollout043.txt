    for _ in range(8):
        if time.time() - start_time > budget_s - 30:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=150, maxiter=80,
                                             mutation=(0.5, 0.8), recombination=0.9, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(15):
        if time.time() - start_time > budget_s - 15:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
