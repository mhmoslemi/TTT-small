import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulse(freq, duty, phase):
        x = np.linspace(0, 2, n_points)
        sig = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((sig + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty, phase):
        x = np.linspace(0, 2, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1 + phase) * duty, 0, 1)
    
    guesses = [h]
    for freq in [2, 3, 4, 5]:
        for duty in [0.4, 0.5]:
            phase = np.random.uniform()
            guesses.append(gen_pulse(freq, duty, phase))
            guesses.append(gen_saw(freq, duty, phase))
    guesses.append(np.random.uniform(0, 1, n_points))
    guesses.append(np.zeros(n_points) + 0.5)
    guesses.append(np.zeros(n_points) + 0.3)
    guesses.append(np.zeros(n_points) + 0.7)
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-10})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 10:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=50,
                                             mutation=(0.8, 1.0), recombination=0.9, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 5:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
