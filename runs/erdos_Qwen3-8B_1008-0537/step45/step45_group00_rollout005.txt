import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    guesses = [h]
    for freq in range(2, 15):
        guesses.append(gen_pulses(freq, 0.5))
    for duty in np.linspace(0.2, 0.8, 6):
        guesses.append(gen_saw(3, duty))
    for duty in np.linspace(0.2, 0.8, 6):
        guesses.append(gen_triangle(3, duty))

    def random_guess():
        base = np.random.rand(n_points)
        base /= np.sum(base)
        base *= target_sum
        return np.clip(base, 0, 1)

    guesses.extend([random_guess() for _ in range(4)])
    guesses.append(np.concatenate([np.ones(20), np.zeros(n_points - 20)]))
    guesses.append(np.concatenate([np.zeros(20), np.ones(n_points - 20)]))

    def slsqp_guess(g):

        return res.x if res.success else g

    for g in guesses:
