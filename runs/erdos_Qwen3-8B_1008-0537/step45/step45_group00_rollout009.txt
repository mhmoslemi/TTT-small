import scipy.optimize as optimize
import time

    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    guesses = [h]
    freqs = [2, 3, 4, 5, 6, 7, 8, 9, 10]
    for freq in freqs:
        for duty in np.linspace(0.2, 0.8, 5):
            guesses.append(gen_pulses(freq, duty))
    for freq in freqs:
        for duty in np.linspace(0.2, 0.8, 5):
            guesses.append(gen_saw(freq, duty))
    for freq in freqs:
        for duty in np.linspace(0.2, 0.8, 5):
            guesses.append(gen_triangle(freq, duty))
    for _ in range(3):
        random_guess = np.random.rand(n_points)
        random_guess = (target_sum / np.sum(random_guess)) * random_guess
        guesses.append(random_guess)
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-11, 'disp': False})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)

    def de_optimize():
