import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_periodic(freq, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.sin(2 * np.pi * freq * x + phase) + 1) * 0.5, 0, 1)
    
    def gen_anti_periodic(freq, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip((1 - np.sin(2 * np.pi * freq * x + phase)) * 0.5, 0, 1)
    
    initial_guesses = [
        h,
        gen_periodic(3, np.random.uniform()),
        gen_anti_periodic(3, np.random.uniform()),
        gen_periodic(5, np.random.uniform()),
        gen_anti_periodic(5, np.random.uniform()),
        np.random.uniform(0, 1, n_points),
        np.zeros(n_points) + 0.5,
        np.zeros(n_points) + 0.3,
        np.zeros(n_points) + 0.7
    ]
    
    def slsqp_guess(g, maxiter=200):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x if res.success else g
    
    for g in initial_guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=30,
                                             mutation=(0.8, 1.2), recombination=0.85, tol=1e-10)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
