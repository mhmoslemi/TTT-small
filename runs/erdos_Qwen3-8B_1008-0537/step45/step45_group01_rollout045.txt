import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.sin(2 * np.pi * freq * x + phase) + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1 + phase) * duty, 0, 1)
    
    def gen_triangle(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1 + phase) * duty
    
    guesses = [h]
    for freq in [1, 2, 3, 4, 5]:
        guesses.append(gen_pulses(freq, 0.5, np.random.uniform()))
    for duty in [0.3, 0.4, 0.5]:
        guesses.append(gen_saw(2, duty, np.random.uniform()))
    for duty in [0.3, 0.4, 0.5]:
        guesses.append(gen_triangle(2, duty, np.random.uniform()))
    guesses.append(np.random.uniform(0, 1, n_points))
    guesses.append(np.zeros(n_points) + 0.5)
    guesses.append(np.zeros(n_points) + 0.3)
    guesses.append(np.zeros(n_points) + 0.7)
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-12})
        return res.x if res.success else g
    
    for g in guesses:

        c = obj(h_opt)
