            for i in range(n):
                j = i + m
                if 0 <= j < n:
                    current_sum += h_vals[i] * (1 - h_vals[j])
            overlap = current_sum * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    start_time = time.time()
    
    def gen_sine(freq, amp):
        x = np.linspace(0, 1, n_points)
        return np.clip(amp * np.sin(2 * np.pi * freq * x) + 0.5, 0, 1)
    
    def gen_pulses(freq, duty):

        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
