def run(seed=42, budget_s=100, **kwargs):

            shifted = np.zeros_like(h_vals)

                x = i * dx
                shifted_x = x + k * dx
                if 0 <= shifted_x <= 2:
                    j = int(shifted_x / dx)
                    if 0 <= j < n_points:
                        shifted[i] = h_vals[j]
                    else:
                        shifted[i] = 0
                else:
                    shifted[i] = 0
            overlap = np.dot(h_vals, 1 - shifted) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    def cons(h_vals):

        return 1 - np.abs(2 * freq * x - 1 + phase) * duty
    
    def gen_saw(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1 + phase) * duty, 0, 1)
