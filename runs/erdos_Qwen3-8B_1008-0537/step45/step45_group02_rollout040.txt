import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def refine(r):
        res = optimize.minimize(
            obj, r, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': cons}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8}
        )
        return res.x if res.success else r
    
    def evolve():
        res = optimize.differential_evolution(
            obj, bounds, popsize=30, maxiter=50, mutation=(0.5, 1.0),
            recombination=0.9, tol=1e-9, mutation_func='best/1'
        )
        return res.x
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = evolve()
        h_opt = refine(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
