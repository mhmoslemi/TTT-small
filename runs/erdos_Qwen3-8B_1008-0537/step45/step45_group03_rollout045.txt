        K = n_points // 2 + 1
        positions = np.zeros(n_points)
        spacing = n_points / K
        for i in range(K):
            idx = int(round(i * spacing))
            if idx >= n_points:
                idx = n_points - 1
            positions[idx] = 1
        dx = 2.0 / n_points
        current = K * dx
        delta = current - 1.0
        if delta > 0:
            idx = np.where(positions == 1)[0][0]
            positions[idx] = 1 - delta / dx
        return positions
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]

    best_c5 = obj(best_h)
    start_time = time.time()
    
    def slsqp_guess(g):

                               constraints=[{'type': 'eq', 'fun': cons}],
