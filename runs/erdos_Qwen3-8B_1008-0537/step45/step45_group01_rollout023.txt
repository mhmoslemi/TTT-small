import scipy.optimize as optimize

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

        return np.clip((np.sin(2 * np.pi * freq * x + phase) + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1 + phase) * duty, 0, 1)
    
    def gen_triangle(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1 + phase) * duty
    
    def gen_blocks(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(np.kaiser(x.size, 5)[np.newaxis, :] * (np.sin(2 * np.pi * freq * x + phase) > 0), 0, 1)
    
    guesses = [h]

    for freq in [1, 2, 3, 4, 5]:
