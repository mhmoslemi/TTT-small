import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-12})
        return res.x if res.success else r
    
    # Initial population-based search with精细化
    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=40, maxiter=40,
                                             mutation=(0.6, 0.95), recombination=0.8, tol=1e-9)
        h_opt = refine(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    # Intensive refinement with multiple passes
    for _ in range(6):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    # Final greedious check
    h_greedy = best_h.copy()
    for i in range(n_points):
        for j in range(n_points):
            if h_greedy[i] < 1 and h_greedy[j] < 1:
                h_greedy[i] = min(1, h_greedy[i] + 0.01)
                h_greedy[j] = min(1, h_greedy[j] + 0.01)
                if np.sum(h_greedy) > target_sum:
                    h_greedy[i] -= 0.01
                    h_greedy[j] -= 0.01
                elif np.sum(h_greedy) < target_sum:
                    h_greedy[i] += 0.01
                    h_greedy[j] += 0.01
                break
    
    return best_h, best_c5, n_points
