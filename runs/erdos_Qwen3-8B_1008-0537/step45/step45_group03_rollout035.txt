import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_sine(freq, amp):
        x = np.linspace(0, 1, n_points)
        return np.clip(amp * np.sin(2 * np.pi * freq * x) + 0.5, 0, 1)
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    def gen_rect(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave > 0) * duty, 0, 1)
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6]:
        for duty in [0.3, 0.5, 0.7]:
            guesses.append(gen_saw(freq, duty))
            guesses.append(gen_triangle(freq, duty))
            guesses.append(gen_pulses(freq, duty))
            guesses.append(gen_rect(freq, duty))
    for freq in [1, 2, 3, 4, 5]:
        guesses.append(gen_sine(freq, 0.6))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-10})
        return res.x if res.success else g
    
    for g in guesses:

        c = obj(h_opt)
