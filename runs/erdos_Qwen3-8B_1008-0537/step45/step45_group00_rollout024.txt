    def obj(h_vals):

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_random():
        r = np.random.rand(n_points)
        r = (target_sum / np.sum(r)) * r
        return r
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x) + 1
        return np.clip((wave * duty) / 2, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    for duty in np.linspace(0.2, 0.8, 5):
        guesses.append(gen_triangle(3, duty))
    
    def slsqp_guess(g):
