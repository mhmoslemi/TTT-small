import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.sin(2 * np.pi * freq * x + phase) + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1 + phase) * duty, 0, 1)
    
    def gen_triangle(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1 + phase) * duty
    
    guesses = [h]
    for freq in range(1, 15):
        guesses.append(gen_pulses(freq, 0.5, np.random.uniform()))
    for duty in np.linspace(0.2, 0.8, 5):
        guesses.append(gen_saw(5, duty, np.random.uniform()))
    for duty in np.linspace(0.2, 0.8, 5):
        guesses.append(gen_triangle(5, duty, np.random.uniform()))
    guesses.append(np.random.uniform(0, 1, n_points))
    guesses.append(np.zeros(n_points) + 0.5)
    guesses.append(np.zeros(n_points) + 0.3)
    guesses.append(np.zeros(n_points) + 0.7)
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 400, 'ftol': 1e-12})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=100,

        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,

        return res.x if res.success else r

    return best_h, best_c5, n_points
