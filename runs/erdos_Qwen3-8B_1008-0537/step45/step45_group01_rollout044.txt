import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def slsqp_opt(h_opt):
        res = optimize.minimize(obj, h_opt, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-12})
        return res.x if res.success else h_opt

    # Initial SLSQP refinement
    best_h = slsqp_opt(best_h)
    best_c5 = obj(best_h)

    # Differential evolution
    if time.time() - start_time < budget_s - 30:
        res_de = optimize.differential_evolution(obj, bounds, popsize=30, maxiter=50, mutation=(0.5, 1.0), recombination=0.9, tol=1e-9)
        best_h = slsqp_opt(res_de.x)
        best_c5 = obj(best_h)

    # Perturb and re-optimize
    if time.time() - start_time < budget_s - 10:
        h_pert = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h_pert = slsqp_opt(h_pert)
        c_pert = obj(h_pert)
        if c_pert < best_c5:
            best_h, best_c5 = h_pert, c_pert

    # Final refinement
    for _ in range(2):
        if time.time() - start_time > budget_s - 5:
            break
        best_h = slsqp_opt(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
