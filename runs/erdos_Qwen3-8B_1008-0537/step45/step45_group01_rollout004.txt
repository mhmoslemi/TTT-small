import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, phase, noise=0.05):
        x = np.linspace(0, 1, n_points)
        pulse = (np.sin(2 * np.pi * freq * x + phase) + 1) * 0.5
        return np.clip(pulse + np.random.normal(0, noise, n_points), 0, 1)
    
    def gen_saw(freq, phase):
        x = np.linspace(0, 1, n_points)
        ret = 1 - np.abs(2 * freq * x - 1 + phase) * 0.5
        return np.clip(ret, 0, 1)
    
    def gen_triangle(freq, phase):
        x = np.linspace(0, 1, n_points)
        ret = 1 - np.abs(2 * freq * x - 1 + phase)
        return np.clip(ret, 0, 1)
    
    guesses = [h]
    guesses.append(gen_pulses(3, np.random.uniform()))
    guesses.append(gen_saw(3, np.random.uniform()))
    guesses.append(gen_triangle(2, np.random.uniform()))
    guesses.append(np.random.uniform(0, 1, n_points) * 0.5 + 0.5)
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-12})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    if time.time() - start_time < budget_s - 20:
        res = optimize.differential_evolution(obj, bounds, popsize=40, maxiter=30,
                                             mutation=(0.7, 0.9), recombination=0.85, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 10:

        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
