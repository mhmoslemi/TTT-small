        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

    def generate_binary(freq, duty, phase=0):

        return np.clip(np.sin(2 * np.pi * freq * x + phase) > 0, 0, 1) * duty
    
    def generate_sparse(density):
