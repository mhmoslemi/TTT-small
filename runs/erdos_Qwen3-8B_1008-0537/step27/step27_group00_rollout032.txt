import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty
    
    def generate_random():
        return np.random.rand(n_points) * 0.5 + 0.25
    
    guesses = [h]
    for p in [3, 4, 5, 6, 7, 8]:
        guesses.append(generate_periodic(p))
    for f, d in [(3, 0.2), (4, 0.3), (5, 0.4), (6, 0.5), (7, 0.2), (8, 0.3)]:
        guesses.append(generate_binary(f, d))
    for _ in range(3):
        guesses.append(generate_random())
    
    for guess in guesses:
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=50, mutation=(0.5, 0.9), recombination=0.8, tol=1e-9, args=(), callback=None, poltergeist=False, seed=None, strategy='best1bin', init='random', atol=1e-9, xtol=1e-9, disp=False, vectorized=True, mutation_strength=0.5, mutation_type='exponential', crossover_prob=0.9, crossover_type='binomial', mutation_min=0, mutation_max=1, recombination_min=0, recombination_max=1, mutation_func=None, crossover_func=None, selection_func=None, termination_func=None, 
        initial_population=guess)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:

        if time.time() - start_time >= budget_s - 20:

        res = optimize.minimize(objective, best_h, method='Powell', bounds=bounds,

            best_h = res.x
