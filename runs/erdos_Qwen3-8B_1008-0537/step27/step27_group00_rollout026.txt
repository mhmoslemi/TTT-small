import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0.0, 0, 1) * duty
    
    def generate_sparse():
        idx = np.random.choice(n_points, size=5, replace=False)
        return 0.0 * np.ones(n_points)
    
    def generate_ripple():
        return np.clip(np.sin(2 * np.pi * np.arange(n_points) / (n_points / 2.5)) ** 2, 0, 1)
    
    guesses = [h]
    for p in [3, 4, 5, 6, 7, 8]:
        guesses.append(generate_periodic(p))
    for f, d in [(3, 0.2), (4, 0.3), (5, 0.4), (6, 0.1), (7, 0.25)]:
        guesses.append(generate_binary(f, d))
    guesses.append(generate_sparse())
    guesses.append(generate_ripple())
    guesses.append(np.random.rand(n_points))
    guesses.append(0.6 * np.ones(n_points))
    guesses.append(0.4 * np.ones(n_points))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-10})
