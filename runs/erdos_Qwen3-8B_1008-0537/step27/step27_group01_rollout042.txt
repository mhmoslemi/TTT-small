import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Diverse structured initial guesses
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)

        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * mod, 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty
    
    def generate_sawtooth(freq):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points)), 0, 1)
    
    def generate_triangular(freq):
        return np.clip(0.5 + 0.5 * np.abs(np.sin(2 * np.pi * freq * np.arange(n_points))), 0, 1)
    
    initial_guesses = [initial_guess]

        initial_guesses.append(generate_periodic(period))

        initial_guesses.append(generate_binary(freq, duty))
    
    # Optimize initial guesses
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
