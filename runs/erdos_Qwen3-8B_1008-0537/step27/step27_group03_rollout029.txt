import scipy.optimize as optimize

def run(seed=42, budget_s=100, **kwargs):

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def generate_periodic(period, phase=0.0, amplitude=1.0):
        return np.clip(0.5 + 0.5 * amplitude * np.sin(2 * np.pi * np.arange(n_points) / period + phase), 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip((((np.arange(n_points) * freq) % 1) < duty) * 1.0, 0, 1)
    
    def generate_modulated(period, mod, phase=0.0):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase) * mod, 0, 1)
    
    def generate_random_walk(step=0.2):
        h = np.zeros(n_points)
        h[0] = np.random.uniform(0, 1)
        for i in range(1, n_points):
            h[i] = np.clip(h[i-1] + np.random.uniform(-step, step), 0, 1)
        return h
    
    initial_guesses = [initial_guess]
    for period in range(4, 9):

    initial_guesses.append(generate_random_walk())
    
    for guess in initial_guesses:
        result = optimize.minimize(
