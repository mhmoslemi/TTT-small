import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Diversity in initial guesses
    def generate_binary(duty, length):
        return np.clip(np.sin(2 * np.pi * np.arange(n_points) / length) > duty, 0, 1)
    
    def generate_sparse():
        return np.array([0.5 if i % 5 == 0 else 0 for i in range(n_points)])
    
    def generate_irregular():
        h = np.zeros(n_points)
        for _ in range(n_points // 2):
            idx = np.random.randint(n_points)
            h[idx] = 1
        return h
    
    def generate_sawtooth():
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / 10), 0, 1)
    
    initial_guesses = [initial_guess, generate_sawtooth(), generate_sparse(), generate_irregular()]
    
    # Expand with more binary patterns
    for freq in range(1, 4):
        initial_guesses.append(generate_binary(0.5, freq * 5))
    
    # Optimization: multiple steps
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            if objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
    
    # Differential evolution with adaptive strategy
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=25,
            maxiter=80,
            mutation=(0.8, 1.2),
            recombination=0.9,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Final dense refinement
    while time.time() - start_time < budget_s - 30:
        try:
