import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    except:

        initial_guess[::2] = 1.0
        initial_guess = np.clip(initial_guess, 0, 1)
    
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_modulated(period, gain):
        base = np.sin(2 * np.pi * np.arange(n_points) / period)
        return np.clip(0.5 + 0.5 * base * gain, 0, 1)
    
    # Validate initial_guess
    if np.sum(initial_guess) * dx < target_sum - 1e-8 or np.sum(initial_guess) * dx > target_sum + 1e-8:

        initial_guess = np.clip(initial_guess, 0, 1)

    # First refine initial guess
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
    )
    if result_slsqp.success:

            best_h = result_slsqp.x

    initial_guesses.append(generate_modulated(10, 0.6))

    # Optimize diverse initial guesses
    for guess in initial_guesses:
        result = optimize.minimize(

            recombination=0.8,

            vectorized=True
        )
        if result_de.success:
