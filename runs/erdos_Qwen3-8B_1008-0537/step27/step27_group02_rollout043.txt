import time

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period, amplify=0.5):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * amplify, 0, 1)
    
    def generate_binary(freq, duty, offset=0):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points) + offset) > 0, 0, 1) * duty
    
    initial_guesses = [initial_guess]

    for guess in initial_guesses:
