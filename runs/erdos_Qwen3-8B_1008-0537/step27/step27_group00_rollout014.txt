import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period, phase):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase), 0, 1)
    
    def generate_binary(freq, duty, phase):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points) + phase) > 0, 0, 1) * duty
    
    def generate_random_sparse(density):
        return np.clip(np.random.rand(n_points) < density, 0, 1)
    
    guesses = []
    for p in [3, 4, 5, 6, 7, 8, 9]:
        guesses.append(generate_periodic(p, np.random.rand()))
    for f, d in [(3, 0.2), (4, 0.3), (5, 0.4), (6, 0.5), (7, 0.1), (9, 0.2)]:
        guesses.append(generate_binary(f, d, np.random.rand()))
    guesses.append(generate_random_sparse(0.3))
    guesses.append(generate_random_sparse(0.5))
    guesses.append(generate_random_sparse(0.7))
    guesses.append(np.random.rand(n_points))
    guesses.append(generate_periodic(4, 0.5))
    guesses.append(generate_binary(5, 0.3, 0.5))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=25,
                                             mutation=(0.8, 1.0), recombination=0.6, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='Powell', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 20})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
