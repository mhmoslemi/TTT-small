import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.6 + 0.4 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_modulated(period, mod):
        return np.clip(0.6 + 0.4 * np.sin(2 * np.pi * np.arange(n_points) / period) * mod, 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points)) > 0) * duty, 0, 1)
    
    def generate_shifted(pulse, shift):
        return np.roll(pulse, shift)
    
    def generate_two_pulse(period1, period2, shift):
        pulse1 = generate_periodic(period1)
        pulse2 = generate_periodic(period2)
        return np.clip((pulse1 + np.roll(pulse2, shift)) / 2, 0, 1)
    
    def generate_ternary(freq, duty1, duty2):
        base = generate_binary(freq, duty1)
        return np.clip(base + (np.sin(2 * np.pi * freq * np.arange(n_points)) > 0) * duty2, 0, 1)
    
    initial_guesses = [initial_guess]
    
    for period in [3, 4, 5, 6, 7, 8]:
        initial_guesses.append(generate_periodic(period))
    
    for period, mod in [(4, 0.7), (5, 0.6), (6, 0.5), (7, 0.4)]:
        initial_guesses.append(generate_modulated(period, mod))
    
    for freq, duty in [(5, 0.2), (6, 0.3), (7, 0.4), (8, 0.5)]:
        initial_guesses.append(generate_binary(freq, duty))
    
    for freq, duty1, duty2 in [(5, 0.3, 0.5), (6, 0.4, 0.5), (7, 0.2, 0.7)]:
        initial_guesses.append(generate_ternary(freq, duty1, duty2))
    
    for shift in [n_points//4, n_points//2]:
        initial_guesses.append(generate_shifted(initial_guess, shift))
    
    for period1, period2, shift in [(4, 6, 10), (5, 7, 15), (6, 8, 20)]:
        initial_guesses.append(generate_two_pulse(period1, period2, shift))
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=100,
            mutation=(0.8, 0.95),
            recombination=0.85,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 30:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 30, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
