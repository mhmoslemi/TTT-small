import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_periodic(period, amp, off):
        return np.clip(off + amp * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)

    def generate_modulated(period, mod):
        return np.clip(0.6 + 0.4 * np.sin(2 * np.pi * np.arange(n_points) / period) * mod, 0, 1)

    def generate_binary(freq, duty):
        return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points)) > 0) * duty, 0, 1)

    initial_guesses = [initial_guess.copy()]

    for period in [3,4,5,6,7]:
        initial_guesses.append(generate_periodic(period, 0.6, 0.4))
    
    for period, mod in [(4,0.7),(5,0.8),(6,0.6),(7,0.5)]:
        initial_guesses.append(generate_modulated(period, mod))
    
    for freq, duty in [(5,0.2),(6,0.4),(7,0.3),(5,0.3)]:
        initial_guesses.append(generate_binary(freq, duty))
    
    for _ in range(4):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        scale = target_sum / perturbed_sum
        perturbed *= scale
        initial_guesses.append(perturbed)

    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)

    try:

            popsize=60,

            vectorized=True
        )
        if result_de.success:

                options={'maxiter': 150, 'ftol': 1e-10}

            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
