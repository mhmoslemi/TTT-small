    np.random.seed(seed)
    # Use initial_h_values if available, otherwise generate a structured binary guess
    try:
        n_points = len(initial_h_values)
    except NameError:
        # Default to 45 points as a safe assumption
        n_points = 45
    
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def generate_binary_initial(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        interval_count = 8
        interval_len = 1.0 / interval_count
        for i in range(interval_count):
            start = i * 0.5 + 0.25
            end = start + interval_len
            start_idx = int(np.floor(start / dx))
            end_idx = int(np.ceil(end / dx))
            h[start_idx:end_idx] = 1.0
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (target_sum / sum_h)
        return h

    # Use the generated initial guess or the provided one
    h = generate_binary_initial(n_points) if 'initial_h_values' not in locals() else np.array(initial_h_values, dtype=float)

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # First iteration: SLSQP with initial guess
    res = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': constraint_sum}],
                           options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
    if res.success and objective(res.x) < best_c5:
        best_c5 = objective(res.x)
        best_h = res.x

    # Differential evolution for global search
    for _ in range(10):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=30,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x

    # Refinement with Powell method
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='Powell', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 20})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    return best_h, best_c5, n_points
