import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except:
        n_points = 60  # Increased from 45 for better resolution
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float) if 'initial_h_values' in kwargs else np.random.rand(n_points)
    initial_guess = np.clip(initial_guess * target_sum / np.sum(initial_guess), 0, 1)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_oscillator(freq, amp, phase=0):
        return np.clip(0.5 + amp * np.sin(2 * np.pi * freq * np.arange(n_points) / n_points + phase), 0, 1)
    
    def generate_sparse(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points) / n_points) > 0, 0, 1) * duty
    
    def generate_random_binary():
        return np.clip(np.random.rand(n_points) / np.sum(np.random.rand(n_points)) * target_sum, 0, 1)
    
    initial_guesses = [initial_guess]
    for freq in [3, 4, 5, 6, 7, 8, 9, 10]:
        initial_guesses.append(generate_oscillator(freq, 0.8))
        initial_guesses.append(generate_oscillator(freq, 0.6, np.pi/2))
    for freq, duty in [(3, 0.2), (4, 0.3), (5, 0.4), (6, 0.5), (7, 0.6), (8, 0.15)]:
        initial_guesses.append(generate_sparse(freq, duty))
    initial_guesses.append(generate_random_binary())
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=80,
            maxiter=100,

        )
