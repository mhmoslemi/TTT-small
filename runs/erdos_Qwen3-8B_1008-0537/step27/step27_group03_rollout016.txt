import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(period, wave_type):
        if wave_type == 'tri':
            t = np.arange(n_points) / period
            return np.clip(0.5 - np.abs(t - 0.5), 0, 1)
        elif wave_type == 'saw':
            t = np.arange(n_points) / period
            return np.clip(0.2 + 0.8 * (t - np.floor(t)), 0, 1)
        elif wave_type == 'sin':
            return np.clip(0.6 + 0.4 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
        elif wave_type == 'mod':
            return np.clip(0.6 + 0.4 * np.sin(2 * np.pi * np.arange(n_points) / period) * 0.7, 0, 1)
        elif wave_type == 'bin':
            freq, duty = 6, 0.2
            return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points)) > 0) * duty, 0, 1)
        else:
            return np.zeros(n_points)
    
    initial_guesses = [initial_guess]
    for period in [4, 5, 6, 7, 8]:
        initial_guesses.append(generate_wave(period, 'sin'))
        initial_guesses.append(generate_wave(period, 'mod'))
        initial_guesses.append(generate_wave(period, 'tri'))
        initial_guesses.append(generate_wave(period, 'saw'))
    for wave_type in ['sin', 'mod', 'tri', 'saw']:
        initial_guesses.append(generate_wave(4, wave_type))
        initial_guesses.append(generate_wave(5, wave_type))
        initial_guesses.append(generate_wave(6, wave_type))
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
