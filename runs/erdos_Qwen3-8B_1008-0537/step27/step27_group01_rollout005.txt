import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period, phase=0.0):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase), 0, 1)
    
    def generate_modulated(period, mod, phase=0.0):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase) * mod, 0, 1)
    
    def generate_binary(freq, duty, phase=0.0):
        return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points) + phase) > 0) * duty, 0, 1)
    
    def generate_sparse():
        return np.zeros(n_points)
    
    initial_guesses = [initial_guess]
    
    for period in [3, 4, 5, 6, 7]:
        initial_guesses.append(generate_periodic(period))
    for period, mod in [(3, 0.8), (4, 0.7), (5, 0.6), (6, 0.5)]:
        initial_guesses.append(generate_modulated(period, mod))
    for freq, duty in [(3, 0.3), (4, 0.25), (5, 0.2), (6, 0.15)]:
        initial_guesses.append(generate_binary(freq, duty))
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success:
            if objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,

            popsize=35,
            maxiter=120,

            recombination=0.95,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,

                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 30:
