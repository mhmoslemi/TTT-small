        return np.sum(h_values) - target_sum

        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / periods), 0, 1)

        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / periods) * mods, 0, 1)
