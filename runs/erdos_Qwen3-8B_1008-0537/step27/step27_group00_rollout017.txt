import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in locals() else 45
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float) if 'initial_h_values' in locals() else np.zeros(n_points)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_three_intervals():
        t = 1.0 / 3.0
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(3):
            start = i * (2.0 / 3.0) - t / 2
            end = start + t
            start_idx = int(start / dx) 
            end_idx = int(end / dx) + 1
            if start_idx < 0:
                start_idx = 0
            if end_idx > n_points:
                end_idx = n_points
            h[start_idx:end_idx] = 1
        return h
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty
    
    guesses = [
        np.random.rand(n_points),
        generate_three_intervals(),
        generate_binary(5, 0.5),
        generate_binary(4, 0.4),
        generate_binary(6, 0.3),
        np.zeros(n_points)
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=30,

        if res.success:

                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:

            best_h = res.x
