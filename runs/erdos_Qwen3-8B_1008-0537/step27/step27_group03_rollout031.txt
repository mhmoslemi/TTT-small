        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def generate_periodic(period, phase=0):
        return np.clip(0.6 + 0.4 * np.sin(2 * np.pi * np.arange(n_points) / period + phase), 0, 1)
    
    def generate_modulated(period, phase=0, mod=0.7):
        return np.clip(0.6 + 0.4 * np.sin(2 * np.pi * np.arange(n_points) / period + phase) * mod, 0, 1)

        return np.clip(tri, 0, 1)
