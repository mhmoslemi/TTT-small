import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    try:
        n_points = len(initial_h_values)
    except:
        n_points = 45  # fallback
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty
    
    def generate_sparse():
        idx = np.random.choice(n_points, size=4, replace=False)
        return np.zeros(n_points) + 0.3
    
    def generate_filtered(period, mod):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * mod, 0, 1)
    
    initial_guesses = [initial_guess]
    for freq, duty in [(5, 0.2), (6, 0.25), (7, 0.15), (4, 0.35)]:
        initial_guesses.append(generate_binary(freq, duty))
    for period, mod in [(5, 0.8), (6, 0.7), (7, 0.6), (8, 0.5), (9, 0.4)]:
        initial_guesses.append(generate_filtered(period, mod))
    initial_guesses.append(generate_sparse())
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if result.success:
            if objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 20:
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=80,
                maxiter=100,
                mutation=(0.7, 0.95),

                tol=1e-9,
                vectorized=True
            )
            if result_de.success:
                result_slsqp = optimize.minimize(

                    options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
