import scipy.optimize as optimize
import time

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_random_pattern():
        return np.random.uniform(0, 1, n_points)

        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_tapered(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * np.exp(-0.1 * np.arange(n_points)), 0, 1)
