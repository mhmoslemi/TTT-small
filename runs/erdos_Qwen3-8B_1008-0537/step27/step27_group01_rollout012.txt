import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_modulated(period, mod):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * mod, 0, 1)
    
    def generate_binary_pattern(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty
    
    def generate_perturbed(g):
        return np.clip(g + 0.1 * np.random.randn(n_points), 0, 1)
    
    initial_guesses = [initial_guess]
    for period in [3,4,5,6,7,8,9]:
        initial_guesses.append(generate_periodic(period))
    for period, mod in [(3,0.9),(4,0.8),(5,0.7),(6,0.6),(7,0.5),(8,0.4),(9,0.3)]:
        initial_guesses.append(generate_modulated(period, mod))
    for freq, duty in [(5,0.2),(4,0.3),(3,0.4),(6,0.25),(7,0.2),(2,0.6),(8,0.15)]:
        initial_guesses.append(generate_binary_pattern(freq, duty))
    for g in initial_guesses[:5]:
        initial_guesses.append(generate_perturbed(g))
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_h = result.x
            best_c5 = objective(result.x)
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=40,
            maxiter=150,
            mutation=(0.6, 0.9),
            recombination=0.9,
            tol=1e-9,
            vectorized=True
        )
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result_slsqp.success and objective(result_slsqp.x) < best_c5:
            best_h = result_slsqp.x
            best_c5 = objective(result_slsqp.x)
    except:
        pass
    
    while time.time() - start_time < budget_s - 20:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='Powell',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 50}
            )
            if result.success and objective(result.x) < best_c5:
                best_h = result.x
                best_c5 = objective(result.x)
        except:
            break
    
    return best_h, best_c5, n_points
