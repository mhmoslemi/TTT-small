def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Run SLSQP with the initial guess
    res_slsqp = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                 options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10})
    if res_slsqp.success and objective(res_slsqp.x) < best_c5:
        best_c5 = objective(res_slsqp.x)
        best_h = res_slsqp.x
    
    # Run differential evolution on best_h
    # Run DE for multiple iterations
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res_de = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=50,
                                                 mutation=(0.5, 1.0), recombination=0.8, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                          constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                          options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Run Powell on best_h
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        res_powell = optimize.minimize(objective, best_h, method='Powell', bounds=bounds,
                                       constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                       options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 30})
        if res_powell.success and objective(res_powell.x) < best_c5:
            best_c5 = objective(res_powell.x)
            best_h = res_powell.x
    
    return best_h, best_c5, n_points

import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Run SLSQP with the initial guess
    res_slsqp = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                 options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10})
    if res_slsqp.success and objective(res_slsqp.x) < best_c5:
        best_c5 = objective(res_slsqp.x)
        best_h = res_slsqp.x
    
    # Run differential evolution on best_h
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res_de = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=50,
                                                 mutation=(0.5, 1.0), recombination=0.8, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                          constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                          options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Run Powell on best_h
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        res_powell = optimize.minimize(objective, best_h, method='Powell', bounds=bounds,
                                       constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                       options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 30})
        if res_powell.success and objective(res_powell.x) < best_c5:
            best_c5 = objective(res_powell.x)
            best_h = res_powell.x
    
    return best_h, best_c5, n_points
