import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period, phase):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase), 0, 1)
    
    def generate_binary(freq, duty, phase):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points) + phase) > 0, 0, 1) * duty
    
    def generate_sparse(density):
        return np.clip(np.random.rand(n_points) < density, 0, 1)
    
    guesses = []
    for p in [4, 5, 6, 7, 8, 9, 10]:
        for phase in np.linspace(0, 2*np.pi, 3):
            guesses.append(generate_periodic(p, phase))
    for f, d in [(6, 0.2), (5, 0.3), (4, 0.4), (7, 0.1), (8, 0.15), (9, 0.25)]:
        for phase in np.linspace(0, 2*np.pi, 3):
            guesses.append(generate_binary(f, d, phase))
    for d in [0.2, 0.35, 0.5]:
        guesses.append(generate_sparse(d))
    guesses.append(np.random.rand(n_points))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
