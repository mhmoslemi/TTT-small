import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Add perturbed initial guess
    perturbed = initial_guess + 0.1 * np.random.randn(n_points)
    perturbed = np.clip(perturbed, 0, 1)
    initial_guesses = [initial_guess, perturbed]
    
    # Basinhopping for global search
    try:
        result = optimize.basinhopping(
            fun=objective,
            x0=initial_guess,
            niter=30,
            T=1.0,
            stepsize=0.1,
            minimizer_kwargs={
                'method': 'SLSQP',
                'bounds': bounds,
                'constraints': [{'type': 'eq', 'fun': constraint_sum}],
                'options': {'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
            }
        )
        if result.success:
            if objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
    except:
        pass
    
    # Differential evolution with tighter parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=20,
            maxiter=50,
            mutation=(0.6, 0.8),
            recombination=0.9,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Final Powell refinement with adaptive stopping
    while time.time() - start_time < budget_s - 30:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='Powell',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 50}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
