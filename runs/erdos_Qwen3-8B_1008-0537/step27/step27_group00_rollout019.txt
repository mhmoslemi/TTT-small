import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty
    
    def generate_sparse():
        idx = np.random.choice(n_points, size=3, replace=False)
        return np.zeros(n_points) if np.random.rand() < 0.5 else np.zeros(n_points)
    
    def generate_pulses(num_pulses, spacing):
        pulses = np.zeros(n_points)
        for i in range(num_pulses):
            pos = i * spacing
            if pos + 0.5 < n_points:
                pulses[round(pos):round(pos + 0.5)] = 1.0
        return pulses
    
    def generate_initial(n_points):
        return np.random.rand(n_points)
    
    guesses = []
    for p in [4, 5, 6, 7, 9, 10]:
        guesses.append(generate_periodic(p))
    for f, d in [(6, 0.2), (5, 0.3), (4, 0.4), (7, 0.1), (8, 0.15), (9, 0.25)]:
        guesses.append(generate_binary(f, d))
    guesses.append(generate_sparse())
    guesses.append(generate_pulses(4, 1.2))
    guesses.append(generate_pulses(5, 1.5))
    guesses.append(generate_pulses(6, 1.8))
    guesses.append(generate_initial(n_points))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(15):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=40,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(4):
        if time.time() - start_time >= budget_s - 10:

        if res.success and objective(res.x) < best_c5:
