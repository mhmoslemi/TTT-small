import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except:
        n_points = 45  # fallback
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Diverse structured initial guesses
    def gen_sine(freq, amp):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points) / n_points) * amp, 0, 1)
    def gen_square(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points) / n_points) > 0, 0, 1) * duty
    def gen_sparse(density):
        return np.clip(np.random.rand(n_points) < density, 0, 1)
    def gen_rect(freq):
        return np.clip(np.mod(np.arange(n_points), freq) < (freq / 2), 0, 1)
    
    initial_guesses = [initial_guess]
    
    for freq in [5, 6, 7, 8, 9]:
        initial_guesses.append(gen_sine(freq, 0.8))
        initial_guesses.append(gen_sine(freq, 0.6))

        initial_guesses.append(gen_square(freq, 0.45))
    for d in [0.2, 0.3, 0.4]:
        initial_guesses.append(gen_sparse(d))
    for f in [4, 5, 6, 7]:
        initial_guesses.append(gen_rect(f))
    initial_guesses.append(gen_rect(3))

    # Optimization
    for guess in initial_guesses:
        result = optimize.minimize(

    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
