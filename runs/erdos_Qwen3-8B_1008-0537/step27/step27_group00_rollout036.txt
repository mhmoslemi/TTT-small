import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_vals):

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = objective(best_h)

        wave = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase)
        return np.clip(wave, 0, 1)
    
    def generate_binary(freq, duty, phase=0.0):
        wave = np.sin(2 * np.pi * freq * np.arange(n_points) + phase)
        wave = np.clip(wave > 0, 0, 1) * duty
        return wave
    
    def generate_randomized():
