import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_base():
        phase = np.random.uniform(0, 2)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * (np.arange(n_points) - phase) / n_points), 0, 1)
    
    def generate_lowfreq():
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / 5), 0, 1)
    
    def generate_highfreq():
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / 10), 0, 1)
    
    initial_guesses = [h.copy(), generate_base(), generate_lowfreq(), generate_highfreq()]
    
    for guess in initial_guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds, 
                               constraints=[{'type': 'eq', 'fun': constraint_sum}], 
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time < budget_s - 20:
        res_de = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=20, 
                                               mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
