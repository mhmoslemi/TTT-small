import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        x = np.arange(n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * x / period), 0, 1)
    
    def generate_modulated(period, mod):
        x = np.arange(n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * x / period) * mod, 0, 1)
    
    def generate_binary_pattern(freq, duty):
        x = np.arange(n_points)
        return np.clip(np.sin(2 * np.pi * freq * x) > 0, 0, 1) * duty
    
    def generate_random_initial_guess():
        h = np.random.uniform(0, 1, n_points)
        scaling = (target_sum) / np.sum(h)
        h = h * scaling
        return np.clip(h, 0, 1)
    
    initial_guesses = [initial_guess]
    for period in [3, 4, 5, 6, 7, 8]:
        initial_guesses.append(generate_periodic(period))
    for period, mod in [(3, 0.7), (4, 0.7), (5, 0.6), (6, 0.5), (7, 0.4), (8, 0.3)]:
        initial_guesses.append(generate_modulated(period, mod))
    for freq, duty in [(3, 0.3), (4, 0.2), (5, 0.4), (6, 0.25), (7, 0.2), (8, 0.5)]:
        initial_guesses.append(generate_binary_pattern(freq, duty))
    for _ in range(3):
        initial_guesses.append(generate_random_initial_guess())
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
