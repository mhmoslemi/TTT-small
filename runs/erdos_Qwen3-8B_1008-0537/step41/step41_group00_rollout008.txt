import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(freq, amp, wave_type):
        t = np.linspace(0, 1, n_points) * 2
        if wave_type == 'sine':
            wave = amp * np.sin(2 * np.pi * freq * t)
        elif wave_type == 'triangle':
            wave = amp * np.abs(2 * (t % 2 - 1))
        elif wave_type == 'square':
            wave = amp * np.sign(np.sin(2 * np.pi * freq * t))
        else:
            wave = amp * np.cos(freq * t)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    guesses = [
        generate_wave(3, 0.4, 'sine'), generate_wave(4, 0.35, 'sine'),
        generate_wave(5, 0.3, 'sine'), generate_wave(3, 0.3, 'triangle'),
        generate_wave(4, 0.25, 'triangle'), generate_wave(5, 0.22, 'triangle'),
        generate_wave(3, 0.3, 'square'), generate_wave(4, 0.25, 'square'),
        generate_wave(5, 0.22, 'square'), np.random.rand(n_points),
        h.copy(), np.flipud(h), np.random.rand(n_points) * 0.5 + 0.5
    ]
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 15:
            break
        res_de = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=30,
                                             mutation=(0.6, 0.95), recombination=0.8, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 60, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
        return res.x if res.success else h_init
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 5:
            break
        best_h = refine_with_slsqp(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
