import scipy.optimize as optimize
import time

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_rippled(freq, amplitude):
        t = np.linspace(0, 2, n_points)
        wave = amplitude * np.sin(2 * np.pi * freq * (t - 1))
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_square(freq, phase):
        t = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * (t - 1))
        wave = np.where(wave >= 0, 1, -1)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_bipolar(freq, amp):
        t = np.linspace(0, 2, n_points) * 2
        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    guesses = [generate_rippled(3, 0.4), generate_rippled(4, 0.35), generate_rippled(5, 0.3)]
    guesses += [generate_square(1, 0.1), generate_square(2, 0.2), generate_square(3, 0.3)]

    guesses += [np.random.rand(n_points), h.copy()]
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,

        if res.success and objective(res.x) < best_c5:

            best_h = res.x
    
    for _ in range(3):
