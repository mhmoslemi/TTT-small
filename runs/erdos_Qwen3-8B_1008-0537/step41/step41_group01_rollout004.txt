import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

    def generate_sweep():
        t = np.linspace(0, 1, n_points)
        wave = np.zeros_like(t)

            wave += np.sin(2 * np.pi * i * t) * (0.1 / i)
