import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_block(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.25 + 0.75 * np.abs(wave), 0, 1)
    
    def generate_square(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sign(np.sin(2 * np.pi * freq * t))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_random_perturbation(h_base):
        return np.clip(h_base + 0.1 * np.random.randn(n_points), 0, 1)
    
    def generate_triangular(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * (2 * np.abs(np.sin(2 * np.pi * freq * t)) - 1)
        return np.clip(np.abs(wave), 0, 1)
    
    guesses = [h]
    for f in range(2, 8):
        guesses += [generate_sine(f, 0.6), generate_block(f, 0.5), generate_square(f, 0.7), generate_triangular(f, 0.6)]
    guesses += [generate_random_perturbation(h)]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:

            best_h = res.x

        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=30,
