import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_rippled(freq, amplitude):
        t = np.linspace(0, 1, n_points) * 2
        wave = amplitude * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_bipolar(freq, amp):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    # Add structured binary guesses
    def generate_binary_guess():
        binary = np.zeros(n_points)
        binary[:n_points//2] = 1.0
        return binary
    
    def generate_regular_guess():
        step = n_points // 2
        regular = np.zeros(n_points)
        for i in range(step):
            regular[i::step] = 1.0
        return regular
    
    guesses = [
        generate_rippled(3, 0.4), generate_rippled(4, 0.35), generate_rippled(5, 0.3),
        generate_bipolar(3, 0.3), generate_bipolar(4, 0.25), generate_bipolar(5, 0.22),
        np.random.rand(n_points), h.copy(), generate_binary_guess(), generate_regular_guess()
    ]
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):

                best_h = res_slsqp.x

        return res.x if res.success else h_init
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 5:
