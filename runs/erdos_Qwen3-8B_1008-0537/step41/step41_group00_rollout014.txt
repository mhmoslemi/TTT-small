import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_square(freq, amp):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * np.sign(np.sin(2 * np.pi * freq * t))
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_rect(freq, amp):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * np.abs(np.sin(2 * np.pi * freq * t))
        return np.clip(0.5 + 0.5 * (1 - wave), 0, 1)
    
    def generate_random():
        return np.random.rand(n_points)
    
    def generate_initial_spaced(freq):
        t = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * (1 - wave), 0, 1)
    
    guesses = [generate_rect(3, 0.3), generate_rect(2, 0.4), generate_rect(1, 0.5),
               generate_square(4, 0.2), generate_square(3, 0.3), generate_square(2, 0.4),
               generate_initial_spaced(4), generate_initial_spaced(5), generate_initial_spaced(1),
               generate_random(), h.copy()]
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,

                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
