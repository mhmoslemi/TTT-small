import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_random(h_base, std=0.05):
        noise = np.random.randn(n_points) * std
        return np.clip(h_base + noise, 0, 1)
    
    def generate_sine(freq, amp, phase):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * freq * t + phase)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    guesses = [h]
    guesses.append(generate_sine(3, 0.6, 0))
    guesses.append(generate_sine(4, 0.5, np.pi/2))
    guesses.append(generate_sine(5, 0.4, np.pi))
    guesses.append(generate_random(h))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        result = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=30,
                                             mutation=(0.7, 1.3), recombination=0.8, tol=1e-9)
        if result.success:
            res_slsqp = optimize.minimize(objective, result.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-11, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 15:

        res = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'gtol': 1e-9, 'maxiter': 60, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
