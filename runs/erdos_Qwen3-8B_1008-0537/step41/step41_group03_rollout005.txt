import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

            wave = np.abs(2 * np.pi * freq * t - np.pi)

            wave = 1 - 2 * wave

            wave = (2 * np.pi * freq * t - np.pi) % (2 * np.pi)
            wave /= (2 * np.pi)
            wave = np.clip(wave, 0, 1)
        return np.clip(wave, 0, 1)
