import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_ramped(freq, amp, phase):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * np.sin(2 * np.pi * freq * t + phase)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_sawtooth(freq, amp):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * (2 * (t % 1) - 1)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_random_perturbation(h_val, scale=0.05):
        return np.clip(h_val + np.random.normal(0, scale, n_points), 0, 1)
    
    guesses = []
    for freq in [2, 3, 4, 5]:
        for amp in [0.3, 0.4, 0.5]:
            for phase in [0, np.pi/2, np.pi]:
                guesses.append(generate_ramped(freq, amp, phase))
    for freq in [1, 2, 3]:
        guesses.append(generate_sawtooth(freq, 0.3))
    guesses += [generate_random_perturbation(h), h.copy(), np.random.rand(n_points)]
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):

        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=50,
