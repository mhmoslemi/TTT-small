import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)

        t = np.linspace(0, 1, n_points) * 2
        wave1 = amp1 * np.sin(2 * np.pi * freq1 * t)
        wave2 = amp2 * np.sin(2 * np.pi * freq2 * t)
        return np.clip(0.5 + 0.5 * (1 + wave1 + wave2), 0, 1)
    
    def generate_gapped(pulse_width, gap_width, duty_cycle):
        t = np.linspace(0, 1, n_points) * 2
        indices = np.where((t % (pulse_width + gap_width)) < pulse_width)
        return np.clip(0.5 + 0.5 * (1 + (indices.astype(float) / n_points)), 0, 1)
    
    guesses = [np.random.rand(n_points), h.copy()]
    for f1 in [2, 3, 4, 5, 6]:
        for f2 in [f1 + 1, f1 - 1, f1 * 2, f1 // 2]:
            for amp1 in [0.3, 0.4, 0.5]:

    for pw in [0.25, 0.3, 0.4]:
        for gw in [0.4, 0.5, 0.6]:

                guesses.append(generate_gapped(pw, gw, dc))
    
    for guess in guesses:

        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
