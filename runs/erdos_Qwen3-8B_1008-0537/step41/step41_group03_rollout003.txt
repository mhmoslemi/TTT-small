import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(freq, duty, waveform_type='square'):
        t = np.linspace(0, 1, n_points)
        if waveform_type == 'square':
            wave = 0.5 * (np.sign(np.sin(2 * np.pi * freq * t)) + 1)
            wave = 0.5 + 0.5 * wave * duty
        elif waveform_type == 'triangle':
            wave = 2 * np.abs(2 * np.pi * freq * t - np.pi) / np.pi
            wave = 1 - wave
        elif waveform_type == 'sawtooth':
            wave = 1 - (2 * np.pi * freq * t) % 1
        wave = np.clip(wave, 0, 1)
        return wave
    
    def perturb(h_base, scale=0.1):
        return np.clip(h_base + scale * np.random.randn(n_points), 0, 1)
    
    guesses = [h]
    for freq in [1, 2, 3, 4, 5, 6]:
        guesses.append(generate_wave(freq, 0.5, 'square'))
        guesses.append(generate_wave(freq, 0.3, 'square'))
        guesses.append(generate_wave(freq, 0.7, 'square'))
    for freq in [1, 2, 3, 4, 5]:
        guesses.append(generate_wave(freq, 0.5, 'triangle'))
        guesses.append(generate_wave(freq, 0.3, 'triangle'))
        guesses.append(generate_wave(freq, 0.7, 'triangle'))
        guesses.append(generate_wave(freq, 0.5, 'sawtooth'))
        guesses.append(generate_wave(freq, 0.3, 'sawtooth'))
        guesses.append(generate_wave(freq, 0.7, 'sawtooth'))
    guesses.append(perturb(h))
    guesses.append(perturb(h, 0.15))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=40,
                                             mutation=(0.8, 1.2), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_bfgs(h_init):
