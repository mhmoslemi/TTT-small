import scipy.optimize as optimize
import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

        t = np.linspace(0, 1, n_points)
        wave = 0.5 * (np.sign(np.sin(2 * np.pi * freq * t)) + 1)
        wave = 0.5 + 0.5 * wave * duty
        return np.clip(wave, 0, 1)
    
    def generate_tri(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.abs(2 * np.pi * freq * t - np.pi)
        wave /= np.pi
        wave = 1 - 2 * wave
        return np.clip(wave, 0, 1)
    
    def generate_sawtooth(freq):
        t = np.linspace(0, 1, n_points)
        wave = 2 * np.pi * freq * t - np.pi

        wave = 1 - wave
        return np.clip(wave, 0, 1)
