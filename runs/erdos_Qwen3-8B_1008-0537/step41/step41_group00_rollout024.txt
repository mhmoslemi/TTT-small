import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_rippled(freq, amp, phase):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * np.sin(2 * np.pi * freq * t + phase)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_bipolar(freq, amp, phase):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * np.sin(2 * np.pi * freq * t + phase)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_sawtooth(freq, amp, phase):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * (2 * (t % 1) - 1) + phase
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_triangle(freq, amp, phase):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * np.abs(2 * (t % 1) - 1) + phase
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    guesses = [
        generate_rippled(3, 0.4, 0.0),
        generate_rippled(4, 0.35, np.pi/2),
        generate_rippled(5, 0.3, np.pi),

        generate_bipolar(4, 0.25, np.pi/3),
        generate_bipolar(5, 0.22, np.pi/6),

        generate_triangle(2, 0.4, 0.0),
        np.random.rand(n_points),
        h.copy()
    ]
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:

        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
