import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_block(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.25 + 0.75 * np.abs(wave), 0, 1)
    
    def generate_triangle(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.abs(2 * np.pi * freq * t - np.pi)
        return np.clip(0.5 - 0.5 * wave / np.pi, 0, 1)
    
    def generate_square(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sign(np.sin(2 * np.pi * freq * t))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_random_perturbation(h_base):
        return np.clip(h_base + 0.05 * np.random.randn(n_points), 0, 1)
    
    guesses = [h]
    for f in range(2, 6):
        guesses += [
            generate_sine(f, 0.6), generate_block(f, 0.5),
            generate_triangle(f, 0.7), generate_square(f, 0.5)
        ]
    guesses += [generate_random_perturbation(h)]
    
    for guess in guesses:
        try:
            res = optimize.minimize(
                objective, guess, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        except:
            pass
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        try:
            res = optimize.differential_evolution(
                objective, bounds, popsize=50, maxiter=40,
                mutation=(0.8, 1.2), recombination=0.8, tol=1e-9
            )
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
        except:
            pass
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
