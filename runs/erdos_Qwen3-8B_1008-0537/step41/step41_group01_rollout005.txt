import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp, phase):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * freq * t + phase)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_square(freq, amp, phase):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sign(np.sin(2 * np.pi * freq * t + phase))
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_triangle(freq, amp, phase):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.abs(np.sin(2 * np.pi * freq * t + phase))
        return np.clip(0.5 - 0.5 * wave, 0, 1)
    
    guesses = [h]
    guesses += [generate_sine(f, a, p) for f, a, p in [(2, 0.6, 0), (3, 0.5, np.pi/2), (4, 0.4, np.pi)]]
    guesses += [generate_square(f, a, p) for f, a, p in [(2, 0.6, 0), (3, 0.5, np.pi/2), (4, 0.4, np.pi)]]
    guesses += [generate_triangle(f, a, p) for f, a, p in [(2, 0.6, 0), (3, 0.5, np.pi/2), (4, 0.4, np.pi)]]
    guesses += [np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:

    for _ in range(2):

        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=50,

                best_h = res_slsqp.x
    
    def refine_with_bfgs(h_init):
