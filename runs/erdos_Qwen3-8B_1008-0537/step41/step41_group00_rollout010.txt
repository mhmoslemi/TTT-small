        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = objective(best_h)

        t = np.linspace(0, 1, n_points) * 2

        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_tri(f, a, offset=0.5):
        t = np.linspace(0, 1, n_points) * 2
        wave = a * (2 * np.abs(2 * np.pi * f * t - np.pi) - np.pi) / np.pi

        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
