import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(freq, shape='square', duty=0.5):
        t = np.linspace(0, 1, n_points)
        if shape == 'square':
            wave = 0.5 * (np.sign(np.sin(2 * np.pi * freq * t)) + 1)
            wave = 0.5 + 0.5 * wave * duty
        elif shape == 'triangle':
            wave = np.abs(2 * np.pi * freq * t - np.pi)
            wave /= np.pi
            wave = 1 - 2 * wave
        elif shape == 'sawtooth':
            wave = (2 * np.pi * freq * t - np.pi) / np.pi
            wave = np.clip(wave, -1, 1)
            wave = (wave + 1) / 2
        wave = np.clip(wave, 0, 1)
        return wave
    
    guesses = [h]
    for f in [1, 2, 3, 4, 5, 0.5, 1.5, 2.5, 3.5, 4.5]:
        guesses += [generate_wave(f, 'square', 0.5), generate_wave(f, 'square', 0.7), generate_wave(f, 'square', 0.3)]
        guesses += [generate_wave(f, 'triangle'), generate_wave(f, 'sawtooth')]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=30,
