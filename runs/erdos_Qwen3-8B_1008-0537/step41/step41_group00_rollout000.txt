import time

        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_random():
        return np.random.rand(n_points)
    
    def generate_uniform():
        return np.repeat([0, 1], n_points // 2 + 1)[:n_points]
