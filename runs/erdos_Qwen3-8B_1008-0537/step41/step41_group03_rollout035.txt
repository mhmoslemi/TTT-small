import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_square(freq, duty):
        t = np.linspace(0, 1, n_points)
        wave = 0.5 * (np.sign(np.sin(2 * np.pi * freq * t)) + 1)
        wave = 0.5 + 0.5 * wave * duty
        return np.clip(wave, 0, 1)
    
    def generate_triangle(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.abs(2 * np.pi * freq * t - np.pi)
        wave /= np.pi
        wave = 1 - 2 * wave
        return np.clip(wave, 0, 1)
    
    def generate_sine(freq):
        t = np.linspace(0, 1, n_points)
        wave = 0.5 * (1 + np.sin(2 * np.pi * freq * t))
        return np.clip(wave, 0, 1)
    
    def generate_random_guess():
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * target_sum
        return np.clip(h_rand, 0, 1)
    
    def generate_random_perturbation(h_base):
        return np.clip(h_base + 0.05 * np.random.randn(n_points), 0, 1)
    
    guesses = [h]
    guesses += [generate_square(f, 0.5) for f in [1, 2, 3, 4, 5]]
    guesses += [generate_triangle(f) for f in [1, 2, 3, 4, 5]]
    guesses += [generate_sine(f) for f in [1, 2, 3, 4, 5]]
    guesses += [generate_random_guess() for _ in range(3)]
    guesses += [generate_random_perturbation(h)]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=40,
                                             mutation=(0.8, 1.2), recombination=0.9, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_bfgs(h_init):
        res = optimize.minimize(objective, h_init, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'gtol': 1e-9, 'maxiter': 60})
        return res.x if res.success else h_init

    return best_h, best_c5, n_points
