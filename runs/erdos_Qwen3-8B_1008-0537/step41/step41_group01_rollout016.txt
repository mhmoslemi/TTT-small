import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_chirp(t):
        freq = 2 * np.pi * (0.3 + 0.7 * np.sin(np.pi * t))
        return np.clip(0.5 + 0.5 * np.sin(freq), 0.0, 1.0)
    
    def generate_finite_pulse(t):
        return np.clip(0.5 + 0.5 * np.heaviside(t - 0.3, 0.5) * np.heaviside(0.7 - t, 0.5), 0.0, 1.0)
    
    def generate_phase_shaped(t, amp, freq, phase):
        return np.clip(0.5 + 0.5 * amp * np.sin(2 * np.pi * freq * t + phase), 0.0, 1.0)
    
    guesses = [h]
    
    # Generate diverse initial guesses with structured patterns
    t_vals = np.linspace(0, 1, n_points)
    guesses += [generate_chirp(t_vals)]
    guesses += [generate_finite_pulse(t_vals)]
    guesses += [generate_phase_shaped(t_vals, 0.7, 4, 0)]
    guesses += [generate_phase_shaped(t_vals, 0.6, 3, np.pi/2)]
    guesses += [generate_phase_shaped(t_vals, 0.5, 5, np.pi)]
    guesses += [generate_phase_shaped(t_vals, 0.6, 2, 0.6)]
    
    # Add random perturbations from initial guess
    for _ in range(2):
        noise = np.random.randn(n_points) * 0.1
        guesses.append(np.clip(h + noise, 0.0, 1.0))
    
    for guess in guesses:
        try:
            res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8,
                                            'disp': False, 'iprint': -1})
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        except:
            pass
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:
            break
        res_de = optimize.differential_evolution(objective, bounds, popsize=75, maxiter=50,
                                                mutation=(0.8, 1.2), recombination=0.9,
                                                tol=1e-8, disp=False)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8,
                                                  'disp': False, 'iprint': -1})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Final refinement with L-BFGS-B
    for _ in range(2):
        if time.time() - start_time >= budget_s - 15:
