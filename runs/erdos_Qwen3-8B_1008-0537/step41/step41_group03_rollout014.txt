import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(freq, func_type, *args):
        t = np.linspace(0, 1, n_points)
        if func_type == 'square':
            wave = 0.5 * (np.sign(np.sin(2 * np.pi * freq * t)) + 1)
            wave = 0.5 + 0.5 * wave * args[0]
        elif func_type == 'tri':
            wave = np.abs(2 * np.pi * freq * t - np.pi)
            wave /= np.pi
            wave = 1 - 2 * wave
        elif func_type == 'sine':
            wave = np.sin(2 * np.pi * freq * t)
            wave = 0.5 + 0.5 * wave
        elif func_type == 'pulse':
            wave = np.zeros_like(t)
            wave[::2] = 1.0
        wave = np.clip(wave, 0, 1)
        return wave
    
    guesses = [h]
    guesses += [generate_wave(f, 'square', 0.5) for f in [1, 2, 3, 4]]
    guesses += [generate_wave(f, 'tri', 0.5) for f in [1, 2, 3, 4]]
    guesses += [generate_wave(f, 'sine', 0.5) for f in [1, 2, 3, 4]]
    guesses += [generate_wave(f, 'pulse', 0.5) for f in [1, 2, 3, 4]]
    guesses += [np.clip(h + 0.05 * np.random.randn(n_points), 0, 1)]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success and objective(res.x) < best_c5:

            best_h = res.x
    
    for _ in range(3):
