        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

        elif waveform == 'tri':
            wave = np.abs(2 * np.pi * freq * t - np.pi) / np.pi
            wave = 1 - 2 * wave
        else:
            wave = np.sin(2 * np.pi * freq * t)
        return np.clip(wave, 0, 1)
