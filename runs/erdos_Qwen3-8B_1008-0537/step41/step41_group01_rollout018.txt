import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq, amp, phase, base=None):
        t = np.linspace(0, 1, n_points)
        wave = amp * np.sin(2 * np.pi * freq * t + phase)
        if base is not None:
            wave += base
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    guesses = [h]
    for f in [1.2, 1.5, 2.0, 2.4]:
        for amp in [0.6, 0.5, 0.4]:
            phase = np.random.uniform(0, 2*np.pi)
            guesses.append(generate_sine(f, amp, phase))
    for amp in [0.6, 0.5, 0.4]:
        guesses.append(generate_sine(2.0, amp, 0, base=0.5))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'gtol': 1e-9, 'maxiter': 150, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    def differential_evolution_with_refinement(popsize=50, maxiter=50):
        res = optimize.differential_evolution(objective, bounds, popsize=popsize, maxiter=maxiter,
                                             mutation=(0.9, 1.1), recombination=0.8, tol=1e-9)
        if res.success:
            h0 = res.x
            res = optimize.minimize(objective, h0, method='L-BFGS-B', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'gtol': 1e-9, 'maxiter': 100, 'eps': 1e-9})
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        return best_h
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        best_h = differential_evolution_with_refinement(popsize=40, maxiter=40)
    
    def refine_with_bfgs():
        res = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'gtol': 1e-9, 'maxiter': 80, 'eps': 1e-9})
        return res.x if res.success else best_h
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 15:
            break
        best_h = refine_with_bfgs()
        if objective(best_h) < best_c5:

    return best_h, best_c5, n_points
