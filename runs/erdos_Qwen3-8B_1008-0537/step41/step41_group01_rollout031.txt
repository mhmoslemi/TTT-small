import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    constraint = lambda h: np.sum(h) - target_sum

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_spread(n_points):
        h = np.zeros(n_points, dtype=float)
        num_ones = n_points // 2
        idx = np.sort(np.random.choice(n_points, size=num_ones, replace=False))
        h[idx] = 1.0
        return h

    def generate_sin(freq, phase):
        t = np.linspace(0, 1, n_points)
        wave = 0.5 * (1 + np.sin(2 * np.pi * freq * t + phase))
        return np.clip(wave, 0, 1)

    guesses = [h]
    guesses += [generate_sin(1.5, np.pi/4), generate_sin(2.5, np.pi/2)]
    guesses += [generate_binary_spread(n_points)]
    guesses += [np.random.rand(n_points) * 0.5]

    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=[(0, 1)]*n_points,
                               constraints=[{'type': 'eq', 'fun': constraint}],
                               options={'maxiter': 150, 'ftol': 1e-12})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, [(0, 1)]*n_points, 
                                             popsize=25, maxiter=30, tol=1e-9)
        res = optimize.minimize(objective, res.x, method='SLSQP', bounds=[(0, 1)]*n_points,
                               constraints=[{'type': 'eq', 'fun': constraint}],
                               options={'maxiter': 70, 'ftol': 1e-11})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    for _ in range(3):
        if time.time() - start_time >= budget_s - 15:
            break
        res = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                               constraints=[{'type': 'eq', 'fun': constraint}],
                               options={'gtol': 1e-9, 'maxiter': 60})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    return best_h, best_c5, n_points
