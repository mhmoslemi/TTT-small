import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_rippled(freq, amplitude):
        t = np.linspace(0, 1, n_points) * 2
        wave = amplitude * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_bipolar(freq, amp):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * (1 - wave), 0, 1)
    
    def generate_triangle(freq, amp):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * (2 * np.abs(t - np.floor(t + 0.5) - 0.5) - 1)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    # Generate more diverse initial guesses
    guesses = [
        generate_rippled(3, 0.35),
        generate_rippled(4, 0.3),
        generate_rippled(5, 0.25),
        generate_bipolar(2, 0.3),
        generate_bipolar(3, 0.25),
        generate_triangle(2, 0.25),
        generate_triangle(3, 0.2),
        generate_triangle(4, 0.15)
    ]
    # Add random perturbations to initial_h_values and include it
    guesses += [np.random.rand(n_points), h.copy()]
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 250, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential evolution as global search
    for _ in range(3):
        if time.time() - start_time >= budget_s - 15:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=30,
                                             mutation=(0.6, 0.85), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 60, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Final refinement
    for _ in range(2):
        if time.time() - start_time >= budget_s - 5:
            break
        res = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,

                               options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
