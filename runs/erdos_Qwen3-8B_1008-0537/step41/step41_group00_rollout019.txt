import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_rippled(freq, amp):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    def generate_bipolar(freq, amp):
        t = np.linspace(0, 1, n_points) * 2
        wave = amp * np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * (1 + wave), 0, 1)
    
    # Enhanced initial guesses
    guesses = [generate_rippled(3, 0.4), generate_rippled(4, 0.35), generate_rippled(5, 0.3)]
    guesses += [generate_bipolar(3, 0.3), generate_bipolar(4, 0.25), generate_bipolar(5, 0.22)]
    for freq in [2, 3, 4, 5, 6]:
        for amp in [0.25, 0.3, 0.35, 0.4, 0.5]:
            guesses.append(generate_rippled(freq, amp))
    for freq in [2, 3, 4, 5, 6]:
        for amp in [0.2, 0.25, 0.3, 0.35, 0.4]:
            guesses.append(generate_bipolar(freq, amp))
    guesses.append(np.random.rand(n_points))
    guesses.append(h.copy())
    
    # Quick refinement of initial guess
    def refine_guess(guess):
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
        return res.x if res.success else guess
    
    for g in guesses:
        if time.time() - start_time >= budget_s - 30:

        res = optimize.minimize(objective, g, method='SLSQP', bounds=bounds,

        if res.success and objective(res.x) < best_c5:

    # Differential evolution with adjusted parameters

    # Refinement with SLSQP
