    start_time = time.time()

    def de_step():
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=100, mutation=(0.3, 0.7), recombination=0.5, tol=1e-9)
        return res.x

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else r

    # Generate a more structured initial guess
    def structured_initial_guess(n_points):
        sample_positions = np.linspace(0, 2, n_points)
        positions = np.linspace(0, 2, 23)
        guess = np.zeros(n_points)
        for pos in positions:
            idx = np.argmin(np.abs(sample_positions - pos))
            if guess[idx] == 0:
                guess[idx] = 1.0
            if np.sum(guess) >= 23:
                break
        while np.sum(guess) < 22.5:
            idx = np.argmax(guess)
            guess[idx] += 0.1
        while np.sum(guess) > 23:
            idx = np.argmin(guess)
            guess[idx] -= 0.1
        return guess

    structured_guess = structured_initial_guess(n_points)
    initial_h = np.copy(structured_guess)

    # Initial optimization
    for _ in range(2):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_step()
        h_opt = refine(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Further refinement
    for _ in range(4):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
