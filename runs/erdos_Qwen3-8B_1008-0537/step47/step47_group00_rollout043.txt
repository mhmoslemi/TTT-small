import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate evenly spaced initial guess
    num_ones = n_points // 2 + 1
    indices = np.linspace(0, n_points - 1, num_ones, endpoint=False).astype(int)
    h = np.zeros(n_points)
    h[indices] = 1.0
    h[indices[-1]] = 0.5  # Adjust to meet sum constraint

    def obj(h_vals):
        corr = np.correlate(h_vals, 1.0 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def de_step():
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=30, mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-10})
        return res.x if res.success else r

    for _ in range(2):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_step()
        h_opt = refine(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
