import time

    n_points = len(h)
    target_sum = n_points / 2.0

    def obj(h_vals):

        return max_corr + penalty

    def cons(h_vals):

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

                               constraints=[{'type': 'eq', 'fun': cons}],
