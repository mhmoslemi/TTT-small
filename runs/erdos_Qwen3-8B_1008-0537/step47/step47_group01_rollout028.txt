import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Adjust for faster execution
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 0.25  # Initial uniform guess
    h = h * (n_points / 2.0)  # Ensure ∫h dx = 1
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - n_points / 2.0

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_sine(freq, phase, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2.0)
        return h

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else r

    guesses = [h]
    for freq in range(1, 20):
        guesses.append(gen_pulse(freq, 0.4))
        guesses.append(gen_saw(freq, 0.3))
        guesses.append(gen_sine(freq, 0.5, 0.6))
    for phase in np.linspace(0, 2*np.pi, 10):
        guesses.append(gen_sine(5, phase, 0.5))
    for _ in range(15):
        guesses.append(gen_random())

    for g in guesses:
        h_opt = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': cons}],
                                 options={'maxiter': 200, 'ftol': 1e-10}).x
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=20,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(10):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = optimize.minimize(obj, h_opt, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': cons}],
                                 options={'maxiter': 100, 'ftol': 1e-10}).x
        c = obj(h_opt)
        if c < best_c5:

        best_h = refine(best_h)
