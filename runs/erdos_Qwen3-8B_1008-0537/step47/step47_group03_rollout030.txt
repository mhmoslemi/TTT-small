        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def gen_evenly_spaced_blocks(m):
        dx = 2.0 / n_points
        block_length = 1.0 / m
        spacing = 0.01
        current_pos = 0.0
        h = np.zeros(n_points)
        for _ in range(m):
            start = current_pos
            end = current_pos + block_length
            start_idx = int(start / dx)
            end_idx = int(end / dx) + 1
            start_idx = max(0, min(start_idx, n_points - 1))
            end_idx = min(end_idx, n_points - 1)
            h[start_idx:end_idx+1] = 1.0
            current_pos += block_length + spacing
        return h

    guesses = [h]
    for freq in [3, 4, 5, 6]:
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.3, 0.4, 0.5, 0.6]:
        guesses.append(gen_saw(3, duty))
        guesses.append(gen_triangle(3, duty))
    for _ in range(5):
        guesses.append(gen_random())
    guesses.append(gen_evenly_spaced_blocks(5))
    guesses.append(gen_evenly_spaced_blocks(10))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 80, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=80, maxiter=50,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 80, 'ftol': 1e-10})
        return res.x if res.success else r

    for _ in range(15):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
