import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def mirror_h(h_vals):
        n = len(h_vals)
        mid = n // 2
        return np.concatenate([h_vals[:mid], h_vals[mid:][::-1]])

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 400, 'ftol': 1e-10})
        h_refined = res.x if res.success else r
        return mirror_h(h_refined)

    # Initial SLSQP optimization
    res = optimize.minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons}],
                           options={'maxiter': 300, 'ftol': 1e-10})
    best_h = refine(res.x)
    best_c5 = obj(best_h)

    # Differential Evolution
    if time.time() - start_time < budget_s - 30:
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=30,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        best_h = refine(res.x)
        best_c5 = obj(best_h)

    # Final refinement
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
