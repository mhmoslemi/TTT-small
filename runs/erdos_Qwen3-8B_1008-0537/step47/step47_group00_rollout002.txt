import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    def obj(h_vals):

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def de_step():
        res = optimize.differential_evolution(obj, bounds, popsize=200, maxiter=80, mutation=(0.7, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else r

    initial_h = h.copy()
    for _ in range(4):
