import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_sine(freq, phase, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def gen_modulated(freq, phase, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_sparse_pulses(freq):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        wave = np.abs(wave)
        return np.clip(wave * 0.8 + 0.1, 0, 1)

    def gen_chirp(freq1, freq2, phase):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * (freq1 + (freq2 - freq1) * x) * x + phase)
        return np.clip((wave + 1) * 0.5, 0, 1)

    guesses = [h]
    for freq in range(2, 11):
        guesses.append(gen_pulses(freq, 0.5))
        guesses.append(gen_saw(freq, 0.25))
        guesses.append(gen_triangle(freq, 0.5))
        guesses.append(gen_sine(freq, np.pi * np.random.rand(), 0.7))
    for phase in np.linspace(0, 2*np.pi, 12):
        guesses.append(gen_sine(3, phase, 0.6))
    for freq in [2, 4, 6, 8]:
        guesses.append(gen_modulated(freq, np.random.rand() * 2*np.pi, 0.6))
    for _ in range(15):
        guesses.append(gen_random())

    def slsqp_guess(g):
        try:
            res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 300, 'ftol': 1e-9})
            return res.x if res.success else g
        except:
            return g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        try:
            res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=30,
                                                 mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
            return res.x
        except:
            return best_h

    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        try:
            res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 400, 'ftol': 1e-9})
            return res.x if res.success else r
        except:
            return r

            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
