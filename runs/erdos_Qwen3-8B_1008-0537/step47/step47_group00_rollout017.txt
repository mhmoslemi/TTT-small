import time

    bounds = [(0., 1.) for _ in range(n_points)]

    best_c5 = obj(best_h)
    start_time = time.time()

    def de_step(popsize=50, maxiter=30):

        return res.x

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'disp': False})
        return res.x if res.success else r

    def random_init():
