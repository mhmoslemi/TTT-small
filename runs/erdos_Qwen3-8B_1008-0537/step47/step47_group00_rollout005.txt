import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    bounds = [(0., 1.) for _ in range(n_points)]
    
    def de_step(popsize=150, maxiter=100, mutation=(0.7, 1.0)):
        return optimize.differential_evolution(obj, bounds, popsize=popsize, maxiter=maxiter, mutation=mutation, recombination=0.8, tol=1e-6)
    
    def refine(r):
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = optimize.minimize(obj, r, method='COBYLA', bounds=bounds, constraints=cons, options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else r
    
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def perturb(h_vals, noise=0.1):
        return np.clip(h_vals + np.random.normal(0, noise, size=h_vals.shape), 0, 1)
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_step()
        h_opt = refine(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        h_opt = refine(best_h)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    return best_h, best_c5, n_points
