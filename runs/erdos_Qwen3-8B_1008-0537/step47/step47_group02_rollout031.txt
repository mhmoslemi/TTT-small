import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def de_search(g):
        try:
            res = optimize.differential_evolution(obj, bounds, popsize=150, maxiter=100,
                                                  mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
            return res.x
        except:
            return g

    try:
        h_opt = de_search(h)
        h_opt = optimize.minimize(obj, h_opt, method='SLSQP', bounds=bounds,
                                  constraints=[{'type': 'eq', 'fun': cons}],
                                  options={'maxiter': 100, 'ftol': 1e-9}).x

        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    except:
        pass

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        try:
            h_opt = optimize.minimize(obj, best_h, method='SLSQP', bounds=bounds,
                                      constraints=[{'type': 'eq', 'fun': cons}],
                                      options={'maxiter': 50, 'ftol': 1e-9}).x
            c = obj(h_opt)
            if c < best_c5:
                best_h, best_c5 = h_opt, c
        except:
            pass

    return best_h, best_c5, n_points
