import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def de_step(h_start):
        res = optimize.differential_evolution(obj, bounds, popsize=150, maxiter=50, mutation=(0.5, 0.95), recombination=0.8, tol=1e-9, x0=h_start)
        return res.x

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else r

    # Generate multiple starting points by perturbing the initial guess
    perturbations = []
    for _ in range(3):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbations.append(perturbed)

    # Run optimization on each perturbed starting point
    for p in perturbations:
        h_opt = de_step(p)
        h_opt = refine(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Additional refinement steps
    for _ in range(5):

            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:

    return best_h, best_c5, n_points
