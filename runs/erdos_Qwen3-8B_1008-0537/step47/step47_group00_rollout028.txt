import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj_de(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr) * dx
        sum_penalty = 1e-3 * np.abs(np.sum(h_vals) - target_sum)
        return max_corr + sum_penalty

    def obj_eval(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj_eval(best_h)
    start_time = time.time()

    def de_step():
        res = optimize.differential_evolution(obj_de, bounds, popsize=200, maxiter=30, mutation=(0.3, 0.6), recombination=0.5, tol=1e-7)
        return res.x

    def refine(r):
        res = optimize.minimize(obj_eval, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-10})
        return res.x if res.success else r

    initial_h = h.copy()
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_step()
        h_opt = refine(h_opt)
        c = obj_eval(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj_eval(best_h) < best_c5:
            best_c5 = obj_eval(best_h)

    return best_h, best_c5, n_points
