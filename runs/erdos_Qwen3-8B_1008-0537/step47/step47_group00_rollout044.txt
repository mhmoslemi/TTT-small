    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def de_step():
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=70, mutation=(0.6, 0.95), recombination=0.8, tol=1e-8)
        return res.x

    def refine(r):
        cons_func = {'type': 'eq', 'fun': cons}
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[cons_func],
                               options={'maxiter': 150, 'ftol': 1e-10, 'gtol': 1e-9})
        return res.x if res.success else r

    initial_h = h.copy()
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_step()
        h_opt = refine(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(6):
        if time.time() - start_time > budget_s - 20:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
