import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def slsqp_refine(g, maxiter=200):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': maxiter, 'ftol': 1e-9})
        return res.x if res.success else g

    def de_search(guess):
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=10,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9, args=(guess,))
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        try:
            improved = de_search(best_h)
            improved = slsqp_refine(improved, maxiter=100)
            c = obj(improved)
            if c < best_c5:
                best_h, best_c5 = improved, c
        except:
            pass

    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_refine(best_h, maxiter=200)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
