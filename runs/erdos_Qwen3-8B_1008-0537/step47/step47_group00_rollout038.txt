import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    n_points = len(h)

    def obj(h_vals):

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def de_step():
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=100, mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else r

    initial_h = h.copy()
    for _ in range(2):
        if time.time() - start_time > budget_s - 30:

            break
        best_h = refine(best_h)
