import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        x = (x * freq) % 1
        wave = np.abs(2 * x - 1)
        return np.clip((1 - wave) * duty, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        x = (x * freq) % 1
        return np.clip(1 - (x * duty), 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        x = (x * freq) % 1
        return np.clip(1 - np.abs(2*x - 1) * duty, 0, 1)

    def gen_sine(freq, phase, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(2, 6):
        guesses.append(gen_pulse(freq, 0.5))
        guesses.append(gen_saw(freq, 0.3))
        guesses.append(gen_triangle(freq, 0.4))
    for phase in np.linspace(0, 2*np.pi, 8):
        guesses.append(gen_sine(3, phase, 0.6))
    for _ in range(8):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=50,
                                             mutation=(0.4, 0.8), recombination=0.7, tol=1e-8)
        return res.x

    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 400, 'ftol': 1e-10})
        return res.x if res.success else r

    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
