    for _ in range(2):
        if time.time() - start_time > budget_s - 30:

        h_opt = de_guess()

        if c < best_c5:

    # Refine with more SLSQP steps
    def refine(r):

        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_structured():
        base = np.zeros(n_points)
        base[::2] = 1.0
        total = np.sum(base)
        if total > 0:
            base = base * (target_sum / total)
        return base

    guesses = [h, gen_structured()]
    for _ in range(2):
        if time.time() - start_time > budget_s - 50:
            break
        for g in guesses:
            res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 20, 'ftol': 1e-10})
            if res.success:
                h_opt = res.x
                c = obj(h_opt)
                if c < best_c5:
                    best_h, best_c5 = h_opt, c

    def de_guess():
        return optimize.differential_evolution(obj, bounds, popsize=20, maxiter=10,
                                              mutation=(0.5, 0.95), recombination=0.8, tol=1e-9).x

    for _ in range(2):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        res = optimize.minimize(obj, h_opt, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 30, 'ftol': 1e-10})
        if res.success:
            c = obj(res.x)
            if c < best_c5:
                best_h, best_c5 = res.x, c

    for _ in range(15):
        if time.time() - start_time > budget_s - 10:
            break
        res = optimize.minimize(obj, best_h, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 10, 'ftol': 1e-10})
        if res.success:
            best_h = res.x
            if obj(best_h) < best_c5:
                best_c5 = obj(best_h)

    return best_h, best_c5, n_points
