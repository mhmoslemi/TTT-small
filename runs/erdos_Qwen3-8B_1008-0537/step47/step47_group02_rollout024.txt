import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else np.random.rand(45)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_pattern(freq, duty, type_func):
        x = np.linspace(0, 1, n_points)
        if type_func == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
            return np.clip((wave + 1) * duty / 2, 0, 1)
        elif type_func == 'saw':
            return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif type_func == 'triangle':
            return 1 - np.abs(2 * freq * x - 1) * duty
        else:
            return np.clip(np.random.rand(n_points) * duty, 0, 1)

    def slsqp_refine(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-9})
        return res.x if res.success else g

    def de_search():
        def obj_wrapper(x):
            return obj(x)
        res = optimize.differential_evolution(obj_wrapper, bounds, popsize=150, maxiter=40,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    guesses = [h]
    for freq in range(2, 8):
        guesses.append(generate_pattern(freq, 0.45, 'pulse'))
    for duty in [0.3, 0.4, 0.5, 0.6]:
        guesses.append(generate_pattern(3, duty, 'saw'))
        guesses.append(generate_pattern(3, duty, 'triangle'))
    for _ in range(5):
        guesses.append(np.random.rand(n_points) * 0.5)
        guesses.append((np.abs(np.random.rand(n_points) - 0.5) < 0.35) * 1.0)

    for g in guesses:
        if time.time() - start_time > budget_s - 30:
            break
        try:
            h_opt = de_search()
            h_opt = slsqp_refine(h_opt)
            c = obj(h_opt)
            if c < best_c5:
                best_h, best_c5 = h_opt, c
        except:

        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
