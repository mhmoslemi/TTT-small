            obj, bounds, popsize=150, maxiter=30, mutation=(0.5, 0.9), recombination=0.8, tol=1e-9
        )
        return res.x

    def refine(r):
        res = optimize.minimize(
            obj, r, method='L-BFGS-B', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': cons}],
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        return res.x if res.success else r

    initial_h = h.copy()
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_step()
        h_opt = refine(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
