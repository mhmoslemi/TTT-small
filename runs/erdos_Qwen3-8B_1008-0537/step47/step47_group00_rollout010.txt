import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def de_step():
        res = optimize.differential_evolution(obj, bounds, popsize=150, maxiter=80, mutation=(0.6, 0.9), recombination=0.7, tol=1e-9)
        return res.x

    def refine(r):
        res = optimize.minimize(obj, r, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else r

    initial_h = h.copy()
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:

        if c < best_c5:

    for _ in range(5):

            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
