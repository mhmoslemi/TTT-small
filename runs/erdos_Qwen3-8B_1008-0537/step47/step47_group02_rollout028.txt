import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_pattern(freq, duty, type_func):
        x = np.linspace(0, 1, n_points)
        if type_func == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
            return np.clip((wave + 1) * duty / 2, 0, 1)
        elif type_func == 'saw':
            return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif type_func == 'triangle':
            return 1 - np.abs(2 * freq * x - 1) * duty
        else:
            return np.random.rand(n_points) * duty

    def slsqp_refine(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else g

    guesses = [h]
    for freq in range(3, 7):
        guesses.append(generate_pattern(freq, 0.5, 'pulse'))
    for duty in [0.2, 0.4, 0.6]:
        guesses.append(generate_pattern(3, duty, 'saw'))
        guesses.append(generate_pattern(3, duty, 'triangle'))

    for g in guesses:
        if time.time() - start_time > budget_s - 40:
            break
        try:
            res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=80,
                                                  mutation=(0.5, 1.0), recombination=0.8, tol=1e-9)
            h_opt = res.x
            h_opt = slsqp_refine(h_opt)
            c = obj(h_opt)
            if c < best_c5:
                best_h, best_c5 = h_opt, c
        except:
            pass

    for _ in range(15):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
