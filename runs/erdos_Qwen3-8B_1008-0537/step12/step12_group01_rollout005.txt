import scipy.optimize as optimize
import time

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.1):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    for _ in range(10):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.15)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
