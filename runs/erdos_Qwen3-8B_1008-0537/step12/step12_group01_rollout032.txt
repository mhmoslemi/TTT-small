import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.inf
        scale = target_sum / current_sum
        h_scaled = h * scale
        h_clipped = np.clip(h_scaled, 0, 1)
        current_clipped_sum = np.sum(h_clipped)
        if current_clipped_sum == 0:
            return np.inf
        scale_clipped = target_sum / current_clipped_sum
        h_final = h_clipped * scale_clipped
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.7,
            maxiter=100,
            seed=seed,
            disp=False
        )
        best_h = result.x
        best_c5 = objective(best_h)
    except:
        pass
    
    return best_h, best_c5, n_points
