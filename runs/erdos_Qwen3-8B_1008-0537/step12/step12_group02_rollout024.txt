        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-8,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    if time.time() - start_time < budget_s - 10:
        # Refine with SLSQP
        def constraint_sum(h_values):
            return np.sum(h_values) - target_sum
        
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result_slsqp.success:
            best_h = result_slsqp.x
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
