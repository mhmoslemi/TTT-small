import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    def optimize_with_method(method, x0, maxiter=500, scale=0.03):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                return result.x, current_c5
        return None, best_c5
    
    result, best_c5 = optimize_with_method('COBYLA', initial_guess, maxiter=400)
    if result is not None:
        best_h = result
    
    for _ in range(4):
        if time.time() - start_time >= budget_s - 10:
            break
        perturbed = generate_perturbation(best_h, scale=0.05)
        result, current_c5 = optimize_with_method('SLSQP', perturbed, maxiter=300, scale=0.02)
        if result is not None:
            best_h = result
            best_c5 = current_c5
        perturbed = generate_perturbation(best_h, scale=0.03)
        result, current_c5 = optimize_with_method('Nelder-Mead', perturbed, maxiter=200, scale=0.015)
        if result is not None:
            best_h = result
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
