import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective_with_penalty(h_values):
        h = np.array(h_values)
        sum_h = np.sum(h)
        penalty = 1e3 * abs(sum_h - target_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Differential evolution to find a good initial guess
    result_diff = optimize.differential_evolution(
        objective_with_penalty,
        bounds,

        popsize=20,
        tol=1e-6,

        recombination=0.8,
        x0=initial_guess
    )

    if best_c5_diff < best_c5:
        best_h = best_h_diff
        best_c5 = best_c5_diff
    
    # Refine with SLSQP
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=best_h,

        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
    )
    
    if result_slsqp.success:

    for _ in range(3):
