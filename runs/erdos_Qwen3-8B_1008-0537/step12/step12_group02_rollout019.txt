    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.03):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    def optimize_with_method(method, x0, maxiter=500):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            return result.x
        return None
    
    def optimize_with_de(x0):
        result = optimize.differential_evolution(
            fun=objective,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed
        )
        if result.success:
            return result.x
        return None
    
    result = optimize_with_method('COBYLA', initial_guess, maxiter=300)
    if result is not None:
        best_h = result
        best_c5 = objective(best_h)
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        perturbed = generate_perturbation(best_h, scale=0.02)
        result = optimize_with_method('SLSQP', perturbed, maxiter=300)
        if result is not None:
            best_c5 = objective(result)
            best_h = result
    
    # Try differential evolution for a single iteration
    de_result = optimize_with_de(best_h)
    if de_result is not None and objective(de_result) < best_c5:
        best_c5 = objective(de_result)
        best_h = de_result
    
    return best_h, best_c5, n_points
