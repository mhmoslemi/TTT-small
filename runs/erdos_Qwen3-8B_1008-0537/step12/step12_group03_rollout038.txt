    binary_pattern = np.clip(binary_pattern, 0.0, 1.0)
    perturbed = generate_perturbation(binary_pattern, scale=0.07)
    if optimize_with_method('SLSQP', perturbed):
        pass
    
    # Perturb and optimize with SLSQP
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.07)
        if optimize_with_method('SLSQP', perturbed):
            pass
    
    # Final perturb and optimize with COBYLA
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.04)
        if optimize_with_method('COBYLA', perturbed):
            pass
    
    return best_h, best_c5, n_points
