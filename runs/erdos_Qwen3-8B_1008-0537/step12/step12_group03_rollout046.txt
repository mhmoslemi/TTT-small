        return max_corr + sum_penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Differential evolution
    if time.time() - start_time < budget_s - 20:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed
        )
        if result_de.success:
            current_h = result_de.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s - 10:
        def constraint_sum(h_values):
            return np.sum(h_values) - target_sum
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result_slsqp.success:
            current_h = result_slsqp.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    return best_h, best_c5, n_points
