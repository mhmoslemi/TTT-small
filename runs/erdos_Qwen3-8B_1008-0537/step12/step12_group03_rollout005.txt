import scipy.optimize as optimize

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum

        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)

        try:
            result = optimize.minimize(
                fun=objective,
                x0=x0,
                method=method,

                options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )

            return False
        return False
