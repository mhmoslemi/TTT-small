            fun=objective,
            x0=x0,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            return result.x
        return None

    result = optimize_with_method('SLSQP', initial_guess, maxiter=300)
    if result is not None:
        best_h = result
        best_c5 = objective(best_h)

    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        perturbed = generate_perturbation(best_h, scale=0.02)
        result = optimize_with_method('COBYLA', perturbed, maxiter=200)
        if result is not None:
            best_c5 = objective(result)
            best_h = result

    return best_h, best_c5, n_points
