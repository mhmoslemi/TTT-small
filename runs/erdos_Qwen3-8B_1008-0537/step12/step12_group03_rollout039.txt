        h = np.array(h_values)
        g = 1.0 - h
        h_fft = np.fft.fft(h)
        g_fft = np.fft.fft(g)
        corr_fft = h_fft * np.conj(g_fft)
        corr = np.fft.ifft(corr_fft).real
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.05):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    def optimize_with_method(method, x0, **kwargs):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=x0,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                return True
        except:
            return False
        return False
    
    if optimize_with_method('SLSQP', initial_guess):
        pass
    
    if optimize_with_method('L-BFGS-B', initial_guess):
        pass
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.07)
        if optimize_with_method('COBYLA', perturbed):
            pass
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.04)
        if optimize_with_method('SLSQP', perturbed):
            pass
    
    return best_h, best_c5, n_points
