import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate uniform initial guess
    m = int(target_sum)
    remaining = target_sum - m
    x = np.linspace(0, 2, m + 1)
    idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
    uniform_guess = np.zeros(n_points)
    for i in range(m):
        uniform_guess[idxs[i]] = 1.0
    if remaining > 0:
        idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1)))
        uniform_guess[idx] += remaining
    
    # Initial guesses
    initial_guesses = [initial_guess, uniform_guess]
    
    for initial in initial_guesses:
        try:
            # Differential evolution
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=50,
                maxiter=400,
                mutation=(0.5, 1.0),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                workers=1
            )
            if result_de.success:
                # COBYLA refinement
                result_cobyla = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='COBYLA',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_cobyla.success:
                    current_c5 = objective(result_cobyla.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_cobyla.x
        except:
            pass
    
    # Additional refinements
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        try:
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
