import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.03):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    def optimize_with_method(method, x0, maxiter=500):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            return result.x
        return None
    
    # Alternative initial guess: evenly distributed 1s with one adjusted to meet sum constraint
    if initial_guess.size == 0:
        mid = n_points // 2 + 1
        indices = np.linspace(0, n_points, mid, endpoint=False).astype(int)
        h_initial = np.zeros(n_points)
        h_initial[indices] = 1.0
        current_sum = np.sum(h_initial)
        delta = (n_points / 2) - current_sum
        if delta > 0:
            h_initial[np.argmin(h_initial)] += delta
        else:
            h_initial[np.argmax(h_initial)] -= delta
        initial_guess = h_initial
    
    result = optimize_with_method('L-BFGS-B', initial_guess, maxiter=300)
    if result is not None:
        best_h = result
        best_c5 = objective(best_h)
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        perturbed = generate_perturbation(best_h, scale=0.02)
        result = optimize_with_method('SLSQP', perturbed, maxiter=150)
        if result is not None:
            best_c5 = objective(result)
            best_h = result
    
    return best_h, best_c5, n_points
