        remaining = target_sum - m
        xs = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), xs)
        for i in range(m):
            initial_h[idxs[i]] = 1.0
        if remaining > 0:
            x = np.max(xs) * m / (m + 1)
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - x))
            initial_h[idx] += remaining
        initial_h = initial_h.astype(float)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_h.copy()
    best_c5 = np.inf

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=60,
        maxiter=150,
        mutation=(0.7, 1.0),
        recombination=0.8,
        tol=1e-8,
        vectorized=True
    )
    if result_de.success:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x

    x = np.linspace(0, 2, int(target_sum) + 1)
    idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
    uniform_guess = np.zeros(n_points)
    for i in range(int(target_sum)):
        uniform_guess[idxs[i]] = 1.0
    if (target_sum - int(target_sum)) > 0:
        x_uniform = np.max(x) * int(target_sum) / (int(target_sum) + 1)
        idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - x_uniform))
        uniform_guess[idx] += (target_sum - int(target_sum))
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=uniform_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_c5 = objective(result_slsqp.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x

    start_time = time.time()
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
