import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def obj(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def perturb(h, scale):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)

    def optimize(method, x0, maxiter=1000):
        result = optimize.minimize(
            fun=obj, x0=x0, method=method,
            bounds=bounds, constraints=[{'type': 'eq', 'fun': cons}],
            options={'maxiter': maxiter, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if result.success and obj(result.x) < best_c5:
            return result.x
        return None

    # First refinement with COBYLA
    result = optimize('COBYLA', h, maxiter=500)
    if result is not None:
        best_h = result
        best_c5 = obj(best_h)

    # Iterative refinement with SLSQP
    for _ in range(4):
        if time.time() - start_time >= budget_s - 10:
            break
        h = perturb(best_h, 0.02)
        result = optimize('SLSQP', h, maxiter=300)
        if result is not None:
            best_c5 = obj(result)
            best_h = result

    return best_h, best_c5, n_points
