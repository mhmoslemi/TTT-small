import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.05):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    def optimize_with_method(method, x0, **kwargs):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=x0,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                return True
        except:
            return False
        return False
    
    # Initial optimization with SLSQP
    if optimize_with_method('SLSQP', initial_guess):
        pass
    
    # Initial optimization with COBYLA
    if optimize_with_method('COBYLA', initial_guess):

    for _ in range(5):
