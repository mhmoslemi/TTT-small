import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    def objective(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(15):
        if time.time() - start_time >= budget_s - 20:
            break
        pert = np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(initial_guess + pert, 0, 1)
        res = optimize.minimize(
            fun=objective, x0=perturbed,
            method='COBYLA', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-11, 'eps': 1e-9}
        )
        if res.success:
            curr_c5 = objective(res.x)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = res.x
    
    for _ in range(15):
        if time.time() - start_time >= budget_s - 20:
            break
        pert = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(best_h + pert, 0, 1)
        res = optimize.minimize(
            fun=objective, x0=perturbed,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-11, 'eps': 1e-9}
        )
        if res.success:
            curr_c5 = objective(res.x)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = res.x
    
    return best_h, best_c5, n_points
