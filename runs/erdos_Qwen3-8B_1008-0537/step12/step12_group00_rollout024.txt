import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def uniform_guess():
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        uniform = np.zeros(n_points)
        for i in range(m):
            uniform[idxs[i]] = 1.0
        if remaining > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1)))
            uniform[idx % n_points] += remaining
        return uniform
    
    def random_guess():
        m = int(target_sum)
        remaining = target_sum - m
        random = np.zeros(n_points)
        indices = np.random.choice(n_points, size=m, replace=False)
        random[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            random[idx] += remaining
        return random
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
