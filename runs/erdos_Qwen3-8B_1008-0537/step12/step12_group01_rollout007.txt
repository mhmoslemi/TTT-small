import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.1):
        noise = np.random.normal(0, scale, size=n_points)
        perturbed = np.clip(h + noise, 0, 1)
        return perturbed
    
    def optimize_from_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x
        return guess
    
    for _ in range(20):
        if time.time() - start_time >= budget_s - 20:
            break

            perturbed = generate_perturbation(best_h, scale=0.15)
        else:

        new_h = optimize_from_guess(perturbed)
        current_c5 = objective(new_h)
        if current_c5 < best_c5:
