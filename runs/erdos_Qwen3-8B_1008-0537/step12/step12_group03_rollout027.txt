                options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                return True
        except:
            return False
        return False
    
    # Initial optimization with differential evolution
    def differential_evolution_optimize(x0):
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            x0=x0,
            popsize=15,
            maxiter=100,
            tol=1e-8,
            seed=seed
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            return True
        return False
    
    if differential_evolution_optimize(initial_guess):
        pass
    
    # Local optimization with SLSQP
    if optimize_with_method('SLSQP', best_h):
        pass
    
    # Perturb and optimize with COBYLA
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.07)
        if optimize_with_method('COBYLA', perturbed):
            pass
    
    # Final perturb and optimize with L-BFGS-B
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.04)
        if optimize_with_method('L-BFGS-B', perturbed):
            pass
    
    return best_h, best_c5, n_points
