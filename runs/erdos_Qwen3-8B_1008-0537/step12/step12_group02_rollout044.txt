        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.03):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    def optimize_with_method(method, x0, maxiter=500):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success:
            return result.x
        return None
    
    # Initial optimization with COBYLA
    result = optimize_with_method('COBYLA', initial_guess, maxiter=500)
    if result is not None:
        best_h = result
        best_c5 = objective(best_h)
    
    # Extended perturbation and optimization with SLSQP
    for _ in range(5):  # Increase the number of iterations
        if time.time() - start_time >= budget_s - 10:
            break
        perturbed = generate_perturbation(best_h, scale=0.02)
        result = optimize_with_method('SLSQP', perturbed, maxiter=300)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    return best_h, best_c5, n_points
