        perturbed = initial_guess + np.random.normal(0, 0.07, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        if optimize_with_method('SLSQP', perturbed):
            pass
    
    # Random guess optimization
    best_h = random_guess.copy()
    best_c5 = objective(best_h)
    if optimize_with_method('COBYLA', random_guess):
        pass
    if optimize_with_method('L-BFGS-B', random_guess):
        pass
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = random_guess + np.random.normal(0, 0.07, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        if optimize_with_method('SLSQP', perturbed):
            pass
    
    # Final perturb and optimize
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = best_h + np.random.normal(0, 0.04, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        if optimize_with_method('COBYLA', perturbed):
            pass
    
    return best_h, best_c5, n_points
