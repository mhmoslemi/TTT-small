import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def generate_perturbation(h, scale=0.05, seed=None):
        rng = np.random.RandomState(seed)
        return np.clip(h + rng.normal(0, scale, size=n_points), 0, 1)
    
    for _ in range(15):

        perturbed = generate_perturbation(best_h, scale=0.1, seed=seed)
