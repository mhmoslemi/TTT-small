import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbations(h, scales):
        return [np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1) for scale in scales]
    
    scales = [0.1, 0.05, 0.02, 0.01]
    perturbations = generate_perturbations(initial_guess, scales)
    
    for pert in perturbations:
        result = optimize.minimize(
            fun=objective,
            x0=pert,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    for _ in range(20):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = np.clip(initial_guess + np.random.normal(0, 0.02, size=n_points), 0, 1)
        result = optimize.minimize(

        )
        if result.success:
