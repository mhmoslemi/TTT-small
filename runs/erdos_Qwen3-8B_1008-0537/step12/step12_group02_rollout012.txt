import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_c = np.max(corr) * dx
        penalty = 1e3 * np.abs(np.sum(h) - target_sum)
        return max_c + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_initial():
        n_ones = int(target_sum)
        initial = np.zeros(n_points, dtype=float)
        initial[:n_ones] = 1.0
        np.random.shuffle(initial)
        return initial

    initial_guess = generate_initial()

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        result = optimize.minimize(
            fun=objective,
            x0=initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_h = result.x
            best_c5 = objective(best_h)
        initial_guess = best_h + np.random.normal(0, 0.02, size=n_points)
        initial_guess = np.clip(initial_guess, 0, 1)

    return best_h, best_c5, n_points
