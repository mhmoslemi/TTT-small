import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_regular_guess(num_ones):
        spacing = n_points / num_ones
        indices = np.round(np.arange(0, n_points, spacing)).astype(int)
        indices = np.unique(indices)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_symmetric_guess(num_ones):
        if num_ones % 2 == 0:
            mid = n_points // 2
            indices = np.sort(np.random.choice(mid, num_ones // 2, replace=False))
            indices = np.concatenate([indices, n_points - indices - 1])
        else:
            mid = n_points // 2
            indices = np.sort(np.random.choice(mid, (num_ones - 1) // 2, replace=False))
            indices = np.concatenate([indices, n_points - indices - 1, [mid]])
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess]
    for _ in range(2):
        for num_ones in [22, 23, 24, 25, 26, 27, 28, 30]:

    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,

                bounds=bounds,
