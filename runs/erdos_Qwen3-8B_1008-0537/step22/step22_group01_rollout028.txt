import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=100, **kwargs):

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def generate_well_spaced_guess():

        if n_points % 2:

        h = np.zeros(n_points)

            pos = 2.0 * i / m
            idx = int(pos / dx)
            if 0 <= idx < n_points:
                h[idx] = 1.0

        if idx < n_points:
            h[idx] = 0.5
