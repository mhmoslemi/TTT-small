import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_varied_guess(num_ones, pattern='random'):
        if pattern == 'regular':
            indices = np.linspace(0, n_points - 1, num_ones, endpoint=False).astype(int)
            h = np.zeros(n_points)
            h[indices] = 1.0
            remaining = np.where(h == 0)[0]
            needed = max(0, int(target_sum - np.sum(h)))
            if needed > 0 and len(remaining) > 0:
                idxs = np.random.choice(remaining, size=needed, replace=False)
                h[idxs] = 0.5
            return h
        else:
            indices = np.random.choice(n_points, num_ones, replace=False)
            h = np.zeros(n_points)
            h[indices] = 1.0
            remaining = np.where(h == 0)[0]
            needed = max(0, int(target_sum - np.sum(h)))
            if needed > 0 and len(remaining) > 0:
                idxs = np.random.choice(remaining, size=needed, replace=False)
                h[idxs] = 0.5
            return h
    
    initial_guesses = [initial_guess]

    def optimize_guess(guess):
        try:
            result = optimize.minimize(

                x0=guess,
                method='SLSQP',
                bounds=bounds,

    try:
        result_de = optimize.differential_evolution(
