import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_regular_guess(num_ones):
        spacing = n_points / num_ones
        indices = np.round(np.arange(0, n_points, spacing)).astype(int)
        indices = np.unique(indices)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_varied_guess(num_ones, spacing='random'):
        if spacing == 'regular':
            return generate_regular_guess(num_ones)
        indices = np.sort(np.random.choice(n_points, num_ones, replace=False))
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_chebyshev_guess(num_ones):
        from numpy.polynomial import chebyshev
        if num_ones == 0:
            return np.zeros(n_points)
        if num_ones == n_points:
            return np.ones(n_points)
        c = np.zeros(num_ones)
        c[0] = 1.0
        c[1] = 0.0
        for k in range(2, num_ones):
            c[k] = 0.5 * (1.0 - (k % 2))
        p = chebyshev.Chebyshev(c)
        x = np.linspace(-1, 1, n_points)
        h = p(x)
        h = np.maximum(0, np.minimum(1, h))
        h = np.round(h * 100) / 100
        if np.sum(h) < target_sum - 1e-6:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def optimized_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
