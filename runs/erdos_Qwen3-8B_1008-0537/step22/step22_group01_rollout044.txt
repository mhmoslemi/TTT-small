    def generate_well_spaced_guess():

            m += 1
        h = np.zeros(n_points)
        for i in range(m):
            pos = 2.0 * i / m
            idx = int(pos / dx)
            if 0 <= idx < n_points:
                h[idx] = 1.0
        idx = int(2.0 / dx)
        if idx < n_points:
            h[idx] = 0.5

        return random_vals

    def optimize_guess(guess):

            result = optimize.minimize(
