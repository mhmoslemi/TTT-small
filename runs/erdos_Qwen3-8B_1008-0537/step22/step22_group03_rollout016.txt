import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def get_initial_guesses(num_ones, spacing='random'):
        indices = np.sort(np.random.choice(n_points, num_ones, replace=False))
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = np.zeros(n_points)
    best_c5 = 1.0
    
    start_time = time.time()
    
    def optimize_guess(start):
        try:
            result = optimize.minimize(
                fun=objective, x0=start,
                method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for num_ones in [20, 22, 24, 25, 28]:
        for _ in range(3):
            guess = get_initial_guesses(num_ones)
            result = optimize_guess(guess)
            if result is not None:
                c = objective(result)
                if c < best_c5:
                    best_c5 = c
                    best_h = result
    
    if time.time() - start_time < budget_s - 50:
        try:
            result_de = optimize.differential_evolution(
                fun=objective, bounds=bounds,
                popsize=50, maxiter=100, mutation=(0.5, 1.0),
                recombination=0.9, tol=1e-10, vectorized=True, seed=seed
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective, x0=result_de.x,
                    method='SLSQP', bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    c = objective(result_slsqp.x)
                    if c < best_c5:
                        best_c5 = c
                        best_h = result_slsqp.x
        except:
            pass
    
    while time.time() - start_time < budget_s - 20:
        try:
            result = optimize.minimize(
