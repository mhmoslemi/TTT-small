import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def reg_guess():
        h = np.zeros(n_points)
        m = int(n_points / 2)
        for i in range(m):
            idx = int((2 * i) / m * n_points / 2)
            if 0 <= idx < n_points:
                h[idx] = 1.0
        return h
    
    def rand_guess():
        h = np.random.rand(n_points)
        h -= h.mean()
        h /= np.abs(h).max() + 1e-6
        h = np.clip(h, 0, 1)
        factor = (target_sum - np.sum(h)) / np.sum(h)
        h = np.clip(h * factor, 0, 1)
        return h
    
    guesses = [h, reg_guess(), rand_guess()]
    
    def de_opt(guess):
        try:
            res = optimize.differential_evolution(
                obj, bounds, popsize=30, maxiter=100, mutation=(0.9, 1.2),
                recombination=0.5, tol=1e-10, seed=seed)
            return res.x if res.success else guess
        except:
            return guess
    
    for g in guesses:
        g = de_opt(g)
        try:
            res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 50, 'ftol': 1e-10})
            if res.success and obj(res.x) < best_c5:
                best_c5 = obj(res.x)
                best_h = res.x
        except:
            pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            res = optimize.minimize(obj, best_h, method='L-BFGS-B',
