import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

            h[indices] = 1.0
            diff = target_sum - np.sum(h)

                zeros = np.where(h == 0)[0]
                add = int(diff)
                idxs = np.random.choice(zeros, size=add, replace=False)
                h[idxs] = 0.5
            return h

                zeros = np.where(h == 0)[0]

                h[idxs] = 0.5
