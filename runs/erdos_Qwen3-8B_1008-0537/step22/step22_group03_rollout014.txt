import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_pattern(num_ones, spacing='regular'):
        if spacing == 'regular':
            spacing = n_points / num_ones
            indices = np.round(np.arange(0, n_points, spacing)).astype(int)
            indices = np.unique(indices)
            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:
                idx = np.random.choice(np.where(h == 0)[0], size=1)
                h[idx] = 0.5
            return h
        elif spacing == 'clustered':
            cluster_size = num_ones // 2
            cluster_centers = np.linspace(0, 2, cluster_size)
            h = np.zeros(n_points)
            for c in cluster_centers:
                idx = np.round(c / dx).astype(int)
                h[idx] = 1.0
            if np.sum(h) < target_sum:
                idx = np.random.choice(np.where(h == 0)[0], size=1)
                h[idx] = 0.5
            return h
    
    initial_guesses = [initial_guess]
    for num_ones in [22, 23, 24, 25, 26]:
        initial_guesses.append(generate_pattern(num_ones, 'regular'))
        initial_guesses.append(generate_pattern(num_ones, 'clustered'))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-8}
            )
            return result.x if result.success else None

        if result_de.success:
