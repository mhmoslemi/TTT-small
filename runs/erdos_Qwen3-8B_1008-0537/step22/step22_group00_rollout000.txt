import scipy.optimize as optimize
import time

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def generate_initial_guesses(num_guesses=5):
        guesses = []
        base = initial_guess.copy()

            perturbation = np.random.normal(0, 0.1, n_points)
            perturbation = np.clip(perturbation, -0.5, 0.5)
            guess = base + perturbation
            guess = np.clip(guess, 0.0, 1.0)
            guess = np.where(np.sum(guess) < target_sum, guess + 0.05 * (target_sum - np.sum(guess)), guess)
            guesses.append(guess)
        return guesses

    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,

            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:

            best_c5 = objective(result)

        result_de = optimize.differential_evolution(
            fun=objective,

            tol=1e-9,
            vectorized=True,

        )
