import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def generate_well_spaced_guess():
        m = int(n_points / 2)
        if n_points % 2:
            m += 1
        h = np.zeros(n_points)
        for i in range(m):
            pos = 2.0 * i / m
            idx = int(pos / dx)
            if 0 <= idx < n_points:
                h[idx] = 1.0
        idx = int(2.0 / dx)
        if idx < n_points:
            h[idx] = 0.5
        return h
    
    def generate_spiral_guess():
        h = np.zeros(n_points)
        for i in range(n_points):
            angle = 2 * np.pi * i / n_points
            pos = 1.0 + 0.5 * np.cos(angle)
            idx = int(pos / dx)
            if 0 <= idx < n_points:
                h[idx] = 1.0
        return h
    
    def generate_random_guess():
        h = np.random.rand(n_points)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = (h * 2 - 1) * 0.5 + 0.5
        h_norm = np.sum(h) / target_sum
        return h / h_norm
    
    def optimize_guess(guess, method='SLSQP', maxiter=200):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': maxiter, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    initial_guesses = [initial_guess, generate_well_spaced_guess(), generate_spiral_guess(), generate_random_guess()]

        result_de = optimize.differential_evolution(
