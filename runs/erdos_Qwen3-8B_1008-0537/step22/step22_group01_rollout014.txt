import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_well_spaced_guess():
        m = int(n_points / 2)
        if n_points % 2:
            m += 1
        h = np.zeros(n_points)
        for i in range(m):
            pos = 2.0 * i / m
            idx = int(pos / dx)
            if 0 <= idx < n_points:
                h[idx] = 1.0
        idx = int(2.0 / dx)
        if idx < n_points:
            h[idx] = 0.5
        return h
    
    def generate_random_binary():
        if n_points % 2 == 0:
            half = n_points // 2
            idx = np.random.choice(n_points, half, replace=False)
            h = np.zeros(n_points)
            h[idx] = 1.0
            return h
        else:
            half = n_points // 2
            idx = np.random.choice(n_points, half, replace=False)
            h = np.zeros(n_points)
            h[idx] = 1.0
            remaining = np.setdiff1d(np.arange(n_points), idx)
            idx_to_set = np.random.choice(remaining)
            h[idx_to_set] = 0.5
            return h
    
    initial_guesses = [initial_guess, generate_well_spaced_guess(), generate_random_binary()]
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:

    try:
        result_de = optimize.differential_evolution(

        if result_de.success:

                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
