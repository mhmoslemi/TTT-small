import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def generate_varied_guess(num_ones, spacing='regular'):
        if spacing == 'regular':
            spacing = n_points / num_ones
            indices = np.round(np.arange(0, n_points, spacing)).astype(int)
            indices = np.unique(indices)
            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:
                idx = np.random.choice(np.where(h == 0)[0], size=1)
                h[idx] = 0.5
            return h
        indices = np.random.choice(n_points, num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess]
    for num_ones in [10, 15, 20, 25, 30, 35, 40]:
        initial_guesses.append(generate_varied_guess(num_ones, 'regular'))
        initial_guesses.append(generate_varied_guess(num_ones, 'random'))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:

    for guess in initial_guesses:

        if result_de.success:
