        if current_sum < target_sum:
            idx = np.where(guess == 0)[0]
            to_add = int(target_sum - current_sum)
            if len(idx) >= to_add:
                guess[idx[:to_add]] = 0.5
        elif current_sum > target_sum:
            idx = np.where(guess == 1)[0]
            to_remove = int(current_sum - target_sum)
            if len(idx) >= to_remove:
                guess[idx[:to_remove]] = 0.5
        return guess
    
    initial_guesses = [initial_guess]
    for num_ones in [15, 20, 25, 30, 35, 40]:
        initial_guesses.append(generate_varied_guess(num_ones))
    initial_guesses.append(create_alternating_guess(n_points))
    initial_guesses.append(create_block_guess(n_points))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=200,
            mutation=(0.8, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    if time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
