import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_structured_guess(num_ones, pattern):
        if pattern == 'even':
            step = n_points // num_ones
            indices = np.arange(num_ones) * step
            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:
                zeros = np.where(h == 0)[0]
                add = max(0, int(target_sum - np.sum(h)))
                idxs = np.random.choice(zeros, size=add, replace=False)
                h[idxs] = 0.5
            return h
        elif pattern == 'fib':
            h = np.zeros(n_points)
            idxs = np.round(np.arange(num_ones) * (n_points / num_ones) * (1 + np.sqrt(5))/2).astype(int)
            h[idxs] = 1.0
            if np.sum(h) < target_sum:
                zeros = np.where(h == 0)[0]
                add = max(0, int(target_sum - np.sum(h)))
                idxs = np.random.choice(zeros, size=add, replace=False)
                h[idxs] = 0.5
            return h
        else:
            return np.random.choice([0, 1], size=n_points, p=[(n_points - num_ones)/n_points, num_ones/n_points])
    
    initial_guesses = [initial_guess]
    for num_ones in [20, 25, 30, 35]:
        initial_guesses.append(generate_structured_guess(num_ones, 'even'))
        initial_guesses.append(generate_structured_guess(num_ones, 'fib'))
        initial_guesses.append(np.random.choice([0, 1], size=n_points, p=[(n_points - num_ones)/n_points, num_ones/n_points]))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-8}
            )
            return result.x if result.success else None
        except:

    for guess in initial_guesses:

    try:
        result_de = optimize.differential_evolution(

        )
        if result_de.success:
            result_slsqp = optimize.minimize(
