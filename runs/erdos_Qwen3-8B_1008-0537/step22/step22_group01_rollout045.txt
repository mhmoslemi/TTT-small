import scipy.optimize as optimize
import time

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum

            pos = 2.0 * i / m
            idx = int(pos / dx)
            if 0 <= idx < n_points:
                h[idx] = 1.0
        idx = int(2.0 / dx)
        if idx < n_points:
            h[idx] = 0.5
