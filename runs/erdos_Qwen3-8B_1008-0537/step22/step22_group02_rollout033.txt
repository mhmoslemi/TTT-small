import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_regular_guess(m):
        step = 2.0 / m
        positions = np.arange(m) * step
        indices = np.round(positions / (2.0 / n_points) * n_points).astype(int)
        indices = np.unique(indices)
        h = np.zeros(n_points)
        h[indices] = 1.0
        current_sum = np.sum(h)
        if current_sum < target_sum:
            zeros = np.where(h == 0)[0]
            add = max(0, int(target_sum - current_sum))
            idxs = np.random.choice(zeros, size=add, replace=False)
            h[idxs] = 0.5
        return h
    
    initial_guesses = [initial_guess]
    for m in [22, 23, 24]:
        initial_guesses.append(generate_regular_guess(m))
    for m in [20, 25, 30]:
        indices = np.random.choice(n_points, m, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        current_sum = np.sum(h)
        if current_sum < target_sum:
            zeros = np.where(h == 0)[0]
            add = max(0, int(target_sum - current_sum))
            idxs = np.random.choice(zeros, size=add, replace=False)
            h[idxs] = 0.5
        initial_guesses.append(h)
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:

        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(

            popsize=150,

            vectorized=True,

        if result_de.success:
            result_slsqp = optimize.minimize(

            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)

    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
