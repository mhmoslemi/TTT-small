import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_varied_guess(num_ones, pattern='random'):
        if pattern == 'regular':
            spacing = n_points / num_ones
            indices = np.round(np.arange(0, num_ones) * spacing).astype(int)
            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:
                zeros = np.where(h == 0)[0]
                if len(zeros) > 0:
                    add = max(0, int(target_sum - np.sum(h)))
                    idxs = np.random.choice(zeros, size=add, replace=False)
                    h[idxs] = 0.5
            return h
        else:
            indices = np.random.choice(n_points, num_ones, replace=False)
            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:
                zeros = np.where(h == 0)[0]
                if len(zeros) > 0:
                    add = max(0, int(target_sum - np.sum(h)))

                    h[idxs] = 0.5
            return h
    
    initial_guesses = [initial_guess]
    for num_ones in [20, 25, 30, 35, 40]:
        initial_guesses.append(generate_varied_guess(num_ones, 'regular'))
        initial_guesses.append(generate_varied_guess(num_ones, 'random'))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}

            vectorized=True,

        if result_de.success:
            result_slsqp = optimize.minimize(

                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}

            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
