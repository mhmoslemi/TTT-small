import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_varied_guess(num_ones, pattern='random'):
        if pattern == 'regular':

            h[indices] = 1.0
            if np.sum(h) < target_sum:
                zeros = np.where(h == 0)[0]
                if len(zeros) > 0:
                    add = max(0, int(target_sum - np.sum(h)))
                    idxs = np.random.choice(zeros, size=add, replace=False)
                    h[idxs] = 0.5
