import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_regular_guess(num_ones):
        spacing = n_points // num_ones
        h = np.zeros(n_points)
        for i in range(num_ones):
            idx = i * spacing
            if idx < n_points:
                h[idx] = 1.0
        return h
    
    initial_guesses = [initial_guess]
    for num_ones in [15, 20, 25, 30, 35, 40]:
        initial_guesses.append(generate_regular_guess(num_ones))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
