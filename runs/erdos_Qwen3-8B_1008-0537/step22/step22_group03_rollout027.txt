import scipy.optimize as optimize
import time

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_padded_guess(num_ones, spacing='regular'):
        if spacing == 'regular':
            spacing = n_points / num_ones
            indices = np.round(np.arange(0, n_points, spacing)).astype(int)
            indices = np.unique(indices)
        else:
            indices = np.random.choice(n_points, num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess]
    for _ in range(2):
        for num_ones in [22, 23, 24, 25]:
            initial_guesses.append(generate_padded_guess(num_ones, 'regular'))
        for num_ones in [20, 21, 26, 27]:
            initial_guesses.append(generate_padded_guess(num_ones, 'random'))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],

        result_de = optimize.differential_evolution(

        )

                best_c5 = objective(result_slsqp.x)
