import scipy.optimize as optimize
import time

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_varied_guess(num_ones, pattern='regular'):

            h[indices] = 1.0
            if np.sum(h) < target_sum:
                zeros = np.where(h == 0)[0]
                add = max(0, int(target_sum - np.sum(h)))

                h[idxs] = 0.5
            return h

                zeros = np.where(h == 0)[0]
                add = max(0, int(target_sum - np.sum(h)))
                idxs = np.random.choice(zeros, size=add, replace=False)
                h[idxs] = 0.5
            return h

            if np.sum(h) < target_sum:

            result = optimize.minimize(
