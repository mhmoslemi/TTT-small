import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_diverse_guesses(num=5):
        guesses = [initial_guess]
        for _ in range(num):
            if np.random.rand() < 0.4:
                idxs = np.random.choice(n_points, size=25, replace=False)
                h = np.zeros(n_points)
                h[idxs] = 1.0
                if np.sum(h) < target_sum:
                    idx = np.random.choice(np.where(h == 0)[0], size=1)
                    h[idx] = 0.5
                guesses.append(h)
            else:
                h = np.random.uniform(0.4, 0.6, n_points)
                h = h / np.sum(h) * target_sum
                guesses.append(h)
        return guesses
    
    def generate_block_guesses(num=3):
        guesses = []
        for _ in range(num):
            m = np.random.randint(3, 6)  # Number of blocks
            block_length = 2.0 / m
            h_block = np.zeros(n_points)
            for i in range(m):
                start = int(i * block_length / dx)
                end = int((i+1) * block_length / dx)
                if np.random.rand() < 0.5:
                    h_block[start:end] = 1.0
            sum_h = np.sum(h_block)
            if sum_h > 0:
                h_block = h_block * target_sum / sum_h
            guesses.append(h_block)
        return guesses
    
    initial_guesses = generate_diverse_guesses(5) + generate_block_guesses(5)
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=100,
            mutation=(0.5, 0.8),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 20:
        try:
            result = optimize.minimize(

                best_c5 = objective(result.x)
                best_h = result.x
        except:

    return best_h, best_c5, n_points
