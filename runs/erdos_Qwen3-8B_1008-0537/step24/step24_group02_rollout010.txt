import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_pattern():
        num_ones = int(target_sum)
        spacing = 2.0 / num_ones
        positions = np.arange(0, 2, spacing)
        indices = np.round(positions / dx).astype(int)
        unique_indices = np.unique(indices)
        unique_indices = unique_indices[(unique_indices >= 0) & (unique_indices < n_points)]
        h = np.zeros(n_points)
        h[unique_indices] = 1.0
        while np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        while np.sum(h) > target_sum:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.0
        return h
    
    initial_guesses = [initial_guess]
    for _ in range(4):
        initial_guesses.append(generate_pattern())
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None:
            current_c5 = objective(result)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=300,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,

        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,

                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 100:

            result = optimize.minimize(
