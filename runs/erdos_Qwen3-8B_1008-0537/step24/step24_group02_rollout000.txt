import time

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_pattern():
        num_ones = np.random.randint(10, 25)

            h[idx] += 1.0
            if np.sum(h) > target_sum:
                idx = np.random.choice(np.where(h > 0)[0])
                h[idx] -= 1.0
        while np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] += 0.5
        return h
    
    initial_guesses = [initial_guess, generate_pattern()]
    for _ in range(3):
        initial_guesses.append(generate_pattern())
    
    for guess in initial_guesses:
        try:
            result = optimize.minimize(
                fun=objective,

                method='SLSQP',
                bounds=bounds,

                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
