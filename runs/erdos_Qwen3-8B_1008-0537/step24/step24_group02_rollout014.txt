import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def obj(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx
    
    def cons(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_symmetric():
        num = np.random.randint(15, 30)
        half = (n_points - 1) // 2
        sym = np.zeros(n_points)
        for i in range(num):
            idx = np.random.randint(0, half + 1)
            sym[idx] = 1.0
            if idx < half:
                sym[n_points - 1 - idx] = 1.0
        if np.sum(sym) < target_sum:
            idx = np.random.choice(np.where(sym == 0)[0])
            sym[idx] = 0.5
        return sym
    
    def gen_random():
        n = np.random.randint(15, 30)
        idx = np.random.choice(n_points, size=n, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 0.5
        return h
    
    def optimize_guess(guess):
        try:
            res = optimize.minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=[cons], options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9})
            return res.x if res.success else None
        except:
            return None
    
    initial_guesses = [initial_guess, gen_random(), gen_symmetric()]
    for _ in range(3):
        initial_guesses.append(gen_random())
        initial_guesses.append(gen_symmetric())
    
    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None and obj(res) < best_c5:
            best_c5 = obj(res)
            best_h = res
