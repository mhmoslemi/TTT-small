        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_pattern_spaced():
        m = n_points // 2

        h = np.zeros(n_points)
        h[indices] = 1.0
        sum_h = np.sum(h)
        diff = target_sum - sum_h
        if diff > 0:
            idx = np.random.choice(np.where(h == 0)[0], size=int(diff), replace=False)
            h[idx] = 0.5

            h[idx] = 0.0
