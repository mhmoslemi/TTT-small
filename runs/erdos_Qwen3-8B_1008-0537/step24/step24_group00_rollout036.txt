    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    # Generate a new structured initial guess with evenly spaced 1s
    m = int(target_sum)
    if n_points % 2 == 1:
        m = (n_points // 2) + 1
    positions = np.linspace(0, 2, m, endpoint=False)
    indices = np.round(positions / dx).astype(int)
    indices = np.unique(np.clip(indices, 0, n_points - 1))
    structured_guess_even = np.zeros(n_points)
    structured_guess_even[indices] = 1.0
    # Adjust sum if needed
    sum_val = np.sum(structured_guess_even)
    if abs(sum_val - target_sum) > 1e-6:
        diff = target_sum - sum_val
        zero_idx = np.where(structured_guess_even == 0)[0]
        one_idx = np.where(structured_guess_even == 1)[0]
        if diff > 0:
            if len(zero_idx) > 0:
                structured_guess_even[zero_idx[0]] = 1.0
        else:
            if len(one_idx) > 0:
                structured_guess_even[one_idx[0]] = 0.0
    initial_guesses = [initial_guess, structured_guess_even]
    
    for m in [2, 3, 4, 5, 6, 7]:
        structured_h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            part = int(x * m)
            if part < int(2 * m / 3):
                structured_h[i] = 1.0
        structured_h[-1] = 0.5
        initial_guesses.append(structured_h)
    
    rand_h = np.zeros(n_points)
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    rand_h[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        rand_h[idx] += remaining
    initial_guesses.append(rand_h)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=300,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
