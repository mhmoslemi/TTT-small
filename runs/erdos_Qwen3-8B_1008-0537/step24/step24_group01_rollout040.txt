        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_diverse_guesses(num=5):
        guesses = [initial_guess]
        # Structured alternating 1 and 0 guess
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if diff != 0:
            idx = np.random.choice(np.where(h == 0)[0] if diff > 0 else np.where(h == 1)[0])
            h[idx] += diff if diff > 0 else -diff
        guesses.append(h)
        # Add random guesses
        for _ in range(num - 1):
            if np.random.rand() < 0.3:
                h = np.zeros(n_points)
                h[np.random.choice(n_points, size=15, replace=False)] = 1.0
                if np.sum(h) < target_sum:
                    idx = np.random.choice(np.where(h == 0)[0], size=1)
                    h[idx] = 0.5
                guesses.append(h)
            else:
                h = np.random.uniform(0.4, 0.6, n_points)
                h = h / np.sum(h) * target_sum
                guesses.append(h)
        return guesses
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    initial_guesses = generate_diverse_guesses(10)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=150,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 20:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
