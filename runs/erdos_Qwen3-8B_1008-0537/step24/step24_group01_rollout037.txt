import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_structured_guesses(num=10):
        guesses = []
        for _ in range(num):
            h = np.zeros(n_points)
            k = np.random.randint(1, 4)
            for i in range(n_points // k):
                idx_start = i * k
                idx_end = idx_start + k
                idx_end = min(idx_end, n_points)
                h[idx_start:idx_end] = 1.0
            if np.sum(h) < target_sum:
                idx = np.random.choice(np.where(h == 0)[0])
                h[idx] = 0.5
            guesses.append(h)
        return guesses
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    initial_guesses = generate_structured_guesses(5)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    for _ in range(3):
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=15,
                maxiter=100,
                mutation=(0.7, 0.9),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                    best_c5 = objective(result_slsqp.x)
                    best_h = result_slsqp.x
        except:
            pass
    
    while time.time() - start_time < budget_s - 20:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
