import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_pattern(modifier=1.0):
        num_ones = int(10 + 5 * np.random.rand())
        spacing = n_points // num_ones
        indices = np.arange(num_ones) * spacing
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum * (1 - 1e-5):
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5 * modifier
        return h
    
    initial_guesses = [initial_guess, generate_pattern(0.9), generate_pattern(1.1), generate_pattern(0.8), generate_pattern(1.2)]
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    def evolve_population(pop, max_iter=200):
        pop_size = len(pop)
        for _ in range(max_iter):
            for i in range(pop_size):
                a, b, c = np.random.choice(range(pop_size), 3, replace=False)
                mutant = pop[a] + np.random.rand()*(pop[b] - pop[c])
                mutant = np.clip(mutant, 0, 1)
                mutant[0] = 0.5
                mutant[-1] = 0.5
                if np.sum(mutant) < target_sum * 0.9:
                    idx = np.random.choice(np.where(mutant == 0)[0], size=1)
                    mutant[idx] = 0.5
                if np.sum(mutant) > target_sum * 1.1:
                    idx = np.random.choice(np.where(mutant == 1)[0], size=1)
                    mutant[idx] = 0.5
                fit = objective(mutant)
                if fit < objective(pop[i]):
                    pop[i] = mutant
        return pop
    
    pop_size = 20
    population = [generate_pattern() for _ in range(pop_size)]
    for _ in range(15):
        population = evolve_population(population)
        best_pop = min(population, key=objective)
        if objective(best_pop) < best_c5:
            best_h = best_pop.copy()
            best_c5 = objective(best_pop)
        if time.time() - start_time > budget_s - 50:
            break
    
    best_h = np.clip(best_h, 0, 1)
    best_h[0] = 0.5
    best_h[-1] = 0.5
    if np.sum(best_h) < target_sum:
        idx = np.random.choice(np.where(best_h == 0)[0], size=1)
        best_h[idx] = 0.5
    elif np.sum(best_h) > target_sum:
        idx = np.random.choice(np.where(best_h == 1)[0], size=1)
        best_h[idx] = 0.5
    
    for _ in range(10):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
