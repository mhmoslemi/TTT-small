import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    def generate_structured_guesses():
        guesses = []
        # Single segment
        k = n_points // 2
        h = np.zeros(n_points)
        h[:k] = 1.0
        if n_points % 2:
            h[-1] = 0.5
        guesses.append(h)
        # Two segments
        k1 = n_points // 3
        h = np.zeros(n_points)
        h[:k1] = 1.0
        h[k1 * 2:(k1 * 2)+k1] = 1.0
        if n_points % 2:
            h[-1] += (target_sum - np.sum(h)) * 2
        guesses.append(h)
        # Random single segment
        start = np.random.randint(0, n_points)
        k = n_points // 2
        h = np.zeros(n_points)
        h[start:start+k] = 1.0
        if n_points % 2:
            h[-1] = 0.5
        guesses.append(h)
        return guesses
    
    structured_guesses = generate_structured_guesses()
    # Random guesses
    random_guesses = []
    for _ in range(3):
        h = np.zeros(n_points)
        idxs = np.random.choice(n_points, size=int(target_sum), replace=False)
        h[idxs] = 1.0
        if n_points % 2:
            extra = target_sum - np.sum(h)
            idx = np.random.choice(n_points, size=1, replace=False)
            h[idx] += extra
        random_guesses.append(h)

            best_c5 = objective(result)
            best_h = result

            break
