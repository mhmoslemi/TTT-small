    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=[{'type':'eq', 'fun':constraint_sum}], options={'maxiter':250, 'ftol':1e-10, 'eps':1e-9})
            return res.x if res.success else None
        except:
            return None
    
    def generate_pattern():
        K = np.random.randint(15, 30)
        idxs = np.arange(K) * (n_points // K)
        h = np.zeros(n_points)
        h[idxs] = 1.0
        if np.sum(h) < target_sum:
            z = np.random.choice(np.where(h==0)[0], 1)
            h[z] = 0.5
        return h
    
    def generate_structured(K):
        pos = np.linspace(0, 2, K, endpoint=False) + 0.5*(2/K)
        idxs = np.round(pos * (n_points/2)).astype(int)
        idxs = np.clip(idxs, 0, n_points-1)
        h = np.zeros(n_points)
        h[idxs] = 1.0
        sm = np.sum(h)
        if sm > target_sum:
            over = int(np.ceil(sm - target_sum))
            for _ in range(over):
                i = np.random.randint(n_points)
                if h[i] == 1:
                    h[i] = 0.5
        elif sm < target_sum:
            need = int(np.ceil(target_sum - sm))
            for _ in range(need):
                i = np.random.randint(n_points)
                if h[i] == 0:
                    h[i] = 0.5
        return h
    
    initial_guesses = [initial_guess, generate_pattern(), generate_structured(23), generate_structured(22)]
    
    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None and objective(res) < best_c5:
            best_c5 = objective(res)
            best_h = res
    
    try:
        res_de = optimize.differential_evolution(objective, bounds, popsize=150, maxiter=300, mutation=(0.7, 0.9), recombination=0.8, tol=1e-10, vectorized=True, seed=seed)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds, constraints=[{'type':'eq', 'fun':constraint_sum}], options={'maxiter':250, 'ftol':1e-10, 'eps':1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 100:
        try:
            res = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=[{'type':'eq', 'fun':constraint_sum}], options={'maxiter':100, 'ftol':1e-10, 'eps':1e-9})
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        except:
            break
    
    return best_h, best_c5, n_points
