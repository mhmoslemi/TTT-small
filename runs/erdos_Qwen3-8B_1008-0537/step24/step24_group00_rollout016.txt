import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def obj(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def cons(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            res = optimize.minimize(
                fun=obj,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': cons}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            return res.x if res.success else None
        except:
            return None
    
    def generate_structured_h(m, threshold):
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            part = int(x * m)
            if part < threshold * m:
                h[i] = 1.0
        h[-1] = 0.5
        return h
    
    initial_guesses = [initial_guess]
    
    for m in range(1, 11):
        if m % 2 == 0:
            threshold = 0.7
        else:
            threshold = 0.6
        structured_h = generate_structured_h(m, threshold)
        initial_guesses.append(structured_h)
    
    for _ in range(5):
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            h[idx] += remaining
        initial_guesses.append(h)
    
    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None and obj(res) < best_c5:
            best_c5 = obj(res)
            best_h = res
    
    try:
        result_de = optimize.differential_evolution(
            fun=obj,
            bounds=bounds,
            popsize=150,
            maxiter=300,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            res = optimize.minimize(
                fun=obj,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': cons}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if res.success and obj(res.x) < best_c5:
                best_c5 = obj(res.x)
                best_h = res.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            res = optimize.minimize(
                fun=obj,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': cons}],
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if res.success and obj(res.x) < best_c5:
                best_c5 = obj(res.x)
                best_h = res.x
        except:
            break
    
    return best_h, best_c5, n_points
