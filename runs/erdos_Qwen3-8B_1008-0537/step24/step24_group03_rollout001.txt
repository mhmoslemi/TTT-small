import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def generate_greedy_pattern(num_blocks=4):
        h = np.zeros(n_points)
        block_size = n_points // num_blocks
        for i in range(num_blocks):
            start = i * block_size
            end = start + block_size
            h[start:end] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_random_pattern():
        h = np.random.rand(n_points)
        h = (h / np.sum(h)) * target_sum
        return h
    
    def generate_bounded_pattern():
        h = np.zeros(n_points)
        ones = np.random.choice(n_points, size=int(target_sum), replace=False)
        h[ones] = 1.0
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = generate_greedy_pattern()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None

    ]
    
    for guess in initial_guesses:

            best_c5 = objective(result)
            best_h = result

        result_de = optimize.differential_evolution(
            fun=objective,

            tol=1e-10,
            vectorized=True,

        )

            result_slsqp = optimize.minimize(

                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
