import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_pattern():
        num_ones = np.random.randint(10, 35)
        spacing = n_points // num_ones
        indices = np.arange(num_ones) * spacing
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_random():
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * target_sum
        return h
    
    initial_guesses = [initial_guess.copy(), generate_pattern(), generate_random()]
    for _ in range(3):
        initial_guesses.append(generate_pattern())
        initial_guesses.append(generate_random())
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    def differential_evolution():
        try:
            result = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=50,
                maxiter=100,
                mutation=(0.5, 0.8),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result.x,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                    best_c5 = objective(result_slsqp.x)
                    best_h = result_slsqp.x
            return best_c5
        except:
            return best_c5
    
    best_c5 = differential_evolution()
    
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
