import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_guesses(num=15):
        guesses = [initial_h_values]
        for _ in range(num):
            if np.random.rand() < 0.5:
                idxs = np.random.choice(n_points, size=20, replace=False)
                h = np.zeros(n_points)
                h[idxs] = 1.0
                if np.sum(h) < target_sum:
                    idx = np.random.choice(np.where(h == 0)[0], size=1)
                    h[idx] = 0.5
                guesses.append(h)
            else:
                h = np.random.uniform(0.4, 0.6, n_points)
                h = h / np.sum(h) * target_sum
                guesses.append(h)
        return guesses

    initial_guesses = generate_guesses(15)

    for guess in initial_guesses:
        res = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=250,
            mutation=(0.6, 0.9),

            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            res_slsqp = optimize.minimize(
                fun=objective,

                method='SLSQP',

                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
