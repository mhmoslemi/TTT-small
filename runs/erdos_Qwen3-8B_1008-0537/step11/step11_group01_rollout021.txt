    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)

    initial_guesses = [initial_h_values]

    h_log = np.array(initial_h_values)
    idx = np.argsort(np.log(np.abs(h_log)))
    h_log_sorted = h_log[idx].copy()
    h_log_sorted[::3] = 1.0
    h_log_sorted[1::3] = 0.5
    h_log_sorted[2::3] = 0.0
    h_log_sorted = np.clip(h_log_sorted, 0, 1)
    h_log_sorted = h_log_sorted * target_sum / np.sum(h_log_sorted)
    initial_guesses.append(h_log_sorted)

    def generate_regular_guess(h):
        n = len(h)
        regular = np.zeros(n)
        for i in range(n):
            if i % 3 == 0:
                regular[i] = 1.0
        regular = np.clip(regular * target_sum / np.sum(regular), 0, 1)
        return regular
    initial_guesses.append(generate_regular_guess(initial_h_values))

    def generate_random_guess(n):
        h = np.random.rand(n)
        h = np.clip(h * target_sum / np.sum(h), 0, 1)
        return h
    initial_guesses.append(generate_random_guess(n_points))

    start_time = time.time()
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        if time.time() - start_time > budget_s - 50:
            break

    return best_h, best_c5, n_points
