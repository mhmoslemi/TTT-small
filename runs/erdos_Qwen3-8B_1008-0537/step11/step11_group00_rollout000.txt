import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.05):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    def objective_with_penalty(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        return c5 + 1e3 * constraint_violation
    
    def optimize_with_slsqp(x0, penalty_scale=1e3, maxiter=1000, ftol=1e-10):
        try:
            result = optimize.minimize(
                fun=objective_with_penalty,
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-9}
            )
            if result.success:
                return result.x
        except:
            pass
        return x0
    
    # Initial optimization with penalty
    best_h = optimize_with_slsqp(best_h, penalty_scale=1e2, maxiter=1000, ftol=1e-10)
    best_c5 = objective(best_h)
    
    # Perturb and refine
    for _ in range(10):
        if time.time() - start_time >= budget_s - 10:

        perturbed = generate_perturbation(best_h, scale=0.08)

        current_c5 = objective(refined)
        if current_c5 < best_c5:
