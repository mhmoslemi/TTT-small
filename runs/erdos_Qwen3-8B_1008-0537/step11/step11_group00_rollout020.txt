import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        return c5 + 1e5 * constraint_violation
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            disp=False,
            initial_population=initial_guess
        )
        if result_de.success:
            current_c5 = objective(result_de.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_de.x
    except:
        pass
    
    try:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-11, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    except:
        pass
    
    return best_h, best_c5, n_points
