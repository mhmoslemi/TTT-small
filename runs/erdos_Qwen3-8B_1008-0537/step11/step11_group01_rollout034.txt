import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    try:
        n_points = len(initial_h_values)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)
        half = n_points // 2
        if n_points % 2 == 0:
            h[:half] = 1
            h[half:] = 0
        else:
            h[:half] = 1
            h[half] = 0.5
            h[half+1:] = 0
        initial_h_values = h
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_uniform_guess(h):
        h_uniform = np.zeros_like(h)
        idx = np.argsort(np.arange(n_points))
        cnt = 0
        for i in range(n_points):
            if cnt < target_sum:
                h_uniform[i] = 1.0
                cnt += 1
            else:
                h_uniform[i] = 0.0
        h_uniform[-1] = max(0, target_sum - np.sum(h_uniform))
        return h_uniform
    
    def generate_log_guess(h):
        log_vals = np.log(np.abs(h))
        idx = np.argsort(log_vals)
        h_sorted = h[idx]
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted
    
    def generate_uniform_initial_guess():
        h = np.zeros(n_points)
        cnt = 0
        for i in range(n_points):
            if cnt < target_sum:
                h[i] = 1.0
                cnt += 1
            else:
                h[i] = 0.0
        h[-1] = max(0, target_sum - np.sum(h))
        return h
    
    initial_guess_uniform = generate_uniform_initial_guess()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e5 * constraint_violation  # Stronger penalty
        return c5 + penalty
    
    results = []
    
    for guess in [initial_guess, initial_guess_uniform]:
        try:
            result_de = optimize.differential_evolution(
                fun=objective_de,
                bounds=bounds,
                popsize=25,
                maxiter=150,
                mutation=(0.5, 1.0),
                recombination=0.9,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
