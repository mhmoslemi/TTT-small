import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    best_h = initial_guess.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_sine_guess(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.sin(np.pi * x / 2)
        h = np.clip(h, 0, 1)
        h = h / np.max(h)
        return h
    
    def generate_padded_guess(h):
        return np.concatenate([h[::2], h[1::2]])
    
    def generate_log_guess(h):
        idx = np.argsort(np.log(np.abs(h)))
        h_sorted = h[idx].copy()
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted
    
    sin_guess = generate_sine_guess(n_points)
    sin_guess = (sin_guess * target_sum) / np.sum(sin_guess)
    
    def optimize_start(point):
        res = optimize.minimize(
            fun=objective,
            x0=point,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 800, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return res
    
    points = [initial_guess, generate_padded_guess(initial_guess), generate_log_guess(initial_guess), sin_guess]
    
    for p in points:
        res = optimize_start(p)
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        res = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    return best_h, best_c5, n_points
