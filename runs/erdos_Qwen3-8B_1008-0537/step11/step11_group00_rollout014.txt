import time

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum

        perturbed = perturbed / np.sum(perturbed) * target_sum
        return perturbed
    
    def optimize_with_method(method, x0, **kwargs):
        try:
            result = optimize.minimize(

                x0=x0,
                method=method,
                bounds=bounds,

            )
