        h = np.array(h_values)

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 600, 'ftol': 1e-11, 'eps': 1e-9}
        )
        return result.x if result.success else guess
    
    best_h = initial_h_values.copy()
