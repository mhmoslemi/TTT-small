import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 45
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Generate structured initial guess: alternate 1 and 0, with 0.5 in the middle
    h = np.zeros(n_points)
    h[::2] = 1.0
    h[1::2] = 0.0
    h[n_points // 2] = 0.5
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Use DE with adjusted parameters for better exploration
    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=40,
        maxiter=300,
        mutation=(0.7, 1.3),
        recombination=0.9,
        tol=1e-10,
        vectorized=True,
        seed=seed
    )
    
    if result_de.success:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    # Perturbations with controlled noise to escape local optima
    for _ in range(2):
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success and objective(new_guess.x) < best_c5:
            best_c5 = objective(new_guess.x)
            best_h = new_guess.x
    
    # Final refinement
    final_result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    
    if final_result.success and objective(final_result.x) < best_c5:
        best_c5 = objective(final_result.x)
        best_h = final_result.x
    
    return best_h, best_c5, n_points
