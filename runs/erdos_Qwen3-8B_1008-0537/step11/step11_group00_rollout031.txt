        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_initial(n_points):
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1
        return h
    
    def generate_perturbation(h, scale=0.05):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    def optimize_with_penalty(h0):
        result = optimize.minimize(
            fun=lambda x: objective(x) + 1e5 * abs(np.sum(x) - target_sum),
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success:
            return result.x
        return h0
    
    def optimize_without_penalty(h0):
        result = optimize.minimize(
            fun=objective,
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success:

        return h0

    best_h = optimize_with_penalty(initial_guess)
    best_c5 = objective(best_h)
