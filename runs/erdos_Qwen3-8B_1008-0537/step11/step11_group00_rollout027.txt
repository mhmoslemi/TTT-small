import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def penalty(obj_val, h):
        return obj_val + 1e4 * (np.sum(h) - target_sum) ** 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            args=(),
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-12,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    except:
        pass
    
    try:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    except:
        pass
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = np.clip(best_h + np.random.normal(0, 0.03, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
