import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_regular_guess(n):
        m = n // 2
        pattern = np.zeros(n, dtype=float)
        pattern[:m] = 1.0
        if n % 2 == 1:
            pattern[m] = 0.5
        return pattern
    
    initial_guesses = [
        initial_guess.copy(),
        initial_guess.copy()[::2], 
        initial_guess.copy()[::3], 
        generate_regular_guess(n_points)
    ]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e4 * constraint_violation
        return c5 + penalty
    
    results = []
    for guess in initial_guesses:
        try:
            result = optimize.differential_evolution(
                fun=objective_de,
                bounds=bounds,
                popsize=30,
                maxiter=150,
                mutation=(0.5, 1.0),
                recombination=0.9,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result.success:
                results.append(result)
        except:
            pass
    
    for res in results:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=res.x,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    def perturb(h, scale=0.002):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 30:
            break
        perturbed = perturb(best_h)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
