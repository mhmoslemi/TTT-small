import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_checkerboard_guess(h):
        h_sorted = np.sort(h)
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted
    
    def generate_sine_guess(h):
        h_sorted = np.sort(h)
        h_sorted = 0.5 * (1 + np.sin(np.linspace(0, np.pi, n_points)))
        return h_sorted
    
    def generate_spread_guess(h):
        h_sorted = np.sort(h)
        h_sorted = np.interp(range(n_points), range(n_points), np.linspace(0, 1, n_points))
        return h_sorted
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        return c5 + 1e5 * constraint_violation
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=25,
            maxiter=500,
            mutation=(0.9, 1.1),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 700, 'ftol': 1e-12, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass

        options={'maxiter': 700, 'ftol': 1e-12, 'eps': 1e-9}

    if result_slsqp_initial.success:

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    result_slsqp_checkerboard = optimize.minimize(
        fun=objective,
        x0=generate_checkerboard_guess(initial_guess),
        method='SLSQP',

        options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
