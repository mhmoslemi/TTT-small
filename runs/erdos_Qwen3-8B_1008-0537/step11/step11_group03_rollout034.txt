    # Final refinement
    best_h = optimize_with_guess(best_h)
    best_c5 = objective(best_h)
    
    while time.time() - start_time < budget_s - 50:
        best_h = evaluate_perturbations(best_h, steps=5)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
