import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Assume initial_h_values is provided, fallback to shuffled binary if not available
    try:
        h_initial = np.array(initial_h_values, dtype=float)
    except:
        n_points = 45
        h_initial = np.zeros(n_points)
        h_initial[:n_points//2] = 1.0
        np.random.shuffle(h_initial)
    
    n_points = len(h_initial)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.05):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    try:
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    except:
        pass
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.03)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
