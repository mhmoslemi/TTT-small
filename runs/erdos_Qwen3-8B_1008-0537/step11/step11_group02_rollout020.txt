import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_log_guess(h):
        h_sorted = np.sort(h)
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted
    
    def generate_binary_guess(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if 0.25 <= x < 0.75 or 1.25 <= x < 1.75:
                h[i] = 1.0
        return h
    
    def generate_uniform_guess(n_points):
        return np.full(n_points, 0.5)
    
    def optimize_from_guess(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x
        return x0
    
    def perturb_guess(h):
        perturbed = h + np.random.normal(0, 0.0015, size=n_points)
        return np.clip(perturbed, 0, 1)
    
    def run_initial_guesses():
        nonlocal best_h, best_c5
        guesses = [
            initial_guess,
            generate_log_guess(initial_guess),
            generate_binary_guess(n_points),
            generate_uniform_guess(n_points)
        ]
        for guess in guesses:
            result = optimize_from_guess(guess)
            current_c5 = objective(result)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result
    
    def run_perturbations():
        nonlocal best_h, best_c5
        while time.time() - start_time < budget_s - 50:
            perturbed = perturb_guess(best_h)
            result = optimize_from_guess(perturbed)
            current_c5 = objective(result)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result
    
    try:
        run_initial_guesses()
    except:
        pass
    
    try:
        run_perturbations()
    except:
        pass
    
    return best_h, best_c5, n_points
