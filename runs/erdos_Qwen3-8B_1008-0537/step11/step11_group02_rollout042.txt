    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        return c5 + 1e4 * constraint_violation
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=15,
            maxiter=200,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
