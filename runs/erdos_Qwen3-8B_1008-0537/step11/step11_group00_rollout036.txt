        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
    )
    if result_slsqp.success:
        current_h = result_slsqp.x
        current_c5 = objective_no_penalty(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Perturbations
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = np.clip(best_h + np.random.normal(0, 0.03, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective_no_penalty,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if result.success:
            current_h = result.x
            current_c5 = objective_no_penalty(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    return best_h, best_c5, n_points
