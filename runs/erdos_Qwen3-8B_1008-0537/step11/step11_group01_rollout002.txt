import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_padded_guess(h):
        h_padded = np.zeros_like(h)
        h_padded[::2] = h[::2]
        h_padded[1::2] = h[1::2]
        return h_padded
    
    def generate_log_spaced_guess(h):
        idx = np.argsort(np.log(np.abs(h)))
        h_sorted = h[idx].copy()
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted

    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e3 * constraint_violation  # Reduced penalty
        return c5 + penalty
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=50,
            maxiter=300,
            mutation=(0.7, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass

    result_slsqp_initial = optimize.minimize(

        x0=initial_guess,
        method='SLSQP',

    result_slsqp_padded = optimize.minimize(

    if result_slsqp_padded.success:
        current_c5 = objective(result_slsqp_padded.x)

    log_guess = generate_log_spaced_guess(initial_guess)
    result_slsqp_log = optimize.minimize(
