    start_time = time.time()
    
    def generate_alternating_guess(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 2 == 0 else 0.0
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum > target_sum:
            h[0] = 0.5
        else:
            h[0] = 0.5
        return h
    
    def generate_rectangular_guess(n_points):
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        return h
    
    def generate_uneven_guess(n_points):
        h = np.zeros(n_points)
        h[::3] = 1.0
        h[1::3] = 0.5
        h[2::3] = 0.0
        return h
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        constraint_violation = abs(np.sum(h) - target_sum)
        return c5 + 1e5 * constraint_violation
    
    initial_guesses = [
        initial_guess,
        generate_alternating_guess(n_points),
        generate_rectangular_guess(n_points),
        generate_uneven_guess(n_points)
    ]
    
    results = []
    for guess in initial_guesses:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=20,
            maxiter=150,
            mutation=(0.9, 1.1),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                results.append((result_slsqp.x, objective(result_slsqp.x)))
    
    if results:
        best_h, best_c5 = min(results, key=lambda x: x[1])
    
    pad_guess = np.zeros_like(best_h)
    pad_guess[::2] = best_h[::2]
    pad_guess[1::2] = best_h[1::2]
    result_padded = optimize.minimize(
        fun=objective,
        x0=pad_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_padded.success:
        current_c5 = objective(result_padded.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_padded.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0015, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
