            break
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        res = optimize.minimize(
            fun=objective, x0=perturbed, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    return best_h, best_c5, n_points
