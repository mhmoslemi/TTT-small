        h = np.where(h < 0.5, 0, 1)
        sum_h = np.sum(h)
        adjustment = n_points / 2 - sum_h
        if adjustment > 0:
            indices = np.where(h == 0)[0]
            if indices.size > 0:
                h[indices[0]] = 1
            else:
                h[-1] = 1
        else:
            indices = np.where(h == 1)[0]
            if indices.size > 0:
                h[indices[0]] = 0
            else:
                h[-1] = 0
        return h
    
    def generate_padded_guess(h):
        h_padded = np.zeros_like(h)
        h_padded[::2] = h[::2]
        h_padded[1::2] = h[1::2]
        return h_padded
    
    def generate_log_spaced_guess(h):
        idx = np.argsort(np.log(np.abs(h)))
        h_sorted = h[idx].copy()
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted
    
    initial_guesses = [
        initial_guess,
        generate_alternating_guess(n_points),
        generate_random_guess(n_points),
        generate_padded_guess(best_h),
        generate_log_spaced_guess(initial_guess),
    ]
    
    for guess in initial_guesses:
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=25,
                maxiter=200,
                mutation=(0.5, 1.0),
                recombination=0.9,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    padded_guess = generate_padded_guess(best_h)
    result_slsqp_padded = optimize.minimize(
        fun=objective,
        x0=padded_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_padded.success:
        current_c5 = objective(result_slsqp_padded.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_padded.x
    
    log_guess = generate_log_spaced_guess(initial_guess)
    result_slsqp_log = optimize.minimize(
        fun=objective,
        x0=log_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_log.success:
        current_c5 = objective(result_slsqp_log.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_log.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
