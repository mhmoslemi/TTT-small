import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_padded_guess(h):
        h_padded = np.zeros_like(h)
        h_padded[::2] = h[::2]
        h_padded[1::2] = h[1::2]
        return h_padded
    
    def generate_log_spaced_guess(h):
        h_sorted = np.sort(h)
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted
