    # Function to run DE and SLSQP on a given initial guess
    def optimize_guess(start_guess):
        # Differential Evolution
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=25,
            maxiter=150,
            mutation=(0.6, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                return result_slsqp.x, current_c5
        # Fallback to initial guess
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=start_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            return result_slsqp.x, current_c5
        return start_guess, objective(start_guess)
    
    # Optimize initial guess
    best_h, best_c5 = optimize_guess(initial_guess)
    
    # Optimize regular pattern guess
    reg_h, reg_c5 = optimize_guess(regular_h)
    if reg_c5 < best_c5:
        best_h, best_c5 = reg_h, reg_c5
    
    # Perturb and refine
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_h, best_c5 = new_guess.x, current_c5
    
    return best_h, best_c5, n_points
