import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=100,
            maxiter=150,

            recombination=0.8,
            tol=1e-10,
            vectorized=True

        if result_de.success:

                method='SLSQP',

                options={'maxiter': 600, 'ftol': 1e-12, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:

    if result_slsqp_initial.success:

    random_guess = np.zeros(n_points)

        random_guess[idx] += remaining
    result_slsqp_random = optimize.minimize(
