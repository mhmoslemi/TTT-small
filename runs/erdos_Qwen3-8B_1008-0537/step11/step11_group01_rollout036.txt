import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    result_de = optimize.differential_evolution(
        fun=objective_de,
        bounds=bounds,
        popsize=40,
        maxiter=250,
        mutation=(0.8, 1.2),
        recombination=0.8,
        tol=1e-8,
        vectorized=True,
        seed=seed
    )
    
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=result_de.x,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    
    if result_slsqp.success:
        current_c5 = objective(result_slsqp.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
    
    def generate_alternating_guess(h):
        idx = np.argsort(np.abs(h))
        h_sorted = h[idx].copy()
        step = n_points // 3
        h_sorted[::step] = 1.0
        h_sorted[1::step] = 0.5
        h_sorted[2::step] = 0.0
        return h_sorted
    
    alt_guess = generate_alternating_guess(initial_guess)
    result_slsqp_alt = optimize.minimize(
        fun=objective,
        x0=alt_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    
    if result_slsqp_alt.success:
        current_c5 = objective(result_slsqp_alt.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_alt.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,

            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}

                best_h = new_guess.x
    
    return best_h, best_c5, n_points
