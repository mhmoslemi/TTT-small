import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Symmetric initial guess
    symmetric_guess = initial_h_values.copy()
    for i in range(n_points // 2):
        symmetric_guess[i] = symmetric_guess[n_points - 1 - i]
    
    initial_guesses = [initial_h_values.copy(), symmetric_guess.copy()]
    initial_guesses.append(np.zeros(n_points))
    initial_guesses.append(np.ones(n_points))
    
    best_h = initial_h_values.copy()
    best_c5 = 1.0
    start_time = time.time()
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        guess = np.array(guess, dtype=float)
        guess = np.clip(guess, 0, 1)
        guess = np.round(guess, 5)
        if np.abs(np.sum(guess) - target_sum) > 1e-5:
            continue
        
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=30,
                maxiter=300,
                mutation=(0.9, 1.1),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = np.round(perturbed, 5)
        if np.abs(np.sum(perturbed) - target_sum) > 1e-5:
            continue
        
        try:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
    
    return best_h, best_c5, n_points
