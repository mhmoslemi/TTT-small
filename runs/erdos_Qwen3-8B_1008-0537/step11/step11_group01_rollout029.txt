        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e3 * constraint_violation  # Reduced penalty for better exploration
        return c5 + penalty
    
    # Generate additional initial guesses
    additional_guesses = []
    k = n_points // 2
    # Binary random guess
    positions = np.random.choice(n_points, size=k, replace=False)
    binary_guess = np.zeros(n_points)
    binary_guess[positions] = 1.0
    additional_guesses.append(binary_guess)
    # Even pattern guess
    even_guess = np.zeros(n_points)
    even_guess[::2] = 1.0
    additional_guesses.append(even_guess)
    # Interleaved guess
    interleaved_guess = np.zeros(n_points)
    interleaved_guess[::3] = 1.0
    interleaved_guess[1::3] = 0.5
    interleaved_guess[2::3] = 0.0
    additional_guesses.append(interleaved_guess)
    
    # Combine initial guesses
    initial_pop = [initial_guess] + additional_guesses
    # Run DE with initial population
    result_de = optimize.differential_evolution(
        fun=objective_de,
        bounds=bounds,
        popsize=30,
        maxiter=150,
        mutation=(0.7, 1.0),
        recombination=0.9,
        tol=1e-10,
        vectorized=True,
        seed=seed,
        x0=initial_pop
    )
    if result_de.success:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
