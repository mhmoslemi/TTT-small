        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_uniform_initial():
        return np.full(n_points, 0.5)
    
    def optimize_with_slsqp(x0, maxiter=300, ftol=1e-10, eps=1e-9):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': ftol, 'eps': eps}
        )
        if result.success:
            return result.x
        return x0
    
    # Initial guess from initial_h_values
    best_h = optimize_with_slsqp(best_h, maxiter=300)
    best_c5 = objective(best_h)
    
    # Generate uniform initial guess
    uniform_guess = generate_uniform_initial()
    uniform_result = optimize_with_slsqp(uniform_guess, maxiter=300)
    current_c5 = objective(uniform_result)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = uniform_result
    
    # Random guess
    def random_guess():
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        guess = np.zeros(n_points)
        guess[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            guess[idx] += remaining
        return guess
    
    random_guess_result = optimize_with_slsqp(random_guess(), maxiter=300)
    current_c5 = objective(random_guess_result)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = random_guess_result
    
    # Differential Evolution with enhanced parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=300,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            de_result = optimize_with_slsqp(result_de.x, maxiter=300)
            current_c5 = objective(de_result)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = de_result
    except:
        pass
    
    # Local refinement with perturbations
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0015, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize_with_slsqp(perturbed, maxiter=200)
        current_c5 = objective(new_guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_guess
    
    return best_h, best_c5, n_points
