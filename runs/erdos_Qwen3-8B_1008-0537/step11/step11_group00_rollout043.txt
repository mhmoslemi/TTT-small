    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.05):
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    def objective_with_penalty(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        return c5 + 1e6 * constraint_violation
    
    try:
        result_cobyla = optimize.minimize(
            fun=objective_with_penalty,
            x0=initial_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'disp': False}
        )
        if result_cobyla.success:
            current_c5 = objective(result_cobyla.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_cobyla.x
    except:
        pass
    
    try:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 400, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    except:
        pass
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.03)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
