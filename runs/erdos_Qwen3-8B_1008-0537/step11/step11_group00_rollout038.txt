        current_c5 = objective(current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current
    
    # Generate random initial guesses
    for _ in range(2):
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        initial_guesses.append(h)
    
    for guess in initial_guesses[3:]:
        current = optimize_once(guess)
        current_c5 = objective(current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current
    
    # Perturb the best solution and optimize
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.03)
        current = optimize_once(perturbed)
        current_c5 = objective(current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current
    
    return best_h, best_c5, n_points
