import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Improved differential evolution with more iterations and population
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=60,
            maxiter=300,
            mutation=(0.8, 1.2),
            recombination=0.9,
            tol=1e-10,
            vectorized=False
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 600, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    # Random even distribution guess
    random_guess = np.zeros(n_points)
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    indices = np.linspace(0, n_points - 1, num_ones, endpoint=False).astype(int)
    random_guess[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        random_guess[idx] += remaining
    result_slsqp_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 400, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_random.success:
        current_c5 = objective(result_slsqp_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_random.x
    
    while time.time() - start_time < budget_s - 100:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
