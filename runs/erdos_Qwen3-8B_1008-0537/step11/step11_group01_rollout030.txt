    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate alternative initial guess: even spaced
    def generate_even_guess(n_points):
        n_ones = n_points // 2
        if n_points % 2 == 1:
            n_ones += 1
        dx_val = 2.0 / n_points
        pos = np.linspace(0, 2, n_ones, endpoint=False)
        h = np.zeros(n_points)
        for i, p in enumerate(pos):
            idx = int(p / dx_val)
            if idx >= n_points:
                idx = n_points - 1
            h[idx] = 1.0
        return h
    
    even_guess = generate_even_guess(n_points)
    initial_guesses = [initial_guess, even_guess]
    
    for init in initial_guesses:
        # Run differential evolution
        def objective_de(h_values):
            h = np.array(h_values)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            constraint_violation = abs(np.sum(h) - target_sum)
            penalty = 1e5 * constraint_violation
            return c5 + penalty
        
        try:
            result_de = optimize.differential_evolution(
                fun=objective_de,
                bounds=bounds,
                popsize=50,
                maxiter=100,
                mutation=(0.5, 1.0),
                recombination=0.9,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass
    
    # Perturb and refine
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
