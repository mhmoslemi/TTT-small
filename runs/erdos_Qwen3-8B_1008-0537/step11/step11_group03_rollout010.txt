import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=30,
            maxiter=300,
            mutation=(0.7, 1.0),
            recombination=0.5,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0015, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)

            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )

    return best_h, best_c5, n_points
