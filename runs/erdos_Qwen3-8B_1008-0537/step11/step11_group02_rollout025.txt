            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            results.append((new_guess.x, objective(new_guess.x)))
    
    best_h, best_c5 = min(results, key=lambda x: x[1])
    
    return best_h, best_c5, n_points
