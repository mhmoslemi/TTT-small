                best_c5 = current_c5
                best_h = result.x
    except:
        pass
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h)
        try:
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options=options
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            continue
    
    return best_h, best_c5, n_points
