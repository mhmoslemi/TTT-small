        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_log.success:
        current_c5 = objective(result_slsqp_log.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_log.x
    
    checkerboard_guess = generate_checkerboard_guess(n_points)
    result_slsqp_checkerboard = optimize.minimize(
        fun=objective,
        x0=checkerboard_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_checkerboard.success:
        current_c5 = objective(result_slsqp_checkerboard.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_checkerboard.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.003, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
