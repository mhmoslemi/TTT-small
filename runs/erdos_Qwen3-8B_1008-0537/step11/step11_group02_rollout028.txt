import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
        n_points = len(h_initial)
    else:
        # Generate a default starting pattern if no initial_h_values available
        n_points = 45
        h_initial = np.zeros(n_points)
        h_initial[::2] = 1.0
        h_initial[1::2] = 0.5
        h_initial = np.clip(h_initial, 0, 1)
    
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_h = h_initial.copy()
    best_c5 = np.inf
    start_time = time.time()

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Generate structured initial guesses
    guesses = [h_initial.copy()]
    # Binary alternating pattern
    guess_binary = np.zeros(n_points)
    guess_binary[::2] = 1.0
    guess_binary[1::2] = 0.0
    guesses.append(guess_binary)
    # Ternary spaced pattern
    guess_ternary = np.zeros(n_points)
    guess_ternary[::3] = 1.0
    guess_ternary[1::3] = 0.5
    guess_ternary[2::3] = 0.0
    guesses.append(guess_ternary)

    # Evaluate all initial guesses
    for h in guesses:
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # DE with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=40,
            maxiter=400,
            mutation=(0.6, 1.2),
            recombination=0.8,
            tol=1e-9,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_h = result_slsqp.x
                    best_c5 = current_c5
    except:
        pass

    # Local refinement
    result_slsqp_ref = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
    if result_slsqp_ref.success:
        current_c5 = result_slsqp_ref.fun
        if current_c5 < best_c5:
            best_h = result_slsqp_ref.x
            best_c5 = current_c5

    # Perturbation loop
    for _ in range(20):
        if time.time() - start_time > budget_s - 10:
            break
        perturbed = best_h + np.random.normal(0, 0.0015, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result_slsqp_pert = optimize.minimize(
            fun=objective,

            method='SLSQP',
