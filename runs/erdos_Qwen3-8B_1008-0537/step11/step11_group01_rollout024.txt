        if ones_count < num_ones:
            binary[-1] = 1.0
        else:
            binary[-1] = 0.0
        return binary

    def generate_padded_guess(h):
        h_padded = np.zeros_like(h)
        h_padded[::2] = h[::2]
        h_padded[1::2] = h[1::2]
        return h_padded

    def generate_log_spaced_guess(h):
        idx = np.argsort(np.log(np.abs(h)))
        h_sorted = h[idx].copy()
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted

    initial_guesses = [
        initial_guess,
        generate_binary_guess(n_points),
        generate_padded_guess(initial_guess),
        generate_log_spaced_guess(initial_guess)
    ]
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x

    return best_h, best_c5, n_points
