            constraints=[{'type': 'eq', 'fun': constraint_sum}] + symmetry_constraints,
            options={'maxiter': 500, 'ftol': 1e-11, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    except:
        pass
    
    try:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}] + symmetry_constraints,
            options={'maxiter': 300, 'ftol': 1e-11, 'eps': 1e-9}

        if result_slsqp.success:

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    except:
        pass
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.03)
        result = optimize.minimize(
            fun=objective,

            method='SLSQP',
            bounds=bounds,

        if result.success:

            if current_c5 < best_c5:

                best_h = result.x
    
    return best_h, best_c5, n_points
