import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h, scale=0.05, seed=seed):
        np.random.seed(seed)
        return np.clip(h + np.random.normal(0, scale, size=n_points), 0, 1)
    
    def optimize_with_method(method, x0, **kwargs):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=x0,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={**kwargs}
            )
            if result.success:
                return result.x
        except:
            pass
        return None
    
    # Initial optimization with COBYLA
    result = optimize_with_method('COBYLA', initial_guess, tol=1e-10, maxiter=300)
    if result is not None:
        current_c5 = objective(result)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result
    
    # Perturbation and further optimization
    for i in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.03, seed=seed + i)
        result = optimize_with_method('SLSQP', perturbed, maxiter=200, ftol=1e-11, eps=1e-9)
        if result is not None:
            current_c5 = objective(result)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result
    
    return best_h, best_c5, n_points
