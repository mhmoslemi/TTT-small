import scipy.optimize as optimize

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_padded_guess(h):
        h_padded = np.zeros_like(h)
        h_padded[::2] = h[::2]
        h_padded[1::2] = h[1::2]
        return h_padded
    
    def generate_log_spaced_guess(h):
        idx = np.argsort(np.log(np.abs(h)))
        h_sorted = h[idx].copy()
        h_sorted[::3] = 1.0
        h_sorted[1::3] = 0.5
        h_sorted[2::3] = 0.0
        return h_sorted

    def objective_de(h_values):

        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e5 * constraint_violation  # Stronger penalty
        return c5 + penalty

        result_de = optimize.differential_evolution(
