import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * 2.0 / n_points
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbations(h, scales):
        perturbations = []
        for scale in scales:
            for _ in range(3):
                perturb = h + np.random.normal(0, scale, size=n_points)
                perturb = np.clip(perturb, 0, 1)
                perturbations.append(perturb)
        return perturbations
    
    scales = [0.03, 0.05, 0.07, 0.1]
    initial_perturbations = generate_perturbations(initial_guess, scales)
    
    def optimize_single(x0):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-11, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                return True
        except:
            return False
        return False
    
    for p in initial_perturbations:
        if time.time() - start_time >= budget_s - 20:
            break
        optimize_single(p)
    
    for _ in range(10):
        if time.time() - start_time >= budget_s - 20:
            break
        perturb = best_h + np.random.normal(0, 0.06, size=n_points)
        perturb = np.clip(perturb, 0, 1)
        optimize_single(perturb)
    
    return best_h, best_c5, n_points
