import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    # Generate a new initial guess with even-indexed ones for sparse coverage
    initial_guess = np.zeros(n_points, dtype=float)
    initial_guess[::2] = 1.0
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        violation = abs(np.sum(h) - target_sum)
        penalty = 1e3 * violation  # Mild penalty to maintain feasibility
        return c5 + penalty
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=25,
            maxiter=150,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass

    result_slsqp = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_c5 = objective(result_slsqp.x)
        if current_c5 < best_c5:

            best_h = result_slsqp.x

    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)

        new_guess = optimize.minimize(
