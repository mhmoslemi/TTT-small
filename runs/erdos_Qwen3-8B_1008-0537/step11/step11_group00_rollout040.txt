        return np.max(corr) * dx

    def objective_with_penalty(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e6 * abs(np.sum(h) - target_sum)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    try:
        result = optimize.differential_evolution(
            objective_with_penalty, bounds,
            strategy='best1bin', maxiter=100, tol=1e-7,
            popsize=15, mutation=(0.5, 1), recombination=0.8,
            disp=False)
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    except:
        pass

    try:
        result = optimize.minimize(
            objective, best_h,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9})
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    except:
        pass

    if time.time() - start_time < budget_s - 10:
        initial_guess = initial_h_values
        try:
            result = optimize.differential_evolution(
                objective_with_penalty, bounds,
                strategy='best1bin', maxiter=50, tol=1e-7,
                popsize=15, mutation=(0.5, 1), recombination=0.8,
                disp=False)
            if result.success:
                current_h = result.x
                current_c5 = objective(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        except:
            pass

    return best_h, best_c5, n_points
