import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_even_guess(n_points):
        m = int(n_points // 2)
        positions = np.linspace(0, n_points - 1, m, endpoint=False)
        positions = np.round(positions).astype(int)
        positions = np.unique(positions)
        guess = np.zeros(n_points)
        guess[positions] = 1.0
        current_sum = np.sum(guess)
        if abs(current_sum - target_sum) > 1e-7:
            if current_sum < target_sum:
                pos = np.where(guess == 0)[0]
                if pos.size > 0:
                    guess[pos[0]] = 1.0
            else:
                pos = np.where(guess == 1)[0]
                if pos.size > 0:
                    guess[pos[0]] = 0.0
        return guess
    
    def generate_random_guess():
        random_guess = np.zeros(n_points)

        return random_guess
    
    initial_guesses = [initial_guess, generate_even_guess(n_points), generate_random_guess()]
    
    for start_guess in initial_guesses:
        try:

                popsize=60,

                tol=1e-9,
                vectorized=True,
                polish=False
            )
            if result_de.success:
                result_slsqp = optimize.minimize(

    random_guess = generate_random_guess()
    result_slsqp_random = optimize.minimize(

    if result_slsqp_random.success:
