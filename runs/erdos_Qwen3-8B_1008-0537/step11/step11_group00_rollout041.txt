    if result_cobyla.success:
        current_c5 = objective(result_cobyla.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_cobyla.x
    
    # Perturbation-based local search
    for _ in range(5):
        if time.time() - start_time >= budget_s - 20:
            break
        perturbed = generate_perturbation(best_h, scale=0.03)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'disp': False, 'maxiter': 100}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
