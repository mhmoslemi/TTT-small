import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    max_iterations = 30
    current_iter = 0

    while current_iter < max_iterations and time.time() - start_time < budget_s:
        current_iter += 1
        noise_scale = 0.01 * (1 - current_iter / max_iterations)
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        # Generate binary initial guess
        m = n_points // 2
        tmp = np.zeros(n_points)
        for i in range(m):
            tmp[i * 2] = 1.0
        if n_points % 2 == 1:
            tmp[-1] = tmp[-1] + (target_sum - np.sum(tmp))
        tmp = np.clip(tmp, 0, 1)
        tmp_sum = np.sum(tmp)
        if abs(tmp_sum - target_sum) < 1e-6:
            perturbed_guess = tmp.copy()
        elif tmp_sum < target_sum:
            delta = target_sum - tmp_sum
            for i in range(n_points):
                if perturbed_guess[i] < 1:
                    perturbed_guess[i] += delta
                    delta = 0
                    break
        elif tmp_sum > target_sum:
            delta = tmp_sum - target_sum
            for i in range(n_points):
                if perturbed_guess[i] > 0:
                    perturbed_guess[i] -= delta
                    delta = 0
                    break

        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
