    initial_evals = []
    for guess in [initial_guess, alternate_initial_guess, structured_initial_guess, random_initial_guess]:
        current_c5 = objective(guess)
        initial_evals.append((current_c5, guess))
    
    initial_evals.sort()
    best_c5, best_h = initial_evals[0]
    
    start_time = time.time()
    for iteration in range(50):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.02 * (1 - iteration / 50)
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
