        current_iter += 1
        noise_scale = 0.01 * (1 - current_iter / max_iterations)
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='trust-constr',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={
                'maxiter': 300,
                'ftol': 1e-9,
                'gtol': 1e-9,
                'eps': 1e-8,
                'verbose': 0
            }
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
