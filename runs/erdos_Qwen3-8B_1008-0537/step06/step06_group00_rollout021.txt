        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    best_even_h = even_initial_guess.copy()
    best_even_c5 = objective(best_even_h)
    if best_even_c5 < best_c5:
        best_h = best_even_h
        best_c5 = best_even_c5
    best_random_h = random_initial_guess.copy()
    best_random_c5 = objective(best_random_h)
    if best_random_c5 < best_c5:
        best_h = best_random_h
        best_c5 = best_random_c5
    
    for iteration in range(200):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.02 * (1 - iteration / 200)
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
