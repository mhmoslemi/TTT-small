import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Prepare initial guess by fixing the last variable
    initial_guess = np.array(initial_h_values[:-1], dtype=float)
    initial_last = target_sum - np.sum(initial_guess)
    initial_last = np.clip(initial_last, 0, 1)
    initial_guess = np.concatenate([initial_guess, [initial_last]])
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e-3 * abs(np.sum(h) - target_sum)
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Run differential evolution with optimized parameters
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=150,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    elapsed_time = time.time() - start_time
    
    # Ensure we don't exceed the budget
    if elapsed_time > budget_s:
        best_h = initial_guess
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
