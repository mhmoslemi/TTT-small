            options={'xatol': 1e-6, 'ftol': 1e-9, 'maxiter': 100}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
