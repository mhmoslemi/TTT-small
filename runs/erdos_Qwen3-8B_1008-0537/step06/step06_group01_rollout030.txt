        # Use different perturbation strategy for each start
        if idx == 0:
            noise_scale = 0.03 * (1 - idx / len(starts))
        else:
            noise_scale = 0.05 * (1 - idx / len(starts))
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(start + perturb, 0, 1)

        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
