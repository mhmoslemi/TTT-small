import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    max_iterations = 80  # Increased for more exploration
    current_iter = 0
    noise_decay = 1.0  # Larger initial noise

    while current_iter < max_iterations and time.time() - start_time < budget_s:
        current_iter += 1
        noise_scale = 0.02 * noise_decay * (1 - current_iter / max_iterations)
        noise_decay *= 0.95  # Slow decay for global exploration
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
