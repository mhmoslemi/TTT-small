            idx = int(x / dx)
            if 0 <= idx < n:
                guess[idx] = 1.0
        remaining = target_sum - m
        for i in range(n):
            if guess[i] < 1.0:
                guess[i] += remaining
                remaining = 0
                break
        return guess
    
    evenly_spaced_guess = generate_evenly_spaced_guess(n_points, target_sum)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x, objective(result.x)
        return guess, objective(guess)
    
    guesses = [initial_guess, alternate_initial_guess, random_initial_guess, evenly_spaced_guess]
    for guess in guesses:
        curr_h, curr_c5 = optimize_guess(guess)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h
    
    return best_h, best_c5, n_points
