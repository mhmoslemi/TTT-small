        h = np.array(h_values)
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    local_result = minimize(
        fun=local_objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-9, 'disp': False}
    )

    best_h = local_result.x
    best_c5 = local_objective(best_h)

    return best_h, best_c5, n_points
