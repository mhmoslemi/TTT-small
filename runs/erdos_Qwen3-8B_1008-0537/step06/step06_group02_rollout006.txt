import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h_values):

        c5 = np.max(corr) * dx

        return c5 + sum_penalty

    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,

        popsize=15,

        mutation=(0.3, 0.9),
        recombination=0.8,
        seed=seed,
        maxiter=50,
        polish=True,
        disp=False
    )

    best_h = result.x
    best_c5 = objective(best_h)

    # Early return if time is exhausted
