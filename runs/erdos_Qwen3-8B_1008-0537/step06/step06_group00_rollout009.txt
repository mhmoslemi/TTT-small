        guess = np.zeros(n)
        guess[indices] = 1.0

                if guess[i] < 1.0:
                    guess[i] += remaining
                    remaining = 0
                    break
        return guess
    
    random_initial_guess = generate_random_guess(n_points)
    
    def objective(h_values):
