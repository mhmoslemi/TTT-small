import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        penalty = 1e5 * (np.sum(h) - target_sum) ** 2
        return max_overlap + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for i in range(120):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.05 * (1 - i / 120)
        perturb = np.random.normal(0, noise_scale, n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(perturbed_guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_guess

    return best_h, best_c5, n_points
