import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_random_guess(n):
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            for i in range(n):
                if guess[i] < 1.0:
                    guess[i] += remaining
                    remaining = 0
                    break
        return guess
    
    random_initial_guess = generate_random_guess(n_points)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    alt_h = np.zeros(n_points, dtype=float)
    alt_h[:n_points//2] = 1
    alt_c5 = objective(alt_h)
    if alt_c5 < best_c5:
        best_h = alt_h
        best_c5 = alt_c5
    
    rand_h = random_initial_guess.copy()
    rand_c5 = objective(rand_h)
    if rand_c5 < best_c5:
        best_h = rand_h
        best_c5 = rand_c5
    
    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
