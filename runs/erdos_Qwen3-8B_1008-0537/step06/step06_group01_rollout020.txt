import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_val = np.sum(h)
        penalty = 1e4 * (sum_val - target_sum) ** 2
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Use differential evolution with constraint handling via penalty
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False
    )
    
    # Early exit if time budget is exceeded
    if time.time() - start_time > budget_s:
        return initial_guess, objective(initial_guess), n_points
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Final refinement with SLSQP for precision
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    def refined_objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = optimize.minimize(
        fun=refined_objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = refined_objective(best_h)
    
    return best_h, best_c5, n_points
