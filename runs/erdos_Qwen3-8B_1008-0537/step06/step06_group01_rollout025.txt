import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
    else:
        n_points = 450
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = np.array([0.5]*n_points)

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for i in range(120):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.025 * (1.0 - i / 120.0)
        if 'initial_h_values' in globals():
            perturbed_guess = initial_h_values + np.random.normal(0, noise_scale, n_points)
        else:
            perturbed_guess = initial_guess + np.random.normal(0, noise_scale, n_points)
        perturbed_guess = np.clip(perturbed_guess, 0.0, 1.0)

        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
