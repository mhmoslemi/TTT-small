import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic-inspired cluster search with adaptive perturbations
    for _ in range(60):
        if time.time() - start_time > budget_s:
            break
        # Generate diverse perturbations from current best
        perturbations = []
        for i in range(5):
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            perturb = np.clip(best_h + noise, 0, 1)
            # Ensure sum constraint by adjusting last point
            perturb[-1] = target_sum - np.sum(perturb[:-1])
            perturbations.append(perturb)

        # Optimize each perturbation and track best
        for perturb in perturbations:
            result = optimize.minimize(
                fun=objective,
                x0=perturb,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x

    return best_h, best_c5, n_points
