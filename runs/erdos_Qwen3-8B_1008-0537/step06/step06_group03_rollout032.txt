import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    def generate_candidate():
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h * target_sum / np.sum(h), 0, 1)
        return h

    candidates = [np.array(initial_h_values, dtype=float)]
    for _ in range(3):
        candidates.append(generate_candidate())

    best_h = candidates[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for idx, guess in enumerate(candidates):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        if time.time() - start_time > budget_s - 1:
            break

    return best_h, best_c5, n_points
