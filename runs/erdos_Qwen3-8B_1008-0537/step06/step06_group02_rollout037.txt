import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    remaining_time = budget_s

    # Generate multiple initial guesses
    num_starts = 4
    start_points = []
    for _ in range(num_starts):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(initial_guess + noise, 0, 1)
        start_points.append(perturbed)

    # Use L-BFGS-B for faster convergence
    for start in start_points:
        if time.time() - start_time >= budget_s:
            break
        result = optimize.minimize(
            fun=objective,
            x0=start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Early stopping based on remaining time
        elapsed = time.time() - start_time
        remaining_time = budget_s - elapsed
        if remaining_time < 10:
            break

    return best_h, best_c5, n_points
