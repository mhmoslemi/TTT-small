    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        best_this = guess.copy()
        best_c5_this = objective(best_this)
        if best_c5_this < best_c5:
            best_c5 = best_c5_this
            best_h = best_this
    
    for i in range(10):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.05 * (1 - i / 10)
        for guess in initial_guesses:
            guess = np.clip(guess, 0, 1)
            perturbed_guess = guess + np.random.normal(0, noise_scale, size=n_points)
            perturbed_guess = np.clip(perturbed_guess, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return best_h, best_c5, n_points

def objective(h_values):
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
