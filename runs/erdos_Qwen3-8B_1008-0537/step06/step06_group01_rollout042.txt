import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Basinhopping parameters: global optimization with local refinement
    # Adjust the number of iterations based on time
    niter = min(30, int((budget_s - 5) / 1))  # Account for initial setup

    result = optimize.basinhopping(
        fun=objective,
        x0=initial_guess,
        minimizer_kwargs={
            'method': 'SLSQP',
            'bounds': bounds,
            'constraints': [{'type': 'eq', 'fun': constraint_sum}],
            'options': {
                'maxiter': 200,
                'ftol': 1e-10,
                'eps': 1e-9,
                'disp': False
            }
        },
        niter=niter,
        T=1.0,
        stepsize=0.1,
        # For real-time control, we'll further refine the best solution
        # after basinhopping completes
    )

    best_h = result.x
    best_c5 = result.fun

    # Refinement phase: local optimization of the best candidate found
    result_refine = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={
            'maxiter': 100,
            'ftol': 1e-10,
            'eps': 1e-9,
            'disp': False
        }
    )
    best_h = result_refine.x
    best_c5 = result_refine.fun

    # Ensure we stay under the time limit
    if time.time() - start_time > budget_s - 5:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
