        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    initial_guesses = [initial_guess.copy(), 
                      np.zeros(n_points), 
                      generate_random_guess(n_points), 
                      generate_spread_guess(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    best_alt = np.zeros(n_points)
    best_alt[:n_points//2] = 1.0
    best_alt_c5 = objective(best_alt)
    if best_alt_c5 < best_c5:
        best_h = best_alt.copy()
        best_c5 = best_alt_c5
    
    for guess in [generate_random_guess(n_points), generate_spread_guess(n_points)]:
        current_c5 = objective(guess)
        if current_c5 < best_c5:
            best_h = guess.copy()
            best_c5 = current_c5
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.02 * (1 - _ / 100)
        perturbed = np.clip(best_h + np.random.normal(0, noise_scale, n_points), 0, 1)
        result = optimize.minimize(objective, perturbed, 
                                  method='COBYLA', bounds=bounds, 
                                  constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                  options={'maxiter': 200, 'ftol': 1e-10})
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
