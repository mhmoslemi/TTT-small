import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def create_alternate_guess(n):
        half = n // 2
        guess = np.zeros(n)
        guess[:half] = 1.0
        return guess
    
    alternate_initial_guess = create_alternate_guess(n_points)
    
    def generate_random_guess(n):
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            guess[np.random.choice(n, size=1, replace=False)] += remaining
        return guess
    
    random_initial_guess = generate_random_guess(n_points)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    best_alt_h = alternate_initial_guess.copy()
    best_alt_c5 = objective(best_alt_h)
    if best_alt_c5 < best_c5:
        best_h = best_alt_h
        best_c5 = best_alt_c5
    
    best_random_h = random_initial_guess.copy()
    best_random_c5 = objective(best_random_h)
    if best_random_c5 < best_c5:
        best_h = best_random_h
        best_c5 = best_random_c5
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_guess = np.clip(best_h + perturbation, 0.0, 1.0)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
