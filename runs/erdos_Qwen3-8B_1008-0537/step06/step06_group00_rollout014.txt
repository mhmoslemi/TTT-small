import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_spaced_guess():
        num_ones = int(target_sum + 0.5)
        indices = np.argsort(np.random.rand(n_points))
        guess = np.zeros(n_points)
        guess[indices[:num_ones]] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            guess[indices[num_ones]] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        initial = generate_spaced_guess()
        result = optimize.minimize(
            fun=objective,
            x0=initial,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
