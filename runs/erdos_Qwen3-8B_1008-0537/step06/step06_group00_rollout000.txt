import scipy.optimize as optimize
import time

    def create_periodic_guess(n):
        period = n // 5
        guess = np.zeros(n)
        for i in range(n):
            if (i % period) < period // 2:
                guess[i] = 1.0
        return guess
    
    periodic_initial = create_periodic_guess(n_points)
    
    def generate_uniform_guess(n):
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            for i in range(n):
                if guess[i] < 1.0:
                    guess[i] += remaining
                    remaining = 0
                    break
        return guess
    
    uniform_initial = generate_uniform_guess(n_points)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    best_periodic_h = periodic_initial.copy()

    if best_periodic_c5 < best_c5:

    best_uniform_h = uniform_initial.copy()
    best_uniform_c5 = objective(best_uniform_h)
    if best_uniform_c5 < best_c5:
        best_h = best_uniform_h
        best_c5 = best_uniform_c5
    
    for iteration in range(150):
