        recombination=0.9,
        polish=True,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        best_h = initial_guess
        best_c5 = objective(best_h)
    return best_h, best_c5, n_points
