        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        # Use a decreasing perturbation scale
        noise_scale = 0.02 + 0.01 * np.exp(-0.05 * _)
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        # Use COBYLA for constrained optimization
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'disp': False, 'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
