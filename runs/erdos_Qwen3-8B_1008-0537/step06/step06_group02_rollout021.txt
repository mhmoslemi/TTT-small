import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import scipy.optimize as optimize
    import time

    def initial_h_values(n_points):
        n = n_points
        return np.concatenate([np.zeros(n // 3 + 1), np.ones(n // 3), np.zeros(n // 3)])

    def objective(h_values):
        h = np.array(h_values, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    n_points = 45
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = initial_h_values(n_points)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for i in range(150):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.02 * (1 - i / 150)
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-7, 'disp': False}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
