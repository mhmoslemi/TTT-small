        best_h = best_structured_h
        best_c5 = best_structured_c5
    best_random_h = random_initial_guess.copy()
    best_random_c5 = objective(best_random_h)
    if best_random_c5 < best_c5:
        best_h = best_random_h
        best_c5 = best_random_c5
    
    for iteration in range(100):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.01 * (1 - iteration / 100)
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
