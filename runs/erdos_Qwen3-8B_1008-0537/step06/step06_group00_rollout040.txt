import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_random_guess(n):
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            for i in range(n):
                if guess[i] < 1.0:
                    guess[i] += remaining
                    remaining = 0
                    break
        return guess
    
    random_initial_guess = generate_random_guess(n_points)
    
    def generate_periodic_guess(n, period):
        if period == 0:
            return np.zeros(n)
        h = np.zeros(n)
        for i in range(n):
            pos = i * dx
            if (pos - 1.0) % period < dx:
                h[i] = 1.0
        return h
    
    periodic_guess = generate_periodic_guess(n_points, 0.5)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    def optimize_single(initial_guess):
        result = optimize.minimize(
            fun=objective,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success:
            return result.x, objective(result.x)
        return initial_guess, np.inf
    
    initial_guesses = [initial_guess, random_initial_guess, periodic_guess]
    for guess in initial_guesses:
        current_h, current_c5 = optimize_single(guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            perturbed_guess = np.clip(best_h + 0.01 * np.random.randn(n_points), 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return best_h, best_c5, n_points
