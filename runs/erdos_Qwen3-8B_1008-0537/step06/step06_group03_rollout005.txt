import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import random
    random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    max_iterations = 80
    current_iter = 0
    warm_start = True

    while current_iter < max_iterations and time.time() - start_time < budget_s:
        current_iter += 1
        noise_scale = 0.05 * (1 - current_iter / max_iterations)
        if current_iter % 5 == 0:
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        else:
            perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        if current_iter % 10 == 0 and warm_start:
            warm_start = False
            perturb = np.random.uniform(-0.3, 0.3, size=n_points)
            perturbed_guess = np.clip(best_h + perturb, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='L-BFGS-B',

                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
            )
