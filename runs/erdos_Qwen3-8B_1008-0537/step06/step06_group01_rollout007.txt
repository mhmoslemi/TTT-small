import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Generate multiple diverse initial guesses
    n_initial = 5  # number of diverse initial guesses
    for i in range(n_initial):
        if time.time() - start_time > budget_s:
            break
        # Perturb with different noise scales
        noise_scale = 0.1 + 0.05 * np.random.rand()
        noise = np.random.normal(0, noise_scale, size=n_points)
        perturbed = np.clip(initial_guess + noise, 0, 1)
        # Enforce sum constraint
        perturbed = (perturbed * target_sum) / np.sum(perturbed)
        # Run optimization
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
        )
        if result.success:
            c5 = objective(result.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x

    # Fallback to initial guess if all optimizations fail
    if best_c5 > objective(initial_guess):
        best_h = initial_guess
        best_c5 = objective(initial_guess)
    
    return best_h, best_c5, n_points
