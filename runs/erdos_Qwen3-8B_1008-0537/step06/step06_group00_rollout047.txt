import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_alternate_guess(n):
        return np.zeros(n, dtype=float)
    
    # Alternate initial guess with a block of 1s
    alternate_initial_guess = generate_alternate_guess(n_points)
    alternate_initial_guess[:n_points//2] = 1
    
    def generate_random_guess(n):
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            for i in range(n):
                if guess[i] < 1.0:
                    guess[i] += remaining
                    remaining = 0
                    break
        return guess
    
    random_initial_guess = generate_random_guess(n_points)
    
    def generate_structured_guess(n_points):
        m = 10
        l = 1.0 / m
        g = 1.0 / (m - 1)
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (l + g)
            end = start + l
            for idx in range(n_points):
                x = idx * dx
                if start <= x <= end:
                    h[idx] = 1.0
        # Normalize to ensure correct sum
        actual_sum = np.sum(h) * dx
        scaling = target_sum / actual_sum if actual_sum > 0 else 1.0
        return np.clip(h * scaling, 0, 1)
    
    structured_initial_guess = generate_structured_guess(n_points)
    
    initial_guesses = [
        initial_guess, alternate_initial_guess, random_initial_guess, structured_initial_guess
    ]
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Evaluate initial guesses
    best_h = None
    best_c5 = float('inf')
    for guess in initial_guesses:
        if np.sum(guess) * dx < 0.99 * target_sum or np.sum(guess) * dx > 1.01 * target_sum:
            continue
        current_c5 = objective(guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = guess.copy()
    
    start_time = time.time()
    for iteration in range(200):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.05 * (1 - iteration / 200)
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
