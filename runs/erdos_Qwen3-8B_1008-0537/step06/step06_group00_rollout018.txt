            disp=False,
            mutation=(0.5, 1.0)
        )
        best_de_h = result.x
        best_de_c5 = objective(best_de_h)
        if best_de_c5 < best_c5:
            best_c5 = best_de_c5
            best_h = best_de_h
    
    return best_h, best_c5, n_points
