import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    # Alternative initial guess: half 1s, half 0s, adjusted for sum
    def generate_regular_initial():
        half = n_points // 2
        h = np.zeros(n_points)
        h[:half] = 1.0
        if n_points % 2:
            h[-1] += 0.5
        if np.sum(h) != target_sum:
            delta = target_sum - np.sum(h)
            h[-int(delta*2):] += delta
        return h

    initial_guesses = [initial_guess, generate_regular_initial()]

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    max_iterations = 60
    current_iter = 0

    while current_iter < max_iterations and time.time() - start_time < budget_s:
        current_iter += 1
        noise_scale = 0.05 * (1 - current_iter / max_iterations)
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        # Use trust-constr for better convergence
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='trust-constr',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'gtol': 1e-8, 'maxiter': 200}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
