def run(seed=42, budget_s=100, **kwargs):

    m = n_points // 2
    ideal_positions = np.linspace(0, 2, m)
    positions = np.linspace(0, 2, n_points)
    selected_indices = np.round(ideal_positions * n_points / 2).astype(int)
    even_h = np.zeros(n_points)
    even_h[selected_indices] = 1
    even_h[selected_indices[0]] = 0.5  # Adjust to meet sum constraint

    # Choose initial guess with lower initial C5
    def compute_initial_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    c5_1 = compute_initial_c5(h1)
    c5_2 = compute_initial_c5(h2)
    initial_guess = h1 if c5_1 < c5_2 else h2

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
