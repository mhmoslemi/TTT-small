import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2

        scale = target_sum / sum_candidate

        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=20,
        tol=1e-7,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        polish=True,
        disp=False
    )

    best_h = result_de.x
    best_c5 = objective(best_h)

    # Refinement with local optimization
    def objective_local(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds_local = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = optimize.minimize(
        fun=objective_local,

        method='SLSQP',

    )

    if result_slsqp.success:

        best_c5 = objective_local(best_h)
