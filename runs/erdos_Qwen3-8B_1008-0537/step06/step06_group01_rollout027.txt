import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_variants(h):
        variants = [h.copy()]
        variants.append(h[::-1].copy())  # reversed
        variants.append(np.roll(h, 2).copy())  # shifted
        variants.append(np.random.permutation(h).copy())  # shuffled
        return variants

    # Generate diverse starting points
    starting_points = generate_variants(initial_guess)
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Evaluate all starting points
    for guess in starting_points:
        current_h = guess.copy()
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

    # Perturb and optimize
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.03 * (1 - _ / 10)
        perturbed_guesses = [np.clip(best_h + np.random.normal(0, noise_scale, n_points), 0, 1) for _ in range(5)]
        for guess in perturbed_guesses:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()

    return best_h, best_c5, n_points
