import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    max_iterations = 80
    current_iter = 0
    noise_scale = 0.05
    decay_factor = 0.995

    while current_iter < max_iterations and time.time() - start_time < budget_s:
        current_iter += 1
        # Generate diverse perturbations
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        # Ensure sum constraint is approximately met
        perturbed_guess[0] = (target_sum - np.sum(perturbed_guess[1:])) / 1.0

        # Use Nelder-Mead for global exploration
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'xatol': 1e-10, 'fatol': 1e-10, 'maxiter': 200}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        # Decay noise scale for fine-tuning
        noise_scale *= decay_factor

    return best_h, best_c5, n_points
