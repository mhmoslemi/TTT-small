import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    # Generate other initial guesses
    alternate_initial_guess = np.zeros(n_points)
    alternate_initial_guess[:n_points//2] = 1.0
    
    def generate_random_guess(n):
        # Randomly select positions to set to 1
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            for i in range(n):
                if guess[i] < 1.0:
                    guess[i] += remaining
                    remaining = 0
                    break
        return guess
    
    random_initial_guess = generate_random_guess(n_points)
    
    # Objective function
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_deviation = np.sum(h) - target_sum
        return max_corr + 1e4 * sum_deviation**2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Run optimization from each initial guess
    start_time = time.time()
    for initial in [initial_guess, alternate_initial_guess, random_initial_guess]:
        if time.time() - start_time > budget_s:
            break
        # Run differential evolution from this initial guess
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=20,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            seed=seed,
            x0=initial
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points

import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_alternate_guess(n):
        guess = np.zeros(n)
        guess[:n//2] = 1.0
        return guess
    
    def generate_random_guess(n):
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            for i in range(n):
                if guess[i] < 1.0:
                    guess[i] += remaining
                    remaining = 0
                    break
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_deviation = np.sum(h) - target_sum
        return max_corr + 1e4 * sum_deviation**2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for init in [initial_guess, generate_alternate_guess(n_points), generate_random_guess(n_points)]:
        if time.time() - start_time > budget_s:
            break

            bounds=bounds,

            maxiter=50,

            seed=seed,
            x0=init

            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
