    initial_guess = np.array(initial_h_values, dtype=float)
    
    # Alternate initial guess: half ones
    alternate_initial_guess = np.zeros(n_points, dtype=float)
    alternate_initial_guess[:n_points//2] = 1.0
    
    # Random initial guess
    def generate_random_guess():
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        guess = np.zeros(n_points)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            for i in range(n_points):
                if guess[i] < 1.0:
                    guess[i] += remaining
                    remaining = 0
                    break
        return guess
    
    random_initial_guess = generate_random_guess()
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Run from multiple initial guesses
    for guess in [initial_guess, alternate_initial_guess, random_initial_guess]:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Early exit if time is up
    if time.time() - start_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    # Add more random restarts
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        guess = generate_random_guess()
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
