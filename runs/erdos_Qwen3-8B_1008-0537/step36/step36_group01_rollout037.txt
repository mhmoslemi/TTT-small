    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guess(period, offset):
        x = np.linspace(0, 2, n_points)
        val = 0.5 * (1 + np.sin(2 * np.pi * (x - offset) / period))
        val = np.clip(val, 0, 1)
        return val
    
    def generate_block_guess(_):
        x = np.linspace(0, 2, n_points)
        h_vals = np.zeros(n_points)
        for i in range(n_points):
            if x[i] < 1.0:
                h_vals[i] = 1.0
        return h_vals
    
    def generate_random_guess(_):
        h_vals = np.random.rand(n_points)
        h_vals = np.sort(h_vals)
        h_vals = (h_vals <= 0.5).astype(float)
        scale = target_sum / np.sum(h_vals)
        h_vals *= scale
        return h_vals
    
    guesses = [generate_guess(p, o) for p, o in [(4, 0.3), (5, 0.1), (6, 0.2), (8, 0.4)]]
    guesses.append(generate_block_guess(0))
    guesses.append(generate_random_guess(0))
    guesses.append(np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time < budget_s - 50:
        res_de = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=40,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 20, 'ftol': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
