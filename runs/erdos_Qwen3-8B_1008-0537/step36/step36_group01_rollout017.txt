import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guess(period, offset, symmetric=False):
        x = np.linspace(0, 2, n_points)
        if symmetric:
            out = np.zeros_like(x)
            for i in range(n_points):
                out[i] = 0.5 * (1 + np.sin(2 * np.pi * (x[i] - offset) / period))
                if i > n_points - 1 - i:
                    out[i] = out[n_points - 1 - i]
            return np.clip(out, 0, 1)
        else:
            return np.clip(0.5 * (1 + np.sin(2 * np.pi * (x - offset) / period)), 0, 1)
    
    guesses = []
    for p, o in [(4, 0.3), (5, 0.1), (6, 0.2), (8, 0.4), (10, 0.5)]:
        guesses.append(generate_guess(p, o))
        guesses.append(generate_guess(p, o, symmetric=True))
    guesses.append(np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]))
    guesses.append(create_symmetric_guess(initial_h_values))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success and objective(res.x) < best_c5:

def create_symmetric_guess(h):
