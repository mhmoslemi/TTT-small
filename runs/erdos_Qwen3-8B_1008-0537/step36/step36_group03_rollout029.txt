    m = n_points // 2
    initial_guess = generate_binary_initial(n_points, m)
    initial_sum = np.sum(initial_guess)
    if initial_sum < target_sum - 1e-6:
        zero_indices = np.where(initial_guess == 0)[0]
        initial_guess[zero_indices[0]] = 1.0
    elif initial_sum > target_sum + 1e-6:
        one_indices = np.where(initial_guess == 1)[0]
        initial_guess[one_indices[0]] = 0.0
    h = initial_guess.astype(float)

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    res_slsqp = optimize.minimize(objective, best_h, method='SLSQP',
                                 bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                 options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
    if res_slsqp.success:
        best_h = res_slsqp.x
        best_c5 = objective(best_h)

    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res_de = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=50,
                                                 mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res_de.success:
            res_slsqp_final = optimize.minimize(objective, res_de.x, method='SLSQP',
                                                bounds=bounds,
                                                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                                options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp_final.success and objective(res_slsqp_final.x) < best_c5:
                best_h = res_slsqp_final.x
                best_c5 = objective(best_h)

    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        res_refine = optimize.minimize(objective, best_h, method='SLSQP',
                                      bounds=bounds,
                                      constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                      options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8})
        if res_refine.success and objective(res_refine.x) < best_c5:
            best_h = res_refine.x
            best_c5 = objective(best_h)

    return best_h, best_c5, n_points
