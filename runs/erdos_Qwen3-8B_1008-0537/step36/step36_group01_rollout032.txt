    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guess(period, offset):
        x = np.linspace(0, 2, n_points)
        val = 0.5 * (1 + np.sin(2 * np.pi * (x - offset) / period))
        return np.clip(val, 0, 1)
    
    guesses = []
    for p, o in [(4, 0.3), (5, 0.1), (6, 0.2), (8, 0.4), (10, 0.5)]:
        guesses.append(generate_guess(p, o))
    guesses.append(np.array([0.5 if i % 2 else 0.5 for i in range(n_points)]))
    h_rand = np.random.rand(n_points) * (target_sum / np.sum(np.random.rand(n_points)))
    guesses.append(h_rand)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time < budget_s - 60:
        res_de = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=60,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 80, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(7):
        if time.time() - start_time >= budget_s - 15:
            break
        res = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 25, 'ftol': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
