import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guess(period, offset):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 * (1 + np.sin(2 * np.pi * (x - offset) / period)), 0, 1)
    
    # Diverse initial guesses
    guesses = []
    for p in [2.5, 3.0, 4.0, 5.0, 6.0, 7.5]:
        for o in np.linspace(0, 2, 5):
            guesses.append(generate_guess(p, o))
    guesses.append(np.array([0.5 if i % 2 == 0 else 0.0 for i in range(n_points)]))
    guesses.append(np.array([0.0 if i % 2 == 0 else 0.5 for i in range(n_points)]))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    if time.time() - start_time < budget_s - 60:
        res_de = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=40,
                                             mutation=(0.6, 0.95), recombination=0.85, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 80, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success:
                current_c5 = objective(res_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res_slsqp.x
    
    for _ in range(7):

        res = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds,
