import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guess(period, offset):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 * (1 + np.sin(2 * np.pi * (x - offset) / period)), 0, 1)
    
    def generate_binary_guess(n_points, num_blocks):
        block_size = 2.0 / num_blocks
        h = np.zeros(n_points)
        for i in range(num_blocks):
            length = int(block_size * n_points / 2)
            start = int(i * n_points / num_blocks)

                end = n_points
            h[start:end] = 1.0
        dx = 2.0 / n_points
        total_area = np.sum(h) * dx
        scale = 1.0 / total_area
        return h * scale

    guesses.append(np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]))

    for guess in guesses:

            res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],

            if res.success and objective(res.x) < best_c5:
