import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guess(period, offset, shape='sine'):
        x = np.linspace(0, 2, n_points)
        if shape == 'sine':
            return np.clip(0.5 * (1 + np.sin(2 * np.pi * (x - offset) / period)), 0, 1)
        elif shape == 'square':
            return np.clip(0.5 * (1 + np.sign(np.sin(2 * np.pi * (x - offset) / period))), 0, 1)
        elif shape == 'triangle':
            return np.abs(np.sin(2 * np.pi * (x - offset) / period)) * 0.5 + 0.5
        elif shape == 'random':
            return np.random.uniform(0, 1, n_points)
    
    guesses = []
    for p, o in [(4, 0.3), (5, 0.1), (6, 0.2), (8, 0.4)]:
        guesses.append(generate_guess(p, o, 'sine'))

        guesses.append(generate_guess(p, o, 'triangle'))
    guesses.append(np.random.uniform(0, 1, n_points))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
