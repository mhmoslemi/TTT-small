import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guess(period, offset, kind):
        x = np.linspace(0, 2, n_points)
        if kind == 'sine':
            return np.clip(0.5 * (1 + np.sin(2 * np.pi * (x - offset) / period)), 0, 1)
        elif kind == 'interval':
            m = int(period)
            l = 1.0 / m
            gap = (2.0 - m * l) / (m - 1)
            h_guess = np.zeros(n_points)
            for i in range(n_points):
                for k in range(m):
                    start = k * (l + gap) if k > 0 else 0.0
                    end = start + l
                    if start <= x[i] < end:
                        h_guess[i] = 1.0
                        break
            return h_guess
    
    guesses = [
        generate_guess(4, 0.3, 'sine'),
        generate_guess(5, 0.1, 'sine'),
        generate_guess(6, 0.2, 'sine'),
        generate_guess(8, 0.4, 'sine')
    ]
    for m in [4,5,6,7]:

    guesses.append(np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time < budget_s - 50:
        res_de = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=30,
