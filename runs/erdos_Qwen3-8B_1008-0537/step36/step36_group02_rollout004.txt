import scipy.optimize as optimize
import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

        elif wave_type == 'triangle':
            return np.clip(0.5 * (1 + np.abs(np.sin(2 * np.pi * x / period))), 0, 1)
        elif wave_type == 'sawtooth':
            return np.clip(0.5 * (1 + np.sin(2 * np.pi * x / period) if np.sin(2 * np.pi * x / period) > 0 else 0), 0, 1)
        else:
            return np.clip(0.5 * (1 + np.sin(2 * np.pi * x / period)), 0, 1)

        guesses.append(generate_initial(p, 'sawtooth'))

    guesses.append(h)
    
    for guess in guesses:
