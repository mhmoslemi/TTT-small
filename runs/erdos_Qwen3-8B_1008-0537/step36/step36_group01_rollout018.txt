import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guess(period, offset):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 * (1 + np.sin(2 * np.pi * (x - offset) / period)), 0, 1)
    
    def binary_guess(num_segments):
        x = np.linspace(0, 2, n_points)
        intervals = np.linspace(0, 2, num_segments + 1)
        seg_len = 1.0 / num_segments
        h = np.zeros(n_points)
        for i in range(num_segments):
            start = intervals[i]
            end = intervals[i+1]
            mask = (x >= start) & (x <= end)
            h[mask] = 1.0
        h = np.interp(x, x, h)
        return h
    
    guesses = [
        generate_guess(4, 0.3), generate_guess(5, 0.1), generate_guess(6, 0.2),
        generate_guess(8, 0.4), generate_guess(10, 0.3), generate_guess(12, 0.1),

        binary_guess(5), binary_guess(6), binary_guess(7)
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time < budget_s - 50:

                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:

            best_h = res.x
    
    return best_h, best_c5, n_points
