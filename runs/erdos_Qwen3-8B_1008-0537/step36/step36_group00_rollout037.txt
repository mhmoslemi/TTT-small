import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_block_basis(f, phase=0):
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * f * x + phase))
        return np.clip(h, 0, 1)
    
    def generate_triplet_basis(f):
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.abs(np.sin(2 * np.pi * f * x))) * 0.7 + 0.3
        return np.clip(h, 0, 1)
    
    def generate_optimized_guess(h_guess):
        res = optimize.minimize(objective, h_guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-10})
        return res.x
    
    guesses = [np.array(initial_h_values)]
    guesses += [generate_block_basis(f) for f in [0.5, 0.25, 0.125, 0.333, 0.666, 0.75, 0.4, 0.2]]
    guesses += [generate_block_basis(f, phase=np.pi/2) for f in [0.5, 0.25]]
    guesses += [generate_triplet_basis(f) for f in [0.5, 0.25, 0.333, 0.666]]
    
    for guess in guesses:
        h_guess = generate_optimized_guess(guess)
        c5 = objective(h_guess)
        if c5 < best_c5:
            best_h, best_c5 = h_guess, c5
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=20,
                                             mutation=(0.8, 0.95), recombination=0.8, tol=1e-9)
        if res.success:
            h_guess = generate_optimized_guess(res.x)
            c5 = objective(h_guess)
            if c5 < best_c5:
                best_h, best_c5 = h_guess, c5
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:

        if res.success and objective(res.x) < best_c5:

            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
