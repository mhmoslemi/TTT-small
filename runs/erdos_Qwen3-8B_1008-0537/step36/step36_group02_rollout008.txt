import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_tiling(period, mode='sin'):
        x = np.linspace(0, 2, n_points)
        if mode == 'sin':
            return np.clip(0.5 * (1 + np.sin(2 * np.pi * x / period)), 0, 1)
        elif mode == 'saw':
            return np.clip(0.5 * (1 + np.abs(np.sin(2 * np.pi * x / period))**0.5), 0, 1)
        elif mode == 'square':
            return np.clip(0.5 * (1 + np.sign(np.sin(2 * np.pi * x / period))), 0, 1)
        return np.zeros(n_points)
    
    initial_guesses = [h]
    for p in [6, 8, 10]:
        initial_guesses.append(generate_tiling(p, 'sin'))
        initial_guesses.append(generate_tiling(p, 'square'))
        initial_guesses.append(generate_tiling(p, 'saw'))
    
    for guess in initial_guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    def evolve_c5(h_init):
        res = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=50,
                                             mutation=(0.5, 0.8), recombination=0.9, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-10})
            return res_slsqp.x if res_slsqp.success else res.x
        return h_init
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 40:
            break
        best_h = evolve_c5(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    def refine_cobyla(h_init):
        res = optimize.minimize(objective, h_init, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'rhobeg': 0.1, 'eps': 1e-9})
        return res.x if res.success else h_init
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_cobyla(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
