import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        n = len(h_vals)
        max_overlap = 0.0
        for s in range(n):
            total = 0.0
            for x in range(n - s):
                total += h_vals[x] * (1.0 - h_vals[x + s])
            overlap = total * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(f, phase=0):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x + phase)
        return np.clip(h, 0, 1)
    
    def generate_random():
        h = np.random.rand(n_points)
        h -= h.mean()
        h /= np.std(h)
        h *= 0.7
        h += 0.3
        return np.clip(h / 2 + 0.5, 0, 1)
    
    def generate_symmetric():
        h = np.zeros(n_points)
        for i in range(n_points // 2 + 1):
            h[i] = 0.5 * (1 + np.sin(2 * np.pi * i / (n_points - 1)))
        for i in range(1, n_points // 2 + 1):
            h[n_points - i] = h[i]
        return np.clip(h, 0, 1)
    
    def generate_triangular(f):
        x = np.linspace(0, 2, n_points)
        h = 0.5 * np.abs(np.sin(2 * np.pi * f * x)) + 0.5
        return np.clip(h, 0, 1)
    
    guesses = [np.array(initial_h_values)]
    guesses += [generate_sine(f) for f in [0.5, 0.25, 0.125, 0.333, 0.666, 0.75, 0.4, 0.2]]
    guesses += [generate_sine(f, phase) for f in [0.5, 0.25] for phase in [np.pi/2, np.pi, 3*np.pi/2]]
