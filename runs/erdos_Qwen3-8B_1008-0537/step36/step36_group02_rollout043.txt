            return np.clip(0.5 * (1 + val), 0, 1)

            t_mod = np.mod(t, 2)
            val = np.zeros_like(t)
            val[t_mod < 1] = t_mod[t_mod < 1]
            val[t_mod >= 1] = 1 - (t_mod[t_mod >= 1] - 1)

            return np.clip(0.5 * (1 + val), 0, 1)
