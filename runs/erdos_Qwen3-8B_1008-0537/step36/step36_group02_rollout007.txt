import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_initial(period, wave_type='sine'):
        x = np.linspace(0, 2, n_points)
        if wave_type == 'square':
            return np.clip(0.5 * (1 + np.sign(np.sin(2 * np.pi * x / period))), 0, 1)
        elif wave_type == 'triangle':
            return np.clip(0.5 * (1 + np.abs(np.sin(2 * np.pi * x / period))), 0, 1)
        else:
            return np.clip(0.5 * (1 + np.sin(2 * np.pi * x / period)), 0, 1)
    
    guesses = []
    for p in [2, 3, 4, 5, 6, 7, 8, 9, 10]:
        guesses.append(generate_initial(p, 'sine'))
        guesses.append(generate_initial(p, 'square'))
        guesses.append(generate_initial(p, 'triangle'))
    guesses.append(np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]))
    guesses.append(h)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    def differential_evolution_wrapper():
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=40,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9,
                                             polish=True, seed=seed)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        differential_evolution_wrapper()
    
    def refine_with_cobyla(h_init):

        return res.x if res.success else h_init
