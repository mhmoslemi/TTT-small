import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_random_guess(n):
        h = np.random.rand(n)
        desired_sum = n / 2
        if np.isclose(np.sum(h), 0):
            h = np.random.rand(n)
        scale = desired_sum / np.sum(h)
        return scale * h
    
    def generate_evenly_spaced(m):
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        return np.zeros(n_points)
    
    guesses = [np.array([0.5]*n_points)]
    
    for m in [10, 12, 14, 16]:
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        guesses.append(h_guess)
    
    guesses.append(generate_random_guess(n_points))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,

        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=80,

            if res_slsqp.success and objective(res_slsqp.x) < best_c5:

                best_h = res_slsqp.x
    
    def refine_with_slsqp(h_init):

        return res.x if res.success else h_init
    
    for _ in range(2):

        best_h = refine_with_slsqp(best_h)

    return best_h, best_c5, n_points
