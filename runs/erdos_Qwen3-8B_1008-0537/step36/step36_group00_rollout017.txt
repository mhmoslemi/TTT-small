import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(f, phase=0, amplitude=0.5):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + amplitude * np.sin(2 * np.pi * f * x + phase)
        return np.clip(h, 0, 1)
    
    def generate_triangular(f, amplitude=1.0):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + amplitude * np.abs(np.sin(2 * np.pi * f * x))
        return np.clip(h, 0, 1)
    
    def generate_random():
        h = np.random.rand(n_points) * 2 - 1
        h = (h - h.min()) / (h.max() - h.min())
        h = (h * 0.4 + 0.6)
        return np.clip(h, 0, 1)
    
    def generate_modulated_sine(f1, f2, phase):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * f1 * x) * np.sin(2 * np.pi * f2 * x + phase)
        return np.clip(h, 0, 1)
    
    def generate_alternating():
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / (n_points - 1))
        return np.clip(h, 0, 1)
    
    guesses = [np.array(initial_h_values)]
    guesses += [generate_sine(f, phase=0) for f in [0.5, 0.333, 0.25, 0.166, 0.125]]
    guesses += [generate_sine(f, phase=np.pi/2) for f in [0.5, 0.333, 0.25, 0.166, 0.125]]
    guesses += [generate_triangular(f) for f in [0.5, 0.333, 0.25, 0.166, 0.125]]
    guesses += [generate_modulated_sine(f1, f2, phase) for f1 in [0.5, 0.333] for f2 in [0.25, 0.166] for phase in [0, np.pi/2]]
    guesses += [generate_alternating()]
    guesses += [generate_random()]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,

        if res.success and objective(res.x) < best_c5:

        if res.success:
