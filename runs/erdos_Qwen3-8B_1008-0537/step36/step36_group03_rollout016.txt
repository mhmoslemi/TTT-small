import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary(n_ones):
        idxs = np.sort(np.random.choice(n_points, n_ones, replace=False))
        h_tmp = np.zeros(n_points)
        h_tmp[idxs] = 1.0
        if np.sum(h_tmp) < target_sum:
            h_tmp[np.argmax(h_tmp)] = 0.5
        elif np.sum(h_tmp) > target_sum:
            h_tmp[np.argmin(h_tmp)] = 0.5
        return h_tmp
    
    def generate_spaced(m):
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_tmp = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_tmp[i] = 1.0
        return h_tmp
    
    guesses = [np.array([0.5]*n_points)]
    for m in [10, 12, 14, 16]:
        guesses.append(generate_spaced(m))
    for n_ones in [15, 20, 25]:
        guesses.append(generate_binary(n_ones))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=80,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_slsqp(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
