import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guess(period, offset):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 * (1 + np.sin(2 * np.pi * (x - offset) / period)), 0, 1)
    
    def generate_square(period, offset):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 + 0.5 * np.sign(np.sin(2 * np.pi * (x - offset) / period)), 0, 1)
    
    guesses = [generate_guess(p, o) for p, o in [(4, 0.3), (5, 0.1), (6, 0.2), (8, 0.4), (10, 0.5)]]
    guesses.extend([generate_square(p, o) for p, o in [(4, 0.3), (5, 0.1), (6, 0.2)]])
    guesses.append(np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time < budget_s - 50:

        if res_de.success:
