import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_spaced(m):
        h = np.zeros(n_points)
        spacing = 2.0 / m
        for i in range(m):
            pos = i * spacing
            idx = int(pos / dx + 0.5)
            if 0 <= idx < n_points:
                h[idx] = 1
        return h
    
    def generate_binary_rand(m):
        idx = np.random.choice(n_points, m, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h

    def generate_fibonacci(m):
        h = np.zeros(n_points)
        pos = np.zeros(m)
        for i in range(m):
            pos[i] = np.sqrt((2 * i + 1) / (2 * m))
        pos *= 2.0
        for i in range(m):
            idx = int(pos[i] / dx + 0.5)
            if 0 <= idx < n_points:
                h[idx] = 1
        return h
    
    guesses = [
        generate_binary_spaced(15), generate_binary_spaced(20),
        generate_binary_rand(15), generate_binary_rand(20),
        generate_fibonacci(15), generate_fibonacci(20),
        np.random.rand(n_points), np.random.rand(n_points) * 0.5 + 0.5, h,
        np.zeros(n_points), np.ones(n_points)
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:

    def refine_with_neldermead(h_init):
