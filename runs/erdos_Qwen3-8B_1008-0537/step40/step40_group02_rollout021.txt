import scipy.optimize as optimize
import time

        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_uniform_spaced(m):
        h = np.zeros(n_points)
        idx = np.round(np.linspace(0, n_points-1, m)).astype(int)
        h[idx] = 1.0
        return h
    
    def generate_random_spaced(m):
        idx = np.random.choice(n_points, m, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        return h
    
    def optimize_from_guess(guess):
        res = optimize.minimize(
            objective, guess, method='SLSQP',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': max_iter, 'ftol': 1e-12, 'eps': 1e-9}
        )

            return res.x
        return guess
