import scipy.optimize as optimize
import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

        h = np.zeros(n_points)
        spacing = 2.0 / m
        for i in range(m):
            pos = i * spacing
            idx = int(pos / dx + 0.5)
            if 0 <= idx < n_points:
                h[idx] = 1
        return h
    
    def generate_binary_rand(m):
        idx = np.random.choice(n_points, m, replace=False)
        h = np.zeros(n_points)

        if res.success and objective(res.x) < best_c5:
