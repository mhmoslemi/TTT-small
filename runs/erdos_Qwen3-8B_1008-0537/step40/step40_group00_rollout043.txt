import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def generate_even_intervals(m):
        w = 1.0 / m
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        centers = np.linspace(w/2, 2 - w/2, m)
        for center in centers:
            start = center - w/2
            end = center + w/2
            _start = np.searchsorted(x, start)
            _end = np.searchsorted(x, end)
            h[_start:_end] = 1
        return h

    def generate_pulses(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def generate_saw(freq, duty):
        x = np.linspace(0, 2, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def generate_triangle(freq, duty):
        x = np.linspace(0, 2, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def generate_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h
    h = generate_even_intervals(5)
    n_points = len(h)
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-9})
        return res.x if res.success else g

    guesses = [h]
    for freq in range(3, 10):
        guesses.append(generate_pulses(freq, 0.5))
    for duty in [0.2, 0.4, 0.6]:
        guesses.append(generate_saw(3, duty))
    for duty in [0.2, 0.4, 0.6]:
        guesses.append(generate_triangle(3, duty))
    for _ in range(5):
        guesses.append(generate_random())

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():

                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(4):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-9})
        return res.x if res.success else r
