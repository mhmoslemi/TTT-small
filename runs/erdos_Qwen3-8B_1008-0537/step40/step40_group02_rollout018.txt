import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_spaced(spacing):
        return np.array([1 if i % spacing == 0 else 0 for i in range(n_points)], dtype=float)
    
    def generate_binary_rand(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        return np.array([1 if i in idx else 0 for i in range(n_points)], dtype=float)
    
    def generate_perturb(original, factor=0.1):
        noise = np.random.normal(0, factor, n_points)
        return np.clip(original + noise, 0, 1)
    
    guesses = [
        generate_binary_spaced(3), generate_binary_spaced(4),
        generate_binary_spaced(5), generate_binary_spaced(6),
        generate_binary_rand(22), generate_binary_rand(23),
        h, generate_perturb(h), np.random.rand(n_points)
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP',
                               bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=40,

        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP',
                                        bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:

            best_h = res.x
