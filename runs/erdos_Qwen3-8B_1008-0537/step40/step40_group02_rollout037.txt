import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_spaced(spacing):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % spacing == 0:
                h[i] = 1
        return h
    
    def generate_binary_rand(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h
    
    def perturb_h(h_vals, sigma=0.05, n_perturb=5):
        h = h_vals.copy()
        for _ in range(n_perturb):
            idx = np.random.choice(n_points, 1, replace=False)
            delta = np.random.normal(0, sigma)
            h[idx] += delta
            h = np.clip(h, 0, 1)
            scale = target_sum / np.sum(h)
            h *= scale
        return h
    
    guesses = [
        generate_binary_spaced(2), generate_binary_spaced(3), generate_binary_spaced(4),

    ]
    
    for guess in guesses:
        try:
            res = optimize.minimize(
                objective, guess, method='SLSQP',

            )
            if res.success and objective(res.x) < best_c5:

                best_h = res.x
