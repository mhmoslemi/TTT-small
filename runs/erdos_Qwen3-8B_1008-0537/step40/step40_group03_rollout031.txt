import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_spaced(spacing, max_ones):
        h = np.zeros(n_points)
        cnt = 0
        for i in range(n_points):
            if i % spacing == 0:
                h[i] = 1
                cnt += 1
                if cnt > max_ones:
                    break
        return h
    
    def generate_binary_rand(count):
        idx = np.random.choice(n_points, count, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h
    
    guesses = [
        generate_binary_spaced(3, n_points//2), generate_binary_spaced(4, n_points//2),
        generate_binary_spaced(5, n_points//2), generate_binary_spaced(6, n_points//2),
        generate_binary_rand(n_points//2), generate_binary_rand(n_points//2 + 1),
        np.random.rand(n_points), h,
        np.zeros(n_points), np.ones(n_points)
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'disp': False, 'rhobeg': 0.1, 'catol': 1e-6})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=50,
                                             mutation=(0.9, 1.0), recombination=0.7, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_neldermead(h_init):
        res = optimize.minimize(objective, h_init, method='Nelder-Mead', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xatol': 1e-8, 'fatol': 1e-8, 'maxiter': 30})
        return res.x if res.success else h_init

        if time.time() - start_time >= budget_s - 10:

        best_h = refine_with_neldermead(best_h)

            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
