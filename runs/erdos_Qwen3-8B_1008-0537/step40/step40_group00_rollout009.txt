import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 30  # Smaller for faster evaluation
    dx = 2.0 / n_points
    target_sum = n_points * 0.5
    initial_h = np.zeros(n_points)
    initial_h[::2] = 1.0  # Start with alternating pattern
    h = initial_h.copy()
    h = h * target_sum / np.sum(h)
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_pattern(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def generate_saw(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty + phase, 0, 1)

    def generate_random():
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        return h

    guesses = [
        h,
        generate_pattern(3, 0.5, 0),
        generate_pattern(5, 0.7, np.pi/4),
        generate_saw(2, 0.6, 0.2),
        generate_saw(3, 0.4, 0.5),
        generate_random(),
        generate_pattern(3, 0.3, np.pi/2)
    ]

    def slsqp_guess(g, iterations=300):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': iterations, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess(popsize=20, maxiter=50):
        res = optimize.differential_evolution(obj, bounds, popsize=popsize, maxiter=maxiter,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(4):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r, iterations=300):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],

        best_h = refine(best_h)
