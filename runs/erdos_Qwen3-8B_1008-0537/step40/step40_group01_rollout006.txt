import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_binary_rand(m):
        idx = np.random.choice(n_points, m, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h

    def generate_spaced(m):
        h = np.zeros(n_points)
        for i in range(m):
            pos = i * (2.0 / m)
            idx = int(pos / dx + 0.5)
            if 0 <= idx < n_points:
                h[idx] = 1
        return h

    guesses = [
        generate_binary_rand(22), generate_binary_rand(23),
        generate_spaced(22), generate_spaced(23),
        np.random.rand(n_points), h,
        np.zeros(n_points), np.ones(n_points)
    ]

    for guess in guesses:
        try:
            res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        except:

        res = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=20,
