import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_rand(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h
    
    guesses = [
        generate_binary_rand(20), generate_binary_rand(21),
        h, np.random.rand(n_points)
    ]
    
    for guess in guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 40:
            break
        res = optimize.differential_evolution(
            objective, bounds, popsize=30, maxiter=30,
            mutation=(0.5, 0.8), recombination=0.8, tol=1e-9
        )
        if res.success:
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP',
                bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.minimize(
            objective, best_h, method='Nelder-Mead',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'xatol': 1e-8, 'fatol': 1e-8, 'maxiter': 50}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
