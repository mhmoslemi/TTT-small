    def constraint_sum(h_vals):

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Differential Evolution for global search
    res_de = optimize.differential_evolution(objective, bounds, popsize=20, maxiter=30, mutation=(0.5, 0.8), recombination=0.8, tol=1e-9)

                                     constraints=[{'type': 'eq', 'fun': constraint_sum}],
