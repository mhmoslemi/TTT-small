import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_uniform_spaced(m):
        dx = 2.0 / n_points
        positions = np.linspace(0, 2, m, endpoint=False)
        indices = np.round(positions / dx).astype(int)
        indices = np.unique(np.clip(indices, 0, n_points - 1))
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    
    def generate_binary_spaced(spacing):
        h = np.zeros(n_points)
        cnt = 0
        for i in range(n_points):
            if i % spacing == 0:
                h[i] = 1
                cnt += 1
                if cnt > n_points // 2:
                    break
        return h
    
    def generate_binary_rand(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        return h
    
    guesses = [
        generate_uniform_spaced(22), generate_uniform_spaced(23),
        generate_binary_spaced(3), generate_binary_spaced(4),

        generate_binary_rand(22), generate_binary_rand(23),
        np.random.rand(n_points), h,
        np.zeros(n_points), np.ones(n_points)
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:

        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=40,

                best_h = res_slsqp.x
