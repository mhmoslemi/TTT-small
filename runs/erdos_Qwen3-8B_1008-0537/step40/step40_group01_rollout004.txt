import time

                pos = i * spacing
                idx = int(pos / dx + 0.5)
                if 0 <= idx < n_points:
                    h[idx] = 1
            return h

                pos = (i * 2.0 / m - 1.0)

                if 0 <= idx < n_points:
                    h[idx] = 1
            return h

            h[idx] = 1
