            if current_c5 < best_c5:

                best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(
            objective, bounds, popsize=60, maxiter=50,
            mutation=(0.6, 0.95), recombination=0.85, tol=1e-10
        )
        if res.success:
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP',
                bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 350, 'ftol': 1e-12, 'eps': 1e-9}
            )
            if res_slsqp.success:
                current_c5 = objective(res_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res_slsqp.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 5:
            break
        res = optimize.minimize(
            objective, best_h, method='Nelder-Mead',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'xatol': 1e-9, 'fatol': 1e-9, 'maxiter': 50}
        )
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    return best_h, best_c5, n_points
