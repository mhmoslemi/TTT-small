import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_wave(freq, duty, wave_type='saw'):
        x = np.linspace(0, 1, n_points)
        if wave_type == 'saw':
            wave = np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif wave_type == 'triangle':
            wave = 1 - np.abs(2 * freq * x - 1) * duty
        else:
            wave = np.sin(2 * np.pi * freq * x)
            wave = np.clip((wave + 1) * duty / 2, 0, 1)
        return wave

    def generate_initial(guesses, freqs, duties, types):
        for f in freqs:
            for d in duties:
                for t in types:
                    yield generate_wave(f, d, t)

    def slsqp_guess(g, iterations=200):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': iterations, 'ftol': 1e-10})
        return res.x if res.success else g

    def de_guess(popsize=50, maxiter=50):
        res = optimize.differential_evolution(obj, bounds, popsize=popsize, maxiter=maxiter,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    guesses = [h]
    
    # Structure: 3 frequencies, 3 duties, 2 wave types
    for f in [2, 3, 4]:
        for d in [0.25, 0.5, 0.75]:
            for t in ['saw', 'triangle']:
                guesses.append(generate_wave(f, d, t))
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(3):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(4):
        if time.time() - start_time > budget_s - 10:
            break
        h_opt = slsqp_guess(best_h, iterations=300)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    return best_h, best_c5, n_points
