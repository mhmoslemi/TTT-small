import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_square(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave > 0) * duty, 0, 1)

    def generate_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def generate_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def generate_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 12):
        guesses.append(generate_square(freq, 0.5))
        guesses.append(generate_saw(freq, 0.5))
        guesses.append(generate_triangle(freq, 0.5))
    for freq in range(2, 6):
        guesses.append(generate_square(freq, 0.3))
        guesses.append(generate_saw(freq, 0.3))
        guesses.append(generate_triangle(freq, 0.3))
    for _ in range(5):
        guesses.append(generate_random())

    def slsqp_guess(g, iterations=500):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],

        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess(popsize=50, maxiter=100):
        res = optimize.differential_evolution(obj, bounds, popsize=popsize, maxiter=maxiter,

        return res.x

    for _ in range(4):
        if time.time() - start_time > budget_s - 20:

        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)

            best_h, best_c5 = h_opt, c

    def refine(r, iterations=300):
