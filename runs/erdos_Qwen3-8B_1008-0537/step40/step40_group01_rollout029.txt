import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_spaced(m):
        h = np.zeros(n_points)
        spacing = 2.0 / m
        for i in range(m):
            pos = i * spacing
            idx = int(pos / dx + 0.5)
            if 0 <= idx < n_points:
                h[idx] = 1
        return h
    
    def generate_binary_rand(m):
        idx = np.random.choice(n_points, m, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h
    
    guessed_h_list = [
        generate_binary_spaced(22), generate_binary_spaced(23), generate_binary_spaced(24),
        generate_binary_rand(22), generate_binary_rand(23), generate_binary_rand(24),
        np.random.rand(n_points), h,

    ]
    
    for guess in guessed_h_list:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success and objective(res.x) < best_c5:
