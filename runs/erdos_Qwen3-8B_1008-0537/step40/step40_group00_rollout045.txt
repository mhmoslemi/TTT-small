    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def make_guess(f, freq, duty, n_bases=5):
        x = np.linspace(0, 1, n_points)
        base = np.zeros(n_points)
        for i in range(n_bases):
            base += np.clip(f(freq * x + i * 0.25), 0, 1)
        return base

    guesses = [h]
    for freq in range(3, 10):
        for duty in [0.25, 0.5, 0.75]:
            guesses.append(make_guess(lambda x: np.sin(x), freq, duty))
            guesses.append(make_guess(lambda x: np.abs(2*x-1), freq, duty))
    for _ in range(5):
        guesses.append(generate_random())

    def slsqp_guess(g, iters=200):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': iters, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess(popsize=50, maxiter=150):
        res = optimize.differential_evolution(obj, bounds, popsize=popsize, maxiter=maxiter,
                                             mutation=(0.7, 1.0), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(4):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r, iters=400):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': iters, 'ftol': 1e-9})
        return res.x if res.success else r

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
