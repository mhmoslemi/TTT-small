import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    target_sum = n_points / 2
    dx = 2.0 / n_points
    # Generate initial binary vector with sum(h) = target_sum
    initial_h = np.zeros(n_points)
    idxs = np.random.choice(n_points, int(target_sum), replace=False)
    initial_h[idxs] = 1.0
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_spaced(m):
        h = np.zeros(n_points)
        spacing = 2.0 / m
        for i in range(m):
            pos = i * spacing
            idx = int(pos / dx + 0.5)
            if 0 <= idx < n_points:
                h[idx] = 1
        return h
    
    def generate_binary_rand(m):
        idxs = np.random.choice(n_points, m, replace=False)
        h = np.zeros(n_points)
        h[idxs] = 1
        return h
    
    guesses = [
        generate_binary_spaced(22), generate_binary_spaced(23),
        generate_binary_rand(22), generate_binary_rand(23),
        initial_h, np.random.rand(n_points),
        np.zeros(n_points), np.ones(n_points)
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=30,

                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],

                best_h = res_slsqp.x
    
    def refine_with_neldermead(h_init):
