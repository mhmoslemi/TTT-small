        return np.max(corr) * dx

        h = np.zeros(n_points)
        cnt = 0
        for i in range(n_points):
            if i % spacing == 0:
                h[i] = 1.0
                cnt += 1
                if cnt > n_points // 2:
                    break
        return h
    
    def generate_binary_rand(ones):

        h = np.zeros(n_points)
        h[idx] = 1.0
        return h
