import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_spaced(spacing):
        h = np.zeros(n_points)
        cnt = 0
        for i in range(n_points):
            if i % spacing == 0:
                h[i] = 1
                cnt += 1
                if cnt > n_points // 2:
                    break
        return h
    
    def generate_binary_rand(ones):
        idx = np.random.choice(n_points, ones, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h
    
    def generate_even_spaced(m):
        positions = np.linspace(0, 2, m)
        h = np.zeros(n_points)
        for x in positions:
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n_points)))
            h[idx] = 1
        return h
    
    def generate_randomized_spaced(m):
        positions = np.random.rand(m) * 2
        h = np.zeros(n_points)
        for x in positions:
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n_points)))
            h[idx] = 1
        return h
    
    guesses = [
        generate_binary_spaced(3), generate_binary_spaced(4),
        generate_binary_spaced(5), generate_binary_spaced(6),
        generate_binary_rand(22), generate_binary_rand(23),
        generate_even_spaced(23), generate_randomized_spaced(23),
        np.random.rand(n_points), h,
        np.zeros(n_points), np.ones(n_points),
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=50,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_neldermead(h_init):
        res = optimize.minimize(objective, h_init, method='Nelder-Mead', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xatol': 1e-8, 'fatol': 1e-8, 'maxiter': 40})
        return res.x if res.success else h_init
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_neldermead(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
