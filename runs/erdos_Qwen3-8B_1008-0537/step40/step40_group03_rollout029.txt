import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        return res.x if res.success else h_init
    
    def refine_with_de(h_init):
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=100,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        return res.x if res.success else h_init
    
    def refine_with_nm(h_init):
        res = optimize.minimize(objective, h_init, method='Nelder-Mead', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 40})
        return res.x if res.success else h_init
    
    # Initial refinement with SLSQP
    best_h = refine_with_slsqp(best_h)
    best_c5 = objective(best_h)
    
    # Differential evolution to explore global optima
    if time.time() - start_time < budget_s - 10:
        best_h = refine_with_de(best_h)
        best_c5 = objective(best_h)
    
    # Final refinement with Nelder-Mead
    if time.time() - start_time < budget_s - 5:
        best_h = refine_with_nm(best_h)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
