import scipy.optimize as optimize
import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_evenly_spaced(k):
        h = np.zeros(n_points)
        count = 0
        for i in range(n_points):
            if i * 2 % (2*k) == 0:
                h[i] = 1.0
                count += 1
                if count > target_sum:
                    break
        return h
    
    def generate_perturbed(h_base, noise=0.1):
        h = h_base.copy()
        idx = np.random.choice(n_points, int(n_points * noise), replace=False)
        h[idx] += np.random.uniform(-0.5, 0.5, size=len(idx))
        h = np.clip(h, 0, 1)
        return h
    
    guesses = [
        generate_evenly_spaced(3), generate_evenly_spaced(4),

    ]
    
    for guess in guesses:
