import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_spaced(spacing):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % spacing == 0:
                h[i] = 1
        return h
    
    def generate_binary_rand(ones):
        return np.zeros(n_points) if ones == 0 else np.random.choice(n_points, ones, replace=False)
    
    initial_guesses = [
        generate_binary_spaced(3), generate_binary_spaced(5), generate_binary_spaced(7),
        generate_binary_rand(22), generate_binary_rand(24), 
        generate_binary_rand(18), generate_binary_rand(20),
        h, np.random.rand(n_points)
    ]
    
    for guess in initial_guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time < budget_s - 30:
        res_de = optimize.differential_evolution(
            objective, bounds, popsize=60, maxiter=30,
            mutation=(0.9, 1.1), recombination=0.8, tol=1e-8
        )
        if res_de.success:
            res_slsqp = optimize.minimize(
                objective, res_de.x, method='SLSQP',
                bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 400, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    if time.time() - start_time < budget_s - 10:
        res_nm = optimize.minimize(
            objective, best_h, method='Nelder-Mead',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 100}
        )
        if res_nm.success and objective(res_nm.x) < best_c5:
            best_c5 = objective(res_nm.x)
            best_h = res_nm.x
    
    return best_h, best_c5, n_points
