import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def generate_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def generate_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def generate_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def generate_harmonic(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * (freq * x + phase) - 1) * duty, 0, 1)

    guesses = [h]
    for freq in range(2, 7):
        guesses.append(generate_pulses(freq, 0.5))
        for duty in [0.2, 0.5, 0.8]:
            guesses.append(generate_saw(freq, duty))
    for duty in [0.3, 0.6]:
        for phase in [0, 0.25, 0.5, 0.75]:
            guesses.append(generate_harmonic(3, duty, phase))
    for _ in range(5):
        guesses.append(generate_random())

    def slsqp_guess(g, iterations=250):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': iterations, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess(popsize=50, maxiter=100):
        res = optimize.differential_evolution(obj, bounds, popsize=popsize, maxiter=maxiter,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h, iterations=200)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
