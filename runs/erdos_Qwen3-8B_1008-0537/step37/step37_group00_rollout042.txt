    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbed_guess(base, std=0.05):
        return np.clip(base + np.random.normal(0, std, n_points), 0, 1)
    
    def generate_binary_guess(n):
        is_ones = np.random.rand(n) < 0.5
        ones = np.sum(is_ones)
        adjust = 22 - ones
        if adjust > 0:
            idxs = np.where(~is_ones)[0][:adjust]
            is_ones[idxs] = True
        elif adjust < 0:
            idxs = np.where(is_ones)[0][:abs(adjust)]
            is_ones[idxs] = False
        if np.sum(is_ones) < 22.5:
            idx = np.random.choice(np.where(is_ones)[0])
            is_ones[idx] = 0.5
        else:
            idx = np.random.choice(np.where(is_ones)[0])
            is_ones[idx] = 0.5
        return is_ones.astype(float)
    
    guesses = [np.array([0.5]*n_points), initial_h_values.copy()]
    guesses.append(generate_binary_guess(n_points))
    for _ in range(4):
        guesses.append(generate_perturbed_guess(initial_h_values))
    
    for guess in guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(
            objective, bounds, popsize=60, maxiter=100,
            mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(
            objective, best_h, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
