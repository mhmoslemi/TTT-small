import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate diverse initial guesses
    guesses = [np.array(initial_h_values)]
    for _ in range(4):
        idxs = np.random.choice(n_points, n_points // 2, replace=False)
        guess = np.zeros(n_points)
        guess[idxs] = 1.0
        guesses.append(guess)
    for _ in range(3):
        idxs = np.random.permutation(n_points)
        guess = np.sin(np.pi * idxs / n_points)
        guess = np.clip((guess + 1) / 2, 0, 1)
        guesses.append(guess)
    
    # Differential evolution with adjusted parameters
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=25, maxiter=20,
                                             mutation=(0.7, 0.8), recombination=0.5, tol=1e-8)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Refinement with adaptive steps
    for _ in range(4):
