import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate structured initial guesses with even spacing
    def generate_even_guess(m_ones, n_points):
        indices = np.round(np.linspace(0, n_points - 1, m_ones)).astype(int)
        h_guess = np.zeros(n_points)
        h_guess[indices] = 1.0
        return h_guess
    
    # Initial guesses: even spacing, random, and Monte Carlo
    guesses = [h.copy()]
    m_ones = int(target_sum) + 1
    if m_ones > 0:
        guesses.append(generate_even_guess(m_ones, n_points))
    guesses.append(np.random.uniform(0, 1, n_points))
    guesses.append(np.random.binomial(1, 0.5, n_points))
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    # Run initial optimizations
    for guess in guesses:
        if time.time() - start_time >= budget_s - 60:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential evolution with adaptive parameters
    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=30,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            refined = refine_with_slsqp(res.x)
            if objective(refined) < best_c5:
                best_c5 = objective(refined)
                best_h = refined
    
    # Final refinement with aggressive SLSQP
    for _ in range(5):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_slsqp(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
