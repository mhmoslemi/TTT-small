            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = res.x
    
    # Differential evolution with adjusted parameters for more exploration
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(
            objective, bounds, popsize=40, maxiter=30,
            mutation=(0.6, 0.8), recombination=0.8, tol=1e-9,
            strategy='best1bin'
        )
        if res.success:
            refined = refine_with_slsqp(res.x)
            curr_c5 = objective(refined)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = refined
    
    # Final refinement with aggressive SLSQP
    for _ in range(5):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_slsqp(best_h)
        curr_c5 = objective(best_h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
    
    return best_h, best_c5, n_points
