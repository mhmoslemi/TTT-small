import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_patterns():
        guesses = []
        for m in [9, 12, 15, 18]:
            x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
            idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
            h_guess = np.zeros(n_points)
            for i in idxs:
                if i < n_points:
                    h_guess[i] = 1.0
            guesses.append(h_guess)
        guesses.append(np.ones(n_points) * 0.5)
        for _ in range(3):
            noise = np.random.uniform(0, 1, n_points)
            guesses.append(np.clip(noise, 0, 1))
        # Regular spaced pattern
        n_ones = int(np.round(target_sum))
        spacing = 2.0 / n_ones
        def loc_to_idx(x):
            return int(x / spacing * n_points)
        h_reg = np.zeros(n_points)
        for i in range(n_ones):
            x = i * spacing
            idx = loc_to_idx(x)
            if idx < n_points:
                h_reg[idx] = 1.0
        guesses.append(h_reg)
        return guesses
    
    guesses = generate_patterns()
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):

        res_de = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=30,
