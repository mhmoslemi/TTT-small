import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate structured initial guesses with periodic patterns
    def generate_periodic_guess(n_points, m):
        w = 1.0 / m
        d = 1.0 / (m - 1)
        h_guess = np.zeros(n_points)
        for i in range(m):
            start = i * (w + d) - d
            end = start + w
            # Map to [0, 2]
            start_idx = int(np.clip(start / dx, 0, n_points - 1))
            end_idx = int(np.clip(end / dx, 0, n_points - 1))
            h_guess[start_idx:end_idx] = 1.0
        return h_guess
    
    guesses = []
    # Add initial guess
    guesses.append(h.copy())
    
    # Add structured guesses for different m values
    for m in [2, 3, 4, 5]:
        h_guess = generate_periodic_guess(n_points, m)
        # Normalize to ensure sum is target_sum
        if np.sum(h_guess) != target_sum:
            h_guess = h_guess * (target_sum / np.sum(h_guess))
        guesses.append(h_guess)
    
    # Add randomized and binary guesses
    guesses.append(np.random.uniform(0, 1, n_points))
    guesses.append(np.random.binomial(1, 0.5, n_points))
    
    # Function to refine with SLSQP
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    # Run initial optimizations
    for guess in guesses:
        if time.time() - start_time >= budget_s - 60:

        if res.success and objective(res.x) < best_c5:
