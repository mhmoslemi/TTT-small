import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    best_c5 = objective(best_h)
    start_time = time.time()

        return np.clip(base + np.random.normal(0, std, n_points), 0, 1)
    
    guesses = [np.array([0.5]*n_points), initial_h_values.copy()]
    for _ in range(6):
