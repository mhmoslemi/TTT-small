import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

    best_c5 = objective(best_h)

                h_guess[idx] = 1.0
        return h_guess
    
    guesses = []
    for num_spikes in [10, 15, 20, 25]:

    def refine_with_slsqp(h_init):

                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-8})
