import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate diverse initial guesses
    guesses = []
    
    # Uniform spacing of 1s
    num_ones = n_points // 2
    spaced_indices = np.linspace(0, n_points - 1, num_ones, endpoint=False).astype(int)
    h_guess_spaced = np.zeros(n_points)
    h_guess_spaced[spaced_indices] = 1.0
    guesses.append(h_guess_spaced)
    
    # Random binary pattern
    guesses.append(np.random.binomial(1, 0.5, n_points))
    
    # Single long interval
    start_idx = n_points // 4
    end_idx = 3 * n_points // 4
    h_guess_long = np.zeros(n_points)
    h_guess_long[start_idx:end_idx] = 1.0
    guesses.append(h_guess_long)
    
    # Multi-interval pattern
    interval_length = n_points // 3
    h_guess_multi = np.zeros(n_points)

    h_guess_multi[interval_length*2:interval_length*3] = 1.0
    h_guess_multi[interval_length*4:interval_length*5] = 1.0
    guesses.append(h_guess_multi)
    
    # Random uniform
    guesses.append(np.random.uniform(0, 1, n_points))
    
    # Function to refine with SLSQP
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    # Run initial optimizations
    for guess in guesses:
        if time.time() - start_time >= budget_s - 60:

        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential evolution for global search
    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:

        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=40,

        if res.success:

            if objective(refined) < best_c5:

                best_h = refined
    
    # Final refinement
    for _ in range(4):
        if time.time() - start_time >= budget_s - 10:

        best_h = refine_with_slsqp(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
