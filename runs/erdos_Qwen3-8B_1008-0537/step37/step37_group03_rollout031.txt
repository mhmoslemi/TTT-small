import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate diverse initial guesses
    guesses = [np.array(initial_h_values)]  # Add the initial guess
    n_ones = n_points // 2
    # Regular pattern guess
    positions = np.linspace(0, 2, n_ones, endpoint=False)
    idxs = np.round(positions / dx).astype(int)
    regular_guess = np.zeros(n_points)
    regular_guess[idxs] = 1.0
    guesses.append(regular_guess)
    
    # Random binary vectors
    for _ in range(3):
        idxs = np.random.choice(n_points, n_points // 2, replace=False)
        guess = np.zeros(n_points)
        guess[idxs] = 1.0
        guesses.append(guess)
    
    # For each guess, run differential evolution
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        # Differential evolution with adjusted parameters
        res = optimize.differential_evolution(
            objective, bounds, popsize=50, maxiter=20, 
            mutation=(0.8, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            # Refine with SLSQP
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
