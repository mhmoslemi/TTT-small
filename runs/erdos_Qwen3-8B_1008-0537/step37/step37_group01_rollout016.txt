import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate structured guesses with varied m values
    m_values = [8, 10, 12, 14, 16, 18, 20, 22]
    guesses = []
    for m in m_values:
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        guesses.append(h_guess)
    
    # Add randomized and binary guesses
    guesses.append(np.random.uniform(0, 1, n_points))
    guesses.append(np.random.binomial(1, 0.5, n_points))
    
    # Function to refine with SLSQP
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    # Run initial optimizations
    for guess in guesses:
        if time.time() - start_time >= budget_s - 60:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential evolution with adjusted parameters
    for _ in range(3):  # Perform two rounds of DE with varying parameters
        if time.time() - start_time >= budget_s - 30:

        mut = (0.6, 0.9) if _ == 0 else (0.5, 0.8)

        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=30,

        if res.success:
