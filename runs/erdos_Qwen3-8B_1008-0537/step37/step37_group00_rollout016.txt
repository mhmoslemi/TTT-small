import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbed_guess(base, std=0.05, scale=1.5):
        return np.clip(base + np.random.normal(0, std * scale, n_points), 0, 1)
    
    def generate_uniform_guess():
        uniform = np.full(n_points, 0.5)
        uniform[:int(n_points/2)] += 0.5
        return np.clip(uniform, 0, 1)
    
    initial_guesses = [np.array([0.5]*n_points), initial_h_values.copy(), generate_uniform_guess()]
    for _ in range(2):
        initial_guesses.append(generate_perturbed_guess(initial_h_values))
    
    for guess in initial_guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 60:

        res = optimize.differential_evolution(

        if res.success:
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
