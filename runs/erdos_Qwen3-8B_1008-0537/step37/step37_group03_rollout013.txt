    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate initial perturbations
    num_perturbations = 10
    for _ in range(num_perturbations):
        if time.time() - start_time >= budget_s - 60:
            break
        # Create a perturbed version
        perturbed_h = best_h.copy()
        # Randomly select some indices to perturb
        indices = np.random.choice(n_points, size=5, replace=False)
        perturbed_h[indices] += np.random.uniform(-0.2, 0.2, size=5)
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        # Run SLSQP
        res = optimize.minimize(objective, perturbed_h, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Further refinement
    for _ in range(5):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
