import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate structured initial guesses with periodic patterns
    def generate_periodic_pattern(period, num_ones):
        pattern = np.zeros(period, dtype=float)
        pattern[:num_ones] = 1.0
        return np.tile(pattern, n_points // period + 1)[:n_points]
    
    guesses = []
    for m in [10, 15, 20]:
        for k in [m//2, m//3, m//4]:
            if k > 0:
                guess = generate_periodic_pattern(m, k)
                guesses.append(guess)
    
    # Add randomized and binary initial guesses
    guesses.append(np.random.uniform(0, 1, n_points))
    guesses.append(np.random.binomial(1, 0.5, n_points))
    
    # Function to refine with SLSQP
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-8, 'disp': False})
        return res.x if res.success else h_init
    
    # Run initial optimizations
    for guess in guesses:
        if time.time() - start_time >= budget_s - 60:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8, 'disp': False})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Global search with differential evolution
    for _ in range(3):

        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=30,
