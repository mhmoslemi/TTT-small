import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_patterns():
        guesses = []
        # Binary patterns
        for m in [9, 12, 15, 18]:
            x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
            idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
            h_guess = np.zeros(n_points)
            for i in idxs:
                if i < n_points:
                    h_guess[i] = 1.0
            guesses.append(h_guess)
        # Uniform distribution
        guesses.append(np.ones(n_points) * 0.5)
        # Random patterns
        for _ in range(1):
            noise = np.random.uniform(0, 1, n_points)
            guesses.append(np.clip(noise, 0, 1))
        # Evenly spaced binary patterns
        m_ones = n_points // 2
        x = np.linspace(0, 2, m_ones + 1)[:-1] + 0.5 * (1.0 / (m_ones + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        guesses.append(h_guess)
        return guesses
    
    guesses = generate_patterns()
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:

        res_de = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=40,

                best_h = res_slsqp.x
    
    def refine_with_slsqp(h_init):
