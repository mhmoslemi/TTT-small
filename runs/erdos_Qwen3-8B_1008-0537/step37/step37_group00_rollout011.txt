import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guesses():
        guesses = [best_h.copy(), initial_h.copy()]
        if n_points < 45:
            guesses.append(np.full(n_points, 0.5))
            guesses.append(np.zeros(n_points))
            guesses.append(np.ones(n_points))
            for _ in range(3):
                guesses.append(np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1))
            for _ in range(3):
                guesses.append(np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1))
        return guesses
    
    guesses = generate_guesses()
    
    for guess in guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(
            objective, bounds, popsize=30, maxiter=50,
            mutation=(0.5, 0.8), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint}],
                options={'maxiter': 60, 'ftol': 1e-9, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:

            options={'maxiter': 30, 'ftol': 1e-9, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
