import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate structured initial guesses
    for m in range(30, 40):
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 / (m + 1)
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        if time.time() - start_time >= budget_s - 60:
            break
        res = optimize.minimize(objective, h_guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential evolution with aggressive parameters
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, 
                                             popsize=30, maxiter=30,
                                             mutation=(0.8, 1.0), recombination=0.9, tol=1e-9)
        if res.success:
            refined = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                       constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                       options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}).x
            if objective(refined) < best_c5:
                best_c5 = objective(refined)
                best_h = refined
    
    # Fine-tuning with gradient descent
    for _ in range(4):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,

        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
