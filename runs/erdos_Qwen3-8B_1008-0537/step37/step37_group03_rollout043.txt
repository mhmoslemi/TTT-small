                        term = h_vals[i] * (1 - h_vals[j])

                        term = h_vals[i] * 1.0

                integral += term
            integral *= dx
            max_c5 = max(max_c5, integral)
        return max_c5
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
