import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate perturbed initial guesses
    perturbations = [np.clip(np.random.normal(0, 0.05, n_points), 0, 1) for _ in range(4)]
    guesses = [h + p for p in perturbations]
    guesses.append(h.copy())  # original guess
    
    # Run SLSQP on each guess
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Further refinement
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                               options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
