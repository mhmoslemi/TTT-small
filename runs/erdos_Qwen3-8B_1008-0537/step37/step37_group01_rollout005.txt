import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_initial_guesses():
        guesses = []
        # Structured guesses: spaced intervals
        for m in [10, 12, 14, 16, 18]:
            step = 2.0 / (m + 1)
            positions = np.arange(0, 2, step) + step / 2
            idxs = np.searchsorted(positions, np.linspace(0, 2, n_points))
            h_guess = np.zeros(n_points)
            for i in idxs:
                if i < n_points:
                    h_guess[i] = 1.0
            guesses.append(h_guess)
        # Randomized guesses
        guesses.append(np.random.uniform(0, 1, n_points))
        guesses.append(np.random.binomial(1, 0.5, n_points))
        # Symmetric guess
        h_sym = np.zeros(n_points)
        for i in range(n_points // 2):
            h_sym[i] = h_sym[-i-1] = 1.0
        guesses.append(h_sym)
        return guesses
    
    guesses = generate_initial_guesses()
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    # Initial optimization
    for guess in guesses:
        if time.time() - start_time >= budget_s - 60:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Global search with DE
    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=40,

    # Final refinement
    for _ in range(4):

        best_h = refine_with_slsqp(best_h)
