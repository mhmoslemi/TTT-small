import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_structured_h(m):
        x = np.linspace(0, 2, n_points)
        period = 2.0 / m
        interval_length = 1.0 / m
        h = np.zeros_like(x)
        for i in range(n_points):
            pos = x[i]
            if pos % period < interval_length:
                h[i] = 1.0
        return h
    
    structured_guess = generate_structured_h(5)
    
    def generate_perturbed_guess(base, std=0.05):
        return np.clip(base + np.random.normal(0, std, n_points), 0, 1)
    
    guesses = [
        np.array([0.5]*n_points),
        initial_h_values.copy(),
        structured_guess.copy(),
        np.random.rand(n_points)
    ]
    
    for _ in range(2):
        guesses.append(generate_perturbed_guess(initial_h_values))
    
    for guess in guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(
            objective, bounds, popsize=60, maxiter=100,

            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP', bounds=bounds,
