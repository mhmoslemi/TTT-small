import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_patterns():
        guesses = []
        # Binary patterns
        for m in [22, 23]:
            step = 2.0 / m
            positions = np.linspace(0, 2 - step, m, endpoint=False) + step / 2
            idxs = np.searchsorted(np.linspace(0, 2, n_points), positions)
            h_guess = np.zeros(n_points)
            for i in np.unique(idxs):
                if i < n_points:
                    h_guess[i] = 1.0
            guesses.append(h_guess)
        # Symmetric patterns
        h_sym = np.zeros(n_points)
        for i in range(n_points):
            h_sym[i] = h[np.min([i, 2 * n_points - 1 - i])]
        guesses.append(h_sym)
        # Perturb initial guess
        noise = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h + noise, 0, 1)
        h_perturb = h_perturb * (target_sum / np.sum(h_perturb))
        guesses.append(h_perturb)
        # Uniform distribution
        guesses.append(np.ones(n_points) * 0.5)
        return guesses
    
    guesses = generate_patterns()
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res_de = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=30,
                                               mutation=(0.5, 0.8), recombination=0.7, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_slsqp(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
