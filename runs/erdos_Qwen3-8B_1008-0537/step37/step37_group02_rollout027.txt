import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_patterns():
        guesses = []
        # Binary patterns
        for m in [9, 12, 15, 18, 21, 24]:
            x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
            idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
            h_guess = np.zeros(n_points)
            for i in idxs:
                if i < n_points:
                    h_guess[i] = 1.0
            guesses.append(h_guess)
        # Uniform distribution
        guesses.append(np.ones(n_points) * 0.5)
        # Random patterns
        for _ in range(5):
            noise = np.random.uniform(0, 1, n_points)
            guesses.append(np.clip(noise, 0, 1))
        # Symmetric patterns
        for _ in range(3):
            sym = np.zeros(n_points)
            mid = n_points // 2
            for i in range(mid):
                sym[i] = 1.0
                sym[n_points - 1 - i] = 1.0
            guesses.append(sym)
        # Block patterns
        for block in [2, 3, 4, 5]:
            h_guess = np.zeros(n_points)
            for i in range(0, n_points, block * 2):
                h_guess[i:i+block] = 1.0
            guesses.append(h_guess)
        return guesses
    
    guesses = generate_patterns()
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)

        res_de = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=50,

                best_h = res_slsqp.x
    
    def refine_with_slsqp(h_init):
