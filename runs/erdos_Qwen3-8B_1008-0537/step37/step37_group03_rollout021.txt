    def generate_valid_guess():
        required_sum = n_points / 2
        m = int(required_sum)
        guess = np.zeros(n_points)
        idxs = np.random.choice(n_points, m, replace=False)
        guess[idxs] = 1.0
        current_sum = np.sum(guess)
        diff = required_sum - current_sum
        remaining_indices = np.setdiff1d(np.arange(n_points), idxs)
        if len(remaining_indices) > 0:
            idx = np.random.choice(remaining_indices)
            guess[idx] = diff
        return guess
    
    # Generate initial guesses
    guesses = [np.array(initial_h_values).copy()]
    for _ in range(5):
        guesses.append(generate_valid_guess())
    for _ in range(5):
        guesses.append(np.full(n_points, 0.5))
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        # Differential evolution
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=50,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-10)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                          constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                          options={'maxiter': 40, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Refine with SLSQP
    for _ in range(5):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
