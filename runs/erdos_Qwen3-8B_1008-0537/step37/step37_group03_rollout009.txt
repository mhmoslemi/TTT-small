import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate multiple initial guesses
    guesses = [np.array(initial_h_values)]
    for _ in range(10):
        idxs = np.random.choice(n_points, n_points // 2, replace=False)
        guess = np.zeros(n_points)
        guess[idxs] = 1.0
        guesses.append(guess)
    for _ in range(5):
        idxs = np.random.choice(n_points, n_points // 2, replace=False)
        guess = np.zeros(n_points)
        guess[idxs] = 1.0
        guesses.append(guess)
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        # Differential evolution
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=50,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Refine further with SLSQP
    for _ in range(5):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
