import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbed_guess(base, std=0.03):
        return np.clip(base + np.random.normal(0, std, n_points), 0, 1)
    
    initial_guesses = [np.array([0.5]*n_points), initial_h_values.copy()]
    for _ in range(2):
        initial_guesses.append(generate_perturbed_guess(initial_h_values))
    
    for guess in initial_guesses:
        try:
            res = optimize.minimize(
                objective, guess, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)

            pass
