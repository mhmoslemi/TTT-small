import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate diverse initial guesses
    guesses = [np.array(initial_h_values)]
    for _ in range(3):
        idxs = np.random.choice(n_points, n_points // 2, replace=False)
        guesses.append(np.zeros(n_points))
        guesses[-1][idxs] = 1.0
    # Add a regular spaced guess
    regular_guess = np.zeros(n_points)
    k = int(np.ceil(n_points / 2))
    for i in range(k):
        regular_guess[i * 2] = 1.0
    if np.sum(regular_guess) > target_sum:
        regular_guess[-1] = 0.0
    guesses.append(regular_guess)
    # Add a checkerboard-like guess
    checker_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            checker_guess[i] = 1.0
    if np.sum(checker_guess) > target_sum:
        checker_guess[-1] = 0.0
    guesses.append(checker_guess)
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 40:
            break
        # Initial simplex optimization
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
