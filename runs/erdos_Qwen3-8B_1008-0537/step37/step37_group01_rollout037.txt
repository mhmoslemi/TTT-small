import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        max_overlap = 0.0
        for k in range(-n_points + 1, n_points):
            k_val = k * dx
            if abs(k_val) > 2.0:
                continue
            corr = 0.0
            for i in range(n_points):
                x = i * dx
                x_k = x + k_val
                if 0 <= x_k < 2:
                    j = int(np.round(x_k / dx))
                    if 0 <= j < n_points:
                        corr += h_vals[i] * (1.0 - h_vals[j])
            max_overlap = max(max_overlap, corr)
        return max_overlap * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate structured guesses
    for m in [10, 12, 14, 16]:
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        h_guess = np.random.permutation(h_guess)
        h_guess = np.clip(h_guess, 0, 1)
        if np.sum(h_guess) * dx < 0.95 * target_sum or np.sum(h_guess) * dx > 1.05 * target_sum:
            continue
        guesses.append(h_guess)
    
    guesses = [h] + guesses + [np.random.uniform(0, 1, n_points), np.random.binomial(1, 0.5, n_points)]
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 60:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential evolution
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=30,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            refined = refine_with_slsqp(res.x)
            if objective(refined) < best_c5:
                best_c5 = objective(refined)
                best_h = refined
    
    # Final refinement
    for _ in range(4):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_slsqp(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
