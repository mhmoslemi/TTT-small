import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_patterns():
        guesses = []
        # Regular binary pattern
        m = n_points // 2
        pos = np.linspace(0, 2, m + 1)[:-1] + 0.5 / (m + 1)
        idxs = np.searchsorted(pos, np.linspace(0, 2, n_points))
        idxs = np.clip(idxs, 0, n_points - 1)
        h_guess = np.zeros(n_points)
        for i in idxs:

        for m in [9, 12, 15]:

            h_guess = np.zeros(n_points)
            for i in idxs:
                h_guess[i] = 1.0
            guesses.append(h_guess)
        # Uniform
        guesses.append(np.ones(n_points) * 0.5)

        for _ in range(3):

    guesses = generate_patterns()
