import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate diverse initial guesses
    guesses = [np.array(initial_h_values)]
    
    # Evenly spaced pattern
    spacing = 2.0 / (n_points // 2 + 1)
    spaced_indices = np.arange(0, n_points, spacing * n_points / 2)
    spaced_indices = np.round(spaced_indices).astype(int)
    spaced_h = np.zeros(n_points)
    for i in spaced_indices:
        if 0 <= i < n_points:
            spaced_h[i] = 1.0
    guesses.append(spaced_h)
    
    # Random cluster
    cluster_size = n_points // 2
    cluster_idx = np.random.choice(n_points, cluster_size, replace=False)
    cluster_h = np.zeros(n_points)
    cluster_h[cluster_idx] = 1.0
    guesses.append(cluster_h)
    
    # Random distribution
    rand_h = np.zeros(n_points)
    rand_idx = np.random.choice(n_points, n_points // 2, replace=False)
    rand_h[rand_idx] = 1.0
    guesses.append(rand_h)
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        # Differential evolution for global search
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=30,

        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
