        total_area = 1.0
        m = 8
        interval_length = total_area / m
        spacing = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (interval_length / spacing) + 0.5 * (interval_length / spacing)
            end = start + interval_length / spacing
            start_idx = int(start)
            end_idx = int(end)
            h[start_idx:end_idx] = 1.0
        return h
    
    guesses = [initial_h_values.copy(), generate_spaced_guess(n_points)]
    
    for _ in range(3):
        guesses.append(generate_perturbed_guess(initial_h_values))
    
    for guess in guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(
            objective, bounds, popsize=30, maxiter=60,
            mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(
            objective, best_h, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
