import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2

    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    guesses = [np.array(initial_h_values)]
    # Randomly perturbed initial guess
    perturb = np.random.normal(0, 0.1, size=n_points)
    guesses.append(np.clip(h + perturb, 0, 1))
    # Randomly spaced guess
    idxs = np.random.choice(n_points, size=n_points // 2, replace=False)
    guess = np.zeros(n_points)
    guess[idxs] = 1.0
    guesses.append(guess)
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        # Differential evolution with tighter parameters
        res = optimize.differential_evolution(
            objective, bounds, popsize=20, maxiter=20,

        )
        if res.success:
            # Refine with SLSQP
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Final refinement
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
