import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_vals):

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

            idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
            h_guess = np.zeros(n_points)
            for i in idxs:
                if i < n_points:
                    h_guess[i] = 1.0

            sym_guess = np.zeros(n_points)
            for i in idxs:
                if i < n_points:
                    sym_guess[i] = 1.0
                    if n_points - 1 - i >= 0:
                        sym_guess[n_points - 1 - i] = 1.0
            h_total = np.sum(sym_guess)
            if h_total > target_sum:
                h_total -= 1
                sym_guess[np.argmax(sym_guess)] = 0
            guesses.append(sym_guess)
        return guesses

    def refine_with_slsqp(h_init):
