import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate more structured initial guesses
    structured_guesses = []
    for m in [5, 8, 12, 16]:  # finer spacing for more structured guesses
        l = 1.0 / m
        s = 1.0 / (m - 1)
        start = 0.0
        while start + l < 2.0:
            end = start + l
            # Find indices in h that fall within [start, end)
            x = np.linspace(0, 2, n_points)
            idx_start = np.searchsorted(x, start)
            idx_end = np.searchsorted(x, end)
            if idx_end < n_points:
                structured_guesses.append(np.zeros(n_points))
                for i in range(idx_start, idx_end):
                    structured_guesses[-1][i] = 1.0
            start += s + l
    
    # Add randomized and binomial guesses
    structured_guesses += [np.random.uniform(0, 1, n_points), np.random.binomial(1, 0.5, n_points)]
    
    # Function to refine with SLSQP
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    # Run initial optimizations
    for guess in structured_guesses:
        if time.time() - start_time >= budget_s - 60:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential evolution for global search with tighter parameters
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=30,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)

                best_h = refined

        best_h = refine_with_slsqp(best_h)
