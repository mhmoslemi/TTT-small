import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate diverse initial guesses
    guesses = [initial_h.copy()]
    for _ in range(10):  # More guesses for exploration
        num_ones = n_points // 2
        indices = np.random.choice(n_points, num_ones, replace=False)
        guess = np.zeros(n_points)
        guess[indices] = 1.0
        guesses.append(guess)
    
    # Run SLSQP on each initial guess
    for guess in guesses:
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Refine with more SLSQP iterations
    for _ in range(5):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
