import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate initial guesses: initial + perturbed variations
    guesses = [h.copy()]
    for _ in range(3):
        perturbation = np.random.normal(0, 0.1, n_points)
        guess = h + perturbation
        guess = np.clip(guess, 0.0, 1.0)
        guesses.append(guess)
    
    # Differential evolution with adaptive parameters
    for guess in guesses:
        if time.time() - start_time >= budget_s - 30:
            break
        # Adaptive differential evolution with restarts
        res = optimize.differential_evolution(
            objective, bounds, popsize=20, maxiter=30,
            mutation=(0.5, 0.9), recombination=0.8, tol=1e-8,
            args=(), workers=1, polish=True,
            strategy='best1bin'
        )
        if res.success:
            # Refine with SLSQP with tighter constraints
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-8}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Further refinement with SLSQP
    for _ in range(5):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(
            objective, best_h, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
