import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbed_guess(base, std=0.05):
        return np.clip(base + np.random.normal(0, std, n_points), 0, 1)
    
    def generate_binary_guess():
        binary = np.random.choice([0, 1], size=n_points)
        s = np.sum(binary)
        scale = (target_sum - s) / (n_points - s) if s < n_points else 0
        return np.clip(binary + scale, 0, 1)
    
    guesses = [np.array([0.5]*n_points), initial_h_values.copy(), generate_binary_guess()]
    
    for _ in range(4):
        guesses.append(generate_perturbed_guess(initial_h_values))
        guesses.append(generate_binary_guess())
    
    for guess in guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
