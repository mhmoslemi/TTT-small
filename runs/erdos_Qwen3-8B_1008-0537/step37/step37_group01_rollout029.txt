import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    guesses = []
    # Generate structured initial guesses
    for m in [12, 14, 16, 18, 20]:
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        guesses.append(h_guess)
    
    # Add randomized initial guess
    guesses.append(np.random.uniform(0, 1, n_points))
    guesses.append(np.random.binomial(1, 0.5, n_points))
    
    # Add even spaced guess
    num_blocks = n_points // 2
    spacing = n_points // num_blocks
    h_even = np.zeros(n_points)
    for i in range(num_blocks):
        idx = int(i * spacing)
        if idx < n_points:
            h_even[idx] = 1.0
    guesses.append(h_even)
    
    # Function to refine with SLSQP
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    # Run initial optimizations
    for guess in guesses:
        if time.time() - start_time >= budget_s - 60:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential evolution with adjusted parameters
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=30,

    # Final refinement with tighter tolerances
    for _ in range(4):

        best_h = refine_with_slsqp(best_h)
