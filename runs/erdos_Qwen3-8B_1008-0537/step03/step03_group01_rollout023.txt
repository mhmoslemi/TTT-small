def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    original_h = np.array(initial_h_values)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def rescale(h):
        s = np.sum(h)
        if s == 0:
            return h
        factor = target_sum / s
        scaled = h * factor
        return np.clip(scaled, 0, 1)
    
    best_h = original_h.copy()
    best_c5 = compute_c5(best_h)
    best_time = time.time()
    
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.9
    
    population = [rescale(original_h)]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = rescale(h)
        population.append(h)
    
    for gen in range(generations):
        if time.time() - best_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        parents = []
        for _ in range(population_size):
            a, b = np.random.randint(0, population_size, 2)
            if fitness[a] < fitness[b]:
                parents.append(population[a])
            else:
                parents.append(population[b])
        
        offspring = []

            if i+1 >= population_size:
