import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_penalty = 1000 * abs(np.sum(h) * dx - 1)
        return max_corr + sum_penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Differential evolution for global search
    if time.time() - start_time < budget_s:
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    # Local refinement with constrained optimization
    if time.time() - start_time < budget_s:
        def constraint_sum(h_values):
            return np.sum(h_values) * dx - target_sum
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
