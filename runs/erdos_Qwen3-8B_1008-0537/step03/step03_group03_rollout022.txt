import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    result = None

    # Generate random starts
    num_starts = 5
    for _ in range(num_starts):
        if time.time() - start_time > budget_s:
            break
        # Generate random h with correct sum
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        result = optimize.minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    # Perturb initial guess
    perturb = np.random.normal(0, 0.05, size=n_points) * 0.5
    perturbed_guess = np.clip(initial_h_values + perturb, 0, 1)
    result = optimize.minimize(
        fun=objective,
        x0=perturbed_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9}
    )
    if result.success:
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    return best_h, best_c5, n_points
