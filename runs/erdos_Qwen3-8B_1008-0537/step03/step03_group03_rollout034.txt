import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    # Generate a few random initial guesses
    def random_guess():
        r = np.random.rand(n_points)
        r = r / np.sum(r) * target_sum
        return r
    
    random_guesses = [random_guess() for _ in range(3)]
    all_guesses = [initial_guess] + random_guesses
    
    for guess in all_guesses:
        if time.time() - start_time > budget_s:
            break
        # Run COBYLA with moderate iterations
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Final run with SLSQP for refinement
    if time.time() - start_time < budget_s:
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 600, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
