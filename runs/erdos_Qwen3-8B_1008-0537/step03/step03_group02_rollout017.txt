import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h0 = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = h0.copy()
    best_c5 = objective(best_h)
    result = None
    
    # Use diverse perturbations and more iterations
    for _ in range(20):  # More iterations for exploration
        if time.time() - start_time > budget_s:
            break
        # Use larger random perturbation for diversification
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_guess = np.clip(h0 + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',  # More robust optimization
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 1500, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:

            if current_val < best_c5:
                best_c5 = current_val
                best_h = result.x
                h0 = best_h  # Use current best as new starting point
    
    # Final refinement with nearby perturbations
    perturb = np.random.uniform(-0.01, 0.01, size=n_points)
