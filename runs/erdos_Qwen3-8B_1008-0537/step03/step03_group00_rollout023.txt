import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        perturb_scale = np.random.uniform(0.05, 0.15)
        perturb = np.random.uniform(-perturb_scale, perturb_scale, size=n_points)
        perturbed = np.clip(initial_guess + perturb, 0, 1)
        res = optimize.minimize(
            objective, perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success:
            current = objective(res.x)
            if current < best_c5:
                best_c5 = current
                best_h = res.x
                initial_guess = best_h
    
    return best_h, best_c5, n_points
