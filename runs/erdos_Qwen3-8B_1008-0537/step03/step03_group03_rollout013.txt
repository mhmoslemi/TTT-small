import scipy.optimize as optimize
import time

    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Perturb and optimize with multiple strategies
    for step in range(6):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.07 * (1 if step < 3 else 0.5), size=n_points) * 0.6
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        # Use SLSQP for more precision
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
