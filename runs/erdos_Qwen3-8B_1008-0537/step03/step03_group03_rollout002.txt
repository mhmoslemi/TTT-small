import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Increase perturbation range and use more diverse starting points
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Use larger perturbations for broader exploration
        perturb = np.random.normal(0, 0.15, size=n_points) * 0.3
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        # Use SLSQP with tighter tolerances for dominance
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-9, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Final refined optimization with even stricter parameters
    perturb = np.random.normal(0, 0.05, size=n_points) * 0.1
    perturbed_guess = np.clip(best_h + perturb, 0, 1)
    result = optimize.minimize(
        fun=objective,

        method='SLSQP',

    )
    if result.success:
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
