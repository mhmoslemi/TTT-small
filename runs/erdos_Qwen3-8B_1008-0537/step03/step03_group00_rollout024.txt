import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(initial_guess, 1 - initial_guess, mode='full')) * dx
    start_time = time.time()
    
    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def constraint(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb_scale = 0.05 + 0.1 * np.random.rand()
        perturbed = np.clip(initial_guess + np.random.normal(0, perturb_scale, n_points), 0, 1)
        res = optimize.minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
                initial_guess = best_h
    
    return best_h, best_c5, n_points
