import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        polish=True,

    )

        constraints=cons,

    )
