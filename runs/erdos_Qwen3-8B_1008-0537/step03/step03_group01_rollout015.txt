import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_diff = np.abs(np.sum(h) - target_sum)
        return max_corr + 1e5 * sum_diff  # Penalize sum deviation
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Differential evolution with adaptive parameters
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-6,
        maxiter=100,
        polish=True,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Early stopping if time exceeds budget
    if time.time() - start_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    # Additional refinement with local search
    result_local = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        tol=1e-8,
        options={'maxiter': 50}
    )
    best_h = result_local.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
