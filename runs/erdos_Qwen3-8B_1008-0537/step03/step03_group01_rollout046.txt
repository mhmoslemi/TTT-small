import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points

    initial_params = initial_h_values[:n_points-1]
    last_val = target_sum - np.sum(initial_params)
    initial_params = np.concatenate([initial_params, [last_val]])

    def objective(params):

        if h[-1] < 0 or h[-1] > 1:

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = optimize.differential_evolution(
