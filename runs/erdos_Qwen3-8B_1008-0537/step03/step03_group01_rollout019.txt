```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e6 * abs(np.sum(h) - target_sum)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_c5 = float('inf')
    best_h = np.array(initial_h_values, dtype=float)

    # Run differential evolution with a tailored strategy
    result = optimize.differential_evolution(
        func=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False
    )

    best_h = result.x
    best_c5 = objective(best_h)

    # Final clipping and normalization
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - target_sum) > 1e-5:
        best_h = best_h / sum_h * target_sum

    return best_h, objective(best_h), n_points
```
