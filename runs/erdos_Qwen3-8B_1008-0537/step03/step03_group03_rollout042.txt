            break
        perturb = np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
