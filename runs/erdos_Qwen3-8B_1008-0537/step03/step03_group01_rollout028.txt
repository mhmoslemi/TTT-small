    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_penalty = 1000 * (np.sum(h) - target_sum)**2
        return max_corr + sum_penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.9,
        popsize=20,
        maxiter=50,
        tol=1e-6,
        polish=True
    )
    
    best_h = result.x
    h = best_h
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_guess, objective(initial_guess) - 1000 * (np.sum(initial_guess) - target_sum)**2, n_points
    
    return best_h, best_c5, n_points
