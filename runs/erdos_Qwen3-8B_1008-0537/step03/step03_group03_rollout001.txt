import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points

    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(10):  # More thorough exploration
        if time.time() - start_time > budget_s:
            break
        # Random perturbation
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(initial_guess + perturb * 0.2, 0, 1)
        result = optimize.minimize(
            objective, perturbed, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            initial_guess = best_h

    # Final fine-tuning
    result = optimize.minimize(
        objective, best_h, bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='SLSQP', options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-8}
    )
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x

    return best_h, best_c5, n_points
