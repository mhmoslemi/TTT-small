import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_param):
        h = np.array(h_param)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_param):
        return np.sum(h_param) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    def optimize_once(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            tol=1e-9,
            options={'maxiter': 2000, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h[:] = result.x
        return best_c5
    
    # Early stopping based on time
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb initial guess
        perturb = np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed = np.clip(initial_guess + perturb, 0, 1)
        optimize_once(perturbed)
        # Use updated guess for next iteration
        initial_guess = best_h
    
    return best_h, best_c5, n_points
