import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import scipy.optimize as optimize
    import time

    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    initial_guesses = [np.array(initial_h_values, dtype=float)]
    # Generate several random initial points
    for _ in range(5):
        guess = np.random.rand(n_points)
        guess /= np.sum(guess)
        guess *= target_sum
        initial_guesses.append(guess)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)

    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
