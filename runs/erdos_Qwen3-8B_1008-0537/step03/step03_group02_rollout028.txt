import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

        corr = np.correlate(h, 1 - h, mode='full')

        current_sum = np.sum(h)
        scaling_factor = (target_sum) / current_sum
        h = h * scaling_factor
        h = np.clip(h, 0, 1)
        return h
    
    methods = ['L-BFGS-B', 'SLSQP', 'COBYLA']
    trial_count = 0
    
    while trial_count < 5 and time.time() - start_time < budget_s:

            h_guess = initial_guess.copy()

            h_guess = generate_random_guess()
        perturb = np.random.normal(0, 0.02, size=n_points)
        h_guess = np.clip(h_guess + perturb, 0, 1)
        
        for method in methods:
            if time.time() - start_time >= budget_s:
                break
            result = optimize.minimize(
                fun=objective,
                x0=h_guess,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
