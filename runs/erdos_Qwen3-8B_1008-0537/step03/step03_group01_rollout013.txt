import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Generate diverse initial guesses
    guesses = [initial_guess]
    base_perturb = np.random.normal(0, 0.02, n_points)
    for _ in range(3):
        perturbed = np.clip(initial_guess + base_perturb * np.random.rand() * 0.5, 0, 1)
        perturbed = (perturbed * target_sum) / np.sum(perturbed)
        guesses.append(perturbed)
    
    # Add a uniform random guess
    uniform = np.random.uniform(0, 1, n_points)
    uniform = (uniform * target_sum) / np.sum(uniform)
    guesses.append(uniform)
    
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        # Optimize with COBYLA for robustness
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Further refine with SLSQP
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
