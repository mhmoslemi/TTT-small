import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    def perturb_guess(h, scale=0.05):
        noise = np.random.normal(0, scale, size=len(h))
        return np.clip(h + noise, 0, 1)
    
    def random_start():
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else np.random.rand(n_points)
        return np.clip(h, 0, 1)
    
    # Initial random start for diversity
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = random_start()
        result = optimize.minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Perturb initial guess and refine
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h = perturb_guess(initial_guess, 0.03)
        result = optimize.minimize(
            fun=objective,
            x0=h,

            options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-10}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Basinhopping for global search
    def basinhopping_optimize(x0):
        res = optimize.basinhopping(
