import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Enhanced exploration with early stopping and diverse initializations
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        # Varying perturbation strategies
        noise_scale = np.random.uniform(0.05, 0.2)
        perturb = np.random.normal(0, noise_scale, size=n_points) * 0.5
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        # Use L-BFGS-B for faster convergence
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    # Final refinement with SLSQP
    if time.time() - start_time < budget_s:
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            best_c5 = objective(result.x)
            best_h = result.x
    
    return best_h, best_c5, n_points
