import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import scipy.optimize as optimize
    import time
    
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e4 * np.abs(np.sum(h) - target_sum)
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Differential Evolution for global search
    try:
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            polish=True,
            disp=False
        )
        if result.success:
            best_h = result.x
            best_c5 = objective(best_h)
    except:
        pass
    
    # Local refinement with SLSQP
    if time.time() - start_time < budget_s - 10:
        perturb = np.random.normal(0, 0.05, size=n_points) * 0.5
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result_slsqp.success:
            best_h = result_slsqp.x
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
