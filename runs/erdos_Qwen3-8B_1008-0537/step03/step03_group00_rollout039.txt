import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import scipy.optimize as optimize
    import time

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective_and_gradient(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        j = np.argmax(corr)
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            term1 = 0.0
            if i <= n - 1 - j:
                term1 = (1 - h[i + j]) * dx
            term2 = 0.0
            if i >= j:
                term2 = -h[i - j] * dx
            grad[i] = term1 + term2
        return max_corr * dx, grad

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective_and_gradient(best_h)[0]

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        result = optimize.minimize(
            fun=lambda h: objective_and_gradient(h)[0],
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            jac=True,
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective_and_gradient(result.x)[0]
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h

    return best_h, best_c5, n_points
