import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    # Run multiple optimizations with different methods and perturbation strategies
    for _ in range(20):  # Increased to 20 iterations
        if time.time() - start_time > budget_s:
            break
        # Perturb using a smaller and decaying step
        step_size = max(0.005, 0.01 * (0.9 ** _))
        perturb = np.random.normal(0, step_size, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:

            if current_c5 < best_c5:
