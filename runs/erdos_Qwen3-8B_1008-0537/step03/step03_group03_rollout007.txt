import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def cons(x):
        return np.sum(x) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h_perturb = h + np.random.uniform(-0.1, 0.1, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        
        res = optimize.minimize(
            fun=obj,
            x0=h_perturb,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': cons}],
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if res.success:
            current_c5 = obj(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
                h = best_h
    
    perturb = np.random.uniform(-0.15, 0.15, n_points)
    h_perturb = np.clip(h + perturb, 0, 1)
    res = optimize.minimize(
        fun=obj,
        x0=h_perturb,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': cons}],
        options={'maxiter': 700, 'ftol': 1e-9, 'eps': 1e-9}
    )
    if res.success:
        current_c5 = obj(res.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = res.x
    
    return best_h, best_c5, n_points
