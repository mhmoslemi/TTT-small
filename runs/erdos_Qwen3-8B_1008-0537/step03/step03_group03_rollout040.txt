import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guesses = []
    
    # Add initial guess from provided data
    initial_guesses.append(np.array(initial_h_values, dtype=float))
    
    # Generate simple step pattern
    def generate_simple_step(n):
        h = np.zeros(n)
        half = n // 2
        h[:half] = 1.0
        h[half] = target_sum - half
        return h
    initial_guesses.append(generate_simple_step(n_points))
    
    # Generate random binary pattern
    def generate_random_binary(n):
        req_sum = n / 2
        idx = np.random.choice(n, size=int(n // 2), replace=False)
        h = np.zeros(n)
        h[idx] = 1.0
        needed = req_sum - len(idx)
        non_idx = np.setdiff1d(np.arange(n), idx)
        chosen = non_idx[0]
        h[chosen] += needed
        return h
    initial_guesses.append(generate_random_binary(n_points))
    
    best_h = initial_guesses[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        # Optimizer 1: COBYLA
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        # Optimizer 2: SLSQP
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
