import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_candidate):
        h = np.array(h_candidate)
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return float('inf')
        scale_factor = target_sum / current_sum
        h_normalized = h_clipped * scale_factor
        h_normalized = np.clip(h_normalized, 0.0, 1.0)
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        tol=1e-6,
        seed=seed,
        x0=initial_guess

    elapsed_time = time.time() - start_time
    best_h = result.x
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
