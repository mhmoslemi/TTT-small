        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': max_iter, 'ftol': 1e-10, 'eps': 1e-9}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    return best_h, best_c5, n_points
