import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import scipy.optimize as optimize
    import time

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    np.random.seed(seed)
    start_time = time.time()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)
    
    # Generate initial candidates
    initial_guesses = [best_h.copy()]
    # Add a random start
    random_guess = np.random.rand(n_points)
    random_guess = random_guess * target_sum / random_guess.sum()
    initial_guesses.append(random_guess)
    # Add a periodic start
    periodic_guess = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5
    periodic_guess = periodic_guess * target_sum / periodic_guess.sum()
    initial_guesses.append(periodic_guess)
    # Add a shuffled start
    shuffled_guess = np.random.permutation(best_h)
    shuffled_guess = shuffled_guess * target_sum / shuffled_guess.sum()
    initial_guesses.append(shuffled_guess)

    def optimize_candidate(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            return result.x, current_c5
        return guess, objective(guess)

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        current_h, current_c5 = optimize_candidate(guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

    return best_h, best_c5, n_points
