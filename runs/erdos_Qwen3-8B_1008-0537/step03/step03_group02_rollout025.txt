import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_guess = np.array(initial_h_values, dtype=np.float64)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        penalty = 1e5 * (np.sum(h) - target_sum)**2
        return current_c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Run global optimization
    try:
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=10, tol=1e-6,
            maxiter=100, mutation=(0.5, 1.0), recombination=0.7
        )
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    except:
        pass
    
    elapsed = time.time() - start_time
    if elapsed <= budget_s:
        # Local refinement
        try:
            result_slqp = minimize(
                objective, best_h, method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
            )
            if result_slqp.success:
                best_h = result_slqp.x
                best_c5 = objective(best_h)
        except:
            pass
    
    return best_h, best_c5, n_points
