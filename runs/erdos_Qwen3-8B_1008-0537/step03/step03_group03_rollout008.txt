import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    def generate_random_guess():
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        return h
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    initial_guesses = [initial_guess]
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        guess = generate_random_guess()
        initial_guesses.append(guess)
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'disp': False}
        )
        if result.success:
            curr = objective(result.x)
            if curr < best_c5:
                best_c5 = curr
                best_h = result.x
    
    if time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.1, size=n_points) * 0.3
        perturbed = np.clip(best_h + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            curr = objective(result.x)
            if curr < best_c5:
                best_c5 = curr
                best_h = result.x
    
    return best_h, best_c5, n_points
