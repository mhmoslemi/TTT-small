        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # COBYLA refinement
    result_cobyla = optimize.minimize(
        fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
        x0=best_h,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        options={'maxiter': 200, 'disp': False}
    )
    if result_cobyla.success:
        current_h = result_cobyla.x
        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # SLSQP refinement
    result_slsqp = optimize.minimize(
        fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_h = result_slsqp.x
        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
