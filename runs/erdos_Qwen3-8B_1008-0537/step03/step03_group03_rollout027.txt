import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    # Enhanced perturbation and optimization
    for _ in range(10):  # More iterations
        if time.time() - start_time > budget_s:
            break
        # Varying perturbations
        noise_scale = np.random.uniform(0.05, 0.15)
        perturb = np.random.normal(0, noise_scale, size=n_points) * 0.6
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        # Use hybrid optimization
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h  # Update initial guess for next iterations
    
    # Final aggressive optimization
    perturb = np.random.normal(0, 0.15, size=n_points) * 0.3
    perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
    result = optimize.minimize(
