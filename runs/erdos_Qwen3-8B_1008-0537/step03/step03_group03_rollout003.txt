import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    # Initialize with perturbations and random starting points
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Perturb initial guess with varying noise
        perturb = np.random.normal(0, 0.05, size=n_points) * 0.5
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        # Run TRUST-CONSTR with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='trust-constr',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'disp': False, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    # Add a final run with higher perturbation and TRUST-CONSTR
    perturb = np.random.normal(0, 0.1, size=n_points) * 0.3
    perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
    result = optimize.minimize(
        fun=objective,

        method='trust-constr',

    )
    if result.success:

        if current_c5 < best_c5:

    return best_h, best_c5, n_points
