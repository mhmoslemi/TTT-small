    def perturb(h, scale=0.05):
        return np.clip(h + np.random.normal(0, scale, h.shape), 0, 1)
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        current_h = perturb(best_h)
        result = optimize.minimize(
            fun=objective,
            x0=current_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
