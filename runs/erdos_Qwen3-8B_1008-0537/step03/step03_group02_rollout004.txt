import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guesses = [np.array(initial_h_values, dtype=float)]
    
    # Generate a random valid initial guess
    def generate_random_guess():
        rng = np.random.Generator(np.random.PCG64(seed))
        guess = rng.uniform(0, 1, size=n_points)
        guess = guess * (target_sum / np.sum(guess))
        guess = np.clip(guess, 0, 1)
        return guess
    
    initial_guesses.append(generate_random_guess())
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)
    
    for _ in range(25):  # 25 iterations
        if time.time() - start_time > budget_s:
            break
        initial_guess = initial_guesses[np.random.randint(len(initial_guesses))]
        result = optimize.minimize(
            fun=objective,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guesses[-1] = result.x
    
    return best_h, best_c5, n_points
