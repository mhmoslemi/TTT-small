import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    initial_guesses = []
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Generate diverse initial guesses
        np.random.seed(seed + 1000 * _)
        candidates = np.random.uniform(0, 1, n_points)
        candidates = np.sort(candidates)
        cuts = np.cumsum([1] * n_points) / n_points

        candidates = np.append(candidates, np.repeat(0, n_points - len(candidates)))
        candidates = candidates[:n_points]
        candidates = np.divide(candidates, np.sum(candidates)) * target_sum
        initial_guesses.append(candidates)

    # Evaluate initial guesses
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = optimize.minimize(
            fun=objective,

        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:

    # Refine with SLSQP
