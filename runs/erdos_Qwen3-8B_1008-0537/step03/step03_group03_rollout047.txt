import time

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_random_guess(n, target_sum):
        h = np.random.rand(n)
        scale = target_sum / h.sum()
        h *= scale
        h = np.clip(h, 0.0, 1.0)
        return h
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    # Generate starting points
    starting_points = [best_h]
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points) * 0.5

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

        h *= scale
        h = np.clip(h, 0.0, 1.0)

    start_time = time.time()

        result = optimize.minimize(
