import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guesses = [np.array(initial_h_values, dtype=float)]
    for _ in range(5):
        h = np.random.uniform(0, 1, n_points)
        h *= (target_sum / np.sum(h))
        initial_guesses.append(h)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    best_h = np.copy(initial_guesses[0])
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 400, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    if time.time() - start_time < budget_s - 5:
        perturb = np.random.normal(0, 0.05, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    return best_h, best_c5, n_points
