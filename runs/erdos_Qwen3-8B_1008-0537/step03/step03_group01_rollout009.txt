import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Generate an alternative initial guess with random fluctuations
    alt_initial_guess = np.clip(initial_guess + np.random.normal(0, 0.05, size=n_points), 0, 1)
    
    # Run multiple optimizations with different starting points
    for _ in range(8):
        if time.time() - start_time > budget_s:
            break
        # Alternate starting methods to avoid local optima
        if _ % 2 == 0:
            current_guess = initial_guess.copy()
        else:
            current_guess = alt_initial_guess.copy()
        
        result = optimize.minimize(
            fun=objective,
            x0=current_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    return best_h, best_c5, n_points
