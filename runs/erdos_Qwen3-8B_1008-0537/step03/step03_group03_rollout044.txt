        n_points = 45  # Default number of points
        # Generate a binary initial guess with 22 or 23 ones
        num_ones = n_points // 2
        initial_guess = np.zeros(n_points)
        random_indices = np.random.choice(n_points, size=num_ones, replace=False)
        initial_guess[random_indices] = 1.0
        initial_h_values = initial_guess

    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - (n_points / 2.0)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    # Evaluate multiple starting points
    starting_points = [initial_guess]
    # Add a randomized binary starting point
    num_random = 1
    for _ in range(num_random):
        num_ones = n_points // 2
        random_indices = np.random.choice(n_points, size=num_ones, replace=False)
        random_guess = np.zeros(n_points)
        random_guess[random_indices] = 1.0
        starting_points.append(random_guess)
    
    # Run optimization for each starting point
    for guess in starting_points:
        perturbed_guess = np.clip(guess, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Additional refinement with perturbations
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Add random noise to the current best
        perturb = np.random.normal(0, 0.05, size=n_points) * 0.2
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
