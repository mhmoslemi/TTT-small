        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    iteration = 0
    
    # Use a more efficient optimization by scaling perturbations
    for _ in range(30):  # 30 iterations to explore
        if time.time() - start_time > budget_s:
            break
        # Use different perturbation strategy: larger initial steps, then smaller
        if iteration < 10:
            std = 0.05  # Larger perturbation early on
        else:
            std = 0.01  # Smaller later
        perturb = np.random.normal(0, std, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h  # Update initial guess for next iteration
            iteration += 1
    
    return best_h, best_c5, n_points
