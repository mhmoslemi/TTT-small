        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    # Optimization methods to try
    methods = ['SLSQP', 'L-BFGS-B', 'COBYLA', 'Nelder-Mead', 'differential_evolution']
    
    while time.time() - start_time < budget_s:
        # Randomly choose optimization method
        method = np.random.choice(methods)
        
        if method == 'differential_evolution':
            # Define objective with penalty for constraint violation
            def obj_de(h_values):
                h = np.array(h_values)
                corr = np.correlate(h, 1 - h, mode='full')

                penalty = 1e4 * np.abs(np.sum(h) - target_sum)
                return max_corr + penalty
            
            # Generate random initial guess that satisfies sum constraint
            h_rand = np.random.rand(n_points)

            result = optimize.differential_evolution(
