        random_h = random_h * target_sum / np.sum(random_h)
        random_guesses.append(random_h)
    
    all_initial_guesses = [initial_guess] + random_guesses
    
    # Run optimization for each initial guess
    for guess in all_initial_guesses:
        # Perturb the guess slightly to avoid same starting point
        perturb = np.random.normal(0, 0.05, size=n_points) * 0.5
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        
        # Run COBYLA optimization
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h  # Use updated guess
    
    # Add a final run with a higher perturbation
    perturb = np.random.normal(0, 0.1, size=n_points) * 0.3
    perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
    result = optimize.minimize(
        fun=objective,
        x0=perturbed_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
    )
    if result.success:
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
