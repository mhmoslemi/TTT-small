import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Random guess generator
    def generate_random_guess():
        random_part = np.random.uniform(0, 1, n_points-1)
        sum_part = np.sum(random_part)
        if sum_part == 0:
            last_val = 1.0
        else:
            last_val = max(0.0, min(1.0, (target_sum - sum_part)))
        return np.concatenate([random_part, [last_val]])
    
    # Run multiple trials with different optimization methods
    for _ in range(15):  # 15 trials to explore diverse regions
        if time.time() - start_time > budget_s:
            break
        # Generate random guess
        current_guess = generate_random_guess()
        # Use L-BFGS-B for faster convergence
        result = optimize.minimize(
            fun=objective,
            x0=current_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_val = objective(result.x)
            if current_val < best_c5:
                best_c5 = current_val
                best_h = result.x
        # Try COBYLA in case L-BFGS-B gets stuck
        result_cobyla = optimize.minimize(
            fun=objective,
            x0=current_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result_cobyla.success:
            current_val_cobyla = objective(result_cobyla.x)
            if current_val_cobyla < best_c5:
                best_c5 = current_val_cobyla
                best_h = result_cobyla.x
    
    return best_h, best_c5, n_points
