```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Multi-phase optimization
    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        
        # Dynamic perturbation strategy
        perturb = np.random.normal(0, 0.05 if phase == 0 else 0.02, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        # Execute SLSQP with adaptive settings
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={
                'maxiter': 100 if phase == 0 else 50,
                'ftol': 1e-9,
                'eps': 1e-8,
                'disp': False
            }
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    return best_h, best_c5, n_points
```
