import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Initial guess 1: provided initial_h_values
    initial_guess1 = np.array(initial_h_values, dtype=float)
    # Initial guess 2: checkerboard pattern with adjusted value
    checkerboard = np.zeros(n_points)
    checkerboard[::2] = 1.0
    sum_check = np.sum(checkerboard)
    diff = target_sum - sum_check
    if diff != 0:
        idx = np.where(checkerboard == 0)[0][0]
        checkerboard[idx] += diff
    initial_guess2 = checkerboard

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    best_h = initial_guess1.copy()
    best_c5 = objective(best_h)

    # Candidate initial guesses
    initial_guesses = [initial_guess1, initial_guess2]

    for guess in initial_guesses:
        perturb_scale = np.random.uniform(0.02, 0.1)
        perturb = np.random.normal(0, perturb_scale, size=n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 1000, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    # Additional trials with random perturbations of the best guess
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        perturb_scale = np.random.uniform(0.01, 0.05)
        perturb = np.random.normal(0, perturb_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
