import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import scipy.optimize as optimize
    import time

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e4 * np.abs(np.sum(h) - target_sum)
        return max_corr + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Run differential evolution with penalty
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    result_x = result.x
    result_c5 = objective(result_x)
    if result_c5 < best_c5:
        best_h = result_x
        best_c5 = result_c5

    # Refine with SLSQP
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    result_refined = optimize.minimize(
        fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    best_h = result_refined.x
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return best_h, best_c5, n_points
