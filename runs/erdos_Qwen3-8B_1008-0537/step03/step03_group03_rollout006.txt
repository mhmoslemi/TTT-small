import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e5 * (np.sum(h) - target_sum)**2
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Run differential evolution within budget
    def run_diff_evolution():
        result = optimize.differential_evolution(
            objective, bounds, strategy='best1bin',
            maxiter=100, popsize=20, tol=1e-6,
            mutation=(0.5, 1), recombination=0.8,
            seed=seed, vectorized=True
        )
        return result.x if result.success else best_h
    
    while time.time() - start_time < budget_s - 10:
        result = run_diff_evolution()
        if np.allclose(np.sum(result), target_sum, atol=1e-3):
            best_h = result
            best_c5 = objective(result)
            break
    
    # Refine with SLSQP
    def refine_func(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds_refine = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    try:
        result_refine = optimize.minimize(
            refine_func, best_h, method='SLSQP',
            bounds=bounds_refine, constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result_refine.success:
            best_h = result_refine.x
            best_c5 = objective(best_h)
    except:
        pass
    
    return best_h, best_c5, n_points
