import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # First, run differential evolution
    try:
        result_diff = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            constraints=cons,
            popsize=15,
            maxiter=100,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        best_h = result_diff.x
        best_c5 = objective(best_h)
    except:
        pass
    
    # Refine with SLSQP
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-9, 'eps': 1e-9}
    )
    current_c5 = objective(result_slsqp.x)
    if current_c5 < best_c5:
        best_h = result_slsqp.x
        best_c5 = current_c5
    
    # Final refinement with a simple descent
    result_final = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9}
    )
    current_c5 = objective(result_final.x)
    if current_c5 < best_c5:
        best_h = result_final.x
        best_c5 = current_c5
    
    return best_h, best_c5, n_points
