import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e4 * abs(np.sum(h) - target_sum)
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Use differential evolution for global search with decayed penalty
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=25,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Refine with SLSQP to escape local optima
    if time.time() - start_time < budget_s * 0.7:
        def constraint_sum(h_values):
            return np.sum(h_values) - target_sum
        
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            best_h = result_slsqp.x
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
