import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_h_values
    best_c5 = objective(best_h)

    # Generate a uniform initial guess
    uniform_guess = np.full(n_points, 0.5)
    best_c5_uniform = objective(uniform_guess)
    best_h_uniform = uniform_guess

    # Evaluate best so far
    best_h, best_c5 = (best_h_uniform, best_c5_uniform) if best_c5_uniform < best_c5 else (best_h, best_c5)

    start_time = time.time()
    for _ in range(6):  # Run multiple optimizations
        if time.time() - start_time > budget_s:
            break
        # Use SLSQP for refinement
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={
                'maxiter': 200,
                'ftol': 1e-10,
                'eps': 1e-10,
                'disp': False
            }
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Alternatively, try uniform guess
        result_uniform = optimize.minimize(
            fun=objective,

            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],

            }
        )

                best_h = result_uniform.x

    return best_h, best_c5, n_points
