import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def obj(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def cons(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        # Multi-stage perturbation: global, local, and hybrid
        perturb = np.zeros_like(initial_guess)
        for _ in range(3):
            perturb += np.random.normal(0, 0.05, n_points)
            perturb += np.random.uniform(-0.05, 0.05, n_points)
            perturb /= 3
        guess = np.clip(initial_guess + perturb, 0, 1)
        result = optimize.minimize(
            fun=obj,
            x0=guess,
