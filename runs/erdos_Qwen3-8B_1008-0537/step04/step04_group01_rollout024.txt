import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    try:
        result = optimize.differential_evolution(
            fun=objective,

            popsize=10,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tolerance=1e-6,
            maxiter=100,
            disp=False,
            polish=True,
            seed=seed,
            vectorize=True,
        )
    except:
        result = optimize.minimize(
            fun=objective,
            x0=initial_h_values,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-9}
        )

    best_h = result.x
    best_c5 = objective(best_h)
