import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    iterations = 0
    
    # Use a diverse set of initial perturbations and optimized methods
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Randomly perturb the guess
        perturb_scale = np.random.uniform(0.01, 0.15)
        perturb = np.random.normal(0, perturb_scale, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        # Use L-BFGS-B for faster convergence on smooth problems
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={
                'maxiter': 200,
                'ftol': 1e-10,
                'eps': 1e-9,
                'disp': False
            }
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
            iterations += 1
    
    # Add a final refinement with SLSQP for precision
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={
            'maxiter': 300,
            'ftol': 1e-12,
            'eps': 1e-9,
            'disp': False
        }
    )
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
