        scale = target_sum / current_sum
        h *= scale
        return h

        h = [1.0 if i % 2 == 0 else 0.0 for i in range(n_points)]
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = np.array(h) * scale
        return h
    
    initial_guesses = [
        np.array(initial_h_values, dtype=float),
        generate_random_initial(),
        generate_block_initial()
    ]
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Make a copy to avoid overwriting
        current_guess = guess.copy()
        # Perform optimization from this guess
        for _ in range(10):  # fewer iterations per guess to save time
            if time.time() - start_time > budget_s:
                break
            perturb_scale = np.random.uniform(0.01, 0.15)
            perturb = np.random.normal(0, perturb_scale, size=n_points)
            perturbed_guess = np.clip(current_guess + perturb, 0, 1)
            
            result = optimize.minimize(

                    best_h = result.x
