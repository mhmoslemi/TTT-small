    def create_even_guess(n_points):
        m = n_points // 2
        required_sum = n_points / 2
        dx = 2.0 / n_points
        positions = np.linspace(0, 2, m + 1)[1:-1]
        h = np.zeros(n_points)

            x_i = i * dx
            for pos in positions:
                if abs(x_i - pos) < dx / 2:
                    h[i] = 1
                    break
        current_sum = np.sum(h) * dx
        diff = required_sum - current_sum
        if diff != 0:
            h[-1] += diff / dx
            h[-1] = np.clip(h[-1], 0, 1)
        return h
    
    even_initial_guess = create_even_guess(n_points)
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()

        perturb = np.random.normal(0, noise_scale, size=n_points)
