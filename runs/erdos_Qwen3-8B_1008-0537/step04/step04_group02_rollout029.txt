        args=(),
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        init='latinhypercube',
        seed=seed
    )
    
    if result.success:
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
