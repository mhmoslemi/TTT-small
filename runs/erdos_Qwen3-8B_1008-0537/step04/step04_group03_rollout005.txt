import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h0 = np.array(initial_h_values, dtype=float)
    one_minus_h = 1.0 - h0

    def objective(h):
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h0.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(40):
        if time.time() - start_time > budget_s:
            break
        # Adaptive noise scaling
        noise_scale = max(0.005, 0.01 * (0.9 ** _))
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)

        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )

        if result.success:
            curr_c5 = objective(result.x)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = result.x

    return best_h, best_c5, n_points
