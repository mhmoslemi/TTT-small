import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Generate a few random initial guesses
    def random_guess():
        h = np.random.uniform(0, 1, n_points)
        scale = target_sum / np.sum(h)
        return np.clip(h * scale, 0, 1)
    
    guesses = [initial_guess]
    for _ in range(3):
        guesses.append(random_guess())
    
    # Try each guess with COBYLA and SLSQP
    for guess in guesses:
        for method in ['L-BFGS-B', 'COBYLA']:
            if time.time() - start_time > budget_s:
                break
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method=method,
                bounds=bounds,

                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
