        # Penalty for sum deviation
        sum_dev = np.abs(np.sum(h) - target_sum)
        return max_corr + 1e6 * sum_dev

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def create_evenly_spaced(n):

        return h

    alt_guess = create_evenly_spaced(n_points)

    # Use differential evolution
    result = optimize.differential_evolution(

        popsize=15,

        disp=False,

        best_c5 = final_c5

    # Local refinement with SLSQP
    def constraint_sum(h_values):

    result_slsqp = optimize.minimize(
