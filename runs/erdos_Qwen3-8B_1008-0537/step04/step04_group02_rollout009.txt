import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the initial guess if available
    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        # If no initial guess is available, create a random one with sum n_points / 2
        n_points = 45
        h = np.random.rand(n_points)
        target_sum = n_points / 2
        h = h / np.sum(h) * target_sum

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)

    # Use COBYLA for robustness and bounds handling
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        # Generate diverse perturbations
        perturb = np.random.normal(0, 0.075, size=n_points)
        perturbed_guess = np.clip(h + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                h = best_h  # Update the current guess

    return best_h, best_c5, n_points
