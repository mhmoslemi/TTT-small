import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Use a more intensive search with adaptive perturbations and multiple iterations
    for iteration in range(100):
        if time.time() - start_time > budget_s:
            break
        # Adaptive perturbation scale based on iteration
        perturb_scale = np.clip(0.1 * (1 - iteration / 100), 0.01, 0.15)
        perturb = np.random.normal(0, perturb_scale, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={
                'maxiter': 200,
                'ftol': 1e-10,
                'eps': 1e-9,
                'disp': False
            }
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    return best_h, best_c5, n_points
