import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def create_even_spaced_guess(n):
        m = n // 2
        if m == 0:
            return np.zeros(n)
        positions = np.linspace(0, 2, m, endpoint=False)
        positions = np.round(positions / dx).astype(int)
        positions = np.unique(positions)
        guess = np.zeros(n)
        for i in positions:
            if 0 <= i < n:
                guess[i] = 1.0
        return guess
    
    initial_guess = np.array(initial_h_values, dtype=float)
    even_spaced_guess = create_even_spaced_guess(n_points)
    alternate_initial_guess = np.zeros(n_points)
    alternate_initial_guess[:n_points//2] = 1.0
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    candidates = [initial_guess, even_spaced_guess, alternate_initial_guess]
    best_h = None
    best_c5 = float('inf')
    
    # Evaluate initial candidates
    for guess in candidates:
        current_c5 = objective(guess.copy())
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = guess.copy()
    
    # Optimize from best candidate
    for _ in range(40):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.01 * np.random.rand() + 0.001
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
