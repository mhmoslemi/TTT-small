        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 1e3 * abs(sum_h - target_sum)
        return max_overlap + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_h = initial_guess
        best_c5 = objective(best_h)
    else:
        best_h = result.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
