import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_h_penalty = 1e6 * abs(np.sum(h) - target_sum)
        return max_corr + sum_h_penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Use differential evolution for global search
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False,
        seed=seed
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Refine with SLSQP
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
