import numpy as np

    def constraint_sum(h_values):

import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    initial_guess = np.array(initial_h_values, dtype=float)
    
    # Generate alternate and random initial guesses
    alternate_initial_guess = np.zeros(n_points)
    alternate_initial_guess[:n_points//2] = 1.0
    idxs = np.random.choice(n_points, size=int(n_points//2), replace=False)
    random_initial_guess = np.zeros(n_points)
    random_initial_guess[idxs] = 1.0
    
    initial_guesses = [initial_guess, alternate_initial_guess, random_initial_guess]
    initial_c5s = [objective(g) for g in initial_guesses]
    best_initial_idx = np.argmin(initial_c5s)
    best_h = initial_guesses[best_initial_idx].copy()
    best_c5 = initial_c5s[best_initial_idx]
    
    start_time = time.time()
    
    # Optimize from the best initial guess
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 400, 'ftol': 1e-10, 'eps': 1e-9}
    )
    
    if result.success:
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Additional perturbations
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.015 + 0.005 * np.random.rand()
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
