import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    best_iter = 0
    
    # Generate structured initial guess
    def generate_structured_guess():
        idx = np.argsort(np.random.rand(n_points))

        h[idx[:int(n_points/2)]] = 1
        h = np.clip(h + (np.random.rand(n_points) - 0.5) * 0.1, 0, 1)
        h = np.clip(h * (n_points / 2) / np.sum(h), 0, 1)
        return h
    
    # Alternative initial guess
    alt_guess = generate_structured_guess()
    candidates = [initial_guess, alt_guess]
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        for guess in candidates:
            result = optimize.minimize(
                fun=objective,
