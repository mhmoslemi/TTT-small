def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2

    def objective(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        max_overlap = 0.0
        for m in range(-(n_points - 1), n_points):
            sum_overlap = 0.0
            for i in range(n_points):
                j = i + m
                if 0 <= j < n_points:
                    sum_overlap += h_values[i] * (1 - h_values[j])
            max_overlap = max(max_overlap, sum_overlap * dx)
        return max_overlap

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    for _ in range(20):
        if time.time() - start_time > budget_s:

        perturb_scale = np.random.uniform(0.01, 0.15)

        result = optimize.minimize(
