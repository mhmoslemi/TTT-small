import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)
    
    initial_guesses = [best_h]
    # Add binary pattern
    num_ones = n_points // 2
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary_guess = np.zeros(n_points)
    binary_guess[indices] = 1.0
    initial_guesses.append(binary_guess)
    # Add random guess
    random_guess = np.random.rand(n_points)
    random_guess = np.clip(random_guess, 0, 1)
    random_guess = (random_guess * n_points - np.sum(random_guess)) / (n_points - target_sum)
    initial_guesses.append(random_guess)
    
    for guess in initial_guesses:
        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    return best_h, best_c5, n_points
