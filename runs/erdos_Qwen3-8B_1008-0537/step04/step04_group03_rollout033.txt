import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def optimize_guess(h):
        res = optimize.minimize(obj, h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': cons}],

    best_h = h0

    best_c5 = obj(best_h)
