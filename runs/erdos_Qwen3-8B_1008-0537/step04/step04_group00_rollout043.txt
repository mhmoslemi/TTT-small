import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    def generate_rand_guess():
        num_ones = int(np.floor(target_sum))
        remaining = target_sum - num_ones
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, num_ones, replace=False)
        h[indices] = 1
        if remaining > 1e-10:
            idx = np.argmin(h)
            h[idx] = remaining
        return h
    
    initial_guesses = [initial_guess.copy()]
    for _ in range(5):
        initial_guesses.append(generate_rand_guess())
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    
    return best_h, best_c5, n_points
