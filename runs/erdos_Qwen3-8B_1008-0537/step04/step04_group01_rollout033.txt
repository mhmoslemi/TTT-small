import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Ensure initial guess has correct sum
    initial_guess = np.array(initial_h_values, dtype=float)
    if not np.isclose(np.sum(initial_guess), target_sum):
        scale = target_sum / np.sum(initial_guess)
        initial_guess = initial_guess * scale
        initial_guess = np.clip(initial_guess, 0, 1)
    
    def objective_de(h_values):
        h = np.array(h_values)
        h_sum = np.sum(h)
        if h_sum == 0 or h_sum == target_sum:
            return float('inf')
        scale = target_sum / h_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    # Use differential evolution with early stopping
    result = optimize.differential_evolution(
        fun=objective_de,
        bounds=bounds,
        strategy='best1bin',
        popsize=30,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-8,
        maxiter=500,
        polish=True,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective_de(best_h)
    
    # Handle timeout by checking elapsed time
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # Fall back to initial guess
        best_h = initial_guess
        best_c5 = objective_de(best_h)
    
    return best_h, best_c5, n_points
