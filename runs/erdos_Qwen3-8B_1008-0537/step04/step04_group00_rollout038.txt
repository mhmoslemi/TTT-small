import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        polish=True
    )
    
    best_h = result.x
    best_h_normalized = best_h / np.sum(best_h) * target_sum
    best_c5 = objective(best_h_normalized)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_guess, objective(initial_guess), n_points
    
    return best_h_normalized, best_c5, n_points
