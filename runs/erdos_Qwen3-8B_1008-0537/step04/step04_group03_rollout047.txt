import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    # Alternate initial guess
    alternate_initial = np.zeros(n_points, dtype=float)
    alternate_initial[:n_points//2] = 1
    
    # Random initial guess
    random_initial = np.random.rand(n_points)
    random_initial = random_initial / np.sum(random_initial) * target_sum

    # Evaluate initial guesses
    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    best_h = initial_guess.copy()
    best_c5 = evaluate(best_h)
    best_alt = evaluate(alternate_initial)
    best_rand = evaluate(random_initial)
    
    # Select best initial guess
    if best_alt < best_c5:
        best_h = alternate_initial.copy()
        best_c5 = best_alt
    if best_rand < best_c5:
        best_h = random_initial.copy()
        best_c5 = best_rand
    
    # Optimization setup
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint_sum = lambda x: np.sum(x) - target_sum
    
    start_time = time.time()
    best_result = best_h.copy()
    best_c5_result = best_c5
    
    # Run optimization with COBYLA
    result = optimize.minimize(
        fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5_result:
            best_result = result.x
            best_c5_result = current_c5
    
    # Apply perturbations and refine with SLSQP
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, size=n_points)
        perturbed_guess = np.clip(best_result + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5_result:
                best_result = result.x
                best_c5_result = current_c5
    
    return best_result, best_c5_result, n_points
