import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 1e4 * (sum_h - target_sum) ** 2
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Try multiple optimization strategies
    strategies = [
        ('differential_evolution', optimize.differential_evolution,
         {'strategy': 'best1bin', 'maxiter': 50, 'popsize': 15, 'tol': 1e-6}),
        ('SLSQP', optimize.minimize,
         {'method': 'SLSQP', 'bounds': bounds,
          'constraints': [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
          'options': {'maxiter': 200, 'ftol': 1e-10}})
    ]
    
    for name, solver, params in strategies:
        if time.time() - start_time > budget_s:
            break
        if name == 'differential_evolution':
            result = solver(
                fun=objective,
                bounds=bounds,
                **params
            )
            if result.success:
                current_h = result.x
                current_c5 = objective(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        else:
            result = solver(
                fun=objective,
                x0=initial_guess,
                **params
            )
            if result.success:
                current_h = result.x
                current_c5 = objective(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        initial_guess = best_h
    
    return best_h, best_c5, n_points
