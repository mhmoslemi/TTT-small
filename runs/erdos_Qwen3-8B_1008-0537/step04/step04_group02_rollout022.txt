import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Multi-start with adaptive perturbation
    for _ in range(40):  # Increased to 40 iterations for better exploration
        if time.time() - start_time > budget_s:
            break
        # Adaptive perturbation: scale based on current solution
        perturb_scale = 0.15 * np.std(best_h)
        perturb = np.random.normal(0, perturb_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        
        # Use L-BFGS-B with bounds and constraints
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={
                'maxiter': 100,
                'ftol': 1e-9,
                'eps': 1e-9,
                'disp': False
            }
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Further refinement with SLSQP
    for _ in range(20):  # Refinement phase
        if time.time() - start_time > budget_s:
            break
        # Small perturbation to avoid local optima
        perturb = np.random.normal(0, 0.02, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

            }
        )
