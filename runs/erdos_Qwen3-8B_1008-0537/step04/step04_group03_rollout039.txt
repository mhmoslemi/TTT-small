import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Generate initial guesses
    initial_guess = np.array(initial_h_values, dtype=float)
    np.random.seed(seed)
    random_guess = np.random.rand(n_points)
    random_guess_sum = np.sum(random_guess)
    scaling_factor = (target_sum) / random_guess_sum
    random_guess = np.clip(random_guess * scaling_factor, 0, 1)
    alternate_initial_guess = np.zeros(n_points)
    alternate_initial_guess[:n_points//2] = 1
    initial_guesses = [initial_guess, random_guess, alternate_initial_guess]

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    start_time = time.time()

    num_iterations = 20
    noise_scale = 0.01
    decay_factor = 0.99

    for i in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        for guess in initial_guesses:
            perturb = np.random.normal(0, noise_scale, size=n_points)
            perturbed_guess = np.clip(guess + perturb, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        noise_scale *= decay_factor

    return best_h, best_c5, n_points
