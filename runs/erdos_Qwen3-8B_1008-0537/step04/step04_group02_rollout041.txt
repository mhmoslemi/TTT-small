    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Run multiple optimization stages with different methods and parameters
    for _ in range(15):  #Increased iterations for better convergence
        if time.time() - start_time > budget_s:
            break
        
        # Perturb initial guess with different noise strategies
        perturb_small = np.random.normal(0, 0.03, n_points)
        perturb_large = np.random.normal(0, 0.08, n_points)
        perturbed_guess = np.clip(initial_guess + 0.5 * perturb_small + 0.5 * perturb_large, 0, 1)
        
        # Use L-BFGS-B for faster convergence with bounds
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'gtol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    return best_h, best_c5, n_points
