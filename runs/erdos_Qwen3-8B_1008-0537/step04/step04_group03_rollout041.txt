import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def create_spaced_intervals(n_points, m):
        t = 1.0 / m
        d = 1.0 / (m - 1)
        intervals = []
        current = 0.0
        for _ in range(m):
            intervals.append((current, current + t))
            current += t + d
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for (s, e) in intervals:
                if s <= x < e:
                    h[i] = 1.0
                    break
        return h

    # Generate spaced intervals initial guess
    h_spaced = create_spaced_intervals(n_points, 5)

    # Initial guesses
    h_initial = initial_h_values
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    c5_initial = objective(h_initial)
    c5_spaced = objective(h_spaced)
    if c5_spaced < c5_initial:
        best_h = h_spaced
    else:
        best_h = h_initial

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_c5 = c5_initial

    for _ in range(50):  # Increased iterations

        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        result = optimize.minimize(

            method='SLSQP',

                best_h_final = result.x

    return best_h_final, best_c5, n_points
