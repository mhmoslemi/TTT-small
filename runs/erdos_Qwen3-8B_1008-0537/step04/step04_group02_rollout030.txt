import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Better initial guess: evenly spaced 1s
    n_ones = n_points // 2
    if n_points % 2 == 1:
        n_ones += 1
    positions = np.linspace(0, 2, n_ones, endpoint=False)
    h_initial = np.zeros(n_points)
    for pos in positions:
        idx = int(np.round(pos / dx))
        idx = np.clip(idx, 0, n_points - 1)
        h_initial[idx] = 1
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    
    # Use L-BFGS-B and multiple random restarts
    for _ in range(15):
        if time.time() - start_time > budget_s:
            break
        # Perturb initial guess
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_guess = np.clip(h_initial + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                h_initial = best_h
    
    return best_h, best_c5, n_points
