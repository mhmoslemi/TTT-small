        initial_guess[idx:] = 0.0
    
    n_points = len(initial_guess)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e5 * (np.sum(h) - target_sum)**2
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Run differential evolution to explore global minima
    try:
        result = optimize.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False,
            seed=seed
        )
        best_h_candidate = result.x
        # Perform SLSQP refinement
        def constraint_sum(h_values):
            return np.sum(h_values) - target_sum
        result_slsqp = optimize.minimize(
            fun=lambda x: objective(x),
            x0=best_h_candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        best_h = result_slsqp.x
        # Compute final C5
        h = np.array(best_h)
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx
    except:
        # Fallback to initial guess if differential evolution fails
        pass
    
    # Ensure sum constraint is satisfied
    if np.abs(np.sum(best_h) - target_sum) > 1e-6:
        def constraint_sum(h_values):
            return np.sum(h_values) - target_sum
        result_slsqp = optimize.minimize(
            fun=lambda x: objective(x),
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        best_h = result_slsqp.x
        h = np.array(best_h)
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return best_h, best_c5, n_points
