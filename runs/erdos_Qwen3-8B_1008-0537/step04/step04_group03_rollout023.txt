import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def create_uniform_guess(n):
        return np.zeros(n, dtype=float)
    
    def create_alternate_guess(n):
        return np.zeros(n, dtype=float)
    
    def create_period_guess(n, period):
        guess = np.zeros(n, dtype=float)
        for i in range(n):
            if i % period < n // 2:
                guess[i] = 1.0
        return guess
    
    alternate_initial_guess = create_alternate_guess(n_points)
    period_initial_guess = create_period_guess(n_points, 3)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    best_alt_h = alternate_initial_guess.copy()
    best_alt_c5 = objective(best_alt_h)
    best_per_h = period_initial_guess.copy()
    best_per_c5 = objective(best_per_h)
    
    if best_alt_c5 < best_c5:
        best_h = best_alt_h
        best_c5 = best_alt_c5
    
    if best_per_c5 < best_c5:
        best_h = best_per_h
        best_c5 = best_per_c5
    
    for _ in range(15):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.01 + 0.01 * np.random.rand()
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,

            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        
        if result.success:
