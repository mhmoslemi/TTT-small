import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    warm_start = initial_guess.copy()
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        # Vary perturbation strength and use warm start
        perturb_scale = np.random.uniform(0.005, 0.02)
        perturb = np.random.normal(0, perturb_scale, size=n_points)
        perturbed_guess = np.clip(warm_start + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={
                'maxiter': 1000,
                'ftol': 1e-10,
                'eps': 1e-9,
                'disp': False
            }
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                warm_start = best_h
    
    return best_h, best_c5, n_points
