import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)

    # Generate random starting points using Dirichlet distribution
    np.random.seed(seed)
    num_start_points = 5
    start_points = [best_h.copy()]
    for _ in range(num_start_points):
        dirichlet_samples = np.random.dirichlet([1.0]*n_points, size=1)
        start_point = dirichlet_samples[0] * target_sum
        start_points.append(start_point)

    # Run optimizations on each starting point
    for start_point in start_points:
        if time.time() - start_time > budget_s:
            break
        result = optimize.minimize(
            fun=objective,
            x0=start_point,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5

import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_h.copy()

    np.random.seed(seed)

    num_start = 5
    start_points = [initial_h.copy()]
    for _ in range(num_start):
        dirichlet = np.random.dirichlet([1.0]*n_points, size=1)
        start_point = dirichlet[0] * target_sum
        start_points.append(start_point)
    
    # Run optimizations on each starting point

            bounds=bounds,

            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
