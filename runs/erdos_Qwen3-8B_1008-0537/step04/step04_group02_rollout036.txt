        perturb = np.random.normal(0, 0.1, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    return best_h, best_c5, n_points
