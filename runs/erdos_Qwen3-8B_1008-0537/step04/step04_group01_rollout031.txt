import time

    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_binary_initial():
        num_ones = n_points // 2 + (1 if n_points % 2 else 0)
        spacing = n_points / num_ones
        idx = np.round(np.arange(num_ones) * spacing).astype(int)
        h = np.zeros(n_points)
        h[idx] = 1.0
        return h

    def generate_random():
        h = np.random.uniform(0, 1, n_points)
        h *= (target_sum / np.sum(h))
        return h

    initial_guesses = [np.array(initial_h_values), generate_binary_initial(), generate_random()]

    best_h = np.array(initial_h_values)
    best_c5 = objective(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        perturbed = np.clip(guess + np.random.normal(0, 0.1, n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_points,
