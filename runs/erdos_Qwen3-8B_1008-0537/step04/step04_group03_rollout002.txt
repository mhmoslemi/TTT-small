    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    random_guess = np.random.rand(n_points)
    random_guess -= random_guess.mean()

    best_h = initial_guess.copy()
    best_c5 = float('inf')
    
    def objective(h_values):
        h = np.array(h_values, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    for _ in range(3):

        # Use a more aggressive optimizer
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
