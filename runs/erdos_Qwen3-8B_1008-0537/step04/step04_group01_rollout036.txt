import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    perturbation_scales = [0.15, 0.1, 0.05, 0.025]
    current_scale_idx = 0
    
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        perturb_scale = perturbation_scales[current_scale_idx]
        perturb = np.random.normal(0, perturb_scale, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={
                'maxiter': 400,
                'disp': False,
                'rhobeg': 0.5,
                'rhoend': 0.05
            }
        )
        
        if result.success:

                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
                current_scale_idx = min(current_scale_idx + 1, len(perturbation_scales) - 1)
    
    return best_h, best_c5, n_points
