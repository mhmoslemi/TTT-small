        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-4,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Refine with SLSQP for local optimization
    h_refine = best_h.copy()
    h_refine = h_refine * (target_sum / np.sum(h_refine))
    
    def refine_objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    result_refine = optimize.minimize(
        fun=refine_objective,
        x0=h_refine,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    
    best_h_refined = result_refine.x
    best_c5_refined = refine_objective(best_h_refined)
    best_h = best_h_refined
    best_c5 = best_c5_refined
    
    # Check budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
