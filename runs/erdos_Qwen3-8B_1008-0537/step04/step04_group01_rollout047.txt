import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the initial_h_values if available, otherwise create a default
    try:
        n_points = len(initial_h_values)
    except NameError:
        n_points = 100  # Default n_points if parent construction not available
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guesses = []
    
    # Generate multiple initial guesses
    np.random.seed(seed)
    # Add the original initial_h_values
    initial_guesses.append(np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else np.random.rand(n_points))
    
    # Add a random guess with sum ≈ target_sum
    random_guess = np.random.rand(n_points) / 2
    random_guess = (random_guess - random_guess.mean()) * (target_sum / np.sum(random_guess)) if np.sum(random_guess) != 0 else random_guess
    random_guess = np.clip(random_guess, 0, 1)
    initial_guesses.append(random_guess)
    
    # Add a binary guess if n_points is even
    if n_points % 2 == 0:
        binary_guess = np.zeros(n_points)
        binary_guess[:n_points//2] = 1.0
        initial_guesses.append(binary_guess)
    
    # Define objective function
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Define constraint
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Early exit if time is up
        if time.time() - start_time > budget_s:
            break
        # Optimize this guess
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={
                'maxiter': 800,
                'ftol': 1e-12,
                'eps': 1e-9,
                'disp': False
            }
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
