import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_diff = np.abs(np.sum(h) - target_sum)
        penalty = sum_diff * 1e4
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    # Differential Evolution for global search
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=40,
        polish=True,
        x0=initial_guess
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check time and refine with SLSQP if time permits
    if time.time() - start_time < budget_s - 10:
        def constraint_sum(h_values):
            return np.sum(h_values) - target_sum
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        best_h = result_slsqp.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
