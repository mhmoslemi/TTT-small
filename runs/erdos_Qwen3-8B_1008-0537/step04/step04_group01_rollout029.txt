        recombination=0.7,
        seed=seed,
        polish=True,
        disp=False
    )
    
    # Ensure the best solution is considered
    if result.success:
        final_h = result.x
        final_c5 = objective(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    else:
        # Fallback to initial guess if optimization fails
        final_h = initial_guess
        final_c5 = best_c5
    
    return best_h, best_c5, n_points
