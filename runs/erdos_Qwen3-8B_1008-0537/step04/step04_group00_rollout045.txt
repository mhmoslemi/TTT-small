import numpy as np

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = target_sum / total
        h_scaled = h * scale
        h_clipped = np.clip(h_scaled, 0.0, 1.0)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_clipped
        scale_final = target_sum / sum_clipped
        return h_clipped * scale_final

    def objective(h_values):
        h_norm = normalize(h_values)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)

    try:
        result = optimize.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=10,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-10,
            polish=True
        )
        if result.success:
            best_h = result.x
            best_c5 = objective(best_h)
