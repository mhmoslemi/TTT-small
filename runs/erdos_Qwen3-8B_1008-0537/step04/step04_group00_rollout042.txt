import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 45
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Generate a sinusoidal initial guess
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 + 0.5 * np.sin(np.pi * x)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial.astype(float)
    initial_guess = h_initial

    def objective(h_values):
        h = np.array(h_values, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None

    # Use more efficient and varied optimization
    for _ in range(40):
        if time.time() - start_time > budget_s:
            break
        # Perturb with adaptive scaling
        perturb_scale = np.random.uniform(0.005, 0.02)
        perturb = np.random.normal(0, perturb_scale, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h  # Use updated guess

    return best_h, best_c5, n_points
