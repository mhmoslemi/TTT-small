import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        current_sum = np.sum(h)
        penalty = 1e4 * abs(current_sum - target_sum)
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Use differential evolution for global search
    result = optimize.differential_evolution(
        func=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=150,
        popsize=20,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    # Optional: refine with local optimization
    def objective_local(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    result_local = optimize.minimize(
        fun=objective_local,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
    )
    
    best_h = result_local.x
    best_c5 = objective_local(best_h)
    
    return best_h, best_c5, n_points
