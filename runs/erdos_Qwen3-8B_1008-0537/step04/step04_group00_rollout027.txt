```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e5 * (np.sum(h) - target_sum)**2
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    # Use differential evolution for global exploration
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        seed=seed
    )
    
    if result.success:
        candidate = result.x
        c5 = objective(candidate)
        if c5 < best_c5:
            best_h = candidate
            best_c5 = c5
    
    # Final refinement with SLSQP
    def objective_clean(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds_clean = [(0.0, 1.0) for _ in range(n_points)]
    result_clean = optimize.minimize(
        fun=objective_clean,
        x0=best_h,
        method='SLSQP',
        bounds=bounds_clean,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
    )
    
    if result_clean.success:
        best_h = result_clean.x
        best_c5 = objective_clean(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
```
