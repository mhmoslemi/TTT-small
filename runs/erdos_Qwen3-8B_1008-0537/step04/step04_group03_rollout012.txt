        best_h = random_guess
        best_c5 = random_c5
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.05 * np.random.rand()
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        
        result = optimize.minimize(
            fun=evaluate,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-5, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = evaluate(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
