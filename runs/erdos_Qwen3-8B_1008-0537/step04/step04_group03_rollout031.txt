    best_c5 = objective(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        guess_sum = np.sum(guess)
        if abs(guess_sum - target_sum) > 1e-6:
            scale = target_sum / guess_sum
            guess = np.clip(guess * scale, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
