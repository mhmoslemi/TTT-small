                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    # Try a random start
    np.random.seed(seed + 100)
    random_guess = np.random.rand(n_points)
    random_guess = random_guess / np.sum(random_guess) * target_sum
    result = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result.success and objective(result.x) < best_c5:
        best_c5 = objective(result.x)
        best_h = result.x
    
    return best_h, best_c5, n_points
