        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    best_alt_h = alternate_initial_guess.copy()
    best_alt_c5 = objective(best_alt_h)

    if best_alt_c5 < best_c5:
        best_h = best_alt_h
        best_c5 = best_alt_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.01 + 0.005 * np.random.rand()
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
