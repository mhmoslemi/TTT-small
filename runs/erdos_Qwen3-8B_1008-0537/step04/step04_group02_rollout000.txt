import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Initialize with given starting point
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    initial_guess = best_h
    
    # Define optimization parameters
    max_iter = 15  # More iterations with adaptive perturbation
    perturbation_scale = 0.075  # Initial scale
    decay_rate = 0.95  # Perturbation decay per iteration
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Generate a new perturbed guess
        perturb = np.random.normal(0, perturbation_scale, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        # Use COBYLA for bounded optimization
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
        # Decay perturbation scale for better local refinement
        perturbation_scale *= decay_rate
