import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def generate_even_guess(m):
        indices = np.linspace(0, n_points - 1, m, endpoint=False, dtype=int)
        h = np.zeros(n_points)
        h[indices] = 1.0
        diff = target_sum - np.sum(h)
        if diff > 0:
            idx = np.random.choice(np.where(h == 0)[0], size=int(diff), replace=False)
            h[idx] += 0.5
        elif diff < 0:
            idx = np.random.choice(np.where(h == 1)[0], size=int(-diff), replace=False)
