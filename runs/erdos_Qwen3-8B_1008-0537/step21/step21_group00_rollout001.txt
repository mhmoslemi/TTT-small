import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(x):

        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

        return np.sum(x) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    
    def spaced_guess(num_ones):
        indices = np.linspace(0, n_points - 1, num_ones).astype(int)
        h_guess = np.zeros(n_points)
        h_guess[indices] = 1.0
        if np.sum(h_guess) < target_sum:
            idx = np.random.choice(np.where(h_guess == 0)[0], size=1)
            h_guess[idx] = 0.5
        return h_guess
    
    initial_guesses = [
        h, spaced_guess(20), spaced_guess(21), spaced_guess(22),
        spaced_guess(23), spaced_guess(24), spaced_guess(25)
    ]
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective, x0=guess, method='SLSQP',
                bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None

    for guess in initial_guesses:

            best_c5 = objective(result)
            best_h = result
    
    try:
        de_result = optimize.differential_evolution(

        )
        if de_result.success:
