import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def generate_varied_guess(m, spacing='structured'):

            for x in positions:
                idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - x))
                h[idx] = 1.0
            if np.sum(h) < target_sum:
                idx = np.random.choice(np.where(h == 0)[0], size=1)
                h[idx] = 0.5
            return h
        else:
            indices = np.random.choice(n_points, m, replace=False)
            h = np.zeros(n_points)
            h[indices] = 1.0

                idx = np.random.choice(np.where(h == 0)[0], size=1)
                h[idx] = 0.5
            return h
    
    initial_guesses = [initial_guess]
    for m in [22, 23]:  # Exact target summation
        initial_guesses.append(generate_varied_guess(m, spacing='structured'))

    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
