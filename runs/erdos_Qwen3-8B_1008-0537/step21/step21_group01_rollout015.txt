import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_spaced_guess(num_ones):

        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_random_guess():

        if excess > 0:
            indices = np.where(h == 1)[0]
            to_reduce = np.random.choice(indices, size=int(excess), replace=False)
            h[to_reduce] = 0.5
        elif excess < 0:
            indices = np.where(h == 0)[0]
            to_increase = np.random.choice(indices, size=int(-excess), replace=False)
            h[to_increase] = 0.5
        return h
    
    initial_guesses = [initial_guess]

    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,

                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
