import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def generate_varied_guess(num_ones, spacing='random'):
        indices = np.random.choice(n_points, num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_structured_guess(m):
        dx = 2.0 / n_points
        positions = np.linspace(0, 2, m + 1)[1:]
        indices = np.round(positions / dx).astype(int)
        indices = np.clip(indices, 0, n_points - 1)
        h = np.zeros(n_points)
        h[indices] = 1.0
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum > target_sum:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] -= (current_sum - target_sum)
        return h
    
    initial_guesses = [initial_guess]
    for _ in range(2):

    def optimize_guess(guess):

            result = optimize.minimize(

                x0=guess,

                bounds=bounds,
