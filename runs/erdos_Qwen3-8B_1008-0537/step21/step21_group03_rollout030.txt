import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_symmetric_guess(num_ones):
        if num_ones <= 0:
            return np.zeros(n_points)
        mid = n_points // 2
        indices = []
        for _ in range(num_ones // 2):
            i = np.random.randint(0, mid)
            if i not in indices and (n_points - 1 - i) not in indices:
                indices.append(i)
                indices.append(n_points - 1 - i)
        if num_ones % 2 == 1:
            indices.append(mid)
        h = np.zeros(n_points)
        for i in indices:
            h[i] = 1.0
        if np.sum(h) < target_sum:
            idxs = np.where(h == 0)[0]
            idx = np.random.choice(idxs, size=1)
            h[idx] = 0.5
        return h
    
    def generate_varied_guess(num_ones):
        indices = np.random.choice(n_points, num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess]
    for num_ones in [20, 22, 24, 26]:
        initial_guesses.append(generate_symmetric_guess(num_ones))
        initial_guesses.append(generate_varied_guess(num_ones))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:

    for guess in initial_guesses:

        if result is not None and objective(result) < best_c5:

            best_h = result
    
    try:
        result_de = optimize.differential_evolution(

        if result_de.success:

            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
