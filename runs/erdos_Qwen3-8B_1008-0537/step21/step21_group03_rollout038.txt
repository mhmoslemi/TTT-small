def run(seed=42, budget_s=100, **kwargs):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

            guess = np.zeros(n_points)
            indices = np.random.choice(n_points, num_ones, replace=False)
            guess[indices] = 1.0
            if np.sum(guess) < target_sum:
                idx = np.random.choice(np.where(guess == 0)[0], size=1)
                guess[idx] = 0.5
            guesses.append(guess)
        # Add a structured guess with periodic intervals
        num_ones = int(target_sum + 0.5)
        spacing = n_points / num_ones
        indices = np.arange(0, n_points, spacing).astype(int)
