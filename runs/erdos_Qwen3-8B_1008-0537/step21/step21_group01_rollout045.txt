import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 40
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_h = np.zeros(n_points)
    best_c5 = 1.0

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess: checkerboard pattern
    h = np.zeros(n_points)
    h[::2] = 1.0
    current_sum = np.sum(h)
    if current_sum < target_sum:
        idx = np.random.choice(np.where(h == 0)[0], 1)
        h[idx] = 0.5
    best_h = h.copy()
    best_c5 = objective(best_h)

    # Generate random initial guess
    def generate_random_guess(n):
        h = np.zeros(n)
        ones = int(np.round(target_sum))
        idx = np.random.choice(n, ones, replace=False)
        h[idx] = 1.0
        while np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], 1)
            h[idx] = 0.5
        return h

    initial_guesses = [generate_random_guess(n_points)]

    start_time = time.time()
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x

    # Differential evolution for global search
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=150,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass

    # Final refinement
    while time.time() - start_time < budget_s - 50:
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x

    return best_h, best_c5, n_points
