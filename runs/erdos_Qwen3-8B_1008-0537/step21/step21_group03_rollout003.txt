import scipy.optimize as optimize
import time

    def objective(h_values):
        h = np.array(h_values)

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_initial_guesses():
        guesses = []
        for _ in range(4):
            h_rand = np.random.rand(n_points)
            h_rand = (h_rand < 0.7).astype(float)
            h_rand /= np.sum(h_rand)
            h_rand *= target_sum
            h_rand += np.random.rand(n_points) * 0.05
            h_rand = np.clip(h_rand, 0, 1)
            guesses.append(h_rand)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:

    try:
        result_de = optimize.differential_evolution(
            fun=objective,

            mutation=(0.8, 1.0),

            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,

        pass
