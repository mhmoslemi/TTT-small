import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_varied_guess(num_ones, spacing='random'):

            indices = np.random.choice(n_points, num_ones, replace=False)
        elif spacing == 'uniform':
            indices = np.linspace(0, n_points, num_ones, endpoint=False, dtype=int)
            indices = np.sort(indices)
        else:
            indices = np.random.choice(n_points, num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess]

            initial_guesses.append(generate_varied_guess(num_ones, spacing='uniform'))
    
    def optimize_guess(guess):
