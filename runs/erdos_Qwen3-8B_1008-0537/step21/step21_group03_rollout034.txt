```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h0 = np.array(initial_h_values, dtype=float)
    
    def objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h0.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        # Differential evolution with moderate parameters
        result = optimize.differential_evolution(
            objective, bounds, popsize=100, maxiter=150,
            mutation=(0.7, 1.0), recombination=0.7, tol=1e-10,
            vectorized=True, seed=seed)
        if result.success:
            best_h = result.x
            best_c5 = objective(best_h)
    except:
        pass
    
    # Refine with SLSQP
    try:
        result = optimize.minimize(
            objective, best_h, method='SLSQP',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9})
        if result.success and objective(result.x) < best_c5:
            best_h = result.x
            best_c5 = objective(best_h)
    except:
        pass
    
    # Local optimization with L-BFGS-B
    try:
        result = optimize.minimize(
            objective, best_h, method='L-BFGS-B',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
        if result.success and objective(result.x) < best_c5:
            best_h = result.x
            best_c5 = objective(best_h)
    except:
        pass
    
    # Final refinement with SLSQP
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                objective, best_h, method='SLSQP',
                bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}],
                options={'maxiter': 20, 'ftol': 1e-10, 'eps': 1e-9})
            if result.success:
                best_h = result.x
                best_c5 = objective(best_h)
        except:
            break
    
    return best_h, best_c5, n_points
```
