        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_varied_guess(num_ones, spacing='random'):

            positions = np.linspace(0, 2, num_ones, endpoint=False)
            indices = np.round(positions / dx).astype(int)
            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:
                idx = np.random.choice(np.where(h == 0)[0], size=1)
                h[idx] = 0.5

        else:

            h = np.zeros(n_points)
            h[indices] = 1.0

                h[idx] = 0.5
            return h
    
    initial_guesses = [initial_guess]
