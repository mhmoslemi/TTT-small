import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_evenly_spaced(num_ones):
        indices = np.linspace(0, n_points-1, num_ones, dtype=int)
        h = np.zeros(n_points)
        h[indices] = 1.0
        diff = target_sum - np.sum(h)
        if diff > 0:
            idx = np.random.choice(np.where(h == 0)[0], size=int(diff), replace=False)
            h[idx] = 0.5
        return h
    
    def generate_random(num_ones):
        indices = np.random.choice(n_points, num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        diff = target_sum - np.sum(h)
        if diff > 0:
            idx = np.random.choice(np.where(h == 0)[0], size=int(diff), replace=False)
            h[idx] = 0.5
        return h
    
    def generate_clustered(num_ones):
        cluster_size = num_ones // 2
        cluster_pos = np.random.choice(n_points, 2, replace=False)
        h = np.zeros(n_points)
        h[cluster_pos[0]:cluster_pos[0] + cluster_size] = 1.0
        h[cluster_pos[1]:cluster_pos[1] + cluster_size] = 1.0
        diff = target_sum - np.sum(h)
        if diff > 0:
            idx = np.random.choice(np.where(h == 0)[0], size=int(diff), replace=False)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess]
    for _ in range(2):
        for num_ones in [15, 20, 25, 30]:
            initial_guesses.append(generate_evenly_spaced(num_ones))
            initial_guesses.append(generate_random(num_ones))
            initial_guesses.append(generate_clustered(num_ones))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=150,
            mutation=(0.5, 1.0),
            recombination=0.7,
            tol=1e-9,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:

    while time.time() - start_time < budget_s - 50:

                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
