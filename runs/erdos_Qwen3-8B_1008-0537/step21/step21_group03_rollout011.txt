import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def initial_guess_sparse():
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, 15, replace=False)
        h[idx] = 1.0
        return h
    
    def initial_guess_even():
        h = np.zeros(n_points)
        idx = np.arange(n_points)[::3]
        h[idx] = 1.0
        return h
    
    def initial_guess_distinct():
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, 20, replace=False)
        h[idx] = 1.0
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    initial_guesses = [np.random.rand(n_points), np.zeros(n_points) + 0.5]
    for _ in range(3):
        initial_guesses.append(initial_guess_sparse())
        initial_guesses.append(initial_guess_even())
        initial_guesses.append(initial_guess_distinct())
    
    best_h = np.random.rand(n_points)
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None:
            current_c5 = objective(result)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=200,
            mutation=(0.9, 1.1),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
