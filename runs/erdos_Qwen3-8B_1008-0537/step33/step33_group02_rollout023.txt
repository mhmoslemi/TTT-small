import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_square(freq):
        x = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sign(np.sin(2 * np.pi * freq * x)), 0, 1)

    guesses = [h]
    for freq in [3, 4, 5, 6, 7, 8]:
        guesses.append(gen_pulses(freq, 0.5))
        guesses.append(gen_square(freq))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
        guesses.append(gen_triangle(3, duty))
    guesses.append(np.random.rand(n_points))
    guesses.append(np.random.rand(n_points) * 0.5 + 0.25)
    guesses.append(np.random.rand(n_points) * 0.3 + 0.15)
    guesses.append(np.random.rand(n_points) * 0.7 + 0.1)

    def perturb_and_norm(h_base, scale=0.1):
        perturb = np.random.normal(0, scale, n_points)
        perturb_h = h_base + perturb
        perturb_h = (perturb_h - np.min(perturb_h)) / (np.max(perturb_h) - np.min(perturb_h))
        perturb_h = (perturb_h * target_sum / np.sum(perturb_h)) * (n_points / 2)
        return np.clip(perturb_h, 0, 1)

    for _ in range(3):
        guesses.append(perturb_and_norm(h))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g

        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

        if time.time() - start_time > budget_s - 15:

        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=100,

        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
