    def obj(h_vals):

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()

    def gen_periodic(freq, duty):

        wave = 0.5 * (1 - np.cos(2 * np.pi * freq * x))
        return np.clip(wave * duty, 0, 1)
    
    def gen_markov(density):
        return np.random.rand(n_points) < density
    
    guesses = [h]

        guesses.append(gen_periodic(freq, 0.5))
    for duty in [0.3, 0.45, 0.6]:

    for density in [0.45, 0.55]:
        guesses.append(gen_markov(density))
    
    def slsqp_guess(g):
