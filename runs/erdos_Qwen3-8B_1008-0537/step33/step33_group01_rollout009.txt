import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty, offset):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * (x - offset))
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty, offset):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * (x - offset) - 1) * duty, 0, 1)

    def gen_triangle(freq, duty, offset):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * (x - offset) - 1) * duty

    guesses = [h]
    for freq in range(3, 9):
        guesses.append(gen_pulse(freq, 0.5, 0.25))
        guesses.append(gen_pulse(freq, 0.5, 0.75))
        guesses.append(gen_saw(freq, 0.2, 0.25))
        guesses.append(gen_saw(freq, 0.4, 0.25))
        guesses.append(gen_saw(freq, 0.6, 0.25))
        guesses.append(gen_saw(freq, 0.8, 0.25))
        guesses.append(gen_triangle(freq, 0.2, 0.25))
        guesses.append(gen_triangle(freq, 0.4, 0.25))
        guesses.append(gen_triangle(freq, 0.6, 0.25))
        guesses.append(gen_triangle(freq, 0.8, 0.25))
    guesses.extend([np.random.uniform(0, 1, n_points) for _ in range(3)])
    guesses.append(np.roll(h, 1))
    guesses.append(np.roll(h, n_points // 3))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

        res = optimize.differential_evolution(obj, bounds, popsize=120, maxiter=60,

        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
