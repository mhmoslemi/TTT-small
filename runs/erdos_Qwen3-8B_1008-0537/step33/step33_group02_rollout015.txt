import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr * dx)
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 0.5 * (1 - np.cos(2 * np.pi * freq * x))
        return np.clip(wave * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 1 - np.abs(2 * freq * x - 1)
        return np.clip(wave * duty, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 1 - 2 * np.abs(freq * x - np.floor(freq * x + 0.5))
        return np.clip(wave * duty, 0, 1)
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 7]:
        guesses.append(gen_pulse(freq, 0.5))
        guesses.append(gen_saw(freq, 0.3))
        guesses.append(gen_triangle(freq, 0.3))
    for duty in [0.15, 0.25, 0.35, 0.5]:
        guesses.append(gen_saw(3, duty))
        guesses.append(gen_triangle(3, duty))
    for _ in range(2):
        guesses.append(np.random.rand(n_points))
        guesses.append(1 - np.random.rand(n_points))
    random_guess = np.random.rand(n_points)
    random_guess = (target_sum / np.sum(random_guess)) * random_guess
    guesses.append(random_guess)
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g
    
    for g in guesses:

        c = obj(h_opt)

    for _ in range(3):

        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=100,

        if c < best_c5:

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
