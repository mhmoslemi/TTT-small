import scipy.optimize as optimize
import time

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty, phase=0):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty, phase=0):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1 + phase) * duty, 0, 1)
    
    def gen_triangle(freq, duty, phase=0):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1 + phase) * duty
    
    guesses = [h]
    for freq in [1, 2, 3, 4, 5, 6]:
        guesses.append(gen_pulses(freq, 0.3))
        guesses.append(gen_saw(freq, 0.3))
        guesses.append(gen_triangle(freq, 0.3))
    for duty in [0.25, 0.3, 0.35, 0.4, 0.45]:
        guesses.append(gen_pulses(3, duty))
        guesses.append(gen_saw(3, duty))
        guesses.append(gen_triangle(3, duty))
    random_guess = np.random.rand(n_points)
    random_guess = (target_sum / np.sum(random_guess)) * random_guess
    guesses.append(random_guess)
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-11})
        return res.x if res.success else g
    
    for g in guesses:
