import scipy.optimize as optimize
import time

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_binary():
        idx = np.random.choice(n_points, size=22, replace=False)
        h_bin = np.zeros(n_points)
        h_bin[idx] = 1.0
        h_bin[np.random.randint(n_points)] += 0.5
        return h_bin
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        return wave
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 1 - np.abs(2 * freq * x - 1) * duty
        return np.clip(wave, 0, 1)

    for freq in [3, 4, 5, 6, 7, 8]:
        guesses.append(gen_saw(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_triangle(3, duty))

    guesses.append(random_guess)
    
    def slsqp_guess(g):
