import scipy.optimize as optimize
import time

    def obj(h_vals):

        return np.max(corr) * dx

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulse(freq, width):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(np.pi * freq * x)
        return np.clip((wave + 1) * width / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    guesses = [h]

        guesses.append(gen_pulse(freq, 0.6))

        guesses.append(gen_triangle(3, duty))
    guesses.append(np.random.uniform(0, 1, n_points))
    
    def slsqp_guess(g):

                               constraints=[{'type': 'eq', 'fun': cons}],

        return res.x if res.success else g
    
    for g in guesses:
