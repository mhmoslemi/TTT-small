import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60  # Smaller number for efficiency
    target_sum = n_points // 2
    dx = 2.0 / n_points

    # Create a binary initial guess with alternating 1s and 0s
    h = np.zeros(n_points)
    h[::2] = 1.0
    h = h.astype(float)

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_binary():
        h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else h
        return h

    def gen_sparse():
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, target_sum, replace=False)
        h[indices] = 1.0
        return h

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [best_h]
    for _ in range(5):
        guesses.append(gen_binary())
    for _ in range(5):
        guesses.append(gen_sparse())
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=30,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-8)
        return res.x

    for _ in range(3):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
