import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freqs, duty):
        x = np.linspace(0, 1, n_points)
        waves = []
        for f in freqs:
            wave = np.sin(2 * np.pi * f * x)
            waves.append(np.clip((wave + 1) * duty / 2, 0, 1))
        return np.array(waves)

    def gen_saw(freqs, duty):
        x = np.linspace(0, 1, n_points)
        waves = []
        for f in freqs:
            wave = 1 - np.abs(2 * f * x - 1) * duty
            waves.append(np.clip(wave, 0, 1))
        return np.array(waves)

    def gen_triangle(freqs, duty):
        x = np.linspace(0, 1, n_points)
        waves = []
        for f in freqs:
            wave = 1 - np.abs(2 * f * x - 1) * duty
            waves.append(np.clip(wave, 0, 1))
        return np.array(waves)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 11):
        guesses.append(gen_pulses([freq], 0.5))
    for freq in range(3, 7):
        for duty in [0.2, 0.3, 0.4, 0.5]:
            guesses.append(gen_saw([freq], duty))
            guesses.append(gen_triangle([freq], duty))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=60,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(4):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
