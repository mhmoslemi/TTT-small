    def obj(h_vals):

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()

        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * (1 - duty), 0, 1)
    
    def gen_rect(freq):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) / 2, 0, 1)
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6]:
        guesses.append(gen_rect(freq))

    for freq in [2, 3, 4]:
