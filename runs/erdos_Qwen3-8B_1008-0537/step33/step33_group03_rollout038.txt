import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(f, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * f * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(f, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * f * x - 1) * duty, 0, 1)

    def gen_triangle(f, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * f * x - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for f in range(3, 8):
        guesses.append(gen_pulses(f, 0.5))
        guesses.append(gen_saw(f, 0.3))
    for d in [0.2, 0.4, 0.6]:
        guesses.append(gen_triangle(3, d))
    for _ in range(4):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-12})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:

    def de_guess():
