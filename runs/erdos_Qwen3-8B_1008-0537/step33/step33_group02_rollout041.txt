def gen_intervals(m):
    # m is the number of intervals
    l = 1.0 / m  # length of each interval
    g = 1.0 / (m + 1)  # gap between intervals
    starts = []
    ends = []
    # Start the first interval at g
    current_start = g
    for _ in range(m):
        starts.append(current_start)
        ends.append(current_start + l)
        # move to next interval: add l (interval length) and g (gap)
        current_start = ends[-1] + g
    # Now, create the step function
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for i in range(n_points):
        x_i = x[i]
        for start, end in zip(starts, ends):
            if start <= x_i <= end:
                h[i] = 1.0
                break
    return h

        for _ in range(m):
            starts.append(current_start)
            ends.append(current_start + l)
            current_start = ends[-1] + g
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            x_i = x[i]
            for s, e in zip(starts, ends):
                if s <= x_i <= e:
                    h[i] = 1.0
                    break
        return h

    guesses = [h]
    for freq in [3, 4, 5, 6, 7, 8]:
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_triangle(3, duty))
    for m in [3, 4, 5, 6, 7, 8]:
        guesses.append(gen_intervals(m))

    random_guess = np.random.rand(n_points)
    random_guess = (target_sum / np.sum(random_guess)) * random_guess
    guesses.append(random_guess)

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=100,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-11})
        return res.x if res.success else r

    for _ in range(4):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
