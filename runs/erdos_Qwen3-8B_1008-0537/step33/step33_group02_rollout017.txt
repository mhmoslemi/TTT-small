import scipy.optimize as optimize
import time

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_wave(freq, duty, phase=0,ftype='saw'):
        x = np.linspace(0, 1, n_points)
        if ftype == 'saw':
            wave = np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif ftype == 'pulse':
            wave = np.clip((np.sin(2 * np.pi * freq * x) + 1) * duty / 2, 0, 1)
        elif ftype == 'tri':
            wave = 1 - np.abs(2 * freq * x - 1) * duty
        elif ftype == 'sin':
            wave = np.clip(np.sin(2 * np.pi * freq * x + phase), 0, 1)
        return np.clip(wave, 0, 1)
    
    guesses = [h]
    for freq in [3, 4, 5, 6, 7, 8]:
        guesses.append(gen_wave(freq, 0.5, ftype='pulse'))
        guesses.append(gen_wave(freq, 0.5, ftype='saw'))
        guesses.append(gen_wave(freq, 0.5, ftype='tri'))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_wave(3, duty, ftype='saw'))
        guesses.append(gen_wave(3, duty, ftype='tri'))
        guesses.append(gen_wave(3, duty, ftype='pulse'))
    random_guess = np.random.rand(n_points)
    random_guess = (target_sum / np.sum(random_guess)) * random_guess
    guesses.append(random_guess)
    
    def slsqp_guess(g):
