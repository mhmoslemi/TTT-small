import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 500, 'ftol': 1e-10})
        return res.x if res.success else r
    
    def gen_binary_initial(n_points):
        total_ones = n_points // 2
        y = np.zeros(n_points)
        positions = np.linspace(0, 2, total_ones + 1)[:-1]
        dx = 2.0 / n_points
        for i in range(n_points):
            y[i] = 1.0 if np.any(np.abs(positions - i * dx) < dx * 0.5) else 0.0
        return y
    
    guesses = [h]
    guesses.append(gen_binary_initial(n_points))
    
    for g in guesses:
        h_opt = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': cons}],

        c = obj(h_opt)
        if c < best_c5:
