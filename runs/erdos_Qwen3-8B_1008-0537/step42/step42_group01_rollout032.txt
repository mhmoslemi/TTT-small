import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Initial DE optimization
    res_de = optimize.differential_evolution(objective, bounds, 
                                            popsize=30, maxiter=100,
                                            mutation=(0.5, 0.8), recombination=0.8, tol=1e-9)
    if res_de.success:
        best_h = res_de.x
        best_c5 = objective(best_h)
    
    # Refinement with SLSQP
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res_slsqp = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,
                                     constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                     options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10})
        if res_slsqp.success:
            best_h = res_slsqp.x
            best_c5 = objective(best_h)
    
    # Final refinement with tighter tolerances
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        res_slsqp = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,
                                     constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                     options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
        if res_slsqp.success:
            best_h = res_slsqp.x
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
