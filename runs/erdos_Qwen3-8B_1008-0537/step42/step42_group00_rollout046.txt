    dx = 2.0 / n_points
    target_sum = n_points / 2

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_random():
        r = np.random.rand(n_points)
        return np.clip((r - np.mean(r)) / np.std(r) * 0.5 + 0.5, 0, 1)

    def gen_symmetric():
        half = n_points // 2
        half_vals = np.random.rand(half)
        half_vals = np.clip((half_vals - np.mean(half_vals)) / np.std(half_vals) * 0.5 + 0.5, 0, 1)
        return np.concatenate([half_vals, half_vals[::-1]])

    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 7, 8]:
        for duty in [0.2, 0.5, 0.7, 0.8]:
            guesses.append(gen_pulse(freq, duty))
    for freq in [2, 3, 4, 5, 6, 7, 8]:
        for duty in [0.2, 0.5, 0.8]:
            guesses.append(gen_saw(freq, duty))
    for _ in range(5):
        guesses.append(gen_random())
    guesses.append(gen_symmetric())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-8})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(4):
        if time.time() - start_time > budget_s - 30:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=80, maxiter=100,
                                             mutation=(0.6, 0.95), recombination=0.8, tol=1e-8)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        res = optimize.minimize(obj, best_h, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200})
        best_h = res.x if res.success else best_h
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
