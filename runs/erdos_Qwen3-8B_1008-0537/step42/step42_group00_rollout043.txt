    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(2):
        if time.time() - start_time > budget_s - 30:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=80, maxiter=100,
                                             mutation=(0.6, 0.95), recombination=0.8, tol=1e-8)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        res = optimize.minimize(obj, best_h, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200})
        best_h = res.x if res.success else best_h
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
