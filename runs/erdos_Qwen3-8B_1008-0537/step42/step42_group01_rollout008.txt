import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60  # Increased points for better resolution
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = np.random.rand(n_points)
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_pattern(freq, phase, amp):
        t = np.linspace(0, 2, n_points)
        wave = amp * np.sin(2 * np.pi * freq * t + phase)
        return np.clip(wave, 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > duty, 0, 1)
    
    guesses = [
        generate_pattern(f, p, 0.5) for f, p in [(1.5, 0.3), (2.5, 0.7), (3.5, 0.4), (4.5, 0.5)]
    ]
    guesses += [
        generate_binary(f, d) for f, d in [(1.2, 0.2), (2.4, 0.4), (3.6, 0.3), (4.8, 0.2)]
    ]
    guesses += [
        np.random.rand(n_points) for _ in range(2)
    ]
    guesses.append(np.linspace(0, 1, n_points))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:

        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=25,
