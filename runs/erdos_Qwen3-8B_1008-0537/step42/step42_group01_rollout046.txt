def run(seed=42, budget_s=100, **kwargs):

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Filter valid shifts k in [0, 2]
        valid_corr = corr[(n_points - 1):]  # Only consider shifts k >= 0
        return np.max(valid_corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate structured initial guesses
    def generate_binary_pattern(n_points, duty):
        period = int(np.ceil(n_points * duty))
        pattern = np.zeros(n_points, dtype=int)
        for i in range(n_points):
            if i % period == 0:
                pattern[i] = 1
        return (pattern.astype(float) / max(1, np.sum(pattern))) * target_sum
    
    guesses = [
        np.sin(np.linspace(0, 2 * np.pi, n_points)) / 2 + 0.5,
        np.clip(np.sin(np.linspace(0, 4 * np.pi, n_points)) / 2 + 0.5, 0, 1),
        np.clip(np.sin(np.linspace(0, 6 * np.pi, n_points)) / 2 + 0.5, 0, 1),
        np.repeat([1, 0], n_points // 2 + (n_points % 2)),
        generate_binary_pattern(n_points, 0.4),

        generate_binary_pattern(n_points, 0.3),
        np.random.rand(n_points)
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
