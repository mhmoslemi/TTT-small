import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_binary_initial(n_points):
        total_ones = n_points // 2
        y = np.zeros(n_points)
        positions = np.linspace(0, 2, total_ones + 1)[:-1]
        dx = 2.0 / n_points
        for i in range(n_points):
            y[i] = 1.0 if np.any(np.abs(positions - i * dx) < dx * 0.5) else 0.0
        return y
    
    def gen_wave(freq, duty):
        def wave_func(x):
            x = np.clip(x, 0, 2)
            return (1 - np.abs(2 * freq * x - 1) * duty)
        return wave_func
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 10]:
        guesses.append(gen_wave(freq, 0.5)(np.linspace(0, 2, n_points)))
    for duty in [0.3, 0.4, 0.5]:
        guesses.append(gen_wave(3, duty)(np.linspace(0, 2, n_points)))
    guesses.append(gen_binary_initial(n_points))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'disp': False})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def refiner(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
