import scipy.optimize as optimize
import time

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_binary_initial(n_points):
        total_ones = n_points // 2
        y = np.zeros(n_points)
        positions = np.linspace(0, 2, total_ones + 1)[:-1]
        dx = 2.0 / n_points
        for i in range(n_points):
            y[i] = 1.0 if np.any(np.abs(positions - i * dx) < dx * 0.5) else 0.0
        return y
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    def gen_sinusoidal(frequencies, duts):
        x = np.linspace(0, 2, n_points)
        wave = np.zeros_like(x)
        for f, d in zip(frequencies, duts):
            wave += np.clip(np.sin(2 * np.pi * f * x) + 1, 0, 1) * d
        return wave / np.max(wave)
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 10]:
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_triangle(3, duty))
    guesses.append(gen_binary_initial(n_points))

    def slsqp_guess(g):

    def refine(r):
