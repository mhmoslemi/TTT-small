import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulse(freq, duty, offset=0):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2 + offset, 0, 1)
    
    def gen_saw(freq, duty, offset=0):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty + offset, 0, 1)
    
    def gen_shifted(h_base, shift):
        shifted = np.roll(h_base, shift)
        return np.clip((shifted - np.min(shifted)) / (np.max(shifted) - np.min(shifted)) * 0.8 + 0.1, 0, 1)
    
    guesses = [h]
    freqs = [2,3,4,5,6,7,8]
    duties = [0.3, 0.5, 0.7]
    for freq in freqs:
        for duty in duties:
            guesses.append(gen_pulse(freq, duty))
    for freq in freqs:
        for duty in duties:
            guesses.append(gen_saw(freq, duty))
    for _ in range(5):
        guesses.append(gen_shifted(h, np.random.randint(n_points//2)))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-8})
        return res.x if res.success else g
    
    for g in guesses:

        c = obj(h_opt)
