import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_lattice(space):
        x = np.linspace(0, 2, n_points)
        wave = np.zeros_like(x)
        for i in range(1, len(space)):
            wave += np.sin(2 * np.pi * space[i] * x) / (i + 1)
        return np.clip((wave + 1) / 2, 0, 1)
    
    def gen_pulses(f, d):
        x = np.linspace(0, 2, n_points)
        wave = np.zeros_like(x)
        for i in range(1, 10):
            wave += np.sin(2 * np.pi * f * x + np.pi * i * d) / (i + 1)
        return np.clip((wave + 1) / 2, 0, 1)
    
    guesses = [h]
    
    for f in [2, 3, 4, 5, 6, 7, 8]:
        guesses.append(gen_pulses(f, 0.5))
        guesses.append(gen_pulses(f, 0.3))
    
    for f in [1.5, 2.5, 3.5, 4.5]:
        guesses.append(gen_lattice([f, 0.5]))
    
    for _ in range(3):
        guesses.append(np.clip(np.random.rand(n_points), 0.3, 0.7))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-6})
        return res.x if res.success else g
    
    for g in guesses:
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=100,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-6)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    return best_h, best_c5, n_points
