import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_rand():
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        return np.clip(h_rand, 0, 1)

    def gen_sym():
        h_sym = np.zeros(n_points)
        h_sym[:n_points//2] = 1
        return h_sym

    def gen_rand_sym():
        h_sym = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        h_sym = h_sym * (target_sum / np.sum(h_sym))
        return np.clip(h_sym, 0, 1)

    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 15]:
        for duty in [0.2, 0.4, 0.6, 0.8]:
            guesses.append(gen_pulse(freq, duty))
    for freq in [3, 4, 5, 6, 7]:
        for duty in [0.15, 0.35, 0.5, 0.65]:
            guesses.append(gen_saw(freq, duty))
            guesses.append(gen_tri(freq, duty))
    guesses.append(gen_rand())
    guesses.append(gen_sym())
    guesses.append(gen_rand_sym())

    def slsqp_guess(g):

                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g

    for g in guesses:

        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

        if time.time() - start_time > budget_s - 15:

        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=50,

        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else r

        best_h = refine(best_h)

    return best_h, best_c5, n_points
