import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_binary_initial(n_points):
        total_ones = n_points // 2
        y = np.zeros(n_points)
        dx_sub = 0.1
        for i in range(n_points):
            y[i] = 1.0 if np.abs(0.5 - dx * i) < dx_sub else 0.0
        return y
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    def gen_random_binary():
        n_ones = int(np.round(n_points / 2))
        indices = np.random.choice(n_points, size=n_ones, replace=False)
        h_rand = np.zeros(n_points)
        h_rand[indices] = 1.0
        return h_rand
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 10]:
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_triangle(3, duty))
    guesses.append(gen_binary_initial(n_points))
    guesses.append(gen_random_binary())
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-11})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)

        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
