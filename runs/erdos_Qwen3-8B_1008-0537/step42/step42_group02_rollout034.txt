import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_binary_initial(n_points):
        total_ones = n_points // 2
        positions = np.linspace(0, 2, total_ones + 1)[:-1]

        y = np.zeros(n_points)
        for i in range(n_points):
            for pos in positions:
                if np.abs(i * dx - pos) < dx * 0.5:
                    y[i] = 1.0
                    break
        return y
    
    def gen_random_binary(n_points):
        total_ones = n_points // 2

        y[indices] = 1.0
        return y
    
    def gen_pulses(freq, duty, n_points):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty, n_points):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
