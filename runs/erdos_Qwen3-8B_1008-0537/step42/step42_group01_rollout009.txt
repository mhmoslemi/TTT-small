import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_pattern(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * (np.sin(2 * np.pi * freq * t) + 1)
        return np.clip(wave, 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > duty, 0, 1)
    
    guesses = [generate_pattern(f, 0.5) for f in [3, 4, 5, 6, 7]]
    guesses += [generate_binary(f, d) for f, d in [(3, 0.25), (4, 0.4), (5, 0.3), (6, 0.15)]]
    guesses += [generate_pattern(f, amp) for f in [5, 6] for amp in [0.7, 0.8]]
    guesses += [np.random.rand(n_points) * 0.5 + 0.25 for _ in range(3)]
    guesses.append(h)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=40,

        if res.success:

                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 60, 'ftol': 1e-12, 'eps': 1e-10})
