import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_symmetric(n_points, func, *args, **kwargs):
        x = np.linspace(0, 2, n_points)
        y = func(x, *args, **kwargs)
        y = (y + np.fliplr([y])) / 2
        y[:n_points//2] = y[n_points//2+1:]
        y[n_points//2] = y[n_points//2]
        return y
    
    def gen_binary(n_points):
        total_ones = n_points // 2
        y = np.zeros(n_points)
        for i in range(n_points):
            if i < total_ones:
                y[i] = 1.0
            else:
                y[i] = 0.0
        y = gen_symmetric(n_points, lambda x, y: y, y)
        return y
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 10]:
        guesses.append(gen_symmetric(n_points, gen_pulses, freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_symmetric(n_points, gen_saw, 3, duty))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_symmetric(n_points, gen_triangle, 3, duty))
    guesses.append(gen_binary(n_points))
    
    def slsqp_guess(g):

                               constraints=[{'type': 'eq', 'fun': cons}],
