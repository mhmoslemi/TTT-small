import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(freq, amp, type='sine'):
        t = np.linspace(0, 1, n_points)
        if type == 'sine':
            wave = amp * (np.sin(2 * np.pi * freq * t) + 1)
        elif type == 'square':
            wave = amp * (np.sign(np.sin(2 * np.pi * freq * t)) + 1)
        elif type == 'triangle':
            wave = amp * (2 * (np.abs(np.sin(2 * np.pi * freq * t)) - 0.5) + 1)
        elif type == 'sawtooth':
            wave = amp * (np.sin(2 * np.pi * freq * t) / np.sin(2 * np.pi * freq * t) * 0.5 + 0.5)
        return np.clip(wave, 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0 * duty, 0, 1)
    
    guesses = []
    for freq in range(1, 15):
        guesses.append(generate_wave(freq, 0.5, 'sine'))
        guesses.append(generate_wave(freq, 0.5, 'square'))
        guesses.append(generate_wave(freq, 0.5, 'triangle'))
        guesses.append(generate_wave(freq, 0.5, 'sawtooth'))
    
    for freq, duty in [(3, 0.2), (4, 0.4), (5, 0.3), (6, 0.2), (7, 0.1), (8, 0.6)]:

    guesses += [np.full(n_points, 0.5) for _ in range(2)]
    guesses.append(h)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,

        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
