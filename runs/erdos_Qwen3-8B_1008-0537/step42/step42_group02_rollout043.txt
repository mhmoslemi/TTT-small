import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_binary_initial(n_points):
        total_ones = n_points // 2
        positions = np.linspace(0, 2, total_ones + 1)[:-1]
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if np.any(np.abs(positions - i * dx) < dx * 0.5) else 0.0
        return h

    def gen_random_binary(n_points):
        total_ones = n_points // 2
        positions = np.random.choice(n_points, size=total_ones, replace=False)
        h = np.zeros(n_points)
        h[positions] = 1.0
        sum_h = np.sum(h)
        required_sum = target_sum
        delta = required_sum - sum_h
        if delta > 0:
            indices = np.where(h == 0)[0]
            h[indices[:int(delta)]] += 0.5
        elif delta < 0:
            indices = np.where(h == 1)[0]
            h[indices[:int(-delta)]] -= 0.5
        return h

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 10]:

    for duty in [0.2, 0.3, 0.4, 0.5]:

    for duty in [0.2, 0.3, 0.4, 0.5]:

    guesses.append(gen_binary_initial(n_points))
    guesses.append(gen_random_binary(n_points))

    def slsqp_guess(g):

        return res.x if res.success else g

        c = obj(h_opt)
