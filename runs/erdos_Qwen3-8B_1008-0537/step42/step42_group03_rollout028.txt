        return np.max(corr) * dx

    def cons(h_vals):

    bounds = [(0., 1.) for _ in range(n_points)]

    best_c5 = obj(best_h)

        indices = np.argsort(np.abs(x - np.linspace(0, 2, num_ones)))
        h = np.zeros(n_points)
        h[indices[:num_ones]] = 1
        return h

    def gen_random():
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        return np.clip(h_rand, 0, 1)

    guesses = [h]

    def slsqp_guess(g):
