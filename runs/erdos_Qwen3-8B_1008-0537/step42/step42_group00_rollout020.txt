import scipy.optimize as optimize
import time

    def obj(h_vals):

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulse(freq, duty, phase=0):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * (freq * x + phase))
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty, phase=0):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1 + phase) * duty, 0, 1)
    
    def gen_random():
        r = np.random.rand(n_points)
        return np.clip((r - np.mean(r)) / np.std(r) * 0.5 + 0.5, 0, 1)
    
    guesses = [h]
    for freq in [2,3,4,5,6,7,8,9,10]:

                guesses.append(gen_saw(freq, duty, phase))
    for _ in range(10):
        guesses.append(gen_random())
    
    def slsqp_guess(g):
