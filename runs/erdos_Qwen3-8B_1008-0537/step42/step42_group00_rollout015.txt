import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_random():
        return np.clip((np.random.rand(n_points) - np.mean(np.random.rand(n_points))) / np.std(np.random.rand(n_points)) * 0.4 + 0.5, 0, 1)
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-8})
        return res.x if res.success else g
    
    guesses = [h]
    for _ in range(10):
        guesses.append(gen_random())
    for freq in range(2, 10):
        for duty in [0.3, 0.5, 0.7]:
            x = np.linspace(0, 1, n_points)
            wave = np.sin(2 * np.pi * freq * x)
            guesses.append(np.clip((wave + 1) * duty / 2, 0, 1))
    for freq in range(2, 10):
        for duty in [0.3, 0.5, 0.8]:
            x = np.linspace(0, 1, n_points)
            guesses.append(np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1))
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=50,
                                             mutation=(0.5, 0.8), recombination=0.8, tol=1e-6)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        res = optimize.minimize(obj, best_h, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150})
        best_h = res.x if res.success else best_h
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
