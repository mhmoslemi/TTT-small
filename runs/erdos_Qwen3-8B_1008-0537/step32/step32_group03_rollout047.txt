            ind = np.argsort(h)[::-1]
            h[ind[:int(diff)]] -= 1.0
        elif diff < 0:
            ind = np.argsort(h)
            h[ind[:int(-diff)]] += 1.0
        return h
    
    def gen_random_binary(n_p):
        h = np.random.choice([0, 1], size=n_p)
        total = np.sum(h)
        diff = total - n_p / 2
        if diff > 0:
            ind = np.argsort(h)[::-1]
            h[ind[:int(diff)]] = 0
        elif diff < 0:
            ind = np.argsort(h)
            h[ind[:int(-diff)]] = 1
        return h
    
    def gen_uniform_block():
        return np.ones(n_points) / 2
    
    guesses = [h]
    for freq in [3, 4, 5, 6, 7, 8]:
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_triangle(3, duty))
    guesses.append(gen_alternating(n_points))
    guesses.append(gen_random_binary(n_points))
    guesses.append(gen_uniform_block())
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=30,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else r
    
    for _ in range(4):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
