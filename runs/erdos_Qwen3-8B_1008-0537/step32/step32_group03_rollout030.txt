import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_binary_guess(n_points):
        num_ones = n_points // 2
        indices = np.argsort(np.random.rand(n_points))
        binary = np.zeros(n_points)
        binary[indices[:num_ones]] = 1
        return binary
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    guesses = [h]
    for _ in range(3):
        guesses.append(gen_binary_guess(n_points))
    for freq in [3, 4, 5]:
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.3, 0.4, 0.5]:
        guesses.append(gen_triangle(3, duty))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-10})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)

        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=30,

        c = obj(h_opt)
