import time

        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        wave *= 0.5
        wave += 0.5
        return np.clip(np.tile(wave, (len(phases), 1)) * duty, 0, 1)
    
    def gen_saw(freq, duty, phases):
        x = np.linspace(0, 1, n_points)
        wave = 1 - np.abs(2 * freq * x - 1) * duty
        return np.clip(np.tile(wave, (len(phases), 1)), 0, 1)
    
    def generate_random_guess():
        h = np.random.rand(n_points)
        total = np.sum(h)
        h *= (target_sum) / total
        return h

            guesses.append(gen_pulses(freq, d, phases))

    def slsqp_guess(g):

    for g in guesses:
