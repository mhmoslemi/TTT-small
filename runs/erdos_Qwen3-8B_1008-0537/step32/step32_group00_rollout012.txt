import scipy.optimize as optimize

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 2 * np.abs(2 * freq * x - 1) - 1
        return np.clip(wave * duty, 0, 1)
    
    def gen_random_guess():
        h_rand = np.random.rand(n_points)
        target_sum = n_points / 2
        current_sum = np.sum(h_rand)
        h_scaled = h_rand * target_sum / current_sum
        h_scaled = np.clip(h_scaled, 0, 1)
        remaining = target_sum - np.sum(h_scaled)
        h_scaled += remaining / n_points
        return h_scaled
    
    guesses = [h]
    for freq in [1, 2, 3, 4, 5, 6, 7, 8]:
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(4, duty))
        guesses.append(gen_tri(4, duty))
    guesses.append(gen_random_guess())
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g
