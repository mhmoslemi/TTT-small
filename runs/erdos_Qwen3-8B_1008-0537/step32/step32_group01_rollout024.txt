import scipy.optimize as optimize
import time

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)

    def gen_pulse(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_tri(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_random_guess():
        h = np.random.rand(n_points)
        total = np.sum(h)
        h *= (target_sum) / total
        return h
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 7, 8]:
        guesses.append(gen_pulse(freq, 0.5, 0))
        guesses.append(gen_pulse(freq, 0.5, np.pi/2))
        guesses.append(gen_tri(freq, 0.5, 0))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_tri(3, duty, 0))
        guesses.append(gen_tri(3, duty, np.pi/2))
    for _ in range(5):
        guesses.append(gen_random_guess())
    
    def slsqp_guess(g):
        try:
            res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],

        except:
            return g
