            wave = np.sin(2 * np.pi * freq * x)

            wave = 2 * (np.abs(2 * freq * x - 1) - 1) * duty

        return np.clip((wave + 1) * amp / 2, 0, 1)

    for _ in range(4):

    def slsqp_guess(g):
