        elif wave_type == 'tri':
            wave = np.abs(2 * freq * x - 1)
            wave = np.clip(duty * (wave - np.floor(wave)) * 2, 0, 1)
        elif wave_type == 'sin':
            wave = np.sin(2 * np.pi * freq * x)

        else:
            wave = np.zeros_like(x)
        return wave
