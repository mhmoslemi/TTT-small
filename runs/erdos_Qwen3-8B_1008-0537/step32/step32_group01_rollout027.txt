import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 2 * np.abs(freq * x - np.floor(freq * x + 0.5)) - 1
        return np.clip(1 - wave * duty, 0, 1)
    
    def generate_random_guess():
        h = np.random.rand(n_points)
        total = np.sum(h)
        h *= (target_sum) / total
        return h
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 7, 8, 9, 10]:
        guesses.append(gen_pulses(freq, 0.5))
    for freq in [2, 3, 4, 5]:

            guesses.append(gen_saw(freq, duty))
    for freq in [2, 3, 4, 5]:

    for _ in range(5):

    def slsqp_guess(g):
