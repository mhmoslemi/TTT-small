import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_varied(freq, duty, phase, shape):
        x = np.linspace(0, 1, n_points)
        if shape == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
        elif shape == 'saw':
            wave = 1 - np.abs(2 * freq * x - 1)
        elif shape == 'tri':
            wave = 2 * np.abs(2 * freq * x - 1) - 1
        elif shape == 'rand':
            wave = np.random.rand(n_points)
        wave = np.clip((wave + 1) * duty / 2, 0, 1)
        return wave
    
    def generate_guesses():
        guesses = []
        for freq in [3, 4, 5, 6, 7, 8]:
            guesses.append(gen_varied(freq, 0.5, 0, 'pulse'))
            guesses.append(gen_varied(freq, 0.2, 0, 'saw'))
        for duty in [0.2, 0.3, 0.4, 0.5]:
            guesses.append(gen_varied(3, duty, 0, 'saw'))
            guesses.append(gen_varied(3, duty, 0, 'tri'))
        return guesses
    
    guesses = generate_guesses()
    guesses.append(h)
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=40,
                                             mutation=(0.6, 0.95), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],

        return res.x if res.success else r

        best_h = refine(best_h)
        if obj(best_h) < best_c5:

    return best_h, best_c5, n_points
