import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 2 * np.abs(2 * freq * x - 1) - 1
        return np.clip(wave * duty, 0, 1)
    
    def gen_rand():
        mask = np.zeros(n_points, dtype=bool)
        np.random.shuffle(mask)
        mask[:int(n_points / 2)] = True
        return np.where(mask, 1.0, 0.0)
    
    def gen_spread():
        X = np.random.uniform(0, 2, n_points)
        idx = np.argsort(X)
        val = np.zeros(n_points)
        val[idx[:int(n_points / 2)]] = 1.0
        return val
    
    guesses = [h]
    for freq in [3, 4, 5, 6, 7, 8]:
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
        guesses.append(gen_tri(3, duty))
    guesses.append(gen_rand())
    guesses.append(gen_spread())
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=40,
                                             mutation=(0.5, 1.0), recombination=0.5, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],

        best_h = refine(best_h)

    return best_h, best_c5, n_points
