import time

    def obj(h_vals):

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()

    start_time = time.time()
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)

        return np.clip(wave * duty, 0, 1)

        grid = np.zeros(n_points)
        for x in positions:
            idx = int(x / dx)
            if 0 <= idx < n_points:
                grid[idx] = 1
        total = np.sum(grid)
        grid = grid * (target_sum / total) if total > 0 else grid

    def slsqp_guess(g):
