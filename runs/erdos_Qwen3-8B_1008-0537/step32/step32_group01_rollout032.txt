import scipy.optimize as optimize
import time

    def obj(h_vals):

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]

        if n_points % 2 != 0:
            return np.zeros(n_points)
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
        return h
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * x / 2)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 2, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def generate_random_guess():

        total = np.sum(h)
        h *= (target_sum) / total

    for freq in [2, 3, 4, 5, 6, 7, 8]:

    guesses.append(gen_binary_guess())
    
    def slsqp_guess(g):
