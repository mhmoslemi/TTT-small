import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 2 * np.abs(2 * freq * x - 1) - 1
        return np.clip(wave * duty, 0, 1)
    
    guesses = [h]
    for freq in range(2, 11):
        for duty in np.linspace(0.2, 0.8, 7):
            guesses.append(gen_pulses(freq, duty))
            guesses.append(gen_saw(freq, duty))
            guesses.append(gen_tri(freq, duty))
    for _ in range(5):
        noise = np.random.uniform(-0.1, 0.1, size=n_points)
        guesses.append(h + noise)
        guesses.append(np.clip(h + noise, 0, 1))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=30, maxiter=30,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-8)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else r
    
    for _ in range(3):

        best_h = refine(best_h)
        if obj(best_h) < best_c5:

    return best_h, best_c5, n_points
