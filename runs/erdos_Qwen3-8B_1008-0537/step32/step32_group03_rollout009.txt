import scipy.optimize as optimize
import time

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    def gen_modified_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 1 - np.abs(2 * freq * x - 1)
        return np.clip(wave * duty + (1 - duty) * (1 - wave), 0, 1)
    
    def gen_chirp(freq):
        x = np.linspace(0, 1, n_points)
        return np.clip(np.sin(2 * np.pi * freq * x) + 1, 0, 1) * 0.5
    
    guesses = [h]
    for freq in [3, 4, 5, 6, 7, 8]:

    def slsqp_guess(g, extra_iter=200):
