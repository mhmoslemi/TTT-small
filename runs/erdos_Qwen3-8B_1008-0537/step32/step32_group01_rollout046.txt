import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 1 - np.abs(2 * freq * x - 1)
        return np.clip(wave * duty, 0, 1)
    
    def gen_sine(freq, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase), 0, 1)
    
    def generate_random_guess():
        h = np.random.rand(n_points)
        total = np.sum(h)
        h *= (target_sum) / total
        return h
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 7, 8, 10]:
        guesses.append(gen_pulse(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
        guesses.append(gen_tri(3, duty))
    for freq in [1, 2, 3, 5, 7, 10]:
        guesses.append(gen_sine(freq, np.random.uniform(-np.pi, np.pi)))
    for _ in range(5):
        guesses.append(generate_random_guess())
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-11, 'eps': 1e-8})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def de_params(popsize=20, mut=(0.5, 1.0), recomb=0.8):
        return {
            'popsize': popsize,
            'mutation': mut,
            'recombination': recomb,
            'tol': 1e-9,
            'maxiter': 50
        }
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, **de_params())
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def refine(r):
        res = optimize.minimize(obj, r, method='COBYLA', bounds=bounds,
