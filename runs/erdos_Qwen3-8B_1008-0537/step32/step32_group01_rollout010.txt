import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_periodic(freq, length, phase=0):
        t = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * t + phase)
        return np.clip((wave + 1) * 0.5, 0, 1)

    def gen_modulated(freq, duty, phase=0):
        t = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * t + phase)
        return np.clip(0.5 + 0.5 * wave * duty, 0, 1)

    def generate_random_guess():
        h = np.random.rand(n_points)
        total = np.sum(h)
        h *= (target_sum) / total
        return h

    guesses = [best_h]
    for freq in [2, 3, 4, 5, 6, 7, 8, 9, 10]:
        guesses.append(gen_periodic(freq, 5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_modulated(3, duty))
    for _ in range(10):
        guesses.append(generate_random_guess())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(3):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=30,
                                             mutation=(0.6, 0.95), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:

    def refine(r):
