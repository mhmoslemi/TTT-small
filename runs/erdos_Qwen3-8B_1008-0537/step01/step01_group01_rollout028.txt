            fun=objective,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
        # Perturb initial guess slightly for next run
        initial_guess += np.random.randn(n_points) * 1e-3
        initial_guess = np.clip(initial_guess, 0.0, 1.0)
        np.random.seed(int(time.time()))

    return best_h, best_c5, n_points
