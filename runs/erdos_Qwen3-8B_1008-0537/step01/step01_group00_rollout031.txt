    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / h_sum

    def objective(x):
        # Compute correlation between x and (1 - x)
        corr = correlate(x, 1.0 - x, mode='full')
        # Find the maximum value in the correlation
        return np.max(corr)

    def constraint(x):
        return np.sum(x) - n_points / 2

    # Optimize using L-BFGS-B with bounds and constraint
    result = minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint},
        options={
