    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx].copy()
        if compute_c5(current_best) < best_c5:
            best_h, best_c5 = current_best.copy(), compute_c5(current_best)
        new_pop = [current_best]
        for _ in range(population_size - 1):
            if random.random() < 0.5:
                h = current_best.copy()
                for i in range(n_points):
                    if random.random() < 0.1:
                        h[i] += np.random.normal(0, 0.1)
                        h[i] = np.clip(h[i], 0.0, 1.0)
                total = np.sum(h)
                scale = (n_points / 2) / total if total != 0 else 1.0
                h = h * scale
                h = np.clip(h, 0.0, 1.0)
                new_pop.append(h)
            else:
                h = random.choice(population).copy()
                for i in range(n_points):
                    if random.random() < 0.1:
                        h[i] += np.random.normal(0, 0.1)
                        h[i] = np.clip(h[i], 0.0, 1.0)
                total = np.sum(h)
                scale = (n_points / 2) / total if total != 0 else 1.0
                h = h * scale
                h = np.clip(h, 0.0, 1.0)
                new_pop.append(h)
        population = new_pop
    return (best_h.tolist(), best_c5, n_points)
