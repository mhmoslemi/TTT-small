import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2.0
    
    def objective(h):
        corr = np.correlate(h, 1-h, mode='full')
        c5 = np.max(corr * dx)
        # Penalize sum deviation
        penalty = 1e6 * abs(np.sum(h) - sum_required)
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Differential evolution with manual parameters
    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=50, popsize=15, tol=1e-6,
        mutation=(0.5, 1), recombination=0.8, disp=False
    )
    
    # Refinement with SLSQP to precisely hit the sum constraint
    best_h = result.x
    sum_h = np.sum(best_h)
    if abs(sum_h - sum_required) > 1e-5:
        def refine_obj(h):
            corr = np.correlate(h, 1-h, mode='full')
            return np.max(corr * dx)
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required})
        res = minimize(
            refine_obj, best_h,
            bounds=bounds, constraints=cons,
            method='SLSQP', options={'maxiter': 100, 'ftol': 1e-8}
        )
        best_h = res.x
    
    final_c5 = objective(best_h)
    return (best_h, final_c5, n_points)
