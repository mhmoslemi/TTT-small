import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values)
    dx = 2.0 / n_points
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    population_size = 100
    population = [initial_h.copy() + np.random.normal(0, 0.05, n_points) for _ in range(population_size)]
    def scale_h(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h
    for _ in range(100):
        if time.time() > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        new_pop = [best_h]
        for _ in range(population_size-1):
            p1, p2 = np.random.choice(population, 2, replace=False)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = p1[i]
                else:
                    child[i] = p2[i]
            if np.random.rand() < 0.2:
                for i in range(n_points):
                    if np.random.rand() < 0.2:
                        child[i] += np.random.normal(0, 0.02)
                        child[i] = np.clip(child[i], 0.0, 1.0)
            child = scale_h(child)
            new_pop.append(child)
        population = new_pop
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return (population[best_idx].tolist(), fitness[best_idx], n_points)
