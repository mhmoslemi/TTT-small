import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        if isinstance(corr, np.ndarray):
            return np.max(corr * dx)
        else:
            # Fallback for older numpy versions
            return np.max(corr * dx)

    bounds = Bounds(0.0, 1.0)
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required})
    
    options = {
        'maxiter': 250, 'ftol': 1e-10, 'disp': False,
        'iprint': -1, 'options': {'eps': 1e-8, 'maxls': 100}
    }
    result = minimize(objective, h_initial, bounds=bounds, constraints=constraints, method='trust-constr', options=options)

    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
