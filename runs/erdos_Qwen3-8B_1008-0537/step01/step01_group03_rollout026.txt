import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

        objective, bounds,

        maxiter=50, popsize=15, tol=1e-3,

        seed=seed, vectorized=True
    )
    
    best_h = result.x
    sum_best = np.sum(best_h)
    h_normalized = best_h / sum_best * sum_required if sum_best > 0 else best_h

    max_overlap = np.max(corr * dx)
