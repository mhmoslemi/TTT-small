import time

        return np.max(corr) * dx

    population_size = 100
    population = [initial_h.copy() + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]

    start_time = time.time()
    while time.time() - start_time < budget_s:

        new_pop = [best_h]
