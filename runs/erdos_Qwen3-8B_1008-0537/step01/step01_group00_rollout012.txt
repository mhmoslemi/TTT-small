        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        disp=False
    )
    # Extract the optimal h_values
    h_opt = result.x
    # Scale to meet the sum constraint
    sum_h_opt = np.sum(h_opt)
    scale = (n_points / 2) / sum_h_opt
    h_opt_scaled = h_opt * scale
    # Compute the correlation
    corr = np.correlate(h_opt_scaled, 1.0 - h_opt_scaled, mode='full')
    max_corr_value = np.max(corr)
    # Compute the actual c5_bound
    c5_bound = max_corr_value * (2.0 / n_points)
    # Return the result
    return (h_opt_scaled, c5_bound, n_points)
