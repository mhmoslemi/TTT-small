        return np.sum(x) - n_points / 2
    
    # Optimize using L-BFGS-B with tighter settings
    result = minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint},
        options={
            'maxiter': 300,
            'ftol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
    )
    
    h_opt = result.x
    corr = np.correlate(h_opt, 1.0 - h_opt, mode='full')
    max_corr_value = np.max(corr)
    c5_bound = max_corr_value * dx
    
    return (h_opt, c5_bound, n_points)
