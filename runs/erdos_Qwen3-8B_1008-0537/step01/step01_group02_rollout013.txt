    population = [h_initial.copy() + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
    for i in range(population_size):
        h = population[i]
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        population[i] = h

    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()

        new_population = [best_h]
        for _ in range(population_size - 1):
            p1, p2, p3 = random.sample(population, 3)
            mutant = p1 + 0.5 * (p2 - p3)
            trial = np.copy(best_h)
            for i in range(n_points):
                if random.random() < 0.5:
                    trial[i] = mutant[i]
            total = np.sum(trial)
            scale = (n_points / 2) / total if total != 0 else 1.0
            trial = trial * scale
            trial = np.clip(trial, 0.0, 1.0)
            new_population.append(trial)
        population = new_population

    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return (population[best_idx].tolist(), fitness[best_idx], n_points)
