import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population_size = 100
    population = [initial_h.copy() + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
    for i in range(population_size):
        h = population[i]
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        population[i] = h

    def tournament_selection(size=5):
        indices = np.random.choice(len(population), size, replace=False)
        return np.argmin([compute_c5(population[i]) for i in indices])

    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        new_pop = [best_h]
        for _ in range(population_size - 1):
            p1_idx = tournament_selection(5)
            p2_idx = tournament_selection(5)
            p1, p2 = population[p1_idx], population[p2_idx]
            child = np.array([p1[i] if random.random() < 0.5 else p2[i] for i in range(n_points)])
            for i in range(n_points):
                if random.random() < 0.15:
                    child[i] += np.random.normal(0, 0.075)
                    child[i] = np.clip(child[i], 0.0, 1.0)
            total = np.sum(child)
            scale = (n_points / 2) / total if total != 0 else 1.0
            child = child * scale
            child = np.clip(child, 0.0, 1.0)
            new_pop.append(child)
        population = new_pop

    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return (population[best_idx].tolist(), fitness[best_idx], n_points)
