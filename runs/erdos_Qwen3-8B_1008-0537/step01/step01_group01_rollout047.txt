    def generate_population(size):
        population = []
        for _ in range(size):
            h = np.random.uniform(0, 1, n_points)
            scale = target_sum / np.sum(h)
            h = h * scale
            population.append(h)
        return np.array(population)

    def select_parents(population, fitnesses):
        selected = []
        for _ in range(POP_SIZE // 2):
            tournament = np.random.choice(range(POP_SIZE), size=TOURNAMENT_SIZE)
            best_idx = tournament[np.argmin(fitnesses[tournament])]
            selected.append(population[best_idx])
        return np.array(selected)
    
    # Crossover
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        child[indices] = parent2[indices]
        return child
    
    # Mutation
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < MUTATION_RATE:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    # Initialize population
    population = generate_population(POP_SIZE)
    best_h = initial_guess.copy()
    best_c5 = evaluate_fitness(best_h)
    
    start_time = time.time()
    
    for gen in range(GENERATIONS):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness for all individuals
        fitnesses = np.array([evaluate_fitness(ind) for ind in population])
        
        # Select parents
        parents = select_parents(population, fitnesses)
        
        # Generate offspring
        offspring = []
        for i in range(POP_SIZE // 2):
            parent1 = parents[i]
            parent2 = parents[POP_SIZE // 2 + i]
            child = crossover(parent1, parent2)
            child = mutate(child)

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_population(size):
        pop = []
        for _ in range(size):
            h = np.random.rand(n_points)
            scale = target_sum / np.sum(h)
            h = h * scale
            pop.append(h)
        return np.array(pop)

    def evaluate_fitness(h):
        return objective(h)

    def select_parents(pop, fitnesses, n_parents):
        parents = []
        for _ in range(n_parents):
            idx = np.random.choice(len(pop), p=fitnesses / np.sum(fitnesses))
            parents.append(pop[idx])
        return parents

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        mask = np.random.rand(n_points) < 0.5
        child[mask] = parent2[mask]
        return child

    def mutate(individual, mutation_rate=0.1):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.05, 0.05)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
