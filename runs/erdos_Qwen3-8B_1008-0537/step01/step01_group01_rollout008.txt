import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        if np.sum(h) < target_sum - 1e-6 or np.sum(h) > target_sum + 1e-6:
            penalty = 1e6 * (np.sum(h) - target_sum) ** 2
        else:
            penalty = 0
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Differential evolution
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    if result.success:
        cand_h = result.x
        cand_c5 = objective(cand_h)
        if cand_c5 < best_c5:
            best_h = cand_h
            best_c5 = cand_c5
    
    # SLSQP refinement
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    if result_slsqp.success:
        final_h = result_slsqp.x
        final_c5 = objective(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return best_h, best_c5, n_points
