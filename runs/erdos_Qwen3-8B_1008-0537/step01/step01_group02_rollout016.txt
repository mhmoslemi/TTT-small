import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values)
    dx = 2.0 / n_points
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    initial_population = [initial_h.copy() + np.random.normal(0, 0.1, n_points) for _ in range(40)]
    for h in initial_population:
        total = np.sum(h)
        if total != 0:
            h *= (n_points / 2) / total
        h = np.clip(h, 0.0, 1.0)
    population = initial_population
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        new_pop = [best_h]
        for _ in range(len(population) -1):
            p1, p2 = random.choices(population, k=2)
            child = p1.copy()
            for i in range(n_points):
                if np.random.rand() < 0.4:
                    child[i] = p2[i]
                    if np.random.rand() < 0.2:
                        child[i] += np.random.normal(0, 0.1)
                        child[i] = np.clip(child[i], 0.0, 1.0)
            total = np.sum(child)
            if total != 0:
                child *= (n_points / 2) / total
            child = np.clip(child, 0.0, 1.0)
            new_pop.append(child)
        population = new_pop
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return (population[best_idx].tolist(), fitness[best_idx], n_points)
