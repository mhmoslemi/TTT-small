import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for h in population:
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h[:] = h * scale
        h[:] = np.clip(h, 0.0, 1.0)
    
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        
        # Local search around best_h
        for _ in range(5):
            perturbed = best_h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    perturbed[i] += np.random.normal(0, 0.02)

            if compute_c5(perturbed) < compute_c5(best_h):

        for _ in range(49):
