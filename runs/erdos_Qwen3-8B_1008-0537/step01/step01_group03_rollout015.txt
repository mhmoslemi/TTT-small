import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr * dx)
        penalty = 1e5 * abs(np.sum(h) - required_sum)
        return max_corr + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective, bounds,
        popsize=20, maxiter=50, tol=1e-8,
        mutation=(0.5, 1.0), recombination=0.9,
        seed=seed
    )
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
