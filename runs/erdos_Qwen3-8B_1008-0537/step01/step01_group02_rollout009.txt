import random
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values)
    dx = 2.0 / n_points
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    population_size = 100  # Increased population size
    population = [initial_h.copy() + np.random.normal(0, 0.05, n_points) for _ in range(population_size)]
    for i in range(population_size):
        h = population[i]
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        population[i] = h
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        new_pop = []
        for _ in range(population_size):
            p1, p2 = random.choices(population, k=2)
            child = np.array([p1[i] * 0.5 + p2[i] * 0.5 for i in range(n_points)])

        population = new_pop
    fitness = [compute_c5(h) for h in population]
