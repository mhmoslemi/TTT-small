import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    init_sum = np.sum(initial_guess)
    adjustment = (target_sum - init_sum) / n_points
    initial_guess += adjustment
    initial_guess = np.clip(initial_guess, 0.0, 1.0)

    def objective(h_values):
        h = np.array(h_values)
        penalty = 1e6 * abs(np.sum(h) - target_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=30,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_guess, objective(initial_guess), n_points

    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
