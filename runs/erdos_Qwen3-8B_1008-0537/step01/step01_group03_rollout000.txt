import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    def objective(h_candidate):
        current_sum = np.sum(h_candidate)
        if current_sum == 0:
            return float('inf')
        scaled_h = h_candidate * (required_sum / current_sum)
        corr = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15, tol=1e-8, mutation=(0.5, 1), recombination=0.7, seed=seed
    )

    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
