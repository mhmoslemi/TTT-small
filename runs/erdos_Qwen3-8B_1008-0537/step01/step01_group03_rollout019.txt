import numpy as np
from scipy.optimize import minimize
from scipy.optimize._optimize import OptimizeResult

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required})

    # Use trust-constr with adaptations for better convergence
    options = {
        'maxiter': 200,
        'ftol': 1e-8,
        'disp': False,
        'gtol': 1e-9,
        'eps': 1e-8,
        'xscale': 'scaled',
        'iprint': -1
    }

    result = minimize(
        objective,
        h_initial,
        method='trust-constr',
        bounds=bounds,
        constraints=constraints,
        options=options
    )

    best_h = result.x
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
