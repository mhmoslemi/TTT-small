import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    try:
        n_points = len(initial_h_values)
    except:
        n_points = 40
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float) if 'initial_h_values' in locals() else np.random.rand(n_points)
    initial_guess = np.clip(initial_guess * target_sum / np.sum(initial_guess), 0, 1) if 'initial_h_values' in locals() else initial_guess

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 0.5:
            break
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-6}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        time.sleep(0.05)
    
    return best_h, best_c5, n_points
