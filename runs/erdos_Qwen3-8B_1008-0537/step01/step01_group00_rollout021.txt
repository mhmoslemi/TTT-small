        constraints={'type': 'eq', 'fun': constraint},
        options={
            'maxiter': 200,
            'ftol': 1e-9,
            'eps': 1e-9,
            'disp': False
        }
    )
    
    h_opt = result.x
    corr = np.correlate(h_opt, 1.0 - h_opt, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h_opt, c5_bound, n_points)
