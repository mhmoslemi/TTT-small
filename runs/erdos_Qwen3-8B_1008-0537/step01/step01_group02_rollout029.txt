import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        scale = (n_points / 2.0) / total
        return np.clip(h * scale, 0.0, 1.0)

    population_size = 40
    population = [initial_h.copy() + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = normalize(population[i])

    DE_F = 0.8
    DE_CR = 0.9
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    while time.time() - start_time < budget_s:
        new_pop = []
        for i in range(population_size):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + DE_F * (b - c)
            trial = np.copy(population[i])

                    trial[j] = mutant[j]
