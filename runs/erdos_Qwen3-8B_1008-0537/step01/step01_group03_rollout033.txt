import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    perturbation = np.random.uniform(-0.05, 0.05, size=h_initial.shape)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    options = {'maxiter': 1000, 'ftol': 1e-8, 'disp': False}
    result = minimize(objective, h_initial, bounds=bounds, constraints=constraints, method='SLSQP', options=options)
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
