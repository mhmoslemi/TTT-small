        population[i] = h
    best_solution = population[0].copy()
    best_fitness = compute_c5(best_solution)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx].copy()
        if fitness[best_idx] < best_fitness:
            best_fitness = fitness[best_idx]
            best_solution = current_best.copy()
        new_pop = [current_best]
        for _ in range(population_size - 1):
            p1, p2 = random.choices(population, k=2)
            child = np.array([p1[i] if random.random() < 0.5 else p2[i] for i in range(n_points)])
            for i in range(n_points):
                if random.random() < 0.1:
                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0.0, 1.0)
            total = np.sum(child)
            scale = (n_points / 2) / total if total != 0 else 1.0
            child = child * scale
            child = np.clip(child, 0.0, 1.0)
            new_pop.append(child)
        population = new_pop
    return (best_solution.tolist(), best_fitness, n_points)
