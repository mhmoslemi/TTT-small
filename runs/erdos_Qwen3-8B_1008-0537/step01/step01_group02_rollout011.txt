import random
import time

def run(seed=42, budget_s=100, **kwargs):

        return np.max(corr) * dx
    population_size = 100
    population = [initial_h.copy() + np.random.normal(0, 0.05, n_points) for _ in range(population_size)]
    for i in range(population_size):
        h = population[i]
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        population[i] = h
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        curr_best_idx = np.argmin(fitness)
        curr_best_h = population[curr_best_idx].copy()
        if fitness[curr_best_idx] < best_c5:
            best_h = curr_best_h

        new_pop = [curr_best_h]
