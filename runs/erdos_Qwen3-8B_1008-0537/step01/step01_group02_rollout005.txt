import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values)
    dx = 2.0 / n_points
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    population_size = 100
    # Start with a diversified population
    population = [initial_h.copy() + np.random.normal(0, 0.05, n_points) for _ in range(population_size)]
    for i in range(population_size):
        h = population[i]
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        population[i] = h
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        new_pop = [best_h]
        # Generate new individuals
        for _ in range(population_size - 1):
            p1, p2 = random.choices(population, k=2)
            # Crossover using linear blend
            child = 0.5 * p1 + 0.5 * p2
            # Local mutation: flip part of the vector

            indices = np.random.choice(n_points, num_mut, replace=False)
            child[indices] += np.random.normal(0, 0.03, num_mut)
            child = np.clip(child, 0.0, 1.0)
            # Normalize
            total = np.sum(child)
            scale = (n_points / 2) / total if total != 0 else 1.0
            child = child * scale
            new_pop.append(child)
        population = new_pop
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
