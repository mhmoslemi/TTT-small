import numpy as np
from scipy.optimize import minimize

# Pre-imported helpers
evaluate_erdos_solution = lambda h: 0.0  # Placeholder; actual function is in environment
initial_h_values = np.random.rand(45) * 1.0  # Placeholder; actual value is in environment

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / h_sum

    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr)

    def constraint(x):
        return np.sum(x) - n_points / 2

    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint},
        options={
            'maxiter': 200,
            'ftol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
    )
    
    h_opt = result.x
    corr = np.correlate(h_opt, 1.0 - h_opt, mode='full')
    max_corr_value = np.max(corr)
    c5_bound = max_corr_value * dx
    
    return (h_opt, c5_bound, n_points)
