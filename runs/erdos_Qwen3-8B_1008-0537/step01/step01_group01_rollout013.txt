import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

        penalty = penalty_coeff * (np.sum(h) - target_sum) ** 2

    start_time = time.time()
    
    # Step 1: Differential Evolution for global search
    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=50
    )
    
    best_h = result_de.x
    best_c5 = objective(best_h)
    
    # Step 2: SLSQP for local refinement
    def objective_slsqp(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    result_slsqp = optimize.minimize(
        fun=objective_slsqp,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    final_h = result_slsqp.x
    final_c5 = objective_slsqp(final_h)
    
    # Ensure the sum is correct (in case of numerical errors)
    final_h = np.clip(final_h, 0, 1)
    if np.isclose(np.sum(final_h), 0):
        final_h = initial_guess
    else:
        final_h = (final_h * (n_points / 2)) / np.sum(final_h) * np.sum(final_h)
    
    # Check elapsed time and return
    if time.time() - start_time > budget_s:

    else:
