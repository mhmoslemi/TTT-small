import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable if available
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        sum_penalty = 1000 * (np.sum(h) - target_sum) ** 2  # Penalize deviation from target sum
        return max_corr * dx + sum_penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Run differential evolution with restarts
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,

                initial_guess = best_h  # Use updated guess for future runs
