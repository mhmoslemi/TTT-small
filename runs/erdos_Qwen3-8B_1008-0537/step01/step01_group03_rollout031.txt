    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required})
    
    def perturb_h(h, seed=None):
        np.random.seed(seed)
        return np.clip(h + np.random.normal(0, 0.05, h.shape), 0, 1)
    
    best_c5 = float('inf')
    best_h = h_initial
    
    for seed_pert in range(3):
        h_perturbed = perturb_h(h_initial, seed_pert)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if abs(sum_perturbed - sum_required) > 1e-5:
            scale = sum_required / sum_perturbed
            h_perturbed *= scale
        
        options = {'maxiter': 300, 'ftol': 1e-8, 'disp': False}
        result = minimize(objective, h_perturbed, bounds=bounds, constraints=constraints, method='L-BFGS-B', options=options)
        result_h = result.x
        result_c5 = objective(result_h)
        
        if result_c5 < best_c5:
            best_c5 = result_c5
            best_h = result_h
    
    return (best_h, best_c5, n_points)
