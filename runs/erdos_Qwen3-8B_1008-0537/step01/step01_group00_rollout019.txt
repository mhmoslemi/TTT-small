        h = h * (n_points / 2) / h_sum

    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr)

    def constraint(x):
        return np.sum(x) - n_points / 2

    # Optimize using COBYLA with bounds and constraint
    result = minimize(
        fun=objective,
        x0=h,
        method='COBYLA',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint},
        options={
            'maxiter': 200,
            'ftol': 1e-6,
            'eps': 1e-6,
            'disp': False
        }
    )
    
    h_opt = result.x
    corr = np.correlate(h_opt, 1.0 - h_opt, mode='full')
    max_corr_value = np.max(corr)
    c5_bound = max_corr_value * dx
    
    return (h_opt, c5_bound, n_points)
