    for _ in range(num_perturbations - 1):
        p = np.random.uniform(-0.05, 0.05, n_points)
        h_perturbed = np.clip(h_initial + p, 0.0, 1.0)
        result = optimize(h_perturbed)
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_result = result.x
    
    return (best_result, best_c5, n_points)
