        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )
    if result_SLSQP.success:
        current_c5 = objective_SLSQP(result_SLSQP.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_SLSQP.x

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(1):
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            perturb = np.clip(perturb, -0.05, 0.05)
            new_guess = initial_guess + perturb
            new_guess = np.clip(new_guess, 0.0, 1.0)
            new_guess = new_guess * (target_sum / np.sum(new_guess)) if np.sum(new_guess) != 0 else new_guess
            result_perturb = optimize.minimize(
                fun=objective_SLSQP,
                x0=new_guess,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result_perturb.success:
                current_c5 = objective_SLSQP(result_perturb.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_perturb.x

    return best_h, best_c5, n_points
