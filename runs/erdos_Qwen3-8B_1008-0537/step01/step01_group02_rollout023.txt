    for i in range(population_size):
        h = population[i]
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        population[i] = h
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        new_pop = [best_h]
        for _ in range(population_size - 1):
            p1, p2 = random.choices(population, k=2)
            child = np.array([p1[i] if random.random() < 0.5 else p2[i] for i in range(n_points)])
            corr = np.correlate(child, 1 - child, mode='full')
            worst_k = np.argmax(corr)
            if worst_k < len(corr):
                shifted_h = np.roll(child, worst_k)
                overlap = child * (1 - shifted_h)
                idxs = np.argsort(overlap)[::-1][:int(n_points * 0.1)]
                for i in idxs:
                    if child[i] > 0.0:
                        child[i] *= 0.95
                        child[i] = np.clip(child[i], 0.0, 1.0)
            total = np.sum(child)
            scale = (n_points / 2) / total if total != 0 else 1.0
            child = child * scale
            child = np.clip(child, 0.0, 1.0)
            new_pop.append(child)
        population = new_pop
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return (population[best_idx].tolist(), fitness[best_idx], n_points)
