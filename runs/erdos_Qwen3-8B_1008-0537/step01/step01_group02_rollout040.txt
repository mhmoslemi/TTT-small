            total = np.sum(child)
            scale = (n_points / 2) / total if total != 0 else 1.0
            child = child * scale
            child = np.clip(child, 0., 1.)
            new_pop.append(child)
        population = new_pop

    fitness = np.array([compute_c5(h) for h in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()

    for _ in range(10):
        h_local = best_h.copy()
        mutate = np.random.rand(n_points) < 0.1
        for i in range(n_points):
            if mutate[i]:
                h_local[i] += np.random.normal(0, 0.02)
                h_local[i] = np.clip(h_local[i], 0., 1.)
        total = np.sum(h_local)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h_local = h_local * scale
        h_local = np.clip(h_local, 0., 1.)
        if compute_c5(h_local) < compute_c5(best_h):
            best_h = h_local

    return (best_h.tolist(), compute_c5(best_h), n_points)
