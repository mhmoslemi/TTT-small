import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Optimized genetic algorithm with blending crossover and adaptive mutation
    population_size = 100
    population = [initial_h.copy() + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
    
    def clip_and_scale(h):
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        return np.clip(h, 0.0, 1.0)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population and select best
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        new_pop = [best_h]  # Keep best solution

        # Generate new solutions via crossover and mutation
        for _ in range(population_size - 1):
            p1, p2 = np.random.choice(population, size=2, replace=False)
            # Arithmetic crossover with learning rate
            alpha = np.random.uniform(0.1, 0.9)
            child = alpha * p1 + (1 - alpha) * p2
            # Adaptive mutation
            if np.random.rand() < 0.15:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            child = clip_and_scale(child)
            new_pop.append(child)
        
        population = new_pop

    # Final evaluation
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return (population[best_idx].tolist(), fitness[best_idx], n_points)
