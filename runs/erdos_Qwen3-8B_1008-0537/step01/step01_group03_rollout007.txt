import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]

    # Use COBYLA for derivative-free optimization, which handles bounds and constraints
    options = {
        'maxiter': 300,  # Increased iteration count for better convergence
        'ftol': 1e-5,    # Slightly relaxed tolerance for faster results
        'disp': False
    }
    result = minimize(objective, h_initial, bounds=bounds, constraints=constraints, method='COBYLA', options=options)

    best_h = result.x
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
