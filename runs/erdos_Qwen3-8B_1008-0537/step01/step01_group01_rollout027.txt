        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_vars)]
    best_h = initial_h.copy()
    best_c5 = objective(x0)
    
    start_time = time.time()
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Apply perturbation to avoid local minima
        perturb = np.random.RandomState(seed).uniform(-0.05, 0.05, size=n_vars)
        x0_perturb = np.clip(x0 + perturb, 0.0, 1.0)
        
        result = optimize.minimize(
            fun=objective,
            x0=x0_perturb,
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_h = np.zeros(n_points)
            current_h[:n_vars] = result.x
            current_h[-1] = target_sum - np.sum(result.x)
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                x0 = result.x
    
    return best_h, best_c5, n_points
