import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    best_h = initial_guess.copy()
    best_c5 = float('inf')
    
    def perturb_guess(h, scale=0.03):
        noise = np.random.normal(0, scale, h.shape)
        p = h + noise
        p = np.clip(p, 0, 1)
        sum_p = np.sum(p)
        if sum_p == 0:
            return h
        factor = target_sum / sum_p
        return p * factor
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Use multiple initial guesses
    initial_guesses = [initial_guess.copy()]
    for _ in range(4):
        initial_guesses.append(perturb_guess(initial_guesses[-1]))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = optimize.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-8}
        )
        if res.success:
            curr_c5 = objective(res.x)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = res.x
        time.sleep(0.05)
    
    return best_h, best_c5, n_points
