import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values)
    dx = 2.0 / n_points
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    population_size = 100
    population = [initial_h.copy() + np.random.normal(0, 0.05, n_points) for _ in range(population_size)]
    for i in range(population_size):
        h = population[i]
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        population[i] = h
    start_time = time.time()
    while time.time() - start_time < budget_s:
        def tournament_selection():
            idxs = np.random.choice(range(population_size), size=5, replace=False)
            best_idx = idxs[np.argmin([compute_c5(population[i]) for i in idxs])]
            return population[best_idx]
        best = tournament_selection()
        new_pop = [best]
        for _ in range(population_size - 1):
            p1, p2 = tournament_selection(), tournament_selection()
