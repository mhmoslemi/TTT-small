import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2.0
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
    
    # Multiple starting points to escape local optima
    def perturb(h, seed):
        np.random.seed(seed)
        return h + np.random.uniform(-0.05, 0.05, size=len(h))
    
    def generate_initial():
        # Create a binary pattern alternating 0.5
        binary_pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                binary_pattern[i] = 0.5
        return binary_pattern
    
    # Precompute initial guesses with randomness
    initial_guesses = [h_initial]
    for i in range(10):
        initial_guesses.append(perturb(generate_initial(), seed + i))
    
    best_h = np.copy(h_initial)
    best_c5 = objective(h_initial)
    
    for guess in initial_guesses:
        # Use L-BFGS-B with more iterations
        options = {'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        result = minimize(objective, guess, bounds=bounds, constraints=constraints, method='L-BFGS-B', options=options)
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
