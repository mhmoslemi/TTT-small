import random
import time

def run(seed=42, budget_s=100, **kwargs):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population with initial_h and some perturbations
    population_size = 100
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = initial_h.copy() + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        population.append(h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Generate new population
        new_pop = [best_h]
        for _ in range(population_size - 1):
            # Select two parents
            p1, p2 = random.choices(population, k=2)

            child = 0.5 * p1 + 0.5 * p2

                    child[i] = np.clip(child[i], 0.0, 1.0)

            total = np.sum(child)
