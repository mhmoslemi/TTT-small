import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initial population
    population_size = 100
    population = [initial_h.copy() + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
    
    # Evaluate initial population
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        current_fitness = [compute_c5(h) for h in population]
        current_best = np.argmin(current_fitness)
        new_population = []
        
        # Elitism: carry over the best individual
        new_population.append(population[current_best].copy())
        
        # Generate new individuals
        for _ in range(population_size - 1):
            # Select parents
            p1, p2 = np.random.choice(population, size=2, replace=False)
            
            # Crossover
            child = np.array([p1[i] if np.random.rand() < 0.5 else p2[i] for i in range(n_points)])
            
            # Mutation
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0.0, 1.0)
            
            # Normalize to maintain constraint
            total = np.sum(child)
            scale = (n_points / 2) / total if total != 0 else 1.0
            child = child * scale
            child = np.clip(child, 0.0, 1.0)
            
            new_population.append(child)
        
        population = new_population
        current_fitness = [compute_c5(h) for h in population]
        current_best = np.argmin(current_fitness)
        
        if current_fitness[current_best] < best_c5:
            best_c5 = current_fitness[current_best]
            best_h = population[current_best].copy()
    
    return (best_h.tolist(), best_c5, n_points)
