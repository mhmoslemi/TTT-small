        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            perturbed[i] += np.random.normal(0, 0.05)
            perturbed[i] = np.clip(perturbed[i], 0.0, 1.0)
        total = np.sum(perturbed)
        if total == 0:
            scale = 1.0
        else:
            scale = (n_points / 2) / total
        perturbed = perturbed * scale
        perturbed = np.clip(perturbed, 0.0, 1.0)
        new_c5 = compute_c5(perturbed)
        if new_c5 < best_c5:
            h = perturbed
            best_c5 = new_c5
        else:
            delta = best_c5 - new_c5
            prob = np.exp(-delta / T)
            if np.random.random() < prob:
                h = perturbed
                best_c5 = new_c5
        T *= cooling_rate
        step_count += 1
    return (h.tolist(), best_c5, n_points)
