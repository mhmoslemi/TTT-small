import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    dx = 2.0 / n_points
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    population_size = 100
    population = [h_initial.copy() + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
    for i in range(population_size):
        h = population[i]
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        population[i] = h
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h_curr = population[best_idx].copy()
        if fitness[best_idx] < best_c5:
            best_h, best_c5 = best_h_curr, fitness[best_idx]
        new_pop = [best_h_curr]
        while len(new_pop) < population_size:
            p1, p2 = np.random.choice(population, size=2, replace=False)
            child = np.array([p1[i] if np.random.rand() < 0.5 else p2[i] for i in range(n_points)])
            if np.random.rand() < 0.1:
                for i in range(n_points):
                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0.0, 1.0)
            total = np.sum(child)
            scale = (n_points / 2) / total if total != 0 else 1.0
            child = child * scale
            child = np.clip(child, 0.0, 1.0)
            new_pop.append(child)
        population = new_pop
    return (best_h.tolist(), best_c5, n_points)
