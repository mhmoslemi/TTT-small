import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable if available
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    # Generate initial guesses
    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guess_alt = np.zeros(n_points)
    for i in range(n_points):
        initial_guess_alt[i] = 1 if i % 2 == 0 else 0
    # Objective function
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    # Constraint
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    # Bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    # List of optimization methods to try
    methods = ['SLSQP', 'L-BFGS-B']
    for method in methods:
        if time.time() - start_time > budget_s:
            break
        # Use different initial guess for different methods
        if method == 'SLSQP':
            x0 = initial_guess
        else:
            x0 = initial_guess_alt
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Introduce small perturbation for diversity
        if method == 'L-BFGS-B':
            perturbed = best_h + np.random.normal(0, 0.05, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    return best_h, best_c5, n_points
