import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

        max_corr = np.max(corr * dx)
        penalty = 1e6 * (np.sum(h) - sum_required)**2
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = differential_evolution(objective, bounds, 
                                   strategy='best1bin', 
                                   maxiter=100, popsize=15, 
                                   tol=1e-8, mutation=(0.5, 1), 
                                   recombination=0.8, 
                                   disp=False)
    
    best_h = result.x
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    return (best_h, best_c5, n_points)
