import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_wave(freq, duty, wave_type):
        x = np.linspace(0, 1, n_points)
        if wave_type == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
            return np.clip((wave + 1) * duty / 2, 0, 1)
        elif wave_type == 'saw':
            return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif wave_type == 'triangle':
            return 1 - np.abs(2 * freq * x - 1) * duty
        else:
            return np.random.rand(n_points)
    
    guesses = []
    guesses.append(h)
    for freq in [3, 5, 7, 10]:
        for duty in [0.25, 0.4, 0.5, 0.6]:
            guesses.append(gen_wave(freq, duty, 'pulse'))
            guesses.append(gen_wave(freq, duty, 'saw'))
            guesses.append(gen_wave(freq, duty, 'triangle'))
    for _ in range(3):
        guesses.append(np.random.rand(n_points))
    
    def slsqp_guess(g):
        try:
            res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 300, 'ftol': 1e-11})
            return res.x if res.success else g
        except:
            return g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 40:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=30, maxiter=50,
                                             mutation=(0.6, 0.8), recombination=0.9, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def refine(r):
        try:
