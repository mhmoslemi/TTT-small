import scipy.optimize as optimize
import time

    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_binary():
        h_bin = np.zeros(n_points)
        np.random.shuffle(h_bin)
        h_bin[:n_points//2] = 1
        return h_bin

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for _ in range(2):
        guesses.append(gen_binary())
    for _ in range(3):
        guesses.append(gen_random())
    for _ in range(5):
        guesses.append(gen_binary())

    def slsqp_guess(g):

                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
