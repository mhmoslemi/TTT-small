            pos += spacing
        intervals = [((a, b), 1.0) for a, b in intervals if a < 2.0]
        result = np.zeros(n_points)
        for a, b in intervals:
            idx_a = int(a / dx)
            idx_b = int(b / dx)
            result[idx_a:idx_b] = 1.0
        return result
