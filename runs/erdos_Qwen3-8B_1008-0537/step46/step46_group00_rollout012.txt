import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)

    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulse(f, d):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * f * x - 1) * d, 0, 1)
    
    def gen_saw(f, d):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * f * x - 1) * d, 0, 1)
    
    def gen_random():
        r = np.random.rand(n_points)
        r = (target_sum / np.sum(r)) * r
        return np.clip(r, 0, 1)
    
    guesses = [h]
    for f in [3, 4, 5, 6, 7, 8]:
        guesses.append(gen_pulse(f, 0.5))
    for d in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, d))
    for _ in range(4):
        guesses.append(gen_random())
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-12})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):

            break
        res = optimize.differential_evolution(obj, bounds, popsize=90, maxiter=80,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
