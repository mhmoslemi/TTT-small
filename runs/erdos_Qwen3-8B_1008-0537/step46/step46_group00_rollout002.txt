import scipy.optimize as optimize
import time

    def obj(h_vals):

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(f, d):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * f * x)
        return np.clip((wave + 1) * d / 2, 0, 1)
    
    def gen_saw(f, d):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * f * x - 1) * d, 0, 1)
    
    def gen_tri(f, d):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * f * x - 1) * d
    
    guesses = [h]
    for f in [3, 4, 5, 6, 7, 8]:
        guesses.append(gen_pulses(f, 0.5))
    for d in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, d))
        guesses.append(gen_tri(3, d))
    guesses.append(np.random.rand(n_points))
    guesses[-1] = (target_sum / np.sum(guesses[-1])) * guesses[-1]
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g
    
    for g in guesses:

    def refine(r):
