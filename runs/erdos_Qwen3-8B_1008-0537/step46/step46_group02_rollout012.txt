import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        sine = np.sin(2 * np.pi * freq * x)
        return np.clip((sine + 1) * duty / 2, 0, 1)

    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_rect(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.cos(np.pi * freq * (x - 0.5)) + 1) * duty / 2, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(1, 4):
        guesses.append(gen_pulse(freq, 0.5))
        guesses.append(gen_tri(freq, 0.5))
        guesses.append(gen_saw(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        for freq in [2, 3]:
            guesses.append(gen_pulse(freq, duty))

            guesses.append(gen_saw(freq, duty))
    for _ in range(8):
        guesses.append(gen_random())

    def slsqp_guess(g):
