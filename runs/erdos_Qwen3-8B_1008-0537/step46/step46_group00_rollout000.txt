import time

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_saw(f, d):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * f * x - 1) * d, 0, 1)

        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * f * x - 1) * d
    
    guesses = [h]

    guesses[-1] = (target_sum / np.sum(guesses[-1])) * guesses[-1]
    
    def slsqp_guess(g):

                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g
