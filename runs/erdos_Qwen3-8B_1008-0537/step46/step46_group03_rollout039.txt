import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def slsqp_opt(g):
        try:
            res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 300, 'ftol': 1e-11})
            return res.x if res.success else g
        except:
            return g
    
    def cobyla_refine(g):
        try:
            res = optimize.minimize(obj, g, method='COBYLA', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 200})
            return res.x if res.success else g
        except:
            return g
    
    # Initial refinement with SLSQP
    best_h = slsqp_opt(best_h)
    best_c5 = obj(best_h)
    
    # Differential evolution for global exploration
    if time.time() - start_time < budget_s - 30:
        try:
            res_de = optimize.differential_evolution(obj, bounds, popsize=30, maxiter=50,
                                                   mutation=(0.5, 0.85), recombination=0.8, tol=1e-9)
            best_h = slsqp_opt(res_de.x)
            best_c5 = obj(best_h)
        except:
            pass
    
    # Refine with COBYLA
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = cobyla_refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
