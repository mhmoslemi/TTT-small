import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_flipped(h):
        return np.flip(h)

    def gen_shifted(h):
        return np.roll(h, 1)

    def gen_perturbed(h):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        return np.clip(h + noise, 0, 1)

    guesses = [h, gen_flipped(h), gen_shifted(h), gen_perturbed(h)]
    for g in guesses:
        h_opt = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': cons}],
                                 options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-6}).x
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=40,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = optimize.minimize(obj, h_opt, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': cons}],
                                 options={'maxiter': 100, 'ftol': 1e-10}).x
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(15):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = optimize.minimize(obj, best_h, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': cons}],
                                 options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-6}).x
        c = obj(best_h)
        if c < best_c5:
            best_c5 = c

    return best_h, best_c5, n_points
