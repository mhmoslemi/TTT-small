import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-12})
        return res.x if res.success else g

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=150,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        return res.x

    # Initial refinement from initial guess
    best_h = slsqp_guess(best_h)
    best_c5 = obj(best_h)

    # Differential evolution with adaptive parameters
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:

        h_opt = de_guess()
