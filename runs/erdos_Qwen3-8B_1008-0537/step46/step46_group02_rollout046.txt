import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def make_two_pulse(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if (x <= 0.5 or x >= 1.0) and (x <= 1.5):
                h[i] = 1.0
        return h

    def make_three_pulse(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if (x <= 1/3) or (2/3 <= x <= 1.0) or (4/3 <= x <= 5/3):
                h[i] = 1.0
        return h

    def make_four_pulse(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if (x <= 0.25 or x >= 0.5) and (x <= 0.75 or x >= 1.0) and (x <= 1.25 or x >= 1.5):
                h[i] = 1.0
        return h

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    guesses.append(make_two_pulse(n_points))
    guesses.append(make_three_pulse(n_points))
    guesses.append(make_four_pulse(n_points))
    for _ in range(3):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=50,
                                             mutation=(0.7, 0.9), recombination=0.9, tol=1e-9)
        return res.x

    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else r

            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
