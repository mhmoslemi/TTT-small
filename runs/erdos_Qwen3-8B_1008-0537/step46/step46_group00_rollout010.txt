import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(f, d):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * f * x)
        return np.clip((wave + 1) * d / 2, 0, 1)
    
    def gen_saw(f, d):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * f * x - 1) * d, 0, 1)
    
    def gen_tri(f, d):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * f * x - 1) * d
    
    guesses = [h]
    for f in [2.5, 3, 3.5, 4, 4.5, 5]:
        guesses.append(gen_pulses(f, 0.5))
    for d in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, d))
        guesses.append(gen_tri(3, d))
    guesses.append(np.random.rand(n_points))
    guesses[-1] = (target_sum / np.sum(guesses[-1])) * guesses[-1]
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 250, 'ftol': 1e-10})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

        res = optimize.differential_evolution(obj, bounds, popsize=80, maxiter=80,
