        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]

    best_c5 = obj(best_h)

        return np.clip((x - 0.5) * k, 0, 1)

    def gen_staircase(m, d):
        x = np.linspace(0, 1, n_points)
        steps = np.sum(np.floor(x * m) == np.floor(x * (m - 1)))
        return np.clip(1 - steps * d, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
