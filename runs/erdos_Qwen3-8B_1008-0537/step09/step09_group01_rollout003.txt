import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_structured_guess(n):
        m = n // 2
        remaining = n - 2 * m
        if remaining:
            m += 1
        dx = 2.0 / n
        ideal_positions = np.linspace(0, 2, m + 1)[:-1]
        grid_indices = np.round(ideal_positions / dx).astype(int)
        grid_indices = np.unique(np.clip(grid_indices, 0, n-1))
        guess = np.zeros(n)
        guess[grid_indices] = 1.0
        current_sum = np.sum(guess)
        if current_sum < target_sum:
            guess[-1] += (target_sum - current_sum)
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        structured_guess = generate_structured_guess(n_points)
        result_cobyla = optimize.minimize(
            fun=objective,
            x0=structured_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'disp': False}
        )
        if result_cobyla.success:
            current_c5 = objective(result_cobyla.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_cobyla.x
    except:
        pass
    
    result_cobyla_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'disp': False}
    )
    if result_cobyla_initial.success:
        current_c5 = objective(result_cobyla_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_cobyla_initial.x
    
    random_guess = np.zeros(n_points)
    num_ones = n_points // 2 + (n_points % 2)
    random_guess[:num_ones] = 1.0
    current_sum = np.sum(random_guess)
    if current_sum < target_sum:
        random_guess[-1] += (target_sum - current_sum)
    result_cobyla_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='COBYLA',

        options={'maxiter': 300, 'disp': False}
    )
    if result_cobyla_random.success:

        if current_c5 < best_c5:
            best_c5 = current_c5

    while time.time() - start_time < budget_s - 30:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0, 1)
