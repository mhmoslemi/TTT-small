import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_structured_guess(n):
        m = n // 2
        remaining = n - 2 * m
        if remaining:
            m += 1
        dx = 2.0 / n
        ideal_positions = [i * (2 / (m + 1)) for i in range(m)]
        grid_indices = []
        for pos in ideal_positions:
            index = round(pos / dx)
            while index in grid_indices or index < 0 or index >= n:
                index += 1
            grid_indices.append(index)
        grid_indices = [min(max(i, 0), n-1) for i in grid_indices]
        grid_indices = np.unique(grid_indices)
        guess = np.zeros(n)
        guess[grid_indices] = 1.0
        current_sum = np.sum(guess)
        if current_sum < target_sum:
            for i in range(n):
                if guess[i] == 0:
                    guess[i] += (target_sum - current_sum)
                    break
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        structured_guess = generate_structured_guess(n_points)
        result_slsqp_structured = optimize.minimize(
            fun=objective,
            x0=structured_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if result_slsqp_structured.success:
            current_c5 = objective(result_slsqp_structured.x)
