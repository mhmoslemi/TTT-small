import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * 2 / n_points
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_uniform_guess(n):
        target_sum = n / 2
        num_int = int(target_sum)
        rem = target_sum - num_int
        pos = np.linspace(0, n-1, num_int, endpoint=False).astype(int)
        guess = np.zeros(n)
        guess[pos] = 1.0
        if rem > 0:
            idx = np.random.choice(n, 1, replace=False)
            guess[idx] += rem
        return guess
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=20,
            maxiter=30,
            mutation=(0.8, 1.2),

            tol=1e-8,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,

                options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-9}
            )
            if result_slsqp.success:
                if objective(result_slsqp.x) < best_c5:
                    best_c5 = objective(result_slsqp.x)

    result_slsqp = optimize.minimize(

    )
