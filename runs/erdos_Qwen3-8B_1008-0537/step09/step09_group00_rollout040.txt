    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_two_clusters(n, t_sum):
        m = int(t_sum)
        first_half = m // 2
        second_half = m - first_half
        guess = np.zeros(n)
        for i in range(first_half):
            x = (i + 0.5) / (first_half + 1)
            idx = int(round(x * (n / 2)))
            if idx < n:
                guess[idx] = 1.0
        for i in range(second_half):
            x = 1.0 + (i + 0.5) / (second_half + 1)
            idx = int(round(x * (n / 2)))
            if idx < n:
                guess[idx] = 1.0
        s = np.sum(guess)
        if s < t_sum:
            guess[0] += t_sum - s
        else:
            guess[-1] -= s - t_sum
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_from_guess(gue):
        res = optimize.minimize(
            fun=objective,
            x0=gue,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success:
            c5 = objective(res.x)
            return res.x, c5
        return best_h, best_c5
    
    guess1 = generate_two_clusters(n_points, target_sum)
    guess2 = generate_two_clusters(n_points, target_sum)
    
    if np.sum(guess1) < target_sum:
        guess1[0] += target_sum - np.sum(guess1)
    elif np.sum(guess1) > target_sum:
        guess1[-1] -= np.sum(guess1) - target_sum
    
    if np.sum(guess2) < target_sum:
        guess2[0] += target_sum - np.sum(guess2)
    elif np.sum(guess2) > target_sum:
        guess2[-1] -= np.sum(guess2) - target_sum
    
    for g in [guess1, guess2]:
        h, c5 = optimize_from_guess(g)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    while time.time() - start_time < budget_s - 20:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0.0, 1.0)
        while np.sum(new_guess) < target_sum - 1e-6:
            new_guess += 1e-3
        while np.sum(new_guess) > target_sum + 1e-6:
            new_guess -= 1e-3
        new_guess = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        ).x
        c5 = objective(new_guess)
        if c5 < best_c5:
            best_h, best_c5 = new_guess, c5
    
    return best_h, best_c5, n_points
