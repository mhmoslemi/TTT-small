import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_spaced_guess(n):
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        positions = np.linspace(0, 2, num_ones, endpoint=False)
        guess = np.zeros(n)
        for pos in positions:
            idx = int(np.round(pos * n / 2))
            guess[idx] = 1.0
        if remaining > 0:
            pos = np.random.uniform(0, 2)
            idx = int(np.round(pos * n / 2))
            guess[idx] += remaining
        return guess
    
    def generate_random_guess(n):
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        positions = np.random.choice(n, size=num_ones, replace=False)

        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    for _ in range(3):
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=20,
            maxiter=50,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,

                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:

                if current_c5 < best_c5:
