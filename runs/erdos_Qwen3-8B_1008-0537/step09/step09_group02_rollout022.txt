        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_even_guess():
        n_ones = n_points // 2
        positions = np.linspace(0, n_points, n_ones + 1, endpoint=False).astype(int)
        guess = np.zeros(n_points)
        guess[positions] = 1.0
        return guess
    
    def generate_random_guess():
        target_sum = n_points / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n_points, size=num_integers, replace=False)
        guess = np.zeros(n_points)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n_points, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    for init in [initial_guess, generate_even_guess(), generate_random_guess()]:
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,

                maxiter=100,
                mutation=(0.5, 1.0),

                tol=1e-10,

                x0=init
            )
            if result_de.success:
                result_slsqp = optimize.minimize(

                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
