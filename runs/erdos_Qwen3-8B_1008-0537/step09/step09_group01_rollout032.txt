    def generate_structured_guess(n):
        num_ones = int(np.round(n / 2))
        spacing = 2.0 / num_ones
        positions = np.linspace(0, 2 - spacing, num_ones)
        grid_indices = np.round(positions / dx).astype(int)
        grid_indices = np.clip(grid_indices, 0, n-1)
        grid_indices = np.unique(grid_indices)
        guess = np.zeros(n)
        guess[grid_indices] = 1.0
        current_sum = np.sum(guess)
        if current_sum < target_sum:
            for i in range(n):
                if guess[i] == 0:
                    guess[i] += (target_sum - current_sum)
                    break
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
