import time

    def generate_structured_guess(n):

        remaining = n - 2 * m
        if remaining:
            m += 1
        dx = 2.0 / n
        ideal_positions = [i * (2 / (m + 1)) for i in range(m)]
        grid_indices = []
        for pos in ideal_positions:
            index = int(pos / dx + 0.5)
            while index in grid_indices or index < 0 or index >= n:
                index += 1
            grid_indices.append(index)
        grid_indices = np.unique(grid_indices)
        guess = np.zeros(n)
        guess[grid_indices] = 1.0
        current_sum = np.sum(guess)
        if current_sum < target_sum:
            for i in range(n):
                if guess[i] == 0:
                    guess[i] += (target_sum - current_sum)
                    break
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
