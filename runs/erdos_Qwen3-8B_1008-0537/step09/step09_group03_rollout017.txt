import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e2 * constraint_violation  # Reduced penalty
        return c5 + penalty
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=30,
            maxiter=200,
            mutation=(0.7, 1.2),
            recombination=0.8,
            tol=1e-8,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    random_guess = np.random.rand(n_points)
    random_guess = (random_guess * n_points / 2).astype(float)
    random_guess = np.round(random_guess, 5)
    result_slsqp_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='SLSQP',

        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_random.success:
        current_c5 = objective(result_slsqp_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_random.x
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.005, size=n_points)
