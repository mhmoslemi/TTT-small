        guess[grid_indices] = 1.0
        current_sum = np.sum(guess)
        if current_sum < target_sum:
            for i in range(n):
                if guess[i] == 0:
                    guess[i] = 1.0
                    break
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Optimize structured guess
    structured_guess = generate_structured_guess(n_points)
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=structured_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 400, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_c5 = objective(result_slsqp.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
    
    # Optimize initial guess
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 400, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_c5 = objective(result_slsqp.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
    
    # Perturb and refine
    while time.time() - start_time < budget_s - 20:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0.0, 1.0)
        result = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
