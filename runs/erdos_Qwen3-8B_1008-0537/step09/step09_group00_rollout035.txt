    optimized_guess = generate_optimized_guess(n_points, target_sum)
    result = optimize.minimize(
        fun=objective,
        x0=optimized_guess,
        method='trust-constr',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9, 'verbose': 0}
    )
    if result.success:
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    while time.time() - start_time < budget_s - 30:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='trust-constr',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9, 'verbose': 0}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
