        m = int(target_sum)
        spacing = 2.0 / m if m > 0 else 0
        positions = [int(round(x * (n / 2))) for x in np.linspace(0, 2, m)]
        guess = np.zeros(n)
        for idx in positions:
            if 0 <= idx < n:
                guess[idx] = 1.0
        fractional = target_sum - m
        if fractional > 1e-6:
            guess[0] += fractional
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        optimized_guess = generate_optimized_guess(n_points, target_sum)
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=100,
            mutation=(0.8, 1.2),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    optimized_guess = generate_optimized_guess(n_points, target_sum)
    result_slsqp_optimized = optimize.minimize(
        fun=objective,
        x0=optimized_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_optimized.success:
        current_c5 = objective(result_slsqp_optimized.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_optimized.x
    
    while time.time() - start_time < budget_s - 20:
        noise = np.random.normal(0, 0.01, size=n_points)
        new_guess = best_h + noise
        new_guess = np.clip(new_guess, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        ).x
        current_c5 = objective(new_guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_guess
    
    return best_h, best_c5, n_points
