import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_optimized_guess(n, target_sum):
        m = int(target_sum)
        spacing = 2.0 / (m + 1)
        guess = np.zeros(n)
        for i in range(m):
            x = (i + 0.5) * spacing
            idx = int(round(x * (n / 2)))
            if 0 <= idx < n:
                guess[idx] = 1.0
        if target_sum - m > 0:
            guess[0] += target_sum - m
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_starting_guesses():
        guesses = []
        # Add initial guess
        guesses.append(initial_guess.copy())
        # Add optimized guess
        guess = generate_optimized_guess(n_points, target_sum)
        guesses.append(guess)
        # Add perturbed guesses
        for _ in range(3):
            noise = np.random.normal(0, 0.02, n_points)
            perturbed = initial_guess + noise
            perturbed = (perturbed / np.sum(perturbed)) * target_sum
            guesses.append(perturbed)
        return guesses
    
    starting_guesses = generate_starting_guesses()
    
    for guess in starting_guesses:
        # Run differential evolution
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=20,
                maxiter=30,
                mutation=(0.7, 1.0),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                callback=lambda x: None
            )
            if result_de.success:
                # Run SLSQP on the DE result
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass
    
    # Run additional local search
    while time.time() - start_time < budget_s - 10:
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = best_h + noise
        perturbed = (perturbed / np.sum(perturbed)) * target_sum
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
