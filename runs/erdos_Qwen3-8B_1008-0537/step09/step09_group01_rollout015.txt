    random_guess[:num_ones] = 1.0
    current_sum = np.sum(random_guess)
    if current_sum < target_sum:
        random_guess[-1] += (target_sum - current_sum)
    result_de_random = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    if result_de_random.success:
        current_c5 = objective(result_de_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_de_random.x
    
    # Final refinement with SLSQP
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
