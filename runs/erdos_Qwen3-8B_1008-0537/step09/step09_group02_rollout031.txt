import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_initial_guess():
        num_ones = int(n_points / 2)
        num_zeros = int(n_points / 2)
        remaining = n_points - num_ones - num_zeros
        ones = [1.0] * num_ones
        zeros = [0.0] * num_zeros
        if remaining > 0:
            zeros.append(0.5)
        guess = ones + zeros
        np.random.shuffle(guess)
        return np.array(guess)
    
    initial_guesses = [initial_guess, generate_binary_initial_guess()]
    random_guess = np.random.uniform(0.0, 1.0, n_points)
    random_guess = (random_guess - np.min(random_guess)) / (np.max(random_guess) - np.min(random_guess))
    random_guess = (random_guess * n_points / 2) / np.sum(random_guess) * n_points / 2
    initial_guesses.append(random_guess)
    
    for guess in initial_guesses:
        try:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
    
    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=30,
        maxiter=50,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-10,
        vectorized=True
    )
    if result_de.success:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    while time.time() - start_time < budget_s - 20:
        perturbation = np.random.normal(0, 0.01, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0, 1)
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
