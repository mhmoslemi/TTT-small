import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    
    def generate_optimized_guess(n, target_sum):
        m = int(target_sum)
        positions = []
        spacing = 2.0 / (m + 1)
        for i in range(m):
            x = (i + 0.5) * spacing
            idx = int(round(x * (n / 2)))
            positions.append(idx)
        guess = np.zeros(n)
        for idx in positions:
            if 0 <= idx < n:
                guess[idx] = 1.0
        if target_sum - m > 0:
            guess[0] += target_sum - m
        return guess
    
    def generate_random_guess(n, target_sum):
        m = int(target_sum)
        remaining = target_sum - m
        positions = np.random.choice(n, size=m, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 1e-10:
            idx = np.random.randint(0, n)
            guess[idx] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    initial_guesses = [initial_h, generate_optimized_guess(n_points, target_sum), generate_random_guess(n_points, target_sum)]
    
    for guess in initial_guesses:
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=30,
                maxiter=20,

                recombination=0.9,
                tol=1e-10,
                vectorized=True,
                x0=guess
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass
    
    while time.time() - start_time < budget_s - 20:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0, 1)
        result_slsqp = optimize.minimize(

            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
