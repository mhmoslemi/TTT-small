import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_random_uniform():
        h = np.zeros(n_points)
        h[:int(n_points/2)] = 1.0
        np.random.shuffle(h)
        return h

    def generate_random_spaced():
        h = np.zeros(n_points)
        idxs = np.random.choice(n_points, size=int(n_points/2), replace=False)
        h[idxs] = 1.0
        return h

    guesses = [initial_guess, generate_random_uniform(), generate_random_spaced()]

    for guess in guesses:
        res = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x

    if time.time() - start_time < budget_s - 20:
        for _ in range(5):
            perturb = np.random.normal(0, 0.01, size=n_points)
            new_guess = best_h + perturb
            new_guess = np.clip(new_guess, 0, 1)
            res = optimize.minimize(
                fun=objective,
                x0=new_guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if res.success:
                current_c5 = objective(res.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x

    return best_h, best_c5, n_points
