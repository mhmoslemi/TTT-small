import time

    dx = 2.0 / n_points

        h = np.zeros(n)
        cnt = int(np.round(target_sum))
        h[::2] = 1.0
        if cnt < n // 2:
            h[0] -= 1.0
        elif cnt > n // 2:
            h[-1] += 1.0
        return h

    initial_guesses = [np.array(initial_h_values, dtype=float), generate_uniform_guess(n_points, target_sum)]

    best_h = None
    best_c5 = float('inf')

    start_time = time.time()

    for guess in initial_guesses:
        try:
            result_de = optimize.differential_evolution(

                popsize=30,
                maxiter=100,
                mutation=(0.8, 1.2),

                tol=1e-10,
                vectorized=True,
                seed=seed

            if result_de.success:

                    bounds=[(0.0, 1.0) for _ in range(n_points)],
                    constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
                    options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:

                    if current_c5 < best_c5:
