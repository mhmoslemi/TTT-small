import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def generate_structured_guess(n_points):

            grid_indices.append(index)

        guess = np.zeros(n_points)
        guess[grid_indices] = 1.0
        current_sum = np.sum(guess)
        diff = target_sum - current_sum
        if abs(diff) > 1e-6:
            zeros = np.where(guess == 0)[0]
            if diff > 0:
                if zeros.size > 0:
