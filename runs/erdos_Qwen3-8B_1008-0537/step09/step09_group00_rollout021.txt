import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    optimized_guess = np.zeros(n_points)
    optimized_guess[::2] = 1.0
    optimized_guess[1::2] = 0.5
    optimized_guess = (optimized_guess * target_sum) / np.sum(optimized_guess)
    result_slsqp_optimized = optimize.minimize(
        fun=objective,
        x0=optimized_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-9}
    )
    if result_slsqp_optimized.success:
        current_c5 = objective(result_slsqp_optimized.x)
        if current_c5 < best_c5:

            best_h = result_slsqp_optimized.x
    
    while time.time() - start_time < budget_s - 30:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)

        new_guess = new_guess * (target_sum / np.sum(new_guess)) if np.sum(new_guess) != 0 else new_guess
        result = optimize.minimize(
