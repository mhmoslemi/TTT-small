import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_structured_guess(n):
        m = int(target_sum)
        remaining = target_sum - m
        ideal_positions = np.linspace(0, 2, m + 1)[:-1]
        if remaining > 0:
            ideal_positions = np.concatenate([ideal_positions, [2 - remaining]])
        guess = np.zeros(n)
        for pos in ideal_positions:
            idx = int(round(pos / dx))
            while idx < 0 or idx >= n:
                idx += 1
            guess[idx] += 1.0
        guess /= np.sum(guess) * dx
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        structured_guess = generate_structured_guess(n_points)
        result = optimize.minimize(
            fun=objective,
            x0=structured_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    except:
        pass
    
    def random_initial():
        h = np.zeros(n_points)
        num_ones = int(np.round(target_sum))
        h[:num_ones] = 1.0
        curr_sum = np.sum(h)
        if curr_sum < target_sum:
            h[-1] += (target_sum - curr_sum)
        h /= np.sum(h) * dx
        return h
    
    for _ in range(3):
        h = random_initial()
        result = optimize.minimize(
            fun=objective,

            method='SLSQP',
            bounds=bounds,

            if current_c5 < best_c5:

                best_h = result.x
