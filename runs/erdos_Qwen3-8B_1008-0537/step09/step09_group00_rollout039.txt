            mutation=(0.5, 1.5),
            recombination=0.8,
            tol=1e-8,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    optimized_guess = np.copy(best_h)
    result_slsqp_optimized = optimize.minimize(
        fun=objective,
        x0=optimized_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
    )
    if result_slsqp_optimized.success:
        current_c5 = objective(result_slsqp_optimized.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_optimized.x
    
    while time.time() - start_time < budget_s - 25:
        perturb = np.random.normal(0, 0.02, size=n_points)
        new_guess = best_h + perturb
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess[0] = np.clip(new_guess[0], 0.0, 1.0)
        new_guess[-1] = np.clip(new_guess[-1], 0.0, 1.0)
        new_guess = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        ).x
        current_c5 = objective(new_guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_guess
    
    return best_h, best_c5, n_points
