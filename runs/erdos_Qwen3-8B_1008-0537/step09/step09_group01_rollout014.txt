            new_guess[-1] -= (current_sum - target_sum)
        result_slsqp_perturb = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp_perturb.success:
            current_c5 = objective(result_slsqp_perturb.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp_perturb.x
    
    return best_h, best_c5, n_points
