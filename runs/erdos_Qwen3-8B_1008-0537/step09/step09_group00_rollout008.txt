import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def generate_optimized_guess(n, target_sum):
        m = int(target_sum)
        f = target_sum - m
        spacing = 2.0 / (m + 1)
        positions = [i * spacing for i in range(m + 1)]
        guess = np.zeros(n)
        for x in positions:
            idx = int(round(x * (n / 2)))
            if 0 <= idx < n:
                guess[idx] += 1.0
        if f > 0:
            idx = int(round(positions[0] * (n / 2)))
            if 0 <= idx < n:
                guess[idx] += f
        return guess

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    optimized_guess = generate_optimized_guess(n_points, target_sum)
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=optimized_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_c5 = objective(result_slsqp.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x

    result_slsqp_initial = optimize.minimize(

        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x

    while time.time() - start_time < budget_s - 20:
