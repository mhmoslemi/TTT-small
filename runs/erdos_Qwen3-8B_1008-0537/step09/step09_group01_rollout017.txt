        perturbation = np.random.normal(0, 0.01, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='trust-constr',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'gtol': 1e-8, 'eps': 1e-9, 'maxiter': 100}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
