import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_optimized_guess(n, target_sum):
        m = int(target_sum)
        spacing = 2.0 / (m + 1)
        positions = [(i + 0.5) * spacing for i in range(m)]
        guess = np.zeros(n)
        for x in positions:
            idx = int(round(x * (n / 2)))
            if 0 <= idx < n:
                guess[idx] = 1.0
        if target_sum - m > 0:
            guess[0] += target_sum - m
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        # Multi-start SLSQP with perturbations
        perturbations = [0.01, 0.02, 0.03]
        for pert in perturbations:
            perturbed = initial_guess + np.random.normal(0, pert, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)
            while np.sum(perturbed) < target_sum - 1e-6:
                perturbed += np.random.normal(0, 0.01, size=n_points)
            while np.sum(perturbed) > target_sum + 1e-6:
                perturbed -= np.random.normal(0, 0.01, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
