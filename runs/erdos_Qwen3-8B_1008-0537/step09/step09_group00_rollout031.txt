            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        ).x
        current_c5 = objective(new_guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_guess
    
    return best_h, best_c5, n_points
