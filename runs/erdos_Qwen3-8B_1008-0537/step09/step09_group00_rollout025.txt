import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    initial_guesses = [np.array(initial_h_values, dtype=float)]
    np.random.seed(seed)
    random_guess = np.random.dirichlet([1.0] * n_points) * target_sum
    initial_guesses.append(random_guess)

    m = int(target_sum)
    spacing = 2.0 / (m + 1)
    positions = [(i + 0.5) * spacing for i in range(m)]
    indices = np.round(np.array(positions) * n_points / 2).astype(int)
    spread_guess = np.zeros(n_points)
    for idx in indices:
        if 0 <= idx < n_points:
            spread_guess[idx] = 1.0
    if target_sum - m > 0:
        spread_guess[0] += target_sum - m
    initial_guesses.append(spread_guess)

    best_h = initial_guesses[0]
    best_c5 = objective(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        res = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
        for _ in range(5):
            perturbed = best_h + np.random.normal(0, 0.01, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if total == 0:
                continue
            perturbed = perturbed / total * target_sum
            res = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x

    return best_h, best_c5, n_points
