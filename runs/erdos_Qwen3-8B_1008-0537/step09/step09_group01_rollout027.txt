    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        checkered_guess = generate_checkered_guess(n_points)
        result_cobyla_checkered = optimize.minimize(
            fun=objective,
            x0=checkered_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 400, 'disp': False}
        )
        if result_cobyla_checkered.success:
            current_c5 = objective(result_cobyla_checkered.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_cobyla_checkered.x
    except:
        pass
    
    result_cobyla_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 400, 'disp': False}
    )
    if result_cobyla_initial.success:
        current_c5 = objective(result_cobyla_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_cobyla_initial.x
    
    random_guess = np.zeros(n_points)
    num_ones = n_points // 2 + (n_points % 2)
    random_guess[:num_ones] = 1.0
    current_sum = np.sum(random_guess)
    if current_sum < target_sum:
        random_guess[-1] += (target_sum - current_sum)
    result_cobyla_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 400, 'disp': False}
    )
    if result_cobyla_random.success:
        current_c5 = objective(result_cobyla_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_cobyla_random.x
    
    while time.time() - start_time < budget_s - 20:
        new_guess = optimize.minimize(
            fun=objective,
            x0=best_h + np.random.normal(0, 0.01, size=n_points),
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'disp': False}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
