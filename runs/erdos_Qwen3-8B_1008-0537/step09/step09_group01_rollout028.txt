        if current_sum < target_sum:
            for i in range(n):
                if guess[i] == 0:
                    guess[i] += (target_sum - current_sum)
                    break
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        relevant_corr = corr[-n_points:]
        return np.max(relevant_corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        structured_guess = generate_structured_guess(n_points)
        result_slsqp_structured = optimize.minimize(
            fun=objective,
            x0=structured_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result_slsqp_structured.success:
            current_c5 = objective(result_slsqp_structured.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp_structured.x
    except:
        pass
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    random_guess = np.zeros(n_points)
    num_ones = n_points // 2 + (n_points % 2)
    random_guess[:num_ones] = 1.0
    current_sum = np.sum(random_guess)
    if current_sum < target_sum:
        random_guess[-1] += (target_sum - current_sum)
    result_slsqp_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    if result_slsqp_random.success:
        current_c5 = objective(result_slsqp_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_random.x
    
    while time.time() - start_time < budget_s - 20:
        perturbation = np.random.normal(0, 0.02, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = optimize.optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
