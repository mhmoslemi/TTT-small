import numpy as np

def generate_even_spaced_guess(n_points):
    m = n_points // 2
    dx = 2.0 / n_points
    grid = np.zeros(n_points, dtype=float)
    for i in range(m):
        x = 2.0 * i / m
        j = int(round(x / dx))
        if 0 <= j < n_points:
            grid[j] = 1.0
    total = np.sum(grid)
    while total < m:
        j = np.argmin(grid)
        grid[j] = 1.0
        total += 1
    while total > m:
        j = np.argmax(grid)
        grid[j] = 0.0
