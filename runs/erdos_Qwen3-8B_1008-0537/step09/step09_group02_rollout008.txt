import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=30,
            maxiter=200,
            mutation=(0.6, 1.0),
            recombination=0.8,
            tol=1e-12,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    random_guesses = [generate_random_guess(n_points) for _ in range(5)]
    for guess in random_guesses:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    while time.time() - start_time < budget_s - 30:
        new_guess = optimize.minimize(
            fun=objective,
            x0=best_h + np.random.normal(0, 0.01, size=n_points),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
