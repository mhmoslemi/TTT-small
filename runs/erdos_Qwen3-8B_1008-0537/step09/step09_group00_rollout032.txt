    while time.time() - start_time < budget_s - 20:
        perturb = np.random.normal(0, 0.05, size=n_points)
        new_guess = np.clip(best_h + perturb, 0.0, 1.0)
        sum_new = np.sum(new_guess)
        adjustment = (target_sum - sum_new) / n_points
        new_guess += adjustment
        result = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
