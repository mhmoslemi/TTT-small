import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_with(method, x0):
        res = optimize.minimize(
            fun=objective,
            x0=x0,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success:
            return res.x
        return x0
    
    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=20,
        maxiter=30,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-10,
        vectorized=True
    )
    best_h = optimize_with('COBYLA', result_de.x)
    best_c5 = objective(best_h)
    
    best_h = optimize_with('COBYLA', initial_guess)
    best_c5 = min(best_c5, objective(best_h))
    
    random_guess = np.zeros(n_points)
    random_guess[:int(target_sum)] = 1.0
    best_h = optimize_with('COBYLA', random_guess)
    best_c5 = min(best_c5, objective(best_h))
    
    while time.time() - start_time < budget_s - 20:
        h_perturb = best_h + np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = optimize_with('COBYLA', h_perturb)
        current_c5 = objective(h_perturb)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturb
    
    return best_h, best_c5, n_points
