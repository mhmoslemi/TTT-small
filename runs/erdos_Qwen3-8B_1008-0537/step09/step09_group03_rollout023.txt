            popsize=30,
            maxiter=200,
            mutation=(0.5, 0.9),
            recombination=0.95,
            tol=1e-8,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 400, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    binary_guess = generate_binary_guess(n_points)
    result_slsqp_binary = optimize.minimize(

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_binary.x
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    random_guess = generate_random_guess(n_points)
    result_slsqp_random = optimize.minimize(

    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.005, size=n_points)
