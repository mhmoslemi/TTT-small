    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_structured_guess(n):
        num_ones = n // 2 + (n % 2)
        positions = np.linspace(0, 2, num_ones + 1)[1:-1]
        indices = np.round(positions / dx).astype(int)
        indices = np.unique(np.clip(indices, 0, n-1))
        guess = np.zeros(n)
        guess[indices] = 1.0
        current_sum = np.sum(guess)
        if current_sum < target_sum:
            for i in range(n):
                if guess[i] == 0:
                    guess[i] += (target_sum - current_sum)
                    break
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        structured_guess = generate_structured_guess(n_points)
        result_cobyla = optimize.minimize(
            fun=objective,
            x0=structured_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'disp': False}
        )
        if result_cobyla.success:
            current_c5 = objective(result_cobyla.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_cobyla.x
    except:
        pass
    
    result_cobyla_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'disp': False}
    )
    if result_cobyla_initial.success:
        current_c5 = objective(result_cobyla_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_cobyla_initial.x
    
    random_guess = np.zeros(n_points)
    num_ones = n_points // 2 + (n_points % 2)
    random_guess[:num_ones] = 1.0
    current_sum = np.sum(random_guess)
    if current_sum < target_sum:
        random_guess[-1] += (target_sum - current_sum)
    result_cobyla_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'disp': False}
    )
    if result_cobyla_random.success:
        current_c5 = objective(result_cobyla_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_cobyla_random.x
    
    while time.time() - start_time < budget_s - 20:
        perturbation = np.random.normal(0, 0.01, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        result_cobyla_perturb = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'disp': False}
        )
        if result_cobyla_perturb.success:
            current_c5 = objective(result_cobyla_perturb.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_cobyla_perturb.x
    
    return best_h, best_c5, n_points
