import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_sine_guess():
        x = np.linspace(0, 2, n_points)
        sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
        return sine_guess
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    initial_guesses = [initial_guess, generate_sine_guess(), generate_random_guess(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e2 * constraint_violation  # Reduced penalty
        return c5 + penalty
    
    def objective_local(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    for guess in initial_guesses:
        try:
            result_de = optimize.differential_evolution(
                fun=objective_de,
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                popsize=20,
                maxiter=150,
                mutation=(0.5, 1.0),
                recombination=0.9,
                tol=1e-10,
                vectorized=True
            )
            if result_de.success:
                result_cobyla = optimize.minimize(
                    fun=objective_local,
                    x0=result_de.x,
                    method='COBYLA',
                    bounds=[(0.0, 1.0) for _ in range(n_points)],
                    constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_cobyla.success:
                    current_c5 = objective_local(result_cobyla.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_cobyla.x
        except:
            pass
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize.minimize(
            fun=objective_local,
            x0=perturbed,
            method='COBYLA',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],

        )
        if result.success:

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
