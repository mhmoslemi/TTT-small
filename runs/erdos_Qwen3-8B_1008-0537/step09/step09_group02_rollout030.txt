import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def square_wave_guess(n, dx):
        h = np.zeros(n)
        for i in range(n):
            x = i * dx
            mod_x = x % 1.0
            if mod_x < 0.5:
                h[i] = 1.0
            else:
                h[i] = 0.0
        return h
    
    h_square = square_wave_guess(n_points, dx)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_from_guess(x0):
        res = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return res.success, res.x, objective(res.x)
    
    # Initial optimization from initial_guess and square_wave
    for guess in [initial_guess, h_square]:
        success, res_x, res_c5 = optimize_from_guess(guess)
        if success and res_c5 < best_c5:
            best_c5 = res_c5
            best_h = res_x
    
    # Differential Evolution
    try:
        res_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=20,
            maxiter=100,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if res_de.success:
            success, res_x, res_c5 = optimize_from_guess(res_de.x)
            if success and res_c5 < best_c5:
                best_c5 = res_c5
                best_h = res_x
    except:
        pass
    
    # Perturb the best solution
    while time.time() - start_time < budget_s - 20:
        perturb = np.random.normal(0, 0.01, size=n_points)
        new_guess = best_h + perturb
        new_guess = np.clip(new_guess, 0.0, 1.0)
        success, res_x, res_c5 = optimize_from_guess(new_guess)
        if success and res_c5 < best_c5:
            best_c5 = res_c5
            best_h = res_x
    
    return best_h, best_c5, n_points
