import numpy as np
import scipy.signal as signal
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = signal.fftconvolve(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=20,
            maxiter=80,
            mutation=(0.7, 1.0),
            recombination=0.8,
            tol=1e-8,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    def generate_regular_guess(n, m):
        positions = np.linspace(0, 2, m, endpoint=False)
        guess = np.zeros(n)
        for i in range(m):
            idx = np.argmin(np.abs(np.linspace(0, 2, n) - positions[i]))
            guess[idx] = 1.0
        return guess
    
    guesses = [initial_guess]
    for m in range(4, 8):
        guess = generate_regular_guess(n_points, m)
        guesses.append(guess)
    
    for guess in guesses:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    while time.time() - start_time < budget_s - 20:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)

            fun=objective,
            x0=new_guess,
            method='SLSQP',

            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
