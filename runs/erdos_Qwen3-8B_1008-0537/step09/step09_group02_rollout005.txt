import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def optimize_with_basinhopping(x0):
        result = optimize.basinhopping(
            fun=objective,
            x0=x0,
            niter=50,
            T=1.0,
            stepsize=0.5,
            minimizer_kwargs={
                'method': 'SLSQP',
                'bounds': bounds,
                'constraints': [{'type': 'eq', 'fun': constraint_sum}],
                'options': {'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            },
            seed=seed
        )
        return result.x
    
    # Initial basinhopping run

        current_c5 = objective(result_bh)

            best_h = result_bh.copy()
    except:
        pass
    
    # Random guess
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)

        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess

    except:

    # Local refinement
