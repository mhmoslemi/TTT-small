            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    # Final refinement
    if time.time() - start_time < budget_s - 30:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
