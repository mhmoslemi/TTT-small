            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    random_guess = generate_random_guess(n_points)
    result = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result.success:
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    while time.time() - start_time < budget_s - 50:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0.0, 1.0)
        result = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
