import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2

    def generate_structured_initial(m=10):
        spacing = 2.0 / m
        interval_length = 1.0 / m
        centers = [(i - 0.5) * spacing for i in range(1, m + 1)]
        intervals = [(c - interval_length / 2, c + interval_length / 2) for c in centers]
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for (start, end) in intervals:
                if start <= x <= end:
                    h[i] = 1.0
                    break
        return h

    def generate_random_initial():
        indices = np.random.permutation(n_points)
        random_h = np.zeros(n_points)
        random_h[indices[:int(n_points / 2)]] = 1.0
        return random_h

    initial_guesses = [h_initial, generate_structured_initial(), generate_random_initial()]

    best_c5 = float('inf')
    best_h = None

    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr * dx)

        bounds = [(0.0, 1.0) for _ in range(n_points)]

        options = {'maxiter': 500, 'ftol': 1e-9, 'disp': False}
        result = minimize(objective, guess, bounds=bounds, constraints=constraints, method='L-BFGS-B', options=options)

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:

                best_h = result.x

    return (best_h, best_c5, n_points)
