    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturbation, 0.0, 1.0)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    return best_h, best_c5, n_points
