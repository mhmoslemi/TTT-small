def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2

    def random_binary_initial():
        idx = np.random.choice(n_points, size=int(n_points/2), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h

    def generate_structured_initial(period=4):
        spacing = 2.0 / period
        centers = [(i - 0.5) * spacing for i in range(1, period + 1)]
        intervals = [(c - 0.25 * spacing, c + 0.25 * spacing) for c in centers]
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for (start, end) in intervals:
                if start <= x <= end:
                    h[i] = 1.0
                    break
        return h

    def objective(h, dx=dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def constraints(h):
        return np.sum(h) - sum_required

    def evaluate_c5(h, dx=dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    initial_guesses = [h_initial, generate_structured_initial(), random_binary_initial()]

    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points

    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraints}]
        result = minimize(objective, guess, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8, options={'maxiter': 300})
        if result.success:

            if current_c5 < best_c5:
