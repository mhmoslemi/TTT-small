import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    current_guess = initial_guess.copy()
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Perturb current guess
        perturbed_guess = current_guess.copy() + np.random.normal(0, 0.05, n_points)
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        if np.sum(perturbed_guess) < 1e-6:
            perturbed_guess += np.random.uniform(0, 1e-6, n_points)
        perturbed_guess = perturbed_guess * (target_sum) / np.sum(perturbed_guess)
        
        # Use L-BFGS-B with tighter tolerance and higher iterations
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                current_guess = best_h.copy()
    
    return best_h, best_c5, n_points
