import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    sum_required = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values, dtype=float)
        # Normalize to ensure sum is correct, avoiding division by zero
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        h = h * sum_required / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    # Run differential evolution to explore global optima
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=15,
        maxiter=300,
        tol=1e-5,
        mutation=(0.5, 0.9),
        recombination=0.8,
        seed=seed,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Early exit if time is exhausted
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    # Refine with SLSQP
    def constraint(h_values):
        return np.sum(h_values) - sum_required
    
    result_refine = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    best_h = result_refine.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
