import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2

    initial_guesses = [np.array(initial_h_values, dtype=float)]
    
    # Generate a more evenly spaced binary initial guess
    number_of_ones = n_points // 2
    spacing = n_points / number_of_ones
    binary_guess = np.zeros(n_points, dtype=float)
    for i in range(number_of_ones):
        pos = int(round(i * spacing))
        if pos < n_points:
            binary_guess[pos] = 1
    initial_guesses.append(binary_guess)
    
    # Additional structured initial guess (alternating pattern)
    alt_guess = np.zeros(n_points, dtype=float)
    for i in range(n_points):
        alt_guess[i] = 1 if i % 3 == 0 else 0.5
    initial_guesses.append(alt_guess)
    
    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for guess in initial_guesses:
        for _ in range(3):  # Limit iterations per guess
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.05, size=n_points)
            perturbed_guess = np.clip(guess + perturb, 0, 1)

                constraints=[{'type': 'eq', 'fun': constraint_sum}],

                if current_c5 < best_c5:

                    best_h = result.x
    
    return best_h, best_c5, n_points

def objective(h_values):
    h = np.array(h_values, dtype=float)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def constraint_sum(h_values):
    return np.sum(h_values) - target_sum
