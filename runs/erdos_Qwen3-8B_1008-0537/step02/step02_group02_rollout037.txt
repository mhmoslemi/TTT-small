def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=np.float64)
    np.random.seed(seed)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        alpha = 100  # Tuned to balance smoothness and accuracy
        soft_max = np.log(np.sum(np.exp(alpha * corr))) / alpha
        return soft_max * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None

    # Multi-start optimization with DE for better global exploration
    for _ in range(4):
        if time.time() - start_time > budget_s:
            break
        # Generate diverse initial guess for DE
        pop = np.random.uniform(0, 1, (20, n_points))
        for i in range(pop.shape[0]):
            pop[i] = np.clip(pop[i], 0, 1)
            pop[i] = np.where(np.sum(pop[i]) < target_sum - 0.1, 
                              pop[i] * (target_sum / np.sum(pop[i])), pop[i])
            pop[i] = np.where(np.sum(pop[i]) > target_sum + 0.1, 
                              pop[i] * (target_sum / np.sum(pop[i])), pop[i])
        result = optimize.differential_evolution(
            fun=objective, 
            bounds=bounds, 
            constraints=[{'type': 'eq', 'fun': constraint_sum}], 
            popsize=15, 
            strategy='best1bin', 
            mutation=(0.5, 1), 
            recombination=0.9, 
            tol=1e-6,
            maxiter=100,
            seed=seed + _,
            disp=False
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
