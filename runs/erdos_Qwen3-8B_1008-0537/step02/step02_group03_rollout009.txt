import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Use basin-hopping for global search
    niter = 50  # increased number of iterations for thorough search
    T = 1.0  # initial temperature
    stepsize = 0.5  # step size for perturbation
    
    # Set up minimizer_kwargs with constraint
    minimizer_kwargs = {
        'method': 'SLSQP',
        'bounds': bounds,
        'constraints': [{'type': 'eq', 'fun': constraint_sum}],
        'options': {'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    }
    
    try:
        result = optimize.basinhopping(
            objective,
            initial_guess,
            niter=niter,
            T=T,
            stepsize=stepsize,
            minimizer_kwargs=minimizer_kwargs,
            disp=False,
            niter_success=10,
            # Early stopping if budget is exceeded
            callback=lambda x: None if (time.time() - start_time) < budget_s else None
        )
    except:
        pass
    
    # Extract best solution if available
    if hasattr(result, 'x'):
        best_h = result.x
        best_c5 = objective(best_h)
    elif len(best_h) < n_points:
        best_h = initial_guess.copy()
        best_c5 = objective(best_h)
    
    # Ensure not exceeding time budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    # Final fallback to initial guess if not improved
    if best_c5 >= objective(initial_guess):
        best_h = initial_guess.copy()
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
