    best_c5 = objective(best_h)
    result = None
    
    # Run multiple optimizations with different methods
    for _ in range(8):
        if time.time() - start_time > budget_s:
            break
        # Use different methods and perturbations
        perturb = np.random.normal(0, 0.05, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        
        # Try COBYLA
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'disp': False, 'maxiter': 500, 'ftol': 1e-9}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
        
        # Try L-BFGS-B
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'disp': False, 'maxiter': 500, 'ftol': 1e-9}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_guess = best_h
    
    return best_h, best_c5, n_points
