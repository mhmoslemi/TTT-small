import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable if available
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Run optimizations from multiple initial guesses
    num_trials = 5
    for trial in range(num_trials):
        if time.time() - start_time > budget_s:
            break
        # First trial uses initial_guess, others use random guesses
        if trial == 0:
            x0 = initial_guess
        else:
            # Generate random initial guess with Dirichlet distribution
            rng = np.random.default_rng(seed + trial)
            dirichlet = rng.dirichlet(np.ones(n_points))
            random_h = dirichlet * target_sum
            x0 = random_h
        # Run optimization
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
