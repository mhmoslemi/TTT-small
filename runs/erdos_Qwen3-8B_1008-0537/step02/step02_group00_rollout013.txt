import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        sum_val = np.sum(h)
        sum_penalty = 1e5 * abs(sum_val - target_sum)
        return max_corr * dx + sum_penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Differential evolution with tuned parameters
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=30,
        mutation=(0.6, 1.0),
        recombination=0.8,
        tol=1e-6,
        polish=True,
        seed=seed
    )
    
    if result.success:
