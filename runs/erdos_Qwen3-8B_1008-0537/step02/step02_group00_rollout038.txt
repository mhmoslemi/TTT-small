    if time.time() - start_time < budget_s:
        # Refine with SLSQP
        result_slsqp = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result_slsqp.success:
            best_h = result_slsqp.x
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
