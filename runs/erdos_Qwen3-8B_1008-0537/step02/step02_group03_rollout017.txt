import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess
    best_c5 = objective(best_h)
    
    # Generate diverse initial guesses
    initial_guesses = [
        initial_guess.copy(),
        initial_guess.copy() + np.random.normal(0, 0.05, n_points),
        np.random.rand(n_points),
        initial_guess.copy() * 0.5 + np.random.rand(n_points) * 0.5,
        np.random.rand(n_points) * 0.5 + 0.5,
    ]
    
    # Normalize guesses to meet the sum constraint
    for i in range(len(initial_guesses)):
        guess = initial_guesses[i].copy()
        guess = np.clip(guess, 0, 1)
        if np.isclose(np.sum(guess), 0):
            guess = np.random.rand(n_points) * (target_sum / n_points)
        else:
            guess = guess * (target_sum / np.sum(guess))
        initial_guesses[i] = guess
    
    # Run optimization for each guess
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,

            if current_c5 < best_c5:
