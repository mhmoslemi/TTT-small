import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 500 * abs(np.sum(h) - target_sum)
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Differential evolution with adaptive mutation
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.9, 1.1),
        recombination=0.8,
        tol=1e-6,
        maxiter=50,
        polish=True,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
