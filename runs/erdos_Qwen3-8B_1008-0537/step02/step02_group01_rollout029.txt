                if start <= x <= end:
                    h[i] = 1.0
                    break
        return h

    def generate_random():
        rand_indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h = np.zeros(n_points)
        h[rand_indices] = 1.0
        return h

    initial_guesses = [h_initial, generate_structured(10), generate_structured(5), generate_structured(15), generate_random()]

    best_c5 = float('inf')
    best_h = None

    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr * dx)
            penalty = 1000 * (np.sum(h) - sum_required)**2
            return c5 + penalty

        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, maxiter=50, tol=1e-5, mutation=(0.5, 1), recombination=0.8, disp=False)
        if result.success:
            c5 = result.fun - 1000 * (np.sum(result.x) - sum_required)**2
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x

    return (best_h, best_c5, n_points)
