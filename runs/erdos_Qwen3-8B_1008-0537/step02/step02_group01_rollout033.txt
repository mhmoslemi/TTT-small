    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2

    def generate_structured_initial(m):
        interval_length = 1.0 / m
        spacing = 2.0 / m
        centers = [i * spacing for i in range(1, m + 1)]
        intervals = [(c - interval_length / 2, c + interval_length / 2) for c in centers]
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for (start, end) in intervals:
                if start <= x <= end:
                    h[i] = 1.0
                    break
        return h

    def generate_random_initial(n_points):
        h = np.random.rand(n_points)
        h = np.round(h).astype(float)
        desired_sum = n_points / 2
        current_sum = np.sum(h)
        if current_sum > 0:
            h *= desired_sum / current_sum
        return h

    initial_guesses = [h_initial, generate_structured_initial(2), generate_structured_initial(3), 
                       generate_structured_initial(4), generate_random_initial(n_points)]

    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points

    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr * dx)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
        
        options = {'maxiter': 500, 'ftol': 1e-6, 'disp': False}
        result = minimize(objective, guess, bounds=bounds, constraints=constraints, 
                          method='SLSQP', options=options)
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return (best_h, best_c5, n_points)
