    perturbed_guess = initial_guess + np.random.normal(0, 0.1, n_points)
    perturbed_guess = np.clip(perturbed_guess, 0, 1)
    initial_guesses.append(perturbed_guess)
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        guess = np.array(random.choice(initial_guesses))
        method = random.choice(['SLSQP', 'L-BFGS-B', 'Nelder-Mead', 'COBYLA', 'differential_evolution'])
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
            )
        except:
            continue
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
