    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    np.random.seed(seed)
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    current_guess = initial_guess.copy()
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturbed = current_guess + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed) < 1e-6:
            perturbed += np.random.uniform(0, 1e-6, n_points)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                current_guess = best_h
    
    return best_h, best_c5, n_points
