    def compute_c5(h):
        n_points = len(h)
        max_integral = 0.0
        dx = 2.0 / n_points
        for m in range(n_points):
            i_max = n_points - m
            if i_max <= 0:
                continue
            valid_h = h[:i_max]
            shifted_h = h[m:m + i_max]
            integral = np.dot(valid_h, 1 - shifted_h) * dx
            max_integral = max(max_integral, integral)
        return max_integral

    def generate_random_h():

        h = (h - np.mean(h)) / np.std(h)
