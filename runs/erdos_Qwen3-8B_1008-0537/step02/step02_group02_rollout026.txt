import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(6):  # 6 trials to explore diverse regions
        if time.time() - start_time > budget_s:
            break
        # Randomized initialization
        rng = np.random.RandomState(seed + _)
        random_guess = rng.uniform(0, 1, size=n_points)
        scaled_guess = random_guess * (target_sum / np.sum(random_guess))
        res = optimize.minimize(
            objective, scaled_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'disp': False}
        )
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
        
        # Perturb best solution with added noise
        perturbation = rng.normal(0, 0.05, size=n_points)
        perturbed_guess = np.clip(best_h + perturbation, 0.0, 1.0)
        res = optimize.minimize(
            objective, perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if res.success:

            if current_c5 < best_c5:

        time.sleep(0.1)
