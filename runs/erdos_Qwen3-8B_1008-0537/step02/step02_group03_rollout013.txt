    best_c5 = float('inf')
    start_time = time.time()
    
    # Evaluate initial guesses
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Additional refinement if time allows
    if time.time() - start_time < budget_s:
        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            perturbed = best_h.copy() + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            if np.sum(perturbed) < 1e-6:
                perturbed += np.random.uniform(0, 1e-6, n_points)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return best_h, best_c5, n_points
