import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * abs(np.sum(h) - target_sum)
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Basin-hopping with controlled iterations
    result = optimize.basinhopping(
        fun=objective,
        x0=initial_guess,
        niter=100,  # 100 iterations
        T=1.0,
        stepsize=0.5,
        minimize=True,
        disp=False,
        minimizer_kwargs={
            'method': 'L-BFGS-B',
            'bounds': bounds,
            'ftol': 1e-8,
            'maxiter': 100
        }
    )
    
    if result.success:
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Normalize to ensure sum matches target_sum
    sum_h = np.sum(best_h)
    if abs(sum_h - target_sum) > 1e-6:
        best_h = best_h * (target_sum / sum_h)
    
    return best_h, best_c5, n_points
