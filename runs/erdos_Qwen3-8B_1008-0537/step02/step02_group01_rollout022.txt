import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2

    def generate_m_intervals(m, spacing_fraction=0.3):
        interval_length = 1.0 / m
        spacing = (2.0 - m * interval_length) * spacing_fraction
        centers = [(i + 0.5) * (spacing + interval_length) for i in range(m)]
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for c in centers:
                if c - interval_length / 2 <= x <= c + interval_length / 2:
                    h[i] = 1.0
                    break
        return h

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
    best_c5 = float('inf')
    best_h = None

    initial_guesses = [h_initial]
    for m in range(3, 16):
        initial_guesses.append(generate_m_intervals(m))
    for m in [5, 7, 13, 17]:
        initial_guesses.append(generate_m_intervals(m, spacing_fraction=0.5))

    for guess in initial_guesses:
        options = {'maxiter': 250, 'ftol': 1e-8, 'disp': False}
        result = minimize(objective, guess, bounds=bounds, constraints=constraints, method='SLSQP', options=options)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return (best_h, best_c5, n_points)
