import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(x):
        h_vals = np.array(x)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Differential evolution for global search
    try:
        res = optimize.differential_evolution(
            objective, bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            strategy='best1bin', popsize=15, mutation=(0.5, 1), recombination=0.8,
            tol=1e-8, maxiter=100, disp=False, polish=False)
        if res.success:
            current_h = res.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5, best_h = current_c5, current_h
    except:
        pass
    
    # Local refinement with SLSQP
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        res = optimize.minimize(
            objective, best_h, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9})
        if res.success:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
