import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2

    def generate_structured(m=10):
        spacing = 2.0 / m
        centers = [(i - 0.5) * spacing for i in range(1, m + 1)]
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for c in centers:
                if abs(x - c) < spacing / 2:
                    h[i] = 1.0
                    break
        return h

    def generate_random_binary():
        n_ones = n_points // 2
        indices = np.random.permutation(n_points)[:n_ones]
        h = np.zeros(n_points, dtype=np.float64)
        h[indices] = 1.0
        return h

    initial_guesses = [h_initial, generate_structured(10), generate_random_binary()]

    best_c5 = float('inf')
    best_h = None
    best_n = n_points

    for guess in initial_guesses:
        def objective(h):
            corr = correlate(h, 1.0 - h, mode='full')
            return np.max(corr * dx)

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]

        res = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x

    return (best_h, best_c5, n_points)
