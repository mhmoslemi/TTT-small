import numpy as np
from scipy.optimize import minimize, basinhopping

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2
    
    def generate_structured_initial(m):
        spacing = 2.0 / m
        interval_length = 1.0 / m
        centers = [(i - 0.5) * spacing for i in range(1, m + 1)]
        intervals = [(c - interval_length / 2, c + interval_length / 2) for c in centers]
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for (start, end) in intervals:
                if start <= x <= end:
                    h[i] = 1.0
                    break
        return h
    
    initial_guesses = [h_initial]
    for m in [5, 10, 15, 20, 25]:
        initial_guesses.append(generate_structured_initial(m))
    
    best_c5 = float('inf')
    best_h = None
    
    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr * dx)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
        
        def optimize():
            result = minimize(objective, guess, bounds=bounds, constraints=constraints, method='L-BFGS-B', options={'maxiter': 100})
            if result.success:
                return result.x
            return np.copy(guess)
        
        best_in_guess = optimize()
        corr = np.correlate(best_in_guess, 1 - best_in_guess, mode='full')
        current_c5 = np.max(corr * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_in_guess
    
    return (best_h, best_c5, n_points)
