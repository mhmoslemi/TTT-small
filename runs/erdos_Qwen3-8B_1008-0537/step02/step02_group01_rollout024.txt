import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_required = n_points / 2.0

    def generate_structured_initial(m, n_points):
        spacing = 2.0 / m
        interval_length = 1.0 / m
        centers = [(i - 0.5) * spacing for i in range(1, m + 1)]
        intervals = [(c - interval_length / 2, c + interval_length / 2) for c in centers]
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for (start, end) in intervals:
                if start <= x <= end:
                    h[i] = 1.0
                    break
        return h

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]

    def run_optimization(guess, method='SLSQP'):
        options = {'maxiter': 500, 'ftol': 1e-8, 'disp': False}
        result = minimize(objective, guess, bounds=bounds, constraints=constraints, method=method, options=options)
        return result.x, result.fun if result.success else None

    initial_guesses = [initial_h_values]
    for m in [5, 10, 15, 20]:
        initial_guesses.append(generate_structured_initial(m, n_points))

    best_c5 = float('inf')
    best_h = None

    for guess in initial_guesses:
        h_opt, c5 = run_optimization(guess, method='L-BFGS-B')
        if h_opt is not None and c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    return (best_h, best_c5, n_points)
