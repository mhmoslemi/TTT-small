    def objective(h_values):
        h = np.array(h_values, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        sum_h = np.sum(h)
        penalty = 1e6 * (sum_h - target_sum)**2
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )

    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)

    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        perturbation = np.random.normal(0, 0.1, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturbation, 0.0, 1.0)
        curr_c5 = objective(perturbed_guess)
        if curr_c5 < best_c5:
            best_h = perturbed_guess
            best_c5 = curr_c5

    return best_h, best_c5, n_points
