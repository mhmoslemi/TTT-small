        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        result = optimize.minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = np.max(np.correlate(result.x, 1 - result.x, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
