    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-8,
        polish=True,
        disp=False,
        vectorsize=n_points
    )
    
    # Adjust for time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return result.x, objective(result.x), n_points
    
    return result.x, objective(result.x), n_points
