import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h_values = [0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333, 0.6667, 0.5, 0.5, 0.3333]
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    for _ in range(8):
        if time.time() - start_time > budget_s:
            break
        # Generate diverse perturbations
        if np.random.rand() < 0.5:
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
        else:
            perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(initial_guess + perturbation, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        
        # Refine with L-BFGS-B
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        initial_guess = best_h.copy()
    
    return best_h, best_c5, n_points
