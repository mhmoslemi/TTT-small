        h_initial = np.zeros(n_points)
        total_ones = n_points // 2
        for i in range(total_ones):
            pos = round(i * (n_points / total_ones))
            if 0 <= pos < n_points:
                h_initial[pos] = 1.0
        h_binary = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_binary)
        if sum_h < total_ones - 1e-6:
            h_binary[0] = 1.0
        h_initial = h_binary

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = h_initial.copy()

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    current_guess = initial_guess.copy()

    for _ in range(6):
        if time.time() - start_time > budget_s:
            break
        while True:
            perturbed_guess = current_guess.copy() + np.random.normal(0, 0.03, n_points)
            perturbed_guess = np.clip(perturbed_guess, 0, 1)
            if np.sum(perturbed_guess) < 1e-6:
                perturbed_guess += np.random.uniform(0, 1e-6, n_points)
            perturbed_guess = perturbed_guess * target_sum / np.sum(perturbed_guess)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    current_guess = best_h.copy()
                break

    return best_h, best_c5, n_points
