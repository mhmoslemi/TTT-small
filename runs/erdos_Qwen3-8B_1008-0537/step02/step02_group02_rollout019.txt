import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) if corr.size > 0 else 0.0
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Differential evolution parameters tuned for efficiency
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        popsize=20,
        mutation=(0.5, 0.9),
        recombination=0.8,
        tol=1e-6,
        seed=seed,
        maxiter=200,
        polish=True
    )
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > 0.95 * budget_s:
        # Early exit if time is nearly exhausted
        return best_h, best_c5, n_points
    
    # Post-processing: refine with local minimization
    bounds_local = [(0.0, 1.0) for _ in range(n_points)]
    result_local = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds_local,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-7, 'maxiter': 50}
    )
    
    if result_local.success:
        best_h = result_local.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
