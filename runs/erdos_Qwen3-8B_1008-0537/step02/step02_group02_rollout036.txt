import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e6 * (np.sum(h) - target_sum)**2
        return max_corr * dx + penalty
    
    start_time = time.time()
    result = optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        maxiter=300,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
