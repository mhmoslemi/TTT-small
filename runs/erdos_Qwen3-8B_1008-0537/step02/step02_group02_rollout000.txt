import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(xvec, dx):
        h = xvec
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(xvec):
        return np.sum(xvec) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h, dx)
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        pert = np.random.normal(0, 0.03, n_points)
        perturbed = np.clip(h + pert, 0.0, 1.0)
        result = optimize.minimize(
            fun=lambda x: objective(x, dx),
            x0=perturbed,
            method='trust-constr',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 500, 'gtol': 1e-8, 'verbose': 0}
        )
        if result.success:
            current_c5 = objective(result.x, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                h = best_h
    
    return best_h, best_c5, n_points
