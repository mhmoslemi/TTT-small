    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr * dx)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
        
        options = {'maxiter': 500, 'ftol': 1e-8, 'disp': False}
        result = minimize(objective, guess, bounds=bounds, constraints=constraints, method='SLSQP', options=options)
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
