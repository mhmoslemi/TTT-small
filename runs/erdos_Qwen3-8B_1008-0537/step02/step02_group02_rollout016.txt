import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is provided globally
    n_points = len(initial_h_values)
    target_sum = n_points / 2

    # Run differential evolution with penalty
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=20,
        maxiter=50,
        tol=1e-8,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        polish=True,
        vectorized=True
    )
    
    if result.success:
        current_h = result.x

    # Final check on initial guess
