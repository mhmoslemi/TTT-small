import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable if available
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    result = None
    
    # Run multiple optimizations to escape local minima
    for _ in range(10):  # Increased number of iterations
        if time.time() - start_time > budget_s:
            break
        # Perturb the initial guess to escape local minima
        if n_points > 0:
            perturbation = np.random.normal(0, 0.03, size=n_points)
            perturbed_guess = np.clip(initial_guess + perturbation, 0.0, 1.0)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    initial_guess = best_h  # Use updated guess
    
    return best_h, best_c5, n_points
