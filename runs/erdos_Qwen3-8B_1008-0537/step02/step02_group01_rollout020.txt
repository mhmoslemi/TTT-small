def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h_initial)
    dx = 2.0 / n_points

                if c - l / 2 <= x <= c + l / 2:
                    h[i] = 1.0
                    break
        return h
    
    initial_guesses = [h_initial]
    for m in [5, 6, 7, 8, 9, 10]:
        initial_guesses.append(generate_structured_initial(m))
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points
    
    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr * dx)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
