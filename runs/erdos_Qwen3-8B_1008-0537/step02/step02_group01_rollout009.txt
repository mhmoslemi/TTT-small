import numpy as np
from scipy.optimize import minimize, DifferentialEvolution
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_required = n_points / 2
    
    def generate_structured(m=10):
        spacing = 2.0 / m
        interval_length = 1.0 / m
        centers = [(i - 0.5) * spacing for i in range(1, m + 1)]
        intervals = [(c - interval_length / 2, c + interval_length / 2) for c in centers]
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for (start, end) in intervals:
                if start <= x <= end:
                    h[i] = 1.0
                    break
        return h
    
    def generate_random_binary():
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h = np.zeros(n_points, dtype=np.float64)
        h[indices] = 1.0
        return h
    
    initial_guesses = [h_initial, generate_structured(5), generate_structured(20), generate_random_binary()]
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points
    
    start_time = time.time()
    
    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr * dx)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
