import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)
    
    # Generate some random initial guesses
    num_initial_guesses = 5
    initial_guesses = [initial_h_values]
    for _ in range(num_initial_guesses):
        # Generate a random guess that meets the sum constraint
        h = np.random.rand(n_points)
        current_sum = np.sum(h)
        if current_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * (target_sum / current_sum)
        initial_guesses.append(h.tolist())
    
    # Try optimizing from each initial guess
    results = []
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid exact initial values
        perturb = np.random.normal(0, 0.05, size=n_points)
        perturbed_guess = np.clip(np.array(guess) + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            results.append((current_c5, result.x))
    
    # Select the best result
    best_c5 = min([x[0] for x in results]) if results else objective(best_h)
    best_h = [x[1] for x in results][np.argmin([x[0] for x in results])] if results else best_h
    
    return best_h, best_c5, n_points
