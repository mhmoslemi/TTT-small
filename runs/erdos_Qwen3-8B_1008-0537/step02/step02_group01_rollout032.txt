        generate_structured_initial(5),
        generate_structured_initial(10),
        generate_structured_initial(15),
        generate_random_initial()
    ]

    best_c5 = float('inf')
    best_h = None

    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr * dx)

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
        options = {'maxiter': 300, 'ftol': 1e-7, 'disp': False}
        result = minimize(objective, guess, bounds=bounds, constraints=constraints, method='SLSQP', options=options)

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    if best_h is not None:
        perturbed_h = best_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(perturbed_h + noise, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed_h)
        diff = sum_required - sum_perturbed
        if abs(diff) > 1e-6:
            indices = np.random.choice(n_points, size=int(abs(diff)), replace=False)
            perturbed_h[indices] += np.sign(diff)
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)

        result = minimize(objective, perturbed_h, bounds=bounds, constraints=constraints, method='SLSQP', options=options)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return (best_h, best_c5, n_points)
