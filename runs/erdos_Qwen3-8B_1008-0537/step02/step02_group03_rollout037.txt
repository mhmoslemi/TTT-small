    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Use basinhopping with SLSQP local optimization
    def minimize_with_slsqp(x0):
        res = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        return res.x if res.success else x0
    
    # Basinhopping to escape local optima
    result = optimize.basinhopping(
        fun=objective,
        x0=initial_guess,
        minimizer_kwargs={'method': 'L-BFGS-B', 'bounds': bounds},
        niter=50,
        T=1.0,
        stepsize=0.5,
        # Local optimization with SLSQP
        seed=seed
    )
    
    # Extract best solution within budget
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 0:
        return best_h, best_c5, n_points
    
    # Final optimization with SLSQP
    final_result = optimize.minimize(
        fun=objective,
        x0=result.x,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )
    best_h = final_result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
