        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed
    )

    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
