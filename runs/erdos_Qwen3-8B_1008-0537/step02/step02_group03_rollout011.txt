    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Generate initial and diverse starting points
    starting_points = [initial_guess.copy()]
    for _ in range(2):
        random_start = np.random.uniform(0, 1, n_points)
        random_start = random_start * target_sum / np.sum(random_start)
        starting_points.append(random_start)
    
    for start_guess in starting_points:
        result = optimize.minimize(
            fun=objective,
            x0=start_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Further refine with perturbation
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        perturbed_guess = best_h + np.random.normal(0, 0.05, n_points)
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        if np.sum(perturbed_guess) < 1e-6:
            perturbed_guess += np.random.uniform(0, 1e-6, n_points)
        perturbed_guess = perturbed_guess * target_sum / np.sum(perturbed_guess)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
