        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # We need to handle sum constraint by enforcing it as a bound
    # So create a new x that satisfies the sum constraint
    def de_constraint(x):
        return np.sum(x) - target_sum
    
    # Run differential evolution with bounds and constraints
    result_de = differential_evolution(
        fun=de_objective,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': de_constraint}],
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        tol=1e-6,
        polish=True
    )
    
    best_c5 = min(best_c5, result_de.fun)
    best_h = result_de.x
    
    # Fallback to SLSQP if DE fails
    if not result_de.success:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_c5 = min(best_c5, result_slsqp.fun)
        best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
