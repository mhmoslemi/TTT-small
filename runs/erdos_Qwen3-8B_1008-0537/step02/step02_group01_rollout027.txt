        options = {'maxiter': 300, 'ftol': 1e-8, 'disp': False}
        try:
            result = minimize(objective, guess, bounds=bounds, constraints=constraints, method='SLSQP', options=options)
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    return (best_h, best_c5, n_points)
