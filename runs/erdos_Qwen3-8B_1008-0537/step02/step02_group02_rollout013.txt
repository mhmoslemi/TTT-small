import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Differential evolution with penalty for constraint
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 0.8),
        recombination=0.7,
        tol=1e-6,
        polish=True,
        maxiter=50,
        seed=seed,
        disp=False
    )
    
    if result.success:
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # SLSQP refinement
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = optimize.minimize(

        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-9}
    )

        best_h = result_slsqp.x
        best_c5 = objective(best_h)
