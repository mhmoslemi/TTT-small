        objective, initial_guess, niter=50, T=1.0,
        minimizer_kwargs={
            'method': 'SLSQP',
            'bounds': bounds,
            'constraints': [constraint],
        },
        # Reduce the temperature and increase iterations for better exploration
        # but within budget constraints
    )
    
    # Check if the result is successful
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Early exit if time is up
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    # Additional refinement using local optimization
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[constraint],
        options={'ftol': 1e-10, 'maxiter': 100}
    )
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
