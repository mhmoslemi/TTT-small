        initial_guesses.append(generate_structured_initial(m))
    initial_guesses.append(generate_random_initial())

    best_c5 = float('inf')
    best_h = None

    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr * dx)

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]

        if np.random.rand() < 0.5:
            method = 'L-BFGS-B'
            options = {'maxiter': 400, 'ftol': 1e-8, 'disp': False}
        else:
            method = 'SLSQP'
            options = {'maxiter': 300, 'ftol': 1e-8, 'disp': False}

        result = minimize(objective, guess, bounds=bounds, constraints=constraints, method=method, options=options)

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return (best_h, best_c5, n_points)
