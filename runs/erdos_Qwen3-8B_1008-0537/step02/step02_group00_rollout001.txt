import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=np.float64)
    
    def objective(h_values):
        h = np.array(h_values, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Use multiple methods to escape local optima
    for method in ['L-BFGS-B', 'SLSQP', 'Nelder-Mead']:
        if time.time() - start_time > budget_s:
            break
        # Perturb and optimize
        perturb = np.random.normal(0, 0.05, size=n_points)
        guess = np.clip(initial_guess + perturb, 0, 1).copy()
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                initial_guess = best_h
    
    return best_h, best_c5, n_points
