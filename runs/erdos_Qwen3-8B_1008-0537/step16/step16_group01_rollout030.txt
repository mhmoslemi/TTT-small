import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guess *= target_sum / np.sum(initial_guess)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_guesses(num_guesses=10):
        guesses = [initial_guess.copy()]
        # Add a structured binary guess
        binary_guess = np.zeros(n_points, dtype=float)
        binary_guess[::2] = 1.0
        guesses.append(binary_guess)
        # Add random guesses
        for _ in range(num_guesses - 2):
            h = np.random.uniform(0, 1, n_points)
            h = h * (target_sum / np.sum(h))
            guesses.append(h)
        return guesses

    initial_guesses = generate_guesses(15)

    for guess in initial_guesses:
        result = optimize.minimize(

            method='L-BFGS-B',

            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}

        if result.success:

            if current_c5 < best_c5:

    try:
