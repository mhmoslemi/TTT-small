import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    # Perturb initial guess
    perturb = np.random.normal(0, 0.05, n_points)
    perturbed = np.clip(initial_guess + perturb, 0.0, 1.0)
    perturbed = perturbed * target_sum / np.sum(perturbed)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = perturbed.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess, maxiter=300, ftol=1e-8):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    # Global search with differential evolution
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=300,
            mutation=(0.6, 0.95),
            recombination=0.85,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_cobyla = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'disp': False}
            )
            if result_cobyla.success and objective(result_cobyla.x) < best_c5:
                best_c5 = objective(result_cobyla.x)
                best_h = result_cobyla.x
    except:
        pass
    
    # Local refinement with BFGS
    while time.time() - start_time < budget_s - 30:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
