import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guesses = [np.array(initial_h_values, dtype=float)]
    
    # Generate random initial guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        scale = target_sum / np.sum(guess)
        guess *= scale
        guess = np.clip(guess, 0.0, 1.0)
        initial_guesses.append(guess)
    
    best_h = initial_guesses[0].copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Differential Evolution
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=100,
                maxiter=200,
                mutation=(0.5, 0.9),
                recombination=0.8,
                tol=1e-9,
                vectorized=True,
                seed=seed + initial_guesses.index(guess)
            )
            if result_de.success:
                refined = result_de.x
                # SLSQP refinement
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=refined,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass
        
        # L-BFGS-B local refinement
        elapsed = time.time() - start_time
        if elapsed < budget_s - 30:
            try:
                result = optimize.minimize(
                    fun=objective,
                    x0=guess,
                    method='L-BFGS-B',
                    bounds=bounds,
