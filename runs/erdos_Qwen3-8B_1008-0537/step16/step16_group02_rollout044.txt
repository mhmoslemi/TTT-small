    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_spread_guess():
        num_ones = int(target_sum) + (1 if target_sum - int(target_sum) > 0 else 0)
        indices = np.linspace(0, n_points, num_ones, endpoint=False).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1.0
        sum_h = np.sum(h)
        if abs(sum_h - target_sum) > 1e-8:
            diff = target_sum - sum_h
            if diff > 0:
                h[np.argmin(h)] += diff
            else:
                h[np.argmax(h)] -= abs(diff)
        return h
    
    def generate_random_guess():
        num_ones = int(target_sum)
        h = np.zeros(n_points)
        h[np.random.choice(n_points, size=num_ones, replace=False)] = 1.0
        sum_h = np.sum(h)
        if sum_h < target_sum:
            extra = target_sum - sum_h
            idx = np.random.choice(n_points, size=int(extra), replace=False)
            h[idx] += 1.0
        else:
            extra = sum_h - target_sum
            idx = np.random.choice(n_points, size=int(extra), replace=False)
            h[idx] -= 1.0
        return h
    
    initial_guesses = [initial_guess]
    spread_guess = generate_spread_guess()
    initial_guesses.append(spread_guess)
    for _ in range(2):
        initial_guesses.append(generate_random_guess())
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def optimize_guess(guess, method='SLSQP'):

            fun=objective,
            x0=guess,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

            best_c5 = objective(result)
            best_h = result

            fun=objective,
            bounds=bounds,
            popsize=80,
            maxiter=300,
            mutation=(0.5, 0.95),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(

            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x

        pass
    
    while time.time() - start_time < budget_s - 60:
        try:
            result = optimize.minimize(

                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],

                best_h = result.x
