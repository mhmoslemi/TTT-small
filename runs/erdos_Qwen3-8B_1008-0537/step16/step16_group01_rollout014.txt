    def objective(h_values):
        h = np.array(h_values)
        overlap = np.zeros(n_points * 2 - 1)
        for i in range(n_points):
            for j in range(n_points):
                x = i * dx
                y = j * dx
                k = (x - y) / dx
                if 0 <= k <= 2:

                    r = int(i)
                    if l < n_points and r < n_points:
                        overlap[int(k)] += h[i] * (1 - h[j]) * dx
        return np.max(overlap)

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    best_h = initial_guess.copy()
