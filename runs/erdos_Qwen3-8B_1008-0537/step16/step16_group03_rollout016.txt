import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def optimize_guess(guess, maxiter=500, ftol=1e-8):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-9}
        )
        return result.x if result.success else None

    try:
        # Global search with differential evolution with adjusted parameters
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=100,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            best_h = result_de.x
            best_c5 = objective(best_h)
    except:
        pass

    # Local refinement with SLSQP
    try:
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success and objective(result_slsqp.x) < best_c5:
            best_h = result_slsqp.x
            best_c5 = objective(best_h)
    except:
        pass

    # Final fine-tuning with L-BFGS-B
    try:
        while time.time() - start_time < budget_s - 30:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,

        pass
