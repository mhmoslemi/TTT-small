import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess, method, maxiter=100, ftol=1e-10):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    # Local refinement with multiple methods
    for _ in range(5):
        try:
            result_slsqp = optimize_guess(best_h, 'SLSQP', maxiter=100, ftol=1e-10)
            if result_slsqp is not None and objective(result_slsqp) < best_c5:
                best_c5 = objective(result_slsqp)
                best_h = result_slsqp
        except:
            pass
        
        try:
            result_bfgs = optimize_guess(best_h, 'L-BFGS-B', maxiter=50, ftol=1e-10)
            if result_bfgs is not None and objective(result_bfgs) < best_c5:
                best_c5 = objective(result_bfgs)
                best_h = result_bfgs
        except:
            break
    
    # Perturb and refine
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        try:
            result_bfgs = optimize_guess(perturbed_h, 'L-BFGS-B', maxiter=50, ftol=1e-10)
            if result_bfgs is not None and objective(result_bfgs) < best_c5:
                best_c5 = objective(result_bfgs)
                best_h = result_bfgs
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed < budget_s - 30:
        try:
            result_bfgs = optimize_guess(best_h, 'L-BFGS-B', maxiter=50, ftol=1e-10)
            if result_bfgs is not None and objective(result_bfgs) < best_c5:
                best_c5 = objective(result_bfgs)
                best_h = result_bfgs
        except:
            pass
    
    return best_h, best_c5, n_points
