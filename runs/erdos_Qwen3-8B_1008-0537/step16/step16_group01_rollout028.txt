import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def generate_uniform_guesses(num=10):
        return [np.random.uniform(0, 1, n_points) * (target_sum / np.sum(np.random.uniform(0, 1, n_points))) for _ in range(num)]

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)

    start_time = time.time()

    def optimize_with_de():
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=300,
            mutation=(0.6, 0.9),
            recombination=0.7,
            tol=1e-10,
            vectorized=True,
            polish=False
        )
        if result.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                nonlocal best_c5, best_h
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x

    optimize_with_de()

    for _ in range(3):
        guesses = generate_uniform_guesses()
        for guess in guesses:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                if objective(result.x) < best_c5:
                    best_c5 = objective(result.x)
                    best_h = result.x

    while time.time() - start_time < budget_s - 10:
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x

    return best_h, best_c5, n_points
