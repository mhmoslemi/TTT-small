import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guesses(num_guesses=15):
        guesses = [initial_guess.copy()]
        for _ in range(num_guesses - 1):
            h = np.zeros(n_points)
            num_ones = int(target_sum)
            idxs = np.argsort(np.random.rand(n_points))
            h[idxs[:num_ones]] = 1.0
            if (target_sum - num_ones) > 0:
                idx = np.random.choice(n_points, size=1, replace=False)
                h[idx] += 1.0
            guesses.append(h)
            # Add a symmetric guess
            sym_h = np.zeros(n_points)
            idxs = np.argsort(np.random.rand(n_points))
            sym_ones = int(target_sum)
            sym_h[idxs[:sym_ones]] = 1.0
            if (target_sum - sym_ones) > 0:
                idx = np.random.choice(n_points, size=1, replace=False)
                sym_h[idx] += 1.0
            sym_h = np.minimum(1.0, np.maximum(0.0, (sym_h + sym_h[::-1]) / 2))
            guesses.append(sym_h)
        return guesses
    
    initial_guesses = generate_guesses()
    
    def optimize_guess(guess, maxiter=400, ftol=1e-9):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=120,
            maxiter=400,
            mutation=(0.7, 1.0),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 60:
        try:
            result = optimize.minimize(
