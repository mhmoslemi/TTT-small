import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_guesses(num_guesses=10):
        guesses = [initial_guess.copy()]
        m = int(n_points / 2)
        # Evenly spaced guess
        indices = np.linspace(0, n_points-1, m, endpoint=False).astype(int)
        h_even = np.zeros(n_points)
        h_even[indices] = 1.0
        guesses.append(h_even)
        # Alternating guess (even n_points only)
        if n_points % 2 == 0:
            h_alt = np.zeros(n_points)
            h_alt[::2] = 1.0
            if sum(h_alt) == m:
                guesses.append(h_alt)
        for _ in range(num_guesses - len(guesses)):
            h = np.zeros(n_points)
            idxs = np.argsort(np.random.rand(n_points))
            h[idxs[:m]] = 1.0
            idx = np.random.choice(n_points, size=1, replace=False)
            h[idx] += 1.0 if m > np.sum(h) else 0
            guesses.append(h)
        return guesses
    
    initial_guesses = generate_guesses()
    
    def optimize_guess(guess, maxiter=200, ftol=1e-9):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result

        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=120,
            mutation=(0.5, 0.95),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,

            result = optimize.minimize(
                fun=objective,
                x0=best_h,

                bounds=bounds,

                best_h = result.x

            break
    
    return best_h, best_c5, n_points
