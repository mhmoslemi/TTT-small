import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def generate_guesses(num_guesses=15):
        guesses = [initial_guess.copy()]
        for _ in range(num_guesses - 1):
            rand = np.random.rand(n_points)
            idxs = np.argsort(rand)
            h = initial_guess.copy()
            h[idxs[:int(target_sum)]] = 1.0
            if target_sum - int(target_sum) > 0:
                h[idxs[int(target_sum)]] += target_sum - int(target_sum)
            guesses.append(h)
        return guesses
    
    initial_guesses = generate_guesses()
    
    def optimize_guess(guess, maxiter=400, ftol=1e-9):
        result = optimize.minimize(

            method='SLSQP',

            constraints=[{'type': 'eq', 'fun': constraint_sum}],
