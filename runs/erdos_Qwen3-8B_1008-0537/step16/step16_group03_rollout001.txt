import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess, maxiter=150, ftol=1e-8):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
        )
        return result.x if result.success else None
    
    # Generate 3 perturbed initial guesses
    perturbations = [np.random.normal(0, 0.05, n_points) for _ in range(3)]
    for pert in perturbations:

        guess = guess * (target_sum - np.sum(guess)) / (target_sum - np.sum(guess)) if np.sum(guess) != target_sum else guess
        guess[np.abs(guess - 0.5) < 1e-6] = 0.5  # Avoid numerical instability
        
        # Run differential evolution
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=60,
                maxiter=250,
                mutation=(0.5, 0.9),
                recombination=0.8,
                tol=1e-9,
                vectorized=True,
                x0=guess
            )
            if result_de.success:
                result_cobyla = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='COBYLA',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 150, 'ftol': 1e-9, 'disp': False}
                )
                if result_cobyla.success and objective(result_cobyla.x) < best_c5:
                    best_c5 = objective(result_cobyla.x)
