import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def generate_structured_guess(num_segments=4):
        intervals = np.linspace(0, 2, num_segments + 1)
        h = np.zeros(n_points)
        for i in range(num_segments):
            start = intervals[i]
            end = intervals[i+1]
            start_idx = int(start / dx)
            end_idx = int(end / dx)
            h[start_idx:end_idx] = 1.0
        h *= target_sum / np.sum(h)
        return h

    def generate_random_guesses(num_guesses=20):
        guesses = []
        for _ in range(num_guesses):
            h = np.random.uniform(0, 1, n_points)
            h = h * (target_sum / np.sum(h))
            guesses.append(h)
        return guesses

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = np.array(initial_h_values, dtype=float) * (target_sum / np.sum(initial_h_values))
    best_c5 = objective(best_h)
    start_time = time.time()

    initial_guesses = [best_h] + generate_structured_guess(4) + generate_random_guesses(10)

    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x

    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=40,
            maxiter=150,
            mutation=(0.6, 0.9),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success and objective(result_de.x) < best_c5:
            best_h = result_de.x
            best_c5 = objective(best_h)
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass

    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
