import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    def generate_initial_guess():
        m_ones = n_points // 2
        positions = np.linspace(0, 2, n_points)
        h_vals = np.zeros(n_points)
        spacing = 2 / m_ones
        for i in range(m_ones):
            pos = i * spacing + spacing / 2
            idx = np.argmin(np.abs(positions - pos))
            h_vals[idx] = 1
        current_sum = np.sum(h_vals)
        if current_sum < target_sum:
            idx = np.where(h_vals == 0)[0][0]
            h_vals[idx] += target_sum - current_sum
        return h_vals

    initial_guess = generate_initial_guess()
    initial_guess *= target_sum / np.sum(initial_guess)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

            guesses.append(h)
        return guesses

    initial_guesses = generate_guesses(10)

    for guess in initial_guesses:
