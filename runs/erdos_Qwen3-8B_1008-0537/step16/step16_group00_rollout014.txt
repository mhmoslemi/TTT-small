import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None

            alt_pattern[i] = 0.0
    # Adjust for target_sum
    current_sum = round(np.sum(alt_pattern))
    if current_sum < target_sum:
        alt_pattern[-1] += (target_sum - current_sum)
    elif current_sum > target_sum:
        alt_pattern[-1] -= (current_sum - target_sum)
    structured_guesses.append(alt_pattern)
    
    # Random distribution
