import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_structured_guess(period):
        guess = np.zeros(n_points)
        for i in range(n_points):
            if (i // period) % 2 == 0:
                guess[i] = 1.0
        guess[-1] = (target_sum - np.sum(guess)) * 2
        return guess
    
    initial_guesses = [best_h]
    
    for period in [2, 3, 4, 5]:
        structured = generate_structured_guess(period)
        initial_guesses.append(structured)
    
    for _ in range(5):
        rand_h = np.zeros(n_points)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        rand_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            rand_h[idx] += remaining
        initial_guesses.append(rand_h)
    
    def optimize_guess(guess):
        try:
            res = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12}
            )
            return res.x if res.success else None
        except:
            return None
    
    for guess in initial_guesses:
        h = optimize_guess(guess)
        if h is not None and objective(h) < best_c5:
            best_c5 = objective(h)
            best_h = h
    
    try:
        res_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=300,
            mutation=(0.7, 1.2),
            recombination=0.85,
            tol=1e-10,
            vectorized=True
        )
        if res_de.success:
            h = optimize.minimize(
                fun=objective,

                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12}
            ).x
            if objective(h) < best_c5:
                best_c5 = objective(h)
                best_h = h
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
