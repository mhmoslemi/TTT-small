        positions = np.linspace(0, n_points-1, m)
        indices = np.round(positions).astype(int)
        for idx in indices:
            if 0 <= idx < n_points:
                h[idx] = 1.0
        sum_h = np.sum(h)
        remaining = m - sum_h
        if remaining > 0:
            idx = np.random.choice(n_points, size=remaining, replace=False)
            h[idx] += 1.0
        return h
    
    initial_guesses.append(generate_even_pattern(n_points))
    initial_guesses.append(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]))
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=200,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
