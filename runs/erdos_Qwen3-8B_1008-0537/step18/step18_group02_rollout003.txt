import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    # Create structured guesses
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h[i] = 1.0
    structured_h[-1] = 0.5
    
    structured_h4 = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            structured_h4[i] = 1.0
    structured_h4[-1] = 0.5
    
    structured_h5 = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            structured_h5[i] = 1.0
    structured_h5[-1] = 0.5
    
    first_half_guess = np.zeros(n_points)
    num_ones = int(n_points / 2)
    first_half_guess[:num_ones] = 1.0
    remaining = target_sum - num_ones
    if remaining > 0:
        first_half_guess[num_ones - 1] += remaining
    
    # Random guess
    random_guess = np.zeros(n_points)
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    random_guess[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        random_guess[idx] += remaining
    
    initial_guesses = [
        initial_guess,
        structured_h,

        structured_h5,
        first_half_guess,

    ]

            best_h = result

            fun=objective,
            bounds=bounds,
            popsize=50,

            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,

        )

            result_slsqp = optimize.minimize(
                fun=objective,
