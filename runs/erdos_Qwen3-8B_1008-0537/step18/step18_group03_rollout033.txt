import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def create_evenly_spaced_guess(n_points):
        n_ones = int(n_points / 2)
        remaining = n_points / 2 - n_ones

        h = np.zeros(n_points)
        for x in positions:
            idx = int(x * n_points / 2)
            if idx < 0:
                idx = 0
            elif idx >= n_points:
                idx = n_points - 1
            h[idx] += 1.0
        if remaining > 0:

            h[idx] += remaining
        return h
    
    structured_h = create_evenly_spaced_guess(n_points)
    
    initial_guesses = [initial_guess, structured_h]
    
    def optimize_guess(guess, method='SLSQP'):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for _ in range(2):
