import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_evenly_spaced_guess():
        num_ones = int(target_sum) + 1
        idx = np.argsort(np.random.rand(n_points))
        selected = idx[:num_ones]
        h = np.zeros(n_points)
        h[selected] = 1.0
        return h
    
    def generate_clustered_guess():
        num_ones = int(target_sum)
        center = int(n_points / 4)
        h = np.zeros(n_points)
        h[center-5:center+6] = 1.0
        idx = np.setdiff1d(np.arange(n_points), np.arange(center-5, center+6))
        h[idx[:(num_ones - 22)]] = 1.0
        return h
    
    initial_guesses = [initial_guess]
    for _ in range(2):
        idx = np.random.choice(n_points, size=int(target_sum), replace=False)
        rand_h = np.zeros(n_points)
        rand_h[idx] = 1.0
        initial_guesses.append(rand_h)
    initial_guesses.append(generate_evenly_spaced_guess())
    initial_guesses.append(generate_clustered_guess())
    
    for guess in initial_guesses:

        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)

        result_de = optimize.differential_evolution(

            maxiter=150,
            mutation=(0.8, 1.2),

            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,

                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
