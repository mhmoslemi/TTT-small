                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_evenly_spaced_guess():
        m = int(target_sum)
        remaining = target_sum - m
        interval = 2.0 / m
        positions = []
        for i in range(m):
            pos = i * interval
            idx = int(pos * n_points / 2)
            if 0 <= idx < n_points:
                positions.append(idx)
        positions = list(set(positions))
        h = np.zeros(n_points)
        for idx in positions:
            if idx < n_points:
                h[idx] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            h[idx] += remaining
        return h
    
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0
    structured_h[-1] = 0.5
    initial_guesses = [initial_guess, structured_h]
    
    for _ in range(2):
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        rand_h = np.zeros(n_points)
        rand_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            rand_h[idx] += remaining
        initial_guesses.append(rand_h)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=50,
                maxiter=200,
                mutation=(0.6, 0.9),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                    best_c5 = objective(result_slsqp.x)
                    best_h = result_slsqp.x
        except:
            pass
    
    return best_h, best_c5, n_points
