    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    initial_guesses = [initial_guess]
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0
    structured_h[-1] = 0.5
    initial_guesses.append(structured_h)
    
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    rand_h = np.zeros(n_points)
    rand_h[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        rand_h[idx] += remaining
    initial_guesses.append(rand_h)
    
    m = int(target_sum) + 1
    positions = np.linspace(0, 2, m + 1)[:-1]
    indices = np.clip(np.round(positions / dx).astype(int), 0, n_points-1)
    spread_h = np.zeros(n_points)
    for i in range(m):
        if indices[i] < n_points:
            spread_h[indices[i]] = 1.0
    if np.sum(spread_h) > target_sum:
        spread_h[-1] -= abs(np.sum(spread_h) - target_sum)
    elif np.sum(spread_h) < target_sum:
        spread_h[-1] += abs(np.sum(spread_h) - target_sum)
    initial_guesses.append(spread_h)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=200,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
