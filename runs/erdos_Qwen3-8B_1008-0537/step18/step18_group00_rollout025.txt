import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    # Create structured and evenly spaced initial guesses
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0
    structured_h[-1] = 0.5
    
    # Evenly spaced pattern
    ideal_positions = np.linspace(0, 2.0, 23)
    grid_positions = np.linspace(0, 2.0, n_points)
    distances = np.abs(grid_positions[:, None] - ideal_positions[None, :])
    closest_indices = np.argmin(distances, axis=1)
    evenly_spaced_h = np.zeros(n_points)
    evenly_spaced_h[closest_indices] = 1.0
    
    initial_guesses = [initial_guess, structured_h, evenly_spaced_h]
    
    # Generate additional random and structured guesses
    for _ in range(3):
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        rand_h = np.zeros(n_points)
        rand_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            rand_h[idx] += remaining
        initial_guesses.append(rand_h)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=300,
            mutation=(0.8, 0.98),
            recombination=0.95,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',

                best_c5 = objective(result.x)
                best_h = result.x
        except:
