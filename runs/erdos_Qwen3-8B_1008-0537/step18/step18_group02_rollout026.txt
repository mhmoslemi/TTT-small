        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def generate_random_initial():
        m = int(target_sum)

        positions = np.random.choice(n_points, size=m, replace=False)
        h = np.zeros(n_points)
