        positions = np.linspace(0, 2, m + 1)[:-1]
        for x in positions:
            i = np.argmin(np.abs(x - np.linspace(0, 2, n_points)))
            h[i] = 1
        current_sum = np.sum(h)
        residual = target_sum - current_sum
        if residual > 1e-6:
            h[-1] += residual
        elif residual < -1e-6:
            h[0] -= residual
        return h
    
    structured_guesses = [
        np.zeros(n_points),
        np.zeros(n_points),
        np.zeros(n_points),
        np.zeros(n_points)
    ]
    
    for i in range(4):
        if i == 0:
            m = int(target_sum)
            structured_guesses[i] = generate_evenly_spaced(m)
        elif i == 1:
            m = int(target_sum * 0.6)
            structured_guesses[i] = generate_evenly_spaced(m)
            res = target_sum - np.sum(structured_guesses[i])
            structured_guesses[i][-1] += res
        elif i == 2:
            m = int(target_sum * 0.8)
            structured_guesses[i] = generate_evenly_spaced(m)
            res = target_sum - np.sum(structured_guesses[i])
            structured_guesses[i][-1] += res
        else:
            m = int(target_sum * 0.4)
            structured_guesses[i] = generate_evenly_spaced(m)
            res = target_sum - np.sum(structured_guesses[i])
            structured_guesses[i][-1] += res
    
    initial_guesses = [initial_guess] + structured_guesses
    
    for guess in initial_guesses:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            pass
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=200,
            maxiter=250,
            mutation=(0.7, 0.95),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
