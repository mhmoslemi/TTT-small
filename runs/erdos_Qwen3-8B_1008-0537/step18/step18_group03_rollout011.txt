        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    structured_h = np.zeros(n_points)

            structured_h[i] = 1.0

        else:
            structured_h[i] = 0.0

    positions_continuous = np.linspace(0, 2, m + 1)
    positions_discrete = np.arange(n_points) * dx
    indices = np.searchsorted(positions_discrete, positions_continuous)
    structured_h2 = np.zeros(n_points)
    structured_h2[indices[:m]] = 1.0
    sum_ones = np.sum(structured_h2)
    diff = target_sum - sum_ones
    if diff > 0:
        structured_h2[np.random.choice(n_points, size=int(diff), replace=False)] += 1.0
