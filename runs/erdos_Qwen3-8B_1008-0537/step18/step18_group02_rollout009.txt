import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def generate_periodic_initial(period):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if (i % period) < 1:
                pattern[i] = 1.0
        while np.sum(pattern) < target_sum - 1e-6:
            idx = np.where(pattern == 0)[0][0]
            pattern[idx] = 1.0
        while np.sum(pattern) > target_sum + 1e-6:
            idx = np.where(pattern == 1)[0][0]
            pattern[idx] = 0.0
        return pattern
    
    initial_guesses = [initial_guess]
    for period in [3, 5, 7]:
        if period <= n_points:
            initial_guesses.append(generate_periodic_initial(period))

        rand_h = np.zeros(n_points)
        rand_h[indices] = 1.0
        initial_guesses.append(rand_h)
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,

                bounds=bounds,

        except:

    try:

        )
        if result_de.success:
            result_slsqp = optimize.minimize(
