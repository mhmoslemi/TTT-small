import scipy.optimize as optimize

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 400, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    structured_h = np.zeros(n_points)

        h = np.zeros(n_points)
        count = 0
        for i in range(n_points):
            if i % period == 0:
                h[i] = 1.0
                count += 1
            elif i == n_points - 1:
                h[i] = 0.5
        return h

        initial_guesses.append(h)

        rand_h[indices] = 1.0

            rand_h[idx] += remaining
