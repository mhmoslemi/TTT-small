import scipy.optimize as optimize
import time

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()

        x = np.linspace(0, 1, n_points)
        if wave_type == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
            return np.clip((wave + 1) * duty / 2, 0, 1)
        elif wave_type == 'saw':
            return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif wave_type == 'tri':
            return 1 - np.abs(2 * freq * x - 1) * duty
    
    guesses = [h]

    def make_symmetric(h_in):
        h_out = np.zeros_like(h_in)
        for i in range(n_points):
            j = int(2 * (n_points // 2) - i)
            if j < 0 or j >= n_points:
                continue
            h_out[i] = h_in[j]
        return h_out
    
    guesses.append(make_symmetric(h))
    
    # More focused waveforms

        guesses.append(gen_wave(3, duty, 'tri'))

    for _ in range(3):
        guesses.append(np.random.uniform(0, 1, n_points))
    
    def slsqp_guess(g):
