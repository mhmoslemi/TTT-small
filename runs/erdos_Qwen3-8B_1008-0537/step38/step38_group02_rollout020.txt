import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_wave(freq, duty, wave_type):
        x = np.linspace(0, 1, n_points)
        if wave_type == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
            return np.clip((wave + 1) * duty / 2, 0, 1)
        elif wave_type == 'saw':
            return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif wave_type == 'tri':
            return 1 - np.abs(2 * freq * x - 1) * duty
    
    guesses = [h]
    for freq in [3, 4, 5, 6, 7, 8]:
        guesses.append(gen_wave(freq, 0.5, 'pulse'))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_wave(3, duty, 'saw'))
        guesses.append(gen_wave(3, duty, 'tri'))
    for freq in [2, 3, 4, 5]:
        guesses.append(gen_wave(freq, 0.5, 'tri'))
    for _ in range(10):
        guesses.append(np.random.uniform(0, 1, n_points))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-10})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:

        res = optimize.differential_evolution(obj, bounds, popsize=80, maxiter=50,
