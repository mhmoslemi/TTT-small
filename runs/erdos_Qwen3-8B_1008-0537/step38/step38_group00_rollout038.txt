        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

    best_c5 = objective(best_h)

        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points) / 2), 0, 1)
    
    def generate_triangle(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.abs(2 * freq * t - 1)
        return np.clip(1 - wave, 0, 1)
    
    def generate_custom(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_binary_uniform():
        indices = np.arange(n_points)
        selected = np.random.choice(indices, size=n_points // 2, replace=False)
        h = np.zeros(n_points, dtype=float)
        h[selected] = 1.0
        return h
    
    guesses = [

        if res.success and objective(res.x) < best_c5:

            best_h = res.x
