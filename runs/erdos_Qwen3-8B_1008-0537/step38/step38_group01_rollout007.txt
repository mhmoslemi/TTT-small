import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_varied_pulses(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * (freq * x + phase))
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_circular_pulse(period, offset):
        x = np.linspace(0, 1, n_points)
        return np.clip(np.sin(2 * np.pi * (x * period - offset)) * 0.5 + 0.5, 0, 1)

    initial_guesses = [h]
    for freq in [3, 4, 5, 6, 7, 8, 9, 10]:
        initial_guesses.append(gen_varied_pulses(freq, 0.5, np.random.rand()))
    for phase in np.linspace(0, 1, 10):
        initial_guesses.append(gen_circular_pulse(4, phase))
    initial_guesses.append(np.random.rand(n_points))

    def slsqp_guess(g):
        try:
            res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 150, 'ftol': 1e-10, 'disp': False})
            return res.x if res.success else g
        except:
            return g

    for g in initial_guesses:
        if np.allclose(np.sum(g), target_sum):
            h_opt = slsqp_guess(g)
            c = obj(h_opt)
            if c < best_c5:
                best_h, best_c5 = h_opt, c

    for _ in range(2):
        if time.time() - start_time > budget_s - 10:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=20,
                                             mutation=(0.6, 0.9), recombination=0.9, tol=1e-9, disp=False)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
