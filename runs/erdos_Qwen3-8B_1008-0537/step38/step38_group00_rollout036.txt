import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary():
        n = len(initial_h_values)
        binary = np.zeros(n)
        binary[:int(n/2)] = 1.0
        binary[int(n/2)] += 0.5
        return binary
    
    def generate_sine(freq):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points) / 2), 0, 1)
    
    guesses = [generate_sine(3), generate_sine(4), generate_sine(5), 
               generate_binary(), np.random.rand(n_points), h]
    
    def refine_with_cobyla(h_init):
        res = optimize.minimize(objective, h_init, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'disp': False})
        return res.x if res.success else h_init
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'disp': False})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 60:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=60,
                                             mutation=(0.7, 0.95), recombination=0.85, tol=1e-9)
        if res.success:
            best_h = refine_with_cobyla(res.x)
            if objective(best_h) < best_c5:
                best_c5 = objective(best_h)
    
    for _ in range(4):
        if time.time() - start_time >= budget_s - 20:
            break
        best_h = refine_with_cobyla(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
