import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def create_binary_pattern(n, duty=0.5, pulse_width=0.2):
        x = np.linspace(0, 2, n)
        idx = np.argsort(np.abs(x - 1))
        n_pulses = int(n * duty)
        selected = idx[:n_pulses]
        h_binary = np.zeros(n)
        for i in selected:
            h_binary[i] = 1
        return h_binary

    class InitialGuesses:
        def __init__(self, base_h, n_points):
            self.base_h = base_h
            self.n_points = n_points
            self.guesses = [base_h.copy()]
            self._generate_waveforms()
            self._generate_binary_patterns()
            self._generate_random()
        
        def _generate_waveforms(self):
            for freq in [4, 5, 6, 7, 8, 10]:
                for duty in [0.2, 0.3, 0.4, 0.5]:
                    x = np.linspace(0, 2, self.n_points)
                    wave = np.sin(2 * np.pi * freq * x)
                    wave = np.clip((wave + 1) * duty / 2, 0, 1)
                    self.guesses.append(wave)
                    wave = np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
                    self.guesses.append(wave)
        
        def _generate_binary_patterns(self):
            for duty in [0.4, 0.5, 0.6]:
                self.guesses.append(create_binary_pattern(self.n_points, duty=duty))
        
        def _generate_random(self):
            for _ in range(10):
                rand_h = np.random.uniform(0, 1, self.n_points)
                self.guesses.append(rand_h)
    
    ig = InitialGuesses(h, n_points)
    guesses = ig.guesses

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-6})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
