import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_pulsed():
        h = np.zeros(n_points)
        h[::2] = 1.0
        return h
    
    def generate_sine(freq):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points) / 2), 0, 1)
    
    def generate_triangle(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.abs(2 * freq * t - 1)
        return np.clip(1 - wave, 0, 1)
    
    def generate_pulses(num_pulses):
        h = np.zeros(n_points)
        if num_pulses == 0:
            return h
        idxs = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
        h[idxs] = 1.0
        return h
    
    guesses = [
        generate_pulsed(), 
        generate_sine(3), 
        generate_triangle(3), 
        np.random.rand(n_points), 
        h, 
        generate_pulses(n_points // 4), 
        generate_pulses(n_points // 2), 
        generate_pulses(n_points // 3)
    ]
    
    for guess in guesses:
        if np.sum(guess) < 0.5 * n_points - 1e-6:
            guess = guess.copy()
            guess[guess < 1e-6] = 0.0
            guess[guess > 1 - 1e-6] = 1.0
            guess = guess + (np.sum(guess) - 0.5 * n_points) / (n_points / 2)
        res = optimize.minimize(objective, guess, bounds=bounds, method='SLSQP',
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=40,
                                             mutation=(0.7, 0.95), recombination=0.85, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 60, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,

        return res.x if res.success else h_init
    
    for _ in range(3):

        best_h = refine_with_slsqp(best_h)
