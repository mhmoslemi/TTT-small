import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary():
        h_bin = np.random.rand(n_points) > 0.5
        h_bin = h_bin.astype(float)
        scale = target_sum / np.sum(h_bin)
        h_bin *= scale
        return h_bin
    
    def generate_sine(freq):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * t), 0, 1)
    
    def generate_triangle(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.abs(2 * freq * t - 1)
        return np.clip(1 - wave, 0, 1)
    
    def generate_custom(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    guesses = [generate_binary(), generate_sine(3), generate_triangle(3), generate_custom(2), np.random.rand(n_points), h]
    
    for guess in guesses:
        try:
            res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'maxiter': 400, 'ftol': 1e-12, 'eps': 1e-10})
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        except:
            pass
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        try:
            res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=60,
                                                  mutation=(0.8, 0.95), recombination=0.85, tol=1e-9)
            if res.success:
                res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                             constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                             options={'maxiter': 60, 'ftol': 1e-12, 'eps': 1e-10})
                if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                    best_c5 = objective(res_slsqp.x)
                    best_h = res_slsqp.x
        except:
            pass
    
    def refine_with_slsqp(h_init):

    for _ in range(3):
