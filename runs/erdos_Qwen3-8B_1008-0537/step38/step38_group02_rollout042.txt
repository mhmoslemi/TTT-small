        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=20,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
