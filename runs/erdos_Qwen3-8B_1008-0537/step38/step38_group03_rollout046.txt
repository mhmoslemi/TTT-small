        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty
    
    def generate_triangular(freq, amp):
        t = np.linspace(0, 1, n_points)
        wave = amp * (2 * t - 1) * np.abs(1 - 2 * np.abs(2 * freq * t - 1))
        return np.clip(wave, 0, 1)
    
    def generate_sawtooth(freq, duty):
        t = np.linspace(0, 1, n_points)
        wave = 2 * duty * (1 - np.abs(2 * freq * t - 1))
        return np.clip(wave, 0, 1)
    
    def generate_cosine(freq):
        return np.clip(0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n_points)), 0, 1)
    
    def generate_intervals(n_points, m):
        length = 1.0 / m
        h = np.zeros(n_points)
        current = 0.0
        for _ in range(m):
            start = current
            end = current + length
            start_idx = int(np.round(start / dx))
            end_idx = int(np.round(end / dx))
            if start_idx < n_points and end_idx < n_points:
                h[start_idx:end_idx] = 1.0
            current += length + 0.1
        return h
    
    guesses = [generate_periodic(p) for p in [3, 4, 5, 7, 9]]
    guesses += [generate_binary(f, d) for f, d in [(3, 0.2), (4, 0.4), (5, 0.3), (6, 0.2), (7, 0.3)]]
    guesses += [generate_triangular(f, a) for f, a in [(3, 0.6), (4, 0.5), (5, 0.4), (6, 0.3)]]
    guesses += [generate_sawtooth(f, d) for f, d in [(3, 0.3), (4, 0.25), (5, 0.2), (6, 0.15)]]
    guesses += [generate_cosine(f) for f in [3, 4, 5, 6]]
    guesses += [generate_intervals(n_points, m) for m in [2, 3, 4]]
    guesses.append(np.random.rand(n_points))
    guesses.append(h)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=100, maxiter=100,
                                             mutation=(0.7, 0.95), recombination=0.85, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 60, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
        return res.x if res.success else h_init
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_slsqp(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
