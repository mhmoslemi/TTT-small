    start_time = time.time()
    
    def generate_even_spaced_ones(m):
        positions = np.linspace(0, 2, m, endpoint=False)
        indices = np.round((positions / dx)).astype(int)
        indices = np.unique(indices)
        ones_mask = np.zeros(n_points)
        ones_mask[indices] = 1
        ones_mask = np.minimum(ones_mask, 1)
        ones_mask = np.maximum(ones_mask, 0)
        excess = np.sum(ones_mask) - m
        if excess != 0:
            idx = np.argsort(np.abs(ones_mask - 0.5))
            ones_mask[idx[:abs(excess)]] += -np.sign(excess)
        return ones_mask
    
    def generate_sawtooth(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 1 - np.abs(2 * freq * x - 1) * duty
        return np.clip(wave, 0, 1)
    
    def generate_pulsewave(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        pulse = (wave + 1) * duty / 2
        return np.clip(pulse, 0, 1)
    
    # Initial guess: even spaced ones
    m = int(np.round(n_points / 2))
    even_guess = generate_even_spaced_ones(m)
    guesses = [even_guess]
    for freq in [3, 4, 5, 6, 7, 8]:
        guesses.append(generate_pulsewave(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(generate_sawtooth(3, duty))
    for _ in range(10):
        guesses.append(np.random.uniform(0, 1, n_points))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=50,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(4):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
