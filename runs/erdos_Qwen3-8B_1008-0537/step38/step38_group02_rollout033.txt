import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

        if wave_type == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
            return np.clip((wave + 1) * duty / 2, 0, 1)
        elif wave_type == 'saw':
            return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif wave_type == 'tri':
            return 1 - np.abs(2 * freq * x - 1) * duty
    
    guesses = [h]

        guesses.append(gen_wave(3, duty, 'tri'))
    for duty in [0.3, 0.6]:
        guesses.append(gen_wave(4, duty, 'saw'))
        guesses.append(gen_wave(4, duty, 'tri'))
    guesses += [np.random.uniform(0, 1, n_points) for _ in range(10)]
    
    def slsqp_guess(g):
