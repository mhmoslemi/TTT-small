import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_sine(freq):
        x = np.linspace(0, 2, n_points)
        return 0.5 + 0.5 * np.sin(2 * np.pi * freq * x)

    def gen_pulse(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    guesses = [h]
    for freq in [2, 3, 4, 5]:
        guesses.append(gen_sine(freq))
    for freq, duty in [(3, 0.25), (3, 0.5), (4, 0.5)]:
        guesses.append(gen_pulse(freq, duty))
    for _ in range(5):
        guesses.append(np.random.uniform(0, 1, n_points))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 400, 'ftol': 1e-12})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=80,
                                             mutation=(0.6, 0.98), recombination=0.7, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
