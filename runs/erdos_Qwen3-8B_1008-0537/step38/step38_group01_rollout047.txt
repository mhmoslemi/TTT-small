import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_random():
        while True:
            r = np.random.rand(n_points)
            if np.all(r >= 0) and np.all(r <= 1):
                return r
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    def gen_binary(frac_ones):
        return np.random.choice([0, 1], size=n_points, p=[1-frac_ones, frac_ones])
    
    def gen_binary_sparse(frac_ones):
        return np.zeros(n_points) + (np.random.rand(n_points) < frac_ones)
    
    initial_guesses = [h]
    for freq in [2,3,4,5,6,7,8,9,10]:
        initial_guesses.append(gen_pulses(freq, 0.5))
    for freq in [2,3,4,5,6,7,8,9,10]:
        initial_guesses.append(gen_saw(freq, 0.5))
    for freq in [2,3,4,5,6,7,8,9,10]:
        initial_guesses.append(gen_triangle(freq, 0.5))
    for frac in [0.5, 0.4, 0.3, 0.25, 0.2]:
        initial_guesses.append(gen_binary_sparse(frac))
    initial_guesses.append(gen_binary(0.5))
    initial_guesses.append(gen_random())
    
    def slsqp_guess(g):

                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 500, 'ftol': 1e-12})
        return res.x if res.success else g

        c = obj(h_opt)
        if c < best_c5:

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
