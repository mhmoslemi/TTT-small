import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(freq, wave_type):
        t = np.linspace(0, 1, n_points)
        if wave_type == 'sine':
            wave = 0.5 + 0.5 * np.sin(2 * np.pi * freq * t)
        elif wave_type == 'triangle':
            wave = np.abs(2 * freq * t - 1)
        elif wave_type == 'custom':
            wave = 0.5 + 0.5 * np.sin(2 * np.pi * freq * t)
        return np.clip(wave, 0, 1)
    
    guesses = [
        generate_wave(1, 'sine'), generate_wave(2, 'sine'), generate_wave(3, 'sine'), generate_wave(4, 'sine'),
        generate_wave(1, 'triangle'), generate_wave(2, 'triangle'), generate_wave(3, 'triangle'), generate_wave(4, 'triangle'),
        generate_wave(1, 'custom'), generate_wave(2, 'custom'), generate_wave(3, 'custom'), generate_wave(4, 'custom'),
        np.random.rand(n_points), h
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=80,
                                             mutation=(0.7, 0.95), recombination=0.85, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
        return res.x if res.success else h_init
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_slsqp(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
