        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)

        return np.random.choice([0, 1], size=n_points, p=[1-frac_ones, frac_ones])
    
    def generate_periodic(freq):
        return np.abs(np.sin(np.linspace(0, np.pi, n_points)) * freq) / freq
