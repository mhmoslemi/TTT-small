import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * t), 0, 1)
    
    def generate_triangle(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.abs(2 * freq * t - 1)
        return np.clip(1 - wave, 0, 1)
    
    def generate_sawtooth(freq):
        t = np.linspace(0, 1, n_points)
        wave = t * freq
        return np.clip(1 - np.sin(wave), 0, 1)
    
    def generate_custom(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * wave, 0, 1)
    
    def generate_random():
        return np.random.rand(n_points)
    
    guesses = [generate_sine(3), generate_triangle(3), generate_sawtooth(1), 
               generate_custom(2), generate_random(), h]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=50,
                                             mutation=(0.8, 1.0), recombination=0.9, tol=1e-9,
                                             strategy='best1bin')
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 80, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
