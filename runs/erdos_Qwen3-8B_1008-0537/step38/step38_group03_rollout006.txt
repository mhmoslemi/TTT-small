import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    target_sum = n_points / 2

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)

    def generate_binary(freq, duty):
        return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points)) > 0) * duty, 0, 1)

    def generate_triangle(freq, amp):
        t = (np.arange(n_points) / n_points)
        wave = amp * (1 - 2 * np.abs(2 * freq * t - 1))
        return np.clip(wave, 0, 1)

    def generate_sawtooth(f, duty):
        t = (np.arange(n_points) / n_points)
        wave = 2 * duty * (1 - np.abs(2 * f * t - 1))
        return np.clip(wave, 0, 1)

    def generate_random():
        return np.random.rand(n_points)

    guesses = [h]
    for p in [3, 4, 5, 6, 7, 8]:
        guesses.append(generate_periodic(p))
    for f, d in [(3, 0.25), (4, 0.3), (5, 0.2), (6, 0.1)]:
        guesses.append(generate_binary(f, d))

        guesses.append(generate_triangle(f, a))
    for f, d in [(3, 0.4), (4, 0.35), (5, 0.3)]:
        guesses.append(generate_sawtooth(f, d))
    guesses.append(generate_random())

    from scipy.optimize import differential_evolution

    for guess in guesses:
        if time.time() - start_time >= budget_s - 10:

        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)

    if time.time() - start_time < budget_s - 20:
        res_de = differential_evolution(objective, bounds, popsize=20, maxiter=50,

            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,

            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
