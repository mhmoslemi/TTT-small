def run(seed=42, budget_s=100, **kwargs):

    def objective(h_vals):

        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)

    def generate_sine(freq):
        t = np.linspace(0, 1, n_points)
        wave = 0.5 + 0.5 * np.sin(2 * np.pi * freq * t)
        return np.clip(wave, 0, 1)
    
    def generate_triangle(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.abs(2 * freq * t - 1)
        return np.clip(1 - wave, 0, 1)
    
    def generate_square():
        t = np.linspace(0, 2, n_points)
        wave = np.zeros(n_points)
        wave[:n_points // 2] = 1.0
        return np.clip(wave, 0, 1)
    
    def generate_random_perturbation(h_base):

        return h_perturbed * scale_factor
    
    guesses = [generate_sine(3), generate_triangle(3), generate_square(), generate_random_perturbation(h), h]
    
    for guess in guesses:
