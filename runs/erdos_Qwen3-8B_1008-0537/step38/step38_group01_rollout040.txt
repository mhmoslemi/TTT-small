        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_even_spaced(m):
        x = np.linspace(0, 2, n_points)
        intervals = []
        for i in range(m):
            start = i * (2.0 / m)
            end = start + 1.0 / m
            intervals.append((start, end))
        h = np.zeros(n_points)
        for i in range(n_points):
            x_val = x[i]
            for (s, e) in intervals:
                if s <= x_val <= e:
                    h[i] = 1.0
                    break
        return h
    
    def gen_random():
        while True:
            r = np.random.rand(n_points)
            if np.all(r >= 0) and np.all(r <= 1):
                return r
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    initial_guesses = [h]
    for freq in [3, 4, 5, 6, 7, 8, 9, 10]:
        initial_guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        initial_guesses.append(gen_saw(3, duty))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        initial_guesses.append(gen_triangle(3, duty))
    for m in [3, 5, 7, 10]:
        initial_guesses.append(gen_even_spaced(m))
    initial_guesses.append(gen_random())
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-10})
        return res.x if res.success else g
    
    for g in initial_guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=150, maxiter=80,
                                             mutation=(0.3, 0.7), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 400, 'ftol': 1e-12})
        return res.x if res.success else r
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
