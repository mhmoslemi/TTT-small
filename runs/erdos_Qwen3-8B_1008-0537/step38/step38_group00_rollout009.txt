import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(freq):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points) / 2), 0, 1)
    
    def generate_triangle(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.abs(2 * freq * t - 1)
        return np.clip(1 - wave, 0, 1)
    
    def generate_square(freq):
        t = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * t)
        return np.clip(0.5 + 0.5 * np.sign(wave), 0, 1)
    
    guesses = [generate_sine(3), generate_triangle(3), generate_square(2), 
               np.random.rand(n_points), h, generate_sine(4), generate_triangle(4)]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):

        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=50,
