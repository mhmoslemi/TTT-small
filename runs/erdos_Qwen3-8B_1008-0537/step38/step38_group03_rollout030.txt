import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_guesses():
        guesses = []
        # Pulse-based patterns
        for freq in [3, 4, 5, 6]:
            t = np.linspace(0, 1, n_points)
            wave = np.sin(2 * np.pi * freq * t)
            wave = np.clip(wave > 0, 0, 1) * 0.6
            guesses.append(wave)
        # Phase-shifted pulses
        for d in [0.3, 0.5, 0.7]:
            t = np.linspace(0, 1, n_points)
            wave = np.clip(np.sin(2 * np.pi * 3 * t + np.pi * d) > 0, 0, 1) * 0.8
            guesses.append(wave)
        # Randomized binary
        for _ in range(3):
            bin_h = np.random.rand(n_points)
            bin_h = np.clip((bin_h - 0.5) * 2, 0, 1)
            bin_h = bin_h / np.sum(bin_h) * target_sum
            guesses.append(bin_h)
        # Original guess
        guesses.append(h)
        return guesses

    guesses = generate_guesses()

    for guess in guesses:
        res = optimize.minimize(
            objective, guess, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 800, 'ftol': 1e-12, 'eps': 1e-9, 'disp': False}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(
            objective, bounds, popsize=20, maxiter=60, 
            mutation=(0.7, 0.95), recombination=0.85, tol=1e-9
        )
        if res.success:
            res_slsqp = optimize.minimize(
                objective, res.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 60, 'ftol': 1e-12, 'eps': 1e-9, 'disp': False}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x

    def refine(h_init):
        res = optimize.minimize(
            objective, h_init, method='SLSQP', bounds=bounds,
