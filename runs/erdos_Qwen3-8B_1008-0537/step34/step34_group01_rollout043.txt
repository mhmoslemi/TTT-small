    def obj(h_vals):

        return np.max(corr) * dx

    def cons(h_vals):

    bounds = [(0., 1.) for _ in range(n_points)]

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1), 0, 1)

    def gen_triangle(freq):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1)

    def gen_sine(freq, phase):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x + phase)
        return np.clip((wave + 1) / 2, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def gen_split(frac):

        h[:int(frac * n_points)] = 1
        return h
