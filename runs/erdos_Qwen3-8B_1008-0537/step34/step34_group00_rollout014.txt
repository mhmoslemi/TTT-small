import scipy.optimize as optimize
import time

    def obj(h_vals):

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_clock(freq):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) / 2, 0, 1)

    def gen_block(freq):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1), 0, 1)

    def gen_mixed(freq1, freq2):
        x = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq1 * x) + 0.5 * np.sin(2 * np.pi * freq2 * x), 0, 1)

    guesses = [h]
    for freq in range(2, 6):
        guesses.append(gen_clock(freq))
        guesses.append(gen_block(freq))
    for freq in range(2, 5):
        guesses.append(gen_mixed(freq, freq + 1))

    def de_guess():
