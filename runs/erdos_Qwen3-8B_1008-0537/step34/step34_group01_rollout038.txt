    def generate_structured(m):
        L = 1.0 / m
        S = 1.0 / (m - 1) if m > 1 else 0.0
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (L + S)
            end = start + L
            msk = (x >= start) & (x <= end)
            h[msk] = 1.0
        return h

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for m in [5, 6, 7]:
        guesses.append(generate_structured(m))
    for freq in range(3, 8):
        guesses.append(np.clip((np.sin(2 * np.pi * freq * np.linspace(0, 1, n_points)) + 1) * 0.5, 0, 1))
    for phase in np.linspace(0, 2*np.pi, 5):
        guesses.append(np.clip(1 - np.abs(2 * 3 * np.linspace(0, 1, n_points) - 1) * 0.6, 0, 1))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 400, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=50,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(2):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
