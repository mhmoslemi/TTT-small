import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(pos, width):
        return np.clip(np.maximum(0, np.minimum(1, 2 * (np.abs(np.linspace(0, 2, n_points) - pos) <= width) - 1)), 0, 1)

    def gen_sine(freq, phase):
        x = np.linspace(0, 2, n_points)
        return np.clip((np.sin(2 * np.pi * freq * x + phase) + 1) / 2, 0, 1)

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-6})
        return res.x if res.success else r

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=50,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    def make_guesses():
        guesses = [best_h.copy()]
        for _ in range(5):
            g = h.copy()
            g += np.random.normal(0, 0.1, n_points)
            g = np.clip(g, 0, 1)
            g = g * target_sum / np.sum(g)
            guesses.append(g)
        for f in [0.4, 0.5, 0.6]:
            g = gen_sine(f, np.random.uniform(0, 2 * np.pi))
            guesses.append(g)
        for pos in [0.2, 0.6, 1.0]:
            for width in [0.15, 0.25, 0.35]:
                guesses.append(gen_pulse(pos, width))
        return guesses

    guesses = make_guesses()

    for g in guesses:
        if time.time() - start_time > budget_s - 10:
            break
        h_opt = refine(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        h_opt = de_guess()
        h_opt = refine(h_opt)
        c = obj(h_opt)

            best_h, best_c5 = h_opt, c

            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:

    return best_h, best_c5, n_points
