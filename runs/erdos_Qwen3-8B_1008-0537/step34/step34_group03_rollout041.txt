import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_wave(freq, shape='saw'):
        x = np.linspace(0, 1, n_points)
        if shape == 'saw':
            wave = 1 - np.abs(2 * freq * x - 1)
        elif shape == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
        elif shape == 'tri':
            wave = 1 - np.abs(2 * freq * x - 1)
        return np.clip((wave + 1) * 0.5, 0, 1)

    def gen_rand():
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        return h

    guesses = [h]
    for freq in range(1, 10):
        guesses.append(gen_wave(freq, 'saw'))
        guesses.append(gen_wave(freq, 'pulse'))
        guesses.append(gen_wave(freq, 'tri'))
    for _ in range(5):
        guesses.append(gen_rand())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 250, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=60,
                                             mutation=(0.6, 0.95), recombination=0.7, tol=1e-9)
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
