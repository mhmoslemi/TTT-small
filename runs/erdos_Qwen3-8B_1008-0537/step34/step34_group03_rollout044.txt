import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        max_c5 = 0.0
        n = len(h_vals)
        for delta in range(-n + 1, n):
            if delta == 0:
                val = np.sum(h_vals * (1 - h_vals)) * dx
            elif delta > 0:
                val = np.sum(h_vals[:n - delta] * (1 - h_vals[delta:])) * dx
            else:
                delta_abs = -delta
                val = np.sum(h_vals[delta_abs:] * (1 - h_vals[:n - delta_abs])) * dx
            if val > max_c5:
                max_c5 = val
        return max_c5

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 9):
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_triangle(3, duty))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
