import scipy.optimize as optimize
import time

        valid_corr = corr[valid_idx]
        return np.max(valid_corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]

    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_square(freq, duty):

        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_random():

        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 10):

    for _ in range(10):
        guesses.append(gen_random())

    def slsqp_guess(g):
