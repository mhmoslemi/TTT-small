import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        return np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty, phase):
        t = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * t + phase)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty, phase):
        t = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * t - 1) * duty, 0, 1)

    def gen_triangle(freq, duty, phase):
        t = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * t - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def gen_symmetric(m):
        h = np.zeros(n_points)
        for i in range(m):
            h[i] = 1
            h[n_points - i - 1] = 1
        return h

    def gen_modulated(f, phase, a):
        t = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.cos(2 * np.pi * f * t + phase) * a, 0, 1)

    guesses = [h]
    for freq in range(2, 6):
        guesses.append(gen_pulses(freq, 0.5, np.pi / 4))
        guesses.append(gen_saw(freq, 0.5, np.pi / 4))
        guesses.append(gen_triangle(freq, 0.6, np.pi / 6))
    for phase in np.linspace(0, 2 * np.pi, 5):
        guesses.append(gen_modulated(3, phase, 0.7))
    for m in range(3, 6):
        guesses.append(gen_symmetric(m))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP',
                               bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(2):
        if time.time() - start_time > budget_s - 30:
            break
        res = optimize.differential_evolution(obj, bounds,
                                             popsize=20, maxiter=20,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-8)
        h_opt = res.x
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
