import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip(wave * duty + 1, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        saw = 1 - np.abs(2 * freq * x - 1) * duty
        return np.clip(saw, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        tri = 1 - np.abs(2 * freq * x - 1) * duty
        return np.clip(tri, 0, 1)

    def gen_random(h_ref):
        h = h_ref + np.random.rand(n_points) * 0.15 - 0.075
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    guesses = [h]
    for freq in range(3, 9):
        guesses.append(gen_pulse(freq, 0.5))
        guesses.append(gen_saw(freq, 0.25))
        guesses.append(gen_triangle(freq, 0.4))
    for phase in np.linspace(0, 2*np.pi, 5):
        guesses.append(gen_pulse(3, 0.6) * np.cos(phase + 0.5 * np.pi))
    for _ in range(3):
        guesses.append(gen_random(h))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 400, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
