import scipy.optimize as optimize
import time

    def obj(h_vals):

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty, shift=0.0):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * (x - shift))
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty, shift=0.0):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * (x - shift) - 1) * duty, 0, 1)

    def gen_triangle(freq, duty, shift=0.0):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * (x - shift) - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(2, 5):
        guesses.append(gen_pulses(freq, 0.5, shift=0.3))
    for duty in [0.2, 0.4, 0.6]:
        guesses.append(gen_saw(3, duty, shift=0.2))

    for _ in range(3):

    def slsqp_guess(g):
