import scipy.optimize as optimize
import time

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def refine(h_vals, num_iters=70):
        res = optimize.minimize(obj, h_vals, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': num_iters, 'ftol': 1e-10})
        return res.x if res.success else h_vals

    def evolve(popsize=100, maxiter=60):
        res = optimize.differential_evolution(obj, bounds, popsize=popsize,
                                             maxiter=maxiter, mutation=(0.5, 0.95),
                                             recombination=0.8, tol=1e-9)
        return res.x

    best_h = refine(best_h)
