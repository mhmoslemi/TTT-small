import scipy.optimize as optimize
import time

    def obj(h_vals):

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]

    best_c5 = obj(best_h)

        mid = n_points // 2
        h[:mid] = np.flip(h[:mid])
        h[mid+1:] = np.flip(h[mid+1:])
        return h

    def generate_sine_guess(freq, phase, duty):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * (x - phase))
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def generate_pulse_guess(pos, width):
        x = np.linspace(0, 2, n_points)
        return np.clip(np.clip(np.maximum(0, 1 - np.abs(x - pos)/width), 0, 1), 0, 1)

    def slsqp_guess(g):
