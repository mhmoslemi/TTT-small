import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_sinusoidal(freq, duty, phase):
        x = np.linspace(0, 1, n_points)
        wave = 0.5 * (1 + np.cos(2 * np.pi * freq * x + phase))
        return np.clip(wave * duty, 0, 1)

    def generate_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    initial_guesses = [h]
    for freq in [2, 3, 4]:
        initial_guesses.append(generate_sinusoidal(freq, 0.5, 0))
        initial_guesses.append(generate_sinusoidal(freq, 0.5, np.pi))
    for _ in range(5):
        initial_guesses.append(generate_random())

    def optimize_guess(g, maxiter=100, ftol=1e-10):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': maxiter, 'ftol': ftol})
        return res.x if res.success else g

    for g in initial_guesses:
        h_opt = optimize_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def differential_evolution(maxiter=100, popsize=50):
        res = optimize.differential_evolution(obj, bounds, popsize=popsize, maxiter=maxiter,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(3):
        if time.time() - start_time > budget_s - 30:

        h_opt = differential_evolution()

        c = obj(h_opt)
        if c < best_c5:

    def refine(iterations=10, maxiter=50):
