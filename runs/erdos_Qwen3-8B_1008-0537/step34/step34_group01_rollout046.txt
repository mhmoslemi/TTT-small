import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_interval_guess(n):
        m = 5
        w = 1.0 / m
        s = 1.0 / (m - 1)
        intervals = []
        start = 0.0
        for _ in range(m):
            end = start + w
            intervals.append((start, end))
            start = end + s
        x_grid = np.linspace(0, 2, n)
        h = np.zeros(n)
        for i, x in enumerate(x_grid):
            for start, end in intervals:
                if start <= x <= end:
                    h[i] = 1.0
                    break
        target_sum = n / 2
        h = h * target_sum / np.sum(h)
        return h

    guesses = [h]
    for freq in range(3, 8):
        guesses.append(gen_pulses(freq, 0.5))
        guesses.append(gen_saw(freq, 0.3))
        guesses.append(gen_triangle(freq, 0.4))
    for phase in np.linspace(0, 2*np.pi, 5):
        guesses.append(gen_sine(3, phase, 0.6))
    guesses.append(gen_interval_guess(n_points))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 400, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=50,
                                             mutation=(0.8, 1.0), recombination=0.9, tol=1e-9)
        return res.x

    for _ in range(2):

        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,

        best_h = refine(best_h)
        if obj(best_h) < best_c5:

    return best_h, best_c5, n_points
