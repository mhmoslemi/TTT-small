import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty, phase):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * (x - phase))
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_triangle(freq, duty, phase):
        x = np.linspace(0, 2, n_points)
        return 1 - np.abs(2 * freq * (x - phase) - 1) * duty
    
    def gen_saw(freq, duty, phase):
        x = np.linspace(0, 2, n_points)
        return np.clip(1 - np.abs(2 * freq * (x - phase) - 1) * duty, 0, 1)
    
    guesses = [h]
    freqs = [2, 3, 4, 5]
    duties = [0.4, 0.5, 0.6]
    phases = [0., np.pi/4, np.pi/2]
    for f in freqs:
        for d in duties:
            for p in phases:
                guesses.append(gen_pulses(f, d, p))
                guesses.append(gen_triangle(f, d, p))
                guesses.append(gen_saw(f, d, p))
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-9})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 20:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=50,

            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
