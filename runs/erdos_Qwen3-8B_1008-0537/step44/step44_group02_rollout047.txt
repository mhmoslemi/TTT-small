import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_structured_h(n_points, m):
        dx = 2.0 / n_points
        w = 1.0 / m
        s = 1.0 / (m - 1) if m > 1 else 0.0
        h = np.zeros(n_points)
        current_start = 0.0
        for i in range(m):
            current_end = current_start + w
            start_idx = int(current_start / dx)
            end_idx = int(current_end / dx)
            h[start_idx:end_idx + 1] = 1.0
            current_start = current_end + s
        return h

    guesses = [h]
    for m in [3, 4]:
        guesses.append(generate_structured_h(n_points, m))

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_square(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((np.sign(wave) + 1) * duty / 2, 0, 1)

    for freq in range(3, 11):
        guesses.append(gen_pulses(freq, 0.5))
        guesses.append(gen_square(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_square(3, duty))

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    for _ in range(5):
        guesses.append(gen_random())
