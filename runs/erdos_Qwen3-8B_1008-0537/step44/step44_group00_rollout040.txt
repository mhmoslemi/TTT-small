        positions = np.linspace(0, 2, num_pulses + 1)[1:-1]
        pulse_positions = np.random.choice(positions, size=num_pulses, replace=False)
        pulse_widths = np.random.uniform(2/(n_points * num_pulses), 2/(n_points * num_pulses), num_pulses)
        h_pulse = np.zeros(n_points)
        for i in range(num_pulses):
            left = np.searchsorted([0] + positions, pulse_positions[i] - pulse_widths[i]/2)
            right = np.searchsorted([0] + positions, pulse_positions[i] + pulse_widths[i]/2)
            h_pulse[left:right] = 1
        return h_pulse
    
    def gen_random_binary():
        return np.random.permutation(n_points) < n_points // 2
    
    guesses = [h, gen_pulse(3, 0.5), gen_pulse(2, 0.6), gen_random_binary()]
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-9})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 20:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=50,
                                             mutation=(0.5, 0.85), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
