import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_wave(freq, shape='triangle', duty=0.5):
        x = np.linspace(0, 1, n_points)
        if shape == 'triangle':
            wave = 1 - np.abs(2 * freq * x - 1) * duty
        elif shape == 'saw':
            wave = 1 - np.abs(2 * freq * x - 1) * duty
        elif shape == 'square':
            wave = np.clip((np.sign(np.sin(2 * np.pi * freq * x)) + 1) * duty / 2, 0, 1)
        elif shape == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
            wave = np.clip((wave + 1) * duty / 2, 0, 1)
        return np.clip(wave, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in [2, 3, 4]:
        for shape in ['triangle', 'saw', 'square', 'pulse']:
            guesses.append(gen_wave(freq, shape, 0.5))
    for _ in range(10):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=60,
                                             mutation=(0.7, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 250, 'ftol': 1e-11})
        return res.x if res.success else r

    for _ in range(15):
        if time.time() - start_time > budget_s - 5:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
