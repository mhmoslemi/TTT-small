import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_symmetric(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_rect(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(np.sin(2 * np.pi * freq * x) > 0.5, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def gen_checkerboard():
        h = np.zeros(n_points)
        cnt = 0
        for i in range(n_points):
            if cnt < int(n_points / 4):
                h[i] = 1
            elif cnt < int(n_points / 2):
                h[i] = 0
            elif cnt < int(3 * n_points / 4):
                h[i] = 1
            else:
                h[i] = 0
            cnt = (cnt + 1) % n_points
        return h

    def gen_bipulse():
        h = np.zeros(n_points)
        h[:int(n_points / 4)] = 1
        h[-int(n_points / 4):] = 1
        return h

    guesses = [h]
    for freq in [2, 3, 4]:
        guesses.append(gen_symmetric(freq, 0.5))
        guesses.append(gen_tri(freq, 0.25))
        guesses.append(gen_rect(freq, 0.5))
    guesses.append(gen_checkerboard())
    guesses.append(gen_bipulse())
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=50,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],

    return best_h, best_c5, n_points
