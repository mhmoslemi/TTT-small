def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 100
        x = np.linspace(0, 2, n_points)
        h = (1 + np.sin(np.pi * x)) / 2
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        desired_sum = n_points / 2
        scaling = desired_sum / current_sum
        h *= scaling
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_sine_guess(n_points):
        x = np.linspace(0, 2, n_points)
        h = (1 + np.sin(np.pi * x)) / 2
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        desired_sum = n_points / 2
        scaling = desired_sum / current_sum
        return h * scaling

    def generate_square_guess(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros_like(x)
        h[x < 1] = 1.0
        current_sum = np.sum(h)
        desired_sum = n_points / 2
        scaling = desired_sum / current_sum
        return h * scaling

    def generate_random_guess(n_points):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [best_h]
    guesses.append(generate_sine_guess(n_points))
    guesses.append(generate_square_guess(n_points))
    for _ in range(5):
        guesses.append(generate_random_guess(n_points))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(
            obj,
            bounds,
            popsize=60,
            maxiter=100,
            mutation=(0.5, 0.9),
            recombination=0.8,
            tol=1e-9
        )
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else r

    for _ in range(15):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
