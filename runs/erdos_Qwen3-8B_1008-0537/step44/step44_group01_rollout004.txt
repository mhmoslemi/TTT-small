import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_uniform(n_segments):
        x = np.linspace(0, 2, n_points)
        segments = np.zeros_like(x)
        step = 2.0 / n_segments
        for i in range(n_segments):
            start = i * step
            end = start + step
            mask = (x >= start) & (x < end)
            segments[mask] = 1
        return segments * (target_sum / n_segments)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for n_segments in range(3, 5):
        for phase in [0.1, 0.3, 0.5, 0.7]:
            guesses.append(gen_uniform(n_segments))
    for _ in range(10):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-11})
        return res.x if res.success else r

    for _ in range(15):
        if time.time() - start_time > budget_s - 20:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
