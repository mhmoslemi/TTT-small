import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_tiling(freq, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(np.sin(2 * np.pi * freq * x + phase) + 1, 0, 1) * 0.5

    def gen_flywheel(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_grid(theta, phi):
        return np.clip(np.cos(np.pi * theta) + np.sin(np.pi * phi), 0, 1) * 0.5

    guesses = [h]
    for freq in range(2, 15):
        for phase in [0, np.pi/4, np.pi/2, 3*np.pi/4]:
            guesses.append(gen_tiling(freq, phase))
    for freq in range(3, 12):
        for duty in [0.2, 0.4, 0.6]:
            guesses.append(gen_flywheel(freq, duty))
    for theta in np.linspace(0, 1, 10):
        for phi in np.linspace(0, 1, 10):
            guesses.append(gen_grid(theta, phi))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-11})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=100,
                                             mutation=(0.5, 0.95), recombination=0.7, tol=1e-9)
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
