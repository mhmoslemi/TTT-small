import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_square(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(np.abs(2 * freq * x - 1) * duty - 0.5, 0, 1)

    def gen_modulated(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_set(freq, duty):
        intervals = np.linspace(0, 1, int(freq) + 1)
        x = np.linspace(0, 1, n_points)
        return np.sum([np.abs(x - i) < 0.5 / freq for i in intervals], axis=0) * duty

    def gen_rectangular(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_finite(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(np.clip((np.sin(2 * np.pi * freq * x) + 1) * duty / 2, 0, 1), 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in [2, 3, 4, 5]:
        guesses.append(gen_square(freq, 0.5))
        guesses.append(gen_modulated(freq, 0.5))

    for duty in [0.2, 0.4, 0.6, 0.8]:

    for _ in range(5):

    def slsqp_guess(g):
