import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def symmetric_square(n_points, duty=0.5):
        x = np.linspace(0, 2, n_points)
        mid = 1.0
        length = 2 * duty
        return np.where((x >= mid - length) & (x <= mid + length), 1.0, 0.0)

    def symmetric_triangle(n_points, duty=0.5):
        x = np.linspace(0, 2, n_points)
        mid = 1.0
        length = 2 * duty
        wave = np.clip(1 - np.abs(2 * (x - mid) / length), 0, 1)
        return wave

    def symmetric_pulse(n_points, freq=4):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * (x - 1))
        return np.where(wave > 0, 1.0, 0.0)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    guesses.append(symmetric_square(n_points))
    guesses.append(symmetric_triangle(n_points, 0.4))
    guesses.append(symmetric_pulse(n_points, 3))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=120, maxiter=80,
                                             mutation=(0.4, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(15):
        if time.time() - start_time > budget_s - 20:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
