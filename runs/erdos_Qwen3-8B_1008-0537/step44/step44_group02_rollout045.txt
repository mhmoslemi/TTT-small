import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_uniform(n_points, m):
        l = 1.0 / m
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(m):
            start = i * l
            end = start + l
            start_idx = np.searchsorted(x, start)
            end_idx = np.searchsorted(x, end)
            h[start_idx:end_idx] = 1
        return h

    def generate_pulse(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def generate_square(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((np.sign(wave) + 1) * duty / 2, 0, 1)

    guesses = [h]
    for m in [5, 10, 15, 20, 25]:
        guesses.append(generate_uniform(n_points, m))
    for freq in [2, 3, 4]:
        guesses.append(generate_pulse(freq, 0.5))
        guesses.append(generate_square(freq, 0.5))
    for duty in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(generate_pulse(2, duty))
        guesses.append(generate_square(2, duty))
    for _ in range(5):
        guesses.append(np.random.rand(n_points))
        guesses[-1] /= np.sum(guesses[-1]) * target_sum

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-12})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=40,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
