import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.sin(2 * np.pi * freq * x) + 1) * duty / 2, 0, 1)

    def gen_square(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.sign(np.sin(2 * np.pi * freq * x)) + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    guesses = [h]
    for freq in range(3, 15):
        guesses.append(gen_pulse(freq, 0.5))
        guesses.append(gen_square(freq, 0.5))

        guesses.append(gen_triangle(3, duty))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):

    for g in guesses:
