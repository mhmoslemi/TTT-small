import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_wave(f, d, w):
        x = np.linspace(0, 1, n_points)
        t = 2 * np.pi * f * x
        if w == 'square':
            wave = np.clip((np.sign(np.sin(t)) + 1) * d / 2, 0, 1)
        elif w == 'pulse':
            wave = np.clip(np.sin(t) * d / 2 + d / 2, 0, 1)
        elif w == 'saw':
            wave = np.clip(1 - np.abs(2 * f * x - 1) * d, 0, 1)
        elif w == 'tri':
            wave = np.clip(1 - np.abs(2 * f * x - 1) * d, 0, 1)
        else:
            wave = np.zeros(n_points)
        return wave

    def gen_rand():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 16):
        guesses.append(gen_wave(freq, 0.5, 'square'))
        guesses.append(gen_wave(freq, 0.5, 'pulse'))
    for dc in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_wave(3, dc, 'saw'))
        guesses.append(gen_wave(3, dc, 'tri'))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(5):
        if time.time() - start_time > budget_s - 20:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=150, maxiter=70,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        h_opt = res.x
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
