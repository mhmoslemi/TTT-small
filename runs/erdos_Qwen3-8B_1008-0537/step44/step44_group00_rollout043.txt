    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_two_pulses(n):
        dx = 2.0 / n
        pulse_width = 0.5
        spacing = 1.0
        pulses = []
        x_start = 0
        for _ in range(2):
            x_end = x_start + pulse_width
            i_low = int(x_start / dx)
            i_high = int(x_end / dx) + 1
            i_low = max(0, i_low)
            i_high = min(n, i_high)
            pulses.append(range(i_low, i_high))
            x_start = x_end + spacing
        h = np.zeros(n)
        for indices in pulses:
            for i in indices:
                h[i] = 1
        return h

    def gen_random_50(n):
        h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        if np.sum(h) > n / 2:
            for i in range(n):
                if h[i] == 1:
                    h[i] = 0
                    break
        elif np.sum(h) < n / 2:
            for i in range(n):
                if h[i] == 0:
                    h[i] = 1
                    break
        return h

    def gen_pulses(f, d):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * f * x)
        return np.clip((wave + 1) * d / 2, 0, 1)

    def gen_triangle(f, d):
        x = np.linspace(0, 2, n_points)
        return 1 - np.abs(2 * f * x - 1) * d

    def gen_saw(f, d):
        x = np.linspace(0, 2, n_points)
        return np.clip(1 - np.abs(2 * f * x - 1) * d, 0, 1)

    guesses = [h]
    guesses.append(gen_two_pulses(n_points))
    guesses.append(gen_random_50(n_points))
    guesses.append(gen_pulses(3, 0.5))
    guesses.append(gen_triangle(3, 0.5))
    guesses.append(gen_saw(3, 0.5))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(2):
        if time.time() - start_time > budget_s - 20:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=50,
                                             mutation=(0.6, 0.95), recombination=0.8, tol=1e-9)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
