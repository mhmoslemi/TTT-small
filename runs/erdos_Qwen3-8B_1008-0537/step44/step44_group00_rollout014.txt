import scipy.optimize as optimize
import time

    def obj(h_vals):

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = 1 - np.abs(2 * freq * x - 1)
        return np.clip(wave * duty, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = 1 - np.abs(2 * freq * x - 1)
        return np.clip(wave * duty, 0, 1)
    
    def gen_random_binary():
        h_rand = np.zeros(n_points)
        h_rand[:int(target_sum)] = 1
        np.random.shuffle(h_rand)
        return h_rand

    guesses.append(gen_saw(3, 0.5))
    
    def slsqp_guess(g):
