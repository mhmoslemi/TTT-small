                mutation=(0.5, 0.8), recombination=0.9, tol=1e-9
            )
            return res.x
        except:
            return best_h

    def slsqp_refine(g):
        res = optimize.minimize(
            obj, g, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': cons}],
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        return res.x if res.success else g

    # Initial evolution
    h_opt = diff_evolve()
    h_opt = slsqp_refine(h_opt)
    c = obj(h_opt)
    if c < best_c5:
        best_h, best_c5 = h_opt, c

    # Intensive refinement
    for _ in range(12):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_refine(best_h)
        current_c5 = obj(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        if current_c5 < 0.3808:
            break

    return best_h, best_c5, n_points
