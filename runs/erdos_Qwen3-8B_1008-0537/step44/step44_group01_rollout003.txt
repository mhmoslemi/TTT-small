import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def obj(h_vals):

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_square(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_custom(freq, shape):
        x = np.linspace(0, 1, n_points)
        if shape == 'saw':
            return np.clip(1 - np.abs(2 * freq * x - 1) * 0.5, 0, 1)
        elif shape == 'triangle':
            return 1 - np.abs(2 * freq * x - 1)
        else:
            return np.sin(2 * np.pi * freq * x)

    guesses = [h]
    for freq in range(2, 8):
        guesses.append(gen_square(freq, 0.5))
        guesses.append(gen_custom(freq, 'saw'))
        guesses.append(gen_custom(freq, 'triangle'))
    for _ in range(10):
        guesses.append(np.random.rand(n_points))
        guesses.append(np.random.rand(n_points) * 2 - 1)

    def slsqp_guess(g):

                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g

    for g in guesses:
