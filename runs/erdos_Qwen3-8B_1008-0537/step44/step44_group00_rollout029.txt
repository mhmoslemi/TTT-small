import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulses(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_triangle(freq, duty):
        x = np.linspace(0, 2, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 2, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_square(freq, duty):
        x = np.linspace(0, 2, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * (1 - duty), 0, 1)
    
    guesses = [h]
    guesses.append(gen_pulses(2, 0.4))
    guesses.append(gen_triangle(2, 0.4))
    guesses.append(gen_saw(2, 0.4))
    guesses.append(gen_square(2, 0.4))
    guesses.append(gen_pulses(3, 0.5))
    guesses.append(gen_triangle(3, 0.5))
    guesses.append(gen_saw(3, 0.5))
    guesses.append(gen_square(3, 0.5))
    guesses.append(gen_pulses(4, 0.6))
    guesses.append(gen_triangle(4, 0.6))
    guesses.append(gen_saw(4, 0.6))
    guesses.append(gen_square(4, 0.6))
    
    def slsqp_guess(g, maxiter=100, ftol=1e-7):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': maxiter, 'ftol': ftol})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def evolve(popsize=50, maxiter=50, mutation=(0.6, 0.95), recomb=0.8, tol=1e-6):
        res = optimize.differential_evolution(obj, bounds, popsize=popsize, maxiter=maxiter,
                                             mutation=mutation, recombination=recomb, tol=tol)
        return res.x
    
    for _ in range(2):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = evolve(popsize=50, maxiter=50, mutation=(0.6, 0.95), recomb=0.8, tol=1e-6)
        h_opt = slsqp_guess(h_opt, maxiter=100, ftol=1e-7)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        h_opt = slsqp_guess(best_h, maxiter=100, ftol=1e-7)
        if obj(h_opt) < best_c5:
            best_h, best_c5 = h_opt, obj(h_opt)
    
    return best_h, best_c5, n_points
