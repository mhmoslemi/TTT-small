import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_square(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((np.sign(wave) + 1) * duty / 2, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 500, 'ftol': 1e-10})
        return res.x if res.success else g

    def de_guess(g):
        res = optimize.differential_evolution(obj, bounds, popsize=80, maxiter=50,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9,
                                             workers=-1, strategy='best1bin')
        return res.x

    # Optimizing initial guess
    best_h = slsqp_guess(best_h)
    best_c5 = obj(best_h)

    # Run DE with initial guess
    for _ in range(2):
        if time.time() - start_time > budget_s - 10:
            break
        h_opt = de_guess(best_h)
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Refinement with SLSQP
    for _ in range(10):
        if time.time() - start_time > budget_s - 5:

        best_h = slsqp_guess(best_h)
