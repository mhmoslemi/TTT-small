import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = initial_h_values.astype(float)
    best_c5 = obj(best_h)
    start_time = time.time()

    def generate_uniform_pulses(k):
        if k < 2:
            return np.zeros(n_points)
        w = 1.0 / k
        d = 1.0 / (k - 1)
        pulse_starts = [i * (w + d) for i in range(k)]
        pulse_ends = [s + w for s in pulse_starts]
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(k):
            start = pulse_starts[i]
            end = pulse_ends[i]
            j_start = np.searchsorted(x, start, side='right')
            j_end = np.searchsorted(x, end, side='left')
            if j_start < j_end:
                h[j_start:j_end] = 1.0
        return h

    def gen_pulses(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [initial_h_values]
    for k in range(2, 11):
        guesses.append(generate_uniform_pulses(k))
    for freq in range(2, 6):
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.2, 0.4, 0.6, 0.8]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.2, 0.4, 0.6, 0.8]:
        guesses.append(gen_triangle(3, duty))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=50,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else r

    for _ in range(15):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
