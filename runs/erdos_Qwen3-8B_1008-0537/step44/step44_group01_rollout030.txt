import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 30
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h_values = np.array([0.34, 0.31, 0.30, 0.55, 0.58, 0.62, 0.69, 0.72, 0.73, 0.74, 
                                 0.56, 0.76, 0.74, 0.78, 0.85, 0.81, 0.87, 0.67, 0.66, 0.72, 
                                 0.69, 0.65, 0.88, 0.68, 0.65, 0.53, 0.53, 0.76, 0.80, 0.87], dtype=float)
    h = initial_h_values.copy()
    h /= np.sum(h) * dx
    h *= target_sum / (np.sum(h) * dx)

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty, h):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_random(h):
        h = h.copy()
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(2, 5):
        guesses.append(gen_pulses(freq, 0.5, h))
    for duty in [0.1, 0.3, 0.5, 0.7]:
        guesses.append(gen_saw(4, duty))
    for duty in [0.1, 0.3, 0.5, 0.7]:
        guesses.append(gen_triangle(3, duty))
    for _ in range(4):
        guesses.append(gen_random(h))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-11})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():

                                             mutation=(0.4, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(2):
        if time.time() - start_time > budget_s - 30:

        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)

    def refine(r):
