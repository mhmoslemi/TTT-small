import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_binary():
        h = np.zeros(n_points)
        idx = np.argsort(np.random.rand(n_points))
        h[idx[:int(target_sum)]] = 1.0
        return h

    def gen_wave(freq, phase, duty, type_):
        x = np.linspace(0, 1, n_points)
        if type_ == 'saw':
            wave = 1 - np.abs(2 * freq * x - 1) * duty
        elif type_ == 'tri':
            wave = 1 - np.abs(2 * freq * x - 1) * duty
        elif type_ == 'pul':
            wave = np.sin(2 * np.pi * freq * x + phase)
            wave = np.clip((wave + 1) * duty / 2, 0, 1)
        elif type_ == 'squ':
            wave = np.sin(2 * np.pi * freq * x + phase)
            wave = np.clip((np.sign(wave) + 1) * duty / 2, 0, 1)
        return np.clip(wave, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for _ in range(3):
        guesses.append(gen_binary())
    for freq in range(2, 6):
        for phase in [0, np.pi/2, np.pi]:
            guesses.append(gen_wave(freq, phase, 0.5, 'pul'))
            guesses.append(gen_wave(freq, phase, 0.5, 'squ'))
    for type_ in ['saw', 'tri']:
        for duty in [0.2, 0.4, 0.6, 0.8]:
            guesses.append(gen_wave(3, 0, duty, type_))
    for _ in range(2):
        guesses.append(gen_random())

    def slsqp_guess(g):
