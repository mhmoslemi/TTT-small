import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective_penalty(h):
        corr = np.correlate(h, 1 - h, mode='full')
        obj = np.max(corr * dx)
        sum_h = np.sum(h)
        penalty = 1e5 * (sum_h - target_sum) ** 2
        return obj + penalty

    # Global Optimization with DE
    result_de = differential_evolution(
        objective_penalty,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed
    )

    h_initial = result_de.x
    # Local Optimization with L-BFGS-B
    result = minimize(
        objective,
        h_initial,
        method='L-BFGSB',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        tol=1e-5,
        options={'maxiter': 100, 'disp': False}
    )

    if result.success:
        h_values = result.x
        c5_bound = result.fun
    else:
        h_values = result_de.x
        c5_bound = objective_penalty(h_values)

    return (h_values, c5_bound, n_points)
