        return np.sum(h_vec) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_func}]
    result = minimize(
        objective, h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 100}
    )
    best_h = result.x if result.success else h
    return (best_h.tolist(), objective(best_h), n_points)
