import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guess with two intervals of 1s
    interval_length = 0.5
    num_samples_per_interval = int(round(interval_length / dx))
    h_initial = np.zeros(n_points)
    start1 = 0
    end1 = start1 + num_samples_per_interval
    start2 = n_points - num_samples_per_interval
    end2 = start2 + num_samples_per_interval
    h_initial[start1:end1] = 1.0
    h_initial[start2:end2] = 1.0
    
    # Define objective function
    def objective(h):
        c = correlate(h, 1 - h, mode='full')
        c5 = np.max(c * dx)
        return c5
    
    # Define constraint: sum(h) = n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Perform optimization
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons)
    
    # Return the best solution
    h_values = result.x
