import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h_arr = np.array(h)
        cross_corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(cross_corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

    h_initial = initial_h.copy()
    start_time = time.time()

    result = opt.minimize(objective, h_initial, method='Nelder-Mead', 
                          bounds=bounds, constraints=cons, options={'maxiter': 3000, 'ftol': 1e-10})

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points

    return result.x, objective(result.x), n_points
