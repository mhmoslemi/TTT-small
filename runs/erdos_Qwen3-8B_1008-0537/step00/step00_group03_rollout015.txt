from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    h0 = initial_h.copy()

    result = minimize(
        objective,
        h0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-6}
    )

    if result.success:
        h_values = result.x
        c5_bound = result.fun
    else:
        h_values = initial_h
        c5_bound = objective(initial_h)

    return (h_values, c5_bound, n_points)
