import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is available as a global variable
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()

    def objective(h_values):
        ones_minus_h = 1.0 - h_values
        conv = np.correlate(h_values, ones_minus_h, mode='full')
        max_overlap = np.max(conv) * dx
        return max_overlap

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    # Define constraints and bounds
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Start optimization
    start_time = time.time()
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8)
    elapsed = time.time() - start_time

    # Early exit if budget is exceeded
    if elapsed > budget_s:
        return (initial_h, objective(initial_h), n_points)

    # Return best solution
    best_h = result.x
    c5_bound = result.fun
    return (best_h, c5_bound, n_points)
