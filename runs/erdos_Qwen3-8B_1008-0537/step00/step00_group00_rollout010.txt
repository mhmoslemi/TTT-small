from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = np.abs(np.sum(h) - n_points / 2)
        return max_corr * dx + 1000 * penalty

    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False
    )
    elapsed = time.time() - start_time
