from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    n_points = 200
    dx = 2.0 / n_points
    # Initialize h with initial_h_values if available, else random
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_initial = len(h_initial)
        x_initial = np.linspace(0, 2, n_initial)
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_initial, h_initial)
    else:
        h = np.random.rand(n_points)
        h = (h / np.sum(h)) * (n_points / 2)
        h = np.clip(h, 0, 1)
    # Objective function
    def objective(h_vec):
        conv = correlate(h_vec, 1. - h_vec, mode='full')
        return np.max(conv) * dx
    # Constraints
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]
    # Optimization
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons,
                      options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
