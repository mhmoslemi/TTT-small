        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

    # Use COBYLA which handles bounds and non-smooth functions
    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-6}
    )

    h_values = result.x
    c5_bound = result.fun
    return (h_values, c5_bound, n_points)
