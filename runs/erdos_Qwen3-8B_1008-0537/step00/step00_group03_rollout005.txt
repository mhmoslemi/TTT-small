        h_initial = h_initial / np.sum(h_initial) * n_points / 2
    
    dx = 2.0 / n_points
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    T = 100.0
    cooling_rate = 0.995
    max_iter = 10000
    best_h = h_initial.copy()
    best_c5 = evaluate(best_h)
    
    for _ in range(max_iter):
        T *= cooling_rate
        if T < 1e-3:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        new_h = h_initial + perturbation
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        target_sum = n_points / 2
        diff = target_sum - current_sum
        if abs(diff) > 1e-6:
            idx = np.random.randint(n_points)
            new_h[idx] += diff
            new_h = np.clip(new_h, 0, 1)
        new_c5 = evaluate(new_h)
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / T):
            h_initial = new_h
            best_c5 = new_c5
    
    return (h_initial, best_c5, n_points)
