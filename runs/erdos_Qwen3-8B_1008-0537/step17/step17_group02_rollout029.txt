import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_structured_guess(m):
        structured_h = np.zeros(n_points)
        period = n_points // 2
        for i in range(n_points):
            if (i * m) % period < period // 2:
                structured_h[i] = 1.0
        structured_h[-1] = 0.5
        return structured_h
    
    initial_guesses = [initial_guess]
    for m in [3, 4, 5, 7, 9]:
        initial_guesses.append(generate_structured_guess(m))
    
    n_initial = len(initial_guesses)
    for i in range(n_initial):
        guess = initial_guesses[i]
        res = optimize_guess(guess)
        if res is not None and objective(res) < best_c5:
            best_c5 = objective(res)
            best_h = res
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,

            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            res = optimize_guess(result_de.x)

    except:
