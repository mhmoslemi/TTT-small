        dx = 2.0 / n_points
        target_sum = n_points / 2
        number_of_ones = int(target_sum)
        remaining = target_sum - number_of_ones
        ideal_positions = []
        for i in range(number_of_ones):
            ideal_x = i * (2.0 / number_of_ones)
            sample_index = round(ideal_x / dx)
            if 0 <= sample_index < n_points:
                ideal_positions.append(sample_index)
        unique_positions = list(set(ideal_positions))
        binary_guess = np.zeros(n_points)
        for idx in unique_positions:
            binary_guess[idx] = 1.0
        if remaining > 0:
            idx_to_adjust = np.random.choice(unique_positions)
            binary_guess[idx_to_adjust] = 0.5
        return binary_guess
    
    def generate_periodic_guess():
        structured_h = np.zeros(n_points)
        for i in range(n_points):
            if i % 4 == 0:
                structured_h[i] = 1.0
            elif i % 4 == 1:
                structured_h[i] = 0.5
            else:
                structured_h[i] = 0.0
        structured_h[-1] = 0.5
        return structured_h
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    initial_guesses = [initial_guess]
    
    structured_h = generate_periodic_guess()
    initial_guesses.append(structured_h)
    
    binary_guess = generate_binary_guess()
    initial_guesses.append(binary_guess)
    
    dense_h = np.full(n_points, 0.5)
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    dense_h[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        dense_h[idx] += remaining
    initial_guesses.append(dense_h)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=150,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
