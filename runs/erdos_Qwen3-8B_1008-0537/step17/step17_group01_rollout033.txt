    initial_guesses = [initial_guess]
    
    # Generate structured periodic guess with different period
    structured_h = np.zeros(n_points)
    period = 5
    for i in range(n_points):
        if i % period == 0:
            structured_h[i] = 1.0
        elif i % period == 1 or i % period == 2:
            structured_h[i] = 0.5
        else:
            structured_h[i] = 0.0
    structured_h[-1] = 0.5
    initial_guesses.append(structured_h)
    
    # Generate evenly spaced binary guess
    m = 23
    positions = np.linspace(0, n_points, m + 1)[1:-1]
    indices = np.round(positions).astype(int)
    indices = np.unique(indices)
    h_binary = np.zeros(n_points)
    h_binary[indices] = 1.0
    delta = (m * dx) - 1.0
    if delta > 0:
        h_binary[indices[0]] -= delta
    initial_guesses.append(h_binary)
    
    # Generate dense random guess
    dense_h = np.full(n_points, 0.5)
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    dense_h[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        dense_h[idx] += remaining
    initial_guesses.append(dense_h)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with tuned parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=150,
            mutation=(0.8, 0.95),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with L-BFGS-B and more iterations
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
