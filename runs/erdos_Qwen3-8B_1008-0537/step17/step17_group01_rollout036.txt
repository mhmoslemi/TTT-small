    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess, method='L-BFGS-B'):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    initial_guesses = [initial_guess]
    
    # Structured periodic guess
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h[i] = 1.0
        elif i % 3 == 1:
            structured_h[i] = 0.5
        else:
            structured_h[i] = 0.0
    structured_h[-1] = 0.5
    initial_guesses.append(structured_h)
    
    # Random binary guess
    bits = np.random.rand(n_points)
    bits = (bits <= 1.0 / target_sum).astype(float)
    bits = np.clip(bits * target_sum, 0, 1)
    initial_guesses.append(bits)
    
    # Dense random guess
    dense_h = np.full(n_points, 0.5)
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    dense_h[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        dense_h[idx] += remaining
    initial_guesses.append(dense_h)
    
    # Regular spaced binary guess
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    if num_ones > 0:
        step = n_points // num_ones
        regular_h = np.zeros(n_points)
        for i in range(num_ones):
            idx = i * step
            if idx < n_points:
                regular_h[idx] = 1.0
        if remaining > 0:
            idx = np.random.choice(np.where(regular_h == 0)[0], size=1, replace=False)
            regular_h[idx] += remaining
        initial_guesses.append(regular_h)
    
    # Evaluate initial guesses
    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None:
            c5 = objective(res)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res
    
    # Differential evolution with hexagonal grid
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=90,
            maxiter=120,
            mutation=(0.8, 0.95),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            res = optimize_guess(result_de.x, 'L-BFGS-B')
            if res is not None and objective(res) < best_c5:
                best_c5 = objective(res)
                best_h = res
    except:
        pass
    
    # Local refinement with dynamic steps
    total_time = time.time() - start_time
    while total_time < budget_s - 20:
        try:
            res = optimize_guess(best_h, 'L-BFGS-B')
            if res is not None and objective(res) < best_c5:
                best_c5 = objective(res)
                best_h = res
            total_time = time.time() - start_time
        except:
            break
    
    return best_h, best_c5, n_points
