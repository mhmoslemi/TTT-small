            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    initial_guesses = [initial_guess]
    
    # Structured spaced guess
    m = n_points // 2
    remaining = target_sum - m
    spaced_h = np.zeros(n_points)
    for i in range(m):
        idx = int(i * n_points / m)
        spaced_h[idx] = 1.0
    if remaining > 0:
        spaced_h[0] += remaining
    initial_guesses.append(spaced_h)
    
    # Binary randomized guess
    bits = np.random.randint(0, 2, size=n_points)
    bits = bits / np.sum(bits) * target_sum
    bits = np.clip(bits, 0, 1)
    initial_guesses.append(bits)
    
    # Dense random guess
    dense_h = np.full(n_points, 0.5)
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    dense_h[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        dense_h[idx] += remaining
    initial_guesses.append(dense_h)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with tighter parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=30,
            maxiter=100,
            mutation=(0.6, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with L-BFGS-B
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
