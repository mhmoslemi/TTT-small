import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    initial_guesses = [initial_guess]
    
    def generate_periodic(n_points):
        h = np.zeros(n_points)
        h[::2] = 1.0
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    initial_guesses.append(generate_periodic(n_points))
    
    def generate_spread(n_points):
        h = np.zeros(n_points)
        idx = np.argsort(np.random.uniform(0, 1, n_points))
        idx = idx[:int(n_points/2)]
        h[idx] = 1.0
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    initial_guesses.append(generate_spread(n_points))
    
    def generate_random(n_points):
        h = np.zeros(n_points)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h[indices] = 1.0
        if remaining > 1e-8:
            h[np.random.choice(n_points)] += remaining
        return h
    for _ in range(4):
        initial_guesses.append(generate_random(n_points))
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=500,
            mutation=(0.5, 1.0),
            recombination=0.85,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(

                x0=best_h,
