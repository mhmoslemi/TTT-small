import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    initial_guesses = [initial_guess]
    
    # Structured guess: shifted triangular wave
    structured_h = np.zeros(n_points)

        pos = i / (n_points - 1) * 2
        if 0 <= pos < 1:
            structured_h[i] = pos * 0.5
        elif 1 <= pos < 2:
            structured_h[i] = 1 - (pos - 1) * 0.5
    structured_h = np.clip(structured_h, 0, 1)
    structured_h[-1] = 0.5
    initial_guesses.append(structured_h)
    
    # Structured binary guess: 1, 0, 1, 0...
    binary_h = np.zeros(n_points)
    binary_h[np.arange(n_points) % 2 == 0] = 1.0
    binary_h[-1] = 0.5
    initial_guesses.append(binary_h)
    
    # Structured sparse guess: 3 alternating blocks
    sparse_h = np.zeros(n_points)
    region_size = n_points // 3
    for i in range(3):
        start = i * region_size
        end = (i + 1) * region_size
        if i % 2 == 0:
            sparse_h[start:end] = 1.0
    sparse_h = np.clip(sparse_h, 0, 1)
    sparse_h[-1] = 0.5
    initial_guesses.append(sparse_h)
    
    for guess in initial_guesses:

            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with increased population
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=150,
            mutation=(0.5, 0.8),

            tol=1e-10,
            vectorized=True
        )
        if result_de.success:

                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}

                best_c5 = objective(result_slsqp.x)
