        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_even_sample(n_points):
        h = np.zeros(n_points)

        h[idx] = 1.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:

                h[np.random.choice(n_points)] += diff

                h[i] = 0.0
