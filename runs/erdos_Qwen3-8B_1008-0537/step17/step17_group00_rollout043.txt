        idx = np.argsort(np.random.uniform(0, 1, n_points))
        idx = idx[:int(n_points/2)]
        h[idx] = 1.0
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    initial_guesses.append(generate_spread(n_points))
    
    # Evenly spaced initial guess
    def generate_evenly_spaced(n_points):
        h = np.zeros(n_points)
        num_ones = int(np.round(target_sum))
        indices = np.linspace(0, n_points - 1, num_ones, endpoint=False).astype(int)
        h[indices] = 1.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[np.random.choice(n_points)] += diff
        return h
    initial_guesses.append(generate_evenly_spaced(n_points))
    
    # Random initial guesses
    for _ in range(5):
        def generate_random(n_points):
            h = np.zeros(n_points)
            num_ones = int(np.round(target_sum))
            remaining = target_sum - num_ones
            indices = np.random.choice(n_points, size=num_ones, replace=False)
            h[indices] = 1.0
            if remaining > 1e-8:
                h[np.random.choice(n_points)] += remaining
            return h
        initial_guesses.append(generate_random(n_points))
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution optimization with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=200,
            maxiter=500,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with L-BFGS-B
    while time.time() - start_time < budget_s - 10:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
