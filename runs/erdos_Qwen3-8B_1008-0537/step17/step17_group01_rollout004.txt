import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        return result.x if result.success else None
    
    def generate_structured_h(period=4):
        h = np.zeros(n_points)
        for i in range(n_points):
            val = 1.0 if i % period == 0 else 0.0
            if i % (period // 2) == 0:
                val = 0.5
            h[i] = val
        scale = target_sum / np.sum(h)
        h *= scale
        h[-1] = (target_sum - np.sum(h[:-1])) / 1.0
        return h
    
    initial_guesses = [initial_guess]
    initial_guesses.append(generate_structured_h(4))
    initial_guesses.append(generate_structured_h(5))
    initial_guesses.append(np.random.rand(n_points))
    initial_guesses.append(np.full(n_points, 0.5))
    initial_guesses.append(np.random.rand(n_points) * 0.5)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=250,
            mutation=(0.5, 0.9),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
