    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=250,
            mutation=(0.5, 0.9),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_nm = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='Nelder-Mead',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'xatol': 1e-10, 'fatol': 1e-10}
            )
            if result_nm.success and objective(result_nm.x) < best_c5:
                best_c5 = objective(result_nm.x)
                best_h = result_nm.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result_nm = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='Nelder-Mead',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'xatol': 1e-10, 'fatol': 1e-10}
            )
            if result_nm.success and objective(result_nm.x) < best_c5:
                best_c5 = objective(result_nm.x)
                best_h = result_nm.x
        except:
            break
    
    return best_h, best_c5, n_points
