        if remaining > 1e-8:
            h[np.random.choice(n_points)] += remaining
        return h
    
    initial_guesses = [initial_guess]
    for _ in range(3):
        initial_guesses.append(generate_random(n_points))
    initial_guesses.append(generate_spread(n_points))
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    def mutate(parent):
        child = parent.copy()
        idxs = np.random.choice(n_points, size=2, replace=False)
        child[idxs[0]] = np.clip(child[idxs[0]] + np.random.uniform(-0.1, 0.1), 0.0, 1.0)
        child[idxs[1]] = np.clip(child[idxs[1]] + np.random.uniform(-0.1, 0.1), 0.0, 1.0)
        return child
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def evolve(population, generations=30):
        for _ in range(generations):
            new_pop = []
            for _ in range(len(population) // 2):
                p1, p2 = np.random.choice(population, size=2, replace=False)
                child1 = crossover(p1, p2)
                child1 = mutate(child1)
                child2 = crossover(p1, p2)
                child2 = mutate(child2)
                new_pop.extend([child1, child2])
            population = new_pop
            population = [np.clip(p, 0.0, 1.0) for p in population]
            population.sort(key=objective)
            population = population[:len(population) // 2]
        return population
    
    population = [initial_guess]
    for _ in range(2):
        population.extend([generate_random(n_points), generate_spread(n_points)])
    population = evolve(population)
    
    for guess in population:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
