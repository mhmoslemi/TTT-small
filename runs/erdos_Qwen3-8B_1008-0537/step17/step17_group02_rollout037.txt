import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
        )
        return result.x if result.success else None
    
    # Generate diverse initial guesses

            spaced = np.zeros(n_points)
            for i in range(1, m+1):
                idx = int((i - 1) * n_points / m)
                spaced[idx] = 1.0
            spaced[-1] = 0.5
            guesses.append(spaced)
        # Block pattern
        block_size = n_points // 10
        block = np.zeros(n_points)
        for i in range(1, 7):
            idx = int((i - 1) * n_points / 6)
            block[idx:idx+block_size] = 1.0
        block[-1] = 0.5
        guesses.append(block)
        return guesses
    
    structured_guesses = generate_structured_guesses()
    initial_guesses = [initial_guess] + structured_guesses
    
    # Random initial guesses
    num_random = 5

        rand_h[indices] = 1.0

            idx = np.random.choice(n_points, size=1, replace=False)
            rand_h[idx] += remaining
        initial_guesses.append(rand_h)

    # Differential evolution with adjusted parameters
    try:
        result_de = optimize.differential_evolution(

        if result_de.success:
            result_slsqp = optimize.minimize(
