import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    structured_h = np.zeros(n_points, dtype=float)

    even_initial = np.zeros(n_points, dtype=float)
    m = n_points // 2
    positions = np.linspace(0, 2, m + 1)
    for i in range(1, m + 1):

        idx = int(pos / dx)
        if idx < n_points:

    initial_guesses = [initial_guess, structured_h, even_initial]
    
    for _ in range(3):

        rand_h[indices] = 1.0
        if remaining > 0:

        initial_guesses.append(rand_h)
