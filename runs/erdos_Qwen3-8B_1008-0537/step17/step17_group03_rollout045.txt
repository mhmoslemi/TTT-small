    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0
    structured_h[-1] = 0.5
    
    m = int(n_points / 2)
    positions = np.linspace(0, 2, m)
    indices = np.round(positions / dx).astype(int)
    indices = np.unique(indices)
    uniform_guess = np.zeros(n_points)
    uniform_guess[indices] = 1.0
    current_sum = np.sum(uniform_guess)
    delta = target_sum - current_sum
    if delta > 0:
        for i in range(n_points):
            if uniform_guess[i] < 1.0:
                uniform_guess[i] += delta
                delta = 0
                break
    elif delta < 0:
        for i in range(n_points-1, -1, -1):
            if uniform_guess[i] > 0.0:
                uniform_guess[i] -= -delta
                delta = 0
                break
    
    initial_guesses = [initial_guess, structured_h, uniform_guess]
    
    for _ in range(3):
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        rand_h = np.zeros(n_points)
        rand_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            rand_h[idx] += remaining
        initial_guesses.append(rand_h)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=75,
            maxiter=200,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
