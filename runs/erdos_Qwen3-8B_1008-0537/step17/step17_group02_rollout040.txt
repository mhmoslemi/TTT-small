            structured.append(h)
        return structured
    
    def generate_random_initials(num=5):
        random = []
        for _ in range(num):
            num_ones = int(target_sum) + np.random.randint(-2, 3)
            remaining = target_sum - num_ones
            if remaining < 0:
                remaining = 0
                num_ones = int(target_sum)
            trial = np.zeros(n_points)
            indices = np.random.choice(n_points, size=num_ones, replace=False)
            trial[indices] = 1.0
            if remaining > 0:
                idx = np.random.choice(n_points, size=1, replace=False)
                trial[idx] += remaining
            random.append(trial)
        return random
    
    initial_guesses = [initial_guess]
    initial_guesses.extend(generate_structured_initials())
    initial_guesses.extend(generate_random_initials(10))
    
    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None and objective(res) < best_c5:
            best_c5 = objective(res)
            best_h = res
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=200,
            maxiter=300,
            mutation=(0.9, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            res_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            res = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        except:
            break
    
    return best_h, best_c5, n_points
