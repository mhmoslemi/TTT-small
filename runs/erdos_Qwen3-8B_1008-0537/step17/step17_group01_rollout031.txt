    reg_binary[idx] = 1.0
    initial_guesses.append(reg_binary)
    
    # Biased binary pattern for testing
    m = int(target_sum)
    idx = np.argsort(np.random.rand(n_points))
    alt_binary = np.zeros(n_points)
    alt_binary[idx[:m]] = 1.0
    initial_guesses.append(alt_binary)
    
    # Dense random guess
    dense_h = np.full(n_points, 0.5)
    num_ones = int(target_sum)
    remaining = target_sum - num_ones
    idx = np.random.choice(n_points, size=num_ones, replace=False)
    dense_h[idx] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)
        dense_h[idx] += remaining
    initial_guesses.append(dense_h)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with tighter parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=200,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with L-BFGS-B
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
