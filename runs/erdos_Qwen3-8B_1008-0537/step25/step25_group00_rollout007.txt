import scipy.optimize as optimize

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_tiled(n_points):

            h[i] = 1.0
        for i in range(step, 2*step):
            h[i] = 0.5
        for i in range(2*step, 3*step):
            h[i] = 0.0
        for i in range(3*step, n_points):
            h[i] = 0.5
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_distinct(n_points):

            h[idx[i]] = 1.0

            h[idx[i]] = 0.5

            h[-1] += diff
        return h
    
    initial_guesses = [initial_guess, generate_tiled(n_points), generate_distinct(n_points)]
    
    def optimize_guess(guess):
        result = optimize.minimize(
