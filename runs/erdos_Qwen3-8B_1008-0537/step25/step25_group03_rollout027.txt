import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic_h(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if (i % period) == 0 else 0.5 if (i % period) == 1 else 0.0
        h[-1] = 0.5
        return h
    
    def generate_distributed_ones(num_ones):
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h = np.full(n_points, 0.5)
        h[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            h[idx] += remaining
        return h
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    def diff_evolve():
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=30,
            maxiter=150,
            mutation=(0.6, 0.95),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        return result.x if result.success else None
    
    initial_guesses = [initial_guess]
    
    for p in [4, 7, 10]:
        initial_guesses.append(generate_periodic_h(p))
    
    for n in [22, 23]:
        initial_guesses.append(generate_distributed_ones(n))
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    if time.time() - start_time < budget_s - 30:
        optimized = diff_evolve()
        if optimized is not None:
            result = optimize.minimize(
                fun=objective,
                x0=optimized,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
    
    while time.time() - start_time < budget_s - 10:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
