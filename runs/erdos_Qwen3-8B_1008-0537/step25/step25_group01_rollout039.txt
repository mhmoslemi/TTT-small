import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    initial_guesses = [initial_guess]
    
    # Generate structured periodic guess
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        structured_h[i] = 1.0 if i % 5 == 0 else 0.0
    structured_h[-1] = 0.5
    initial_guesses.append(structured_h)
    
    # Generate cosine-based guess
    x = np.linspace(0, 2, n_points)
    cosine_h = 0.5 + 0.5 * np.cos(np.pi * x / 2)
    total_integral = np.sum(cosine_h) * dx
    scaling_factor = 1.0 / total_integral
    cosine_h *= scaling_factor
    cosine_h = np.clip(cosine_h, 0, 1)
    initial_guesses.append(cosine_h)
    
    # Generate structured binary guess
    bits = np.random.randint(0, 2, size=n_points)
    bits = np.clip(bits / np.sum(bits) * target_sum, 0, 1)
    initial_guesses.append(bits)
    
    # Generate blocky guess
    block_size = n_points // 4
    block_h = np.zeros(n_points)
    for i in range(4):

    block_h[-1] = 0.5
    initial_guesses.append(block_h)
    
    # Generate random mutation guess
    mutant = initial_guess.copy()
    idx = np.random.choice(n_points, size=5, replace=False)

    initial_guesses.append(mutant)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with tighter parameters
    try:
        result_de = optimize.differential_evolution(

            bounds=bounds,
            popsize=50,

            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(

                method='SLSQP',

                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:

                best_h = result_slsqp.x
