    # Generate random binary guess
    bin_guess = np.zeros(n_points)
    ones_needed = int(target_sum)
    bin_guess[:ones_needed] = 1.0
    if target_sum - ones_needed > 0:
        bin_guess[-1] = target_sum - ones_needed
    initial_guesses.append(bin_guess)
    
    # Generate blocky guess
    block_size = n_points // 4
    block_h = np.zeros(n_points)
    for i in range(4):
        block_h[i * block_size:(i + 1) * block_size] = 1.0
    block_h[-1] = 0.5
    initial_guesses.append(block_h)
    
    # Generate random mutation guess
    mutant = initial_guess.copy()
    idx = np.random.choice(n_points, size=5, replace=False)
    mutant[idx] = np.random.uniform(0.2, 0.8, size=5)
    initial_guesses.append(mutant)
    
    # Generate structured binary guess with alternating 0 and 1
    alt_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            alt_guess[i] = 1.0
    current_sum = np.sum(alt_guess)
    diff = target_sum - current_sum
    alt_guess[-1] += diff
    alt_guess[-1] = np.clip(alt_guess[-1], 0, 1)
    initial_guesses.append(alt_guess)
    
    # Additional guess: more spaced out periodic pattern
    structured_h2 = np.zeros(n_points)
    period = n_points // 3
    for i in range(n_points):
        structured_h2[i] = 1.0 if i % period <= 1 else 0.0
    initial_guesses.append(structured_h2)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=150,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with L-BFGS-B and more iterations
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
