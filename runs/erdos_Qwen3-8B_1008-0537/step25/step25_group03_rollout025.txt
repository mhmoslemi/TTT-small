import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    def generate_periodic_h(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if (i % period) == 0 else 0.5 if (i % period) == 1 else 0.0
        h[-1] = 0.5
        return h
    
    def generate_binary_h():
        h = np.zeros(n_points)
        ones = np.random.choice(n_points, size=int(target_sum), replace=False)
        h[ones] = 1.0
        h[np.random.choice(n_points, size=1)] += target_sum - int(target_sum)
        return h
    
    def generate_dense_h():
        h = np.full(n_points, 0.5)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            h[idx] += remaining
        return h
    
    initial_guesses = [initial_guess]
    for period in [2, 3, 4, 5, 6, 7]:
        initial_guesses.append(generate_periodic_h(period))
    initial_guesses.append(generate_binary_h())
    initial_guesses.append(generate_dense_h())
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=40,
            maxiter=60,
            mutation=(0.6, 0.95),
            recombination=0.85,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 30:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
