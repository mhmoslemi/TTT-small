import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_even_spread_binary():
        num_ones = int(target_sum) + (1 if target_sum % 1 else 0)
        idxs = np.sort(np.random.choice(n_points, size=num_ones, replace=False))
        h = np.zeros(n_points)
        h[idxs] = 1.0
        return h
    
    def generate_periodic(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = max(0.0, min(1.0, 0.5 * (1 + np.cos(2 * np.pi * i / period))))
        return h
    
    initial_guesses = [initial_guess]
    for period in [4, 5, 6, 8, 10]:
        initial_guesses.append(generate_periodic(period))
    initial_guesses.append(generate_even_spread_binary())
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None and objective(res) < best_c5:
            best_c5 = objective(res)
            best_h = res
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=100,
            mutation=(0.6, 0.8),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            res = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            res = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'rhobeg': 0.15, 'rhoend': 1e-5}
            )
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        except:
            break
    
    return best_h, best_c5, n_points
