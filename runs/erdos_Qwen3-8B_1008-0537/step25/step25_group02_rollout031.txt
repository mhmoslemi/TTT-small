import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_periodic(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = max(0.0, min(1.0, 1.0 - 0.5 * np.sin(2 * np.pi * i / period)))
        h[-1] = 0.5
        return h
    
    def generate_uniform():
        h = np.zeros(n_points)
        num_ones = int(target_sum + 0.5)
        h[::int(n_points / num_ones)] = 1.0
        if np.sum(h) < target_sum:
            h[np.random.choice(n_points, size=1, replace=False)] += 1.0 - (target_sum - np.sum(h))
        return h

    for period in [4, 5, 6, 7, 8]:
        initial_guesses.append(generate_periodic(period))

        result_de = optimize.differential_evolution(
