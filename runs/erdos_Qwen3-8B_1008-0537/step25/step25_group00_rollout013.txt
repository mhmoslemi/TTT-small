import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_uniform(n_points):
        h = np.zeros(n_points)
        indices = np.argsort(np.random.uniform(0, 1, n_points))
        num_ones = int(target_sum) + (1 if np.random.rand() < 0.5 else 0)
        h[indices[:num_ones]] = 1.0
        h[indices[num_ones:]] = 0.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[indices[0]] += diff
        return h

            h[i] = 1.0

            h[i] = 0.5
        for i in range(2 * block_size, n_points):
            h[i] = 0.0

        return h
