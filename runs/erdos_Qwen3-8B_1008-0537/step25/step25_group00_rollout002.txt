import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_annular(n_points):
        h = np.zeros(n_points)
        idx = np.argsort(np.random.uniform(0, 1, n_points))
        for i in range(n_points // 3):
            h[idx[i]] = 1.0
        for i in range(n_points // 3, 2 * n_points // 3):
            h[idx[i]] = 0.5
        for i in range(2 * n_points // 3, n_points):
            h[idx[i]] = 0.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_stratified(n_points):
        h = np.zeros(n_points)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h[indices] = 1.0
        if remaining > 1e-8:
            h[np.random.choice(n_points)] += remaining
        return h
    
    def generate_sine(n_points):
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        val = 0.5 + 0.5 * np.sin(np.pi * x)
        h = np.clip(val, 0, 1)
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    initial_guesses = [initial_guess]
    initial_guesses.append(generate_annular(n_points))
    initial_guesses.append(generate_stratified(n_points))
    initial_guesses.append(generate_sine(n_points))
    
    for _ in range(3):
