import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = max(0.0, min(1.0, 1.0 - 0.5 * np.sin(2 * np.pi * i / period)))
        h[-1] = 0.5
        return h
    
    def generate_regular(n_points, target_sum):
        h = np.zeros(n_points)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        positions = np.linspace(0, 2, num_ones + 1)[1:-1]
        indices = np.round(positions / dx).astype(int) % n_points
        h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points)
            h[idx] += remaining
        return h
    
    def generate_uniform(n_points, target_sum):
        h = np.zeros(n_points)
        num_ones = int(target_sum)
        idxs = np.random.choice(n_points, size=num_ones, replace=False)
        h[idxs] = 1.0
        if target_sum - num_ones > 0:
            idx = np.random.choice(n_points)
            h[idx] += target_sum - num_ones
        return h
    
    initial_guesses = [initial_guess]
    for period in [4, 5, 6, 8, 10]:
        initial_guesses.append(generate_periodic(period))
    initial_guesses.append(generate_regular(n_points, target_sum))
    initial_guesses.append(generate_uniform(n_points, target_sum))
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x if result.success else None
    
    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None and objective(res) < best_c5:
            best_h, best_c5 = res, objective(res)
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=120,
            maxiter=150,
            mutation=(0.85, 1.05),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            res = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-8}
            )
            if res.success and objective(res.x) < best_c5:
                best_h, best_c5 = res.x, objective(res.x)
    except:
        pass
    
    def refinement(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'rhobeg': 0.1, 'rhoend': 1e-4}
        )
        return result.x if result.success else guess
    
    while time.time() - start_time < budget_s - 10:
        best_h = refinement(best_h)
    
    return best_h, best_c5, n_points
