import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic_h(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            idx = i % period
            if idx == 0:
                h[i] = 1.0
            elif idx == 1:
                h[i] = 0.5
            else:
                h[i] = 0.0
        h[-1] = 0.5
        return h
    
    def generate_binary_h():
        bits = np.random.randint(0, 2, size=n_points)
        bits = bits / np.sum(bits) * target_sum
        bits = np.clip(bits, 0, 1)
        return bits
    
    def generate_dense_h():
        dense_h = np.full(n_points, 0.5)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        dense_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            dense_h[idx] += remaining
        return dense_h
    
    initial_guesses = [initial_guess]
    initial_guesses.append(generate_periodic_h(3))
    initial_guesses.append(generate_periodic_h(5))
    initial_guesses.append(generate_binary_h())
    initial_guesses.append(generate_dense_h())
    initial_guesses.append(generate_periodic_h(2))
    initial_guesses.append(generate_periodic_h(4))
    initial_guesses.append(generate_periodic_h(7))
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}

        result_de = optimize.differential_evolution(

            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,

                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
