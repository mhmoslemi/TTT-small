            pos = (i + 0.5) * spacing
            idx = int(np.round(pos * n_points / 2))
            indices[idx] = True
        h = np.zeros(n_points)
        h[indices] = 1.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_periodic(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_annular(n_points):
        h = np.zeros(n_points)
        idx = np.argsort(np.random.uniform(0, 1, n_points))
        for i in range(n_points // 4):
            h[idx[i]] = 1.0
        for i in range(n_points // 4, 3 * n_points // 4):
            h[idx[i]] = 0.5
        for i in range(3 * n_points // 4, n_points):
            h[idx[i]] = 0.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_random(n_points):
        h = np.zeros(n_points)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h[indices] = 1.0
        if remaining > 1e-8:
            h[np.random.choice(n_points)] += remaining
        return h
    
    initial_guesses = [initial_guess]
    initial_guesses.append(generate_periodic(n_points))
    initial_guesses.append(generate_annular(n_points))
    initial_guesses.append(generate_random(n_points))
    initial_guesses.append(generate_evenly_spaced(n_points))
    
    for _ in range(4):
        initial_guesses.append(generate_random(n_points))
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=300,
            mutation=(0.6, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 30:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
