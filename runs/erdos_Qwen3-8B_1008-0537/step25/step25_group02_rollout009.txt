import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 350, 'ftol': 1e-12, 'eps': 1e-9}
        )
        return result.x if result.success else None
    
    # Generate structured initial guesses
    def generate_periodic(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = max(0.0, min(1.0, 1.0 - 0.5 * np.sin(2 * np.pi * i / period)))
        h[-1] = 0.5
        return h
    
    def generate_triangular(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            idx = int((i * period) % n_points)
            h[i] = max(0.0, min(1.0, 0.5 * (1 + np.cos(2 * np.pi * idx / period))))
        return h
    
    def generate_sawtooth(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            idx = int((i * period) % n_points)
            h[i] = max(0.0, min(1.0, (idx + 1) / period))
        return h
    
    initial_guesses = [initial_guess]

        initial_guesses.append(generate_periodic(period))
        initial_guesses.append(generate_triangular(period))
        initial_guesses.append(generate_sawtooth(period))

    dense_h[indices] = 1.0

        dense_h[idx] += remaining

    # Differential evolution with adjusted parameters

            vectorized=True,
            seed=seed
        )
        if result_de.success:
