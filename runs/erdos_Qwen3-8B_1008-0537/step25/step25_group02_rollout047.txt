    initial_guesses.append(generate_half(n_points))
    initial_guesses.append(generate_single_peak(n_points))
    
    bits = np.random.randint(0, 2, size=n_points)
    bits = bits / np.sum(bits) * target_sum
    bits = np.clip(bits, 0, 1)
    initial_guesses.append(bits)
    
    def generate_dense(n_points):
        dense_h = np.full(n_points, 0.5)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        dense_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            dense_h[idx] += remaining
        return dense_h
    
    initial_guesses.append(generate_dense(n_points))
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=300,
            mutation=(0.8, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'rhobeg': 0.1, 'rhoend': 1e-4}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
