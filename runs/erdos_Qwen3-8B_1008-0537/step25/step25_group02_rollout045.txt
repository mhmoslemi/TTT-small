        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    def generate_periodic(period, n_points, dx):
        h = np.zeros(n_points)
        period_points = np.ceil(n_points / period)
        for i in range(n_points):
            h[i] = max(0.0, min(1.0, 1.0 - 0.5 * np.sin(2 * np.pi * i / period)))
        return h
    
    def generate_binary_even(n_points, target_sum):
        m = int(target_sum)
        positions = np.linspace(0, 2, m + 1)[:-1]
        h = np.zeros(n_points)
        for i in range(n_points):
            dist = np.min(np.abs(positions - i * dx))
            if dist < 1e-6:
                h[i] = 1.0
        remaining = target_sum - np.sum(h)
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            h[idx] += remaining
        elif remaining < 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            h[idx] += remaining
        return h
    
    initial_guesses = [initial_guess]
    for period in [4, 5, 6, 7, 8]:
        initial_guesses.append(generate_periodic(period, n_points, dx))
    
    for _ in range(3):
        bits = np.random.randint(0, 2, size=n_points)
        bits = bits / np.sum(bits) * target_sum
        bits = np.clip(bits, 0, 1)
        initial_guesses.append(bits)
    
    initial_guesses.append(generate_binary_even(n_points, target_sum))
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=150,
            mutation=(0.8, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'rhobeg': 0.1, 'rhoend': 1e-4}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
