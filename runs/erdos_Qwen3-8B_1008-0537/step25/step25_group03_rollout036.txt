import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    def generate_periodic_h(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if (i % period) == 0 else 0.5 if (i % period) == 1 else 0.0
        h[-1] = 0.5
        return h
    
    def generate_binary_h():
        bits = np.random.randint(0, 2, size=n_points)
        bits = bits / np.sum(bits) * target_sum
        bits = np.clip(bits, 0, 1)
        return bits
    
    def generate_dense_h():
        dense_h = np.full(n_points, 0.5)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        dense_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            dense_h[idx] += remaining
        return dense_h
    
    def generate_sparse_h():
        sparse_h = np.zeros(n_points)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        sparse_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            sparse_h[idx] += remaining
        return sparse_h
    
    def generate_varied_h():
        varied_h = np.linspace(0, 1, n_points).flatten()
        varied_h = np.clip(varied_h / np.max(varied_h) * target_sum, 0, 1)

    initial_guesses = [initial_guess]

        )
