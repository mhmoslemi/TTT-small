import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    # Generate structured initial guesses
    initial_guesses = [initial_guess]
    
    # Structured periodic
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        structured_h[i] = 1.0 if i % 5 == 0 else 0.0
    structured_h[-1] = 0.5
    initial_guesses.append(structured_h)
    
    # Structured blocky
    block_size = n_points // 4
    block_h = np.zeros(n_points)
    for i in range(4):
        block_h[i * block_size:(i + 1) * block_size] = 1.0
    block_h[-1] = 0.5
    initial_guesses.append(block_h)
    
    # Structured binary with spacing
    spaced_bits = np.zeros(n_points)
    cnt = 0
    for i in range(n_points):
        if cnt < target_sum:
            spaced_bits[i] = 1.0
            cnt += 1
        else:
            spaced_bits[i] = 0.0
    initial_guesses.append(spaced_bits)
    
    # Mutant of initial guess
    mutant = initial_guess.copy()
    idx = np.random.choice(n_points, size=5, replace=False)
    mutant[idx] = np.random.uniform(0.2, 0.8, size=5)
    initial_guesses.append(mutant)
    
    # Optimize initial guesses
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with optimized parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=30,
            maxiter=150,
            mutation=(0.9, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with L-BFGS-B
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
