import scipy.optimize as optimize
import time

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_periodic_h(period):
        h = np.zeros(n_points)

        h[-1] = 0.5
        return h
    
    def generate_binary_h():

        bits = bits / np.sum(bits) * target_sum

        return bits
    
    def generate_dense_h():
        dense_h = np.full(n_points, 0.5)
        num_ones = int(target_sum)

        dense_h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            dense_h[idx] += remaining
        return dense_h

        return np.clip(h + noise, 0, 1)
    
    def optimize_guess(guess, method='SLSQP'):
        result = optimize.minimize(

            method=method,
            bounds=bounds,

            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
