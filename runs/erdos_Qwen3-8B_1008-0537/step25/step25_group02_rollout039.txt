import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    def generate_periodic(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = max(0.0, min(1.0, 1.0 - 0.5 * np.sin(2 * np.pi * i / period)))
        h[-1] = 0.5
        return h
    
    def generate_steps(num_steps):
        h = np.zeros(n_points)
        step = 2.0 / num_steps
        for i in range(num_steps):
            start = i * step
            end = (i + 1) * step
            start_idx = int(start / dx)
            end_idx = int(end / dx)
            if start_idx < n_points:
                h[start_idx:end_idx] = 1.0
        h = np.clip(h, 0, 1)
        return h
    
    initial_guesses = [initial_guess]
    for period in [4, 5, 6, 8, 10]:
        initial_guesses.append(generate_periodic(period))
    
    # Add step patterns
    for steps in [3, 4, 5, 6, 7]:
        initial_guesses.append(generate_steps(steps))
    
    # Add random binary pattern
    bits = np.random.randint(0, 2, size=n_points)
    bits = bits / np.sum(bits) * target_sum
    bits = np.clip(bits, 0, 1)
    initial_guesses.append(bits)
    
    # Add sparse distribution
    dense_h = np.full(n_points, 0.5)
    num_ones = int(target_sum)

    indices = np.random.choice(n_points, size=num_ones, replace=False)
    dense_h[indices] = 1.0
    if remaining > 0:
        idx = np.random.choice(n_points, size=1, replace=False)

    initial_guesses.append(dense_h)

            best_h = result
    
    # Differential evolution with optimized parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=250,

            recombination=0.85,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',

                constraints=[{'type': 'eq', 'fun': constraint_sum}],
