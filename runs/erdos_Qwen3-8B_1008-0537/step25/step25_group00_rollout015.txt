import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-10}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_block(n_points):
        h = np.zeros(n_points)
        num_blocks = 5
        block_size = n_points // num_blocks
        for i in range(num_blocks):
            h[i * block_size:(i + 1) * block_size] = 1.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_modulated(n_points):
        h = np.zeros(n_points)
        freq = 2
        for i in range(n_points):

        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[np.argmax(h)] += diff
        return h
    
    initial_guesses = [initial_guess]
    initial_guesses.append(generate_block(n_points))
    initial_guesses.append(generate_modulated(n_points))
    initial_guesses.append(generate_periodic(n_points))
    initial_guesses.append(generate_annular(n_points))

        result_de = optimize.differential_evolution(

            vectorized=True,
