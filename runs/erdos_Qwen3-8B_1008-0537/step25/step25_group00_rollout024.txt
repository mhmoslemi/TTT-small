import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)

    def objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_symmetric(h):
        h_sym = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h_sym[i] = h[i]
            else:
                j = n_points - 1 - i
                h_sym[i] = h[j]
        return h_sym

    initial_guesses = [initial_h.copy(), generate_symmetric(initial_h).copy()]

    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective, x0=guess, method='SLSQP',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None

    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None and objective(res) < best_c5:
            best_h, best_c5 = res, objective(res)

    try:
        result_de = optimize.differential_evolution(
            fun=objective, bounds=bounds, popsize=30, maxiter=150,
            mutation=(0.6, 0.9), recombination=0.8, tol=1e-10, vectorized=True
        )
        if result_de.success and objective(result_de.x) < best_c5:
            best_h, best_c5 = result_de.x, objective(result_de.x)
    except:
        pass

    while time.time() - start_time < budget_s - 20:
        try:
            res = optimize.minimize(
                fun=objective, x0=best_h, method='L-BFGS-B',
                bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if res.success and objective(res.x) < best_c5:
                best_h, best_c5 = res.x, objective(res.x)
        except:
            break

    return best_h, best_c5, n_points
