import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_periodic_h(period, phase=0.0):
        h = np.zeros(n_points)
        for i in range(n_points):

        h[-1] = 0.5
        return h

        h = np.zeros(n_points)

    def optimize_guess(guess, method, maxiter=200):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method=method,
            bounds=bounds,
