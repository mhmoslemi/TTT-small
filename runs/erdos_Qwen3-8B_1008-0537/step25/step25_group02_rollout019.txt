import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_periodic(period):
        h = np.zeros(n_points)

            h[i] = 0.5 * (1 + np.cos(2 * np.pi * i / period))
        h = np.clip(h, 0, 1)
        return h
    
    def generate_binary():
        num_ones = int(np.ceil(target_sum))
        if num_ones > n_points:

        remaining = target_sum - num_ones
        if remaining > 0:
            idx = np.random.choice(n_points, size=int(remaining), replace=False)
            h[idx] += 1.0
        h = np.clip(h, 0, 1)
        return h
    
    initial_guesses = [initial_guess]

    def optimize_guess(guess):
        result = optimize.minimize(

            x0=guess,

    for guess in initial_guesses:
