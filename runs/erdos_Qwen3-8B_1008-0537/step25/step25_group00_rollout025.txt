import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    best_h = initial_guess.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_bipartition(n_points):
        h = np.zeros(n_points)
        start = np.random.uniform(0.0, 1.0)
        end = start + 1.0
        for i in range(n_points):
            x = i * dx
            if start <= x < end:
                h[i] = 1.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h

    def generate_uniform(n_points):
        h = np.zeros(n_points)
        ones = int(target_sum)
        h[:ones] = 1.0
        diff = target_sum - ones
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h

    initial_guesses = [initial_guess]
    initial_guesses.append(generate_bipartition(n_points))
    initial_guesses.append(generate_uniform(n_points))
    for _ in range(3):
        initial_guesses.append(np.random.uniform(0, 1, n_points))

    for guess in initial_guesses:
        res = optimize.minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    try:
        result_de = optimize.differential_evolution(
            objective,
            bounds,
            popsize=100,
            maxiter=300,
            mutation=(0.6, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            res_slsqp = optimize.minimize(
                objective,
                result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    except:
        pass

    while time.time() - start_time < budget_s - 30:
        try:
            res = optimize.minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        except:
            break

    return best_h, best_c5, n_points
