        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    initial_guesses = [initial_guess]
    
    # Generate even spaced binary guess
    m = int(n_points / 2)
    if n_points % 2 == 1:
        m += 1
    positions = np.linspace(0, 2, m + 1)[1:-1]
    indices = np.unique(np.round(positions / dx).astype(int))
    initial_guess_even = np.zeros(n_points, dtype=float)
    initial_guess_even[indices] = 1.0
    initial_guesses.append(initial_guess_even)
    
    # Generate periodic guess
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        structured_h[i] = 1.0 if i % 5 == 0 else 0.0
    structured_h[-1] = 0.5
    initial_guesses.append(structured_h)
    
    # Generate blocky guess
    block_size = n_points // 4
    block_h = np.zeros(n_points)
    for i in range(4):
        block_h[i * block_size:(i + 1) * block_size] = 1.0
    block_h[-1] = 0.5
    initial_guesses.append(block_h)
    
    # Generate random mutation guess
    mutant = initial_guess.copy()
    idx = np.random.choice(n_points, size=5, replace=False)
    mutant[idx] = np.random.uniform(0.2, 0.8, size=5)
    initial_guesses.append(mutant)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=40,
            maxiter=120,
            mutation=(0.6, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement with L-BFGS-B
    while time.time() - start_time < budget_s - 40:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
