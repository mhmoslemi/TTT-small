import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    initial_guesses = [initial_guess]

        for i in range(n_points):
            structured_h[i] = 1.0 if i % period == 0 else 0.0
        structured_h[-1] = 0.5
        initial_guesses.append(structured_h)
    
    # Generate binary patterns
    bits = np.random.randint(0, 2, size=n_points)

        block_h[-1] = 0.5
        initial_guesses.append(block_h)
    
    # Generate random mutations
    mutant = initial_guess.copy()
    idx = np.random.choice(n_points, size=5, replace=False)
