        return np.sum(h_values) - target_sum

    def generate_annular(n_points):

        for i in range(n_points // 4):
            h[idx[i]] = 1.0
        for i in range(n_points // 4, 3 * n_points // 4):
            h[idx[i]] = 0.75
        for i in range(3 * n_points // 4, n_points):
            h[idx[i]] = 0.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h

            h[i] = 1.0 if (i % 3) == 0 else 0.0
        diff = target_sum - np.sum(h)

            h[-1] += diff
        return h
