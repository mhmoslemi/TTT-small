            block_h[i * s:(i + 1) * s] = 1.0
        block_h[-1] = 0.5
        initial_guesses.append(block_h)
    
    # Random binary guesses
    for _ in range(2):
        bits = np.random.randint(0, 2, size=n_points)
        bits = np.clip(bits / np.sum(bits) * target_sum, 0, 1)
        initial_guesses.append(bits)
    
    # Mutant
    mutant = initial_guess.copy()
    idx = np.random.choice(n_points, size=5, replace=False)
    mutant[idx] = np.random.uniform(0.2, 0.8, size=5)
    initial_guesses.append(mutant)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    # Differential evolution with adjusted parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=30,
            maxiter=150,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    # Local refinement
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
