import scipy.optimize as optimize
import time

        return np.sum(h) - target_sum

        elif wave_type == 'tri':

        else:
            return np.clip(0.5 + 0.5 * np.cos(2 * np.pi * freq * x), 0, 1)
    
    def generate_sparse(n=3):
        idx = np.random.choice(n_points, size=n, replace=False)
        base = np.zeros(n_points)
        base[idx] = 1.0
        return np.clip(base, 0, 1)
