import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
            )
            return result.x if result.success else None
        except:
            return None
    
    def generate_block(n_points, block_size):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % block_size == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_sinusoidal(n_points, freq):
        h = np.sin(2 * np.pi * freq * np.linspace(0, 2, n_points))
        h = (h + 1) / 2
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    initial_guesses = []
    initial_guesses.append(initial_guess)
    
    for block_size in [3, 4, 5, 7]:
        initial_guesses.append(generate_block(n_points, block_size))
    
    for freq in [0.2, 0.3, 0.4, 0.25, 0.35]:
        initial_guesses.append(generate_sinusoidal(n_points, freq))
    
    for _ in range(3):
        guess = np.random.rand(n_points)
        guess = np.clip(guess, 0, 1)
        guess *= (target_sum / np.sum(guess))

    for guess in initial_guesses:

        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,

            tol=1e-10,

        )
        if result_de.success:
