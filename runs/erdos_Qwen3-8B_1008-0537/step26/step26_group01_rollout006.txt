import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    structured_h = np.zeros(n_points)
    interval = int(np.ceil(2 * n_points / 5))
    for i in range(n_points):
        if i % interval == 0:
            structured_h[i] = 1.0
        elif i % interval == 1:
            structured_h[i] = 0.5
        else:
            structured_h[i] = 0.0
    structured_h[-1] = 0.5
    initial_guesses = [initial_guess, structured_h]
    
    interval2 = int(np.ceil(n_points / 4))
    structured_h2 = np.zeros(n_points)
    for i in range(n_points):
        if i % interval2 == 0:

            structured_h2[i] = 0.5
        else:
            structured_h2[i] = 0.0
    structured_h2[-1] = 0.5
    initial_guesses.append(structured_h2)
    
    for _ in range(2):
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        rand_h = np.zeros(n_points)
        rand_h[indices] = 1.0
        if remaining > 0:

            rand_h[idx] += remaining
        initial_guesses.append(rand_h)

            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,

            tol=1e-9,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,

                bounds=bounds,

                options={'maxiter': 400, 'ftol': 1e-9, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:

    while time.time() - start_time < budget_s - 100:
