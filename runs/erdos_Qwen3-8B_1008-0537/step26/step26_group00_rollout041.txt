import scipy.optimize as optimize

        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_binary(freq, duty, phase=0):

    def generate_symmetric(freq):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points//2:
                h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * freq * i / (n_points//2))

                h[i] = 1 - h[n_points - i - 1]
        return np.clip(h, 0, 1)
    
    initial_guesses = [initial_guess]

    for guess in initial_guesses:
        result = optimize.minimize(
