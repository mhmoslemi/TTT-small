import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_modulated(period, mod):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * mod, 0, 1)
    
    def generate_binary(freq, duty, phase=0):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points) + phase) > 0, 0, 1) * duty
    
    def generate_binary_splitted(freq, duty, phase=0):
        return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points) + phase) > 0) * duty + 
                      (np.sin(2 * np.pi * (freq + 0.5) * np.arange(n_points) + phase) > 0) * (1 - duty), 0, 1)
    
    def generate_sparse_periodic(period, duty):
        idx = np.arange(n_points)
        sel = (idx % period < duty * period)
        return np.clip(np.where(sel, 1, 0), 0, 1)
    
    initial_guesses = [initial_guess]
    for period in [3.5, 4.5, 5.5, 6.5]:
        initial_guesses.append(generate_periodic(period))
    for period, mod in [(4, 0.8), (5, 0.7), (6, 0.6), (7, 0.5)]:
        initial_guesses.append(generate_modulated(period, mod))
    for freq, duty in [(5, 0.3), (6, 0.2), (4, 0.4), (5.5, 0.2)]:
        initial_guesses.append(generate_binary(freq, duty))
    for freq, duty in [(5, 0.25), (6, 0.3), (4, 0.35), (5.5, 0.2)]:
        initial_guesses.append(generate_binary_splitted(freq, duty))
    for period, duty in [(4, 0.4), (5, 0.3), (6, 0.25), (7, 0.2)]:
        initial_guesses.append(generate_sparse_periodic(period, duty))
    initial_guesses.append(1.0 * np.random.rand(n_points))
    initial_guesses.append(0.5 * np.random.rand(n_points))
    initial_guesses.append(0.8 * np.random.rand(n_points))
    
    for guess in initial_guesses:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success:
                curr_c5 = objective(result.x)
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = result.x
        except:
            pass
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=100,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,

                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success:
                curr_c5 = objective(result_slsqp.x)
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 30:
        try:
            result = optimize.minimize(
