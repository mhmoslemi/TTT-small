        L = 1.0 / m
        centers = np.linspace(0.5, 1.5, m)
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for center in centers:
            start = center - L / 2
            end = center + L / 2
            for i in range(n_points):
                if start <= x[i] < end:
                    h[i] = 1.0
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    initial_guesses = [initial_guess]
    
    for block_size in [3, 4, 5]:
        initial_guesses.append(generate_block(n_points, block_size))
    
    for m in [3, 4, 5]:
        initial_guesses.append(generate_balanced(n_points, m))
    
    for freq in [0.2, 0.25, 0.3, 0.35, 0.4]:
        initial_guesses.append(np.sin(2 * np.pi * freq * np.linspace(0, 2, n_points)))
        initial_guesses[-1] = (initial_guesses[-1] + 1) / 2
        current_sum = np.sum(initial_guesses[-1])
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            initial_guesses[-1][-1] += diff
    
    for _ in range(2):
        initial_guesses.append(np.random.rand(n_points))
        initial_guesses[-1] = np.clip(initial_guesses[-1], 0, 1)
        initial_guesses[-1] *= (target_sum / np.sum(initial_guesses[-1]))
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=200,
            mutation=(0.8, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            polish=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
