import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5*np.sin(2*np.pi*np.arange(n_points)/period), 0, 1)
    
    def generate_modulated(period, mod):
        return np.clip(0.5 + 0.5*np.sin(2*np.pi*np.arange(n_points)/period)*mod, 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(0.5 + 0.5*np.sin(2*np.pi*freq*np.arange(n_points)) > 0, 0, 1) * duty
    
    def generate_random_binary():
        n = n_points
        count = int(n / 2)
        idx = np.random.choice(n, size=count, replace=False)
        h = np.zeros(n)
        h[idx] = 1
        if count < n / 2:
            h[np.random.choice(n - count)] += (n/2 - count) / (n - count)
        return np.clip(h, 0, 1)
    
    # Diverse generation
    initial_guesses = [initial_guess]
    for p in [3, 4, 5, 6]: initial_guesses.append(generate_periodic(p))
    for p, m in [(3, 0.6), (4, 0.7), (5, 0.5)]: initial_guesses.append(generate_modulated(p, m))
    for f, d in [(3, 0.4), (4, 0.3), (5, 0.2)]: initial_guesses.append(generate_binary(f, d))
    initial_guesses.append(generate_random_binary())
    
    # Optimization steps
    for guess in initial_guesses:
        try:
            res = optimize.minimize(objective, guess, method='SLSQP', 
                                   bounds=bounds, constraints=[constraint_sum],
                                   options={'maxiter': 150, 'ftol': 1e-12})
            if res.success and objective(res.x) < best_c5:
                best_c5, best_h = objective(res.x), res.x
        except: pass
    
    try:
        res_de = optimize.differential_evolution(objective, bounds, 
                                                 popsize=60, maxiter=80, 
                                                 mutation=(0.75, 0.95), recombination=0.9, tol=1e-9,
                                                 vectorized=True)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP',
                                          bounds=bounds, constraints=[constraint_sum],
                                          options={'maxiter': 200, 'ftol': 1e-12})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5, best_h = objective(res_slsqp.x), res_slsqp.x
    except: pass
    
    # Refinement
    while time.time() - start_time < budget_s - 10:
        try:
            res = optimize.minimize(objective, best_h, method='Powell',
                                   bounds=bounds, constraints=[constraint_sum],
                                   options={'xtol': 1e-9, 'maxiter': 40})
            if res.success and objective(res.x) < best_c5:
                best_c5, best_h = objective(res.x), res.x
        except: break
    
    return best_h, best_c5, n_points
