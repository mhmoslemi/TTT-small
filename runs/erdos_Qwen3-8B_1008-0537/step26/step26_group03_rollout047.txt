            return result.x if result.success else None
        except:
            return None

    initial_guesses = [initial_guess]

    def generate_even_spaced():
        num_ones = 22
        num_half = 1
        positions = np.linspace(0, 2, num_ones + 1)[:-1]
        indices = np.round(positions / dx).astype(int)
        unique_indices = np.unique(indices)
        h = np.zeros(n_points)
        for idx in unique_indices[:num_ones]:
            h[idx] = 1.0
        if num_half > 0:
            for i in range(n_points):
                if h[i] == 0:
                    h[i] = 0.5
                    break
        return h

    initial_guesses.append(generate_even_spaced())

    def generate_block(n_points, block_size):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % block_size == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h

    for block_size in [3, 4, 5, 6, 7]:
        initial_guesses.append(generate_block(n_points, block_size))

    def generate_sinusoidal(n_points, freq):
        h = np.sin(2 * np.pi * freq * np.linspace(0, 2, n_points))
        h = (h + 1) / 2
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h

    for freq in [0.2, 0.25, 0.3, 0.35, 0.4]:
        initial_guesses.append(generate_sinusoidal(n_points, freq))

    for _ in range(3):
        initial_guesses.append(np.random.rand(n_points))
        initial_guesses[-1] = np.clip(initial_guesses[-1], 0, 1)
        initial_guesses[-1] *= (target_sum / np.sum(initial_guesses[-1]))

    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result

    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=200,
            mutation=(0.8, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass

    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break

    return best_h, best_c5, n_points
