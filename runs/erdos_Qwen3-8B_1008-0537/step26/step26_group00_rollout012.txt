import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def generate_periodic(period, amplitude=0.5):
        return np.clip(0.5 + amplitude * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_modulated(period, amp, freq):
        return np.clip(0.5 + amp * np.sin(2 * np.pi * freq * np.arange(n_points) / period), 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty

        idx = np.random.choice(n_points, size=4, replace=False)
        return np.zeros(n_points).astype(float)

    for period in [5, 6, 7, 8, 9]:
        initial_guesses.append(generate_periodic(period))
    for period, amp in [(5, 0.6), (6, 0.5), (7, 0.4)]:
        initial_guesses.append(generate_modulated(period, amp, 1.0))
    for freq, duty in [(6, 0.25), (5, 0.35), (4, 0.45)]:
        initial_guesses.append(generate_binary(freq, duty))
    initial_guesses.append(generate_sparse())
    
    def check_guess(guess):
        h = np.array(guess)
        if np.sum(h) < target_sum - 1e-6 or np.sum(h) > target_sum + 1e-6:
            return np.inf
        return objective(h)
    
    for guess in initial_guesses:
        try:
            res = optimize.minimize(

        result_de = optimize.differential_evolution(

        )
