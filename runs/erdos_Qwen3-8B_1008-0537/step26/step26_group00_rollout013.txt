import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_symmetric(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_sparse(period, density):
        idx = np.arange(n_points)
        np.random.shuffle(idx)
        return np.zeros(n_points) + np.where(idx[:int(density * n_points)] % 2 == 0, 1, 0)
    
    initial_guesses = [initial_guess]
    for period in [4, 5, 6, 7, 9]:
        initial_guesses.append(generate_symmetric(period))
    for period in [4, 5, 6, 7]:
        initial_guesses.append(generate_sparse(period, 0.25))
    initial_guesses.append(generate_sparse(3, 0.33))
    initial_guesses.append(generate_sparse(8, 0.15))
    bits = np.random.rand(n_points)
    bits = np.clip(bits / np.sum(bits) * target_sum, 0, 1)
    initial_guesses.append(bits)
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,

            popsize=40,

        )

                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
