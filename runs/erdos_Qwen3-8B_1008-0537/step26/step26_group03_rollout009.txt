import scipy.optimize as optimize
import time

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_block(n_points, block_size):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % block_size == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_sinusoidal(n_points, freq):
        h = np.sin(2 * np.pi * freq * np.linspace(0, 2, n_points))
        h = (h + 1) / 2
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_random_binary(n_points):
        h = np.random.rand(n_points)
        h = np.round(h)
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h += np.sign(diff)
        return h
    
    def generate_alternating_blocks(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 1.0

        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_spaced_ones(n_points):
        positions = np.linspace(0, 2, n_points)
        ones = np.zeros(n_points)
        ones[np.argsort(positions[:int(n_points/2)])] = 1
        current_sum = np.sum(ones)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            ones[-1] += diff
        return ones
    
    initial_guesses = [initial_guess]

        initial_guesses.append(generate_sinusoidal(n_points, freq))

    def optimize_guess(guess):

            result = optimize.minimize(
