        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros(n_points)
        scale = target_sum / current_sum
        h *= scale
        return np.clip(h, 0, 1)
    
    def generate_modulated(period, mod):
        h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * mod
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros(n_points)
        scale = target_sum / current_sum
        h *= scale
        return np.clip(h, 0, 1)
    
    def generate_binary_pattern(freq, duty):
        idx = np.random.choice(n_points, size=int(duty * n_points), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h
    
    def generate_sparse_pattern():
        idx = np.random.choice(n_points, size=2, replace=False)
        h = np.zeros(n_points)
        h[idx] = 0.5
        return np.clip(h, 0, 1)
    
    initial_guesses = [initial_guess]
    
    for period in [3, 4, 5, 6]:
        initial_guesses.append(generate_periodic(period))
    for period, mod in [(3, 0.7), (4, 0.6), (5, 0.5)]:
        initial_guesses.append(generate_modulated(period, mod))
    for freq, duty in [(4, 0.2), (5, 0.3), (6, 0.4)]:
        initial_guesses.append(generate_binary_pattern(freq, duty))
    
    initial_guesses.append(generate_sparse_pattern())
    bits = np.zeros(n_points)
    for _ in range(int(target_sum)):
        idx = np.random.randint(n_points)
        bits[idx] = 1
    initial_guesses.append(bits)
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if result.success:
            if objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=100,
            mutation=(0.7, 0.9),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 30:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='Powell',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 50}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
