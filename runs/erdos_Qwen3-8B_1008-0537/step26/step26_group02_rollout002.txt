import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(n):
        h = np.zeros(n)
        h[::3] = 1.0
        h[1::3] = 0.5
        h[2::3] = 0.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_block(n):
        h = np.zeros(n)
        b = int(n / 5)
        for i in range(5):
            h[i*b:(i+1)*b] = 1.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_sparse(n):
        h = np.zeros(n)
        idxs = np.random.choice(n, size=int(target_sum), replace=False)
        h[idxs] = 1.0
        if target_sum - int(target_sum) > 1e-8:
            h[np.random.choice(n)] += target_sum - int(target_sum)
        return h
    
    def generate_random(n):
        h = np.zeros(n)
        cnt = int(target_sum)
        used = np.random.choice(n, cnt, replace=False)
        h[used] = 1.0
        rem = target_sum - cnt
        if rem > 1e-8:
            h[np.random.choice(n)] += rem
        return h
    
    initial_guesses = [initial_guess]
    initial_guesses.append(generate_periodic(n_points))
    initial_guesses.append(generate_block(n_points))
    initial_guesses.append(generate_sparse(n_points))
    for _ in range(3):
        initial_guesses.append(generate_random(n_points))
    
    def optimize_guess(guess):
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=[{'type':'eq','fun':constraint_sum}])
        return res.x if res.success else None
    
    for guess in initial_guesses:
        res = optimize_guess(guess)
        if res is not None and objective(res) < best_c5:
            best_c5 = objective(res)
            best_h = res

        de_res = optimize.differential_evolution(objective, bounds, popsize=100, mutation=(0.7, 1.0), recombination=0.8, maxiter=200, tol=1e-10, vectorized=True)
        if de_res.success:
            slsqp_res = optimize.minimize(objective, de_res.x, method='SLSQP', bounds=bounds, constraints=[{'type':'eq','fun':constraint_sum}], options={'maxiter':300})
            if slsqp_res.success and objective(slsqp_res.x) < best_c5:
                best_c5 = objective(slsqp_res.x)
                best_h = slsqp_res.x

        pass
