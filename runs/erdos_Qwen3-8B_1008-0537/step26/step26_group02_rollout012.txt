import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    def generate_block(n_points):
        h = np.zeros(n_points)
        block_size = n_points // 5
        for i in range(5):
            h[i*block_size:(i+1)*block_size] = 1.0
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-8:
            h[-1] += target_sum - current_sum
        return h
    
    def generate_random(n_points):
        h = np.zeros(n_points)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h[indices] = 1.0
        if remaining > 1e-8:
            h[np.random.choice(n_points)] += remaining
        return h
    
    initial_guesses = [initial_guess, generate_block(n_points), generate_random(n_points)]
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    def optimize_de():
        try:
            result = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=40,
                maxiter=100,
                mutation=(0.8, 1.0),
                recombination=0.8,
                tol=1e-10,
                vectorized=True
            )
            if result.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10}
                )
                if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                    best_c5 = objective(result_slsqp.x)
                    best_h = result_slsqp.x
        except:
            pass
    
    optimize_de()
    
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
