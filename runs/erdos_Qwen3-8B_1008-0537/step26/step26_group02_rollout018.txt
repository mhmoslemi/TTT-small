import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
        )
        return result.x if result.success else None
    
    initial_guesses = [initial_guess]
    
    def generate_periodic(n_points):
        h = np.zeros(n_points)
        h[::3] = 1.0
        h[1::3] = 0.5
        h[2::3] = 0.0
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    initial_guesses.append(generate_periodic(n_points))

        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    initial_guesses.append(generate_block(n_points))
    
    def generate_alternating(n_points):
        h = np.zeros(n_points)

            if i % 4 == 0:
                h[i] = 1.0
            elif i % 4 == 1:
                h[i] = 0.5
            elif i % 4 == 2:
                h[i] = 0.0
            else:
                h[i] = 1.0
        current_sum = np.sum(h)

        return h
    initial_guesses.append(generate_alternating(n_points))
    
    def generate_random(n_points):
        h = np.zeros(n_points)
        num_ones = int(target_sum)

        result_de = optimize.differential_evolution(
