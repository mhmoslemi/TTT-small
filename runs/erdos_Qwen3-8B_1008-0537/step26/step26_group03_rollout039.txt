import scipy.optimize as optimize
import time

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    def generate_block(n_points, block_size, phase=0):
        h = np.zeros(n_points)
        for i in range(n_points):
            if (i + phase) % block_size == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_sinusoidal(n_points, freq, phase=0):

        h = (h + 1) / 2
        h += phase
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            h[-1] += diff
        return h
    
    def generate_random(n_points):
        h = np.random.rand(n_points)

        return h
    
    initial_guesses = [initial_guess]

    def optimize_guess(guess):

            result = optimize.minimize(
