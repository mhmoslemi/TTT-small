import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_modulated(period, mod):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * mod, 0, 1)
    
    def generate_binary_pattern(freq, duty):
        return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points)) > 0) * duty, 0, 1)
    
    def generate_random_binary():
        cnt = int(target_sum)
        idx = np.random.choice(n_points, size=cnt, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return h
    
    initial_guesses = [initial_guess]
    for period in [4, 5, 6, 7, 8]:
        initial_guesses.append(generate_periodic(period))
    for period, mod in [(4, 0.7), (5, 0.6), (6, 0.5), (7, 0.4)]:
        initial_guesses.append(generate_modulated(period, mod))
    for freq, duty in [(6, 0.2), (5, 0.3), (4, 0.4), (7, 0.1)]:
        initial_guesses.append(generate_binary_pattern(freq, duty))
    initial_guesses.append(generate_random_binary())
    
    for guess in initial_guesses:
        try:
            res = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',

                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if res.success and objective(res.x) < best_c5:

        except:

        result = optimize.differential_evolution(
