import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    alternate_h = np.zeros(n_points)
    alternate_h[:n_points//2] = 1.0

    def objective(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    best_alt_h = alternate_h.copy()
    best_alt_c5 = objective(best_alt_h)
    if best_alt_c5 < best_c5:
        best_h, best_c5 = best_alt_h, best_alt_c5

    n_iter = 0
    max_iter = 100
    noise_scale = 0.05
    decay = 0.995

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        n_iter += 1
        noise = np.random.normal(0, noise_scale, n_points)
        perturbed = np.clip(best_h + noise, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        noise_scale *= decay

    return best_h, best_c5, n_points
