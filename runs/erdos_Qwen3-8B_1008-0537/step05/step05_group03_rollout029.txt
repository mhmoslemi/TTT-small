            fun=objective,
            bounds=bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            options={'mutation': (0.5, 1), 'recombination': 0.8}
        )
    except:
        result = {'x': initial_guess}

    best_h = np.clip(result['x'], 0, 1)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        scale_factor = target_sum / sum_h
        best_h = best_h * scale_factor

    # Final adjustment to ensure sum is exactly target_sum
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        scale_factor = target_sum / sum_h
        best_h = best_h * scale_factor

    # Recompute C5 after adjustment
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        best_h = initial_guess
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    return best_h, best_c5, n_points
