import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Multi-start with diverse perturbations
    max_iter = 40
    noise_scale = 0.02
    perturbation_seed = seed
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Initialize with diverse perturbations
        np.random.seed(perturbation_seed)
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)
        perturbation_seed += 1
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Adjust perturbation parameters dynamically
        noise_scale *= 0.94
    
    return best_h, best_c5, n_points
