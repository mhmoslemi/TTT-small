        x0=initial_guess,
        vectorize=True
    )
    
    best_h = np.zeros(n_points)
    best_h[:n_points-1] = result.x
    best_h[-1] = target_sum - np.sum(result.x)
    h_values = best_h
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    best_c5 = np.max(corr) * dx
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h_values, objective(initial_h_values), n_points
    
    return best_h, best_c5, n_points
