import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h0 = np.array(initial_h_values, dtype=float)
    
    def objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h0.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Use multiple diverse optimization approaches
    methods = ['L-BFGS-B', 'SLSQP', 'COBYLA']
    for meth in methods:
        if time.time() - start_time > budget_s:
            break
        # Create perturbed version with larger initial exploration
        perturb = np.random.normal(0, 0.08, size=n_points)
        guess = np.clip(h0 + perturb, 0, 1)
        result = optimize.minimize(
            objective, guess, method=meth,
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x

    # Give a final fine-tuning pass with COBYLA
    if time.time() - start_time < budget_s - 2:
        result = optimize.minimize(
            objective, best_h, method='COBYLA',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x

    return best_h, best_c5, n_points
