        h = np.zeros(n_points)
        for i in range(n_points):
            x_i = i * dx
            for pos in positions:
                if abs(x_i - pos) < dx / 2:
                    h[i] = 1
                    break
        current_sum = np.sum(h) * dx
        diff = required_sum - current_sum
        if diff != 0:
            h[-1] += diff / dx
            h[-1] = np.clip(h[-1], 0, 1)
        return h
    
    even_initial_guess = create_even_guess(n_points)
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    even_best_h = even_initial_guess.copy()
    even_best_c5 = objective(even_best_h)
    
    if even_best_c5 < best_c5:
        best_h = even_best_h
        best_c5 = even_best_c5
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.003 + 0.002 * np.random.rand()
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, n_points), 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
