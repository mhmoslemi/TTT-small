    current_sum = np.sum(initial_guess) * dx
    diff = target_sum - current_sum
    if diff != 0:
        initial_guess[-1] += diff / dx
        initial_guess[-1] = np.clip(initial_guess[-1], 0, 1)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-5,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
