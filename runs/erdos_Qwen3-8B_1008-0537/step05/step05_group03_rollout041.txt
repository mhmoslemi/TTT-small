import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)

    def objective(x):
        x = np.array(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)

    max_iter = 30
    noise_scale = 0.05

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)

        res = optimize.minimize(
            objective, perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )

        if res.success:
            curr_c5 = objective(res.x)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = res.x

        noise_scale *= 0.95

    return best_h, best_c5, n_points
