import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import os
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e5 * (np.sum(h) - target_sum) ** 2
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.9,
        maxiter=100,
        tol=1e-8,
        args=(),
        callback=None,
        disp=False,
        polish=True,
        polish_x0=[best_h],
        polish_maxiter=50
    )
    
    final_h = result.x
    final_c5 = objective(final_h)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    # Final refinement with SLSQP
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],

    )

    final_c5 = objective(final_h)
