    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in initial_guesses:
        for _ in range(2):  # Two rounds of optimization
            if time.time() - start_time > budget_s:
                break
            # Adaptive noise scaling to balance exploration and exploitation
            perturb = np.random.normal(0, 0.05, size=n_points)
            perturbed_guess = np.clip(guess + perturb, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
            # Reduce noise scale for further refinement
            perturb *= 0.95
            guess = result.x

    return best_h, best_c5, n_points
