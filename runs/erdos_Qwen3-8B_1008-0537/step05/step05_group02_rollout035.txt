    start_time = time.time()

    # Use trust-constr for better constrained optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='trust-constr',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={
            'maxiter': 200,
            'tol': 1e-10,
            'verbose': 0
        }
    )

    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
