import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = float('inf')
    start_time = time.time()

    # Generate multiple starting points
    def generate_uniform():
        return np.full(n_points, 0.5)

    def generate_sparse():
        ones = np.random.choice(n_points, size=n_points//2, replace=False)
        sparse = np.zeros(n_points)
        sparse[ones] = 1.0
        return sparse

    start_points = [
        initial_guess,
        generate_uniform(),
        generate_sparse()
    ]

    for start_idx in range(len(start_points)):
        start_point = start_points[start_idx]
        for _ in range(20):
            if time.time() - start_time > budget_s:
                break
            noise_scale = 0.05 + 0.05 * np.random.rand()
            perturbed_guess = start_point + noise_scale * np.random.randn(n_points)
            perturbed_guess = np.clip(perturbed_guess, 0, 1)
            perturbed_guess = perturbed_guess * (target_sum / np.sum(perturbed_guess))
            
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.01 * (1 - _ / 10)
        perturbed_guess = best_h + noise_scale * np.random.randn(n_points)
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        perturbed_guess = perturbed_guess * (target_sum / np.sum(perturbed_guess))
        
        result = optimize.minimize(
            fun=objective,
