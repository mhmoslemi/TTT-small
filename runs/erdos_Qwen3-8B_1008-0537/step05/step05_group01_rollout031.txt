import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    initial_guesses = []
    initial_guesses.append(np.array(initial_h_values, dtype=float))
    
    alternate_initial_guess = np.zeros(n_points, dtype=float)
    alternate_initial_guess[:n_points//2] = 1
    initial_guesses.append(alternate_initial_guess)
    
    random_initial_guess = np.random.rand(n_points)
    random_initial_guess[-1] = (target_sum - np.sum(random_initial_guess)) / dx
    random_initial_guess = np.clip(random_initial_guess, 0, 1)
    initial_guesses.append(random_initial_guess)
    
    best_h = None
    best_c5 = float('inf')
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
