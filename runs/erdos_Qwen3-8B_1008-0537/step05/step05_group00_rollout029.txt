            h[-1] = np.clip(h[-1], 0, 1)
        return h

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    even_initial = create_even_guess(n_points)
    even_c5 = objective(even_initial)
    if even_c5 < best_c5:
        best_h, best_c5 = even_initial, even_c5

    for _ in range(15):
        if time.time() - start_time > budget_s:
            break
        np.random.seed(int(time.time()))
        perturb = np.random.normal(0, 0.01, n_points)
        guess = np.clip(best_h + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_h, best_c5 = result.x, current_c5

    return best_h, best_c5, n_points
