        h = np.zeros(n_points)

                    h[i] = 1
                    break
        current_sum = np.sum(h) * dx
        diff = required_sum - current_sum
        if diff != 0:

        return h

    bounds = [(0.0, 1.0) for _ in range(n_points)]
