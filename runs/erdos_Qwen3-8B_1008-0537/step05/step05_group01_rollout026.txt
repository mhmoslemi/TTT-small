    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    if best_h is not None:
        for _ in range(2):
            if time.time() - start_time > budget_s:
                break
            noise_scale = 0.005 * np.random.rand()
            perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return best_h, best_c5, n_points
