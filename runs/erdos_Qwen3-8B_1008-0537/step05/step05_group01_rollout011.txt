import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def create_random_guess():
        guess = np.random.rand(n_points)
        guess /= np.sum(guess)
        guess *= target_sum
        return guess
    
    random_guess = create_random_guess()
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    if np.max(objective(random_guess)) < best_c5:
        best_h = random_guess.copy()
        best_c5 = objective(best_h)
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.01 + 0.005 * np.random.rand()
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
