import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def create_even_guess(n_points):
        m = n_points // 2
        required_sum = n_points / 2
        dx = 2.0 / n_points
        positions = np.linspace(0, 2, m + 1)[1:-1]
        h = np.zeros(n_points)
        for i in range(n_points):
            x_i = i * dx
            for pos in positions:
                if abs(x_i - pos) < dx / 2:
                    h[i] = 1
                    break
        current_sum = np.sum(h) * dx
        diff = required_sum - current_sum
        if diff != 0:
            h[-1] += diff / dx
            h[-1] = np.clip(h[-1], 0, 1)
        return h
    
    def random_initial_guess(n_points):
        m = int(n_points / 2)
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, m, replace=False)
        h[indices] = 1.0
        if n_points % 2 == 1:
            rand_idx = np.random.randint(n_points)
            if h[rand_idx] == 1:
                h[rand_idx] = 0.5
            else:
                h[rand_idx] = 0.5
        return h
    
    even_initial_guess = create_even_guess(n_points)
    random_initial_guess = random_initial_guess(n_points)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    even_best_h = even_initial_guess.copy()
    even_best_c5 = objective(even_best_h)
    random_best_h = random_initial_guess.copy()
    random_best_c5 = objective(random_best_h)
    
    if even_best_c5 < best_c5:
        best_h = even_best_h
        best_c5 = even_best_c5
    if random_best_c5 < best_c5:

        best_c5 = random_best_c5
    
    for _ in range(20):

        noise_scale = 0.005 + 0.005 * np.random.rand()

        result = optimize.minimize(
