import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values, dtype=float)
    h_alt = np.zeros(n_points)
    h_alt[:n_points//2] = 1.0
    h_random = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h_random[indices] = 1.0

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    init_guesses = [h_initial, h_alt, h_random]
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    for guess in init_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.1:
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            noise_scale = 0.003 + 0.001 * np.random.rand()
            perturb = np.random.normal(0, noise_scale, size=n_points)
            perturbed_guess = np.clip(best_h + perturb, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if result.success:

                if current_c5 < best_c5:
