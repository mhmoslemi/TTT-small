            current_h = np.zeros(n_points)
            current_h[:-1] = result.x
            last_current = target_sum - np.sum(result.x)
            last_current = np.clip(last_current, 0, 1)
            current_h[-1] = last_current
            current_c5 = objective(result.x)

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                initial_reduced = result.x  # Update for next iteration

    return best_h, best_c5, n_points
