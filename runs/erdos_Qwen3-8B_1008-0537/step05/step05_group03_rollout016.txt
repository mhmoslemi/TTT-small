    result = optimize.basinhopping(
        objective,
        initial_guess,
        niter=20,
        T=1.0,
        stepsize=0.5,
        minimizer_kwargs=minimizer_kwargs,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check remaining time for additional refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        result = optimize.basinhopping(
            objective,
            best_h,
            niter=10,
            T=0.5,
            stepsize=0.2,
            minimizer_kwargs=minimizer_kwargs,
            disp=False
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
