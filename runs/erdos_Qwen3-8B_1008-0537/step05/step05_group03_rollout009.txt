        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    def generate_structured_guess(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        length_segment = int(0.5 / dx)
        start = 0
        end = start + length_segment
        if end > n_points:
            end = n_points
        h[start:end] = 1.0
        start = n_points - length_segment
        end = n_points
        if start < 0:
            start = 0
        h[start:end] = 1.0
        return h
    
    # Try a few different initial guesses
    initial_guesses = [initial_guess, generate_structured_guess(n_points)]
    
    # For each initial guess, run optimization with L-BFGS-B
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
