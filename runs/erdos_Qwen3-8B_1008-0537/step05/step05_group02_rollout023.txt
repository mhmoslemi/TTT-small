import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Direct optimization with COBYLA
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'disp': False}
    )
    if result.success:
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    # Perturbation loop with adaptive steps
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.05 * (1 - _ / 20)
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(initial_guess + perturb, 0, 1)

        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
