import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Basinhopping with SLSQP as local optimizer
    def local_minimizer(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x
        return x0
    
    # Control basinhopping iterations within budget
    max_iter = 50
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        result = optimize.basinhopping(
            fun=objective,
            x0=best_h,
            niter=1,
            T=1.0,
            stepsize=0.1,
            minimizer_kwargs={
                'method': 'SLSQP',
                'bounds': bounds,
                'constraints': [{'type': 'eq', 'fun': constraint_sum}],
                'options': {'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            },
            callback=None
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
