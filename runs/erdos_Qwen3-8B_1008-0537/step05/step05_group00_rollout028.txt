import scipy.optimize as optimize

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is provided
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Create initial guesses
    def create_even_guess(n_points):
        m = n_points // 2
        positions = np.linspace(0, 2, m + 1)[1:-1]
        h = np.zeros(n_points)
        for i in range(n_points):
            x_i = i * dx
            for pos in positions:
                if abs(x_i - pos) < dx / 2:
                    h[i] = 1
                    break
        current_sum = np.sum(h) * dx
        diff = target_sum - current_sum
        if diff != 0:
            h[-1] += diff / dx
            h[-1] = np.clip(h[-1], 0, 1)
        return h

    even_initial_guess = create_even_guess(n_points)

    initial_guess = np.array(initial_h_values, dtype=float)

    # Generate random initial guess
    np.random.seed(seed)
    random_initial_guess = np.random.rand(n_points)
    random_initial_guess = random_initial_guess / np.sum(random_initial_guess) * target_sum
    random_initial_guess = np.clip(random_initial_guess, 0, 1)

    # Define objective function
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint function
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # List of initial guesses
    initial_guesses = [initial_guess, even_initial_guess, random_initial_guess]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    for guess in initial_guesses:
        # Run optimization from this guess
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points

                if abs(x_i - pos) < dx / 2:
                    h[i] = 1
                    break
        current_sum = np.sum(h) * dx
        diff = target_sum - current_sum
        if diff != 0:
            h[-1] += diff / dx
            h[-1] = np.clip(h[-1], 0, 1)
        return h
    
    even_initial_guess = create_even_guess(n_points)
    initial_guess = np.array(initial_h_values, dtype=float)
    
    np.random.seed(seed)
    random_initial_guess = np.random.rand(n_points)
    random_initial_guess = random_initial_guess / np.sum(random_initial_guess) * target_sum
    random_initial_guess = np.clip(random_initial_guess, 0, 1)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    initial_guesses = [initial_guess, even_initial_guess, random_initial_guess]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    for guess in initial_guesses:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
