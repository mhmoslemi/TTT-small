import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def create_even_guess(n_points):
        m = n_points // 2
        required_sum = n_points / 2
        dx = 2.0 / n_points
        positions = np.linspace(0, 2, m + 1)[1:-1]
        h = np.zeros(n_points)
        for i in range(n_points):
            x_i = i * dx
            for pos in positions:
                if abs(x_i - pos) < dx / 2:
                    h[i] = 1
                    break
        current_sum = np.sum(h) * dx
        diff = required_sum - current_sum
        if diff != 0:
            h[-1] += diff / dx
            h[-1] = np.clip(h[-1], 0, 1)
        return h
    
    def generate_random_guess(n_points):
        required_sum = n_points / 2
        m = int(required_sum)
        remaining = required_sum - m
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, m, replace=False)
        h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, 1)
            h[idx] += remaining
            h[idx] = np.clip(h[idx], 0, 1)
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    even_guess = create_even_guess(n_points)
    even_c5 = objective(even_guess)
    if even_c5 < best_c5:
        best_h = even_guess
        best_c5 = even_c5
    
    random_guess = generate_random_guess(n_points)
    random_c5 = objective(random_guess)
    if random_c5 < best_c5:
        best_h = random_guess
        best_c5 = random_c5
    
    start_time = time.time()
    for _ in range(15):
        if time.time() - start_time > budget_s - 0.1:
            break
        noise_scale = 0.003 + 0.005 * np.random.rand()
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, n_points), 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
