    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        # Define tighter bounds for better convergence
        tight_bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Create a perturbed guess with adaptive noise
        perturb = np.random.normal(0, 0.01 + 0.005 * np.random.rand(), size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=tight_bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
