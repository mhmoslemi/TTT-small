import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initial population
    pop_size = 20
    population = []
    initial_h = np.array(initial_h_values, dtype=float)
    population.append(initial_h)
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        h_rand = np.clip(h_rand, 0, 1)
        population.append(h_rand)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            a_idx, b_idx, c_idx = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[a_idx], population[b_idx], population[c_idx]
            F = np.random.rand() * 0.8 + 0.2
            mutant = a + F * (b - c)
            trial = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros(n_points)
            else:
                scale = target_sum / trial_sum
                trial = trial * scale
            trial = np.clip(trial, 0, 1)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
            new_population.append(trial)
        population = new_population
    
    return best_h, best_c5, n_points
