    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Evaluate alternate guess
    current_c5 = objective(alt_guess)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = alt_guess
    
    # Evaluate random guess
    current_c5 = objective(rand_guess)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = rand_guess
    
    # Additional optimization steps
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        # Perturb current best
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
