import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Initial guesses
    initial_guess = np.array(initial_h_values, dtype=float)
    alt_guess = np.zeros(n_points)
    alt_guess[:n_points//2] = 1
    uniform_guess = np.full(n_points, 0.5)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def evaluate_guess(g):
        return objective(g)
    
    # Evaluate initial candidates
    candidates = [initial_guess, alt_guess, uniform_guess]
    best_h = min(candidates, key=evaluate_guess)
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        # Perturb with cosine-based structure
        noise_scale = 0.05 + 0.01 * np.random.rand()
        perturb = np.sin(np.linspace(0, 2*np.pi, n_points)) * noise_scale
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        
        # Optimize with COBYLA
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'disp': False}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
