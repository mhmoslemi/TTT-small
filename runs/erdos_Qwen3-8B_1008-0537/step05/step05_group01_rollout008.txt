import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h0 = np.array(initial_h_values, dtype=float)
    
    def generate_binary_initial(n, num_ones):
        indices = np.random.choice(n, size=num_ones, replace=False)
        return np.zeros(n, dtype=float).astype(float) + (indices[:, np.newaxis] == np.arange(n)[:, np.newaxis]).astype(float)
    
    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def constraint(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = h0.copy()
    best_c5 = objective(best_h)
    
    # Multiple random binary initial guesses
    for _ in range(2):
        h = generate_binary_initial(n_points, n_points // 2)
        sol = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
        if sol.success:
            c = objective(sol.x)
            if c < best_c5:
                best_c5 = c
                best_h = sol.x
    
    # Progressive perturbations and iterations
    for i in range(50):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.01 * (1.0 - i / 50.0)
        perturb = np.random.normal(0, noise_scale, n_points)
        h = np.clip(best_h + perturb, 0, 1)
        
        sol = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}], 
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9})
        if sol.success:
            c = objective(sol.x)
            if c < best_c5:
                best_c5 = c
                best_h = sol.x
    
    return best_h, best_c5, n_points
