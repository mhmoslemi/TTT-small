import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        sum_diff = np.sum(h) - target_sum
        return c5 + 1e4 * np.abs(sum_diff)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        best_h = result.x
        best_c5 = np.max(np.correlate(result.x, 1 - result.x, mode='full')) * dx
        return best_h, best_c5, n_points

    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return best_h, best_c5, n_points
