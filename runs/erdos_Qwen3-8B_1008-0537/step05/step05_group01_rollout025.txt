            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.003, size=n_points)
            perturbed = np.clip(best_h + perturb, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return best_h, best_c5, n_points
