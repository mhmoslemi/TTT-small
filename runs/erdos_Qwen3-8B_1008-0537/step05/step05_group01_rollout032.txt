import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def generate_binary_guess(n_points):
        n_ones = n_points // 2
        if n_points % 2:
            n_ones += 1
        h = np.zeros(n_points)
        h[:n_ones] = 1.0
        return h
    
    def generate_random_guess(n_points):
        h_rand = np.random.rand(n_points)
        sum_rand = np.sum(h_rand)
        if sum_rand == 0:
            scale = 1.0
        else:
            scale = target_sum / sum_rand
        h_scaled = h_rand * scale
        h_clipped = np.clip(h_scaled, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped != 0:
            scale_adj = target_sum / sum_clipped
