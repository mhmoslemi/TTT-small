        dx = 2.0 / n_points

        h = np.zeros(n_points)
        for i in range(n_points):
            x_i = i * dx

                if abs(x_i - pos) < dx / 2:
                    h[i] = 1
                    break
        current_sum = np.sum(h) * dx
        diff = required_sum - current_sum
        if diff != 0:
            h[-1] += diff / dx
            h[-1] = np.clip(h[-1], 0, 1)
        return h
    
    def create_random_guess(n_points):
        h = np.random.rand(n_points)

        return h
