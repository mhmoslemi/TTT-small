            x_i = i * dx
            for pos in positions:
                if abs(x_i - pos) < dx / 2:
                    h[i] = 1
                    break
        current_sum = np.sum(h) * dx
        diff = target_sum - current_sum
        if diff != 0:
            h[-1] += diff / dx
            h[-1] = np.clip(h[-1], 0, 1)
        return h

    best_h = even_initial_guess.copy()

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
