    if n_points % 2 == 0:
        symmetric_guess = np.zeros(n_points)

            symmetric_guess[n_points - 1 - i] = initial_guess[i]

        symmetric_guess = np.zeros(n_points)
        for i in range(n_points // 2):
            symmetric_guess[i] = initial_guess[i]
            symmetric_guess[n_points - 1 - i] = initial_guess[i]
        symmetric_guess[n_points // 2] = initial_guess[n_points // 2]
    initial_guess = symmetric_guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    # Use COBYLA for better constraint handling
    for _ in range(30):

        # Random perturbation
