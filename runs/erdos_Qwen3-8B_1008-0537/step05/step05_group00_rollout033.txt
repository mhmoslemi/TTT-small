        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate
    
    start_time = time.time()
    best_h = best_h.copy()
    best_c5 = objective(best_h)
    max_iter = 50
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.005 * np.random.rand()
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
