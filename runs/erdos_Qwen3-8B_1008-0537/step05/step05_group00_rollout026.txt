            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success:
            curr = res.x
            curr_c5 = objective(curr)
            if curr_c5 < best_c5:
                best_h, best_c5 = curr, curr_c5

    return best_h, best_c5, n_points
