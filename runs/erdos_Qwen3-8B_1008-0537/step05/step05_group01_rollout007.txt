import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def create_pattern(n, periodicity=4):
        idx = np.arange(n)
        return np.sin(2 * np.pi * idx / periodicity) * 0.5 + 0.5

    def create_alternate_guess(n):
        return np.zeros(n, dtype=float)

    alternate_initial_guess = create_alternate_guess(n_points)
    alternate_initial_guess[:n_points//2] = 1

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    best_alt_h = alternate_initial_guess.copy()
    best_alt_c5 = objective(best_alt_h)
    
    if best_alt_c5 < best_c5:
        best_h = best_alt_h
        best_c5 = best_alt_c5
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.01 + 0.005 * np.random.rand()
        perturb = np.random.normal(0, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                print(f"New best C5: {best_c5}")
    
    return best_h, best_c5, n_points
