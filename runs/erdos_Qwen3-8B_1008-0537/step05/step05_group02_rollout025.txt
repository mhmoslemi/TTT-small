```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values, dtype=float)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.inf
        h_normalized = h * (target_sum / h_sum)
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        args=(),
        popsize=15,
        maxiter=100,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.7,
        polish=True,
        disp=False
    )

    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
```
