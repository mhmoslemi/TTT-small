import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_periodic(n):
        return np.array([1 if i % 4 < 2 else 0 for i in range(n)])
    
    initial_guesses = [
        initial_guess,
        np.zeros(n_points) + 0.5,
        generate_periodic(n_points),
        np.array([1 if i % 3 < 1 else 0 for i in range(n_points)]),
        np.array([1 if i % 5 < 2 else 0 for i in range(n_points)])
    ]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for guess in initial_guesses:
        for perturb_scale in [0.05, 0.01, 0.005]:
            perturbed = np.clip(guess + np.random.normal(0, perturb_scale, n_points), 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
