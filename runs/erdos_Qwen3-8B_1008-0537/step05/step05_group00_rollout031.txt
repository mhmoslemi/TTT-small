        idx = np.random.choice(n_points, num_ones, replace=False)
        h[idx] = 1
        current_sum = np.sum(h)
        diff = required_sum - current_sum
        if diff != 0:
            h[-1] += diff
            h[-1] = np.clip(h[-1], 0, 1)
        return h
    
    random_initial_guess = random_initial_guess(n_points)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    def optimize_from_guess(guess):
        perturb = np.random.normal(0, 0.003, size=n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            return result.x, current_c5
        return guess, best_c5
    
    candidates = [initial_guess, even_initial_guess, random_initial_guess]
    for guess in candidates:
        h, c5 = optimize_from_guess(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    for _ in range(15):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.002, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
