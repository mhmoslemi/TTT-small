    def generate_tri(f, d):
        t = np.linspace(0, 2, n_points)
        y = 0.5 + 0.5 * np.sin(2 * np.pi * f * t)
        y = np.clip(y, 0, 1)
        y *= d
        y += (1 - d) * np.random.rand(n_points)
        return y
    
    guesses = [np.copy(h)]
    for f in [0.5, 0.75, 1.0, 1.25, 1.5, 1.75]:
        guesses.append(generate_bin(f, 0.5))
        guesses.append(generate_bin(f, 0.3))
        guesses.append(generate_bin(f, 0.2))
        guesses.append(generate_bin(f, 0.1))
    for d in [0.8, 0.6, 0.4, 0.3, 0.25, 0.2]:
        guesses.append(generate_tri(1.0, d))
        guesses.append(generate_tri(1.5, d))
        guesses.append(generate_tri(0.5, d))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 40:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=50,
                                             mutation=(0.6, 0.95), recombination=0.9, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='Powell', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xtol': 1e-8, 'ftol': 1e-8, 'maxiter': 50})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
