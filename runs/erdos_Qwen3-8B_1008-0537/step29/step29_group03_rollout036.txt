import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(h_init, sigma=0.1):
        return np.clip(h_init + sigma * np.random.randn(n_points), 0, 1)
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10})
        return res.x if res.success else h_init
    
    def refine_with_powell(h_init):
        res = optimize.minimize(objective, h_init, method='Powell', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 40})
        return res.x if res.success else h_init
    
    def optimize_with_de(h_init):
        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=50,
                                             mutation=(0.5, 0.8), recombination=0.8, tol=1e-9)
        return res.x if res.success else h_init
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        best_h = refine_with_slsqp(best_h)

        best_h = optimize_with_de(best_h)

    best_h = refine_with_powell(best_h)
    final_c5 = objective(best_h)
