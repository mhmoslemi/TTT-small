import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(period, duty):
        t = np.linspace(0, 2, n_points)
        y = 0.5 + 0.5 * np.sin(2 * np.pi * t / period)
        y = np.clip(y, 0, 1)
        y *= duty
        y += (1 - duty) * np.random.rand(n_points)
        return y
    
    def generate_binary(freq, duty):
        t = np.linspace(0, 2, n_points)
        y = np.sin(2 * np.pi * freq * t) > 0
        y = np.clip(y.astype(float) * duty, 0, 1)
        return y
    
    guesses = [np.copy(h)]
    for p in [4, 5, 6, 7, 8, 9]:
        guesses.append(generate_wave(p, 0.5))
    for f, d in [(6, 0.2), (5, 0.3), (4, 0.4), (7, 0.1), (8, 0.25), (9, 0.3)]:
        guesses.append(generate_binary(f, d))
    
    for guess in guesses:
        res_slsqp = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                    options={'maxiter': 400, 'ftol': 1e-12, 'eps': 1e-10})
        if res_slsqp.success and objective(res_slsqp.x) < best_c5:
            best_c5 = objective(res_slsqp.x)
            best_h = res_slsqp.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res_de = optimize.differential_evolution(objective, bounds, popsize=80, maxiter=30,
                                               mutation=(0.5, 0.9), recombination=0.7, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                        constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                        options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        res_powell = optimize.minimize(objective, best_h, method='Powell', bounds=bounds,
                                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                    options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 30})
        if res_powell.success and objective(res_powell.x) < best_c5:
            best_c5 = objective(res_powell.x)
            best_h = res_powell.x
    
    return best_h, best_c5, n_points
