import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_even():
        indices = np.arange(n_points)
        selected = indices[::2][:23]  # 23 ones for 45 samples
        selected = np.sort(np.concatenate([selected, indices[1::2][:22]]))
        return np.zeros(n_points), np.zeros(n_points)
    
    def generate_binary_random():
        indices = np.random.permutation(n_points)[:23]
        h_rand = np.zeros(n_points)
        h_rand[indices] = 1.0
        return h_rand, np.zeros(n_points)
    
    def generate_periodic(period):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * x / period), 0, 1), np.zeros(n_points)
    
    def generate_tent():
        x = np.linspace(0, 2, n_points)
        return np.clip(1.0 - np.abs(x - 1.0) / 1.0, 0, 1), np.zeros(n_points)
    
    def generate_triangular():
        x = np.linspace(0, 2, n_points)

    def perturb(guess):
        noise = np.random.normal(0, 0.05, size=n_points)
        return np.clip(guess + noise, 0, 1)
    
    guesses = [np.array(initial_h_values)]
    guesses += [generate_periodic(p)[0] for p in [4, 5, 6, 7, 8, 9, 10]]
    guesses += [generate_binary_random()[0], generate_binary_even()[0]]

    guesses.append(np.random.rand(n_points))
    
    for guess in guesses:
