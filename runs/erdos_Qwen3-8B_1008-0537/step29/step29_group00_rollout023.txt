import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary(freq, duty):
        t = np.linspace(0, 2, n_points)
        y = np.sin(2 * np.pi * freq * t) > 0
        y = np.clip(y.astype(float) * duty, 0, 1)
        return y
    
    def generate_offset(freq, duty, offset):
        t = np.linspace(0, 2, n_points)
        y = np.sin(2 * np.pi * freq * t + offset) > 0
        y = np.clip(y.astype(float) * duty, 0, 1)
        return y
    
    guesses = [np.copy(h)]
    for f in [2, 3, 4, 5]:
        guesses.append(generate_binary(f, 0.5))
    for f, d in [(2, 0.2), (3, 0.3), (4, 0.5), (5, 0.1)]:
        guesses.append(generate_binary(f, d))
    for f, d in [(2, 0.5), (3, 0.5), (4, 2.0), (5, 1.0)]:
        guesses.append(generate_offset(f, d, np.pi/2))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):

        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=30,
