import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(n_points, m=5, phase=0.0):
        dx = 2.0 / n_points
        l = 1.0 / m
        s = (2 - m * l) / (m - 1)
        intervals = []
        x = 0
        for _ in range(m):
            start = x
            end = x + l
            intervals.append((start, end))
            x = end + s
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for start, end in intervals:

                    break
        h = np.roll(h, int(phase / dx))
        return h
    
    def generate_binary(freq, duty):
        t = np.linspace(0, 2, n_points)
        y = np.sin(2 * np.pi * freq * t) > 0
        y = np.clip(y.astype(float) * duty, 0, 1)
        return y
    
    guesses = [np.copy(h)]
    for m in [3, 4, 5, 6]:
        guesses.append(generate_periodic(n_points, m))
    for f in [3, 4, 5, 6]:
        guesses.append(generate_binary(f, 0.3))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
