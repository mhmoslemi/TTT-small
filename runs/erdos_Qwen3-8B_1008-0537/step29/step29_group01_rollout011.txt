import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)

    def generate_pulses(freq, amp, offset):
        t = np.linspace(0, 1, n_points)
        return np.clip(amp * np.abs(np.sin(2 * np.pi * freq * t + offset)), 0, 1)
    
    def generate_piecewise(freq, duty):
        t = np.linspace(0, 1, n_points)
        return np.clip(np.abs(np.sin(2 * np.pi * freq * t)) * duty, 0, 1)
    
    def generate_fade(freq, width):
        t = np.linspace(0, 1, n_points)
        wave = np.clip(2 * (1 - np.abs(2 * freq * t - 1)) * width, 0, 1)
        return wave
    
    guesses = [generate_pulses(f, a, o) for f, a, o in [(1.5, 0.5, 0.3), (1.2, 0.6, 0.7), (1.8, 0.4, 0.1)]]
    guesses += [generate_piecewise(f, d) for f, d in [(2.0, 0.3), (1.5, 0.5), (1.3, 0.2)]]
    guesses += [generate_fade(f, w) for f, w in [(2.0, 0.3), (1.5, 0.4), (1.2, 0.25)]]
    guesses.append(np.random.rand(n_points))
    guesses.append(h)
    guesses.append((1 - np.abs(np.linspace(-1, 1, n_points))) * 0.5)
    
    for guess in guesses:
