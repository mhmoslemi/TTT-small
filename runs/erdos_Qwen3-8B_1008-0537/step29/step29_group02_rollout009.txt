import scipy.optimize as optimize
import time

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

    best_c5 = objective(best_h)

        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_binary(f, d):
        return np.clip(np.sign(np.sin(2 * np.pi * f * np.arange(n_points))) * d, 0, 1)
    
    def perturb(guess):
        noise = np.random.normal(0, 0.05, size=n_points)
        return np.clip(guess + noise, 0, 1)

    guesses += [perturb(g) for g in guesses[:5]]
    guesses.append(np.random.rand(n_points))
    
    for guess in guesses:
