import time

        return np.max(corr) * dx

    best_c5 = objective(best_h)

        theta = 2 * np.pi * np.arange(n_points) / period
        return np.clip(0.5 + 0.5 * np.sin(theta), 0, 1)
    
    def generate_binary(freq, amp):
        wave = np.sin(2 * np.pi * freq * np.arange(n_points))
        return np.clip(amp * wave, 0, 1)

        noise = np.random.normal(0, 0.05, size=n_points)
        return np.clip(guess + noise, 0, 1)
