import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Initial SLSQP optimization
    res = optimize.minimize(objective, h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': constraint_sum}],
                           options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9})
    if res.success:
        best_h = res.x
        best_c5 = objective(best_h)
    
    # Differential evolution refinement
    if time.time() - start_time < budget_s - 20:
        res_de = optimize.differential_evolution(
            objective, bounds, popsize=30, maxiter=50, mutation=(0.6, 0.9),
            recombination=0.8, tol=1e-9, disp=False)
        if res_de.success:
            res_slsqp = optimize.minimize(
                objective, res_de.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_h, best_c5 = res_slsqp.x, objective(res_slsqp.x)
    
    # Powell's method for final refinement
    if time.time() - start_time < budget_s - 5:
        res_powell = optimize.minimize(
            objective, best_h, method='Powell', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 30})
        if res_powell.success and objective(res_powell.x) < best_c5:
            best_h, best_c5 = res_powell.x, objective(res_powell.x)
    
    return best_h, best_c5, n_points
