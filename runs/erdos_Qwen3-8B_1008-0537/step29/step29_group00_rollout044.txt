    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_binary_freq(freq, duty):
        t = np.linspace(0, 2, n_points)
        y = np.sin(2 * np.pi * freq * t) > 0
        y = np.clip(y.astype(float) * duty, 0, 1)
        return y
    
    def generate_pulse(duty, spacing):
        t = np.linspace(0, 2, n_points)
        y = np.zeros_like(t)
        for i in range(n_points):
            if i % spacing == 0:
                y[i] = duty
        return y
    
    initial_guesses = [np.copy(h)]
    for freq in [3, 4, 5, 6]:
        initial_guesses.append(generate_binary_freq(freq, 0.3))
    for duty in [0.2, 0.4, 0.6, 0.8]:
        initial_guesses.append(generate_pulse(duty, 5))
    for duty in [0.2, 0.4, 0.6, 0.8]:
        initial_guesses.append(generate_pulse(duty, 7))
    
    for guess in initial_guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=100,
                                             mutation=(0.5, 0.9), recombination=0.7, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='Powell', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 50})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
