import scipy.optimize as optimize

        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = objective(best_h)

        t = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * t / period)
        wave = (wave + 1) / 2
        wave = np.clip(wave * duty, 0, 1)
        return wave
    
    def generate_binary(freq, duty):
        t = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * t) > 0
        wave = np.clip(wave.astype(float) * duty, 0, 1)
        return wave
    
    def generate_random_pulses(num_pulses, duty):

            start = max(0, idx - width // 2)
            end = min(n_points, idx + width // 2)
            pulses[start:end] += duty
        return np.clip(pulses, 0, 1)

    for guess in guesses:
