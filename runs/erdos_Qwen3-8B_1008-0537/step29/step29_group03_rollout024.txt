import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty
    
    def generate_triangular():
        return np.clip(0.5 * (1 + np.cos(2 * np.pi * np.arange(n_points) / 9)), 0, 1)
    
    def generate_random_perturbation():
        return np.clip(h + np.random.uniform(-0.2, 0.2, n_points), 0, 1)
    
    guesses = [generate_periodic(p) for p in [2, 3, 5, 7, 9, 12, 15]]
    guesses += [generate_binary(f, d) for f, d in [(2, 0.3), (3, 0.5), (4, 0.2), (5, 0.4), (6, 0.1), (7, 0.3), (8, 0.2)]]
    guesses += [generate_triangular(), generate_random_perturbation()]
    guesses.append(initial_h_values.copy())
    guesses.append(np.random.rand(n_points) * 0.5 + 0.25)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 400, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 25:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=150, maxiter=40,

                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_neldermead(h_init):
