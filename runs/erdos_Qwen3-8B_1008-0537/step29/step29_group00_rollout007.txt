import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values
    n_points = len(h_initial)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(h_initial, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(period, duty, noise=0.0):
        t = np.linspace(0, 2, n_points)
        y = 0.5 + 0.5 * np.sin(2 * np.pi * t / period)
        y = np.clip(y * duty + (1 - duty) * noise, 0, 1)
        return y
    
    def generate_binary(f, d):
        t = np.linspace(0, 2, n_points)
        y = np.sin(2 * np.pi * f * t) > 0
        y = np.clip(y.astype(float) * d, 0, 1)
        return y
    
    guesses = [np.copy(h)]
    for p in [3, 4, 5, 6, 7, 8, 9]:
        guesses.append(generate_wave(p, 0.5))
    for f, d in [(3, 0.2), (4, 0.3), (5, 0.4), (6, 0.1), (7, 0.2), (8, 0.3), (9, 0.2)]:
        guesses.append(generate_binary(f, d))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 50:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=50,
                                             mutation=(0.5, 0.9), recombination=0.7, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 20:

        res = optimize.minimize(objective, best_h, method='Powell', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 50})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
