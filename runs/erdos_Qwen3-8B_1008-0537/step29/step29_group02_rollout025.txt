import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(periods, phase):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / periods + phase), 0, 1)
    
    def generate_binary(frequencies, depths):
        return np.clip(np.sign(np.sin(2 * np.pi * frequencies * np.arange(n_points))) * depths, 0, 1)
    
    def perturb(guess):
        noise = np.random.normal(0, 0.08, size=n_points)
        return np.clip(guess + noise, 0, 1)
    
    guesses = [np.array(initial_h_values)]
    guesses += [generate_periodic(p, np.random.uniform(0, 2*np.pi)) for p in [3,4,5,6,7,8,9]]
    guesses += [generate_binary(f, d) for f, d in zip(np.random.uniform(0.2, 0.8, 5), np.random.uniform(0.1, 0.5, 5))]
    guesses += [perturb(g) for g in guesses[-5:]]
    guesses.append(np.random.rand(n_points))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
