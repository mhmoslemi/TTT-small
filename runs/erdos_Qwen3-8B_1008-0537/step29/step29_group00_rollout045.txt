def generate_pulses(n_points, m_pulses):
    pulse_length = 1.0 / m_pulses
    spacing = (2.0 - m_pulses * pulse_length) / (m_pulses + 1)
    centers = np.linspace(spacing/2, 2.0 - spacing/2, m_pulses)
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for center in centers:
        start = max(0, center - pulse_length/2)
        end = min(2.0, center + pulse_length/2)
        start_idx = int(start / dx)
        end_idx = int(end / dx)
        if start_idx < end_idx:
