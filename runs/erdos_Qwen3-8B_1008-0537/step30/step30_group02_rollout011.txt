import time

        return np.max(corr) * dx

        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 * (1 + np.sin(2 * np.pi * x / period)), 0, 1)
    
    guesses = [generate_initial(p) for p in [3, 4, 5, 6, 7, 8, 9, 10]]
