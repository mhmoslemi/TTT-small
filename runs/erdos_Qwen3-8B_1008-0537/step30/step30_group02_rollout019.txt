import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 * (1 + np.sin(2 * np.pi * x / period)), 0, 1)
    
    guesses = [generate_initial(p) for p in [4, 5, 6, 7, 8]]
    guesses.append(np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]))
    guesses.append(h)
    guesses.append(np.random.uniform(0, 1, n_points))
    guesses.append(np.array([0.5 + 0.5 * np.cos(np.pi * i / (n_points // 2)) for i in range(n_points)]))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:

            best_h = res.x
