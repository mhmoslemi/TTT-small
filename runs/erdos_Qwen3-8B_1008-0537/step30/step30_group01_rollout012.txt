        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

    best_c5 = objective(best_h)

        h = np.zeros(n_points)
        for i in range(n_points // 2 + 1):

        for i in range(1, n_points // 2 + 1):
            h[n_points - i] = h[i]
        return np.clip(h, 0, 1)
    
    def generate_sine(f):

    for guess in guesses:
