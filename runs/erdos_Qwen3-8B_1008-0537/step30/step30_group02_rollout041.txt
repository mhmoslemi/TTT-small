import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_square_wave(n_points):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 * (1 + np.sign(np.sin(2 * np.pi * (x - 1.0) / 1.0))), 0, 1)
    
    def generate_binary(n_points):
        h = np.random.choice([0, 1], size=n_points)
        diff = target_sum - np.sum(h)
        if diff > 0:
            h[-int(diff)] = 1
        elif diff < 0:
            h[-int(diff)] = 0
        return h
    
    guesses = [np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)])]
    guesses.append(generate_square_wave(n_points))
    guesses.append(generate_binary(n_points))
    guesses.append(h)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:

        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=40,

        if res.success:
