import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_initial(period, base=0.5):
        x = np.linspace(0, 2, n_points)
        return np.clip(base + 0.5 * (1 - np.cos(2 * np.pi * x / period)), 0, 1)
    
    def generate_square(period):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 * (1 + np.sign(np.sin(2 * np.pi * x / period))), 0, 1)
    
    guesses = [
        generate_initial(4), generate_initial(5), generate_initial(6), generate_initial(7), generate_initial(8),
        generate_square(4), generate_square(5), generate_square(6), generate_square(7), generate_square(8),
        np.zeros(n_points) + 0.5
    ]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10, 'disp': False})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=40,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9, disp=False)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10, 'disp': False})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_powell(h_init):
        res = optimize.minimize(objective, h_init, method='Powell', bounds=bounds,
