import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(f):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x)
        return np.clip(h, 0, 1)
    
    def generate_mixed_sines(f_list):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for f in f_list:
            h += 0.5 * np.sin(2 * np.pi * f * x)
        h = np.clip((h / len(f_list)) + 0.5, 0, 1)
        return h
    
    def generate_step_mixed(widths):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if (i % (n_points // 3)) < widths[i % (n_points // 3)] else 0.0
        return h
    
    guesses = [np.array(initial_h_values)]
    guesses += [generate_sine(f) for f in [0.5, 0.25, 0.333, 0.2, 0.4, 0.1, 0.15, 0.6]]
    guesses += [generate_mixed_sines([0.5, 0.333, 0.25])]
    guesses += [generate_mixed_sines([0.666, 0.333, 0.5])]
    guesses += [generate_step_mixed([1, 2, 1, 1, 1])]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(4):
        if time.time() - start_time >= budget_s - 20:

        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=50,

            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
