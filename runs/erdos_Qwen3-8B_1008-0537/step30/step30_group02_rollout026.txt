import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_initial(period):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 * (1 + np.sin(2 * np.pi * x / period)), 0, 1)
    
    guesses = [generate_initial(p) for p in [4, 5, 6, 7, 8]]
    guesses.append(np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]))
    guesses.append(h)
    
    # Additional initial guesses
    random_guess = np.random.rand(n_points)
    random_guess = random_guess * (target_sum / random_guess.sum())
    guesses.append(random_guess)
    
    binary_guess = np.zeros(n_points)
    binary_guess[:n_points // 2] = 1.0
    binary_guess = binary_guess * (target_sum / binary_guess.sum())
    guesses.append(binary_guess)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=60,

        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})

                best_h = res_slsqp.x
    
    def refine_with_powell(h_init):
