    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = objective(best_h)

        h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x)
        return np.clip(h, 0, 1)

    def generate_symmetric():
        h = np.zeros(n_points)
        for i in range(n_points // 2 + 1):

        for i in range(1, n_points // 2 + 1):

    for guess in guesses:
