import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_freq(freq, duty=0.5):
        t = np.linspace(0, 1, n_points)
        wave = 0.5 + 0.5 * np.sin(2 * np.pi * freq * t)
        wave = np.clip(wave * duty, 0, 1)
        return wave
    
    def generate_binary(freq, duty=0.5):
        t = np.linspace(0, 1, n_points)
        wave = np.clip(np.sin(2 * np.pi * freq * t) > 0, 0, 1) * duty
        return wave
    
    def generate_sawtooth(freq, duty=0.5):
        t = np.linspace(0, 1, n_points)
        wave = 2 * duty * (1 - np.abs(2 * freq * t - 1))
        return np.clip(wave, 0, 1)
    
    def generate_triangular(freq, amp=0.5):
        t = np.linspace(0, 1, n_points)
        wave = amp * (2 * t - 1) * np.abs(1 - 2 * np.abs(2 * freq * t - 1))
        return np.clip(wave, 0, 1)
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    guesses = [generate_periodic(p) for p in [3, 4, 5, 6, 7, 8]]
    guesses += [generate_binary(f, d) for f, d in [(3, 0.2), (4, 0.4), (5, 0.3), (6, 0.2), (7, 0.1)]]
    guesses += [generate_sawtooth(f, d) for f, d in [(3, 0.3), (4, 0.25), (5, 0.2), (6, 0.15), (7, 0.1)]]
    guesses += [generate_triangular(f, a) for f, a in [(3, 0.6), (4, 0.5), (5, 0.4), (6, 0.3), (7, 0.2)]]
    guesses += [generate_freq(f, d) for f, d in [(3, 0.2), (4, 0.4), (5, 0.3), (6, 0.2), (7, 0.1)]]
    guesses.append(np.random.rand(n_points))
    guesses.append(h)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=80, maxiter=40,

        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,

                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
