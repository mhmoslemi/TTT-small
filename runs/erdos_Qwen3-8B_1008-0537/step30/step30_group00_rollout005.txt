        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = objective(best_h)

    def generate_periodic(period, amp):
        return np.clip(0.5 + amp * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)

    def generate_binary(freq, duty):
        return np.clip(np.sign(np.sin(2 * np.pi * freq * np.arange(n_points))) * duty, 0, 1)

    for p in [4, 5, 6, 7, 8, 9]:
        best_h = optimize.minimize(objective, generate_periodic(p, 0.5), method='SLSQP',

                                   bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
