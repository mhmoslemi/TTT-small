import time

    best_c5 = objective(best_h)

        if shape == 'sine':
            return np.clip(0.5 * (1 + np.sin(2 * np.pi * x / period)), 0, 1)
        elif shape == 'square':
            return np.clip(0.5 * (1 + np.sign(np.sin(2 * np.pi * x / period))), 0, 1)
        elif shape == 'triangle':
            return np.clip((2 / np.pi) * np.arctan(np.sin(2 * np.pi * x / period)) + 0.5, 0, 1)
        return np.clip(0.5 * (1 + np.sin(2 * np.pi * x / period)), 0, 1)
    
    guesses = [generate_initial(p, 'sine') for p in [4, 5, 6, 7, 8]]
    guesses.append(generate_initial(3, 'square'))
    guesses.append(generate_initial(3, 'triangle'))
    guesses.append(np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]))
    guesses.append(h)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
