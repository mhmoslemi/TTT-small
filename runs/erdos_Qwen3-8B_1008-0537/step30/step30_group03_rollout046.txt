    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Differential evolution with aggressive parameters
    if time.time() - start_time < budget_s - 30:
        try:
            res_de = optimize.differential_evolution(
                objective, bounds, strategy='best1bin',
                popsize=100, maxiter=50, mutation=(0.7, 0.9),
                recombination=0.8, tol=1e-9, workers=-1
            )
            if res_de.success:
                best_h = res_de.x
                best_c5 = objective(best_h)
        except:
            pass
    
    # SLSQP refinement with tight tolerances
    if time.time() - start_time < budget_s - 20:
        try:
            res_slsqp = optimize.minimize(
                objective, best_h, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if res_slsqp.success:
                best_h = res_slsqp.x
                best_c5 = objective(best_h)
        except:
            pass
    
    # Powell refinement for final polishing
    if time.time() - start_time < budget_s - 10:
        try:
            res_powell = optimize.minimize(
                objective, best_h, method='Powell', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 50}
            )
            if res_powell.success:
                best_h = res_powell.x
                best_c5 = objective(best_h)
        except:
            pass
    
    return best_h, best_c5, n_points
