import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_binary(f, d):
        return np.clip(np.sign(np.sin(2 * np.pi * f * np.arange(n_points))) * d, 0, 1)
    
    def perturb(guess):
        noise = np.random.normal(0, 0.05, size=n_points)
        return np.clip(guess + noise, 0, 1)
    
    initial_guesses = [np.array(initial_h_values)]
    
    for p in [4, 5, 6, 7, 8, 9]:
        initial_guesses.append(generate_periodic(p))
    
    for f, d in [(4, 0.2), (5, 0.3), (6, 0.4), (7, 0.1), (8, 0.25), (9, 0.3), (10, 0.25)]:
        initial_guesses.append(generate_binary(f, d))
    
    for g in initial_guesses[:5]:
        initial_guesses.append(perturb(g))
    
    initial_guesses.append(np.random.rand(n_points))
    
    for guess in initial_guesses:
        try:
            res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        except:
            pass
    
    for _ in range(4):
        if time.time() - start_time >= budget_s - 30:

        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=50,
