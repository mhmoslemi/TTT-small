import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_periodic(period, amps):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * amps, 0, 1)
    
    def generate_binary(freq, amp):
        return np.clip(np.sign(np.sin(2 * np.pi * freq * np.arange(n_points))) * amp, 0, 1)
    
    initial_guesses = [np.array(initial_h_values)]
    
    for p in range(3, 11):

            initial_guesses.append(generate_binary(freq, amp))
    
    for perturb in [0.02, 0.04, 0.06]:
        for g in initial_guesses[:5]:
            initial_guesses.append(np.clip(g + np.random.normal(0, perturb, n_points), 0, 1))
    
    initial_guesses = list(set(map(tuple, initial_guesses)))[:20]
    initial_guesses = [np.array(g) for g in initial_guesses]
