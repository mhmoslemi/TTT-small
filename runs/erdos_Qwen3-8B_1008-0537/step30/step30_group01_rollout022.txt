import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(f):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x)
        return np.clip(h, 0, 1)
    
    def generate_symmetric():
        h = np.zeros(n_points)
        for i in range(n_points // 2 + 1):
            h[i] = 0.5 * (1 + np.sin(2 * np.pi * i / (n_points - 1)))
        for i in range(1, n_points // 2 + 1):
            h[n_points - i] = h[i]
        return np.clip(h, 0, 1)
    
    def generate_alternating(m):
        h = np.zeros(n_points)
        step = n_points // m
        for i in range(n_points):
            if (i // step) % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        return np.clip(h, 0, 1)
    
    guesses = [np.array(initial_h_values)]
    guesses += [generate_sine(f) for f in [0.5, 0.25, 0.125, 0.333, 0.666]]
    guesses += [generate_symmetric()]
    guesses += [generate_alternating(5), generate_alternating(10)]
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-11, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):

        res = optimize.differential_evolution(objective, bounds, popsize=50, maxiter=40,
