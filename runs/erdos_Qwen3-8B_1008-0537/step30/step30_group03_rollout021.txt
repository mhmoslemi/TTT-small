import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        x = np.linspace(0, 1, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * x * period), 0, 1)
    
    def generate_binary(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave > 0) * duty, 0, 1)
    
    def generate_triangular(freq, amp):
        x = np.linspace(0, 1, n_points)
        wave = amp * (2 * x - 1) * np.abs(1 - 2 * np.abs(2 * freq * x - 1))
        return np.clip(wave, 0, 1)
    
    def generate_sawtooth(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = 2 * duty * (1 - np.abs(2 * freq * x - 1))
        return np.clip(wave, 0, 1)
    
    guesses = [generate_periodic(p) for p in [3, 4, 5, 6, 7, 8, 9, 10]]
    guesses += [generate_binary(f, d) for f, d in [(3, 0.25), (4, 0.5), (5, 0.3), (6, 0.25)]]
    guesses += [generate_triangular(f, a) for f, a in [(3, 0.6), (4, 0.5), (5, 0.4), (6, 0.3)]]
    guesses += [generate_sawtooth(f, d) for f, d in [(3, 0.3), (4, 0.2), (5, 0.25), (6, 0.3)]]
    guesses.append(np.random.rand(n_points) * 0.5 + 0.5)
    guesses.append(h)
    
    for guess in guesses:
        try:
            res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
            if res.success and objective(res.x) < best_c5:
                best_c5 = objective(res.x)
                best_h = res.x
        except:
            pass
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:

        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=30,
