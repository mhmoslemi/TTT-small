import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_symmetric(f):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * f * (x - 1)), 0, 1)
    
    def generate_two_freqs(f1, f2):
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 + 0.5 * (np.sin(2 * np.pi * f1 * (x - 1)) + np.sin(2 * np.pi * f2 * (x - 1))), 0, 1)
    
    guesses = [np.array(initial_h_values)]
    for f in [0.5, 0.333, 0.25, 0.2, 0.166, 0.1, 0.083]:
        guesses.append(generate_symmetric(f))
    for f1, f2 in [(0.5, 0.25), (0.5, 0.333), (0.333, 0.25), (0.25, 0.166), (0.2, 0.166), (0.166, 0.1)]:
        guesses.append(generate_two_freqs(f1, f2))
    guesses.append(np.clip(0.5 * np.ones(n_points), 0, 1))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=40,
                                             mutation=(0.7, 0.95), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 70, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 30, 'catol': 1e-9, 'ftol': 1e-9})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
