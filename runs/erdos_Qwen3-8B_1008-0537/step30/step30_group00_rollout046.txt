import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        x = np.arange(n_points) * dx
        values = 0.5 + 0.5 * np.sin(2 * np.pi * x / period)
        sum_values = np.sum(values)
        if sum_values != 0:
            values *= (n_points / 2) / sum_values
        return np.clip(values, 0, 1)
    
    def generate_binary(f, d):
        x = np.arange(n_points) * dx
        try:
            values = d * np.sign(np.sin(2 * np.pi * f * x))
        except:

        values = np.clip(values, 0, 1)
        sum_values = np.sum(values)
        if sum_values != 0:
            values *= (n_points / 2) / sum_values
        return values
    
    def perturb(guess):
        noise = np.random.normal(0, 0.02, size=n_points)
        return np.clip(guess + noise, 0, 1)
    
    initial_guesses = [np.array(initial_h_values)]
    
    for p in [3, 4, 5, 6, 7, 8, 9]:
        initial_guesses.append(generate_periodic(p))
    
    for f, d in [(3, 0.3), (4, 0.25), (5, 0.2), (6, 0.15), (7, 0.1), (8, 0.05)]:

    for g in initial_guesses[:5]:
        initial_guesses.append(perturb(g))
    
    initial_guesses.append(np.random.rand(n_points))
    
    for guess in initial_guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success and objective(res.x) < best_c5:
