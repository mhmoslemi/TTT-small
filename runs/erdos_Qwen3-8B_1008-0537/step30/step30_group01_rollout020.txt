import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_sine(f, shift=0.0):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x + shift)
        return np.clip(h, 0, 1)
    
    def generate_symmetric(n_segments=15):
        h = np.zeros(n_points)
        for i in range(n_segments):
            mid = i * (n_points // n_segments)
            h[mid] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_segments)
        return np.clip(h, 0, 1)
    
    def generate_square(freq):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sign(np.sin(2 * np.pi * freq * x))
        return np.clip(h, 0, 1)
    
    guesses = [np.array(initial_h_values)]
    guesses += [generate_sine(f, shift=0.3) for f in [0.5, 0.25, 0.125, 0.333, 0.666, 0.2, 0.4]]
    guesses += [generate_sine(f, shift=0.7) for f in [0.5, 0.25, 0.125, 0.333, 0.666]]
    guesses += [generate_symmetric(n_segments=10), generate_symmetric(n_segments=20)]

    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9})
        if res.success and objective(res.x) < best_c5:
