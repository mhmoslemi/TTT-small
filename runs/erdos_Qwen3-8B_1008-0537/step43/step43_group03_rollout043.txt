    def refine(r):
        res = optimize.minimize(obj, r, method='Nelder-Mead', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200})
        return res.x if res.success else r

    for _ in range(3):
        if time.time() - start_time > budget_s - 5:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
