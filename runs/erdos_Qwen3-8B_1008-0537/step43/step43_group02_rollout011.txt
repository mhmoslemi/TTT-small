import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_block(freq, duty, n_blocks):
        x = np.linspace(0, 1, n_points)
        wave = np.zeros_like(x)
        for i in range(n_blocks):
            start = i * (1 / n_blocks) * 2
            end = (i + 1) * (1 / n_blocks) * 2
            wave[(x >= start) & (x < end)] = 1 if i % 2 == 0 else 0
        return np.clip(wave * duty, 0, 1)

    def gen_repeating(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 7, 8, 10, 12, 14, 16]:
        guesses.append(gen_repeating(freq, 0.4))
        guesses.append(gen_block(freq, 0.3, 4))
    for _ in range(3):
        guesses.append(np.random.rand(n_points))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-8})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(2):
        if time.time() - start_time > budget_s - 10:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=50,
                                             mutation=(0.7, 0.95), recombination=0.8, tol=1e-8)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-9})
        return res.x if res.success else r

            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
