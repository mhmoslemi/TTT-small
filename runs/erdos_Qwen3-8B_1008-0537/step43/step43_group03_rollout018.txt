import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_staircase(frac, width):
        x = np.linspace(0, 2, n_points)
        h = np.zeros_like(x)
        for i in range(n_points):
            if (x[i] < 0.5 * frac * width or
                x[i] > 0.5 * (1 + 2 * (frac - 1)) * width):
                h[i] = 1.0
        return h

    def gen_random(frac=0.5):
        r = np.random.rand(n_points)
        return np.clip((r - np.mean(r)) / np.std(r) * frac + 0.5, 0, 1)

    guesses = [h]
    for freq in [2, 3, 5, 7]:
        guesses.append(gen_pulse(freq, 0.5))
        guesses.append(gen_staircase(1.0, 0.3))
        guesses.append(gen_staircase(2.0, 0.4))
    for _ in range(2):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(2):
        if time.time() - start_time > budget_s - 15:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=30,
                                             mutation=(0.6, 0.95), recombination=0.8, tol=1e-8)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
