import scipy.optimize as optimize
import time

    def obj(h_vals):

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty):
        x = np.linspace(0, 2, n_points)
        t = (x - 0.5) * 2
        wave = np.sin(2 * np.pi * freq * t)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 2, n_points)
        t = x / 2
        return np.clip(1 - np.abs(2 * freq * t - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 2, n_points)
        t = x / 2
        return 1 - np.abs(2 * freq * t - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 10):
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.3, 0.4, 0.5, 0.6]:

    for duty in [0.3, 0.4, 0.5, 0.6]:

    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
