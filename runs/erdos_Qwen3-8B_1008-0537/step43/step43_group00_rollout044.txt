    np.random.seed(seed)
    n_points = 30
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Generate initial guesses
    h_binary = np.zeros(n_points, dtype=float)
    for i in range(n_points):
        if i % 2 == 0:
            h_binary[i] = 1
    initial_guesses = [h_binary]

    # Generate pulse and triangle patterns
    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    for freq in [3,5,7]:
        initial_guesses.append(gen_pulse(freq, 0.5))

    # Generate triangles
    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty
    for freq in [3,5,7]:
        initial_guesses.append(gen_triangle(freq, 0.5))

    # Add some random guesses
    def gen_random():
        h = np.random.rand(n_points)
        h[...] = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
        return h
    for _ in range(3):
        initial_guesses.append(gen_random())

    # Optimization functions
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h_binary.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    # Use SLSQP on initial guesses
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-11})
        return res.x if res.success else g

    for g in initial_guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Differential evolution
    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=100,

        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 20:

        if c < best_c5:

import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pattern(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.sin(2 * np.pi * freq * x) + 1) * duty / 2, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return np.clip(h, 0, 1)

    guesses = [h]
    for freq in [3, 4, 5]:
        guesses.append(gen_pattern(freq, 0.5))
    for freq in [3, 4]:
        guesses.append(gen_triangle(freq, 0.4))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=80,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(5):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
