        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_staircase(frac, width):
        x = np.linspace(0, 2, n_points)
        h = np.zeros_like(x)
        for i in range(n_points):
            if (x[i] < 0.5 * frac * width or
                x[i] > 0.5 * (1 + 2 * (frac - 1)) * width):
                h[i] = 1.0
        return h

    def gen_segments(m, n_points):
        l = 1.0 / m
        s = (2.0 - 1.0) / (m - 1)
        x = np.linspace(0, 2, n_points)
        h = np.zeros_like(x)
        current = 0.0
        for _ in range(m):
            start = current
            end = current + l
            mask = (x >= start) & (x <= end)
            h[mask] = 1.0
            current = end + s
        return h

    def gen_random(frac=0.5):
        r = np.random.rand(n_points)
        return np.clip((r - np.mean(r)) / np.std(r) * frac + 0.5, 0, 1)

    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 7, 8]:
        guesses.append(gen_pulse(freq, 0.5))
        guesses.append(gen_staircase(1.0, 0.3))
        guesses.append(gen_staircase(2.0, 0.4))
    for _ in range(3):
        guesses.append(gen_random())
    for m in [5, 4, 3]:
        guesses.append(gen_segments(m, n_points))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],

        return res.x if res.success else g

        c = obj(h_opt)
        if c < best_c5:

        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=50,

        c = obj(h_opt)
        if c < best_c5:

    def refine(r):
        res = optimize.minimize(obj, r, method='COBYLA', bounds=bounds,

    return best_h, best_c5, n_points
