import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    # Generate diverse initial guesses
    guesses = [h]
    for freq in [3, 4, 5]:
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        guess = np.clip((wave + 1) * 0.6 / 2, 0, 1)
        guesses.append(guess)
    for frac in [0.6, 0.7, 0.8]:
        x = np.linspace(0, 2, n_points)
        h = np.zeros_like(x)
        for i in range(n_points):
            if x[i] < frac or x[i] > 2 - frac:
                h[i] = 1.0
        guesses.append(h)
    for _ in range(2):
        guesses.append(np.clip((np.random.rand(n_points) - 0.5) * 0.6 + 0.5, 0, 1))

    # SLSQP refinement
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:

        c = obj(h_opt)
        if c < best_c5:

    def de_run():

        c = obj(h_opt)
        return h_opt, c

    for _ in range(3):

        h_opt, c = de_run()
