        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

    best_c5 = obj(best_h)

        x = np.linspace(0, 2, n_points)
        h = np.zeros_like(x)

            if (x[i] % spacing < frac * spacing):
                h[i] = 1.0
        return h

    def gen_rand():
        r = np.random.rand(n_points) + 0.5
        return np.clip((r - 0.5) * 2.0 / np.std(r) + 0.5, 0, 1)

        return np.sin(theta * np.linspace(0, 2, n_points)) ** 2

    def refine(r):

                               constraints=[{'type': 'eq', 'fun': cons}],

    guesses = [h]

        guesses.append(gen_block(0.5, 2.0 / freq))

        guesses.append(gen_block(0.3, 1.0))
