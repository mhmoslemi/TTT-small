import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.sin(2 * np.pi * freq * x) + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.abs(np.sin(2 * np.pi * freq * x)) + 1) * duty / 2, 0, 1)

    def gen_random():
        return np.clip((np.random.rand(n_points) - 0.5) * 0.6 + 0.5, 0, 1)

    def gen_modulated(freqs, duties):
        waves = []
        for f, d in zip(freqs, duties):
            waves.append(gen_pulse(f, d))
        return np.clip(np.mean(waves, axis=0), 0, 1)

    guesses = [h]
    for f in [2, 3, 4, 5, 6, 8, 10, 12, 15, 20]:
        guesses.append(gen_pulse(f, 0.4))
        guesses.append(gen_saw(f, 0.3))
        guesses.append(gen_tri(f, 0.5))
    guesses.append(gen_modulated([2, 4], [0.5, 0.5]))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        if np.allclose(g, best_h): continue
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(3):

        res = optimize.differential_evolution(obj, bounds, popsize=120, maxiter=150,

        c = obj(h_opt)
        if c < best_c5:

    def refine(r):
