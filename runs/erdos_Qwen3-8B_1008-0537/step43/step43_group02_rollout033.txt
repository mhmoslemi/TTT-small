import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freqs, duty):
        x = np.linspace(0, 1, n_points)
        waves = []
        for freq in freqs:
            wave = np.sin(2 * np.pi * freq * x)
            waves.append(np.clip((wave + 1) * duty / 2, 0, 1))
        return np.stack(waves).mean(axis=0)

    def gen_saw(freqs, duty):
        x = np.linspace(0, 1, n_points)
        waves = []
        for freq in freqs:
            wave = 1 - np.abs(2 * freq * x - 1) * duty
            waves.append(np.clip(wave, 0, 1))
        return np.stack(waves).mean(axis=0)

    def gen_tri(freqs, duty):
        x = np.linspace(0, 1, n_points)
        waves = []
        for freq in freqs:
            wave = np.abs(np.sin(2 * np.pi * freq * x))
            waves.append(np.clip((wave + 1) * duty / 2, 0, 1))
        return np.stack(waves).mean(axis=0)

    def gen_random():
        r = np.random.rand(n_points)
        return np.clip((r - np.mean(r)) / np.std(r) * 0.5 + 0.5, 0, 1)

    guesses = [h]
    for freqs in [[1], [2], [3], [4], [5], [6], [7], [8], [10], [12]]:
        guesses.append(gen_pulses(freqs, 0.35))
        guesses.append(gen_saw(freqs, 0.3))
        guesses.append(gen_tri(freqs, 0.45))

    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 600, 'ftol': 1e-10, 'eps': 1e-6})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        res = optimize.differential_evolution(obj, bounds, popsize=150, maxiter=100,
                                             mutation=(0.85, 0.98), recombination=0.95, tol=1e-8)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-6})
        return res.x if res.success else r

    for _ in range(8):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
