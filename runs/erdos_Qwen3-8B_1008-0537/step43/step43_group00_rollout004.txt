import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_binary():
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        return h

    def gen_square(f):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * f * x)
        h = np.clip((wave + 1) / 2, 0, 1)
        return h

    def gen_saw(f, d):
        x = np.linspace(0, 1, n_points)
        h = np.clip(1 - np.abs(2 * f * x - 1) * d, 0, 1)
        return h

    guesses = [h, gen_binary(), gen_square(3), gen_saw(3, 0.5)]

    def slsqp_guess(g):
        try:
            res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 150, 'ftol': 1e-10})
            return res.x
        except:
            return g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        try:
            res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=100,
                                                 mutation=(0.5, 0.9), recombination=0.8)
            return res.x
        except:
            return np.random.rand(n_points)

    for _ in range(3):

        h_opt = de_guess()
