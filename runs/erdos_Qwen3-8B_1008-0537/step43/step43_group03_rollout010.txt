        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_staircase(frac, width):

        h = np.zeros_like(x)

            if (x[i] < 0.5 * frac * width or

                h[i] = 1.0
        return h
    
    def gen_random(frac=0.5):

    def slsqp_guess(g):
