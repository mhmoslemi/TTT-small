import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_perturbed():
        noise = np.random.normal(0, 0.05, n_points)
        return np.clip(h + noise, 0, 1)

    def gen_carve(frac):
        idx = np.argsort(np.random.rand(n_points))
        return np.clip(np.zeros(n_points), 0, 1).astype(float)

    def gen_patch(frac):
        idx = np.random.choice(n_points, int(frac * n_points), replace=False)
        return np.clip(np.zeros(n_points), 0, 1).astype(float)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_patch_hole(frac):
        idx = np.random.choice(n_points, int(frac * n_points), replace=False)
        return np.clip(1.0 + np.random.normal(0, 0.1, n_points), 0, 1)

    def gen_dither():
        return np.clip(np.random.rand(n_points) * 0.2 + 0.5, 0, 1)

    guesses = [h]
    for _ in range(3):
        guesses.append(gen_perturbed())
    for freq in [2, 3, 4, 5]:
        guesses.append(gen_saw(freq, 0.2))
        guesses.append(gen_saw(freq, 0.3))
    for _ in range(2):
        guesses.append(gen_dither())
        guesses.append(gen_patch_hole(0.3))
    for _ in range(2):
        guesses.append(gen_patch(0.3))

    def slsqp_guess(g):
        try:
            res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 200, 'ftol': 1e-10})
            return res.x if res.success else g
        except:
            return g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_params():
        return (0.5, 1.0), 0.8, 30

    for _ in range(3):
        if time.time() - start_time > budget_s - 15:
            break
        popsize = min(80, int(n_points * 0.5))
        res = optimize.differential_evolution(obj, bounds, popsize=popsize, maxiter=50, mutation=de_params()[0], recombination=de_params()[1], tol=1e-8)
        h_opt = slsqp_guess(res.x)

    return best_h, best_c5, n_points
