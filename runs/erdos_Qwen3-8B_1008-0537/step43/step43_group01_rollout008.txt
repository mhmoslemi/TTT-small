import scipy.optimize as optimize
import time

    def obj(h_vals):

        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)
    
    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
    
    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty * 2, 0, 1)
    
    def gen_random():
        r = np.random.rand(n_points)
        return np.clip((r - np.mean(r)) / np.std(r) * 0.5 + 0.5, 0, 1)
    
    def gen_stair(n_steps):
        steps = np.linspace(0, 1, n_steps + 1)
        h = np.zeros(n_points)
        for i in range(n_steps):
            start = np.searchsorted(steps, i / n_steps) * dx
            end = np.searchsorted(steps, (i + 1) / n_steps) * dx
            h[start:end] = 1 / n_steps
        return h
    
    def gen_random_stair():
        n_steps = np.random.randint(3, 6)
        return gen_stair(n_steps)
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 7, 8]:

    for _ in range(2):

    def slsqp_guess(g):
