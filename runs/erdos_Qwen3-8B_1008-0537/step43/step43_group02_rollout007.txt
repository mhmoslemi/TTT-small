import scipy.optimize as optimize
import time

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.sin(2 * np.pi * freq * x) + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_tri(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.abs(np.sin(2 * np.pi * freq * x))
        return np.clip((wave + 1) * duty / 2, 0, 1)

    guesses = [h]
    for freq in [2, 3, 4, 5, 6, 7, 8, 10, 12, 15]:
        guesses.append(gen_pulse(freq, 0.4))
        guesses.append(gen_saw(freq, 0.5))
        guesses.append(gen_tri(freq, 0.6))
    for _ in range(4):
        guesses.append(gen_pulse(5, 0.5))
        guesses.append(gen_saw(5, 0.5))
        guesses.append(gen_tri(5, 0.6))

    def slsqp_guess(g):

                               constraints=[{'type': 'eq', 'fun': cons}],
