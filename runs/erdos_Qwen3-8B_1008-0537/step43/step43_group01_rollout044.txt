import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def cons(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_wave(freq, phase, duty, type_):
        x = np.linspace(0, 1, n_points)
        if type_ == 'pulse':
            return np.clip((np.sin(2 * np.pi * freq * x) + 1) * duty / 2, 0, 1)
        elif type_ == 'saw':
            return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif type_ == 'tri':
            return np.clip(1 - np.abs(2 * freq * x - 1) * duty * 2, 0, 1)
        elif type_ == 'square':
            return np.clip(0.5 + 0.5 * np.sign(np.sin(2 * np.pi * freq * x + phase)), 0, 1)
    
    def gen_random():
        r = np.random.rand(n_points)
        return np.clip((r - np.mean(r)) / np.std(r) * 0.5 + 0.5, 0, 1)
    
    guesses = [h]
    for freq in [2, 3, 4, 5, 6]:
        for p in [0.3, 0.5, 0.7]:
            guesses.append(gen_wave(freq, 0, p, 'pulse'))
            guesses.append(gen_wave(freq, 0, p, 'saw'))
            guesses.append(gen_wave(freq, 0, p, 'tri'))
            guesses.append(gen_wave(freq, np.pi/2, p, 'square'))
    for _ in range(3):
        guesses.append(gen_random())
    
    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-9})
        return res.x if res.success else g
    
    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    def de_run():
        if time.time() - start_time > budget_s - 30:
            return
        res = optimize.differential_evolution(obj, bounds, popsize=80, maxiter=50,
                                             mutation=(0.6, 1.0), recombination=0.8, tol=1e-8)
        h_opt = slsqp_guess(res.x)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
    
    for _ in range(3):
        de_run()
    
    def refine(r):
        res = optimize.minimize(obj, r, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200})
        return res.x if res.success else r
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
