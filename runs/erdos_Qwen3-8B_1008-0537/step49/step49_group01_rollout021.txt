import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate additional initial guesses
    guesses = [initial_h.copy()]
    for m in [10, 12, 14, 16, 20, 24, 28]:
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        h_guess = np.clip(h_guess, 0, 1)
        h_guess = h_guess * (target_sum / np.sum(h_guess))
        guesses.append(h_guess)
    
    # Add random binary pattern
    rand_guess = np.random.uniform(0, 1, n_points)
    rand_guess = np.clip(rand_guess, 0, 1)
    rand_guess = rand_guess * (target_sum / np.sum(rand_guess))
    guesses.append(rand_guess)
    
    # Initial optimization with SLSQP
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time >= budget_s - 60:
        return best_h, best_c5, n_points
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    # Differential evolution with adjusted parameters
    def optimize_with_de():
        res = optimize.differential_evolution(objective, bounds, popsize=80, maxiter=100,
                                             mutation=(0.5, 0.9), recombination=0.7, tol=1e-9)
        if res.success:
            refined = refine_with_slsqp(res.x)
            if objective(refined) < best_c5:
                best_c5 = objective(refined)
                best_h = refined
            return True
        return False
    
    # Run DE with time budget
    for _ in range(3):
        if time.time() - start_time >= budget_s - 30:
            break
        if optimize_with_de():
            continue
    
    return best_h, best_c5, n_points
