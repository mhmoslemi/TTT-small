import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_evenly_spaced(m):
        x = np.linspace(0, 2, m)
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        return h_guess
    
    def generate_binary(m):
        x = np.linspace(0, 2, m + 1)[:-1]
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs[:m]:
            if i < n_points:
                h_guess[i] = 1.0
        return h_guess
    
    guesses = [initial_h_values.copy()]
    for m in [22, 23]:
        guesses.append(generate_evenly_spaced(m))
        guesses.append(generate_binary(m))
    for _ in range(3):
        noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed = np.clip(initial_h_values + noise, 0, 1)
        guesses.append(perturbed)
    
    for guess in guesses:
        if time.time() - start_time >= budget_s - 50:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time >= budget_s - 30:
        return best_h, best_c5, n_points
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    def differential_evolution_with_refinement():
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=40,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = refine_with_slsqp(res.x)
            if objective(res_slsqp) < best_c5:
                best_c5 = objective(res_slsqp)
                best_h = res_slsqp
        return
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:

        differential_evolution_with_refinement()
