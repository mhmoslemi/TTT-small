        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=50, mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Refine final solution
    for _ in range(2):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,
                                  constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                  options={'maxiter': 20, 'ftol': 1e-10, 'eps': 1e-8}).x
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
