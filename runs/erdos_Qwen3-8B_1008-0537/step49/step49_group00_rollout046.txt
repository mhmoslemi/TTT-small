    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 1, n_points)
        return 1 - np.abs(2 * freq * x - 1) * duty

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def generate_symmetric_pulses(num_pulses):
        l = 1.0 / num_pulses
        half_len = l / 2
        centers = np.linspace(0.5 - (num_pulses // 2) * l, 0.5 + (num_pulses // 2) * l, num_pulses)
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for center in centers:
            start = max(0, center - half_len)
            end = min(2, center + half_len)
            idx_start = np.searchsorted(x, start)
            idx_end = np.searchsorted(x, end) + 1
            h[idx_start:idx_end] = 1
        return h

    guesses = [h]
    for freq in range(2, 6):
        guesses.append(gen_pulse(freq, 0.5))
    for duty in [0.25, 0.5, 0.75]:
        guesses.append(gen_saw(3, duty))
        guesses.append(gen_triangle(3, duty))
    for _ in range(10):
        guesses.append(gen_random())
    guesses.append(generate_symmetric_pulses(5))
    guesses.append(generate_symmetric_pulses(7))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=30, maxiter=20,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(2):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
