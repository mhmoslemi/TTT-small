    dx = 2.0 / n_points
    h = initial_h.copy()
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_uniform_guess(n_points):
        count_ones = n_points // 2
        if n_points % 2:
            count_ones += 1
        positions = np.linspace(0, 2, count_ones + 1)[:-1]
        sample_positions = np.arange(n_points) * dx
        idxs = np.array([np.argmin(np.abs(positions - sample_positions)) for _ in range(count_ones)])
        uniform_guess = np.zeros(n_points)
        for idx in np.unique(idxs):
            if 0 <= idx < n_points:
                uniform_guess[idx] = 1.0
        return uniform_guess
    
    guesses = [initial_h.copy()]
    guesses.append(generate_uniform_guess(n_points))
    for _ in range(4):
        noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        guesses.append(perturbed)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time >= budget_s - 60:
        return best_h, best_c5, n_points
    
    for _ in range(4):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=70,
                                             mutation=(0.5, 0.9), recombination=0.7, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='L-BFGS-B', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    return best_h, best_c5, n_points
