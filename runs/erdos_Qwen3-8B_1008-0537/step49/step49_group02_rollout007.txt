import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate structured guesses
    guesses = [h]
    for m in [10, 12, 14, 16, 20, 25, 30]:
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        guesses.append(h_guess)
    
    # Add symmetric guesses
    h_sym = np.zeros(n_points)
    for i in range(n_points):
        h_sym[i] = h[np.min([i, n_points - i - 1])]
    guesses.append(h_sym)
    
    # Add random guesses
    for _ in range(2):
        h_rand = np.random.rand(n_points)
        h_rand = (h_rand - h_rand.mean()) / h_rand.std()
        h_rand = np.clip(h_rand, 0, 1)
        h_rand *= (target_sum / np.sum(h_rand))
        guesses.append(h_rand)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 80, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Differential evolution
    for _ in range(2):
        if time.time() - start_time >= budget_s - 40:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=50, mutation=(0.6, 0.9), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 40, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    # Refinement
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds,

        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
