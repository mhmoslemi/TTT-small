import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def perturb(h, scale=0.05):
        noise = np.random.normal(0, scale, h.shape)
        return np.clip((h + noise) / np.sum(h) * target_sum, 0, 1)

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-9})
        return res.x if res.success else g

    def de_guess(g):
        res = optimize.differential_evolution(obj, bounds, popsize=20, maxiter=20,
                                             mutation=(0.5, 0.9), recombination=0.8, tol=1e-9)
        return res.x

    def refine(h, n_refines=5):
        for _ in range(n_refines):
            if time.time() - start_time > budget_s - 10:
                break
            h = slsqp_guess(h)
            c = obj(h)
            if c < best_c5:
                best_h, best_c5 = h, c
        return h

    initial_perturb = perturb(h)
