        mirrored = h[:half][::-1] if n % 2 == 0 else h[:half][::-1]
        if n % 2 == 0:
            return np.concatenate([h[:half], mirrored])
        else:
            return np.concatenate([h[:half], [h[half]], mirrored])
    guesses.append(generate_symmetric(initial_h))
    
    # Generate random perturbations
    for _ in range(2):
        noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        guesses.append(perturbed)
    
    # Additional spaced guess
    m = 10
    x_pts = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
    idxs = np.searchsorted(x_pts, np.linspace(0, 2, n_points))
    h_guess = np.zeros(n_points)
    for i in idxs:
        if i < n_points:
            h_guess[i] = 1.0
    guesses.append(h_guess)
    
    # Optimize each guess
    for guess in guesses:
        res = optimize.minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    # Early return if time is nearly up
    if time.time() - start_time >= budget_s - 60:
        return best_h, best_c5, n_points
    
    # Refinement with differential evolution
    def refine_with_slsqp(h_init):
        res = optimize.minimize(
            objective,
            h_init,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
        )
        return res.x if res.success else h_init
    
    # Run differential evolution with tighter parameters
    for _ in range(4):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(
            objective,
            bounds,
            popsize=30,
            maxiter=50,
            mutation=(0.5, 0.9),
            recombination=0.7,
            tol=1e-9
        )
        if res.success:
            res_slsqp = refine_with_slsqp(res.x)
            current_c5 = objective(res_slsqp)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res_slsqp
    
    return best_h, best_c5, n_points
