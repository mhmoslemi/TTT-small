import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_evenly_spaced(m):
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        h_guess = np.zeros(n_points)
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        return h_guess
    
    def generate_binary_spaced(m):
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h_guess[i] = 1.0
        return h_guess
    
    guesses = []
    guesses.append(initial_h.copy())
    guesses.append(generate_evenly_spaced(60))
    guesses.append(generate_binary_spaced(60))
    for m in [8, 10, 12]:
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        guesses.append(h_guess)
    for _ in range(3):
        noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        guesses.append(perturbed)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
