import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate structured initial guesses
    def generate_binary_spaced():
        x = np.linspace(0, 2, n_points) * (n_points + 1)
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in range(1, n_points // 2 + 1):
            if i < n_points:
                h_guess[i] = 1.0
        return h_guess
    
    def generate_evenly_spaced():
        x = np.linspace(0, 2, n_points + 1)[:-1] + 0.5 * (1.0 / (n_points + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        return h_guess
    
    def generate_perturbed():
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        return np.clip(initial_h + perturb, 0, 1)
    
    guesses = [initial_h.copy()]
    guesses.append(generate_binary_spaced())
    guesses.append(generate_evenly_spaced())
    guesses.append(generate_perturbed())
    
    for _ in range(3):
        guesses.append(generate_perturbed())
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:

            best_h = res.x
