        m = int(np.ceil(n_points / 2))
        h_guess = np.zeros(n_points)
        h_guess[:m] = 1.0
        current_sum = np.sum(h_guess)
        if current_sum > target_sum:
            extra = current_sum - target_sum
            for i in range(m):
                if extra > 0:
                    h_guess[i] -= extra
                    extra = 0
                else:
                    break
        elif current_sum < target_sum:
            deficit = target_sum - current_sum
            for i in range(m):
                if deficit > 0:
                    h_guess[i] += deficit
                    deficit = 0
                else:
                    break
        return h_guess
    
    guesses = [h.copy()]
    for _ in range(5):
        guesses.append(generate_random_guess())
    for m in [6, 8, 10, 12, 14]:
        guesses.append(generate_clustered_guess(m))
    guesses.append(generate_regular_guess())
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    def refine_with_cobyla(h_init):
        res = optimize.minimize(objective, h_init, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_cobyla(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
