import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def gen_greedy_guess():
        h_guess = np.zeros(n_points)
        for i in range(n_points):
            h_guess[i] = 1.0
        np.random.shuffle(h_guess)
        h_guess = np.sort(h_guess)
        h_guess = np.reshape(h_guess[:n_points//2], n_points)
        h_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points//2), h_guess)
        return h_guess
    
    def gen_binary_guess():
        idxs = np.unique(np.sort(np.random.choice(n_points, size=n_points//2, replace=False)))
        h_guess = np.zeros(n_points)
        h_guess[idxs] = 1.0
        return h_guess
    
    guesses = [h.copy(),
               np.ones(n_points) * 0.5,
               np.zeros(n_points),
               gen_binary_guess(),
               gen_greedy_guess()]

    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=30, maxiter=50, mutation=(0.2, 0.8), recombination=0.8, tol=1e-9)
        if res.success:
            res_slsqp = optimize.minimize(objective, res.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 60, 'ftol': 1e-10, 'eps': 1e-8})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    def refine_with_slsqp(h_init):
