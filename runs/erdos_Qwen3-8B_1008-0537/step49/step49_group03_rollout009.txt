import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_random_guess():
        return np.random.rand(n_points)
    
    def generate_clustered_guess(n_clusters):
        clusters = np.linspace(0, 2, n_clusters, endpoint=False)
        spacing = (2 / n_points) * np.random.rand(n_clusters)
        h_guess = np.zeros(n_points)
        for i in range(n_clusters):
            idx = np.searchsorted(np.linspace(0, 2, n_points), clusters[i] + spacing[i])
            h_guess[idx] = 1.0
        return h_guess
    
    def generate_regular_guess(n_ones):
        positions = np.linspace(0, 2, n_ones + 1)
        h_guess = np.zeros(n_points)
        for x in positions[1:]:
            idx = np.searchsorted(np.linspace(0, 2, n_points), x)
            if idx < n_points:
                h_guess[idx] = 1.0
        return h_guess
    
    guesses = [h.copy()]
    for _ in range(3):
        guesses.append(generate_random_guess())
    for m in [6, 8, 10]:
        guesses.append(generate_clustered_guess(m))
    n_ones = n_points // 2
    guesses.append(generate_regular_guess(n_ones))
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        return res.x if res.success else h_init
    
    for _ in range(2):

        best_h = refine_with_slsqp(best_h)
