import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_phase_shifting(freq, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(np.sin(2 * np.pi * (freq * x + phase)), 0, 1)

    def gen_periodic(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_binary(pulse_width):
        x = np.linspace(0, 1, n_points)
        idx = np.floor(x * n_points).astype(int)
        return np.where((idx % (pulse_width * n_points)) < pulse_width, 1, 0)

    guesses = [h]
    for freq in range(2, 8):
        guesses.append(gen_phase_shifting(freq, 0.25))
    for duty in [0.2, 0.4, 0.6, 0.8]:
        guesses.append(gen_periodic(3, duty))
    for width in [0.2, 0.3, 0.4, 0.5]:
        guesses.append(gen_binary(width))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-9})
        return res.x if res.success else g

        c = obj(h_opt)
