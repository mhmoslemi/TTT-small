import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    guesses = []

    # Uniform guess
    guesses.append(np.full(n_points, 0.5))

    # Structured guesses
    for m in [5, 10, 15]:
        # Spread m intervals evenly
        step = 2.0 / (m + 1)
        positions = np.linspace(step, 2.0 - step, m)
        h_guess = np.zeros(n_points)
        for pos in positions:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos))
            h_guess[idx] = 1.0
        guesses.append(h_guess)

    # Randomized pattern
    def random_pattern():
        h_rand = np.random.rand(n_points)
        h_rand = (h_rand - h_rand.mean()) / h_rand.std()
        h_rand = np.clip(h_rand, 0, 1)
        h_rand *= (target_sum / np.sum(h_rand))
        return h_rand
    guesses.append(random_pattern())

    # Add structured periodic guess
    m = 5
    step = 2.0 / m
    h_guess = np.zeros(n_points)
    for i in range(m):
        idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (i + 0.5) * step))
        h_guess[idx] = 1.0
    guesses.append(h_guess)

    # Optimize with SLSQP and refinement
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],

        if res.success and objective(res.x) < best_c5:

    # Differential evolution with refinement
    for _ in range(2):
