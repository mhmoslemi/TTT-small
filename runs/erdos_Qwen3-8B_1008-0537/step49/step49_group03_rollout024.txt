import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_alternating_guess():
        guess = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                guess[i] = 1.0
        t_sum = np.sum(guess)
        if abs(t_sum - target_sum) < 1e-6:
            return guess
        scale = target_sum / t_sum
        return guess * scale

    def generate_spaced_guess(num_blocks):
        guess = np.zeros(n_points)
        block_len = n_points // num_blocks
        for i in range(num_blocks):
            start = i * block_len
            end = start + block_len
            guess[start:end] = 1.0
        t_sum = np.sum(guess)
        if abs(t_sum - target_sum) < 1e-6:
            return guess
        scale = target_sum / t_sum
        return guess * scale

    def generate_shifted_guess(base_guess):
        shifted = np.roll(base_guess, 1)
        t_sum = np.sum(shifted)
        if abs(t_sum - target_sum) < 1e-6:
            return shifted
        scale = target_sum / t_sum
        return shifted * scale

    initial_guesses = [h.copy()]
    initial_guesses.append(generate_alternating_guess())
    initial_guesses.append(generate_spaced_guess(6))
    initial_guesses.append(generate_spaced_guess(10))
    initial_guesses.append(generate_shifted_guess(initial_guesses[1]))

    for guess in initial_guesses:
        res = optimize.minimize(objective, guess, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    def refine_with_cobyla(h_init):
        res = optimize.minimize(objective, h_init, method='COBYLA', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init

    for _ in range(3):
        if time.time() - start_time >= budget_s - 10:
            break
        best_h = refine_with_cobyla(best_h)
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)

    return best_h, best_c5, n_points
