        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        S = n_points / 2
        K = np.sum(h_guess)
        delta = S - K
        if delta != 0:
            h_guess[-1] += delta
            h_guess[-1] = np.clip(h_guess[-1], 0, 1)
        guesses.append(h_guess)
    for _ in range(3):
        num_ones = int(np.round(target_sum))
        random_indices = np.random.choice(n_points, size=num_ones, replace=False)
        h_guess = np.zeros(n_points)
        h_guess[random_indices] = 1.0
        delta = target_sum - np.sum(h_guess)
        if delta != 0:
            h_guess[-1] += delta
            h_guess[-1] = np.clip(h_guess[-1], 0, 1)
        guesses.append(h_guess)
    for _ in range(3):
        noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        guesses.append(perturbed)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time >= budget_s - 60:
        return best_h, best_c5, n_points
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    for _ in range(4):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=80, maxiter=100,
                                             mutation=(0.5, 0.9), recombination=0.7, tol=1e-9)
        if res.success:
            res_slsqp = refine_with_slsqp(res.x)
            if objective(res_slsqp) < best_c5:
                best_c5 = objective(res_slsqp)
                best_h = res_slsqp
    
    return best_h, best_c5, n_points
