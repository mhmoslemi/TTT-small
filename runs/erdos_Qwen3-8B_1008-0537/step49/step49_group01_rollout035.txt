            start = k * (l + s)
            end = start + l
            lower = max(0, int(np.ceil(start / dx)))
            upper = int(np.floor((end - 1e-9) / dx))
            if lower <= upper:
                h[lower:upper+1] = 1.0
        return h
    
    guesses = [initial_h.copy()]
    for m in [5, 7, 9]:
        guesses.append(generate_structured_guess(m))
    
    for _ in range(3):
        noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        guesses.append(perturbed)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time >= budget_s - 60:
        return best_h, best_c5, n_points
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    for _ in range(4):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=70,
                                             mutation=(0.5, 0.9), recombination=0.7, tol=1e-9)
        if res.success:
            res_slsqp = refine_with_slsqp(res.x)
            if objective(res_slsqp) < best_c5:
                best_c5 = objective(res_slsqp)
                best_h = res_slsqp
    
    return best_h, best_c5, n_points
