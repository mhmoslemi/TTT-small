import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate diverse initial guesses
    guesses = [h.copy()]
    
    # Regular pattern: evenly spaced 1's
    def generate_regular(m):
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        h_g = np.zeros(n_points)
        for i in range(m):
            pos = x[i]
            idx = int(pos * n_points / 2)
            if idx < n_points:
                h_g[idx] = 1.0
        return h_g
    
    for m in [10, 12, 14, 16]:
        guesses.append(generate_regular(m))
    
    # Random perturbations
    for _ in range(3):
        noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed = np.clip(h + noise, 0, 1)
        guesses.append(perturbed)
    
    # Optimization loop
    for guess in guesses:
        if time.time() - start_time >= budget_s - 60:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Refinement with DE
    if time.time() - start_time < budget_s - 20:
        res_de = optimize.differential_evolution(
            objective, bounds, popsize=50, maxiter=50,
            mutation=(0.5, 0.9), recombination=0.7, tol=1e-9
        )
        if res_de.success:
            res_slsqp = optimize.minimize(
                objective, res_de.x, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    return best_h, best_c5, n_points
