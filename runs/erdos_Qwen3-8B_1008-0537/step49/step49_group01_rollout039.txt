    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_evenly_spaced(m):
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        return h_guess
    
    def generate_binary_spaced(m):
        x = np.linspace(0, 2, m + 1)[:-1] + 0.5 * (1.0 / (m + 1))
        idxs = np.searchsorted(x, np.linspace(0, 2, n_points))
        h_guess = np.zeros(n_points)
        for i in idxs:
            if i < n_points:
                h_guess[i] = 1.0
        return h_guess
    
    def generate_low_discrepancy(m):
        base = 2
        h_guess = np.zeros(n_points)
        for i in range(1, m + 1):
            num = i
            denom = base
            fraction = 0.0
            while num > 0:
                fraction += (num % base) / denom
                num //= base
                denom *= base
            pos = 2 * fraction
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos))
            h_guess[idx] += 1.0 / m
        return h_guess
    
    guesses = [initial_h.copy()]
    for m in [10, 12, 14, 16]:
        guesses.append(generate_evenly_spaced(m))
    for m in [10, 12, 14, 16]:
        guesses.append(generate_binary_spaced(m))
    guesses.append(generate_low_discrepancy(23))
    for _ in range(3):
        noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        guesses.append(perturbed)
    
    for guess in guesses:
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time >= budget_s - 60:
        return best_h, best_c5, n_points
    
    def refine_with_slsqp(h_init):
        res = optimize.minimize(objective, h_init, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        return res.x if res.success else h_init
    
    for _ in range(4):
        if time.time() - start_time >= budget_s - 20:
            break
        res = optimize.differential_evolution(objective, bounds, popsize=60, maxiter=70,
                                             mutation=(0.5, 0.9), recombination=0.7, tol=1e-9)
        if res.success:
            res_slsqp = refine_with_slsqp(res.x)
            if objective(res_slsqp) < best_c5:
                best_c5 = objective(res_slsqp)
                best_h = res_slsqp
    
    return best_h, best_c5, n_points
