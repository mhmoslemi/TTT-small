import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h_initial = np.array(initial_h_values, dtype=float)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_initial_guesses(num_ones, spacing='random'):
        if spacing == 'regular':
            spacing_val = 2.0 / num_ones
            indices = np.arange(0, n_points, spacing_val)
            indices = np.unique(np.round(indices).astype(int))
            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:
                idx = np.random.choice(np.where(h == 0)[0], size=1)
                h[idx] = 0.5
            return h
        else:
            indices = np.random.choice(n_points, num_ones, replace=False)
            h = np.zeros(n_points)
            h[indices] = 1.0
            if np.sum(h) < target_sum:
                idx = np.random.choice(np.where(h == 0)[0], size=1)
                h[idx] = 0.5
            return h
    
    initial_guesses = [h_initial.copy()]
    ones_range = range(15, 45, 5)
    for num_ones in ones_range:
        initial_guesses.append(generate_initial_guesses(num_ones, 'regular'))
        initial_guesses.append(generate_initial_guesses(num_ones, 'random'))
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 1000, 'ftol': 1e-12, 'eps': 1e-9}
            )
            return result.x if result.success else None

    for guess in initial_guesses:

            tol=1e-10,
            vectorized=True,

        )
        if result_de.success:
            result_slsqp = optimize.minimize(

            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
