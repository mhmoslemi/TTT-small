import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_regular_guess(num_ones):
        spacing = 2.0 / num_ones
        indices = np.round(np.arange(0, n_points, spacing)).astype(int)
        indices = np.unique(indices)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_symmetric_guess(num_ones):
        spacing = 2.0 / num_ones
        positions = np.arange(0, n_points, spacing)
        indices = np.round(positions).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_perturbed_guess(base_guess):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(base_guess + perturb, 0, 1)
        return perturbed
    
    initial_guesses = [initial_guess]
    for _ in range(2):
        for num_ones in [n_points // 2 - 5, n_points // 2, n_points // 2 + 5]:
            if num_ones > 0:
                initial_guesses.append(generate_regular_guess(num_ones))
                initial_guesses.append(generate_symmetric_guess(num_ones))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:

    for guess in initial_guesses:

            best_h = result
    
    try:
        result_de = optimize.differential_evolution(

            tol=1e-10,
            vectorized=True,

        if result_de.success:

            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
