import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_evenly_spaced_guess(num_ones):
        if num_ones == 0 or num_ones == n_points:
            return np.zeros(n_points)
        spacing = 2.0 * (num_ones - 1) / n_points
        positions = np.arange(num_ones) * spacing
        indices = np.round(positions / dx).astype(int)
        indices = np.unique(np.clip(indices, 0, n_points - 1))
        if len(indices) < num_ones:
            indices = np.random.choice(n_points, num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess]
    for num_ones in [15, 20, 25, 30, 35]:
        initial_guesses.append(generate_evenly_spaced_guess(num_ones))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,

                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
