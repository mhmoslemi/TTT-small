import scipy.optimize as optimize
import time

        return np.max(corr) * dx
    
    def constraint(h_values):
        return np.sum(h_values) - target_sum

    best_h = h.copy()

    def generate_perturbed_guess(base_h, sigma=0.1):
        perturb = np.random.normal(0, sigma, n_points)
        h = base_h + perturb
        h = np.clip(h, 0, 1)
        h[ np.sum(h) < target_sum ] = 1.0
        h[ np.sum(h) > target_sum ] = 0.0
        return h
    
    def generate_regularity_guess(num_ones):
        spacing = n_points / num_ones
        indices = np.unique(np.round(np.arange(0, n_points, spacing)).astype(int))
        h = np.zeros(n_points)
        h[indices] = 1.0
        while np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 0.5
        while np.sum(h) > target_sum:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.5
        return h
    
    def optimize_guess(guess, method='SLSQP'):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    def differential_evolution_run():
        try:
            result = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=150,
                maxiter=150,

                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            pass
