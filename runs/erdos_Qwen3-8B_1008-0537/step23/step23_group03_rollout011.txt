import scipy.optimize as optimize
import time

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbed_guess(base_guess, perturb_factor=0.2):
        perturb = np.random.normal(0, perturb_factor, n_points)
        perturb = np.clip(perturb, -0.5, 0.5)
        return np.clip(base_guess + perturb, 0.0, 1.0)
    
    initial_guesses = [initial_guess]
    for i in range(3):
        initial_guesses.append(generate_perturbed_guess(initial_guess))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
