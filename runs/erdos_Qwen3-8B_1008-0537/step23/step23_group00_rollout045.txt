        h_perturbed[idxs] += np.random.uniform(-noise_scale, noise_scale, size=len(idxs))
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        if np.sum(h_perturbed) < target_sum:
            idx = np.random.choice(np.where(h_perturbed == 0)[0], size=1)
            h_perturbed[idx] = 0.5
        return h_perturbed
    
    def generate_random_guess():
        h_random = np.random.rand(n_points)
        h_random = h_random * (target_sum / np.sum(h_random))
        return h_random
    
    initial_guesses = [initial_guess, generate_perturbed_guess(initial_guess)]
    initial_guesses.append(generate_random_guess())
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=200,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-9,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
