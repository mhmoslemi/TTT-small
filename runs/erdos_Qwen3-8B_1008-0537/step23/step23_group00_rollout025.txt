import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbed_guess(h, noise_scale=0.1):
        h_perturbed = h.copy()
        idxs = np.random.choice(n_points, size=min(10, n_points), replace=False)
        h_perturbed[idxs] += np.random.uniform(-noise_scale, noise_scale, size=len(idxs))
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        if np.sum(h_perturbed) < target_sum - 1e-6:
            idx = np.random.choice(np.where(h_perturbed < 1e-6)[0], size=1)
            h_perturbed[idx] = 0.5
        return h_perturbed
    
    initial_guesses = [initial_guess]
    n_uniform = n_points // 2
    h_uniform = np.zeros(n_points)
    h_uniform[:n_uniform] = 1.0
    initial_guesses.append(h_uniform)

    initial_guesses.append(h_random)
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],

    try:
