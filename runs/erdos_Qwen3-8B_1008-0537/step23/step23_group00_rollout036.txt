import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbed_guess(h, noise_scale=0.1):
        h_perturbed = h.copy()
        idxs = np.random.choice(n_points, size=min(10, n_points), replace=False)
        h_perturbed[idxs] += np.random.uniform(-noise_scale, noise_scale, size=len(idxs))
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        if np.sum(h_perturbed) < target_sum:
            idx = np.random.choice(np.where(h_perturbed == 0)[0], size=1)
            h_perturbed[idx] = 0.5
        return h_perturbed
    
    def generate_initial_guesses():
        guesses = [initial_guess]
        uniform = np.full(n_points, 0.5)
        guesses.append(uniform)
        step = np.array([1.0 if i < n_points//2 else 0.0 for i in range(n_points)])
        guesses.append(step)
        for _ in range(3):
            guesses.append(generate_perturbed_guess(initial_guess, noise_scale=0.2))
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=300,
            mutation=(0.6, 1.0),
            recombination=0.8,
            tol=1e-9,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,

                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 100:
        try:
            result = optimize.minimize(
