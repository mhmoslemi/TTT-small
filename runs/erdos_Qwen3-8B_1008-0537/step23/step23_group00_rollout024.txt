import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def generate_random_guess():
        idxs = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[idxs] = 1.0
        return h
    
    def generate_structured_guess():
        # Spread out the 1s as evenly as possible
        h = np.zeros(n_points)
        indices = np.linspace(0, n_points-1, n_points//2, endpoint=False).astype(int)
        h[indices] = 1.0
        return h
    
    initial_guesses = [initial_guess, generate_random_guess(), generate_structured_guess()]
    
    # Objective function
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Constraint function
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

            result = optimize.minimize(
                fun=objective,
                x0=guess,
