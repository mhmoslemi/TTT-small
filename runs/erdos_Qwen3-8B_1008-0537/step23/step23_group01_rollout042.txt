import scipy.optimize as optimize
import time

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def generate_regular_guess(num_ones):
        spacing = 2.0 / num_ones
        positions = np.arange(0, 2, spacing)
        h = np.zeros(n_points)
        for pos in positions:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos))
            h[idx] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    def generate_varied_guess(num_ones):
        indices = np.random.choice(n_points, num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0], size=1)
            h[idx] = 0.5
        return h
    
    initial_guesses = [initial_guess]

        initial_guesses.append(generate_varied_guess(num_ones))
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
