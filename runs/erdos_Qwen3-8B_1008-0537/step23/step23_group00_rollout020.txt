import scipy.optimize as optimize

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def generate_perturbed_guess(h, noise_scale=0.1):
        h_perturbed = h.copy()
        idxs = np.random.choice(n_points, size=min(10, n_points), replace=False)
        h_perturbed[idxs] += np.random.uniform(-noise_scale, noise_scale, size=len(idxs))
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        if np.sum(h_perturbed) < target_sum:
            idx = np.random.choice(np.where(h_perturbed == 0)[0], size=1)
            h_perturbed[idx] = 0.5
        return h_perturbed
    
    def generate_random_guess():
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        return h
    
    initial_guesses = [initial_guess, generate_perturbed_guess(initial_guess), generate_perturbed_guess(initial_guess)]
    initial_guesses.append(generate_random_guess())
    
    def optimize_guess(guess):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
            )
            return result.x if result.success else None
        except:
            return None
    
    for guess in initial_guesses:

            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,

        )
        if result_de.success:
            result_slsqp = optimize.minimize(

                best_c5 = objective(result_slsqp.x)

    except:

    while time.time() - start_time < budget_s - 100:
