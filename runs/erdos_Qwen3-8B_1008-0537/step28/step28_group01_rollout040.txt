import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        n_points = len(initial_h_values)

        n_points = 60

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_phase_modulated(period, phase_shift, amp):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase_shift) * amp, 0, 1)
    
    def generate_staggered(period, low, high):
        return np.clip(np.sin(2 * np.pi * np.arange(n_points) / period) > 0, low, high)
    
    def generate_sparse(density):
        idx = np.random.choice(n_points, size=3, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return np.clip(0.5 + (h - 0.5) * (1 - density), 0, 1)
    
    initial_guesses = [initial_guess]
    for p in [5, 6, 7, 8, 9]:
        initial_guesses.append(generate_phase_modulated(p, np.pi/2, 0.7))

    for d in [0.2, 0.3, 0.4, 0.5]:
        initial_guesses.append(generate_sparse(d))
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if result.success:
            if objective(result.x) < best_c5:
