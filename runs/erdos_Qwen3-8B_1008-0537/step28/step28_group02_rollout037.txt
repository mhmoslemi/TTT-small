import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def cons(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    def gen_sine(freq):
        return np.clip(0.5 + 0.5 * np.sin(np.pi * freq * np.arange(n_points) * dx), 0, 1)
    
    def gen_binary(freq, duty):
        return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points) * dx) > 0) * duty, 0, 1)
    
    def gen_periodic(period):
        return np.clip(0.6 + 0.4 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def gen_random():
        h = np.random.rand(n_points)
        s = target_sum / np.sum(h)
        return np.clip(h * s, 0, 1)
    
    guesses = [initial_guess, gen_random()]
    for p in [3, 4, 5, 6, 7]:
        guesses.append(gen_periodic(p))
    for p, m in [(3, 0.7), (4, 0.8), (5, 0.9)]:
        guesses.append(gen_periodic(p) * m)
    for f, d in [(6, 0.2), (5, 0.3), (4, 0.4)]:
        guesses.append(gen_binary(f, d))
    for f in [2, 3, 4]:
        guesses.append(gen_sine(f))
    
    for guess in guesses:
        res = optimize.minimize(obj, guess, bounds=bounds, constraints=[cons], method='SLSQP', options={'maxiter': 100})
        if res.success and obj(res.x) < best_c5:
            best_c5, best_h = obj(res.x), res.x
    
    try:
        de_res = optimize.differential_evolution(obj, bounds, strategy='best1bin', maxiter=50, popsize=80, tol=1e-9, mutation=(0.8, 1.0), recombination=0.8)
        res = optimize.minimize(obj, de_res.x, bounds=bounds, constraints=[cons], method='SLSQP', options={'maxiter': 100})
        if res.success and obj(res.x) < best_c5:
            best_c5, best_h = obj(res.x), res.x
    except:
        pass
