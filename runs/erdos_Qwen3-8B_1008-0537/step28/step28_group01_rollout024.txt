import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except:
        n_points = 45  # Match initial H for better finetuning

    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate diverse initial guesses based on advanced patterns
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_triwave(period):
        return np.clip(0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / period)), 0, 1)
    
    # Optimize initial guesses with adaptive parameters
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if result.success:
            return result.x
        return guess
    
    initial_guesses = [initial_guess]
    for period in [4, 5, 6, 7, 8, 9, 10]:
        initial_guesses.append(generate_periodic(period))
    initial_guesses.append(generate_triwave(7))
    
    # Refine with SLSQP
    for guess in initial_guesses:
        refined = optimize_guess(guess)
        current_c5 = objective(refined)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = refined
    
    # Differential evolution with tight parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=120,
            mutation=(0.8, 0.95),
            recombination=0.8,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:

                method='SLSQP',

                options={'maxiter': 180, 'ftol': 1e-12, 'eps': 1e-10}
            )

                best_h = refined.x
