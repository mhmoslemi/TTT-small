import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period, phase):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase), 0, 1)
    
    def generate_modulated(period, phase):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase) * 0.3, 0, 1)
    
    def generate_random():
        h = np.random.rand(n_points)
        scaling = target_sum / np.sum(h)
        h = np.clip(h * scaling, 0, 1)
        if np.sum(h) != target_sum:
            h += (target_sum - np.sum(h)) / n_points
        return h
    
    initial_guesses = [initial_guess]
    for period in [4, 5, 6, 7, 8]:
        initial_guesses.append(generate_periodic(period, 0))
        initial_guesses.append(generate_periodic(period, np.pi/2))
        initial_guesses.append(generate_modulated(period, 0))
        initial_guesses.append(generate_modulated(period, np.pi/2))
    initial_guesses.append(generate_random())
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=80,
            maxiter=100,

            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
