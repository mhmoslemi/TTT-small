import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_modulated(period, mod):
        x = np.arange(n_points)
        wave = 0.5 + 0.5 * np.sin(2 * np.pi * x / period * mod)
        return np.clip(wave, 0, 1)
    
    def generate_binary_freq(freq, duty):
        x = np.arange(n_points)
        wave = np.sin(2 * np.pi * freq * x) > 0
        return np.clip(wave * duty, 0, 1)
    
    def generate_sparse(density):
        h = np.zeros(n_points)
