import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except:
        n_points = 60
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_wave(freq, modulator=0.5):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) * freq) * modulator, 0, 1)
    
    def generate_sparse(density=0.3):
        idx = np.random.choice(n_points, size=3, replace=False)
        return np.clip(0.5 + (np.zeros(n_points) if np.random.rand() < 0.5 else np.ones(n_points)) * density, 0, 1)
    
    def generate_shifted(freq, shift):
        return generate_wave(freq, modulator=0.5) * np.roll(np.ones(n_points), shift)
    
    initial_guesses = [initial_guess]
    for f in [0.333, 0.4, 0.5, 0.666, 0.75, 0.9]:
        initial_guesses.append(generate_wave(f))
    for f in [0.5, 0.6, 0.7, 0.8, 1.0]:
        initial_guesses.append(generate_shifted(f, int(np.random.randint(0, n_points))))
    initial_guesses.append(generate_sparse())
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    remaining_time = budget_s - (time.time() - start_time)
    
    if remaining_time > 30:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=100,
            mutation=(0.6, 0.95),
            recombination=0.9,
            tol=1e-9,
            vectorized=True
        )

            best_c5 = objective(result_de.x)
