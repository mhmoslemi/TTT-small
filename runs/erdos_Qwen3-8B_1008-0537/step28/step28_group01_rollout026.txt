import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except:
        n_points = 60
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h0 = np.array(initial_h_values, dtype=float)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h0.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate diverse initial guesses
    def generate_wave(freq, amp, phase):
        return np.clip(0.5 + amp * np.sin(2*np.pi*freq*np.arange(n_points) + phase), 0, 1)
    
    guesses = [h0]
    for freq in [2, 3, 4, 5, 6]:
        guesses.append(generate_wave(freq, 0.6, np.random.rand() * np.pi))
    for freq in [1.5, 2.5, 3.5]:
        guesses.append(generate_wave(freq, 0.4, np.random.rand() * np.pi))
    
    for guess in guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    # Enhanced differential evolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=30,
                maxiter=100,
                mutation=(0.6, 0.8),
                recombination=0.9,
                tol=1e-9,
                vectorized=True,
                polish=True
            )
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x

            pass

    if remaining > 5:
