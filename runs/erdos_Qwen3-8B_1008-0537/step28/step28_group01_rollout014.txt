import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_perturbation(base, scale=0.1):
        pert = np.random.normal(0, scale, n_points)
        return np.clip(base + pert, 0, 1)
    
    def generate_periodic(period, phase=0):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase), 0, 1)

        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points) + offset) > 0, 0, 1) * duty

        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * mod, 0, 1)
    
    initial_guesses = [
        initial_guess,

    ]
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
