import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        x = np.arange(n_points) * dx
        wave = 0.6 + 0.4 * np.sin(2 * np.pi * x / period)
        return np.clip(wave, 0, 1)
    
    def generate_modulated(period, mod):
        x = np.arange(n_points) * dx
        wave = 0.6 + 0.4 * np.sin(2 * np.pi * x / period) * mod
        return np.clip(wave, 0, 1)
    
    def generate_binary(freq, duty):
        x = np.arange(n_points) * dx
        wave = (np.sin(2 * np.pi * freq * x) > 0) * duty
        return np.clip(wave, 0, 1)
    
    def generate_random():
        h = np.random.rand(n_points)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        scaling = target_sum / np.sum(h)
        h = np.clip(h * scaling, 0, 1)
        if np.sum(h) != target_sum:
            h += (target_sum - np.sum(h)) / n_points
        return h
    
    initial_guesses = [initial_guess, generate_random()]
    
    # Add more diversified initial guesses
    for period in [3, 4, 5, 6, 7]:
        initial_guesses.append(generate_periodic(period))

    for guess in initial_guesses:
