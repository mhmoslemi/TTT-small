import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_waveform(period, freq, phase, duty, mod, shape):
        if shape == 'sawtooth':
            wave = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points) / period)
        elif shape == 'triangle':
            wave = 0.5 + 0.5 * np.abs(np.sin(2 * np.pi * freq * np.arange(n_points) / period))
        elif shape == 'square':
            wave = (np.sin(2 * np.pi * freq * np.arange(n_points) / period) > 0) * 1.0
        else:
            wave = np.clip(0.6 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n_points) / period), 0, 1)
        wave = np.clip(wave * mod + phase, 0, 1)
        wave = np.where(wave < duty, 0, wave)
        return wave
    
    initial_guesses = [initial_guess]
    for period in [4, 5, 6, 7, 8, 9]:
        initial_guesses.append(generate_waveform(period, 1, 0.5, 0.5, 0.7, 'sawtooth'))
    for period, mod in [(4, 0.7), (5, 0.6), (6, 0.5), (7, 0.4), (8, 0.3), (9, 0.2)]:
        initial_guesses.append(generate_waveform(period, 1, 0.3, 0.3, mod, 'triangle'))
    for freq, duty in [(6, 0.2), (5, 0.3), (7, 0.1), (8, 0.4), (9, 0.5)]:
        initial_guesses.append(generate_waveform(10, freq, 0.5, duty, 0.8, 'square'))
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=150,
            maxiter=120,
            mutation=(0.9, 1.0),
            recombination=0.85,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 250, 'ftol': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    while time.time() - start_time < budget_s - 30:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
