import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        t = np.linspace(0, 2, n_points)
        return np.clip(0.6 + 0.4 * np.sin(2 * np.pi * t / period), 0, 1)
    
    def generate_modulated(period, mod):
        t = np.linspace(0, 2, n_points)
        return np.clip(0.6 + 0.4 * np.sin(2 * np.pi * t / period) * mod, 0, 1)
    
    def generate_binary(freq, duty):
        t = np.linspace(0, 2, n_points)
        return np.clip((np.sin(2 * np.pi * freq * t) > 0) * duty, 0, 1)
    
    def generate_triangular(period):
        t = np.linspace(0, 2, n_points)
        return np.clip(0.5 + 0.5 * np.abs(np.sin(2 * np.pi * t / period)), 0, 1)
    
    def generate_sawtooth(period):
        t = np.linspace(0, 2, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * t / period), 0, 1)
    
    initial_guesses = [initial_guess]
    for period in [3, 4, 5, 6, 7, 8, 9, 10]:
        initial_guesses.append(generate_periodic(period))

    for freq, duty in [(3, 0.3), (4, 0.2), (5, 0.1), (6, 0.4), (7, 0.3)]:
        initial_guesses.append(generate_binary(freq, duty))
    initial_guesses.extend([generate_triangular(5), generate_sawtooth(6)])
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
