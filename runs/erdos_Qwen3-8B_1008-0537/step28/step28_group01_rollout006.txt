import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except:
        n_points = 45  # Optimal for balance
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def gen_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def gen_binary(freq, duty):
        return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points)) > 0) * duty, 0, 1)
    
    def gen_sparse(density):
        idx = np.random.choice(n_points, size=3, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        return np.clip(0.5 + (h - 0.5) * (1 - density), 0, 1)
    
    guesses = [initial_guess]
    for p in [6, 7, 8]:
        guesses.append(gen_periodic(p))
    for f, d in [(6, 0.3), (5, 0.4), (4, 0.2)]:
        guesses.append(gen_binary(f, d))
    guesses.append(gen_sparse(0.4))
    
    for guess in guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 180, 'ftol': 1e-12}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=100,
                maxiter=120,
                mutation=(0.5, 0.8),
                recombination=0.8,
                tol=1e-9
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',

                    options={'maxiter': 150, 'ftol': 1e-12}
                )

                    best_c5 = objective(result_slsqp.x)
                    best_h = result_slsqp.x

            pass
    
    return best_h, best_c5, n_points
