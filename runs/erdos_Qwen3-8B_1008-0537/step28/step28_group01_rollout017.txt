import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except:
        n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_binary(freq, duty):
        return np.clip(np.sin(2 * np.pi * freq * np.arange(n_points)) > 0, 0, 1) * duty
    
    initial_guesses = [initial_guess]
    for p in [4, 5, 6, 7, 8, 9, 10]:
        initial_guesses.append(generate_periodic(p))
    for f, d in [(4, 0.2), (5, 0.3), (6, 0.4), (7, 0.5), (8, 0.25)]:
        initial_guesses.append(generate_binary(f, d))
    for _ in range(3):
        initial_guesses.append(np.random.rand(n_points))
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            nonlocal best_c5, best_h
            best_c5 = objective(result.x)
            best_h = result.x
    
    for guess in initial_guesses:
        optimize_guess(guess)
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=80,
                maxiter=100,
                mutation=(0.6, 0.95),
                recombination=0.9,
                tol=1e-9,
                vectorized=True
            )
            if result_de.success:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-10}
                )
                if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                    best_c5 = objective(result_slsqp.x)
                    best_h = result_slsqp.x
        except:
            pass
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 20:
