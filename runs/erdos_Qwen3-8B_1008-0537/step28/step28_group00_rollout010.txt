import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period, phase=0.0):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * (np.arange(n_points) + phase) / period), 0, 1)
    
    def generate_binary(freq, duty, phase=0.0):
        return np.clip(np.sin(2 * np.pi * freq * (np.arange(n_points) + phase)) > 0, 0, 1) * duty
    
    # Enhanced initial guesses
    guesses = [h.copy()]
    for p in [4, 5, 6, 7, 8, 9]:
        guesses.append(generate_periodic(p))
    for f, d in [(6, 0.2), (5, 0.3), (4, 0.4), (7, 0.1), (8, 0.25), (9, 0.35)]:
        guesses.append(generate_binary(f, d))
    for _ in range(3):
        guesses.append(np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1))
    
    for guess in guesses:
        if time.time() - start_time > budget_s - 50:
            break
        res = optimize.minimize(objective, guess, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': constraint_sum}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-10})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    if time.time() - start_time < budget_s - 50:
        res_de = optimize.differential_evolution(objective, bounds, popsize=40, maxiter=40,
                                             mutation=(0.7, 0.95), recombination=0.8, tol=1e-9)
        if res_de.success:
            res_slsqp = optimize.minimize(objective, res_de.x, method='SLSQP', bounds=bounds,
                                         constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                         options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-10})
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_c5 = objective(res_slsqp.x)
                best_h = res_slsqp.x
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 10:

        res = optimize.minimize(objective, best_h, method='Powell', bounds=bounds,

                               options={'xtol': 1e-9, 'ftol': 1e-9, 'maxiter': 30})
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
