    best_c5 = objective(best_h)

        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * x / period), 0, 1)
    
    def generate_triangle(period):

        return np.clip(0.5 - 0.5 * wave, 0, 1)
    
    def generate_binary(freq, duty):
