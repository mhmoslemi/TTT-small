import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period), 0, 1)
    
    def generate_modulated(period, mod):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period) * mod, 0, 1)
    
    def generate_square(freq, duty):
        return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points)) > 0) * duty, 0, 1)
    
    def generate_tri(freq):
        t = np.arange(n_points)
        return np.clip((2 * np.pi * freq * (t - t // (2 * np.pi * freq))) % (2 * np.pi) / np.pi, 0, 1)
    
    def generate_pulse_train(num_pulses, spacing):
        t = np.arange(n_points)
        pulse_width = 0.3
        pulses = np.zeros_like(t)
        for i in range(num_pulses):
            start = int(i * spacing)
            end = int(start + pulse_width * n_points)
            pulses[start:end] = 1
        return np.clip(pulses / num_pulses * (target_sum / np.sum(pulses)), 0, 1)
    
    def generate_random():
        h = np.random.rand(n_points)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        scaling = target_sum / np.sum(h)
        h = np.clip(h * scaling, 0, 1)
        if np.sum(h) != target_sum:
            h += (target_sum - np.sum(h)) / n_points
        return h
    
    initial_guesses = [initial_guess, generate_random()]
    for period in [4, 5, 6, 7, 8]:
        initial_guesses.append(generate_periodic(period))
    for period, mod in [(4, 0.6), (5, 0.7), (6, 0.8), (7, 0.9)]:
        initial_guesses.append(generate_modulated(period, mod))
    for freq, duty in [(5, 0.2), (6, 0.3), (7, 0.4), (8, 0.5)]:
        initial_guesses.append(generate_square(freq, duty))
    for freq in [5, 6, 7]:
        initial_guesses.append(generate_tri(freq))
    for num_pulses, spacing in [(3, 2), (4, 3), (5, 4)]:
        initial_guesses.append(generate_pulse_train(num_pulses, spacing))
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=100,
            maxiter=150,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,

                options={'maxiter': 150, 'ftol': 1e-10}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
