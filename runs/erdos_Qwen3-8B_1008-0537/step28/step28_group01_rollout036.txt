import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        n_points = len(initial_h_values)

        n_points = 60

    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_periodic(period, phase):
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / period + phase), 0, 1)
    
    def generate_triwave(period):
        return np.clip(0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / period)), 0, 1)
    
    def generate_binary(freq, duty, offset):
        return np.clip((np.sin(2 * np.pi * freq * np.arange(n_points) + offset) > 0) * duty, 0, 1)

        h[idx] = 1

    def generate_scaled_wave(freq, amp=0.5):
        return np.clip(0.5 + amp * np.sin(2 * np.pi * freq * np.arange(n_points)), 0, 1)
