        return h

        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h.copy()]
    for freq in range(3, 15):
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.25, 0.5, 0.75]:
        guesses.append(gen_saw(3, duty))
        guesses.append(gen_triangle(3, duty))
    for m in [2, 4, 6]:
        guesses.append(gen_spaced(m))
    for _ in range(15):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=50,
                                             mutation=(0.6, 0.95), recombination=0.8, tol=1e-10)
        return res.x

    for _ in range(4):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(6):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
