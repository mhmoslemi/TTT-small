import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_wave(freq, duty, wave_type='saw'):
        x = np.linspace(0, 1, n_points)
        if wave_type == 'saw':
            wave = 1 - np.abs(2 * freq * x - 1) * duty
        elif wave_type == 'pulse':
            wave = np.sin(2 * np.pi * freq * x)
            wave = np.clip((wave + 1) * duty / 2, 0, 1)
        elif wave_type == 'triangle':
            wave = 1 - np.abs(2 * freq * x - 1) * duty
        else:
            wave = np.zeros_like(x)
        return np.clip(wave, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h.copy()]
    for freq in range(3, 12):
        guesses.append(gen_wave(freq, 0.5, 'pulse'))
        guesses.append(gen_wave(freq, 0.5, 'saw'))
    for duty in [0.25, 0.33, 0.4, 0.5]:
        guesses.append(gen_wave(3, duty, 'saw'))
        guesses.append(gen_wave(3, duty, 'pulse'))
    for _ in range(10):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-9})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=200, maxiter=80,
                                             mutation=(0.6, 0.95), recombination=0.8, tol=1e-9)
        return res.x

        if time.time() - start_time > budget_s - 40:
