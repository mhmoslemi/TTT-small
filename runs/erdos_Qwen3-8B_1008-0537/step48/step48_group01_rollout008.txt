import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_sine(freq, amp, phase):
        x = np.linspace(0, 1, n_points)
        return np.clip(amp * np.sin(2 * np.pi * freq * x + phase) + 0.5, 0, 1)

    def gen_rect(pulse_width):
        x = np.linspace(0, 1, n_points)
        return np.clip(np.sum([np.heaviside(x - i * 0.5 - pulse_width/2, 1) for i in range(4)], axis=0), 0, 1)

    def gen_triangular():
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * x - 1), 0, 1)

    def gen_binary():
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 4 == 0 else 0
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in [0.5, 1.0, 1.5]:
        guesses.append(gen_sine(freq, 0.8, 0))
        guesses.append(gen_sine(freq, 0.8, np.pi/2))
    guesses.append(gen_rect(0.25))
    guesses.append(gen_rect(0.35))
    guesses.append(gen_triangular())
    guesses.append(gen_binary())
    for _ in range(3):
        guesses.append(np.random.rand(n_points))
    for _ in range(3):
        guesses.append(gen_sine(1.5, 0.7, 0))

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-8})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=50,
                                             mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
        return res.x

    for _ in range(2):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-8})
        return res.x if res.success else r

    for _ in range(7):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
