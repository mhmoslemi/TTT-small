import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_sine(f, d):
        x = np.linspace(0, 2, n_points)
        wave = 0.5 + 0.5 * np.sin(2 * np.pi * f * x)
        return np.clip(wave, 0, 1)

    def gen_rect(f, d):
        x = np.linspace(0, 2, n_points)
        wave = (np.sin(2 * np.pi * f * x) + 1) * d / 2
        return np.clip(wave, 0, 1)

    def gen_triangle(f, d):
        x = np.linspace(0, 2, n_points)
        wave = 1 - np.abs(2 * f * x - 1)
        return np.clip(wave * d, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h.copy()]
    for freq in range(1, 16):
        guesses.append(gen_sine(freq, 0.5))
    for freq in range(1, 15):
        guesses.append(gen_rect(freq, 0.6))
    for freq in range(1, 12):
        guesses.append(gen_triangle(freq, 0.5))
    for _ in range(10):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 250, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=120, maxiter=80,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-10)
        return res.x

    for _ in range(3):
        if time.time() - start_time > budget_s - 20:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-10})
        return res.x if res.success else r

    for _ in range(6):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
