            constraints=[{'type': 'eq', 'fun': cons}],
            options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
        )
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(
            obj,
            bounds,
            popsize=50,
            maxiter=50,
            mutation=(0.5, 0.95),
            recombination=0.8,
            tol=1e-9,
            disp=False
        )
        return res.x

    for _ in range(2):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(
            obj,
            r,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': cons}],
            options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
        )
        return res.x if res.success else r

    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
