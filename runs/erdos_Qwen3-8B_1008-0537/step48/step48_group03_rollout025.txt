import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.sin(2 * np.pi * freq * x) + 1) * 0.5, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    guesses = [h.copy()]
    for freq in [2, 3, 4]:
        guesses.append(gen_pulse(freq))
    for duty in [0.25, 0.35, 0.45]:
        guesses.append(gen_saw(3, duty))

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 500, 'ftol': 1e-10, 'disp': False})
        return res.x if res.success else r

    for g in guesses:
        h_opt = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                                 constraints=[{'type': 'eq', 'fun': cons}],
                                 options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}).x
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(4):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = optimize.minimize(obj, best_h, method='SLSQP', bounds=bounds,
