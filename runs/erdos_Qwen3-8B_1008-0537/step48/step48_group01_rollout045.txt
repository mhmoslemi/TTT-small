import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = np.array(initial_h_values, dtype=float)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulses(freq, duty):
        x = np.linspace(0, 2, n_points)
        T = 2.0 / freq
        on_time = T * duty
        wave = np.zeros_like(x)
        for i in range(len(x)):
            t = x[i]
            period = int(t / T)
            rem = t - period * T
            if rem < on_time:
                wave[i] = 1.0
        return wave

    def gen_saw(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = 1 - np.abs(2 * freq * x - 1) * duty
        return np.clip(wave, 0, 1)

    def gen_triangle(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = 1 - np.abs(2 * freq * x - 1) * duty
        return np.clip(wave, 0, 1)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 7):
        guesses.append(gen_pulses(freq, 0.5))
    for duty in [0.3, 0.4, 0.5]:
        guesses.append(gen_saw(3, duty))
        guesses.append(gen_triangle(3, duty))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        try:
            res = optimize.minimize(obj, g, method='SLSQP',
                                   bounds=bounds, constraints=[{'type': 'eq', 'fun': cons}],
                                   options={'maxiter': 150, 'ftol': 1e-8})
            return res.x
        except:
            return g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        try:
            res = optimize.differential_evolution(obj, bounds, popsize=50, maxiter=50,
                                                 mutation=(0.5, 0.95), recombination=0.8, tol=1e-9)
            return res.x
        except:

    for _ in range(2):
        if time.time() - start_time > budget_s - 30:

        h_opt = de_guess()

        c = obj(h_opt)

    def refine(r):
