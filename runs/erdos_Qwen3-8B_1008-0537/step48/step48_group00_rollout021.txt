        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_binary_step(m):

        positions = np.zeros(m)
        positions[:m//2] = 1
        positions[m//2:] = 0
        np.random.shuffle(positions)
        h = np.zeros(n_points)

        for i in range(m):
            start = int(i * n_points / m)
            end = int((i + 1) * n_points / m)
            h[start:end] = positions[i]
        return h

    def gen_random_binary():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for m in range(4, 12):
        if n_points % m == 0:
            guesses.append(gen_binary_step(m))
    for _ in range(5):
        guesses.append(gen_random_binary())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],

        return res.x if res.success else g

    for g in guesses:
