import time

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip((np.sin(2 * np.pi * freq * x) + 1) * duty / 2, 0, 1)

    def gen_pulse2(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(np.abs(np.sin(2 * np.pi * freq * x)) * duty, 0, 1)

    def gen_random(n=5):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-12})
        return res.x if res.success else r

    guesses = [h]
    for freq in range(2, 10):
        guesses.append(gen_pulse(freq, 0.5))
    for freq in range(2, 10):
        guesses.append(gen_pulse2(freq, 0.6))
