import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_pulse(freq, phase):
        t = np.linspace(0, 2, n_points)
        return np.clip((np.sin(2 * np.pi * freq * t + phase) + 1) * 0.5, 0, 1)

    def gen_rect(width, pos):
        h = np.zeros(n_points)
        start = int(pos - width / 2)
        end = int(pos + width / 2)
        h[start:end] = 1.0
        return np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points), h)

    def gen_saw(freq, duty):
        t = np.linspace(0, 2, n_points)
        wave = np.clip(1 - np.abs(2 * freq * t - 1) * duty, 0, 1)
        return wave

    def gen_triangle(freq, duty):
        t = np.linspace(0, 2, n_points)
        wave = np.clip(1 - np.abs(2 * freq * t - 1) * duty, 0, 1)
        return wave

    def gen_sparse():
        h = np.zeros(n_points)
        n = int(np.sqrt(n_points))
        for i in np.random.choice(n_points, size=n, replace=False):
            h[i] = 1.0
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 8):
        guesses.append(gen_pulse(freq, np.random.uniform(0, 2*np.pi)))
    for width in [0.4, 0.5, 0.6]:
        guesses.append(gen_rect(width, np.random.uniform(0, 2)))
    for duty in [0.3, 0.4, 0.6, 0.7]:
        guesses.append(gen_saw(3, duty))
    for duty in [0.3, 0.4, 0.6, 0.7]:
        guesses.append(gen_triangle(3, duty))
    for _ in range(5):
        guesses.append(gen_sparse())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-4})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=40, maxiter=30,
                                             mutation=(0.5, 0.8), recombination=0.8, tol=1e-10)
        return res.x

    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_guess(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
