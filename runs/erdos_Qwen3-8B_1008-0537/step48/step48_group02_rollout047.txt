import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_structured(m):
        l = 1/m
        s = 1/(m-1)
        start_positions = []
        current = 0.0
        for _ in range(m):
            start_positions.append(current)
            current += l
            if _ < m-1:
                current += s
        h = np.zeros(n_points)
        for start in start_positions:
            start_idx = int(start / dx)
            end = start + l
            end_idx = int(end / dx) + 1
            h[start_idx:end_idx] = 1.0
            h = np.clip(h, 0, 1)
        return h

    guesses = [h]
    for m in [3, 4, 5, 6, 7, 8, 9]:
        guesses.append(gen_structured(m))
    for freq in range(3, 10):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        guesses.append(np.clip((wave + 1) * 0.5, 0, 1))
    for duty in [0.25, 0.5, 0.75]:
        x = np.linspace(0, 1, n_points)
        guesses.append(np.clip(1 - np.abs(2 * 3 * x - 1) * duty, 0, 1))
    for _ in range(5):
        guesses.append(np.random.rand(n_points))
        guesses[-1] = guesses[-1] / np.sum(guesses[-1]) * target_sum

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-12})
        return res.x if res.success else g

    for g in guesses:
        h_opt = slsqp_guess(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def de_guess():
        res = optimize.differential_evolution(obj, bounds, popsize=60, maxiter=40,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-10)
        return res.x

    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess()
        h_opt = slsqp_guess(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def refine(r):
        res = optimize.minimize(obj, r, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-12})
        return res.x if res.success else r

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
