import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def de_guess(g):
        res = optimize.differential_evolution(obj, bounds, popsize=100, maxiter=50, mutation=(0.8, 1.0), recombination=0.9, tol=1e-8)
        return res.x

    def slsqp_refine(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 100, 'ftol': 1e-10})
        return res.x if res.success else g

    # Initial refinement
    best_h = slsqp_refine(best_h)
    best_c5 = obj(best_h)

    # Differential evolution
    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        h_opt = de_guess(best_h)
        h_opt = slsqp_refine(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Final refinement
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = slsqp_refine(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
