import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 180  # More samples for better resolution
    h = np.array(initial_h_values, dtype=float)
    h = np.resize(h, n_points)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_blocks(freq, duty):
        x = np.linspace(0, 1, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_saw(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)

    def gen_rect(freq, duty):
        x = np.linspace(0, 1, n_points)
        return np.clip(2 * np.abs(freq * x - 0.5) - 1 + duty, 0, 1)

    def gen_rand_perturb(h):
        noise = np.random.normal(0, 0.1, n_points)
        return np.clip(h + noise, 0, 1)

    intial_guesses = [np.copy(best_h), gen_rect(3, 0.5), gen_saw(3, 0.6), gen_blocks(4, 0.4)]
    for _ in range(3):
        intial_guesses.append(gen_rand_perturb(intial_guesses[-1]))

    def optimize_with_gd(guess):
        res = optimize.minimize(obj, guess, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 150, 'ftol': 1e-10})
        return res.x if res.success else guess

    for g in intial_guesses:
        h_opt = optimize_with_gd(g)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    def differential_evolution_best():
        res = optimize.differential_evolution(obj, bounds, popsize=30, maxiter=30,
                                             mutation=(0.6, 0.9), recombination=0.8, tol=1e-10)
        return res.x

    for _ in range(4):
        if time.time() - start_time > budget_s - 40:
            break
        h_opt = differential_evolution_best()
        h_opt = optimize_with_gd(h_opt)
        c = obj(h_opt)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = optimize_with_gd(best_h)
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return best_h, best_c5, n_points
