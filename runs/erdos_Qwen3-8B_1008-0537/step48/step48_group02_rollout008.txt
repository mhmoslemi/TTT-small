import scipy.optimize as optimize
import time

    def obj(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_symmetric_pulses(freq, duty):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * freq * x)
        return np.clip((wave + 1) * duty / 2, 0, 1)

    def gen_periodic(freq):
        x = np.linspace(0, 2, n_points)
        wave = np.fmod(freq * x, 1)
        return 0.5 * (1 + np.cos(np.pi * wave))

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 10):
        guesses.append(gen_symmetric_pulses(freq, 0.5))
    for freq in range(3, 10):
        guesses.append(gen_periodic(freq))
    for _ in range(5):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='L-BFGS-B', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 300, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:

        c = obj(h_opt)
        if c < best_c5:

        h_opt = de_guess()
