import time

    def obj(h_vals):

        return np.max(corr) * dx

    def cons(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def gen_wave(freq, shape, duty):
        x = np.linspace(0, 2, n_points)
        if shape == 'saw':
            return np.clip(1 - np.abs(2 * freq * x - 1) * duty, 0, 1)
        elif shape == 'tri':
            return 1 - np.abs(2 * freq * x - 1) * duty
        elif shape == 'sin':
            return 0.5 * (1 - np.cos(2 * np.pi * freq * x)) * duty
        return np.zeros(n_points)

    def gen_random():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h

    guesses = [h]
    for freq in range(3, 9):
        guesses.append(gen_wave(freq, 'saw', 0.4))
        guesses.append(gen_wave(freq, 'tri', 0.5))
        guesses.append(gen_wave(freq, 'sin', 0.6))
    for _ in range(3):
        guesses.append(gen_random())

    def slsqp_guess(g):
        res = optimize.minimize(obj, g, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': cons}],
                               options={'maxiter': 500, 'ftol': 1e-10})
        return res.x if res.success else g

    for g in guesses:

        return res.x

    for _ in range(2):
