            h[indices] = 1.0
            return h

    initial_guesses = [initial_h_values, generate_uniform_guess(), generate_random_guess()]
    best_h = initial_guesses[0]
    best_c5 = objective(best_h)
    
    start_time = time.time()
    for guess in initial_guesses:
        h = np.array(guess)
        res = optimize.minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        res = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x

    return best_h, best_c5, n_points
