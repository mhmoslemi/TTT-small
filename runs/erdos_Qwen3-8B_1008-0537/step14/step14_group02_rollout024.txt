import scipy.optimize as optimize
import time

    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    np.random.seed(seed)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_initial():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h
    
    initial_guesses = [initial_guess, np.zeros(n_points), generate_initial()]
    
    def optimize_from_guess(guess):
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=50,
                maxiter=300,
                mutation=(0.7, 1.0),
                recombination=0.8,
                tol=1e-10,
                vectorized=True
            )
            if result_de.success:
                result_cobyla = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='COBYLA',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_cobyla.success:
                    return result_cobyla.x
        except:
            pass
        return guess
    
    for guess in initial_guesses:

        current_c5 = objective(res)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = res
    
    # Perturb and optimize
    while time.time() - start_time < budget_s - 50:
