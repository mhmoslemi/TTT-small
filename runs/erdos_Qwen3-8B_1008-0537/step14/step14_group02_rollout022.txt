import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=100,
            maxiter=500,
            mutation=(0.5, 0.9),
            recombination=0.7,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_cobyla = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 1000, 'ftol': 1e-12, 'eps': 1e-9}
            )
            if result_cobyla.success:
                current_c5 = objective(result_cobyla.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
    except:
        pass
    
    binary_initial = np.zeros(n_points)

    binary_initial[:m] = 1.0
    remaining = target_sum - m
    if remaining > 0:
        binary_initial[m] += remaining
    result_cobyla_binary = optimize.minimize(

        x0=binary_initial,

        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
