        h = np.zeros(n_points)
        dx = 2.0 / n_points
        delta = 2.0 / m
        for i in range(m):
            x = i * delta
            idx = int(round(x / dx))
            if 0 <= idx < n_points:
                h[idx] += 1.0
        total = np.sum(h)
        if total > m:
            h /= total / m
        return h
    
    m_uniform = int(target_sum)
    uniform_guess = create_uniform_guess(n_points, m_uniform)
    uniform_guess[np.argmin(np.abs(np.linspace(0, 2, n_points) - (2 * m_uniform) / (m_uniform + 1))) % n_points] += (target_sum - m_uniform)
    
    def optimize_with_slsqp(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x, objective(result.x)
        return x0, objective(x0)
    
    best_h, best_c5 = optimize_with_slsqp(initial_guess)
    best_h, best_c5 = optimize_with_slsqp(uniform_guess)
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess, new_c5 = optimize_with_slsqp(perturbed)
        if new_c5 < best_c5:
            best_h, best_c5 = new_guess, new_c5
    
    return best_h, best_c5, n_points
