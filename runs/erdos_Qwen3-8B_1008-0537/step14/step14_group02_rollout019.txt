def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Random initial guess
    def generate_random_h():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h
    
    def optimize_h(h0):
        try:
            res_de = optimize.differential_evolution(
                objective, bounds, strategy='best1bin', popsize=30, maxiter=100, tol=1e-10,
                mutation=(0.5, 1.0), recombination=0.8, vectorized=True
            )
            if res_de.success:
                res = optimize.minimize(
                    objective, res_de.x, method='SLSQP', bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'ftol': 1e-10, 'maxiter': 200}
                )
                return res.x
        except:
            pass
        try:
            res = optimize.minimize(objective, h0, method='SLSQP', bounds=bounds,
                                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                    options={'ftol': 1e-10, 'maxiter': 200})
            return res.x
        except:
            pass
        return h0
    
    # Multiple initial guesses
    for _ in range(2):
        if time.time() - start_time > budget_s - 50:
            break
        h = generate_random_h()
        h_opt = optimize_h(h)
        c5 = objective(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    # Binary initial guess
    binary_initial = np.zeros(n_points)
    m = n_points // 2
    binary_initial[:m] = 1.0
    remaining = target_sum - m
    if remaining > 0:
        binary_initial[m] += remaining
    h_opt = optimize_h(binary_initial)
    c5 = objective(h_opt)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_opt
    
    # Perturb and optimize
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.005, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        h_opt = optimize.minimize(
            objective, perturbed, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'ftol': 1e-10, 'maxiter': 100}
        ).x
        c5 = objective(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    return best_h, best_c5, n_points
