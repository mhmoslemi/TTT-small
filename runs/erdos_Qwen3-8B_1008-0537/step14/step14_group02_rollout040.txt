import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        # Compute valid k shifts
        k_values = np.arange(0, 2.0, dx)
        max_c5 = 0.0
        for k in k_values:
            # Compute the overlap integral for this shift
            # Only consider x where x and x +k are in [0,2]
            corr = np.zeros(len(h))
            for i in range(len(h)):
                x = i * dx
                x_k = x + k
                if 0 <= x_k <= 2.0:
                    corr[i] = h[i] * (1.0 - h[np.int_(x_k / dx)])
            max_c5 = max(max_c5, np.sum(corr) * dx)
        return max_c5
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

        result_de = optimize.differential_evolution(

            vectorized=True
        )
