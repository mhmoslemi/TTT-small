    ideal_indices = np.round(positions / dx).astype(int).clip(0, n_points - 1)
    even_guess = np.zeros(n_points)
    for idx in ideal_indices:
        if idx < n_points:
            even_guess[idx] = 1.0
    remaining_pos = 2 * m / (m + 1)

    even_guess[remaining_idx] += remaining

    # Start with the even_guess
    best_h = even_guess
    best_c5 = objective(best_h)
    # Also try initial_h_values
    initial_guess = np.array(initial_h_values, dtype=float)
    best_c5_initial = objective(initial_guess)

        best_h = initial_guess
        best_c5 = best_c5_initial

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
