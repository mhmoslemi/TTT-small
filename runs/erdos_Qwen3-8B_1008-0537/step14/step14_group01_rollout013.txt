import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.zeros(n_points)
    m = int(target_sum)
    remaining = target_sum - m
    x = np.linspace(0, 2, m + 1)
    idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
    for i in range(m):
        initial_guess[idxs[i]] = 1.0
    if remaining > 0:
        pos = np.linspace(0, 2, n_points)
        best_idx = np.argmin(np.abs(pos - 1.0))
        initial_guess[best_idx] += remaining

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def optimize_slsqp(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-6}
        )
        return result

    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=50,
        maxiter=100,
        mutation=(0.8, 1.2),
        recombination=0.7,
        tol=1e-10,
        vectorized=True
    )
    if result_de.success:
        best_c5 = objective(result_de.x)
        best_h = result_de.x
    
    result_slsqp = optimize_slsqp(result_de.x)
    if result_slsqp.success:
        best_c5 = objective(result_slsqp.x)
        best_h = result_slsqp.x

    result_slsqp_initial = optimize_slsqp(initial_guess)
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x

    result_slsqp_uniform = optimize_slsqp(initial_guess)
    if result_slsqp_uniform.success:
        current_c5 = objective(result_slsqp_uniform.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_uniform.x

    while time.time() - start_time < budget_s - 30:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize_slsqp(perturbed)
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
