import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_even_spaced_guess():
        m = int(target_sum)
        remaining = target_sum - m
        positions = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), positions)
        uniform_guess = np.zeros(n_points)
        for i in range(m):
            uniform_guess[idxs[i]] = 1.0
        if remaining > 0:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (2 - 1e-8)))
            uniform_guess[idx % n_points] += remaining
        return uniform_guess
    
    def generate_random_guess():
        m = int(target_sum)
        remaining = target_sum - m
        random_guess = np.zeros(n_points)
        for i in range(m):
            random_guess[np.random.randint(n_points)] = 1.0
        if remaining > 0:
            random_guess[np.random.randint(n_points)] += remaining
        return random_guess
    
    def optimize_from_guess(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )

    init_guesses = [initial_guess, generate_even_spaced_guess(), generate_random_guess()]

            maxiter=40,

            recombination=0.9,

            x0=guess
        )
        if result_de.success:
            result_slsqp = optimize.minimize(

                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
