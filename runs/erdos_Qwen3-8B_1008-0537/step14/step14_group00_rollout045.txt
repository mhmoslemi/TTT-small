import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=100, **kwargs):

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_guess(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x if result.success else None

    initial_guesses = [initial_guess]

        indices = np.random.choice(n_points, size=num_ones, replace=False)
        random_guess[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            random_guess[idx] += remaining
        initial_guesses.append(random_guess)
    
    # Add evenly spaced initial guess
    m = int(target_sum)
    if m >= 1:
        positions = np.linspace(0, 2, m, endpoint=False)

        indices = np.unique(indices)
        h_even = np.zeros(n_points)
        for idx in indices:
            h_even[idx] = 1.0
        current_sum = np.sum(h_even)
        delta = target_sum - current_sum
        if abs(delta) > 1e-6:
            for i in range(n_points):
                if h_even[i] == 0:
                    h_even[i] += delta
