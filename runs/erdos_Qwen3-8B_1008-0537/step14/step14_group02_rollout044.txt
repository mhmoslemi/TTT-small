        dx = 2.0 / n_points

            pos = (i + 0.5) * dx
            idx = int(pos / dx)
            if 0 <= idx < n_points:
                h[idx] += 1.0
        current_sum = np.sum(h)
        extra = target_sum - current_sum
        h[0] += extra
        return h
