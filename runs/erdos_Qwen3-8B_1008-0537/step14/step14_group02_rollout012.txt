import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_initial():
        r = np.random.rand(n_points)
        r = r * target_sum / np.sum(r)
        return r
    
    def optimize_with_de(x0):
        try:
            result = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=50,
                maxiter=500,
                mutation=(0.7, 1.0),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                x0=x0
            )
            if result.success:
                return result.x
        except:
            return x0
    
    def optimize_with_cobyla(x0):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=x0,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 700, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                return result.x
        except:
            return x0
    
    def perturb_and_optimize(x0):
        while time.time() - start_time < budget_s - 100:
            perturbed = x0 + np.random.normal(0, 0.001, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)
            new_guess = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if new_guess.success:
                current_c5 = objective(new_guess.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = new_guess.x
        return best_h

            best_c5 = current_c5
            best_h = cobyla_result
    
    best_h = perturb_and_optimize(best_h)
