            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        return result.x if result.success else None
    
    def generate_spread_guess():
        m = int(target_sum)
        remaining = target_sum - m
        interval = 2.0 / m
        h = np.zeros(n_points)
        for i in range(m):
            pos = i * interval
            idx = int(pos / dx)
            if 0 <= idx < n_points:
                h[idx] = 1.0
        if remaining > 0:
            for i in range(n_points):
                if h[i] == 0.0:
                    h[i] += remaining
                    break
        return h
    
    def generate_random_guess():
        h = np.zeros(n_points)
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h[indices] = 1.0
        if remaining > 0:
            idx = np.random.choice(n_points, size=1, replace=False)
            h[idx] += remaining
        return h
    
    initial_guesses = []
    for _ in range(2):
        initial_guesses.append(generate_spread_guess())
    for _ in range(3):
        initial_guesses.append(generate_random_guess())
    initial_guesses.append(initial_guess)
    
    for guess in initial_guesses:
        result = optimize_guess(guess)
        if result is not None and objective(result) < best_c5:
            best_c5 = objective(result)
            best_h = result
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=250,
            mutation=(0.9, 1.1),
            recombination=0.8,
            tol=1e-9,
            vectorized=True,
            polish=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result_slsqp.success and objective(result_slsqp.x) < best_c5:
                best_c5 = objective(result_slsqp.x)
                best_h = result_slsqp.x
    except:
        pass
    
    for _ in range(2):
        if time.time() - start_time >= budget_s - 30:
            break
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success and objective(result.x) < best_c5:
                best_c5 = objective(result.x)
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
