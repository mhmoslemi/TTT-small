        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

        m = int(target_sum)

            uniform_guess[pos % n_points] += remaining
        return uniform_guess
    
    def generate_random_guess():
        r = np.random.rand(n_points)
        r /= np.sum(r) * target_sum
        return r
    
    guesses = [initial_guess, generate_uniform_guess(), generate_random_guess()]

            pass
    
    result_slsqp_initial = optimize.minimize(
