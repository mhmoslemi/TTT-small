    dx = 2.0 / n_points
    np.random.seed(seed)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    initial_guesses = []

import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    initial_guesses = []
    initial_guesses.append(np.array(initial_h_values, dtype=float))
    
    binary = np.zeros(n_points)
    m = n_points // 2
    binary[:m] = 1.0
    remaining = target_sum - m
    if remaining > 0:
        binary[m] += remaining
    initial_guesses.append(binary)
    
    dirichlet = np.random.dirichlet([1.0]*n_points, size=1)[0] * target_sum
    initial_guesses.append(dirichlet)
    
    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for guess in initial_guesses:
        try:
            de_res = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=50,
                maxiter=200,
                mutation=(0.7, 1.0),
                recombination=0.8,
                tol=1e-10,
                vectorized=True
            )
            if de_res.success:
                cobyla_res = optimize.minimize(
                    fun=objective,
                    x0=de_res.x,
                    method='COBYLA',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if cobyla_res.success:
                    current_c5 = objective(cobyla_res.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = cobyla_res.x
        except:
            pass
        
        try:
            cobyla_res = optimize.minimize(
                fun=objective,
                x0=guess,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if cobyla_res.success:
                current_c5 = objective(cobyla_res.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = cobyla_res.x
        except:
            pass
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
