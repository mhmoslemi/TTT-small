import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def generate_initial_guess():
        if np.random.rand() < 0.25:
            # Binary initial guess
            binary_initial = np.zeros(n_points)
            m = n_points // 2
            binary_initial[:m] = 1.0
            remaining = target_sum - m
            if remaining > 0:
                binary_initial[m] += remaining
            return binary_initial
        elif np.random.rand() < 0.25:
            # Random initial guess
            random_initial = np.random.uniform(0, 1, n_points)
            random_initial = random_initial / np.sum(random_initial) * target_sum
            return random_initial
        else:
            return initial_guess
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 5:
            break
        initial = generate_initial_guess()
        try:
            result_de = optimize.differential_evolution(
                fun=objective_de,
                bounds=bounds,
                popsize=50,
                maxiter=300,
                mutation=(0.7, 1.0),
                recombination=0.8,
                tol=1e-10,
                vectorized=True
            )
            if result_de.success:
                result_cobyla = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='COBYLA',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_cobyla.success:
                    current_c5 = objective(result_cobyla.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_cobyla.x
        except:
            pass
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 5:
            break
        perturbed = best_h + np.random.normal(0, 0.003, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
