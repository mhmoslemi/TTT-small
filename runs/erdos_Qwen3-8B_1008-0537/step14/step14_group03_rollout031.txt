    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    # Generate alternative initial guesses
    m = int(target_sum)
    remaining = target_sum - m
    x = np.linspace(0, 2, m + 1)
    idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
    uniform_guess = np.zeros(n_points)
    for i in range(m):
        uniform_guess[idxs[i]] = 1.0
    if remaining > 0:
        uniform_guess[np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1))) % n_points] += remaining
    np.random.seed(seed)
    random_guess = np.random.rand(n_points)
    random_guess = random_guess / np.sum(random_guess) * target_sum
    initial_guesses = [initial_guess, uniform_guess, random_guess]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    # Run DE on each initial guess
    for guess in initial_guesses:
        try:
            result_de = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=15,
                maxiter=150,
                mutation=(0.8, 1.0),
                recombination=0.9,
                tol=1e-9,
                vectorized=True,
                polish=True
            )
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success and result_slsqp.fun < best_c5:
                best_c5 = result_slsqp.fun
                best_h = result_slsqp.x
        except:
            pass

    # Local search with perturbation
    start_time = time.time()
    for _ in range(15):
        if time.time() - start_time > budget_s - 50:
            break
        perturbed = best_h + np.random.normal(0, 0.0005, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success and new_guess.fun < best_c5:
            best_c5 = new_guess.fun
            best_h = new_guess.x

    return best_h, best_c5, n_points
