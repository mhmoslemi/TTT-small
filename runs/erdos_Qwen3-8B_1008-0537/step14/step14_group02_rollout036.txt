        perturbed = best_h + np.random.normal(0, 0.001, size=n_points)

        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
