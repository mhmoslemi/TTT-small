import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate a more spread-out initial guess
    m = int(target_sum)
    remaining = target_sum - m
    x_ideal = np.linspace(0, 2, m + 1)[:-1]
    sample_positions = np.linspace(0, 2, n_points)
    idxs = np.array([np.argmin(np.abs(sample_positions - xi)) for xi in x_ideal])
    uniform_guess = np.zeros(n_points)
    for i in range(m):
        uniform_guess[idxs[i]] = 1.0
    if remaining > 0:
        add_pos = 2.0 * m / (m + 1)
        idx = np.argmin(np.abs(sample_positions - add_pos))
        uniform_guess[idx] += remaining
    
    # Try optimizing from the new uniform guess
    result_slsqp_uniform = optimize.minimize(
        fun=objective,
        x0=uniform_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_uniform.success:
        current_c5 = objective(result_slsqp_uniform.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_uniform.x
    
    # Try initial guess
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,

        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:

    # Differential evolution with fewer iterations to save time

        vectorized=True
    )
    if result_de.success:
        result_slsqp_de = optimize.minimize(
            fun=objective,
            x0=result_de.x,
            method='SLSQP',

            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}

        if result_slsqp_de.success:

                best_c5 = current_c5
                best_h = result_slsqp_de.x
    
    # Local perturbation and refinement
    perturb_steps = 3
    for _ in range(perturb_steps):

    return best_h, best_c5, n_points
