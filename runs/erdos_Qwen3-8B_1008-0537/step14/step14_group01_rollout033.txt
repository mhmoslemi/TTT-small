import scipy.optimize as optimize
import time

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_uniform_guess():
        m = int(target_sum)
        remaining = target_sum - m
        dx = 2.0 / n_points
        positions = np.linspace(dx/2, 2 - dx/2, m)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), positions)
        uniform_guess = np.zeros(n_points)
        for i in range(m):
            uniform_guess[idxs[i]] = 1.0
        if remaining > 1e-6:
            avg_pos = (2 - dx/2) * m / (m + 1)
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - avg_pos))
            uniform_guess[idx] += remaining
        return uniform_guess
    
    try:

        result = optimize.minimize(

            x0=uniform_guess,
            method='SLSQP',
