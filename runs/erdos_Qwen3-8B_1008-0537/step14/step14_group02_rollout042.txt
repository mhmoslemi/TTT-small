import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=100,
            maxiter=500,
            mutation=(0.5, 1.5),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            polish=False
        )
        if result_de.success:
            result_cobyla = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_cobyla.success:
                current_c5 = objective(result_cobyla.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
    except:
        pass
    
    # Random binary initial guess
    binary_initial = np.zeros(n_points, dtype=float)
    num_ones = int(np.round(target_sum))
    selected_indices = np.random.choice(n_points, num_ones, replace=False)
    binary_initial[selected_indices] = 1.0
    remaining = target_sum - num_ones
    if abs(remaining) > 1e-6:
        if remaining > 0:
            idx = np.random.randint(n_points)
            binary_initial[idx] += remaining
        else:
            idx = np.random.randint(n_points)
            binary_initial[idx] += remaining
    result_cobyla_binary = optimize.minimize(
        fun=objective,
        x0=binary_initial,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_cobyla_binary.success:
        current_c5 = objective(result_cobyla_binary.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_cobyla_binary.x
    
    # Perturb and optimize
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.01, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
