import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    # Try basinhopping for global optimization

        )
        if result_bh.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_bh.x,
                method='SLSQP',
                bounds=bounds,

                options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    # Try uniform guess
    m = int(target_sum)
    remaining = target_sum - m
    x = np.linspace(0, 2, m + 1)
    idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
    uniform_guess = np.zeros(n_points)
    for i in range(m):

    if remaining > 0:
        uniform_guess[np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1))) % n_points] += remaining
    result_slsqp_uniform = optimize.minimize(
        fun=objective,
        x0=uniform_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_uniform.success:

            best_c5 = current_c5
            best_h = result_slsqp_uniform.x
    
    # Perturb around the best solution
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0002, size=n_points)

import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Basinhopping for global optimization
    try:
        result_bh = optimize.basinhopping(
            fun=objective, 
            x0=initial_guess, 
            niter=50, 
            T=1.0, 
            stepsize=0.1, 
            take_last=True
        )
        if result_bh.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_bh.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    # Uniform guess optimization
    m = int(target_sum)
    remaining = target_sum - m
    x = np.linspace(0, 2, m + 1)
    idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
    uniform_guess = np.zeros(n_points)
    for i in range(m):
        uniform_guess[idxs[i]] = 1.0
    if remaining > 0:
        uniform_guess[np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1))) % n_points] += remaining
    result_slsqp_uniform = optimize.minimize(
        fun=objective,
        x0=uniform_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_uniform.success:
        current_c5 = objective(result_slsqp_uniform.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_uniform.x
    
    # Local perturbations
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
