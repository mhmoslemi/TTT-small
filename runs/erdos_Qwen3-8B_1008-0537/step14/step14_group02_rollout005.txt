import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=60,
            maxiter=350,
            mutation=(0.75, 1.0),
            recombination=0.85,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_cobyla = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 600, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_cobyla.success:
                current_c5 = objective(result_cobyla.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
    except:
        pass

        h[:m] = 1.0

        if remaining > 0:
            h[m:] += remaining / (n_points - m)
        return h
    
    result_cobyla_uniform = optimize.minimize(
