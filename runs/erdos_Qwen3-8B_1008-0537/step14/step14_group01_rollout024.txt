        idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - 1.0))
        pos = np.append(pos, idx)
    random = np.zeros(n_points)
    random[pos] = 1.0
    initial_guesses.append(random)

    best_h = initial_guesses[0]
    best_c5 = objective(best_h)

    # Evaluate all initial guesses
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    # Perturb and refine
    for _ in range(5):
        perturbed = best_h + np.random.normal(0, 0.0001, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return best_h, best_c5, n_points
