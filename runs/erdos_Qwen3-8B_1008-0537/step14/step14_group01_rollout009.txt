import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_uniform_guess():
        m = int(target_sum)
        remaining = target_sum - m
        pos = np.linspace(0, 2, m + 1)
        pos = pos[:-1]
        idxs = np.searchsorted(np.linspace(0, 2, n_points), pos)
        guess = np.zeros(n_points)
        for i in range(m):
            guess[idxs[i]] = 1.0
        if remaining > 0:
            for i in range(m):
                guess[idxs[i]] += remaining / m
        return guess
    
    uniform_guess = generate_uniform_guess()
    
    def optimize_slsqp(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result
    
    def perturb(x, std=0.0002):
        return np.clip(x + np.random.normal(0, std, size=len(x)), 0, 1)
    
    def evaluate_candidate(x):
        nonlocal best_c5
        try:
            res = optimize_slsqp(x)
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
            return res.x
        except:
            return x
    
    # Initial optimization
    best_h = evaluate_candidate(initial_guess)
    best_h = evaluate_candidate(uniform_guess)
    
    # Differential evolution
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=50,
            maxiter=300,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            best_h = evaluate_candidate(result_de.x)
    except:
        pass
    
    # Perturbation
    while time.time() - start_time < budget_s - 50:
        perturbed = perturb(best_h)
        best_h = evaluate_candidate(perturbed)
    
    return best_h, best_c5, n_points
