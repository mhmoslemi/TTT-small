    result_cobyla_random = optimize.minimize(
        fun=objective,
        x0=random_initial,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 600, 'ftol': 1e-8, 'eps': 1e-9}
    )
    if result_cobyla_random.success:
        current_c5 = objective(result_cobyla_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_cobyla_random.x
    
    # Perturb and optimize
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0005, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        new_guess = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
