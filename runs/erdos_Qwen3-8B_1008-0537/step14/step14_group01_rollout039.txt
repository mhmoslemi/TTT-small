import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate a new initial guess with even spacing
    m = int(n_points / 2)
    positions = np.linspace(0, 2, m + 1)[:-1]
    even_guess = np.zeros(n_points)
    for i in range(m):
        idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - positions[i]))
        even_guess[idx] = 1.0
    # Adjust for exact sum
    remaining = target_sum - m
    if remaining > 0:

        even_guess[idx % n_points] += remaining
    even_guess = np.clip(even_guess, 0, 1)
    
    # Try even spacing guess
    result_slsqp_even = optimize.minimize(
        fun=objective,

        method='SLSQP',

    if result_slsqp_even.success:

            tol=1e-9,
            vectorized=True
        )
        if result_de.success:
