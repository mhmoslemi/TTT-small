import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guesses = [initial_h_values.copy()]
    
    # Generate evenly spaced initial guess
    m = int(target_sum)
    evenly_spaced = np.zeros(n_points)
    positions = np.linspace(0, 2, m)
    for p in positions:
        idx = int(p / dx)
        if idx < n_points:
            evenly_spaced[idx] = 1.0
    initial_guesses.append(evenly_spaced)
    
    # Binary initial guess
    binary_initial = np.zeros(n_points)
    m = n_points // 2
    binary_initial[:m] = 1.0
    remaining = target_sum - m
    if remaining > 0:
        binary_initial[m] += remaining
    initial_guesses.append(binary_initial)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = np.zeros(n_points)
    best_c5 = 1.0
    start_time = time.time()
    
    def objective_de(h_values):
        return objective(h_values)
    
    for guess in initial_guesses:
        try:
            result_de = optimize.differential_evolution(
                fun=objective_de,
                bounds=bounds,
                popsize=50,
                maxiter=200,
                mutation=(0.7, 1.0),
                recombination=0.8,
                tol=1e-10,
                vectorized=True,
                seed=seed
            )
            if result_de.success:
                result_cobyla = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_cobyla.success:
                    current_c5 = objective(result_cobyla.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_cobyla.x
        except:
            pass
    
    # Perturb and optimize
    while time.time() - start_time < budget_s - 50:
        if np.random.random() < 0.5:
            perturbed = best_h + np.random.normal(0, 0.001, size=n_points)
        else:
            perturbed = best_h * (1 - np.random.uniform(0, 0.2, size=n_points))
        perturbed = np.clip(perturbed, 0, 1)
        try:
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
