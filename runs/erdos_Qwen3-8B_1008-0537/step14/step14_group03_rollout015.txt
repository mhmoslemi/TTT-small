import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    target_sum = n_points / 2
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def run_optimize(x0):
        try:
            de_result = optimize.differential_evolution(
                fun=objective,
                bounds=bounds,
                popsize=40,
                maxiter=60,
                mutation=(0.5, 0.9),
                recombination=0.9,
                tol=1e-10,
                vectorized=True
            )
            if de_result.success:
                slsqp_result = optimize.minimize(
                    fun=objective,
                    x0=de_result.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 400, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if slsqp_result.success:
                    return slsqp_result.x
        except:
            pass
        return x0
    
    guesses = [
        initial_guess,
        np.zeros(n_points),
        np.zeros(n_points)
    ]
    m = n_points // 2
    uniform_idx = np.searchsorted(np.linspace(0, 2, n_points), np.linspace(0, 2, m + 1))
    guesses[1][uniform_idx[:m]] = 1.0
    random_idx = np.random.choice(n_points, size=m, replace=False)
    guesses[2][random_idx] = 1.0
    
    for guess in guesses:
        res = run_optimize(guess)
        curr_c5 = objective(res)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = res
    
    while time.time() - start_time < budget_s - 10:
        perturbed = best_h + np.random.normal(0, 0.0001, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        res = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
