import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=np.float64)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate uniform initial guess
    m = int(target_sum)
    remaining = target_sum - m
    x = np.linspace(0, 2, m + 1)
    idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
    uniform_guess = np.zeros(n_points)
    for i in range(m):
        uniform_guess[idxs[i]] = 1.0
    if remaining > 0:
        pos = (np.max(x) - np.min(x)) * m / (m + 1)
        idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos))
        uniform_guess[idx] += remaining
    
    # Generate structured guess with 1s spaced evenly
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    scaled_structured = structured_guess * (target_sum / (np.sum(structured_guess) * dx))
    scaled_structured = np.clip(scaled_structured, 0, 1)
    
    candidates = [initial_guess, uniform_guess, scaled_structured]
    
    def run_optimization(start_guess):
        result = optimize.minimize(
            fun=objective,
            x0=start_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x
        return start_guess
    
    for guess in candidates:
        current_h = run_optimization(guess)
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

    return best_h, best_c5, n_points
