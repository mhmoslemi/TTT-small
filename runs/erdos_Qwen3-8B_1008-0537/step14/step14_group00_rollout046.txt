        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    def generate_structured_guess(n_points):
        K = n_points // 2
        dx = 2.0 / n_points
        ideal_positions = np.linspace(0, 2, K, endpoint=False)
        sample_indices = np.round(ideal_positions / dx).astype(int)
        sample_indices = sample_indices[sample_indices < n_points]
        sample_indices = np.unique(sample_indices)
        h = np.zeros(n_points)
        h[sample_indices] = 1.0
        # Ensure sum matches target_sum
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-8:
            pos = 0
            h[pos] += diff
        return h
    
    structured_guess = generate_structured_guess(n_points)
    initial_guesses = [initial_guess, structured_guess]
    
    def optimize_guess(guess):
        result = optimize.minimize(
