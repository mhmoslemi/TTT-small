```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = np.array(initial_h_values)
    best_c5 = objective(best_h)
    start_time = time.time()

    def create_sine(n_points):
        x = np.linspace(0, 2, n_points)
        return 0.5 * (1 + np.sin(np.pi * x))

    for init in [np.array(initial_h_values), create_sine(n_points)]:
        res = optimize.minimize(
            objective, init, method='COBYLA',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.001, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        res = optimize.minimize(
            objective, perturbed, method='COBYLA',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return best_h, best_c5, n_points
```
