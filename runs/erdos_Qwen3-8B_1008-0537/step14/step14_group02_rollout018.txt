import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points

    def objective(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def refine_with_cobyla(x0):
        res = optimize.minimize(
            objective,
            x0,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        return res.x if res.success else x0

    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=20,
            maxiter=100,
            mutation=(0.6, 0.9),
            recombination=0.7,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            best_h = refine_with_cobyla(result_de.x)
            best_c5 = objective(best_h)
    except:
        pass

    def binary_guess():
        binary = np.zeros(n_points)
        binary[:n_points//2] = 1.0
        return refine_with_cobyla(binary)

    best_h = refine_with_cobyla(initial_guess)
    best_c5 = objective(best_h)
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 50:
            break
        h_perturbed = best_h + np.random.normal(0, 0.005, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = optimize.minimize(
            objective,
            h_perturbed,

            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-9}
        ).x
        c5 = objective(h_perturbed)
        if c5 < best_c5:
            best_h = h_perturbed
            best_c5 = c5

    return best_h, best_c5, n_points
