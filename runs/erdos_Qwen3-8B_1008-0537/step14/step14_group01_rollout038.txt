import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_uniform_guess():
        uniform_guess = np.zeros(n_points)
        m = int(target_sum)
        remaining = target_sum - m
        x = np.linspace(0, 2, m + 1)
        idxs = np.searchsorted(np.linspace(0, 2, n_points), x)
        for i in range(m):
            uniform_guess[idxs[i]] = 1.0
        if remaining > 0:
            x_pos = np.max(x) - np.min(x) * m / (m + 1)
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - x_pos))
            uniform_guess[idx] += remaining
        return uniform_guess
    
    def optimize_with_cobyla(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 1000, 'disp': False}
        )
        if result.success:
            return result.x
        return x0
    
    initial_guesses = [
        initial_guess.copy(),
        generate_uniform_guess()
    ]
    
    for guess in initial_guesses:
        result = optimize_with_cobyla(guess)
        current_c5 = objective(result)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result
    
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0005, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize_with_cobyla(perturbed)
        current_c5 = objective(result)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result
    
    return best_h, best_c5, n_points
