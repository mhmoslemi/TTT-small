import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # SLSQP initial refinement
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    
    # Differential evolution with tuned parameters
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=60,
            maxiter=300,
            mutation=(0.8, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp_de = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp_de.success and result_slsqp_de.fun < best_c5:
                best_c5 = result_slsqp_de.fun
                best_h = result_slsqp_de.x
    except:
        pass
    
    # Local refinement with BFGS
    while time.time() - start_time < budget_s - 50:
        try:
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        except:
            break
    
    return best_h, best_c5, n_points
