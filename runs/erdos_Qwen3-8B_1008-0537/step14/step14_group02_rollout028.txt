import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = objective(best_h)

    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=50,
            maxiter=600,
            mutation=(0.8, 1.0),

                options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_cobyla.success:

    m = n_points // 2
    binary_initial[:m] = 1.0
    remaining = target_sum - m
    if remaining > 0:
        binary_initial[m] += remaining
    result_cobyla_binary = optimize.minimize(
