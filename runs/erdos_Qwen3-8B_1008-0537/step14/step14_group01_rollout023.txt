        idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - (np.max(x) - np.min(x)) * m / (m + 1))) % n_points
        uniform_guess[idx] += remaining
    
    # Random guess
    np.random.seed(seed)
    random_guess = np.random.rand(n_points)
    random_guess = random_guess / np.sum(random_guess) * target_sum
    
    # List of initial guesses
    initial_guesses = [initial_guess, uniform_guess, random_guess]
    
    # Function to run optimization on a given guess
    def run_optimization(guess):
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            return result.x, current_c5
        return None, float('inf')
    
    # Run optimization for each initial guess
    for guess in initial_guesses:
        h, c5 = run_optimization(guess)
        if h is not None:
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    
    # Perturb the best solution and refine
    while time.time() - start_time < budget_s - 50:
        perturbed = best_h + np.random.normal(0, 0.0002, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
