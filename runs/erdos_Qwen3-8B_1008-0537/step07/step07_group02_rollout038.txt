    result = optimize.differential_evolution(
        objective,
        bounds,
        popsize=20,
        maxiter=50,
        tol=1e-10,
        mutation=(0.5, 1.0),
        recombination=0.9,
        strategy='best1bin',
        seed=seed,
        polish=True,
        disp=False,
    )

    best_h = result.x
    best_c5 = objective(best_h)

    # SLSQP refinement
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )

    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = objective(best_h)

    # Perturbation with exponential cooling
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        decay = 0.9 ** _
        perturb = np.random.normal(0, decay * 0.05, size=n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        res = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res.success:
            best_c5 = objective(res.x)
            best_h = res.x

    return best_h, best_c5, n_points
