    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_h.copy()
    best_c5 = 0.0
    
    start_time = time.time()
    
    # Differential Evolution for global search
    result_de = optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        callback=None,
        args=()
    )
    de_time = time.time() - start_time
    if de_time < budget_s:
        h_best = result_de.x
        # Local refinement with L-BFGS-B
        result_local = optimize.minimize(
            fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            x0=h_best,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'disp': False}
        )
        if result_local.success:
            best_c5 = result_local.fun
            best_h = result_local.x
        else:
            best_c5 = np.max(np.correlate(h_best, 1 - h_best, mode='full')) * dx
    else:
        # Fall back to initial_h if DE took too long
        h_best = initial_h
        best_c5 = np.max(np.correlate(h_best, 1 - h_best, mode='full')) * dx
    
    return best_h, best_c5, n_points
