import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def create_guess():
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i] = 1.0
        excess = target_sum - np.sum(h)
        if excess > 0:
            h[-1] += excess
        elif excess < 0:
            h[-1] -= -excess
        return h

    def generate_symmetric_guess():
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i] = 1.0
        c = target_sum - np.sum(h)
        if c > 0:
            h[-1] += c
        elif c < 0:
            h[-1] -= -c
        return h

    def generate_perturbed_guess(base):
        perturb = np.random.normal(0, 0.05, size=n_points)
        return np.clip(base + perturb, 0, 1)

    initial_guesses = [
        np.array(initial_h_values, dtype=float),
        np.zeros(n_points) + 0.5,
        np.array([0.0] * (n_points // 2) + [1.0] * (n_points // 2)),
        create_guess(), generate_symmetric_guess()
    ]

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)

    start_time = time.time()

    for guess in initial_guesses[1:]:
        c5 = objective(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = guess.copy()

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        for guess in initial_guesses[:2]:
            result = optimize.minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        perturbed = generate_perturbed_guess(best_h)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:

            if current_c5 < best_c5:
