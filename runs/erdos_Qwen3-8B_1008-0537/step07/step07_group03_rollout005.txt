import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    def constraint_jacobian(h_values):
        return np.ones_like(h_values)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start with SLSQP for better convergence
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        # Structured perturbations to avoid local traps
        noise = np.ones(n_points) * (np.random.rand(n_points) - 0.5) * 0.1
        perturbed = np.clip(initial_h + noise, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_jacobian}],
            options={'maxiter': 80, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # COBYLA refinement for local optimization
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 150, 'disp': False}
    )
    if result.success:
        best_c5 = objective(result.x)
        best_h = result.x
    
    return best_h, best_c5, n_points
