import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    starting_points = []
    starting_points.append(best_h)
    starting_points.append(generate_random_guess(n_points))
    for _ in range(8):
        perturb = np.random.normal(0, 0.05, size=n_points)
        starting_points.append(best_h + perturb)
    
    for guess in starting_points:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    for _ in range(50):
        if time.time() - start_time >= budget_s - 5:
            break
        perturb = np.random.normal(0, 0.01, size=n_points)
        new_guess = best_h + perturb
        result = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
