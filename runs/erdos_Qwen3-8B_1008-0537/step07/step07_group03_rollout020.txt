    start_time = time.time()
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                pattern[i] = 1.0
            else:
                pattern[i] = 0.0
        current_sum = np.sum(pattern)
        target = n_points / 2
        diff = target - current_sum
        if diff > 0:
            pattern[-int(diff):] = 1.0
        elif diff < 0:
            pattern[:int(-diff)] = 0.0
        perturbed = pattern
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'disp': False}
    )
    if result.success:
        best_c5 = objective(result.x)
        best_h = result.x
    
    return best_h, best_c5, n_points
