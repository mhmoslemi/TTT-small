            guess[idx] = 1.0
        current_sum = np.sum(guess)
        remaining = target_sum - current_sum
        if remaining > 0:
            idx = np.random.randint(n)
            guess[idx] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        def objective_de(h_values):
            h = np.array(h_values)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            constraint_violation = abs(np.sum(h) - target_sum)
            penalty = 1e6 * constraint_violation
            return c5 + penalty
        
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=20,
            maxiter=80,
            mutation=(0.7, 1.1),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    structured_guess = generate_structured_guess(n_points)
    result_slsqp_structured = optimize.minimize(
        fun=objective,
        x0=structured_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_structured.success:
        current_c5 = objective(result_slsqp_structured.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_structured.x
    
    random_guess = np.random.rand(n_points)
    random_guess = (random_guess * target_sum) / np.sum(random_guess)
    result_slsqp_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_random.success:
        current_c5 = objective(result_slsqp_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_random.x
    
    while time.time() - start_time < budget_s - 20:
        perturbation = np.random.normal(0, 0.01, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        result_slsqp_perturb = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp_perturb.success:
            current_c5 = objective(result_slsqp_perturb.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp_perturb.x
    
    return best_h, best_c5, n_points
