import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start with SLSQP and varied perturbations
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Gaussian noise with dynamic scaling
        noise_std = 0.07 * (1.0 / np.sqrt(n_points))
        noise = np.random.normal(0, noise_std, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    # Refinement with SLSQP
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
    )
    if result.success:
        best_c5 = objective(result.x)
        best_h = result.x
    
    return best_h, best_c5, n_points
