import scipy.optimize as optimize
import time

    def create_alternate_guess(n):
        guess = np.zeros(n)
        guess[:n//2] = 1.0
        return guess
    
    def create_spaced_guess(n):
        guess = np.zeros(n)
        spacing = 1.0
        num_ones = int(target_sum + 0.5)
        for i in range(n):
            if i % int(n / num_ones) == 0:
                guess[i] = 1.0
        current_sum = np.sum(guess)
        diff = target_sum - current_sum
        if diff > 0:
            guess[-1] += diff
        elif diff < 0:
            guess[-1] -= -diff
        return guess
    
    def create_random_guess(n):
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            guess[-1] += remaining
        return guess

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
