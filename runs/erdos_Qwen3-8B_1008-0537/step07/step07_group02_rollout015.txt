    initial_guess = np.array(initial_h_values, dtype=float)

        alternate_guess[i] = 1.0 if i % 2 == 0 else 0.0
    current_sum = np.sum(alternate_guess)
    diff = target_sum - current_sum
    if diff > 0:
        alternate_guess[-1] += diff
    elif diff < 0:
        alternate_guess[-1] -= -diff

    random_guess = np.zeros(n_points)
    num_ones = int(target_sum + 0.5)
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    random_guess[indices] = 1.0
    current_sum = np.sum(random_guess)
    diff = target_sum - current_sum
    if diff > 0:
        random_guess[-1] += diff
    elif diff < 0:
        random_guess[-1] -= -diff

    def even_guess(n):
        m = n // 2
        guess = np.zeros(n)
        for i in range(m):
            pos = int(2 * i / m * n)
            guess[pos] = 1.0
        current_sum = np.sum(guess)
        diff = target_sum - current_sum
        if diff > 0:
            guess[-1] += diff
        elif diff < 0:
            guess[-1] -= -diff
        return guess

    even_guess_val = even_guess(n_points)

    initial_guesses = [initial_guess, alternate_guess, random_guess, even_guess_val]

    def objective(h_values):

        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()

    start_time = time.time()
    for candidate in best_candidates:
        result = optimize.minimize(
