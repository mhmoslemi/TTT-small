import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def generate_even_spaced_guess(n_points):
        m = n_points // 2
        k = n_points / m
        indices = []
        for i in range(n_points):
            if i % k < k / 2:
                indices.append(i)
        indices = sorted(indices)[:m]
        guess = np.zeros(n_points)
        for i in indices:
            guess[i] = 1.0
        return guess

    initial_guesses = [np.array(initial_h_values), 
                      generate_alternate_guess(n_points),
                      generate_random_guess(n_points), 
                      generate_alternating_guess(n_points),
                      generate_even_spaced_guess(n_points)]
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)
    
    for guess in initial_guesses[1:]:
        current_c5 = objective(guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = guess.copy()
    
    start_time = time.time()
    
    for candidate in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=candidate,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.02 + 0.01 * np.cos(np.pi * _ / 20)
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',

    return best_h, best_c5, n_points
