import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        # Enhanced DE with tighter constraints
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=30,
            maxiter=100,
            mutation=(0.6, 0.9),
            recombination=0.8,
            tol=1e-8,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    result_slsqp_initial = optimize.minimize(

        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:

            best_h = result_slsqp_initial.x
    
    # Improved random guess with uniform spacing
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(np.floor(target_sum))
        remaining = target_sum - num_integers
        positions = np.linspace(0, n-1, n, endpoint=False)
        np.random.shuffle(positions)
        guess = np.zeros(n)
        guess[:num_integers] = 1.0
        if remaining > 0:
            guess[np.random.choice(n, size=1, replace=False)] += remaining
        return guess
    
    random_guess = generate_random_guess(n_points)
    result_slsqp_random = optimize.minimize(

        perturbation = np.random.normal(0, 0.05, size=n_points)
