import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def generate_structured_guess(num_blocks):
        h = np.zeros(n_points)
        block_width = 1.0 / num_blocks
        for i in range(num_blocks):
            center = (i + 0.5) * (2.0 / num_blocks)
            start = center - block_width / 2
            end = center + block_width / 2
            idx = np.where((np.arange(n_points) * dx >= start) & (np.arange(n_points) * dx <= end))[0]
            h[idx] = 1.0
        return h
    
    initial_guesses = [np.array(initial_h_values)]
    for num_blocks in [1, 2, 3]:
        initial_guesses.append(generate_structured_guess(num_blocks))
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    
    for guess in initial_guesses:
        current_c5 = objective(guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = guess.copy()
    
    start_time = time.time()
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if result.success:
            best_h = result.x
            best_c5 = objective(best_h)
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.02 * (1 - _ / 20)
        perturbed = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
