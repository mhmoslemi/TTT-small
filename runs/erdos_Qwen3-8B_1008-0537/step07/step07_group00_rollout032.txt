            )

            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x

            # Adaptive noise reduction for final phase
            if phase == 2:
                noise_scale *= 0.5

    return best_h, best_c5, n_points
