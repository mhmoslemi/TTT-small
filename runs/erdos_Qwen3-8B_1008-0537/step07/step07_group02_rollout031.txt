            guess[n-1-i] = guess[i]
        current_sum = np.sum(guess)
        diff = target_sum - current_sum
        if diff > 0:
            guess[-1] += diff
        elif diff < 0:
            guess[-1] -= -diff
        return guess
    
    symmetric_guess = create_symmetric_guess(n_points)
    
    def generate_random_guess(n):
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            guess[-1] += remaining
        return guess
    
    random_guess = generate_random_guess(n_points)
    
    def generate_alternating_guess(n):
        guess = np.zeros(n)
        for i in range(n):
            guess[i] = 1.0 if i % 2 == 0 else 0.0

            guess[-1] += diff

            guess[-1] -= -diff

        return np.max(corr) * dx
    
    def constraint_sum(h_values):
