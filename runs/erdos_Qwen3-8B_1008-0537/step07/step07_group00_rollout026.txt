import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        idx = np.random.choice(n_points, size=3, replace=False)
        perturb = np.zeros(n_points)
        perturb[idx] = np.random.uniform(-0.2, 0.2, size=3)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)

        result = optimize.minimize(
            fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'ftol': 1e-10, 'maxiter': 100, 'disp': False}
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
