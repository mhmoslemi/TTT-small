def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_binary_guess(num_ones):
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        guess = np.zeros(n_points)
        guess[indices] = 1
        return guess

    def get_initial_guesses():
        guesses = [np.array(initial_h_values, dtype=float)]
        num_ones = n_points // 2 + (1 if n_points % 2 else 0)
        guesses.append(generate_binary_guess(num_ones))
        guesses.append(generate_binary_guess(num_ones - 1))
        guesses.append(generate_binary_guess(num_ones + 1))
        return guesses

    initial_guesses = get_initial_guesses()
    best_h = initial_guesses[0]
    best_c5 = objective(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        h = guess.copy()
        for i in range(150):
            if time.time() - start_time > budget_s:
                break
            noise_scale = 0.05 * (1 - i / 150)
            perturb = np.random.uniform(-noise_scale, noise_scale, size=n_points)
            perturbed_guess = np.clip(h + perturb, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    return best_h, best_c5, n_points
