def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start optimization with SLSQP
    for _ in range(5):  # Increase number of perturbations
        if time.time() - start_time > budget_s:
            break
        # Random perturbation with uniform noise
        noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )

    # Refinement with SLSQP
    result = optimize.minimize(
        fun=objective,

        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
    )
    if result.success:
        best_c5 = objective(result.x)
        best_h = result.x
    
    return best_h, best_c5, n_points
