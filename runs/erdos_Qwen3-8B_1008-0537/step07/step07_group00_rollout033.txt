import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        value = np.max(corr) * dx
        penalty = 1e5 * abs(np.sum(h) - target_sum)
        return value + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use differential evolution with a hybrid strategy
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        x0=initial_guess,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-7,
        maxiter=100,
        seed=seed
    )

    # Refine with SLSQP
    best_h = result.x
    best_c5 = objective(best_h)

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds_refine = [(0.0, 1.0) for _ in range(n_points)]

    result_refine = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds_refine,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
    )

    final_h = result_refine.x
    final_c5 = objective(final_h)
    return final_h, final_c5, n_points
