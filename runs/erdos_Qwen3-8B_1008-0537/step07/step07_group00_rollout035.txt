            return h
        delta = target - s
        adjustment = delta / len(h)
        h_adjusted = h + adjustment
        h_adjusted = np.clip(h_adjusted, 0, 1)
        return h_adjusted

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Use DE with adjusted sum to ensure constraint
    result = optimize.differential_evolution(
        fun=lambda x: objective(adjust_sum(x)),
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        polish=True,
        seed=seed
    )

    # Evaluate the best solution found
    final_h = adjust_sum(result.x)
    final_c5 = objective(final_h)
    return final_h, final_c5, n_points
