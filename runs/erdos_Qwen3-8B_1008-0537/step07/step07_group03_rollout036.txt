            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        if result.success and objective(result.x) < best_c5:
            best_c5 = objective(result.x)
            best_h = result.x
    
    # Refinement with L-BFGS-B for faster local tuning
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
    )
    if result.success:
        best_c5 = objective(result.x)
        best_h = result.x
    
    return best_h, best_c5, n_points
