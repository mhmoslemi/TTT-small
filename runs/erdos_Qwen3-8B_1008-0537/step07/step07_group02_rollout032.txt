import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use initial_h_values as provided
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Define initial guesses
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_penalty = 1e6 * np.abs(np.sum(h) - target_sum)
        return max_corr + sum_penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Evaluate initial guess
    best_h = initial_guess.copy()
    best_c5 = objective_de(best_h) - 1e6 * np.abs(np.sum(best_h) - target_sum)
    
    # Run differential evolution on initial guess
    result_de = optimize.differential_evolution(
        fun=objective_de,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    best_h = result_de.x
    best_c5 = objective_de(best_h) - 1e6 * np.abs(np.sum(best_h) - target_sum)
    
    # Refine with SLSQP
    def objective_slsqp(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    result_slsqp = optimize.minimize(
        fun=objective_slsqp,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    best_h = result_slsqp.x
    best_c5 = objective_slsqp(best_h)
    
    # Perturb and refine
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.05 * (1 - _ / 20)
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective_slsqp,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective_slsqp(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
