    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    # Generate checkerboard-like initial guess
    def generate_checkerboard_initial(n):
        h = np.zeros(n)
        for i in range(n):
            if i % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        # Adjust for target_sum
        adjustment = np.sum(h) - target_sum
        if adjustment != 0:
            h[-1] -= adjustment
        return h
    
    checkerboard_guess = generate_checkerboard_initial(n_points)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        # DE with adjusted penalty
        def objective_de(h_values):
            h = np.array(h_values)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            constraint_violation = abs(np.sum(h) - target_sum)
            penalty = 1e3 * constraint_violation
            return c5 + penalty
        
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=20,
            maxiter=80,
            mutation=(0.5, 0.7),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    # Optimize checkerboard initial guess
    result_slsqp_checkerboard = optimize.minimize(
        fun=objective,
        x0=checkerboard_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_checkerboard.success:
        current_c5 = objective(result_slsqp_checkerboard.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_checkerboard.x
    
    # Local perturbations
    while time.time() - start_time < budget_s - 30:
        new_guess = optimize.minimize(
            fun=objective,
            x0=best_h + np.random.normal(0, 0.01, size=n_points),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
