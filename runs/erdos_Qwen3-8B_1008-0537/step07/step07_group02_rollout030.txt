    for guess in initial_guesses:
        current_c5 = objective(guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = guess.copy()
            best_candidates.append(guess)
    
    start_time = time.time()
    
    # Perform local optimization on each candidate
    for candidate in best_candidates:
        result = optimize.minimize(
            fun=objective,
            x0=candidate,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Further refinement with perturbation
    for _ in range(40):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.1 * (1 - _ / 40)
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
