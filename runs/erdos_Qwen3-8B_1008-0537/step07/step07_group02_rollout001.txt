import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    def make_guess():
        h = np.zeros(n_points)
        indices = np.random.permutation(n_points)
        fill = int(target_sum + 0.5)
        h[indices[:fill]] = 1.0
        h[-1] += target_sum - fill
        return h

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    h0 = initial_h_values.astype(float)
    h0 = np.clip(h0, 0, 1)

    def optimize_once(h):
        res = optimize.minimize(
            objective, h, bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            method='L-BFGS-B', tol=1e-8, options={'maxiter': 200})
        return res.x if res.success else h
