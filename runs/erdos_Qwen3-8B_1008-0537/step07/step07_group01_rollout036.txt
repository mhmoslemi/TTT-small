            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def evaluate_h(h):
        try:
            res = optimize.minimize(
                fun=objective,
                x0=h,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if res.success:
                return objective(res.x)
        except:
            return np.inf
        return np.inf
    
    # Evaluate initial guess
    c5_initial = evaluate_h(initial_guess)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = initial_guess
    
    # Evaluate even guess
    even_guess = generate_even_guess(n_points)
    c5_even = evaluate_h(even_guess)
    if c5_even < best_c5:
        best_c5 = c5_even
        best_h = even_guess
    
    # Evaluate random guess
    random_guess = generate_random_guess(n_points)
    c5_random = evaluate_h(random_guess)
    if c5_random < best_c5:
        best_c5 = c5_random
        best_h = random_guess
    
    # Perturbations
    while time.time() - start_time < budget_s - 20:
        perturbation = np.random.normal(0, 0.01, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        c5_new = evaluate_h(new_guess)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_guess
    
    return best_h, best_c5, n_points
