import scipy.optimize as optimize
import time

        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        def objective_de(h_values):
            h = np.array(h_values)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx

            penalty = 1e5 * constraint_violation
            return c5 + penalty
        
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=25,
            maxiter=40,
            mutation=(0.6, 1.2),

            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',

                options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:

    except:
