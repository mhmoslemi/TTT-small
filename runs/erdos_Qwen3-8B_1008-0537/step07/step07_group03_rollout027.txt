    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    best_h = alt_initial_h.copy()
    best_c5 = objective(best_h)

    # Global search using differential evolution
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-6,
        maxiter=50,
        disp=False
    )
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)

    # Refinement with COBYLA
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'disp': False}
    )
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)

    # Local perturbation and refinement
    if time.time() - start_time < budget_s * 0.8:
        for _ in range(2):
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(best_h + noise, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'disp': False}
            )
            if result.success and objective(result.x) < best_c5:
                best_h = result.x
                best_c5 = objective(best_h)

    return best_h, best_c5, n_points
