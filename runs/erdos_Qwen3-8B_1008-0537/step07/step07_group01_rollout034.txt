                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    random_guess = np.random.rand(n_points)
    random_guess = (random_guess / np.sum(random_guess) * target_sum)
    result_slsqp_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_random.success:
        current_c5 = objective(result_slsqp_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_random.x
    
    while time.time() - start_time < budget_s - 20:
        perturb = np.random.normal(0, 0.01, size=n_points)
        new_guess = best_h + perturb
        new_guess = np.clip(new_guess, 0, 1)
        new_guess = (new_guess / np.sum(new_guess) * target_sum)
        result = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
