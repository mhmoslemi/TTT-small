import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Generate sinusoidal initial guess for exploration
    x_vals = np.linspace(0, 2, n_points)
    h_sine = 0.5 + 0.5 * np.sin(np.pi * x_vals)
    h_sine = np.clip(h_sine, 0, 1)
    h_sine = h_sine * (target_sum / np.sum(h_sine)) if np.sum(h_sine) > 0 else h_sine
    
    def optimize_start(h_start):
        result = optimize.minimize(
            fun=objective,
            x0=h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            return result.x, objective(result.x)
        return h_start, objective(h_start)
    
    def refine(h_start):
        result = optimize.minimize(
            fun=objective,
            x0=h_start,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'disp': False}
        )
        if result.success:
            return result.x, objective(result.x)
        return h_start, objective(h_start)
    
    # Multiple starting points
    starts = [initial_h, h_sine]
    
    for h_start in starts:
        if time.time() - start_time > budget_s:
            break
        h_opt, c5 = optimize_start(h_start)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        h_opt, c5 = refine(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    return best_h, best_c5, n_points
