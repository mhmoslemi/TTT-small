        idxs = np.random.choice(n_points, num_ones, replace=False)
        guess[idxs] = 1.0
        if num_ones < target_sum:
            guess[-1] += target_sum - num_ones
        elif num_ones > target_sum:
            guess[-1] -= num_ones - target_sum
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
