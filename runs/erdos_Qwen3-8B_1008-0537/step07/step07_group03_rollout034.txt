import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    
    # Generate a random binary starting guess
    random_binary = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    sum_binary = np.sum(random_binary)
    # Adjust to meet sum constraint
    if abs(sum_binary - target_sum) > 1e-6:
        diff = target_sum - sum_binary
        idx = np.argmin(random_binary)
        random_binary[idx] += diff
    random_binary = np.clip(random_binary, 0, 1)
    random_h = np.array(random_binary, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start with diverse perturbations
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        np.random.seed(seed + _)  # Vary seeds
        noise = np.random.laplace(0, 0.05, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        res = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    # Local refinement
    res = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100}
    )
    if res.success:
        best_c5 = objective(res.x)
        best_h = res.x
    
    # Final refinement with SLSQP
    res = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 50}
    )
    if res.success:
        best_c5 = objective(res.x)
        best_h = res.x
    
    return best_h, best_c5, n_points
