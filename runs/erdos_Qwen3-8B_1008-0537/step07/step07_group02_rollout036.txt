    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    best_candidates = [best_h]
    
    # Evaluate all initial guesses
    for guess in initial_guesses:
        current_c5 = objective(guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = guess.copy()
            best_candidates.append(guess)
    
    start_time = time.time()
    
    def optimize_candidate(candidate):
        result = optimize.minimize(
            fun=objective,
            x0=candidate,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x
        return candidate
    
    # Local optimization
    for candidate in best_candidates:
        updated = optimize_candidate(candidate)
        current_c5 = objective(updated)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = updated
    
    # Perturbation refinement
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.05 * (1 - _ / 30)
        perturbed = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        updated = optimize_candidate(perturbed)
        current_c5 = objective(updated)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = updated
    
    return best_h, best_c5, n_points
