import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def create_uniform_guess(n):
        num_ones = n // 2
        guess = np.zeros(n)
        step = 2.0 / n
        for i in range(num_ones):
            pos = i * step
            idx = int(pos / dx)
            if idx >= n:
                idx = n - 1
            guess[idx] = 1.0
        current_sum = np.sum(guess)
        if current_sum < num_ones:
            guess[-1] += (num_ones - current_sum)
        elif current_sum > num_ones:
            guess[-1] -= (current_sum - num_ones)
        return guess

    def generate_random_guess(n):
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            guess[-1] += remaining
        return guess

    def generate_alternating_guess(n):
        guess = np.zeros(n)
        for i in range(n):
            guess[i] = 1.0 if i % 2 == 0 else 0.0
        current_sum = np.sum(guess)
        diff = target_sum - current_sum
        if diff > 0:
            guess[-1] += diff
        elif diff < 0:
            guess[-1] -= -diff
        return guess

    initial_guesses = [np.array(initial_h_values, dtype=float), create_uniform_guess(n_points), generate_random_guess(n_points), generate_alternating_guess(n_points)]

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)

    for guess in initial_guesses:
        current_c5 = objective(guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = guess.copy()

    start_time = time.time()

    def optimize_candidate(candidate, method='SLSQP'):
        result = optimize.minimize(
            fun=objective,
            x0=candidate,
            method=method,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x
        return candidate

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        best_h = optimize_candidate(best_h, 'SLSQP')

    for _ in range(15):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.05 * (1 - _ / 15)
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        best_h = optimize_candidate(perturbed_guess, 'COBYLA')

    return best_h, best_c5, n_points
