import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    initial_guess = np.array(initial_h_values)
    
    def create_alternate_guess(n):
        guess = np.zeros(n)
        guess[:n//2] = 1.0
        guess[-1] += 0.5 if n % 2 else 0.0
        return guess
    
    def generate_random_guess(n):
        num_ones = int(target_sum + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            guess[-1] += remaining
        return guess
    
    def generate_periodic_guess(n):
        freq = 3
        guess = np.zeros(n)
        for i in range(n):
            guess[i] = 1.0 if ((i * freq) % 2) < 1 else 0.0
        current_sum = np.sum(guess)
        diff = target_sum - current_sum
        guess[-1] += diff
        return guess
    
    initial_guesses = [initial_guess, create_alternate_guess(n_points), generate_random_guess(n_points), generate_periodic_guess(n_points)]
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for guess in initial_guesses:
        current_c5 = objective(guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = guess.copy()
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'xatol': 1e-5, 'fatol': 1e-5}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:

    for _ in range(20):
