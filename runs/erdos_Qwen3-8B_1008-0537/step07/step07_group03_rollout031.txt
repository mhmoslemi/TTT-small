import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start optimization with structured perturbations
    for _ in range(5):  # More perturbations for diversity
        if time.time() - start_time > budget_s:
            break
        # Random perturbations with different noise types
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
