import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    # Generate multiple initial guesses
    np.random.seed(seed)
    initial_guesses = [initial_h.copy()]
    for _ in range(3):
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        diff = target_sum - np.sum(h)
        if diff > 0:
            indices = np.argsort(h)
            for i in range(n_points):
                if diff <= 0:
                    break
                h[indices[i]] = np.clip(h[indices[i]] + diff / (n_points - i), 0, 1)
                diff -= h[indices[i]] - np.clip(h[indices[i]], 0, 1)
        elif diff < 0:
            indices = np.argsort(h)
            for i in range(n_points):
                if diff >= 0:
                    break
                h[indices[i]] = np.clip(h[indices[i]] + diff / (n_points - i), 0, 1)
                diff -= h[indices[i]] - np.clip(h[indices[i]], 0, 1)
        initial_guesses.append(np.clip(h, 0, 1))
    
    best_c5 = float('inf')
    best_h = initial_h
    
    start_time = time.time()
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-9, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    if time.time() - start_time < budget_s:
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='COBYLA',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
