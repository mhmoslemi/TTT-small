import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_uniform_guess(n):
        m = n // 2
        positions = np.linspace(0, 2, m + 1)[:-1]
        positions = (positions * n) / 2
        positions = np.round(positions).astype(int)
        positions = np.unique(positions)
        guess = np.zeros(n)
        if len(positions) > m:
            guess[positions[:m]] = 1.0
        else:
            guess[positions] = 1.0
        sum_guess = np.sum(guess)
        diff = target_sum - sum_guess
        if abs(diff) > 1e-5:
            guess[0] += diff
        return guess
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_with_slsqp(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if result.success:
            return result.x, objective(result.x)
        return x0, objective(x0)
    
    candidates = [initial_guess]
    try:
        uniform_guess = generate_uniform_guess(n_points)
        candidates.append(uniform_guess)
    except:
        pass
    
    for _ in range(3):
        random_guess = generate_random_guess(n_points)
        candidates.append(random_guess)
    
    results = []
    for guess in candidates:
        h, c5 = optimize_with_slsqp(guess)
        results.append((h, c5))
    
    for h, c5 in results:
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    if time.time() - start_time < budget_s - 10:
        def perturbative_optimize(x0):
            result = optimize.minimize(
                fun=objective,
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-9}
            )
            if result.success:
                return result.x, objective(result.x)
            return x0, objective(x0)
        
        for _ in range(5):
            new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
            h, c5 = perturbative_optimize(new_guess)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    
    return best_h, best_c5, n_points
