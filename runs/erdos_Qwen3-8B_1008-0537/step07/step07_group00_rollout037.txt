    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = np.abs(np.sum(h) - target_sum)
        penalty = 1e4 * constraint_violation
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Run differential evolution with focus on early iterations
    result = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False,
        workers=-1
    )
    
    # Evaluate the best solution found
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Early exit if time is up
    if time.time() - start_time > budget_s - 5:
        return best_h, best_c5, n_points
    
    # Refine with local optimization
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        tol=1e-6,
        options={'maxfun': 100, 'ftol': 1e-9}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    return best_h, best_c5, n_points
