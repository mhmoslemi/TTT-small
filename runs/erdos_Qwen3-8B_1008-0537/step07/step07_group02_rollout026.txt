            pos = int(i * (n / num_ones))
            guess[pos] = 1.0

        return guess

                guess[i] = 1.0
        current_sum = np.sum(guess)
        diff = target_sum - current_sum
        if diff > 1e-6:
            guess[-1] += diff
        elif diff < -1e-6:
            guess[-1] -= -diff
        return guess

    initial_guesses = [
        np.array(initial_h_values, dtype=float),
        np.random.permutation(n_points)[:int(target_sum)].astype(float),

    def objective(h_values):
