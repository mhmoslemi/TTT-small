            nelder_mead_options={'xatol': 1e-8},
            maxiter=100
        )
        if result_shgo.success:
            best_shgo = result_shgo.x
            current_c5 = objective(best_shgo)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = best_shgo
    except:
        pass
    
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_c5 = objective(result_slsqp.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    def generate_random_guess(n):
        target_sum = n / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n, size=num_integers, replace=False)
        guess = np.zeros(n)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    random_guess = generate_random_guess(n_points)
    result_slsqp_random = optimize.minimize(
        fun=objective,
        x0=random_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_random.success:
        current_c5 = objective(result_slsqp_random.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_random.x
    
    while time.time() - start_time < budget_s - 20:
        new_guess = best_h.copy()
        new_guess += np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0.0, 1.0)
        result_slsqp_perturb = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp_perturb.success:
            current_c5 = objective(result_slsqp_perturb.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp_perturb.x
    
    return best_h, best_c5, n_points
