import time

def run(seed=42, budget_s=100, **kwargs):

    def generate_binary_guess(n, density=0.6):
        num_ones = int(density * n + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            guess[-1] += remaining
        return guess

    def generate_random_guess(n, density=0.5):
        num_ones = int(density * n + 0.5)
        indices = np.random.choice(n, size=num_ones, replace=False)
        guess = np.zeros(n)
        guess[indices] = 1.0
        remaining = target_sum - num_ones
        if remaining > 0:
            guess[-1] += remaining
        return guess

    def generate_alternating_guess(n):
        guess = np.zeros(n)
        for i in range(n):
            guess[i] = 1.0 if i % 2 == 0 else 0.0
        current_sum = np.sum(guess)
        diff = target_sum - current_sum
        if diff > 0:
            guess[-1] += diff
        elif diff < 0:
            guess[-1] -= -diff
        return guess

    def objective(h_values):
