import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def generate_uniform_guess(n):
        num_ones = n // 2
        positions = np.linspace(0, 2, n)
        idx = np.argsort(np.abs(positions - 1.0))
        guess = np.zeros(n)
        guess[idx[:num_ones]] = 1.0
        return guess
    
    def generate_bipartite_guess(n):
        guess = np.zeros(n)
        guess[:n//2] = 1.0
        guess[n//2:] = 0.0
        return guess
    
    initial_guesses = [
        np.array(initial_h_values, dtype=float),
        generate_uniform_guess(n_points),
        generate_bipartite_guess(n_points),
        np.random.choice([0,1], size=n_points, p=[0.5,0.5]).astype(float),
        np.zeros(n_points) + 0.5
    ]
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for guess in initial_guesses:
        res = optimize.minimize(
            objective, guess, method='L-BFGS-B', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05 * (1 - _/30), n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        res = optimize.minimize(
            objective, perturbed, method='L-BFGS-B', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if res.success and objective(res.x) < best_c5:
            best_c5 = objective(res.x)
            best_h = res.x
    
    return best_h, best_c5, n_points
