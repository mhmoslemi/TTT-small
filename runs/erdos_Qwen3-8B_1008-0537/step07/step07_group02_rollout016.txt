import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_uniform_guess(n, m):
        positions = []
        interval = n / m
        for i in range(m):
            pos = int(i * interval)
            positions.append(pos)
        positions = np.unique(positions)
        positions = positions[:m]
        guess = np.zeros(n)
        for pos in positions:
            if 0 <= pos < n:
                guess[pos] = 1.0
        current_sum = np.sum(guess)
        diff = target_sum - current_sum
        if diff > 0:
            guess[-1] += diff
        elif diff < 0:
            guess[-1] -= -diff
        return guess
    
    uniform_guess = generate_uniform_guess(n_points, int(n_points//2))
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for guess in [initial_guess, uniform_guess]:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.04 * (1 - _ / 50)
        perturbed_guess = np.clip(best_h + np.random.normal(0, noise_scale, size=n_points), 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
