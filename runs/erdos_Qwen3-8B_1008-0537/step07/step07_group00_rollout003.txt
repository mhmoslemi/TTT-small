import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def generate_random_initial(n_points):
        h = np.random.uniform(0, 1, n_points)
        h *= target_sum / np.sum(h)
        return h

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Use initial guess and a few random starts
    starts = [initial_guess]
    for _ in range(2):
        starts.append(generate_random_initial(n_points))
    
    for start in starts:
        result = optimize.minimize(
            fun=objective,
            x0=start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    # Fine-tune with local search
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        noise_scale = 0.05
        perturb = np.random.uniform(-noise_scale, noise_scale, size=n_points)
        perturbed_guess = np.clip(best_h + perturb, 0, 1)
        
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
