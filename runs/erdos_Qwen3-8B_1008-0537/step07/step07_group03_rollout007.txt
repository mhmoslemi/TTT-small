import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Enhanced multi-start with structured perturbations
    for _ in range(5):  # Increased number of starts
        if time.time() - start_time > budget_s:
            break
        # Structured perturbation: shift nearby samples
        perturb = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                perturb[i] = np.random.uniform(-0.05, 0.05)
            else:
                perturb[i] = np.random.uniform(-0.02, 0.02)
        perturbed = np.clip(initial_h + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,

            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
