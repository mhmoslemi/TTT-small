import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    def generate_structured_guess():  # Jittered uniform distribution
        num_ones = int(target_sum + 0.5)
        positions = np.linspace(0, 2, num_ones, endpoint=False)
        positions += np.random.uniform(-0.1, 0.1, size=num_ones)
        grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for pos in positions:
            idx = np.argmin(np.abs(grid - pos))
            h[idx] = 1.0
        if abs(np.sum(h) - target_sum) > 1e-6:
            h[np.random.choice(n_points, size=1)] += (target_sum - np.sum(h))
        return h
    
    def objective(h_values):
        return evaluate(np.array(h_values))
    
    try:
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=20,
            maxiter=50,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    structured_guess = generate_structured_guess()
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=structured_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_c5 = result_slsqp.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
    
    while time.time() - start_time < budget_s - 20:
        perturbed = best_h + np.random.normal(0, 0.01, size=n_points)

            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:

            if current_c5 < best_c5:
