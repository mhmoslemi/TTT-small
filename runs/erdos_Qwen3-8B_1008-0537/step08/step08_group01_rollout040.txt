    def generate_even_guess(n, target_sum):
        m = int(target_sum)
        additional = target_sum - m
        positions = []
        spacing = 2.0 / (m + 1)
        for i in range(1, m + 1):
            x = i * spacing
            idx = int(round(x * (n / 2)))
            positions.append(idx)
        start = np.zeros(n)
        for i in positions:
            if 0 <= i < n:
                start[i] = 1.0
        if additional > 0 and positions:
            start[positions[0]] += additional
        return start
    
    def generate_regular_guess(n, target_sum):
        num_ones = int(target_sum)
        remaining = target_sum - num_ones
        step = 2.0 / (num_ones + 1)
        positions = [i * step for i in range(1, num_ones + 1)]
        idxs = [int(round(x * n / 2)) for x in positions]
        start = np.zeros(n)
        for i in idxs:
            if 0 <= i < n:
                start[i] = 1.0
        if remaining > 0:
            closest = min(idxs, key=lambda x: abs(x - positions[-1]))
            closest_idx = idxs.index(closest)
            start[closest_idx] += remaining
        return start
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
