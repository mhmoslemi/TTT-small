import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import random
    import copy
    
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Generate additional initial guesses
    def generate_initial_guesses():
        guesses = [initial_guess.copy()]
        # Random uniform
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        guesses.append(guess)
        # Alternating pattern
        guess = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                guess[i] = 1.0
            else:
                guess[i] = 0.0
        guess = guess * (target_sum / np.sum(guess))
        guesses.append(guess)
        return guesses

    initial_guesses = generate_initial_guesses()

    for guess in initial_guesses:
        h_start = copy.copy(guess)
        for i in range(200):
            if time.time() - start_time > budget_s:
                break
            # Adaptive perturbation
            noise_scale = max(0.01, 0.1 * (1 - i / 200))
            perturb = np.random.uniform(-noise_scale, noise_scale, size=n_points)
            perturbed_guess = np.clip(h_start + perturb, 0, 1)

            result = optimize.minimize(
                fun=objective,
                x0=perturbed_guess,
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'disp': False}
            )

            if result.success:
                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                h_start = result.x

    # Final refinement
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'disp': False}
    )

    if result.success:
        best_c5 = objective(result.x)
        best_h = result.x

    return best_h, best_c5, n_points
