import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 1e3 * np.abs(sum_h - target_sum)
        return max_corr + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False,
        vectorized=True
    )
    if time.time() - start_time > budget_s - 5:
        return initial_guess, objective(initial_guess), n_points

    best_candidate = result_de.x
    sum_candidate = np.sum(best_candidate)
    scale = target_sum / sum_candidate
    scaled_candidate = np.clip(best_candidate * scale, 0.0, 1.0)

    def refined_objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result_l_bfgs = optimize.minimize(
        fun=refined_objective,
        x0=scaled_candidate,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        options={'maxiter': 50, 'disp': False}
    )

    best_h = result_l_bfgs.x
    best_c5 = refined_objective(best_h)

    return best_h, best_c5, n_points
