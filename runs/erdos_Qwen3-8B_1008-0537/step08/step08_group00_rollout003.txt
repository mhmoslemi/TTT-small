import scipy.optimize as optimize

        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.linspace(0, n-1, n, endpoint=False)
        shift = np.random.uniform(0, 1)
        h = np.zeros(n)
        for i in range(num_integers):
            idx = int((positions + shift) % n)
            h[idx] = 1.0
        if remaining > 0:
            idx = np.random.choice(n)
            h[idx] += remaining
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

        return c5 + penalty
