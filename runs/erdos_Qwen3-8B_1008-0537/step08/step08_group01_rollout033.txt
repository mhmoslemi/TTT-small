import time

def run(seed=42, budget_s=100, **kwargs):

        for i in range(center):
            start[i] = 1.0
            start[n - 1 - i] = 0.0
        start[center] = 0.5
        return start

    symmetric_guess = generate_symmetric_guess(n_points, target_sum)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Optimize symmetric guess
    result_slsqp_sym = optimize.minimize(

        x0=symmetric_guess,
        method='SLSQP',

    )
    if result_slsqp_sym.success:
        current_c5 = objective(result_slsqp_sym.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
