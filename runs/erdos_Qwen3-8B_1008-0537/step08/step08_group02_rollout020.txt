import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

        corr = np.correlate(h, 1 - h, mode='full')

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Basin-hopping global optimization
    result_bh = optimize.basinhopping(
        fun=objective,
        x0=initial_guess,
        niter=50,
        T=1.0,
        stepsize=0.5,
        minimizer_kwargs={

            'bounds': bounds,
            'constraints': [{'type': 'eq', 'fun': constraint_sum}],
            'options': {'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        },
        polish=True,
        niter_success=10
    )
    
    if result_bh.success:
        best_h = result_bh.x
        best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    result_slsqp = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp.success:
        current_c5 = objective(result_slsqp.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
    
    # Additional refinement with random perturbations
    while time.time() - start_time < budget_s - 20:

            fun=objective,
            x0=best_h + np.random.normal(0, 0.01, size=n_points),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )

            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
