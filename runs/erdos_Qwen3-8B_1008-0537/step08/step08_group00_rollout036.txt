    result_slsqp = optimize_with_slsqp(initial_guess)
    current_c5 = objective(result_slsqp)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = result_slsqp

    # Local optimization from even_guess
    result_slsqp = optimize_with_slsqp(even_guess)
    current_c5 = objective(result_slsqp)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = result_slsqp

    # Local optimization from random_guess
    result_slsqp = optimize_with_slsqp(random_guess)
    current_c5 = objective(result_slsqp)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = result_slsqp

    while time.time() - start_time < budget_s - 20:
        new_guess = optimize_with_slsqp(best_h + np.random.normal(0, 0.01, size=n_points))
        if new_guess is not None:
            current_c5 = objective(new_guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess

    return best_h, best_c5, n_points
