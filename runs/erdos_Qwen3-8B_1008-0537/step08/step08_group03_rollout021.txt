    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Adaptive exploration with Nelder-Mead and L-BFGS-B
    for i in range(100):
        if time.time() - start_time > budget_s:
            break
        noise_scale = max(0.02, 0.3 * (1 - i / 100))
        perturbed_guess = np.clip(best_h + np.random.uniform(-noise_scale, noise_scale, size=n_points), 0, 1)

        # Nelder-Mead for global search
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_guess,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'disp': False}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    # Final refinement with L-BFGS-B
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'disp': False}
    )

    if result.success:
        best_c5 = objective(result.x)
        best_h = result.x

    return best_h, best_c5, n_points
