import scipy.optimize as optimize
import time

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_uniform_guess(n, target_sum):
        m = int(target_sum)
        additional = target_sum - m
        positions = np.floor(np.linspace(0, n-1, m+1)).astype(int)
        start = np.zeros(n)
        start[positions] = 1.0
        if additional > 0:
            start[positions[0]] += additional
        return start
    
    def run_slsqp(x0, maxiter=300):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-11, 'eps': 1e-9}
        )
        if result.success:
            return result
        return None
    
    def run_de(popsize=20, maxiter=100):
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=popsize,
            maxiter=maxiter,
            mutation=(0.7, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        return result

    if initial_result and initial_result.success:

    uniform_guess = generate_uniform_guess(n_points, target_sum)
