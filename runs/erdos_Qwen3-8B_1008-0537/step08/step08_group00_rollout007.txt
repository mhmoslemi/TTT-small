        h = np.array(h_values)

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum

        def objective_de(h_values):
            h = np.array(h_values)
            corr = np.correlate(h, 1 - h, mode='full')

            constraint_violation = abs(np.sum(h) - target_sum)
            penalty = 1e4 * constraint_violation
            return c5 + penalty
        
        result_de = optimize.differential_evolution(
            fun=objective_de,
