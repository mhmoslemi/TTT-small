        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        # Use structured guess as initial guess
        structured_guess = generate_structured_guess(n_points)
        result_slsqp_initial = optimize.minimize(
            fun=objective,
            x0=structured_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if result_slsqp_initial.success:
            current_c5 = objective(result_slsqp_initial.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp_initial.x
    except:
        pass
    
    # Use initial guess and refine
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    # Perturbations
    while time.time() - start_time < budget_s - 20:
        new_guess = optimize.minimize(
            fun=objective,
            x0=best_h + np.random.normal(0, 0.01, size=n_points),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
