import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def generate_random_guess(target_sum):
        h = np.random.uniform(0, 1, n_points)
        scale = target_sum / np.sum(h)
        h *= scale
        return h
    
    def generate_structured_guess(n):
        num_ones = int(target_sum + 0.5)
        positions = np.linspace(0, 2, num_ones, endpoint=False)
        grid = np.linspace(0, 2, n)
        h = np.zeros(n)
        for pos in positions:
            idx = np.argmin(np.abs(grid - pos))
            h[idx] = 1.0
        if abs(np.sum(h) - target_sum) > 1e-6:
            h[np.random.choice(n, size=1)] += (target_sum - np.sum(h))
        return h
    
    initial_guess = np.array(initial_h_values, dtype=float)
    random_guess = generate_random_guess(target_sum)
    structured_guess = generate_structured_guess(n_points)
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        return c5 + 1e2 * constraint_violation
    
    for guess in [initial_guess, random_guess, structured_guess]:
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=30,
            maxiter=200,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            workers=1
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    
    while time.time() - start_time < budget_s - 30:
        new_guess = optimize.minimize(
            fun=objective,
            x0=best_h + np.random.normal(0, 0.01, size=n_points),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
