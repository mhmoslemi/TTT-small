import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_with_restarts(x0, maxiter=500, ftol=1e-10):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-9}
        )
        if result.success:
            return result.x, result.fun
        return x0, objective(x0)
    
    # Restart with initial guess
    best_h, best_c5 = optimize_with_restarts(initial_guess)

    perturbed_guess = initial_guess + perturb
    perturbed_guess = np.clip(perturbed_guess, 0, 1)
    perturbed_guess = perturbed_guess * (target_sum / np.sum(perturbed_guess)) if np.sum(perturbed_guess) > 0 else perturbed_guess
    best_h, best_c5 = optimize_with_restarts(perturbed_guess)
    
    # Random guess
    def generate_random_guess():
        target_sum = n_points / 2
        num_integers = int(target_sum)
        remaining = target_sum - num_integers
        positions = np.random.choice(n_points, size=num_integers, replace=False)
        guess = np.zeros(n_points)
        guess[positions] = 1.0
        if remaining > 0:
            pos = np.random.choice(n_points, size=1, replace=False)
            guess[pos] += remaining
        return guess
    
    random_guess = generate_random_guess()
    best_h, best_c5 = optimize_with_restarts(random_guess)
    
    # Local search around best_h
    for _ in range(3):
