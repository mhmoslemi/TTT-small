import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Start with initial guess
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def generate_even_guess(n):
        return np.zeros(n) + 0.5

    def generate_pattern_guess(n, step):
        pattern = np.zeros(n)
        for i in range(n):
            if i % step == 0:
                pattern[i] = 1.0
        return pattern

    def optimize_from_guess(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x, objective(result.x)
        return x0, best_c5

    # Multi-start optimization
    guesses = [initial_guess]
    guesses.append(generate_even_guess(n_points))
    guesses.append(generate_pattern_guess(n_points, 3))
    guesses.append(generate_pattern_guess(n_points, 5))

    for guess in guesses:
        current_h, current_c5 = optimize_from_guess(guess)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    # Differential evolution for global search
    result_de = optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=10,
        maxiter=50,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-10,
        vectorized=True
    )

    # Refine with SLSQP
    final_result = optimize.minimize(
        fun=objective,
        x0=result_de.x,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
    )

    if final_result.success and objective(final_result.x) < best_c5:
        best_h = final_result.x
        best_c5 = objective(final_result.x)

    # Final refinement with random perturbations
    while time.time() - start_time < budget_s - 20:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0, 1)
        current_h, current_c5 = optimize_from_guess(new_guess)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
