import scipy.optimize as optimize
import time

        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    try:
        def generate_structured_guess(n):

            h = np.zeros(n)
            for pos in positions:
                idx = np.argmin(np.abs(grid - pos))
                h[idx] = 1.0
            if abs(np.sum(h) - target_sum) > 1e-6:
                h[np.random.choice(n, size=1)] += (target_sum - np.sum(h))
            return h

            tol=1e-10,
            vectorized=True
        )
        if result_de_initial.success:
            best_h = result_de_initial.x
            best_c5 = objective(best_h)
        
        result_slsqp_initial = optimize.minimize(
