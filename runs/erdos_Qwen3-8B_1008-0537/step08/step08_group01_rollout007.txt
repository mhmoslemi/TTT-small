import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_binary_guess(n):
        guess = np.zeros(n)
        guess[::2] = 1.0
        if n % 2 == 1:
            guess[-1] = 0.0
        if np.sum(guess) > target_sum:
            guess[-1] = 0.0
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def evaluate_guess(guess):
        nonlocal best_h, best_c5
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    def perturb_guess(guess, step=0.02):
        return guess + np.random.normal(0, step, size=n_points)
    
    guesses = [
        initial_guess,
        generate_binary_guess(n_points)
    ]
    
    for guess in guesses:
        evaluate_guess(guess)
    
    if time.time() - start_time < budget_s - 40:
        for _ in range(5):
            new_guess = perturb_guess(best_h)
            evaluate_guess(new_guess)
    
    return best_h, best_c5, n_points
