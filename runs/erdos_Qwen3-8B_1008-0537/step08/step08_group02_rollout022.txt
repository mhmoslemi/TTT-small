import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

        grid = np.linspace(0, 2, n_points)

            idx = np.argmin(np.abs(grid - pos))
            h[idx] = 1.0
        if abs(np.sum(h) - target_sum) > 1e-6:
            h[np.random.choice(n_points, size=1)] += (target_sum - np.sum(h))
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_from_guess(x0):
        result_de = optimize.differential_evolution(
            fun=objective, 
            bounds=bounds,
            popsize=25,
            maxiter=100,
            mutation=(0.7, 1.2),
            recombination=0.9,
            tol=1e-8,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
