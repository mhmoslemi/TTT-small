import scipy.optimize as optimize
import time

    def generate_optimized_guess(n, target_sum):
        m = int(target_sum)
        positions = []
        spacing = 2.0 / (m + 1)
        for i in range(m):
            x = (i + 0.5) * spacing
            idx = int(round(x * (n / 2)))
            positions.append(idx)
        guess = np.zeros(n)
        for idx in positions:
            if 0 <= idx < n:
                guess[idx] = 1.0
        if target_sum - m > 0:
            guess[0] += target_sum - m
        return guess
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
