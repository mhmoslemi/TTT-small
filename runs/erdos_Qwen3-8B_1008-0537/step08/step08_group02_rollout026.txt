    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_random_guess():
        num_ones = int(target_sum + 0.5)
        positions = np.random.choice(n_points, size=num_ones, replace=False)
        h = np.zeros(n_points)
        h[positions] = 1.0
        if abs(np.sum(h) - target_sum) > 1e-6:
            h[np.random.choice(n_points)] += (target_sum - np.sum(h))
        return h
    
    def generate_structured_guess():
        num_ones = int(target_sum + 0.5)
        grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(num_ones):
            pos = (i + 0.5) / num_ones * 2
            idx = np.argmin(np.abs(grid - pos))
            h[idx] = 1.0
        if abs(np.sum(h) - target_sum) > 1e-6:
            h[np.random.choice(n_points)] += (target_sum - np.sum(h))
        return h
    
    guesses = [initial_guess, generate_random_guess(), generate_structured_guess()]
    
    for guess in guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    while time.time() - start_time < budget_s - 20:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        result = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
