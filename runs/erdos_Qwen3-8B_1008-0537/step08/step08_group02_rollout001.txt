import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_structured_guess(n, m=None):
        if m is None:
            m = int(target_sum + 0.5)
        num_ones = int(m + 0.5)
        positions = np.linspace(0, 2, num_ones, endpoint=False)
        grid = np.linspace(0, 2, n)
        h = np.zeros(n)
        for pos in positions:
            idx = np.argmin(np.abs(grid - pos))
            h[idx] = 1.0
        if abs(np.sum(h) - target_sum) > 1e-6:
            h[np.random.choice(n, size=1)] += (target_sum - np.sum(h))
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        penalty = 1e4 * constraint_violation
        return c5 + penalty
    
    structured_guesses = [generate_structured_guess(n_points, m) for m in [int(target_sum), int(target_sum)+1, int(target_sum)-1]]
    
    for guess in structured_guesses:

            fun=objective_de,

            popsize=30,
            maxiter=200,
            mutation=(0.8, 1.5),

            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
