    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for guess in initial_guesses:
        result = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={
                'maxiter': 200, 
                'ftol': 1e-10, 
                'eps': 1e-9, 
                'disp': False
            }
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    for _ in range(5):
        if time.time() - start_time >= budget_s - 10:
            break
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0.0, 1.0)
        result = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={
                'maxiter': 100, 
                'ftol': 1e-10, 
                'eps': 1e-9, 
                'disp': False
            }
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
