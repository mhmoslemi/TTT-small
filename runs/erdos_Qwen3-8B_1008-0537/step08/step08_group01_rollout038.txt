        interval_end2 = 1.75

        start2 = int(np.ceil(interval_start2 / dx))
        end2 = int(np.floor(interval_end2 / dx))
        start1 = max(0, start1)
        end1 = min(n-1, end1)
        start2 = max(0, start2)
        end2 = min(n-1, end2)

        h[start2:end2+1] = 1
        return h

        h = np.array(h_values)
