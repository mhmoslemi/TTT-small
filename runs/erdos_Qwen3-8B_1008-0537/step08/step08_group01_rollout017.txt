import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_even_guess(n, target_sum):
        m = int(target_sum)
        additional = target_sum - m
        positions = []
        spacing = 2.0 / (m + 1)
        for i in range(1, m + 1):
            x = i * spacing
            idx = int(round(x * (n / 2)))
            positions.append(idx)
        start = np.zeros(n)
        for i in positions:
            if 0 <= i < n:
                start[i] = 1.0
        if additional > 0 and positions:
            start[positions[0]] += additional
        return start
    
    def generate_random_guess(n, target_sum):
        h = np.random.rand(n)
        h = (h * 2.0 - 1.0) * 0.1 + 0.5
        h *= (target_sum / np.sum(h)) if np.sum(h) > 0 else 1.0
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_with_slsqp(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result
    
    def optimize_with_de(x0):
        result = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=15,
            maxiter=50,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-10,
            vectorized=True,
            x0=x0
        )
        return result
    
    candidates = [
        initial_guess,
        generate_even_guess(n_points, target_sum),
        generate_random_guess(n_points, target_sum)
    ]
    
    for x0 in candidates:
        try:
            de_result = optimize_with_de(x0)
            slsqp_result = optimize_with_slsqp(de_result.x)
            current_c5 = objective(slsqp_result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = slsqp_result.x
        except:
            pass
    
    while time.time() - start_time < budget_s - 20:
        new_guess = optimize_with_slsqp(best_h + np.random.normal(0, 0.01, size=n_points))
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
