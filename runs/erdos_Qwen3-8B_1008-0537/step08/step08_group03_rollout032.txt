        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5_value = np.max(corr) * dx
        penalty = 1e5 * abs(np.sum(h) - target_sum)
        return c5_value + penalty
    
    def objective_without_penalty(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Differential evolution with penalty
    result_de = optimize.differential_evolution(
        fun=objective_with_penalty,

        strategy='best1bin',
        popsize=20,
        maxiter=50,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,

        init='latinhypercube',
        x0=initial_guess
