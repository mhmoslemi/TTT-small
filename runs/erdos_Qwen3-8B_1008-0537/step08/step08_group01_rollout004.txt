import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 90  # Double the points for better resolution while keeping under 1k
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.zeros(n_points)
    initial_guess[::2] = 1.0  # Simple alternating pattern as base

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def optimize_with_method(method, x0, **kwargs):
        try:
            result = optimize.minimize(
                fun=objective,
                x0=x0,
                method=method,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 150, 'ftol': 1e-8, 'eps': 1e-9}
            )
            if result.success:
                return result
        except:
            pass
        return optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
