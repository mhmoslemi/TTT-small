    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        even_guess = generate_even_guess(n_points, target_sum)
        result_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=15,
            maxiter=30,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        
        def objective_de(h_values):
            h = np.array(h_values)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            constraint_violation = abs(np.sum(h) - target_sum)
            penalty = 1e6 * constraint_violation
            return c5 + penalty
        
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=15,
            maxiter=30,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-10,
            vectorized=True
        )
        
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                current_c5 = objective(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    except:
        pass
    
    result_slsqp_initial = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_initial.success:
        current_c5 = objective(result_slsqp_initial.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_initial.x
    
    even_guess = generate_even_guess(n_points, target_sum)
    result_slsqp_even = optimize.minimize(
        fun=objective,
        x0=even_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_even.success:
        current_c5 = objective(result_slsqp_even.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_even.x
    
    while time.time() - start_time < budget_s - 20:
        new_guess = optimize.minimize(
            fun=objective,
            x0=best_h + np.random.normal(0, 0.01, size=n_points),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
