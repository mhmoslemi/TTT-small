import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)

        grid = np.linspace(0, 2, n)

            idx = np.argmin(np.abs(grid - p))
            h[idx] = 1.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-6:
            h[np.random.choice(n, size=1)] += diff
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

            x0=initial_guess,
            niter=45,
            T=0.8,
            stepsize=0.5,
            minimizer_kwargs={
                'method': 'SLSQP',
                'bounds': bounds,
                'constraints': [{'type': 'eq', 'fun': constraint_sum}],
                'options': {'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
            }

            if current_c5 < best_c5:
