        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'disp': False}
    )

    if result_refine.success:
        best_c5 = result_refine.fun
        best_h = result_refine.x

    return best_h, best_c5, n_points
