import time

        h = np.array(h_values)

        return np.max(corr) * dx

        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

            grid = np.linspace(0, 2, n_points)
            h = np.zeros(n_points)
            for pos in positions:
                idx = np.argmin(np.abs(grid - pos))
                h[idx] = 1.0
            if abs(np.sum(h) - target_sum) > 1e-6:
                h[np.random.choice(n_points, size=1)] += (target_sum - np.sum(h))
            guesses.append(h)
        np.random.seed(seed + 1)
        guesses.append((initial_h + np.random.normal(0, 0.1, n_points)) * (1.0 + 1e-3))
        return guesses

    best_h = initial_h.copy()

    start_time = time.time()

    guesses = generate_initial_guesses()
    for guess in guesses:
        result = optimize.minimize(
