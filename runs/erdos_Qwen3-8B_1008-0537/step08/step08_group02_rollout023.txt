import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def generate_structured_guess(num_ones):
        positions = np.linspace(0, 2, num_ones, endpoint=False)
        grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for pos in positions:
            idx = np.argmin(np.abs(grid - pos))
            h[idx] = 1.0
        if abs(np.sum(h) - target_sum) > 1e-6:
            h[np.random.choice(n_points)] += (target_sum - np.sum(h))
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_de(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        constraint_violation = abs(np.sum(h) - target_sum)
        return c5 + 1e3 * constraint_violation
    
    def optimize_guess(initial_h):
        result_de = optimize.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            popsize=20,
            maxiter=200,
            mutation=(0.7, 1.2),
            recombination=0.9,
            tol=1e-10,
            vectorized=True
        )
        if result_de.success:
            result_slsqp = optimize.minimize(
                fun=objective,
                x0=result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if result_slsqp.success:
                return result_slsqp.x, objective(result_slsqp.x)
        return initial_h, objective(initial_h)
    
    guesses = [
        generate_structured_guess(22),
        generate_structured_guess(23),
        generate_structured_guess(24),
        initial_h_values
    ]
    
    for guess in guesses:
        h, c5 = optimize_guess(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    while time.time() - start_time < budget_s - 20:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:

                best_h = result.x
    
    return best_h, best_c5, n_points
