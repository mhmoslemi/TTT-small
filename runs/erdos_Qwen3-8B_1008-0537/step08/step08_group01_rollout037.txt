        if additional > 0:
            positions.append((m + 0.5) * (n / 2))
        start = np.zeros(n)
        for x in positions:
            idx = int(round(x))
            if 0 <= idx < n:
                start[idx] = 1.0
        if additional > 0:
            start[positions[0]] += additional
        return start

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h_values.copy()
