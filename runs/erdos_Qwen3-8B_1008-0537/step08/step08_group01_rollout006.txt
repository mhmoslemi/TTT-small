import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def generate_perturbed_guess(h, noise_scale=0.05):
        return h + np.random.normal(0, noise_scale, size=n_points)
    
    def generate_spread_guess(n, target_sum):
        m = int(target_sum)
        add = target_sum - m
        positions = []
        for i in range(1, m+1):
            pos = 2.0 * i / (m + 1)
            idx = int(round(pos * (n / 2)))
            positions.append(idx)
        start = np.zeros(n)
        for i in positions:
            if 0 <= i < n:
                start[i] = 1.0
        if add > 0 and positions:
            start[positions[0]] += add
        return start
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_vals):
        return np.sum(h_vals) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        res_de = optimize.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=25,
            maxiter=100,
            mutation=(0.8, 1.2),
            recombination=0.9,
            tol=1e-10,
            vectorized=True,
            seed=seed
        )
        if res_de.success:
            best_h = res_de.x
            best_c5 = objective(best_h)
        
        res_slsqp = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res_slsqp.success:
            best_h = res_slsqp.x
            best_c5 = objective(best_h)
        
        for _ in range(2):
            perturbed = generate_perturbed_guess(best_h, 0.02)
            res_slsqp = optimize.minimize(

                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint}],
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
            )
            if res_slsqp.success and objective(res_slsqp.x) < best_c5:
                best_h = res_slsqp.x
                best_c5 = objective(best_h)
        
        spread_guess = generate_spread_guess(n_points, target_sum)
        res_slsqp = optimize.minimize(

            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if res_slsqp.success and objective(res_slsqp.x) < best_c5:
            best_h = res_slsqp.x
            best_c5 = objective(best_h)
    except:
