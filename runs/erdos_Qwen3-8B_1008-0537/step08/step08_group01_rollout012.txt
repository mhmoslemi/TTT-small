import scipy.optimize as optimize
import time

    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)

    try:
        # Generate even guess
        even_guess = np.zeros(n_points)
        m = int(target_sum)
        additional = target_sum - m

            x = i * spacing
            idx = int(round(x * (n_points / 2)))
            if 0 <= idx < n_points:
                even_guess[idx] = 1.0
        if additional > 0 and m > 0:
            even_guess[0] += additional
        
        # Differential evolution with penalty
        def objective_de(h_values):
            h = np.array(h_values)
            corr = np.correlate(h, 1 - h, mode='full')

            penalty = 1e6 * constraint_violation
            return c5 + penalty

            bounds=bounds,

            vectorized=True
        )
        
        # SLSQP refinement
        result_slsqp = optimize.minimize(
