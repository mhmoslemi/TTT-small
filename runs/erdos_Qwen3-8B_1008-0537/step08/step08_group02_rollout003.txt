import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_structured_guess(n, num_ones):
        grid = np.linspace(0, 2, n)
        ones = np.random.choice(n, size=num_ones, replace=False)
        h = np.zeros(n)
        h[ones] = 1.0
        return h
    
    def optimize_from_guess(guess):
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result_slsqp.success:
            return result_slsqp.x, objective(result_slsqp.x)
        return guess, objective(guess)
    
    # Start with initial guess
    best_h, best_c5 = optimize_from_guess(initial_guess)
    
    # Try structured guesses with different numbers of ones
    for num_ones in [int(target_sum - 0.5), int(target_sum), int(target_sum + 0.5)]:
        if 0 < num_ones < n_points:
            guess = generate_structured_guess(n_points, num_ones)
            h, c5 = optimize_from_guess(guess)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    # Try random/near-initial perturbations
    for _ in range(2):
        perturb = np.random.normal(0, 0.05, size=n_points)
        guess = initial_guess + perturb

        h, c5 = optimize_from_guess(guess)
