        best_c5 = objective(result_e.x)
        best_h = result_e.x
    
    # Refine with perturbations
    while time.time() - start_time < budget_s - 20:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        new_guess = np.clip(new_guess, 0, 1)
        new_guess_sum = np.sum(new_guess)
        if abs(new_guess_sum - target_sum) > 1e-6:
            new_guess = new_guess * (target_sum / new_guess_sum)
        result_p = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_p.success and objective(result_p.x) < best_c5:
            best_c5 = objective(result_p.x)
            best_h = result_p.x
    
    return best_h, best_c5, n_points
