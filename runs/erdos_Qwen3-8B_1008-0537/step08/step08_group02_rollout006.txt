import scipy.optimize as optimize
import time

    def generate_structured_guess(n):
        num_ones = int(target_sum + 0.5)
        grid = np.linspace(0, 2, n)
        h = np.zeros(n)
        if num_ones <= 0:
            return h
        positions = np.random.choice(n, size=num_ones, replace=False)
        h[positions] = 1.0
        if abs(np.sum(h) - target_sum) > 1e-6:
            if num_ones < n:
                idx = np.argmin(np.abs(grid - 1.5))
                h[idx] += (target_sum - np.sum(h))
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()

    start_time = time.time()
    
    try:
        result_slsqp_initial = optimize.minimize(
            fun=objective,

            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
