            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    structured_guess = generate_structured_guess(n_points)
    result_slsqp_struc = optimize.minimize(
        fun=objective,
        x0=structured_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_struc.success:
        current_c5 = objective(result_slsqp_struc.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_struc.x
    
    while time.time() - start_time < budget_s - 20:
        new_guess = optimize.minimize(
            fun=objective,
            x0=best_h + np.random.normal(0, 0.01, size=n_points),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if new_guess.success:
            current_c5 = objective(new_guess.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_guess.x
    
    return best_h, best_c5, n_points
