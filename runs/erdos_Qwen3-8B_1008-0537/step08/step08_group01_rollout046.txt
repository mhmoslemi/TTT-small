import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Generate even guess
    def generate_even_guess(n, target_sum):
        m = int(target_sum)
        additional = target_sum - m
        positions = []

        for i in range(1, m + 1):
            x = i * spacing
            idx = int(round(x * (n / 2)))
            positions.append(idx)
        start = np.zeros(n)
        for i in positions:
            if 0 <= i < n:
                start[i] = 1.0
        if additional > 0 and positions:

        return start

        vectorized=True,

    )
    
    if result_de.success:

                best_h = result_slsqp.x
    
    # Try even guess
    result_slsqp_even = optimize.minimize(

            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    result_slsqp_even = optimize.minimize(
        fun=objective,
        x0=even_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )
    if result_slsqp_even.success:
        current_c5 = objective(result_slsqp_even.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp_even.x
    
    while time.time() - start_time < budget_s - 20:
        perturbed = best_h + np.random.normal(0, 0.01, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result_new = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_new.success:
            current_c5 = objective(result_new.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_new.x
    
    return best_h, best_c5, n_points
