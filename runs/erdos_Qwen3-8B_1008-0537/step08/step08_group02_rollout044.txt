import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guesses = [initial_h_values]
    # Create more structured guesses
    def gen_cluster_guess(n, t_sum, num_clusters):
        cluster_len = 2.0 / num_clusters
        positions = []
        for i in range(num_clusters):
            start = i * cluster_len
            end = (i+1) * cluster_len
            cnt = int(t_sum / num_clusters)
            if i == num_clusters - 1:
                cnt += t_sum - num_clusters * cnt
            spacing = end - start
            for j in range(cnt):
                pos = start + j * spacing / (cnt - 1)
                positions.append(pos)
        grid = np.linspace(0, 2, n)
        h = np.zeros(n)
        for pos in positions:
            idx = np.argmin(np.abs(grid - pos))
            h[idx] += 1.0
        adj = t_sum - np.sum(h)
        if abs(adj) > 1e-6:
            idx = np.random.choice(n)
            h[idx] += adj
        return h
    # Add structured guesses
    initial_guesses.append(gen_cluster_guess(n_points, target_sum, 2))
    initial_guesses.append(gen_cluster_guess(n_points, target_sum, 3))
    initial_guesses.append(gen_cluster_guess(n_points, target_sum, 4))
    initial_guesses.append(gen_cluster_guess(n_points, target_sum, 5))
    initial_guesses.append(gen_cluster_guess(n_points, target_sum, 10))
    best_h = initial_guesses[0]
    best_c5 = float('inf')
    start = time.time()
    
    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    for guess in initial_guesses:
        x0 = np.array(guess, dtype=float)
        res = optimize.minimize(
            objective, x0, method='SLSQP', bounds=[(0,1)]*n_points,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],

        )

                best_h = res.x

        res = optimize.minimize(
