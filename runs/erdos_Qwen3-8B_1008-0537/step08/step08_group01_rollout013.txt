import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=float)
    
    def generate_even_guess(n, target_sum):
        m = int(target_sum)
        additional = target_sum - m
        spacing = 2.0 / (m + 1)
        pos = np.arange(1, m + 1) * spacing
        x = np.round(pos * (n / 2)).astype(int)
        res = np.zeros(n)
        for i in x:
            if 0 <= i < n:
                res[i] = 1.0
        if additional > 0 and m > 0:
            res[x[0]] += additional
        return res
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        even_guess = generate_even_guess(n_points, target_sum)
        
        # Use SLSQP directly with initial guess
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
        
        # Try even guess
        result_slsqp_even = optimize.minimize(
            fun=objective,
            x0=even_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp_even.success:
            current_c5 = objective(result_slsqp_even.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp_even.x
        
        # Perturb best_h and refine
        for _ in range(5):
            if time.time() - start_time > budget_s - 20:
                break
            perturb = np.random.normal(0, 0.01, size=n_points)
            perturbed = best_h + perturb
            perturbed = np.clip(perturbed, 0.0, 1.0)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],

                if current_c5 < best_c5:

    except:

    return best_h, best_c5, n_points
