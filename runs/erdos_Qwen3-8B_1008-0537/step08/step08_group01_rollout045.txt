            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            return result.x, objective(result.x)
        return x0, objective(x0)
    
    # Try initial guess
    h_opt, c5 = evaluate_guess(initial_guess)
    if c5 < best_c5:
        best_h, best_c5 = h_opt, c5
    
    # Try even guess
    even_guess = generate_even_guess(n_points, target_sum)
    h_opt, c5 = evaluate_guess(even_guess)
    if c5 < best_c5:
        best_h, best_c5 = h_opt, c5
    
    # Try random guess
    np.random.seed(seed)
    random_guess = np.random.rand(n_points)
    random_guess = random_guess / np.sum(random_guess) * target_sum
    h_opt, c5 = evaluate_guess(random_guess)
    if c5 < best_c5:
        best_h, best_c5 = h_opt, c5
    
    # Try finer perturbations
    for _ in range(2):
        if time.time() - start_time > budget_s - 10:
            break
        perturb = np.random.normal(0, 0.01, size=n_points)
        new_guess = best_h + perturb
        new_guess = np.clip(new_guess, 0, 1)
        new_guess = new_guess * (target_sum / np.sum(new_guess))
        h_opt, c5 = evaluate_guess(new_guess)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
    
    return best_h, best_c5, n_points
