import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use provided initial H values if available; otherwise, construct a uniform one
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        n_points = 100  # Default to 100 points if no initial_h_values provided
        initial_h = np.full(n_points, 0.5)
    n_points = len(initial_h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Multi-start optimization with adaptive perturbation
    for _ in range(10):
        if time.time() - start_time > budget_s - 30:
            break  # Reserve time for final refinement
        # Perturb the current best guess
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    # Final refinement with SLSQP
    result = optimize.minimize(

        x0=best_h,
        method='SLSQP',

        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
