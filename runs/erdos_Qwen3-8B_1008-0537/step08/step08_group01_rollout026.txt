            if result_de.success and result_de.status == 0:
                result_slsqp = optimize.minimize(
                    fun=objective,
                    x0=result_de.x,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
                )
                if result_slsqp.success:
                    current_c5 = objective(result_slsqp.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
        except:
            pass

    while time.time() - start_time < budget_s - 20:
        perturb = np.random.normal(0, 0.05, n_points)
        new_guess = best_h + perturb
        new_guess = np.clip(new_guess, 0.0, 1.0)
        sum_new = np.sum(new_guess)
        delta = (target_sum - sum_new) / n_points
        new_guess += delta
        result_slsqp = optimize.minimize(
            fun=objective,
            x0=new_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result_slsqp.success:
            current_c5 = objective(result_slsqp.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x

    return best_h, best_c5, n_points
