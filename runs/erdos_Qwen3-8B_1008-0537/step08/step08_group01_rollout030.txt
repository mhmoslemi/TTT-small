        for i in range(0, n, 2):
            h[i] = 1.0
        current_sum = np.sum(h)
        if current_sum < target_sum:
            h[-1] += target_sum - current_sum
        elif current_sum > target_sum:
            h[0] -= current_sum - target_sum
        return h
    
    def objective(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_from_guess(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        return
    
    optimize_from_guess(initial_guess)
    binary_guess = generate_binary_guess(n_points, target_sum)
    optimize_from_guess(binary_guess)
    
    while time.time() - start_time < budget_s - 20:
        new_guess = best_h + np.random.normal(0, 0.01, size=n_points)
        optimize_from_guess(new_guess)
    
    return best_h, best_c5, n_points
