    m = np.max(vals)
    if eps > 0:
        sse = m + (1.0/eps) * np.log(np.sum(np.exp(eps*(vals - m))))
    else:
        sse = m
    return sse

def h_to_param(h, n):
    """Invert param_to_h approximately for initialization"""
    # h[i] = sigmoid(x[i] - offset)
    # So x[i] = logit(h[i]) + offset
    # We need to find x and offset such that sigmoid(x[i]-offset) ≈ h[i]
    # This is underdetermined, so just set offset = 0
    # x[i] = logit(clip(h[i], eps, 1-eps))
    eps = 1e-6
    h_c = np.clip(h, eps, 1-eps)
    x = np.log(h_c / (1 - h_c))
    return x

    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old)

    h_new = np.clip(h_new, 0, 1)

    # Adjust
    diff = target - np.sum(h_new)

                h_new = np.clip(h_new, 0, 1)
            else:
                break
    return h_new

    x = x - np.mean(x)  # Center
    
    best_x = x.copy()
    best_val = compute_c5(h_init, n)

            break
        if time.time() - t_start > max_time * 0.7:
            break
    
    h_best = param_to_h(best_x, n)
    return h_best, compute_c5(h_best, n)

    dx = 2.0 / n

def param_to_h(x, n):
    target = n / 2.0
    lo, hi = -30.0, 30.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        h = 1.0 / (1.0 + np.exp(-np.clip(x - mid, -50, 50)))
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = 1.0 / (1.0 + np.exp(-np.clip(x - (lo + hi) * 0.5, -50, 50)))

def obj_from_x(x, n, eps):
    h = param_to_h(x, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
