import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, n, temp):
    h = project_feasible(h, n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    if temp < 1e-13:
        return m
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

    h = project_feasible(h, n)
