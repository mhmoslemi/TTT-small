def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def compute_correlations(h):
    return np.correlate(h, 1-h, mode="full")

def repair(h, target_sum):
    """Ensure sum(h) = target_sum and 0 <= h <= 1"""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            mask = h < 1
        else:
            mask = h > 0
        n_adjust = np.sum(mask)
        if n_adjust > 0:
            h[mask] += diff / n_adjust
            h = np.clip(h, 0, 1)
    return h

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def local_search(h, n, dx, target_sum, max_iters, rng):
    """Local search by swapping bits"""
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(max_iters):
        # Pick a random 0 (or 0.5) and a random 1 (or 0.5)

        pos_1 = np.where(h > 0.5)[0]

        j = rng.choice(pos_1)
