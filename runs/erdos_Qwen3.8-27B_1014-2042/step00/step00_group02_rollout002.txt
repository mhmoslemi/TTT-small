def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    fval = float(np.max(corr) * dx)
    m = int(np.argmax(corr))
    k = m - n + 1
    grad = np.zeros(n)
    for j in range(n):
        i1 = j - k
        if 0 <= i1 < n:
            grad[j] += (1.0 - h[i1])
        j2 = j + k
        if 0 <= j2 < n:
            grad[j] += (-h[j2])
    return fval, grad * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h_proj)

        free = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if np.any(free):

    diff = target - np.sum(h_proj)

        h_proj[i_best] = float(np.clip(h_proj[i_best] + diff, 0.0, 1.0))
    return h_proj

def refine_with_slqp(h0, n):

    dx = 2.0 / n

        return c + np.log(np.sum(np.exp(corr - c))) / tau

                    grad[j] += wm * (-h[j2])
        return grad
