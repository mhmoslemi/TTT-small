import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def project_h(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(np.max(corr))
    shift = np.clip((corr - cmax) / T, -500.0, 500.0)
    return cmax + T * float(np.log(np.sum(np.exp(shift))))

def obj_smooth(x, n, T):
    target = n / 2.0
    h = project_h(x, target)
    return c5_smooth(h, T)

def obj_direct(x, n):
    target = n / 2.0
    h = project_h(x, target)
    return c5_value(h)

def upsample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return project_h(h.copy(), n_new / 2.0)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)

        h0 = np.full(45, 0.5)

    n0 = len(h0)

    best_h = h0.copy()
    best_c5 = c5_value(best_h)

    # Phase 1: Smooth Nelder-Mead at n0
    for T in [0.05, 0.01, 0.003, 0.001]:
        if time.time() - t_start > budget_s * 0.30:

        res = minimize(obj_smooth, h0, args=(n0, T), method='Nelder-Mead',

        if c5_opt < best_c5:

    # Phase 1b: Direct Nelder-Mead at n0

        res = minimize(obj_direct, h0, args=(n0,), method='Nelder-Mead',
