        xs_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        xs_new = np.linspace(0, 2, n_target, endpoint=False) + 1.0 / n_target
        h_up = np.interp(xs_new, xs_old, best_h, period=2.0)
        h_up = np.clip(h_up, 0, 1)
        h_up = project_h(h_up, n_target / 2.0)
        h_opt_u, c5_u = optimize_gradient(h_up, t0, budget_s)
        if c5_u < best_c5:
            best_c5 = c5_u
            best_h = h_opt_u.copy()
            h_ls_u, c5_lu = local_search(best_h, 800, seed + 777)
            if c5_lu < best_c5:
                best_c5 = c5_lu
                best_h = h_ls_u.copy()

    # Phase 5: Final polish with high-alpha gradient + local search
    if time.time() - t0 < budget_s - 20:
        n_final = len(best_h)
        h_polish = best_h.copy()
        for a in [2000, 5000, 10000]:
            lr = 4.0 / (a * n_final)
            for it in range(300):
                if time.time() - t0 > budget_s - 15:
                    break
                g = grad_softmax(h_polish, a)
                h_new = project_h(h_polish - lr * g, n_final / 2.0)
                h_polish = h_new
        c5_polish = compute_c5(h_polish)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish.copy()

    h_ls_final, c5_final = local_search(best_h, 3000, seed + 9999)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_ls_final.copy()

    h_pair_final, c5_pf = pair_local_search(best_h, 2000, seed + 9999)
    if c5_pf < best_c5:
        best_c5 = c5_pf
        best_h = h_pair_final.copy()

    return best_h, best_c5, len(best_h)
