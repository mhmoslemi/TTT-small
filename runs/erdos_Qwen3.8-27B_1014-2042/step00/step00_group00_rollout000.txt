import numpy as np
import time

def c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def feas(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n * 0.5
    d = t - h.sum()
    if abs(d) > 1e-13:
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = int(m.sum())
        if c > 0:
            h[m] += d / c
            h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-14:
            i = int(np.argmin(h)) if d > 0 else int(np.argmax(h))
            h[i] = float(np.clip(h[i] + d, 0.0, 1.0))
    return h

def local_search(h, n, rng, tl, ns=None, s_init=0.06):
    bh = h.copy()
    bv = c5_val(bh)
    t0 = time.time()
    s = s_init
    if ns is None:
        ns = max(2, n // 8)
    while time.time() - t0 < tl:
        hn = bh.copy()
        ix = rng.integers(0, n, size=ns)
        hn[ix] += rng.uniform(-s, s, size=ns)
        if rng.random() < 0.25:
            b = int(rng.integers(0, n))
            L = int(rng.integers(1, max(2, n // 6)))
            b = min(b, n - 1)
            actual = min(L, n - b)
            if actual > 0:
                hn[b:b + actual] += rng.uniform(-s, s, size=actual)
        hn = feas(hn, n)
        v = c5_val(hn)
        if v < bv:
            bh, bv = hn, v
            s = min(0.12, s * 1.03)
        else:
            s = max(0.003, s * 0.97)
    return bh, bv

def sa_search(h, n, rng, tl, t_init=0.025, t_floor=5e-6, sigma=0.04):
    bh = h.copy()
    bv = c5_val(bh)
    ch = h.copy()
    cv = bv
    T = t_init
    t0 = time.time()
    while time.time() - t0 < tl and T > t_floor:
        cn = ch + rng.normal(0.0, sigma, n)
        cn = feas(cn, n)
        v = c5_val(cn)
        delta = v - cv
        accept = delta < 0 or rng.random() < np.exp(-delta / T)
        if accept:
            ch, cv = cn, v
            if v < bv:
                bh, bv = cn.copy(), v
        T *= 0.9992
    return bh, bv

def coord_descent(h, n, rng, tl):
    bh = h.copy()
    bv = c5_val(bh)
    t0 = time.time()
    order = list(range(n))
    while time.time() - t0 < tl:
        rng.shuffle(order)
        improved = False
        for i in order:
            t_save = bh[i]
            for trial in [t_save + 0.03, t_save - 0.03, t_save + 0.01, t_save - 0.01,
                          0.0, 1.0, t_save + 0.005, t_save - 0.005]:
                bh[i] = float(np.clip(trial, 0.0, 1.0))
                bh = feas(bh, n)
                v = c5_val(bh)
                if v < bv:
                    bv = v
                    improved = True
                    break
            if not improved:
                bh[i] = t_save
        if not improved:
            break
    return bh, bv

        hs, vs = sa_search(bh, bn, rng, ta, t_init=0.03, sigma=0.035)

            bh, bc = hs, vs

    # Phase 4: Final local polish
