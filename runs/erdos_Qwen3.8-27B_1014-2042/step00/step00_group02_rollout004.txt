import numpy as np
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def make_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h = h + diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def subgradient(h, n):
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    max_val = cor.max()
    active = np.where(cor >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        if s >= 0:
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] -= h[s:]
        else:
            grad[-s:] += 1.0 - h[:n + s]
            grad[-s:] -= h[s:]
    grad *= dx / len(active)
    return grad

def sa_optimize(h_init, n, time_limit, rng, T0=0.004, T1=1e-8, cool=0.9997):
    h = h_init.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t0 = time.time()
    T = T0
    stall = 0
    ss = 0.04
    while time.time() - t0 < time_limit and T > T1:
        h_try = h.copy()
        move = rng.random()
        if move < 0.55:
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i2 == i1:
                i2 = (i2 + 1) % n
            eps = rng.uniform(-ss, ss)
            h_try[i1] += eps
            h_try[i2] -= eps
        elif move < 0.80:
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            i3 = int(rng.integers(0, n))
            if i2 == i1:
                i2 = (i2 + 1) % n
            if i3 == i1 or i3 == i2:
                i3 = (i3 + 2) % n
            e1 = rng.uniform(-ss, ss)
            e2 = rng.uniform(-ss, ss)
            h_try[i1] += e1
            h_try[i2] += e2
            h_try[i3] -= (e1 + e2)
        elif move < 0.90:
            shift = int(rng.integers(1, max(2, n // 10) + 1))
            h_try = np.roll(h_try, shift)
        else:
            b = int(rng.integers(2, max(3, n // 4) + 1))
            nb = n // b
            if nb > 1:
                blocks = h_try[:nb * b].reshape(nb, b)
                perm = rng.permutation(nb)
                h_try[:nb * b] = blocks[perm].flatten()
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        delta = c5_try - c5
        if delta < 0 or (T > T1 and rng.random() < math.exp(-delta / T)):

def gradient_descent(h_init, n, time_limit):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    lr = 0.5
    while time.time() - t0 < time_limit:
        grad = subgradient(h, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        h_try = h - lr * grad
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            h = h_try
            best_h = h.copy()
            best_c5 = c5_try
            lr = min(lr * 1.02, 10.0)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_c5
