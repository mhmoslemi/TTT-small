            x_old = np.arange(len(best[0])) * (2.0 / len(best[0]))
            x_new = np.arange(n_try) * dx_t
            h_try = np.interp(x_new, x_old, best[0])
        else:
            h_try = np.full(n_try, 0.5)
        h_try = proj_feasible(h_try, n_try, target_t)

        h_opt = do_optimize(h_try, n_try, 500, 0.005)
        update(h_opt, n_try)
        if elapsed() > budget_s * 0.88:
            h_ref = do_optimize(h_opt, n_try, 200, 0.001)
            update(h_ref, n_try)

    # Phase 4: final refinement on best
    if elapsed() < budget_s * 0.92 and best[0] is not None:
        n_f = best[2]
        for eps_f in [0.001, 0.0003]:
            if elapsed() > budget_s * 0.97:
                break
            h_f = do_optimize(best[0], n_f, 400, eps_f)
            update(h_f, n_f)

    # Final feasibility check
    if best[0] is None:
        best[0] = np.full(n_base, 0.5)
        best[2] = n_base

    n_final = best[2]
    best[0] = proj_feasible(best[0], n_final, float(n_final) / 2.0)
    best[1] = compute_c5(best[0], 2.0 / n_final)

    return best[0], best[1], n_final
