    bounds = [(0.0, 1.0)] * n

    def do_lbfgs(h_start, tau, pen, maxiter):
        res = minimize(smooth_obj_grad, h_start, jac=True, args=(tau, dx, pen),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_feasible(res.x, n)

    # Phase 1: Coarse to fine L-BFGS-B
    for phase, tau in enumerate([15, 40, 80, 160, 320, 640]):
        if time.time() - t0 > budget_s - 40:
            break
        pen = 20.0 + 10.0 * phase
        h_new = do_lbfgs(best_h, tau, pen, 400)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        print(f"  L-BFGS tau={tau}: C5={c5:.6f} (best {best_c5:.6f})")

    # Phase 2: Subgradient refinement
    for it in range(300):
        if time.time() - t0 > budget_s - 40:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        sg = np.zeros(n)
        i_lo = max(0, -lag)
        i_hi = min(n - 1, n - 1 - lag)
        for i in range(i_lo, i_hi + 1):
            j1 = i + lag
            sg[j1] += (1.0 - best_h[i])
            sg[i] -= best_h[j1]
        sg *= dx
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-14:
            break
        step = 0.02 / sg_norm
        h_try = best_h - step * sg
        h_try = project_feasible(h_try, n)
        c5 = compute_c5(h_try, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try
            if it % 50 == 0:
                print(f"  Subgrad it={it}: C5={c5:.6f}")
        else:
            step *= 0.5
            h_try = best_h - step * sg
            h_try = project_feasible(h_try, n)
            c5 = compute_c5(h_try, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try

    # Phase 3: Random restarts
    remaining = budget_s - (time.time() - t0)
    n_restart = max(10, int(remaining / 15))
    for r in range(n_restart):
        if time.time() - t0 > budget_s - 10:
            break
        scale = 0.03 + 0.12 * (r % 7) / 7.0
        h_try = best_h + scale * rng.randn(n)
        h_try = project_feasible(h_try, n)
        h_try = do_lbfgs(h_try, 200, 300.0, 200)
        c5 = compute_c5(h_try, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()
            print(f"  Restart {r}: C5={c5:.6f}")

    # Phase 4: Fine-tune best with high tau
    if time.time() - t0 < budget_s - 15:
        h_final = do_lbfgs(best_h, 1000, 500.0, 500)
        c5 = compute_c5(h_final, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()
        print(f"  Final tune: C5={c5:.6f}")

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    print(f"Final: n={n}, C5={best_c5:.6f}, elapsed={time.time()-t0:.1f}s")
    return best_h, best_c5, n
