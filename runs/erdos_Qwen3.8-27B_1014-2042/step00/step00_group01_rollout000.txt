        h = fix_sum(h, len(h))

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Phase 1: SLSQP on parent (long run)
    if time_left() > 120:
        h_opt = slsqp_optimize(best_h, max_iter=800, ftol=1e-14)
        try_update(h_opt)
    
    # Phase 2: Nelder-Mead refinement
    if time_left() > 80:
        h_nm = nelder_mead_optimize(best_h, max_iter=4000)
        try_update(h_nm)
    
    # Phase 3: SLSQP again on Nelder-Mead result
    if time_left() > 80:
        h_opt = slsqp_optimize(best_h, max_iter=500, ftol=1e-14)
        try_update(h_opt)
    
    # Phase 4: COBYLA
    if time_left() > 80:
        h_cob = cobyla_optimize(best_h, max_iter=3000)
        try_update(h_cob)
    
    # Phase 5: Multi-start perturbations
    n = len(best_h)
    scales = [0.02, 0.05, 0.1, 0.15, 0.2, 0.08, 0.12, 0.25]
    for scale in scales:
        if time_left() < 60:
            break
        for trial in range(2):
            if time_left() < 50:
                break
            h0 = best_h + rng.normal(0, scale, size=n)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = fix_sum(h0, n)
            h_opt = slsqp_optimize(h0, max_iter=300, ftol=1e-13)
            c = try_update(h_opt)
            if c < best_c5 + 1e-8:
                h_nm2 = nelder_mead_optimize(h_opt, max_iter=2000)
                try_update(h_nm2)
    
    # Phase 6: Random coordinate perturbations
    if time_left() > 60:
        for _ in range(10):
            if time_left() < 40:
                break
            h0 = best_h.copy()
            idx = rng.choice(n, size=max(3, n//4), replace=False)
            h0[idx] += rng.normal(0, 0.08, size=len(idx))
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = fix_sum(h0, n)
            h_opt = slsqp_optimize(h0, max_iter=250)
            try_update(h_opt)
    
    # Phase 7: Coordinate descent polish
    if time_left() > 30:
        h_cd = coordinate_descent(best_h, n_iter=800, step=0.005)
        try_update(h_cd)
    
    # Phase 8: Final SLSQP + NM cycle
    if time_left() > 40:
        h_final = slsqp_optimize(best_h, max_iter=400, ftol=1e-15)
        try_update(h_final)
    if time_left() > 30:
        h_nm_final = nelder_mead_optimize(best_h, max_iter=3000)
        try_update(h_nm_final)
    if time_left() > 20:
        h_final2 = slsqp_optimize(best_h, max_iter=300, ftol=1e-15)
        try_update(h_final2)
    
    # Phase 9: Try slightly different resolutions by upsampling
    if time_left() > 100:
        for n_trial in [50, 55, 60]:
            if time_left() < 80:
                break
            n_orig = len(best_h)
            # Upsample via linear interpolation
            x_orig = np.linspace(0, 2, n_orig, endpoint=False)
            x_new = np.linspace(0, 2, n_trial, endpoint=False)
            h_up = np.interp(x_new, x_orig, best_h)
            h_up = fix_sum(h_up, n_trial)
            h_opt = slsqp_optimize(h_up, max_iter=400)
            c_up = compute_c5(h_opt)
            if c_up < best_c5 - 1e-8:
                best_c5 = c_up
                best_h = h_opt.copy()
    
    # Final validation and correction
    n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(best_h)
    if abs(s - target) > 1e-10:
        # Iterative correction
        for _ in range(100):
            best_h += (target - s) / n
            best_h = np.clip(best_h, 0.0, 1.0)
            s = np.sum(best_h)
            if abs(s - target) < 1e-12:
                break
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n
