        ni = best[i] + d
        nj = best[j] - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        old_i, old_j = best[i], best[j]
        best[i] = ni
        best[j] = nj
        tc = _c5(best)
        if tc < best_c5 - 1e-13:
            best_c5 = tc
        else:
            best[i] = old_i
            best[j] = old_j
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.ones(45) * 0.5

    n = len(h0)
    s = n / 2.0
    h0 = _project(h0, s)

    best = h0.copy()
    best_c5 = _c5(best)

    # Phase 1: Differential Evolution (~45% budget)
    t_de_end = t0 + budget_s * 0.45
    P = 50
    pop = np.empty((P, n))
    pop[0] = best.copy()
    for i in range(1, P):
        pop[i] = _project(rng.uniform(0.0, 1.0, n), s)

    gen = 0
    while time.time() < t_de_end:
        F = max(0.35, 0.85 * (0.9992 ** gen))
        CR = 0.8
        for i in range(P):
            others = list(range(P))
            others.remove(i)
            a, b, c = rng.choice(others, 3, replace=False)
            mut = pop[a] + F * (pop[b] - pop[c])
            mut = _project(mut, s)
            trial = pop[i].copy()
            mask = rng.random(n) < CR
            if not mask.any():
                mask[int(rng.integers(n))] = True
            trial[mask] = mut[mask]
            trial = _project(trial, s)
            tc = _c5(trial)
            pc = _c5(pop[i])
            if tc < pc:
                pop[i] = trial
                if tc < best_c5:
                    best_c5 = tc
                    best = trial.copy()
        # Inject best into population periodically
        if gen % 20 == 0:
            pop[int(rng.integers(P))] = best.copy()
        gen += 1

    # Refresh best from pop
    for i in range(P):
        vc = _c5(pop[i])
        if vc < best_c5:
            best_c5 = vc
            best = pop[i].copy()

    # Phase 2: Local search with pair swaps (~40% budget)
    t_ls_end = t0 + budget_s * 0.87
    step = 0.04
    while time.time() < t_ls_end:
        best, best_c5 = _revert_and_try(best, n, s, rng, t_ls_end, best_c5, step)
        # Occasionally inject perturbation to escape local opt
        if rng.random() < 0.01:
            step = 0.05

    # Phase 3: Fine-grained polishing (remaining time)
    t_fin_end = t0 + budget_s - 5
    step_fine = 0.005
    while time.time() < t_fin_end:
        best, best_c5 = _revert_and_try(best, n, s, rng, t_fin_end, best_c5, step_fine)
        step_fine = max(0.0005, step_fine * 0.95)

    # Final feasibility check and correction
    drift = s - best.sum()
    if abs(drift) > 1e-12:
        # Spread correction evenly among interior points
        interior = np.where((best > 0.01) & (best < 0.99))[0]
        if len(interior) > 0:
            adj = drift / len(interior)
            best[interior] = np.clip(best[interior] + adj, 0.0, 1.0)
        # Residual fix
        drift2 = s - best.sum()
        if abs(drift2) > 1e-12:
            k = int(np.argmax(1.0 - np.abs(best - 0.5)))
            best[k] = max(0.0, min(1.0, best[k] + drift2))

    best = np.clip(best, 0.0, 1.0)
    final_c5 = _c5(best)
    if final_c5 > best_c5:
        final_c5 = best_c5

    return best, final_c5, n
