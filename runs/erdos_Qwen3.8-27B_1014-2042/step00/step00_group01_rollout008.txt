    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        lo, hi = 0.0, 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if np.sum(np.clip(h - mid, 0, 1)) > target:
                lo = mid
            else:
                hi = mid
        return np.clip(h - 0.5 * (lo + hi), 0, 1)
    else:
        lo, hi = 0.0, 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if np.sum(np.clip(h + mid, 0, 1)) < target:
                lo = mid
            else:
                hi = mid
        return np.clip(h + 0.5 * (lo + hi), 0, 1)

def cd_sweep(h, target, rng, n_iterations=10000, deadline=None):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    k_vals = np.arange(-(n - 1), n)
    stall = 0
    for _ in range(n_iterations):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            h[i] = 0.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            mask1 = (k_vals >= -i) & (k_vals <= n - 1 - i)
            idx1 = np.clip(i + k_vals, 0, n - 1)
            term1 = np.where(mask1, 1.0 - h[idx1], 0.0)
            mask2 = (k_vals >= i - (n - 1)) & (k_vals <= i) & (k_vals != 0)
            idx2 = np.clip(i - k_vals, 0, n - 1)
            term2 = np.where(mask2, h[idx2], 0.0)
            grad = term1 - term2
            xg = np.linspace(0.0, 1.0, 65)
            vals = corr[None, :] + grad[None, :] * xg[:, None]
            h[i] = xg[np.argmin(np.max(vals, axis=1))]
        h = project_to_constraint(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            stall = 0
        else:
            stall += 1
            if stall >= 300:
                break
    return h, best_c5

def slqp_polish(h, n, target, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(c) * dx)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        if time.time() > deadline - 10:
            return h, compute_c5(h)
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-14})
        x = project_to_constraint(res.x, target)
        c5 = compute_c5(x)
        if c5 < 10.0:
            return x, c5
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 15

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(47, 0.5)
    n0 = len(h0)
    target0 = n0 / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Coordinate descent from initial
    h1, c1 = cd_sweep(h0, target0, rng, 40000, deadline)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Multi-start with perturbations
    for s in range(30):
        if time.time() > deadline - 80:
            break
        h_s = best_h.copy()
        mask = rng.random(n0) < 0.5
        h_s[mask] += rng.normal(0, 0.35, mask.sum()).clip(-1.0, 1.0)
        h_s = project_to_constraint(h_s, target0)
        h_s, c_s = cd_sweep(h_s, target0, rng, 12000, deadline)
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s

    # Phase 3: SLSQP polish on best
    if time.time() < deadline - 20:
        hp, cp = slqp_polish(best_h, n0, target0, deadline)
        if cp < best_c5:
            best_c5, best_h = cp, hp

    # Phase 4: Try nearby resolutions
    for n_try in [48, 50, 60, 80]:
        if time.time() > deadline - 30:
            break
        target_try = n_try / 2.0
        h_try = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                          np.linspace(0, 2, n0, endpoint=False), best_h)
        h_try = project_to_constraint(h_try, target_try)
        h_try, c_try = cd_sweep(h_try, target_try, rng, 20000, deadline)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try
            n0 = n_try
            target0 = target_try

    # Phase 5: Final SLSQP on best across all resolutions
    if time.time() < deadline - 5:
        nf = len(best_h)
        hf, cf = slqp_polish(best_h, nf, nf / 2.0, deadline)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    # Final validation
    nf = len(best_h)
    best_h = project_to_constraint(best_h, nf / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, nf)
