    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(max_iter):
        r = rng.random()
        if r < 0.5:
            h_new = make_transfer(h_cur.copy(), n, rng, T)
        elif r < 0.8:
            h_new = targeted_move(h_cur.copy(), n, rng, T)
        else:
            i = int(rng.integers(n))
            h_temp = h_cur.copy()
            h_temp[i] += rng.normal(0, T * 0.5)
            h_new = project_feasible(h_temp, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
            T = max(T * 0.999, 1e-5)
        else:
            T = max(T * 0.9999, 1e-5)
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float).copy()
    n = len(h_init)
    target_sum = n / 2.0
    h_init = project_feasible(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Simulated Annealing (55% budget)
    phase1_end = t0 + budget_s * 0.55
    T = 0.03
    T_min = 5e-5
    no_improve = 0
    patience = 8000

    while time.time() < phase1_end:
        r = rng.random()
        if r < 0.35:
            h_new = make_transfer(best_h.copy(), n, rng, T)
        elif r < 0.70:
            h_new = targeted_move(best_h.copy(), n, rng, T)
        elif r < 0.85:
            # Multi-transfer
            h_temp = best_h.copy()
            for _ in range(2):
                h_temp = make_transfer(h_temp, n, rng, T * 0.7)
            h_new = h_temp
        else:
            # Binary perturbation
            i, j = rng.choice(n, 2, replace=False)
            h_temp = best_h.copy()
            h_temp[i] = 1.0 - h_temp[i]
            h_temp[j] = 1.0 - h_temp[j]
            h_new = project_feasible(h_temp, n)

        c5_new = compute_c5(h_new)
        delta_c = c5_new - best_c5

        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-10)):
            best_h = h_new
            best_c5 = c5_new
            no_improve = 0
            T = max(T * 0.9997, T_min)
        else:
            no_improve += 1
            T = max(T * 0.99995, T_min)

        if no_improve >= patience:
            no_improve = 0
            T = min(0.08, T * 3.0)

    # Phase 2: Local refinement (20% budget)
    phase2_end = t0 + budget_s * 0.75
    h_cur = best_h.copy()
    T2 = 0.015
    while time.time() < phase2_end:
        h_cur, c5_cur = local_refine(h_cur, n, rng, T2, max_iter=5000)
        if c5_cur < best_c5:
            best_h = h_cur.copy()
            best_c5 = c5_cur
            T2 *= 0.8
        else:
            T2 *= 0.85
        T2 = max(T2, 5e-5)

    # Phase 3: Binary rounding exploration (10% budget)
    phase3_end = t0 + budget_s * 0.85
    h_bin = binary_round(best_h, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h = h_bin
        best_c5 = c5_bin

    # Polish the binary solution
    while time.time() < phase3_end:
        h_cur2, c5_cur2 = local_refine(best_h.copy(), n, rng, 0.005, max_iter=3000)
        if c5_cur2 < best_c5:
            best_h = h_cur2
            best_c5 = c5_cur2
        else:
            break

    # Phase 4: Multi-start refinement (10% budget)
    phase4_end = min(t0 + budget_s * 0.95, t0 + budget_s - 5)
    n_starts = 5
    for s in range(n_starts):
        if time.time() > phase4_end:
            break
        # Perturbed start from best
        h_start = best_h.copy()
        for _ in range(3):
            h_start = make_transfer(h_start, n, rng, 0.05)
        h_start = binary_round(h_start, n)
        h_cur3, c5_cur3 = local_refine(h_start, n, rng, 0.01, max_iter=5000)
        if c5_cur3 < best_c5:
            best_h = h_cur3
            best_c5 = c5_cur3

    # Final: ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
