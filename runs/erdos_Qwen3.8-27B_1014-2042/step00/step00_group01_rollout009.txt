import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

    rng = np.random.default_rng(seed)
    
    def tleft():
        return budget_s - (time.time() - t0)

        h_best = np.array(initial_h_values, dtype=float).copy()
    else:

        h_best = np.ones(n_best) * 0.5
    
    n = len(h_best)
    target = n / 2.0
    h_best = project_feasible(h_best, target)
    c5_best = compute_c5(h_best)

    for beta in [50, 100, 200]:

                          options={'maxiter':300, 'ftol':1e-14, 'iprint':0})
            if res.x is not None:
                h_try = project_feasible(res.x, target)
                c5_try = compute_c5(h_try)
                if c5_try < c5_best:
                    h_best = h_try
                    c5_best = c5_try
        except Exception:
            pass

        c5_cd = c5_best
        for sweep in range(200):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def slsqp_optimize(h0, target, beta, maxiter):
    n = len(h0)
    dx = 2.0 / n
    def obj(h):
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        return smooth_max_val(c, beta)
    def cons_fun(h):
        return h.sum() - target
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':cons_fun}],
                   options={'maxiter':maxiter, 'ftol':1e-15, 'iprint':0})
    return res.x

def coord_descent(h0, target, n_sweeps=100, deltas=None):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    if deltas is None:
        deltas = [0.05, -0.05, 0.02, -0.02, 0.1, -0.1, 0.01, -0.01, 0.2, -0.2]
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            best_v = h[i]
            best_c = c5
            for dv in deltas:
                v = h[i] + dv
                if v < -1e-9 or v > 1.0 + 1e-9:
                    continue
                h[i] = max(0.0, min(1.0, v))
                h = project_feasible(h, target)
                c = compute_c5(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = h[i]
                    improved = True
                    h[i] = best_v
                else:
                    h[i] = best_v
            h = project_feasible(h, target)
            c5 = best_c
        if not improved:
            break
    return h, c5

def swap_anneal(h0, target, n_steps, temp_start, temp_end, rng):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for step in range(n_steps):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        temp = temp_start * (temp_end / temp_start) ** (step / max(n_steps, 1))
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(temp, 1e-12)):
            c5_cur = c5_new
            if c5_cur < best_c5 - 1e-14:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def noise_anneal(h0, target, n_steps, temp_start, temp_end, rng):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for step in range(n_steps):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        noise = rng.gauss(0, 0.05)
        h[i] += noise
        h[j] -= noise
        h = project_feasible(h, target)
        c5_new = compute_c5(h)
        temp = temp_start * (temp_end / temp_start) ** (step / max(n_steps, 1))
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(temp, 1e-12)):
            c5_cur = c5_new

                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i] -= noise
            h[j] += noise
            h = project_feasible(h, target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    return h_best, c5_final, len(h_best)
