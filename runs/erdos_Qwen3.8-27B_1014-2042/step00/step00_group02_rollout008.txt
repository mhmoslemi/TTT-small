import numpy as np
from scipy.optimize import minimize


def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def _smooth_max(vals, T):
    m = vals.max()
    return float(m + T * np.log(np.sum(np.exp((vals - m) / T))))


def _repair(h, ts):
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    for _ in range(60):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += d * r / s
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += d * h / s
        h = np.clip(h, 0.0, 1.0)
    return h


class _Obj:
    def __init__(self, T, n):
        self.T = T
        self.dx = 2.0 / n
    def __call__(self, x):
        h = np.clip(x, 0.0, 1.0)
        c = np.correlate(h, 1.0 - h, mode='full')
        return _smooth_max(c, self.T) * self.dx


class _SumEq:
    def __init__(self, ts):
        self.ts = ts
    def fun(self, x):
        return np.sum(x) - self.ts
    def jac(self, x):
        return np.ones(len(x))


def _slsqp(hi, T, ts, n, mi=500):

                     constraints=cs, options={'maxiter': mi, 'ftol': 1e-15})
        if r.x is not None:

    except Exception:
        pass
    return _repair(hi, ts)
