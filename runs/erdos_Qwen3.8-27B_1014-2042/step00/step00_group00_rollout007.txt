import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, ts):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(2000):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
            if not m.any():
                break
            h[m] += d / m.sum()
        else:
            m = h > 1e-15
            if not m.any():
                break
            h[m] += d / m.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, beta):
    mx = c.max()
    if mx < 1e-15:
        return 0.0
    s = beta * (c - mx)
    sm = s.max()
    return mx + (sm + np.log(np.exp(s - sm).sum())) / beta

def make_obj(n, ts, beta):
    dx = 2.0 / n
    def obj(x):
        h = project(x, ts)
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        return smooth_max(c, beta)
    return obj

def make_obj_exact(n, ts):
    dx = 2.0 / n
    def obj(x):
        h = project(x, ts)
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        return float(c.max())
    return obj
