import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_simplex(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mu = 0.5 * (lo + hi)
        hp = np.clip(h - mu, 0.0, 1.0)
        s = np.sum(hp)
        if s > target_sum:
            lo = mu
        else:
            hi = mu
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_corr_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[lag:] += (1.0 - h[:m])
            g[:m] -= h[lag:]
    else:
        L = -lag
        m = n - L
        if m > 0:
            g[:m] += (1.0 - h[L:])
            g[L:] -= h[:m]
    return g

def subgradient_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    fmax = np.max(corr)
    tol = max(1e-8, fmax * 1e-6)
    active = np.where(corr >= fmax - tol)[0]
    g = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        g += grad_corr_lag(h, lag)
    g /= len(active)
    return g, fmax

def smooth_objective(h, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr / temp
    m = np.max(shifted)
    p = np.exp(shifted - m)
    p /= np.sum(p)
    f_val = m + np.log(np.sum(p))
    dx = 2.0 / n
    return f_val * dx

def smooth_gradient(h, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr / temp
    m = np.max(shifted)
    p = np.exp(shifted - m)
    p /= np.sum(p)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
            continue
        lag = k - (n - 1)
        g += p[k] * grad_corr_lag(h, lag)
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.ones(45) * 0.5

    n = len(h_start)
    target = n / 2.0
    h_start = project_box_simplex(np.clip(h_start, 0.0, 1.0), target)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)

    def elapsed_frac():
        return (time.time() - t0) / budget_s

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Smooth optimization from start
    h = h_start.copy()
    temperature = 2.0
    lr = 0.1
    max_iters_phase1 = 2000
    for it in range(max_iters_phase1):
        if time_left() < budget_s * 0.3:
            break
        g = smooth_gradient(h, temperature)
        f_curr = smooth_objective(h, temperature)
        alpha = lr
        improved = False
        for _ in range(12):
            h_new = project_box_simplex(h - alpha * g, target)
            f_new = smooth_objective(h_new, temperature)
            if f_new < f_curr - 1e-12:
                h = h_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            lr = max(lr * 0.7, 1e-10)
            if lr < 1e-10:
                break
            continue
        lr = min(lr * 1.05, 0.5)
        if it % 50 == 0:
            temperature *= 0.95
        c5_now = evaluate_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()

    # Phase 2: Hard max subgradient with restarts
    max_restarts = 30
    for attempt in range(max_restarts):
        if time_left() < budget_s * 0.15:
            break

        if attempt == 0:
            h = best_h.copy()
        elif attempt % 3 == 1:
            h = best_h + rng.randn(n) * 0.03
            h = project_box_simplex(h, target)
        elif attempt % 3 == 2:
            h = best_h.copy()
            idx = rng.choice(n, size=max(2, n // 8), replace=False)
            h[idx] = 1.0 - h[idx]
            h = project_box_simplex(h, target)
        else:
            # Random-ish restart preserving sum
            h = np.clip(0.5 + rng.randn(n) * 0.3, 0.0, 1.0)
            h = project_box_simplex(h, target)

        lr = 0.05
        stagnant = 0
        for it in range(5000):

            dx = 2.0 / n

            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
                stagnant = 0
            else:

            alpha = lr
            improved = False
            for _ in range(15):
                h_new = project_box_simplex(h - alpha * g, target)
                g2, fmax2 = subgradient_max_corr(h_new)
                if fmax2 < fmax - 1e-10:

                    improved = True

                alpha *= 0.4
            if not improved:

                if lr < 1e-10:

                if stagnant > 200:
                    break
        # After each restart, if stuck, try a temperature step
        if best_c5 > 0.3808 and time_left() > budget_s * 0.15:
            temperature = 0.5
            for it in range(300):
                if time_left() < budget_s * 0.12:
                    break
                g = smooth_gradient(h, temperature)
                f_curr = smooth_objective(h, temperature)
                alpha = lr
                improved = False
                for _ in range(10):
                    h_new = project_box_simplex(h - alpha * g, target)
                    f_new = smooth_objective(h_new, temperature)
                    if f_new < f_curr - 1e-12:
                        h = h_new
                        improved = True
                        break
                    alpha *= 0.5
                if not improved:
                    break
                c5_now = evaluate_c5(h)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
            temperature *= 0.8

    # Final polish: very small steps on best solution
    h = best_h.copy()
    lr = 1e-4
    for it in range(2000):
        if time_left() < 5.0:
            break
        g, fmax = subgradient_max_corr(h)
        alpha = lr
        improved = False
        for _ in range(10):

            _, fmax2 = subgradient_max_corr(h_new)

                h = h_new
                fmax = fmax2
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
        c5_now = evaluate_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()

    # Verify and return
    best_h = project_box_simplex(np.clip(best_h, 0.0, 1.0), target)
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
