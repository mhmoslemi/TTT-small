import numpy as np
from scipy.optimize import minimize
import time

def project_capped(v, S):
    """Project v onto {x: 0<=x_i<=1, sum(x)=S} via bisection water-filling."""
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-14:
        mask = (x > 1e-9) & (x < 1.0 - 1e-9)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
    return np.clip(x, 0.0, 1.0)


def c5_val(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, 'full')
    return c.max() * dx


def smooth_obj(x, beta, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, 'full') * dx
    m = c.max()
    sm = m + np.log(np.sum(np.exp(beta * (c - m)))) / beta
    sv = x.sum() - n * 0.5
    return sm + penalty * sv * sv


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        parent = np.array(initial_h_values, dtype=float)
    except NameError:
        parent = np.full(45, 0.5)
        parent[8:37] = 1.0

    n = len(parent)
    S = n / 2.0
    best_h = project_capped(parent, S)
    best_c5 = c5_val(best_h)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with increasing smoothing
    for beta in [3, 8, 20, 50, 120]:
        if time.time() - t0 > budget_s - 50:
            break
        res = minimize(smooth_obj, best_h.copy(), method='L-BFGS-B',
                       bounds=bounds, args=(beta, 50.0),
                       options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-7})
        h_new = project_capped(res.x, S)
        c5_new = c5_val(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Coordinate descent (coarse)
    deltas_coarse = [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.25, -0.25]
    for sweep in range(200):
        if time.time() - t0 > budget_s - 40:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > budget_s - 40:
                break
            orig = best_h[i]
            for d in deltas_coarse:
                nv = orig + d
                if nv < 0.0 or nv > 1.0:
                    continue
                h_try = best_h.copy()
                h_try[i] = nv
                h_try = project_capped(h_try, S)
                c5_try = c5_val(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break

    for r in range(50):

            h_start[swap_i[0]], h_start[swap_i[1]] = h_start[swap_i[1]], h_start[swap_i[0]]

        h_start = project_capped(h_start, S)
        for beta in [8, 30, 80]:

            h_new = project_capped(res.x, S)

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h_start = h_new

    # Phase 4: Fine coordinate descent
    deltas_fine = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.04, -0.04, 0.06, -0.06]
    for sweep in range(300):
        if time.time() - t0 > budget_s - 10:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > budget_s - 10:
                break

            for d in deltas_fine:
                nv = orig + d
                if nv < 0.0 or nv > 1.0:

                h_try[i] = nv
                h_try = project_capped(h_try, S)
                c5_try = c5_val(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break
        if not improved:
            break

    # Final L-BFGS-B polish
    if time.time() - t0 < budget_s - 5:
        for beta in [50, 100, 200]:
            res = minimize(smooth_obj, best_h.copy(), method='L-BFGS-B',
                           bounds=bounds, args=(beta, 100.0),
                           options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
            h_new = project_capped(res.x, S)
            c5_new = c5_val(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Verify constraints
    final_c5 = c5_val(best_h)
    assert abs(best_h.sum() - S) < 1e-6, f"Sum constraint violated: {best_h.sum()} vs {S}"
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)

    return best_h, final_c5, n
