def compute_gradient(h, T):
    n = len(h)
    dx = 2.0/n
    c_full = np.correlate(h, 1-h, mode="full") * dx
    c_max = c_full.max()
    w = np.exp((c_full - c_max) / T)
    w /= w.sum()
    
    grad = np.zeros(n)
    for j in range(len(c_full)):
        s = j - (n-1)
        # ∂total_j/∂h_t = (1-h_{t+s}) for valid t+s, minus h_{t-s} for valid t-s
        if w[j] < 1e-10:
            continue
        # First term: (1-h[t+s]) for t where 0≤t<n and 0≤t+s<n
        t_range_1 = np.arange(max(0, -s), min(n, n-s))
        if len(t_range_1) > 0:
            grad[t_range_1] += w[j] * (1 - h[t_range_1 + s])
        # Second term: -h[t-s] for t where 0≤t<n and 0≤t-s<n
        t_range_2 = np.arange(max(0, s), min(n, n+s))
        if len(t_range_2) > 0:
            grad[t_range_2] -= w[j] * h[t_range_2 - s]
    
    return dx / T * grad

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = c.max()
    if T < 1e-6:
        return cm
    return cm + T * np.log(np.sum(np.exp((c - cm) / T)))

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = c_full.max()
    w = np.exp((c_full - cm) / T)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(1, 2 * n - 1):  # skip j=0 (shift=-(n-1), minimal)
        if w[j] < 1e-12:
            continue
        s = j - (n - 1)
        # First term: (1-h[t+s]) for valid t
        t1_lo = max(0, -s)
        t1_hi = min(n, n - s)
        if t1_hi > t1_lo:
            grad[t1_lo:t1_hi] += w[j] * (1.0 - h[t1_lo + s:t1_hi + s])
        # Second term: -h[t-s] for valid t
        t2_lo = max(0, s)
        t2_hi = min(n, n + s)
        if t2_hi > t2_lo:
            grad[t2_lo:t2_hi] -= w[j] * h[t2_lo - s:t2_hi - s]
    return (dx / T) * grad

def optimize_smooth(h, n, time_limit):
    target = n / 2.0
    h = h.copy()
    h = project_box_sum(h, target)
    start = time.time()
    lr = 0.01
    for T in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() - start > time_limit:
            break
        for _ in range(200):
            if time.time() - start > time_limit:
                break
            g = smooth_grad(h, T)
            h_new = h - lr * g
            h_new = project_box_sum(h_new, target)
            # Simple line search / step size adaptation
            if smooth_obj(h_new, T) < smooth_obj(h, T):
                h = h_new
            else:
                lr *= 0.7
            if lr < 1e-8:
                lr = 0.01
                break
    # Final polish
    h = project_box_sum(h, target)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h)

def try_binary(n, time_limit):

    if n % 2 != 0:
        return None
    target = n // 2
    start = time.time()
    best = None

            break
        h = np.zeros(n)
        h[np.random.choice(n, target, replace=False)] = 1.0
        val = eval_c5(h)
        if val < best_val:
            best_val = val
            best = h.copy()

        for _ in range(2000):
            if time.time() - start > time_limit:
                break
            i = np.random.randint(n)
            # Find a 0 to swap

            if len(zeros) == 0:

            j = zeros[np.random.randint(len(zeros))]
            best[i], best[j] = best[j], best[i]
            val = eval_c5(best)
            if val < best_val:
                best_val = val
            else:
                best[i], best[j] = best[j], best[i]
    
    return best if best is not None else None, best_val

        n = len(best_h)
        target = n / 2.0
        remaining = max(5, budget_s * 0.9 - (time.time() - t0))

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = c.max()
    if T < 1e-8:
        return cm
    c_shifted = (c - cm) / T
    c_shifted = np.clip(c_shifted, -50, 50)
    return cm + T * np.log(np.sum(np.exp(c_shifted)))

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = c_full.max()
    w = np.exp(np.clip((c_full - cm) / T, -50, 50))
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-12:
            continue
        s = j - (n - 1)
        t_lo1 = max(0, s)
        t_hi1 = min(n, n + s)
        if t_hi1 > t_lo1:
            grad[t_lo1:t_hi1] += wj * (1.0 - h[t_lo1 - s:t_hi1 - s])
        t_lo2 = max(0, -s)
        t_hi2 = min(n, n - s)
        if t_hi2 > t_lo2:
            grad[t_lo2:t_hi2] -= wj * h[t_lo2 + s:t_hi2 + s]
    return (dx / T) * grad

def optimize_smooth(h, time_limit):
    n = len(h)
    target = n / 2.0
    h = project_box_sum(h.copy(), target)
    start = time.time()
    best_h = h.copy()
    best_obj = smooth_obj(h, 0.01)
    for T in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() - start > time_limit:
            break
        lr = 0.05
        stall = 0
        for it in range(300):
            if time.time() - start > time_limit:
                break
            g = smooth_grad(h, T)
            h_new = h - lr * g
            h_new = project_box_sum(h_new, target)
            obj_new = smooth_obj(h_new, T)
            if obj_new < smooth_obj(h, T):
                h = h_new
                stall = 0
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_h = h.copy()
            else:
                lr *= 0.6
                stall += 1
                if stall > 15 or lr < 1e-9:
                    break
    return project_box_sum(best_h, target)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) + 0.5
    x_new = np.arange(n_new) + 0.5
    x_new = x_new * (n_old - 0.5) / (n_new - 0.5)
    return np.interp(x_new, x_old, h)

def binary_local_search(h, time_limit, rng):
    n = len(h)
    best_h = h.copy()
    best_val = eval_c5(h)
    start = time.time()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    temp = 0.01
    for _ in range(20000):
        if time.time() - start > time_limit:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        v = eval_c5(h)
        if v < best_val or rng.rand() < np.exp(-(v - best_val) / temp):
            best_val = v
            best_h = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i], h[j] = h[j], h[i]
        temp *= 0.9999
    return best_h, best_val

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float)

        for trial in range(3):

            h_rand = np.clip(0.5 + 0.3 * rng.randn(n), 0, 1)
            h_rand = project_box_sum(h_rand, n / 2.0)
            h_opt = optimize_smooth(h_rand, max(3.0, budget_s * 0.02))
            c5 = eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Binary search for even sizes
    for n in [44, 46, 48, 50, 52, 56]:
        if time.time() - t0 > budget_s * 0.92:
            break
        target = n // 2
        h_bin = np.zeros(n)
        h_bin[:target] = 1.0
        h_bin = h_bin[rng.permutation(n)]
        h_best_bin, v_best = binary_local_search(h_bin, max(5.0, budget_s * 0.015), rng)
        if v_best < best_c5:

    best_c5 = eval_c5(best_h)
    assert 0 <= np.min(best_h) and np.max(best_h) <= 1
    assert abs(np.sum(best_h) - len(best_h) / 2.0) < 1e-6
    return best_h, best_c5, len(best_h)
