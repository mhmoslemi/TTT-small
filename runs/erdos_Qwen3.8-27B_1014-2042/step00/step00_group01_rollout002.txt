from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c.max()

def smooth_c5(h, n, beta=50.0):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return (1.0/beta) * np.log(np.sum(np.exp(beta * c)))

def smooth_grad(c, beta):
    p = np.exp(beta * c)
    p /= p.sum()
    return p  # weights for each lag

def grad_smooth(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    p = np.exp(beta * c)
    p /= p.sum()
    # c has 2n-1 entries, lag index k corresponds to shift -(n-1)+k
    # c[k] = sum_i h[i] * (1-h[i + lag_k]) * dx where lag_k = k - (n-1)
    grad = np.zeros(n)
    for lag_idx in range(2*n-1):
        lag = lag_idx - (n-1)
        w = p[lag_idx] * dx
        if w < 1e-12:
            continue
        if lag >= 0:

            for i in range(n - lag):
                grad[i] += w * (1 - h[i+lag])
                grad[i+lag] -= w * h[i]
        else:
            lag_abs = -lag
            # c = sum_{i=lag_abs}^{n-1} h[i] * (1-h[i-lag_abs])
            for i in range(lag_abs, n):
                grad[i] += w * (1 - h[i-lag_abs])
                grad[i-lag_abs] -= w * h[i]
