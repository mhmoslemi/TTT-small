import numpy as np
from scipy.optimize import minimize
import time


def project_to_simplex(h, target_sum):
    n = len(h)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def obj_and_grad(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    obj = corr[idx] * dx
    lag = (n - 1) - idx
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    else:
        m = -lag
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    return obj, grad * dx


def pgd_loop(h, target_sum, n_pts, max_iters, time_limit):
    lr = 0.03
    dx = 2.0 / n_pts
    t0 = time.time()
    best_obj = compute_c5(h)
    stall = 0
    for _ in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        obj, grad = obj_and_grad(h, dx)
        h_new = h - lr * grad
        h_new = project_to_simplex(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_obj - 1e-14:
            best_obj = c5_new
            h = h_new
            stall = 0
            lr = min(lr * 1.02, 0.15)
        else:
            stall += 1
            lr *= 0.93
        if lr < 1e-7:
            break
    return h, best_obj


def slsqp_optimize(h0, n_pts, max_iter=400):
    dx = 2.0 / n_pts
    target = n_pts / 2.0
    bounds = [(0.0, 1.0)] * n_pts

    def _con_fun(h):
        return np.sum(h) - target

    constraints = [{'type': 'eq', 'fun': _con_fun}]
    res = minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    if np.all(np.isfinite(res.x)):
        h_c = project_to_simplex(np.clip(res.x, 0, 1), target)
        return h_c, compute_c5(h_c)
    return h0, compute_c5(h0)
