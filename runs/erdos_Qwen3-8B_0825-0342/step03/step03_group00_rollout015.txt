import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Set a higher n_points for better resolution, but limit to 150 to stay within budget
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize with a more strategic starting point
    initial_h = np.zeros(n_points)
    initial_h[:n_points//2] = 1.0
    initial_h[n_points//2:] = 0.0
    initial_h = initial_h * (1 / (np.sum(initial_h) * dx))
    
    # Define the objective function to minimize the maximum overlap
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Define the constraint: sum(h) * dx == 1
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use SLSQP method with a more aggressive initial try
    result = optimize.minimize(
        objective_func,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func},
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Perform a few perturbations to refine
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
        current_c5 = evaluate_erdos_solution(h_perturbed)
        if current_c5 < best_c5:
            best_h = h_perturbed
            best_c5 = current_c5
    
    # Use a regularized version to avoid overfitting and improve smoothness
    def objective_with_regularization(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        regularizer = np.sum(np.abs(np.gradient(h)))  # Regularize smoothness
        return np.max(corr) + 0.005 * regularizer
    
    result = optimize.minimize(
        objective_with_regularization,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func},
        options={'ftol': 1e-9, 'maxiter': 150}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Final check for local refinements
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
        current_c5 = evaluate_erdos_solution(h_perturbed)
        if current_c5 < best_c5:
            best_h = h_perturbed
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
