```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a higher-resolution initial guess based on the known construction
    n_points = 120
    initial_h = initial_h_values().copy()
    initial_h = np.resize(initial_h, n_points)
    dx = 2.0 / n_points
    initial_h = initial_h * (1 / (np.sum(initial_h) * dx))
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    best_time = 0
    time_start = time.time()
    
    # Define the objective function
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Define the constraints: sum(h) * dx == 1
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    # Bounds for h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize with SLSQP
    result = optimize.minimize(
        objective_func,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func},
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    h_opt = result.x
    current_c5 = evaluate_erdos_solution(h_opt)
    best_h = h_opt
    best_c5 = current_c5
    
    # Refine with guided descent
    for _ in range(5):
        # Add regularization to the objective to encourage smoothness
        def objective_with_regularization(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            regularizer = np.sum(np.abs(np.gradient(h)))  # Regularize smoothness
            return np.max(corr) + 0.01 * regularizer
        
        result = optimize.minimize(
            objective_with_regularization,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        best_h = result.x.copy()
        current_c5 = evaluate_erdos_solution(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    # Perturb the best solution and refine
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        if current_c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = current_c5_perturbed
    
    # Final refinement with regularization
    def final_objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    result = optimize.minimize(
        final_objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func},
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    best_h = result.x.copy()
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Check time budget
    elapsed_time = time.time() - time_start
    if elapsed_time < budget_s:
        # Try a denser grid if time allows
        n_points = 150
        initial_h = best_h.copy()
        initial_h = np.resize(initial_h, n_points)
        dx = 2.0 / n_points
        initial_h = initial_h * (1 / (np.sum(initial_h) * dx))
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize with SLSQP
        result = optimize.minimize(
            objective_func,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        if current_c5 < final_c5:
            best_h = h_opt.copy()
            final_c5 = current_c5
    
    return best_h, final_c5, n_points
```
