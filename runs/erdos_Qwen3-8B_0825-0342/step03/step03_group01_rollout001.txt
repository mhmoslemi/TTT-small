import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Use a more structured initial guess based on piecewise constant functions
    def generate_initial_guess(n_points):
        # Create a piecewise constant function with alternating high and low values
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 0.25
            elif i % 3 == 1:
                h[i] = 0.75
            else:
                h[i] = 0.5
        # Normalize the function to ensure the integral is 1
        dx = 2.0 / n_points
        total_area = np.sum(h) * dx
        h = h * (1 / total_area)
        return h

    # Initial guess based on the provided initial_h_values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    
    # Try different n_points to find the best solution
    n_points_list = [n_points, n_points + 10, n_points + 20, n_points + 30, n_points + 40, n_points + 50]
    n_points_list = [np.clip(n, 40, 150) for n in n_points_list]  # Ensure we stay in the relevant range
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    elapsed_time = 0
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate a more structured initial guess
        h = generate_initial_guess(n_points)
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Use L-BFGS-B optimizer with perturbations to avoid local minima
        result = optimize.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': 1e-7, 'maxiter': 200}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Add a small random perturbation to the solution to escape local minima
        h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
