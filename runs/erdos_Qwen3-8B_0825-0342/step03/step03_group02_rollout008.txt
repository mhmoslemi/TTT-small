import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate
import random

def run(seed=42, budget_s=60, **kwargs):
    # Pre-loaded initial construction
    initial_h_values = np.array([0.4925, 0.5833, 0.6458, 0.6865, 0.7135, 0.7342, 0.7495, 0.7602, 0.7673, 0.7713,
                                 0.7722, 0.7701, 0.7652, 0.7578, 0.7482, 0.7367, 0.7235, 0.7086, 0.6923, 0.6746,
                                 0.6559, 0.6364, 0.6162, 0.5954, 0.574, 0.5522, 0.5301, 0.5076, 0.4847, 0.4616,
                                 0.4382, 0.4145, 0.3905, 0.3663, 0.3419, 0.3173, 0.2926, 0.2677, 0.2428, 0.2179,
                                 0.193, 0.168, 0.143, 0.118, 0.093, 0.068, 0.043, 0.018, 0.0])

    # Try multiple n_points values for better performance and exploration
    n_points_list = [64, 96, 128]
    best_c5 = float('inf')
    best_h_values = None
    best_n_points = 0

    # Set seed for reproducibility
    random.seed(seed)

    for n_points in n_points_list:
        # Initial construction with n_points samples
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        # Initialize with a symmetric version of the initial_h_values
        for i in range(n_points):
            h[i] = initial_h_values[i % len(initial_h_values)]
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)

        time_limit = budget_s
        start_time = time.time()

        def compute_c5(h_values):
            h = np.array(h_values)
            shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
            one_minus_h = 1 - h
            corr = correlate(shifted_h, one_minus_h, mode='full')
            return np.max(corr) * dx

        # Initialize with the current h
        best_h = h.copy()
        best_c5_current = compute_c5(best_h)

        # Optimization loop with different strategies
        for _ in range(10):  # Multiple iterations to explore
            # Random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            new_h = best_h + perturbation
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            c5_candidate = compute_c5(new_h)
            if c5_candidate < best_c5_current:
                best_c5_current = c5_candidate
                best_h = new_h.copy()

            # Gradient descent with L-BFGS-B
            def objective_func(h_values):
                return compute_c5(h_values)

            result = optimize.minimize(
                objective_func,
                best_h,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)]
            )
            if result.success:
                h_candidate = result.x
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = compute_c5(h_candidate)
                if c5_candidate < best_c5_current:
                    best_c5_current = c5_candidate
                    best_h = h_candidate.copy()
            else:
                break

            # Check if time is exceeded
            if time.time() - start_time > time_limit:
                break

        # Update best solution
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h_values = best_h.copy()
            best_n_points = n_points

    return best_h_values, best_c5, best_n_points
