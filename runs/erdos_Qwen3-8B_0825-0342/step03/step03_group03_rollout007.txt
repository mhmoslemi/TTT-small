import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=60, **kwargs):
    # Pre-loaded initial construction (n=49 samples)
    initial_h_values = np.array([0.4925, 0.5833, 0.6458, 0.6865, 0.7135, 0.7342, 0.7495, 0.7602, 0.7673, 0.7713,
                                 0.7722, 0.7701, 0.7652, 0.7578, 0.7482, 0.7367, 0.7235, 0.7086, 0.6923, 0.6746,
                                 0.6559, 0.6364, 0.6162, 0.5954, 0.574, 0.5522, 0.5301, 0.5076, 0.4847, 0.4616,
                                 0.4382, 0.4145, 0.3905, 0.3663, 0.3419, 0.3173, 0.2926, 0.2677, 0.2428, 0.2179,
                                 0.193, 0.168, 0.143, 0.118, 0.093, 0.068, 0.043, 0.018, 0.0])

    # Start with 49 points as initial
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    time_limit = budget_s
    start_time = time.time()

    def compute_c5(h_values):
        h = np.array(h_values)
        shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
        one_minus_h = 1 - h
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Initial best values
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    # Try uniform distribution
    uniform_h = np.full(n_points, 0.5)
    c5_uniform = compute_c5(uniform_h)
    if c5_uniform < best_c5:
        best_c5 = c5_uniform
        best_h = uniform_h.copy()

    # Try alternating pattern
    alt_h = np.zeros(n_points)
    for i in range(n_points):
        alt_h[i] = 0.5 if i % 2 == 0 else 0.0
    c5_alt = compute_c5(alt_h)
    if c5_alt < best_c5:
        best_c5 = c5_alt
        best_h = alt_h.copy()

    # Try random restarts with different n_points
    for n in [49, 64, 96, 128]:
        new_n = n
        new_dx = 2.0 / new_n
        new_h = np.random.uniform(0, 1, new_n)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (new_n / 2)  # Normalize to meet area constraint
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()
            n_points = new_n
            dx = new_dx

    # Try optimizing with random perturbations
    def random_perturbation(h_values, perturbation_scale=0.05):
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=len(h_values))
        new_h = h_values + perturbation
        new_h = np.clip(new_h, 0, 1)
        return new_h

    # Local search with random perturbations
    for _ in range(5):
        new_h = random_perturbation(best_h)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h

    # Use Nelder-Mead simplex algorithm for optimization
    result = minimize(
        lambda x: compute_c5(x),
        best_h,
        method='Nelder-Mead',
        bounds=[(0, 1) for _ in range(len(best_h))],
        options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 100}
    )

    if result.success:
        h_candidate = result.x
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate

    # Local search with random perturbations again
    for _ in range(3):
        new_h = random_perturbation(best_h)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h

    # Check time and return best results
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
