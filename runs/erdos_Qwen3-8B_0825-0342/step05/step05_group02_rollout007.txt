```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from the initial construction
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if sum_h * dx != 1:
        initial_h = initial_h * (1 / (sum_h * dx))
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    elapsed_time = 0
    start_time = time.time()
    
    # Try a few different n_points to find the best solution
    n_points_list = [80, 100, 120, 150]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate a more structured initial guess with a wave-like shape
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * x / 2)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h * dx != 1:
            h = h * (1 / (sum_h * dx))
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        # Use a more sophisticated optimization method with tighter constraints
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
            options={'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Perturb the solution to explore nearby regions
        h_perturbed = h_opt + np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
```
