import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Try different n_points starting from 120 and increasing
    n_points_list = [120, 140, 160, 180, 200]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 120
    
    for n_points in n_points_list:
        if time.time() > budget_s:
            break
        
        # Initial guess
        h_initial = initial_h_values()
        h_initial = np.resize(h_initial, n_points)
        h_initial = h_initial * (n_points / 120)  # Adjust for different n_points
        h_initial = h_initial * (1 / (np.sum(h_initial) * 2.0 / n_points))
        
        # Define objective and constraint
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        def constraint_func(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) * (2.0 / n_points) - 1.0
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use a local optimization method with multiple starting points
        def optimize_with_perturbations(h_initial, max_iter=50, perturbation_scale=0.05):
            best_h = h_initial.copy()
            best_c5 = evaluate_erdos_solution(best_h)
            
            for _ in range(max_iter):
                # Perturb the current best solution
                perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
                h_perturbed = best_h + perturbation
                h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
                h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
                
                # Local optimization
                result = optimize.minimize(
                    objective_func,
                    h_perturbed,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_func},
                    options={'ftol': 1e-8, 'maxiter': 100}
                )
                
                h_opt = result.x
                current_c5 = evaluate_erdos_solution(h_opt)
                
                # Update best solution if improved
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt.copy()
            
            return best_h, best_c5
        
        # Run optimization
        start_time = time.time()
        h_opt, c5 = optimize_with_perturbations(h_initial, max_iter=50, perturbation_scale=0.05)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n_points = n_points
        
        # Refinement step
        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            
            # Perturb the best_h
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
            
            # Local optimization
            result = optimize.minimize(
                objective_func,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_func},
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            h_opt = result.x
            current_c5 = evaluate_erdos_solution(h_opt)
            
            # Update best solution if improved
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
    
    return best_h, best_c5, best_n_points
