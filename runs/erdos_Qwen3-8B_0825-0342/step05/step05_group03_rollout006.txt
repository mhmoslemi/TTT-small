import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define the number of samples and the population size for genetic algorithm
    n_points = 120
    population_size = 50
    generations = 100
    mutation_rate = 0.1

    # Initial population
    initial_population = []
    for _ in range(population_size):
        h = initial_h_values()
        h = np.resize(h, n_points)
        h = h * (1 / (np.sum(h) * 2.0 / n_points))
        initial_population.append(h)
    
    best_h = initial_population[0]
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    # Genetic algorithm loop
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness for all individuals
        fitness = []
        for h in initial_population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
        
        # Select the best individuals for crossover
        indices = np.argsort(fitness)
        selected_indices = indices[:int(population_size * 0.6)]
        selected_population = [initial_population[i] for i in selected_indices]
        
        # Crossover to generate new individuals
        new_population = []
        for _ in range(population_size):
            parent1 = random.choice(selected_population)
            parent2 = random.choice(selected_population)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):
            for j in range(n_points):
                if random.random() < mutation_rate:
                    new_population[i][j] += np.random.uniform(-0.1, 0.1)
                    new_population[i][j] = np.clip(new_population[i][j], 0, 1)
            # Maintain the integral constraint
            new_population[i] = new_population[i] * (1 / (np.sum(new_population[i]) * 2.0 / n_points))
        
        # Replace the population
        initial_population = new_population
        
        # Update the best solution
        for h in initial_population:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
