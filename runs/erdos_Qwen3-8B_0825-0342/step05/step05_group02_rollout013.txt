# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Start from a known good construction: 80 samples
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define the n_points grid to try
    n_points_list = [80, 100, 120, 140, 160]
    
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = len(best_h)
    elapsed_time = 0
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initial guess: sinusoidal shape
        x = np.linspace(0, 2, n_points)
        h_initial = 0.5 + 0.5 * np.sin(np.pi * x)
        h_initial = np.clip(h_initial, 0, 1)
        dx = 2.0 / n_points
        sum_h = np.sum(h_initial)
        if sum_h * dx != 1:
            h_initial = h_initial * (1 / (sum_h * dx))
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        # Optimization with constraints
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Perturb the solution to explore nearby regions
        h_perturbed = h_opt + np.random.uniform(-0.03, 0.03, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Return the best solution found
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, best_n_points
