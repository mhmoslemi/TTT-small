import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Try different n_points values in a more refined range
    n_points_list = [50, 60, 70, 80, 90, 100, 120, 150]
    
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = 50
    best_time = 0
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initial guess: alternating blocks of 1 and 0
        block_size = n_points // 4
        initial_h = np.zeros(n_points)
        initial_h[0:block_size] = 1.0
        initial_h[block_size:2*block_size] = 0.0
        initial_h[2*block_size:3*block_size] = 1.0
        initial_h[3*block_size:] = 0.0
        
        # Normalize to satisfy the integral constraint
        dx = 2.0 / n_points
        initial_h = initial_h * (1 / (np.sum(initial_h) * dx))
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        # Define the constraints: sum(h) * dx == 1
        def constraint_func(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) * dx - 1
        
        # Use SLSQP optimizer for more robust optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Initial attempt with SLSQP
        result = optimize.minimize(
            objective_func,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        # Perturb the solution to explore nearby points
        for _ in range(3):
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = h_opt + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
            current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
            
            if current_c5_perturbed < current_c5:
                current_c5 = current_c5_perturbed
                h_opt = h_perturbed
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Final refinement with guided local search for the best solution so far
        if best_c5 < 0.381:
            def objective_with_regularization(h_flat):
                h = h_flat.reshape(n_points)
                corr = np.correlate(h, 1 - h, mode="full") * dx
                # Regularize smoothness
                regularizer = np.sum(np.abs(np.gradient(h)))  
                return np.max(corr) + 0.01 * regularizer
            
            result = optimize.minimize(
                objective_with_regularization,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_func},
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            best_h = result.x.copy()
            best_c5 = evaluate_erdos_solution(best_h)
    
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, best_n_points
