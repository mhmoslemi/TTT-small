import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial construction
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    elapsed_time = 0
    start_time = time.time()
    
    # Define the function to evaluate the overlap
    def evaluate_overlap(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Use a more refined optimization approach
    result = optimize.minimize(
        evaluate_overlap,
        h_initial,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-9, 'maxiter': 150}
    )
    
    h_opt = result.x
    current_c5 = evaluate_erdos_solution(h_opt)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_opt.copy()
        best_n_points = n_points
    
    # Optional: Add perturbations to explore better
    h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
    
    if current_c5_perturbed < best_c5:
        best_c5 = current_c5_perturbed
        best_h = h_perturbed.copy()
        best_n_points = n_points
    
    # Consider other n_points values
    for n_points in [45, 50, 60, 70, 80, 90, 100, 120, 150, 180, 210]:
        if time.time() - start_time > budget_s:
            break
        
        if n_points <= len(best_h):
            continue  # Skip if already tried
        
        # Construct a sinusoidal initial guess for better distribution
        h = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5
        h = np.clip(h, 0, 1)
        
        # Ensure that the integral constraint is satisfied
        dx = 2.0 / n_points
        sum_h = np.sum(h)
        if sum_h * dx != 1:
            h = h * (1 / (sum_h * dx))
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        # Use L-BFGS-B optimizer with perturbations to avoid local minima
        result = optimize.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': 1e-9, 'maxiter': 150}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Optional: Add perturbations to explore better
        h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
