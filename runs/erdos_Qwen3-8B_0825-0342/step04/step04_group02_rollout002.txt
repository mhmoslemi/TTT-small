```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = len(best_h)
    elapsed_time = 0
    start_time = time.time()
    
    # Try multiple n_points
    n_points_list = [45, 50, 60, 70, 80, 90, 100, 120, 150, 200]
    
    # Strategy: Use a multi-start optimization with different initializations and objective functions
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initial guess: sinusoidal with normalization
        h_initial = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5
        h_initial = np.clip(h_initial, 0, 1)
        dx = 2.0 / n_points
        sum_h = np.sum(h_initial)
        if sum_h * dx != 1:
            h_initial = h_initial * (1 / (sum_h * dx))
        
        # Define objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        # Use a more advanced optimization approach with multiple iterations and perturbations
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 50}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        # Perturb and optimize again
        for _ in range(3):
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = np.clip(h_opt + perturbation, 0, 1)
            result_perturbed = optimize.minimize(
                objective_func,
                h_perturbed,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 50}
            )
            h_perturbed_opt = result_perturbed.x
            current_c5_perturbed = evaluate_erdos_solution(h_perturbed_opt)
            
            if current_c5_perturbed < current_c5:
                h_opt = h_perturbed_opt
                current_c5 = current_c5_perturbed
        
        # Check if this is the best solution found so far
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
```
