import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Strategy:
    # 1. Start with a better initial guess based on known optimal patterns.
    # 2. Use a fine-grained optimization with multiple starting points.
    # 3. Use a combination of different optimization methods to avoid local minima.
    # 4. Test multiple n_points to find the best solution within the time budget.
    
    # Try different n_points to find the best solution
    n_points_list = [45, 50, 60, 70, 80, 90, 100, 120, 150]  # Try more values for better exploration
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = 45
    elapsed_time = 0
    
    start_time = time.time()
    
    # Initial guess from the provided initial_h_values
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    # Use a combination of different optimization methods
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # First optimization with L-BFGS-B
    result = optimize.minimize(
        objective_func,
        h_initial,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-7, 'maxiter': 100}
    )
    
    h_opt = result.x
    current_c5 = evaluate_erdos_solution(h_opt)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_opt.copy()
        best_n_points = n_points
    
    # Perturb and optimize again
    h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
    
    if current_c5_perturbed < best_c5:
        best_c5 = current_c5_perturbed
        best_h = h_perturbed.copy()
        best_n_points = n_points
    
    # Try other n_points
    for n_points in n_points_list:
        if n_points <= len(best_h):
            continue  # Skip if already tried
        
        if time.time() - start_time > budget_s:
            break
        
        # Better initial guess: alternating pattern with controlled overlap
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 0.5
            elif i % 3 == 0:
                h[i] = 0.6
            else:
                h[i] = 0.4
        
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        sum_h = np.sum(h)
        if sum_h * dx != 1:
            h = h * (1 / (sum_h * dx))
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        # Use L-BFGS-B optimizer with perturbations to avoid local minima
        result = optimize.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': 1e-7, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Optional: Add perturbations to explore better
        h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
