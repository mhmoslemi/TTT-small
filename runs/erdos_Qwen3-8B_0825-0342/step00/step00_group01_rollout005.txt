            overlaps.append(overlap * dx)
        return np.max(overlaps)

        n = len(h)

        # Maximize the overlaps
        max_overlap = cp.Max(overlaps)
        
        # Objective: minimize max_overlap
        objective = cp.Minimize(max_overlap)
        
        # Solve the problem
        prob = cp.Problem(objective, [integral_const])
