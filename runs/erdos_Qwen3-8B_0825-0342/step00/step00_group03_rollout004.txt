```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points
    
    # Initial guess: use the initial_h_values, rescaled to fit n_points
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) > 0:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                initial_h[i] = 1.0
            else:
                initial_h[i] = 0.0
    
    # Ensure sum of h is n_points / 2
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Optimization variables
    h = cp.Variable(n_points)
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]
    
    # Objective function: minimize the maximum overlap integral
    # Calculate the correlation
    def objective(h_values):
        h_vals = h_values.reshape(-1)
        corr = signal.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    # Optimizer settings
    options = {
        'max_iter': 100,
        'eps': 1e-6,
        'verbose': False
    }
    
    # Run optimization
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    for _ in range(5):  # Run multiple times for better exploration
        # Define the optimization problem
        prob = cp.Problem(cp.Minimize(cp.Max([cp.sum(cp.multiply(h[i], 1 - h[i + k])) for k in range(n_points)])), constraints)
        prob.solve(solver=cp.SCS, **options)
        
        # Check if the solution is feasible
        if prob.status != 'optimal':
            continue
        
        # Evaluate the solution
        current_h = h.value
        # Compute the correlation
        corr = signal.correlate(current_h, 1 - current_h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Ensure the solution satisfies the constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Compute final C5 value
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Check time budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        print("Time budget exceeded, returning best solution found so far.")
    
    return (best_h.tolist(), final_c5, n_points)
```
