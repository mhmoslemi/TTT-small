import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Starting point: use the initial_h_values
    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=np.float64)
    
    # Define dx
    dx = 2.0 / n_points
    
    # Objective: minimize the maximum overlap integral
    # For numerical optimization, use a grid search with some refinement
    # Start with a grid of h values near the initial solution
    
    # Define the function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        # Ensure h is in [0, 1] and normalize to sum to n_points / 2
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * (n_points / 2)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Optimization: use scipy.optimize to find a better solution
    # We use a bounded optimization with the initial guess and some random perturbation
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess: slightly perturb the initial solution
    h_perturbed = h_initial.copy()
    perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
    h_perturbed += perturbation
    h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
    h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
    
    # Use L-BFGS-B to optimize the overlap
    result = opt.minimize(
        evaluate_overlap,
        h_perturbed,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-6, 'maxiter': 100}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    c5_bound = evaluate_overlap(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), c5_bound, n_points)
