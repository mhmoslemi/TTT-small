import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points_initial = len(initial_h_values)
    h = np.array(initial_h_values).copy()
    n_points = n_points_initial
    
    # Try a few different n_points starting from the initial
    for n_points in [n_points_initial, n_points_initial * 2, n_points_initial * 3]:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Ensure n_points is even for symmetry
        if n_points % 2 != 0:
            n_points += 1
        
        # Create a grid of x values
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initialize h with uniform distribution (but adjust for integral constraint)
        h = np.zeros(n_points)
        h[:n_points // 2] = 1
        h = h / (n_points / 2)
        
        # Define the objective function: maximize the overlap integral
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            # Compute 1 - h(x + k)
            # We use a sliding window with circular boundary conditions
            shift = np.roll(h, 1)
            overlap = np.correlate(h, 1 - shift, mode="full")
            c5 = np.max(overlap) * dx
            return c5
        
        # Define the constraints: sum(h) == n_points / 2
        def constraints(h_flat):
            return np.sum(h_flat) - n_points / 2
        
        # Optimization using scipy's minimize
        h_initial = (np.random.rand(n_points) * 2 - 1) * 0.5
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        result = opt.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=1e-6,
            options={'maxiter': 100}
        )
        
        if result.success:
            h_opt = result.x
            c5_bound = objective(h_opt)
            
            # Try to further refine with a more accurate solver
            h_refined = cp.Variable(n_points)
            constraints_refined = [cp.sum(h_refined) == n_points / 2]
            obj_refined = cp.Maximize(cp.max(cp.correlate(h_refined, 1 - cp.roll(h_refined, 1), mode="full")) * dx)
            prob_refined = cp.Problem(obj_refined, constraints_refined)
            
            try:
                prob_refined.solve(solver='ECOS', verbose=False)
                h_opt = h_refined.value
                c5_bound = objective(h_opt)
            except:
                pass
            
            # Evaluate the solution
            c5_bound = evaluate_erdos_solution(h_opt, n_points, dx)
            if c5_bound < 0.380876:
                return (h_opt.tolist(), c5_bound, n_points)
    
    # If no better solution found, return the initial one
    h_initial = initial_h_values
    c5_initial = evaluate_erdos_solution(h_initial, len(h_initial), 2.0 / len(h_initial))
    return (h_initial, c5_initial, len(h_initial))
