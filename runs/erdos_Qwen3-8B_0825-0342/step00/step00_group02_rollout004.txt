```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 45
    h_values = initial_h_values
    dx = 2.0 / n_points
    
    # Evaluate initial C5
    c5_initial = evaluate_erdos_solution(h_values, dx)
    
    # Try increasing number of points for better resolution
    for n_points in [60, 75, 90, 105, 120, 135, 150, 165, 180]:
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        
        # Resample to n_points
        h_resampled = np.zeros(n_points)
        h_resampled[::3] = h_values
        h_resampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_values)), h_values)
        h_resampled = np.clip(h_resampled, 0, 1)
        dx = 2.0 / n_points
        sum_h = np.sum(h_resampled)
        
        # Adjust to ensure integral is 1
        if sum_h != n_points / 2:
            scale_factor = (n_points / 2) / sum_h
            h_resampled *= scale_factor
        
        # Evaluate C5
        c5 = evaluate_erdos_solution(h_resampled, dx)
        
        if c5 < c5_initial:
            h_values = h_resampled
            c5_initial = c5
            print(f"Improved C5: {c5_initial} with {n_points} points")
    
    # Further refinement using optimization
    n_points_opt = 120
    dx_opt = 2.0 / n_points_opt
    h_opt = np.random.uniform(0, 1, n_points_opt)
    sum_h_opt = np.sum(h_opt)
    if sum_h_opt != n_points_opt / 2:
        scale_factor = (n_points_opt / 2) / sum_h_opt
        h_opt *= scale_factor
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != n_points_opt / 2:
            scale_factor = (n_points_opt / 2) / sum_h
            h = h * scale_factor
        c5 = evaluate_erdos_solution(h, dx_opt)
        return c5
    
    # Use L-BFGS-B optimizer
    result = opt.minimize(objective, h_opt, bounds=((0, 1),) * n_points_opt, method='L-BFGS-B', options={'maxiter': 100})
    
    if result.success:
        h_opt = np.clip(result.x, 0, 1)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt != n_points_opt / 2:
            scale_factor = (n_points_opt / 2) / sum_h_opt
            h_opt = h_opt * scale_factor
        c5_opt = evaluate_erdos_solution(h_opt, dx_opt)
        if c5_opt < c5_initial:
            h_values = h_opt
            c5_initial = c5_opt
    
    # Final evaluation
    dx_final = 2.0 / len(h_values)
    c5_final = evaluate_erdos_solution(h_values, dx_final)
    
    return h_values, c5_final, len(h_values)
```
