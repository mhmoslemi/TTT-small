```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial parameters
    n_points = 45  # Start with the initial sample size
    h = initial_h_values.copy()
    h = h[:n_points]  # Ensure we're not overstepping the initial array

    # Normalize to ensure the integral is 1
    h = h * (n_points / 2) / np.sum(h)

    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_n = n_points

    # Try increasing the number of points up to a reasonable limit
    for n_points in range(50, 200):
        # Define the decision variables
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        # Objective: minimize the maximum overlap integral
        # Note: We approximate the max using a convex relaxation
        overlap = cp.built_in('max', [cp.multiply(h, 1 - cp.roll(h, k)) for k in range(1, n_points)])
        objective = cp.Minimize(overlap)

        # Solve the problem
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver='ECOS', verbose=False)
            if prob.status == 'optimal':
                h_vals = h.value
                current_c5 = evaluate_erdos_solution(h_vals)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_vals.copy()
                    best_n = n_points
            else:
                print(f"Optimal solution not found for n={n_points}")
        except:
            print(f"Solver failed for n={n_points}, skipping.")

        # Check if we've exceeded the budget
        if time.time() - start_time > budget_s:
            break

    # Return the best found solution
    return best_h, best_c5, best_n
```
