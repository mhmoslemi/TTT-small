```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 samples, can be increased if time permits
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        dx = 2.0 / n_points
        h_array = np.array(h)
        h_rolled = np.roll(h_array, shift=1, axis=0)
        overlap = np.correlate(h_array, 1 - h_rolled, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to optimize h to minimize the maximum overlap
    def optimize_h():
        nonlocal best_c5, best_h
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        obj = cp.Minimize(cp.Max(cp.multiply(h, 1 - cp.roll(h, 1))))
        problem = cp.Problem(obj, constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status == 'optimal':
                h_values = np.round(h.value, 5).tolist()
                c5 = evaluate_overlap(h_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
        except:
            pass
    
    # Initial guess
    h_initial = initial_h_values.copy()
    if len(h_initial) != n_points:
        h_initial = np.random.rand(n_points)
        h_initial = (h_initial * n_points / 2).tolist()
    best_h = h_initial
    best_c5 = evaluate_overlap(best_h)
    
    # Iterative optimization with cooling schedule
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        optimize_h()
        # Perturb the solution slightly
        perturbation = np.random.rand(n_points) * 0.1 - 0.05
        h_perturbed = np.clip(np.array(best_h) + perturbation, 0, 1)
        c5_perturbed = evaluate_overlap(h_perturbed.tolist())
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.tolist()
    
    # Final check with a more precise solver
    h = cp.Variable(n_points)
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]
    obj = cp.Minimize(cp.Max(cp.multiply(h, 1 - cp.roll(h, 1))))
    problem = cp.Problem(obj, constraints)
    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status == 'optimal':
            h_values = np.round(h.value, 5).tolist()
            c5 = evaluate_overlap(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values
    except:
        pass
    
    return best_h, best_c5, n_points
```
