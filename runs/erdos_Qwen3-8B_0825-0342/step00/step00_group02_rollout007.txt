def run(seed=42, budget_s=60, **kwargs):
    n_points = 100  # Start with 100 points for speed
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    start_time = time.time()
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(overlap)
    
    # Optimization: use differential evolution for global search
    result = optimize.differential_evolution(
        objective,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        tol=1e-6,
        polish=True,
        disp=False,
        x0=h_initial
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Check if we can improve with a larger n_points
    if time.time() - start_time < budget_s - 5:
        n_points = 150
        h_initial = best_h.copy()
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        result = optimize.differential_evolution(
            objective,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            strategy='best1bin',
            maxiter=80,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            tol=1e-6,
            polish=True,
            disp=False,
            x0=h_initial
        )
        
        best_h = result.x
        best_c5 = result.fun
    
    # Final check with n_points = 200
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        h_initial = best_h.copy()
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        result = optimize.differential_evolution(
            objective,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            tol=1e-6,
            polish=True,
            disp=False,
            x0=h_initial
        )
        
        best_h = result.x
        best_c5 = result.fun
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * (2.0 / n_points)
    
    return (best_h, best_c5, n_points)
