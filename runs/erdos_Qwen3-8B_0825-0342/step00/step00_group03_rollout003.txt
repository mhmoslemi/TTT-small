def run(seed=42, budget_s=60, **kwargs):

    h = np.random.rand(n_points) * dx * n_points  # Initial guess with random values in [0, 1]

    # Set up the optimization problem
    # Variables: h_values, a vector of size n_points
    h_values = cp.Variable(n_points)
    constraints = []
    constraints.append(cp.sum(h_values) == n_points / 2)  # Ensure the integral equals 1

    # Evaluate the overlap integral for all possible shifts
    # This is a simplified approach and can be optimized with more advanced methods
    def evaluate_overlap(h):
        h_values = h.copy()
        overlap = np.zeros(n_points)
        for k in range(n_points):
            if k < n_points:
                overlap[k] = np.dot(h_values, (1 - h_values[k:])) * dx
            else:
                overlap[k] = 0
            if k >= n_points:
                overlap[k] = np.dot(h_values, (1 - h_values[k - n_points:])) * dx
        return np.max(overlap)

    # Optimization with a simpler approach (local search with random perturbations)
    best_h = h.copy()
    best_c5 = evaluate_overlap(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Perturb the current solution
        perturbation = np.random.rand(n_points) * 0.1 - 0.05
        new_h = best_h + perturbation

        if new_c5 < best_c5:
            best_h = new_h
