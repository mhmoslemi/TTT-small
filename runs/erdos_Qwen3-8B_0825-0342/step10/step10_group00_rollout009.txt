```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define a grid of sample points
    n_points_list = [150, 180, 200, 220, 240, 260, 280, 300, 320, 350, 380, 400]
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = len(best_h)
    elapsed_time = 0
    start_time = time.time()
    
    # Function to evaluate the overlap integral
    def evaluate_c5(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to generate a random perturbation
    def perturb(h_flat, perturbation_scale=0.05):
        h_perturbed = h_flat + np.random.uniform(-perturbation_scale, perturbation_scale, size=len(h_flat))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        return h_perturbed
    
    # Function to optimize using SLSQP
    def optimize_h(h_initial, n_points):
        dx = 2.0 / n_points
        sum_h = np.sum(h_initial) * dx
        if sum_h != 1:
            h_initial = h_initial * (1 / (sum_h))
        
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        return result.x
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate an initial guess
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = optimize_h(h_initial, n_points)
        
        # Perturb slightly and optimize
        h_perturbed = perturb(h_initial)
        h_perturbed = optimize_h(h_perturbed, n_points)
        
        # Evaluate both the initial and perturbed solutions
        current_c5 = evaluate_c5(h_initial)
        current_c5_perturbed = evaluate_c5(h_perturbed)
        
        # Update the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_initial.copy()
            best_n_points = n_points
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
```
