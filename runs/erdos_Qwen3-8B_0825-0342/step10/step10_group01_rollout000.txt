```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random
from scipy.optimize import Bounds

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial construction
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    elapsed_time = 0
    start_time = time.time()
    
    # Try with different n_points values
    n_points_list = [120, 140, 160, 180, 200, 220, 240, 260, 280, 300]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Interpolate initial guess to new n_points
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h * dx != 1:
            h_initial = h_initial * (1 / (sum_h * dx))
        
        # Define optimization problem
        h = cp.Variable(n_points, lb=0, ub=1)
        constraints = [cp.sum(h) == n_points / 2]
        objective = cp.Minimize(cp.Max(cp.correlate(h.value, 1 - h.value, mode="full") * dx))
        
        # Solve using CVXPY with solver
        try:
            result = cp.solve(objective, constraints, solver='MOSEK', verbose=False)
            if result is None:
                continue
            h_opt = h.value
            current_c5 = evaluate_erdos_solution(h_opt)
            
            # Perturb the solution to avoid local minima
            h_perturbed = h_opt + np.random.uniform(-0.02, 0.02, size=n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
                best_n_points = n_points
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = h_perturbed.copy()
                best_n_points = n_points
        except:
            continue
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
```
