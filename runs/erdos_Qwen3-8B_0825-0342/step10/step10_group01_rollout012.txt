```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with initial construction
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    elapsed_time = 0
    start_time = time.time()
    
    # Try with different n_points values
    n_points_list = [100, 120, 140, 160, 180, 200, 220, 240, 260, 280, 300]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Interpolate initial guess to new n_points
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h * dx != 1:
            h_initial = h_initial * (1 / (sum_h * dx))
        
        # Define objective: minimize the maximum overlap
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        # Add more constraints for better optimization
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
        
        # Use a different optimization method for better exploration
        # We'll try SLSQP with more aggressive options and also use differential evolution
        result_slsqp = optimize.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-8, 'maxiter': 200, 'disp': False}
        )
        result_de = optimize.differential_evolution(
            objective_func,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.9,
            disp=False
        )
        
        h_opt_slsqp = result_slsqp.x
        current_c5_slsqp = evaluate_erdos_solution(h_opt_slsqp)
        
        h_opt_de = result_de.x
        current_c5_de = evaluate_erdos_solution(h_opt_de)
        
        # Perturb the solution to avoid local minima
        h_perturbed_slsqp = h_opt_slsqp + np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturbed_slsqp = np.clip(h_perturbed_slsqp, 0, 1)
        current_c5_perturbed_slsqp = evaluate_erdos_solution(h_perturbed_slsqp)
        
        h_perturbed_de = h_opt_de + np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturbed_de = np.clip(h_perturbed_de, 0, 1)
        current_c5_perturbed_de = evaluate_erdos_solution(h_perturbed_de)
        
        # Update best solution
        if current_c5_slsqp < best_c5:
            best_c5 = current_c5_slsqp
            best_h = h_opt_slsqp.copy()
            best_n_points = n_points
        if current_c5_de < best_c5:
            best_c5 = current_c5_de
            best_h = h_opt_de.copy()
            best_n_points = n_points
        if current_c5_perturbed_slsqp < best_c5:
            best_c5 = current_c5_perturbed_slsqp
            best_h = h_perturbed_slsqp.copy()
            best_n_points = n_points
        if current_c5_perturbed_de < best_c5:
            best_c5 = current_c5_perturbed_de
            best_h = h_perturbed_de.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
```
