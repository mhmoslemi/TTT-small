import numpy as np

import time

    np.random.seed(seed)

    # Try different n_points values (smaller for speed)
    n_points_candidates = [42, 60, 80, 100, 120, 140, 160, 180]
    best_c5 = float('inf')
    best_h = initial_h
    
    for n_points in n_points_candidates:
        # Generate a random initial guess around the initial solution
        h = np.random.rand(n_points) * 2 - 1
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        
        # Define the optimization variables
        x = cp.Variable(n_points)
        dx = 2.0 / n_points
        
        # Objective: minimize the maximum overlap integral
        # We use a convex relaxation for the max operation
        overlaps = [cp.multiply(x, 1 - cp.roll(x, k)) for k in range(1, n_points)]
        overlap_integrals = [cp.sum(overlap) * dx for overlap in overlaps]
        max_overlap = cp.Max(overlap_integrals)
        
        # Constraints
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        
        # Problem setup
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        
        # Solve
        start_time = time.time()
        try:
            prob.solve(solver=cp.ECOS, max_iter=1000)
            if prob.status == 'OPTIMAL':
                current_h = x.value
                current_c5 = max_overlap.value
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        except:
            pass
        
        # Check elapsed time
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break
    
    # Final check with the best solution
    dx = 2.0 / len(best_h)
    overlaps = [np.sum(np.multiply(best_h, 1 - np.roll(best_h, k))) * dx for k in range(1, len(best_h))]
    best_c5 = max(overlaps)
    
    return (best_h, best_c5, len(best_h))
