    # Start with a smaller number of points for faster computation
    n_points = 64
    dx = 2.0 / n_points
    time_limit = budget_s
    start_time = time.time()

    # Create initial h_values with a more structured pattern
    h_values = np.zeros(n_points)
    for i in range(n_points):
        # Create a piecewise constant function with alternating regions
        # First region: 0.5
        if i < n_points // 4:
            h_values[i] = 0.5
        # Second region: 0.75
        elif i < n_points // 2:
            h_values[i] = 0.75
        # Third region: 0.25
        elif i < 3 * n_points // 4:
            h_values[i] = 0.25
        # Fourth region: 0.0
        else:
            h_values[i] = 0.0

    # Normalize h_values to ensure the integral is 1
    h_values = h_values * (n_points / 2) / np.sum(h_values)

    def compute_c5(h):
        h = np.array(h)
        shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
        one_minus_h = 1 - h
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h_values)
    best_h = h_values.copy()

    # Define optimization variables and objective
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]

    # Try to minimize the max overlap integral
    def minimize_max_overlap(h_values):
        h_values = np.array(h_values)
        shifted_h = np.concatenate([h_values[-n_points + 1:], h_values], axis=0)
        one_minus_h = 1 - h_values
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Optimization loop with multiple iterations and random restarts
    for _ in range(10):
        # Use Nelder-Mead to find a better solution
        result = optimize.minimize(
            minimize_max_overlap,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)]
        )
        if result.success:
            h_candidate = result.x
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
        else:
            break

        # Add some random perturbation to escape local minima
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()

        # Check time limit
        if time.time() - start_time > time_limit - 5:
            break

    # Final check
    if time.time() - start_time < time_limit:
        # Try a few more random restarts
        for _ in range(5):
            # Randomly initialize h
            h_init = np.random.uniform(0, 1, n_points)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            c5_init = compute_c5(h_init)
            if c5_init < best_c5:
                best_c5 = c5_init
                best_h = h_init.copy()

    return best_h, best_c5, n_points
