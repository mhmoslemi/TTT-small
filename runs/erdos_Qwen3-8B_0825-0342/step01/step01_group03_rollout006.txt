```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points for faster iteration
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Start from a known good initial guess
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate((h, h[:1]))  # To handle full convolution
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to compute the objective (we minimize C5)
    def objective(h_values):
        return compute_c5(h_values)
    
    # Function to compute the constraint: sum(h) * dx == 1
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Function for the optimization
    def optimize_h(h_start):
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        objective = cp.Minimize(compute_c5(h))
        
        # Use SCS solver for faster convergence
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4, max_iter=1000)
        except:
            pass
        
        if prob.status == "optimal":
            return h.value
        else:
            return h_start
    
    # Optimization loop
    best_h = h
    best_c5 = compute_c5(h)
    iterations = 0
    max_iterations = 20
    while time.time() - start_time < budget_s and iterations < max_iterations:
        # Optimize with current h
        h = optimize_h(h)
        
        # Compute C5
        current_c5 = compute_c5(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
        
        iterations += 1
        print(f"Iteration {iterations}, C5 = {best_c5}")
    
    # Refine with more points if time allows
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        dx = 2.0 / n_points
        h = np.copy(initial_h_values[:n_points])
        best_h = optimize_h(h)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
```
