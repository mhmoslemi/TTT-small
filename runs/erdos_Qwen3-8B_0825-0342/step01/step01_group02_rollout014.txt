import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Start with 45 points
    initial_h = initial_h_values()
    h = initial_h.copy()
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    # Optimization strategy: use a gradient descent with adaptive learning rate and local search
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    # Local search with simulated annealing
    def simulated_annealing(h, temp=100.0, cooling_rate=0.99, iterations=1000):
        best_h = h.copy()
        best_c5 = evaluate_erdos_solution(best_h)
        current_h = h.copy()
        current_c5 = best_c5
        for _ in range(iterations):
            if time.time() - start_time > budget_s:
                break
            # Random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed_h = np.clip(current_h + perturbation, 0, 1)
            perturbed_c5 = evaluate_erdos_solution(perturbed_h)
            # Acceptance probability
            if perturbed_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - perturbed_c5) / temp):
                current_h = perturbed_h
                current_c5 = perturbed_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
            temp *= cooling_rate
        return best_h, best_c5
    
    # Start with a refined initial guess
    h = simulated_annealing(h)[0]
    
    # Use a more focused gradient descent after local search
    def objective_func_with_gradient(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5_val = np.max(corr)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for k in range(n_points):
            if corr[k] == c5_val:
                for i in range(n_points):
                    grad[i] += (1 - np.roll(h, k))[i] * (2.0 / n_points)
                    grad[i] -= h[i] * (2.0 / n_points)
        return c5_val, grad.flatten()
    
    # Gradient descent optimization
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Use L-BFGS-B optimizer
        result = optimize.minimize(
            lambda x: objective_func(x),
            h,
            method='L-BFGS-B',
            jac=lambda x: objective_func_with_gradient(x)[1],
            bounds=[(0, 1) for _ in range(n_points)]
        )
        
        h = result.x
        current_c5 = evaluate_erdos_solution(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
