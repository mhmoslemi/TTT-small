    def evaluate_overlap(h):
        h = h.copy()
        h = np.pad(h, (0, len(h) - 1), 'constant')  # To handle full convolution
        corr = correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr * (2.0 / len(h)))

    def optimize_h(h, max_iter=100, step_size=0.01):
        h = h.copy()
        for _ in range(max_iter):
            # Compute objective function and gradients
            obj = evaluate_overlap(h)
            # Small perturbation to estimate gradient
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_perturbed = h.copy()
                h_perturbed[i] += 1e-5
                obj_perturbed = evaluate_overlap(h_perturbed)
                grad[i] = (obj_perturbed - obj) / 1e-5
            # Update h in the direction of descent
            h_new = h - step_size * grad
            # Clamp h to [0, 1]
            h_new = np.clip(h_new, 0, 1)
            # Normalize h to maintain integral constraint
            h_new = h_new / np.sum(h_new) * (len(h) / 2)
            # Check if improvement
            if evaluate_overlap(h_new) < obj:
                h = h_new
            else:
                break
        return h

    h = initial_h_values.copy()
    h = h / np.sum(h) * (len(h) / 2)
    h = optimize_h(h)

    c5 = evaluate_overlap(h)
    best_c5 = c5
    best_h = h.copy()

    for _ in range(max_attempts):
