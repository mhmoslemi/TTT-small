import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    # Set up parameters
    n_points = 100  # Using 100 points for a good balance between speed and accuracy
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values.npy' in locals() else np.zeros(n_points)
    
    # Use the initial construction if available
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def objective(h_values):
        h = h_values.reshape(-1, 1)
        # Compute the overlaps for all lags
        overlaps = []
        for k in range(1, n_points):
            shifted_h = np.roll(h, shift=k, axis=0)
            overlap = h * (1 - shifted_h)
            overlaps.append(np.sum(overlap * dx))
        return np.max(overlaps)
    
    # Define bounds for the optimization
    bounds = Bounds(0, 1)
    
    # Set up the optimization
    start_time = time.time()
    result = minimize(
        objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure the integral constraint is satisfied
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h, best_c5, n_points)
