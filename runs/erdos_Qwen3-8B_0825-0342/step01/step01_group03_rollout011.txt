```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess with 42 samples
    n_points_initial = 42
    h_initial = initial_h_values.copy()
    h_initial = np.round(h_initial, 6)  # Ensure numerical stability
    
    # Let's try a few different resolutions to find a good balance between accuracy and speed
    # We'll prioritize smaller numbers of samples (less than 1k)
    n_points_list = [64, 128, 256, 512]
    
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(h_initial)
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)
        
        # Start with the initial guess and perturb it
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to sum to n_points / 2
        
        # Define the variables
        h_var = cp.Variable(n_points, nonnegativity=True)
        h_var = cp.Variable(n_points, nonnegativity=True)
        
        # Objective: minimize the maximum overlap
        # We'll use a big-M approach to convert the max into a linear constraint
        M = 1.0
        overlap = cp.Variable(1)
        
        # Define constraints for the overlap
        for k in range(1, n_points):
            if k > n_points - k:
                break
            # Integral of h(x)*(1 - h(x + k)) dx
            # Handle wrap-around with periodic extension
            h_shifted = np.roll(h_var, k)
            overlap_k = cp.sum(h_var * (1 - h_shifted)) * dx
            overlap = cp.Minimize(overlap_k)
            M = cp.maximum(M, overlap_k)
        
        # Objective: minimize the maximum overlap
        objective = cp.Minimize(M)
        
        # Constraints: sum(h) = n_points / 2, 0 <= h <= 1
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, verbose=False, max_iter=100)
            if result is not None and result < best_c5:
                best_h = h_var.value
                best_c5 = result
        except:
            pass
        
        # Evaluate the current solution
        current_c5 = evaluate_erdos_solution(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    return best_h, best_c5, len(best_h)
```
