import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from functools import lru_cache

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(signal.correlate(h, 1 - h, mode="full")) * (2 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    sum_h = n_points / 2.0
    
    # Initial guess
    h_initial = initial_h_values[:n_points].copy()
    h_initial = np.clip(h_initial, 0, 1)
    
    # Objective function to minimize C5
    def objective(h):
        h = np.clip(h, 0, 1)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Constraints: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - sum_h
    
    # Optimization using scipy.optimize
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint}],
        tol=1e-6
    )
    
    # Extract the best solution found
    h_values = result.x
    c5_bound = result.fun
    
    # Check if we can refine further
    if time.time() - start_time < budget_s - 1:
        # Try with more points
        n_points += 50
        dx = 2.0 / n_points
        sum_h = n_points / 2.0
        h_initial = np.linspace(0.5, 0.5, n_points)
        result = optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            tol=1e-6
        )
        h_values = result.x
        c5_bound = result.fun
    
    # Ensure result is within time limit
    if time.time() - start_time > budget_s:
        return h_values, c5_bound, n_points
    
    return h_values, c5_bound, n_points
