```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a higher-resolution initial guess
    initial_h = initial_h_values()
    n_initial = len(initial_h)
    dx_initial = 2.0 / n_initial
    sum_initial = np.sum(initial_h)
    if sum_initial * dx_initial != 1:
        initial_h = initial_h * (1 / (sum_initial * dx_initial))
    
    # Try increasing the resolution in a more strategic way
    n_points_list = [n_initial * 2, n_initial * 3, n_initial * 4, n_initial * 5]
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_initial
    elapsed_time = 0
    start_time = time.time()
    
    # Optimize with increasing resolution
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Interpolate initial guess to new resolution
        x_old = np.linspace(0, 2, n_initial)
        x_new = np.linspace(0, 2, n_points)
        h_new = np.interp(x_new, x_old, initial_h)
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if sum_h * (2.0 / n_points) != 1:
            h_new = h_new * (1 / (sum_h * (2.0 / n_points)))
        
        # Define objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Use SLSQP with bounds and constraints
        constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            objective_func,
            h_new,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Optional: Add perturbations
        h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
```
