```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random
import matplotlib.pyplot as plt

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess based on known optimal patterns
    # Create a step function with alternating regions of high and low values
    n_points = 300
    h = np.zeros(n_points)
    for i in range(n_points):
        # Create a pattern that alternates between high and low values
        # Ensure the total integral is 1 over [0, 2]
        if i % 3 == 0:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points_list = [180, 200, 220, 240, 260, 280, 300]
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = len(best_h)
    elapsed_time = 0
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Create a better initial guess based on known optimal patterns
        # Construct a step function with alternating regions of high and low values
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            # Create a pattern with more varied structure
            # Ensure the total integral is 1 over [0, 2]
            if i % 5 < 2:
                h_initial[i] = 1.0
            elif i % 5 < 4:
                h_initial[i] = 0.5
            else:
                h_initial[i] = 0.0
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h * (2.0 / n_points) != 1:
            h_initial = h_initial * (1 / (sum_h * (2.0 / n_points)))
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Use SLSQP with bounds and constraints
        constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Optional: Add perturbations
        h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
```
