```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 120
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define a more diverse initial guess
    def generate_diverse_initial_guesses(n_guesses=5):
        guesses = []
        for _ in range(n_guesses):
            h = np.zeros(n_points)
            # Construct diverse initial guesses with different periodic patterns
            for i in range(n_points):
                if i % 5 == 0:
                    h[i] = 1.0
                elif i % 5 == 1:
                    h[i] = 0.5
                elif i % 5 == 2:
                    h[i] = 0.25
                elif i % 5 == 3:
                    h[i] = 0.75
                else:
                    h[i] = 0.0
            # Add offset for diversity
            offset = np.random.uniform(-0.1, 0.1, size=n_points)
            h += offset
            h = np.clip(h, 0.0, 1.0)
            h = h * (1 / (np.sum(h) * 2.0 / n_points))
            guesses.append(h)
        return guesses
    
    # Use a genetic algorithm for global search
    from scipy.optimize import differential_evolution
    
    def genetic_optimization():
        nonlocal best_h, best_c5
        guesses = generate_diverse_initial_guesses(n_guesses=5)
        result = differential_evolution(
            objective_func,
            bounds,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination='blend',
            tol=1e-6,
            polish=True,
            seed=seed,
            strategy='best1bin',
            maxiter=50
        )
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
        
        # Local refinement with SLSQP
        result = optimize.minimize(
            objective_func,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Perform the optimization
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        genetic_optimization()
        
        # Local refinement with perturbations
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
        
        result = optimize.minimize(
            objective_func,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
```
