import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define the initial solution with 300 samples
    n_points_initial = 300
    h_initial = initial_h_values()
    h_initial = np.interp(np.linspace(0, 2, n_points_initial), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h * (2.0 / n_points_initial) != 1:
        h_initial = h_initial * (1 / (sum_h * (2.0 / n_points_initial)))
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points_initial
    elapsed_time = 0
    start_time = time.time()
    
    # Try different n_points in a refined way
    n_points_list = [150, 200, 250, 300, 350, 400]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Interpolate the current best solution to the new n_points
        x_current = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points)
        h_interpolated = np.interp(x_new, x_current, best_h)
        h_interpolated = np.clip(h_interpolated, 0, 1)
        sum_h = np.sum(h_interpolated)
        if sum_h * (2.0 / n_points) != 1:
            h_interpolated = h_interpolated * (1 / (sum_h * (2.0 / n_points)))
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Use SLSQP with bounds and constraints
        constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Initial optimization
        result = optimize.minimize(
            objective_func,
            h_interpolated,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Optional: Add random perturbation for exploration
        h_perturbed = h_opt + np.random.uniform(-0.03, 0.03, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_h = np.sum(h_perturbed)
        if sum_h * (2.0 / n_points) != 1:
            h_perturbed = h_perturbed * (1 / (sum_h * (2.0 / n_points)))
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
