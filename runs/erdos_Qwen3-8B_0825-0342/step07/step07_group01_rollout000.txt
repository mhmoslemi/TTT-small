import numpy as np
import scipy.optimize as optimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from the current construction
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    # Define a helper function to compute the overlap
    def compute_overlap(h):
        n_points = len(h)
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Define a function to optimize
    def optimize_overlap(n_points, h_initial):
        h = np.zeros(n_points)
        h[:] = h_initial
        sum_h = np.sum(h)
        if sum_h * (2.0 / n_points) != 1:
            h = h * (1 / (sum_h * (2.0 / n_points)))
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Use SLSQP with bounds and constraints
        constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            objective_func,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        # Check if we have a better solution
        if result.success:
            return result.x, compute_overlap(result.x)
        else:
            return h, compute_overlap(h)
    
    # Try different numbers of points
    n_points_list = [180, 200, 220, 240, 260, 280, 300]
    best_h = initial_h_values()
    best_c5 = compute_overlap(best_h)
    best_n_points = len(best_h)
    elapsed_time = 0
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Interpolate initial guess to the new number of points
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h * (2.0 / n_points) != 1:
            h_initial = h_initial * (1 / (sum_h * (2.0 / n_points)))
        
        # Optimize
        h_opt, current_c5 = optimize_overlap(n_points, h_initial)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Add random perturbations
        h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_h_perturbed = np.sum(h_perturbed)
        if sum_h_perturbed * (2.0 / n_points) != 1:
            h_perturbed = h_perturbed * (1 / (sum_h_perturbed * (2.0 / n_points)))
        current_c5_perturbed = compute_overlap(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = compute_overlap(best_h)
    
    return best_h, final_c5, best_n_points
