import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points_list = [100, 120, 140, 160, 180, 200, 220, 240, 260, 280, 300]
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = len(best_h)
    elapsed_time = 0
    
    start_time = time.time()
    
    # Initial guess from the given initial_h_values
    initial_guess = initial_h_values()
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_guess)
    if sum_h * dx != 1:
        initial_guess = initial_guess * (1 / (sum_h * dx))
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to optimize the step function
    def optimize_step_function(n_points, initial_guess, budget_s):
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
        
        # Initial guess
        h_initial = initial_guess.copy()
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_guess)), initial_guess)
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h * dx != 1:
            h_initial = h_initial * (1 / (sum_h * dx))
        
        # Use bounded optimization with SLSQP
        result = optimize.minimize(
            evaluate_overlap,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success:
            h_opt = result.x
            c5 = evaluate_overlap(h_opt)
            return h_opt, c5
        else:
            return h_initial, evaluate_overlap(h_initial)
    
    # Try different values of n_points
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Compute dx and initial guess
        dx = 2.0 / n_points
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_guess)), initial_guess)
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h * dx != 1:
            h_initial = h_initial * (1 / (sum_h * dx))
        
        # Optimize
        h_opt, c5 = optimize_step_function(n_points, h_initial, budget_s)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Perturb the solution for further improvement
        h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        c5_perturbed = evaluate_overlap(h_perturbed)
        
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
