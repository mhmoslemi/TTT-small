```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 120
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_time = 0
    start_time = time.time()
    
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use a more refined and adaptive optimization approach
    def optimize_solution(h_initial):
        nonlocal best_h, best_c5
        # Local optimization with SLSQP
        result = optimize.minimize(
            evaluate,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        h_opt = result.x
        c5 = evaluate_erdos_solution(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
    
    def perturb_solution(h):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
        return h_perturbed
    
    # Run multiple local optimizations with perturbations
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        # Perturb and optimize
        h_perturbed = perturb_solution(best_h)
        optimize_solution(h_perturbed)
        
        # Also try a few random starting points
        for _ in range(3):
            h_rand = np.random.uniform(0.0, 1.0, size=n_points)
            h_rand = h_rand * (1 / (np.sum(h_rand) * (2.0 / n_points)))
            optimize_solution(h_rand)
    
    # Final optimization with more refined steps
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Local refinement
        result = optimize.minimize(
            evaluate,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
```
