```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Step 1: Use initial construction as a starting point
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    # Step 2: Try a larger n_points for better resolution
    n_points_list = [200, 240, 280, 320, 360, 400]
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    elapsed_time = 0
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Interpolate initial solution to new n_points
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if sum_h * dx != 1:
            h_new = h_new * (1 / (sum_h * dx))
        
        # Step 3: Define objective function and constraints
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        # Constraints: volume constraint and bounds
        constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Step 4: Use more advanced optimization with different tolerances
        result = optimize.minimize(
            objective_func,
            h_new,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 200, 'eps': 1e-8}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Step 5: Perturb solution to explore nearby regions
        h_perturbed = h_opt + np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Step 6: Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
```
