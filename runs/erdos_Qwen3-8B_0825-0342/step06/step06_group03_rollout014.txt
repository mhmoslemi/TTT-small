import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 120
    best_h = initial_h_values()
    best_h = np.resize(best_h, n_points)
    dx = 2.0 / n_points
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Function to evaluate the error
    def evaluate_error(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to evaluate the constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the objective function to minimize the maximum overlap
    def objective_func(h_flat):
        return evaluate_error(h_flat)
    
    # Use sequential quadratic programming for local optimization
    def optimize_with_sqp(h_initial):
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x, result.fun
    
    # Multi-start optimization with perturbation and smoothing
    def multi_start_optimization():
        population_size = 50
        best_error = float('inf')
        best_solution = np.zeros(n_points)
        
        for _ in range(population_size):
            if time.time() - start_time > budget_s:
                break
            
            # Generate a random initial guess
            h_initial = np.random.uniform(0, 1, size=n_points)
            h_initial = h_initial * (1 / (np.sum(h_initial) * dx))
            
            # Optimize
            h_opt, error = optimize_with_sqp(h_initial)
            
            # Update best solution if improved
            if error < best_error:
                best_error = error
                best_solution = h_opt.copy()
        
        return best_solution, best_error
    
    # Initialize the best solution and best C5
    best_solution = best_h
    best_c5 = best_c5
    
    # Run the multi-start optimization
    start_time = time.time()
    best_solution, best_c5 = multi_start_optimization()
    
    # Iterative refinement with perturbation and smoothing
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the best solution
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = best_solution + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
        
        # Optimize the perturbed solution
        h_opt, error = optimize_with_sqp(h_perturbed)
        
        # Update best solution if improved
        if error < best_c5:
            best_c5 = error
            best_solution = h_opt.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_solution)
    return best_solution, final_c5, n_points
