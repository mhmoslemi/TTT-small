import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points_list = [60, 80, 100, 120, 140, 160, 180, 200]
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = 45
    elapsed_time = 0
    
    start_time = time.time()
    
    # Start from the initial construction and refine
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with a better guess based on previous best
        if best_n_points < n_points:
            # Interpolate previous best to current n_points
            x_prev = np.linspace(0, 2, best_n_points)
            x_curr = np.linspace(0, 2, n_points)
            h_interpolated = np.interp(x_curr, x_prev, best_h)
            h_interpolated = np.clip(h_interpolated, 0, 1)
            sum_h = np.sum(h_interpolated)
            if sum_h * (2.0 / n_points) != 1:
                h_interpolated = h_interpolated * (1 / (sum_h * (2.0 / n_points)))
        else:
            # Use initial construction for smaller n_points
            h_interpolated = initial_h_values()
            h_interpolated = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_interpolated)), h_interpolated)
            h_interpolated = np.clip(h_interpolated, 0, 1)
            sum_h = np.sum(h_interpolated)
            if sum_h * (2.0 / n_points) != 1:
                h_interpolated = h_interpolated * (1 / (sum_h * (2.0 / n_points)))
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Use sequential least squares programming with bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
        
        result = optimize.minimize(
            objective_func,
            h_interpolated,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Optional: Add perturbations
        h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
