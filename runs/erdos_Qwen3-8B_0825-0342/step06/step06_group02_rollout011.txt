import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define a range of n_points to explore
    n_points_list = [80, 100, 120, 140, 160]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 120
    
    start_time = time.time()
    
    # Start with a higher resolution for better optimization
    n_points = 120
    if time.time() - start_time > budget_s:
        return initial_h_values(), evaluate_erdos_solution(initial_h_values()), n_points
    
    # Initial guess
    initial_h = initial_h_values()
    initial_h = np.resize(initial_h, n_points)
    initial_h = initial_h * (n_points / 120)  # resize and scale to maintain integral
    
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use a more sophisticated optimization strategy
    def optimize_h(h_initial):
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        # Refinement with simulated annealing
        from scipy.optimize import basinhopping
        
        def basinhopping_objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        def basinhopping_constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) * (2.0 / n_points) - 1.0
        
        result_basin = basinhopping(
            basinhopping_objective,
            result.x,
            T=1.0,
            stepsize=0.1,
            niter=30,
            minimizer_kwargs={
                'method': 'SLSQP',
                'bounds': bounds,
                'constraints': {'type': 'eq', 'fun': basinhopping_constraint},
                'options': {'ftol': 1e-8, 'maxiter': 100}
            }
        )
        
        h_basin = result_basin.x
        current_c5 = evaluate_erdos_solution(h_basin)
        return h_basin, current_c5
    
    # Main optimization loop
    h_opt, current_c5 = optimize_h(initial_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_opt
        best_n_points = n_points
    
    # Try different starting points and n_points
    for n_points in [80, 100, 140, 160]:
        if time.time() - start_time > budget_s:
            break
        
        # Generate different initial guesses
        initial_h = np.random.uniform(0, 1, n_points)
        initial_h = initial_h * (1 / (np.sum(initial_h) * 2.0 / n_points))
        
        def optimize_h_new(h_initial):
            result = optimize.minimize(
                objective_func,
                h_initial,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_func},
                options={'ftol': 1e-8, 'maxiter': 200}
            )
            
            # Refinement with simulated annealing
            def basinhopping_objective(h_flat):
                h = h_flat.reshape(n_points)
                corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
                return np.max(corr)
            
            def basinhopping_constraint(h_flat):
                h = h_flat.reshape(n_points)
                return np.sum(h) * (2.0 / n_points) - 1.0
            
            result_basin = basinhopping(
                basinhopping_objective,
                result.x,
                T=1.0,
                stepsize=0.1,
                niter=30,
                minimizer_kwargs={
                    'method': 'SLSQP',
                    'bounds': bounds,
                    'constraints': {'type': 'eq', 'fun': basinhopping_constraint},
                    'options': {'ftol': 1e-8, 'maxiter': 100}

            h_basin = result_basin.x
            current_c5 = evaluate_erdos_solution(h_basin)
            return h_basin, current_c5
        
        h_opt, current_c5 = optimize_h_new(initial_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt
            best_n_points = n_points
    
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, best_n_points
