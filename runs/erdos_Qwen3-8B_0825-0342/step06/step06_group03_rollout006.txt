import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a larger number of samples
    n_points = 120
    best_h = initial_h_values()
    best_h = np.resize(best_h, n_points)
    dx = 2.0 / n_points
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Function to evaluate the error
    def evaluate_error(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to evaluate the constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to perform local optimization
    def local_optimize(initial_guess):
        result = optimize.minimize(
            evaluate_error,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x
    
    # Function to perform simulated annealing with local improvements
    def simulated_annealing():
        start_time = time.time()
        temperature = 1.0
        cooling_rate = 0.995
        iterations = 0
        max_iterations = 10000
        best_h = best_h.copy()
        best_c5 = evaluate_erdos_solution(best_h)
        
        # Initial guess
        h = np.random.uniform(0, 1, n_points)
        h = h * (1 / (np.sum(h) * dx))
        
        while iterations < max_iterations and time.time() - start_time < budget_s:
            iterations += 1
            # Perturb the solution
            perturbation = np.random.normal(0, 0.05, n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
            
            # Evaluate the error
            current_error = evaluate_error(h_perturbed)
            if current_error < best_c5:
                best_c5 = current_error
                best_h = h_perturbed.copy()
            
            # Local optimization
            h_local = local_optimize(h_perturbed)
            local_error = evaluate_error(h_local)
            if local_error < best_c5:
                best_c5 = local_error
                best_h = h_local.copy()
            
            # Acceptance criteria
            if current_error < best_c5:
                h = h_perturbed
            else:
                delta = current_error - best_c5
                if np.random.rand() < np.exp(-delta / temperature):
                    h = h_perturbed
            
            temperature *= cooling_rate
        
        return best_h, best_c5
    
    # Run simulated annealing
    best_h, best_c5 = simulated_annealing()
    
    # Final refinement with perturbations and local optimization
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the solution
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
        
        # Optimize the perturbed solution
        h_opt = local_optimize(h_perturbed)
        current_c5 = evaluate_error(h_opt)
        
        # Check if the new solution is better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
