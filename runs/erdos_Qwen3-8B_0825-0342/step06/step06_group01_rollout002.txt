import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from the initial construction
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if sum_h * dx != 1:
        initial_h = initial_h * (1 / (sum_h * dx))
    
    # Try a finer grid of n_points
    n_points_list = [100, 120, 140, 160, 180, 200]
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    elapsed_time = 0
    start_time = time.time()
    
    # Function to evaluate and return C5
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to enforce integral constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Optimization function with bounds and constraint
    def optimize_h(initial_guess, n_points):
        h_initial = initial_guess.copy()
        sum_h = np.sum(h_initial)
        if sum_h * dx != 1:
            h_initial = h_initial * (1 / (sum_h * dx))
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_func}]
        
        result = optimize.minimize(
            evaluate,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 150}
        )
        
        # Perturb and check if better
        h_perturbed = result.x + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_h_perturbed = np.sum(h_perturbed)
        if sum_h_perturbed * dx != 1:
            h_perturbed = h_perturbed * (1 / (sum_h_perturbed * dx))
        
        current_c5 = evaluate(result.x)
        perturbed_c5 = evaluate(h_perturbed)
        
        if perturbed_c5 < current_c5:
            return h_perturbed, perturbed_c5
        else:
            return result.x, current_c5
    
    # Try increasing n_points
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Interpolate initial guess to new n_points
        interpolated_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        interpolated_h = np.clip(interpolated_h, 0, 1)
        sum_h = np.sum(interpolated_h)
        if sum_h * dx != 1:
            interpolated_h = interpolated_h * (1 / (sum_h * dx))
        
        h_opt, current_c5 = optimize_h(interpolated_h, n_points)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    
    return best_h, final_c5, best_n_points
