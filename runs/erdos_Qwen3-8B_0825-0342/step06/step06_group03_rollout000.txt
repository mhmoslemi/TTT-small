import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a higher number of samples for better accuracy
    n_points = 120
    best_h = initial_h_values()
    best_h = np.resize(best_h, n_points)
    dx = 2.0 / n_points
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Function to evaluate the error
    def evaluate_error(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to evaluate the constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define a more sophisticated starting point based on a pattern
    def improved_initial_h_values():
        pattern = [0.5] * 10 + [0.75] * 10 + [1.0] * 10 + [0.75] * 10 + [0.5] * 10
        return np.resize(pattern, n_points)
    
    best_h = improved_initial_h_values()
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use a more sophisticated optimization approach with gradient descent and
    # simulated annealing to avoid local minima
    def simulated_annealing():
        T = 1.0
        Tmin = 1e-5
        alpha = 0.995
        best_h = best_h.copy()
        best_c5 = best_c5
        start_time = time.time()
        
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
            
            # Perturb the current solution
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
            
            # Evaluate the perturbed solution
            current_c5 = evaluate_error(h_perturbed)
            
            # Accept the new solution based on simulated annealing
            delta = current_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                best_h = h_perturbed
                best_c5 = current_c5
            
            T *= alpha
        
        return best_h, best_c5
    
    # Perform simulated annealing
    best_h, best_c5 = simulated_annealing()
    
    # Iterative refinement with local optimization
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        
        # Local optimization using SLSQP
        result = optimize.minimize(
            evaluate_error,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_error(h_opt)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
