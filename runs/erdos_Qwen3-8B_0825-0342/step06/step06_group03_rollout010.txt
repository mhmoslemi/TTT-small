import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Use a slightly larger number of samples for better accuracy
    n_points = 120
    best_h = initial_h_values()
    best_h = np.resize(best_h, n_points)
    dx = 2.0 / n_points
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Function to evaluate the error
    def evaluate_error(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to evaluate the constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize with SLSQP for global convergence
    def optimize_h(h_initial):
        result = optimize.minimize(
            evaluate_error,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        return result.x
    
    # Refine with multiple starting points
    start_time = time.time()
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    # Try multiple perturbations around the initial solution
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the initial solution
        h_perturbed = best_h + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
        
        # Optimize the perturbed solution
        h_opt = optimize_h(h_perturbed)
        c5 = evaluate_error(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
    
    # Try a more robust initial guess using a random walk strategy
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        
        # Create a random walk initial guess
        h_random = np.random.uniform(0, 1, n_points)
        h_random = h_random * (1 / (np.sum(h_random) * dx))
        
        # Optimize the random walk solution
        h_opt = optimize_h(h_random)
        c5 = evaluate_error(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
