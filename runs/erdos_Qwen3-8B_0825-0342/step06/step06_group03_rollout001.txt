import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 120
    dx = 2.0 / n_points
    
    # Start from the initial construction
    h_initial = initial_h_values()
    h_initial = np.resize(h_initial, n_points)
    h_initial = h_initial * (1 / (np.sum(h_initial) * dx))
    best_c5 = evaluate_erdos_solution(h_initial)
    best_h = h_initial.copy()
    
    # Function to evaluate error
    def evaluate_error(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to evaluate constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform simulated annealing for global optimization
    def simulated_annealing():
        temp = 1.0
        temp_min = 0.01
        cooling_rate = 0.995
        best_h = copy.deepcopy(best_h)
        best_c5 = float('inf')
        
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
            
            # Generate a perturbed solution
            h_perturbed = best_h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    h_perturbed[i] += np.random.uniform(-0.1, 0.1)
                    h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
            h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
            
            # Compute error
            current_c5 = evaluate_error(h_perturbed)
            
            # Acceptance probability
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed.copy()
            else:
                delta = current_c5 - best_c5
                if np.random.rand() < np.exp(-delta / temp):
                    best_h = h_perturbed.copy()
                    best_c5 = current_c5
            
            temp *= cooling_rate
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = simulated_annealing()
    
    # Iterative refinement with local optimization
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        
        # Use SLSQP for local optimization
        result = optimize.minimize(
            evaluate_error,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_error(h_opt)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
