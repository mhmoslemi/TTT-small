import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a larger number of samples for better accuracy
    n_points = 120
    best_h = initial_h_values()
    best_h = np.resize(best_h, n_points)
    dx = 2.0 / n_points
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Function to evaluate the error
    def evaluate_error(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to evaluate the constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define a more efficient and refined optimization approach
    def optimize_solution(start_h):
        result = optimize.minimize(
            evaluate_error,
            start_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        return result.x, result.fun
    
    # Define a more sophisticated heuristic to improve the solution
    def refine_solution(h, n_refine=5):
        for _ in range(n_refine):
            if time.time() - start_time > budget_s:
                break
            # Perturb the solution
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
            
            # Optimize perturbed solution
            h_opt, c5_opt = optimize_solution(h_perturbed)
            
            # Update best solution if improved
            if c5_opt < evaluate_error(h):
                h = h_opt.copy()
        return h
    
    # Main optimization loop with improved heuristics
    start_time = time.time()
    best_h, best_c5 = initial_h_values(), evaluate_erdos_solution(best_h)
    
    # Refine the initial solution
    best_h = refine_solution(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try additional refinements with different strategies
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        
        # Generate a new random starting point
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (1 / (np.sum(random_h) * dx))
        
        # Optimize the random starting point
        h_opt, c5_opt = optimize_solution(random_h)
        
        # Update best solution if improved
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Final refinement
    best_h = refine_solution(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
