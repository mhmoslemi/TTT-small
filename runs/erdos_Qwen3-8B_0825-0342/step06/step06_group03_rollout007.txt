import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Use a larger number of samples for better accuracy
    n_points = 120
    best_h = initial_h_values()
    best_h = np.resize(best_h, n_points)
    dx = 2.0 / n_points
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Function to evaluate the error
    def evaluate_error(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to evaluate the constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a combination of genetic algorithm and local optimization
    def genetic_algorithm():
        population_size = 50
        mutation_rate = 0.1
        generations = 100
        best_population = []
        best_error = float('inf')
        
        # Initialize population
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * (1 / (np.sum(population, axis=1) * dx))[:, np.newaxis]
        
        for _ in range(generations):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate all population members
            errors = np.array([evaluate_error(ind) for ind in population])
            current_best_idx = np.argmin(errors)
            current_best = population[current_best_idx]
            current_best_error = errors[current_best_idx]
            
            # Update best solution
            if current_best_error < best_error:
                best_error = current_best_error
                best_population = current_best
            
            # Select parents
            parent_indices = np.argsort(errors)[:population_size // 2]
            parents = population[parent_indices]
            
            # Generate offspring
            offspring = []
            for parent1 in parents:
                for parent2 in parents:
                    child = np.zeros(n_points)
                    for i in range(n_points):
                        if np.random.rand() < 0.5:
                            child[i] = parent1[i]
                        else:
                            child[i] = parent2[i]
                    # Apply mutation
                    for i in range(n_points):
                        if np.random.rand() < mutation_rate:
                            child[i] = np.random.uniform(0, 1)
                    child = child * (1 / (np.sum(child) * dx))
                    offspring.append(child)
            
            # Replace some population members
            population = np.vstack((parents, offspring))
            population = population[:population_size]
        
        return best_population, best_error
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    
    # Iterative refinement with perturbation and smoothing
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb best_h
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
        
        # Optimize perturbed solution
        result = optimize.minimize(
            evaluate_error,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_error(h_opt)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
