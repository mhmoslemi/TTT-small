import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Use a larger number of samples for better accuracy
    n_points = 120
    best_h = initial_h_values()
    best_h = np.resize(best_h, n_points)
    dx = 2.0 / n_points
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to evaluate the error
    def evaluate_error(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to evaluate the constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    # Define a function to generate an initial guess based on a pattern
    def generate_patterned_initial_guess(n_points):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            segment = i % 5
            if segment == 0:
                pattern[i] = 0.75
            elif segment == 1:
                pattern[i] = 0.5
            elif segment == 2:
                pattern[i] = 0.25
            elif segment == 3:
                pattern[i] = 0.5
            else:
                pattern[i] = 0.75
        pattern = pattern * (1 / (np.sum(pattern) * dx))
        return pattern
    
    # Initial guess
    best_h = generate_patterned_initial_guess(n_points)
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    # Use a hybrid approach: simulate annealing and local optimization
    def simulated_annealing():
        T = 1.0
        T_min = 1e-5
        cooling_rate = 0.95
        num_iterations = 1000
        best_h_sa = best_h.copy()
        best_c5_sa = best_c5
        
        for _ in range(num_iterations):
            if time.time() - start_time > budget_s:
                break
            # Generate a perturbed version
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = best_h_sa + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * dx))
            current_error = evaluate_error(h_perturbed)
            
            # Acceptance probability
            delta = current_error - best_c5_sa
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                best_h_sa = h_perturbed
                best_c5_sa = current_error
        
            T *= cooling_rate
        
        return best_h_sa, best_c5_sa
    
    # Run simulated annealing
    best_h, best_c5 = simulated_annealing()
    
    # Perform local optimization with SLSQP
    result = optimize.minimize(
        evaluate_error,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func},
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = evaluate_error(best_h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
