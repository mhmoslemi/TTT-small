import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 120
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    # Function to compute the integral constraint
    def compute_integral(h):
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define a function for optimization
    def optimize_h(h_initial):
        result = optimize.minimize(
            compute_overlap,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': compute_integral},
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        return result
    
    # Define an improved initial guess with more structure
    def improved_initial_h_values():
        n_points = 120
        h = np.zeros(n_points)
        # Introduce more structured peaks and valleys
        for i in range(n_points):
            if i % 6 == 0:
                h[i] = 1.0
            elif i % 6 == 1:
                h[i] = 0.6
            elif i % 6 == 2:
                h[i] = 0.3
            elif i % 6 == 3:
                h[i] = 0.5
            elif i % 6 == 4:
                h[i] = 0.7
            else:
                h[i] = 0.0
        # Normalize the integral to 1
        h = h * (1 / (np.sum(h) * 2.0 / n_points))
        return h
    
    # Use a variety of starting points to escape local optima
    initial_guesses = [
        initial_h_values(),
        improved_initial_h_values(),
        np.random.uniform(0, 1, n_points),
        np.zeros(n_points),
        np.ones(n_points),
    ]
    
    # Normalize the random and zero-initialized guesses
    for i in range(len(initial_guesses)):
        if initial_guesses[i].sum() != 0:
            initial_guesses[i] = initial_guesses[i] * (1 / (np.sum(initial_guesses[i]) * 2.0 / n_points))
    
    # Perform a series of optimizations starting from different points
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = optimize_h(guess)
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Refinement step with local optimization
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Perturb the best_h
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
        
        # Local optimization
        result = optimize.minimize(
            compute_overlap,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': compute_integral},
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        h_opt = result.x
        current_c5 = compute_overlap(h_opt)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = compute_overlap(best_h)
    return best_h, final_c5, n_points
