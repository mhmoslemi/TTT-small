```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points_list = [45, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = 45
    elapsed_time = 0
    
    start_time = time.time()
    
    def optimize_for_n_points(n_points):
        dx = 2.0 / n_points
        # Start with a refined version of the initial h_values
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values())), initial_h_values())
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h * dx != 1:
            h_initial = h_initial * (1 / (sum_h * dx))
        
        # Define the objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        # Use SLSQP with bounds and constraints
        constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        # Perturb
        h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        # Return best of the two
        if current_c5 < current_c5_perturbed:
            return h_opt, current_c5
        else:
            return h_perturbed, current_c5_perturbed
    
    # Try different n_points
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        if n_points < len(best_h):
            continue  # Skip smaller points
        
        h_opt, current_c5 = optimize_for_n_points(n_points)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
    
    # Also try a few random perturbations on the best current solution
    h_perturbed = best_h + np.random.uniform(-0.05, 0.05, size=len(best_h))
    h_perturbed = np.clip(h_perturbed, 0, 1)
    current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
    
    if current_c5_perturbed < best_c5:
        best_c5 = current_c5_perturbed
        best_h = h_perturbed.copy()
    
    return best_h, best_c5, best_n_points
```
