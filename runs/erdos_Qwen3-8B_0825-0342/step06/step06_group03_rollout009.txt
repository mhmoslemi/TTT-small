import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a moderate number of samples
    n_points = 80
    best_h = initial_h_values()
    best_h = np.resize(best_h, n_points)
    dx = 2.0 / n_points
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Function to evaluate the error
    def evaluate_error(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to evaluate the constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define a custom penalty function for the constraint
    def penalty_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.abs(np.sum(h) * dx - 1)
    
    # Use a Bayesian optimization approach with Scipy
    def bayesian_optimization():
        from scipy.optimize import minimize, Bounds
        from skopt import BayesianOptimization, GaussianProcessRegressor
        from skopt.learning import GaussianProcessRegressor
        from skopt import Optimizer
        from skopt.space import Space, Real
        
        # Define the search space
        space = Space([Real(0, 1, name=f'x{i}') for i in range(n_points)])
        
        # Define the objective function with constraint handling
        def objective_function(x):
            h_flat = x.reshape(n_points)
            # Apply constraint penalty
            penalty = penalty_func(h_flat)
            # Evaluate error
            error = evaluate_error(h_flat)
            return error + 1000 * penalty  # Penalize constraint violations
        
        # Set up the Bayesian optimization
        optimizer = Optimizer(space, base_estimator=GaussianProcessRegressor(
            alpha=1e-5, n_restarts_optimizer=25), n_points=20)
        
        # Run the optimization
        start_time = time.time()
        while time.time() - start_time < budget_s:
            # Get next point to evaluate
            x = optimizer.ask()
            # Evaluate the objective
            result = objective_function(x)
            # Tell the optimizer the result
            optimizer.tell(x, result)
            # Update best solution
            best_x = optimizer.x
            best_error = optimizer.func_vals.min()
            
            # Convert best solution to h_values
            best_h_candidate = best_x.reshape(n_points)
            best_h_candidate = best_h_candidate * (1 / (np.sum(best_h_candidate) * dx))
            current_c5 = evaluate_erdos_solution(best_h_candidate)
            
            # Update best_h and best_c5 if improved
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = best_h_candidate.copy()
        
        return best_h, best_c5
    
    # Run Bayesian optimization
    best_h, best_c5 = bayesian_optimization()
    
    # Iterative refinement with local optimization
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        
        # Apply local optimization to current best solution
        result = optimize.minimize(
            evaluate_error,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_error(h_opt)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
