```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 120
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    # Evaluate initial solution
    print(f"Initial C₅: {best_c5:.6f}")
    
    # Define objective and constraint
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define a local optimizer and constraint handling
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x
    
    # Try different perturbations and local optimization
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the current solution
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
        
        # Local optimization with the perturbed solution
        h_opt = local_optimization(h_perturbed)
        current_c5 = evaluate_erdos_solution(h_opt)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            print(f"Improved C₅: {best_c5:.6f}")
        else:
            print(f"No improvement: {current_c5:.6f}")
    
    # Try using a different starting point (e.g., randomized binary)
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        
        # Construct a random binary initial guess
        h_random = np.random.choice([0.0, 1.0], size=n_points)
        h_random = h_random * (1 / (np.sum(h_random) * (2.0 / n_points)))
        
        # Local optimization
        h_opt = local_optimization(h_random)
        current_c5 = evaluate_erdos_solution(h_opt)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            print(f"Improved C₅: {best_c5:.6f}")
        else:
            print(f"No improvement: {current_c5:.6f}")
    
    # Try using a periodic initial guess
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        
        # Construct a periodic initial guess
        h_periodic = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h_periodic[i] = 1.0
            elif i % 5 == 1:
                h_periodic[i] = 0.5
            elif i % 5 == 2:
                h_periodic[i] = 0.25
            elif i % 5 == 3:
                h_periodic[i] = 0.75
            else:
                h_periodic[i] = 0.0
        h_periodic = h_periodic * (1 / (np.sum(h_periodic) * (2.0 / n_points)))
        
        # Local optimization
        h_opt = local_optimization(h_periodic)
        current_c5 = evaluate_erdos_solution(h_opt)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            print(f"Improved C₅: {best_c5:.6f}")
        else:
            print(f"No improvement: {current_c5:.6f}")
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    print(f"Final C₅: {final_c5:.6f}")
    return best_h, final_c5, n_points
```
