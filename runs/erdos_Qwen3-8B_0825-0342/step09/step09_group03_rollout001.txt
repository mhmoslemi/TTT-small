import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200  # More points for finer optimization
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    # Define the objective and constraint
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define a more advanced optimization strategy with multiple stages
    def multi_stage_optimization(h_initial, time_budget):
        nonlocal best_h, best_c5
        time_remaining = time_budget
        
        # Stage 1: Global optimization with basin hopping
        from scipy.optimize import basinhopping
        
        def basinhopping_objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        def basinhopping_constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) * (2.0 / n_points) - 1.0
        
        result = basinhopping(
            basinhopping_objective,
            h_initial,
            T=1.0,
            stepsize=0.1,
            niter=30,
            minimizer_kwargs={
                'method': 'SLSQP',
                'bounds': bounds,
                'constraints': {'type': 'eq', 'fun': basinhopping_constraint},
                'options': {'ftol': 1e-8, 'maxiter': 100}
            },
            disp=False
        )
        
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
        time_remaining -= time.time() - start_time
        
        # Stage 2: Local optimization with SLSQP
        result = optimize.minimize(
            objective_func,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
        time_remaining -= time.time() - start_time
        
        # Stage 3: Perturbation and refinement
        for _ in range(5):
            if time_remaining <= 0:
                break
            
            # Perturb the solution
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
            
            # Local optimization
            result = optimize.minimize(
                objective_func,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_func},
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            h_opt = result.x
            current_c5 = evaluate_erdos_solution(h_opt)
            
            # Update best solution if improved
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
            
            time_remaining -= time.time() - start_time
        
        return best_h, best_c5
    
    # Run optimization with time budget
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        
        best_h, best_c5 = multi_stage_optimization(best_h, budget_s - (time.time() - start_time))
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
