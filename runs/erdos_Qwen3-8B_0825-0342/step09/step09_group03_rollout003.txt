import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 120
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    # Define objective and constraint
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return corr
    
    # Function to compute the maximum overlap
    def compute_c5(h_flat):
        corr = evaluate_overlap(h_flat)
        return np.max(corr)
    
    # Function to refine a solution with local optimization
    def refine_solution(h_initial):
        nonlocal best_c5
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        h_opt = result.x
        current_c5 = compute_c5(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt
        return h_opt
    
    # Function to perform perturbation and refine
    def perturb_and_refine(h_initial):
        nonlocal best_c5
        # Generate a perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        # Normalize to maintain integral constraint
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
        # Refine the perturbed solution
        h_opt = refine_solution(h_perturbed)
        return h_opt
    
    # Function to use simulated annealing for global optimization
    def simulated_annealing(initial_guess):
        T = 1.0
        T_min = 1e-3
        alpha = 0.995
        current_solution = initial_guess.copy()
        current_c5 = compute_c5(current_solution)
        best_solution = current_solution.copy()
        best_c5 = current_c5
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
            # Generate a new candidate solution
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            candidate = current_solution + perturbation
            candidate = np.clip(candidate, 0.0, 1.0)
            candidate = candidate * (1 / (np.sum(candidate) * (2.0 / n_points)))
            candidate_c5 = compute_c5(candidate)
            # Acceptance probability
            delta = candidate_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_solution = candidate.copy()
                current_c5 = candidate_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = current_solution.copy()
            T *= alpha
        return best_solution
    
    # Function to use a particle swarm optimization (PSO) variant
    def pso_optimization(initial_guess):
        from pyswarm import pso
        def pso_objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        def pso_constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) * (2.0 / n_points) - 1.0
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # PSO optimization with bounds and constraints
        result = pso(pso_objective, bounds, constraints=pso_constraint, n=20, maxiter=50)
        return result[0]
    
    # Main optimization workflow
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Use PSO for global optimization
        h_initial = initial_h_values()
        h_initial = h_initial * (1 / (np.sum(h_initial) * (2.0 / n_points)))
        h_pso = pso_optimization(h_initial)
        # Refine the PSO result
        h_opt = refine_solution(h_pso)
        # Perturb and refine
        h_perturbed = perturb_and_refine(h_opt)
        # Use simulated annealing for further refinement
        h_sa = simulated_annealing(h_perturbed)
        # Update best solution
        current_c5 = compute_c5(h_sa)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_sa.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
