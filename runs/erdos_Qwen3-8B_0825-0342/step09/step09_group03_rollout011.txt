```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Try with different n_points: 120, 150, 180
    n_points_list = [120, 150, 180]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 120
    
    for n_points in n_points_list:
        # Initialize with a good starting point
        h_initial = initial_h_values()
        h_initial = np.resize(h_initial, n_points)  # Resize for current n_points
        h_initial = h_initial * (n_points / 120)  # Adjust to match total integral
        
        # Define objective and constraint
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        def constraint_func(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) * (2.0 / n_points) - 1.0
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use advanced optimization with adaptive strategies and gradient-based methods
        def adaptive_optimization(h_initial):
            nonlocal best_h, best_c5
            # Global optimization with differential evolution
            result = optimize.differential_evolution(
                objective_func,
                bounds,
                args=(),
                strategy='best1bin',
                popsize=15,
                maxiter=50,
                tol=1e-8,
                mutation=(0.5, 1),
                recombination=0.9,
                seed=seed,
                constraints=[{'type': 'eq', 'fun': constraint_func}]
            )
            
            h_opt = result.x
            current_c5 = evaluate_erdos_solution(h_opt)
            
            # Local refinement with SLSQP
            result = optimize.minimize(
                objective_func,
                h_opt,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_func},
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            h_opt = result.x
            current_c5 = evaluate_erdos_solution(h_opt)
            
            return h_opt, current_c5
        
        # Run optimization
        start_time = time.time()
        h_opt, current_c5 = adaptive_optimization(h_initial)
        if time.time() - start_time > budget_s:
            break
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, best_n_points
```
