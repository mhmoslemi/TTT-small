import numpy as np
import scipy.optimize as optimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess: start from the given initial construction
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    # Try different number of points and use a more sophisticated optimization strategy
    n_points_list = [180, 200, 220, 240, 260, 280, 300]
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    elapsed_time = 0
    start_time = time.time()
    
    # Try a different approach: use a genetic algorithm with mutation and crossover
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    def genetic_algorithm():
        population_size = 50
        mutation_rate = 0.1
        crossover_rate = 0.8
        generations = 100
        
        # Initialize population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * (1 / (np.sum(h) * dx)) for h in population]
        
        for generation in range(generations):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate fitness
            fitness = [evaluate_erdos_solution(h) for h in population]
            
            # Select parents
            parents = np.argsort(fitness)[:population_size // 2]
            parents = [population[i] for i in parents]
            
            # Crossover
            new_population = []
            for i in range(population_size // 2):
                if np.random.rand() < crossover_rate:
                    parent1 = parents[i % len(parents)]
                    parent2 = parents[(i + 1) % len(parents)]
                    child = np.zeros(n_points)
                    for j in range(n_points):
                        if np.random.rand() < 0.5:
                            child[j] = parent1[j]
                        else:
                            child[j] = parent2[j]
                    new_population.append(child)
                else:
                    new_population.append(parents[i % len(parents)])
            
            # Mutation
            for i in range(len(new_population)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        new_population[i][j] = np.random.uniform(0, 1)
                new_population[i] = np.clip(new_population[i], 0, 1)
                new_population[i] = new_population[i] * (1 / (np.sum(new_population[i]) * dx))
            
            population = new_population
        
        # Find the best solution in the population
        best_h = min(population, key=evaluate_erdos_solution)
        return best_h
    
    # Try genetic algorithm for different n_points
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Interpolate initial guess to new n_points
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h * dx != 1:
            h_initial = h_initial * (1 / (sum_h * dx))
        
        # Run genetic algorithm
        h_opt = genetic_algorithm()
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Perturb solution for better exploration
        h_perturbed = h_opt + np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
