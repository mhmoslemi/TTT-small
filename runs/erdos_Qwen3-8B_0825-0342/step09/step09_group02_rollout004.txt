import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points_candidate = [150, 180, 200, 220, 240, 260, 280, 300]
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = len(best_h)
    elapsed_time = 0
    start_time = time.time()
    
    for n_points in n_points_candidate:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize h with interpolation from the best solution so far
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h * (2.0 / n_points) != 1:
            h_initial = h_initial * (1 / (sum_h * (2.0 / n_points)))
        
        # Use a more sophisticated optimization method
        def objective_func(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Use L-BFGS-B as a more efficient solver for bound-constrained problems
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Add a small amount of perturbation to avoid local minima
        h_perturbed = h_opt + np.random.uniform(-0.01, 0.01, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_h_perturbed = np.sum(h_perturbed)
        if sum_h_perturbed * (2.0 / n_points) != 1:
            h_perturbed = h_perturbed * (1 / (sum_h_perturbed * (2.0 / n_points)))
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, best_n_points
