```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 120
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use a combination of global search and local optimization with different strategies
    def global_search(initial_guess):
        nonlocal best_h, best_c5
        # Basinhopping for global search
        from scipy.optimize import basinhopping
        
        def basinhopping_objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        def basinhopping_constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) * (2.0 / n_points) - 1.0
        
        result = basinhopping(
            basinhopping_objective,
            initial_guess,
            T=1.0,
            stepsize=0.1,
            niter=20,
            minimizer_kwargs={
                'method': 'SLSQP',
                'bounds': bounds,
                'constraints': {'type': 'eq', 'fun': basinhopping_constraint},
                'options': {'ftol': 1e-8, 'maxiter': 100}
            }
        )
        
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
        
        # Local refinement with SLSQP
        result = optimize.minimize(
            objective_func,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Use a genetic algorithm for global search
    def genetic_algorithm(population_size=20, generations=20, mutation_rate=0.1):
        nonlocal best_h, best_c5
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * (1 / (np.sum(population, axis=1) * (2.0 / n_points))[:, np.newaxis])
        
        for _ in range(generations):
            if time.time() - start_time > budget_s:
                break
            # Evaluate fitness
            fitness = np.array([evaluate_erdos_solution(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            
            # Selection
            selected_indices = np.argsort(fitness)[:population_size // 2]
            selected_population = population[selected_indices]
            
            # Crossover
            new_population = []
            for _ in range(population_size // 2):
                parent1 = selected_population[np.random.choice(selected_indices)]
                parent2 = selected_population[np.random.choice(selected_indices)]
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                new_population.append(child)
            
            # Mutation
            for i in range(len(new_population)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        new_population[i][j] = np.clip(new_population[i][j] + np.random.uniform(-0.1, 0.1), 0, 1)
                # Normalize
                total = np.sum(new_population[i]) * (2.0 / n_points)
                new_population[i] = new_population[i] / np.clip(total, 1e-12, np.inf)
            
            population = np.vstack((selected_population, new_population))
        
        # After genetic algorithm, do local optimization
        result = optimize.minimize(
            objective_func,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Run global search
    global_search(best_h)
    
    # Run genetic algorithm
    genetic_algorithm()
    
    # Perturb and refine
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Perturb
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
        
        # Local optimization
        result = optimize.minimize(
            objective_func,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        h_opt = result.x
        current_c5 = evaluate_erdos_solution(h_opt)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
```
