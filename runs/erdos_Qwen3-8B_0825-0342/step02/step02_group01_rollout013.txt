```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load("initial_h_values.npy")  # Placeholder for initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can increase if needed
    
    # Initial solution
    h_initial = initial_h_values[:n_points] if len(initial_h_values) >= n_points else initial_h_values
    h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant') if len(h_initial) < n_points else h_initial
    h_initial = h_initial[:n_points]
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial / (np.sum(h_initial) * (2.0 / n_points)) * (n_points / 2)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial.reshape(-1)
    c5_initial = evaluate_erdos_solution(h_initial)
    best_c5 = c5_initial
    best_h = h_initial.copy()
    
    # Define a function to evaluate the overlap integral for a given h
    def evaluate_overlap(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        return np.max(overlap)
    
    # Define a function to update h based on gradient descent
    def update_h(h, learning_rate):
        h = h.copy()
        h = np.clip(h, 0, 1)
        overlap = evaluate_overlap(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_new = h.copy()
            h_new[i] += 1e-5
            h_new = np.clip(h_new, 0, 1)
            overlap_new = evaluate_overlap(h_new)
            grad[i] = (overlap_new - overlap) / 1e-5
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        return h
    
    # Gradient descent optimization
    learning_rate = 0.01
    max_iterations = 500
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        h = update_h(best_h, learning_rate)
        c5 = evaluate_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Try random perturbations
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * (2.0 / n_points)) * (n_points / 2)
        h = np.clip(h, 0, 1)
        c5 = evaluate_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Try higher resolution
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h / (np.sum(best_h) * (2.0 / n_points)) * (n_points / 2)
        best_h = np.clip(best_h, 0, 1)
        c5 = evaluate_overlap(best_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
    
    return best_h, best_c5, n_points
```
