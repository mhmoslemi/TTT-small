import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with a reasonable number of points
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        initial_h_values[i] = 1.0 if i % 5 == 0 else 0.0  # Simple initial guess
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h_expanded = np.concatenate([h, np.zeros(n_points)])
        one_minus_h = 1.0 - h
        one_minus_h_expanded = np.concatenate([one_minus_h, np.zeros(n_points)])
        corr = np.correlate(h_expanded, one_minus_h_expanded, mode="full")
        return np.max(corr * dx)
    
    # Objective function to minimize
    def objective(h_values):
        return compute_c5(h_values)
    
    # Constraints: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Try different numbers of points up to a maximum of 1000
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time.time() - start_time > budget_s:
            break
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h.reshape(-1)
        
        # Optimize using scipy
        res = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        h_opt = res.x
        c5 = compute_c5(h_opt)
        
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
    
    return best_h, best_c5, n_points
