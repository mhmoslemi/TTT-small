import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.optimize import Bounds

# Pre-loaded initial construction (n=49 samples)
initial_h_values = np.array([0.4925, 0.5833, 0.6458, 0.6865, 0.7135, 0.7342, 0.7495, 0.7602, 0.7673, 0.7713,
                             0.7722, 0.7701, 0.7652, 0.7578, 0.7482, 0.7367, 0.7235, 0.7086, 0.6923, 0.6746,
                             0.6559, 0.6364, 0.6162, 0.5954, 0.574, 0.5522, 0.5301, 0.5076, 0.4847, 0.4616,
                             0.4382, 0.4145, 0.3905, 0.3663, 0.3419, 0.3173, 0.2926, 0.2677, 0.2428, 0.2179,
                             0.193, 0.168, 0.143, 0.118, 0.093, 0.068, 0.043, 0.018, 0.0])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Start with a different number of points than the initial construction
    n_points_list = [50, 60, 70]  # Try different resolutions for better exploration
    best_c5 = float('inf')
    best_h = np.array([])
    best_n_points = 0
    
    for n_points in n_points_list:
        # Define the grid
        dx = 2.0 / n_points
        time_limit = budget_s
        start_time = time.time()
        
        # Initial guess based on the initial construction rescaled
        if n_points == 50:
            h_initial = np.zeros(n_points)
            h_initial[:49] = initial_h_values[:49] * (n_points / 49)
            h_initial[49] = 0.5 * (1 - (n_points / 49))
        elif n_points == 60:
            h_initial = np.zeros(n_points)
            h_initial[:49] = initial_h_values[:49] * (n_points / 49)
            h_initial[49:] = 0.5 * (n_points - 49) / n_points
        else:  # n_points == 70
            h_initial = np.zeros(n_points)
            h_initial[:49] = initial_h_values[:49] * (n_points / 49)
            h_initial[49:] = 0.5 * (n_points - 49) / n_points
        
        # Normalize to satisfy the integral constraint
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        h_initial = np.clip(h_initial, 0, 1)
        
        # Optimization
        def compute_c5(h):
            h = np.array(h)
            shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0) if n_points > 1 else h
            one_minus_h = 1 - h
            corr = correlate(shifted_h, one_minus_h, mode='full')
            return np.max(corr) * dx

        # Define objective function and constraints
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]

        def objective_func(h_values):
            h_values = np.array(h_values)
            shifted_h = np.concatenate([h_values[-n_points + 1:], h_values], axis=0) if n_points > 1 else h_values
            one_minus_h = 1 - h_values
            corr = correlate(shifted_h, one_minus_h, mode='full')
            return np.max(corr) * dx

        # Use SLSQP with bounds and constraints
        def minimize_max_overlap(x):
            return objective_func(x)

        # Initial optimization with L-BFGS-B
        result = minimize(
            minimize_max_overlap,
            h_initial,
            method='L-BFGS-B',
            bounds=Bounds(0, 1),
            tol=1e-6
        )

        if result.success:
            h_candidate = result.x.copy()
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n_points = n_points
        else:
            print(f"Optimization failed for n={n_points}, result: {result.message}")
        
        # Simple random perturbation
        for _ in range(3):
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h_initial + perturbation
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h / np.sum(new_h) * (n_points / 2)
            c5_new = compute_c5(new_h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = new_h
                best_n_points = n_points

        # Check time limit and break early if needed
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s * 0.6:  # Allocate 60% of budget for each n_points
            break
    
    return best_h, best_c5, best_n_points
