import scipy.signal as signal
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100  # Start with 100 samples (as per initial_h_values)
    population_size = 50
    generations = 100
    crossover_rate = 0.8
    mutation_rate = 0.1
    dx = 2.0 / n_points
    
    # Initialize population with initial_h_values and some perturbations
    initial_h = initial_h_values()
    initial_h = np.resize(initial_h, n_points)
    initial_h = initial_h * (1 / (np.sum(initial_h) * dx))
    
    population = [initial_h + np.random.uniform(-0.05, 0.05, size=n_points) for _ in range(population_size)]
    population = [np.clip(p, 0, 1) for p in population]
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate the fitness of each candidate
        fitness = [evaluate_erdos_solution(p) for p in population]
        
        # Select the best candidates for crossover
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:population_size // 2]]
        
        # Crossover to create new candidates
        new_population = []
        for i in range(population_size // 2):
            parent1 = selected[i]

            child = np.copy(parent1)
            for j in range(n_points):
                if random.random() < crossover_rate:
                    child[j] = parent2[j]
            new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):

        # Replace the population with the new candidates
        population = new_population
        
        # Update the best solution
        current_best = min(population, key=evaluate_erdos_solution)
        current_c5 = evaluate_erdos_solution(current_best)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
    
    return best_h, best_c5, n_points
