def run(seed=42, budget_s=60, **kwargs):

    np.random.seed(seed)

    start_time = time.time()

    def evaluate_overlap(h):
        h = h.astype(float)
        h_shifted = np.concatenate([np.zeros_like(h), h[:-1]])
        overlap = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to optimize h, given a starting point
    def optimize_h(h_start, max_iterations=50, tolerance=1e-5, n_points=n_points):
        h = h_start.copy()
        best_c5 = evaluate_overlap(h)
        best_h = h.copy()
        
        for _ in range(max_iterations):
            # Compute the current C₅
            current_c5 = evaluate_overlap(h)
            
            # Define variables for optimization
            h_var = cp.Variable(n_points, nonneg=True, lb=0, ub=1)
            constraints = [cp.sum(h_var) == n_points / 2]
            
            # Objective function: minimize the maximum overlap
            # To handle max, we use a convex reformulation
            max_overlap = cp.Parameter()
            max_overlap.value = current_c5
            
            # We minimize the maximum overlap by introducing a new variable
            # that must be greater than or equal to all overlaps
            t = cp.Variable()
            objective = cp.Minimize(t)
            
            # Define the constraints for all shifts
            for k in range(1, n_points):
                overlap_k = cp.sum(h_var * (1 - h_var[k:])) * dx
                constraints.append(overlap_k <= t)
            
            # Solve the convex optimization problem
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver='MOSEK', verbose=False)
                if prob.status == 'OPTIMAL':
                    h = h_var.value
                    current_c5 = t.value
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()

                    break
            except:
                # If solver fails, use the current h
                break
        
        return best_h, best_c5
    
    # Perform the optimization
    best_h, best_c5 = optimize_h(h)

    elapsed_time = time.time() - start_time

        # If time is almost up, return the best solution found
        return (best_h, best_c5, n_points)

        h, _ = optimize_h(h)
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h)
    
    # Return the best solution found
