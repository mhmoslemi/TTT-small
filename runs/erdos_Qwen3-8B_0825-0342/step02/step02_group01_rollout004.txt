import numpy as np

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can be adjusted if time permits
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Initial guess: use initial_h_values if provided, else uniform distribution
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Function to evaluate C5
    def evaluate_c5(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h_shifted = np.zeros_like(h)
        for k in range(1, n_points):
            h_shifted = np.roll(h, -k)
            overlap = np.sum(h * (1 - h_shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    # Function to evaluate and return C5
    def evaluate_and_return(h_values):
        nonlocal max_overlap
        max_overlap = 0.0
        for k in range(1, n_points):
            h_shifted = np.roll(h_values, -k)
            overlap = np.sum(h_values * (1 - h_shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    # Optimization function
    def objective(h_values):
        return evaluate_and_return(h_values)

    # Constraints
    def constraint_sum(h_values):
        return np.sum(h_values) - (n_points / 2)

    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization
    result = minimize(objective, h, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result['x']
    best_c5 = evaluate_and_return(best_h)

    # Refinement with local search
    for _ in range(5):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        c5_perturbed = evaluate_and_return(perturbed_h)
        if c5_perturbed < best_c5:
            best_h = perturbed_h
            best_c5 = c5_perturbed

    # Final evaluation
    final_c5 = evaluate_and_return(best_h)
    return (best_h, final_c5, n_points)
