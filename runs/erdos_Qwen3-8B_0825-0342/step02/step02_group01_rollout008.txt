```python
import numpy as np
import scipy.optimize as opt
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 100
    max_n_points = 500  # Maximum number of points allowed within the budget
    
    # Initial h values
    h_values = np.array(initial_h_values[:n_points], dtype=float)
    
    # Function to evaluate the C5 bound
    def evaluate_c5(h):
        dx = 2.0 / len(h)
        h = np.array(h, dtype=float)
        h = np.concatenate([h, h[:1]])  # Pad to make full convolution work
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Function to optimize the h values
    def optimize_h(current_h, n_points):
        # Define the objective function to minimize the maximum overlap
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            return evaluate_c5(h)
        
        # Define constraints
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0, 1) for _ in range(len(current_h))]
        
        # Initial guess
        x0 = current_h.copy()
        
        # Run optimization
        result = opt.minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP"
        )
        
        if result.success:
            return result.x
        else:
            return current_h  # Return original if optimization fails
    
    # Adaptive optimization loop
    best_c5 = float('inf')
    best_h = h_values
    
    # Start with initial n_points
    n_points = 100
    
    # Try increasing n_points up to max_n_points within budget
    while n_points <= max_n_points and time.time() - start_time < budget_s:
        # Optimize h for current n_points
        current_h = optimize_h(best_h, n_points)
        
        # Evaluate the C5 bound
        current_c5 = evaluate_c5(current_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        
        # Try increasing n_points
        n_points += 50
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    # Return the result
    return (best_h, final_c5, len(best_h))
```
