```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Sample initial solution

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100  # Use 100 points for fast evaluation within the time budget
    dx = 2.0 / n_points
    
    # Initial guess: start with the provided initial solution
    h_initial = initial_h_values[:n_points]  # Use first n_points samples
    # Scale to ensure the integral equals 1
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Define the optimization variables
    h = cp.Variable(n_points, nonneg=True, bounded=(0, 1))
    
    # Objective: minimize the maximum overlap integral over all shifts
    # Note: We can't directly minimize the max over all k, so we use a convex relaxation
    # by introducing slack variables for each shift
    # For optimization, we'll use a convex relaxation that minimizes the max with a bound
    
    # Define a bound on the max overlap
    max_overlap = cp.Variable()
    constraints = [max_overlap >= cp.sum(h * (1 - h[shift:] - h[:n_points - shift])) * dx for shift in range(n_points)]
    
    # Objective is to minimize max_overlap
    prob = cp.Problem(cp.Minimize(max_overlap), constraints)
    
    # Solve the optimization problem
    start_time = time.time()
    result = prob.solve(solver=cp.ECOS, verbose=False)
    elapsed_time = time.time() - start_time
    
    # Extract the optimal h_values
    h_values = h.value
    
    # Ensure the integral constraint is satisfied
    if np.abs(np.sum(h_values) * dx - 1) > 1e-6:
        h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Calculate the final C5 bound
    c5_bound = evaluate_erdos_solution(h_values)
    
    # If we've exceeded the time budget, return the best solution found so far
    if elapsed_time >= budget_s - 1:
        return h_values, c5_bound, n_points
    
    return h_values, c5_bound, n_points
```
