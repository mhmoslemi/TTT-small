```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of points, starting from n_points=100
    # These are chosen to balance speed and accuracy
    n_points_list = [100, 150, 200, 250, 300, 350]
    
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Initial guess: use the initial_h_values and interpolate to n_points
        initial_h = initial_h_values
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Variables
        h = cp.Variable(n_points, bounds=(0, 1))
        
        # Objective: minimize the maximum overlap integral
        overlaps = []
        for k in range(1, n_points):
            # Compute h(x) * (1 - h(x + k)) for all x
            # Wrap around the interval [0, 2]
            h_shifted = np.zeros(n_points)
            for i in range(n_points):
                x = i * dx
                x_shifted = x + k * dx
                if x_shifted > 2:
                    x_shifted -= 2
                idx = int(x_shifted / dx)
                h_shifted[i] = h[idx]
            overlap = cp.sum(h * (1 - h_shifted)) * dx
            overlaps.append(overlap)
        
        # Maximize the overlaps
        objective = cp.Minimize(cp.Max(overlaps))
        constraints = [cp.sum(h) == n_points / 2]
        
        # Solve
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cvxpy.SCS, eps=1e-5, max_iter=100)
            
            if prob.status == "optimal":
                h_values = h.value
                h_values = np.clip(h_values, 0, 1)
                h_values = h_values * (n_points / 2) / np.sum(h_values)
                
                # Evaluate C5
                c5 = evaluate_erdos_solution(h_values, dx)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
        except:
            pass
    
    # If we didn't find a solution, use the initial one
    if best_h is None:
        best_h = initial_h_values
        
    return best_h, best_c5, len(best_h)
```
