    # Use optimization to minimize the max overlap

    # Optimization loop
    max_iterations = 100

        # Compute current C_5
        current_c5 = compute_c5(h)

        if current_c5 < best_c5:
            best_c5 = current_c5

        # Perform a small gradient step
