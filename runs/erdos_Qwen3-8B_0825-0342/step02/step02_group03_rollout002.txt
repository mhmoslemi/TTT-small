import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    # Pre-loaded initial construction (n=49 samples)
    initial_h_values = np.array([0.4925, 0.5833, 0.6458, 0.6865, 0.7135, 0.7342, 0.7495, 0.7602, 0.7673, 0.7713,
                                 0.7722, 0.7701, 0.7652, 0.7578, 0.7482, 0.7367, 0.7235, 0.7086, 0.6923, 0.6746,
                                 0.6559, 0.6364, 0.6162, 0.5954, 0.574, 0.5522, 0.5301, 0.5076, 0.4847, 0.4616,
                                 0.4382, 0.4145, 0.3905, 0.3663, 0.3419, 0.3173, 0.2926, 0.2677, 0.2428, 0.2179,
                                 0.193, 0.168, 0.143, 0.118, 0.093, 0.068, 0.043, 0.018, 0.0])

    # Use the same number of points as the initial construction
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    time_limit = budget_s
    start_time = time.time()

    def compute_c5(h):
        h = np.array(h)
        shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
        one_minus_h = 1 - h
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Try initial construction
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    # Try uniform distribution
    uniform_h = np.full(n_points, 0.5)
    c5_uniform = compute_c5(uniform_h)
    if c5_uniform < best_c5:
        best_c5 = c5_uniform
        best_h = uniform_h.copy()

    # Try alternating pattern
    alt_h = np.zeros(n_points)
    for i in range(n_points):
        alt_h[i] = 0.5 if i % 2 == 0 else 0.0
    c5_alt = compute_c5(alt_h)
    if c5_alt < best_c5:
        best_c5 = c5_alt
        best_h = alt_h.copy()

    # Function to compute the maximum overlap integral
    def objective_func(h_values):
        h_values = np.array(h_values)
        shifted_h = np.concatenate([h_values[-n_points + 1:], h_values], axis=0)
        one_minus_h = 1 - h_values
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Optimization with Sequential Quadratic Programming (SQP)
    def minimize_max_overlap(x):
        return objective_func(x)

    # Use a local optimization method for refinement
    result = minimize(
        minimize_max_overlap,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        tol=1e-6
    )

    if result.success:
        h_candidate = result.x
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate

    # Local search with random perturbations
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h

    # Try a new starting point: sinusoidal pattern
    sin_h = np.sin(np.linspace(0, np.pi, n_points))
    sin_h = (sin_h - np.min(sin_h)) / (np.max(sin_h) - np.min(sin_h))
    sin_h = np.clip(sin_h, 0, 1)
    sin_h = sin_h * 0.5 + 0.5
    c5_sin = compute_c5(sin_h)
    if c5_sin < best_c5:
        best_c5 = c5_sin
        best_h = sin_h.copy()

    # Try a new starting point: piecewise constant pattern
    piecewise_h = np.zeros(n_points)
    for i in range(n_points):
        piecewise_h[i] = 0.5 if i < n_points // 2 else 0.0
    c5_piecewise = compute_c5(piecewise_h)
    if c5_piecewise < best_c5:
        best_c5 = c5_piecewise
        best_h = piecewise_h.copy()

    # Try a new starting point: random binary pattern
    binary_h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    binary_h = binary_h * 0.5 + 0.5
    c5_binary = compute_c5(binary_h)
    if c5_binary < best_c5:
        best_c5 = c5_binary
        best_h = binary_h.copy()

    # Try a new starting point: Gaussian bump pattern
    gaussian_h = np.exp(-((np.linspace(0, 2, n_points) - 1) / 0.5)**2 / 2)
    gaussian_h = (gaussian_h - np.min(gaussian_h)) / (np.max(gaussian_h) - np.min(gaussian_h))
    gaussian_h = np.clip(gaussian_h, 0, 1)
    gaussian_h = gaussian_h * 0.5 + 0.5
    c5_gaussian = compute_c5(gaussian_h)
    if c5_gaussian < best_c5:
        best_c5 = c5_gaussian
        best_h = gaussian_h.copy()

    # Use a small-scale optimization with CVXPY
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]

    # Define objective function
    def cvx_objective(h_values):
        h_values = np.array(h_values)
        shifted_h = np.concatenate([h_values[-n_points + 1:], h_values], axis=0)
        one_minus_h = 1 - h_values
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return cp.Max(corr * dx)

    # Try to optimize
    try:
        problem = cp.Problem(cp.Minimize(cvx_objective(h)), constraints)
        problem.solve(solver=cp.SCS, eps=1e-6)
        if problem.status == 'optimal':
            h_candidate = h.value
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
    except:
        pass

    # Ensure we don't exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > time_limit - 1:
        # Return the best solution before time runs out
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
