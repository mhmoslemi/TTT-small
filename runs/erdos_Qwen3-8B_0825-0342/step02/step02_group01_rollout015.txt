def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h_values
    corr = np.correlate(h_values, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5
