# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Set up the initial population
    n_points = 100
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 20
    
    # Initialize population
    initial_h = initial_h_values()

    dx = 2.0 / n_points

    initial_population = [initial_h.copy() for _ in range(population_size)]

        fitness = [evaluate_erdos_solution(individual) for individual in initial_population]
        
        # Select parents
        parent_indices = np.argsort(fitness)[:int(population_size / 2)]
        parents = [initial_population[i] for i in parent_indices]
        
        # Create next generation
        next_population = []
        for _ in range(population_size):
