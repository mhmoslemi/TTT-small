import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=60, **kwargs):
    # Pre-loaded initial construction (n=49 samples)
    initial_h_values = np.array([0.4925, 0.5833, 0.6458, 0.6865, 0.7135, 0.7342, 0.7495, 0.7602, 0.7673, 0.7713,
                                 0.7722, 0.7701, 0.7652, 0.7578, 0.7482, 0.7367, 0.7235, 0.7086, 0.6923, 0.6746,
                                 0.6559, 0.6364, 0.6162, 0.5954, 0.574, 0.5522, 0.5301, 0.5076, 0.4847, 0.4616,
                                 0.4382, 0.4145, 0.3905, 0.3663, 0.3419, 0.3173, 0.2926, 0.2677, 0.2428, 0.2179,
                                 0.193, 0.168, 0.143, 0.118, 0.093, 0.068, 0.043, 0.018, 0.0])

    # Try smaller n_points for faster optimization
    n_points = 49
    dx = 2.0 / n_points
    time_limit = budget_s
    start_time = time.time()

    def compute_c5(h):
        h = np.array(h)
        shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
        one_minus_h = 1 - h
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Try initial construction
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    # Try uniform distribution
    uniform_h = np.full(n_points, 0.5)
    c5_uniform = compute_c5(uniform_h)
    if c5_uniform < best_c5:
        best_c5 = c5_uniform
        best_h = uniform_h.copy()

    # Try alternating pattern
    alt_h = np.zeros(n_points)
    for i in range(n_points):
        alt_h[i] = 0.5 if i % 2 == 0 else 0.0
    c5_alt = compute_c5(alt_h)
    if c5_alt < best_c5:
        best_c5 = c5_alt
        best_h = alt_h.copy()

    # Try a more structured pattern: block-wise step function
    block_size = n_points // 4
    block_h = np.zeros(n_points)
    for i in range(block_size):
        block_h[i] = 1.0
    for i in range(block_size, 2 * block_size):
        block_h[i] = 0.0
    for i in range(2 * block_size, 3 * block_size):
        block_h[i] = 1.0
    for i in range(3 * block_size, n_points):
        block_h[i] = 0.0
    c5_block = compute_c5(block_h)
    if c5_block < best_c5:
        best_c5 = c5_block
        best_h = block_h.copy()

    # Try a modified pattern with varying step heights
    mod_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            mod_h[i] = 1.0
        elif i % 5 == 1 or i % 5 == 2:
            mod_h[i] = 0.75
        elif i % 5 == 3:
            mod_h[i] = 0.5
        else:
            mod_h[i] = 0.25
    c5_mod = compute_c5(mod_h)
    if c5_mod < best_c5:
        best_c5 = c5_mod
        best_h = mod_h.copy()

    # Try a random initialization
    np.random.seed(seed)
    random_h = np.random.uniform(0, 1, n_points)
    random_h = random_h * (n_points / 2) / np.sum(random_h)
    c5_random = compute_c5(random_h)
    if c5_random < best_c5:
        best_c5 = c5_random
        best_h = random_h.copy()

    # Use a metaheuristic approach - genetic algorithm
    def fitness_function(h):
        h = np.array(h)
        shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
        one_minus_h = 1 - h
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Genetic algorithm parameters
    population_size = 20
    generations = 20
    mutation_rate = 0.1
    crossover_rate = 0.7

    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = population[i] * (n_points / 2) / np.sum(population[i])

    for generation in range(generations):
        # Evaluate fitness
        fitness = [fitness_function(ind) for ind in population]
        # Select fittest individuals
        best_indices = np.argsort(fitness)[:population_size // 2]
        selected = [population[i] for i in best_indices]
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Crossover
            if np.random.rand() < crossover_rate:
                parent1 = np.random.choice(selected)
                parent2 = np.random.choice(selected)
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
            else:
                child = np.random.uniform(0, 1, n_points)
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        population = new_population

        # Update best solution
        for i in range(len(population)):
            if fitness[i] < best_c5:
                best_c5 = fitness[i]
                best_h = population[i].copy()

        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break

    # Use local optimization to refine the best solution
    result = minimize(
        lambda x: compute_c5(x),
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        tol=1e-6
    )

    if result.success:
        h_candidate = result.x
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate

    # Local search with random perturbations
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h

    return best_h, best_c5, n_points
