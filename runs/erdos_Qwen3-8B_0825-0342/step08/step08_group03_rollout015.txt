import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random
from scipy.optimize import basinhopping

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 120
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    # Define objective and constraint
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Global optimization with basinhopping
    def basinhopping_optimization(h_initial):
        # Define the objective function for basinhopping
        def basinhopping_objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Define the constraint for basinhopping
        def basinhopping_constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) * (2.0 / n_points) - 1.0
        
        result = basinhopping(
            basinhopping_objective,
            h_initial,
            T=1.0,
            stepsize=0.1,
            niter=30,
            minimizer_kwargs={
                'method': 'SLSQP',
                'bounds': bounds,
                'constraints': {'type': 'eq', 'fun': basinhopping_constraint},
                'options': {'ftol': 1e-8, 'maxiter': 100}
            }
        )
        
        return result.x
    
    # Local refinement with SLSQP
    def local_refinement(h_initial):
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x
    
    # Adaptive optimization with multiple restarts and perturbations
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        
        # Run basinhopping optimization
        h_initial = best_h.copy()
        h_opt = basinhopping_optimization(h_initial)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (1 / (np.sum(h_opt) * (2.0 / n_points)))
        
        # Refine with local optimization
        h_opt = local_refinement(h_opt)
        
        # Evaluate and update best solution
        current_c5 = evaluate_erdos_solution(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
        
        # Perturb the solution for further exploration
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
        
        # Refine perturbed solution
        h_perturbed = local_refinement(h_perturbed)
        
        # Evaluate and update best solution if improved
        current_c5 = evaluate_erdos_solution(h_perturbed)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
