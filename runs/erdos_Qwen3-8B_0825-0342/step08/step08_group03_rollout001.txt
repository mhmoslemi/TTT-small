```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    # Construct a better initial guess with some periodicity and structure
    n_points = 120
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        elif i % 5 == 2:
            h[i] = 0.25
        elif i % 5 == 3:
            h[i] = 0.75
        else:
            h[i] = 0.0
    # Normalize the integral to 1
    h = h * (1 / (np.sum(h) * 2.0 / n_points))
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Try different n_points values
    n_point_options = [120, 140, 160]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 120
    
    for n_points in n_point_options:
        if time.time() > budget_s - 5:
            break
        
        # Generate initial guess
        h = initial_h_values()
        h = np.resize(h, n_points)
        h = h * (n_points / 2.0) / np.sum(h)
        
        def objective_func(x):
            h_flat = x.reshape(n_points)
            corr = np.correlate(h_flat, 1 - h_flat, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        def constraint_func(x):
            h_flat = x.reshape(n_points)
            return np.sum(h_flat) * (2.0 / n_points) - 1.0
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use a gradient-based optimizer with careful tuning
        result = optimize.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        current_h = result.x
        current_c5 = evaluate_erdos_solution(current_h)
        
        # Perform local refinement
        for _ in range(5):
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = current_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (1 / (np.sum(h_perturbed) * (2.0 / n_points)))
            
            result = optimize.minimize(
                objective_func,
                h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_func},
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, best_n_points
```
