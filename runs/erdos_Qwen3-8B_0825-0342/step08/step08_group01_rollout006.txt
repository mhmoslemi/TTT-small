import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from the initial construction
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    # Try different starting points
    n_points_list = [100, 120, 140, 160, 180, 200, 220, 240, 260, 280, 300]
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    elapsed_time = 0
    
    start_time = time.time()
    
    # Use a different algorithmic approach: Simulated Annealing for global optimization
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initial guess based on current best
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h * dx != 1:
            h_initial = h_initial * (1 / (sum_h * dx))
        
        # Define objective function
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        # Constraints: volume constraint and bounds
        def volume_constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - n_points / 2
        
        # Simulated Annealing for global optimization
        def simulated_annealing(n_points, initial_h, objective_func, volume_constraint, budget_s):
            T = 1.0  # Initial temperature
            T_min = 1e-6  # Minimum temperature
            cooling_rate = 0.995  # Cooling rate
            n_iterations = 1000  # Maximum iterations per temperature
            best_h = initial_h.copy()
            best_c5 = objective_func(best_h)
            start_time_sa = time.time()
            
            while T > T_min and time.time() - start_time_sa < budget_s:
                for _ in range(n_iterations):
                    # Generate a random perturbation
                    h_perturbed = best_h + np.random.uniform(-0.05, 0.05, size=n_points)
                    h_perturbed = np.clip(h_perturbed, 0, 1)
                    
                    # Check constraint
                    if abs(volume_constraint(h_perturbed)) > 1e-6:
                        continue  # Skip if constraint is violated
                    
                    current_c5 = objective_func(h_perturbed)
                    
                    # Acceptance probability
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h_perturbed.copy()
                    else:
                        prob = np.exp((current_c5 - best_c5) / T)
                        if np.random.rand() < prob:
                            best_c5 = current_c5
                            best_h = h_perturbed.copy()
                
                T *= cooling_rate  # Cool down
        
            return best_h, best_c5
        
        # Run simulated annealing
        h_opt, current_c5 = simulated_annealing(n_points, h_initial, objective_func, volume_constraint, budget_s)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Perturb the solution for better exploration
        h_perturbed = h_opt + np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
