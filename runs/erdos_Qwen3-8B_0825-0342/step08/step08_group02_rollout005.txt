import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from the initial construction
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    # Define the objective function
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Use SLSQP with bounds and constraints
    constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(
        objective_func,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    
    h_opt = result.x
    current_c5 = evaluate_erdos_solution(h_opt)
    
    # Try a finer grid
    n_points = 200
    dx = 2.0 / n_points
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_opt)), h_opt)
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    # Define the objective function for the finer grid
    def objective_func_finer(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Use SLSQP with bounds and constraints
    constraints_finer = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
    bounds_finer = [(0, 1) for _ in range(n_points)]
    
    result_finer = optimize.minimize(
        objective_func_finer,
        h_initial,
        method='SLSQP',
        bounds=bounds_finer,
        constraints=constraints_finer,
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    
    h_opt_finer = result_finer.x
    current_c5_finer = evaluate_erdos_solution(h_opt_finer)
    
    if current_c5_finer < current_c5:
        h_opt = h_opt_finer
        current_c5 = current_c5_finer
    
    # Add perturbations
    h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
    
    if current_c5_perturbed < current_c5:
        h_opt = h_perturbed
        current_c5 = current_c5_perturbed
    
    # Try a coarser grid
    n_points = 100
    dx = 2.0 / n_points
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_opt)), h_opt)
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    # Define the objective function for the coarser grid
    def objective_func_coarser(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Use SLSQP with bounds and constraints
    constraints_coarser = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2}]
    bounds_coarser = [(0, 1) for _ in range(n_points)]
    
    result_coarser = optimize.minimize(
        objective_func_coarser,
        h_initial,
        method='SLSQP',
        bounds=bounds_coarser,
        constraints=constraints_coarser,
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    
    h_opt_coarser = result_coarser.x
    current_c5_coarser = evaluate_erdos_solution(h_opt_coarser)
    
    if current_c5_coarser < current_c5:
        h_opt = h_opt_coarser

    h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)

        h_opt = h_perturbed
