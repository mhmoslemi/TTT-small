```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random
import matplotlib.pyplot as plt

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from initial guess
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if sum_h * dx != 1:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    elapsed_time = 0
    start_time = time.time()
    
    # Use a different approach than the original by using CVXPY for a more structured optimization
    n_points_list = [160, 180, 200, 220, 240, 260, 280, 300, 250]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with a piecewise constant function as a starting point
        h_start = np.zeros(n_points)
        h_start[:n_points//3] = 1.0
        h_start[n_points//3:2*n_points//3] = 0.5
        h_start[2*n_points//3:] = 0.0
        h_start = h_start * (1 / (np.sum(h_start) * dx))
        
        # Define the problem using CVXPY
        h = cp.Variable(n_points)
        obj = cp.Minimize(cp.Max(cp.Correlation(h, 1 - h, mode="full") * dx))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        problem = cp.Problem(obj, constraints)
        result = problem.solve(solver='SCIPY', abstol=1e-8, reltol=1e-8, maxiter=100)
        
        h_opt = h.value
        current_c5 = evaluate_erdos_solution(h_opt)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            best_n_points = n_points
        
        # Perturb solution for better exploration
        h_perturbed = h_opt + np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
```
