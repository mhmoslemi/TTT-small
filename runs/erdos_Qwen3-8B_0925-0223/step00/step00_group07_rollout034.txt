import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    current_sum = np.sum(initial_h)
    scale_factor = (n_points / 2) / current_sum
    transformed_h = np.clip(initial_h * scale_factor, 0, 1)

        corr = np.correlate(h, 1 - h, mode='full')

    def constraint_eq(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_eq}]
    
    start_time = time.time()
    result = minimize(objective, transformed_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6, options={'maxiter': 1000})
    elapsed = time.time() - start_time
    
    if not result.success:
        final_h = np.clip(transformed_h, 0, 1)
        corr = np.correlate(final_h, 1 - final_h, mode='full')
        max_corr = np.max(corr[1:-1])
        c5_bound = max_corr * dx
        return (final_h, c5_bound, n_points)
    
    final_h = np.clip(result.x, 0, 1)
    corr = np.correlate(final_h, 1 - final_h, mode='full')
    max_corr = np.max(corr[1:-1])
    c5_bound = max_corr * dx
    
    if elapsed > budget_s:
        print("Time limit exceeded, returning best estimate")
        return (final_h, c5_bound, n_points)
    
    return (final_h, c5_bound, n_points)
