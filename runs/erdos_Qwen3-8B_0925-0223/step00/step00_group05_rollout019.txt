import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h)
    scale_initial = (n_points / 2) / initial_sum if abs(initial_sum) > 1e-12 else 1.0
    scaled_initial = initial_h * scale_initial

    def objective(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return float('inf')
        scale = (n_points / 2) / total
        scaled_h = h * scale
        corr = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    result = optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.8,
        polish=True,
        seed=seed
    )

    best_h = result.x
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum) < 1e-12:
        scaled_best_h = scaled_initial
    else:
        scale_best = (n_points / 2) / best_h_sum
        scaled_best_h = best_h * scale_best

    return (scaled_best_h, result.fun, n_points)
