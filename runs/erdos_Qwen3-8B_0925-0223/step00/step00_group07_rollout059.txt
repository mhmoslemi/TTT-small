    h_guess = np.round(h_guess, 5)  # Reduce floating point noise

    def objective(h):
        h = np.array(h, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points
    result = minimize(objective, h_guess, method='SLSQP',
