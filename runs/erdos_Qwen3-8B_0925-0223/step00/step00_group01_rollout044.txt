import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Clip and scale initial guess
    h_initial = np.clip(initial_h, 0.0, 1.0)
    h_initial = h_initial * (target_sum / np.sum(h_initial))

    # PSO parameters
    n_particles = 50
    max_iterations = 200
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    lambda_p = 1e4
    mu_p = 1e4

    # Initialize particles
    particles = np.random.rand(n_particles, n_points)
    particles = np.clip(particles, 0.0, 1.0)
    velocities = np.random.rand(n_particles, n_points) * v_max * 2 - v_max

    # Best positions and fitness
    best_positions = particles.copy()
    best_fitness = np.inf * np.ones(n_particles)

    # Evaluate initial positions
    for i in range(n_particles):
        fitness = compute_fitness_approx(particles[i], n_points, dx, lambda_p, mu_p)
        best_fitness[i] = fitness
        if fitness < best_fitness[0]:
            best_positions[0, :] = particles[i, :]

    # PSO loop
    start_time = time.time()
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i, :] = w * velocities[i, :] + c1 * r1 * (best_positions[i, :] - particles[i, :]) + c2 * r2 * (best_positions[0, :] - particles[i, :])
            velocities[i, :] = np.clip(velocities[i, :], -v_max, v_max)
            particles[i, :] += velocities[i, :]
            particles[i, :] = np.clip(particles[i, :], 0.0, 1.0)
            particles[i, :] = particles[i, :] * (target_sum / np.sum(particles[i, :]))

        for i in range(n_particles):
            fitness = compute_fitness_approx(particles[i], n_points, dx, lambda_p, mu_p)
            if fitness < best_fitness[i]:
                best_fitness[i] = fitness
                best_positions[i, :] = particles[i, :]

        if np.min(best_fitness) < best_fitness[0]:
            best_fitness[0] = np.min(best_fitness)
            best_positions[0, :] = best_positions[np.argmin(best_fitness), :]

    # Final evaluation
    best_h = best_positions[0, :]
    sum_h = np.sum(best_h)
    best_h = best_h * (target_sum / sum_h)
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_exact_c5(best_h, n_points, dx)
    return (best_h, final_c5, n_points)

def compute_fitness_approx(h, n_points, dx, lambda_p, mu_p):
    sum_h = np.sum(h)
    sum_penalty = lambda_p * abs(sum_h - n_points / 2.0)
    value_penalty = mu_p * np.sum(np.maximum(0, h - 1.0)) + mu_p * np.sum(np.maximum(0, 1.0 - h))
    
    corr = np.correlate(h, 1 - h, mode='full')
    if n_points > 100:
        lags = np.arange(len(corr))
        sampled_lags = lags[::10]
        sampled_corr = corr[sampled_lags]
        max_corr = np.max(sampled_corr) * dx
    else:
        max_corr = np.max(corr) * dx
    return max_corr + sum_penalty + value_penalty

def compute_exact_c5(h, n_points, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr
