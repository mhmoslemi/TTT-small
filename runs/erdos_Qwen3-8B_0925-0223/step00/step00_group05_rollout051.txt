import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        h_initial = np.zeros(70)  # fallback, not used as initial is provided
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return float('inf')
        scale = (n_points / 2.0) / total
        scaled_h = h * scale
        corr = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.8,
        polish=True,
        seed=seed,
    )

    best_h = result.x
    c5_bound = objective(best_h)
    sum_h = np.sum(best_h)
    if sum_h == 0:
        scaled_initial = h_initial * (n_points / 2.0) / np.sum(h_initial)
        best_h = scaled_initial
        c5_bound = objective(best_h)
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    scale = (n_points / 2.0) / sum_h
    final_h = best_h * scale

    return (final_h, c5_bound, n_points)
