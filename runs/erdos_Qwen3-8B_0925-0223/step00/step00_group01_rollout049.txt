        penalty = 1e6 * abs(sum_h - T)
        return c5 + penalty

    bounds = [(0.0, 1.0)] * n_points

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        seed=seed,
        polish=True,
        x0=initial_h_values
    )

    best_h = result.x
    corr_best = np.correlate(best_h, 1 - best_h, mode='full')
    c5_best = np.max(corr_best) * dx
    return (best_h, c5_best, n_points)
