import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    # Use initial_h_values if available, otherwise fallback to a default construction
    h_initial = initial_h_values.copy() if 'initial_h_values' in globals() else np.array([0.5]*70)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(x):
        h = x
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        # Penalty for sum constraint
        penalized_sum = np.sum(h) - n_points / 2
        penalty = 5e3 * abs(penalized_sum)  # Tuned penalty weight for convergence
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization parameters
    maxiter = 100
    tol = 1e-6

    try:
        result = minimize(
            objective, 
            h_initial, 
            method='L-BFGS-B', 
            bounds=bounds, 
            tol=tol, 
            options={'maxiter': maxiter, 'ftol': 1e-8, 'gtol': 1e-8}
        )
    except:
        result = minimize(
            objective, 
            h_initial, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
            tol=tol, 
            options={'maxiter': maxiter}
        )

    if not result.success:
        h_opt = h_initial
        c5_bound = evaluate_erdos_solution(h_initial)
    else:
        h_opt = result.x
        c5_bound = evaluate_erdos_solution(h_opt)

    # Ensure sum constraint is satisfied
    sum_h = np.sum(h_opt)
    if abs(sum_h - n_points / 2) > 1e-6:
        h_opt = h_opt * (n_points / 2) / sum_h

    # Final validation
    c5_bound = evaluate_erdos_solution(h_opt)
    return (h_opt, c5_bound, n_points)
