import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    current_h = np.clip(initial_h_values, 0, 1)
    current_sum = np.sum(current_h)
    current_h = current_h * (n_points / 2) / current_sum

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    best_h = current_h.copy()
    best_c5 = compute_c5(current_h)
    start_time = time.time()
    temp = 1.0

    while (time.time() - start_time) < budget_s:
        perturbed_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        perturbed_h = current_h.copy()
        noise = np.random.normal(0, 0.1, size=n_points//2)
        perturbed_h[perturbed_indices] += noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        sum_perturbed = np.sum(perturbed_h)
        if sum_perturbed == 0:
            continue
        perturbed_h *= (n_points / 2) / sum_perturbed
        perturbed_c5 = compute_c5(perturbed_h)
        current_c5 = compute_c5(current_h)
        delta = perturbed_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = perturbed_h.copy()
            if perturbed_c5 < best_c5:
                best_h = perturbed_h.copy()
                best_c5 = perturbed_c5
        current_time = time.time() - start_time
        temp = 1.0 * (1 - current_time / budget_s)

    final_sum = np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2) / final_sum if final_sum != 0 else 1.0
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
