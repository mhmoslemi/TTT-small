import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    
    assert np.all(0 <= h) and np.all(h <= 1)
    assert np.isclose(np.sum(h), n_points / 2)
    
    best_c5 = compute_c5(h, n_points)
    best_h = h.copy()
    
    max_iterations = 100
    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        
        def objective(h_vec):
            overlap = 0.0
            for i in range(n_points):
                j = (i + lag) % n_points
                overlap += h_vec[i] * (1 - h_vec[j])
            penalty = 1e6 * (np.sum(h_vec) - n_points / 2)
            return overlap + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds)
        
        if result.success:
            new_h = result.x
            if np.isclose(np.sum(new_h), n_points / 2, atol=1e-6):
                h = new_h.copy()
                current_c5 = compute_c5(h, n_points)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                else:
                    break
            else:
                continue
        else:
            continue
        
        if np.isclose(best_c5, compute_c5(h, n_points), atol=1e-6):
            break
    
    return best_h, best_c5, n_points
