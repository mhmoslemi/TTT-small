import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):

        initial_h = initial_h_values

        dx = 2.0 / n_points

        h[:half] = 1.0
        initial_h = h
    n_points = len(initial_h)
    dx = 2.0 / n_points

    h = np.copy(initial_h)
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = target_sum / sum_h
    else:
        scale = 1.0
    h *= scale
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h - target_sum) > 1e-6:
        scale = target_sum / sum_h
        h *= scale
        h = np.clip(h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.copy(h)
    best_c5 = compute_c5(h)

    T_initial = 1.0
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    while True:
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        T = T_initial * (1 - elapsed_time / budget_s)
        perturb_h = np.copy(h)
        n_perturb = int(0.5 * n_points)
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        noise = np.random.normal(0, 0.1, size=len(indices))
        perturb_h[indices] += noise
        perturb_h = np.clip(perturb_h, 0, 1)
        sum_perturb = np.sum(perturb_h)
        if sum_perturb != 0:
            scale = target_sum / sum_perturb
        else:
            scale = 1.0
        perturb_h *= scale
        perturb_h = np.clip(perturb_h, 0, 1)
        current_c5 = compute_c5(h)
        new_c5 = compute_c5(perturb_h)
        delta_c5 = new_c5 - current_c5
        if new_c5 < current_c5 or (np.random.rand() < np.exp(-delta_c5 / T)):
            h = np.copy(perturb_h)
