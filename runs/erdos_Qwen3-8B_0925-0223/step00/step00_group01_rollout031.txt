import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()

        noise = np.random.normal(0, 0.1, size=len(indices))
        h_perturbed = h.copy()
        h_perturbed[indices] += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        scale_perturbed = (n_points / 2) / sum_perturbed
        h_perturbed = h_perturbed * scale_perturbed
        c5_perturbed = np.max(np.correlate(h_perturbed, 1 - h_perturbed, mode='full')) * dx
        c5_current = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
