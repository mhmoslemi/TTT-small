import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_initial = np.sum(initial_h_values)

    h = initial_h_values * scale_factor

    bounds = [(0.0, 1.0)] * n_points

    def objective(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return float('inf')

        scaled = h_clipped * scale
        corr = np.correlate(scaled, 1 - scaled, mode='full')
        return np.max(corr) * dx

    maxiter = 100
    popsize = 15
    mutation = (0.5, 1.0)
    recombination = 0.8
    tol = 1e-6

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=popsize,
        mutation=mutation,
        recombination=recombination,
        tol=tol,
        polish=True
    )

    best_h = result.x
    best_h_scaled = best_h * ((n_points / 2) / np.sum(best_h))
    best_h_scaled = np.clip(best_h_scaled, 0, 1)
    c5_bound = evaluate_erdos_solution(best_h_scaled)

    return (best_h_scaled, c5_bound, n_points)
