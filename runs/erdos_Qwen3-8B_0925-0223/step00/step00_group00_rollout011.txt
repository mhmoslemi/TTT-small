        h_pert = np.clip(h_pert, 0, 1)
        current_c5 = evaluate_erdos_solution(h_pert)
        if current_c5 < best_c5:
            h = h_pert.copy()
            best_c5 = current_c5
        elif np.random.rand() < np.exp((best_c5 - current_c5) / T):
            h = h_pert.copy()
        if current_c5 < best_c5:
            best_h = h.copy()
        T *= cooling_rate
    return best_h, best_c5, n_points
