    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    num_particles = 50
    max_iter = 200
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    np.random.seed(seed)
    particles = np.random.rand(num_particles, n_points)
    particles = particles * 0.1 + h_initial
    velocities = np.random.rand(num_particles, n_points) * v_max
    
    personal_best = particles.copy()
    personal_best_fitness = np.inf * np.ones(num_particles)
    global_best = particles[0].copy()
    global_best_fitness = np.inf
    
    for i in range(num_particles):
        h = np.clip(particles[i], 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h if sum_h > 0 else h
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        sum_diff = np.abs(np.sum(h) - n_points / 2)
        bound_penalty = np.sum(np.maximum(0, h - 1)) + np.sum(np.maximum(0, 1 - h))
        fitness = c5 + 1e4 * sum_diff + 1e4 * bound_penalty
        personal_best_fitness[i] = fitness
        if fitness < global_best_fitness:
            global_best_fitness = fitness
            global_best = h.copy()
    
    start_time = time.time()
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            particles[i] += velocities[i]
            h = np.clip(particles[i], 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h if sum_h > 0 else h
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            sum_diff = np.abs(np.sum(h) - n_points / 2)
            bound_penalty = np.sum(np.maximum(0, h - 1)) + np.sum(np.maximum(0, 1 - h))
            fitness = c5 + 1e4 * sum_diff + 1e4 * bound_penalty
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = h.copy()
                if fitness < global_best_fitness:
                    global_best_fitness = fitness
                    global_best = h.copy()
    
    final_h = global_best
    corr = np.correlate(final_h, 1 - final_h, mode='full')
    c5_bound = np.max(corr) * (2.0 / n_points)
    return final_h, c5_bound, n_points
