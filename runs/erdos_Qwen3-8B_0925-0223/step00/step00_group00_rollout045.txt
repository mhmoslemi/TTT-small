        max_corr = np.max(corr)
        lag_index = np.argmax(corr)
        k = lag_index - n_points + 1
        gradient = np.zeros(n_points)
        for i_idx in range(n_points):
            i_plus_k = (i_idx + k) % n_points
            i_minus_k = (i_idx - k) % n_points
            gradient[i_idx] = (1 - h[i_plus_k]) - h[i_minus_k]
