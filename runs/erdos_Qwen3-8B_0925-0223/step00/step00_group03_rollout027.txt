import numpy as np
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Assume initial_h_values is available as a global variable
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()

    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    # Parameters
    temp = 1.0
    cooling_rate = 0.995
    perturbation_std = 0.05
    iter_max = 10000
    no_impr = 0
    early_stop = 100
    
    for _ in range(iter_max):
        # Select indices to perturb
        num_perturb = np.random.randint(1, 4)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Generate random perturbation
        delta = np.random.randn(num_perturb) * perturbation_std

        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        scale = (n_points / 2) / sum_h_new
        h_new = h_new * scale
        h_new = np.clip(h_new, 0, 1)
        
        # Compute C5 for new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        current_c5 = np.max(corr_new) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            no_impr = 0
        else:
            # Simulated annealing acceptance
            prob = np.exp(-(current_c5 - best_c5) / temp)
            if np.random.rand() < prob:
                best_c5 = current_c5
                best_h = h_new.copy()
                no_impr = 0
            else:
                no_impr += 1
        
        # Cool temperature
        temp *= cooling_rate
        
        # Early stopping
        if no_impr >= early_stop:
            break
    
    # Final normalization and evaluation
    h_values = np.clip(best_h, 0, 1)
    sum_h_val = np.sum(h_values)
    scale_final = (n_points / 2) / sum_h_val

    h_values = np.clip(h_values, 0, 1)
    
    corr_final = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr_final) * dx
