import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        scale_factor = 0
    else:
        scale_factor = (n_points / 2) / sum_h
    h *= scale_factor
    
    def compute_C5(h, n):
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    current_C5 = compute_C5(h, n_points)
    best_h = h.copy()
    best_C5 = current_C5
    start_time = time.time()
    T_initial = 1.0
    
    while (time.time() - start_time) < budget_s:
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        perturbed_h = h.copy()
        perturbed_h[indices] += np.random.normal(0, 0.1, size=len(indices))
        perturbed_h = np.clip(perturbed_h, 0, 1)
        sum_perturbed = np.sum(perturbed_h)
        if sum_perturbed == 0:
            perturbed_h.fill(0.5)
        else:
            scale_factor_p = (n_points / 2) / sum_perturbed
            perturbed_h *= scale_factor_p
        C5_perturbed = compute_C5(perturbed_h, n_points)
        delta = C5_perturbed - current_C5
        T = T_initial * (1 - (time.time() - start_time)/budget_s)

            current_C5 = C5_perturbed
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
