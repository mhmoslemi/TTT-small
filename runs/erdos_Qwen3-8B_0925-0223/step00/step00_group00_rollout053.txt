        if abs(current_c5 - prev_best) < improvement_threshold:
            early_stop_count += 1
            if early_stop_count >= early_stop:
                break
        else:
            early_stop_count = 0
        prev_best = current_c5

    h_sum = np.sum(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    if h_sum > 0:
        best_h *= (target_sum / h_sum)
    corr_best = np.correlate(best_h, 1 - best_h, mode="full")
    c5_bound = np.max(corr_best) * dx
    return (best_h, c5_bound, n_points)
