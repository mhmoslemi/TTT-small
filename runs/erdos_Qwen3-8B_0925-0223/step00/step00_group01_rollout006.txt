    best_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
    start_time = time.time()
    T_initial = 1.0
    while True:
        current_time = time.time() - start_time
        if current_time >= budget_s:
            break
        indices = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        noise = np.random.normal(0, 0.1, size=len(indices))
        perturbed = current_h.copy()
        perturbed[indices] += noise
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed == 0:
            continue
        scale = (n_points / 2) / sum_perturbed
        perturbed *= scale
        new_c5 = np.max(np.correlate(perturbed, 1 - perturbed, mode='full')) * dx
        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        if new_c5 < current_c5:
            current_h = perturbed.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = perturbed.copy()
        else:
            delta = new_c5 - current_c5
            T = T_initial * (1 - current_time / budget_s)
            if np.random.rand() < np.exp(-delta / T):
                current_h = perturbed.copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = perturbed.copy()
    best_h = np.round(best_h, 6)
    sum_best = np.sum(best_h)
    if not np.isclose(sum_best, n_points / 2, atol=1e-6):
        scale = (n_points / 2) / sum_best
        best_h = best_h * scale
    return (best_h, best_c5, n_points)
