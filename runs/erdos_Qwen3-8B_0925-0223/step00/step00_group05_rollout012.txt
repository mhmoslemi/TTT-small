import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    g = initial_h_values - 0.5
    bounds = [(-0.5, 0.5) for _ in range(n_points)]
    
    def objective(g_flat):
        g = g_flat.reshape(n_points)
        corr = correlate(g, g, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    cons = {'type': 'eq', 'fun': lambda g_flat: np.sum(g_flat.reshape(n_points))}

    result = opt.minimize(
        fun=objective,
        x0=g,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    best_g = result.x
    h_values = best_g + 0.5
    corr = correlate(best_g, best_g, mode='full')
    max_corr = np.max(corr)
    c5_bound = max_corr * dx
    
    # Additional checks and fallback
    if np.isnan(c5_bound):
        h_values = initial_h_values
        corr = correlate(initial_h_values - 0.5, initial_h_values - 0.5, mode='full')
        max_corr = np.max(corr)
        c5_bound = max_corr * dx
    
    return (h_values, c5_bound, n_points)
