import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def objective(h):
        h_1_minus_h = 1 - h
        corr = np.correlate(h, h_1_minus_h, mode='full')
        return np.max(corr) * dx
    
    constraints = [{'type': 'eq', 'fun': constraint}]
    
    result = minimize(objective, initial_h, bounds=bounds, constraints=constraints, method='SLSQP')
    
    best_h = result.x if result.success else initial_h
    c5 = objective(best_h)
    
    return (best_h, c5, n_points)
