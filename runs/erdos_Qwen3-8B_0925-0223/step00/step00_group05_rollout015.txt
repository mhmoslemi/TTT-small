import numpy as np
import scipy.optimize as sco
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    initial_h = initial_h_values.copy()
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    if total > 0:
        initial_h *= (n_points / 2) / total
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.inf
        scale = (n_points / 2) / total
        h_scaled = h_clipped * scale
        return evaluate_erdos_solution(h_scaled)

    bounds = [(0.0, 1.0)] * n_points
    start_time = time.time()
    result = sco.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=30,
        mutation=(0.5, 1.0),
        recombination=0.8,
        maxiter=100,
        tol=1e-6,
        polish=True,
        seed=seed
    )
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (initial_h, evaluate_erdos_solution(initial_h), n_points)

    best_h = result.x
    def refine_objective(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.inf
        scale = (n_points / 2) / total
        h_scaled = h_clipped * scale
        return evaluate_erdos_solution(h_scaled)

    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    result_refine = sco.minimize(
        refine_objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50}
    )
    best_h_refine = result_refine.x

    best_h_final = np.clip(best_h_refine, 0, 1)
    total = np.sum(best_h_final)
    if total > 0:
        best_h_final *= (n_points / 2) / total
    c5_bound = evaluate_erdos_solution(best_h_final)
    return (best_h_final, c5_bound, n_points)
