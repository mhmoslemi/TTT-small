        lags = get_lags(n_points)
        max_corr = 0.0
        for lag in lags:
            idx = (n_points - 1) + lag
            if 0 <= idx < len(corr):
                val = corr[idx] * dx
                if val > max_corr:
                    max_corr = val
        sum_penalty = abs(np.sum(h) - n_points / 2)
        bounds_penalty = np.sum(np.maximum(0, h - 1) + np.maximum(0, 1 - h))
        return max_corr + lambda_ * sum_penalty + mu * bounds_penalty

    start_time = time()
    for _ in range(max_iter):
        for i in range(n_particles):
            h = particles[i].copy()
            current_fitness = compute_fitness(h)
            if current_fitness < personal_best[i]:
                personal_best[i] = current_fitness
                personal_best_h[i] = h.copy()
                if current_fitness < best_c5:
                    best_c5 = current_fitness
                    global_best = h.copy()
            # Update velocity and position
            velocity = velocities[i].copy()
            r1 = random.random()
            r2 = random.random()
            velocity = w * velocity + c1 * r1 * (personal_best_h[i] - h) + c2 * r2 * (global_best - h)
            velocity = np.clip(velocity, -v_max, v_max)
            h_new = h + velocity
            h_new = np.clip(h_new, 0, 1)
            sum_h = np.sum(h_new)
            if sum_h == 0:
                h_new = np.zeros(n_points)
            else:
                h_new = h_new * (n_points / 2) / sum_h
            particles[i] = h_new
            velocities[i] = velocity
        # Early termination if time is up
        if time() - start_time > budget_s:
            break

    # Final evaluation of best solution
    final_corr = np.correlate(global_best, 1 - global_best, mode='full')
    final_max = np.max(final_corr) * dx

    return (global_best, final_max, n_points)
