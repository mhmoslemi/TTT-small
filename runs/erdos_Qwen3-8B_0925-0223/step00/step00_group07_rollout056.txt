import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Adjust initial_h to target sum and bounds
    current_sum = np.sum(initial_h)
    scale = target_sum / current_sum if current_sum != 0 else 1.0
    adjusted_h = np.clip(initial_h * scale, 0, 1)
    adjusted_sum = np.sum(adjusted_h)
    if adjusted_sum != 0:
        adjusted_h = adjusted_h * (target_sum / adjusted_sum)
    adjusted_h = np.clip(adjusted_h, 0, 1)

    # Objective function
    def objective(h):
        h = np.array(h)
        ones_minus_h = 1 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx

    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess
    initial_guess = adjusted_h.tolist()

    # Optimization
    start_time = time.time()
    result = minimize(objective, initial_guess, method='Nelder-Mead', bounds=bounds, constraints=cons, options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 100})
    elapsed = time.time() - start_time

    if elapsed >= budget_s:
        h_values = adjusted_h
        c5_bound = objective(adjusted_h)
        return (h_values, c5_bound, n_points)

    # Post-process result
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h)
    if best_sum != 0:
        best_h *= target_sum / best_sum
    best_h = np.clip(best_h, 0, 1)
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
