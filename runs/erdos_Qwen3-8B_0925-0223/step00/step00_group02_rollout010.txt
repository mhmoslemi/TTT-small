import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    target_sum = n_points / 2
    h = h / np.sum(h) * target_sum

    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n_points)
    T = 1000.0
    alpha = 0.99
    max_iter = 5000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.1, 0.1)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h / np.sum(new_h) * target_sum
        new_c5 = compute_c5(new_h, n_points)
        if new_c5 < best_c5:
            h = new_h
            best_h = new_h
            best_c5 = new_c5
        else:
            prob = np.exp((best_c5 - new_c5) / T)
            if np.random.rand() < prob:
                h = new_h
        T *= alpha

    return (best_h, best_c5, n_points)
