import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    # Validate and rescale initial_h
    initial_h = np.clip(initial_h, 0, 1)
    total_sum = np.sum(initial_h)
    scale_factor = (n_points / 2) / total_sum if total_sum != 0 else 1.0
    initial_h *= scale_factor
    best_h = initial_h.copy()
    dx = 2.0 / n_points
    elapsed_time = 0
    max_iterations = int(budget_s / 2)
    
    for _ in range(max_iterations):
        cross_corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_lag_index = np.argmax(cross_corr)
        k = worst_lag_index - (n_points - 1)
        contributions = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            contributions[i] = best_h[i] * (1 - best_h[j]) * dx
        sorted_indices = np.argsort(contributions)[::-1]
        selected_indices = sorted_indices[:int(0.4 * n_points)]
        fixed_indices = np.setdiff1d(np.arange(n_points), selected_indices)
        fixed_sum = np.sum(best_h[fixed_indices])
        required_sum = (n_points / 2) - fixed_sum
        
        sub_bounds = [(0, 1) for _ in range(len(selected_indices))]
        def objective(h_sub):
            temp_h = best_h.copy()
            temp_h[selected_indices] = h_sub
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += temp_h[i] * (1 - temp_h[j]) * dx
            return overlap
        
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - required_sum})
        try:
            result = opt.minimize(objective, best_h[selected_indices], bounds=sub_bounds, constraints=cons, method='SLSQP')
            if result.success:
                new_h_sub = result.x
                temp_h = best_h.copy()
                temp_h[selected_indices] = new_h_sub
                temp_h = np.clip(temp_h, 0, 1)
                total_sum = np.sum(temp_h)
                if total_sum != 0:
                    scale_factor = (n_points / 2) / total_sum
                    temp_h *= scale_factor
                else:
                    temp_h = np.zeros(n_points)
                cross_corr = np.correlate(temp_h, 1 - temp_h, mode='full')
                new_c5 = np.max(cross_corr * dx)
                if new_c5 < np.max(cross_corr * dx):
                    best_h = temp_h.copy()
        except:
            pass
        
        if np.random.rand() < 0.1:
            perturb_indices = np.random.choice(n_points, int(0.05 * n_points), replace=False)
            perturb = np.random.choice([-0.1, 0.1], size=len(perturb_indices))
            new_h = best_h.copy()
            new_h[perturb_indices] += perturb
            new_h = np.clip(new_h, 0, 1)

        elapsed_time += 2
        if elapsed_time >= budget_s:

    cross_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(cross_corr * dx)
    return best_h, final_c5, n_points
