def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    scale = (n_points / 2.0) / initial_sum

        for i in range(num_particles):

            corr = np.correlate(h, one_minus_h, mode='full')

            max_corr = np.max(corr[lags])
