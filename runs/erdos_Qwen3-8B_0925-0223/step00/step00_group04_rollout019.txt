import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in kwargs:
        initial_h = kwargs['initial_h_values']
    else:
        n_points = 50
        initial_h = np.ones(n_points) * 0.5
        initial_h[::2] = 0.0
        initial_h[1::2] = 1.0
        initial_h = np.clip(initial_h, 0, 1)
        total = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2.0) / total
    best_h = initial_h.copy()
    n_points = len(best_h)
    dx = 2.0 / n_points
    elapsed = 0.0
    best_c5 = float('inf')
    start_time = time.time()

    def compute_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5_values = cross_corr * dx
        return np.max(c5_values)

    def optimize_step(h, worst_lag_index):
        k = worst_lag_index - (n_points - 1)
        valid_i = [i for i in range(n_points) if 0 <= i + k < n_points]
        if not valid_i:
            return h
        # Select top 40% contributing indices
        contributions = np.array([h[i] * (1 - h[i + k]) for i in valid_i])
        idxs = np.argsort(contributions)[::-1]
        selected = valid_i[idxs[:int(0.4 * len(valid_i))]]
        selected_set = set(selected)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        def obj(x):
            h_new = h.copy()
            for i, val in zip(selected, x):
                h_new[i] = val
            total = np.sum(h_new)
            if abs(total - n_points / 2.0) > 1e-6:
                return float('inf')
            k_val = worst_lag_index - (n_points - 1)
            overlap = 0.0
            for i in valid_i:
                i_plus_k = i + k_val
                if 0 <= i_plus_k < n_points:
                    overlap += h_new[i] * (1 - h_new[i_plus_k])
            return overlap * dx
        result = minimize(obj, h[selected], method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h_new = h.copy()
            h_new[selected] = result.x
            h_new = np.clip(h_new, 0.0, 1.0)
            total = np.sum(h_new)
            if abs(total - n_points / 2.0) > 1e-6:
                h_new = h.copy()
                h_new[selected] = result.x
                h_new = np.clip(h_new, 0.0, 1.0)

                scale = (n_points / 2.0) / total
