import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    desired_sum = n_points / 2
    dx = 2.0 / n_points

    current_sum = np.sum(h)
    scale = desired_sum / current_sum
    h = h * scale
    h = np.clip(h, 0, 1)
    current_h = h
    best_h = h.copy()
    best_c5 = compute_c5(current_h, dx)

    start_time = time.time()
    T_initial = 1.0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        indices = np.random.choice(n_points, size=int(n_points // 2), replace=False)
        noise = np.random.normal(0, 0.1, n_points)
        perturbed_h = current_h.copy()
        perturbed_h[indices] += noise[indices]
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_sum = np.sum(perturbed_h)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
