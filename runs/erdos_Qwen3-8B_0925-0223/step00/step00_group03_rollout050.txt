import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-6:
        h *= target_sum / current_sum
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    temp = 1.0
    cooling_rate = 0.995
    perturbation_std = 0.05
    start_time = time.time()
    while time.time() - start_time < budget_s:
        num_perturb = np.random.randint(1, 4)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        delta = np.random.randn(num_perturb) * perturbation_std
        h[indices] += delta
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h *= target_sum / current_sum
        corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5)/temp):
            best_h = h.copy()
            best_c5 = new_c5
        temp *= cooling_rate
        if temp < 1e-5:
            break
    return (best_h.tolist(), best_c5, n_points)
