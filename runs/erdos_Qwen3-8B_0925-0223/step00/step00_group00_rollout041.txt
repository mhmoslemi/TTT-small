    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    N_particles = 20
    max_iter = 300
    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    particles = []
    velocities = []
    initial_h = initial_h_values.copy()
    for _ in range(N_particles - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        scale = (n_points / 2) / h_sum
        h = h * scale
        particles.append(h)
    particles.append(initial_h)
    velocities = [np.zeros(n_points) for _ in range(N_particles)]
    pbest_positions = [np.copy(p) for p in particles]
    pbest_values = [np.inf] * N_particles
    global_best_pos = np.copy(particles[0])
    global_best_val = np.inf
    for i in range(N_particles):
        h = particles[i]
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        pbest_values[i] = c5
        if c5 < global_best_val:
            global_best_val = c5
            global_best_pos = h.copy()
    start_time = time.time()
    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for p in range(N_particles):
            h = particles[p]
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            if c5 < pbest_values[p]:
                pbest_values[p] = c5
                pbest_positions[p] = h.copy()
            if c5 < global_best_val:
                global_best_val = c5
                global_best_pos = h.copy()
        for p in range(N_particles):
            r1 = random.random()
            r2 = random.random()
            velocities[p] = omega * velocities[p] + c1 * r1 * (pbest_positions[p] - particles[p]) + c2 * r2 * (global_best_pos - particles[p])
            new_pos = particles[p] + velocities[p]
            new_pos = np.clip(new_pos, 0, 1)
            sum_new = np.sum(new_pos)
            if sum_new == 0:
                new_pos = np.zeros_like(new_pos)
                scale = (n_points / 2) / (sum_new + 1e-10)
            else:
                scale = (n_points / 2) / sum_new
            new_pos *= scale
            particles[p] = new_pos.copy()
    best_h = global_best_pos.copy()
    return (best_h, global_best_val, n_points)
