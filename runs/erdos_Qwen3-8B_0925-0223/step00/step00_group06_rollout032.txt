import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    desired_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    learning_rate = 0.3
    decay_rate = 0.99
    max_iter = 500
    start_time = time.time()
    improved = True
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cross_corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += 1 - h[j + k]
            if 0 <= j - k < n_points:
                grad[j] -= h[j - k]
        # Update h
        h -= learning_rate * grad
        # Clip and normalize
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:

        # Evaluate new C5
