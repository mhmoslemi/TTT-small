    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_points = len(initial_h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h *= scale
        initial_h = h
    N_particles = 20
    max_iter = 300
    omega = 0.7
    c1, c2 = 1.5, 1.5
    particles = [initial_h.copy() + np.random.normal(0, 0.05, n_points) for _ in range(N_particles)]
    velocities = [np.zeros(n_points) for _ in range(N_particles)]
    pbest = [h.copy() for h in particles]
    pbest_vals = [np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points) for h in particles]
    gbest = particles[0].copy()
    gbest_val = pbest_vals[0]
    start_time = time.time()
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        for j in range(N_particles):
            h = particles[j]
            v = velocities[j]
            r1, r2 = np.random.rand(), np.random.rand()
            new_v = omega * v + c1 * r1 * (pbest[j] - h) + c2 * r2 * (gbest - h)
            velocities[j] = new_v
            h += new_v
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            h *= (n_points / 2) / sum_h
            particles[j] = h
            corr = np.correlate(h, 1 - h, mode='full')
            current_val = np.max(corr) * (2.0 / n_points)
            if current_val < pbest_vals[j]:
                pbest[j] = h.copy()
                pbest_vals[j] = current_val
            if current_val < gbest_val:
                gbest = h.copy()
                gbest_val = current_val
    return (gbest, gbest_val, n_points)
