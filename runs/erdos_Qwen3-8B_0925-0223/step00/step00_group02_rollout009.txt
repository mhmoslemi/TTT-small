import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    # Access global initial_h_values
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    # Normalize initial sum
    h_sum = np.sum(h)
    if not np.isclose(h_sum, n_points / 2, atol=1e-6):
        h = h * (n_points / 2) / h_sum
    best_h = h.copy()
    best_c5 = np.inf
    # Optimization parameters
    learning_rate = 0.005
    epsilon = 1e-5
    max_iterations = 1000
    # Gradient descent loop
    for _ in range(max_iterations):
        # Compute current C5
        cross_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(cross_corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Compute gradient using finite differences
        gradient = np.zeros_like(h)
        for i in range(n_points):
            # Perturb up
            h_plus = h.copy()
            h_plus[i] += epsilon
            cross_corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(cross_corr_plus) * dx
            # Perturb down
            h_minus = h.copy()
            h_minus[i] -= epsilon
            cross_corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(cross_corr_minus) * dx
            # Approximate gradient
            gradient[i] = (c5_plus - c5_minus) / (2 * epsilon)
        # Update h
        h -= learning_rate * gradient
        # Clamp to [0,1] and normalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
