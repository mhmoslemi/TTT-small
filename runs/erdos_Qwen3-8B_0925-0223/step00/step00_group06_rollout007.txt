        h *= scale

    dx = 2.0 / n_points
    best_h = h.copy()

    iters = 0

    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        cross_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(np.abs(cross_corr)) * dx
        
        # Find top k indices
        top_indices = np.argsort(np.abs(cross_corr))[-5:]
        # Compute gradients for these k's
        grad = np.zeros(n_points)
        for idx in top_indices:
            k = idx - (n_points - 1)
            # Compute gradient for this k
            for i in range(n_points):
                if i + k < n_points:
                    grad[i] += (1 - h[i + k])
                if i - k >= 0:
                    grad[i] -= h[i - k]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-6:
            grad /= grad_norm
        
        # Update h
        h -= eta * grad
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)

            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Update best solution
        if current_c5 < best_c5:
