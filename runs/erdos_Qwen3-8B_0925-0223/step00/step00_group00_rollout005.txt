                h_new = np.zeros_like(h_new) + 0.5
            particles[i] = h_new
            velocities[i] = v
            current_val = evaluate_erdos_solution(particles[i], n_points)
            if current_val < pbest_vals[i]:
                pbest_vals[i] = current_val
                pbest[i] = particles[i].copy()
                if current_val < gbest_val:
                    gbest_val = current_val
                    gbest = particles[i].copy()

    best_h = gbest
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h, best_c5, len(best_h))
