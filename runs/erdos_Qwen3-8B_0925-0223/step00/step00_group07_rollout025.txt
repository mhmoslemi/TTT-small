import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    # Use global initial_h_values as provided
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Normalize initial values to meet sum constraint
    sum_initial = np.sum(initial_h_values)
    scale = (n_points / 2) / sum_initial
    h = initial_h_values * scale
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    
    # Define objective function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Define constraint for sum of h
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use SLSQP method with numerical gradient
    result = minimize(
        fun=objective,
        x0=h.flatten(),
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    scale_factor = (n_points / 2) / sum_best
    best_h = best_h * scale_factor
    best_h = np.clip(best_h, 0, 1)
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    return (best_h, best_c5, n_points)
