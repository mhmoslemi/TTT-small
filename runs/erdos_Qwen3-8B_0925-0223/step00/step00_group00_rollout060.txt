    else:
        particles = [np.copy(initial_h)]
        velocities = [np.zeros_like(initial_h)]
        for _ in range(num_particles - 1):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h
            particles.append(h)
            velocities.append(np.random.rand(n_points) * 0.1)

    pbest = [np.copy(h) for h in particles]
    pbest_values = [evaluate_C5(h) for h in particles]
    gbest = pbest[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)
    best_h = gbest.copy()
    best_value = gbest_value

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(num_particles):
            h = particles[i]
            curr_value = evaluate_C5(h)
            if curr_value < pbest_values[i]:
                pbest[i] = np.copy(h)
                pbest_values[i] = curr_value
        current_min = np.min(pbest_values)
        if current_min < gbest_value:
            gbest_value = current_min
            gbest = pbest[np.argmin(pbest_values)]
        for i in range(num_particles):
            h = particles[i]
            v = velocities[i]
            r1, r2 = np.random.rand(2)
            v_new = omega * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            velocities[i] = v_new
            h_new = h + v_new
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = np.sum(h_new)
            if sum_h_new > 0:
                h_new = h_new * (n_points / 2) / sum_h_new
            else:
                h_new = np.random.rand(n_points)
                h_new = np.clip(h_new, 0, 1)
                sum_h_new = np.sum(h_new)
                h_new = h_new * (n_points / 2) / sum_h_new
            particles[i] = h_new
        if gbest_value < best_value:
            best_value = gbest_value
            best_h = gbest.copy()

    return (best_h, best_value, n_points)
