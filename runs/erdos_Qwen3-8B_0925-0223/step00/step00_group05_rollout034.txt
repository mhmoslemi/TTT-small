    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = np.clip(initial_h_values, 0, 1)
    S = np.sum(h)
    if S == 0:
        initial_guess = np.zeros(n_points)
    else:
        scale = (n_points / 2) / S
        scaled_initial = np.clip(h * scale, 0, 1)
        initial_guess = scaled_initial
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        S = np.sum(h)
        if S == 0:

        scale = (n_points / 2) / S
        scaled_h = np.clip(h * scale, 0, 1)
        corr = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        return np.max(corr) * dx
    
    result = differential_evolution(

        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
