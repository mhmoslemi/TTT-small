import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 20
    max_iter = 300
    omega = 0.7
    c1, c2 = 1.5, 1.5
    
    # Initialize positions
    positions = [initial_h.copy()]
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        total = np.sum(h)
        scale = (n_points / 2) / total
        h = h * scale
        positions.append(h)
    positions = np.array(positions)
    
    # Initialize velocities
    velocities = np.random.rand(n_particles, n_points)
    
    # Best positions and values
    pbest_positions = positions.copy()
    pbest_values = np.zeros(n_particles)
    gbest_position = positions[0].copy()
    gbest_value = float('inf')
    
    # Evaluate initial fitness
    for i in range(n_particles):
        h = positions[i]
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        pbest_values[i] = c5
        if c5 < gbest_value:
            gbest_value = c5
            gbest_position = h.copy()
    
    best_h = gbest_position
    best_c5 = gbest_value
    
    for _ in range(max_iter):
        for i in range(n_particles):
            h = positions[i]
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = h.copy()
            
            if c5 < gbest_value:
                gbest_value = c5
                gbest_position = h.copy()
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest_positions[i] - positions[i]) + c2 * r2 * (gbest_position - positions[i])
            positions[i] += velocities[i]
            
            # Clip and rescale
            positions[i] = np.clip(positions[i], 0, 1)
            total = np.sum(positions[i])
            if total == 0:
                positions[i] = np.zeros_like(positions[i])
            else:
                scale = (n_points / 2) / total
                positions[i] = positions[i] * scale
        
        if gbest_value < best_c5:
            best_c5 = gbest_value
            best_h = gbest_position
    
    return (best_h, best_c5, n_points)
