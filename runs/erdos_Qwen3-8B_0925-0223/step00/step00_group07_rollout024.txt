import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy')  # Assuming provided initial configuration
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        sum_penalty = 100 * (np.sum(h) - target_sum) ** 2
        bounds_penalty = np.sum(np.maximum(0, h - 1) + np.maximum(0, -h))
        return c5 + sum_penalty + bounds_penalty

    def constraint_sum(h):
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(objective, initial_h_values, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
    best_h = np.clip(result.x, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
