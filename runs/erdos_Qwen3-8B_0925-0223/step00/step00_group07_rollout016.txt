import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    scale_factor = target_sum / np.sum(h)
    h = np.clip(h * scale_factor, 0, 1)
    
    def obj(x):
        h_c = np.clip(x, 0, 1)
        corr = np.correlate(h_c, 1 - h_c, mode='full')
        max_corr = np.max(corr) * dx

        sum_pen = 1e4 * (s - target_sum) ** 2
        bound_pen = 1e4 * np.sum(np.maximum(h_c - 1, 0) + np.maximum(0 - h_c, 0))
        return max_corr + sum_pen + bound_pen
    
    bounds = [(0,1)] * n_points
    result = optimize.minimize(obj, h, method='Nelder-Mead', bounds=bounds, options={'maxiter': 100})
    best_h = np.clip(result.x, 0, 1)
    best_sum = np.sum(best_h)
    best_h = best_h * (target_sum / best_sum)
    best_h = np.clip(best_h, 0, 1)
    
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
