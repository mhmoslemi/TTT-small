        scale = (n_points / 2) / sum_h
        h_values *= scale
        sum_h = n_points / 2

    best_c5 = float('inf')
    best_h = h_values.copy()

    start_time = time.time()
    max_iterations = 10000
    step_size = 0.05
    decay_rate = 0.9999

    for iteration in range(max_iterations):
        corr = np.correlate(h_values, 1 - h_values, mode='full') * dx
        current_c5 = np.max(corr)
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        m = np.argmax(corr)
        k_max = m - (n_points - 1)

        gradient = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k_max < n_points:
                gradient[j] += 1 - h_values[j + k_max]
            if 0 <= j - k_max < n_points:
                gradient[j] -= h_values[j - k_max]

        step = step_size * (-gradient)
        h_values += step
        h_values = np.clip(h_values, 0.0, 1.0)

        sum_h = np.sum(h_values)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h_values *= scale

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()

        step_size *= decay_rate

    sum_h = np.sum(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr)
    return (best_h, c5_bound, n_points)
