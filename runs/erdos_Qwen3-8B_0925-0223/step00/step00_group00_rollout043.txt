        else:
            no_improvement += 1
        if iteration % 50 == 0 and iteration != 0:
            learning_rate *= decay_rate
        if no_improvement >= 20:
            break
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    return (best_h.tolist(), final_c5, n_points)
