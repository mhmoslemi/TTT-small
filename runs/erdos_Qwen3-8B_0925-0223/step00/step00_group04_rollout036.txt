    def compute_c5(h):
        max_ov = 0.0
        for k in range(n_points):
            shifted = np.roll(h, k)
            ov = np.sum(h * (1 - shifted)) * dx
            max_ov = max(max_ov, ov)
        return max_ov

    def objective(h, k):
        shifted = np.roll(h, k)
        return np.sum(h * (1 - shifted)) * dx

    def get_max_k(h):
        max_ov = 0.0
        max_k = 0
        for k in range(n_points):
            shifted = np.roll(h, k)
            ov = np.sum(h * (1 - shifted)) * dx
            if ov > max_ov:
                max_ov = ov
                max_k = k
        return max_k

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Initial checks
