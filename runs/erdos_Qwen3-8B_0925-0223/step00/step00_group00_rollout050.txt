import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    
    if initial_h_values is None:
        n_points = 200
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        scale = (n_points / 2) / h_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        initial_h_values = h
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    T_initial = 1.0
    T_final = 0.01
    cooling_rate = 0.995
    iteration = 0
    start_time = time.time()
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        T = T_initial * (cooling_rate ** iteration)
        if T < T_final:
            break
        
        i = np.random.randint(n_points)
        delta = np.random.uniform(-0.05, 0.05)
        new_h = h.copy()
        new_h[i] += delta
        new_h[i] = np.clip(new_h[i], 0.0, 1.0)
        S_new = np.sum(new_h)
        if S_new != 0:
            scale_factor = (n_points / 2) / S_new
            new_h *= scale_factor
            new_h = np.clip(new_h, 0.0, 1.0)
        
        new_c5 = compute_c5(new_h)
        
        if new_c5 < current_c5:
            h = new_h
            current_c5 = new_c5
        else:
            prob = np.exp((current_c5 - new_c5) / T)
            if random.random() < prob:
                h = new_h
                current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        iteration += 1
    
    return best_h, best_c5, n_points
