def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    n_points = len(h)
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2, atol=1e-6):
        h = h * (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed = 0
    iteration = 0
    no_improv = 0
    while elapsed < budget_s:
        t0 = time()
        cross_corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_lag_idx = np.argmax(cross_corr)
        k = worst_lag_idx - (n_points - 1)
        contributions = [best_h[i] * (1 - best_h[(i + k) % n_points]) for i in range(n_points)]
        sorted_indices = np.argsort(contributions)[::-1]
        top_25 = int(0.25 * n_points)
        sel_indices = sorted_indices[:top_25]
        selected_mask = np.zeros(n_points, dtype=bool)
        selected_mask[sel_indices] = True
        non_sel_mask = ~selected_mask
        target_sum = (n_points / 2) - np.sum(best_h[non_sel_mask])
        x0 = best_h[sel_indices].copy()
        shifted = [(i + k) % n_points for i in sel_indices]
        weights = 1 - best_h[shifted]
        def obj(x):
            return np.dot(x, weights)
        bounds = [(0., 1.) for _ in range(len(sel_indices))]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            best_h[sel_indices] = result.x
        new_c5 = evaluate_erdos_solution(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            no_improv = 0
        else:
            no_improv += 1
        elapsed += time() - t0
        if no_improv >= 20:
            perturb = np.random.choice(np.arange(n_points), size=int(0.1 * n_points), replace=False)
            best_h[perturb] = np.clip(best_h[perturb] + np.random.uniform(-0.1, 0.1), 0, 1)
    return best_h, best_c5, n_points
