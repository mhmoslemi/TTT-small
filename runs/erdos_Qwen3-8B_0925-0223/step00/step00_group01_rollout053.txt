def run(seed=42, budget_s=600, **kwargs):

    from scipy.optimize import differential_evolution

        sum_h = np.sum(h_clip)

        return max_corr + penalty
    
    # Differential Evolution setup
    bounds = [(0.0, 1.0) for _ in range(n_points)]
