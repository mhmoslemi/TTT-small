    def objective(h):
        cross = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(cross)
        sum_val = np.sum(h)
        penalty_sum = 100 * abs(sum_val - target_sum)
        penalty_bounds = 100 * (np.sum(np.maximum(h - 1.0, 0.0)) + np.sum(np.maximum(-h, 0.0)))
        return max_c5 * dx + penalty_sum + penalty_bounds

    result = minimize(
        objective,
        h_initial,
        method='Nelder-Mead',
        options={
            'xatol': 1e-6,
            'ftol': 1e-6,
            'maxiter': 1000,
            'disp': False
        }
    )

    best_h = result.x
    cross = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(cross) * dx
    return (best_h, best_c5, n_points)
