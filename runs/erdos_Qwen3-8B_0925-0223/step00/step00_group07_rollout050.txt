import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    current_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2) / current_sum
    h = np.clip(initial_h_values * scale_factor, 0, 1)
    h_sum = np.sum(h)
    if np.abs(h_sum - n_points / 2) > 1e-6:
        h += np.random.uniform(-1e-6, 1e-6, size=n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr[1:-1]) * dx

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': constraint}]
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 1000})
    h_values = result.x
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr[1:-1]) * dx
    return (h_values, c5_bound, n_points)
