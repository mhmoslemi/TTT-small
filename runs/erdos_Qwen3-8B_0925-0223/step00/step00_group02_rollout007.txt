import numpy as np

def run(seed=42, budget_s=600, **kwargs):

        h = h * (n_points / 2) / total
    best_h = h.copy()
    best_c5 = np.inf
    max_iter = int(budget_s / 0.03)
    no_improvement = 0
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement >= 50:
                break
        lag_idx = np.argmax(overlaps)
        lag = lag_idx - (n_points - 1)
        valid = []
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                valid.append(i)
        contrib = np.array([h[i] * (1 - h[i + lag]) for i in valid])
        top_idx = valid[np.argsort(-contrib)[:5]]
        eps = 0.01 if current_c5 > 0.38 else 0.005
