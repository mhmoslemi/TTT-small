import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    start_time = time.time()
    max_iter = 200
    
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = corr.argmax()
        lag = max_corr_idx - (n_points - 1)
        # Vectorized objective
        def obj(x):
            h_new = x.reshape(n_points)
            shifted = np.roll(h_new, lag)
            return np.dot(h_new, 1 - shifted) * dx
        # Constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        # Optimization
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if res.success:
            h = res.x
            curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return (best_h, best_c5, n_points)
