import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    h = initial_h_values.copy()

        return np.max(corr) * dx

    alpha = 0.995
    max_iter = 10000
    random.seed(seed)
    start_time = time.time()

    for _ in range(max_iter):
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.1, 0.1)
        new_h = h.copy()

        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new == 0:

        new_h = new_h * scale
        new_c5 = compute_c5(new_h, dx)
        current_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:

            best_h = new_h.copy()
