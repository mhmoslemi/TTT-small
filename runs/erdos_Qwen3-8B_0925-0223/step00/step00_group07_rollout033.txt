import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    current_sum = np.sum(h)
    scale = required_sum / current_sum
    h = h * scale
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    def obj(x):
        h_curr = x
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        c5 = np.max(corr) * dx
        sum_penalty = 1000 * np.abs(np.sum(h_curr) - required_sum)
        return c5 + sum_penalty

    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(obj, h, bounds=bounds, method='L-BFGS-B', options={'maxiter': 200})
    if result.success:
        best_h = result.x
    else:
        best_h = h
    best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
