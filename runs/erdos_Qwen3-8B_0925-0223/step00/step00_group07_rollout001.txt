    def obj(x):
        h = np.array(x)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
    
    result = minimize(obj, h_clamped, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
    if not result.success:
        best_h = h_clamped
        best_c5 = obj(best_h)
    else:
        best_h = result.x
        best_c5 = result.fun
    return (best_h, best_c5, n_points)
