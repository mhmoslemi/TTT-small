def run(seed=42, budget_s=600, **kwargs):

    n_points = len(h)
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    scale_factor = (n_points / 2) / current_sum
    h = h * scale_factor
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    learning_rate = 0.01
    time_start = time.time()
    for iteration in range(1000):
        elapsed = time.time() - time_start
        if elapsed > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        corr_middle = corr[1:-1]
        max_corr_idx = np.argmax(corr_middle)
        k = max_corr_idx
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
