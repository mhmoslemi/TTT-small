import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        n_points = len(h_initial)
    else:
        n_points = 70
        h_initial = np.zeros(n_points)
        h_initial[0::2] = 1.0

        scale = -sum_g / n_points
        g_initial += scale
    
    bounds = [(-0.5, 0.5) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x)})
    
    def objective(g):
        corr = np.correlate(g, g, mode='full')
        min_corr = np.min(corr)
        return 0.5 - min_corr * dx
    
    result = opt.minimize(objective, g_initial, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
    
    best_g = result.x
    best_h = best_g + 0.5
