import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    from scipy.optimize import differential_evolution

    if 'initial_h_values' not in globals():
        n_points = 70
        initial_h = np.full(n_points, 0.5)
    else:
        initial_h = initial_h_values
        n_points = len(initial_h)

    target_sum = n_points / 2
    initial_sum = np.sum(initial_h)
    scale_factor = target_sum / initial_sum
    scaled_initial = initial_h * scale_factor

    def objective(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return float('inf')
        scale = (n_points / 2) / total
        scaled_h = h * scale
        corr = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    bounds = [(0.0, 1.0)] * n_points

    result = differential_evolution(
        objective, bounds,
        strategy='best1bin', maxiter=100, popsize=15,
        mutation=(0.5, 1.0), recombination=0.8,
        polish=True, seed=seed, tol=1e-6
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    scale = (n_points / 2) / total if total > 0 else 1.0
    best_h *= scale

    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
