import numpy as np
import scipy.optimize as opt
import scipy.signal as scp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(n_points)
    current_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2) / current_sum
    transformed_h = np.clip(initial_h_values * scale_factor, 0, 1)
    
    def objective(h):
        corr = scp.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1)] * n_points
    result = opt.minimize(objective, transformed_h, method='SLSQP', 
                          constraints={'type': 'eq', 'fun': constraint}, 
                          bounds=bounds, tol=1e-6, options={'maxiter': 1000})
    
    if result.success and np.isfinite(result.fun):
        h_values = np.clip(result.x, 0, 1)
        c5_bound = result.fun
    else:
        h_values = np.clip(transformed_h, 0, 1)
        c5_bound = objective(h_values)

    if not np.isfinite(c5_bound):
        h_values = np.clip(initial_h_values * scale_factor, 0, 1)
        c5_bound = objective(h_values)

    return h_values, c5_bound, n_points
