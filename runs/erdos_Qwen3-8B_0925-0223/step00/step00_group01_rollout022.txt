        perturbed = np.clip(perturbed, 0, 1)
        s = np.sum(perturbed)
        if s == 0:
            scaled = perturbed
        else:
            scaled = perturbed * (n_points / 2) / s
        new_c5 = compute_c5(scaled)
        T = T_initial * (1 - current_time / budget_s)
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h, current_c5 = scaled, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = scaled.copy(), new_c5
    return (best_h, best_c5, n_points)
