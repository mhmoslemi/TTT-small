    total = np.sum(h)
    h = h * (n_points / 2) / total
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    
    learning_rate = 0.1
    decay_rate = 0.99
    max_iter = 500
    no_improvement = 0
    
    for _ in range(max_iter):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        m_max = np.argmax(np.abs(cross_corr))
        k_val = m_max - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k_val < n_points:
                grad[i] += 1 - h[i + k_val]
            if i - k_val >= 0:
                grad[i] -= h[i - k_val]
        
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        total_new = np.sum(h_new)
        if total_new == 0:
            h = h_new
        else:
            h = h_new * (n_points / 2) / total_new
        
        current_cross_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(np.abs(current_cross_corr)) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if _ % 100 == 0 and _ != 0:
            learning_rate *= decay_rate
        
        if no_improvement >= 50:
