from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Pre-imported initial_h_values is a global variable
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.inf
        scaling_factor = target_sum / sum_h
        h_normalized = np.clip(h_clipped * scaling_factor, 0, 1)
        cross_corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(cross_corr * dx)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    
    result = differential_evolution(
        objective,
        bounds,
        popsize=15,
        maxiter=200,
        tol=1e-6,
        seed=seed,
        strategy='best1bin'
    )
    
    best_h = result.x
    best_h_clipped = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h_clipped)
    if sum_best == 0:
        best_h_normalized = np.zeros_like(best_h)
    else:
        scaling_factor = target_sum / sum_best
        best_h_normalized = np.clip(best_h_clipped * scaling_factor, 0, 1)
    
    cross_corr = np.correlate(best_h_normalized, 1 - best_h_normalized, mode='full')
    final_c5 = np.max(cross_corr * dx)
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (best_h_normalized.tolist(), final_c5, n_points)
    return (best_h_normalized.tolist(), final_c5, n_points)
