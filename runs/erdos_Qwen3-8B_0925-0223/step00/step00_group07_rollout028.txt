        h = h * scale_factor
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= 0.99
    
    h = scipy.ndimage.median_filter(best_h, size=5)
    current_sum = np.sum(h)
    if current_sum != 0:
        scale_factor = (n_points / 2) / current_sum
    else:
        scale_factor = 1.0
    h = h * scale_factor
    h = np.clip(h, 0, 1)
    current_c5 = compute_c5(h)
    return h, current_c5, n_points
