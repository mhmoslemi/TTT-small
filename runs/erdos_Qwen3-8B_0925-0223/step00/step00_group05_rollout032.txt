import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    def obj(h):
        h_array = np.array(h)
        one_minus_h = 1.0 - h_array
        corr = np.correlate(h_array, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    start = time.time()
    res = scipy.optimize.minimize(obj, initial_h_values, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 100})
    
    if time.time() - start > budget_s:
        return (initial_h_values, obj(initial_h_values), n_points)
    
    if res.success:
        h_vals = res.x
        c5 = obj(h_vals)
        assert np.allclose(np.sum(h_vals), n_points / 2, atol=1e-6)
        return (h_vals, c5, n_points)
    else:
        return (initial_h_values, obj(initial_h_values), n_points)
