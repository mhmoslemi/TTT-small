def run(seed=42, budget_s=600, **kwargs):

        scale = S / np.sum(h)
        h *= scale
        h = np.clip(h, 0, 1)
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)
    best_C5 = compute_C5(h)
    best_h = h.copy()

    cooling_rate = 0.995
    T_final = 0.01

        delta = np.random.uniform(-0.1, 0.1)
        new_h = h.copy()
        new_h[idx] += delta
