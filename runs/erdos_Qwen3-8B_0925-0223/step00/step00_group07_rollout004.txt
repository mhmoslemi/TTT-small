    learning_rate = 0.01
    max_iter = 1000
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        corr_vals = corr[1:-1]
        if len(corr_vals) == 0:
            break
        max_corr_idx = np.argmax(corr_vals)
        k = max_corr_idx
        idx_plus = (np.arange(n_points) + k) % n_points
        idx_minus = (np.arange(n_points) - k) % n_points
        grad = (1 - h[idx_plus]) - h[idx_minus]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            scale = (n_points / 2.0) / current_sum
        else:
            scale = 1.0
        h *= scale
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    return (best_h, best_c5, n_points)
