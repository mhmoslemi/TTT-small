import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    target_sum = n_points / 2.0
    if not np.isclose(sum_h, target_sum):
        scale = target_sum / sum_h
        h *= scale
    
    num_particles = 50
    max_iter = 200
    c1, c2, w, v_max = 2.0, 2.0, 0.7, 0.1
    lambda_coeff, mu_coeff = 1e4, 1e4
    
    particles = []
    velocities = []
    for _ in range(num_particles):
        perturbation = np.random.randn(n_points) * 0.1
        pos = initial_h_values + perturbation
        pos = np.clip(pos, 0, 1)
        sum_pos = np.sum(pos)
        if sum_pos != 0:
            scale = target_sum / sum_pos
            pos *= scale
        particles.append(pos)
        vel = np.random.uniform(-v_max, v_max, size=n_points)
        velocities.append(vel)
    
    pbest_positions = [np.copy(p) for p in particles]
    pbest_fitness = []
    for i in range(num_particles):
        h = particles[i]
        corr = np.correlate(h, 1 - h, mode='full')
        if n_points >= 100:
            lags_to_check = np.arange(0, len(corr), 50)
        else:
            lags_to_check = np.arange(len(corr))
        c5_val = np.max(corr[lags_to_check]) * dx
        penalty_sum = np.abs(np.sum(h) - target_sum)
        penalty_bounds = np.sum(np.maximum(h - 1, 0)) + np.sum(np.maximum(1 - h, 0))
        fitness = c5_val + lambda_coeff * penalty_sum + mu_coeff * penalty_bounds
        pbest_fitness.append(fitness)
    
    gbest_index = np.argmin(pbest_fitness)
    gbest_pos = np.copy(particles[gbest_index])
    gbest_fitness = pbest_fitness[gbest_index]
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        for i in range(num_particles):
            pos = particles[i]
            vel = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel_new = w * vel + c1 * r1 * (pbest_positions[i] - pos) + c2 * r2 * (gbest_pos - pos)
            vel_new = np.clip(vel_new, -v_max, v_max)

            pos_new = pos + vel_new

            particles[i] = pos_new
