    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        lag_index = np.argmax(corr)
        k = lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_k = (i + k) % n_points
            i_k_minus = (i - k) % n_points
            grad[i] = (1 - h[i_k]) - h[i_k_minus]
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) > 0:
            h *= (target_sum / np.sum(h))
        current_c5 = max_corr * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            early_stop_counter = 0
        else:
            early_stop_counter += 1
        if _ % 50 == 0:
            learning_rate *= 0.99
        if early_stop_counter >= 20:
            break
    return best_h, best_c5, n_points
