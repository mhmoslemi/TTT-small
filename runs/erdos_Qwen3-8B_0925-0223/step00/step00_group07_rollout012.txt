import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    scale = (n_points / 2) / current_sum if current_sum != 0 else 1
    h *= scale
    best_h = h.copy()
    best_c5 = compute_c5(h)
    time_start = time.time()
    iters = 0
    lr = 0.1
    while time.time() - time_start < budget_s:
        iters += 1
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corr[1:-1])
        k = max_idx
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = (i + k) % n_points
            j_minus = (i - k) % n_points
            grad[i] = (1 - h[j_plus]) - h[j_minus]
        grad *= dx
        h -= lr * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale = (n_points / 2) / current_sum if current_sum != 0 else 1
        h *= scale
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        lr *= 0.995
    med = np.median(h.reshape(-1, 2), axis=1)
    best_h = np.repeat(med, 2)
    return (best_h, best_c5, n_points)
