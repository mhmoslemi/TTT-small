    scale = (n_points / 2) / current_sum
    h = h * scale
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != n_points / 2:
        scale = (n_points / 2) / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = 0.0
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    start_time = time.time()
    step_size = 0.1
    iteration = 0
    while (time.time() - start_time) < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        i_max = np.argmax(corr)
        lag = i_max - (n_points - 1)
        grad = np.zeros(n_points)
        for m in range(n_points):
            t1 = 1.0 - h[m + lag] if (m + lag < n_points) else 0.0
            t2 = -h[m - lag] if (m - lag >= 0) else 0.0
            grad[m] = t1 + t2
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            scale = (n_points / 2) / current_sum
            h = h * scale
            h = np.clip(h, 0, 1)
        if np.random.rand() < 0.05:
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if current_sum != 0:
                scale = (n_points / 2) / current_sum
                h = h * scale
                h = np.clip(h, 0, 1)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        step_size *= 0.999
        iteration += 1
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, final_c5, n_points)
