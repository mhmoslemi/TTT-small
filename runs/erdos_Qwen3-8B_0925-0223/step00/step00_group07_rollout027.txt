import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    current_sum = np.sum(h_initial)
    scale_factor = (n_points / 2) / current_sum
    transformed_h = np.clip(h_initial * scale_factor, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr[1:-1])
        return max_corr * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1)] * n_points
    result = minimize(
        fun=objective,
        x0=transformed_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 1000, 'ftol': 1e-6, 'eps': 1e-8}
    )
    
    if result.success and np.all((result.x >= 0) & (result.x <= 1)):
        best_h = result.x
        best_c5 = objective(best_h)
    else:
        best_h = transformed_h
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
