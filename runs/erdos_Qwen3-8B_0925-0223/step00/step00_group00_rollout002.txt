import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Assume initial_h_values is provided
    n_points = len(initial_h_values)
    desired_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    def compute_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    current_c5 = best_c5
    T = 1.0
    cooling_rate = 0.995
    T_final = 0.01
    start_time = time.time()
    while time.time() - start_time < budget_s:
        i = np.random.randint(n_points)
        delta = np.random.uniform(-0.1, 0.1)
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[i] = np.clip(new_h[i], 0.0, 1.0)
        sum_new = np.sum(new_h)
        excess = sum_new - desired_sum
        if excess != 0:
            adjustment = excess / (n_points - 1)
            new_h[~(np.arange(n_points) == i)] -= adjustment
            new_h = np.clip(new_h, 0.0, 1.0)
        new_c5 = compute_c5(new_h)
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        if T < T_final:
            break
        T *= cooling_rate
