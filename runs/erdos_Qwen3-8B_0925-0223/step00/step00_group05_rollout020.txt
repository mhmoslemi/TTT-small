        corr = np.correlate(g_vec, g_vec, mode='full')
        return np.max(corr) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x)}]
    bounds = [(-0.5, 0.5) for _ in range(n_points)]
    
    result = minimize(
        objective,
        g,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    if not result.success:
        h_opt = h_initial
        corr = np.correlate(h_initial - 0.5, h_initial - 0.5, mode='full')
        c5_bound = np.max(corr) * dx
    else:
        h_opt = result.x + 0.5
        corr = np.correlate(result.x, result.x, mode='full')
        c5_bound = np.max(corr) * dx
    
    h_opt = np.clip(h_opt, 0, 1)
    return h_opt, c5_bound, n_points
