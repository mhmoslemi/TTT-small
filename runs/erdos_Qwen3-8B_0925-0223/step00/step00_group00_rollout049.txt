        i = np.random.randint(n_points)
        delta = np.random.uniform(-0.1, 0.1)
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[i] = np.clip(new_h[i], 0, 1)
        new_sum = np.sum(new_h)
        scaling_factor = required_sum / new_sum
        new_h = np.clip(new_h * scaling_factor, 0, 1)
        new_c5 = compute_c5(new_h)
        if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5) / T):
            current_h = new_h.copy()
            current_c5 = new_c5
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        T *= cooling_rate
        if T < T_final:
            break
    return (best_h, best_c5, n_points)
