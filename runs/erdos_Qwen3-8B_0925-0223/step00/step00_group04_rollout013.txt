        if result.success:
            best_h[top_indices] = result.x
            new_c5 = compute_c5(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                improvement_count = 0
            else:
                improvement_count += 1
        else:
            improvement_count += 1
        
        if np.any(best_h < 0) or np.any(best_h > 1):
            best_h[top_indices] = np.clip(result.x, 0, 1)
            new_c5 = compute_c5(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
        
        elapsed_time += time.time() - start_time
        iteration += 1
        
        if improvement_count >= 20:
            perturb_indices = np.random.choice(n_points, int(0.1 * n_points), replace=False)
            best_h[perturb_indices] += np.random.uniform(-0.1, 0.1, size=len(perturb_indices))
            best_h = np.clip(best_h, 0, 1)
            sum_h = np.sum(best_h)
            best_h *= (n_points / 2) / sum_h
            improvement_count = 0
    
    return (best_h, best_c5, n_points)
