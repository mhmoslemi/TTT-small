import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    # Assume initial_h_values is available as a global variable
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    current_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2) / current_sum
    transformed_h = np.clip(initial_h_values * scale_factor, 0, 1)
    transformed_h = np.nan_to_num(transformed_h, nan=0.0, neginf=0.0, posinf=1.0)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    def constraint(h):
        return np.sum(h) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1)] * n_points
    initial_guess = transformed_h

    start_time = time.time()
    max_iter = 1000
    result = minimize(
        objective, initial_guess, method='SLSQP', bounds=bounds, constraints=cons,
        tol=1e-6, options={'maxiter': max_iter, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    elapsed = time.time() - start_time

    if elapsed >= budget_s - 10:

        corr = np.correlate(h_values, 1 - h_values, mode='full')
        c5_bound = np.max(corr[1:-1]) * dx
