import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    
    # Ensure initial solution meets constraints
    h_sum = np.sum(h)

    T_initial = 1.0

    cooling_rate = 0.995

    start_time = time.time()
    
    for step in range(max_iterations):
        if time.time() - start_time > budget_s:

        h_new[idx] += delta
        h_new = np.clip(h_new, 0.0, 1.0)
        
        # Rescale to maintain sum constraint
        h_new_sum = np.sum(h_new)
        scale = (n_points / 2) / h_new_sum if h_new_sum != 0 else 1.0
        h_new *= scale
        
        # Evaluate new C5
        c5_new = compute_c5(h_new, dx)
        
        # Acceptance probability
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
        else:

        T = T_initial * (cooling_rate ** step)
        if T < T_final:

    return best_h, best_c5, n_points

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
