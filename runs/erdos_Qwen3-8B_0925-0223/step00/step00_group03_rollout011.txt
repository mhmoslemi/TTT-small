import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np

    global initial_h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    scale = (n_points / 2) / total
    h = h * scale

    best_h = h.copy()
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)
    start_time = time.time()
    temp = 1.0
    cooling = 0.995
    std_dev = 0.05
    iterations = 0
    max_iter = 10000

    while time.time() - start_time < budget_s and iterations < max_iter:
        num = np.random.randint(1, 4)
        idxs = np.random.choice(n_points, num, replace=False)
        delta = std_dev * np.random.randn(num)
        h_new = h.copy()
        h_new[idxs] += delta
        h_new = np.clip(h_new, 0.0, 1.0)
        total = np.sum(h_new)
        h_new *= (n_points / 2) / total

        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - best_c5) / temp)
            if np.random.rand() < prob:
                h = h_new.copy()
                best_c5 = new_c5

        temp *= cooling
        iterations += 1

    return (h, best_c5, n_points)
