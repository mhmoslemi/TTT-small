import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Scale initial_h to meet sum constraint
    sum_initial = np.sum(initial_h)
    scale_initial = (n_points / 2) / sum_initial if sum_initial else 1.0
    scaled_initial = initial_h * scale_initial
    
    def objective(h):
        h = np.clip(h, 0, 1)
        total_h = np.sum(h)
        if total_h == 0:
            return np.inf
        scale = (n_points / 2) / total_h
        scaled_h = h * scale
        corr = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0)] * n_points
    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   maxiter=100, popsize=15, tol=1e-6,
                                   mutation=(0.5, 1.0), recombination=0.8,
                                   polish=True, disp=False)
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    total_h = np.sum(best_h)
    if total_h != 0:
        best_h *= (n_points / 2) / total_h
    
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
