        for i in top_idx:
            j = j_vals[i]
            if j < 0 or j >= n_points:
                continue
            contrib = h[i] * (1 - h[j])
            if contrib <= 0:
                continue
            max_delta = min(h[i], 1 - h[i]) * (1 - h[i]) / (2 * contrib)
            delta = max(-eps, min(eps, max_delta)) * -1
            h[i] += delta
            h[i] = np.clip(h[i], 0, 1)
        
        total = np.sum(h)
        scale = (n_points / 2) / total
        h *= scale
        h = np.clip(h, 0, 1)
    
    final_corr = np.correlate(h, 1 - h, mode='full') * dx
    final_c5 = np.max(final_corr)
    return (h, final_c5, n_points)
