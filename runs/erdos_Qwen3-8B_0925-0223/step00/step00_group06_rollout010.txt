import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    total = np.sum(h)
    h = h * (n_points / 2) / total
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.inf
    eta = 0.1
    start_time = time.time()
    
    for _ in range(300):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(np.abs(cross_corr))
        k = max_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for m in range(n_points):
            mk = m + k
            if 0 <= mk < n_points:
                grad[m] += (1 - h[mk])
            mk = m - k
            if 0 <= mk < n_points:
                grad[m] -= h[mk]
        
        h = h - eta * grad
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        
        current_c5 = np.max(np.abs(cross_corr)) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        eta *= 0.99
    
    return (best_h, best_c5, n_points)
