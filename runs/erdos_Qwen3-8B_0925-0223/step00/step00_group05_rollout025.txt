import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()

        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
