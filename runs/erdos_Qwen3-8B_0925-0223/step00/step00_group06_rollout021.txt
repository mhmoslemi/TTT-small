    start_time = time.time()
    
    for _ in range(max_iter):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(cross_corr)
        k = m - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            i_minus_k = i - k
            if 0 <= i_plus_k < n_points:
                term1 = 1 - h[i_plus_k]
            else:
                term1 = 0
            if 0 <= i_minus_k < n_points:
                term2 = h[i_minus_k]
            else:
                term2 = 0
            grad[i] = term1 - term2
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        if np.random.random() < perturb_prob:
            noise = np.random.randn(n_points) * 0.01
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        current_cross = cross_corr[m]
        current_c5 = current_cross * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
        lr *= decay
    
    return (best_h, best_c5, n_points)
