import scipy.optimize as opt
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    n_points = len(initial_h)

    elapsed = 0

    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def objective(x, lag_index, h_indices, dx, n_points):
        h = np.zeros(n_points)
        h[h_indices] = x
        h[~h_indices] = h[~h_indices]
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode='full')
        return corr[lag_index] * dx

    def optimize_lag(h, lag_index, h_indices, dx, n_points):
        bounds = [(0, 1) for _ in range(n_points)]
        res = opt.minimize(
            lambda x: objective(x, lag_index, h_indices, dx, n_points),
            x0=h[h_indices],
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        new_h = h.copy()
        new_h[h_indices] = res.x

            return new_h
        return h

    for _ in range(100):
        if elapsed >= budget_s:

        corr = correlate(h, 1 - h, mode='full')
        lag_index = np.argmax(corr)
