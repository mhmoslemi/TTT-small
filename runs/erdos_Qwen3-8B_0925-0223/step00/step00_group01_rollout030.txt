    np.random.seed(seed)
    random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 100
        h = np.full(n_points, 0.5)
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    scale_factor = (n_points / 2.0) / np.sum(h)
    h = h * scale_factor
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    current_h = h.copy()
    T_initial = 1.0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        mask = np.random.rand(n_points) < 0.5
        noise = 0.1 * np.random.normal(0, 1, n_points)
        perturbed = current_h.copy()
        perturbed[mask] += noise[mask]
        perturbed = np.clip(perturbed, 0, 1)
        scaling = (n_points / 2.0) / np.sum(perturbed)
        perturbed = perturbed * scaling
        perturbed_c5 = _compute_c5(perturbed, dx)
        current_c5 = _compute_c5(current_h, dx)
        
        if perturbed_c5 < current_c5:
            current_h = perturbed.copy()
            if perturbed_c5 < best_c5:
                best_c5 = perturbed_c5
                best_h = perturbed.copy()
        else:
            delta = perturbed_c5 - current_c5
            if delta > 0:
                T = T_initial * (1 - (time.time() - start_time) / budget_s)
                if np.random.random() < np.exp(-delta / T):
                    current_h = perturbed.copy()
                    if perturbed_c5 < best_c5:
                        best_c5 = perturbed_c5
                        best_h = perturbed.copy()
    
    # Final checks
    best_h = np.clip(best_h, 0, 1)
    scaling = (n_points / 2.0) / np.sum(best_h)
    best_h = best_h * scaling
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n_points)

def _compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
