import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    current_h = np.array(initial_h, dtype=np.float64)
    current_h = np.clip(current_h, 0.0, 1.0)
    sum_current = np.sum(current_h)
    if sum_current != 0:
        current_h = current_h * (n_points / 2.0) / sum_current

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    T_initial = 1.0
    T_total = budget_s
    start_time = time.time()

    while time.time() - start_time < budget_s:
        time_elapsed = time.time() - start_time
        T = T_initial * (1 - time_elapsed / T_total)

        perturbed_h = current_h.copy()
        indices = np.random.choice(n_points, size=int(n_points * 0.5), replace=False)
        noise = np.random.normal(0, 0.1, len(indices))
        perturbed_h[indices] += noise
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed_h)
        if sum_perturbed == 0:
            continue
        perturbed_h = perturbed_h * (n_points / 2.0) / sum_perturbed

        new_c5 = compute_c5(perturbed_h)
        delta = new_c5 - current_c5

        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = perturbed_h.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5

    best_h_rounded = np.round(best_h, 6)
    return (best_h_rounded, best_c5, n_points)
