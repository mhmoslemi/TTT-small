import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    scaled_initial = initial_h_values * scale

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        h = np.clip(h, 0, 1)

        if total == 0:
            return float('inf')

        scaled_h = h * scale
        corr = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    try:
        result = differential_evolution(
            objective, bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            polish=True,
            seed=seed,
            tol=1e-6,
            x0=scaled_initial
        )
    except:
        result = differential_evolution(
            objective, bounds,
            strategy='best1bin',
            maxiter=100,

            mutation=(0.5, 1.0),
