            best_h = h.copy()
            best_iter = iteration
        if iteration > 0 and np.abs(current_c5 - best_c5) < 1e-8:
            iteration += early_stopping
            break
        if iteration % decay_step == 0 and iteration > 0:
            learning_rate *= decay_rate
        iteration += 1

    initial_cross_corr = np.correlate(initial_h, 1 - initial_h, mode='full')
    initial_c5 = np.max(initial_cross_corr) * dx
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = initial_h.copy()
    
    h_values = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_values)
    if not np.isclose(sum_h, target_sum):
        scale = target_sum / sum_h
        h_values = h_values * scale
        h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
