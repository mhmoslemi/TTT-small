import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    # Assume initial_h_values is available
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0)] * n_points

    # Scale initial_h_values
    sum_initial = np.sum(initial_h_values)
    if sum_initial == 0:
        scaled_initial = np.zeros(n_points)
    else:
        scale_factor = (n_points / 2) / sum_initial
        scaled_initial = initial_h_values * scale_factor

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.inf
        scaled_h = h_clipped * (n_points / 2) / total
        corr = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    start_time = time.time()

    strategy = 'best1bin'
    maxiter = 100
    popsize = 15
    mutation = (0.5, 1.0)
    recombination = 0.8
    polish = True
    tol = 1e-6

    result = differential_evolution(
        objective,
        bounds,
        strategy=strategy,
        maxiter=maxiter,
        popsize=popsize,
        mutation=mutation,
        recombination=recombination,
        polish=polish,
        tol=tol,
        seed=seed

    elapsed_time = time.time() - start_time

        # Return the best solution if time runs out

        c5_bound = result.fun
