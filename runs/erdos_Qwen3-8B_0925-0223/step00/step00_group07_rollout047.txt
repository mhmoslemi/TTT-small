    iteration = 0
    start_time = time.time()
    
    while iteration < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr[1:-1]
        if np.max(max_corr) < 1e-10:
            break
        max_corr_idx = np.argmax(max_corr)
        k = max_corr_idx
        
        gradient = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                gradient[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                gradient[i] -= h[i - k] * dx
        
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale_factor = (n_points / 2.0) / current_sum if current_sum != 0 else 1.0
        h = h * scale_factor
        
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        iteration += 1
        if iteration % 10 == 0:
            learning_rate *= 0.99
        
        if iteration > 10 and abs(best_c5 - current_c5) < 1e-6:
            break
    
    h_filtered = median_filter(best_h, size=5)
    current_sum = np.sum(h_filtered)
    scale_factor = (n_points / 2.0) / current_sum if current_sum != 0 else 1.0
    h_filtered = h_filtered * scale_factor
    h_filtered = np.clip(h_filtered, 0, 1)
    
    final_c5 = evaluate_erdos_solution(h_filtered)
    assert np.isclose(np.sum(h_filtered), n_points / 2.0, atol=1e-6)
    
    return h_filtered, final_c5, n_points
