import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', np.random.rand(80) * 2 - 1)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    current_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2) / current_sum
    h = np.clip(initial_h_values * scale_factor, 0, 1)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    learning_rate = 0.01
    time_start = time.time()
    
    for iter in range(2000):
        if time.time() - time_start > budget_s - 10:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr[1:-1])
        k = max_corr_idx
        gradient = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                gradient[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                gradient[i] -= h[i - k] * dx
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum < 1e-6:
            h = np.clip(h * (n_points / 2) / (np.sum(h) + 1e-6), 0, 1)
        else:
            h = np.clip(h * (n_points / 2) / current_sum, 0, 1)
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if iter % 100 == 0:
            learning_rate *= 0.99
    
    # Apply median filter
    window = 5
    if n_points > window:
        h = np.lib.stride_tricks.sliding_window_view(best_h, window).mean(axis=1)
        h = np.pad(h, (window // 2, window // 2), mode='constant', constant_values=0)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum < 1e-6:
            h = np.clip(h * (n_points / 2) / (np.sum(h) + 1e-6), 0, 1)
        else:
            h = np.clip(h * (n_points / 2) / current_sum, 0, 1)
    
    return (h, best_c5, n_points)
