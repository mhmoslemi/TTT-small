import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Generate a random initial h with n_points=256
        n_points = 256
        h = np.random.rand(n_points)
        h = (h * (n_points / 2)) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    best_n_points = n_points
    
    # Parameters
    max_iterations = 500  # adjust based on time
    learning_rate = 0.1
    decay_rate = 0.99
    perturbation_prob = 0.1
    perturbation_scale = 0.01
    top_lags = 5
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        # Compute cross-correlation
        cross_corr = np.correlate(h, 1 - h, mode='full')
        # Find top lags
        top_k_indices = np.argsort(cross_corr)[::-1][:top_lags]
        
        # Compute gradients for top lags
        gradients = np.zeros(n_points)
        for idx in top_k_indices:
            k = idx - (n_points - 1)
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    gradients[i] += (1 - h[i + k])
                if 0 <= i - k < n_points:
                    gradients[i] -= h[i - k]
        
        # Apply learning rate and update
        h -= learning_rate * gradients
        learning_rate *= decay_rate  # decay learning rate
        
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Stochastic perturbation
        if np.random.rand() < perturbation_prob:
            noise = np.random.normal(0, perturbation_scale, size=h.shape)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Compute current C5
        current_c5 = np.max(np.abs(cross_corr)) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    # Final evaluation on best_h
    final_cross_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(np.abs(final_cross_corr)) * (2.0 / best_n_points)
    
    return (best_h, final_c5, best_n_points)
