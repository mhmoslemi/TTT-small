import numpy as np
from scipy.signal import correlate, medfilt
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    
    # Normalize initial h
    current_sum = np.sum(h)
    scale_factor = (n_points / 2) / current_sum if current_sum != 0 else 1
    h = np.clip(h * scale_factor, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    learning_rate = 0.01
    max_iterations = 1000
    iterations = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 100:
        if iterations >= max_iterations:
            break
        corr = correlate(h, 1 - h, mode='full')
        corr_mid = corr[1:-1]
        max_corr_idx = np.argmax(corr_mid)
        orig_idx = 1 + max_corr_idx
        k = orig_idx - (n_points - 1)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                gradient[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                gradient[i] -= h[i - k] * dx
        
        # Update h
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale_factor = (n_points / 2) / current_sum if current_sum != 0 else 1
        h *= scale_factor
        
        # Check best
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()

        iterations += 1

    h = medfilt(best_h, kernel_size=5)

    h = np.clip(h * scale_factor, 0, 1)
    final_c5 = compute_c5(h)

    max_corr = np.max(corr[1:-1])
    dx = 2.0 / len(h)
