        best_h = best_h * (desired_total / total)
    best_h = np.clip(best_h, 0, 1)
    
    c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    return (best_h, c5, n_points)
