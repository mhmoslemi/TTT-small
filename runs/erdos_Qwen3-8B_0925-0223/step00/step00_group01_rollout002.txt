    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}]

    result = minimize(
        objective,
        h.flatten(),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h_sum = np.sum(best_h)
        normalize_factor = (n_points / 2) / best_h_sum
        best_h = best_h * normalize_factor
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        c5_bound = np.max(corr) * (2.0 / n_points)
        return (best_h, c5_bound, n_points)
    else:
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        normalize_factor = (n_points / 2) / h_sum
        h = h * normalize_factor
        corr = np.correlate(h, 1 - h, mode='full')
        c5_bound = np.max(corr) * (2.0 / n_points)
        return (h, c5_bound, n_points)
