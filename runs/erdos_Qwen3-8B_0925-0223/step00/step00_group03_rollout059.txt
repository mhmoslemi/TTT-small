    dx = 2.0 / n
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    h = h * (n / 2) / sum_h if sum_h != 0 else h
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    for _ in range(10000):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = corr * dx
        current_max_c5 = np.max(c5)
        if current_max_c5 < best_c5:
            best_c5 = current_max_c5
            best_h = h.copy()
        k_max_index = np.argmax(c5)
        lag = k_max_index - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            j_plus = i + lag
            if 0 <= j_plus < n:
                grad[i] += 1 - h[j_plus]
            j_minus = i - lag
            if 0 <= j_minus < n:
                grad[i] -= h[j_minus]
        grad_norm = grad / np.linalg.norm(grad) if np.linalg.norm(grad) > 1e-10 else np.zeros(n)
        step = 0.05 * (1 - _ / 10000)
        h += step * (-grad_norm)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n / 2) / sum_h
        if time.time() - start_time > budget_s:
            break
    return (best_h, best_c5, n)
