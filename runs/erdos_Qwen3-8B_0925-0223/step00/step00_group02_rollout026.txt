import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    current_sum = np.sum(h)
    scaling_factor = target_sum / current_sum
    h = h * scaling_factor
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    T = 1000.0
    cooling_rate = 0.99
    max_iterations = 100000

    start_time = time.time()
    for _ in range(max_iterations):
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.1, 0.1)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        if new_sum == 0:
            new_h = h.copy()
        else:
            new_h *= target_sum / new_sum
        new_c5 = compute_c5(new_h, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        current_c5 = compute_c5(h, n_points)
        if np.random.rand() < np.exp((current_c5 - new_c5) / T):
            h = new_h.copy()
        T *= cooling_rate
        if time.time() - start_time > budget_s:
            break
    return (best_h, best_c5, n_points)
