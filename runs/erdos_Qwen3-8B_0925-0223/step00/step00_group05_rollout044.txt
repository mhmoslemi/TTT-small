import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    from time import time

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    g = np.array(initial_h_values) - 0.5
    sum_g = np.sum(g)
    if abs(sum_g) > 1e-10:
        g *= (1 - sum_g) / (np.sum(g) + 1e-10)
    bounds = [(-0.5, 0.5) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda g: np.sum(g)}]
    
    def objective(g):
        corr = np.correlate(g, g, mode='full')
        return np.max(corr) * dx

    start_time = time()
    result = minimize(objective, g, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
    elapsed = time() - start_time

    if not result.success or elapsed > budget_s:
        best_g = g
    else:
        best_g = result.x

    h_values = best_g + 0.5
    corr = np.correlate(best_g, best_g, mode='full')
    c5_bound = np.max(corr) * dx
    return h_values, c5_bound, n_points
