import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    # Use pre-imported initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values, dtype=float)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    def objective(h):
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(corr) * dx
    
    result = opt.minimize(
        fun=objective,
        x0=h_initial,
        bounds=bounds,
        constraints=constraint,
        method='SLSQP',
        tol=1e-5,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    if result.success:
        h_optimized = result.x
    else:
        h_optimized = h_initial
    
    # Ensure constraints are satisfied
    h_optimized = np.clip(h_optimized, 0.0, 1.0)
    sum_h = np.sum(h_optimized)
    if abs(sum_h - n_points / 2) > 1e-6:
        scaling_factor = (n_points / 2) / sum_h
        h_optimized = h_optimized * scaling_factor
    
    c5_bound = objective(h_optimized)
    
    return (h_optimized.tolist(), c5_bound, n_points)
