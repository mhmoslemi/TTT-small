import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    # Use initial_h_values if available; otherwise, default to a uniform distribution
    initial_h_values = np.array([0.5]*80)  # fallback for non-parent context
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Normalize initial_h_values to meet the sum constraint
    h = initial_h_values.copy()
    s = np.sum(h)
    scale_factor = (n_points / 2) / s
    h = h * scale_factor
    h = np.clip(h, 0, 1)

    # Define the objective function with penalties
    def objective(x):
        h_current = x.reshape(-1)
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        max_corr = np.max(corr)
        # Penalties for sum and bounds constraints
        sum_penalty = 1e4 * (np.sum(h_current) - n_points / 2)**2
        bounds_penalty = 1e4 * np.sum(np.maximum(h_current - 1, 0) + np.maximum(1 - h_current, 0))
        return max_corr * dx + sum_penalty + bounds_penalty

    # Optimization with Nelder-Mead, bounded within [0,1]
    result = minimize(
        objective,
        h,
        method="Nelder-Mead",
        bounds=[(0, 1)] * n_points,
        options={"xatol": 1e-5, "fatol": 1e-5, "maxiter": 1000},
    )

    best_h = result.x
    # Evaluate final C5 to ensure no numerical errors
    final_corr = np.correlate(best_h, 1 - best_h, mode="full")
    best_c5 = np.max(final_corr) * dx

    return (best_h, best_c5, n_points)
