import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)

    # Check and adjust initial sum
    sum_initial = np.sum(initial_h)
    required_sum = n_points / 2

        return max_corr * dx

    h0 = initial_h.copy()
    
    start_time = time()
    result = minimize(

        x0=h0,
        method='SLSQP',
        bounds=bounds,
