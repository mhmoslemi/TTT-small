import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 400
        h = np.random.rand(n_points)
        h -= h.min()
        h /= h.max()
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum != 0:
        h *= target_sum / current_sum
    learning_rate = 0.05
    max_iterations = 1000
    iteration = 0
    start_time = time.time()
    while time.time() - start_time < budget_s:
        correlation = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(correlation)
        lead = n_points - 1
        k = max_index - lead
        m = np.arange(n_points, dtype=int)
        terms1 = m + k
        terms2 = m - k
        valid1 = (terms1 >= 0) & (terms1 < n_points)
        valid2 = (terms2 >= 0) & (terms2 < n_points)
        grad = np.zeros(n_points)
        grad[valid1] += (1 - h[terms1[valid1]])
        grad[valid2] -= h[terms2[valid2]]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            scale_factor = target_sum / current_sum
            h *= scale_factor
        iteration += 1
        if iteration > max_iterations:
            break
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * 2.0 / n_points
    return (h.tolist(), c5, n_points)
