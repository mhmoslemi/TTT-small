    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    h_clamped = np.clip(initial_h, 0, 1)
    sum_clamped = np.sum(h_clamped)
    scaling = (n_points / 2.0) / sum_clamped
    h_clamped = h_clamped * scaling
    h_clamped = np.clip(h_clamped, 0, 1)
    best_h = h_clamped.copy()
    best_c5 = compute_c5(h_clamped)
    
    pop_size = 50
    mutation_rate = 0.2
    elite_size = 5
    max_gen = 1000
    start_time = time.time()
    
    population = []
    for _ in range(pop_size):
        temp = initial_h.copy()
        for i in range(n_points):
            if random.random() < 0.1:
                temp[i] += np.random.normal(0, 0.1)
        temp = np.clip(temp, 0, 1)
        temp_sum = np.sum(temp)
        scaling = (n_points / 2.0) / temp_sum
        temp = temp * scaling
        temp = np.clip(temp, 0, 1)
        population.append(temp)
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        
        evaluated = [(compute_c5(h), h) for h in population]
        evaluated.sort()
        best_current = evaluated[0][0]
        if best_current < best_c5:
            best_c5 = best_current
            best_h = evaluated[0][1]
        
        elites = [h for _, h in evaluated[:elite_size]]
        new_pop = elites[:]
        
        for _ in range(pop_size - elite_size):
            parent = random.choice(population)
            mut = parent.copy()
            for j in range(n_points):
                if random.random() < mutation_rate:
                    mut[j] += np.random.normal(0, 0.1)
            mut = np.clip(mut, 0, 1)
            mut_sum = np.sum(mut)
            scaling = (n_points / 2.0) / mut_sum
            mut = mut * scaling
            mut = np.clip(mut, 0, 1)
            new_pop.append(mut)
        
        population = new_pop
    
    return (best_h, best_c5, n_points)
