    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    
    def normalize_h(vec):
        sum_h = np.sum(vec)
        scale = (n_points / 2) / sum_h
        vec = np.clip(vec * scale, 0, 1)
        sum_clipped = np.sum(vec)
        if sum_clipped < n_points / 2 - 1e-6:
            scale = (n_points / 2) / sum_clipped
            vec = vec * scale
        return vec
    
    h = normalize_h(h)
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    
    start_time = time.time()
    for _ in range(2000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag = np.argmax(corr)
        
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            corr_p = np.correlate(h_p, 1 - h_p, mode='full')
            val_p = corr_p[max_lag] * dx
            h_m = h.copy()
            h_m[i] -= eps
            h_m = normalize_h(h_m)
            corr_m = np.correlate(h_m, 1 - h_m, mode='full')
            val_m = corr_m[max_lag] * dx
            grad[i] = (val_p - val_m) / (2 * eps)
        
        h -= 0.1 * grad
        h = normalize_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, final_c5, n_points)
