                    diff -= a

    return g

def _viol_obj(h, tau):
    c = np.correlate(h, 1.0 - h, mode='full')
    exc = np.maximum(c - tau, 0.0)
    return np.sum(exc * exc)

def _viol_grad(h, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if c[k] <= tau:
            continue
        exc = 2.0 * (c[k] - tau) * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += exc * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= exc * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += exc * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= exc * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _build_bnds(n):
    return [(0.0, 1.0)] * n

def _build_cons():
    return [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]

def _slsqp_lse(h, alpha, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=_build_bnds(n), constraints=_build_cons(),
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _slsqp_viol(h, tau, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(_viol_obj, h, jac=_viol_grad, args=(tau,),
                       method='SLSQP', bounds=_build_bnds(n), constraints=_build_cons(),
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: SLSQP LSE at original resolution
    for alpha in [500, 2000, 10000, 50000]:
        if time.time() - t0 > 100:
            break
        h_new = _slsqp_lse(best_h, alpha, 2000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase B: Upsample to 2x, optimize
    if time.time() - t0 < budget_s - 450:
        n1 = n0 * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha in [500, 2000, 10000, 50000, 200000]:
            if time.time() - t0 > budget_s - 300:
                break
            h_hi = _slsqp_lse(h_hi, alpha, 2000, 1e-19)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1

    # Phase C: Violation-based optimization (key new idea)
    nc = best_n
    if time.time() - t0 < budget_s - 250:
        for tau_frac in [0.999, 0.995, 0.99, 0.985, 0.98]:
            if time.time() - t0 > budget_s - 180:
                break
            tau = best_c5 * nc / 2.0 * tau_frac
            h_new = _slsqp_viol(best_h, tau, 1000, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase D: Multi-start with diverse strategies
    nc = best_n
    if time.time() - t0 < budget_s - 150:
        for _r in range(25):
            if time.time() - t0 > budget_s - 100:
                break
            strat = rng.randint(6)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 5)):
                    bs = rng.randint(5, max(6, nc // 8))
                    st = rng.randint(0, nc - bs)
                    hp[st:st + bs] += 0.03 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 3:
                hp = best_h.copy()
                f = rng.randint(2, nc // 4 + 1)
                amp = 0.008 * rng.rand() * (1 if rng.rand() < 0.5 else -1)
                t_arr = np.arange(nc)
                hp += amp * np.sin(2 * np.pi * f * t_arr / nc + rng.rand() * 2 * np.pi)
                hp = _repair(hp, nc)
            elif strat == 4:
                hp = 0.5 + 0.3 * rng.randn(5)
                hp = np.interp(np.linspace(0, 1, nc), np.linspace(0, 1, 5), hp)
                hp += 0.05 * rng.randn(nc)
                hp = _repair(hp, nc)
            else:
                hp = _repair(best_h * rng.uniform(0.8, 1.2) + 0.01 * rng.randn(nc), nc)
            h_new = _slsqp_lse(hp, 50000, 1500, 1e-19)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase E: Lag-targeted descent with restarts
    if time.time() - t0 < budget_s - 80:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        dx = 2.0 / nc
        for _round in range(300):
            if time.time() - t0 > budget_s - 80:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            k_w = c.argmax()
            sh = (nc - 1) - k_w
            g = np.zeros(nc)
            if sh >= 0:
                i1 = np.arange(nc - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, nc)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, nc)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(nc - sp)
                g[i2] -= h[i2 + sp]
            g *= dx
            improved = False
            for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                h_new = _proj_to_sum(h - step * g, nc * 0.5)
                c5n = _eval(h_new)
                if c5n < c5c - 1e-15:
                    h = h_new
                    c5c = c5n
                    improved = True
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_h = h.copy()
                    break
            if not improved:
                h = _repair(h + 0.001 * rng.randn(nc), nc)
                c5c = _eval(h)

    # Phase F: Fine pairwise search
    if time.time() - t0 < budget_s - 30:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for scale in [0.003, 0.001, 0.0003, 0.0001]:
            if time.time() - t0 > budget_s - 30:
                break
            for _it in range(12000):
                if time.time() - t0 > budget_s - 30:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase G: Final SLSQP polish
    if time.time() - t0 < budget_s - 10:
        h = _slsqp_lse(best_h, 500000, 3000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase H: Ultra-fine touch
    if time.time() - t0 < budget_s - 2:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(25000):
            if time.time() - t0 > budget_s - 2:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 3e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
