    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if abs(d) < 1e-14:
                break
            delta = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += delta
            d -= delta
    return h

def c5val(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx, c

def smooth_obj(x, n, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    # Use top-m power sum for better conditioning
    cs = np.sort(c)[::-1]
    m = min(50, len(cs))
    top = cs[:m]
    # Soft max via power sum
    top_pos = np.maximum(top, 1e-10)
    obj = (np.sum(top_pos ** alpha)) ** (1.0/alpha)
    penalty = lam * (x.sum() - n*0.5)**2
    return obj + penalty

def smooth_obj2(x, n, a, lam):
    # Log-sum-exp version
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    obj = m + np.log(ew.sum()) / a
    penalty = lam * (x.sum() - n*0.5)**2
    return obj, pw

def targeted_grad(h, n, active_idx):
    # Gradient of average of active overlaps
    g = np.zeros(n)
    for ki in active_idx:
        L = ki - (n - 1)
        if L == 0:
            g[:] += 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += 1.0 - h[:n-L]
            g[:n-L] -= h[L:]
        else:
            m = -L
            g[:n-m] += 1.0 - h[m:]
            g[m:] -= h[:n-m]
    g /= len(active_idx)
    return g

def proj_step(h, g, step, n):
    # Projected gradient step with line search
    for _ in range(40):
        hn = repair(h - step * g, n)
        c5, _ = c5val(hn, 2.0/n)
        cn, _ = c5val(h, 2.0/n)
        if c5 < cn - 1e-18:
            return hn, step
        step *= 0.7
        if step < 1e-12:
            break
    return h, step

def targeted_optimize(h, n, t0, timelimit):
    dx = 2.0/n
    bh, bc = h.copy(), c5val(h, dx)[0]
    for topm in [2, 3, 5, 8, 10]:
        h = bh.copy()
        step = 0.05
        stall = 0
        for it in range(500):
            if time.time() - t0 > timelimit:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            cm = c.max()
            thr = cm - 0.005 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > topm:
                idx = np.argsort(c)[::-1][:topm]
            g = targeted_grad(h, n, idx)
            # Remove mean to preserve sum constraint approximately
            g -= g.mean()
            h, step = proj_step(h, g, step, n)
            cn = c5val(h, dx)[0]
            if cn < bc:
                bc, bh = cn, h.copy()
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    break
    return bh, bc

def lbfgs_opt(h, n, t0, timelimit):
    _N = n
    _A = 100.0
    _L = 50.0
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(_A * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / _A + _L * (x.sum() - _N*0.5)**2
    def _jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(_A * (c - m), -500, 500)
        ew = np.exp(s)
        pw = ew / ew.sum()
        g = np.zeros(n)
        for k in range(2*n-1):
            if pw[k] > 1e-14:
                L = k - (n-1)
                if L == 0:
                    g[:] += pw[k] * (1.0 - 2.0 * x)
                elif L > 0:
                    g[L:] += pw[k] * (1.0 - x[:n-L])
                    g[:n-L] -= pw[k] * x[L:]
                else:
                    mm = -L
                    g[:n-mm] += pw[k] * (1.0 - x[mm:])
                    g[mm:] -= pw[k] * x[:n-mm]
        g /= _A
        g += 2.0 * _L * (x.sum() - _N*0.5)
        return g
    
    bh, bc = h.copy(), c5val(h, 2.0/n)[0]
    bds = [(0.0, 1.0)] * n
    for a, lm in [(50, 80), (200, 60), (1000, 40), (5000, 20), (20000, 10), (80000, 5), (200000, 2)]:
        if time.time() - t0 > timelimit:
            break
        _A, _L = float(a), float(lm)
        try:
            r = minimize(_obj, bh.copy(), jac=_jac, method='L-BFGS-B', bounds=bds,
                         options={'maxiter': 300, 'ftol': 1e-19, 'gtol': 1e-16})
            hc = repair(r.x, n)
            c = c5val(hc, 2.0/n)[0]
            if c < bc:
                bc, bh = c, hc
        except:
            pass
    return bh, bc

def discrete_search(h, n, rng, t0, timelimit, total_iters=30000):
    dx = 2.0/n
    bh, bc = h.copy(), c5val(h, dx)[0]
    cc = bc
    deltas = np.array([2e-3, 5e-3, 1e-2, -2e-3, -5e-3, -1e-2, 1e-3, -1e-3])
    it = 0
    while it < total_iters:
        if time.time() - t0 > timelimit:
            break
        it += 1
        i = rng.randint(n)
        d = deltas[rng.randint(len(deltas))]
        ni = h[i] + d
        if 0 <= ni <= 1:
            h[i] = ni
            c = c5val(h, dx)[0]
            if c < cc - 1e-15:
                h = repair(h, n)
                cc = c5val(h, dx)[0]
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i] = ni - d
        else:
            h[i] = ni - d
    return bh, bc

def pair_search(h, n, rng, t0, timelimit, total_iters=20000):
    dx = 2.0/n
    bh, bc = h.copy(), c5val(h, dx)[0]
    cc = bc
    deltas = np.array([1e-2, 4e-3, 1.5e-3, -1e-2, -4e-3, -1.5e-3])
    it = 0
    while it < total_iters:
        if time.time() - t0 > timelimit:
            break
        it += 1
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue
        d = deltas[rng.randint(len(deltas))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5val(h, dx)[0]
            if c < cc - 1e-15:
                h = repair(h, n)
                cc = c5val(h, dx)[0]
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
        else:
            h[i], h[j] = ni - d, nj + d
    return bh, bc

def spectral_eq(h, n, t0, timelimit):
    dx = 2.0/n
    bh, bc = h.copy(), c5val(h, dx)[0]
    for alpha in [0.3, 0.5, 0.7, 0.5, 0.3, 0.15]:
        h = bh.copy()
        for _ in range(200):
            if time.time() - t0 > timelimit:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            target_mag = np.full(n//2+1, np.sqrt(R0/n))
            target_mag[0] = abs(H[0])
            new_mag = (1-alpha)*mag + alpha*target_mag
            hn = np.fft.irfft(new_mag * np.exp(1j*ph), n)
            hr = repair(hn, n)
            cn = c5val(hr, dx)[0]
            if cn < c5val(h, dx)[0] - 1e-18:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc
