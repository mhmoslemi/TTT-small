import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_at_lag(h, sh, n):
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:sh + m])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh
        m = n - sp
        g[sp:] += (1.0 - h[:m])
        g[:m] -= h[sp:sp + m]
    return g

def _subgrad_topk(h, n, top_k):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:top_k]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        g += _grad_at_lag(h, sh, n)
    return g * dx, c.max() * dx

def _pgd_momentum(h, n, nsteps, lr, momentum, top_k, t0, deadline):
    v = np.zeros(n)
    h_cur = h.copy()
    c5 = _eval(h_cur)
    for t in range(nsteps):
        if time.time() > deadline:
            break
        g, _ = _subgrad_topk(h_cur, n, top_k)
        v = momentum * v + (1.0 - momentum) * g
        step = lr
        improved = False
        for _ in range(15):
            h_new = _proj_sum(h_cur - step * v, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new <= c5 + 1e-14:
                h_cur = h_new
                c5 = c5_new
                improved = True
                break
            step *= 0.5
        if not improved:
            v *= 0.5
    return h_cur, c5

def _get_active(h, n, top_k):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:top_k]
    grads = np.zeros((n, top_k))
    for ti, k in enumerate(idx):
        sh = (n - 1) - k
        grads[:, ti] = _grad_at_lag(h, sh, n)
    return c[idx], grads

def _fast_pairwise(h, n, top_k, n_iter, rng, scale, t0, deadline, recompute_every=500):
    h_cur = h.copy()
    c5 = _eval(h_cur)
    dx = 2.0 / n
    c_active, grads = _get_active(h_cur, n, top_k)
    accepted = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h_cur[i] + d
        nj = h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        delta = d * (grads[i] - grads[j])
        new_max = (c_active + delta).max() * dx
        if new_max < c5 - 1e-15:
            h_cur[i], h_cur[j] = ni, nj
            c5 = new_max
            c_active = c_active + delta
            accepted += 1
            if accepted % recompute_every == 0:
                c_active, grads = _get_active(h_cur, n, top_k)
                c5 = _eval(h_cur)
    return h_cur, c5
