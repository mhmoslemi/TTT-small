            best = c_new
            h = h_new
        else:

        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP polish (80s)
    t_end = t0 + 80
    for alpha in [300, 2000, 8000, 30000]:
        if time.time() > t_end or rem() < budget_s - 100:
            break
        h, c = _slsqp_polish(h, dx, alpha, 400, t_end)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 2: Targeted pair optimization (350s)
    t_end = min(t0 + 430, t0 + budget_s - 400)
    h = best_h.copy()
    for outer_batch in range(5):
        if time.time() > t_end or rem() < 350:
            break
        h, c = _pair_opt_pass(h, dx, rng, t_end, n_outer=200, n_gs=50)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        # Small noise injection between batches
        h = best_h + 0.015 * (2.0 * rng.random(n) - 1.0)
        h = _repair(h, tgt)

    # Use best_h for next phase
    h = best_h.copy()
    h, best_c5 = _pair_opt_pass(h, dx, rng, min(t0 + 500, t0 + budget_s - 300), 300, 40)
    if best_c5 < _c5(h, dx):
        pass  # best_c5 already updated inside
    best_h = h.copy()

    # Phase 3: Local search (150s)
    t_end = min(t0 + 650, t0 + budget_s - 250)
    h = best_h.copy()
    h, c = _local_search(h, dx, n, tgt, rng, t_end, 800000, base_scale=0.04)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()

    # Phase 4: Higher resolution
    if rem() > 250:
        for n_new in [2 * best_n, 3 * best_n]:
            if rem() < 180 or n_new > 1200:
                break
            dxx = 2.0 / n_new
            tgtx = n_new / 2.0
            hx = _repair(_interp1d(best_h, n_new), tgtx)
            cx = _c5(hx, dxx)
            # Pair optimization at new resolution
            t_ex = min(t0 + budget_s - 120, time.time() + min(120, rem() - 130))
            hx, cx = _pair_opt_pass(hx, dxx, rng, t_ex, 150, 35)
            # Local search
            t_ex2 = min(t0 + budget_s - 60, time.time() + 40)
            hx, cx = _local_search(hx, dxx, n_new, tgtx, rng, t_ex2, 200000, base_scale=0.02)
            # Final pair pass
            if rem() > 80:
                hx, cx = _pair_opt_pass(hx, dxx, rng, t0 + budget_s - 40, 100, 30)
            if cx < best_c5:
                best_c5 = cx
                best_h = hx.copy()
                best_n = n_new
                best_dx = dxx

    # Phase 5: Final intensive local search + pair opt
    h = best_h.copy()
    nn = len(h)
    ddx = 2.0 / nn
    tggt = nn / 2.0
    best_c5 = _c5(h, ddx)
    t_end = t0 + budget_s - 25
    h, c = _local_search(h, ddx, nn, tggt, rng, t_end, 500000, base_scale=0.02)
    if c < best_c5:
        best_c5 = c
        h = _repair(h, tggt)
    if rem() > 40:
        h, c = _pair_opt_pass(h, ddx, rng, t0 + budget_s - 10, 100, 35)
        if c < best_c5:
            best_c5 = c

    # Final repair and validation
    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
