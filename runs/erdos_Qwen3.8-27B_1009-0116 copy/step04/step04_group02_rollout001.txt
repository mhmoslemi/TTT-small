import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def logsumexp_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_con(x, target):
    return np.sum(x) - target

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, pen),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_opt(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, 0.0), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def subgradient_descent(h, n, deadline, n_outer=200):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h, dx)
    target = n / 2.0
    for outer in range(n_outer):
        if time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        topk = 5
        idx_top = np.argsort(corr)[-topk:]
        d = np.zeros(n)
        for m in idx_top:
            lag = m - (n - 1)
            g = np.zeros(n)
            if lag >= 0:
                valid = slice(0, n - lag)
                g[valid] += (1.0 - h[lag:lag + len(valid)])
                g[lag:] -= h[0:n - lag]
            else:
                ll = -lag
                valid = slice(0, n - ll)
                g[ll:] += (1.0 - h[0:n - ll])
                g[0:n - ll] -= h[ll:]
            d += g
        d /= topk
        d -= np.mean(d)
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-14:
            break
        step = 0.02 / norm_d
        for _ in range(10):
            h_new = h - step * d
            h_new = repair(h_new, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                break
            step *= 0.5
            if step < 1e-10:
                break
    return h, best_c5

def coarse_local(h, n, dx, rng, deadline, n_iters=80000):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for i in range(n_iters):
        if time.time() >= deadline:
            break
        a, b = rng.randint(n), rng.randint(n)
        if a == b:
            continue
        d = rng.uniform(-0.02, 0.02)
        va, vb = h[a] + d, h[b] - d
        if not (0 <= va <= 1 and 0 <= vb <= 1):
            continue
        h[a], h[b] = va, vb
        c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        else:
            h[a], h[b] = va - d, vb + d
    return h, best_c5

def fine_local(h, n, dx, rng, deadline, n_iters=200000):

def block_local(h, n, dx, rng, deadline, n_iters=30000):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for i in range(n_iters):
        if time.time() >= deadline:
            break
        bs = rng.randint(2, 6)
        a = rng.randint(0, n - bs + 1)
        b = rng.randint(0, n - bs + 1)
        if abs(a - b) < bs:
            continue
        eps = rng.uniform(0.002, 0.012) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if not (-1e-9 <= h[a+di]+eps <= 1.0+1e-9 and -1e-9 <= h[b+di]-eps <= 1.0+1e-9):
                ok = False
                break
        if not ok:
            continue
        h[a:a+bs] += eps
        h[b:b+bs] -= eps
        c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        else:
            h[a:a+bs] -= eps
            h[b:b+bs] += eps
    return h, best_c5

def perturb(h, n, rng, kind):

    if kind == 0:
        hp = np.roll(hp, rng.randint(n // 4, 3 * n // 4))
