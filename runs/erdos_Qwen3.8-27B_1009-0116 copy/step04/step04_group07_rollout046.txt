        s = np.sum(np.clip(v + mid, 0.0, 1.0))

            lo = mid
        else:
            hi = mid

    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

    m_max = np.max(c)
    w = np.exp(alpha * (c - m_max))

            if 0 <= j < n:

        stalled = 0
        for it in range(max_iter_per_alpha):
            if time.time() > t_end:

            s = step
            accepted = False

    best = compute_c5(h, n)
    for p in range(n_passes):

            old = h[i]

            # Simple: just adjust h[i] and redistribute the difference uniformly
            for delta in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                new_val = old + delta
                if new_val < 0 or new_val > 1:

                h[i] = new_val

                h_try[~(np.arange(n)==i)] += diff / (n-1)
                h_try = np.clip(h_try, 0, 1)

                    improved = True
                    h = h_try
                    break
                h = h.copy()  # undo (but we modified h[i] already)
                h[i] = old
            else:
                h[i] = old
        if not improved and p > 20:
            break
    return h, best

        best_h = h_new
        best_c5 = c5_new
    h_new, c5_new = swap_descent(best_h, n, t_p4, 50000, rng)
    if c5_new < best_c5 - 1e-14:
        best_h = h_new
        best_c5 = c5_new

    # Phase 5: Try other n values
    t_p5 = t0 + budget_s * 0.88
    for n_alt in [n - 2, n + 2, n + 4, n + 8, n - 4]:
        if time.time() > t_p5 or n_alt < 40 or n_alt > 160 or n_alt == n:
            continue
        h_seed = resample(best_h, n, n_alt)
        h_seed = project(h_seed, n_alt / 2.0)
        h_new, c5_new = proj_grad_momentum(h_seed, n_alt, 50000, t_p5, max_iter=5000, mom=0.80)
        h_new, c5_new = random_coord(h_new, n_alt, t_p5, 50000, rng)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
            best_n = n_alt

    # Phase 6: Simulated annealing + final coordinate descent
    t_p6 = t0 + budget_s * 0.96
    h_new, c5_new = anneal(best_h, best_n, t_p6, rng, n_trials=80000)
    if c5_new < best_c5 - 1e-14:
        best_h = h_new
        best_c5 = c5_new
    h_new, c5_new = random_coord(best_h, best_n, t_p6, 100000, rng)
    if c5_new < best_c5 - 1e-14:
        best_h = h_new
        best_c5 = c5_new

    # Phase 7: Final ultra-fine polish
    t_final = t0 + budget_s * 0.995
    h_new, c5_new = proj_grad_momentum(best_h, best_n, 2000000, t_final, max_iter=3000, mom=0.75)
    if c5_new < best_c5 - 1e-14:
        best_h = h_new
        best_c5 = c5_new
    h_new, c5_new = random_coord(best_h, best_n, t_final, 50000, rng)
    if c5_new < best_c5 - 1e-14:
        best_h = h_new
        best_c5 = c5_new

    best_h = project(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
