                it += 1; continue
            d = rng.uniform(-sigma * 2.5, sigma * 2.5)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                it += 1; continue
            h[i], h[j] = ni, nj
            c5_new = _c5(h, n)
            delta = c5_new - c5_cur
            if delta < 0 or rng.random() < np.exp(-min(delta / temp, 500)):
                c5_cur = c5_new
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        it += 1
    return best_h, best_c5

def _pair_sweep(h, n, t_end, rng):
    h = h.copy()
    c5_cur = _c5(h, n)
    deltas = np.array([0.001, 0.002, 0.004, 0.007, 0.012, 0.02,
                       -0.001, -0.002, -0.004, -0.007, -0.012, -0.02])
    for _sweep in range(500):
        if time.time() > t_end:
            break
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for i in idx:
            if time.time() > t_end:
                break
            for j in idx:
                if j == i:
                    continue
                for d in deltas:
                    ni, nj = h[i] + d, h[j] - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5_cur - 1e-15:
                        c5_cur = c
                        improved = True
                    else:
                        h[i], h[j] = ni - d, nj + d
        if not improved:
            break
    return h, c5_cur

def _ultra_fine(h, n, t_end, rng):
    h = h.copy()
    c5_cur = _c5(h, n)
    tiny = np.array([0.0001, 0.0002, 0.0004, 0.0006, 0.001,
                     -0.0001, -0.0002, -0.0004, -0.0006, -0.001])
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c = _c5(h, n)
        if c < c5_cur:
            c5_cur = c
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    global _SQP_N, _SQP_A
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_n, best_c5 = h.copy(), n, _c5(h, n)

    # Phase 1: Gradient refinement (0-12%)
    t1 = t0 + budget_s * 0.12
    for alpha in [5000, 50000, 200000]:
        if time.time() > t1:
            break
        h_w, c5_w = _grad_refine(best_h.copy(), best_n, alpha, t1, 0.05)
        if c5_w < best_c5:
            best_c5, best_h = c5_w, h_w

    # Phase 2: SLSQP polish (12-25%)
    t2 = t0 + budget_s * 0.25
    _SQP_N = best_n
    for alpha in [80000, 300000]:
        if time.time() > t2:
            break
        _SQP_A = alpha
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(_sqp_obj, best_h.copy(), jac=_sqp_jac, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _sqp_con, 'jac': _sqp_jac_con}],
                           options={'maxiter': 500, 'ftol': 1e-18})
            hc = _repair(res.x, best_n)
            c5 = _c5(hc, best_n)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 3: Simulated annealing - main (25-55%)
    t3 = t0 + budget_s * 0.55
    h_w, c5_w = _sa_optimize(best_h.copy(), best_n, t3, rng,
                             t_init=4e-4, t_final=2e-6, sigma=0.005)
    if c5_w < best_c5:
        best_c5, best_h = c5_w, h_w
    # Gradient refinement after SA
    if time.time() < budget_s * 0.60:
        h_w, c5_w = _grad_refine(best_h.copy(), best_n, 150000, t0 + budget_s * 0.60, 0.02)
        if c5_w < best_c5:
            best_c5, best_h = c5_w, h_w

    # Phase 4: Multi-resolution exploration (60-73%)
    t4 = t0 + budget_s * 0.73
    for n_try in [100, 120, 160, 200, 84, 112]:
        if time.time() > t4:
            break
        if n_try == best_n or time.time() > t4 - 5:
            continue
        x_old = np.linspace(0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h, period=2.0)
        h_new = _repair(h_new, n_try)
        h_new, c5_new = _grad_refine(h_new, n_try, 50000, min(t4, time.time() + 10), 0.03)
        if time.time() < t4 - 10:
            h_new, c5_new = _sa_optimize(h_new, n_try, min(t4, time.time() + 15), rng,
                                         t_init=2e-4, t_final=2e-5, sigma=0.003)
        if c5_new < best_c5:
            best_c5, best_h, best_n = c5_new, h_new, n_try

    # Phase 5: Pair sweep (73-89%)
    t5 = t0 + budget_s * 0.89
    best_h, best_c5 = _pair_sweep(best_h.copy(), best_n, t5, rng)

    # Phase 6: Second SA with tighter parameters (89-96%)
    t6 = t0 + budget_s * 0.96
    h_w, c5_w = _sa_optimize(best_h.copy(), best_n, t6, rng,
                             t_init=1.5e-4, t_final=5e-6, sigma=0.002)
    if c5_w < best_c5:
        best_c5, best_h = c5_w, h_w

    # Phase 7: Ultra-fine final polish (96-99.5%)
    t7 = t0 + budget_s - 5
    best_h, best_c5 = _ultra_fine(best_h.copy(), best_n, t7, rng)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
