The min-max problem: min_h max_k corr_k

4. Iterate

The weighted sum problem is:

        if time.time() > t_end or rem() < 350:
            break
        h = _slsqp(h, a, 600)
        c = _c5(h, dx0)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # Phase 2: L-BFGS refinement at n0 (100s)
    t_end = min(t0 + 250, t0 + budget_s - 300)
    h2 = _lbfgs(best_h, 5000, 500, 4)
    c2 = _c5(h2, dx0)
    if c2 < best_c:
        best_h, best_c = h2, c2
    h2 = _lbfgs(best_h, 50000, 300, 4)
    c2 = _c5(h2, dx0)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Multi-restart with structured perturbations (200s)
    t_end = min(t0 + 450, t0 + budget_s - 250)
    for r in range(8):
        if time.time() > t_end or rem() < 250:
            break
        x = np.linspace(0, 2, n0)
        f = 1.0 + rng.randint(4)
        phi = rng.uniform(0, 2 * np.pi)
        amp = 0.06 * (1.0 - r / 12.0)
        hp = best_h + amp * np.sin(np.pi * f * x + phi)
        hp = _repair(hp, tgt0)
        hp = _slsqp(hp, 3000, 400)
        hp = _slsqp(hp, 20000, 500)
        c = _c5(hp, dx0)
        if c < best_c:
            best_h, best_c = hp.copy(), c

    # Phase 4: Upsample to 2*n0 and optimize (250s)
    n1 = 2 * n0
    dx1 = 2.0 / n1
    tgt1 = n1 / 2.0
    h1 = _repair(_interp(best_h, n1), tgt1)
    c1 = _c5(h1, dx1)
    t_end = min(t0 + 700, t0 + budget_s - 200)
    if c1 < best_c + 0.01 and rem() > 250:
        h1 = _lbfgs(h1, 3000, 400, 3)
        c1 = _c5(h1, dx1)
        for a in [10000, 50000, 200000]:
            if time.time() > t_end or rem() < 200:
                break
            h1 = _slsqp(h1, a, 500)
            c1 = _c5(h1, dx1)
        if c1 < best_c:
            best_h, best_c, best_n = h1.copy(), c1, n1

    # Phase 5: Gradient-guided local search (200s)
    nb = best_n
    dxb = 2.0 / nb
    tgtb = nb / 2.0
    t_end = min(t0 + 900, t0 + budget_s - 120)
    best_h, best_c = _grad_search(best_h, dxb, tgtb, 50000, rng, t_end)

    # Phase 6: Pairwise coordinate descent (100s)
    t_end = min(t0 + 1000, t0 + budget_s - 50)
    best_h, best_c = _pairwise_cd(best_h, dxb, tgtb, 30, t_end)

    # Phase 7: Final SLSQP polish
    if rem() > 30:
        best_h = _slsqp(best_h, 300000, 400)
        best_c = _c5(best_h, dxb)

    # Phase 8: Final micro local search
    if rem() > 10:
        best_h, best_c = _grad_search(best_h, dxb, tgtb, 20000, rng, t0 + budget_s - 2)

    best_h = _repair(best_h, best_n / 2.0)
    best_c = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c, best_n)
