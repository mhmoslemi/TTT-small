def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14: break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

    return np.correlate(h, 1.0 - h, mode='full')

    L = k - (n - 1)

    elif L > 0:

        g[:n-L] -= h[L:]
    else:
        m = -L
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def corr_full(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_k(h, k, n):
    g = np.zeros(n)
    L = k - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

_G = {}

def obj_sm(x):
    n, a = _G['n'], _G['a']
    c = corr_full(x, n)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def jac_sm(x):
    n, a = _G['n'], _G['a']
    c = corr_full(x, n)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-12:
            g += pw[k] * grad_k(x, k, n)
    return g / a

def eq_fun(x):
    return x.sum() - _G['n'] * 0.5

def eq_jac(x):
    return np.ones(_G['n'])

def lbfgs_run(h, n, dx, a_list, t0, lim):
    _G['n'] = n
    bh, bc = h.copy(), c5v(h, dx)
    bds = [(0.0, 1.0)] * n
    for a in a_list:
        if time.time() - t0 > lim:
            break
        _G['a'] = float(a)
        try:
            r = minimize(obj_sm, bh.copy(), jac=jac_sm, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': 400, 'ftol': 1e-20, 'gtol': 1e-17})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def slsqp_run(h, n, dx, a_list, t0, lim):
    _G['n'] = n
    bh, bc = h.copy(), c5v(h, dx)
    bds = [(0.0, 1.0)] * n
    for a in a_list:
        if time.time() - t0 > lim:
            break
        _G['a'] = float(a)
        try:
            r = minimize(obj_sm, bh, method='SLSQP', bounds=bds,
                         constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                         options={'maxiter': 600, 'ftol': 1e-20})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def subgrad_run(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for k_top in [1, 2, 3, 5]:
        if time.time() - t0 > lim:
            break
        h = bh.copy()
        eta = 0.04
        stall = 0
        for _ in range(500):
            if time.time() - t0 > lim:
                break
            c = corr_full(h, n)
            cm = c.max()
            thr = cm - 0.002 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > k_top:
                idx = np.argsort(c)[::-1][:k_top]
            g = np.zeros(n)
            for ki in idx:
                g += grad_k(h, ki, n)
            g /= len(idx)
            g -= g.mean()
            found = False
            for _ in range(35):
                hn = repair(h - eta * g, n)
                cn = c5v(hn, dx)
                if cn < c5v(h, dx) - 1e-18:
                    h = hn
                    found = True
                    if cn < bc:
                        bc, bh = cn, h.copy()
                    eta = min(eta * 1.12, 1.5)
                    stall = 0
                    break
                eta *= 0.7
            if not found:
                stall += 1
                if stall > 12:
                    break
    return bh, bc

def spectral_run(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for alpha in [0.06, 0.12, 0.25, 0.4, 0.55, 0.4, 0.25, 0.12, 0.06]:
        if time.time() - t0 > lim:
            break
        h = bh.copy()
        for _ in range(250):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tgt = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tgt[0] = abs(H[0])
            nm = (1 - alpha) * mag + alpha * tgt
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-16:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def coord_run(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    steps = np.array([0.015, 0.008, 0.004, 0.002, 0.001, -0.015, -0.008, -0.004, -0.002, -0.001])
    for _ in range(50000):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = steps[rng.randint(10)]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc - 1e-15:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def polish_run(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    dt = np.array([2e-4, 4e-4, 7e-4, 1e-3, -2e-4, -4e-4, -7e-4, -1e-3])
    for _ in range(80000):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = dt[rng.randint(8)]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def full_opt(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = lbfgs_run(bh, n, dx, [200, 1000, 5000, 25000, 100000], t0, lim - 200)
    if c < bc:
        bh, bc = r, c
    r, c = slsqp_run(bh, n, dx, [3000, 20000, 80000, 250000], t0, lim - 140)
    if c < bc:
        bh, bc = r, c
    r, c = subgrad_run(bh, n, dx, t0, lim - 100)
    if c < bc:
        bh, bc = r, c
    r, c = spectral_run(bh, n, dx, t0, lim - 70)
    if c < bc:
        bh, bc = r, c
    r, c = lbfgs_run(bh, n, dx, [50000, 200000, 500000], t0, lim - 50)
    if c < bc:
        bh, bc = r, c
    r, c = coord_run(bh, n, dx, rng, t0, lim - 20)
    if c < bc:
        bh, bc = r, c
    r, c = polish_run(bh, n, dx, rng, t0, lim - 3)
    if c < bc:
        bh, bc = r, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n0 = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    dx0 = 2.0 / n0
    bh, bc, bn = h0.copy(), c5v(h0, dx0), n0

    # Main optimization at original resolution
    t_lim = t0 + budget_s * 0.40
    r, c = full_opt(h0, n0, dx0, seed, t0, t_lim)
    if c < bc:
        bh, bc, bn = r, c, n0

    # Try upsampled resolutions
    for mult in [2, 3]:
        n_new = n0 * mult
        if n_new >= 1000:
            break
        rem = budget_s - (time.time() - t0)
        if rem < 150:
            break
        xo = np.linspace(0, 2, n0, endpoint=False)
        xn = np.linspace(0, 2, n_new, endpoint=False)
        hi = repair(np.interp(xn, xo, bh, period=2.0), n_new)
        dxi = 2.0 / n_new
        t_lim = t0 + rem - 40
        r, c = full_opt(hi, n_new, dxi, seed + mult * 100, t0, t_lim)
        if c < bc:
            bh, bc, bn = r, c, n_new

    # Try coarser resolution
    n_coarse = n0 // 2
    rem = budget_s - (time.time() - t0)
    if n_coarse >= 20 and rem > 130:
        xo = np.linspace(0, 2, n0, endpoint=False)
        xn = np.linspace(0, 2, n_coarse, endpoint=False)
        hi = repair(np.interp(xn, xo, bh, period=2.0), n_coarse)
        dxi = 2.0 / n_coarse
        r, c = full_opt(hi, n_coarse, dxi, seed + 500, t0, t0 + rem - 20)
        if c < bc:
            bh, bc, bn = r, c, n_coarse

    # Perturbation restarts at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        rng_p = np.random.RandomState(seed + 777)
        for scale in [0.03, 0.015, 0.06]:
            rem = budget_s - (time.time() - t0)
            if rem < 50:
                break
            hp = repair(bh + scale * rng_p.randn(bn), bn)
            r, c = full_opt(hp, bn, 2.0 / bn, seed + int(scale * 10000), t0, t0 + rem - 10)
            if c < bc:
                bh, bc, bn = r, c, bn

    # Final micro-polish
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        rng_f = np.random.RandomState(seed + 9999)
        bh, bc = polish_run(bh, bn, 2.0 / bn, rng_f, t0, t0 + rem - 2)

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
