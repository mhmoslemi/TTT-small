                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a

def _interp1d(h, n_new):

    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.exp(s).sum()) / alpha + pen * (h.sum() - target) ** 2

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp1d(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _soft_obj(h, alpha, target, pen):
    c = _corr_full(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.exp(s).sum()) / alpha + pen * (h.sum() - target) ** 2

def _slsqp(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, tgt, 0), h,
                     method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lbfgs(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    best, bc = h.copy(), _c5(h, dx)
    for pen in [50, 500, 5000, 50000]:
        try:
            r = minimize(lambda x: _soft_obj(x, alpha, tgt, pen), best,
                         method='L-BFGS-B', bounds=bds,
                         options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
            hc = _repair(r.x, tgt)
            c = _c5(hc, dx)
            if c < bc:
                best, bc = hc, c
        except Exception:
            pass
    return best, bc

def _subgrad(h):
    n = len(h)
    c = _corr_full(h)
    m = c.max()
    active = np.where(c >= m - 0.005 * m)[0]
    if len(active) > 12:
        active = active[np.argsort(c[active])[-12:]]
    grad = np.zeros(n)
    for k in active:
        sh = k - (n - 1)
        if sh >= 0:
            grad[:n-sh] += (1.0 - h[sh:])
            grad[sh:] -= h[:n-sh]
        else:
            s = -sh
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
    grad /= max(1, len(active))
    return grad, c, active

def _grad_search(h, dx, rng, t_end, iters):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    best_h = h.copy()
    h = h.copy()
    for it in range(iters):
        if time.time() > t_end:
            break
        sg, c, active = _subgrad(h)
        d = sg - sg.mean()
        norm = np.linalg.norm(d)
        if norm < 1e-12:
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                dv = 0.005 * (2 * rng.random() - 1)
                h[i] += dv; h[j] -= dv
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best, best_h = cn, h.copy()
                    else:
                        h[i] -= dv; h[j] += dv
            continue
        moved = False
        for step in [0.008, 0.004, 0.002, 0.001, 0.0004]:
            h_new = h - step * d
            if h_new.min() >= -1e-9 and h_new.max() <= 1 + 1e-9:
                hn = _repair(np.clip(h_new, 0, 1), tgt)
                cn = _c5(hn, dx)
                if cn < best - 1e-13:
                    h, best, best_h = hn, cn, hn.copy()

                    c = _corr_full(h)
