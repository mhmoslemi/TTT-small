def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    return float(np.max(_corr(h)) * 2.0 / len(h))

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-10:
        idx = np.argsort(1.0 - h)[::-1] if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _soft_obj(x, alpha, target):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -50, 50)
    return m + np.log(np.exp(s).sum()) / alpha + 2000.0 * (x.sum() - target) ** 2

    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    constr = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt, 'jac': lambda x: np.ones(n)}]
    try:

                     bounds=bds, constraints=constr,

        hc = _repair(r.x, tgt)
        return hc, _c5(hc)

def _grad_corr(h, c_weights):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. h"""
    n = len(h)

        if shift > 0:
            i_hi1 = n - 1 - shift
        elif shift < 0:
            i_lo1 = -shift
        if i_hi1 >= i_lo1:
            idx1 = np.arange(i_lo1, i_hi1 + 1)
            grad[idx1] += wk * (1.0 - h[idx1 - shift])
        # Term 2: -h[i+shift] for valid i
        i_lo2, i_hi2 = 0, n - 1
        if shift > 0:
            i_lo2 = shift
        elif shift < 0:
            i_hi2 = n - 1 + shift
        if i_hi2 >= i_lo2:
            idx2 = np.arange(i_lo2, i_hi2 + 1)
            grad[idx2] -= wk * h[idx2 + shift]
    return grad
