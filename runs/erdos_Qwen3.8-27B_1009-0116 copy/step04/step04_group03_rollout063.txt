import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _to_binary(h, n):

    h_b = (h >= 0.5).astype(float)

        # Flip the ones with lowest "importance" (e.g., smallest neighborhood)
        diff = s - target
        ones = np.where(h_b == 1)[0]
        # Sort by original h value (flip the ones that were closest to 0.5)
        order = np.argsort(h[ones])
        for idx in range(diff):
            h_b[ones[order[idx]]] = 0

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_w(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w

def _grad(h, alpha):
    n = len(h)
    c = _corr(h)
    w = _smax_w(c, alpha)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n-lag] += wk * (1.0 - h[lag:])
            g[lag:] += wk * (-h[:n-lag])
        else:
            L = -lag
            g[L:] += wk * (1.0 - h[:n-L])
            g[:n-L] += wk * (-h[L:])
    return g

def _slsqp(h, n, alpha, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda: np.ones(n)}]
    try:
        res = minimize(
            lambda x: _smax(_corr(x), alpha),
            h, jac=lambda x: _grad(x, alpha),
            method='SLSQP', bounds=bnds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-20}
        )
        return _repair(res.x, n)
    except:
        return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _to_binary(h, n):
    h_b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(h_b.sum())
    if s > target:
        diff = s - target
        ones = np.where(h_b == 1)[0]
        order = np.argsort(h[ones])
        for idx in range(diff):
            h_b[ones[order[idx]]] = 0
    elif s < target:
        diff = target - s
        zeros = np.where(h_b == 0)[0]
        order = np.argsort(h[zeros])[::-1]
        for idx in range(diff):
            h_b[zeros[order[idx]]] = 1
    return h_b

def _interp(h, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_w(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w

def _grad_func(h, alpha):
    n = len(h)
    c = _corr(h)
    w = _smax_w(c, alpha)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n-lag] += wk * (1.0 - h[lag:])
            g[lag:] += wk * (-h[:n-lag])
        else:
            L = -lag
            g[L:] += wk * (1.0 - h[:n-L])
            g[:n-L] += wk * (-h[L:])
    return g

def _obj_smax(x, alpha):
    return _smax(_corr(x), alpha)

def _jac_smax(x, alpha):
    return _grad_func(x, alpha)

def _con_eq(x):
    return x.sum() - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def _slsqp(h, n, alpha, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), jac=_jac_smax,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_sa(h, n, rng, t_end):

    best_c5 = c5
    best_h = h.copy()
    temp = 0.04
    no_imp = 0

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0, 1
        c5_new = _eval(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()
            no_imp = 0
        elif c5_new < c5 - 1e-15:
            c5 = c5_new
            no_imp += 1
        else:
            delta = c5_new - c5
            if delta > 0 and rng.rand() < np.exp(-delta * 15.0 / (temp + 1e-10)):
                c5 = c5_new
                no_imp += 1
            else:
                h[i], h[j] = 1, 0
                no_imp += 1
        if no_imp > 3000:
            temp *= 0.6
            no_imp = 0
            if temp < 0.001:
                temp = 0.03
                h = best_h.copy()
                for _ in range(8 + rng.randint(20)):
                    o = np.where(h == 1)[0]

                    h[hi], h[hj] = 0, 1

def _fudge_search(h, n, rng, t_end):

    best_c5 = c5c

    best_c5 = _eval(h_parent)
    best_n = n_parent

    n_list = [80, 100, 120, 160, 200]
    for n in n_list:
        if _tl() < 80:
            break
        if n == n_parent:
            h_cont = h_parent.copy()
        else:
            h_cont = _repair(_interp(h_parent, n_parent, n), n)
        h_bin = _to_binary(h_cont, n)
        h_rand = np.zeros(n)
        h_rand[rng.choice(n, n // 2, replace=False)] = 1.0

        t1 = time.time() + min(_tl() * 0.45, 100)
        h_a, c5_a = _binary_sa(h_bin, n, rng, t1)
        h_b, c5_b = _binary_sa(h_rand, n, rng, t1)
        h_best_bin = h_a if c5_a <= c5_b else h_b
        c5_best_bin = min(c5_a, c5_b)

        if _tl() > 60:
            t2 = time.time() + min(_tl() * 0.3, 50)
            h_best_bin, c5_best_bin = _binary_greedy(h_best_bin, n, rng, t2)

        if _tl() > 40:
            t3 = time.time() + min(_tl() * 0.15, 30)
            t3 = min(t3, t0 + budget_s - 30)
            h_cont2 = h_best_bin.copy()
            for alpha in [5000, 50000, 500000]:
                if time.time() > t3:
                    break
                h_new = _slsqp(h_cont2, n, alpha, maxiter=400)
                c5_new = _eval(h_new)
                if c5_new < c5_best_bin:
                    c5_best_bin = c5_new
                    h_cont2 = h_new
            if c5_best_bin < _eval(h_best_bin):
                h_best_bin = h_cont2

        if c5_best_bin < best_c5:
            best_c5 = c5_best_bin
            best_h = h_best_bin.copy()
            best_n = n

    if _tl() > 20:
        t_fin = min(time.time() + _tl() * 0.5, t0 + budget_s - 10)
        best_h, best_c5 = _fudge_search(best_h, best_n, rng, t_fin)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
