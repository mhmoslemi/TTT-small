        if delta < 0 or (T > T_min and -delta / T < 50 and rng.random() < np.exp(-delta / T)):
            h_c = h_trial
            c5_c = c5_n
            if c5_n < c5_best:
                c5_best = c5_n
                h_best = h_c.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        # Cooling / kick
        if no_imp > 25000:
            h_c = repair(h_best + rng.normal(0, 0.05, n), n)
            c5_c = c5_val(h_c, dx)
            T = 2e-4
            no_imp = 0
            kick_count += 1
        else:
            T = max(T * 0.99996, T_min)

    # Phase 3: Greedy local search (200s)
    t_ls = min(time.time() + 200, t_start + budget_s - 90)
    hw = h_best.copy()
    cw = c5_best
    for sc in [0.012, 0.005, 0.0018, 0.0006, 0.0002, 0.00006]:
        if time.time() >= t_ls:
            break
        imp = False
        for _ in range(120000):
            if time.time() >= t_ls:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            hw[i] += d
            hw[j] -= d
            if hw[i] < 0 or hw[i] > 1 or hw[j] < 0 or hw[j] > 1:
                hw[i] -= d
                hw[j] += d
                continue
            c5n = c5_val(hw, dx)
            if c5n < cw - 1e-15:
                cw = c5n
                imp = True
                if cw < c5_best:
                    c5_best = cw
                    h_best = hw.copy()
            else:
                hw[i] -= d
                hw[j] += d
        if not imp and sc < 0.003:
            break

    # Phase 4: L-BFGS high-alpha polish (40s)
    if rem() > 55:
        te4 = time.time() + min(40, rem() - 25)
        for al in [20000, 80000, 300000]:
            if time.time() >= te4:
                break
            try:
                res = minimize(smooth_obj, h_best, args=(al, target, 8000),
                              method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': 600, 'ftol': 1e-16})
                hc = repair(res.x, n)
                cc = c5_val(hc, dx)
                if cc < c5_best:
                    h_best, c5_best = hc, cc
            except Exception:
                pass

    # Phase 5: Ultra-fine local search (35s)
    if rem() > 35:
        te5 = time.time() + min(35, rem() - 8)
        hw2 = h_best.copy()
        cw2 = c5_best
        for sc in [1e-4, 3e-5, 1e-5, 3e-6]:
            if time.time() >= te5:
                break
            for _ in range(300000):
                if time.time() >= te5:
                    break
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = sc * (2.0 * rng.random() - 1.0)
                hw2[i] += d
                hw2[j] -= d
                if hw2[i] < 0 or hw2[i] > 1 or hw2[j] < 0 or hw2[j] > 1:
                    hw2[i] -= d
                    hw2[j] += d
                    continue
                c5n = c5_val(hw2, dx)
                if c5n < cw2 - 1e-16:
                    cw2 = c5n
                    if cw2 < c5_best:
                        c5_best = cw2
                        h_best = hw2.copy()
                else:
                    hw2[i] -= d
                    hw2[j] += d

    h_best = repair(h_best, n)
    c5_best = c5_val(h_best, dx)
    return (h_best, c5_best, n)
