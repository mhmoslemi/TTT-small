                if ia == ib:
                    continue
                i, j = int(ii[ia]), int(ii[ib])
                for d in steps:
                    h[i] += d; h[j] -= d
                    if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                        cn = _c5(h, dx)
                        if cn < best:
                            best = cn; h = _project(h, tgt); improved = True; break
                        else:
                            h[i] -= d; h[j] += d
                    else:
                        h[i] -= d; h[j] += d
                if improved:
                    break
            if improved:
                break
        if not improved and it % 100 == 0:
            i = rng.randint(n); j = rng.randint(n)
            if i != j:
                d = 0.003 * (2.0 * rng.random() - 1.0)
                h[i] += d; h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn; h = _project(h, tgt)
                    else:
                        h[i] -= d; h[j] += d
                else:
                    h[i] -= d; h[j] += d
    return h, best

def _grad_search(h, dx, rng, t_end, n_iter=20000):
    n = len(h); tgt = n / 2.0; alpha = 8000.0
    best = _c5(h, dx); h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        sc = np.clip(alpha * (c - m), -500, 500)
        e = np.exp(sc); w = e / e.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            s = k - (n - 1)
            if s >= 0:
                g[s:] += w[k] * (1.0 - h[:n - s])
                g[:n - s] -= w[k] * h[s:]
            else:
                s2 = -s
                g[:n - s2] += w[k] * (1.0 - h[s2:])
                g[s2:] -= w[k] * h[:n - s2]
        i = int(np.argmin(g)); j = int(np.argmax(g))
        done = False
        for d in [0.004, 0.002, 0.001, 0.0005]:
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn; h = _project(h, tgt); done = True; break
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
        if not done:
            i = rng.randint(n); j = rng.randint(n)
            if i != j:
                d = 0.002 * (2.0 * rng.random() - 1.0)
                h[i] += d; h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn; h = _project(h, tgt)
                    else:
                        h[i] -= d; h[j] += d
                else:
                    h[i] -= d; h[j] += d
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _project(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n = len(h); dx = 2.0 / n
    best_h, best_c5 = h.copy(), _c5(h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B with multiple alphas at native res (n=140)
    t_end = t0 + 180
    for alpha in [500, 2000, 8000, 30000, 100000]:
        if time.time() > t_end or rem() < 450:
            break
        h, c = _lbfgs(h, dx, alpha, 250)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 2: Gradient-guided local search at native res
    t_end = min(t0 + 380, t0 + budget_s - 350)
    best_h, best_c5 = _grad_search(best_h, 2.0 / len(best_h), rng, t_end, 20000)

    # Phase 3: Focal search at native res
    t_end = min(t0 + 550, t0 + budget_s - 250)
    best_h, best_c5 = _focal_search(best_h, 2.0 / len(best_h), rng, t_end, 60000)

    # Phase 4: Interpolate to 2n and optimize
    if rem() > 300:
        n2 = 2 * n; dx2 = 2.0 / n2
        h2 = _project(np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, n), best_h), n2 / 2.0)
        c2 = _c5(h2, dx2)
        t_end = min(t0 + budget_s - 150, t0 + 800)
        for alpha in [3000, 10000, 40000]:
            if time.time() > t_end:
                break
            h2, c2 = _lbfgs(h2, dx2, alpha, 200)
        t_end = min(t_end, t0 + budget_s - 80)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 5: Final local search on best
    nb = len(best_h); dxb = 2.0 / nb
    t_end = min(t0 + budget_s - 60, t0 + budget_s - 60)
    best_h, best_c5 = _focal_search(best_h, dxb, rng, t_end, 40000)

    # Phase 6: Final L-BFGS polish
    if rem() > 30:
        best_h, best_c5 = _lbfgs(best_h, 2.0 / len(best_h), 80000, 150)

    # Phase 7: Tiny final local moves
    if rem() > 10:
        best_h, best_c5 = _focal_search(best_h, 2.0 / len(best_h), rng, t0 + budget_s - 2, 10000)

    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
