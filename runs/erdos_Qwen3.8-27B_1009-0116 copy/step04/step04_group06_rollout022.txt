def _subgrad_optimize(h, n, budget_end, n_iters=50000):
    h = h.copy()
    c5_cur = _c5(h, n)
    lr = 0.01
    for it in range(n_iters):
        if time.time() > budget_end:
            break
        c = _all_corr(h, n)  # shape (2n-1,)
        max_c = np.max(c)
        # Active set: correlations within epsilon of max
        eps = 0.01 * max_c
        active = np.where(c > max_c - eps)[0]
        
        # Compute subgradient (average of active gradients)
        g = np.zeros(n)
        for k in active:
            lag = k - (n-1)
            for j in range(n):
                val = 0.0
                if 0 <= j+lag < n:
                    val += (1-h[j+lag])
                if 0 <= j-lag < n:
                    val -= h[j-lag]
                g[j] += val
        g /= len(active)
        
        # Step
        h_new = _project(h - lr * g, n)
        c5_new = _c5(h_new, n)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            lr = min(lr * 1.05, 0.1)  # increase step
        else:
            lr *= 0.7
        
        # Periodic lr decay
        if it % 100 == 99:
            lr *= 0.95
    
    return h, c5_cur

def _corr_grad(h, n, lag):
    """Gradient of correlation at given lag w.r.t. h"""
    g = np.zeros(n)
    # h[j] as h[i]: contributes (1-h[j+lag]) if valid
    # h[j] as h[i+lag]: contributes -h[j-lag] if valid
    lo1, hi1 = max(0, -lag), min(n, n - lag)  # j range where j+lag in [0,n)
    lo2, hi2 = max(0, lag), min(n, n + lag)    # j range where j-lag in [0,n)
    
    if lo1 < hi1:
        g[lo1:hi1] += (1.0 - h[lo1+lag:hi1+lag])
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2-lag:hi2-lag]
    return g

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)

            if abs(diff) < 1e-14:
                break

                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add

                sub = min(-diff, h[idx])

                diff += sub
    return h
