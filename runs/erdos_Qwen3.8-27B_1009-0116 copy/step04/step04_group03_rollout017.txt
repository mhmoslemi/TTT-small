    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    # For each lag k (0 to 2n-2), the actual lag is k-(n-1)
    # corr_k = sum_i h[i]*(1-h[i+k]) for valid i
    # grad h[j] of corr_k = (1-h[j+k])*[j+k<n] - h[j-k]*[j-k>=0]

        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # h[i] appears for i in [0, n-lag-1], contributes (1-h[i+lag])
            # h[i+lag] appears for i in [0, n-lag-1], contributes -h[i]

                g[j + lag2] += wk * (1.0 - x[j])
                g[j] += wk * (-x[j + lag2])

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sf(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    om = 1.0 - x
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            r = n - lag
            g[:r] += wk * om[lag:]
            g[lag:] -= wk * x[:r]
        else:
            lag2 = -lag
            r = n - lag2
            g[lag2:] += wk * om[:r]
            g[:r] -= wk * x[lag2:]
    return g

def _cf(x, n):
    return np.sum(x) - n / 2.0

def _cj(x, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(_sf, h, jac=_gs, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _pg_step(h, n, step, t_end):
    """Projected gradient steps on the max correlation."""
    c5c = _eval(h)
    for _ in range(5000):
        if time.time() > t_end:
            break
        c = _corr(h)
        idx = np.argmax(c)
        lag = idx - (n - 1)
        # Gradient of corr[idx] w.r.t. h
        g = np.zeros(n)
        if lag >= 0:
            r = n - lag
            g[:r] = 1.0 - h[lag:]
            g[lag:] = -h[:r]
        else:
            lag2 = -lag
            r = n - lag2
            g[lag2:] = 1.0 - h[:r]
            g[:r] = -h[lag2:]
        # Projected gradient step (negative gradient to minimize)
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-18:
            h = h_new
            c5c = c5n
            step = min(step * 1.1, 0.05)
        else:
            step *= 0.7
            if step < 1e-7:
                break
    return h, c5c

def _local_search(h, n, rng, t_end):
    c5c = _eval(h)
    scale = 0.005
    no_imp = 0
    for _ in range(300000):
        if time.time() > t_end:
            break
        # Choose move type
        r = rng.random()
        if r < 0.6:
            # Pair swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                no_imp = 0
                scale = min(scale * 1.03, 0.015)
            else:
                h[i], h[j] = ni - d, nj + d
                no_imp += 1
                if no_imp > 200:
                    scale *= 0.8
                    no_imp = 0
                    if scale < 3e-6:
                        break
        elif r < 0.85:
            # Triplet move: h[i]+d1, h[j]+d2, h[k]-d1-d2
            idx = rng.choice(n, 3, replace=False)
            i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
            d1 = scale * rng.randn()
            d2 = scale * rng.randn()
            ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
            if min(ni, nj, nk) < 0 or max(ni, nj, nk) > 1:
                continue
            oi, oj, ok = h[i], h[j], h[k]
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                no_imp = 0
                scale = min(scale * 1.03, 0.015)
            else:
                h[i], h[j], h[k] = oi, oj, ok
                no_imp += 1
                if no_imp > 200:
                    scale *= 0.8
                    no_imp = 0
                    if scale < 3e-6:
                        break
        else:
            # Block move: shift a small block
            blen = rng.randint(2, 5)
            start = rng.randint(0, n - blen)
            d = scale * 0.5 * rng.randn()
            for b in range(blen):
                h[start + b] += d
            # Adjust one random point to maintain sum
            rest = [x for x in range(n) if x not in range(start, start + blen)]
            adj = rest[rng.randint(len(rest))]
            h[adj] -= d * blen
            if h[adj] < 0 or h[adj] > 1:
                for b in range(blen):
                    h[start + b] -= d
                h[adj] += d * blen
                continue
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                no_imp = 0
                scale = min(scale * 1.03, 0.015)
            else:
                for b in range(blen):
                    h[start + b] -= d
                h[adj] += d * blen
                no_imp += 1
                if no_imp > 200:
                    scale *= 0.8
                    no_imp = 0
                    if scale < 3e-6:
                        break
    return h, c5c

from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sf(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())

    g = np.zeros(n)
    om = 1.0 - x
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            r = n - lag
            g[:r] += wk * om[lag:]
            g[lag:] -= wk * x[:r]
        else:
            lag2 = -lag
            r = n - lag2
            g[lag2:] += wk * om[:r]
            g[:r] -= wk * x[lag2:]
    return g

    return np.sum(x) - n / 2.0

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)

        return _repair(h, n)
