    h = np.clip(h, 0.0, 1.0).copy()

            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:

    return h

    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.exp(s).sum()) / alpha + pen * (x.sum() - target) ** 2
