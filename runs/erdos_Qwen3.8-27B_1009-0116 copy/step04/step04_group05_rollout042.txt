                    step = min(step * 1.15, 2.5)
                    stalled = 0
                    break
                step *= 0.65
            if not ok:
                stalled += 1
                if stalled > 12:
                    break
    return bh, bc

def spectral_phase(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.3, 0.2, 0.1, 0.05, 0.03]:
        h = bh.copy()
        for _ in range(400):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a) * mag + a * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-20:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def coord_desc(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    halts = 0
    for _ in range(400):
        if time.time() - t0 > lim:
            break
        imp = False
        for _ in range(1000):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            for d in (0.006, 0.002, 0.0008, 0.0003, -0.006, -0.002, -0.0008, -0.0003):
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = c5v(h, dx)
                    if c < cc - 1e-15:
                        h = repair(h, n)
                        cc = c5v(h, dx)
                        imp = True
                        if cc < bc:
                            bc, bh = cc, h.copy()
                        break
                    h[i], h[j] = ni - d, nj + d
        halts = halts + 1 if not imp else 0
        if halts >= 6:
            break
    return bh, bc

def full_pipeline(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = lbfgs_opt(bh, n, dx, t0, lim - 200)
    if c < bc: bh, bc = r, c
    r, c = spectral_phase(bh, n, dx, t0, lim - 160)
    if c < bc: bh, bc = r, c
    r, c = active_set(bh, n, dx, t0, lim - 120)
    if c < bc: bh, bc = r, c
    r, c = lbfgs_opt(bh, n, dx, t0, lim - 80)
    if c < bc: bh, bc = r, c
    r, c = spectral_phase(bh, n, dx, t0, lim - 50)
    if c < bc: bh, bc = r, c
    r, c = active_set(bh, n, dx, t0, lim - 30)
    if c < bc: bh, bc = r, c
    r, c = coord_desc(bh, n, dx, rng, t0, lim - 10)
    if c < bc: bh, bc = r, c
    r, c = lbfgs_opt(bh, n, dx, t0, lim - 2)
    if c < bc: bh, bc = r, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        nl = len(initial_h_values)
        h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    except NameError:
        nl = 70
        h0 = repair(rng.random(nl), nl)

    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    # Phase 1: Short DE for global exploration
    n_de = nl
    de_lim = t0 + budget_s * 0.25
    if time.time() < de_lim:
        bounds_de = [(0.0, 1.0)] * n_de
        try:
            de_r = differential_evolution(
                de_obj, bounds_de, args=(n_de,),
                seed=seed, maxiter=100, popsize=12,
                tol=1e-8, mutation=(0.4, 0.9), recomb=0.8,
                polish=False, updating='immediate'
            )
            h_de = repair(de_r.x, n_de)
            c_de = c5v(h_de, dx0)
            if c_de < bc:
                bh, bc = h_de, c_de
        except Exception:
            pass

    # Quick local from DE result
    if time.time() < t0 + budget_s * 0.3:
        r, c = lbfgs_opt(bh, bn, dx0, t0, t0 + budget_s * 0.3)
        if c < bc: bh, bc = r, c

    # Phase 2: Multi-start full pipeline
    candidates = [bh.copy()]
    for i in range(5):
        candidates.append(repair(bh + 0.04 * rng.randn(bn), bn))
    for nb in [3, 5, 7]:
        h_blk = np.zeros(bn)
        bs = bn // nb
        for i in range(nb):
            h_blk[i*bs:i*bs+int(bs*0.45)] = 1.0
        candidates.append(repair(h_blk, bn))
    x_lin = np.linspace(0, 1, bn, endpoint=False)
    for freq in [2, 3, 4]:
        h_sin = 0.5 + 0.35 * np.sin(2*np.pi*freq*x_lin)
        candidates.append(repair(h_sin, bn))

    main_lim = t0 + budget_s * 0.72
    for cand in candidates:
        if time.time() > main_lim:
            break
        c_cand = c5v(cand, dx0)
        r, c = full_pipeline(cand, bn, dx0, seed, main_lim)
        if c < bc:
            bh, bc = r, c

    # Phase 3: Extended active set + spectral
    if time.time() < t0 + budget_s * 0.82:
        r, c = spectral_phase(bh, bn, dx0, t0, t0 + budget_s * 0.82)
        if c < bc: bh, bc = r, c
    if time.time() < t0 + budget_s * 0.9:
        r, c = active_set(bh, bn, dx0, t0, t0 + budget_s * 0.9)
        if c < bc: bh, bc = r, c

    # Phase 4: Fine polishing
    if time.time() < t0 + budget_s * 0.95:
        r, c = coord_desc(bh, bn, dx0, rng, t0, t0 + budget_s * 0.95)
        if c < bc: bh, bc = r, c
    if time.time() < t0 + budget_s - 5:
        r, c = lbfgs_opt(bh, bn, dx0, t0, t0 + budget_s - 3)
        if c < bc: bh, bc = r, c

    # Phase 5: Final micro-polish
    if time.time() < t0 + budget_s - 2:
        rng2 = np.random.RandomState(seed + 777)
        for _ in range(3):
            hp = repair(bh + 0.008 * rng2.randn(bn), bn)
            r, c = active_set(hp, bn, dx0, t0, t0 + budget_s - 2)
            if c < bc:
                bh, bc = r, c
            r, c = coord_desc(bh, bn, dx0, rng2, t0, t0 + budget_s - 1)
            if c < bc: bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
