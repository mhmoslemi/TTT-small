    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * (2.0/n)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()

    lo, hi = -3.0, 3.0

        if np.sum(np.clip(h-m, 0.0, 1.0)) > t: lo = m

        hi, hj = h[i], h[j]
        ni, nj = hi + dl, hj - dl
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, n)
            if c < c5c:
                c5c = c
                if c < bc:
                    bc = c
                    bh = h.copy()
            else:
                h[i], h[j] = hi, hj
    return bh, bc

def _slsqp_obj(x):
    n = _GN
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(_GALPHA * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _GALPHA

def _slsqp_jac(x):
    return _grad_smooth(x, _GN, _GALPHA)

def _slsqp_cf(x):
    return float(np.sum(x) - _GN / 2.0)

def _slsqp_cj(x):
    return np.ones(_GN)

def _slsqp_opt(h, n, alpha, maxiter):
    global _GN, _GALPHA
    _GN = n
    _GALPHA = alpha
    cons = [{'type': 'eq', 'fun': _slsqp_cf, 'jac': _slsqp_cj}]
    bounds = [(0.0, 1.0)] * n
    return minimize(_slsqp_obj, h, jac=_slsqp_jac, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-16})

def _topk_pgd(h, n, k, t_end, lr):
    h = h.copy()
    c5p = _c5(h, n)
    while time.time() < t_end:
        c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        idx = np.argsort(c)[-k:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / k
        g = np.zeros(n)
        for kk in idx:
            lag = int(kk) - (n - 1)
            if lag == 0:
                g += (1 - 2 * h) / k
            elif lag > 0:
                g[lag:] += (1 - h[:n - lag]) / k
                g[:n - lag] -= h[lag:] / k
            else:
                l = -lag
                g[:n - l] += (1 - h[l:]) / k
                g[l:] -= h[:n - l] / k
        hn = _repair(h - lr * g, n)
        c5n = _c5(hn, n)
        if c5n < c5p:
            h = hn
            c5p = c5n
        else:
            lr *= 0.7
    return _repair(h, n), _c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h0)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: Nesterov PGD with smooth max (20%)
    t1 = t0 + budget_s * 0.20
    for alpha in [500, 2000, 8000, 30000]:
        if elapsed() > t1:
            break
        hw, c5w = _pgd_nesterov(best_h, n, alpha, min(t1, t0 + 120), lr=0.06, mom=0.8)
        if c5w < best_c5:
            best_c5 = c5w
            best_h = hw

    # Phase 2: Top-k subgradient (10%)
    t2 = t0 + budget_s * 0.30
    for k in [3, 5]:
        if elapsed() > t2:
            break
        hw, c5w = _topk_pgd(best_h, n, k, min(t2, t0 + 60), lr=0.04)
        if c5w < best_c5:
            best_c5 = c5w
            best_h = hw

    # Phase 3: SLSQP polishing (10%)
    t3 = t0 + budget_s * 0.40
    for alpha in [50000, 150000]:
        if elapsed() > t3:
            break
        try:
            res = _slsqp_opt(best_h.copy(), n, alpha, 250)
            hc = _repair(res.x, n)
            c5 = _c5(hc, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    # Phase 4: Simulated annealing exploration (30%)
    t4 = t0 + budget_s * 0.70
    if elapsed() < t4 - 5:
        bh, bc = _anneal(best_h, n, t4 - 3, rng, T0=8e-5)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh

    # Phase 5: Greedy pair search (20%)
    t5 = t0 + budget_s * 0.90
    best_h, best_c5 = _greedy_pairs(best_h, n, t5, rng)

    # Phase 6: Micro search (10%)
    t6 = t0 + budget_s - 3
    best_h, best_c5 = _micro_search(best_h, n, t6, rng)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
