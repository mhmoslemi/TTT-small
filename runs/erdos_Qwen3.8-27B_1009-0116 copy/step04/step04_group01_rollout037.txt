def _active_descent(h, n_act, n_steps, step0):
    """Gradient descent focusing on top n_act lags."""
    n = len(h)
    dx = 2.0 / n
    step = step0
    for _ in range(n_steps):
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[::-1][:n_act]
        g = np.zeros(n)
        for k in idx:
            sh = (n - 1) - k
            if sh >= 0:
                ns = n - sh
                g[sh:] += (1.0 - h[:ns])
                g[:ns] -= h[sh:]
            else:
                s = -sh
                ns = n - s
                g[:ns] += (1.0 - h[s:])
                g[s:] -= h[:ns]
        g *= dx / n_act
        for _bt in range(30):
            h_new = repair(h - step * g, n)
            cn = eval_c5(h_new)
            co = eval_c5(h)
            if cn < co - 1e-15:
                h = h_new
                break
            step *= 0.5
        else:
            break
        step = min(step * 1.1, step0)
    return h

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            a = min(abs(diff), room)
            if a < 1e-14:
                continue
            h[idx] += a if diff > 0 else -a
            diff = target - h.sum()
            if abs(diff) < 1e-13:
                break
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def interp_resize(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_o = np.arange(n_old) * (2.0 / n_old)
    x_n = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_n, x_o, h)

def _og(h, alpha, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    a = np.clip(alpha * (c - m), -500.0, 500.0)
    am = a.max()

    w = e / e.sum()
    f = m + (am + np.log(e.sum())) / alpha

            g[:ns] -= wk * h[sh:]
        else:
            s = -sh
            ns = n - s
            g[:ns] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:ns]
    if pen > 0:
        sv = h.sum() - n * 0.5
        g += 2.0 * pen * sv
    return f, g

    _g_alpha = alpha
    _g_pen = pen
    n = len(h)

                     bounds=bnds, options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-12})
        return r.x
    except Exception:
        return h
