def active_gradient(h, n, top_k=3):
    """Compute gradient of the top-k correlations combined."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Identify top lags
    top_idx = np.argsort(corr)[-top_k:]
    # Compute gradient for each top lag
    # corr[j] = sum_k h[k] * (1-h[k - (j-(n-1))])
    # Let lag = j - (n-1)
    # corr[j] = sum_{k: valid} h[k] * (1 - h[k - lag])
    # d(corr[j])/d(h[i]) = (1 - h[i-lag]) * [i in valid range for k] 
    #                      + sum_{k: k-lag=i} h[k] * (-1) * [i-lag valid]
    #                      = (1 - h[i-lag]) - h[i+lag] (with bounds)
    
    grad = np.zeros(n)
    for j in top_idx:
        lag = j - (n - 1)
        # corr[j] = sum_{k: 0<=k<n, 0<=k-lag<n} h[k] * (1 - h[k-lag])
        for k in range(max(0, lag), min(n, n + lag)):
            # term: h[k] * (1 - h[k-lag])
            # d/dh[k]: (1 - h[k-lag])
            # d/dh[k-lag]: -h[k]
            if 0 <= k - lag < n:
                grad[k] += (1 - h[k - lag])
                grad[k - lag] += (-h[k])
    return grad / top_k  # average gradient of top lags

        else:

    s = h.sum()
    d = target - s
    if abs(d) > 1e-15:

            h[idx] += min(d, 1.0 - h[idx])
        else:
            idx = np.argmin(h)
            h[idx] += max(d, -h[idx])

def soft_max_obj(v, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_solve(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [
        {'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0,
         'jac': lambda v: np.ones(n)}
    ]
    try:
        res = minimize(
            lambda v: soft_max_obj(v, alpha),
            h, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0}
        )
        return repair(res.x, n)
    except Exception:
        return h

            d -= a
        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    for _ in range(200):
        if abs(s - target) < 1e-14:
            break
        if s < 1e-15:
            h[:] = 0.5
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    d = target - h.sum()
    if d > 0:
        order = np.argsort(1.0 - h)
    elif d < 0:
        order = np.argsort(h)
    else:
        return h
    for idx in order:
        if d > 0:
            a = min(d, 1.0 - h[idx])
            h[idx] += a
            d -= a
        else:
            a = min(-d, h[idx])
            h[idx] -= a
            d += a
        if abs(d) < 1e-14:
            break
    return h

def soft_obj(v, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_solve(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(soft_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def corr_grad(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        for k in range(lag, n):
            grad[k] += 1.0 - h[k - lag]
            grad[k - lag] -= h[k]
    else:
        L = -lag
        for k in range(n - L):
            grad[k] += 1.0 - h[k + L]
            grad[k + L] -= h[k]
    return grad
