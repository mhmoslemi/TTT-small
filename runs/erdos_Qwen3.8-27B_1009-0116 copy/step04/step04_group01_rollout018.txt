def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def _smooth_grad(h, c, alpha):
    n = len(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())

            g[i1] += wk * (1.0 - h[i1 + sh])

            i2 = np.arange(0, n - sp)
