def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_topk(h, k_list, n):
    g = np.zeros(n)
    for k in k_list:
        s = abs(k - (n - 1))
        if s == 0:
            g += 1.0 - 2.0 * h
        elif s < n:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
    return g

def _proj(h0, target):
    n = len(h0)
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h0 - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h0 - (lo + hi) * 0.5, 0.0, 1.0)

def _opt_phase(h, n, t0, deadline, top_m=10):
    target = n / 2.0
    dx = 2.0 / n
    best_c5 = _eval(h)
    best_h = h.copy()
    step = 0.008
    momentum = np.zeros(n)
    beta = 0.85
    stall = 0
    max_stall = 200

    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(c.max() * dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                h = _repair(best_h + 0.005 * np.random.RandomState(stall).randn(n), n)
                momentum[:] = 0
                step = 0.005
                stall = 0
                max_stall = int(max_stall * 1.5)
                continue

        thr_idx = np.argsort(c)[-top_m:]
        g = _grad_topk(h, thr_idx, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        momentum = beta * momentum + (1.0 - beta) * g
        mg = np.linalg.norm(momentum)
        if mg < 1e-14:
            break
        direction = momentum / mg

        accepted = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            h_new = _proj(h + step * factor * (-direction), target)
            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-16:
                h = h_new
                accepted = True
                break
        if not accepted:
            step *= 0.5
            if step < 1e-12:
                break
        else:
            step = min(step * 1.03, 0.04)

    return best_h, best_c5

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_polish(h, n, alpha, maxiter, t0, deadline):
    if time.time() - t0 > deadline:
        return h
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        return _smooth_obj(x, alpha)
    def _con(x):
        return x.sum() - n * 0.5
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _con}],
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        if res.fun < _smooth_obj(h, alpha):
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _upsample(h, factor):
    n = len(h)
    h_hi = np.kron(h, np.ones(factor))
    return h_hi

def _asymptote_search(h, n, t0, deadline, rng):
    """Try asymmetric interval constructions."""
    target = n / 2.0
    best_c5 = _eval(h)
    best_h = h.copy()
    while time.time() < deadline:
        n_int = rng.randint(2, 8)
        widths = rng.dirichlet(np.ones(n_int) * 0.5)
        positions = sorted(rng.randint(0, n, size=n_int))
        gap_widths = widths[:n_int - 1]
        h_cand = np.zeros(n)
        for i in range(n_int):
            w = max(1, int(widths[i] * n * (2.0 / n) / 2.0))
            w = max(1, min(w, n - positions[i]))
            h_cand[positions[i]:positions[i] + w] = 1.0
        h_cand = _repair(h_cand, n)
        c5 = _eval(h_cand)
        if c5 < best_c5 + 0.03:
            h_opt, c5_opt = _opt_phase(h_cand, n, t0, min(time.time() + 8, deadline), top_m=8)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Directed optimization at n0
    h, c5 = _opt_phase(h, n0, t0, t0 + 100, top_m=12)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h, n0

    # Phase 2: SLSQP polish at n0
    if _el() < budget_s - 350:
        h2 = _slsqp_polish(best_h, n0, 30000, 2500, t0, t0 + _el() + 120)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h2, n0

    # Phase 3: Upsample to 2x
    if _el() < budget_s - 350:
        n1 = n0 * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n1)
        h_hi, c5 = _opt_phase(h_hi, n1, t0, t0 + _el() + 180, top_m=15)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_hi, n1
        if _el() < budget_s - 150:
            h2 = _slsqp_polish(best_h, best_n, 80000, 2000, t0, t0 + _el() + 100)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h2, best_n

    # Phase 4: Directed optimization at best_n with more top shifts
    if _el() < budget_s - 250:
        nc = best_n
        h, c5 = _opt_phase(best_h, nc, t0, t0 + _el() + 120, top_m=20)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h, nc

    # Phase 5: Multi-start diverse
    if _el() < budget_s - 200:
        nc = best_n
        for trial in range(25):
            if _el() > budget_s - 200:
                break
            strat = trial % 6
            if strat == 0:
                hp = _repair(best_h + 0.012 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.008 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                for _ in range(rng.randint(1, 4)):
                    bs = rng.randint(5, max(6, nc // 6))
                    st = rng.randint(0, nc - bs)
                    hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 3:
                hp = best_h.copy()
                freqs = rng.randint(1, nc // 2 + 1, size=3)
                for f in freqs:
                    ph = rng.uniform(0, 2 * np.pi)
                    hp += 0.004 * rng.uniform(0.5, 1.0) * np.sin(2 * np.pi * f * np.arange(nc) / nc + ph)
                hp = _repair(hp, nc)
            elif strat == 4:
                idx = rng.permutation(nc)
                hp = np.zeros(nc)
                hp[idx[:nc // 2]] = 1.0
                kernel = np.ones(rng.randint(3, 7))
                hp = np.convolve(hp, kernel / kernel.sum(), mode='same')
                hp = _repair(hp, nc)
            else:
                hp, _ = _asymptote_search(best_h, nc, t0, t0 + _el() + 10, rng)

            h_new, c5_new = _opt_phase(hp, nc, t0, t0 + _el() + 25, top_m=10)
            if c5_new < best_c5:
                best_c5, best_h, best_n = c5_new, h_new, nc

    # Phase 6: Pairwise fine search
    if _el() < budget_s - 60:
        nc = best_n
        h = best_h.copy()
        c5c = _eval(h)
        for scale in [0.004, 0.0015, 0.0005, 0.0002]:
            if _el() > budget_s - 60:
                break
            for _it in range(30000):
                if _el() > budget_s - 60:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-16:
                        c5c = c5n
                        if c5n < best_c5:
                            best_c5 = c5n
                            best_h = h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d

    # Phase 7: Final SLSQP
    if _el() < budget_s - 15:
        h2 = _slsqp_polish(best_h, best_n, 150000, 2500, t0, t0 + _el() + 10)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
