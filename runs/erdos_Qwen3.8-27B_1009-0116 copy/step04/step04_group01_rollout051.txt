def _slsqp_opt(h, alpha, maxiter, ftol, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    c_fun = lambda x: x.sum() - n * 0.5
    c_jac = lambda x: np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    try:
        res = minimize(_lse, h, jac=lambda x: _lse_grad(x, alpha),
                       args=(alpha,), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() < t_end:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _lbfgsb_opt(h, alpha, mu, maxiter, t_end):
    n = len(h)
    def obj(x):
        return _lse(x, alpha) + mu * (x.sum() - n * 0.5) ** 2
    def grad(x):
        return _lse_grad(x, alpha) + 2.0 * mu * (x.sum() - n * 0.5)
    try:
        x_res, f_res, i_res = fmin_l_bfgs_b(obj, h.copy(), fprime=grad, m=30,
                                             pgtol=1e-14, maxiter=maxiter,
                                             iprint=0, factr=1e10,
                                             callback=lambda *a: time.time() > t_end)
        if time.time() < t_end:
            return _repair(x_res, n)
    except Exception:
        pass
    return _repair(h, n)

def _proj_grad(h, alpha, nsteps, step0, t_end):
    n = len(h)
    f = _lse(h, alpha)
    step = step0
    for _ in range(nsteps):
        if time.time() > t_end:
            break
        g = _lse_grad(h, alpha)
        g -= g.mean()
        improved = False
        for _ in range(35):
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            fn = _lse(h_new, alpha)
            if fn < f - 1e-6 * step * np.dot(g, h_new - h):
                h, f = h_new, fn
                improved = True
                break
            step *= 0.5
        if not improved:
            break
        step = min(step * 1.1, step0)
    return h

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _coord_descent(h, n_iter, scale, t_end):
    nc = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i, j = np.random.randint(nc), np.random.randint(nc)
        if i == j:
            continue
        d = scale * np.random.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _golden_init(n, rot):
    t = np.arange(n) / n
    h = (np.mod(t * rot, 1.0) < 0.5).astype(float)
    return _repair(h, n)

def _worst_shift_descend(h, alpha, n_rounds, t_end):
    n = len(h)
    dx = 2.0 / n
    c5_now = _eval(h)
    for _ in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_w = c.argmax()
        sh = (n - 1) - k_w
        g = np.zeros(n)
        if sh >= 0:
            l = n - sh
            g[:l] += (1.0 - h[sh:sh + l])
            if sh > 0:
                g[sh:] -= h[:l]
        else:
            sp = -sh
            l = n - sp
            g[sp:] += (1.0 - h[:l])
            g[:l] -= h[sp:sp + l]
        g *= dx
        g -= g.mean()
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_now - 1e-15:
                h, c5_now = h_new, c5n
                break
    return h, c5_now

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    phi = (1.0 + np.sqrt(5.0)) / 2.0

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: L-BFGS-B sweep at n0
    for alpha in [200, 1000, 5000, 20000, 100000]:
        if _el() > 80:
            break
        for mu in [50, 500, 5000]:
            if _el() > 80:
                break
            hn = _lbfgsb_opt(best_h, alpha, mu, 300, t0 + 80)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5, best_h = c5, hn.copy()

    # Phase 2: SLSQP at n0
    for alpha in [20000, 100000, 500000]:
        if _el() > 200:
            break
        hn = _slsqp_opt(best_h, alpha, 800, 1e-19, t0 + 200)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()
            best_n = n0

    # Phase 3: Worst-shift descent
    if _el() < budget_s - 200:
        hn, c5n = _worst_shift_descend(best_h, 10000, 50, t0 + budget_s - 180)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 4: Upsample to 2x
    if _el() < budget_s - 200:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [1000, 10000, 80000]:
            if _el() > budget_s - 150:
                break
            h_hi = _lbfgsb_opt(h_hi, alpha, 2000, 200, t0 + budget_s - 140)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h = c5, h_hi.copy()
                best_n = n1
        if _el() < budget_s - 100:
            h_hi = _proj_grad(h_hi, 80000, 150, 0.004, t0 + budget_s - 80)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h = c5, h_hi.copy()
                best_n = n1

    # Phase 5: Multi-start with diverse initializations
    n_cur = best_n
    alphas = [50000, 200000]
    for _r in range(25):
        if _el() > budget_s - 150:
            break
        strat = np.random.randint(6)
        if strat == 0:
            hp = _repair(best_h + 0.02 * np.random.randn(n_cur), n_cur)
        elif strat == 1:
            hp = _repair(1.0 - best_h + 0.008 * np.random.randn(n_cur), n_cur)
        elif strat == 2:
            hp = _golden_init(n_cur, phi + np.random.randn() * 0.05)
        elif strat == 3:
            hp = best_h.copy()
            for f in np.random.randint(1, n_cur // 2 + 1, size=4):
                hp += 0.008 * np.random.randn() * np.sin(2 * np.pi * f * np.arange(n_cur) / n_cur)
            hp = _repair(hp, n_cur)
        elif strat == 4:
            hp = (best_h + 0.15 * np.random.randn(n_cur) > 0.5).astype(float)
            hp = _repair(hp, n_cur)
        else:
            hp = best_h.copy()
            for _b in range(np.random.randint(2, 5)):
                bs = np.random.randint(3, max(4, n_cur // 6))
                st = np.random.randint(0, n_cur - bs)
                hp[st:st + bs] += 0.04 * np.random.randn(bs)
            hp = _repair(hp, n_cur)
        al = alphas[np.random.randint(len(alphas))]
        hn = _lbfgsb_opt(hp, al, 2000, 150, t0 + budget_s - 140)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()
            best_n = n_cur

    # Phase 6: Coordinate descent (multiple scales)
    if _el() < budget_s - 40:
        for sc in [0.008, 0.003, 0.0008, 0.0002]:
            if _el() > budget_s - 40:
                break
            best_h, best_c5 = _coord_descent(best_h.copy(), 4000, sc, t0 + budget_s - 40)
            best_n = len(best_h)

    # Phase 7: Final SLSQP polish
    if _el() < budget_s - 15:
        hn = _slsqp_opt(best_h, 1000000, 1500, 1e-20, t0 + budget_s - 5)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()
            best_n = len(hn)

    # Phase 8: Ultra-fine coordinate
    if _el() < budget_s - 3:
        best_h, best_c5 = _coord_descent(best_h.copy(), 10000, 5e-5, t0 + budget_s - 1)
        best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
