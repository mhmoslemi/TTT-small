def repair(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (t / s)
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        m = -L
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g
