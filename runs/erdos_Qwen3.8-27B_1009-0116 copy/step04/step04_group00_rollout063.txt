def _coord_minimax_1d(a, b, bounds=(0.0, 1.0)):
    """
    Solve min_{t in [0,1]} max_k (a[k]*t + b[k])
    """
    # The optimal t is at 0, 1, or at an intersection of two lines
    # where the maximum transition happens
    n = len(a)
    # Collect all candidate t values
    candidates = [0.0, 1.0]
    for i in range(n):
        for j in range(i+1, n):
            if abs(a[i] - a[j]) > 1e-15:
                t = (b[j] - b[i]) / (a[i] - a[j])
                if 0.0 <= t <= 1.0:
                    candidates.append(t)
    # Also add the point where each line is minimized (not applicable for linear)
    best_val = np.inf
    best_t = 0.0
    for t in candidates:
        vals = a * t + b
        m = vals.max()
        if m < best_val:
            best_val = m
            best_t = t
    return best_t, best_val

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _coord_descent(h, n, dx, budget_s, t0, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5_val = _c5(h, dx)
    best_h = h.copy()
    best_c5 = c5_val
    k_arr = np.arange(2 * n - 1)
    
    for pass_i in range(5000):
        if time.time() - t0 > budget_s - 30:
            break
        # Compute correlation
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        
        for i in order:
            old = h[i]
            # Compute d[k] for this coordinate
            ia = i + (k_arr - (n - 1))
            ib = i - (k_arr - (n - 1))
            valid_a = (ia >= 0) & (ia <= n - 1)
            valid_b = (ib >= 0) & (ib <= n - 1)
            term_a = np.where(valid_a, 1.0 - h[np.clip(ia, 0, n - 1)], 0.0)
            term_b = np.where(valid_b, h[np.clip(ib, 0, n - 1)], 0.0)
            d = term_a - term_b

            c_trial[:, n-1] = c[n-1] + (T - old) * (1.0 - T - old)
            f = c_trial.max(axis=1)
            # Add small penalty to encourage diversity (optional)
            best_idx = np.argmin(f)
            best_t = T[best_idx]

                improved = True

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

            break
    
    return best_h, best_c5

def _pair_swap(h, n, dx, budget_s, t0, seed=42):
