import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _lag_grad(h, n, L):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += 1.0 - h[M:]
        g[M:] -= h[:n - M]
    return g

def _grad_topK(h, n, K):
    c = _corr(h)
    idx = np.argsort(c)[::-1][:K]
    g = np.zeros(n)
    for k in idx:
        L = k - (n - 1)
        g += _lag_grad(h, n, L)
    g /= float(K)
    return g

def _coord_opt(h, n, dx, time_limit, rng):
    """Coordinate descent: for each i, find best h[i] via grid search."""
    c5c = _c5(h, dx)
    grid = np.array([0.0, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 0.95, 1.0])
    for sweep in range(30):
        if time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() > time_limit:
                break
            old = h[i]
            best_v = old
            best_c = c5c
            candidates = np.concatenate([grid, np.array([max(0, old - 0.05), max(0, old - 0.02),
                                                         min(1, old + 0.02), min(1, old + 0.05)])])
            for v in candidates:
                if abs(v - old) < 1e-10:
                    continue
                h[i] = v
                h = _repair(h, n)
                c = _c5(h, dx)
                if c < best_c:
                    best_c = c
                    best_v = v
                h[i] = old
            if best_v != old:
                h[i] = best_v
                h = _repair(h, n)
                c5c = best_c
                improved = True
        if not improved:
            break
    return h, c5c

def _pairwise_opt(h, n, dx, time_limit, rng):
    """Focused pairwise swap optimization."""
    c5c = _c5(h, dx)
    dlist = np.array([0.0005, 0.001, 0.002, 0.005, 0.012, 0.025,
                      -0.0005, -0.001, -0.002, -0.005, -0.012, -0.025])
    order_idx = np.arange(n)
    for sweep in range(200):
        if time.time() > time_limit:
            break
        g = _grad_topK(h, n, 8)
        order_idx = np.argsort(np.abs(g))[::-1][:40]
        found = False
        for ii2 in range(len(order_idx)):
            i = order_idx[ii2]
            for jj2 in range(ii2 + 1, min(ii2 + 25, len(order_idx))):
                j = order_idx[jj2]
                best_d = 0.0
                best_c = c5c
                for d in dlist:
                    ni = h[i] + d
                    nj = h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best_c:
                            best_c = c
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    found = True
                    break
            if found:
                break
        if not found:
            break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n = len(initial_h_values)
    dx = 2.0 / n
    h = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Projected gradient with adaptive step (top-5 lags)
    step = 0.012
    t_lim1 = t0 + budget_s * 0.25
    for it in range(8000):
        if elapsed() > t_lim1 - t0:
            break
        g = _grad_topK(h, n, 5)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        found = False
        for s_t in [step, step * 0.5, step * 0.25, step * 0.1]:
            h_new = _repair(h - s_t * g, n)
            c_new = _c5(h_new, dx)
            if c_new < best_c5 - 1e-15:
                h = h_new
                best_h = h.copy()
                best_c5 = c_new
                step = min(step * 1.08, 0.04)
                found = True
                break
        if not found:
            step *= 0.4
            if step < 1e-9:
                break

    # Phase 2: Soft-max L-BFGS with increasing alpha
    t_lim2 = t0 + budget_s * 0.40
    bounds = [(0.0, 1.0)] * n
    target_sum = n / 2.0

    def _soft_new(x, alpha):
        c = _corr(x)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -400, 400)
        s -= np.max(s)
        pen = 80.0 * (np.sum(x) - target_sum) ** 2
        return m + np.log(np.sum(np.exp(s))) / alpha + pen

    def _soft_jac(x, alpha):
        c = _corr(x)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -400, 400)
        s -= np.max(s)
        ew = np.exp(s)
        pw = ew / ew.sum()
        g = np.zeros(len(x))
        for k in range(2 * len(x) - 1):
            L = k - (len(x) - 1)
            pk = pw[k]
            if pk < 1e-15:
                continue
            g += pk * _lag_grad(x, len(x), L)
        g /= alpha
        g += 2.0 * 80.0 * (np.sum(x) - target_sum) * np.ones(len(x))
        return g

    for a_val in [2000, 20000, 200000, 2000000]:
        if elapsed() > t_lim2 - t0 - 2:
            break
        try:
            res = minimize(_soft_new, h.copy(), jac=_soft_jac, args=(a_val,),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-14})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
                h = hc
        except Exception:
            pass

    # Phase 3: Coordinate descent
    t_lim3 = t0 + budget_s * 0.60
    h, c5c = _repair(best_h.copy(), n), best_c5
    h, c5c = _coord_opt(h, n, dx, t_lim3, rng)
    if c5c < best_c5:
        best_c5 = c5c
        best_h = h.copy()

    # Phase 4: More gradient steps (top-3 lags, finer)
    h = best_h.copy()
    t_lim4 = t0 + budget_s * 0.72
    step = 0.006
    for it in range(6000):
        if elapsed() > t_lim4 - t0:
            break
        g = _grad_topK(h, n, 3)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        found = False
        for s_t in [step, step * 0.5, step * 0.25, step * 0.1]:
            h_new = _repair(h - s_t * g, n)
            c_new = _c5(h_new, dx)
            if c_new < best_c5 - 1e-15:
                h = h_new
                best_h = h.copy()
                best_c5 = c_new
                step = min(step * 1.05, 0.02)
                found = True
                break
        if not found:
            step *= 0.5
            if step < 1e-10:
                break

    # Phase 5: Pairwise local search
    t_lim5 = t0 + budget_s * 0.88
    h = best_h.copy()
    h, c5c = _pairwise_opt(h, n, dx, t_lim5, rng)
    if c5c < best_c5:
        best_c5 = c5c
        best_h = h.copy()

    # Phase 6: Tiny random perturbations (final polish)
    h = best_h.copy()
    c5c = _c5(h, dx)
    t_end = t0 + budget_s - 2
    tiny = 0.0003
    while time.time() < t_end:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-tiny, tiny)
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c:
                c5c = c
                best_c5 = c
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
