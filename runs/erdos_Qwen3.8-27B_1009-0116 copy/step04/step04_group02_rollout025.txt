from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def logsumexp_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8, 'maxfun': 10000})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def projected_subgradient_step(h, n, step_size, top_m=10):
    """Take a subgradient step on the max correlation."""
    corr = compute_corr(h)
    # Find top M correlations
    top_idx = np.argsort(corr)[-top_m:]
    # Compute gradient of max as average gradient of top M
    grad = np.zeros(n)
    for k_idx in top_idx:
        # k_idx is index in the full correlation (0 to 2n-2)
        # Convert to actual lag
        lag = k_idx - (n - 1)  # lag ranges from -(n-1) to (n-1)
        for p in range(n):
            g = 1.0
            if 0 <= p + lag < n:
                g -= h[p + lag]
            if 0 <= p - lag < n:
                g -= h[p - lag]
            grad[p] += g
    grad /= top_m
    # Take step
    h_new = h - step_size * grad
    h_new = repair(h_new, n)
    return h_new

    """Subgradient-like search focused on active correlations."""
    best_c5 = compute_c5(h, dx)

    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        lag = k_idx - (n - 1)
        # Compute gradient contribution for this lag
        # ∂c_lag/∂h[p] = (1-h[p+lag]) if 0<=p+lag<n else 0, minus h[p-lag] if 0<=p-lag<n else 0
        for p in range(n):
            g = 0.0
            if 0 <= p + lag < n:
                g += (1.0 - h[p + lag])
            if 0 <= p - lag < n:
                g -= h[p - lag]
            grad[p] += g
    grad /= len(top_idx)
    return grad, corr

def slsqp(h, n, alpha, target, maxiter):

def active_grad(h, n, top_m):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_m:]
    g = np.zeros(n)
    for ki in idx:
        lag = int(ki) - (n - 1)
        for p in range(n):
            pl, pm = p + lag, p - lag
            v = 0.0
            if 0 <= pl < n:
                v += 1.0 - h[pl]
            if 0 <= pm < n:
                v -= h[pm]
            g[p] += v
    g /= len(idx)
    return g, corr

                if i != j:
                    d = rng.uniform(-0.015, 0.015)
                    if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                        h[i] += d
                        h[j] -= d
                stale = 0
        else:
            stale = 0
        if stale % 600 == 599:
            h = repair(h, n)
    return h, best

def fine_search(h, n, dx, rng, deadline):
    best = compute_c5(h, dx)
    h = h.copy()
    for sc in (0.001, 0.0003, 0.00008, 0.00002):
        imp = False
        while time.time() < deadline:
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            if not (0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1):
                continue
            h[i] += d
            h[j] -= d
            c = compute_c5(h, dx)
            if c < best - 1e-14:
                best = c
                imp = True
            else:
                h[i] -= d
                h[j] += d
        if not imp:
            break
    return h, best

def perturb(h, n, rng, k):
    hp = h.copy()
    t = np.arange(n) / n
    if k == 0:
        hp = np.roll(hp, int(rng.randint(3, max(4, n // 4))))
    elif k == 1:
        hp += rng.uniform(0.02, 0.07) * np.sin(2 * np.pi * int(rng.randint(1, 6)) * t + rng.uniform(0, 6.28))
    elif k == 2:
        hp += rng.normal(0, 0.02, n)
    elif k == 3:
        hp += rng.uniform(0.01, 0.04) * np.sin(2 * np.pi * 2 * t) * np.sin(2 * np.pi * 3 * t)
    else:
        hp *= rng.uniform(0.85, 1.15)
    return repair(hp, n)

def frac_nelder(h, n, dx, deadline):
    hwork = h.copy()
    frac = np.minimum(hwork, 1.0 - hwork)
    nf = min(8, n)
    fidx = np.argsort(frac)[-nf:]
    x0 = hwork[fidx].copy()

    def obj_fr(x):
        ht = hwork.copy()
        ht[fidx] = np.clip(x, 0, 1)
        ht = repair(ht, n)
        return compute_c5(ht, dx)
    try:
        res = minimize(obj_fr, x0, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-13})
        hwork[fidx] = np.clip(res.x, 0, 1)
        hwork = repair(hwork, n)
    except Exception:
        pass
    return hwork, compute_c5(hwork, dx)

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n)
    bc = compute_c5(h0, dx)
    bh = h0.copy()

    # P1: L-BFGS sweep (70s)
    t_end = ts + 70
    hc = h0.copy()
    for a in (50, 200, 1000, 5000, 20000):
        if time.time() >= t_end:
            break
        for p in (500, 5000):
            if time.time() >= t_end:
                break
            x, c = lbfgs(hc, n, a, tgt, p, 200)
            if c < bc:
                hc, bc = x, c

    # P2: SLSQP (50s)
    t_end = min(ts + 120, time.time() + 50)
    for a in (500, 2000, 10000, 40000):
        if time.time() >= t_end:
            break
        x, c = slsqp(hc, n, a, tgt, 300)
        if c < bc:
            hc, bc = x, c

    # P3: Gradient-driven search (300s)
    rem = budget_s - (time.time() - ts)
    t_end = time.time() + min(300, rem - 350) if rem > 350 else time.time() + min(100, rem - 30)
    hc, bc = gradient_search(hc, n, dx, rng, t_end)

    # P4: Multi-restart (70s)
    rem = budget_s - (time.time() - ts)
    if rem > 130:
        t_end = time.time() + min(70, rem - 130)
        for r in range(5):
            if time.time() >= t_end:
                break
            hp = perturb(hc, n, rng, r)
            x, c = lbfgs(hp, n, 2000, tgt, 5000, 150)
            x, c = slsqp(x, n, 10000, tgt, 200)
            if c < bc:
                hc, bc = x, c

    # P5: Gradient search round 2 (180s)
    rem = budget_s - (time.time() - ts)
    if rem > 100:
        t_end = time.time() + min(180, rem - 100)
        hc, bc = gradient_search(hc, n, dx, rng, t_end)

    # P6: Fractional Nelder-Mead (40s)
    rem = budget_s - (time.time() - ts)
    if rem > 60:
        hc, bc2 = frac_nelder(hc, n, dx, time.time() + min(40, rem - 50))
        if bc2 < bc:
            bc = bc2

    # P7: Fine random search (80s)
    rem = budget_s - (time.time() - ts)
    if rem > 40:
        t_end = time.time() + min(80, rem - 30)
        hc, bc2 = fine_search(hc, n, dx, rng, t_end)
        if bc2 < bc:
            bc = bc2

    # P8: Final L-BFGS + fine polish (remaining - 10s)
    rem = budget_s - (time.time() - ts)
    if rem > 30:
        x, c = lbfgs(hc, n, 50000, tgt, 10000, 150)
        if c < bc:
            hc, bc = x, c
    rem = budget_s - (time.time() - ts)
    if rem > 10:
        hc, bc2 = fine_search(hc, n, dx, rng, time.time() + max(3, rem - 5))
        if bc2 < bc:
            bc = bc2

    bh = repair(hc, n)
    bc = compute_c5(bh, dx)
    return (bh, bc, n)
