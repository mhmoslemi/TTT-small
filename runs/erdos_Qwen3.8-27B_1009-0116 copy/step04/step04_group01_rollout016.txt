def _min_1d_line_max(A, B, x_lo=0.0, x_hi=1.0, iters=60):
    for _ in range(iters):
        if x_hi - x_lo < 1e-12:
            break
        x_mid = 0.5 * (x_lo + x_hi)
        v = A + B * x_mid
        j_max = v.argmax()
        # subgradient
        v_max = v[j_max]
        # find all j with v[j] close to v_max
        close = np.where(np.abs(v - v_max) < 1e-10)[0]
        if close.size > 1:
            sg_lo = B[close].min()
            sg_hi = B[close].max()
            if sg_lo <= 0 <= sg_hi:
                x_lo = x_hi = x_mid
                break
            elif sg_lo > 0:
                x_hi = x_mid
            else:
                x_lo = x_mid
        else:
            if B[j_max] > 0:
                x_hi = x_mid
            elif B[j_max] < 0:
                x_lo = x_mid
            else:
                x_lo = x_hi = x_mid
                break
    return 0.5 * (x_lo + x_hi)
