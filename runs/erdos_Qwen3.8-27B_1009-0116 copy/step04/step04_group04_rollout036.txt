```python
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _logsumexp_obj(h, alpha, target, pen):
    c = _corr_full(h)
    m = c.max()
    # Stable log-sum-exp
    s = c - m
    s = np.clip(s * alpha, -100, 100)
    lse = np.log(np.exp(s).sum()) / alpha
    obj = m + lse + pen * (h.sum() - target) ** 2
    return obj

def _grad_subgrad(h):
    """Compute subgradient of max_k c_k."""
    n = len(h)
    c = _corr_full(h)
    max_c = c.max()
    std_c = c.std()
    active = c >= max_c - 0.2 * max(std_c, 1e-10)
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        s = int(k - (n - 1))
        if s >= 0:
            m = n - s
            grad[:m] += (1.0 - h[s:])
            grad[s:] -= h[:m]
        else:
            m = n + s
            t = -s
            grad[t:] += (1.0 - h[:m])
            grad[:m] -= h[t:]
    return grad
```

```python
        best_d = 0.0
        best_local = best_c

                d = sign * scale
                hi = h[i] + d
                hj = h[j] - d

                        best_local = c
                        best_d = d

            h[j] -= best_d

            d = rng.uniform(-0.03, 0.03)

                    best_c = c
```

```python
def _subgrad_top(h, top_k=10):
    """Compute weighted subgradient using top-k active shifts."""
    n = len(h)
    c = _corr_full(h)
    # Get top_k shifts
    idx = np.argsort(c)[::-1][:top_k]
    # Weight by how close to max
    weights = c[idx] / c[idx].sum()
    grad = np.zeros(n)
    for w, k in zip(weights, idx):
        s = int(k - (n - 1))
        if s >= 0:
            m = n - s
            grad[:m] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:m]
        else:
            m = n + s
            t = -s
            grad[t:] += w * (1.0 - h[:m])
            grad[:m] -= w * h[t:]
    return grad
```

This is O(n * top_k) which is very fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in range(len(idx)):
            if d > 0:
                a = min(d, 1.0 - h[idx[i]]); h[idx[i]] += a; d -= a
            else:
                a = min(-d, h[idx[i]]); h[idx[i]] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj_feasible(h, target):
    n = len(h)
    def _f(lam):
        return np.clip(h - lam, 0.0, 1.0).sum() - target
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if _f(mid) > 0: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _subgrad_top(h, top_k=12):
    n = len(h)
    c = _corr_full(h)
    idx = np.argsort(c)[::-1][:top_k]
    w = c[idx].copy()
    wsum = w.sum()
    if wsum > 0:
        w /= wsum
    grad = np.zeros(n)
    n_half = n - 1
    for wi, k in zip(w, idx):
        s = int(k - n_half)
        if s >= 0:
            m = n - s
            grad[:m] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:m]
        else:
            m = n + s; t = -s
            grad[t:] += wi * (1.0 - h[:m])
            grad[:m] -= wi * h[t:]
    return grad, c

def _grad_desc(h, dx, t_end, max_iter=20000, seed=1):
    n = len(h)
    target = n / 2.0
    best_c = _c5(h, dx)
    h = h.copy()
    eta = 0.1
    best_eta = eta
    no_imp = 0
    rng = np.random.RandomState(seed)
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        grad, _ = _subgrad_top(h, 15)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        h_t = _proj_feasible(h - eta * grad, target)
        c_t = _c5(h_t, dx)
        if c_t < best_c - 1e-16:
            h = h_t; best_c = c_t; no_imp = 0
            if no_imp <= 10 and eta < 0.3:
                eta = min(eta * 1.08, 0.5); best_eta = eta
        else:
            no_imp += 1; eta *= 0.85
            if no_imp > 50:
                h = _proj_feasible(h + 0.03 * (rng.rand(n) - 0.5), target)
                eta = best_eta * 0.6; no_imp = 0
    return h, best_c

def _coord_desc(h, dx, t_end, max_iter=30000, seed=2):
    n = len(h)
    target = n / 2.0
    best_c = _c5(h, dx)
    h = h.copy()
    rng = np.random.RandomState(seed)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        grad, c = _subgrad_top(h, 10)
        i = int(np.argmax(grad)); j = int(np.argmin(grad))
        if i == j:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j: j = (j+1) % n
        best_d = 0.0; best_lc = best_c
        for sc in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]:
            for sgn in [-1.0, 1.0]:
                d = sgn * sc
                hi, hj = h[i]+d, h[j]-d
                if -1e-9 <= hi <= 1+1e-9 and -1e-9 <= hj <= 1+1e-9:
                    h[i], h[j] = hi, hj
                    cv = _c5(h, dx)
                    if cv < best_lc: best_lc = cv; best_d = d
                    h[i], h[j] = h[i]-d, h[j]+d
        if best_d != 0.0:
            h[i] += best_d; h[j] -= best_d; best_c = best_lc
        elif it % 500 == 0:
            i2, j2 = int(rng.randint(n)), int(rng.randint(n))
            if i2 != j2:
                d = rng.uniform(-0.04, 0.04)
                if 0 <= h[i2]+d <= 1 and 0 <= h[j2]-d <= 1:
                    h[i2] += d; h[j2] -= d
                    cv = _c5(h, dx)
                    if cv < best_c: best_c = cv
                    else: h[i2] -= d; h[j2] += d
    return h, best_c

def _bfgs_run(h, dx, alpha, maxiter, target):
    n = len(h)
    def obj(x):
        c = _corr_full(x)
        m = c.max()
        s = np.clip(alpha*(c-m), -80, 80)
        return m + np.log(np.exp(s).sum())/alpha
    def obj2(x):
        return obj(x) + 5000.0*(x.sum()-target)**2
    bds = [(0.0, 1.0)]*n
    try:
        r = minimize(obj2, h, method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        hc = _repair(r.x, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _slsqp_run(h, dx, alpha, maxiter, target):
    n = len(h)
    def obj(x):
        c = _corr_full(x)
        m = c.max()
        s = np.clip(alpha*(c-m), -80, 80)
        return m + np.log(np.exp(s).sum())/alpha
    bds = [(0.0, 1.0)]*n
    cons = [{'type':'eq', 'fun': lambda x: x.sum()-target}]
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _interp1d(h, n_new):
    return np.interp(np.linspace(0,1,n_new), np.linspace(0,1,len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values)/2.0)
    n0 = len(h0); dx0 = 2.0/n0
    best_h, best_c5, best_n = h0, _c5(h0, dx0), n0

    def rem():
        return budget_s - (time.time()-t0)

    # Phase 1: BFGS at n0 (180s)
    t_end = min(t0+180, t0+budget_s-400)
    for a in [2000, 10000, 50000]:
        if time.time() > t_end: break
        hc, c = _bfgs_run(best_h, dx0, a, 500, n0/2.0)
        if c < best_c5: best_h, best_c5 = hc, c

    # Phase 2: Gradient descent (200s)
    t_end = min(t0+380, t0+budget_s-250)
    best_h, best_c5 = _grad_desc(best_h, dx0, t_end, 20000, seed)

    # Phase 3: SLSQP polish (100s)
    t_end = min(t0+480, t0+budget_s-150)
    for a in [20000, 100000]:
        if time.time() > t_end: break
        hc, c = _slsqp_run(best_h, dx0, a, 600, n0/2.0)
        if c < best_c5: best_h, best_c5 = hc, c

    # Phase 4: Coordinate descent (120s)
    t_end = min(t0+600, t0+budget_s-80)
    best_h, best_c5 = _coord_desc(best_h, dx0, t_end, 30000, seed+7)

    # Phase 5: Multi-resolution
    for res in [2, 4]:
        if rem() < 120: break
        n_new = n0 * res
        dx_new = 2.0 / n_new
        h_new = _repair(_interp1d(best_h, n_new), n_new/2.0)
        c_new = _c5(h_new, dx_new)
        t_end = min(t0+budget_s-60, t0+time.time()-t0+min(150, rem()-30))
        # BFGS at new res
        for a in [5000, 30000]:
            if time.time() > t_end: break
            hc, c = _bfgs_run(h_new, dx_new, a, 400, n_new/2.0)
            if c < c_new: h_new, c_new = hc, c
        # Gradient desc
        if time.time() < t_end:
            t_end2 = min(t_end, t_end2 if 't_end2' in dir() else t_end)
            h_new, c_new = _grad_desc(h_new, dx_new, min(time.time()+80, t_end), 8000, seed+res)
        # Coord desc
        if time.time() < t_end:
            h_new, c_new = _coord_desc(h_new, dx_new, t_end, 15000, seed+10+res)
        if c_new < best_c5:
            best_h, best_c5, best_n = h_new, c_new, n_new

    # Phase 6: Final intensive at best resolution
    nb = best_n; dxb = 2.0/nb
    t_end = t0 + budget_s - 30
    best_h, best_c5 = _coord_desc(best_h, dxb, min(time.time()+100, t_end), 20000, seed+99)
    if time.time() < t_end - 15:
        best_h, best_c5 = _grad_desc(best_h, dxb, min(time.time()+60, t_end), 10000, seed+200)

    best_h = _repair(best_h, best_n/2.0)
    best_c5 = _c5(best_h, 2.0/best_n)
    return (best_h, best_c5, best_n)
```
