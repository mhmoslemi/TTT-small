import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    # Proportional scaling with clipping
    for _ in range(100):
        h2 = np.clip(h * target / s, 0.0, 1.0)
        if abs(h2.sum() - target) < 1e-14:
            return h2
        s = h2.sum()
        if s < 1e-12:
            return np.full(n, 0.5)
    # Fallback: greedy
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 0:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff < 1e-14:
                break
    else:
        order = np.argsort(h)
        for idx in order:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if diff > -1e-14:
                break
    return h

    """Project h onto {x: 0<=x<=1, sum x = n/2}"""
    target = n / 2.0
    h = h.copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
    return np.clip(hp, 0.0, 1.0)
