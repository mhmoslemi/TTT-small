import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if abs(d) < 1e-13:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def topk_smooth(corr, k, alpha):
    top = np.sort(corr)[::-1][:k]
    m = top[0]
    s = np.clip(alpha * (top - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def power_mean(corr, p):
    c = np.maximum(corr, 0.0)
    return float(np.sum(c ** (2 * p)) ** (1.0 / (2.0 * p)))

def slsqp_obj(v, k, alpha, n):
    c = np.correlate(v, 1.0 - v, mode='full')
    return topk_smooth(c, k, alpha)

def slsqp_opt(h, n, k, alpha, maxiter=500):
    def obj(v):
        return slsqp_obj(v, k, alpha, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(res.x, n)
    except Exception:
        return h

def finest_cd(h, n, t_end, rng, n_iter=500000):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = np.max(compute_corr(h)) * dx
    steps = [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05]
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        hi, hj = h[i], h[j]
        improved = False
        for d in steps:
            if improved:
                break
            for sign in (1, -1):
                ni, nj = hi + sign * d, hj - sign * d
                if ni < -1e-10 or ni > 1 + 1e-10 or nj < -1e-10 or nj > 1 + 1e-10:
                    continue
                h[i], h[j] = ni, nj
                c5 = np.max(compute_corr(h)) * dx
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
                    break
                h[i], h[j] = hi, hj
    return h, best_c5

def targeted_cd(h, n, t_end, rng, n_iter=300000):

def resample_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, h_old)

def power_mean_slsqp(h, n, p, maxiter=400):
    def obj(v):
        c = np.correlate(v, 1.0 - v, mode='full')
        return power_mean(c, p)

        return h
