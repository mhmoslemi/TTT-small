        c3, d3 = opt_coord(a3, n3, dx3, lambda: remain() - 8)
        if d3 < best_c5:
            best_h, best_c5, n, dx = c3, d3, n3, dx3

    # Try n=200 (upsample)
    if remain() > 55:
        n4 = 200; dx4 = 2.0/n4
        hi = np.interp(np.linspace(0,2,n4,endpoint=False),
                       np.linspace(0,2,n,endpoint=False), best_h)
        hi = _repair(hi, n4)
        a4, b4 = opt_gradient(hi, n4, dx4, lambda: remain() - 25)
        c4, d4 = opt_coord(a4, n4, dx4, lambda: remain() - 8)
        if d4 < best_c5:
            best_h, best_c5, n, dx = c4, d4, n4, dx4

    # Final intense polish
    if remain() > 4:
        hf = best_h.copy(); c5f = _c5(hf, dx); nf = len(hf)
        d_tiny = np.array([0.0001, 0.0003, 0.001, -0.0001, -0.0003, -0.001])
        for _ in range(80000):
            if remain() < 3: break
            i, j = rng.randint(nf), rng.randint(nf)
            if i == j: continue
            d = d_tiny[rng.randint(len(d_tiny))]
            ni, nj = hf[i]+d, hf[j]-d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                hf[i], hf[j] = ni, nj
                c = _c5(hf, dx)
                if c < c5f:
                    c5f = c; best_c5 = c; best_h = hf.copy()
                else:
                    hf[i], hf[j] = ni-d, nj+d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5(best_h, 2.0/len(best_h))
    return (best_h, best_c5, len(best_h))
