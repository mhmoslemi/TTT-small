def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _slsqp(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.exp(s).sum()) / alpha
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _slsqp(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.exp(s).sum()) / alpha
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)

def _pair_cd(h, dx, deadline, n_top=6, gs_iter=20):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    tgt = n / 2.0
    pass_count = 0
    while time.time() < deadline:
        pass_count += 1
        cv = _corr(h)
        top_lags = np.argsort(cv)[-n_top:]
        for lag_idx in top_lags:
            shift = lag_idx - (n - 1)
            if shift == 0:
                continue
            s = abs(shift)
            ilen = n - s
            if ilen < 2:
                continue
            for i in range(ilen):
                if time.time() > deadline:
                    break
                j = i + s
                a0 = h[i]; b0 = h[j]
                t_lo = max(-a0, b0 - 1.0)
                t_hi = min(1.0 - a0, b0)
                if t_hi - t_lo < 1e-10:
                    continue
                lo = t_lo; hi = t_hi
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                h[i] = a0 + c1; h[j] = b0 - c1
                f1 = _c5(h, dx)
                h[i] = a0 + c2; h[j] = b0 - c2
                f2 = _c5(h, dx)
                for _ in range(gs_iter):
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo)
                        h[i] = a0 + c1; h[j] = b0 - c1
                        f1 = _c5(h, dx)
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo)
                        h[i] = a0 + c2; h[j] = b0 - c2
                        f2 = _c5(h, dx)
                t_best = (lo + hi) / 2.0
                h[i] = a0 + t_best; h[j] = b0 - t_best
                c = _c5(h, dx)
                if c < best:
                    best = c
                else:
                    h[i] = a0; h[j] = b0
        h = _repair(h, tgt)
        c = _c5(h, dx)
        if c < best:
            best = c
    return h, best

def _fine_local(h, dx, rng, deadline, n_top=5):

    h = h.copy()

            else:
                if shift > 0:
                    h[j] += d; h[i] -= d
                else:
                    h[i] += d; h[j] -= d
    return h, best

def _multi_restart_slsqp(h, dx, rng, deadline, n_restarts=5):
    n = len(h)
    tgt = n / 2.0
    best_h = h.copy()
    best_c = _c5(h, dx)

        hp = h + 0.05 * (1.0 - r / float(n_restarts + 1)) * (2.0 * rng.random(n) - 1.0)
        hp = _repair(hp, tgt)

                break
            hp, c = _slsqp(hp, dx, alpha, 400)
            if c < best_c:
                best_h, best_c = hp, c
    return best_h, best_c
