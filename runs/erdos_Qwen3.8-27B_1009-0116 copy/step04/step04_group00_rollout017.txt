def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    n = len(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)

        else:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    n = len(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-16:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    if lam > 0:
        g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _sharp_grad(h, power=2.0, top_frac=0.25):
    c = _corr(h)
    n = len(h)
    n_act = max(4, int(top_frac * (2 * n - 1)))
    cs = np.sort(c)[::-1]
    thresh = cs[min(n_act - 1, len(cs) - 1)]
    mask = c >= thresh
    w = np.zeros(2 * n - 1)
    w[mask] = (c[mask] - thresh + 1e-12) ** power
    ws = w.sum()
    if ws < 1e-15:
        w[mask] = 1.0 / max(1, mask.sum())
        ws = 1.0
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        L = k - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _grad_lbfgs(h, n, dx, bounds, deadline):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    alphas = [100, 400, 1500, 6000, 25000, 100000, 400000, 1500000, 6000000, 20000000]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_softobj, best_h.copy(), args=(a, 80.0),
                           jac=_softgrad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-14})
            hc = _repair(res.x, n)
            cv = _c5(hc, dx)
            if cv < best_c5:
                best_c5 = cv
                best_h = hc
        except Exception:
            pass
    return best_h, best_c5

def _grad_desc(h, n, dx, time_budget, deadline):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    h = best_h.copy()
    c5c = best_c5
    step = 0.004
    max_iters = 3000
    for it in range(max_iters):
        if time.time() > deadline or it > max_iters:
            break
        g = _sharp_grad(h, power=2.0, top_frac=0.2)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-14:
            break
        g /= ng
        ok = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1]:
            h_new = _repair(h - s * g, n)
            cn = _c5(h_new, dx)
            if cn < c5c - 1e-15:
                h = h_new
                c5c = cn
                ok = True
                if c5c < best_c5:
                    best_c5 = c5c
                    best_h = h.copy()
                break
        if ok:
            step = min(step * 1.04, 0.018)
        else:
            step *= 0.4
            if step < 1e-9:
                break
    return best_h, best_c5

def _pairwise(h, n, dx, deadline, max_sweeps=80):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    h = best_h.copy()
    c5c = best_c5
    dl = np.array([0.0004, 0.001, 0.004, 0.01, 0.02,
                   -0.0004, -0.001, -0.004, -0.01, -0.02])
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        g = _sharp_grad(h, power=1.5, top_frac=0.3)
        order = np.argsort(np.abs(g))[::-1][:30]
        improved = False
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + 10, len(order))):
                j = order[jj]
                bd = 0.0
                bc = c5c
                for d in dl:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < bc:
                            bc = c
                            bd = d
                        h[i], h[j] = ni - d, nj + d
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    improved = True
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_h = h.copy()
                    break
        if not improved:
            break
    return best_h, best_c5

def _rand_search(h, n, dx, deadline, seed_val=777):
    rng = np.random.RandomState(seed_val)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    h = best_h.copy()
    c5c = best_c5
    n_iters = int(max(0, (deadline - time.time()) * 4000))
    n_iters = min(n_iters, 80000)
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.002, 0.002)
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c:
                c5c = c
                best_c5 = c
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _run_grid(h_init, n, dx, time_left_fn, min_time=30):
    if time_left_fn() < min_time:
        return h_init, _c5(h_init, dx)
    bounds = [(0.0, 1.0)] * n
    deadline1 = time.time() + max(10, time_left_fn() * 0.35 - 30)
    h1, c1 = _grad_lbfgs(h_init, n, dx, bounds, deadline1)
    if time_left_fn() < min_time:
        return h1, c1
    deadline2 = time.time() + max(10, min(60, time_left_fn() * 0.25))
    h2, c2 = _grad_desc(h1, n, dx, time_left_fn(), deadline2)
    if time_left_fn() < min_time:
        return h2, c2
    deadline3 = time.time() + max(5, min(30, time_left_fn() * 0.2))
    h3, c3 = _pairwise(h2, n, dx, deadline3)
    if time_left_fn() < min_time:
        return h3, c3
    deadline4 = time.time() + max(3, min(15, time_left_fn() * 0.15))
    h4, c4 = _rand_search(h3, n, dx, deadline4)
    return h4, c4
