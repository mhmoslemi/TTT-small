from scipy.optimize import minimize
import time

def _c5(h):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/len(h))

def _repair(h, tgt=None):
    n = len(h)
    if tgt is None: tgt = n/2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        d = tgt - h.sum()
        if abs(d) < 1e-14: break
        h = np.clip(h + d/n, 0.0, 1.0)

                a = min(d, 1-h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
    return h

def _interp(h, nn):
    return np.interp(np.linspace(0,1,nn), np.linspace(0,1,len(h)), h)

def _grad_corr(h, alpha):

    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    sm = m + np.log(e.sum()) / alpha
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12: continue
        shift = k - (n-1)
        if shift >= 0:
            if shift == 0:
                grad += w[k] * (1.0 - 2.0*h)
            else:
                grad[shift:] += w[k] * (1.0 - h[:-shift])
                grad[:n-shift] += w[k] * (-h[shift:])
        else:
            s = -shift
            grad[:n-s] += w[k] * (1.0 - h[s:])
            grad[s:] += w[k] * (-h[:n-s])
    return sm, grad

def _lbfgs_opt(h, alpha, maxiter, t_end):
    n = len(h)
    tgt = n/2.0
    bds = [(0.0, 1.0)]*n
    
    def obj(x):
        v, _ = _grad_corr(x, alpha)
        pen = 1e6 * (x.sum() - tgt)**2
        return v + pen
    
    def jac(x):
        _, g = _grad_corr(x, alpha)
        d = x.sum() - tgt
        g += 2e6 * d * np.ones(n)
        return g
    
    best = h.copy()
    best_c = _c5(h)
    try:

        hc = _repair(r.x, tgt)
        c = _c5(hc)
        if c < best_c:
            best, best_c = hc, c
    except Exception:
        pass
    return best, best_c

            s2 = -shift
            grad[:n - s2] += wk * (1.0 - h[s2:])
            grad[s2:] += wk * (-h[:n - s2])
    return val, grad

def _lbfgs(h, alpha, maxit, tgt=None):
    n = len(h)
    if tgt is None:
        tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    pen = 1e5
    def obj(x):
        v, _ = _obj_grad(x, alpha)
        return v + pen * (x.sum() - tgt) ** 2
    def jac(x):
        _, g = _obj_grad(x, alpha)
        g = g + 2.0 * pen * (x.sum() - tgt)
        return g
    try:
        r = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-10})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc)
    except Exception:
        return h, _c5(h)

def _slsqp(h, alpha, maxit, tgt=None):
    n = len(h)
    if tgt is None:
        tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        v, _ = _obj_grad(x, alpha)
        return v
    def jac(x):
        _, g = _obj_grad(x, alpha)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt, 'jac': lambda x: ones}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc)
    except Exception:
        return h, _c5(h)

def _targeted_ls(h, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    best_c = _c5(h)
    h = h.copy()
    scales = [0.008, 0.004, 0.002, 0.001, 0.0005]
    per = max(50, n_iter // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() > t_end:
                return h, best_c
            c = np.correlate(h, 1.0 - h, mode='full')
            m = c.max()
            thresh = m - 0.15 * m
            active = np.where(c > thresh)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            k = active[rng.randint(len(active))]
            shift = k - (n - 1)
            if shift > 0:
                pairs = np.arange(n - shift)
                score = h[pairs + shift] * (1.0 - h[pairs])
                top = np.argsort(score)[-5:]
                p = top[rng.randint(len(top))]
                i, j = int(p), int(p + shift)
                d = sc * (1.0 if h[i] < h[j] else -1.0) * (0.5 + rng.random())
            elif shift < 0:
                s2 = -shift
                pairs = np.arange(n - s2)
                score = h[pairs] * (1.0 - h[pairs + s2])
                top = np.argsort(score)[-5:]
                p = top[rng.randint(len(top))]
                i, j = int(p + s2), int(p)
                d = sc * (1.0 if h[i] < h[j] else -1.0) * (0.5 + rng.random())
            else:
                mid = (h > 0.02) & (h < 0.98)
                if mid.sum() < 2:
                    i, j = rng.randint(n), rng.randint(n)
                else:
                    idxs = np.where(mid)[0]
                    i = idxs[rng.randint(len(idxs))]
                    j = idxs[rng.randint(len(idxs))]
                if i == j:
                    continue
                d = sc * 2.0 * (1.0 if h[i] < 0.5 else -1.0)
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                cn = _c5(h)
                if cn < best_c:
                    best_c = cn
                else:
                    h[i], h[j] = ni - d, nj + d
        h = _repair(h, tgt)
    return h, best_c

def _sim_anneal(h, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    best_h, best_c = h.copy(), _c5(h)
    cur_c = best_c
    T0, T1 = 0.008, 0.00005
    for it in range(n_iter):
        if time.time() > t_end:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_iter))
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.015, 0.015)
        ni, nj = h[i] + d, h[j] - d
        if not (0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0):
            continue
        h[i], h[j] = ni, nj
        cn = _c5(h)
        if cn <= cur_c or rng.random() < np.exp(-(cn - cur_c) / max(T, 1e-12)):
            cur_c = cn
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
        if it % 500 == 0:
            h = _repair(h, tgt)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy())
    n0 = len(h0)
    best_h, best_c5, best_n = h0.copy(), _c5(h0), n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS at n0
    t_end = t0 + 90
    for a in [80, 400, 2000, 8000]:
        if time.time() > t_end or rem() < 500:
            break
        h_new, c = _lbfgs(best_h, a, 600)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Phase 2: SLSQP at n0
    t_end = t0 + 170
    for a in [2000, 10000, 40000]:
        if time.time() > t_end or rem() < 400:
            break
        h_new, c = _slsqp(best_h, a, 500)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Phase 3: Targeted local search at n0
    t_end = t0 + 300
    best_h, best_c5 = _targeted_ls(best_h, rng, t_end, 150000)

    # Phase 4: Interpolate to 2n, optimize
    n1 = 2 * n0
    h1 = _repair(_interp(best_h, n1))
    c1 = _c5(h1)
    t_end = min(t0 + 550, t0 + budget_s - 350)
    for a in [400, 2000, 10000]:
        if time.time() > t_end or rem() < 300:
            break
        h1, c1 = _lbfgs(h1, a, 500)
    for a in [5000, 30000]:
        if time.time() > t_end or rem() < 250:
            break
        h1, c1 = _slsqp(h1, a, 500)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, n1

    # Phase 5: Targeted LS at best res
    t_end = min(t0 + 750, t0 + budget_s - 250)
    best_h, best_c5 = _targeted_ls(best_h, rng, t_end, 250000)

    # Phase 6: Simulated annealing
    t_end = min(t0 + 900, t0 + budget_s - 150)
    h_sa, c_sa = _sim_anneal(best_h, rng, t_end, 80000)
    if c_sa < best_c5:
        best_h, best_c5 = h_sa, c_sa

    # Phase 7: Alternative start - sinusoidal perturbation
    if rem() > 100:
        nb = best_n
        xp = np.linspace(0, 1, nb)
        h_alt = 0.5 + 0.45 * np.sin(2.0 * np.pi * 2.0 * xp)
        h_alt = _repair(h_alt, nb / 2.0)
        h_alt = 0.7 * best_h + 0.3 * h_alt
        h_alt = _repair(h_alt, nb / 2.0)
        t_end2 = min(t0 + budget_s - 80, time.time() + 60)
        h_alt, c_alt = _lbfgs(h_alt, 3000, 400)
        if c_alt < best_c5:
            best_h, best_c5 = h_alt, c_alt
        h_alt, c_alt = _targeted_ls(h_alt, rng, t_end2, 50000)
        if c_alt < best_c5:
            best_h, best_c5 = h_alt, c_alt

    # Phase 8: Final polish
    if rem() > 40:
        t_end = t0 + budget_s - 20
        best_h, best_c5 = _slsqp(best_h, 100000, 400)
    if rem() > 15:
        t_end = t0 + budget_s - 3
        best_h, best_c5 = _targeted_ls(best_h, rng, t_end, 100000)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h)
    return (best_h, best_c5, best_n)
