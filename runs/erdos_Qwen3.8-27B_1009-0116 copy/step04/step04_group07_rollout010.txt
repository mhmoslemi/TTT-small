    return h, best

                    improved = True
                else:
                    h[i] = h[i]  # already back
                    h = repair(h, n)  # restore
        if not improved:
            break
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h = np.minimum(h + room * (diff / tr), 1.0)
        else:
            mass = h.copy()
            tm = mass.sum()
            if tm < 1e-15:
                break
            h = np.maximum(h - mass * (-diff / tm), 0.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def soft_obj(v, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_opt(h, n, alpha, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(soft_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return h

def pair_sweep(h, n, steps, t_end, max_passes=10000):
    h = h.copy()
    best = compute_c5(h, n)
    no_improve = 0
    for p in range(max_passes):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            hi = h[i]
            for j in range(i + 1, n):
                hj = h[j]
                for d in steps:
                    hit = False
                    for sign in (1, -1):
                        ni = hi + sign * d
                        nj = hj - sign * d
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            h[i], h[j] = ni, nj
                            c = compute_c5(h, n)
                            if c < best - 1e-14:
                                best = c
                                improved = True
                                hi, hj = ni, nj
                                hit = True
                                break
                            h[i], h[j] = hi, hj
                    if hit:
                        break
            if time.time() > t_end:
                return h, best
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return h, best

def transfer_n(h, n_old, n_new):
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h2 = np.interp(x_new, x_old, h)
    return repair(h2, n_new)

def perturbed(h, n, sigma, rng):
    h2 = h + sigma * rng.randn(n)
    return repair(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 7)
    best_h = None
    best_c5 = np.inf
    best_n = 64

    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(init_h)
        best_h = repair(init_h, best_n)
        best_c5 = compute_c5(best_h, best_n)
    except NameError:
        best_h = repair(np.full(64, 0.5), 64)
        best_c5 = compute_c5(best_h, 64)
        best_n = 64

    n = best_n
    h_work = best_h.copy()

    # Phase 1: SLSQP refinement (escalating alpha)
    t_limit = t0 + budget_s * 0.12
    for alpha in [3000, 15000, 80000, 400000, 1500000]:
        if time.time() > t_limit:
            break
        h_new = slsqp_opt(h_work, n, alpha, 600)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new
        elif c5_new < compute_c5(h_work, n) - 1e-14:
            h_work = h_new

    # Phase 2: Coarse pair sweep
    t_limit = t0 + budget_s * 0.30
    h_work, c5_w = pair_sweep(h_work, n, [0.08, 0.04, 0.02], t_limit)
    if c5_w < best_c5 - 1e-14:
        best_h, best_c5 = h_work.copy(), c5_w

    # Phase 3: Medium pair sweep
    t_limit = t0 + budget_s * 0.50
    h_work, c5_w = pair_sweep(h_work, n, [0.01, 0.005, 0.002], t_limit)
    if c5_w < best_c5 - 1e-14:
        best_h, best_c5 = h_work.copy(), c5_w

    # Phase 4: Fine pair sweep
    t_limit = t0 + budget_s * 0.68
    h_work, c5_w = pair_sweep(h_work, n, [0.001, 0.0005, 0.0002, 0.0001], t_limit)
    if c5_w < best_c5 - 1e-14:
        best_h, best_c5 = h_work.copy(), c5_w

    # Phase 5: Multi-n exploration
    t_limit = t0 + budget_s * 0.85
    for n_alt in [48, 56, 72, 80, 96]:
        if time.time() > t_limit:
            break
        if n_alt == best_n:
            continue
        h_seed = transfer_n(best_h, best_n, n_alt)
        h_alt = slsqp_opt(h_seed, n_alt, 100000, 400)
        h_alt, c5_alt = pair_sweep(h_alt, n_alt, [0.01, 0.005, 0.002],
                                   min(t0 + budget_s * 0.87, t_limit))
        h_alt, c5_alt = pair_sweep(h_alt, n_alt, [0.001, 0.0005],
                                   min(t0 + budget_s * 0.89, t_limit))
        if c5_alt < best_c5 - 1e-14:
            best_h, best_c5, best_n = h_alt.copy(), c5_alt, n_alt
            h_work = h_alt.copy()
        else:
            h_seed2 = transfer_n(h_work, n, n_alt)
            h_alt2, c5_alt2 = pair_sweep(h_seed2, n_alt, [0.005, 0.002, 0.001],
                                         min(t0 + budget_s * 0.90, t_limit))
            if c5_alt2 < best_c5 - 1e-14:

    # Phase 6: Perturbation + re-optimization on best
    t_limit = t0 + budget_s * 0.94
    best_n = len(best_h)
    for trial in range(6):
        if time.time() > t_limit:
            break
        sigma = 0.02 + 0.015 * trial
        h_p = perturbed(best_h, best_n, sigma, rng)
        h_p = slsqp_opt(h_p, best_n, 200000, 300)
        h_p, c5_p = pair_sweep(h_p, best_n, [0.005, 0.002, 0.001, 0.0005],
                               min(t0 + budget_s * 0.95, t_limit))
        if c5_p < best_c5 - 1e-14:
            best_h, best_c5 = h_p.copy(), c5_p

    # Phase 7: Final ultra-fine polish
    t_limit = t0 + budget_s * 0.99
    best_n = len(best_h)
    h_fin = slsqp_opt(best_h, best_n, 2000000, 400)
    c5_fin = compute_c5(h_fin, best_n)
    if c5_fin < best_c5 - 1e-14:
        best_h, best_c5 = h_fin.copy(), c5_fin

    if c5_fin2 < best_c5 - 1e-14:
        best_h, best_c5 = h_fin2.copy(), c5_fin2

    best_n = len(best_h)
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
