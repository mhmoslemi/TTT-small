def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)

        else:

                diff -= add

def _repair(h, n):

    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()

                break

                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:

def _project(h, n):

        else:

    c = np.correlate(h, 1.0 - h, mode='full')
    # Weight top k lags

            child = _project(child, n)

            fit[i] = _c5(child, n)
        bi = fit.argmin()
        if fit[bi] < best_c5 - 1e-13:
            best_c5 = fit[bi]
            best_h = pop[bi].copy()
            stall = 0
        else:
            stall += 1
        pop[bi] = best_h
        if stall % 10 == 9:
            sigma *= 0.82
        if stall > 25:
            k = (bi + 1 + rng.randint(pop_size - 1)) % pop_size
            pop[k] = _repair(best_h + rng.randn(n) * sigma * 5, n)
            fit[k] = _c5(pop[k], n)
            stall = 0
    return best_h, best_c5

def _pair_search(h, n, end_t, rng):
    h = h.copy()
    c5_cur = _c5(h, n)
    best_c5 = c5_cur
    best_h = h.copy()
    deltas_big = np.array([0.05, 0.02, 0.01, 0.005, -0.05, -0.02, -0.01, -0.005])
    deltas_small = np.array([0.002, 0.001, 0.0005, -0.002, -0.001, -0.0005])
    # Big deltas first
    for dset in [deltas_big, deltas_big, deltas_small, deltas_small, deltas_small]:
        if time.time() > end_t:
            break
        for _pass in range(80):
            if time.time() > end_t:
                break
            improved = False
            idx = np.arange(n)
            rng.shuffle(idx)
            for i in idx:
                if time.time() > end_t:
                    break
                hi = h[i]
                for j in idx:
                    if j == i:
                        continue
                    hj = h[j]
                    for d in dset:
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, n)
                        if c < c5_cur - 1e-15:
                            c5_cur = c
                            improved = True
                            if c < best_c5:
                                best_c5 = c
                                best_h = h.copy()
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
            if not improved:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)
    best_n = n

    # Phase 1: Evolutionary exploration (38% budget)
    e1 = t0 + budget_s * 0.38
    bh, bc = _evolve(best_h, best_n, rng, e1, pop_size=24)
    if bc < best_c5:
        best_h, best_c5 = bh, bc

    # Phase 2: Focused gradient descent on top lags (22% budget)
    e2 = t0 + budget_s * 0.60
    for _rnd in range(8):
        if time.time() > e2:
            break
        if _rnd > 0 and _rnd % 3 == 0:
            h_start = _repair(best_h + 0.015 * rng.randn(best_n), best_n)
        else:
            h_start = best_h.copy()
        bh2, bc2 = _grad_descent(h_start, best_n, rng, e2, lr0=0.006 + 0.002 * rng.rand())
        if bc2 < best_c5:
            best_h, best_c5 = bh2, bc2

    # Phase 3: Adaptive pair search (30% budget)
    e3 = t0 + budget_s * 0.88
    bh3, bc3 = _pair_search(best_h, best_n, e3, rng)
    if bc3 < best_c5:
        best_h, best_c5 = bh3, bc3

    # Phase 4: Secondary resolution attempt
    if time.time() < t0 + budget_s * 0.96:
        n2 = best_n + 40
        if n2 != best_n:
            x1 = np.linspace(0, 2.0, best_n, endpoint=False)
            x2 = np.linspace(0, 2.0, n2, endpoint=False)
            h2 = np.interp(x2, x1, best_h, period=2.0)
            h2 = _repair(h2, n2)
            h2, c2 = _grad_descent(h2, n2, rng, t0 + budget_s * 0.96, lr0=0.004)
            if c2 < best_c5 - 2e-6:
                best_h, best_c5, best_n = h2, c2, n2

    # Final
    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
