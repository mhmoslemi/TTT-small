def softmax_obj_grad(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = alpha * (corr - m)
    sh = np.clip(sh, -500, 500)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        wk = w[k]
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n-lag])
            grad[:n-lag] -= wk * h[lag:]
        else:
            ll = -lag
            grad[:n-ll] += wk * (1.0 - h[ll:])
            grad[ll:] -= wk * h[:n-ll]
    return grad

    if c5_1 < best_c5 - 1e-14:
        best_h, best_c5 = h1.copy(), c5_1
    
    # Phase 2: Perturbation + re-optimize (20% of budget)
    t2 = t0 + budget_s * 0.70
    for trial in range(5):
        if time.time() > t2:
            break
        sigma = 0.02 + 0.02 * trial
        h_p = best_h + sigma * rng.randn(best_n)
        h_p = repair(h_p, best_n)
        h2, c5_2 = systematic_cd(h_p, best_n, min(t2, time.time() + budget_s * 0.04))
        if c5_2 < best_c5 - 1e-14:
            best_h, best_c5 = h2.copy(), c5_2
    
    # Phase 3: Try different n values (15% of budget)
    t3 = t0 + budget_s * 0.85
    for n_alt in [best_n - 4, best_n - 2, best_n + 2, best_n + 4, best_n + 6, best_n + 8]:
        if time.time() > t3 or n_alt < 40 or n_alt > 120:
            continue
        if n_alt == best_n:
            continue
        x_old = np.linspace(0, 1, best_n)
        x_new = np.linspace(0, 1, n_alt)
        h_seed = np.interp(x_new, x_old, best_h)
        h_seed = repair(h_seed, n_alt)
        t_sub = min(t3, time.time() + budget_s * 0.025)
        h3, c5_3 = systematic_cd(h_seed, n_alt, t_sub)
        if c5_3 < best_c5 - 1e-14:
            best_h, best_c5, best_n = h3.copy(), c5_3, n_alt
    
    # Phase 4: Final polish (15% of budget)
    t4 = t0 + budget_s * 0.995
    h4, c5_4 = systematic_cd(best_h, best_n, t4)
    if c5_4 < best_c5 - 1e-14:
        best_h, best_c5 = h4.copy(), c5_4
    
    # Final check
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
