        t_end = time.time() + min(35, tl() - 5)
        h2, c5_2 = _local(best_h.copy(), best_n, rng, t_end)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 5: final SLSQP + fine
    if tl() > 12:
        h4 = _slsqp(best_h.copy(), best_n, 5000000, maxiter=300)
        c5_4 = _eval(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    if tl() > 8:
        t_end = time.time() + min(15, tl() - 3)
        h5, c5_5 = _fine(best_h.copy(), best_n, rng, t_end)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: ultra-fine
    if tl() > 4:
        h = best_h.copy()
        c5c = _eval(h)
        while time.time() < t0 + budget_s - 2:
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * rng.randint(1, 500) * 1e-7
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
