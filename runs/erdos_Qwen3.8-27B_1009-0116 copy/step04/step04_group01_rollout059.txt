import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lp_solve(n):
    """Solve LP relaxation of the min-max problem."""
    # Variables
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    # Build y variables and constraints
    # For each shift sh in [-(n-1), n-1], valid i
    y_vars = {}
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    
    # For each shift, build the constraint sum(y) >= sum(h_valid) - t
    # y_sh[i] = h[i] * h[i+sh]

        valid_iables = []
        valid_h = []
        if sh >= 0:
            for i in range(n - sh):
                y_i = cp.Variable()
                y_vars[(sh, i)] = y_i
                valid_iables.append(y_i)
                valid_h.append(h_var[i])
                constraints.append(y_i >= 0)
                constraints.append(y_i <= h_var[i])
                constraints.append(y_i <= h_var[i + sh])
                constraints.append(y_i >= h_var[i] + h_var[i + sh] - 1)
        else:
            sp = -sh

                y_i = cp.Variable()
                y_vars[(sh, i)] = y_i
                valid_iables.append(y_i)
                valid_h.append(h_var[i])
                constraints.append(y_i >= 0)
                constraints.append(y_i <= h_var[i])
                constraints.append(y_i <= h_var[i - sp])
                constraints.append(y_i >= h_var[i] + h_var[i - sp] - 1)

np.correlate(a, v, mode='full') for a of length na, v of length nv:

For a=h (length n), v=1-h (length n):

For d < 0 (s=-d): n ranges from s to n-1. result[n-s+n-1] = sum_{n=s}^{n-1} h[n](1-h[n-s])

Now ∂/∂x[j] of result[k] where d = k-n+1:
If d ≥ 0:
- x[j] appears as h[n] (first factor) when n=j, valid if 0≤j≤n-1-d, i.e., j+d≤n-1. Contribution: (1-x[j+d])
- x[j] appears as h[n+d] (second factor) when n+d=j, i.e., n=j-d, valid if 0≤j-d≤n-1-d, i.e., d≤j≤n-1. Contribution: -x[j-d]

- x[j] appears as h[n] (first factor) when n=j, valid if s≤j≤n-1. Contribution: (1-x[j-s]) = (1-x[j+d])

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_shift(x, d, n):
    g = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            g[:n-d] += (1.0 - x[d:n])
        if d > 0:
            g[d:] -= x[:n-d]
    else:
        s = -d
        g[s:] += (1.0 - x[:n-s])
        g[:n-s] -= x[s:]
    return g

def _subgrad(h, n_steps=500, top_frac=0.08, step0=0.04):
    n = len(h)
    dx = 2.0 / n
    ts = n / 2.0
    step = step0
    for _ in range(n_steps):
        c = _corr(h)
        c_max = c.max()
        thr = c_max - top_frac * max(c_max, 1e-10)
        active = np.where(c >= thr)[0]
        if len(active) == 0:
            active = np.array([c.argmax()])
        g = np.zeros(n)
        for k in active:
            g += _grad_shift(h, (n - 1) - k, n)
        g *= dx / len(active)
        best_f = c_max
        for _ in range(12):
            h_new = _proj_to_sum(h - step * g, ts)
            f_new = _corr(h_new).max()
            if f_new < best_f - 1e-12:
                h, best_f = h_new, f_new
                break
            step *= 0.5
        step = min(step * 1.08, step0)
    return h

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14: continue
            d = (n - 1) - k
            g_k = _grad_shift(x, d, n)
            g += w[k] * dx * g_k
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_swap(h, n_iters=30000, rng=None, anneal=True):
    n = len(h)
    if rng is None: rng = np.random.RandomState(0)
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum()); tgt = n // 2
    if n1 > tgt:
        idx = np.where(hb == 1)[0]
        idx = idx[np.argsort(h[idx])[:n1 - tgt]]
        hb[idx] = 0
    elif n1 < tgt:
        idx = np.where(hb == 0)[0]
        idx = idx[np.argsort(-h[idx])[:tgt - n1]]
        hb[idx] = 1
    if abs(hb.sum() - tgt) > 0.5:
        return None
    c5 = _eval(hb)
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    for it in range(n_iters):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hb[i] = 0; hb[j] = 1
        new_c5 = _eval(hb)
        if new_c5 < c5 - 1e-13:
            c5 = new_c5
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]
        else:
            accept = anneal and (rng.rand() < np.exp(-(new_c5 - c5) * 3000 / (1 + it * 0.001)))
            if not accept:
                hb[i] = 1; hb[j] = 0
    return hb

def _golden_ratio_binary(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    idx = np.arange(n)
    vals = (idx * phi) % 1.0
    order = np.argsort(vals)
    h[order[:n // 2]] = 1.0
    return h

def _periodic_binary(n, period):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < period // 2:
            h[i] = 1.0
    h = _repair(h, n)
    return (h > 0.5).astype(float)
