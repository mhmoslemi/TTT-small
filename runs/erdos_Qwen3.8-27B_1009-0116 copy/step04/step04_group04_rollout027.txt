from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _c5_grad(h, dx):
    """Gradient of C5 w.r.t. h (subgradient at active lag)."""
    c = _corr_full(h)
    k_star = np.argmax(c)
    n = len(h)
    ell = k_star - (n - 1)  # actual shift
    grad = np.ones(n)
    # subtract h[m+ell] where valid
    if ell >= 0:
        grad[:n-ell] -= h[ell:]
    else:
        grad[-ell:] -= h[:-ell]
    # subtract h[m-ell] where valid
    if ell >= 0:
        grad[ell:] -= h[:n-ell]
    else:
        grad[:-ell] -= h[-ell:]
    return grad * dx

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:

        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:

            if d > 0:

            else:
                a = min(-d, h[i]); h[i] -= a; d += a

import time

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _to_binary(h):

    n = len(h)
    tgt = n / 2.0

    # Fix sum
    excess = h_bin.sum() - tgt

        order = np.argsort(h[ones])  # lowest h value first (closest to 0.5 from above)

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            a = min(abs(d), (1.0 - h[i] if d > 0 else h[i]))
            h[i] += a * (1 if d > 0 else -1)
            d -= a * (1 if d > 0 else -1)
            if abs(d) < 1e-13:
                break
    return h

def _to_binary(h):
    n = len(h)
    tgt = n / 2.0
    h_bin = (h > 0.5).astype(float)
    excess = int(round(h_bin.sum())) - int(tgt)
    if excess > 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        for i in order[:excess]:
            h_bin[i] = 0
    elif excess < 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        for i in order[:(-excess)]:
            h_bin[i] = 1

        oi = rng.randint(len(ones_pos))
        i = zeros_pos[zi]
        j = ones_pos[oi]

        h[j] = 0.0

            best_c = c
            zeros_pos[zi] = j
            ones_pos[oi] = i
        else:
            h[i] = 0.0
            h[j] = 1.0
    return h, best_c

    """Smooth approximation of max(c)."""
    m = c.max()
    return m + np.log(np.exp(alpha * (c - m)).sum()) / alpha

def _soft_obj(h, dx, alpha):
    c = _corr(h) * dx
    return _soft_max(c, alpha)

def _slsqp(h, dx, alpha, maxiter, target):
    n = len(h)
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    def obj(x):
        return _soft_obj(x, dx, alpha)
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, target)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)

def _subgrad_descent(h, dx, t_end, n_iter, target):

    best_h = h.copy()
    best_c = _c5(h, dx)
    lr = 0.005

            grad[ell:] -= h[:n-ell]
        elif ell < 0:
            grad[-ell:] -= h[:-ell]
            grad[:-ell] -= h[-ell:]
        h_new = np.clip(h - lr * grad * dx, 0.0, 1.0)
