        res = minimize(lse_obj_slsqp, h, args=(alpha, target), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(res.x, n)
    except Exception:
        return h

def targeted_local(h, n, dx, rng, deadline, n_iters, top_k=5):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    corr_len = 2 * n - 1
    mid = n - 1
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        corr = get_corr(h)
        worst_idx = np.argsort(corr)[-top_k:]
        found = False
        for ki in worst_idx:
            shift = ki - mid
            if shift == 0:
                for _ in range(4):
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    d = rng.choice([-1, 1]) * rng.uniform(0.01, 0.07)
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c5n = compute_c5(h, dx)
                        if c5n < best_c5 - 1e-15:
                            best_c5 = c5n
                            found = True
                        else:
                            h[i], h[j] = ni - d, nj + d
            else:
                for _ in range(4):
                    i = rng.randint(0, n)
                    j = i + shift
                    if j < 0 or j >= n:
                        j2 = i - shift
                        if 0 <= j2 < n:
                            shift = -shift
                            j = j2
                        else:
                            continue
                    d = rng.choice([-1, 1]) * rng.uniform(0.01, 0.07)
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c5n = compute_c5(h, dx)
                        if c5n < best_c5 - 1e-15:
                            best_c5 = c5n
                            found = True
                        else:
                            h[i], h[j] = ni - d, nj + d
        if not found:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                d = rng.choice([-1, 1]) * rng.uniform(0.004, 0.05)
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = compute_c5(h, dx)
                    if c5n < best_c5 - 1e-15:
                        best_c5 = c5n
                    else:
                        h[i], h[j] = ni - d, nj + d
    return h, best_c5

def anneal_local(h, n, dx, rng, deadline, T_max=0.04, T_min=0.0008):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    c5_cur = best_c5
    T = T_max
    it = 0
    while time.time() < deadline:
        it += 1
        T = T_max * (T_min / T_max) ** min(1.0, it / 30000.0)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([-1, 1]) * rng.uniform(0.004, 0.06)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = compute_c5(h, dx)
        delta = c5n - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c5_cur = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def binarize_nudge(h, n, strength):
    hh = h.copy()
    mask_lo = hh < 0.35
    mask_hi = hh > 0.65
    hh[mask_lo] *= (1.0 - strength)
    hh[mask_hi] += strength * (1.0 - hh[mask_hi])
    return repair(hh, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    h_parent = np.array(initial_h_values, dtype=float)
    n0 = len(h_parent)
    dx0 = 2.0 / n0
    h = repair(h_parent.copy(), n0)
    best_c5 = compute_c5(h, dx0)
    best_h = h.copy()
    best_n = n0
    target0 = n0 / 2.0

    # Phase 1: Quick L-BFGS on parent (30s)
    t_end = t_start + 30
    for alpha in [500, 3000, 15000]:
        if time.time() >= t_end:
            break
        hh = lbfgs_opt(h, n0, alpha, target0, 3000, 400)
        cc = compute_c5(hh, dx0)
        if cc < best_c5:
            best_c5, best_h = cc, hh.copy()
        h = hh

    # Phase 2: Try different n (60s)
    t_end = t_start + 90
    x_old = np.linspace(0, 2, n0, endpoint=False)
    for nTry in [100, 120, 200]:
        if time.time() >= t_end or remaining() < 350:
            break
        dxTry = 2.0 / nTry
        targetTry = nTry / 2.0
        x_new = np.linspace(0, 2, nTry, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_try = repair(h_interp, nTry)
        for alpha in [500, 3000]:
            if time.time() >= t_end:
                break
            h_try = lbfgs_opt(h_try, nTry, alpha, targetTry, 3000, 250)
            h_try = binarize_nudge(h_try, nTry, 0.15)
        cc = compute_c5(h_try, dxTry)
        if cc < best_c5:
            best_c5, best_h, best_n = cc, h_try.copy(), nTry
            h, n0, dx0, target0 = h_try, nTry, dxTry, targetTry
            x_old = np.linspace(0, 2, n0, endpoint=False)

    # Phase 3: Multi-restart (160s)
    n = best_n
    dx = 2.0 / n
    target = n / 2.0
    t_end = t_start + 250
    for r in range(18):
        if time.time() >= t_end or remaining() < 280:
            break
        hp = best_h.copy()
        kind = r % 5
        if kind == 0:
            hp += rng.normal(0, 0.04, n)
        elif kind == 1:
            f = rng.randint(1, 6)
            hp += 0.08 * np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.28))
        elif kind == 2:
            hp = np.roll(hp, rng.randint(n // 5, 4 * n // 5))
        elif kind == 3:
            hp = binarize_nudge(hp, n, 0.3)
        else:
            f1, f2 = rng.randint(1, 4), rng.randint(1, 5)
            t = np.arange(n) / n
            hp += 0.07 * np.sin(2 * np.pi * f1 * t + rng.uniform(0, 6.28)) * np.sin(2 * np.pi * f2 * t)
        hp = repair(hp, n)
        hh = lbfgs_opt(hp, n, 2000, target, 4000, 350)
        hh = slsqp_opt(hh, n, 10000, target, 400)
        cc = compute_c5(hh, dx)
        if cc < best_c5:
            best_c5, best_h = cc, hh.copy()

    # Phase 4: Targeted local search (250s)
    if remaining() > 320:
        t_end = time.time() + min(250, remaining() - 220)
        best_h, best_c5 = targeted_local(best_h, n, dx, rng, t_end, 20000, top_k=6)

    # Phase 5: SLSQP polish + binarize (80s)
    if remaining() > 250:
        t_end = time.time() + min(80, remaining() - 150)
        hh = slsqp_opt(best_h, n, 25000, target, 500)
        cc = compute_c5(hh, dx)
        if cc < best_c5:
            best_h, best_c5 = hh, cc
        hh = lbfgs_opt(hh, n, 50000, target, 5000, 300)
        cc = compute_c5(hh, dx)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 6: Simulated annealing (150s)
    if remaining() > 180:
        t_end = time.time() + min(150, remaining() - 80)
        h_ann, c5_ann = anneal_local(best_h, n, dx, rng, t_end, T_max=0.03, T_min=0.0005)
        if c5_ann < best_c5:
            best_h, best_c5 = h_ann, c5_ann

    # Phase 7: Targeted local search round 2 (120s)
    if remaining() > 80:
        t_end = time.time() + min(120, remaining() - 30)
        best_h, best_c5 = targeted_local(best_h, n, dx, rng, t_end, 12000, top_k=8)

    # Phase 8: Final SLSQP + fine tune (remaining)
    if remaining() > 30:
        t_end2 = time.time() + max(5, min(20, remaining() - 10))
        hh = slsqp_opt(best_h, n, 50000, target, 400)
        cc = compute_c5(hh, dx)
        if cc < best_c5:
            best_h, best_c5 = hh, cc
        best_h, best_c5 = targeted_local(best_h, n, dx, rng, t_end2, 5000, top_k=10)

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
