    return g

                h[i], h[j] = ni - d, nj + d
    return bh, bc

def opt_full(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = phase_subgrad(bh, n, dx, t0, lim - 280)
    if c < bc:
        bh, bc = r, c
    r, c = phase_lbfgs(bh, n, dx, t0, lim - 180)
    if c < bc:
        bh, bc = r, c
    r, c = phase_equalize(bh, n, dx, t0, lim - 130)
    if c < bc:
        bh, bc = r, c
    r, c = phase_slsqp(bh, n, dx, t0, lim - 90)
    if c < bc:
        bh, bc = r, c
    r, c = phase_spectral(bh, n, dx, t0, lim - 55)
    if c < bc:
        bh, bc = r, c
    r, c = phase_subgrad(bh, n, dx, t0, lim - 35)
    if c < bc:
        bh, bc = r, c
    r, c = phase_cd_sa(bh, n, dx, rng, t0, lim - 15)
    if c < bc:
        bh, bc = r, c
    r, c = phase_polish(bh, n, dx, rng, 40000, t0, lim - 3)
    if c < bc:
        bh, bc = r, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    r, c = opt_full(h0, nl, dx0, seed, t0, budget_s - 120)
    if c < bc:
        bh, bc, bn = r, c, nl

    if time.time() - t0 < budget_s - 90:
        rng2 = np.random.RandomState(seed + 77)
        hp = repair(bh + 0.04 * rng2.randn(bn), bn)
        r, c = opt_full(hp, bn, dx0, seed + 300, t0, budget_s - 50)
        if c < bc:
            bh, bc, bn = r, c, bn

    if time.time() - t0 < budget_s - 70:
        rng3 = np.random.RandomState(seed + 155)
        idx_p = rng3.permutation(bn)[:bn // 4]
        hp2 = bh.copy()
        hp2[idx_p] = 1.0 - hp2[idx_p]
        hp2 = repair(hp2, bn)
        r, c = opt_full(hp2, bn, dx0, seed + 600, t0, budget_s - 20)
        if c < bc:
            bh, bc, bn = r, c, bn

    n2 = nl * 2
    if n2 < 1000 and time.time() - t0 < budget_s - 50:
        xo = np.linspace(0, 2, nl, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, bh, period=2.0), n2)
        d2 = 2.0 / n2
        r, c = opt_full(h2, n2, d2, seed + 900, t0, budget_s - 8)
        if c < bc:
            bh, bc, bn = r, c, n2

    if time.time() - t0 < budget_s - 4:
        rng4 = np.random.RandomState(seed + 1234)
        r, c = phase_polish(bh, bn, 2.0 / bn, rng4, 30000, t0, budget_s - 1)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
