import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

_SM = {'a': 100.0, 'l': 50.0, 'n': 140}

def sm_obj(x):
    n = _SM['n']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(_SM['a'] * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _SM['a'] + _SM['l'] * (x.sum() - n * 0.5) ** 2

def sm_jac(x):
    n = _SM['n']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(_SM['a'] * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-14:
            g += pw[k] * grad1(x, k, n)
    g /= _SM['a']
    g += 2.0 * _SM['l'] * (x.sum() - n * 0.5)
    return g

def se_fun(x):
    return x.sum() - _SM['n'] * 0.5

def se_jac(x):
    return np.ones(_SM['n'])

def lbf_phase(h, n, dx, t0, lim):
    _SM['n'] = n
    bds = [(0.0, 1.0)] * n
    bh, bc = h.copy(), c5v(h, dx)
    schedule = [(100, 100, 500), (500, 80, 400), (2000, 60, 350),
                (10000, 40, 250), (50000, 30, 200), (200000, 20, 150),
                (800000, 10, 100), (2000000, 5, 60)]
    for a, lm, miter in schedule:
        if time.time() - t0 > lim:
            break
        _SM['a'], _SM['l'] = float(a), float(lm)
        try:
            r = minimize(sm_obj, bh.copy(), jac=sm_jac, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': miter, 'ftol': 1e-19, 'gtol': 1e-16})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def active_phase(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for mt in [2, 3, 5, 8, 15, 25]:
        h = bh.copy()
        step = 0.06
        stalled = 0
        for _ in range(500):
            if time.time() - t0 > lim:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            cm = c.max()
            thr = cm - 0.003 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            g = np.zeros(n)
            for ki in idx:
                g += grad1(h, ki, n)
            g /= len(idx)
            g -= g.mean()
            ok = False
            cur_c = c5v(h, dx)
            for _b in range(25):
                hn = repair(h - step * g, n)
                cn = c5v(hn, dx)
                if cn < cur_c - 1e-18:
                    h = hn
                    cur_c = cn
                    ok = True
                    if cn < bc:
                        bc, bh = cn, h.copy()
                    step = min(step * 1.1, 1.2)
                    stalled = 0
                    break
                step *= 0.72
            if not ok:
                stalled += 1
                if stalled > 10:
                    break
    return bh, bc

def cd_phase(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    steps_large = [0.015, 0.006, 0.003, 0.0015, -0.015, -0.006, -0.003, -0.0015]
    steps_small = [0.004, 0.002, 0.001, 0.0005, -0.004, -0.002, -0.001, -0.0005]
    steps_tiny = [0.001, 0.0005, 0.0002, 0.0001, -0.001, -0.0005, -0.0002, -0.0001]
    for passes, steps, max_iter in [(30, steps_large, 400), (40, steps_small, 600), (60, steps_tiny, 800)]:
        for _p in range(passes):
            if time.time() - t0 > lim:
                break
            imp = False
            for _ in range(max_iter):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                best_d, best_c = 0, cc
                for d in steps:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = c5v(h, dx)
                        if c < best_c:
                            best_c, best_d = c, d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = repair(h, n)
                    cc = c5v(h, dx)
                    imp = True
                    if cc < bc:
                        bc, bh = cc, h.copy()
            if not imp:
                break
    return bh, bc

def spec_phase(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.2, 0.35, 0.5, 0.35, 0.2, 0.1]:
        h = bh.copy()
        for _ in range(200):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a) * mag + a * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-18:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def opt_h(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = lbf_phase(bh, n, dx, t0, lim - (n * 0.4))
    if c < bc:
        bh, bc = r, c
    r, c = active_phase(bh, n, dx, t0, lim - n * 0.25)
    if c < bc:
        bh, bc = r, c
    r, c = spec_phase(bh, n, dx, t0, lim - n * 0.15)
    if c < bc:
        bh, bc = r, c
    r, c = lbf_phase(bh, n, dx, t0, lim - n * 0.08)
    if c < bc:
        bh, bc = r, c
    r, c = cd_phase(bh, n, dx, rng, t0, lim)
    if c < bc:
        bh, bc = r, c
    return bh, bc
