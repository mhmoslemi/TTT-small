def spec_phase(h, n, dx, t_lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.2, 0.3, 0.5, 0.4, 0.25, 0.15, 0.1]:
        if time.time() > t_lim: break
        h = bh.copy()
        for _ in range(150):
            if time.time() > t_lim: break
            H = np.fft.rfft(h)
            mag, ph = np.abs(H), np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n//2+1, np.sqrt(R0/n))
            tm[0] = abs(H[0])
            nm = (1-a)*mag + a*tm
            hn = np.fft.irfft(nm*np.exp(1j*ph), n)
            hr = repair(hn, n)
            if c5v(hr, dx) < c5v(h, dx) - 1e-18:
                h = hr
                if c5v(h, dx) < bc: bc, bh = c5v(h, dx), h.copy()
            else: break
    return bh, bc

                hn = repair(h2 - step * g, n)
                cn = c5v(hn, dx)
                if cn < c5v(h2, dx) - 1e-18:
                    h2 = hn
                    ok = True
                    if cn < bc:
                        bc, bh = cn, h2.copy()
                    step = min(step * 1.1, 1.5)
                    stalled = 0
                    break
                step *= 0.72
            if not ok:
                stalled += 1
                if stalled > 10:
                    break
    return bh, bc

def spec_phase(h, n, dx, t_lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.2, 0.3, 0.5, 0.4, 0.25, 0.15, 0.1]:
        if time.time() > t_lim:
            break
        h2 = bh.copy()
        for _ in range(150):
            if time.time() > t_lim:
                break
            H = np.fft.rfft(h2)
            mag, ph = np.abs(H), np.angle(H)
            R0 = max(np.dot(h2, h2), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1.0 - a) * mag + a * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            if c5v(hr, dx) < c5v(h2, dx) - 1e-18:
                h2 = hr
                if c5v(h2, dx) < bc:
                    bc, bh = c5v(h2, dx), h2.copy()
            else:
                break
    return bh, bc

def coord_polish(h, n, dx, rng, t_lim):
    bh, bc = h.copy(), c5v(h, dx)
    for _ in range(400):
        if time.time() > t_lim:
            break
        imp = False
        for _ in range(500):
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            for d in (0.004, 0.0015, -0.004, -0.0015, 0.0008, -0.0008):
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    if c5v(h, dx) < bc - 1e-15:
                        h = repair(h, n)
                        bc = c5v(h, dx)
                        imp = True
                        break
                    h[i], h[j] = ni - d, nj + d
        if not imp:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n0 = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    dx0 = 2.0 / n0
    bh, bc, bn = h0.copy(), c5v(h0, dx0), n0
    rng = np.random.RandomState(seed)
    T = float(budget_s)

    if time.time() - t0 < T * 0.28:
        r, c = de_search(h0, bn, dx0, rng, t0 + T * 0.28)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < T * 0.48:
        r, c = local_refine(bh, bn, 2.0 / bn, t0 + T * 0.48)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < T * 0.54:
        r, c = spec_phase(bh, bn, 2.0 / bn, t0 + T * 0.54)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < T * 0.60:
        r, c = local_refine(bh, bn, 2.0 / bn, t0 + T * 0.60)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < T * 0.65:
        r, c = coord_polish(bh, bn, 2.0 / bn, rng, t0 + T * 0.65)
        if c < bc:
            bh, bc = r, c

    for mult in [2, 3]:
        n2 = n0 * mult
        frac = 0.88 if mult == 2 else 0.97
        if n2 >= 1000 or time.time() - t0 > T * frac:
            break
        xo = np.linspace(0, 2, bn, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, bh, period=2.0), n2)
        d2 = 2.0 / n2
        if mult == 2 and time.time() - t0 < T * 0.73:
            r, c = de_search(h2, n2, d2, rng, t0 + T * 0.73)
            if c < bc:
                bh, bc, bn = r, c, n2
            else:
                bh, bn = h2, n2
        else:
            bh, bn = h2, n2
        if time.time() - t0 < T * (0.85 if mult == 2 else 0.95):
            r, c = local_refine(bh, bn, 2.0 / bn, t0 + T * (0.85 if mult == 2 else 0.95))
            if c < bc:
                bh, bc = r, c
        if time.time() - t0 < T * (0.88 if mult == 2 else 0.97):
            r, c = coord_polish(bh, bn, 2.0 / bn, rng, t0 + T * (0.88 if mult == 2 else 0.97))
            if c < bc:
                bh, bc = r, c

    if time.time() - t0 < T - 5:
        r, c = local_refine(bh, bn, 2.0 / bn, t0 + T - 5)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < T - 2:
        r, c = coord_polish(bh, bn, 2.0 / bn, rng, t0 + T - 1)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
