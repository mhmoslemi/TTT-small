    # Phase 9: Ultra-fine final polish
    if rem() > 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        t_ph9 = min(time.time() + min(20, rem() - 3), t0 + budget_s - 2)
        best_h, best_v = fine_ls(best_h, n_f, dx_f, rng, t_ph9)

    best_h = repair(best_h, len(best_h))
    best_v = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_v, len(best_h))
