        _, g = _smooth_obj_grad(h, n, alpha)
        h_new = _project(h - lr * g, n)
        c5_new = _c5(h_new, n)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            lr = min(lr * 1.08, 50.0 / alpha)
        else:
            lr *= 0.75
            if lr < 1e-9:
                break
    return h, c5_cur

def _sa_run(h, n, t_end, rng, T0=0.06, cool=0.992):
    h = h.copy()
    c5_cur = _c5(h, n)
    T = T0
    while time.time() < t_end and T > 0.0003:
        T *= cool
        r = rng.random()
        h_try = h.copy()
        applied = False
        if r < 0.30:
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                d = rng.normal(0, 0.025)
                if 0 <= h_try[i] + d <= 1 and 0 <= h_try[j] - d <= 1:
                    h_try[i] += d
                    h_try[j] -= d
                    applied = True
        elif r < 0.50:
            L = rng.randint(2, min(10, n // 4))
            start = rng.randint(0, n - L)
            if start + L < n:
                h_try[start:start+L] = h[start+1:start+L+1]
                h_try[start+L] = h[start]
                applied = True
            elif start > 0:
                h_try[start:start+L] = h[start-1:start+L-1]
                h_try[start-1] = h[start+L-1]
                applied = True
            if applied:
                h_try = _repair(h_try, n)
        elif r < 0.70:
            L = rng.randint(4, min(18, n // 5))
            start = rng.randint(0, n - L)
            ws = h_try[start:start+L].sum()
            h_try[start:start+L] += rng.normal(0, 0.07, L)
            h_try[start:start+L] = np.clip(h_try[start:start+L], 0, 1)
            sn = h_try[start:start+L].sum()
            if sn > 1e-9:
                h_try[start:start+L] *= ws / sn
            h_try[start:start+L] = np.clip(h_try[start:start+L], 0, 1)
            h_try = _repair(h_try, n)
            applied = True
        elif r < 0.85:
            idx = rng.choice(n, 3, replace=False)
            mv = rng.normal(0, 0.02, 3)
            mv -= mv.sum() / 3.0
            ok = all(0 <= h_try[ii] + dd <= 1 for ii, dd in zip(idx, mv))
            if ok:
                for ii, dd in zip(idx, mv):
                    h_try[ii] += dd
                h_try = _repair(h_try, n)
                applied = True
        else:
            L = rng.randint(3, min(6, n // 4))
            start = rng.randint(0, n - L)
            h_try[start:start+L] = 1.0 - h[start:start+L]
            h_try = _repair(h_try, n)
            applied = True
        if not applied:
            continue
        c5_new = _c5(h_try, n)
        delta = c5_new - c5_cur
        if delta < 0 or (T > 0.001 and rng.random() < np.exp(-min(delta / T, 40.0))):
            h = h_try
            c5_cur = c5_new
    return h, c5_cur

def _local_refine(h, n, t_end, rng):
    c5_cur = _c5(h, n)
    deltas = np.array([0.0005, 0.001, 0.003, 0.01, 0.03,
                       -0.0005, -0.001, -0.003, -0.01, -0.03])
    idx = np.arange(n)
    for _ in range(500):
        if time.time() > t_end:
            break
        improved = False
        rng.shuffle(idx)
        for i in idx:
            if time.time() > t_end:
                break
            hi = h[i]
            for j in idx:
                if j == i:
                    continue
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5_cur - 1e-15:
                        c5_cur = c
                        hi, hj = ni, nj
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
        if not improved:
            break
    return h, c5_cur

def _try_resolution(best_h, best_n, n_try, t_end, rng):
    x_old = np.linspace(0, 2.0, best_n, endpoint=False)
    x_new = np.linspace(0, 2.0, n_try, endpoint=False)
    h_new = np.interp(x_new, x_old, best_h, period=2.0)
    h_new = _repair(h_new, n_try)
    c5_cur = _c5(h_new, n_try)
    h_new, c5_cur = _pgd_run(h_new, n_try, 50000, min(t_end, time.time() + 60), 0.05)
    h_new, c5_cur = _sa_run(h_new, n_try, min(t_end, time.time() + 80), rng, T0=0.04, cool=0.99)
    h_new, c5_cur = _local_refine(h_new, n_try, t_end, rng)
    return h_new, c5_cur, n_try

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h0)
    best_h = h0.copy()
    best_n = n
    best_c5 = _c5(best_h, best_n)

    # Phase 1: Fast PGD (20% budget)
    t1 = t0 + budget_s * 0.20
    for alpha in [3000, 15000, 80000, 400000]:
        if time.time() > t1:
            break
        h_w, c5_w = _pgd_run(best_h, best_n, alpha, t1, 0.1)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_w.copy()

    # Phase 2: SA (25% budget)
    t2 = t0 + budget_s * 0.45
    h_sa, c5_sa = _sa_run(best_h, best_n, t2, rng, T0=0.08, cool=0.993)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 3: PGD polish + SA (20% budget)
    t3 = t0 + budget_s * 0.65
    for alpha in [200000, 1000000, 5000000]:
        if time.time() > t3:
            break
        h_w, c5_w = _pgd_run(best_h, best_n, alpha, t3, 0.04)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_w.copy()
    if time.time() < t3:
        h_sa, c5_sa = _sa_run(best_h, best_n, t3, rng, T0=0.03, cool=0.988)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Try alternative resolutions (15% budget)
    t4 = t0 + budget_s * 0.80
    for n_try in [200, 100, 160, 120]:
        if time.time() > t4:
            break
        if n_try == best_n:
            continue
        h_t, c5_t, nt = _try_resolution(best_h, best_n, n_try, t4, rng)
        if c5_t < best_c5 - 5e-5:
            best_c5 = c5_t
            best_h = h_t
            best_n = nt

    # Phase 5: Multi-restart SA from perturbed states (10% budget)
    t5 = t0 + budget_s * 0.90
    for _r in range(5):
        if time.time() > t5:
            break
        h_pert = _repair(best_h + 0.03 * rng.randn(best_n), best_n)
        h_pert, c5_p = _sa_run(h_pert, best_n, min(t5, time.time() + 40), rng, T0=0.05, cool=0.99)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pert.copy()

    # Phase 6: Final local refinement (remaining time)
    t_end = t0 + budget_s - 5
    best_h, best_c5 = _local_refine(best_h, best_n, t_end, rng)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
