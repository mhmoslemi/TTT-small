        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _soft_obj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(x) - len(x) / 2.0) ** 2

def _soft_grad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _active_grad(h, frac):
    n = len(h)
    c = _corr(h)
    m = np.max(c)
    spread = m - np.min(c) + 1e-15
    thresh = m - frac * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g

def _proj_grad(h, n, dx, step, t_end):
    c5c = _c5(h, dx)
    while time.time() < t_end:
        g = _active_grad(h, 0.05)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-13:
            break
        g /= ng
        found = False
        for s_try in [step, step * 0.5, step * 0.25, step * 0.1, step * 1.5]:
            h_new = _repair(h - s_try * g, n)
            c_new = _c5(h_new, dx)
            if c_new < c5c - 1e-16:
                h = h_new; c5c = c_new; found = True
                break
        if found:
            step = min(step * 1.08, 0.05)
        else:
            step *= 0.5
            if step < 1e-9:
                break
    return h, c5c

def _local_search(h, n, dx, t_end, rng, best_ref):
    c5c = _c5(h, dx)
    dlist = np.array([0.0003, 0.0008, 0.002, 0.005, 0.01, 0.018, 0.03,
                      -0.0003, -0.0008, -0.002, -0.005, -0.01, -0.018, -0.03])
    best_h = h.copy()
    best_c5 = c5c
    stall = 0
    while time.time() < t_end:
        g = _active_grad(h, 0.03)
        pos_idx = np.argsort(-g)[:12]
        neg_idx = np.argsort(g)[:12]
        improved = False
        for i in pos_idx:
            if time.time() > t_end:
                break
            for j in neg_idx:
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        c_new = _c5(h, dx)
                        if c_new < c5c:
                            c5c = c_new; improved = True
                            break
                        h[i], h[j] = ni - d, nj + d
                if improved:
                    break
            if improved:
                break
        if not improved:
            improved = False
            for _ in range(300):
                if time.time() > t_end:
                    break
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = dlist[rng.randint(len(dlist))]
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c_new = _c5(h, dx)
                    if c_new < c5c:
                        c5c = c_new; improved = True
                        break
                    h[i], h[j] = ni - d, nj + d
            if not improved:
                stall += 1
                if stall > 50 and rng.random() < 0.25:
                    i, j = rng.randint(n), rng.randint(n)
                    if i != j:
                        d = rng.uniform(-0.12, 0.12)
                        ni, nj = h[i] + d, h[j] - d
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            h[i], h[j] = ni, nj
                            h = _repair(h, n)
                            c5c = _c5(h, dx)
                stall = 0 if improved else stall + 1
        else:
            stall = 0
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h.copy()
        best_ref[0] = best_c5
        best_ref[1] = best_h
    return best_h, best_c5

def _grad_local(h, n, dx, t_end, rng, best_ref):
    c5c = _c5(h, dx)
    best_h = h.copy()
    best_c5 = c5c
    step = 0.005
    while time.time() < t_end:
        g = _active_grad(h, 0.04)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-13:
            break
        g /= ng
        for s_try in [step, step * 0.5, step * 0.25, step * 2.0]:
            h_new = h - s_try * g
            h_new = _repair(h_new, n)
            c_new = _c5(h_new, dx)
            if c_new < c5c:
                h = h_new; c5c = c_new
                if c5c < best_c5:
                    best_c5 = c5c; best_h = h.copy()
                step = min(step * 1.05, 0.03)
                break
        else:
            step *= 0.5
            if step < 1e-10:
                break
        best_ref[0] = best_c5
        best_ref[1] = best_h
    return best_h, best_c5

def _optimize_n(h0, n, dx, t_end, rng, best_ref):
    bounds = [(0.0, 1.0)] * n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > t_end - 60:
            break
        try:
            res = minimize(_soft_obj, h.copy(), jac=_soft_grad,
                           args=(alpha, 40.0), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 400, 'ftol': 1e-18})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < best_c5:
                best_c5 = c5v; best_h = hc.copy(); h = hc.copy()
        except Exception:
            pass
    t_pg = min(time.time() + 30, t_end - 40)
    h, c5 = _proj_grad(h, n, dx, 0.015, t_pg)
    if c5 < best_c5:
        best_c5 = c5; best_h = h.copy()
    best_h, best_c5 = _grad_local(best_h, n, dx, min(time.time() + 20, t_end - 30), rng, best_ref)
    best_h, best_c5 = _local_search(best_h, n, dx, t_end, rng, best_ref)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

        while time.time() < budget_s - 3:
            g = _active_grad(h, 0.03)

            improved = False
            for i in pos_idx:
                for j in neg_idx:
                    for d in dlist:
                        ni, nj = h[i] + d, h[j] - d
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            h[i], h[j] = ni, nj
                            c_new = _c5(h, dx)
                            if c_new < c5c:
                                c5c = c_new; improved = True
                                break
                            h[i], h[j] = ni - d, nj + d
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                for _ in range(50):
                    i, j = rng.randint(n), rng.randint(n)
                    if i == j:
                        continue
                    d = dlist[rng.randint(len(dlist))]
                    ni, nj = h[i] + d, h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        c_new = _c5(h, dx)
                        if c_new < c5c:
                            c5c = c_new; improved = True
                            break
                        h[i], h[j] = ni - d, nj + d
            if c5c < best_c5:
                best_c5 = c5c; best_h = h.copy()
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
