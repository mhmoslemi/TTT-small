    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        # Distribute to non-saturated elements
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    d = target - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-14:
                break
    return h

def _lsep_obj(h, alpha, pen):
