    if time.time() < t0 + budget_s - 10:
        best_h, best_c5 = _local(best_h, best_n, t0 + budget_s - 5, rng)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
