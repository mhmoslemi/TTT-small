```python
    return float(np.max(corr) * dx)

    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

    return np.sum(x) - target

def repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:

                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def grad_active_max(h, n, eps=0.001):
    """Compute gradient of max_k c_k approximated by active set."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    # Active set: lags where c_k is within eps*dx of max
    active_mask = corr >= cmax - eps  # in units of sum (not integral)
    # Weight by softmax over active set
    weights = np.exp(100 * (corr - cmax))
    weights[~active_mask] = 0
    wsum = weights.sum()
    if wsum < 1e-12:
        weights = active_mask.astype(float) / active_mask.sum()
        wsum = 1.0
    else:
        weights /= wsum
    
    g = np.zeros(n)
    for idx in np.where(active_mask)[0]:
        k = idx - (n - 1)  # actual lag
        w = weights[idx]
        for j in range(n):
            val = 0.0
            if 0 <= j - k < n:
                val += (1.0 - h[j - k])
            if 0 <= j + k < n:
                val -= h[j + k]
            g[j] += w * val
    return g, cmax, active_mask.sum()
```

Hmm, the gradient computation is O(n * |active|) which for n=140 and |active| maybe 20-50 is about 3000-7000 operations. That's fine.

But wait, I realize the gradient formula needs to be right. Let me re-derive:

c_k = sum_{n: valid} h[n+k] * (1-h[n])

where "valid" means 0 <= n < n and 0 <= n+k < n.

∂c_k/∂h[j] = 
  - If j = n+k for some valid n (i.e., n = j-k is valid, 0<=j-k<n): contribute (1-h[n]) = (1-h[j-k])
  - If n = j for some valid k (i.e., j+k is valid, 0<=j+k<n): contribute h[j+k] * (-1) = -h[j+k]

So: ∂c_k/∂h[j] = (1-h[j-k])*I(0<=j-k<n) - h[j+k]*I(0<=j+k<n)

This looks right. Let me verify with a simple case: n=2, h=[a,b], k=0:
c_0 = a*(1-a) + b*(1-b)
∂c_0/∂a = (1-a) - a = 1-2a ✓
∂c_0/∂b = (1-b) - b = 1-2b ✓

For k=1: c_1 = h[1]*(1-h[0]) = b*(1-a)
∂c_1/∂a = -b (j=0: j-k=-1 invalid, j+k=1 valid → -h[1] = -b) ✓
∂c_1/∂b = (1-a) (j=1: j-k=0 valid → (1-h[0])=1-a, j+k=2 invalid) ✓

Great, the formula is correct.

Now, for the gradient projection:
1. Compute g = weighted sum of gradients of active lags
2. Project onto sum-free subspace: g -= mean(g)
3. Zero out components where h is at bound
4. Take step: h -= lr * g
5. Clip and repair

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_con(x, target):
    return np.sum(x) - target

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def grad_active_max(h, n, width=0.002):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    active = corr >= cmax - width
    weights = np.exp(200.0 * (corr - cmax))
    weights[~active] = 0.0
    ws = weights.sum()
    if ws < 1e-15:
        weights = active.astype(float)
        if weights.sum() > 0:
            weights /= weights.sum()
        else:
            weights = np.zeros_like(corr)
            weights[n - 1] = 1.0
    else:
        weights /= ws
    g = np.zeros(n)
    idxs = np.where(weights > 1e-10)[0]
    for idx in idxs:
        k = int(idx) - (n - 1)
        w = weights[idx]
        for j in range(n):
            v = 0.0
            if 0 <= j - k < n:
                v += 1.0 - h[j - k]
            if 0 <= j + k < n:
                v -= h[j + k]
            g[j] += w * v
    return g, cmax

def grad_proj_step(h, n, lr, momentum, vel):
    g, cmax = grad_active_max(h, n)
    g -= g.mean()
    at_lo = h < 1e-8
    at_hi = h > 1.0 - 1e-8
    g[at_lo | at_hi] = 0.0
    norm_g = np.sqrt(np.sum(g * g))
    if norm_g < 1e-15:
        return h, cmax, vel
    vel = momentum * vel + g / norm_g
    h_new = h - lr * vel
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair(h_new, n)
    return h_new, cmax, vel

def fine_local(h, n, dx, rng, deadline, max_iter=3000000):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    scale = 0.004
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        if it % 100000 == 0 and it > 0:
            scale *= 0.7
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * (2.0 * rng.random() - 1.0)
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        h[i], h[j] = ni, nj
        c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, best_c5

def block_local(h, n, dx, rng, deadline, n_iters=500000):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        bs = rng.randint(2, 8)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.012) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if not (-1e-9 <= h[i + di] + eps <= 1.0 + 1e-9):
                ok = False
                break
        if ok:
            for di in range(bs):
                if not (-1e-9 <= h[j + di] - eps <= 1.0 + 1e-9):
                    ok = False
                    break
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if c5n < best_c5 - 1e-14:
            best_c5 = c5n
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best_c5

def slsqp_pass(h, n, alpha, maxiter=200):
    dx = 2.0 / n
    target = n / 2.0
    try:
        res = minimize(smooth_obj, h, args=(alpha, target, 0), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def lbfgs_pass(h, n, alpha, maxiter=200):
    dx = 2.0 / n
    target = n / 2.0
    pen = 5000
    try:
        res = minimize(smooth_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp += rng.normal(0, 0.06, n)
    elif kind == 1:
        f = rng.randint(1, 10)
        hp += 0.1 * np.sin(2.0 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.28))
    elif kind == 2:
        hp = np.roll(hp, rng.randint(7, n // 3))
    elif kind == 3:
        f1, f2 = rng.randint(1, 5), rng.randint(1, 8)
        t = np.arange(n) / n
        hp += 0.08 * np.sin(2*np.pi*f1*t + 0.5) * np.cos(2*np.pi*f2*t + 0.3)
    else:
        mask = rng.random(n) < 0.3
        hp[mask] += rng.normal(0, 0.1, mask.sum())
    return repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):

    h_init = np.array(initial_h_values, dtype=float)
```
