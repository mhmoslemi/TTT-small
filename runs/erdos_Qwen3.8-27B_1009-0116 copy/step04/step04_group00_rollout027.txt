    # d range: max(-h[i], -(1-h[j])) <= d <= min(1-h[i], h[j])
    d_lo = max(-h[i], -(1-h[j]))
    d_hi = min(1-h[i], h[j])
    if d_lo >= d_hi:
        return 0.0
    
    # c_k = correlate(h, 1-h, full)[k]
    # dc_k/dd = dc_k/dh_i - dc_k/dh_j
    c = np.correlate(h, 1-h, mode='full')
    
    # Gradient of c_k w.r.t. h_i:
    # c_k = sum_j h[j]*(1-h[j-L]) where L = k-(n-1)
    # dc_k/dh[i] = (1-h[i-L])*[i-L valid] - h[i+L]*[i+L valid]
    
    def grad_c(idx):
        g = np.zeros(2*n-1)
        for k in range(2*n-1):
            L = k - (n-1)
            gi = 0.0
            if 0 <= idx-L < n:
                gi += 1.0 - h[idx-L]
            if 0 <= idx+L < n:
                gi -= h[idx+L]
            g[k] = gi
        return g

    gj = grad_c(j)
    g = gi - gj  # dc_k/dd

    # This is 1D convex optimization
    
    # Bisection on subgradient
    for _ in range(50):

        if subgrad > 0:
            d_hi = d_mid
        elif subgrad < 0:
            d_lo = d_mid
        else:
            break
    return (d_lo + d_hi) / 2

    best_h, best_c5 = _pairwise_phase(best_h, n, dx, rng, t0, deadline * 0.92, 40)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    remaining = budget_s - 2

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float), n0)

    # Phase A: Optimize at native resolution
    best_h, best_c5 = _optimize_at_resolution(h0, n0, rng, t0, t0 + remaining * 0.40)

    # Phase B: Try n=100
    if time.time() - t0 < remaining * 0.55:
        nB = 100
        hB = np.interp(np.linspace(0, 2, nB, endpoint=False),
                       np.linspace(0, 2, n0, endpoint=False), best_h)
        hBr, c5B = _optimize_at_resolution(_repair(hB, nB), nB, rng, t0, t0 + remaining * 0.65)
        if c5B < best_c5:
            best_h, best_c5, n0 = hBr, c5B, nB

    # Phase C: Try n=180
    if time.time() - t0 < remaining * 0.72:
        nC = 180
        x_from = np.linspace(0, 2, n0, endpoint=False)
        x_to = np.linspace(0, 2, nC, endpoint=False)
        hC = np.interp(x_to, x_from, best_h)
        hCr, c5C = _optimize_at_resolution(_repair(hC, nC), nC, rng, t0, t0 + remaining * 0.85)
        if c5C < best_c5:
            best_h, best_c5, n0 = hCr, c5C, nC

    # Phase D: Final polish at best resolution
    n_final = n0
    dx_final = 2.0 / n_final
    best_h = _repair(best_h, n_final)
    best_c5 = _c5(best_h, dx_final)
    deadline_final = t0 + remaining * 0.96
    best_h, best_c5 = _pairwise_phase(best_h, n_final, dx_final, rng, time.time(), deadline_final, 30)
    best_h, best_c5 = _random_polish(best_h, n_final, dx_final, rng, time.time(), t0 + remaining * 0.99)

    best_h = _repair(best_h, n_final)
    best_c5 = _c5(best_h, dx_final)
    return (best_h, best_c5, n_final)
