def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

    G[0, :] = 1.0 - 2.0 * h

        # Subtract h[j-k] for j >= k in first region: j = k to m-1
        if k < m:

        # Second region: j = m to n-1
        if m < n:
            # j-k for j in [m, n-1] is in [m-k, n-1-k] = [k, n-1-k]
            # Wait: m = n-k, so m-k = n-2k
            # j-k for j in [n-k, n-1] is in [n-2k, n-1-k]
            # We only include where j >= k, i.e., j-k >= 0, i.e., j >= k
            # j >= n-k AND j >= k → j >= max(n-k, k)
            start = max(m, k)  # = max(n-k, k)
            if start < n:

24. **For the pairwise optimization**:
    Given the gradient at current h, the most promising swap is the pair (i,j) that maximizes the potential decrease. The decrease is approximately:
    Δf ≈ step * (∇f_i - ∇f_j)
    
    We want ∇f_i > ∇f_j (decrease h_i, increase h_j) or vice versa.
    Actually, if we do h_i += d, h_j -= d:
    Δf ≈ d * (∇f_i - ∇f_j)  ... wait no.
    f(h + d*e_i - d*e_j) ≈ f(h) + d*(∂f/∂h_i - ∂f/∂h_j)
    To decrease f, we want d*(∂f/∂h_i - ∂f/∂h_j) < 0.
    So if ∂f/∂h_i > ∂f/∂h_j, we want d < 0, i.e., decrease h_i, increase h_j.
    
    The most promising pair has the largest |∂f/∂h_i - ∂f/∂h_j|.

    But this is just first-order. For a better approach, we can compute the exact change for a given pair:
    c_k(h + d*e_i - d*e_j) = c_k(h) + d*(G[k,i] - G[k,j]) + d²*(cross[k,i,j])
    
    Where cross[k,i,j] is the second-order term. Since c_k is bilinear:
    The cross term is: -[i and j are paired in c_k at lag k]
    Specifically, in c_k = ∑ h_p(1-h_{p+k}):
    - h_i appears in term p=i as h_i(1-h_{i+k}), and in term p=i-k as h_{i-k}(1-h_i)
    - h_j similarly
    
    If j = i+k (i.e., they're paired in the bilinear form):
    The term h_i(1-h_{i+k}) = h_i(1-h_j) changes to (h_i+d)(1-(h_j-d)) = h_i(1-h_j) + d(1-h_j) + d*h_i - d²
    So the change is d(1-h_j) + d*h_i - d²
    But this is already captured in G[k,i] = 1-h_{i+k} - h_{i-k}[i≥k] = 1-h_j (for j=i+k, assuming i<k so no second term)
    
    Hmm, the point is that the bilinear term h_i*h_j contributes -d² to c_k when j=i+k (at lag k). So:
    
    Δc_k = d*(G[k,i] - G[k,j]) - d²*[j=i+k or i=j+k at lag k]
    
    The indicator [j=i+k at lag k] means: in the sum for c_k, is there a term h_i * h_j? This happens when j = i+k (term p=i) or i = j+k (term p=j).
    
    So: cross = -1 if (j-i == k or i-j == k) and both are valid indices in the sum.
    
    More precisely:
    - j = i+k: term h_i*h_j exists in c_k if i ≤ n-1-k (i.e., i+k ≤ n-1)
    - i = j+k: term h_j*h_i exists in c_k if j ≤ n-1-k (i.e., j+k ≤ n-1)
    
    So the quadratic coefficient for c_k is:
    b_k = -[i+k ≤ n-1 and j == i+k] - [j+k ≤ n-1 and i == j+k]
    (This is -1, 0, or -2, but actually can it be -2? Only if k=0, but we said k≥1. For k=0, b_k = -2.)
    
    Actually for k=0: c_0 = ∑ h_i(1-h_i) = ∑h_i - ∑h_i²
    Δc_0 for h_i→h_i+d, h_j→h_j-d = d - (h_i+d)² + h_i² + (h_j-d)² - h_j²
    = d - 2h_i*d - d² + 2h_j*d - d²
    = d(1-2h_i+2h_j) - 2d²
    
    So b_0 = -2 (for the pair i,j, but only the diagonal terms contribute).
    
    For k ≥ 1:
    b_k = -[j-i == k and i+k < n] - [i-j == k and j+k < n]
    (each term is 0 or 1, and they can't both be 1 for k ≥ 1)
    
    So the 1D problem for a pair (i,j) is:
    minimize max_k (c_k + d*a_k + d²*b_k)
    subject to 0 ≤ h_i+d ≤ 1, 0 ≤ h_j-d ≤ 1
    
    where a_k = G[k,i] - G[k,j] and b_k is as above.
    
    This is a convex 1D problem (max of convex functions). We can solve it with bisection on d.
    
    Actually, since b_k ≤ 0, the functions c_k + d*a_k + d²*b_k are concave (b_k < 0) or linear (b_k = 0) in d. The max of concave functions is... not necessarily convex or concave.
    
    Wait, b_k ≤ 0, so d²*b_k ≤ 0, meaning the function is concave in d (or linear). The max of concave functions is not in general convex. So bisection on the derivative might not work directly.
    
    However, in practice, the solution is usually at a point where two of the active functions intersect. We can use a simple approach:
    - Evaluate the objective at d = 0, d = ±small, d = ±large
    - Use golden section search or just try a grid of d values

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h):
    n = len(h)
    return float(np.max(_corr(h))) * 2.0 / n

def _repair(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_matrix(h, n):
    """G[k,j] = partial c_k / partial h_j for k=0..n-1, j=0..n-1"""
    G = np.zeros((n, n))
    G[0, :] = 1.0 - 2.0 * h
    for k in range(1, n):
        m = n - k
        G[k, :m] = 1.0 - h[k:k+m]
        if k < m:
            G[k, k:m] -= h[0:m-k]
        if m < n:
            start = max(m, k)
            if start < n:
                G[k, start:n] = -h[start-k:n-k]
    return G

def _subgrad_max(G, c, frac=0.05):
    """Average gradient over top `frac` fraction of lags"""
    n = len(c)
    cm = c.max()
    spread = cm - c.min()
    if spread < 1e-14:
        spread = 1.0
    thresh = cm - frac * spread
    active = np.where(c >= thresh)[0]
    return G[active].mean(axis=0)

def _pairwise_optimize(h, G, c, n, dx, time_fn, n_iter=5000):

    c5_current = c.max() * dx
    best_c5 = c5_current
    best_h = h.copy()

        if time_fn() > 0:
            break
        # Find most promising pair
        # For pair (i,j) with h_i + d, h_j - d:
        # change in c_k = d*(G[k,i]-G[k,j]) + d^2 * b_k
        # Approximate: spread of G values
        g_spread = G.max(axis=1) - G.min(axis=1)  # not quite right

        sg = _subgrad_max(G, c, 0.05)
        order_high = np.argsort(sg)[::-1][:20]
        order_low = np.argsort(sg)[:20]
        
        improved = False
        for ii in order_high:
            for jj in order_low:
                if ii == jj:
                    continue
                # Try several d values
                d_min = max(-h[ii], h[jj] - 1.0)  # h[ii]+d >= 0, h[jj]-d <= 1
                d_max = min(1.0 - h[ii], h[jj])    # h[ii]+d <= 1, h[jj]-d >= 0
                if d_max < d_min:
                    continue
                best_d = 0.0
                best_val = c5_current
                # Try a grid of d values
                for d in np.linspace(d_min, d_max, 11)[1:-1]:
                    h[ii] += d
                    h[jj] -= d
                    c_new = _corr(h)
                    val = c_new.max() * dx
                    if val < best_val:
                        best_val = val
                        best_d = d
                    h[ii] -= d
                    h[jj] += d
                if best_d != 0.0:
                    h[ii] += best_d
                    h[jj] -= best_d
                    h = _repair(h, n)
                    c = _corr(h)
                    c5_current = c.max() * dx
                    # Update G (approximate: recompute)
                    G = _grad_matrix(h, n)
                    if c5_current < best_c5:
                        best_c5 = c5_current
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Try random pairs
            i1, i2 = np.random.randint(n, 2)
            if i1 == i2:
                continue
            d = np.random.uniform(-0.01, 0.01)
            if 0 <= h[i1]+d <= 1 and 0 <= h[i2]-d <= 1:
                h[i1] += d
                h[i2] -= d
                c_new = _corr(h)
                if c_new.max() * dx < c5_current:
                    c5_current = c_new.max() * dx
                    c = c_new
                    G = _grad_matrix(h, n)
                    if c5_current < best_c5:
                        best_c5 = c5_current
                        best_h = h.copy()
                else:
                    h[i1] -= d
                    h[i2] += d
    
    return best_h, best_c5

            best_h[ii] += d
            best_h[jj] -= d
            v = _c5_val(best_h)
            if v < best_v:
                best_v, best_d = v, d
            best_h[ii] -= d
            best_h[jj] += d
        if best_d != 0.0:
            best_h[ii] += best_d
            best_h[jj] -= best_d
            best_h = _repair(best_h, n)
            best_c = _c5_val(best_h)
            improved = True
    return best_h, best_c

def _phase_fine(h, n, dx, time_check, n_iter=30000):
    best_h, best_c = h.copy(), _c5_val(h)
    rng = np.random.RandomState(1)
    deltas = np.array([0.0005, 0.001, 0.002, 0.004, -0.0005, -0.001, -0.002, -0.004])
    for it in range(n_iter):
        if time_check():
            break
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = deltas[rng.randint(len(deltas))]
        ni, nj = best_h[i1] + d, best_h[i2] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            best_h[i1], best_h[i2] = ni, nj
            v = _c5_val(best_h)
            if v < best_c:
                best_c = v
            else:
                best_h[i1], best_h[i2] = ni - d, nj + d
    return best_h, best_c

def _interpolate(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    h_new = np.interp(x_tgt, x_src, h_src)
    return _repair(h_new, n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def time_check():
        return elapsed() > budget_s - 10

    n = len(initial_h_values)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h, best_c = h0.copy(), _c5_val(h0)

    # Phase 1: Soft-max optimization
    if not time_check():
        best_h, best_c = _phase_softmax(best_h, n, dx, bounds, time_check)

    # Phase 2: Subgradient descent
    if not time_check():
        best_h, best_c = _phase_subgrad(best_h, n, dx, time_check, n_steps=4000)

    # Phase 3: Pairwise optimization
    if not time_check():
        best_h, best_c = _phase_pairwise(best_h, n, dx, time_check, n_iter=25000)

    # Phase 4: Try other resolutions
    for n_try in [n - 20, n + 20, n - 10, n + 10, 100, 160]:
        if time_check() or n_try < 40:
            continue
        if abs(n_try - n) > 60:
            continue
        dx_try = 2.0 / n_try
        bounds_try = [(0.0, 1.0)] * n_try
        h_try = _interpolate(best_h, n, n_try)
        c_try = _c5_val(h_try)
        if c_try < best_c:
            best_h, best_c, n, dx = h_try, c_try, n_try, dx_try
            bounds = bounds_try
        if not time_check():
            h_try2, c_try2 = _phase_subgrad(h_try, n_try, dx_try, time_check, n_steps=1500)
            if c_try2 < best_c:
                best_h, best_c, n, dx = h_try2, c_try2, n_try, dx_try
                bounds = bounds_try
            if not time_check():
                h_try3, c_try3 = _phase_pairwise(h_try2, n_try, dx_try, time_check, n_iter=8000)
                if c_try3 < best_c:
                    best_h, best_c, n, dx = h_try3, c_try3, n_try, dx_try

    # Phase 5: Final fine search
    if not time_check():
        best_h, best_c = _phase_pairwise(best_h, n, dx, time_check, n_iter=15000)
    if not time_check():
        best_h, best_c = _phase_fine(best_h, n, dx, time_check, n_iter=40000)

    best_h = _repair(best_h, n)
    best_c = _c5_val(best_h)
    return (best_h, best_c, n)
