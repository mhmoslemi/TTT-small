    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        d = target - s
        if abs(d) < 1e-12:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-12:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

    m = np.max(corr)
    s = np.sum(np.exp(np.clip(alpha * (corr - m), -500, 500)))
    return m + np.log(s) / alpha

    V = np.zeros((n, 2*n-1))
    for k in range(n):
        ip = k + lags
        im = k - lags
        mp = (ip >= 0) & (ip < n)
        mm = (im >= 0) & (im < n)
        V[k, mp] += h[ip[mp]]
        V[k, mm] += h[im[mm]]
    return V

    best = compute_c5(h, n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        V = precompute_V(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        improved = 0
        # Iterate over pairs in random order
        pairs = []
        for m in range(n):
            for j in range(m+1, n):
                pairs.append((m, j))
        rng.shuffle(pairs)
        
        for (m, j) in pairs:
            if time.time() > t_end:
                break
            d = V[j] - V[m]
            # Check if d is all zeros
            if np.max(np.abs(d)) < 1e-14:
                continue
            delta_lo = max(-h[m], -(1.0-h[j]))
            delta_hi = min(1.0-h[m], h[j])
            if delta_hi - delta_lo < 1e-14:
                continue
            # Golden section search on soft_max(corr + delta*d)
            gr = (np.sqrt(5.0)-1.0)/2.0
            a, b = delta_lo, delta_hi
            c_pt = b - gr*(b-a)
            d_pt = a + gr*(b-a)
            fc = soft_max(corr + c_pt*d, alpha)
            fd = soft_max(corr + d_pt*d, alpha)
            for _ in range(40):

                    b = d_pt; d_pt = c_pt; fd = fc
                    c_pt = b - gr*(b-a)
                    fc = soft_max(corr + c_pt*d, alpha)
                else:
                    a = c_pt; c_pt = d_pt; fc = fd
                    d_pt = a + gr*(b-a)
                    fd = soft_max(corr + d_pt*d, alpha)
            opt_delta = (a + b) / 2.0
            if abs(opt_delta) < 1e-14:
                continue
            # Apply update
            h[m] += opt_delta
            h[j] -= opt_delta
            h[m] = np.clip(h[m], 0.0, 1.0)
            h[j] = np.clip(h[j], 0.0, 1.0)
            # Recompute corr
            corr = np.correlate(h, 1.0 - h, mode='full')
            V = precompute_V(h, n)
            c5 = compute_c5(h, n)
            if c5 < best - 1e-12:
                best = c5
                improved += 1
            elif c5 < best + 1e-10:
                improved += 1
        if improved == 0 and sweep > 5:
            break
    return h, best

                        fd = soft_max_val(corr + d_pt*d, alpha)

                h[j] -= opt

                h[j] = np.clip(h[j], 0.0, 1.0)

                V = precompute_V(h, n)
                c5_new = np.max(corr) * (2.0/n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        d = target - s
        if abs(d) < 1e-12:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-12:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def soft_max_val(corr, alpha):
    m = np.max(corr)
    s = np.sum(np.exp(np.clip(alpha * (corr - m), -500, 500)))
    return m + np.log(s) / alpha

def precompute_V(h, n):
    lags = np.arange(2 * n - 1) - (n - 1)
    ks = np.arange(n)[:, None]
    idx1 = ks + lags[None, :]
    idx2 = ks - lags[None, :]
    V = np.zeros((n, 2 * n - 1))
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    V[m1] += h[idx1[m1]]
    V[m2] += h[idx2[m2]]
    return V

def softmax_weights(corr, alpha):
    ac = np.clip(alpha * (corr - np.max(corr)), -500, 500)
    w = np.exp(ac)
    w /= w.sum()
    return w

def slsqp_grad(h, n, alpha, maxiter=1500):
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return soft_max_val(corr, alpha)
    def grad(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        V = precompute_V(v, n)
        w = softmax_weights(corr, alpha)
        return 1.0 - V @ w
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

        V = precompute_V(v, n)
        w = softmax_weights(corr, alpha)
        g = 1.0 - V @ w
        g += 2.0 * pen * (v.sum() - n / 2.0)

                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12, 'maxfun': 50000})
        return repair(res.x, n)

        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass
    if init_h is None:
        init_h = np.full(64, 0.5)
        init_n = 64

    init_h = repair(init_h, init_n)
    best_h = init_h.copy()
    best_c5 = compute_c5(init_h, init_n)
    best_n = init_n
    n = best_n
    h_work = best_h.copy()

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP with analytical gradient, escalating alpha
    for alpha in [2000, 10000, 50000, 200000]:
        if elapsed() > budget_s * 0.15:
            break
        h_new = slsqp_grad(h_work.copy(), n, alpha, 1000)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new
        else:
            h_work = h_new if compute_c5(h_new, n) < compute_c5(h_work, n) else h_work

    # Phase 2: Pair coordinate descent (KEY INNOVATION) - multiple alpha levels
    for alpha in [5000, 20000, 50000, 100000, 200000]:
        t_phase = t0 + budget_s * 0.50
        if elapsed() > t_phase:
            break
        h_cd, c5_cd = pair_cd(h_work, n, alpha, t_phase, rng, n_sweeps=60)
        if c5_cd < best_c5 - 1e-13:
            best_h = h_cd.copy()
            best_c5 = c5_cd
        h_work = h_cd if c5_cd <= compute_c5(h_work, n) else h_work

    # Phase 3: Targeted top-k coordinate descent
    if elapsed() < budget_s * 0.68:
        t_tgt = t0 + budget_s * 0.68
        h_tg, c5_tg = targeted_cd(h_work, n, t_tgt, rng, top_k=5)
        if c5_tg < best_c5 - 1e-13:
            best_h = h_tg.copy()
            best_c5 = c5_tg
        h_work = h_tg if c5_tg <= compute_c5(h_work, n) else h_work

    # Phase 4: L-BFGS-B with analytical gradient and large penalty
    if elapsed() < budget_s * 0.80:
        for alpha, pen in [(100000, 1e6), (300000, 1e7)]:
            if elapsed() > budget_s * 0.80:
                break
            h_lb = lbfgs_grad(h_work.copy(), n, alpha, pen, 6000)
            c5_lb = compute_c5(h_lb, n)
            if c5_lb < best_c5 - 1e-13:
                best_h = h_lb.copy()
                best_c5 = c5_lb
                h_work = h_lb

    # Phase 5: Pair CD on best solution with very high alpha
    if elapsed() < budget_s * 0.88:
        t_cd2 = t0 + budget_s * 0.88
        h_cd2, c5_cd2 = pair_cd(best_h, best_n, 500000, t_cd2, rng, n_sweeps=40)
        if c5_cd2 < best_c5 - 1e-13:
            best_h = h_cd2.copy()
            best_c5 = c5_cd2

    # Phase 6: Try nearby n values
    if elapsed() < budget_s * 0.95:
        for n_alt in [n - 4, n - 2, n + 2, n + 4, n + 8]:
            if elapsed() > budget_s * 0.95:
                break
            if n_alt < 32 or n_alt > 200 or n_alt == n:
                continue
            x_old = np.linspace(0.0, 1.0, n)
            x_new = np.linspace(0.0, 1.0, n_alt)
            h_seed = np.interp(x_new, x_old, best_h)
            h_seed = repair(h_seed, n_alt)
            h_alt = slsqp_grad(h_seed, n_alt, 50000, 500)
            c5_alt = compute_c5(h_alt, n_alt)
            h_alt, c5_alt = pair_cd(h_alt, n_alt, 50000, t0 + budget_s * 0.95, rng, n_sweeps=20)
            if c5_alt < best_c5 - 1e-13:
                best_h = h_alt.copy()
                best_c5 = c5_alt
                best_n = n_alt
                n = n_alt

    # Phase 7: Final ultra-fine polish
    if elapsed() < budget_s * 0.99:
        h_fin = slsqp_grad(best_h.copy(), best_n, 500000, 800)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-13:
            best_h = h_fin.copy()
            best_c5 = c5_fin
        h_fin, c5_fin = pair_cd(best_h, best_n, 500000, t0 + budget_s * 0.995, rng, n_sweeps=30)
        if c5_fin < best_c5 - 1e-13:
            best_h = h_fin.copy()
            best_c5 = c5_fin

    if best_h is None:
        best_n = 64
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
