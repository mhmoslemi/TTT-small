import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lse_raw(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def sum_con(x, target):
    return np.sum(x) - target

def lbfgs_run(h0, n, alpha, target, pen, maxiter, deadline):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h0, args=(alpha, target, pen),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8,
                                'maxfun': 5000})
        if time.time() > deadline:
            return h0, compute_c5(h0, dx)
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except:
        return h0, compute_c5(h0, dx)

def slsqp_run(h0, n, alpha, target, maxiter, deadline):
    dx = 2.0 / n
    try:
        res = minimize(lse_raw, h0, args=(alpha, target),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() > deadline:
            return h0, compute_c5(h0, dx)
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except:
        return h0, compute_c5(h0, dx)

def sqp_cvxpy(h0, n, dx, target, alpha, deadline, max_iter=8):
    """SQP with CVXPY: linearize bilinear form, solve convex subproblem."""
    h = h0.copy()
    c5 = compute_c5(h, dx)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        # Compute Jacobian numerically (n correlations x n variables)
        n_corr = 2 * n - 1
        J = np.zeros((n_corr, n))
        eps = 1e-6
        for j in range(n):
            hp = h.copy()
            hp[j] += eps
            cp_ = np.correlate(hp, 1.0 - hp, mode='full') * dx
            J[:, j] = (cp_ - corr) / eps
        # Linearized: lin_c = corr + J @ (h_new - h) = (corr - J@h) + J@h_new
        offset = corr - J @ h
        # Solve convex: min log_sum_exp(alpha * (offset + J @ h_new))
        hv = cp.Variable(n)
        lin_c = offset + J @ hv
        try:
            obj = cp.Minimize(cp.log_sum_exp(alpha * lin_c))
        except:
            break
        cons = [hv >= 0, hv <= 1, cp.sum(hv) == target]
        prob = cp.Problem(obj, cons)
        t_solve = min(30, max(5, deadline - time.time()))
        solved = False
        for solver in ['ECOS', 'SCS', 'PDLP']:
            try:
                prob.solve(solver=solver, max_time=t_solve)
                if hv.value is not None:
                    h_new = repair(np.clip(hv.value, 0, 1), n)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < c5:
                        h = h_new
                        c5 = c5_new
                    solved = True
                    break
            except:
                continue
        if not solved:
            break
        if time.time() >= deadline:
            break
    return h, c5

def local_search(h, n, dx, rng, deadline, lo, hi):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    scales = [0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001]
    per_scale = 40000
    for scale in scales:
        if scale > hi or scale < lo:
            continue
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i] = ni
            h[j] = nj
            c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
            if c5n < best_c5 - 1e-14:
                best_c5 = c5n
                improved = True
            else:
                h[i] -= d
                h[j] += d
        if not improved and scale < 0.002:
            break
    return h, best_c5

def block_search(h, n, dx, rng, deadline, iters):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for _ in range(iters):
        if time.time() >= deadline:
            break
        bs = rng.randint(2, 6)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.012) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if not (0 <= h[i+di] + eps <= 1) or not (0 <= h[j+di] - eps <= 1):
                ok = False
                break
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if c5n < best_c5 - 1e-14:
            best_c5 = c5n
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best_c5
