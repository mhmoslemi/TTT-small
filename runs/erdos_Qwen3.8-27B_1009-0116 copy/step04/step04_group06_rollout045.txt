def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    # Project onto {h >= 0, h <= 1, sum(h) = n/2}
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    # Bisection for projection
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-13:
        # Distribute to non-bound points
        mask = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
        else:
            # Force fix
            order = np.argsort(hp)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - hp[idx])
                    hp[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, hp[idx])
                    hp[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-13:
                    break
    return hp

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = _corr(x) * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]

    hi, hj = h[i], h[j]

    lo = max(-hi, hj - 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-9) & (hp < 1.0 - 1e-9)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
        else:
            order = np.argsort(hp) if diff < 0 else np.argsort(1.0 - hp)
            for idx in order:
                if abs(diff) < 1e-13:
                    break
                h[i] = idx  # placeholder, won't reach
    return hp

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):

            lo = mid
        else:
            hi = mid

    diff = target - hp.sum()
    if abs(diff) > 1e-13:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = _corr(x) * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _pair_tern(h, n, i, j, ntern):
    hio, hjo = h[i], h[j]
    lo_d = max(-hio, hjo - 1.0)
    hi_d = min(1.0 - hio, hjo)
    if hi_d - lo_d < 1e-14:
        return _c5(h, n)
    a, b = lo_d, hi_d
    best_v = _c5(h, n)
    for _ in range(ntern):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hio + m1
        h[j] = hjo - m1
        v1 = _c5(h, n)
        h[i] = hio + m2
        h[j] = hjo - m2
        v2 = _c5(h, n)
        if v1 < v2:
            b = m2
        else:
            a = m1
    d_opt = (a + b) / 2.0
    h[i] = hio + d_opt
    h[j] = hjo - d_opt
    v_opt = _c5(h, n)
    if v_opt >= best_v:
        h[i] = hio
        h[j] = hjo
        return best_v
    return v_opt

def _con_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)
