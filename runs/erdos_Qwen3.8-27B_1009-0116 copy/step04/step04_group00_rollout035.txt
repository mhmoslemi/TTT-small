import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _project(p, T):
    n = len(p)
    lo, hi = -np.max(p) - 1.0, np.max(p) + 1.0
    for _ in range(120):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(p - mid, 0.0, 1.0)) > T:
            lo = mid
        else:
            hi = mid
    return np.clip(p - 0.5 * (lo + hi), 0.0, 1.0)

def _softmax_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softmax_grad(h, alpha):
    c = _corr(h)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += pk * (1.0 - h[:n - L])
            g[:n - L] -= pk * h[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - h[M:])
            g[M:] -= pk * h[:n - M]
    return g / alpha

def _subgrad(h, frac=0.12):
    c = _corr(h)
    n = len(h)
    thr = np.percentile(c, 100.0 * (1.0 - frac))
    active = np.where(c >= thr)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g

def _optimize_n(h0, n, dx, T, bounds, t0, budget, reserve):
    h = h0.copy()
    bc5 = _c5(h, dx)
    def tl():
        return t0 + budget - reserve - time.time()
    for alpha in [300, 1500, 8000, 40000, 200000, 800000]:
        if tl() < 30:
            break
        try:
            r = minimize(lambda x: _softmax_obj(x, alpha), h,
                         jac=lambda x: _softmax_grad(x, alpha),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 300, 'ftol': 1e-19, 'gtol': 1e-15})
            hc = _project(r.x, T)
            cv = _c5(hc, dx)
            if cv < bc5:
                bc5, h = cv, hc
        except Exception:
            pass
    h = bc5_h = None
    h = _project(h, T)
    c5c = _c5(h, dx)
    step = 0.025
    for _ in range(4000):
        if tl() < 40:
            break
        g = _subgrad(h, 0.10)
        gc = g - g.mean()
        ng = np.linalg.norm(gc)
        if ng < 1e-14:
            break
        gc /= ng
        ok = False
        for s in [step, step * 0.5, step * 0.25, step * 0.125]:
            hn = _project(h - s * gc, T)
            cn = _c5(hn, dx)
            if cn < c5c - 1e-16:
                h, c5c = hn, cn
                ok = True
                if c5c < bc5:
                    bc5 = c5c
                break
        if ok:
            step = min(step * 1.04, 0.06)
        else:
            step *= 0.65
            if step < 1e-7:
                break
    best_h = _project(h, T)
    if c5c < bc5:
        bc5 = c5c
    h = best_h.copy()
    c5c = _c5(h, dx)
    dl = np.array([0.0008, 0.002, 0.005, 0.012, 0.025, -0.0008, -0.002, -0.005, -0.012, -0.025])
    for _sw in range(300):
        if tl() < 25:
            break
        g = _subgrad(h, 0.07)
        idx = np.argsort(np.abs(g))[::-1][:35]
        imp = False
        for ii in range(len(idx)):
            i = idx[ii]
            for jj in range(ii + 1, min(ii + 18, len(idx))):
                j = idx[jj]
                bd, bc = 0.0, c5c
                for d in dl:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        cv = _c5(h, dx)
                        if cv < bc:
                            bc, bd = cv, d
                        h[i], h[j] = ni - d, nj + d
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    h = _project(h, T)
                    c5c = _c5(h, dx)
                    imp = True
                    if c5c < bc5:
                        bc5 = c5c
                    break
        if not imp:
            break
    best_h = _project(h, T)
    if c5c < bc5:
        bc5 = c5c
    h = best_h.copy()
    c5c = _c5(h, dx)
    nr = int(min(40000, max(0, tl() - 10) * 400))
    for _ in range(nr):

        i, j = np.random.randint(n), np.random.randint(n)

        d = np.random.uniform(-0.008, 0.008)
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cv = _c5(h, dx)
            if cv < c5c:
                c5c = cv
                if cv < bc5:
                    bc5 = cv
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    best_h = _project(best_h, T)
    bc5 = _c5(best_h, dx)
    return best_h, bc5
