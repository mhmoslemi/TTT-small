from scipy.optimize import minimize
import time

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * (2.0/n)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n/2.0
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    h *= target/s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14: break
            a = min(abs(diff), h[idx]) if diff < 0 else min(diff, 1.0-h[idx])
            h[idx] += a if diff > 0 else -a
            diff += a if diff < 0 else -a
    return h

def _project(h, n):
    target = n/2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo+hi)*0.5
        if np.sum(np.clip(h-mid, 0, 1)) > target: lo = mid
        else: hi = mid
    hp = np.clip(h-(lo+hi)*0.5, 0, 1)
    d = target - hp.sum()
    if abs(d) > 1e-12:
        m = (hp > 1e-10) & (hp < 1-1e-10)
        if m.sum() > 0:
            hp[m] += d/m.sum()
            hp = np.clip(hp, 0, 1)
    return hp

def _obj(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')*(2.0/n)
    m = c.max()
    s = np.clip(alpha*(c-m), -400, 400)
    return m + np.log(np.sum(np.exp(s)))/alpha

def _grad(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')*(2.0/n)
    m = c.max()
    s = np.clip(alpha*(c-m), -400, 400)
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-13: continue
        lag = k-(n-1)
        if lag == 0: g += w[k]*(1.0-2.0*x)
        elif lag > 0:
            g[lag:] += w[k]*(1.0-x[:n-lag])
            g[:n-lag] -= w[k]*x[lag:]
        else:
            l2 = -lag
            g[:n-l2] += w[k]*(1.0-x[l2:])
            g[l2:] -= w[k]*x[:n-l2]
    return g*(2.0/n)

def _slsqp(h, n, alpha, maxiter):
    def cf(x): return x.sum() - n/2.0
    def cg(x): return np.ones(n)
    res = minimize(_obj, h, args=(n, alpha), jac=_grad, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':cf,'jac':cg}],
                   options={'maxiter':maxiter,'ftol':1e-16})
    return _repair(res.x, n)
