            pass
    return best, bc

def _run_slsqp(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    try:
        def obj(x):
            v, _ = _soft_val_and_grad(x, alpha, dx)
            return v
        def jobj(x):
            v, g = _soft_val_and_grad(x, alpha, dx)
            return v, g
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt, 'jac': lambda x: ones}]
        r = minimize(obj, h, jac=jobj, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _cd_search(h, dx, t_end, n_iter, rng):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    deltas = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_k = np.argsort(c)[-6:]
        found = False
        for kk in top_k:
            if time.time() > t_end:
                break
            shift = kk - (n - 1)
            if shift == 0:
                continue
            if shift > 0:
                ii = np.arange(n - shift)
                jj = ii + shift
            else:
                s = -shift
                jj = np.arange(s, n)
                ii = jj - s
            contribs = h[ii] * (1.0 - h[jj])
            n_try = min(4, len(ii))
            idx = np.argsort(contribs)[-n_try:]
            for idx_ in idx:
                i, j = int(ii[idx_]), int(jj[idx_])
                for d in deltas:
                    h[i] += d
                    h[j] -= d
                    if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                        cn = _c5(h, dx)
                        if cn < best:
                            best = cn
                            h = _repair(h, tgt)
                            found = True
                            break
                    h[i] -= d
                    h[j] += d
                if found:
                    break
            if found:
                break
        if not found:
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                d = rng.uniform(-0.006, 0.006)
                h[i] += d
                h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                        h = _repair(h, tgt)
                    else:
                        h[i] -= d; h[j] += d
                else:
                    h[i] -= d; h[j] += d
    return h, best

def _subgrad_momentum(h, dx, t_end, n_iter, alpha0, alpha1, g0):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    best_h = h.copy()
    h = h.copy()
    vel = np.zeros(n)
    mu = 0.9
    for it in range(n_iter):
        if time.time() > t_end:
            break
        alpha = alpha0 * (alpha1 / alpha0) ** (it / max(1, n_iter))
        _, grad = _soft_val_and_grad(h, alpha, dx)
        g = it / max(1, n_iter)
        step = g0 / (1.0 + g * 8.0)
        vel = mu * vel - step * grad
        h_new = h + vel
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _repair(h_new, tgt)
        cn = _c5(h_new, dx)
        if cn < best:
            best = cn
            best_h = h_new.copy()
        h = h_new
    return best_h, best

def _try_resolution(h_best, c_best, n_ref, dx_ref, t_end, rng):
    n1 = 2 * n_ref
    if n1 > 560:
        return h_best, c_best, n_ref
    dx1 = 2.0 / n1
    h1 = _repair(np.interp(np.linspace(0, 1, n1), np.linspace(0, 1, n_ref), h_best), n1 / 2.0)
    c1 = _c5(h1, dx1)
    if c1 > c_best * 1.002:
        return h_best, c_best, n_ref
    if time.time() < t_end:
        h1, c1 = _run_lbfgs(h1, dx1, 3000, 200, [500, 5000])
    if time.time() < t_end:
        h1, c1 = _run_slsqp(h1, dx1, 20000, 400)
    if time.time() < t_end:
        h1, c1 = _cd_search(h1, dx1, min(t_end, time.time() + 40), 15000, rng)
    if c1 < c_best:
        return h1, c1, n1
    return h_best, c_best, n_ref

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_h, best_c5, best_n = h0, _c5(h0, dx0), n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B with analytic gradient
    t_end = t0 + 120
    best_h, best_c5 = _run_lbfgs(best_h, dx0, 5000, 600, [50, 500, 5000])

    # Phase 2: SLSQP polish with increasing smoothing
    t_end = t0 + 250
    for a in [10000, 50000, 150000]:
        if time.time() > t_end:
            break
        _, c = _run_slsqp(best_h, dx0, a, 700)
        best_h, best_c5 = _run_slsqp(best_h, dx0, a, 700)

    # Phase 3: Subgradient with momentum
    t_end = t0 + 420
    if rem() > 200:
        hs, cs = _subgrad_momentum(best_h, dx0, t_end, 40000, 100, 8000, 0.05)
        if cs < best_c5:
            best_h, best_c5 = hs, cs

    # Phase 4: Directed coordinate descent
    t_end = t0 + 620
    if rem() > 150:
        best_h, best_c5 = _cd_search(best_h, dx0, t_end, 120000, rng)

    # Phase 5: Multi-restart for global exploration
    t_end = min(t0 + 780, t0 + budget_s - 250)
    if rem() > 250:
        n = n0
        tgt = n / 2.0
        for r in range(6):
            if time.time() > t_end:
                break
            amp = 0.07 * (1.0 - r / 8.0)
            hp = best_h + amp * (2.0 * rng.random(n) - 1.0)
            hp = _repair(hp, tgt)
            hp, c = _run_lbfgs(hp, dx0, 3000, 250, [500, 5000])
            if c < best_c5:
                best_h, best_c5 = hp, c

    # Phase 6: Try 2x resolution
    t_end = min(t0 + budget_s - 120, t0 + 950)
    if rem() > 120:
        best_h, best_c5, best_n = _try_resolution(best_h, best_c5, n0, dx0, t_end, rng)

    # Phase 7: Final SLSQP + CD on best
    nb = best_n
    dxb = 2.0 / nb
    if rem() > 60:
        best_h, c = _run_slsqp(best_h, dxb, 300000, 500)
        if c < best_c5:
            best_c5 = c
    if rem() > 25:
        best_h, best_c5 = _cd_search(best_h, dxb, t0 + budget_s - 2, 60000, rng)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
