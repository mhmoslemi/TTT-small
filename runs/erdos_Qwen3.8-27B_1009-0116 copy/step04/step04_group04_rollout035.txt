import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp1d(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _slopes_for(h, i):
    n = len(h)
    nhalf = n - 1
    slopes = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - nhalf
        s = 0.0
        ik = i + k
        if 0 <= ik < n:
            s += 1.0 - h[ik]
        ik2 = i - k
        if 0 <= ik2 < n:
            s -= h[ik2]
        slopes[m] = s
    return slopes

def _minimize_1d_max(base, slopes):
    """Minimize max_m(base[m] + slopes[m]*t) over t in [0,1]."""
    def f(t):
        return float(np.max(base + slopes * t))
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    return res.x

def _coord_descent(h, dx, t_end, max_sweeps=2000, rng=None):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    c = _corr_full(h)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = list(range(n))
        if rng is not None:
            rng.shuffle(order)
        
        for i in order:
            slopes = _slopes_for(h, i)
            base = c - h[i] * slopes
            t_new = _minimize_1d_max(base, slopes)
            if abs(t_new - h[i]) > 1e-10:
                delta = t_new - h[i]
                h[i] = t_new
                c = c + delta * slopes
                c5 = float(np.max(c) * dx)
                if c5 < best - 1e-13:
                    best = c5
                    improved = True
            else:
                c5 = float(np.max(c) * dx)
                if c5 < best:
                    best = c5
        # Repair sum after sweep
        h = _repair(h, tgt)
        c = _corr_full(h)
        c5 = float(np.max(c) * dx)
        if c5 < best:
            best = c5
        if not improved and sweep > 5:
            break
    return h, best

def _pair_cd(h, dx, t_end, max_steps=50000, rng=None):
    """Pair coordinate descent preserving sum."""
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    c = _corr_full(h)
    
    for step in range(max_steps):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # We want h[i] + h[j] = s (preserved)
        # Let t = h[i] - h[i]_old, then h[j] = h[j]_old - t
        # c_new[m] = c[m] + t*(slopes_i[m] - slopes_j[m])
        slopes_i = _slopes_for(h, i)
        slopes_j = _slopes_for(h, j)
        combo = slopes_i - slopes_j
        base = c  # base at t=0
        
        # Minimize max_m(base[m] + t*combo[m]) over t in [h[i]-1, h[j]] (to keep in [0,1])
        t_lo = h[i] - 1.0
        t_hi = h[j]
        if t_lo >= t_hi:
            continue
        
        def f(t):
            return float(np.max(base + t * combo))
        res = minimize_scalar(f, bounds=(t_lo, t_hi), method='bounded',
                              options={'xatol': 1e-10})
        t_new = res.x
        if abs(t_new) > 1e-10:
            h[i] += t_new
            h[j] -= t_new
            c = c + t_new * combo
            c5 = float(np.max(c) * dx)
            if c5 < best:
                best = c5
    h = _repair(h, tgt)
    return h, best

def _lbfgs_run(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = _corr_full(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 0)
        return m + np.log(np.exp(s).sum()) / alpha
    
    try:
        r = minimize(obj, h, method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _slsqp_run(h, dx, alpha, maxiter=400):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    
    def obj(x):
        c = _corr_full(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 0)
        return m + np.log(np.exp(s).sum()) / alpha
    
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)
