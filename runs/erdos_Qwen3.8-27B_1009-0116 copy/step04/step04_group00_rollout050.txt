        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _obj_sm(h, dx, alpha, lam):

    s = alpha * (c - m)
    s -= np.max(s)
    return (m + np.log(np.sum(np.exp(s)))/alpha)*dx + lam*(np.sum(h)-len(h)/2.0)**2

def _grad_sm(h, dx, alpha, lam):
