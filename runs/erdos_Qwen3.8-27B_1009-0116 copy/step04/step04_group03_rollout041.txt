    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_soft_max_obj, h, jac=_soft_max_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _proj_box_sum(res.x, n / 2.0)
    except Exception:
        return _proj_box_sum(h, n / 2.0)

def _local_search(h, n, rng, t_end, scale0=0.012):
    c5c = _eval(h)
    best_h, best_c5 = h.copy(), c5c
    scale, no_imp = scale0, 0
    for _ in range(300000):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.04, 0.035)
            if c5n < best_c5 - 1e-14:
                best_c5, best_h = c5n, h.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            no_imp += 1
            if no_imp > 300:
                scale *= 0.82
                no_imp = 0
                if scale < 5e-7:
                    break
    return best_h, best_c5

def _interp(hs, ns, nd):
    xs = np.linspace(0, 2, ns, endpoint=False)
    xd = np.linspace(0, 2, nd, endpoint=False)
    return np.interp(xd, xs, hs)

def _triple_search(h, n, rng, t_end, scale0=0.005):
    c5c = _eval(h)
    best_h, best_c5 = h.copy(), c5c
    scale, no_imp = scale0, 0
    for _ in range(200000):
        if time.time() > t_end:
            break
        i, j, k = rng.randint(n), rng.randint(n), rng.randint(n)
        if i == j or i == k or j == k:
            continue
        di = scale * rng.randn()
        dj = scale * rng.randn()
        dk = -(di + dj)
        ni, nj, nk = h[i]+di, h[j]+dj, h[k]+dk
        if ni<0 or ni>1 or nj<0 or nj>1 or nk<0 or nk>1:
            continue
        h[i],h[j],h[k] = ni,nj,nk
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
            no_imp = 0
            scale = min(scale*1.03, 0.02)
        else:
            h[i],h[j],h[k] = ni-di, nj-dj, nk-dk
            no_imp += 1
            if no_imp > 400:
                scale *= 0.8
                no_imp = 0
                if scale < 2e-7:
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _tl():
        return budget_s - (time.time() - t0)

    n_par = len(initial_h_values)
    h_par = _proj_box_sum(np.array(initial_h_values, dtype=float).copy(), n_par / 2.0)
    best_h, best_c5, best_n = h_par.copy(), _eval(h_par), n_par

    n_list = sorted(set([40, 60, 80, 100, 120, n_par, 160, 200]))
    n_list = [n for n in n_list if n <= 300]

    inits = {}
    for n in n_list:
        if _tl() < 20:
            break
        cands = [_proj_box_sum(_interp(h_par, n_par, n), n / 2.0)]
        for _ in range(5):
            hp = _interp(h_par, n_par, n) + 0.15 * rng.randn(n)
            cands.append(_proj_box_sum(hp, n / 2.0))
        for fr in [1, 2, 3, 4]:
            for am in [0.25, 0.4, 0.55]:
                hs = 0.5 + am * np.sin(2 * np.pi * fr * np.arange(n) / n)
                cands.append(_proj_box_sum(hs, n / 2.0))
        for bl in [2, 3, 4, 5]:
            if n % bl == 0:
                nrep = n // bl
                hb = np.empty(n)
                for b in range(nrep):
                    hb[b*bl:(b+1)*bl] = 0.5 + 0.35 * (1 if b % 2 == 0 else -1)
                cands.append(_proj_box_sum(hb, n / 2.0))
        bc, bh = 9.0, None
        for hc in cands:
            cv = _eval(hc)
            if cv < bc:
                bc, bh = cv, hc.copy()
        inits[n] = bh

    for n in n_list:
        if _tl() < 40:
            break
        h0 = inits[n]
        c5_0 = _eval(h0)
        if c5_0 > best_c5 + 0.04:
            continue
        h_cur, c5_cur = h0.copy(), c5_0
        alphas = [500, 5000, 50000, 300000]
        for alpha in alphas:
            if _tl() < 25:
                break
            mi = 800 if n <= 80 else 500
            hc = _slsqp_opt(h_cur, n, alpha, maxiter=mi)
            c5n = _eval(hc)
            if c5n < c5_cur - 1e-14:
                h_cur, c5_cur = hc, c5n
        if c5_cur < best_c5:
            best_c5, best_h, best_n = c5_cur, h_cur.copy(), n

    if _tl() > 50:
        t_end = time.time() + min(_tl() - 15, 120)
        h1, c5_1 = _mm_descent(best_h.copy(), best_n, t_end, lr0=0.06)
        if c5_1 < best_c5:
            best_c5, best_h = c5_1, h1.copy()

    if _tl() > 30:
        t_end = time.time() + min(_tl() - 10, 60)
        h2, c5_2 = _local_search(best_h.copy(), best_n, rng, t_end)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    if _tl() > 20:
        t_end = time.time() + min(_tl() - 8, 40)
        h3, c5_3 = _triple_search(best_h.copy(), best_n, rng, t_end)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    if _tl() > 15:
        alpha_f = 500000
        h4 = _slsqp_opt(best_h.copy(), best_n, alpha_f, maxiter=600)
        c5_4 = _eval(h4)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
        t_end2 = time.time() + min(_tl() - 5, 30)
        h5, c5_5 = _local_search(best_h.copy(), best_n, rng, t_end2, scale0=0.004)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    if _tl() > 8:
        t_end3 = time.time() + min(_tl() - 2, 20)
        h6, c5_6 = _triple_search(best_h.copy(), best_n, rng, t_end3, scale0=0.003)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    best_h = _proj_box_sum(best_h, best_n / 2.0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
