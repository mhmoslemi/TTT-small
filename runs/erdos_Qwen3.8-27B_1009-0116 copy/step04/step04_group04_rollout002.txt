import cvxpy as cp

def _lp_iteration(h, n, dx, active_idx, trust_radius):
    target = n / 2.0
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    c_curr = np.correlate(h, 1.0 - h, mode='full')
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        cp.norm(h_var - h, np.inf) <= trust_radius  # trust region
    ]
    
    for m in active_idx:
        c_val = c_curr[m]
        g = _grad_c_fast(m, h, n)
        # Linearized: c[m] ≈ c_val + g · (h_var - h)
        constraints.append(t >= c_val + g @ (h_var - h))
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if h_var.value is not None:
            return h_var.value, t.value
    except:
        pass
    return h, c_curr.max()

        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_c(idx, h, n):
    shift = idx - (n - 1)
    grad = np.zeros(n)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        s = shift
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    else:
        s = -shift
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    return grad

def _lp_step(h, n, delta):
    c_curr = np.correlate(h, 1.0 - h, mode='full')
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        h_var >= 0, h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        h_var >= h - delta, h_var <= h + delta
    ]
    for m in range(2 * n - 1):
        g = _grad_c(m, h, n)
        constraints.append(t >= c_curr[m] + g @ (h_var - h))
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if h_var.value is not None and np.isfinite(h_var.value).all():
            return h_var.value, float(t.value)
    except Exception:
        pass
    return h.copy(), float(c_curr.max())

def _slsqp_polish(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.exp(s).sum()) / alpha + 1e4 * (x.sum() - tgt) ** 2
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt}],
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, tgt)
        return hc, _c5_of(hc, dx)
    except Exception:
        return h.copy(), _c5_of(h, dx)

def _local_search(h, dx, rng, t_end, n_iter):
    n = len(h)
    best = _c5_of(h, dx)
    h = h.copy()
    tgt = n / 2.0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        scale = 0.005 * max(0.3, 1.0 - it / n_iter)
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * (2.0 * rng.random() - 1.0)
        h[i] += d; h[j] -= d
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            c = _c5_of(h, dx)
            if c < best:
                best = c
            else:
                h[i] -= d; h[j] += d
        else:
            h[i] -= d; h[j] += d
    return _repair(h, tgt), best

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0 / 2.0)
    best_h, best_c5, best_n = h0, _c5_of(h0, dx0), n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: LP iterations at n0
    t_end1 = t0 + 450
    delta = 0.03
    stagnation = 0
    prev_c5 = best_c5
    while time.time() < t_end1 and rem() > 350:
        h_new, t_val = _lp_step(best_h, n0, delta)
        h_new = _repair(h_new, n0 / 2.0)
        c_new = _c5_of(h_new, dx0)
        if c_new < best_c5 - 1e-8:
            best_h, best_c5 = h_new, c_new
            stagnation = 0
            if prev_c5 - best_c5 > 0.001:
                delta = max(0.005, delta * 0.7)
        else:
            stagnation += 1
            if stagnation > 30:
                delta = max(0.002, delta * 0.5)
                stagnation = 0
                if delta < 0.003:
                    break
        prev_c5 = best_c5

    # Phase 2: SLSQP polish at n0
    t_end2 = t0 + 600
    for a in [2000, 10000, 50000]:
        if time.time() > t_end2 or rem() < 250:
            break
        best_h, best_c5 = _slsqp_polish(best_h, dx0, a, 500)

    # Phase 3: Different perturbation restart + LP
    t_end3 = t0 + 750
    for r in range(4):
        if time.time() > t_end3 or rem() < 200:
            break
        hp = best_h + 0.04 * (2.0 * rng.random(n0) - 1.0)
        hp = _repair(hp, n0 / 2.0)
        # Short LP on perturbed
        for _ in range(30):
            if time.time() > t_end3 or rem() < 200:
                break
            hp2, _ = _lp_step(hp, n0, 0.02)
            hp2 = _repair(hp2, n0 / 2.0)
            c2 = _c5_of(hp2, dx0)
            if c2 < _c5_of(hp, dx0):
                hp = hp2
            else:
                break
        hp, c_hp = _slsqp_polish(hp, dx0, 20000, 300)
        if c_hp < best_c5:
            best_h, best_c5 = hp, c_hp

    # Phase 4: Interpolate to higher n and optimize
    n1 = min(280, 2 * n0)
    dx1 = 2.0 / n1
    h1 = _repair(_interp(best_h, n1), n1 / 2.0)
    t_end4 = min(t0 + 950, t0 + budget_s - 120)
    for _ in range(40):
        if time.time() > t_end4 or rem() < 80:
            break
        h1n, _ = _lp_step(h1, n1, 0.015)
        h1n = _repair(h1n, n1 / 2.0)
        c1n = _c5_of(h1n, dx1)
        if c1n < _c5_of(h1, dx1):
            h1 = h1n
        else:
            break
    h1, c1 = _slsqp_polish(h1, dx1, 30000, 400)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, n1

    # Phase 5: Local search
    nb = best_n
    dxb = 2.0 / nb
    t_end5 = t0 + budget_s - 30
    best_h, best_c5 = _local_search(best_h, dxb, rng, t_end5, 800000)

    # Final SLSQP
    if rem() > 20:
        best_h, best_c5 = _slsqp_polish(best_h, dxb, 100000, 300)
    if rem() > 5:
        best_h, best_c5 = _local_search(best_h, dxb, rng, t0 + budget_s - 1, 30000)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
