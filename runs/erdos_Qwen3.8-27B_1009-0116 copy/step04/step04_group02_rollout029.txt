import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def compute_influence(h, n):
    """A[i, m] = d(corr[m])/d(h[i]) where corr = correlate(h, 1-h, 'full')"""
    m_total = 2 * n - 1
    A = np.zeros((n, m_total))
    for i in range(n):
        for k in range(-i, n - i):  # i+k in [0, n-1]
            A[i, n - 1 + k] += (1.0 - h[i + k])
        for k in range(-(n-1-i), i+1):  # i-k in [0, n-1], k != 0 handled
            if k != 0:
                A[i, n - 1 + k] += (-h[i - k])
    return A

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_con(x, target):
    return np.sum(x) - target

def local_search_ineff(h, n, dx, rng, deadline, n_trials):
    """Local search using influence matrix for O(1) C5 evaluation per trial."""
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    A = compute_influence(h, n)
    
    for t in range(n_trials):
        if time.time() >= deadline:
            break
        # Random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try both directions
        for sign in [1, -1]:
            # delta = sign * scale
            for scale in [0.01, 0.005, 0.002]:
                d = sign * scale
                ni = h[i] + d
                nj = h[j] - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:

                # Compute new C5 using influence

                new_c5 = np.max(new_corr) * dx
                if new_c5 < best_c5 - 1e-15:
                    # Accept

                    h[j] = nj
                    best_c5 = new_c5
                    corr = new_corr.copy()
                    A = compute_influence(h, n)
                    break  # try other pair
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_con(x, target):
    return np.sum(x) - target

def lbfgs_step(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_step(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def influence_row(h, i, n):
    """Compute A[i, :] = d(corr)/d(h[i]) for a single row."""
    m_total = 2 * n - 1
    row = np.zeros(m_total)
    for k in range(-i, n - i):
        row[n - 1 + k] += (1.0 - h[i + k])
    for k in range(i - n + 1, i + 1):
        if k != 0:
            row[n - 1 + k] += (-h[i - k])
    return row

def fast_local_search(h, n, dx, rng, deadline, n_rounds):
    """Local search: for each trial, pick pair (i,j), find best d by checking kink points."""
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    m_total = 2 * n - 1
    
    # Precompute all influence rows
    A = np.zeros((n, m_total))
    for i in range(n):
        A[i] = influence_row(h, i, n)
    
    improved_recent = 0
    scale = 0.02
    
    for t in range(n_rounds):
        if time.time() >= deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Feasible d range
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-10:
            continue
        # diff vector
        diff = A[i] - A[j]
        # Try to find d that minimizes max(corr + d*diff)
        # Simple approach: try a few candidate d values
        candidates = [(d_lo + d_hi) / 2.0, d_lo + 0.25 * (d_hi - d_lo), d_lo + 0.75 * (d_hi - d_lo)]
        # Also: d that equalizes top 2
        top = np.argsort(corr)[-2:]
        if len(top) == 2:
            m1, m2 = top[0], top[1]
            denom = diff[m1] - diff[m2]
            if abs(denom) > 1e-12:
                d_star = (corr[m2] - corr[m1]) / denom
                if d_lo <= d_star <= d_hi:
                    candidates.append(d_star)
                # Also try slightly offset
                if d_lo <= d_star + 0.001 <= d_hi:
                    candidates.append(d_star + 0.001)
                if d_lo <= d_star - 0.001 <= d_hi:
                    candidates.append(d_star - 0.001)
        
        best_d = None
        best_val = best_c5
        for d in candidates:
            if d < d_lo - 1e-12 or d > d_hi + 1e-12:
                continue
            new_corr = corr + d * diff
            val = np.max(new_corr) * dx
            if val < best_val:
                best_val = val
                best_d = d
        
        if best_d is not None and best_val < best_c5 - 1e-15:
            h[i] += best_d
            h[j] -= best_d
            best_c5 = best_val
            corr = corr + best_d * diff
            # Update A rows i and j
            A[i] = influence_row(h, i, n)
            A[j] = influence_row(h, j, n)
            # Also update rows that reference i or j
            for p in range(n):
                if p != i and p != j:
                    # A[p] depends on h[p+k] and h[p-k]
                    # If p+k == i or p-k == i or p+k == j or p-k == j, A[p] changed
                    for k_idx in range(1, n):
                        if p + k_idx == i or p + k_idx == j or p - k_idx == i or p - k_idx == j:
                            A[p] = influence_row(h, p, n)
                            break
            improved_recent += 1
        else:
            improved_recent += 0.5  # decay
    
        # Adaptive scale (not really used with candidates, but for future)
        if improved_recent > 50:
            scale *= 0.7
        elif improved_recent < 5:
            scale *= 1.3
        improved_recent *= 0.98
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_con(x, target):
    return np.sum(x) - target

def lbfgs_step(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_step(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha

                      bounds=[(0.0, 1.0)] * n,

                      options={'maxiter': maxiter, 'ftol': 1e-15})
