def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:

                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a

    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp1d(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _grad_i(h, i, lags):
    n = len(h)
    g = np.ones(2 * n - 1)
    id1 = lags + i
    m1 = (id1 >= 0) & (id1 < n)
    g[m1] -= h[id1[m1]]
    id2 = i - lags
    m2 = (id2 >= 0) & (id2 < n)
    g[m2] -= h[id2[m2]]
    return g

def _coord_pass(h, n, dx, t_end, n_passes, rng):
    tgt = n / 2.0
    best_c = _c5(h, dx)
    c = _corr_full(h)
    lags = np.arange(-(n - 1), n)
    h = h.copy()
    for p in range(n_passes):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        for i in order:
            g = _grad_i(h, i, lags)
            a = c - h[i] * g
            lo, hi_t = 0.0, 1.0
            for _ in range(25):
                m1 = lo + (hi_t - lo) / 3.0
                m2 = hi_t - (hi_t - lo) / 3.0
                if np.max(a + m1 * g) < np.max(a + m2 * g):
                    hi_t = m2
                else:
                    lo = m1
            t_new = (lo + hi_t) / 2.0
            delta = t_new - h[i]
            if abs(delta) > 1e-14:
                h[i] = t_new
                c += delta * g
        h = _repair(h, tgt)
        c = _corr_full(h)
        cv = _c5(h, dx)
        if cv < best_c:
            best_c = cv
    return h, best_c

def _pair_swap_sa(h, dx, n, rng, t_end, n_iter, T_start=0.004, T_end=0.0002):
    tgt = n / 2.0
    c = _corr_full(h)
    lags = np.arange(-(n - 1), n)
    best_c = float(c.max() * dx)
    best_h = h.copy()
    h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if it % 2000 == 1999:
            c = _corr_full(h)
        T = T_start * (T_end / T_start) ** (it / max(n_iter, 1))
        cmax = c.max()
        thresh = cmax * 0.97
        active = np.where(c > thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        k_idx = int(rng.choice(active))
        shift = k_idx - (n - 1)
        if shift == 0:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j:
                continue
        elif shift > 0:
            i = int(rng.randint(n - shift)); j = i + shift
        else:
            j = int(rng.randint(-shift, n)); i = j + shift
        d = rng.uniform(-0.025, 0.025)
        hi_n = h[i] + d; hj_n = h[j] - d
        if hi_n < -1e-9 or hi_n > 1 + 1e-9 or hj_n < -1e-9 or hj_n > 1 + 1e-9:
            continue
        hi_n = np.clip(hi_n, 0.0, 1.0); hj_n = np.clip(hj_n, 0.0, 1.0)
        d_eff = hi_n - h[i]
        if abs(d_eff) < 1e-12:
            continue
        g_i = _grad_i(h, i, lags)
        g_j = _grad_i(h, j, lags)
        new_c = c + d_eff * g_i - d_eff * g_j
        new_max = float(new_c.max() * dx)
        cur_max = float(cmax * dx)
        if new_max <= cur_max or rng.random() < np.exp((cur_max - new_max) / max(T, 1e-9)):
            h[i] = hi_n; h[j] = hj_n - d_eff
            c = new_c
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
    best_h = _repair(best_h, tgt)
    return best_h, best_c

def _slsqp_run(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        def obj(x):
            co = _corr_full(x)
            m = co.max()
            s = np.clip(alpha * (co - m), -500, 500)
            return m + np.log(np.exp(s).sum()) / alpha
        r = minimize(obj, h, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lbfgs_run(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    best, bc = h.copy(), _c5(h, dx)
    for pen in [500, 5000, 50000]:
        def obj(x):
            co = _corr_full(x)
            m = co.max()
            s = np.clip(alpha * (co - m), -500, 500)
            return m + np.log(np.exp(s).sum()) / alpha + pen * (x.sum() - tgt) ** 2
        try:
            r = minimize(obj, best, method='L-BFGS-B', bounds=bds,
                         options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
            hc = _repair(r.x, tgt)
            cv = _c5(hc, dx)
            if cv < bc:
                best, bc = hc, cv
        except Exception:
            pass
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0); dx0 = 2.0 / n0
    best_h, best_c5, best_n = h0, _c5(h0, dx0), n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent at n0
    t_end = t0 + 80
    h0, c = _coord_pass(h0, n0, dx0, t_end, 3000, rng)
    if c < best_c5:
        best_h, best_c5 = h0, c

    # Phase 2: SLSQP + coord at n0
    t_end = t0 + 220
    for a in [2000, 10000, 50000, 200000]:
        if time.time() > t_end:
            break
        h0, c = _slsqp_run(h0, dx0, a, 400)
        h0, c = _coord_pass(h0, n0, dx0, min(t_end, time.time()), 300, rng)
        if c < best_c5:
            best_h, best_c5 = h0, c

    # Phase 3: L-BFGS at n0
    t_end = t0 + 320
    h0, c = _lbfgs_run(h0, dx0, 2000, 400)
    h0, c = _coord_pass(h0, n0, dx0, t_end, 500, rng)
    if c < best_c5:
        best_h, best_c5 = h0, c

    # Phase 4: SA pair swaps at n0
    t_end = t0 + 450
    best_h, best_c5 = _pair_swap_sa(best_h, 2.0 / best_n, best_n, rng, t_end, 150000)
    best_h, best_c5 = _coord_pass(best_h, best_n, 2.0 / best_n, t_end, 500, rng)

    # Phase 5: Interpolate to 2*n0 and optimize
    n1 = 2 * n0
    dx1 = 2.0 / n1
    h1 = _repair(_interp1d(best_h, n1), n1 / 2.0)
    t_end = min(t0 + 650, t0 + budget_s - 300)
    h1, c1 = _coord_pass(h1, n1, dx1, t_end, 1500, rng)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, n1
    dxb = 2.0 / best_n
    t_end = min(t0 + 750, t0 + budget_s - 200)
    for a in [5000, 30000]:
        if time.time() > t_end:
            break
        best_h, best_c5 = _slsqp_run(best_h, dxb, a, 400)
        best_h, best_c5 = _coord_pass(best_h, best_n, dxb, min(t_end, time.time()), 300, rng)

    # Phase 6: SA pair swaps at best resolution
    t_end = t0 + budget_s - 120
    best_h, best_c5 = _pair_swap_sa(best_h, dxb, best_n, rng, t_end, 200000,
                                     T_start=0.003, T_end=0.0001)
    best_h, best_c5 = _coord_pass(best_h, best_n, dxb, t_end, 500, rng)

    # Phase 7: Final SLSQP + coord
    if rem() > 40:
        best_h, best_c5 = _slsqp_run(best_h, dxb, 300000, 300)
    if rem() > 10:
        best_h, best_c5 = _coord_pass(best_h, best_n, dxb, t0 + budget_s - 5, 300, rng)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
