def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = np.clip(h * target / s, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                h[idx] += min(diff, 1.0 - h[idx])
                diff -= min(diff, 1.0 - h[idx])
            else:
                h[idx] -= min(-diff, h[idx])
                diff += min(-diff, h[idx])
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):

            lo = mid
        else:
            hi = mid

        if mask.any():

            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _sm_grad(x, n, alpha):
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n-lag])
            g[:n-lag] -= wk * x[lag:]
        else:
            l = -lag
            g[:n-l] += wk * (1.0 - x[l:])
            g[l:] -= wk * x[:n-l]
    return g * dxv

def _sm_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _pgd(h, n, alpha, t_end, lr0, niter):
    h = h.copy()
    c5 = _c5(h, n)
    lr = lr0
    for it in range(niter):
        if time.time() > t_end:
            break
        g = _sm_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = lr / gn
        for _ in range(12):
            ht = _project(h - step * g, n)
            ct = _c5(ht, n)
            if ct < c5 - 1e-16:
                h, c5 = ht, ct
                break
            step *= 0.5
        else:
            lr *= 0.75
            if lr < 1e-12:
                break
        if it % 500 == 499:
            lr *= 0.8
    return h, c5

    # Focus on worst lags
    corr_vals = _corr(h, n)
