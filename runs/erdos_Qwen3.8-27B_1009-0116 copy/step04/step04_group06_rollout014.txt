    return h, c5_new

def _local_search(h, n, budget_end, rng, pairs_per_round):
    c5_cur = _c5_val(h, n)
    stagnant = 0
    offset = 0
    while time.time() < budget_end:
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for cnt in range(pairs_per_round):
            if time.time() > budget_end:
                break
            i_idx = idx[cnt % n]
            j_idx = idx[(cnt + 3 + offset) % n]
            if i_idx == j_idx:
                continue
            h_new, c5_new = _pair_ternary(h, n, i_idx, j_idx, c5_cur)
            if h_new is not None:
                h = h_new
                c5_cur = c5_new
                improved = True
        if improved:
            stagnant = 0
        else:
            stagnant += 1
            offset = (offset + 1) % n
            if stagnant > 20:
                break
    return h, c5_cur

def _try_res(h_cur, n_cur, n_new, time_limit):
    if n_new == n_cur or n_new <= 20:
        return None, None
    x_old = np.linspace(0.0, 2.0, n_cur, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_cur)
    h_new = _repair(h_new, n_new)
    c5_new = _c5_val(h_new, n_new)
    end = min(time_limit, time.time() + time_limit - time.time())
    end = time.time() + 20
    h_opt, c5_opt = _fista(h_new, n_new, end, 8000, 0.02, 4000)
    c5_opt = _c5_val(h_opt, n_new)
    return h_opt, c5_opt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_n = n0
    best_c5 = _c5_val(best_h, best_n)

    # Phase 1: FISTA with increasing alpha (25%)
    end1 = t0 + budget_s * 0.25
    h1, c5_1 = _fista(h0, n0, end1, 5000, 0.03, 20000)
    c5_1 = _c5_val(h1, n0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP at multiple alphas (20%)
    end2 = t0 + budget_s * 0.47
    for alpha in [8000, 30000, 100000, 300000]:
        if time.time() > end2:
            break
        _G['n'] = best_n
        _G['alpha'] = alpha
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
        for att in range(2):
            if time.time() > end2:
                break
            if att == 0:
                x0 = best_h.copy()
            else:
                x0 = _repair(best_h + 0.004 * rng.randn(best_n), best_n)
            try:
                res = minimize(_slsqp_obj, x0, jac=_slsqp_grad,
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 350, 'ftol': 1e-17})
                hc = _repair(res.x, best_n)
                c5 = _c5_val(hc, best_n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 3: Multi-resolution (18%)
    end3 = t0 + budget_s * 0.65
    for n_try in [70, 100, 120, 96, 112]:
        if time.time() > end3:
            break
        h_opt, c5_opt = _try_res(best_h, best_n, n_try, end3)
        if c5_opt is not None and c5_opt < best_c5 - 1e-7:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try

    # Phase 4: SLSQP again on best resolution (7%)
    end4 = t0 + budget_s * 0.72
    _G['n'] = best_n
    _G['alpha'] = 200000
    bounds = [(0.0, 1.0)] * best_n
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_obj, best_h, jac=_slsqp_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 500, 'ftol': 1e-17})
        hc = _repair(res.x, best_n)
        c5 = _c5_val(hc, best_n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
    except Exception:
        pass

    # Phase 5: Aggressive ternary pair search (28%)
    end5 = t0 + budget_s * 0.99
    best_h, best_c5 = _local_search(best_h, best_n, end5, rng, pairs_per_round=50)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5_val(best_h, best_n)
    return (best_h, best_c5, best_n)
