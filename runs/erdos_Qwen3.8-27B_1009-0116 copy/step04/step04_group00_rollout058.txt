                ii = i + di
                d = rng.uniform(-0.01, 0.01)
                if 0 <= h[ii] + d <= 1:
                    h[ii] += d
            h = _repair(h, n)
            c = _c5(h, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
            else:
                h = best_h.copy()
    return best_h, best_c5

def _make_initial(n, rng):
    h = np.full(n, 0.5)
    freqs = rng.choice([1, 2, 3, 4, 5], size=3)
    amps = rng.uniform(0.1, 0.3, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    x = np.linspace(0, 2, n, endpoint=False)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(np.pi * f * x + p)
    h = _repair(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    best_h = None
    best_c5 = 1.0
    best_n = 0

    # Main: use initial_h_values
    if 'initial_h_values' in dir() or True:
        try:
            n = len(initial_h_values)
        except NameError:
            n = 140
            initial_h_values = _make_initial(n, rng)
            n = len(initial_h_values)
        dx = 2.0 / n
        h0 = _repair(np.array(initial_h_values, dtype=float), n)
        bounds = [(0.0, 1.0)] * n

        t1_end = budget_s - 350
        h1, c1 = _phase1(h0, n, dx, bounds, min(t0 + t1_end - t0, t0 + t1_end))
        h1, c1 = _phase1(h1, n, dx, bounds, min(elapsed() + 200, budget_s - 250))

        h2, c2 = _phase2(h1, n, dx, min(elapsed() + 120, budget_s - 150))

        h3, c3 = _phase3(h2, n, dx, min(elapsed() + 150, budget_s - 40), rng)

        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
            best_n = n

    # Try n=100
    if elapsed() < budget_s - 120:
        n2 = 100
        dx2 = 2.0 / n2
        if best_n == n2:
            h2_start = best_h.copy()
        elif best_n > 0:
            h2_start = np.interp(np.linspace(0, 2, n2, endpoint=False),
                                 np.linspace(0, 2, best_n, endpoint=False), best_h)
        else:
            h2_start = _make_initial(n2, rng)
        h2_start = _repair(h2_start, n2)
        bounds2 = [(0.0, 1.0)] * n2
        h2p, c2p = _phase1(h2_start, n2, dx2, bounds2, min(elapsed() + 80, budget_s - 80))
        h2q, c2q = _phase2(h2p, n2, dx2, min(elapsed() + 40, budget_s - 40))
        h2r, c2r = _phase3(h2q, n2, dx2, min(elapsed() + 30, budget_s - 10), rng)
        if c2r < best_c5:
            best_c5 = c2r
            best_h = h2r.copy()
            best_n = n2

    # Try n=200 if time allows
    if elapsed() < budget_s - 100:
        n3 = 200
        dx3 = 2.0 / n3
        if best_n == n3:
            h3_start = best_h.copy()
        elif best_n > 0:
            h3_start = np.interp(np.linspace(0, 2, n3, endpoint=False),
                                 np.linspace(0, 2, best_n, endpoint=False), best_h)
        else:
            h3_start = _make_initial(n3, rng)
        h3_start = _repair(h3_start, n3)
        bounds3 = [(0.0, 1.0)] * n3
        h3p, c3p = _phase1(h3_start, n3, dx3, bounds3, min(elapsed() + 50, budget_s - 60))
        h3q, c3q = _phase2(h3p, n3, dx3, min(elapsed() + 30, budget_s - 30))
        h3r, c3r = _phase3(h3q, n3, dx3, min(elapsed() + 20, budget_s - 5), rng)
        if c3r < best_c5:
            best_c5 = c3r
            best_h = h3r.copy()
            best_n = n3

    # Final micro-polish
    if elapsed() < budget_s - 2:
        h = best_h.copy()
        c5c = _c5(h, dx if best_n == len(h) else 2.0 / len(h))
        dx_f = 2.0 / len(h)
        c5c = _c5(h, dx_f)
        tiny = np.array([0.0003, 0.0008, 0.002, -0.0003, -0.0008, -0.002])
        for _ in range(30000):
            if elapsed() > budget_s - 1:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                h[i] += d
                h[j] -= d
                c = _c5(h, dx_f)
                if c < c5c:
                    c5c = c
                    best_c5 = c
                    best_h = h.copy()
                    best_n = best_n
                else:
                    h[i] -= d
                    h[j] += d

    best_h = _repair(best_h, best_n)
    dx_final = 2.0 / best_n
    best_c5 = _c5(best_h, dx_final)
    return (best_h, best_c5, best_n)
