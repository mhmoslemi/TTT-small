import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        d = target - s
        if abs(d) < 1e-10:
            break
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(d) < 1e-10:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
            else:
                a = max(d, -h[idx])
            h[idx] += a
            d -= a
    return h

def binary_c5(b, n):
    """Compute C5 for binary array (0/1 values)."""
    dx = 2.0 / n
    corr = np.correlate(b.astype(float), 1.0 - b.astype(float), mode='full')
    return float(np.max(corr)) * dx, corr

def binary_search(h_bin, n, rng, max_rounds=500, t_end=None):
    """Local search on binary sequence by swapping."""
    h_bin = h_bin.astype(int).copy()
    _, corr = binary_c5(h_bin, n)
    best_c5 = float(np.max(corr)) * 2.0 / n
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    
    for r in range(max_rounds):
        if t_end and time.time() > t_end:
            break
        improved = False
        # Try multiple random swaps per round
        n_attempts = max(200, n)
        for _ in range(n_attempts):
            if t_end and time.time() > t_end:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i], h_bin[j] = 0, 1
            corr_new = np.correlate(h_bin.astype(float), 1.0 - h_bin.astype(float), mode='full')
            c5_new = float(np.max(corr_new)) * 2.0 / n
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                corr = corr_new
                improved = True
                # Update ones/zeros lists
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            else:
                h_bin[i], h_bin[j] = 1, 0
        if not improved and r > 20:
            break
    return h_bin, best_c5

def continuous_refine(h, n, rng, t_end):
    """Refine a (possibly binary) solution using small perturbations."""
    h = h.astype(float).copy()
    best_c5 = compute_c5(h, n)
    
    # Random coordinate descent with various step sizes
    steps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for step in steps:
        improvements = 0
        n_passes = max(100, 3*n)
        for _ in range(n_passes):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n-1)) % n
            delta = step * rng.choice([-1, 1])
            old_i, old_j = h[i], h[j]
            new_i, new_j = old_i + delta, old_j - delta
            if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
                continue
            h[i], h[j] = new_i, new_j
            c5 = compute_c5(h, n)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                improvements += 1
            else:
                h[i], h[j] = old_i, old_j
        if improvements == 0:
            continue
    
    h = repair(h, n)
    best_c5 = compute_c5(h, n)
    return h, best_c5

def soft_max_obj(v, n, alpha):
    """Smooth approximation to max of correlation."""

    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > t_end:
            break
        stall = 0
        n_trials = max(500, 8 * n)
        for _ in range(n_trials):
            if time.time() > t_end:
                break
            # Recompute correlation to find hot lag
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - (n - 1)
            # Choose pair (i, j) that affects lag k
            if rng.rand() < 0.6:
                if k >= 0 and n - k > 1:
                    i = rng.randint(0, n - k)
                    j = i + k
                elif k < 0 and n + k > 1:
                    i = rng.randint(-k, n)
                    j = i + k
                else:
                    i = rng.randint(n)
                    j = (i + rng.randint(1, n)) % n
            else:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(n - 1)) % n
            if i == j:
                continue
            old_i, old_j = h[i], h[j]
            for sign in (1, -1):
                di = sign * step
                ni, nj = old_i + di, old_j - di
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5 = compute_c5(h, n)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    stall = 0
                    break
                h[i], h[j] = old_i, old_j
            else:
                stall += 1
        if stall > n_trials * 0.7:
            continue
    h = repair(h, n)
    best_c5 = compute_c5(h, n)
    return h, best_c5

def swap_cd(h, n, rng, t_end):
    """Try full swaps of h[i] and h[j]."""
    h = h.astype(float).copy()
    best_c5 = compute_c5(h, n)
    n_trials = max(200, 4 * n)
    for _ in range(n_trials):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        hi, hj = h[i], h[j]
        h[i], h[j] = hj, hi
        c5 = compute_c5(h, n)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
        else:
            h[i], h[j] = hi, hj
    h = repair(h, n)
    best_c5 = compute_c5(h, n)
    return h, best_c5

def resample(h_src, n_src, n_dst):
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_new = np.interp(x_new, x_old, h_src)
    return repair(h_new, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 777)
    best_h = None
    best_c5 = np.inf
    best_n = 64

    init_h = None
    init_n = 64
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass

    if init_h is not None:
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, init_n)
        best_n = init_n
    else:
        best_h = repair(np.full(64, 0.5), 64)
        best_c5 = compute_c5(best_h, 64)
        best_n = 64

    # Phase 1: SLSQP on initial (20% budget)
    t1 = t0 + budget_s * 0.18
    h_work = best_h.copy()
    for alpha in [5000, 20000, 80000, 300000, 1000000]:
        if time.time() > t1:
            break
        h_new = slsqp_opt(h_work, best_n, alpha, 500)
        c5_new = compute_c5(h_new, best_n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        h_work = h_new if c5_new < compute_c5(h_work, best_n) else h_work

    # Phase 2: Intensive CD (30% budget)
    t2 = t0 + budget_s * 0.48
    h_out, c5_out = intensive_cd(best_h, best_n, rng, t2)
    if c5_out < best_c5:
        best_c5 = c5_out
        best_h = h_out
    h_out, c5_out = swap_cd(best_h, best_n, rng, t2)
    if c5_out < best_c5:
        best_c5 = c5_out
        best_h = h_out

    # Phase 3: Try nearby n values (20% budget)
    t3 = t0 + budget_s * 0.68
    n_candidates = sorted(set([best_n - 2, best_n, best_n + 2, best_n + 4, best_n - 4]))
    n_candidates = [nn for nn in n_candidates if 40 <= nn <= 120 and nn % 2 == 0 and nn != best_n]
    for n_alt in n_candidates:
        if time.time() > t3:
            break
        h_seed = resample(best_h, best_n, n_alt)
        h_alt = slsqp_opt(h_seed, n_alt, 50000, 300)
        c5_alt = compute_c5(h_alt, n_alt)
        h_alt, c5_alt = intensive_cd(h_alt, n_alt, rng, min(time.time() + 60, t3))
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt
            best_n = n_alt

    # Phase 4: Another SLSQP + CD cycle (15% budget)
    t4 = t0 + budget_s * 0.83
    for alpha in [500000, 2000000, 5000000]:
        if time.time() > t4:
            break
        h_new = slsqp_opt(best_h, best_n, alpha, 600)
        c5_new = compute_c5(h_new, best_n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    h_out, c5_out = intensive_cd(best_h, best_n, rng, t4)
    if c5_out < best_c5:
        best_c5 = c5_out
        best_h = h_out

    # Phase 5: Final ultra-fine CD (15% budget)
    t5 = t0 + budget_s * 0.985
    h_out, c5_out = intensive_cd(best_h, best_n, rng, t5)
    if c5_out < best_c5:
        best_c5 = c5_out
        best_h = h_out
    h_out, c5_out = swap_cd(best_h, best_n, rng, t5)
    if c5_out < best_c5:
        best_c5 = c5_out
        best_h = h_out

    # Final SLSQP if time remains
    if time.time() < t0 + budget_s * 0.995:
        h_new = slsqp_opt(best_h, best_n, 10000000, 200)
        c5_new = compute_c5(h_new, best_n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
