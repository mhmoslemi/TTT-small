                diff -= add

    ht = h_full.copy()
    ht[s1:s1+bs] = x[:bs]
    if bs2 > 0:
        ht[s2:s2+bs2] = x[bs:]
    corr = np.correlate(ht, 1.0 - ht, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    val += 80.0 * (x[:bs].sum() - sum1)**2
    if bs2 > 0:
        val += 80.0 * (x[bs:].sum() - sum2)**2
    return val

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def block_cd(h, n, dx, rng, deadline, alpha, block_size):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    while time.time() < deadline:
        if rng.random() < 0.35 and n > 2 * block_size + 2:
            bs = block_size
            s1 = rng.randint(0, n - bs + 1)
            s2 = rng.randint(0, n - bs + 1)
            if abs(s1 - s2) < bs + 1:
                continue
            b1 = h[s1:s1+bs].copy()
            b2 = h[s2:s2+bs].copy()
            x0 = np.concatenate([b1, b2])
            total = 2 * bs
            try:
                res = minimize(block_obj, x0,
                             args=(h, s1, bs, b1.sum(), alpha, s2, bs, b2.sum()),
                             method='L-BFGS-B', bounds=[(0.0, 1.0)]*total,
                             options={'maxiter': 30, 'ftol': 1e-13})
                ht = h.copy()
                ht[s1:s1+bs] = np.clip(res.x[:bs], 0, 1)
                ht[s2:s2+bs] = np.clip(res.x[bs:], 0, 1)
                new_c5 = compute_c5(ht, dx)
                if new_c5 < best_c5 - 1e-14:
                    h = ht
                    best_c5 = new_c5
            except Exception:
                pass
        else:
            bs = block_size
            s1 = rng.randint(0, n - bs + 1)
            b1 = h[s1:s1+bs].copy()
            x0 = b1.copy()
            try:
                res = minimize(block_obj, x0,
                             args=(h, s1, bs, b1.sum(), alpha, -1, 0, 0.0),
                             method='L-BFGS-B', bounds=[(0.0, 1.0)]*bs,
                             options={'maxiter': 35, 'ftol': 1e-13})
                ht = h.copy()
                ht[s1:s1+bs] = np.clip(res.x, 0, 1)
                diff = b1.sum() - ht[s1:s1+bs].sum()
                if abs(diff) > 1e-10:
                    for di in range(bs):
                        adj = min(max(diff, -(1.0 - ht[s1+di]), -ht[s1+di]), 1.0 - ht[s1+di])
                        ht[s1+di] += adj
                        diff -= adj
                        if abs(diff) < 1e-12:
                            break
                new_c5 = compute_c5(ht, dx)
                if new_c5 < best_c5 - 1e-14:
                    h = ht
                    best_c5 = new_c5
            except Exception:
                pass
    return h, best_c5

def pair_search(h, n, dx, rng, deadline, scales):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for scale in scales:
        improved = False
        while time.time() < deadline:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i] = ni
            h[j] = nj
            c5n = compute_c5(h, dx)
            if c5n < best_c5 - 1e-14:
                best_c5 = c5n
                improved = True
            else:
                h[i] -= d
                h[j] += d
        if not improved and scale < 0.002:
            break
    return h, best_c5

def block_shift_search(h, n, dx, rng, deadline, n_iters):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for _ in range(n_iters):
        if time.time() >= deadline:
            break
        bs = rng.randint(2, 6)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.015) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if h[i+di] + eps < -1e-9 or h[i+di] + eps > 1.0 + 1e-9:
                ok = False
                break
            if h[j+di] - eps < -1e-9 or h[j+di] - eps > 1.0 + 1e-9:
                ok = False
                break
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5n = compute_c5(h, dx)
        if c5n < best_c5 - 1e-14:
            best_c5 = c5n
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    dx0 = 2.0 / n0
    h0 = repair(h_init.copy(), n0)
    c5_0 = compute_c5(h0, dx0)
    best_h, best_c5, best_n = h0.copy(), c5_0, n0

    n = n0
    dx = dx0
    target = n / 2.0
    h = h0.copy()
    c5h = c5_0

    # Phase 1: L-BFGS full (35s)
    t_end = t_start + 35
    for alpha in [800, 4000, 15000]:
        if time.time() >= t_end:
            break
        res = minimize(smooth_obj, h, args=(alpha, target, 5000),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': 350, 'ftol': 1e-16, 'gtol': 1e-9})
        hr = repair(res.x, n)
        cr = compute_c5(hr, dx)
        if cr < c5h:
            h, c5h = hr, cr
    if c5h < best_c5:
        best_h, best_c5, best_n = h.copy(), c5h, n

    # Phase 2: Block CD main (280s)
    if remaining() > 450:
        t_end = time.time() + min(280, remaining() - 150)
        h, c5h = block_cd(h, n, dx, rng, t_end, alpha=2500, block_size=4)
        if c5h < best_c5:
            best_h, best_c5, best_n = h.copy(), c5h, n

    # Phase 3: Pair search (180s)
    if remaining() > 250:
        t_end = time.time() + min(180, remaining() - 100)
        h, c5h = pair_search(h, n, dx, rng, t_end,
                           [0.02, 0.008, 0.003, 0.001, 0.0004])
        if c5h < best_c5:
            best_h, best_c5, best_n = h.copy(), c5h, n

    # Phase 4: Multi-restart (80s)
    if remaining() > 170:
        t_end = time.time() + min(80, remaining() - 80)
        for r in range(20):
            if time.time() >= t_end:
                break
            hp = h.copy()
            kind = r % 4
            if kind == 0:
                hp += rng.normal(0, 0.025, n)
            elif kind == 1:
                f = rng.randint(1, 7)
                hp += 0.04 * np.sin(2*np.pi*f*np.arange(n)/n + rng.uniform(0, 6.28))
            elif kind == 2:
                hp = np.roll(hp, rng.randint(8, n-8))
            else:
                f1, f2 = rng.randint(1, 5), rng.randint(1, 6)
                tt = np.arange(n) / n
                hp += 0.04 * np.sin(2*np.pi*f1*tt) * np.cos(2*np.pi*f2*tt + rng.uniform(0, 6.28))
            hp = repair(hp, n)
            res = minimize(smooth_obj, hp, args=(4000, target, 5000),
                         method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 250, 'ftol': 1e-15})
            hr = repair(res.x, n)
            cr = compute_c5(hr, dx)
            if cr < c5h:
                h, c5h = hr, cr
        if c5h < best_c5:
            best_h, best_c5, best_n = h.copy(), c5h, n

    # Phase 5: Block CD refinement (120s)
    if remaining() > 120:
        t_end = time.time() + min(120, remaining() - 60)
        h, c5h = block_cd(h, n, dx, rng, t_end, alpha=8000, block_size=3)
        if c5h < best_c5:
            best_h, best_c5, best_n = h.copy(), c5h, n

    # Phase 6: Block shift (60s)
    if remaining() > 80:
        t_end = time.time() + min(60, remaining() - 30)
        h, c5h = block_shift_search(h, n, dx, rng, t_end, 20000)
        if c5h < best_c5:
            best_h, best_c5, best_n = h.copy(), c5h, n

    # Phase 7: High-alpha L-BFGS polish (50s)
    if remaining() > 50:
        t_end = time.time() + min(50, remaining() - 20)
        for alpha in [50000, 200000]:
            if time.time() >= t_end:
                break
            res = minimize(smooth_obj, h, args=(alpha, target, 10000),
                         method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 500, 'ftol': 1e-17, 'gtol': 1e-10})
            hr = repair(res.x, n)
            cr = compute_c5(hr, dx)
            if cr < c5h:
                h, c5h = hr, cr
        if c5h < best_c5:
            best_h, best_c5, best_n = h.copy(), c5h, n

    # Phase 8: Ultra-fine pair search (40s)
    if remaining() > 25:
        t_end = time.time() + max(5, min(40, remaining() - 5))
        h, c5h = pair_search(h, n, dx, rng, t_end,
                           [0.0008, 0.0003, 0.0001, 0.00004])
        if c5h < best_c5:
            best_h, best_c5, best_n = h.copy(), c5h, n

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
