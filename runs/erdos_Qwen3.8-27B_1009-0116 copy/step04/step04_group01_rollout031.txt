        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _shift_grad(h, k_idx):
    n = len(h)
    sh = (n - 1) - k_idx
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _pg_mom(h, target, nsteps, step0, mom, bin_period, bin_str):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    for it in range(nsteps):
        c = _corr(h)
        top_k = np.argsort(-c)[:5]
        g = np.zeros(n)
        for ki in top_k:
            g += _shift_grad(h, ki)
        g /= len(top_k) * (n / 2.0)
        v = mom * v - step0 * g
        h = _proj_to_sum(h + v, target)
        if bin_period > 0 and (it + 1) % bin_period == 0:
            h = h + bin_str * (np.round(h) - h)
            h = _proj_to_sum(h, target)
    return h

def _topk_desc(h, target, nsteps, step0, topk):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for _ in range(nsteps):
        c = _corr(h)
        top_idx = np.argsort(-c)[:topk]
        g = np.zeros(n)
        for ki in top_idx:
            g += _shift_grad(h, ki)
        g *= dx / topk
        for sf in [1.0, 0.5, 0.25, 0.1, 0.04, 0.01]:
            hn = _proj_to_sum(h - step0 * sf * g, target)
            cn = _corr(hn)
            if cn.max() < c.max() - 1e-14:
                h = hn
                break
    return h

def _pairwise(h, target, niter, scale, rng):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    for _ in range(niter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-16:
            c5 = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5

def _block_push(h, target, niter, rng):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    for _ in range(niter):
        start = rng.randint(n)
        length = rng.randint(2, max(3, n // 10))
        end = min(start + length, n)
        direction = 1.0 if rng.rand() < 0.5 else -1.0
        delta = direction * rng.uniform(0.02, 0.08)
        h_new = h.copy()
        h_new[start:end] = np.clip(h_new[start:end] + delta, 0, 1)
        h_new = _proj_to_sum(h_new, target)
        c5n = _eval(h_new)
        if c5n < c5 - 1e-15:
            h = h_new
            c5 = c5n
    return h, c5

def _anneal(h, target, niter, T0, T1, rng):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    for it in range(niter):
        T = T0 * (T1 / T0) ** (it / max(1, niter - 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.01 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 or rng.rand() < np.exp(-(c5n - c5) / max(T, 1e-8)):
            c5 = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5
