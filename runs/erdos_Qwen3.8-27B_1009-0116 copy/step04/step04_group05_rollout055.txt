    g = np.zeros(n)

        g[:] = 1.0 - 2.0 * h

        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

    c = np.correlate(x, 1.0 - x, mode='full')

    s = np.clip(s, -500, 500)
    sm = m + np.log(np.sum(np.exp(s))) / a

def active_subgrad(h, n, dx, t0, lim, n_iter=3000):
    """Active-set subgradient method with backtracking"""
    bh, bc = h.copy(), c5v(h, dx)
    h = h.copy()
    alpha = 0.05
    beta = 0.0  # momentum
    h_prev = h.copy()
    
    for it in range(n_iter):
        if time.time() - t0 > lim:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        fmax = c.max()
        # Active set: lags within threshold
        thr = fmax - max(0.002 * c.std(), 1e-6 * fmax)
        idx = np.where(c >= thr)[0]
        if len(idx) == 0:
            idx = np.array([np.argmax(c)])
        # Limit active set size
        if len(idx) > 30:
            idx = np.argsort(c)[::-1][:30]
        
        # Subgradient
        g = np.zeros(n)
        for ki in idx:
            g += grad1(h, ki, n)
        g /= len(idx)
        
        # Nesterov momentum
        if beta > 0:
            y = h + beta * (h - h_prev)
            g_y = np.zeros(n)
            for ki in idx:
                g_y += grad1(y, ki, n)
            g_y /= len(idx)
            g = g_y + beta * (g_y - g)
        
        # Step
        h_new = repair(h - alpha * g, n)
        c_new = c5v(h_new, dx)
        
        if c_new < c5v(h, dx) - 1e-15:
            h_prev = h.copy()
            h = h_new
            alpha = min(alpha * 1.08, 1.0)
            if c_new < bc:
                bc, bh = c_new, h.copy()
            # Update momentum
            beta = (1 + np.sqrt(1 + 4 * beta ** 2)) / (1 + 2 * beta)  # simplified Nesterov
        else:
            alpha *= 0.7
            if alpha < 1e-6:
                # Reset
                alpha = 0.05
                beta = 0
                break
    
    return bh, bc

    return bh, bc

def spec_phase(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a_s in [0.2, 0.35, 0.5, 0.4, 0.25, 0.15, 0.08, 0.04]:
        h = bh.copy()
        for _ in range(300):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a_s) * mag + a_s * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-18:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def coord_phase(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    halts = 0
    for _ in range(300):
        if time.time() - t0 > lim:
            break
        imp = False
        for _ in range(600):
            if time.time() - t0 > lim:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            for d in (0.01, 0.005, 0.002, 0.0008, -0.01, -0.005, -0.002, -0.0008):
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = c5v(h, dx)
                    if c < cc - 1e-15:
                        h = repair(h, n)
                        cc = c5v(h, dx)
                        imp = True
                        if cc < bc:
                            bc, bh = cc, h.copy()
                        break
                    h[i], h[j] = h[i] - d, h[j] + d
        halts = halts + 1 if not imp else 0
        if halts >= 5:
            break
    return bh, bc

def micro_polish(h, n, dx, rng, cnt, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    for _ in range(cnt):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = (rng.rand() - 0.5) * 0.003
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

def full_opt(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = lbfgs_phase(bh, n, dx, t0, lim - min(350, int(0.4 * lim)))
    if c < bc:
        bh, bc = r, c
    r, c = active_phase(bh, n, dx, t0, lim - min(250, int(0.3 * lim)))
    if c < bc:
        bh, bc = r, c
    r, c = slsqp_phase(bh, n, dx, t0, lim - min(150, int(0.2 * lim)))
    if c < bc:
        bh, bc = r, c
    r, c = spec_phase(bh, n, dx, t0, lim - min(80, int(0.12 * lim)))
    if c < bc:
        bh, bc = r, c
    r, c = lbfgs_phase(bh, n, dx, t0, lim - min(40, int(0.06 * lim)))
    if c < bc:
        bh, bc = r, c
    r, c = coord_phase(bh, n, dx, rng, t0, lim - min(15, int(0.03 * lim)))
    if c < bc:
        bh, bc = r, c
    r, c = micro_polish(bh, n, dx, rng, 15000, t0, lim - 2)
    if c < bc:
        bh, bc = r, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    t_rem = budget_s - (time.time() - t0)
    t_n1 = int(0.38 * t_rem)
    t_n2 = int(0.35 * t_rem)
    t_n3 = int(0.20 * t_rem)

    r, c = full_opt(h0, nl, dx0, seed, t0, t_n1)
    if c < bc:
        bh, bc, bn = r, c, nl

    n2 = nl * 2
    if n2 < 1000 and time.time() - t0 < budget_s - t_n3 - 30:
        xo = np.linspace(0, 2, nl, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        hh = repair(np.interp(xn, xo, h0, period=2.0), n2)
        d2 = 2.0 / n2
        r, c = full_opt(hh, n2, d2, seed + 100, t0, int(budget_s - t_n3 - 30))
        if c < bc:
            bh, bc, bn = r, c, n2
        else:
            bh, bc, bn = r, c, n2 if c < bc else (bh, bc, bn)

    n3 = nl * 3
    if n3 < 1000 and time.time() - t0 < budget_s - 25:
        xo = np.linspace(0, 2, bn, endpoint=False)
        xn = np.linspace(0, 2, n3, endpoint=False)
        hh = repair(np.interp(xn, xo, bh, period=2.0), n3)
        d3 = 2.0 / n3
        rem = int(budget_s - 25 - (time.time() - t0))
        if rem > 60:
            rng3 = np.random.RandomState(seed + 300)
            r, c = full_opt(hh, n3, d3, seed + 200, t0, rem)
            if c < bc:
                bh, bc, bn = r, c, n3
            else:
                r2, c2 = coord_phase(hh, n3, d3, rng3, t0, rem - 5)
                if c2 < bc:
                    bh, bc, bn = r2, c2, n3

    if time.time() - t0 < budget_s - 5:
        rng2 = np.random.RandomState(seed + 700)
        hp = repair(bh + 0.01 * rng2.randn(bn), bn)
        r, c = coord_phase(hp, bn, 2.0 / bn, rng2, t0, int(budget_s - 3))
        r, c = micro_polish(r, bn, 2.0 / bn, rng2, 8000, t0, int(budget_s - 1))
        bn_r = bn
        dx_f = 2.0 / bn
        r, c = micro_polish(r, bn_r, dx_f, rng2, 5000, t0, int(budget_s - 1))
        if c < bc:
            bh, bc, bn = r, c, bn_r

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
