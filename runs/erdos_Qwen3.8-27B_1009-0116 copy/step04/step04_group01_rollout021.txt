        for alpha in [500, 5000, 30000]:
            if _elapsed() > budget_s - 250:
                break
            h_hi = _slsqp_with_timeout(h_hi, alpha, 3000, 1e-18, 0)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if best_n == n1:
            for _it in range(3000):
                if _elapsed() > budget_s - 200:
                    break
                g = _topk_grad(best_h, 4)
                for step in [0.02, 0.01, 0.005, 0.001]:
                    h_new = _proj_sum(best_h - step * g, n1)
                    c5n = _eval(h_new)
                    if c5n < best_c5 - 1e-16:
                        best_c5 = c5n
                        best_h = h_new
                        break

    # Phase 5: Multi-start perturbation at best resolution
    n_cur = best_n
    if _elapsed() < budget_s - 150:
        for _r in range(10):
            if _elapsed() > budget_s - 150:
                break
            hp = best_h + 0.02 * rng.randn(n_cur)
            hp = _repair(hp, n_cur)
            h_new = _slsqp_with_timeout(hp, 50000, 2000, 1e-18, 0)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_cur

    # Phase 6: Final SLSQP polish
    if _elapsed() < budget_s - 60:
        h = _slsqp_with_timeout(best_h, 500000, 5000, 1e-20, 0)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        for _it in range(2000):
            if _elapsed() > budget_s - 40:
                break
            g = _topk_grad(best_h, 3)
            for step in [0.01, 0.005, 0.002, 0.0005, 0.0002]:
                h_new = _proj_sum(best_h - step * g, best_n)
                c5n = _eval(h_new)
                if c5n < best_c5 - 1e-17:
                    best_c5 = c5n
                    best_h = h_new
                    break

    # Phase 7: Ultra-fine pairwise
    if _elapsed() < budget_s - 3:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(30000):
            if _elapsed() > budget_s - 3:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 3e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
