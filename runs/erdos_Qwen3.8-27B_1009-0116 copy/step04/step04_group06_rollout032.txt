def _paired_s_values(h, i, j, n):
    """Compute s[t] for all lags t=-(n-1)...(n-1)"""
    # t ranges from -(n-1) to (n-1), total 2n-1 values

def _exact_line_search(c0, s, d_lo, d_hi, M=20):
    """Find d in [d_lo, d_hi] minimizing max_k (c0[k] + d*s[k])
    Only consider top M lags for breakpoints"""
    # Get top M lags by current value
    top_idx = np.argsort(c0)[-M:]
    
    # Candidate d values: boundaries + breakpoints among top M
    candidates = [d_lo, d_hi, 0.0]
    
    # Breakpoints among top M lags
    for a in range(M):
        for b in range(a+1, M):
            ka, kb = top_idx[a], top_idx[b]
            ds = s[ka] - s[kb]
            if abs(ds) > 1e-14:
                d_bp = (c0[kb] - c0[ka]) / ds
                if d_lo <= d_bp <= d_hi:
                    candidates.append(d_bp)
    
    # Also add breakpoints between top M and the current max
    # (in case a non-top-M lag becomes the max)
    
    candidates = np.array(candidates)
    candidates = np.clip(candidates, d_lo, d_hi)
    candidates = np.unique(candidates)  # or just sort and deduplicate
    
    # Evaluate objective at each candidate
    best_d = 0.0
    best_val = np.max(c0)
    for d in candidates:
        vals = c0 + d * s
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_d = d
    
    return best_d, best_val

import numpy as np
import time

def _c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _c5_and_corr(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0/n), corr

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _safe_get(h, idx):
    """Safe array indexing: return 0 for out-of-bounds"""
    n = len(h)
    out = np.zeros(len(idx), dtype=float)
    mask = (idx >= 0) & (idx < n)
    out[mask] = h[idx[mask]]
    return out

def _paired_s(h, i, j, n):
    """Compute s[t] for all lags for paired update (h[i]+=d, h[j]-=d)"""
    t = np.arange(-(n-1), n, dtype=int)  # lag values, shape (2n-1,)
    
    A1 = _safe_get(h, i - t)  # h[i-t]
    A2 = _safe_get(h, i + t)  # h[i+t]
    B1 = _safe_get(h, j - t)  # h[j-t]
    B2 = _safe_get(h, j + t)  # h[j+t]
    
    A = (1.0 - A1) - A2
    B = (1.0 - B1) - B2
    return A - B

def _line_search(c0, s, d_lo, d_hi, M=30):
    """Find d in [d_lo, d_hi] minimizing max(c0 + d*s)"""
    if d_lo >= d_hi:
        return 0.0, c0.max()
    
    # Active set: top M lags
    top = np.argsort(c0)[-M:]
    
    # Base candidates
    candidates = [d_lo, d_hi, 0.0]
    
    # Breakpoints among top M
    for a in range(M):
        for b in range(a+1, M):
            ka, kb = top[a], top[b]
            ds = s[ka] - s[kb]
            if abs(ds) > 1e-12:
                d_bp = (c0[kb] - c0[ka]) / ds
                if d_lo - 1e-12 <= d_bp <= d_hi + 1e-12:
                    candidates.append(d_bp)
    
    candidates = np.array(candidates)
    candidates = np.clip(candidates, d_lo, d_hi)
    
    # Sort and evaluate
    candidates = np.sort(np.unique(candidates))
    
    best_d = 0.0
    best_val = c0.max()
    for d in candidates:
        v = (c0 + d * s).max()
        if v < best_val - 1e-15:
            best_val = v
            best_d = d
    
    return best_d, best_val

    obj = (c0[None, :] + cands[:, None] * s[None, :]).max(axis=1)
    return cands[obj.argmin()]

def _cd_pass(h, n, t, corr, best, budget_end, M, n_rand, n_struct):
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    h = h.copy()
    ops = 0
    for p in range(n_rand + n_struct):
        if time.time() > budget_end:
            break
        if p < n_struct:
            i = p % n
            j = (i + n // 2) % n
            if i == j:
                j = (i + 1) % n
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        s = _s_pair(h, i, j, t)
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dhi - dlo < 1e-14:
            continue
        d = _opt_d(corr, s, dlo, dhi, M)
        if abs(d) > 1e-14:
            h[i] += d
            h[j] -= d
            corr = corr + d * s
            ops += 1
            cm = corr.max() * 2.0 / n
            if cm < best:
                best = cm
        if ops % 80 == 0:
            h = _repair(h, n)
            corr = _corr_full(h)
            cv = _c5(h, n)
            if cv < best:
                best = cv
    return h, corr, best

def _greedy_sweep(h, n, t, corr, best, M):
    h = h.copy()
    improved = False
    for i in range(n):
        si = (1.0 - _safe(h, i - t)) - _safe(h, i + t)
        bi_best = best
        bd_best = 0.0
        bj_best = -1
        for j in range(n):
            if j == i:
                continue
            sj = (1.0 - _safe(h, j - t)) - _safe(h, j + t)
            s = si - sj
            dlo = max(-h[i], h[j] - 1.0)
            dhi = min(1.0 - h[i], h[j])
            if dhi - dlo < 1e-14:
                continue
            d = _opt_d(corr, s, dlo, dhi, M)
            if abs(d) > 1e-14:
                cm = (corr + d * s).max() * 2.0 / n
                if cm < bi_best:
                    bi_best = cm
                    bd_best = d
                    bj_best = j
        if bj_best >= 0:
            h[i] += bd_best
            h[bj_best] -= bd_best
            corr = corr + bd_best * _s_pair(h - np.array([bd_best] if i == bj_best else [0.0]) + (bd_best if i != bj_best else 0.0), i, bj_best, t)
            # Simpler: recompute s before update
            improved = True
            if bi_best < best:
                best = bi_best
    return h, corr, best, improved

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=np.float64).copy(), n0)
    best_h = h0.copy()
    best_n = n0
    best_c5 = _c5(best_h, best_n)
    
    # Phase 1: CD on initial resolution (35% budget)
    ph1_end = t0 + budget_s * 0.30
    t_arr = np.arange(-(best_n - 1), best_n)
    corr = _corr_full(best_h)
    for alpha in [15, 25, 40]:
        if time.time() > ph1_end:
            break
        for _ in range(50):
            if time.time() > ph1_end:
                break
            best_h, corr, best_c5 = _cd_pass(best_h, best_n, t_arr, corr, best_c5,
                                             ph1_end, alpha, 300, 100)
    
    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    
    # Phase 2: Greedy sweeps (20% budget)
    ph2_end = t0 + budget_s * 0.50
    t_arr = np.arange(-(best_n - 1), best_n)
    corr = _corr_full(best_h)
    for _ in range(20):
        if time.time() > ph2_end:
            break
        best_h, corr, best_c5, imp = _greedy_sweep(best_h, best_n, t_arr, corr, best_c5, 20)
        best_h = _repair(best_h, best_n)
        corr = _corr_full(best_h)
        cv = _c5(best_h, best_n)
        if cv < best_c5:
            best_c5 = cv
        if not imp:
            break
    
    # Phase 3: Multi-resolution (25% budget)
    ph3_end = t0 + budget_s * 0.75
    x_old = np.linspace(0, 2.0, best_n, endpoint=False)
    for n_try in [best_n - 10, best_n - 5, best_n + 5, best_n + 10, best_n + 20]:
        if n_try < 40 or time.time() > ph3_end:
            continue
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = _repair(h_new, n_try)
        c5_new = _c5(h_new, n_try)
        # Optimize at new resolution
        t_new = np.arange(-(n_try - 1), n_try)
        corr_new = _corr_full(h_new)
        spend = min(30.0, ph3_end - time.time())
        for _ in range(max(1, int(spend / 3.0))):
            if time.time() > ph3_end:
                break
            h_new, corr_new, c5_new = _cd_pass(h_new, n_try, t_new, corr_new, c5_new,
                                               ph3_end, 30, 200, 80)
        h_new = _repair(h_new, n_try)
        c5_new = _c5(h_new, n_try)
        if c5_new < best_c5 - 1e-8:
            best_c5 = c5_new
            best_h = h_new
            best_n = n_try
            t_arr = np.arange(-(best_n - 1), best_n)
    
    # Phase 4: Fine CD + tiny moves (20% budget)
    ph4_end = t0 + budget_s * 0.95
    t_arr = np.arange(-(best_n - 1), best_n)
    corr = _corr_full(best_h)
    for _ in range(100):
        if time.time() > ph4_end:
            break
        best_h, corr, best_c5 = _cd_pass(best_h, best_n, t_arr, corr, best_c5,
                                         ph4_end, 50, 400, 50)
    
    # Phase 5: Ultra-fine random perturbation + CD
    ph5_end = t0 + budget_s - 10
    h = best_h.copy()
    c5_cur = _c5(h, best_n)
    t_arr = np.arange(-(best_n - 1), best_n)
    corr = _corr_full(h)
    while time.time() < ph5_end:
        i = rng.randint(best_n)
        j = rng.randint(best_n)
        if i == j:
            continue
        d = rng.choice([-0.0002, -0.0001, -0.00005, 0.00005, 0.0001, 0.0002])
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            s = _s_pair(h - d * (np.arange(best_n) == i) - (-d) * (np.arange(best_n) == j), i, j, t_arr)
            c = _c5(h, best_n)
            if c < c5_cur - 1e-15:
                c5_cur = c
                corr = _corr_full(h)
                # Quick CD polish
                h, corr, c5_cur = _cd_pass(h, best_n, t_arr, corr, c5_cur,
                                           min(time.time() + 0.5, ph5_end), 30, 50, 20)
                h = _repair(h, best_n)
                c5_cur = _c5(h, best_n)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    
    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
