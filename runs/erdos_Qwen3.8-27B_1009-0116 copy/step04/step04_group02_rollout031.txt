```python
    """Compute gradient of C5 approximation wrt h"""
    corr = np.correlate(h, 1-h, mode='full')
    m = max(corr)
    # Find active terms (within tolerance of max)
    tol = 1e-6
    active = corr > m - tol
    if not np.any(active):
        active = np.array([corr == m])
    # Gradient of each corr_k
    # corr_k corresponds to lag (k - (n-1)) in 0-indexed
    # For lag L: corr_L = sum_i h[i]*(1-h[i+L])
    # d/dh[j] = (1-h[j+L]) * delta(i=j) - h[j-L] * delta(i+L=j)
    #         = (1-h[j+L]) - h[j-L] (with bounds)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_value(x, alpha, target, pen, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lse_grad(x, alpha, target, pen, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-16:
            continue
        L = k - (n - 1)
        gk = np.zeros(n)
        if L >= 0:
            lc = min(L, n)
            gk[lc:n] += (1.0 - x[0:n-lc])
            gk[0:n-lc] -= x[lc:n]
        else:
            M = min(-L, n)
            gk[0:n-M] += (1.0 - x[M:n])
            gk[M:n] -= x[0:n-M]
        grad += w[k] * gk
    if pen > 0:
        grad += 2.0 * pen * (np.sum(x) - target)
    return grad

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_value, h, args=(alpha, target, pen, n),
                      jac=lambda x, a, t, p, nn: lse_grad(x, a, t, p, nn),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_opt(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_value, h, args=(alpha, target, 0.0, n),
                      jac=lambda x, a, t, p, nn: lse_grad(x, a, t, p, nn),
                      method='SLSQP', bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': lambda x, t: np.sum(x) - t, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def pair_line_search(h, i, j, n):
    """Exact-ish line search for h[i]+=d, h[j]-=d"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = np.zeros(len(corr))
    for k in range(len(corr)):
        L = k - (n - 1)
        if L >= 0:
            vi = (1.0 - h[i - L]) if (0 <= i - L < n - L) else 0.0
            vi -= h[i + L] if (0 <= i + L < n and i < n - L) else 0.0
            vj = (1.0 - h[j - L]) if (0 <= j - L < n - L) else 0.0
            vj -= h[j + L] if (0 <= j + L < n and j < n - L) else 0.0
        else:
            M = -L
            vi = (1.0 - h[i + M]) if (0 <= i + M < n and i < n - M) else 0.0
            vi -= h[i - M] if (0 <= i - M < n and i >= M) else 0.0
            vj = (1.0 - h[j + M]) if (0 <= j + M < n and j < n - M) else 0.0
            vj -= h[j - M] if (0 <= j - M < n and j >= M) else 0.0
        g[k] = vi - vj
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_hi - d_lo < 1e-12:
        return 0.0
    d_grid = np.linspace(d_lo + 1e-9, d_hi - 1e-9, 24)
    best_d, best_v = 0.0, np.max(corr)
    for d in d_grid:
        v = np.max(corr + d * g)
        if v < best_v:
            best_v = v; best_d = d
    return best_d

def gradient_anneal(h, n, dx, rng, deadline, T0, T1, n_iter):
    """Gradient-guided simulated annealing with pair moves and block moves"""
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    n_corr = 2 * n - 1
    for it in range(n_iter):
        if time.time() >= deadline:
            break
        frac = it / max(n_iter, 1)
        T = T0 * (T1 / T0) ** frac
        if rng.random() < 0.7:
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = pair_line_search(h, i, j, n)
            if abs(d) < 1e-12:
                continue
            d += T * rng.normal(0, 0.01)
            ni = h[i] + d; nj = h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i] = ni; h[j] = nj
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
            elif rng.random() > np.exp(-(c5_new - best_c5) / max(T, 1e-12)):
                h[i] = ni - d; h[j] = nj + d
        else:
            bs = rng.randint(2, 6)
            i = rng.randint(0, n - bs)
            j = rng.randint(0, n - bs)
            if abs(i - j) < bs:
                continue
            eps = rng.uniform(-0.08, 0.08) * (1.0 + T * 5)
            ok = True
            for di in range(bs):
                if not (-1e-9 <= h[i+di]+eps <= 1.0+1e-9) or not (-1e-9 <= h[j+di]-eps <= 1.0+1e-9):
                    ok = False; break
            if not ok:
                continue
            h[i:i+bs] += eps; h[j:j+bs] -= eps
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
            elif rng.random() > np.exp(-(c5_new - best_c5) / max(T, 1e-12)):
                h[i:i+bs] -= eps; h[j:j+bs] += eps
    return h, best_c5

def fine_search(h, n, dx, rng, deadline):
    """Ultra-fine pair-based local search with line search"""
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    while time.time() < deadline:
        improved = False
        for _ in range(2000):
            if time.time() >= deadline:
                break
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = pair_line_search(h, i, j, n)
            if abs(d) < 1e-10:
                continue
            h[i] += d; h[j] -= d
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new; improved = True
            else:
                h[i] -= d; h[j] += d
        if not improved:
            break
    return h, best_c5

        try:
            res = minimize(sub_obj, x0, method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-12})
            x_new = np.clip(res.x, 0, 1)
            h[idx] = x_new
            h = repair(h, n)
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
        except Exception:
            pass
    return h, best_c5

def perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(n // 4, 3 * n // 4))
    elif kind == 1:
        f = rng.randint(1, 7)
        ph = rng.uniform(0, 2 * np.pi)
        hp += rng.uniform(0.02, 0.15) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.05, n)
    elif kind == 3:
        f1, f2 = rng.randint(1, 5), rng.randint(2, 7)
        p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
        t = np.arange(n) / n
        hp += rng.uniform(0.03, 0.12) * np.sin(2*np.pi*f1*t+p1) * np.sin(2*np.pi*f2*t+p2)
    else:
        hp *= rng.uniform(0.8, 1.2)
        hp += rng.uniform(-0.02, 0.02)
    return repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    h0_raw = np.array(initial_h_values, dtype=float)
    n = len(h0_raw)
    dx = 2.0 / n
    target = n / 2.0
    h = repair(h0_raw.copy(), n)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    # Phase 1: Quick L-BFGS on parent (45s)
    t_end = t_start + 45
    for alpha in [500, 2000, 8000]:
        if time.time() >= t_end:
            break
        for pen in [200, 2000]:
            if time.time() >= t_end:
                break
            hh, cc = lbfgs_opt(h, n, alpha, target, pen, 200)
            if cc < best_c5:
                h, best_c5, best_h = hh, cc, hh.copy()

    # Phase 2: SLSQP refinement (60s)
    t_end = t_start + 105
    for alpha in [500, 3000, 10000]:
        if time.time() >= t_end:
            break
        hh, cc = slsqp_opt(h, n, alpha, target, 300)
        if cc < best_c5:
            h, best_c5, best_h = hh, cc, hh.copy()

    # Phase 3: Gradient annealing - main exploration (250s)
    t_end = t_start + 355
    h, c5a = gradient_anneal(h, n, dx, rng, t_end, T0=0.004, T1=0.0001, n_iter=5000000)
    if c5a < best_c5:
        best_c5, best_h = c5a, h.copy()

    # Phase 4: L-BFGS polish after annealing (90s)
    t_end = t_start + 445
    for alpha in [2000, 8000, 30000]:
        if time.time() >= t_end:
            break
        hh, cc = lbfgs_opt(h, n, alpha, target, 5000, 300)
        if cc < best_c5:
            h, best_c5, best_h = hh, cc, hh.copy()

    # Phase 5: Multi-restart L-BFGS + SLSQP (180s)
    t_end = t_start + 625
    for r in range(20):
        if time.time() >= t_end or remaining() < 200:
            break
        hp = perturb(h, n, rng, r % 5)
        hh, cc = lbfgs_opt(hp, n, 3000, target, 3000, 200)
        hh, cc = slsqp_opt(hh, n, 10000, target, 250)
        if cc < best_c5:
            h, best_c5, best_h = hh, cc, hh.copy()

    # Phase 6: Block descent (100s)
    if remaining() > 200:
        t_end = time.time() + min(100, remaining() - 150)
        hh, cc = block_descent(h, n, dx, rng, t_end)
        if cc < best_c5:
            h, best_c5, best_h = hh, cc, hh.copy()

    # Phase 7: Fine line-search local search (180s)
    if remaining() > 130:
        t_end = time.time() + min(180, remaining() - 80)
        hh, cc = fine_search(h, n, dx, rng, t_end)
        if cc < best_c5:
            h, best_c5, best_h = hh, cc, hh.copy()

    # Phase 8: Final SLSQP + L-BFGS polish (50s)
    if remaining() > 60:
        hh, cc = slsqp_opt(h, n, 50000, target, 400)
        if cc < best_c5:
            h, best_c5, best_h = hh, cc, hh.copy()
        if remaining() > 30:
            hh, cc = lbfgs_opt(h, n, 100000, target, 10000, 400)
            if cc < best_c5:
                h, best_c5, best_h = hh, cc, hh.copy()

    # Phase 9: Ultra-fine local search (remaining)
    if remaining() > 15:
        t_end = time.time() + max(5, remaining() - 5)
        hh, cc = fine_search(h, n, dx, rng, t_end)
        if cc < best_c5:
            best_c5, best_h = cc, hh.copy()

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
```
