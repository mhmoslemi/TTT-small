    # These are cross-correlations! Can compute with FFT.
    # But for n=140, O(n²) is fine (140*140=19600 ops per i, total 140*19600 ≈ 2.7M).

    for i in range(n):

        # First sum: s from 0 to n-1-i
        for s in range(n - i):
            val += w[n - 1 + s] * (1.0 - h[i + s])
        # Second sum: s from 0 to i
        for s in range(i + 1):
            val -= w[n - 1 - s] * h[i - s]
        g[i] = val
