def weighted_grad(p, h, n):
    """Gradient of sum_k p[k] * corr[k](h)"""
    # corr[k] = sum_i h[i] * (1-h[i+k]) for valid i (in full correlation mode)
    # For np.correlate(h, (1-h), 'full'), the index mapping is:
    # result[k] = sum_i h[i] * (1-h)[i + (k-n+1)] for valid indices
    # Let s = k - n + 1 (shift)
    # result[k] = sum_{i: 0<=i<n, 0<=i+s<n} h[i] * (1-h[i+s])
    
    # d/dh[j] of result[k] = sum_i (1-h[i+s])*δ(i,j) - h[i]*h[i+s]*... 
    # Actually: d/dh[j] sum_i h[i](1-h[i+s]) = (1-h[j+s]) - h[j-s] (with boundary)
    
    n_corr = 2*n - 1
    grad = np.zeros(n)
    for k in range(n_corr):
        if p[k] < 1e-15:
            continue
        s = k - n + 1  # shift
        # For each j, contribution to grad[j]:
        # h[j] appears as first factor when 0<=j<n and 0<=j+s<n: contributes (1-h[j+s])
        # h[j] appears as second factor (in 1-h[j]) when 0<=j-s<n and 0<=j<n (shifted): 
        #   the term is h[j-s]*(1-h[j]), derivative w.r.t. h[j] is -h[j-s]
        for j in range(n):
            gj = 0.0
            if 0 <= j + s < n:
                gj += (1.0 - h[j + s])
            if 0 <= j - s < n:
                gj -= h[j - s]
            grad[j] += p[k] * gj
    
    return grad * (2.0 / n)  # multiply by dx
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(2000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def weighted_grad(p, h, n):
    """Gradient of sum_k p[k] * corr_k(h) where corr = correlate(h,1-h,'full')*dx"""
    dx = 2.0 / n
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        pk = p[k]
        if pk < 1e-18:
            continue
        s = k - n + 1
        if s >= 0:
            # h[j+s] valid for j in [0, n-1-s]
            if n - s > 0:
                grad[:n-s] += pk * (1.0 - h[s:])
            if s > 0 and n - s > 0:
                grad[s:] -= pk * h[:n-s]
        else:
            s2 = -s
            if n - s2 > 0:
                grad[s2:] += pk * (1.0 - h[:n-s2])
                grad[:n-s2] -= pk * h[s2:]
    grad *= dx
    return grad

def fw_minimize(p, h, n, n_iter=500, t_end=None):
    """Frank-Wolfe for min sum_k p[k]*corr_k(h) s.t. 0<=h<=1, sum(h)=n/2"""
    h = h.copy()
    for it in range(n_iter):
        if t_end is not None and time.time() > t_end:
            break
        grad = weighted_grad(p, h, n)
        # Linear minimization oracle: pick n/2 smallest grad entries -> set to 1
        idx = np.argsort(grad)[:n//2]
        v = np.zeros(n)
        v[idx] = 1.0
        t = 2.0 / (it + 2)
        h_new = (1.0 - t) * h + t * v
        h_new = np.clip(h_new, 0.0, 1.0)
        # Check improvement
        F_old = np.dot(p, compute_corr(h, n))
        F_new = np.dot(p, compute_corr(h_new, n))
        if F_new < F_old:
            h = h_new
        else:
            # Line search
            for tt in [0.5*t, 0.25*t, 0.1*t]:
                h_try = (1.0-tt)*h + tt*v
                h_try = np.clip(h_try, 0.0, 1.0)
                F_try = np.dot(p, compute_corr(h_try, n))
                if F_try < F_old:
                    h = h_try
                    break
    return h

def minmax_opt(h, n, t_end, n_outer=50, T0=0.05, cooling=0.95):
    """Alternating: minimize weighted sum, then update weights"""
    h = h.copy()
    n_corr = 2*n - 1
    p = np.ones(n_corr) / n_corr
    T = T0
    for outer in range(n_outer):
        if time.time() > t_end:
            break
        # Inner: minimize F_p(h)
        h = fw_minimize(p, h, n, 300, t_end)
        # Update p
        corr = compute_corr(h, n)
        max_c = np.max(corr)
        T = max(T0 * 0.01, T * cooling)
        p_new = np.exp((corr - max_c) / T)
        p_new /= p_new.sum()
        p = p_new
    return h
```

def targeted_moves(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        corr = compute_corr(h, n)
        max_c = np.max(corr)
        # Find active shifts (within 5% of max)
        active = np.where(corr > 0.95 * max_c)[0]
        improved = False
        for _ in range(5 * n):
            if time.time() > t_end:
                break
            # Pick a random active shift
            k = active[rng.randint(len(active))]
            s = k - n + 1
            # Find the pair that contributes most
            if s >= 0:
                valid_i = np.arange(n - s)
                contrib = h[valid_i] * (1.0 - h[valid_i + s])
            else:
                s2 = -s
                valid_i = np.arange(s2, n)
                contrib = h[valid_i] * (1.0 - h[valid_i - s2])
            if len(valid_i) == 0:
                continue
            i_star = valid_i[np.argmax(contrib)]
            # Try moving mass: decrease h[i_star], increase h[partner]
            partner = i_star + s if s >= 0 else i_star - s
            # Actually we want to increase h[partner] to reduce (1-h[partner])
            # or decrease h[i_star]
            # And compensate elsewhere
            j = rng.randint(n)
            if j == i_star or j == partner:
                continue
            # Try: h[i_star] -= d, h[partner] += d (reduces the product)
            d = 0.01
            hi, hp, hj = h[i_star], h[partner], h[j]
            if hi - d < 0 or hp + d > 1:
                continue
            h[i_star] = hi - d
            h[partner] = hp + d
            # h[j] unchanged (sum preserved)
            c5_new = compute_c5(h, n)
            if c5_new < best - 1e-14:
                best = c5_new
                improved = True
            else:
                h[i_star] = hi
                h[partner] = hp
    
    return h, best

    corr = np.correlate(v, 1.0 - v, mode='full') * dx
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    exps = np.exp(s)
    logsum = m + np.log(np.sum(exps)) / alpha
    # p_k = exp(s_k) / sum(exp(s))
    p = exps / exps.sum()
    # grad = p^T @ grad_corr = sum_k p[k] * grad_corr_k
    grad = weighted_grad(p, v, n)
    return logsum, grad

Wait, but the soft-max includes a constant m = max(corr), so the gradient is:
d/dv dlogsumexp = p^T @ dcorr/dv (the m term cancels because it's a constant with respect to v at the point of evaluation)

Actually no, m = max_k corr_k changes with v. But in the soft-max, we're using:
f(v) = m + (1/alpha) log sum_k exp(alpha(corr_k - m))

df/dv = dm/dv + (1/alpha) * sum_k [alpha * exp(alpha(corr_k-m)) / sum_j exp(alpha(corr_j-m))] * dcorr_k/dv - m' * ...

This is getting messy. Let me just use the standard log-sum-exp formulation without subtracting m:
f(v) = (1/alpha) log sum_k exp(alpha * corr_k)
df/dv = sum_k [alpha * exp(alpha*corr_k) / sum_j exp(alpha*corr_j)] * dcorr_k/dv
= sum_k p_k * dcorr_k/dv
where p_k = exp(alpha*corr_k) / sum_j exp(alpha*corr_j)

For numerical stability, subtract max: p_k = exp(alpha*(corr_k - C)) / sum_j exp(alpha*(corr_j - C)) where C = max(corr)

This is clean. Let me use this.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(2000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def weighted_grad(p, h, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        pk = p[k]
        if pk < 1e-18:
            continue
        s = k - n + 1
        if s >= 0:
            if n - s > 0:
                grad[:n - s] += pk * (1.0 - h[s:])
            if s > 0 and n - s > 0:
                grad[s:] -= pk * h[:n - s]
        else:
            s2 = -s
            if n - s2 > 0:
                grad[s2:] += pk * (1.0 - h[:n - s2])
                grad[:n - s2] -= pk * h[s2:]
    return grad * dx

def softmax_obj_grad(v, n, alpha):
    corr = compute_corr(v, n)
    C = np.max(corr)
    s = alpha * (corr - C)
    s = np.clip(s, -500, 500)
    exps = np.exp(s)
    obj = C + np.log(np.sum(exps)) / alpha
    p = exps / exps.sum()
    grad = weighted_grad(p, v, n)
    return obj, grad

def slsqp_analytic(h, n, alpha, maxiter=800):
    def obj(v):
        o, _ = softmax_obj_grad(v, n, alpha)
        return o
    def grad(v):
        _, g = softmax_obj_grad(v, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = ([{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0,
              'jac': lambda v: np.ones(n)}],)
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons[0], options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(res.x, n)
    except Exception:
        return h

def fw_solve(p, h, n, n_iter=400, t_end=None):
    h = h.copy()
    target = n // 2
    for it in range(n_iter):
        if t_end is not None and time.time() > t_end:
            break
        grad = weighted_grad(p, h, n)
        idx = np.argsort(grad)[:target]
        v = np.zeros(n)
        v[idx] = 1.0
        t = 2.0 / (it + 2)
        h_new = (1.0 - t) * h + t * v
        h_new = np.clip(h_new, 0.0, 1.0)
        F_old = np.dot(p, compute_corr(h, n))
        F_new = np.dot(p, compute_corr(h_new, n))
        if F_new < F_old:
            h = h_new
        else:
            for tt in [0.5 * t, 0.25 * t, 0.1 * t]:
                ht = (1.0 - tt) * h + tt * v
                ht = np.clip(ht, 0.0, 1.0)
                if np.dot(p, compute_corr(ht, n)) < F_old:
                    h = ht
                    break
    return h

def minmax_fw(h, n, t_end, n_outer=40, T0=0.03, cooling=0.93):
    h = h.copy()
    n_corr = 2 * n - 1
    p = np.ones(n_corr) / n_corr
    T = T0
    for _ in range(n_outer):
        if time.time() > t_end:
            break
        h = fw_solve(p, h, n, 300, t_end)
        corr = compute_corr(h, n)
        mx = np.max(corr)
        T = max(T0 * 0.005, T * cooling)
        p_new = np.exp((corr - mx) / T)
        p = p_new / p_new.sum()
    return h

def targeted_descent(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        corr = compute_corr(h, n)
        mx = np.max(corr)
        active = np.where(corr > 0.92 * mx)[0]
        improved = False
        ntrials = 6 * n
        for _ in range(ntrials):
            if time.time() > t_end:
                break
            k = active[rng.randint(len(active))]
            s = k - n + 1
            if s >= 0 and n - s > 1:
                valid = np.arange(n - s)
                contrib = h[valid] * (1.0 - h[valid + s])
            elif s < 0 and n + s > 1:
                s2 = -s
                valid = np.arange(s2, n)
                contrib = h[valid] * (1.0 - h[valid - s2])
            else:
                continue
            i_st = valid[np.argmax(contrib)]
            if s >= 0:
                partner = i_st + s
            else:
                partner = i_st + s
            j = rng.randint(n)
            if j == i_st or j == partner:
                continue
            hi, hp, hj = h[i_st], h[partner], h[j]
            d = 0.008
            if hi < d or hp > 1.0 - d:
                continue
            h[i_st] = hi - d
            h[partner] = hp + d
            c5n = compute_c5(h, n)
            if c5n < best - 1e-14:
                best = c5n
                improved = True
            else:
                h[i_st] = hi
                h[partner] = hp
        if not improved and r > 30:
            break
    return h, best

def pair_descent(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.03, 0.015, 0.007, 0.003, 0.001]
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(10 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hid, hjd = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hid + delta, hjd - delta
                    if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hid, hjd
                if improved:
                    break
        if not improved and r > 50:
            break
    return h, best

def binary_push(h, n):
    """Push values toward 0 or 1, preserving sum"""
    h2 = h.copy()
    target = n / 2.0
    for _ in range(200):
        changed = False
        for i in range(n):
            if h2[i] > 0.5:
                surplus = h2[i] - 1.0
                h2[i] = 1.0
                deficit = 1.0 - h2[i]
                s = h2.sum() - target
                if s > 1e-12:
                    order = np.argsort(h2)
                    rem = s
                    for j in order:
                        if rem < 1e-12:
                            break
                        a = min(rem, h2[j])
                        h2[j] -= a
                        rem -= a
                    changed = True
            elif h2[i] < 0.5:
                h2[i] = 0.0
                s = h2.sum() - target
                if s < -1e-12:
                    order = np.argsort(1.0 - h2)
                    rem = -s
                    for j in order:
                        if rem < 1e-12:
                            break
                        a = min(rem, 1.0 - h2[j])
                        h2[j] += a
                        rem -= a
                    changed = True
        if not changed:
            break
    return repair(h2, n)
