import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:

        else:

    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]

    bnds = [(0.0, 1.0)] * n
    def _con_fun(x):
        return x.sum() - n * 0.5
    def _con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:

                       method='SLSQP', bounds=bnds, constraints=cons,

        return _repair(res.x, n)

For the coordinate optimization:
Given h with h[i] = v, and all other coordinates fixed, the objective is:
f(v) = max_k c[k](v) where c[k](v) is linear in v (since c[k] = sum_j h[j](1-h[j+k]) and h[i] appears linearly in each c[k]).

Wait, is c[k] linear in h[i]? Let's check:
c[k] = sum_{valid j} h[j](1-h[j+k])

If h[i] appears as h[j] where j=i: term is h[i](1-h[i+k]), linear in h[i]
If h[i] appears as h[j+k] where j+k=i, i.e., j=i-k: term is h[i-k](1-h[i]), linear in h[i]

So yes! c[k] is linear in h[i] for each k. Therefore f(v) = max_k (a_k * v + b_k) is the maximum of linear functions, which is convex in v. We can minimize it using ternary search or by finding where the subgradient crosses zero.

Actually, since f is piecewise linear and convex, the minimum is at a "kink" where two (or more) of the a_k*v+b_k cross. We can find it by:
1. For each k, compute a_k (coefficient of v) and b_k (constant term)
2. f(v) = max_k (a_k*v + b_k)
3. The minimum is at the v where the lower envelope of the upper envelope is minimized

For a 1D piecewise-linear convex function, the minimum is where 0 is in the subgradient:
min_k a_k ≤ 0 ≤ max_k a_k (at the kink points)

Actually, more precisely: at the minimum, the set of k's achieving the maximum must include at least one with a_k > 0 and one with a_k < 0 (or one with a_k = 0).

So the algorithm is:
1. For the current h with h[i] = v, compute all c[k] and identify which k's are at the max
2. If all active k's have a_k > 0, decrease v
3. If all active k's have a_k < 0, increase v
4. Otherwise, the minimum is at v (or at the crossing point)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            i1 = np.arange(cnt)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - x.shape[0] * 0.5

def _sum_jac(x):
    return np.ones(x.shape[0])

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    n = len(h)
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n * factor, endpoint=False)
    return np.interp(x_new, x_old, h)

def _topk_grad(h, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(c)[-topk:]
    topk_c = c[topk_idx]

    w_top /= w_top.sum()
    dx = 2.0 / n
    g = np.zeros(n)

            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g
