def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    nkp1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - nkp1
        # Term 1: (1 - x[i+lag]) for valid i
        i1, j1 = max(0, -lag), min(n, n - lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - x[i1+lag:j1+lag])

def _proj_grad(h, n, alpha, step, rng, t_end, max_iter=20000):
    """Projected gradient with momentum."""
    h_cur = h.copy()
    v = np.zeros(n)
    mu = 0.9

    best_h = h_cur.copy()
    lr = step
    for it in range(max_iter):

            break
        g = _gs(h_cur, alpha)

            lr *= 0.9

def _local_search(h, n, rng, t_end):

    try:
        res = minimize(_sf, h.copy(), jac=_gs, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj_grad(h, n, alpha, rng, t_end, max_iter=30000):
    h_cur = h.copy()
    v = np.zeros(n)
    mu = 0.85
    best_c5 = _eval(h_cur)
    best_h = h_cur.copy()
    lr = 0.015
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = _gs(h_cur, alpha)
        g -= g.mean()
        v = mu * v - lr * g
        v -= v.mean()
        h_new = _repair(h_cur + v, n)
        c5_new = _eval(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.005, 0.04)
        else:
            h_cur = best_h.copy()
            v *= 0.3
            lr *= 0.92
        if lr < 1e-9:
            break
    return best_h, best_c5

def _local_search(h, n, rng, t_end, start_scale=0.004):
    c5c = _eval(h)
    best_h = h.copy()
    scale = start_scale
    no_imp = 0
    fine_phase = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if fine_phase == 0 and no_imp > 400:
            fine_phase = 1
            scale = 1e-4
            no_imp = 0
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-17:
            c5c = c5n
            best_h = h.copy()
            no_imp = 0
            scale = min(scale * 1.03, 0.012)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 300:
                scale *= 0.75
                no_imp = 0
                if scale < 2e-5:
                    if fine_phase < 2:
                        fine_phase += 1
                        scale = 1e-5
                        no_imp = 0
                    else:
                        break
    return best_h, c5c

def _grad_local(h, n, rng, t_end, alpha=100000):
    c5c = _eval(h)
    best_h = h.copy()
    no_imp = 0
    for it in range(200000):
        if time.time() > t_end:
            break
        if no_imp > 100:
            no_imp = 0
            continue
        g = _gs(h, alpha)
        g -= g.mean()
        idx = np.argsort(g)
        i = idx[rng.randint(min(20, n))]
        jp = np.argsort(-g)
        j = jp[rng.randint(min(20, n))]
        if i == j:
            continue
        d = 0.005 * (g[i] - g[j])
        d = float(np.clip(d, -0.02, 0.02))
        if d == 0:
            continue
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-17:
            c5c = c5n
            best_h = h.copy()
            no_imp = 0
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
    return best_h, c5c

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _tl():
        return budget_s - (time.time() - t0)

    h_par = np.array(initial_h_values, dtype=float).copy()
    n_par = len(h_par)
    h_par = _repair(h_par, n_par)

    best_h = h_par.copy()
    best_c5 = _eval(h_par)
    best_n = n_par

    n_list = [n_par]
    for cand in [100, 120, 140, 180]:
        if cand != n_par and _tl() > 300:
            n_list.append(cand)

    inits = {n_par: h_par.copy()}
    for n in n_list:
        if n == n_par:
            continue
        if _tl() < 80:
            break
        inits[n] = _repair(_interp(h_par, n_par, n), n)

    # Phase 1: SLSQP with increasing alpha
    for n in sorted(n_list, key=lambda x: _eval(inits[x]) if x in inits else 1.0):
        if _tl() < 60:
            break
        h_cur = inits[n].copy()
        c5_cur = _eval(h_cur)
        if c5_cur > best_c5 + 0.03:
            continue
        for alpha in [1000, 10000, 100000, 500000]:
            if _tl() < 50:
                break
            maxit = 600 if n <= 140 else 350
            h_new = _slsqp(h_cur, n, alpha, maxiter=maxit)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-14:
                h_cur = h_new
                c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n

    # Phase 2: Projected gradient (momentum)
    if _tl() > 120:
        t_e = time.time() + min(100, _tl() - 30)
        h_new, c5_new = _proj_grad(best_h.copy(), best_n, 200000, rng, t_e)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Gradient-guided local search
    if _tl() > 60:
        t_e = time.time() + min(45, _tl() - 25)
        h_new, c5_new = _grad_local(best_h.copy(), best_n, rng, t_e, alpha=300000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Random local search
    if _tl() > 40:
        t_e = time.time() + min(30, _tl() - 20)
        h_new, c5_new = _local_search(best_h.copy(), best_n, rng, t_e)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Final SLSQP polish
    if _tl() > 25:
        h_new = _slsqp(best_h.copy(), best_n, 800000, maxiter=800)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Ultra-fine random perturbation
    if _tl() > 12:
        h = best_h.copy()
        c5c = _eval(h)
        sizes = [1e-6, 2e-6, 5e-6, 1e-7, 3e-7, 1e-8]
        for _ in range(200000):
            if time.time() > t0 + budget_s - 3:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * rng.choice(sizes).item()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5c
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
