from scipy.optimize import minimize, linprog

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

    h0 = _repair(np.array(initial_h_values, dtype=float), n0)
    bounds0 = [(0.0, 1.0)] * n0

    h0 = _lbb(h0, n0, bounds0, t0, budget_s, 0.12)
    best_h, best_c5 = _local_search(h0, n0, dx0, rng, t0, budget_s * 0.40)

    for _hop in range(8):
        if not _locate(t0, budget_s, 0.65):
            break
        sz = 0.015 * (1.0 + 0.5 * _hop)
        h_hop = best_h + rng.normal(0, sz, n0)
        h_hop = _repair(h_hop, n0)
        hh, ch = _local_search(h_hop, n0, dx0, rng, t0, budget_s * 0.65)
        if ch < best_c5:
            best_h, best_c5 = hh, ch

    n1 = 100
    dx1 = 2.0 / n1
    bounds1 = [(0.0, 1.0)] * n1
    x_f = np.linspace(0, 2, n0, endpoint=False)
    x_c = np.linspace(0, 2, n1, endpoint=False)
    h1 = _repair(np.interp(x_c, x_f, best_h), n1)
    h1 = _lbb(h1, n1, bounds1, t0, budget_s, 0.78)
    h1, c1 = _local_search(h1, n1, dx1, rng, t0, budget_s * 0.90)
    if c1 < best_c5:
        best_h, best_c5, n0, dx0 = h1, c1, n1, dx1
        bounds0 = bounds1

    n2 = 80
    dx2 = 2.0 / n2
    bounds2 = [(0.0, 1.0)] * n2
    x_c2 = np.linspace(0, 2, n2, endpoint=False)
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    h2 = _repair(np.interp(x_c2, x_best, best_h), n2)
    if _locate(t0, budget_s, 0.82):
        h2 = _lbb(h2, n2, bounds2, t0, budget_s, 0.86)
        h2, c2 = _local_search(h2, n2, dx2, rng, t0, budget_s * 0.93)
        if c2 < best_c5:
            best_h, best_c5, n0, dx0 = h2, c2, n2, dx2

    h_fin = best_h.copy()
    c_fin = best_c5
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    for _ in range(100000):
        if not _locate(t0, budget_s, 0.995):
            break
        i, j = rng.randint(n_f), rng.randint(n_f)
        if i == j:
            continue
        d = rng.uniform(-0.004, 0.004)
        ni, nj = h_fin[i] + d, h_fin[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h_fin[i], h_fin[j] = ni, nj
            c_try = _c5(h_fin, dx_f)
            if c_try < c_fin:
                c_fin = c_try
                best_c5 = c_try
                best_h = h_fin.copy()
            else:
                h_fin[i], h_fin[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
