        g[M:] -= h[:n - M]
    return g

def _topk_grad(h, k, lam):
    c = _corr(h)
    n = len(h)
    idx = np.argpartition(c, -k)[-k:]
    g = np.zeros(n)
    for ki in idx:
        L = ki - (n - 1)
        g += _lag_grad(h, L, n)
    g /= k
    g += 2.0 * lam * (np.sum(h) - n / 2.0)
    return g

def _softmax_obj(h, alpha, lam):
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(h) - len(h) / 2.0) ** 2

def _softmax_grad(h, alpha, lam):
    c = _corr(h)
    n = len(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        g += pk * _lag_grad(h, L, n)
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(h) - n / 2.0)
    return g

def _momentum_descent(h, n, dx, get_time, k_top=8, max_iter=6000):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    step = 0.025
    mom = np.zeros(n)
    beta = 0.65
    for it in range(max_iter):
        if get_time() < 8:
            break
        g = _topk_grad(h, k_top, 0.0)
        g_c = g - g.mean()
        ng = np.linalg.norm(g_c)
        if ng < 1e-14:
            break
        g_c /= ng
        mom = beta * mom + (1.0 - beta) * g_c
        h_new = _repair(h - step * mom, n)
        c_new = _c5(h_new, dx)
        if c_new < bc5 - 1e-15:
            bc5 = c_new
            bh = h_new.copy()
            h = h_new
            step = min(step * 1.04, 0.06)
        else:
            h = h_new
            step *= 0.88
            if step < 1e-9:
                h = bh.copy()
                step = 0.025
                mom *= 0.3
    return bh, bc5

def _local_search(h, n, dx, get_time, n_iter=25000):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    c5c = bc5
    rng = np.random.RandomState(123)
    stalls = 0
    for it in range(n_iter):
        if get_time() < 5:
            break
        g = _topk_grad(h, 5, 0.0)
        order = np.argsort(np.abs(g))[::-1][:25]
        found = False
        for ii in range(min(12, len(order))):
            i = order[ii]
            for jj in range(max(0, ii - 8), min(ii + 8, len(order))):
                if jj == ii:
                    continue
                j = order[jj]
                best_d = 0.0
                best_c = c5c
                for d in np.array([0.001, 0.003, 0.007, 0.014, -0.001, -0.003, -0.007, -0.014]):
                    ni, nj = h[i] + d, h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best_c:
                            best_c = c
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    found = True
                    if c5c < bc5:
                        bc5 = c5c
                        bh = h.copy()
                    break
            if found:
                stalls = 0
                break
        if not found:
            stalls += 1
            if stalls > 300:
                i, j = rng.randint(n), rng.randint(n)
                if i != j:
                    d = rng.uniform(-0.008, 0.008)
                    ni, nj = h[i] + d, h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        h = _repair(h, n)
                        c5c = _c5(h, dx)
                stalls = 0
                if stalls > 1500:
                    break
    return bh, bc5

def _interp_h(src, src_n, dst_n):
    sg = np.linspace(0, 2, src_n, endpoint=False)
    dg = np.linspace(0, 2, dst_n, endpoint=False)
    return _repair(np.interp(dg, sg, src), dst_n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    h0 = _repair(np.array(initial_h_values, dtype=float), n0)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx0)

    # Phase A: L-BFGS softmax at native resolution
    if rem() > 250:
        bounds = [(0.0, 1.0)] * n0
        for alpha in [3000, 15000, 80000, 400000]:
            if rem() < 150:
                break
            try:
                res = minimize(
                    _softmax_obj, best_h, args=(alpha, 40.0),
                    jac=_softmax_grad, method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 600, 'ftol': 1e-19, 'gtol': 1e-15})
                hc = _repair(res.x, n0)
                cv = _c5(hc, dx0)
                if cv < best_c5:
                    best_c5 = cv
                    best_h = hc
            except Exception:
                pass

    # Phase B: Top-k momentum descent at native resolution
    if rem() > 200:
        hB, cB = _momentum_descent(best_h, n0, dx0, lambda: rem() - 120, k_top=7)
        if cB < best_c5:
            best_c5 = cB
            best_h = hB

    # Phase C: Local search at native resolution
    if rem() > 130:
        hC, cC = _local_search(best_h, n0, dx0, lambda: rem() - 70, n_iter=20000)
        if cC < best_c5:
            best_c5 = cC
            best_h = hC

    # Phase D: Coarse n=60 for global structure
    if rem() > 180:
        nc = 60
        dcc = 2.0 / nc
        hc = _interp_h(best_h, len(best_h), nc)
        bounds_c = [(0.0, 1.0)] * nc
        for alpha in [5000, 30000, 150000]:
            if rem() < 100:
                break
            try:
                res = minimize(
                    _softmax_obj, hc, args=(alpha, 40.0),
                    jac=_softmax_grad, method='L-BFGS-B', bounds=bounds_c,
                    options={'maxiter': 500, 'ftol': 1e-19})
                hc2 = _repair(res.x, nc)
                cv = _c5(hc2, dcc)
                if cv < best_c5:
                    best_c5 = cv
                    best_h = hc2
                    hc = hc2
            except Exception:
                pass
        if rem() > 80:
            hD, cD = _momentum_descent(hc, nc, dcc, lambda: rem() - 40, k_top=5)
            if cD < best_c5:
                best_c5 = cD
                best_h = hD
        if rem() > 35:
            hD2, cD2 = _local_search(hc, nc, dcc, lambda: rem() - 15, n_iter=10000)
            if cD2 < best_c5:
                best_c5 = cD2
                best_h = hD2

    # Phase E: n=100 refinement
    if rem() > 100:
        ne = 100
        dxe = 2.0 / ne
        he = _interp_h(best_h, len(best_h), ne)
        bounds_e = [(0.0, 1.0)] * ne
        for alpha in [10000, 60000]:
            if rem() < 55:
                break
            try:
                res = minimize(
                    _softmax_obj, he, args=(alpha, 40.0),
                    jac=_softmax_grad, method='L-BFGS-B', bounds=bounds_e,
                    options={'maxiter': 400, 'ftol': 1e-19})
                he2 = _repair(res.x, ne)
                cv = _c5(he2, dxe)
                if cv < best_c5:
                    best_c5 = cv
                    best_h = he2
                    he = he2
            except Exception:
                pass
        if rem() > 45:
            hE, cE = _momentum_descent(he, ne, dxe, lambda: rem() - 20, k_top=8)
            if cE < best_c5:
                best_c5 = cE
                best_h = hE
        if rem() > 15:
            hE2, cE2 = _local_search(he, ne, dxe, lambda: rem() - 5, n_iter=8000)
            if cE2 < best_c5:
                best_c5 = cE2
                best_h = hE2

    # Phase F: Final micro-polish
    if rem() > 8:
        nf = len(best_h)
        dxf = 2.0 / nf
        hf = best_h.copy()
        c5f = _c5(hf, dxf)
        tiny = np.array([0.0002, 0.0006, 0.0015, -0.0002, -0.0006, -0.0015])
        n_pol = min(20000, int((rem() - 3) * 3000))
        for _ in range(n_pol):
            if rem() < 2:
                break
            i, j = rng.randint(nf), rng.randint(nf)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = hf[i] + d, hf[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                hf[i], hf[j] = ni, nj
                c = _c5(hf, dxf)
                if c < c5f:
                    c5f = c
                else:
                    hf[i], hf[j] = ni - d, nj + d
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
