For np.correlate(h, 1-h, 'full') at index k:
Let shift = k - (n-1) (the actual spatial shift)
corr[k] = sum_i h[i + shift] * (1 - h[i]) for valid i (when shift >= 0)

The gradient d(corr[k])/dh[j]:
For shift >= 0:
- If j is in the range of i+shift: term is (1-h[i]) where i = j-shift
- If j is in the range of i: term is -h[i+shift] where i = j
So: grad[j] = (1-h[j-shift]) - h[j+shift] (with appropriate bounds)

    else:

        for i in range(n - s):

            grad[i+s] += -h[i]

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        d = target - s
        if abs(d) < 1e-12:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-12:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def soft_max_val(v, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    def obj(v):
        return soft_max_val(v, alpha) + 1e8 * (v.sum() - n / 2.0) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
