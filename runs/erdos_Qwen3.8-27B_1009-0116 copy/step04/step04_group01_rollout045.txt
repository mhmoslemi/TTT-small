def _to_binary(h, n):
    """Project continuous h to binary with exactly n/2 ones."""
    target = n // 2
    # Round: take top n/2 values as 1, rest as 0
    # Tie-break by index
    h_copy = h.copy()
    # Add tiny noise to break ties

    top = np.argsort(scores)[-target:]
    b = np.zeros(n)
    b[top] = 1.0

def _correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _local_search_binary(a, max_passes=50):
    """Local search: swap 1 and 0 to reduce max correlation."""
    n = len(a)
    a = a.astype(float).copy()
    c = _correlation(a)
    cmax = c.max()
    
    for _pass in range(max_passes):
        improved = False
        ones = np.where(a == 1.0)[0]
        zeros = np.where(a == 0.0)[0]
        
        # Try all swaps (or a subset if too many)
        # For n=140: 70*70 = 4900 swaps, each O(n)
        for i in ones:
            for j in zeros:
                # Compute Δc
                # shift = k - (n-1)
                # Δc[k] = a[i-shift] + a[i+shift] - a[j-shift] - a[j+shift]
                # Let's compute for all k at once
                shifts = np.arange(2*n-1) - (n-1)  # range from -(n-1) to (n-1)
                
                # For position i: a[i-shift] is valid when 0 <= i-shift < n
                # a[i+shift] is valid when 0 <= i+shift < n
                ia1 = i - shifts  # valid where 0 <= ia1 < n
                ia2 = i + shifts  # valid where 0 <= ia2 < n
                ja1 = j - shifts
                ja2 = j + shifts
                
                contrib_i = np.where((ia1 >= 0) & (ia1 < n), a[ia1[((ia1 >= 0) & (ia1 < n))]], 0) + \
                           np.where((ia2 >= 0) & (ia2 < n), a[ia2[((ia2 >= 0) & (ia2 < n))]], 0)
                contrib_j = np.where((ja1 >= 0) & (ja1 < n), a[ja1[((ja1 >= 0) & (ja1 < n))]], 0) + \
                           np.where((ja2 >= 0) & (ja2 < n), a[ja2[((ja2 >= 0) & (ja2 < n))]], 0)
                
                dc = contrib_i - contrib_j
                new_c = c + dc
                new_max = new_c.max()
                
                if new_max < cmax - 1e-15:
                    # Accept
                    a[i] = 0.0
                    a[j] = 1.0
                    c = new_c
                    cmax = new_max
                    improved = True
                    ones = np.where(a == 1.0)[0]
                    zeros = np.where(a == 0.0)[0]
                    break
            if improved:
                break
        if not improved:
            break
    
    return a, cmax * 2.0 / n

def _periodic_pattern(n, p, pattern):
    """Create a periodic pattern of length n from a period of length p."""
    reps = n // p
    # pattern has exactly p/2 ones
    result = np.tile(pattern, reps)
    if len(result) < n:
        result = np.concatenate([result, np.zeros(n - len(result))])
    return result[:n]
