    return float(np.max(_corr(h)) * dx)

    n = len(h)
    c = _corr(h)
    med = np.median(c)
    w = np.exp(beta * (c - med))
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-8:
            continue
        s = k - (n - 1)
        if s >= 0:
            g1 = np.zeros(n)
            g1[s:] = 1.0 - h[:n-s]
            g2 = np.zeros(n)
            g2[:n-s] = -h[s:n]

Let me think about what "targeted" means. For the worst lag k*, the overlap is:
c_{k*} = sum_i h[i+k*] * (1-h[i])

This is a bilinear form. To reduce it, we want to "shift" mass so that high values of h don't align with low values at the shifted position.

For a more continuous h, the idea is similar: we want h(x) and h(x+k) to be both high or both low for most x, at the worst k.

    c = _corr(h)
    top = np.argsort(c)[::-1][:5]
    w = np.exp(60.0 * (c[top] - c[top].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top):
        s = abs(int(ki - (n - 1)))
        if s == 0:
            continue
        m = n - s
        g = np.zeros(n)
        g[s:n] += 1.0 - h[:m]
        g[:m] -= h[s:n]
        grad += wi * g
    grad -= grad.mean()
    return grad

def _sgd(h, dx, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr = 0.005
    for _ in range(10000000):
        if time.time() > t_end:
            break
        g = _subgrad(h)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = _repair(np.clip(h - lr * g / gn, 0, 1), tgt)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new
            h = h_new
            lr = min(lr * 1.01, 0.03)
        else:
            lr *= 0.9
            if lr < 1e-6:
                break
    return h, best

def _slsqp(h, dx, alpha, maxiter=400):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _slsqp_multi(h, dx, t_end, alphas, maxiter=400):
    best_h, best_c = h, _c5(h, dx)
    for a in alphas:
        if time.time() > t_end:
            break
        hc, c = _slsqp(best_h, dx, a, maxiter)
        if c < best_c:
            best_h, best_c = hc, c
    return best_h, best_c

def _targeted(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:4]
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            for _ in range(6):
                a = rng.randint(m)
                i1, i2 = a + s, a
                d = rng.uniform(0.002, 0.02)
                h[i1] -= d; h[i2] += d
                if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                    else:
                        h[i1] += d; h[i2] -= d
                else:
                    h[i1] += d; h[i2] -= d
        for _ in range(4):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.012, 0.012)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _fine(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.004, 0.002, 0.001, 5e-04, 2e-04, 1e-04, 5e-05, 2e-05]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.04 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 5), replace=False)
        hp[idx] += 0.08 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.03 * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(2, n // 8), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
                len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    # P1: Subgradient descent
    h, c = _sgd(best_h, dx, t0 + 110)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P2: SLSQP multi-alpha
    h, c = _slsqp_multi(best_h, dx, t0 + 190, [300, 1500, 6000, 20000], 300)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P3: Targeted worst-lag search
    h, c = _targeted(best_h, dx, rng, t0 + 340)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P4: Higher-alpha SLSQP
    h, c = _slsqp_multi(best_h, dx, t0 + 400, [8000, 30000, 100000], 300)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P5: Multi-start with diverse perturbations
    t_end5 = t0 + 600
    for r in range(14):
        rem = budget_s - (time.time() - t0)
        if time.time() > t_end5 or rem < 220:
            break
        hp = _perturb(best_h, rng, r % 4)
        hp, c = _sgd(hp, dx, min(t_end5 - 5, time.time() + 14))
        hp, c = _slsqp(hp, dx, 10000, 250)
        if c < best_c5:
            best_h, best_c5 = hp, c
            hp, c = _targeted(hp, dx, rng, min(t_end5 - 2, time.time() + 8))
            if c < best_c5:
                best_h, best_c5 = hp, c

    # P6: Try 2x resolution
    t_end6 = t0 + 800
    if budget_s - (time.time() - t0) > 230:
        n2, dx2 = 2 * n, 1.0 / n
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        h2, c2 = _sgd(h2, dx2, min(t_end6 - 70, time.time() + 70))
        h2, c2 = _slsqp_multi(h2, dx2, min(t_end6 - 25, time.time() + 70),
                              [1000, 5000, 20000], 250)
        h2, c2 = _targeted(h2, dx2, rng, t_end6)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # P7: Targeted search on best
    nb, dxb = best_n, 2.0 / best_n
    best_h, best_c5 = _targeted(best_h, dxb, rng, t0 + 900)

    # P8: Final SLSQP
    if time.time() < t0 + 950:
        h, c = _slsqp_multi(best_h, dxb, t0 + 950, [20000, 80000, 300000], 200)
        if c < best_c5:
            best_h, best_c5 = h, c

    # P9: Fine local search
    best_h, best_c5 = _fine(best_h, dxb, rng, t0 + 1070)

    # P10: Final SLSQP + micro-search
    if budget_s - (time.time() - t0) > 12:
        h, c = _slsqp(best_h, dxb, 100000, 200)
        if c < best_c5:
            best_h, best_c5 = h, c
    if budget_s - (time.time() - t0) > 3:
        best_h, best_c5 = _fine(best_h, dxb, rng, t0 + budget_s - 1)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
