Now, an important observation: the gradient has a very simple structure. For each shift k, the gradient is just 1-2h evaluated at shifted indices. This means the optimization landscape is quite structured.

Furthermore, note that:
sum_m grad[m] = sum_{j: valid} (1-2h[j]) = num_valid - 2*sum(j: valid) h[j]

The sum of the gradient is NOT zero in general, which means the gradient step will change the integral constraint. That's why we need the repair step.

But for the "central" shifts (k ≈ n-1), the valid range is the full [0,n), and:
sum_m grad[m] = n - 2*sum h[j] = n - 2*(n/2) = 0

So for the central shift k=n-1, the gradient is zero-sum. For edge shifts, it's not.

This is important: the central shift k=n-1 (which corresponds to lag=0, i.e., the "self-overlap") has a zero-sum gradient. The edge shifts don't.

Now let me think about the optimization more carefully.

The C5 objective is:
C5(h) = dx * max_k R_k(h)

where R_k = sum_j h[k-j](1-h[j]) for valid j.

This is the maximum of 2n-1 convex functions (each R_k is convex in h when... wait, R_k is quadratic, not convex in general).

Actually, R_k = sum h[k-j](1-h[j]) = sum h[k-j] - sum h[k-j]h[j]
The first term is linear, the second is a (negative) quadratic form. So R_k is not convex.

The Hessian of R_k w.r.t. h is:
The quadratic part is -sum h[k-j]h[j], which is -h^T A_k h where A_k is a permutation-like matrix.
The Hessian is -2*A_k, which is indefinite (eigenvalues ±2).

So each R_k is neither convex nor concave. The max of non-convex functions is non-convex. The optimization is genuinely hard.

However, for the purposes of gradient-based methods, the key is to have a good gradient, which we now have analytically.

2. **Phase 2**: Cyclic projected subgradient on the true max (use analytical gradient)
3. **Phase 3**: Adaptive coordinate descent
4. **Phase 4**: Try other resolutions
5. **Phase 5**: Final polish

Let me also add a key improvement: after the SLSQP phase, I'll do a "max-equalization" step. The idea:
- Find all shifts k where R_k is within epsilon of the max
- Compute the average gradient over these shifts
- Take a step in that direction
- This tends to flatten the top of the correlation, reducing the max

This is essentially the subgradient of the max function (when the max is achieved at multiple points, the subgradient is the convex hull of the individual gradients).

For the subgradient: when the max is achieved at k*, a subgradient is ∇R_k*. When multiple k achieve the max, any convex combination is a subgradient. The "average" is a natural choice.

```python
def cyclic_subgrad(h, n, t_end, top_m=5, lr_init=0.005):
    h = h.copy()
    best = compute_c5(h, n)
    lr = lr_init
    stall = 0
    
    while time.time() < t_end:
        c_all = compute_all_c(h, n)
        c_max = np.max(c_all)
        top_idx = np.argsort(-c_all)[:top_m]
        
        # Weighted average gradient
        weights = np.zeros(2*n-1)
        for idx in top_idx:
            weights[idx] = 1.0  # uniform weight for simplicity
        w_sum = weights.sum()
        
        avg_grad = np.zeros(n)
        for k in top_idx:
            avg_grad += grad_shift(h, n, k)
        avg_grad /= w_sum
        
        # Projected gradient step
        g_norm = np.linalg.norm(avg_grad)
        if g_norm < 1e-14:
            break
        avg_grad = avg_grad / g_norm
        
        # Try a few step sizes
        best_step = None
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            h_try = h - trial_lr * avg_grad
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = repair(h_try, n)
            c5_try = compute_c5(h_try, n)
            if c5_try < best - 1e-15:
                best_step = (h_try, c5_try, trial_lr)
                break
        
        if best_step is not None:
            h, c5_try, lr_new = best_step
            best = c5_try
            lr = min(lr * 1.5, 0.1)
            stall = 0
        else:
            lr *= 0.3
            stall += 1
            if stall > 50:
                # Perturb and restart
                h = h + 0.02 * np.random.randn(n)
                h = repair(h, n)
                lr = lr_init
                stall = 0
    
    return h, best
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if abs(d) < 1e-13:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def grad_shift(h, n, k):
    grad = np.zeros(n)
    m_lo = max(0, k - n + 1)
    m_hi = min(n - 1, k)
    j_start = k - m_hi
    j_end = k - m_lo
    grad[m_lo:m_hi + 1] = 1.0 - 2.0 * h[j_start:j_end + 1]
    return grad

def soft_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def soft_grad(x, alpha):
```

```python
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def grad_shift(h, n, k):
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return g
    js = k - hi
    je = k - lo
    g[lo:hi + 1] = 1.0 - 2.0 * h[js:je + 1]
    return g

def _soft_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -280, 280)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _soft_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -280, 280)
    e = np.exp(s)
    w = e / np.sum(e)
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-12:
            g += w[k] * grad_shift(x, n, k)
    return g

def _con_fun(x):
    return x.sum() - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def do_slsqp(h, n, alpha, maxiter=400):
    bds = [(0.0, 1.0)] * n
    cns = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        r = minimize(_soft_obj, h, args=(alpha,), jac=_soft_grad,
                     method='SLSQP', bounds=bds, constraints=cns,
                     options={'maxiter': maxiter, 'ftol': 1e-20})
        return repair(r.x, n)
    except Exception:
        return h

def cyclic_subgrad(h, n, t_end, rng, top_m=8, lr0=0.003):

    best = compute_c5(h, n)
    lr = lr0
    stall = 0
    it = 0

        c_raw = compute_all_c_raw(h, n)
        thr = np.max(c_raw)
        ok = np.where(c_raw >= thr - 0.05)[0]
        if len(ok) > top_m:

        if len(ok) == 0:

        it += 1
        if it % 3 == 1:
            ag = grad_shift(h, n, int(np.argmax(c_raw)))
        else:
            ag = np.zeros(n)
            for k in ok:
                ag += grad_shift(h, n, int(k))
            ag /= len(ok)
        gn = np.linalg.norm(ag)
        if gn < 1e-14:
            break
        ag /= gn
        imp = False

def coord_descent(h, n, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    ni = 0
    while time.time() < t_end:
        imp = False
        for _ in range(6 * n):
            if time.time() > t_end:
                return h, best
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hi, hj = h[i], h[j]
            for d in steps:
                for sg in (1, -1):
                    dl = sg * d
                    a, b2 = hi + dl, hj - dl
                    if a < -1e-9 or a > 1 + 1e-9 or b2 < -1e-9 or b2 > 1 + 1e-9:
                        continue
                    h[i], h[j] = a, b2
                    v = compute_c5(h, n)
                    if v < best - 1e-15:
                        best, imp = v, True
                        break
                    h[i], h[j] = hi, hj
                if imp:
                    break
        if not imp:
            ni += 1
            if ni > 25:
                break
            h = repair(h + 0.005 * rng.randn(n), n)
    return h, best
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_c_raw(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def grad_shift(h, n, k):
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return g
    js = k - hi
    je = k - lo
    g[lo:hi + 1] = 1.0 - 2.0 * h[js:je + 1]
    return g

def _soft_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -280, 280)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _soft_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -280, 280)
    e = np.exp(s)
    w = e / np.sum(e)
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-12:
            g += w[k] * grad_shift(x, n, k)
    return g

def _con_fun(x):
    return x.sum() - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def do_slsqp(h, n, alpha, maxiter=400):
    bds = [(0.0, 1.0)] * n
    cns = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        r = minimize(_soft_obj, h, args=(alpha,), jac=_soft_grad,
                     method='SLSQP', bounds=bds, constraints=cns,
                     options={'maxiter': maxiter, 'ftol': 1e-20})
        return repair(r.x, n)
    except Exception:
        return h

    while time.time() < t_end:
        c_raw = compute_all_c_raw(h, n)
        thr = np.max(c_raw)
        ok = np.where(c_raw >= thr - 0.05)[0]
        if len(ok) > top_m:
            ok = np.argsort(-c_raw)[:top_m]
        if len(ok) == 0:
            ok = np.array([np.argmax(c_raw)])
        it += 1
        if it % 3 == 1:
            ag = grad_shift(h, n, int(np.argmax(c_raw)))
        else:
            ag = np.zeros(n)
            for k in ok:
                ag += grad_shift(h, n, int(k))
            ag /= len(ok)
        gn = np.linalg.norm(ag)
        if gn < 1e-14:
            break
        ag /= gn
        imp = False
        for f in (1.0, 0.5, 0.25, 0.1, 0.05):
            ht = np.clip(h - lr * f * ag, 0.0, 1.0)
            ht = repair(ht, n)
            val = compute_c5(ht, n)
            if val < best - 1e-15:
                h, best, imp = ht, val, True
                break
        if imp:
            lr = min(lr * 1.3, 0.05)
            stall = 0
        else:
            lr *= 0.35
            stall += 1
            if stall > 40:
                h = repair(h + 0.02 * rng.randn(n), n)
                lr = lr0
                stall = 0
    return h, best

def coord_descent(h, n, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    ni = 0
    while time.time() < t_end:
        imp = False
        for _ in range(6 * n):
            if time.time() > t_end:
                return h, best
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hi, hj = h[i], h[j]
            for d in steps:
                for sg in (1, -1):
                    dl = sg * d
                    a, b2 = hi + dl, hj - dl
                    if a < -1e-9 or a > 1 + 1e-9 or b2 < -1e-9 or b2 > 1 + 1e-9:
                        continue
                    h[i], h[j] = a, b2
                    v = compute_c5(h, n)
                    if v < best - 1e-15:
                        best, imp = v, True
                        break
                    h[i], h[j] = hi, hj
                if imp:
                    break
        if not imp:
            ni += 1
            if ni > 25:
                break
            h = repair(h + 0.005 * rng.randn(n), n)
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 77)
    best_h, best_c5, best_n = None, np.inf, 64
    try:
        ih = np.array(initial_h_values, dtype=float).copy()
        best_n = len(ih)
        best_h = repair(ih, best_n)
        best_c5 = compute_c5(best_h, best_n)
    except NameError:
        best_h = repair(np.full(64, 0.5), 64)
        best_n = 64
        best_c5 = compute_c5(best_h, best_n)
    n = best_n
    h = best_h.copy()
    for a in (2000, 10000, 50000):
        if time.time() - t0 > budget_s * 0.15:
            break
        hn = do_slsqp(h, n, a, 300)
        v = compute_c5(hn, n)
        if v < best_c5 - 1e-14:
            best_h, best_c5, h = hn.copy(), v, hn
        elif v < compute_c5(h, n) - 1e-14:
            h = hn
    if time.time() - t0 < budget_s * 0.50:
        hn, v = cyclic_subgrad(h, n, t0 + budget_s * 0.50, rng)
        if v < best_c5 - 1e-14:
            best_h, best_c5, h = hn.copy(), v, hn
        else:
            h = hn
    for a in (100000, 500000, 2000000):
        if time.time() - t0 > budget_s * 0.65:
            break
        hn = do_slsqp(h, n, a, 500)
        v = compute_c5(hn, n)
        if v < best_c5 - 1e-14:
            best_h, best_c5, h = hn.copy(), v, hn
        elif v < compute_c5(h, n) - 1e-14:
            h = hn
    if time.time() - t0 < budget_s * 0.80:
        hn, v = coord_descent(h, n, t0 + budget_s * 0.80, rng)
        if v < best_c5 - 1e-14:
            best_h, best_c5, h = hn.copy(), v, hn
        else:
            h = hn
    if time.time() - t0 < budget_s * 0.95:
        for na in (80, 96, 128, 64):
            if time.time() - t0 > budget_s * 0.95:
                break
            if na == best_n:
                continue
            xo = np.arange(best_n, dtype=float)
            xn = np.arange(na, dtype=float)
            hs = repair(np.interp(xn, xo, best_h), na)
            ha, va = cyclic_subgrad(hs, na, t0 + budget_s * 0.95, rng, top_m=6, lr0=0.002)
            ha2, va2 = coord_descent(ha, na, t0 + budget_s * 0.95, rng)
            if va2 < va:
                ha, va = ha2, va2
            if va < best_c5 - 1e-14:
                best_h, best_c5, best_n = ha.copy(), va, na
    if best_h is not None and time.time() - t0 < budget_s * 0.995:
        hn = do_slsqp(best_h, best_n, 1000000, 800)
        v = compute_c5(hn, best_n)
        if v < best_c5 - 1e-14:
            best_h, best_c5 = hn.copy(), v
        hn2, v2 = coord_descent(best_h, best_n, t0 + budget_s * 0.998, rng)
        if v2 < best_c5 - 1e-14:
            best_h, best_c5 = hn2.copy(), v2
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
```
