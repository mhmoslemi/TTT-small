        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g

def _subgrad_opt(h, n, dx, time_fn, max_iter=5000, n_top_frac=0.2):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    step = 0.025
    n_pts = n
    for it in range(max_iter):
        if time_fn() < 3:
            break
        c = _corr(h)
        cm = np.max(c)
        spread = cm - np.min(c)
        if spread < 1e-12:
            break
        ntop = max(5, int(n_top_frac * (2 * n_pts - 1)))
        active = np.argsort(c)[::-1][:ntop]
        g = _grad_lags(h, active)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-14:
            break
        g /= ng
        improved = False
        for s_try in [step, step * 0.6, step * 0.3, step * 0.1, step * 0.05]:
            h_new = _repair(h - s_try * g, n_pts)
            c_new = _c5(h_new, dx)
            if c_new < _c5(h, dx) - 1e-15:
                h = h_new
                improved = True
                if c_new < bc5:
                    bc5 = c_new
                    bh = h.copy()
                break
        if improved:
            step = min(step * 1.02, 0.06)
        else:
            step *= 0.55
            if step < 1e-8:
                break
    return bh, bc5

def _worst_lag_cd(h, n, dx, time_fn, iters=3000, seed=42):
    rng = np.random.RandomState(seed)
    bh = h.copy()
    bc5 = _c5(bh, dx)
    c5c = bc5
    step = 0.004
    n_pts = n
    for it in range(iters):
        if time_fn() < 3:
            break
        c = _corr(h)
        worst_k = np.argmax(c)
        L = worst_k - (n_pts - 1)
        M = abs(L)
        if M == 0:
            idx = rng.randint(n_pts)
            d = step if h[idx] < 0.5 else -step
            h[idx] += d
            if h[idx] < 0:
                h[idx] = 0
            if h[idx] > 1:
                h[idx] = 1
            c_new = _c5(h, dx)
            if c_new < c5c:
                c5c = c_new
                h = _repair(h, n_pts)
                if c5c < bc5:
                    bc5 = c5c
                    bh = h.copy()
            else:
                h[idx] -= d
        else:
            idx = rng.randint(0, n_pts - M) if L > 0 else rng.randint(M, n_pts)
            d = step
            if L > 0:
                if h[idx] >= d and h[idx + M] <= 1 - d:
                    h[idx] -= d
                    h[idx + M] += d
                    c_new = _c5(h, dx)
                    if c_new < c5c:
                        c5c = c_new
                        h = _repair(h, n_pts)
                        if c5c < bc5:
                            bc5 = c5c
                            bh = h.copy()
                    else:
                        h[idx] += d
                        h[idx + M] -= d
            else:
                idx2 = idx - M
                if h[idx] >= d and h[idx2] <= 1 - d:
                    h[idx] -= d
                    h[idx2] += d
                    c_new = _c5(h, dx)
                    if c_new < c5c:
                        c5c = c_new
                        h = _repair(h, n_pts)
                        if c5c < bc5:
                            bc5 = c5c
                            bh = h.copy()
                    else:
                        h[idx] += d
                        h[idx2] -= d
    return bh, bc5

def _fine_cd(h, n, dx, time_fn, iters=8000, seed=123):
    rng = np.random.RandomState(seed)
    bh = h.copy()
    bc5 = _c5(bh, dx)
    c5c = bc5
    n_pts = n
    steps = np.array([0.0003, 0.0008, 0.0015, 0.003, 0.006,
                      -0.0003, -0.0008, -0.0015, -0.003, -0.006])
    for it in range(iters):
        if time_fn() < 2:
            break
        i = rng.randint(n_pts)
        j = rng.randint(n_pts)
        if i == j:
            continue
        d = steps[rng.randint(len(steps))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c:
                c5c = c
                if c5c < bc5:
                    bc5 = c5c
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                h = _repair(h, n_pts)
    return bh, bc5

def _make_sine(n, freqs):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for f in freqs:
        h += 0.5 + 0.5 * np.sin(2 * np.pi * f * x / 2.0)
    h /= len(freqs)
    return _repair(h, n)

def _make_sawtooth(n, period):
    h = np.zeros(n)
    for i in range(n):
        pos = (i % period) / period
        h[i] = 1.0 - pos
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_fn(offset):
        return budget_s - elapsed() - offset

    n_init = len(initial_h_values)
    dx_init = 2.0 / n_init
    h0 = _repair(np.array(initial_h_values, dtype=float), n_init)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx_init)
    best_n = n_init
    best_dx = dx_init

    # Phase 1: Aggressive subgradient on initial
    for ntop_f in [0.15, 0.25, 0.35]:
        if time_fn(100) < 60:
            break
        ht = h0.copy()
        ht, ct = _subgrad_opt(ht, n_init, dx_init, time_fn(100),
                               max_iter=4000, n_top_frac=ntop_f)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 2: Worst-lag coordinate descent
    if time_fn(80) < 50:
        ht = best_h.copy()
        ht, ct = _worst_lag_cd(ht, best_n, best_dx, time_fn(80), iters=5000, seed=seed)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 3: Try alternative resolutions and starting points
    resolutions = [70, 100, 105, 200]
    for n_try in resolutions:
        if time_fn(40) < 40:
            break
        dx_try = 2.0 / n_try
        starts = []
        # Interpolated from best
        x_b = np.linspace(0, 2, best_n, endpoint=False)
        x_t = np.linspace(0, 2, n_try, endpoint=False)
        starts.append(_repair(np.interp(x_t, x_b, best_h), n_try))
        # Sine starting points
        starts.append(_make_sine(n_try, [2]))
        starts.append(_make_sine(n_try, [3]))
        starts.append(_make_sine(n_try, [2, 3]))
        # Sawtooth
        starts.append(_make_sawtooth(n_try, 5))
        starts.append(_make_sawtooth(n_try, 7))

        for hs in starts:
            if time_fn(30) < 30:
                break
            ht, ct = _subgrad_opt(hs, n_try, dx_try, time_fn(30),
                                   max_iter=2500, n_top_frac=0.2)
            ht, ct = _worst_lag_cd(ht, n_try, dx_try, time_fn(20),
                                    iters=2000, seed=seed + n_try)
            if ct < best_c5:
                best_h, best_c5, best_n, best_dx = ht, ct, n_try, dx_try

    # Phase 4: Re-run subgradient on best with finer settings
    if time_fn(30) > 40:
        ht = best_h.copy()
        ht, ct = _subgrad_opt(ht, best_n, best_dx, time_fn(30),
                               max_iter=5000, n_top_frac=0.1)
        ht, ct = _worst_lag_cd(ht, best_n, best_dx, time_fn(20),
                                iters=4000, seed=seed + 7)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 5: Fine coordinate descent polish
    if time_fn(5) > 10:
        ht = best_h.copy()
        ht, ct = _fine_cd(ht, best_n, best_dx, time_fn(3),
                          iters=12000, seed=seed + 99)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 6: Extra fine with very small steps if time allows
    if time_fn(3) > 5:
        ht = best_h.copy()
        ht, ct = _fine_cd(ht, best_n, best_dx, time_fn(2),
                          iters=6000, seed=seed + 200)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_dx)
    return (best_h, best_c5, best_n)
