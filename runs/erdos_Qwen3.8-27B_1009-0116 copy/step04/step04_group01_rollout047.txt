def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    # Proportional scaling then clip and fix
    h *= target / max(s, 1e-15)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute remaining mass
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            room = 1.0 - h[idx] if diff > 0 else h[idx]
            adj = min(abs(diff), room)
            h[idx] += np.sign(diff) * adj
            diff -= np.sign(diff) * adj
            if abs(diff) < 1e-12:

                    c5_new = _eval(h)
                    if c5_new < c5_cur - 1e-16:
                        c5_cur = c5_new
                        if c5_new < _chk.cur:
                            _chk.cur = c5_new
                            _chk.h = h.copy()
                            _chk.n = n
                    else:
                        h[i], h[j] = ni - d, nj + d

    # Phase F: Final SLSQP polish
    if time.time() - t0 < budget_s - 15:
        h_new = _do_slsqp(_chk.h, 1000000, 5000, 1e-20)
        _chk(h_new)

    # Phase G: Ultra-fine touch
    if time.time() - t0 < budget_s - 3:
        h = _chk.h.copy()
        n = len(h)
        c5_cur = _eval(h)
        for it in range(30000):
            if time.time() - t0 > budget_s - 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = 3e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5_new = _eval(h)
                if c5_new < c5_cur - 1e-17:
                    c5_cur = c5_new
                    if c5_new < _chk.cur:
                        _chk.cur = c5_new
                        _chk.h = h.copy()
                        _chk.n = n
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(_chk.h, _chk.n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, _chk.n)
