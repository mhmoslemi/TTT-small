def _repair(h, n):
    # Project onto {0<=h<=1, sum(h)=n/2}
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-10:
        return h
    # Bisection for lambda: sum(clip(h_i - lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s_mid = np.clip(h - mid, 0, 1).sum()
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

    # Subgradient of f_k_star w.r.t. h
    n = len(h)
    k = k_star - (n - 1)  # actual lag
    g = np.zeros(n)
    if k >= 0:

For mode='full', the output c has length M + N - 1 where M, N are lengths of a and v.

The formula is: c[m] = sum_n a[n] * v[n + m - (N-1)]

where m ranges from 0 to M+N-2, and n ranges over valid indices where 0 ≤ n < M and 0 ≤ n+m-(N-1) < N.

In our case, a = h (length n), v = 1-h (length n), N = n.

Let k = m - (n-1), ranging from -(n-1) to n-1.

c[m] = sum_{n: valid} h[n] * (1-h[n+k])

For k ≥ 0: valid n are 0 to n-1-k (since n+k < n)
For k < 0: valid n are -k to n-1 (since n+k ≥ 0)

        = sum_{n=-k}^{n-1} h[n](1-h[n+k]) for k < 0

def _sg(h, m):
    """Subgradient of correlate(h,1-h,'full')[m] w.r.t. h."""
    n = len(h)
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    else:
        kp = -k
        g[kp:] += 1.0 - h[:n - kp]
        g[:n - kp] -= h[kp:]
    return g

def _sgj(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(a * (c - m), -400.0, 400.0)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for m_ in range(2 * n - 1):
        if w[m_] < 1e-13:
            continue
        g += w[m_] * _sg(h, m_)
    return g * (2.0 / n)

def _cf(x, n):
    return np.sum(x) - n * 0.5

def _cj(x, n):
    return np.ones(n)

def _slsqp(h, n, a, mi, t_end):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        r = minimize(_sf, h, jac=_sgj, args=(a,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-20})
        if time.time() > t_end:
            return _proj(h, n)
        return _proj(r.x, n)
    except Exception:
        return _proj(h.copy(), n)

def _ls(h, n, rng, t_end, sc0=0.01):
    hh = h.copy()
    c5 = _ev(hh)
    sc = sc0
    idle = 0
    while time.time() < t_end:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = sc * rng.randn()
        a, b = hh[i] + d, hh[j] - d
        if a < 0.0 or a > 1.0 or b < 0.0 or b > 1.0:
            continue
        hh[i], hh[j] = a, b
        nc = _ev(hh)
        if nc < c5 - 1e-15:
            c5 = nc
            idle = 0
            sc = min(sc * 1.03, 0.025)
        else:
            hh[i], hh[j] = hh[i] - d, hh[j] + d
            idle += 1
            if idle > 250:
                sc *= 0.84
                idle = 0
                if sc < 1e-6:
                    break
    return hh, c5

def _ix(h, ns, nd):
    xs = np.linspace(0.0, 2.0, ns, endpoint=False)
    xd = np.linspace(0.0, 2.0, nd, endpoint=False)
    return _proj(np.interp(xd, xs, h), nd)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_p = _proj(np.asarray(initial_h_values, dtype=np.float64))
    n_p = len(h_p)
    bc5, bh, bn = _ev(h_p), h_p.copy(), n_p

    n_lst = [40, 50, 60, 70, 80, 90, 100, 120, 140]
    inits = {n_p: h_p.copy()}
    for n in n_lst:
        if n == n_p:
            continue
        cands = [_ix(h_p, n_p, n)]
        for f in range(1, 7):
            for am in (0.2, 0.4, 0.65):
                cands.append(_proj(0.5 + am * np.sin(2.0 * np.pi * f * np.arange(n) / n), n))
        cands.append(_proj(0.2 + 0.6 * rng.rand(n), n))
        bv, bc = 9.0, None
        for c in cands:
            v = _ev(c)
            if v < bv:
                bv, bc = v, c.copy()
        inits[n] = bc

    # Phase 1: SLSQP multi-alpha across resolutions
    t1 = t0 + 0.25 * budget_s
    n_all = sorted(set(n_lst + [n_p]))
    for n in n_all:
        if time.time() > t1:
            break
        h0 = inits.get(n, _ix(bh, bn, n))
        for a in (200, 2000, 20000, 200000):
            if time.time() > t1:
                break
            mi = 400 if n <= 80 else 150
            h1 = _slsqp(h0, n, a, mi, t1)
            v = _ev(h1)
            if v < bc5:
                bc5, bh, bn = v, h1.copy(), n
            inits[n] = h1
            h0 = h1

    # Phase 2: PFM exploration
    t2 = t0 + 0.48 * budget_s
    for n in n_all + [bn]:
        if time.time() > t2:
            break
        h0 = inits.get(n, _ix(bh, bn, n))
        for lr, bm in ((0.06, 0.85), (0.10, 0.92), (0.15, 0.94)):
            if time.time() > t2 - 3:
                break
            h1, v = _pfm(h0, n, t2 - 3, lr, bm)
            if v < bc5:
                bc5, bh, bn = v, h1, n
            inits[n] = h1
            h0 = h1

    # Phase 3: Focused SLSQP on current best
    t3 = t0 + 0.58 * budget_s
    for a in (50000, 150000, 400000):
        if time.time() > t3:
            break
        h1 = _slsqp(bh, bn, a, 400, t3)
        v = _ev(h1)
        if v < bc5:
            bc5, bh = v, h1

    # Phase 4: Extended local search
    t4 = t0 + 0.85 * budget_s
    for sc0 in (0.012, 0.005, 0.002, 0.001):
        if time.time() > t4:
            break
        h1, v = _ls(bh, bn, rng, t4, sc0)
        if v < bc5:
            bc5, bh = v, h1

    # Phase 5: Cross-resolution + final polish
    t5 = t0 + 0.96 * budget_s
    for dn in (-10, 0, 10):
        n_t = bn + dn
        if n_t < 20 or time.time() > t5:
            continue
        h1 = _ix(bh, bn, n_t)
        h1 = _slsqp(h1, n_t, 200000, 250, t5)
        v = _ev(h1)
        if v < bc5:
            bc5, bh, bn = v, h1, n_t
    if time.time() < t5:
        h1, v = _ls(bh, bn, rng, t5, 0.0008)
        if v < bc5:
            bc5, bh = v, h1

    if budget_s - (time.time() - t0) > 12:
        h1 = _slsqp(bh, bn, 500000, 300, t0 + budget_s - 3)
        v = _ev(h1)
        if v < bc5:
            bc5, bh = v, h1

    bh = _proj(bh, bn)
    bc5 = _ev(bh)
    return (bh, bc5, bn)
