import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
            else:
                a = min(-d, h[idx]); h[idx] -= a; d += a
    return h

def fw_subgrad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    active = np.where(corr > mx - 0.005 * mx)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += (1.0 - h[s:n])
            grad[s:n] += -h[:m]
        else:
            s2 = -s; m = n - s2
            grad[:m] += -h[s2:n]
            grad[s2:n] += (1.0 - h[:m])
    grad /= len(active)
    return grad

def fw_opt(h, n, n_iters, t_end):
    h = h.copy()
    best = compute_c5(h, n)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        grad = fw_subgrad(h, n)
        order = np.argsort(grad)
        v = np.zeros(n)
        rem = n / 2.0
        for idx in order:
            if rem >= 1.0:
                v[idx] = 1.0; rem -= 1.0
            elif rem > 0:
                v[idx] = rem; rem = 0.0
            else:
                break
        improved = False
        for beta in [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]:
            ht = (1.0 - beta) * h + beta * v
            ht = repair(ht, n)
            ct = compute_c5(ht, n)
            if ct < best - 1e-12:
                h, best, improved = ht, ct, True
                break
        if not improved:
            for beta in [0.002, 0.001]:
                ht = (1.0 - beta) * h + beta * v
                ht = repair(ht, n)
                ct = compute_c5(ht, n)
                if ct < best - 1e-13:
                    h, best = ht, ct
                    break
    return h, best

def pair_search(h, n, n_iters, t_end, rng, use_target=False):
    h = h.copy()
    best = compute_c5(h, n)
    deltas = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    stall = 0
    for it in range(n_iters):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        found = False
        for d in deltas:
            for sg in (1, -1):
                delta = sg * d
                ni, nj = hi + delta, hj - delta
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5 = compute_c5(h, n)
                h[i], h[j] = hi, hj
                if c5 < best - 1e-14:
                    h[i], h[j] = ni, nj
                    best = c5
                    found = True
                    stall = 0
                    break
            if found:
                break
        if not found:
            stall += 1
            if stall > 300:
                h = h + 0.015 * rng.randn(n)
                h = repair(h, n)
                stall = 0
    return h, best

def binary_opt(h_init, n, t_end, rng):
    h = (h_init > 0.5).astype(float)
    tgt = n // 2
    no = int(h.sum())
    if no > tgt:
        oi = np.where(h == 1)[0]
        for idx in rng.choice(oi, no - tgt, replace=False):
            h[idx] = 0.0
    elif no < tgt:
        zi = np.where(h == 0)[0]
        for idx in rng.choice(zi, tgt - no, replace=False):
            h[idx] = 1.0
    best = compute_c5(h, n)
    for it in range(500000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h, n)
        if c5 < best - 1e-14:
            best = c5
        else:
            h[i], h[j] = h[j], h[i]
            if it % 5000 == 4999:
                for _ in range(20):
                    a, b = rng.randint(n), rng.randint(n)
                    if h[a] != h[b]:
                        h[a], h[b] = h[b], h[a]
                        if compute_c5(h, n) < best:
                            best = compute_c5(h, n)
                            break
                        h[a], h[b] = h[b], h[a]
    return h, best

def sys_pair_search(h, n, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    deltas = [0.03, 0.01, 0.005, 0.002, 0.001]
    for rnd in range(500):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if time.time() > t_end:
                break
            hidx = h[idx]
            for d in deltas:
                for sg in (1, -1):
                    delta = sg * d
                    ni = hidx + delta
                    if ni < 0 or ni > 1:
                        continue
                    j = rng.randint(n)
                    if j == idx:
                        continue
                    nj = h[j] - delta
                    if nj < 0 or nj > 1:
                        continue
                    h[idx], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:
                        best = c5
                        improved = True
                        break
                    h[idx], h[j] = hidx, h[j]
                if improved:
                    break
    return h, best

def resample_and_opt(h_best, n_best, n_alt, t_end, rng):
    x_old = np.linspace(0.0, 1.0, n_best)
    x_new = np.linspace(0.0, 1.0, n_alt)
    h_s = np.interp(x_new, x_old, h_best)
    h_s = repair(h_s, n_alt)
    h1, c5_1 = fw_opt(h_s, n_alt, 3000, t_end)
    h2, c5_2 = pair_search(h1, n_alt, 3000, t_end, rng)
    if c5_2 < c5_1:
        h1, c5_1 = h2, c5_2
    return h1, c5_1, n_alt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 54321)
    n0 = 64
    try:
