        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = dts[rng.randint(len(dts))]
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def de_neighborhood(h0, n, t0, lim, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    pop_size = 14
    pop = [h0.copy()]
    pop_c = [c5v(h0, dx)]
    for i in range(pop_size - 1):
        sc = 0.03 + 0.12 * (i / pop_size)
        hi = repair(h0 + sc * rng.randn(n), n)
        pop.append(hi)
        pop_c.append(c5v(hi, dx))
    bi = int(np.argmin(pop_c))
    bh, bc = pop[bi].copy(), pop_c[bi]
    for gen in range(200):
        if time.time() - t0 > lim:
            break
        improved = False
        for i in range(pop_size):
            if time.time() - t0 > lim:
                break
            others = [j for j in range(pop_size) if j != i]
            a, b, c_idx = rng.choice(others, 3, replace=False)
            F = 0.4 + 0.4 * rng.rand()
            CR = 0.7 + 0.25 * rng.rand()
            j0 = rng.randint(n)
            v = pop[a] + F * (pop[b] - pop[c_idx])
            mask = rng.rand(n) < CR
            mask[j0] = True
            trial = np.where(mask, v, pop[i])
            trial = repair(trial, n)
            ct = c5v(trial, dx)
            if ct < pop_c[i]:
                pop[i] = trial
                pop_c[i] = ct
                improved = True
                if ct < bc:
                    bc, bh = ct, trial.copy()
        if not improved and gen > 30:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    if time.time() - t0 < budget_s - 200:
        r, c = lse_lbfgs(bh, nl, t0, budget_s - 150)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 120:
        r, c = active_eq(bh, nl, t0, budget_s - 100)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 80:
        r, c = spectral(bh, nl, t0, budget_s - 70)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 60:
        r, c = pm_lbfgs(bh, nl, t0, budget_s - 50)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 50:
        r, c = lse_lbfgs(bh, nl, t0, budget_s - 40)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 40:
        r, c = active_eq(bh, nl, t0, budget_s - 35)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 30:
        r, c = de_neighborhood(bh, nl, t0, budget_s - 25, seed)
        if c < bc:
            bh, bc = r, c
            if time.time() - t0 < budget_s - 15:
                r2, c2 = lse_lbfgs(bh, nl, t0, budget_s - 12)
                if c2 < bc:
                    bh, bc = r2, c2
                r2, c2 = active_eq(bh, nl, t0, budget_s - 10)
                if c2 < bc:
                    bh, bc = r2, c2

    if time.time() - t0 < budget_s - 20:
        r, c = de_neighborhood(bh, nl, t0, budget_s - 18, seed + 77)
        if c < bc:
            bh, bc = r, c
            if time.time() - t0 < budget_s - 10:
                r2, c2 = active_eq(bh, nl, t0, budget_s - 8)
                if c2 < bc:
                    bh, bc = r2, c2

    if time.time() - t0 < budget_s - 8:
        r, c = spectral(bh, nl, t0, budget_s - 6)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 5:
        r, c = active_eq(bh, nl, t0, budget_s - 4)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 2:
        rng = np.random.RandomState(seed + 5000)
        r, c = fine_cd(bh, nl, rng, t0, budget_s - 0.5, 40000)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
