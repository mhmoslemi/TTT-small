- t ≤ n/2 (redundant but helps)

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
                if abs(d) < 1e-12: break
        else:
            order = np.argsort(h)
            for idx in order:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
                if abs(d) < 1e-12: break
    return h

def solve_lp(h_fixed, n):
    """Solve the LP maximization of t s.t. L_k @ h >= t for all k, 0<=h<=1, sum(h)=n/2"""
    # Variables: h_0...h_{n-1}, t → n+1 variables
    # Objective: minimize -t → c = [0,...,0,-1]
    c = np.zeros(n + 1)
    c[-1] = -1.0  # minimize -t = maximize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: -(L_k @ h) + t <= 0 → [-L_k, 1] @ [h, t] <= 0
    # Number of lags: 2n-1
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    
    for k in range(n_lags):
        d = k - (n - 1)  # shift
        for i in range(n):
            j = i + d  # index in h_fixed
            if 0 <= j < n:
                A_ub[k, i] = -h_fixed[j]
        A_ub[k, n] = 1.0  # coefficient of t
    
    # Upper bound: t <= n/2 (already implied but add for numerical stability)
    # Actually let's not add it, the constraint L_k @ h >= t with L_k[i] <= 1
    # and sum(h) = n/2 already bounds t <= n/2
    
    # Equality constraints: A_eq @ x == b_eq
    # sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[i] <= 1, t >= 0 (and t <= n/2)
    bounds = [(0.0, 1.0)] * n + [(0.0, n / 2.0)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    for k in range(n_lags):
        d = k - (n - 1)
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                A_ub[k, i] = -h_fixed[j]
        A_ub[k, n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

def soft_obj(v, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_opt(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(lambda v: soft_obj(v, alpha), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def lbfgs_opt(h, n, alpha, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(v):
        return soft_obj(v, alpha) + pen * (v.sum() - n / 2.0) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
        return repair(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, t_end, rng, max_rounds=3000):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    stall = 0
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(6 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hi + delta, hj - delta
                    if ni < -1e-10 or ni > 1 + 1e-10 or nj < -1e-10 or nj > 1 + 1e-10:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved:
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0
    return h, best

def deterministic_cd(h, n, t_end, n_passes=3):

    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

                if h[i] != hi:
