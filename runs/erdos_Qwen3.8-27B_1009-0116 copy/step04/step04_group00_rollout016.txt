```python
        if time.time() - t0 > tl:
            break
        # Try random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hb[i], hb[j] = 0, 1
        c5_new = _c5(hb, dx)
        if c5_new < c5c:
            c5c = c5_new
            if c5c < best_c5:
                best_c5 = c5c
                best = hb.copy()
            # Update index sets
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]
        else:
            hb[i], hb[j] = 1, 0
    return best, best_c5
```

But this is very slow - O(n) per swap for the correlation. For n=140, that's 140 mults per swap, which is fine.

Actually, for binary h, the correlation is:
c[k] = ∑ h[i] * (1-h[i+k])

For h ∈ {0,1}, this is just the count of positions where h[i]=1 and h[i+k]=0.

We can compute this faster using FFT, but for n=140 it's not necessary.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softobj(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    if lam > 0:
        g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _active_subgrad(h, n, frac=0.04):
    c = _corr(h)
    cm = np.max(c)
    spread = max(cm - np.min(c), 1e-12)
    thresh = cm - frac * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g

def _nesterov_descent(h, n, dx, tl, rng, max_it=5000):
    t0 = time.time()
    h = h.copy()
    c5c = _c5(h, dx)
    best, best_c5 = h.copy(), c5c
    y = h.copy()
    mom = np.zeros(n)
    beta = 0.85
    lr = 0.012
    flat = 0
    for it in range(max_it):
        if time.time() - t0 > tl:
            break
        g = _active_subgrad(y, n, max(0.02, 0.06 * (1 - it / max(max_it, 1))))
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-14:
            break
        g /= ng
        mom = beta * mom - lr * g
        h_new = _repair(h + mom, n)
        y = h_new + (beta - 1) * (mom / (lr + 1e-30) * lr)
        c5_new = _c5(h_new, dx)
        if c5_new < c5c - 1e-16:
            h = h_new
            c5c = c5_new
            flat = 0
            lr = min(lr * 1.015, 0.045)
            beta = min(beta + 0.001, 0.93)
            if c5c < best_c5:
                best_c5, best = c5c, h.copy()
        else:
            flat += 1
            lr *= 0.92
            beta *= 0.97
            if flat > 60:
                flat = 0
                lr, beta = 0.012, 0.85
                mom = np.zeros(n)
                y = h.copy()
                h = _repair(h + 0.004 * rng.randn(n), n)
                c5c = _c5(h, dx)
    return best, best_c5

def _lbfgs_phase(h, n, dx, tl, rng):
    t0 = time.time()
    bh, bc5 = h.copy(), _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    for a in [3000, 15000, 80000, 400000, 2000000]:
        if time.time() - t0 > tl:
            break
        try:
            res = minimize(lambda x: _softobj(x, a, 50.0, n), bh.copy(),
                           jac=lambda x: _softgrad(x, a, 50.0, n),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-19, 'gtol': 1e-15})
            hc = _repair(res.x, n)
            cv = _c5(hc, dx)
            if cv < bc5:
                bc5, bh = cv, hc
        except Exception:
            pass
    return bh, bc5

def _pairwise_refine(h, n, dx, tl, rng):
    t0 = time.time()
    h = h.copy()
    c5c = _c5(h, dx)
    best, best_c5 = h.copy(), c5c
    dlist = np.array([0.001, 0.003, 0.008, 0.016, 0.032, 0.06,
                      -0.001, -0.003, -0.008, -0.016, -0.032, -0.06])
    while time.time() - t0 < tl:
        g = _active_subgrad(h, n, 0.1)
        order = np.argsort(np.abs(g))[::-1][:35]
        improved = False
        for a in range(len(order)):
            if time.time() - t0 > tl:
                break
            i = order[a]
            for b in range(a + 1, min(a + 12, len(order))):
                j = order[b]
                bd, bc = 0.0, c5c
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        oi, oj = h[i], h[j]
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        h[i], h[j] = oi, oj
                        if c < bc:
                            bc, bd = c, d
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    improved = True
                    if c5c < best_c5:
                        best_c5, best = c5c, h.copy()
                    break
        if not improved:
            break
    return best, best_c5

def _binary_search(h, n, dx, tl, rng):
    t0 = time.time()
    hb = (h > 0.5).astype(float)
    s = hb.sum()
    target = n / 2
    if s < target:
        order = np.argsort(-h)[int(s):]
        hb = (h > 0.5).astype(float)
        for idx in order[:int(target - s)]:
            if hb[idx] == 0:
                hb[idx] = 1
    elif s > target:
        order = np.argsort(h)[int(s - target):]
        hb = (h > 0.5).astype(float)
        for idx in order:
            if hb[idx] == 1:
                hb[idx] = 0
    hb = _repair(hb, n)
    c5c = _c5(hb, dx)
    best, best_c5 = hb.copy(), c5c
    while time.time() - t0 < tl:
        ones = np.where(hb >= 0.99)[0]
        zeros = np.where(hb <= 0.01)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hb[i], hb[j] = 0.0, 1.0
        c5_new = _c5(hb, dx)
        if c5_new < c5c:
            c5c = c5_new
            if c5c < best_c5:
                best_c5, best = c5c, hb.copy()
        else:
            hb[i], hb[j] = 1.0, 0.0
    return best, best_c5

def _random_refine(h, n, dx, tl, rng):
    t0 = time.time()
    h = h.copy()
    c5c = _c5(h, dx)
    best, best_c5 = h.copy(), c5c
    while time.time() - t0 < tl:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.015, 0.015)
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            oi, oj = h[i], h[j]
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c:
                c5c = c
                if c5c < best_c5:
                    best_c5, best = c5c, h.copy()
            else:
                h[i], h[j] = oi, oj
    return best, best_c5

def _interpolate(src, src_n, dst_n):
    src_grid = np.linspace(0, 2, src_n, endpoint=False)
    dst_grid = np.linspace(0, 2, dst_n, endpoint=False)
    return np.interp(dst_grid, src_grid, src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_left(margin=10):
        return budget_s - elapsed() - margin

    # Start from parent
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h, best_c5 = h0.copy(), _c5(h0, dx)

    # Phase 1: L-BFGS at parent resolution
    if time_left(300) > 50:
        best_h, best_c5 = _lbfgs_phase(best_h, n, dx, min(120, time_left(300)), rng)

    # Phase 2: Nesterov subgradient
    if time_left(250) > 80:
        best_h, best_c5 = _nesterov_descent(best_h, n, dx, min(200, time_left(250)), rng)

    # Phase 3: Pairwise refinement
    if time_left(200) > 60:
        best_h, best_c5 = _pairwise_refine(best_h, n, dx, min(120, time_left(200)), rng)

    # Phase 4: Binary local search
    if time_left(200) > 60:
        bh, bc = _binary_search(best_h, n, dx, min(100, time_left(200)), rng)
        if bc < best_c5:
            best_h, best_c5 = bh, bc

    # Phase 5: Nesterov again from potentially improved point
    if time_left(150) > 60:
        best_h, best_c5 = _nesterov_descent(best_h, n, dx, min(100, time_left(150)), rng)

    # Phase 6: More pairwise
    if time_left(100) > 40:
        best_h, best_c5 = _pairwise_refine(best_h, n, dx, min(60, time_left(100)), rng)

    # Try lower resolution for global structure
    n2 = 70
    dx2 = 2.0 / n2
    h2 = _repair(_interpolate(best_h, n, n2), n2)
    if time_left(100) > 40:
        h2, c2 = _nesterov_descent(h2, n2, dx2, min(60, time_left(100)), rng)
        if c2 < best_c5:
            # Upscale back
            best_h = _repair(_interpolate(h2, n2, n), n)
            best_c5 = _c5(best_h, dx)

    # Try another lower resolution
    n3 = 50
    dx3 = 2.0 / n3
    h3 = _repair(_interpolate(best_h, n, n3), n3)
    if time_left(80) > 40:
        h3, c3 = _nesterov_descent(h3, n3, dx3, min(50, time_left(80)), rng)
        if c3 < best_c5:
            best_h = _repair(_interpolate(h3, n3, n), n)
            # Refine at n
            best_h, best_c5 = _pairwise_refine(best_h, n, dx, min(30, time_left(40)), rng)

    # Final random refinement
    if time_left(10) > 20:
        best_h, best_c5 = _random_refine(best_h, n, dx, min(40, time_left(10)), rng)

    # Final pairwise polish
    if time_left(5) > 15:
        best_h, best_c5 = _pairwise_refine(best_h, n, dx, min(20, time_left(5)), rng)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
