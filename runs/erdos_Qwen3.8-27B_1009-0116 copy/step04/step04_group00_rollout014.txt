import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_component(h, k, n):
    L = k - (n - 1)
    if L == 0:
        return 1.0 - 2.0 * h
    elif L > 0:
        g = np.zeros(n)
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
        return g
    else:
        M = -L
        g = np.zeros(n)
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
        return g

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_one(h, k, n):
    L = k - (n - 1)
    if L == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if L > 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _topk_grad(h, k_count):
    c = _corr(h)
    n = len(h)
    if k_count >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -k_count)[-k_count:]
    g = np.zeros(n)
    for ki in idx:
        g += _grad_one(h, ki, n)
    return g / len(idx)
