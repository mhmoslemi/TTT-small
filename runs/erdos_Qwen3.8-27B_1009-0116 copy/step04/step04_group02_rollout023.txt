import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def subgradient(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    kmax = np.argmax(corr)
    offset = kmax - (n - 1)
    grad = np.zeros(n)
    jp = np.arange(n) + offset
    jm = np.arange(n) - offset
    mask_p = (jp >= 0) & (jp < n)
    mask_m = (jm >= 0) & (jm < n)
    grad[mask_p] += 1.0 - h[jp[mask_p]]
    grad[mask_m] -= h[jm[mask_m]]
    return grad

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_con(x, target):
    return np.sum(x) - target

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

        h, c5 = directed_search(h, n, dx, rng, t_end, 100000)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
