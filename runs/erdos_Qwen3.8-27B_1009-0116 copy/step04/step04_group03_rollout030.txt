def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-14: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm_obj(h, alpha, beta, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -50, 50)
    s -= s.max()
    sm = m + np.log(np.sum(np.exp(s))) / alpha
    d = h.sum() - n / 2.0
    return sm + beta * d * d

def _sm_grad(h, alpha, beta, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -50, 50)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-17: continue
        lag = k - (n - 1)
        i_lo, i_hi = max(0, -lag), min(n, n - lag)
        if i_lo >= i_hi: continue
        g[i_lo:i_hi] += wk * (1.0 - h[i_lo+lag:i_hi+lag])
        g[i_lo+lag:i_hi+lag] -= wk * h[i_lo:i_hi]
    g += 2.0 * beta * (h.sum() - n / 2.0)
