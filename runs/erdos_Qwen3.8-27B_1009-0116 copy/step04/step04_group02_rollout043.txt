        c5n = compute_c5(h, dx)
        delta = c5n - cur_c5
        if delta <= 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cur_c5 = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
                last_imp_time = time.time()
        else:
            h[i] -= d
            h[j] += d
        t_elapsed = time.time() - (deadline - t_total_est)
    return repair(best_h, n), best_c5

def fine_local(h, n, dx, rng, deadline):
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    scales = [0.002, 0.0008, 0.0003, 0.00012, 0.00005]
    for sc in scales:
        while time.time() < deadline:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i] = ni
            h[j] = nj
            c5n = compute_c5(h, dx)
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
            else:
                h[i] -= d
                h[j] += d
    return h, best_c5

def block_move(h, n, dx, rng, deadline):
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    while time.time() < deadline:
        bs = rng.randint(2, 5)
        i = rng.randint(0, n - bs)
        j = rng.randint(0, n - bs)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.012) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if h[i + di] + eps < 0 or h[i + di] + eps > 1:
                ok = False
                break
            if h[j + di] - eps < 0 or h[j + di] - eps > 1:
                ok = False
                break
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5n = compute_c5(h, dx)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h0 = repair(h_init.copy(), n)
    c5_0 = compute_c5(h0, dx)

    best_h, best_c5 = h0.copy(), c5_0
    h, cur_c5 = h0.copy(), c5_0

    # Phase 1: L-BFGS multi-alpha (70s)
    t_end = t_start + 70
    for alpha in [50, 200, 1000, 5000, 20000]:
        if time.time() >= t_end:
            break
        for pen in [200, 2000]:
            if time.time() >= t_end:
                break
            hh, cc = lbfgs(h, n, alpha, pen, 300, target)
            if cc < cur_c5:
                h, cur_c5 = hh, cc
    if cur_c5 < best_c5:
        best_h, best_c5 = h.copy(), cur_c5

    # Phase 2: SLSQP (70s)
    t_end = t_start + 140
    for alpha in [500, 3000, 15000]:
        if time.time() >= t_end:
            break
        hh, cc = slsqp(h, n, alpha, 400, target)
        if cc < cur_c5:
            h, cur_c5 = hh, cc
    if cur_c5 < best_c5:
        best_h, best_c5 = h.copy(), cur_c5

    # Phase 3: Simulated annealing main (480s)
    t_end = t_start + 620
    h, cur_c5 = anneal(h, n, dx, rng, 0.012, 0.0003, t_end)
    if cur_c5 < best_c5:
        best_h, best_c5 = h.copy(), cur_c5

    # Phase 4: Multi-restart (180s)
    t_end = t_start + 800
    for r in range(10):
        if time.time() >= t_end:
            break
        hp = h.copy()
        idx = rng.choice(n, size=max(5, n // 6), replace=False)
        hp[idx] += rng.normal(0, 0.04, len(idx))
        hp = repair(hp, n)
        hh, cc = lbfgs(hp, n, 1500, 3000, 200, target)
        if cc < cur_c5:
            h, cur_c5 = hh, cc
        seg_deadline = time.time() + min(15, t_end - time.time())
        h2, cc2 = anneal(h, n, dx, rng, 0.006, 0.0002, seg_deadline)
        if cc2 < cur_c5:
            h, cur_c5 = h2, cc2
    if cur_c5 < best_c5:
        best_h, best_c5 = h.copy(), cur_c5

    # Phase 5: Block local search (80s)
    t_end = t_start + 880
    h, cur_c5 = block_move(h, n, dx, rng, t_end)
    if cur_c5 < best_c5:
        best_h, best_c5 = h.copy(), cur_c5

    # Phase 6: Fine local search (100s)
    t_end = t_start + budget_s - 40
    h, cur_c5 = fine_local(h, n, dx, rng, t_end)
    if cur_c5 < best_c5:
        best_h, best_c5 = h.copy(), cur_c5

    # Phase 7: Final SLSQP (30s)
    if budget_s - (time.time() - t_start) > 35:
        t_end = time.time() + 25
        hh, cc = slsqp(h, n, 50000, 500, target)
        if cc < cur_c5:
            h, cur_c5 = hh, cc
        if cur_c5 < best_c5:
            best_h, best_c5 = h.copy(), cur_c5

    # Phase 8: Ultra-fine
    if budget_s - (time.time() - t_start) > 15:
        h, cur_c5 = fine_local(h, n, dx, rng, t_start + budget_s - 2)
        if cur_c5 < best_c5:
            best_h, best_c5 = h.copy(), cur_c5

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
