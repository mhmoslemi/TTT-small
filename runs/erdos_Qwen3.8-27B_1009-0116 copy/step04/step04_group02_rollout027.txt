                if i == j:
                    continue
                d = rng.uniform(-0.12, 0.12)
                a, b = h[i] + d, h[j] - d
                if a < -0.01 or a > 1.01 or b < -0.01 or b > 1.01:
                    continue
                h[i], h[j] = a, b
                nc = compute_c5(h, dx)
                dd = nc - cc
                if dd < 0 or rng.random() < np.exp(-dd / T):
                    cc = nc
                    if cc < bc:
                        bc = cc; bh = h.copy(); stuck = 0
                    else:
                        stuck += 1
                else:
                    h[i], h[j] = a - d, b + d
            elif r < 0.8:
                bs = rng.randint(2, 5)
                i = rng.randint(0, n - bs); j = rng.randint(0, n - bs)
                if abs(i - j) < bs:
                    continue
                ep = rng.uniform(-0.06, 0.06)
                bi = h[i:i+bs]; bj = h[j:j+bs]
                if np.min(bi + ep) < -0.01 or np.max(bi + ep) > 1.01:
                    continue
                if np.min(bj - ep) < -0.01 or np.max(bj - ep) > 1.01:
                    continue
                h[i:i+bs] += ep; h[j:j+bs] -= ep
                nc = compute_c5(h, dx)
                dd = nc - cc
                if dd < 0 or rng.random() < np.exp(-dd / T):
                    cc = nc
                    if cc < bc:
                        bc = cc; bh = h.copy(); stuck = 0
                    else:
                        stuck += 1
                else:
                    h[i:i+bs] -= ep; h[j:j+bs] += ep
            else:
                ii = rng.randint(n, size=4)
                i, j, i2, j2 = ii[0], ii[1], ii[2], ii[3]
                if i == j or i2 == j2:
                    continue
                d1 = rng.uniform(-0.08, 0.08)
                d2 = rng.uniform(-0.08, 0.08)
                a, b = h[i] + d1, h[j] - d1
                c2, dv = h[i2] + d2, h[j2] - d2
                if min(a, b, c2, dv) < -0.01 or max(a, b, c2, dv) > 1.01:
                    continue
                h[i], h[j], h[i2], h[j2] = a, b, c2, dv
                nc = compute_c5(h, dx)
                dlt = nc - cc
                if dlt < 0 or rng.random() < np.exp(-dlt / T):
                    cc = nc
                    if cc < bc:
                        bc = cc; bh = h.copy(); stuck = 0
                    else:
                        stuck += 1
                else:
                    h[i], h[j], h[i2], h[j2] = a-d1, b+d1, c2-d2, dv+d2
        if stuck > 3000:
            T = max(T, T0 * 0.3)
            stuck = 0
    return bh, bc

def greedy_ls(h, n, dx, rng, deadline, per_scale):
    h = h.copy()
    bc = compute_c5(h, dx)
    scales = [0.004, 0.0015, 0.0006, 0.0002, 0.00007]
    for sc in scales:
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, bc
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            a, b = h[i] + d, h[j] - d
            if a < -1e-9 or a > 1.0 + 1e-9 or b < -1e-9 or b > 1.0 + 1e-9:
                continue
            h[i], h[j] = a, b
            nc = compute_c5(h, dx)
            if nc < bc - 1e-14:
                bc = nc
            else:
                h[i], h[j] = a - d, b + d
    return h, bc

def perturb_restart(h, n, rng):
    hp = h.copy()
    k = rng.randint(3)
    if k == 0:
        hp += rng.normal(0, 0.03, n)
    elif k == 1:
        f = rng.randint(1, 5)
        hp += 0.05 * np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.28))
    else:
        hp *= rng.uniform(0.9, 1.1)
    return repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = repair(h, n)
    bc = compute_c5(h, dx)
    bh = h.copy()

    # Phase 1: L-BFGS polish (40s)
    dl = ts + 40
    for a in [500, 2000, 8000]:
        if time.time() >= dl:
            break
        hh, cc = lbfgs_opt(h, n, a, target, 2000, 300)
        if cc < bc:
            h, bc, bh = hh, cc, hh.copy()

    # Phase 2: Simulated annealing - main workhorse (500s)
    dl = ts + 540
    h, c2 = sim_anneal(h, n, dx, rng, dl)
    if c2 < bc:
        bc = c2; bh = h.copy()

    # Phase 3: Multi-start with perturbation (110s)
    dl = ts + 650
    n_starts = 8
    for r in range(n_starts):
        if time.time() >= dl:
            break
        hp = perturb_restart(bh, n, rng)
        hh, cc = lbfgs_opt(hp, n, 3000, target, 3000, 200)
        dl_r = min(dl, time.time() + 10)
        hh, cc = sim_anneal(hh, n, dx, rng, dl_r)
        if cc < bc:
            bc = cc; bh = hh.copy(); h = hh.copy()

    # Phase 4: L-BFGS high-alpha + SLSQP (130s)
    dl = ts + 780
    for a in [5000, 20000, 80000]:
        if time.time() >= dl:
            break
        hh, cc = lbfgs_opt(h, n, a, target, 3000, 400)
        if cc < bc:
            h, bc, bh = hh, cc, hh.copy()
    if time.time() < dl:
        hh, cc = slsqp_opt(h, n, 30000, target, 500)
        if cc < bc:
            h, bc, bh = hh, cc, hh.copy()

    # Phase 5: Greedy local search (190s)
    dl = ts + 970
    h, c5 = greedy_ls(h, n, dx, rng, dl, 25000)
    if c5 < bc:
        bc = c5; bh = h.copy()

    # Phase 6: SLSQP polish + ultra-fine (remaining)
    dl = ts + budget_s - 5
    if time.time() < dl - 55:
        hh, cc = slsqp_opt(h, n, 60000, target, 400)
        if cc < bc:
            h, bc, bh = hh, cc, hh.copy()
    h, c6 = greedy_ls(h, n, dx, rng, dl, 15000)
    if c6 < bc:
        bc = c6; bh = h.copy()

    bh = repair(bh, n)
    bc = compute_c5(bh, dx)
    return (bh, bc, n)
