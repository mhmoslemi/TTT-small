            continue
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c:
                c5c = c
                bc = c
                best = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best, bc

def _optimize_n(h0, n, dx, time_lim):
    t_end = time_lim
    bh = h0.copy()
    bc5 = _c5(bh, dx)
    bounds = [(0.0, 1.0)] * n
    target_pts = n / 2.0

    alphas = [200, 800, 3000, 15000, 70000, 300000, 1000000]
    for a in alphas:
        if time.time() > t_end - 40:
            break
        try:
            res = minimize(
                lambda x, al=a: _softobj(x, al, 80.0), bh.copy(),
                jac=lambda x, al=a: _softgrad(x, al, 80.0),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 600, 'ftol': 1e-22, 'gtol': 1e-18})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < bc5:
                bc5 = c5v
                bh = hc
        except Exception:
            pass

    bh, bc5 = _subgrad_loop(bh, n, dx, t_end - 60, n_iter=1200)
    bh, bc5 = _coord_descent(bh, n, dx, t_end - 30, n_sweeps=80)
    bh, bc5 = _random_perturb(bh, n, dx, np.random.RandomState(1), t_end - 5, 10000)
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def remaining():
        return budget_s - elapsed()

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 0

    # Primary: use initial_h_values at its native resolution
    if 'initial_h_values' in dir() or True:
        try:
            n0 = len(initial_h_values)
        except NameError:
            n0 = 140
            initial_h_values = np.full(140, 0.5)
        dx0 = 2.0 / n0
        h0 = _repair(np.array(initial_h_values, dtype=float), n0)
        allocated = remaining() * 0.45
        t_end1 = t0 + allocated
        bh, bc5 = _optimize_n(h0, n0, dx0, t_end1)
        if bc5 < best_c5:
            best_c5, best_h, best_n = bc5, bh, n0

    # Try n=100 from interpolated best
    if remaining() > 200:
        n2 = 100
        dx2 = 2.0 / n2
        x_fine = np.linspace(0, 2, best_n, endpoint=False)
        x_coarse = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_coarse, x_fine, best_h)
        h2 = _repair(h2, n2)
        t_end2 = t0 + elapsed() + remaining() * 0.25
        bh2, bc5_2 = _optimize_n(h2, n2, dx2, t_end2)
        if bc5_2 < best_c5:
            best_c5, best_h, best_n = bc5_2, bh2, n2

    # Try n=140 from scratch with a different start (block pattern)
    if remaining() > 150:
        n3 = 140
        dx3 = 2.0 / n3
        x = np.linspace(0, 2, n3, endpoint=False)
        h3 = np.where(x < 0.5, 1.0, np.where(x < 1.0, 0.3, np.where(x < 1.5, 1.0, 0.3)))
        h3 = _repair(h3, n3)
        # Also try a "two-block" pattern
        h3b = np.where((x < 0.4) | (x > 1.6), 1.0, 0.0)
        h3b = _repair(h3b, n3)
        t_end3 = t0 + elapsed() + min(remaining() * 0.2, 120)
        bh3, bc5_3 = _optimize_n(h3, n3, dx3, t_end3)
        if bc5_3 < best_c5:
            best_c5, best_h, best_n = bc5_3, bh3, n3
        if remaining() > 50:
            t_end3b = min(t_end3 + 60, t0 + budget_s - 30)
            bh3b, bc5_3b = _optimize_n(h3b, n3, dx3, t_end3b)
            if bc5_3b < best_c5:
                best_c5, best_h, best_n = bc5_3b, h3b, n3

    # Final polish on best solution
    if remaining() > 10:
        h = best_h.copy()
        n_f = best_n
        dx_f = 2.0 / n_f
        c5c = _c5(h, dx_f)
        t_end_f = t0 + budget_s - 2
        h, c5c = _subgrad_loop(h, n_f, dx_f, t_end_f - remaining() * 0.3, n_iter=500)
        h, c5c = _coord_descent(h, n_f, dx_f, t_end_f, n_sweeps=50)
        if c5c < best_c5:
            best_c5, best_h = c5c, h

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
