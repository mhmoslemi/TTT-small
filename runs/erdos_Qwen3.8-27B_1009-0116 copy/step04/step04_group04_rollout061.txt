    best_c = _c5(h, dx)
    cur_h = h.copy()

    cooling = 0.9995
    iters = 0

    scale = 0.005
    scale_min = 0.00005
    
    while time.time() < t_end:
        iters += 1
        if iters % 100 == 0:
            T = max(T_min, T * cooling)
            scale = max(scale_min, scale * 0.999)
        
        c = _corr(cur_h)
        max_c = c.max()
        # Find active lags (within 5% of max)
        threshold = max_c - 0.02
        active = np.where(c > threshold)[0] - (n - 1)

        if len(active) == 0:
            active = np.array([np.argmax(c) - (n - 1)])
        lag = int(rng.choice(active))

            # Perturb randomly
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.choice([-1, 1])
            cur_h[i] += d
            cur_h[j] -= d
        else:
            if lag > 0:
                ii = np.arange(n - lag)
            else:
                ii = np.arange(n + lag)
            if len(ii) == 0:
                continue

            j = i + lag
            # Determine direction: reduce h[i](1-h[j]) or h[j](1-h[i])
            if cur_h[i] > cur_h[j]:

            else:
                d = scale * rng.choice([1, -1, -1])  # bias to increase h[i]
            cur_h[i] += d
            cur_h[j] -= d
        
        # Check bounds
        if not (0 <= cur_h[i] <= 1 and 0 <= cur_h[j] <= 1):
            cur_h[i] -= d
            cur_h[j] += d
            continue
        
        # Repair and evaluate
        cur_h = _repair(cur_h, target)
        new_c = _c5(cur_h, dx)
        
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] -= d
            cur_h[j] += d
            cur_h = _repair(cur_h, target)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _soft_obj(x, alpha, target):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha + 100*(x.sum()-target)**2

def _slsqp(h, dx, alpha, maxiter, target):
    n = len(h)
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, target), h,
                     method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

        w /= w.sum()

            lr *= 0.9
        else:
            lr = min(lr * 1.01, 0.01)
        cur_c = new_c

import numpy as np
from scipy.optimize import minimize
import time

_obj_alpha = [1000.0]
_obj_target = [70.0]
_cons_target = [70.0]

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _objective(x):
    c = _corr(x)
    m = c.max()
    s = np.clip(_obj_alpha[0] * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / _obj_alpha[0] + 1e4*(x.sum()-_obj_target[0])**2

def _constraint_fn(x):
    return x.sum() - _cons_target[0]

def _slsqp(h, dx, alpha, maxiter, target):
    _obj_alpha[0] = alpha
    _obj_target[0] = target
    _cons_target[0] = target
    n = len(h)
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(_objective, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': _constraint_fn}],
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _grad_c(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        jj = np.arange(lag, n)
        g[jj] += 1.0 - h[jj - lag]
        ii = np.arange(n - lag)
        g[ii] += -h[ii + lag]
    else:
        m = -lag
        ii = np.arange(n - m)
        g[ii] += 1.0 - h[ii + m]
        jj = np.arange(m, n)
        g[jj] += -h[jj - m]
    return g

def _grad_descend(h, dx, target, t_end, n_active=5, alpha=3000):
    n = len(h)
    best_h = h.copy()
    best_c = _c5(h, dx)
    cur_h = h.copy()
    cur_c = best_c
    lr = 0.0005
    while time.time() < t_end:
        c = _corr(cur_h)
        m = c.max()
        w = np.exp(np.clip(alpha * (c - m), -500, 500))
        w /= w.sum()
        g = np.zeros(n)
        active = np.argsort(c)[-n_active:]
        for ki in active:
            lag = int(ki) - (n - 1)
            g += w[int(ki)] * _grad_c(cur_h, lag)
        cur_h_new = cur_h - lr * g
        cur_h_new = _repair(cur_h_new, target)
        new_c = _c5(cur_h_new, dx)
        if new_c < best_c:
            best_c = new_c
            best_h = cur_h_new.copy()
        if new_c > cur_c:
            lr *= 0.92
        else:
            lr = min(lr * 1.02, 0.005)
        cur_c = new_c
        cur_h = cur_h_new
    return best_h, best_c

            j = rng.randint(n)

def _block_search(h, dx, rng, t_end, target):
    n = len(h)
    best_h = h.copy()
    best_c = _c5(h, dx)
    scale = 0.002
    while time.time() < t_end:
        bl = int(rng.choice([1, 2, 3, 4, 6]))
        if bl >= n // 3:
            continue
        i = rng.randint(n - bl)
        j = rng.randint(n - bl)
        if abs(i - j) < bl:
            continue
        d = scale * rng.choice([-1.0, 1.0])
        old_i = h[i:i+bl].copy()
        old_j = h[j:j+bl].copy()
        h[i:i+bl] += d
        h[j:j+bl] -= d
        if np.any(h < -1e-9) or np.any(h > 1 + 1e-9):
            h[i:i+bl] = old_i
            h[j:j+bl] = old_j
            continue
        h = _repair(h, target)
        new_c = _c5(h, dx)
        if new_c < best_c - 1e-15:
            best_c = new_c
            best_h = h.copy()

            h = best_h.copy()
        if rng.random() < 0.005:
            scale *= 0.9
            if scale < 0.0001:
                scale = 0.001
    return best_h, best_c
