def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:

        for idx in order:
            if diff > 0:

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                v = min(diff, 1.0 - h[idx]); h[idx] += v; diff -= v
            else:
                v = min(-diff, h[idx]); h[idx] -= v; diff += v
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        r = minimize(lse_obj, h, args=(alpha, target, pen),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8})
        hc = repair(r.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def _compute_a(h, n):
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)
    a = np.zeros((n, n_lags))
    for i in range(n):
        vp = (i + lags >= 0) & (i + lags < n)
        a[i, vp] = (1.0 - h[i + lags[vp]])
        vn = (i - lags >= 0) & (i - lags < n)
        a[i, vn] -= h[i - lags[vn]]
    return a, lags

def _update_a(a, h, i, lags, n):
    a[i] = 0.0
    vp = (i + lags >= 0) & (i + lags < n)
    a[i, vp] = (1.0 - h[i + lags[vp]])
    vn = (i - lags >= 0) & (i - lags < n)
    a[i, vn] -= h[i - lags[vn]]

def fast_ls(h, n, rng, deadline, max_iters, batch_size, scale0, flr):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c5r = float(np.max(c))
    h = h.copy()
    a, lags = _compute_a(h, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        prog = float(it) / float(max_iters)
        scale = scale0 * (flr / scale0) ** prog
        scale = max(scale, 1e-5)
        pairs = rng.randint(0, n, (batch_size, 2))
        d = (rng.random(batch_size) * 2.0 - 1.0) * scale
        hi = h[pairs[:, 0]] + d
        hj = h[pairs[:, 1]] - d
        valid = (hi >= 0) & (hi <= 1) & (hj >= 0) & (hj <= 1)
        nv = valid.sum()
        if nv < 10:
            d *= 0.05
            hi = h[pairs[:, 0]] + d
            hj = h[pairs[:, 1]] - d
            valid = (hi >= 0) & (hi <= 1) & (hj >= 0) & (hj <= 1)
            nv = valid.sum()
            if nv < 1:
                it += 1
                continue
        vp = pairs[valid]
        vd = d[valid]
        delta = vd[:, None] * (a[vp[:, 0]] - a[vp[:, 1]])
        c_new = c[None, :] + delta
        max_new = c_new.max(axis=1)
        better = max_new < c5r - 1e-15
        if better.any():
            idxs = np.where(better)[0]
            bi = idxs[np.argmin(max_new[better])]
            gi = vp[bi, 0]; gj = vp[bi, 1]; gd = vd[bi]
            h[gi] += gd; h[gj] -= gd
            c = c_new[bi]
            c5r = float(c.max())
            _update_a(a, h, gi, lags, n)
            _update_a(a, h, gj, lags, n)
        it += 1
    return h, c5r * dx

def block_ls(h, n, dx, rng, deadline, n_iters, bmin, bmax):
    c = np.correlate(h, 1.0 - h, mode='full')
    c5r = float(np.max(c))
    h = h.copy()
    for _ in range(n_iters):
        if time.time() >= deadline:
            break
        bs = rng.randint(bmin, bmax + 1)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.015) * (1.0 if rng.random() < 0.5 else -1.0)
        if np.any(h[i:i+bs] + eps < -1e-9) or np.any(h[i:i+bs] + eps > 1.0 + 1e-9):
            continue
        if np.any(h[j:j+bs] - eps < -1e-9) or np.any(h[j:j+bs] - eps > 1.0 + 1e-9):
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        cn = np.max(np.correlate(h, 1.0 - h, mode='full'))
        if cn < c5r - 1e-14:
            c5r = float(cn)
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, c5r * dx

def perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(n//6, 5*n//6))
    elif kind == 1:
        f = rng.randint(1, 8)
        ph = rng.uniform(0, 2*np.pi)
        hp += rng.uniform(0.02, 0.1) * np.sin(2*np.pi*f*np.arange(n)/n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.03, n)
    elif kind == 3:
        t = np.arange(n)/n
        f1, f2 = rng.randint(1,5), rng.randint(1,5)
        p1, p2 = rng.uniform(0,6.28), rng.uniform(0,6.28)
        hp += rng.uniform(0.02,0.08)*np.sin(2*np.pi*f1*t+p1)*np.cos(2*np.pi*f2*t+p2)
    else:
        hp *= rng.uniform(0.8, 1.2)
    return repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    h = repair(h_init.copy(), n)
    c5 = compute_c5(h, 2.0/n)
    best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 1: L-BFGS polish (25s)
    te = t0 + 25
    for alpha in [10000, 50000, 200000]:
        if time.time() >= te: break
        hh, cc = lbfgs(h, n, alpha, n/2.0, 8000, 400)
        if cc < c5: h, c5 = hh, cc
    if c5 < best_c5: best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 2: Fast local search main (380s)
    te = t0 + 405
    h, c5 = fast_ls(h, n, rng, te, 10**9, 3000, 0.015, 0.00003)
    if c5 < best_c5: best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 3: L-BFGS very high alpha (40s)
    te = t0 + 445
    for alpha in [200000, 1000000]:
        if time.time() >= te: break
        hh, cc = lbfgs(h, n, alpha, n/2.0, 10000, 500)
        if cc < c5: h, c5 = hh, cc
    if c5 < best_c5: best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 4: Fast local search fine (280s)
    te = t0 + 725
    h, c5 = fast_ls(h, n, rng, te, 10**9, 5000, 0.003, 0.00001)
    if c5 < best_c5: best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 5: Block local search (60s)
    te = t0 + 785
    h, c5 = block_ls(h, n, 2.0/n, rng, te, 8000, 2, 5)
    if c5 < best_c5: best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 6: Multi-restart (120s)
    te = t0 + 905
    for r in range(25):
        if time.time() >= te: break
        hp = perturb(h, n, rng, r % 5)
        hh, cc = lbfgs(hp, n, 50000, n/2.0, 8000, 200)
        hh, cc2 = fast_ls(hh, n, rng, time.time()+8, 10**9, 3000, 0.005, 0.00002)
        if cc2 < c5: h, c5 = hh, cc2
        if c5 < best_c5: best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 7: Try n=280 if time (60s)
    if time.time() < t0 + budget_s - 80:
        n2 = 2 * n
        h2 = np.repeat(h, n2 // n).astype(float)
        h2 = repair(h2, n2)
        c5_2 = compute_c5(h2, 2.0/n2)
        te2 = min(t0 + budget_s - 20, time.time() + 50)
        if te2 > time.time() + 5:
            h2, c5_2 = fast_ls(h2, n2, rng, te2, 10**9, 2000, 0.01, 0.00002)
            if c5_2 < best_c5:
                best_h, best_c5, best_n = h2.copy(), c5_2, n2

    # Phase 8: Final ultra-fine (remaining)
    te = t0 + budget_s - 5
    if te > time.time() + 3:
        h, c5 = fast_ls(h, n, rng, te, 10**9, 5000, 0.0005, 0.000005)
        if c5 < best_c5: best_h, best_c5, best_n = h.copy(), c5, n

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
