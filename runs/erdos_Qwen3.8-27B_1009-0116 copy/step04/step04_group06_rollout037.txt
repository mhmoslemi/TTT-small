import numpy as np
from scipy.optimize import linprog
import time

def _c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def _gradients(h, n):
    """Compute gradients of all 2n-1 correlation values w.r.t. h."""
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = k - (n-1)
        if lag == 0:
            grads[k] = (1.0 - 2.0*h) * dx
        elif lag > 0:
            g = np.zeros(n)
            g[:n-lag] += (1.0 - h[lag:]) 
            g[lag:] -= h[:n-lag]
            grads[k] = g * dx
        else:
            l2 = -lag
            g = np.zeros(n)
            g[l2:] += 1.0
            g[l2:] -= h[:n-l2]  # h[k-l2] for k>=l2 → h[0:n-l2]
            g[:n-l2] -= h[l2:]  # h[k+l2] for k<=n-1-l2 → h[l2:n]
            grads[k] = g * dx
    return grads

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _mirror_descent(h, n, t_end, lr0, alpha):
    h = h.copy()
    dx = 2.0 / n
    c5, _ = _c5(h, n)
    best_h, best_c5 = h.copy(), c5
    lr = lr0
    stall = 0
    for it in range(100000):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0-h, mode='full') * dx
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -200, 200)
        e = np.exp(s)
        w = e / e.sum()
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-10:
                continue
            lag = k - (n-1)
            if lag == 0:
                g += wk * (1.0 - 2.0*h)
            elif lag > 0:
                g[:n-lag] += wk * (1.0 - h[lag:])
                g[lag:] -= wk * h[:n-lag]
            else:
                l2 = -lag
                g[l2:] += wk
                g[l2:] -= wk * h[:n-l2]
                g[:n-l2] -= wk * h[l2:]
        h_new = _project(h - lr * g * dx, n)

def _lp_polish(h, n, t_end, max_rounds=20):
    """LP-based cutting plane polish."""
    h = h.copy()
    c5, corr = _c5(h, n)
    best_h, best_c5 = h.copy(), c5
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        # Compute gradients
        grads = _gradients(h, n)
        # Build LP
        c_obj = np.zeros(n+1)
        c_obj[-1] = 1.0  # minimize t
        A_ub = np.zeros((2*n-1, n+1))
        b_ub = np.zeros(2*n-1)
        for k in range(2*n-1):
            A_ub[k, :n] = grads[k]
            A_ub[k, n] = -1.0
            intercept = corr[k] - grads[k] @ h
            b_ub[k] = -intercept
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        bounds = [(0,1)]*n + [(0, None)]
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res is None or res.status not in (0, 2):
            break
        h_new = res.x[:n]
        t_new = res.x[n]
        # Line search from h to h_new
        best_local_c5 = c5

                h_best_local = h_try
        if best_local_c5 < c5 - 1e-14:
            h = h_best_local
            c5 = best_local_c5
            corr = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            break
    return best_h, best_c5

            else:
                l2 = k-(n-1)
                g[:n-l2] += wk*(1.0-h[l2:])
                g[l2:] -= wk*h[:n-l2]
        g *= (2.0/n)
        h_new = _project(h - lr*g, n)
        c5_new, _ = _c5(h_new, n)
        if c5_new < c5 - 1e-15:
            h, c5 = h_new, c5_new
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
            stall = 0
        else:
            lr *= 0.93
            stall += 1
            if stall > 400:
                lr = max(lr, 0.0008); stall = 0
        if it % 800 == 799:
            lr *= 0.75
    return best_h, best_c5

def _lp_polish(h, n, t_end, max_rounds=25):
    h = h.copy()
    c5, corr = _c5(h, n)
    best_h, best_c5 = h.copy(), c5
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        grads = _gradients(h, n)
        c_obj = np.zeros(n+1); c_obj[-1] = 1.0
        A_ub = np.zeros((2*n-1, n+1))
        b_ub = np.zeros(2*n-1)
        for k in range(2*n-1):
            A_ub[k, :n] = grads[k]
            A_ub[k, n] = -1.0
            b_ub[k] = -(corr[k] - grads[k] @ h)
        A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        bounds = [(0,1)]*n + [(0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if res is None or res.status not in (0, 2):
            break
        h_dir = res.x[:n]
        improved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h_try = _project(h + frac*(h_dir - h), n)
            c_try, _ = _c5(h_try, n)
            if c_try < c5 - 1e-14:
                h, c5 = h_try, c_try
                corr = np.correlate(h, 1.0-h, mode='full')*(2.0/n)
                improved = True
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
                break
        if not improved:
            break
    return best_h, best_c5

def _local_search(h, n, t_end, rng):
    h = h.copy()
    c5, _ = _c5(h, n)
    best_c5 = c5
    deltas = np.concatenate([np.array([0.001,0.003,0.01,0.03,0.06,0.1]),
                             -np.array([0.001,0.003,0.01,0.03,0.06,0.1])])
    while time.time() < t_end:
        improved = False
        perm = np.arange(n); rng.shuffle(perm)
        for i in perm:
            if time.time() > t_end:
                return h, best_c5
            hi = h[i]
            for j in perm:
                if i == j:
                    continue
                hj = h[j]
                for d in deltas:
                    ni, nj = hi+d, hj-d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c, _ = _c5(h, n)
                    if c < c5 - 1e-13:
                        c5, hi, hj = c, ni, nj
                        improved = True
                        if c < best_c5:
                            best_c5 = c
                        break
                    else:
                        h[i], h[j] = hi, hj
        if not improved:
            break
    return h, best_c5

def _interpolate(h_old, n_old, n_new):
    xo = np.arange(n_old)*2.0/n_old
    xn = np.arange(n_new)*2.0/n_new
    return np.interp(xn, xo, h_old, period=2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float), n0)
    best_h, best_n, best_c5 = h0.copy(), n0, _c5(h0, n0)[0]

    def elapsed():
        return time.time() - t0

    # Phase 1: Mirror descent with corrected gradients
    t1 = t0 + budget_s * 0.22
    for alpha in [100, 500, 2000, 8000]:
        if elapsed() > t1:
            break
        h, c5 = _mirror_descent(best_h, best_n, t1, 0.06, alpha)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: LP-based polishing
    t2 = t0 + budget_s * 0.48
    h, c5 = _lp_polish(best_h, best_n, t2, max_rounds=30)
    if c5 < best_c5:
        best_h, best_c5 = h, c5

    # Phase 3: Local search
    t3 = t0 + budget_s * 0.65
    h, c5 = _local_search(best_h, best_n, t3, rng)
    if c5 < best_c5:
        best_h, best_c5 = h, c5

    # Phase 4: Another LP + local pass
    if elapsed() < budget_s * 0.8:
        h, c5 = _lp_polish(best_h, best_n, t0+budget_s*0.78, max_rounds=15)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h, c5 = _local_search(best_h, best_n, t0+budget_s*0.85, rng)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 5: Try other resolutions
    if elapsed() < budget_s * 0.9:
        for n_try in [100, 120, 160, 200]:
            if elapsed() > budget_s * 0.92:
                break
            if n_try == best_n:
                continue
            ht = _repair(_interpolate(best_h, best_n, n_try), n_try)
            ht, ct = _mirror_descent(ht, n_try, min(t0+budget_s*0.92, t0+25), 0.04, 800)
            ht, ct = _local_search(ht, n_try, min(t0+budget_s*0.94, t0+12), rng)
            if ct < best_c5:
                best_h, best_n, best_c5 = ht, n_try, ct

    # Phase 6: Final micro-search
    if elapsed() < budget_s - 15:
        h, c5 = _lp_polish(best_h, best_n, t0+budget_s-8, max_rounds=8)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        tiny = np.array([0.0002, 0.0005, 0.001, -0.0002, -0.0005, -0.001])
        h = best_h.copy(); c5 = best_c5
        end_t = t0 + budget_s - 3
        while time.time() < end_t:
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i]+d, h[j]-d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c, _ = _c5(h, best_n)
                if c < c5:
                    c5 = c
                    if c < best_c5:
                        best_c5, best_h = c, h.copy()
                else:
                    h[i], h[j] = ni-d, nj+d

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)[0]
    return (best_h, best_c5, best_n)
