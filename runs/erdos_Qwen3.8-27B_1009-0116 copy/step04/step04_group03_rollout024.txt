                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:

            if scale < 5e-7: break
    return h, c5c

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _tl(): return budget_s - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h_init)
    h_init = _repair(h_init, n0)
    best_h = h_init.copy()
    best_c5 = _eval(h_init)
    best_n = n0

    # Multi-stage SLSQP on original resolution
    alphas = [500, 5000, 50000, 200000, 1000000]
    h_cur = h_init.copy()
    for alpha in alphas:
        if _tl() < 80: break
        mit = 600 if n0 <= 100 else 400
        h_new = _slsqp(h_cur, n0, alpha, maxiter=mit)
        c5_new = _eval(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new; best_h = h_new.copy()
        h_cur = h_new.copy()

    # Active descent refinement
    if _tl() > 100:
        t_end = time.time() + min(90, _tl() - 20)
        h_ad, c5_ad = _active_descent(best_h.copy(), best_n, t_end, lr=0.003)
        if c5_ad < best_c5:
            best_c5 = c5_ad; best_h = h_ad.copy()

    # Try other resolutions
    for n_try in [100, 120, 140, 180]:
        if _tl() < 120: break
        h_t = _repair(_interp(best_h, best_n, n_try), n_try)
        ht_c5 = _eval(h_t)
        for alpha in [100000, 500000]:
            if _tl() < 60: break
            h_t = _slsqp(h_t, n_try, alpha, maxiter=350)
            ht_c5 = _eval(h_t)
        if ht_c5 < best_c5:
            best_c5 = ht_c5; best_h = h_t.copy(); best_n = n_try

    # Local search
    if _tl() > 60:
        t_end = time.time() + min(50, _tl() - 10)
        h_ls, c5_ls = _local_search(best_h.copy(), best_n, rng, t_end)
        if c5_ls < best_c5:
            best_c5 = c5_ls; best_h = h_ls.copy()

    # Final SLSQP polish
    if _tl() > 30:
        h_f = _slsqp(best_h.copy(), best_n, 2000000, maxiter=300)
        c5_f = _eval(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f; best_h = h_f.copy()

    # Fine local search
    if _tl() > 20:
        t_end = time.time() + min(15, _tl() - 5)
        h_fs, c5_fs = _local_search(best_h.copy(), best_n, rng, t_end)
        if c5_fs < best_c5:
            best_c5 = c5_fs; best_h = h_fs.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
