    return float(np.max(_corr(h)) * dx)

    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break

    d = target - h.sum()
    if abs(d) > 1e-12:

        for i in idx:
            if d > 0:

                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_lag(h, m):
    """Vectorized gradient of c[m] w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if m >= 0:
        L = n - m  # number of terms
        if L <= 0:
            return g
        # First part: (1-h[i+m]) for i in [0, L-1]
        g[:L] += (1.0 - h[m:m+L])
        # Second part: -h[i-m] for i in [m, n-1]
        g[m:] += -h[:L]
    else:
        m2 = -m
        L = n - m2
        if L <= 0:
            return g
        # First part: (1-h[i+m]) = (1-h[i-m2]) for i in [m2, n-1]
        g[m2:] += (1.0 - h[:L])
        # Second part: -h[i-m] = -h[i+m2] for i in [0, L-1]
        g[:L] += -h[m2:]
    return g

def _active_subgrad(h, dx, top_k, n_iters, lr_init, t_end, rng):
    """Projected subgradient on top-k active lags"""
    n = len(h)
    target = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr = lr_init
    stalled = 0
    for it in range(n_iters):
        if time.time() > t_end:
            break
        c = _corr(h)
        # Get top-k lags (excluding lag 0 which is always small for good solutions)
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute weighted gradient
        top_vals = c[top_idx]
        w = np.exp((top_vals - top_vals.max()) * 10)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, mi in zip(w, top_idx):
            grad += wi * _grad_lag(h, int(mi) - (n - 1))

        hn = _repair(hn, target)

            h = hn
            stalled = 0

                stalled = 0
    return h, best

        h0, c = _slsqp_polish(h0, dx0, a, 300)
        if c < best_c5:
            best_h, best_c5 = h0, c
    
    # Phase 3: Informed search at n0 (60s)
    t_end = t0 + 290
    h0, c = _informed_search(h0, dx0, rng, t_end, 100000)
    if c < best_c5:
        best_h, best_c5 = h0, c
    
    # Phase 4: Multi-restart at n0 (120s)
    t_end = t0 + 410
    for r in range(6):
        if time.time() > t_end or rem() < 400:
            break
        amp = 0.06 * (1.0 - r / 8.0)
        hp = best_h + amp * (2.0 * rng.random(n0) - 1.0)
        hp = _repair(hp, n0 / 2.0)
        hp, c = _subgrad(hp, dx0, 5, 3000, 0.003, t_end, rng)
        hp, c = _slsqp_polish(hp, dx0, 20000, 300)
        if c < best_c5:
            best_h, best_c5 = hp, c
    
    # Phase 5: Interpolate to 280, optimize (250s)
    n1 = 280
    dx1 = 2.0 / n1
    h1 = _repair(_interp(best_h, n1), n1 / 2.0)
    t_end = min(t0 + 660, t0 + budget_s - 250)
    h1, c = _subgrad(h1, dx1, 5, 5000, 0.003, t_end, rng)
    h1, c = _lbfgs_run(h1, dx1, 3000, 200, t_end)
    for a in [5000, 30000]:
        if time.time() > t_end:
            break
        h1, c = _slsqp_polish(h1, dx1, a, 300)
    if c < best_c5:
        best_h, best_c5, best_n = h1, c, n1
    
    # Phase 6: Informed search at best resolution (100s)
    nb = best_n
    dxb = 2.0 / nb
    t_end = t0 + budget_s - 130
    best_h, best_c5 = _informed_search(best_h, dxb, rng, t_end, 150000)
    
    # Phase 7: Final subgradient + SLSQP (80s)
    t_end = t0 + budget_s - 50
    best_h, c = _subgrad(best_h, dxb, 3, 4000, 0.001, t_end, rng)
    if c < best_c5:
        best_c5 = c
    if rem() > 30:
        best_h, c = _slsqp_polish(best_h, dxb, 100000, 300)
        if c < best_c5:
            best_c5 = c
    
    # Phase 8: Final micro-search (30s)
    t_end = t0 + budget_s - 5
    best_h, best_c5 = _informed_search(best_h, dxb, rng, t_end, 50000)
    
    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
