    best_h, best_c = h0.copy(), c5v(h0, dx)
    bounds = [(0.0, 1.0)] * n
    def de_obj(x):
        return c5v(repair(x, n), dx)
    try:
        r = differential_evolution(de_obj, bounds, seed=seed, maxiter=60,
                                   popsize=22, tol=1e-7, mutation=(0.5, 1.0),
                                   recombination=0.85, init='latinhypercube',
                                   polish=False)
        hc = repair(r.x, n)
        c = c5v(hc, dx)
        if c < best_c:
            best_h, best_c = hc, c
    except Exception:
        pass
    return best_h, best_c

def smooth_opt(h, n, dx, t0, lim):
    _GR['n'] = n
    bds = [(0.0, 1.0)] * n
    bh, bc = h.copy(), c5v(h, dx)
    schedule = [(20, 30), (80, 150), (400, 600), (2000, 3000),
                (10000, 15000), (50000, 70000), (200000, 300000),
                (800000, 1000000), (3000000, 4000000)]
    for a, lm in schedule:
        if time.time() - t0 > lim:
            break
        _GR['a'], _GR['l'] = float(a), float(lm)
        try:
            r = minimize(_sm_obj, bh.copy(), jac=_sm_jac, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': 600, 'ftol': 1e-20,
                                              'gtol': 1e-17, 'maxls': 50})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc - 1e-15:
                bc, bh = c, hc
        except Exception:
            pass
        if time.time() - t0 > lim:
            break
        _GR['a'], _GR['l'] = float(a), float(lm)
        try:
            r = minimize(_sm_obj, bh.copy(), jac=_sm_jac, method='SLSQP',
                         bounds=bds,
                         constraints=[{'type': 'eq', 'fun': _se_fun, 'jac': _se_jac}],
                         options={'maxiter': 400, 'ftol': 1e-19})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc - 1e-15:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def active_opt(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for mt in [3, 5, 8, 12, 20, 30]:
        h = bh.copy()
        step = 0.04
        stalled = 0
        for _ in range(500):
            if time.time() - t0 > lim:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            cm = c.max()
            thr = cm - 0.0015 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            g = np.zeros(n)
            for ki in idx:
                g += grad1(h, ki, n)
            g /= len(idx)
            g -= g.mean()
            ok = False
            for _b in range(25):
                hn = repair(h - step * g, n)
                cn = c5v(hn, dx)
                if cn < c5v(h, dx) - 1e-16:
                    h = hn
                    ok = True
                    if cn < bc:
                        bc, bh = cn, h.copy()
                    step = min(step * 1.08, 2.0)
                    stalled = 0
                    break
                step *= 0.72
            if not ok:
                stalled += 1
                if stalled > 5:
                    break
    return bh, bc

def coord_cd(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    halts = 0
    dl = np.array([0.004, 0.0015, 0.0006, -0.004, -0.0015, -0.0006])
    for _ in range(200):
        if time.time() - t0 > lim:
            break
        imp = False
        for _ in range(400):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            for d in dl:
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = c5v(h, dx)
                    if c < cc - 1e-15:
                        h = repair(h, n)
                        cc = c5v(h, dx)
                        imp = True
                        if cc < bc:
                            bc, bh = cc, h.copy()
                        break
                    h[i], h[j] = ni - d, nj + d
        halts = halts + 1 if not imp else 0
        if halts >= 5:
            break
    return bh, bc

def multi_start(h0, n, dx, seed, t0, lim, n_starts=4):
    best_h, best_c = h0.copy(), c5v(h0, dx)
    rng = np.random.RandomState(seed)
    for s in range(n_starts):
        if time.time() - t0 > lim:
            break
        frac = (lim - (time.time() - t0)) / (n_starts - s)
        if frac < 30:
            break
        if s == 0:
            h_s = h0.copy()
        elif s == 1:
            h_s = repair(h0 + 0.08 * rng.randn(n), n)
        elif s == 2:
            h_s = repair(h0 + 0.15 * rng.randn(n), n)
        else:
            h_s = repair(rng.uniform(0, 1, n), n)
        r, c = smooth_opt(h_s, n, dx, t0, t0 + frac * 0.7)
        r2, c2 = active_opt(r, n, dx, t0, t0 + frac * 0.9)
        if c2 < best_c:
            best_h, best_c = r2, c2
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    # Phase 1: DE global exploration
    p1_lim = t0 + 0.08 * (budget_s - 5)
    r, c = de_global(h0, nl, dx0, seed, t0, p1_lim)
    if c < bc:
        bh, bc = r, c

    # Phase 2: Multi-start local optimization
    p2_lim = t0 + 0.45 * (budget_s - 5)
    r, c = multi_start(bh, nl, dx0, seed + 7, t0, p2_lim, n_starts=3)
    if c < bc:
        bh, bc = r, c

    # Phase 3: Deep smooth optimization on best
    p3_lim = t0 + 0.72 * (budget_s - 5)
    r, c = smooth_opt(bh, nl, dx0, t0, p3_lim)
    if c < bc:
        bh, bc = r, c

    # Phase 4: Active set refinement
    p4_lim = t0 + 0.85 * (budget_s - 5)
    r, c = active_opt(bh, nl, dx0, t0, p4_lim)
    if c < bc:
        bh, bc = r, c

    # Phase 5: Another smooth pass
    p5_lim = t0 + 0.93 * (budget_s - 5)
    r, c = smooth_opt(bh, nl, dx0, t0, p5_lim)
    if c < bc:
        bh, bc = r, c

    # Phase 6: Coordinate descent polish
    p6_lim = t0 + 0.99 * (budget_s - 5)
    rng = np.random.RandomState(seed + 100)
    r, c = coord_cd(bh, nl, dx0, rng, t0, p6_lim)
    if c < bc:
        bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
