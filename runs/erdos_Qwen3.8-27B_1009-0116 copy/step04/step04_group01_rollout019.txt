                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_single(k, h):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(s, n)
        g[i1] += (1.0 - h[i1 - s])
        if s > 0:
            i2 = np.arange(n - s)
            g[i2] -= h[i2 + s]
    else:
        sp = -s
        i1 = np.arange(n - sp)
        g[i1] += (1.0 - h[i1 + sp])
        i2 = np.arange(n - sp, n)
        g[i2] -= h[i2 - sp]
    return g

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = _corrs_raw(h) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corrs_raw(h) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _cfun(x, n):
    return x.sum() - n * 0.5

def _cjac(x, n):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': lambda x: _cfun(x, n),
                                     'jac': lambda x: _cjac(x, n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent(h, topm, max_iter, step0=0.005):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - 20:
            break
        c = _corrs_raw(h)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_single(k, h)
        g /= topm
        g *= (2.0 / n)
        found = False
        for _bt in range(35):
            h_new = _proj_to_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-6 * step * np.dot(g, h_new - h):
                h = h_new; f = f_new; found = True
                break
            step *= 0.5
        if not found:
            break
        step = min(step * 1.08, step0)
    return h, f

def _round_robin(h, n_lags, steps_per_lag):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    for _ in range(n_lags):
        if time.time() - _t0 > _budget - 20:
            break
        c = _corrs_raw(h)
        k_star = c.argmax()
        for _s in range(steps_per_lag):
            g = _grad_single(k_star, h) * (2.0 / n)
            found = False
            for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
                h_new = _proj_to_sum(h - step * g, target)
                f_new = _eval(h_new)
                if f_new < f - 1e-14:
                    h = h_new; f = f_new; found = True
                    break
            if not found:
                break
    return h, f

def _upsample_spline(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def _pairwise(h, n_iter, scale, seed_val):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 8:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def _el():
        return time.time() - _t0

    # Phase A: SLSQP at n0
    for alpha in [5000, 50000, 200000]:
        if _el() > 50:
            break
        hn = _slsqp(best_h, alpha, 1500, 1e-18)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5; best_h = hn.copy()

    # Phase B: Upsample 2x with spline, optimize
    n1 = n0 * 2
    h2 = _upsample_spline(best_h, 2)
    if _el() < budget_s - 150:
        for alpha in [10000, 80000]:
            if _el() > budget_s - 120:
                break
            hn = _slsqp(h2, alpha, 2000, 1e-18)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5; best_h = hn.copy(); best_n = n1
        h2 = best_h.copy()

    # Phase C: Active-set descent at best resolution
    nc = len(best_h)
    if _el() < budget_s - 180:
        for topm in [3, 5, 8, 12, 20]:
            if _el() > budget_s - 150:
                break
            hn, fc = _active_descent(best_h, topm, 4000, 0.004)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy(); best_n = nc

    # Phase D: Round-robin on individual lags
    if _el() < budget_s - 120:
        for _rr in range(5):
            if _el() > budget_s - 100:
                break
            hn, fc = _round_robin(best_h, 40, 8)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase E: High-alpha SLSQP polish
    if _el() < budget_s - 80:
        hn = _slsqp(best_h, 500000, 2500, 1e-19)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5; best_h = hn.copy()
        if _el() < budget_s - 50:
            hn = _slsqp(best_h, 1000000, 1500, 1e-20)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5; best_h = hn.copy()

    # Phase F: Multi-start
    nc = len(best_h)
    if _el() < budget_s - 70:
        for _r in range(12):
            if _el() > budget_s - 50:
                break
            hp = best_h + 0.008 * rng.randn(nc)
            hp = _repair(hp, nc)
            hn = _slsqp(hp, 200000, 1000, 1e-18)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5; best_h = hn.copy()

    # Phase G: Active descent again (post-perturbation)
    if _el() < budget_s - 40:
        hn, fc = _active_descent(best_h, 15, 3000, 0.003)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    # Phase H: Fine pairwise rounds
    for scale, nit in [(0.003, 8000), (0.001, 10000), (0.0003, 12000), (0.0001, 10000)]:
        if _el() > budget_s - 10:
            break
        hn, fc = _pairwise(best_h, nit, scale, seed + int(scale * 1e6))
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    # Phase I: Final active + round-robin polish
    if _el() < budget_s - 8:
        hn, fc = _active_descent(best_h, 10, 2000, 0.002)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()
        hn, fc = _round_robin(best_h, 20, 5)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
