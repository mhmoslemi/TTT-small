Let me vectorize the inner loop:

For lag >= 0:
- t[j] += (1-x[j-lag]) for j in [lag, n-1]  → t[lag:] += (1-x[:n-lag])
- t[j] -= x[j+lag] for j in [0, n-1-lag]  → t[:n-lag] -= x[lag:]

- t[j] -= x[j-l2] for j in [l2, n-1]  → t[l2:] -= x[:n-l2]

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 140
_G_ALPHA = 1e5

def _c5_val(h):
    n = _G_N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort((1.0 - h) if diff > 0 else h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-13:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            np.clip(hp, 0.0, 1.0, out=hp)
    return hp

def _smooth_obj(x):
    n = _G_N
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    a = _G_ALPHA
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _smooth_grad(x):
    n = _G_N
    dx = 2.0 / n
    a = _G_ALPHA
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-15:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dx

def _con_fun(x):
    return float(np.sum(x) - _G_N / 2.0)

def _do_slsqp(h0, alpha, maxiter):
    global _G_ALPHA
    _G_ALPHA = alpha
    bounds = [(0.0, 1.0)] * _G_N
    cons = [{'type': 'eq', 'fun': _con_fun}]
    res = minimize(_smooth_obj, h0, jac=_smooth_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-22})
    return res.x

def _sa_optimize(h, n, rng, t_end):
    global _G_N
    _G_N = n
    c5_cur = _c5_val(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = 0.008
    cool = 0.99992
    h = h.copy()
    stall = 0
    while time.time() < t_end:
        if T < 0.0003:
            T = 0.006
            cool = 0.99992
        T *= cool
        r = rng.random()
        if r < 0.65:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d = rng.uniform(-0.025, 0.025)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            oi, oj = h[i], h[j]
            h[i], h[j] = ni, nj
            c5_new = _c5_val(h)
            dE = c5_new - c5_cur
            if dE < -1e-15:
                c5_cur = c5_new
                stall = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            elif dE > 0 and rng.random() > np.exp(-dE / max(T, 1e-12)):
                h[i], h[j] = oi, oj
                stall += 1
            else:
                c5_cur = c5_new
                stall += 1
        elif r < 0.85:

            j = rng.randint(n)
            k = rng.randint(n)

            if min(ni, nj, nk) < 0 or max(ni, nj, nk) > 1:

            oi, oj, ok_ = h[i], h[j], h[k]
            h[i], h[j], h[k] = ni, nj, nk
            c5_new = _c5_val(h)
            dE = c5_new - c5_cur
            if dE < -1e-15:
                c5_cur = c5_new
                stall = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            elif dE > 0 and rng.random() > np.exp(-dE / max(T, 1e-12)):
                h[i], h[j], h[k] = oi, oj, ok_
                stall += 1
            else:
                c5_cur = c5_new
                stall += 1
        else:
            i, j = rng.randint(n, 2)
            if i == j:
                continue
            oi, oj = h[i], h[j]
            h[i], h[j] = oj, oi
            c5_new = _c5_val(h)
            dE = c5_new - c5_cur
            if dE < -1e-15:
                c5_cur = c5_new
                stall = 0

            elif dE > 0 and rng.random() > np.exp(-dE / max(T, 1e-12)):

        if stall > 5000:
            h = best_h.copy()
            c5_cur = best_c5
            T = 0.01
            stall = 0
    return best_h, best_c5

def _coord_descent(h, n, rng, t_end, lr):
    global _G_N
    _G_N = n
    c5_cur = _c5_val(h)
    best_h = h.copy()
    best_c5 = c5_cur
    h = h.copy()
    steps = np.array([0.001, 0.003, 0.01, 0.03, 0.0005, 0.0002,
                      -0.001, -0.003, -0.01, -0.03, -0.0005, -0.0002])
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        for d in steps:
            ni, nj = hi + d, hj - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5_new = _c5_val(h)
            if c5_new < c5_cur - 1e-16:
                c5_cur = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                h[i], h[j] = hi, hj
    return best_h, best_c5

def _multi_res(h, n, rng, t_end):
    global _G_N
    best_h, best_c5, best_n = h.copy(), _c5_val(h), n
    for n_try in [70, 100, 200, 170]:
        if time.time() > t_end:
            break
        _G_N = best_n
        x_old = np.linspace(0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h, period=2.0)
        h_new = _repair(h_new, n_try)
        _G_N = n_try
        h_wr = _project(h_new, n_try)
        try:
            res = _do_slsqp(h_wr, 2e5, 300)
            hc = _repair(res.x, n_try)
            c5 = _c5_val(hc)
            if c5 < best_c5 - 1e-7:
                best_c5 = c5
                best_h = hc
                best_n = n_try
        except Exception:
            pass
    _G_N = best_n
    return best_h, best_c5, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
```
