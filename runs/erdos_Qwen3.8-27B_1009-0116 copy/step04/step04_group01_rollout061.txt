    A = (c_hi - c_lo) / (hi_t - lo_t)
    B = c_lo - A * lo_t
    a, b = lo_t, hi_t
    for _ in range(45):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(A * m1 + B) <= np.max(A * m2 + B):
            b = m2
        else:
            a = m1
    t = 0.5 * (a + b)
    h[i] = t
    h[j] = S - t

def _grad_c_all(h):
    """Compute all 2n-1 gradients of c_k w.r.t. h, shape (2n-1, n)"""
    n = len(h)
    dx = 2.0 / n
    r = 2 * n - 1
    grads = np.zeros((r, n))
    h1 = 1.0 - h
    for k in range(r):
        sh = (n - 1) - k
        if sh >= 0:
            L = n - sh
            grads[k, :L] += dx * h1[sh:sh+L]
            if sh > 0:
                grads[k, sh:sh+L] -= dx * h[:L]
        else:
            sp = -sh
            L = n - sp
            grads[k, sp:sp+L] += dx * h1[:L]
            grads[k, :L] -= dx * h[sp:sp+L]
    return grads

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -50, 50)
    ew = np.exp(s - s.max())
    w = ew / ew.sum()
    # Only use top weights for speed
    top_k = 20
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    h1 = 1.0 - h
    for k in idx:
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            L = n - sh
            g[:L] += wk * h1[sh:sh+L]
            if sh > 0:
                g[sh:sh+L] -= wk * h[:L]
        else:
            sp = -sh
            L = n - sp
            g[sp:sp+L] += wk * h1[:L]
            g[:L] -= wk * h[sp:sp+L]
    return g

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Intensive random pairwise optimization
    stall = 0
    for it in range(500000):
        if _el() > budget_s * 0.45:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            _pair_opt(h, i, j)
        if it % 3000 == 0:
            c5 = _eval(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 3000
            if stall > 60000:
                # Escape: random perturbation
                h = _repair(h + 0.02 * rng.randn(n), n)
                stall = 0
    c5 = _eval(h)
    if c5 < best_c5:
        best_c5 = c5; best_h = h.copy()

    # Phase 2: Smooth gradient descent
    if _el() < budget_s * 0.65:
        h2 = best_h.copy()
        alpha = 3000.0
        step = 0.02
        for it in range(4000):
            if _el() > budget_s * 0.65:
                break
            g = _smooth_grad(h2, alpha)
            h_new = _proj_sum(h2 - step * g, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < _eval(h2) - 1e-16:
                h2 = h_new
                step = min(step * 1.1, 0.05)
            else:
                step *= 0.5
            if it % 200 == 0:
                c5 = _eval(h2)
                if c5 < best_c5:
                    best_c5 = c5; best_h = h2.copy()
            alpha *= 1.005
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2.copy()

    # Phase 3: Second round pairwise (on improved solution)
    if _el() < budget_s * 0.88:
        h = best_h.copy()
        stall = 0
        for it in range(400000):
            if _el() > budget_s * 0.88:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                _pair_opt(h, i, j)
            if it % 3000 == 0:
                c5 = _eval(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5; best_h = h.copy(); stall = 0
                else:
                    stall += 3000
                if stall > 50000:
                    h = _repair(h + 0.01 * rng.randn(n), n)
                    stall = 0
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()

    # Phase 4: Systematic full sweep
    if _el() < budget_s * 0.96:
        h = best_h.copy()
        for sweep in range(8):
            if _el() > budget_s * 0.96:
                break
            for i in range(n):
                for j in range(i + 1, n):
                    _pair_opt(h, i, j)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5 = c5; best_h = h.copy()

    # Phase 5: Final micro-optimization
    if _el() < budget_s - 5:
        h = best_h.copy()
        for it in range(200000):
            if _el() > budget_s - 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                _pair_opt(h, i, j)
            if it % 5000 == 0:
                c5 = _eval(h)
                if c5 < best_c5:
                    best_c5 = c5; best_h = h.copy()
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
