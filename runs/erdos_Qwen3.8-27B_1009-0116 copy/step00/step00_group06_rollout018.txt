    h = h.copy()
    # Bisection on lambda: x_i = clip(h_i - lambda, 0, 1)
    lo, hi = float(np.max(h) - target_sum) - 1.0, float(np.min(h)) + 1.0

        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Small correction

    if abs(diff) > 1e-10:

        if n_pos > 0:
            mask = (x < 1.0) & (x > 0.0)
            x[mask] += diff / n_pos
            x = np.clip(x, 0.0, 1.0)
    return x

    """Compute (1/beta)*log(sum(exp(beta*c)))"""
    m = np.max(c)
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

    """Compute softmax weights for the smooth max"""
    shifted = beta * c
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w /= np.sum(w)
    return w

    c_k = sum_{i valid} h[i] * (1 - h[i + k - n + 1])
