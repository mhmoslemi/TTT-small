def project_box_sum(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} via bisection on lambda."""
    lo = -np.max(h) - 0.01
    hi = np.max(1.0 - h) + 0.01

            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

    ev = np.exp(corr - m)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    if abs(h.sum() - target) < 1e-14:
        return np.clip(h, 0, 1)
    lo = -np.max(h) - 0.1
    hi = np.max(1.0 - h) + 0.1
    for _ in range(90):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def smooth_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    ev = np.exp(c - m)
    return m + np.log(ev.sum()) / t

def smooth_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    om = 1.0 - h
    c = np.correlate(h, om, mode='full') * dx
    m = c.max()
    ev = np.exp(t * (c - m))
    p = ev / ev.sum()
    g = np.zeros(n)
    for L in range(-(n - 1), n):
        k = L + n - 1
        w = p[k] * dx
        if L == 0:
            g += w * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n - L] += w * om[L:]
            g[L:] -= w * h[:n - L]
        else:
            M = -L
            g[M:] += w * om[:n - M]
            g[:n - M] -= w * h[M:]
    return g

def interp_parent(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    xp = np.linspace(0, 2, n, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return project_box_sum(np.interp(xn, xp, h), n_new / 2.0)

def binary_local_search(h_init, time_limit=10.0):
    n = len(h_init)
    ts = int(n / 2)
    x = (h_init >= 0.5).astype(float)
    ones = np.where(x == 1.0)[0]
    zeros = np.where(x == 0.0)[0]
    while len(ones) > ts:
        x[ones[-1]] = 0.0
        ones, zeros = np.where(x == 1.0), np.where(x == 0.0)
    while len(ones) < ts:
        x[zeros[-1]] = 1.0
        ones, zeros = np.where(x == 1.0), np.where(x == 0.0)
    best = compute_c5(x)
    t_s = time.time()
    stall = 0
    for _ in range(5000):
        if time.time() - t_s > time_limit:
            break
        improved = False
        ones = np.where(x == 1.0)[0]
        zeros = np.where(x == 0.0)[0]
        n_try = min(40, max(10, int(time_limit / 0.002 * n / (n * n + 1))))
        for _tr in range(n_try):
            i = zeros[np.random.randint(len(zeros))]
            j = ones[np.random.randint(len(ones))]
            x[j], x[i] = 0.0, 1.0
            v = compute_c5(x)
            if v < best - 1e-14:
                best = v
                improved = True
                ones, zeros = np.where(x == 1.0), np.where(x == 0.0)
                stall = 0
                break
            x[j], x[i] = 1.0, 0.0
        if not improved:
            stall += 1
            if stall > 30:
                break
    return x, best

def gradient_phase(h, n, ts, time_limit=60.0):
    t_s = time.time()
    hr = h.copy()
    t_sm = 200.0
    lr = 1.0
    for _ in range(2000):
        if time.time() - t_s > time_limit:
            break
        g = smooth_grad(hr, t_sm)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        a = lr
        fcur = smooth_obj(hr, t_sm)
        accepted = False
        for _ in range(50):
            cand = project_box_sum(hr - a * g, ts)
            fnew = smooth_obj(cand, t_sm)
            if fnew < fcur - 1e-15 * a * gn:
                hr = cand
                fcur = fnew
                accepted = True
                break
            a *= 0.4
        if not accepted:
            lr *= 0.5
            if lr < 1e-8:
                break
        if _ % 50 == 0:
            t_sm = max(50.0, t_sm * 0.95)
    return hr

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    if 'initial_h_values' in globals():
        hb = np.array(initial_h_values, dtype=float).copy()
    else:
        hb = np.full(47, 0.5)
    n0 = len(hb)
    hb = project_box_sum(hb, n0 / 2.0)
    cb = compute_c5(hb)

    # Phase 1: SLSQP at parent resolution

        ts0 = n0 / 2.0
        try:

                           bounds=bnds,
                           constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts0}],
                           options={'maxiter': 400, 'ftol': 1e-14})
            h2 = project_box_sum(res.x, ts0)
            c2 = compute_c5(h2)
            if c2 < cb:
                hb, cb = h2, c2
        except Exception:
            pass

    # Phase 2: Multi-resolution optimization
    target_res = [n0]

        if nr <= target_res[-1]:
            continue
        if nr > 1000:
            break
        target_res.append(nr)

    for idx, nr in enumerate(target_res[1:]):

        tsr = nr / 2.0
        hr = interp_parent(hb, nr)
        c_r = compute_c5(hr)

            c_r = compute_c5(hr)
            if c_r < cb:

        rem2 = budget_s * 0.80 - (time.time() - t0)

            if c01 < cb:

        # SLSQP polish at this resolution
        rem3 = budget_s * 0.80 - (time.time() - t0)

            except Exception:

    # Phase 3: Random restarts at a mid resolution

            if c_ls < cb:
                hb, cb = x_ls.copy(), c_ls

    # Phase 4: Final SLSQP polish
