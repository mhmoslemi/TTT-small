    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

    target = n / 2.0
    h = np.clip(h, 0, 1)

        diff = target - np.sum(h)

        h = np.clip(h + diff/n, 0, 1)

    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def coord_descent(h, n, dx, max_passes=200, tol=1e-10):
    N = 2 * n - 1
    s_arr = np.arange(N) - (n - 1)
    prev_obj = eval_c5(h)

        max_change = 0.0
        for i in range(n):
            m = 1.0 - h
            corr = np.correlate(h, m, mode='full')
            old = h[i]
            
            a = np.empty(N)
            b = np.empty(N)
            for idx in range(N):

                v1 = 1.0 if (0 <= i - s < n) else 0.0

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    return h

def coord_decent(h, n, max_passes=200):
    N = 2 * n - 1
    s_arr = np.arange(N) - (n - 1)
    zero_idx = n - 1  # index where s=0
    prev_val = eval_c5(h)
    for _ in range(max_passes):
        max_ch = 0.0
        for i in range(n):
            m = 1.0 - h
            corr = np.correlate(h, m, mode='full')
            old = h[i]
            a = np.empty(N)
            b = np.empty(N)
            for idx in range(N):
                s = s_arr[idx]
                if idx == zero_idx:
                    # s=0: quadratic term t*(1-t), linearize
                    a[idx] = corr[idx] - old * (1.0 - old)
                    b[idx] = 1.0 - 2.0 * old
                else:
                    v1 = 1.0 if (0 <= i - s < n) else 0.0
                    v2 = 1.0 if (0 <= i + s < n) else 0.0
                    curr = v1 * old * (1.0 - h[i - int(s)]) + v2 * h[i + int(s)] * (1.0 - old)
                    a[idx] = corr[idx] - curr
                    b[idx] = v1 * (1.0 - h[i - int(s)]) - v2 * h[i + int(s)]
            # Minimize max(a + b*t) over t in [0,1]
            best_v = np.max(a + b * 0.0)  # t=0
            best_t = 0.0
            v1_ = np.max(a + b * 1.0)  # t=1
            if v1_ < best_v:
                best_v, best_t = v1_, 1.0
            for ki in range(N):
                for kj in range(ki + 1, N):
                    d = b[ki] - b[kj]
                    if abs(d) > 1e-14:
                        tc = (a[kj] - a[ki]) / d
                        if 0.0 < tc < 1.0:
                            v = np.max(a + b * tc)
                            if v < best_v:
                                best_v, best_t = v, tc
            h[i] = best_t
            max_ch = max(max_ch, abs(best_t - old))
        if max_ch < 1e-10:
            break
    return h

    if c5_cd < best_c5:

        h_s = project_feasible(res.x, n)
