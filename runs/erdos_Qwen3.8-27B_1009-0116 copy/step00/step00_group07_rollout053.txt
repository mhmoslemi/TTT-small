    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        mask = h < 0.999
    else:
        mask = h > 0.001

    if cnt > 0:
        h[mask] += diff / cnt
    else:
        h += diff / n
