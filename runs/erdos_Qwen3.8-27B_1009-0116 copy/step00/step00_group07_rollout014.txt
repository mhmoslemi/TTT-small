import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def make_binary(h, n, rng):
    b = np.zeros(n)
    if len(h) == n:
        b = (h >= 0.5).astype(float)
    else:
        idx = np.round(np.linspace(0, len(h) - 1, n)).astype(int)
        b = (h[idx] >= 0.5).astype(float)
    target = n // 2
    s = int(b.sum())
    if s > target:
        ones = np.where(b == 1)[0]
        hv = h[ones] if len(h) == n else h[np.minimum(ones, len(h) - 1)]
        order = np.argsort(hv)
        for k in range(s - target):
            b[ones[order[k]]] = 0
    elif s < target:
        zeros = np.where(b == 0)[0]
        hv = h[zeros] if len(h) == n else h[np.minimum(zeros, len(h) - 1)]
        order = np.argsort(-hv)
        for k in range(target - s):
            b[zeros[order[k]]] = 1
    return b


def project(h, target_sum):

        return h
    if diff > 0:
        room = 1.0 - h
    else:
        room = h.copy()
    total = room.sum()
    if total < 1e-14:
        return h
    h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)
