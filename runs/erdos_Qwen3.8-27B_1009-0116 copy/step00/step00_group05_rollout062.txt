```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def coord_sweep(h, n, n_sweeps=100):
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    h = project(h, n)
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            # For fixing all except h[j], compute the effect
            # corr_k changes linearly with h[j]
            # Let's compute: corr_k = sum of terms, some involving h[j]
            # ∂corr_k/∂h[j] is either 0 or ±some value
            # For the max, we want to find h[j] that minimizes max_k (base_k + slope_k * h[j])
            
            # Compute partial derivatives of each corr_k w.r.t. h[j]
            slopes = np.zeros(2*n-1)
            bases = np.arange(2*n-1)
            # We need corr_k with h[j] set to 0, and the slope
            h_saved = h[j]
            h[j] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode="full")
            h[j] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode="full")
            h[j] = h_saved
            slopes = corr1 - corr0
            bases = corr0  # corr when h[j]=0
            
            # Now: corr_k(h[j]=x) = bases[k] + slopes[k] * x
            # We want min over x in [0,1] of max_k (bases[k] + slopes[k]*x) * dx
            # (ignoring sum constraint for now)
            
            # The max of linear functions is a convex piecewise-linear function
            # Optimum is at: a vertex of the upper envelope, or boundary
            # Find the x that minimizes the max
            
            # Evaluate at x=0 and x=1
            f0 = np.max(bases)
            f1 = np.max(bases + slopes)
            
            # Check all "knee" points where two lines cross
            # Line k: bases[k] + slopes[k] * x
            # Crossing of lines a and b: x = (bases[b] - bases[a]) / (slopes[a] - slopes[b])
            # Only valid if 0 <= x <= 1
            
            # For efficiency, only consider lines that are at or near the max
            # For n=70, 139 lines, O(L^2) crossings = ~19000, that's fine
            
            L = len(bases)
            best_f = min(f0, f1)
            best_x = 0.0 if f0 <= f1 else 1.0
            
            # Find crossings
            for i in range(L):
                for jj in range(i+1, L):
                    ds = slopes[i] - slopes[jj]
                    if abs(ds) < 1e-15:
                        continue
                    x_star = (bases[jj] - bases[i]) / ds
                    if x_star < 0 or x_star > 1:
                        continue
                    f_star = np.max(bases + slopes * x_star)
                    if f_star < best_f:
                        best_f = f_star
                        best_x = x_star
            
            h[j] = best_x
            # Re-project for sum constraint
            h = project(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                improved = True
        
        if not improved:
            break
    
    return h, best_c5
```

```python
            bases = corr0
            
            # Find best x in [0,1]
            f0 = np.max(bases)
            f1 = np.max(bases + slopes)
            best_f = min(f0, f1)
            best_x = 0.0 if f0 <= f1 else 1.0
            
            # Vectorized crossing computation
            B = np.outer(np.ones(len(slopes)), bases)  # B[i] = bases
            # Actually simpler:
            b_diff = bases[np.newaxis, :] - bases[:, np.newaxis]  # b_diff[i,j] = bases[i] - bases[j]
            s_diff = slopes[:, np.newaxis] - slopes[np.newaxis, :]  # s_diff[i,j] = slopes[i] - slopes[j]
            
            # x_star[j_pair, i_pair] = b_diff / s_diff
            mask = np.abs(s_diff) > 1e-15
            with np.errstate(divide='ignore', invalid='ignore'):
                x_cross = np.where(mask, b_diff / s_diff, -1)
            
            valid_x = (x_cross >= 0) & (x_cross <= 1)
            valid_x &= mask
            
            if np.any(valid_x):
                valid_crossings = x_cross[valid_x]
                # Evaluate max at each crossing point
                for xc in valid_crossings:
                    f_x = np.max(bases + slopes * xc)
                    if f_x < best_f:
                        best_f = f_x
                        best_x = xc
            
            h_new[j] = best_x
        
        # Project the new solution
        h_candidate = project(h_new, n)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            h = h_candidate
            best_c5 = c5_candidate
    
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.cast if hasattr(h, 'cast') else h, 0.0, 1.0)
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def find_best_x(bases, slopes):
    x_grid = np.linspace(0.0, 1.0, 401)
    vals = bases[np.newaxis, :] + slopes[np.newaxis, :] * x_grid[:, np.newaxis]
    f_grid = np.max(vals, axis=1)
    best_idx = int(np.argmin(f_grid))
    best_x = x_grid[best_idx]
    best_f = f_grid[best_idx]
    if 0 < best_idx < 400:
        x_lo = x_grid[best_idx - 1]
        x_hi = x_grid[best_idx + 1]
        for _ in range(60):
            m1 = x_lo + (x_hi - x_lo) / 3.0
            m2 = x_hi - (x_hi - x_lo) / 3.0
            f_m1 = float(np.max(bases + slopes * m1))
            f_m2 = float(np.max(bases + slopes * m2))
            if f_m1 <= f_m2:
                x_hi = m2
            else:
                x_lo = m1
        best_x = (x_lo + x_hi) / 2.0
        best_f = float(np.max(bases + slopes * best_x))
    return best_x, best_f

def coord_descent(h, n, n_sweeps=200):
    target = n / 2.0
    h = project(h.copy(), n)
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        h_new = h.copy()
        for j in range(n):
            h_new[j] = 0.0
            corr0 = np.correlate(h_new, 1.0 - h_new, mode="full")
            h_new[j] = 1.0
            corr1 = np.correlate(h_new, 1.0 - h_new, mode="full")
            slopes = corr1 - corr0
            best_x, _ = find_best_x(corr0, slopes)
            h_new[j] = best_x
        h_candidate = project(h_new, n)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5 - 1e-14:
            h = h_candidate
            best_c5 = c5_candidate
        else:
            if sweep > 5:
                break
    return h, best_c5

def slsqp_optimize(h, n, max_iter=300, ftol=1e-12):
    target = n / 2.0
    try:
        res = minimize(
            compute_c5, h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
            options={'maxiter': max_iter, 'ftol': ftol}
        )
        x_opt = project(res.x, n)
        c5_opt = compute_c5(x_opt)
        return x_opt, c5_opt
    except Exception:
        return h, compute_c5(h)

def slsqp_smooth(h, n, max_iter=200):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        vals = corr * dx
        m = np.max(vals)
        if m < 1e-15:
            return 0.0
        t = 0.01
        shifted = (vals - m) / t
        shifted -= np.max(shifted)
        return m + t * np.log(np.sum(np.exp(shifted)) / len(shifted))
    try:
        res = minimize(
            obj, h, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-14}
        )
        x_opt = project(res.x, n)
        c5_opt = compute_c5(x_opt)
        return x_opt, c5_opt
    except Exception:
        return h, compute_c5(h)
```
