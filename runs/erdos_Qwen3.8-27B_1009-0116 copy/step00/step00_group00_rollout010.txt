from scipy.optimize import minimize

def compute_c5_bound(h, n):
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return float(np.max(corr) * dx)

def interp_to_new_n(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0/n_old  # cell centers
    # Actually, for step functions, the samples are at cell centers
    x_old = (np.arange(n_old) + 0.5) * (2.0/n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0/n_new)
    h_new = np.interp(x_new, x_old, h_old)
    # Adjust to satisfy constraint
    h_new = np.clip(h_new, 0, 1)
    target = n_new / 2.0
    diff = target - np.sum(h_new)
    h_new += diff / n_new
    h_new = np.clip(h_new, 0, 1)
    # Final adjustment
    h_new = np.clip(h_new, 0, 1)
