import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

def make_obj(n):
    def f(x):
        return compute_c5(x, n)
    return f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = globals().get('initial_h_values', None)
    if init is not None:
        n = len(init)
        h_best = np.array(init, dtype=float).copy()
    else:
        n = 70
        h_best = np.full(n, 0.5)

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def eq_con(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': eq_con}]
    obj_fn = make_obj(n)

    stall = 0
    it = 0
    best_time = t0

    while time.time() - t0 < budget_s - 25:
        it += 1
        # Adaptive perturbation
        if stall < 3:
            sigma = 0.03
        elif stall < 8:
            sigma = 0.08
        elif stall < 20:
            sigma = 0.15
        else:
            sigma = 0.2 + 0.02 * min(stall - 20, 15)

        h_pert = repair(h_best + rng.normal(0, sigma, n), n)
        improved = False

        # Phase 1: SLSQP on exact max objective
        if time.time() - t0 < budget_s - 50:
            try:
                r = minimize(obj_fn, h_pert, method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 250, 'ftol': 1e-13})
                hc = repair(r.x, n)
                c = compute_c5(hc, n)
                if c < best_c5 - 1e-13:
                    h_best, best_c5, improved, best_time = hc, c, True, time.time()
            except Exception:
                pass

        # Phase 2: Smooth log-sum-exp with annealed beta
        if time.time() - t0 < budget_s - 40 and it % 2 == 0:
            beta = 15.0 + 8.0 * (it % 8)
            try:
                r2 = minimize(lambda x: logsumexp_obj(x, n, beta),
                              h_pert, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 180, 'ftol': 1e-11})
                hc2 = repair(r2.x, n)
                c2 = compute_c5(hc2, n)
                if c2 < best_c5 - 1e-13:
                    h_best, best_c5, improved, best_time = hc2, c2, True, time.time()
            except Exception:
                pass

        # Phase 3: Targeted coordinate moves (mass transfer)
        if time.time() - t0 < budget_s - 20 and it % 3 == 0:
            for _ in range(80):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                for eps in (0.02, 0.05, 0.1, 0.15):
                    hp = h_best.copy()
                    hp[i] += eps
                    hp[j] -= eps
                    if hp[i] > 1.0 or hp[j] < 0.0:
                        continue
                    c = compute_c5(hp, n)
                    if c < best_c5 - 1e-13:
                        h_best, best_c5, improved = hp, c, True
                        break

        # Phase 4: Larger structural moves every 10 iters
        if time.time() - t0 < budget_s - 30 and it % 10 == 0:
            # Flip a random block
            start = int(rng.randint(n // 2))
            length = int(rng.randint(3, min(10, n - start)))
            hp = h_best.copy()
            hp[start:start+length] = 1.0 - hp[start:start+length]
            hp = repair(hp, n)
            c = compute_c5(hp, n)
            if c < best_c5 - 1e-13:
                h_best, best_c5, improved = hp, c, True

        # Phase 5: Local polish - very small SLSQP from best
        if time.time() - t0 < budget_s - 20 and stall > 5 and it % 5 == 0:
            try:
                r5 = minimize(obj_fn, h_best + rng.normal(0, 0.01, n),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 400, 'ftol': 1e-14})
                hc5 = repair(r5.x, n)
                c5 = compute_c5(hc5, n)
                if c5 < best_c5 - 1e-14:
                    h_best, best_c5, improved = hc5, c5, True
            except Exception:
                pass

        stall = 0 if improved else stall + 1

    # Final polish
    if time.time() - t0 < budget_s - 5:
        try:
            r_final = minimize(obj_fn, h_best, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-14})
            hf = repair(r_final.x, n)
            cf = compute_c5(hf, n)
            if cf < best_c5:
                h_best, best_c5 = hf, cf
        except Exception:
            pass

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
