            j = int(rng.integers(n))
            if i == j:
                continue
            if (h_cur[i] > 0.5) == (h_cur[j] > 0.5):
                continue
            ht = h_cur.copy()
            ht[i] = 1.0
            ht[j] = 0.0
            c5t = compute_c5(ht, _DX)
            delta = c5t - c5_cur
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-8)):
                h_cur = ht
                c5_cur = c5t
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    h_best = h_cur.copy()
        temp = max(temp * 0.97, 0.002)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            h_best = h_cur.copy()

    # Phase 3: Fine continuous polish
    phase3_end = t0 + budget_s - 3.0
    no_improve3 = 0
    while time.time() < phase3_end and no_improve3 < 20:
        sigma = 0.005 + 0.02 * rng.random()
        h_start = h_best + sigma * rng.standard_normal(n)
        h_start = repair(h_start, _TS)
        try:
            res = minimize(obj_func, h_start, method='Nelder-Mead',
                          options={'maxiter': 3000, 'xatol': 1e-9,
                                   'fatol': 1e-12, 'adaptive': True})
            hc = repair(res.x, _TS)
            c5c = compute_c5(hc, _DX)
            if c5c < best_c5:
                best_c5 = c5c
                h_best = hc.copy()
                no_improve3 = 0
            else:
                no_improve3 += 1
        except Exception:
            no_improve3 += 1

    # Final verification
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = repair(h_best, _TS)
    best_c5 = compute_c5(h_best, _DX)
    return (h_best, best_c5, n)
