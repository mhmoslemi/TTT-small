def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def binarize(h, n):
    """Convert a continuous h to binary, preserving as much structure as possible."""
    # Sort by value, assign n//2 to 1 and rest to 0
    h_copy = h.copy()
    threshold = np.partition(h_copy, n//2)[n//2]
    binary = (h_copy >= threshold).astype(np.float64)
    # Fix count if needed
    cnt = int(binary.sum())
    if cnt > n // 2:
        # Set some 1s to 0s (those closest to threshold)
        ones = np.where(binary == 1.0)[0]
        order = np.argsort(h_copy[ones])[::-1]
        binary[ones[order[:cnt - n//2]]] = 0.0
    elif cnt < n // 2:
        zeros = np.where(binary == 0.0)[0]
        order = np.argsort(h_copy[zeros])[::-1]
        binary[zeros[order[:n//2 - cnt]]] = 1.0
    return binary

def upsample(h, n_new):
    """Upsample h from its current length to n_new using linear interpolation."""
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return h_new

            else:

        shifted = corr - m
        return m + np.log(np.sum(np.exp(a * shifted))) / a

        # Compute gradient of smooth max
        m = np.max(corr_vals)
        shifted = corr_vals - m
        weights = np.exp(alpha * shifted) / np.sum(np.exp(alpha * shifted))
        
        # Gradient: grad[t] = sum_k weights[k] * (dc_k/dh[t])

            lr *= 0.8
            h = best_h.copy()

        ones = np.where(h == 1.0)[0]
        rng = np.random.default_rng(0)
        for idx in ones[rng.choice(len(ones), cnt - n // 2, replace=False)]:
            h[idx] = 0.0
    elif cnt < n // 2:
        zeros = np.where(h == 0.0)[0]
        rng = np.random.default_rng(0)
        for idx in zeros[rng.choice(len(zeros), n // 2 - cnt, replace=False)]:
            h[idx] = 1.0
    return h

def local_search_binary(h, time_limit=30.0, seed=42):
    rng = np.random.default_rng(seed)
    n = len(h)
    h = h.astype(np.float64).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    stagnant = 0
    temp = 0.01

    while time.time() - start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_try = min(len(ones) * len(zeros), 800)
        for _ in range(n_try):
            ii = int(ones[rng.integers(len(ones))])
            jj = int(zeros[rng.integers(len(zeros))])
            h[ii] = 0.0
            h[jj] = 1.0
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                # Accept worse with small probability (simulated annealing)
                if rng.random() < 0.001 and c5 < best_c5 + temp:
                    pass
                else:
                    h[ii] = 1.0
                    h[jj] = 0.0
        if not improved:
            stagnant += 1
            temp *= 0.9
            if stagnant > 30:
                break
        else:
            stagnant = 0
    return best_h, best_c5

def continuous_refine(h, time_limit=30.0, alpha=40.0):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(np.float64).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    def project(v):
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            s = np.sum(np.clip(v - mid, 0.0, 1.0))
            if s > n / 2.0:
                lo = mid
            else:
                hi = mid
        return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

    start = time.time()
    lr = 0.02
    stall = 0

    while time.time() - start < time_limit and lr > 1e-7:
        g = 1.0 - h
        corr_vals = np.correlate(h, g, mode="full") * dx
        m = np.max(corr_vals)
        shifted = corr_vals - m
        wb = np.exp(alpha * shifted)
        wb /= wb.sum()

        # Gradient via convolution
        # A[t] = sum_k wb[k_idx] * g[t-k], B[t] = sum_k wb[k_idx] * h[t+k]
        # A = convolve(g, wb, 'full')[n-1:2n-1]
        A = np.convolve(g, wb, mode='full')[n - 1:2 * n - 1]
        # B[t] = sum_s wb[s] * h[t + s - (n-1)], reindex: = corr of h with wb flipped
        B = np.convolve(h, wb[::-1], mode='full')[n - 1:2 * n - 1]
        grad = A - B

        h_new = project(h - lr * grad)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.1, 0.3)
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if stall > 50:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')

    def elapsed():
        return time.time() - t0

    def update(h, c5):
        nonlocal best_h, best_c5
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Get initial values
    try:
        init = initial_h_values
    except NameError:
        init = None

    # Define n values to try (must be even for binary)
    n_list = [46, 98, 100, 146, 198, 200, 298, 400, 500]

    # Build starting points
    starts_for_n = {}
    for n in n_list:
        starts = []
        if init is not None:
            h_up = upsample(init, n)
            starts.append(binarize_from_h(h_up, n))
        # Thue-Morse
        starts.append(thue_morse_binary(n))
        # Random
        for r in range(2):
            rng = np.random.default_rng(seed + n * 31 + r)
            h_r = np.zeros(n)
            h_r[rng.choice(n, n // 2, replace=False)] = 1.0
            starts.append(h_r)
        starts_for_n[n] = starts

    # Budget allocation
    time_used = 0
    for n in n_list:
        if elapsed() > budget_s - 60:
            break
        remaining = budget_s - 60 - elapsed()
        n_starts = len(starts_for_n[n])
        per_start = max(5.0, min(40.0, remaining / (n_starts * len([x for x in n_list if x >= n]))))
        for s in range(n_starts):
            if elapsed() > budget_s - 30:
                break
            h_s = starts_for_n[n][s]
            h_opt, c5_opt = local_search_binary(h_s, time_limit=per_start, seed=seed + n * 100 + s)
            update(h_opt, c5_opt)

    # Continuous refinement on best solution
    if best_h is not None and elapsed() < budget_s - 15:
        tl = min(90.0, budget_s - 15 - elapsed())
        h_ref, c5_ref = continuous_refine(best_h, time_limit=tl, alpha=40.0)
        update(h_ref, c5_ref)

    # If we have time, try refinement at a higher resolution
    if best_h is not None and elapsed() < budget_s - 10:
        n_cur = len(best_h)
        n_up = min(n_cur * 2, 800)
        if n_up % 2 != 0:
            n_up -= 1
        if n_up > n_cur:
            tl = min(60.0, budget_s - 10 - elapsed())
            h_up = upsample(best_h, n_up)
            # Quick local search on continuous
            h_up_mb = np.clip(h_up, 0.0, 1.0)
            # Enforce sum
            target = n_up / 2.0
            cur_sum = h_up_mb.sum()
            if abs(cur_sum - target) > 1e-10:
                h_up_mb = h_up_mb * (target / cur_sum)
                h_up_mb = np.clip(h_up_mb, 0.0, 1.0)
            h_ref2, c5_ref2 = continuous_refine(h_up_mb, time_limit=tl, alpha=50.0)
            update(h_ref2, c5_ref2)

    # Fallback
    if best_h is None:
        n = 100
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # Final check
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
