import numpy as np
from scipy.optimize import minimize
import time

_DX = 2.0 / 70
_N = 70
_EPS = 0.01
_TGT = 35.0


def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _c5(h, dx):
    return _corr(h, dx).max()


def _sobj(x):
    cv = _corr(x, _DX)
    mx = cv.max()
    return mx + _EPS * np.log(np.sum(np.exp((cv - mx) / _EPS)))


def _sgrad(x):
    cv = _corr(x, _DX)
    mx = cv.max()
    w = np.exp((cv - mx) / _EPS)
    w /= w.sum()
    n = _N
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * _DX
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g


def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _efun(x):
    return x.sum() - _TGT


def _ejac(x):
    return np.ones(len(x))


def _slsqp(h0, dx, n, target, eps, maxiter=300):
    global _DX, _N, _EPS, _TGT
    _DX = dx
    _N = n
    _EPS = float(eps)
    _TGT = float(target)
    res = minimize(
        _sobj, h0, jac=_sgrad,
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
        method='SLSQP',
        options={'maxiter': maxiter, 'ftol': 1e-16}
    )
    return _proj(res.x, target)


def _subgrad_step(h, dx, target, lr):
    cv = _corr(h, dx)
    mx = cv.max()
    active = np.where(cv >= mx - 1e-10)[0]
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    g /= max(len(active), 1)
    return _proj(h - lr * g, target)


def _optimize_one(h_init, dx, n, target, rng, t_end):
    best_h = h_init.copy()
    best_c5 = _c5(h_init, dx)
    h = h_init.copy()
    for eps in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > t_end - 25:
            break
        try:
            h_new = _slsqp(h, dx, n, target, eps, maxiter=400)
        except Exception:
            break
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    lr = 0.2
    for it in range(6000):
        if time.time() > t_end - 12:
            break
        h_new = _subgrad_step(h, dx, target, lr)
        c_new = _c5(h_new, dx)
        c_cur = _c5(h, dx)
        if c_new <= c_cur:
            h = h_new
            lr = min(lr * 1.01, 1.5)
        else:
            lr = max(lr * 0.98, 0.005)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        if it % 100 == 50:
            h = _proj(best_h + rng.normal(0, 0.04, n), target)
    if time.time() < t_end - 5:
        for eps in [0.0005, 0.0002, 0.0001]:
            if time.time() > t_end - 2:
                break
            try:
                h2 = _slsqp(best_h, dx, n, target, eps, maxiter=300)
            except Exception:
                break
            c2 = _c5(h2, dx)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.RandomState(seed)
    n = 70
    dx = 2.0 / n
    target = n / 2.0

        if len(h0) != n:

    else:

    h0 = _proj(h0, target)
    t1_end = t0 + (deadline - t0) * 0.60
    best_h, best_c5 = _optimize_one(h0, dx, n, target, rng, t1_end)
    t2_end = t0 + (deadline - t0) * 0.88
    n_restarts = 0
    while time.time() < t2_end:
        n_restarts += 1
        amp = rng.uniform(0.03, 0.18)
        h_pert = _proj(best_h + rng.normal(0, amp, n), target)
        c_pert = _c5(h_pert, dx)
        if c_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c_pert
        sub_end = min(time.time() + min(25.0, t2_end - time.time()), t2_end)

            two, c5o = _optimize_one(h_pert, dx, n, target, rng, sub_end)
            if c5o < best_c5:
                best_h = two
                best_c5 = c5o
    if time.time() < deadline - 15:
        hp, cp = _optimize_one(best_h, dx, n, target, rng, deadline)
        if cp < best_c5:
            best_h, best_c5 = hp, cp
    return best_h, best_c5, n
