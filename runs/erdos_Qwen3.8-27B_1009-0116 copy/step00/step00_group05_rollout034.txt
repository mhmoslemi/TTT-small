import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode="full"))) * (2.0/n)

def _proj(h, tgt):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > 1e-15:
        h *= tgt / s
    h = np.clip(h, 0, 1)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        m = (h > 1e-9) & (h < 1-1e-9)
        if m.any():
            h[m] = np.clip(h[m] + d/m.sum(), 0, 1)
    return h

def _grad_lag(h, k):
    """Gradient of corr at lag k (could be negative)."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # corr = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[m]: (1-h[m-k]) for m in [k, n-1]  -  h[m+k] for m in [0, n-1-k]
        mask1 = np.arange(n) >= k
        mask2 = np.arange(n) <= n-1-k
        g[mask1] += (1.0 - h[np.arange(n)[mask1] - k])
        g[mask2] -= h[np.arange(n)[mask2] + k]
    else:
        # k < 0, let m = -k
        # corr = sum_{j=0}^{n-1+m} h[j]*(1-h[j+m])  where m = -k
        m_idx = -k
        mask1 = np.arange(n) < n - m_idx  # h[j] term
        mask2 = np.arange(n) >= m_idx     # (1-h[j+m]) term, j+m = index
        g[mask1] += (1.0 - h[np.arange(n)[mask1] + m_idx])
        g[mask2] -= h[np.arange(n)[mask2] - m_idx]
    return g

def _grad_step(h, tgt, lr=0.01):
    n = len(h)
    corr = np.correlate(h, 1.0-h, mode="full")
    k_max = np.argmax(corr) - (n-1)
    g = _grad_lag(h, k_max)
    h_new = h - lr * g
    h_new = _proj(h_new, tgt)
    return h_new, k_max

            if i == j: continue

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)

def _proj(h, tgt):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h * (tgt / s)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        m = (h > 1e-9) & (h < 1.0 - 1e-9)
        if m.any():
            h[m] = np.clip(h[m] + d / m.sum(), 0.0, 1.0)
    return h

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    idx = np.arange(n)
    if k >= 0:
        m1 = idx >= k
        m2 = idx <= n - 1 - k
        g[m1] += 1.0 - h[idx[m1] - k]
        g[m2] -= h[idx[m2] + k]
    else:
        m_abs = -k
        m1 = idx < n - m_abs
        m2 = idx >= m_abs
        g[m1] += 1.0 - h[idx[m1] + m_abs]
        g[m2] -= h[idx[m2] - m_abs]
    return g

def _grad_descend(h, tgt, rng, t_end, c5):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5
    cur = h.copy()
    cur_c5 = c5
    lr = 0.005
    for _ in range(200000):
        if time.time() >= t_end:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        k_max = int(np.argmax(corr)) - (n - 1)
        g = _grad_lag(cur, k_max)
        g -= g.mean()
        for trial_lr in (lr, lr * 0.5, lr * 2.0):
            h_t = cur - trial_lr * g
            h_t = _proj(h_t, tgt)
            c5_t = _c5(h_t)
            if c5_t < cur_c5:
                cur = h_t
                cur_c5 = c5_t
                if c5_t < best_c5:
                    best_h = h_t.copy()
                    best_c5 = c5_t
                break
    return best_h, best_c5

def _pair_search(h, tgt, rng, t_end, c5):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5
    deltas = np.array([0.03, -0.03, 0.07, -0.07, 0.15, -0.15, 0.01, -0.01])
    while time.time() < t_end:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        for d in deltas:
            ht = best_h.copy()
            ht[i] += d
            ht[j] -= d
            ht = _proj(ht, tgt)
            c5t = _c5(ht)
            if c5t < best_c5 - 1e-14:
                best_h = ht
                best_c5 = c5t
                break
    return best_h, best_c5

def _anneal(h, tgt, rng, t_end, c5, T0=0.04):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5
    cur = h.copy()
    cur_c5 = c5
    it = 0
    while time.time() < t_end:
        it += 1
        T = max(T0 * (0.9995 ** it), 1e-7)
        r = rng.random()
        if r < 0.4:
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            d = rng.uniform(-0.08, 0.08)
            ht = cur.copy()
            ht[i] += d
            ht[j] -= d
        elif r < 0.7:
            i = rng.integers(n)
            j = rng.integers(n)
            k = rng.integers(n)
            if i == j or i == k or j == k:
                continue
            ht = cur.copy()
            d1 = rng.uniform(-0.06, 0.06)
            d2 = rng.uniform(-0.06, 0.06)
            d3 = -(d1 + d2)
            ht[i] += d1
            ht[j] += d2
            ht[k] += d3
        else:
            ht = cur + rng.normal(0, 0.03, n)
            ht -= ht.mean() - cur.mean()
        ht = _proj(ht, tgt)
        c5t = _c5(ht)
        if c5t <= cur_c5 or rng.random() < np.exp(-(c5t - cur_c5) / T):

            cur_c5 = c5t

                best_c5 = c5t

            for d in offsets:
                ht = best_h.copy()
                ht[i] += d
                ht[j] -= d
                ht = _proj(ht, tgt)

                    best_c5 = c5t
                    break
    return best_h, best_c5

    t2 = t0 + budget_s * 0.47

    # Phase 4: anneal from perturbed best (20%)
    t4 = t0 + budget_s * 0.78
    hp = best_h.copy()
    perts = rng.integers(0, n, 3)
    hp[perts[0]] += 0.2

    hp[perts[2]] += 0.1
    hp = _proj(hp, tgt)
    hp, c5p = _anneal(hp, tgt, rng, t4, _c5(hp))
    if c5p < best_c5:

    t6 = t0 + budget_s * 0.985

    if time.time() < t0 + budget_s * 0.995:
        t7 = t0 + budget_s * 0.995
        best_h, best_c5 = _pair_search(best_h, tgt, rng, t7, best_c5)

    # Final projection check
    best_h = _proj(best_h, tgt)
    best_c5 = _c5(best_h)

    return best_h, best_c5, n
