def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def grad_c5_k(h, k):
    """Gradient of the k-th correlation value w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        # term: (1-h[j-k]) if 0 <= j-k < n
        if 0 <= j - k < n:
            val += (1 - h[j - k])
        # term: -h[j+k] if 0 <= j+k < n
        if 0 <= j + k < n:
            val -= h[j + k]
        grad[j] = val
    return grad

def project_onto_constraint(h, n):
    """Project onto {sum(h)=n/2, 0<=h<=1}"""
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - n/2) < 1e-12:
            break
        # Distribute equally to non-saturated components
        diff = (n/2 - s)
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(free) == 0:
            h = np.clip(h, 0, 1)
            break
        h[free] += diff / np.sum(free)
    return h

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(c)
    return m + tau * np.log(np.sum(np.exp((c - m) / tau)) / len(c))

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx

    # Phase 2: LSE-smoothed refinement at original resolution
    if elapsed() < budget_s * 0.35:
        h_smooth = try_lse_slsqp(best_h, n, t0, budget_s)
        c5 = compute_c5(h_smooth)
        if c5 < best_c5:
            best_h = h_smooth
            best_c5 = c5

    # Phase 3: Increase resolution and optimize
    for factor in [2, 2, 2, 2]:
        if elapsed() > budget_s * 0.85:
            break
        n_new = n * factor
        if n_new > 900:
            break
        h_up = interpolate_up(best_h, n, n_new)
        h_up = project_h(h_up, n_new)
        # Optimize at higher resolution
        h_opt = try_slsqp(h_up, n_new, t0, budget_s, max_iter=1500)
        c5_opt = compute_c5(h_opt)
        if c5_opt >= best_c5 and elapsed() < budget_s * 0.5:
            h_opt = try_lse_slsqp(h_opt, n_new, t0, budget_s)
            c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            n = n_new
        else:
            # Keep the higher-res interpolated even if not better (for next scale)
            n = n_new
            best_h = h_up

    # Phase 4: Final SLSQP polish at best resolution
    if elapsed() < budget_s * 0.75 and n == len(best_h):
        h_final = try_slsqp(best_h, n, t0, budget_s, max_iter=5000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Phase 5: If budget remains, try subgradient on active constraint
    if elapsed() < budget_s * 0.6 and n < 300:
        h = best_h.copy()
        dx = 2.0 / n
        for it in range(50000):
            if elapsed() > budget_s * 0.88:
                break
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            k_star = int(np.argmax(corr))
            shift = k_star - (n - 1)
            grad = np.zeros(n)
            for j in range(n):
                g = 0.0
                if 0 <= j - shift < n:
                    g += (1.0 - h[j - shift])
                if 0 <= j + shift < n:
                    g -= h[j + shift]
                grad[j] = g
            lr = 0.05 * dx / (1.0 + 1e-4 * it)
            h_new = project_h(h - lr * grad, n)
            h = h_new
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    # Final validation and evaluation
    best_h = project_h(best_h, len(best_h))
    final_c5 = evaluate_erdos_solution(best_h)
    if not np.isfinite(final_c5):
        final_c5 = best_c5
    return best_h.tolist(), float(final_c5), len(best_h)
