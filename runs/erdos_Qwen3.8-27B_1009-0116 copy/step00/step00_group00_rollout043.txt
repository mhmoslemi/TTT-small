from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def enforce_integral(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        remaining = diff
        for i in order:
            add = min(1.0 - h[i], remaining)
            h[i] += add
            remaining -= add
            if remaining < 1e-14:
                break
    else:
        order = np.argsort(-h)
        remaining = -diff
        for i in order:
            sub = min(h[i] + 1e-14, remaining)
            h[i] -= sub
            remaining -= sub
            if remaining < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr * dx
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta)

    x_p = np.linspace(0, 2, len(parent))
    x_t = np.linspace(0, 2, n_target)
    return np.interp(x_t, x_p, parent)

def slsqp_optimize(h0, n, beta, maxiter, ftol):
    dx = 2.0 / n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: obj_smooth(x, n, beta),
            h0, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        return res
    except Exception:
        return None

    """Simple coordinate descent / random perturbation local search."""
    best = h.copy()
    best_c5 = compute_c5(best)
    for s in range(steps):
        i = np.random.randint(n)
        delta = np.random.uniform(-step_size, step_size)
        new_val = best[i] + delta
        if new_val < 0 or new_val > 1:
            continue
        h_trial = best.copy()
        h_trial[i] = new_val
        h_trial = enforce_integral(h_trial, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best = h_trial
            best_c5 = c5_trial
