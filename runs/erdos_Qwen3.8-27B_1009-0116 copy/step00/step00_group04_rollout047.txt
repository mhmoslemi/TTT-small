import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    mask = (h > 0.0) & (h < 1.0)
    cnt = int(mask.sum())
    if cnt > 0:
        h[mask] = h[mask] + diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_descent(h, n, rng, max_iter, lr0, lr_decay):
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(best, n)
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        s = idx - (n - 1)
        grad = np.zeros(n)
        for m in range(n):
            ms = m + s
            if 0 <= ms < n:
                grad[m] += 1.0 - h[ms]
            mms = m - s
            if 0 <= mms < n:
                grad[m] -= h[mms]
        lr = lr0 * lr_decay ** it
        h = project_feasible(h - lr * grad, n)
        c = compute_c5(h, n)
        if c < best_c5:
            best_c5 = c
            best = h.copy()
    return best, best_c5

def slsqp_polish(h, n, max_iter=500):
    beta = 200.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        # smooth max
        c = beta * corr
        cmax = c.max()
        return float((cmax + np.log(np.sum(np.exp(c - cmax)))) / beta * (2.0 / n))
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        c = beta * corr
        cmax = c.max()
        weights = np.exp(c - cmax) / np.sum(np.exp(c - cmax))
        # gradient of smooth max
        grad = np.zeros(n)
        for k in range(len(corr)):
            if weights[k] < 1e-15:
                continue
            s = k - (n - 1)
            for m in range(n):

                if 0 <= ms < n:
                    grad[m] += weights[k] * (1.0 - x[ms])
                mms = m - s
                if 0 <= mms < n:
                    grad[m] -= weights[k] * x[mms]
        return grad * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    if res.success or res.fun < compute_c5(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    h = h + (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    cnt = max(int(mask.sum()), 1)
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def _integ_eq(x, n):
    return x.sum() - n * 0.5

def _integ_jac(x, n):
    return np.ones(len(x))

def _smax_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    c = beta * corr
    m = c.max()
    return float((m + np.log(np.sum(np.exp(c - m)))) / beta * (2.0 / n))

def _smax_grad(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    c = beta * corr
    m = c.max()
    w = np.exp(c - m)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        for i in range(n):
            ms = i + s
            if 0 <= ms < n:
                grad[i] += w[k] * (1.0 - x[ms])
            ms2 = i - s
            if 0 <= ms2 < n:
                grad[i] -= w[k] * x[ms2]
    return grad * dx

def slsqp_opt(h0, n, maxiter, beta):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _integ_eq, 'jac': _integ_jac, 'args': (n,)}]
    res = minimize(_smax_obj, h0, method='SLSQP',
                   jac=_smax_grad, args=(n, beta),
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

def subgrad_opt(h0, n, n_iter, lr0):
    h = h0.copy()
    best = h.copy()
    best_c = compute_c5(best, n)
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = int(np.argmax(corr))
        s = k - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            ms = i + s
            if 0 <= ms < n:
                grad[i] += 1.0 - h[ms]
            ms2 = i - s
            if 0 <= ms2 < n:
                grad[i] -= h[ms2]
        lr = lr0 / (1.0 + 0.02 * it)
        h = project_feasible(h - lr * grad, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best = h.copy()
    return best, best_c

def binary_search(h0, n, rng, n_iter):
    h = (h0 > 0.5).astype(float)
    for _ in range(200):
        if h.sum() > n * 0.5:
            h[int(rng.integers(n))] = 0.0
        elif h.sum() < n * 0.5:
            h[int(rng.integers(n))] = 1.0
        else:
            break
    best = h.copy()
    best_c = compute_c5(best, n)
    for _ in range(n_iter):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h, n)
        if c <= best_c:
            best_c = c
            best = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best, best_c
