```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def project(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Water-filling to adjust sum
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        # Distribute diff to elements not at bounds
        free_up = np.sum(h < 1.0)
        free_dn = np.sum(h > 0.0)
        if diff > 0:
            mask = h < 1.0
            h[mask] += diff / free_up
        else:
            mask = h > 0.0
            h[mask] += diff / free_dn
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    
    # Ensure feasibility
    h0 = project(h0, n0)
    best_h = h0.copy()
    best_c5, _ = compute_c5(best_h)
    
    def elapsed():
        return time.time() - start
    
    # Try Nelder-Mead on the initial
    def objective(h):
        c5, _ = compute_c5(h)
        return c5
    
    # Also try different n values
    for n_try in [n0, n0*2, n0*3, 90, 100, 120, 150, 200, 300, 500]:
        if elapsed() > budget_s * 0.8:
            break
        if n_try == n0:
            h_start = h0.copy()
        else:
            # Interpolate h0 to n_try points
            x_orig = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_start = np.interp(x_new, x_orig, h0)
            h_start = project(h_start, n_try)
        
        c5_start, _ = compute_c5(h_start)
        
        # Nelder-Mead
        x0 = h_start.copy()
        result = minimize(objective, x0, method='Nelder-Mead',
                          options={'maxiter': 5000, 'xatol': 1e-6, 'fatol': 1e-7})
        
        if result.fun < best_c5:
            # Project and verify
            h_opt = project(result.x, n_try)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
            else:
                best_h = result.x
                best_c5 = result.fun
    
    return best_h, best_c5, len(best_h)
```

```python
def softmax_gradient(h, alpha=100):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlation values
    corr = np.correlate(h, 1.0 - h, mode="full") * dx  # shape (2n-1,)
    # Softmax weights
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    # Compute gradient: weighted sum of d corr[ℓ] / d h[j]
    # corr[ℓ] for ℓ = -(n-1), ..., 0, ..., (n-1)
    # d corr[ℓ] / d h[j] = (1-h[j+ℓ])*1_{valid} - h[j-ℓ]*1_{valid}
    grad = np.zeros(n)
    for ell in range(-(n-1), n):
        w = weights[ell + n - 1]
        if w < 1e-10:
            continue
        # (1-h[j+ell]) for valid j+ell
        j_plus = np.arange(n) + ell
        valid_plus = (j_plus >= 0) & (j_plus < n)
        grad[valid_plus] += w * (1 - h[j_plus[valid_plus]])
        # -h[j-ell] for valid j-ell
        j_minus = np.arange(n) - ell
        valid_minus = (j_minus >= 0) & (j_minus < n)
        grad[valid_minus] -= w * h[j_minus[valid_minus]]
    return grad * dx
```

```python
def soft_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    e = np.exp(alpha * (corr - mc))
    s = np.sum(e)
    w = e / s

    threshold = 1e-8
    active = np.where(w > threshold)[0]
    ph = 1.0 - h
    grad = np.zeros(n)
    for m in active:
        ell = m - (n - 1)
        wi = w[m]
        lo_a = max(0, -ell)
        hi_a = min(n, n - ell)
        if lo_a < hi_a:
            grad[lo_a:hi_a] += wi * ph[lo_a + ell: hi_a + ell]
        lo_b = max(0, ell)
        hi_b = min(n, n + ell)
        if lo_b < hi_b:
            grad[lo_b:hi_b] -= wi * h[lo_b - ell: hi_b - ell]
    return mc, grad * dx
```
