```python
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    
    # Gradient of c_{k_star} w.r.t. h
    grad = np.zeros(n)
    offset = k_star - (n - 1)
    for i in range(n):
        j = i + offset  # = i + k_star - (n-1)
        if 0 <= j < n:
            # h[i] appears as first factor: contributes (1-h[j])
            grad[i] += (1.0 - h[j])
            # h[j] appears as second factor: contributes -h[i]
            grad[j] -= h[i]
    
    # Take step
    h_new = h - lr * grad
    # Project to feasible set
    h_new = np.clip(h_new, 0, 1)
    # Rescale for integral constraint
    target = n / 2.0
    total = np.sum(h_new)
    if total > 0:
        h_new = h_new * target / total
        h_new = np.clip(h_new, 0, 1)
        # Final correction
        total = np.sum(h_new)
        if abs(total - target) > 1e-10 and total > 0:
            h_new = h_new * target / total
    return h_new
```

```python
import numpy as np
import time
from scipy.optimize import minimize, bounds as scipy_bounds

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        h = np.full(n, 0.5)
        return h
    h = h * (target / total)
    h = np.clip(h, 0.0, 1.0)
    # Iterate to converge
    for _ in range(10):
        total = np.sum(h)
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # Distribute difference to non-saturated variables
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_free = np.sum(free)
        if n_free == 0:
            # All saturated, relax slightly
            h = np.clip(h + diff / n, 0, 1)
            continue
        h[free] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_optimize(h, dx, n, t_limit):
    """Subgradient method with top-m averaging."""
    t_start = time.time()
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    m_top = 5  # Use top 5 lags
    lr = 0.01
    momentum = np.zeros(n)
    speed = 0.9
    
    while time.time() - t_start < t_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Get top m lags
        top_idx = np.argsort(-corr)[:m_top]
        # Weighted gradient
        grad = np.zeros(n)
        weights = corr[top_idx] - np.min(corr[top_idx]) + 0.1  # soft weights
        weights /= np.sum(weights)
        
        for w, k in zip(weights, top_idx):
            offset = k - (n - 1)
            for i in range(n):
                j = i + offset
                if 0 <= j < n:
                    grad[i] += w * (1.0 - h[j])
                    grad[j] += w * (-h[i])
        
        # Momentum
        momentum = speed * momentum - lr * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        
        c5 = evaluate_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
        lr = max(0.001, lr * 0.9999)
    
    return best_h, best_c5
```

```python
    grad = np.zeros(n)
    offsets = np.arange(2 * n - 1) - (n - 1)
    for wt, k in zip(w, top):
        off = int(offsets[k])
        fs = max(0, -off)
        fe = min(n - 1, n - 1 - off)
        ir = np.arange(fs, fe + 1)
        jr = ir + off
        grad[ir] += wt * (1.0 - h[jr])
        grad[jr] -= wt * h[ir]
    return project_feasible(h - lr * grad, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    best_h = initial_h_values.astype(float).copy()
    best_c5 = evaluate_c5(best_h, dx)

    # Phase 1: Coordinate descent with restarts (50% time)
    t1 = t0 + 0.5 * budget_s
    stuck = 0
    perturb_scales = [0.01, 0.02, 0.03, 0.05, 0.01]
    pidx = 0
    while time.time() < t1:
        hn = coord_sweep(best_h, n)
        c5 = evaluate_c5(hn, dx)
        if c5 < best_c5 - 1e-13:
            best_c5, best_h = c5, hn
            stuck = 0
        else:
            stuck += 1
            if stuck >= 3:
                for attempt in range(8):
                    scale = perturb_scales[(pidx + attempt) % len(perturb_scales)]
                    p = project_feasible(best_h + np.random.randn(n) * scale, n)
                    ps = coord_sweep(p, n)
                    cp = evaluate_c5(ps, dx)
                    if cp < best_c5:
                        best_c5, best_h = cp, ps
                        stuck = 0
                        break
                pidx = (pidx + 1) % len(perturb_scales)
                if stuck >= 8:
                    stuck = 0

    # Phase 2: Subgradient with periodic coordinate polish (30% time)
    t2 = t0 + 0.8 * budget_s
    hs = best_h.copy()
    lr = 0.005
    step_c = 0
    while time.time() < t2:
        hs = subgrad_step(hs, n, lr, 5)
        cs = evaluate_c5(hs, dx)
        if cs < best_c5:
            best_c5, best_h = cs, hs.copy()
        step_c += 1
        if step_c % 150 == 0:
            hs = coord_sweep(hs, n)
            cs = evaluate_c5(hs, dx)
            if cs < best_c5:
                best_c5, best_h = cs, hs.copy()
        lr = max(0.001, lr * 0.9999)

    # Phase 3: Final coordinate polish (20% time)
    t3 = t0 + budget_s - 5
    while time.time() < t3:
        hn = coord_sweep(best_h, n)
        c5 = evaluate_c5(hn, dx)
        if c5 < best_c5 - 1e-13:
            best_c5, best_h = c5, hn
        else:
            # One more perturbation attempt
            p = project_feasible(best_h + np.random.randn(n) * 0.005, n)
            ps = coord_sweep(p, n)
            cp = evaluate_c5(ps, dx)
            if cp < best_c5:
                best_c5, best_h = cp, ps
            else:
                break

    best_h = project_feasible(best_h, n)
    best_c5 = evaluate_c5(best_h, dx)
    return best_h, best_c5, n
```
