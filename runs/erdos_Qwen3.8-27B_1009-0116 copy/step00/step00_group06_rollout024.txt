def evaluate(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    lo = -1.5
    hi = 1.5
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def compute_grad(h, corr, temp):
    n = len(h)
    half = corr[:n]
    max_c = np.max(half)
    w = np.exp((half - max_c) / temp)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # r_0 term
    if w[0] > 1e-12:
        grad += w[0] * (1 - 2 * h)
    for ell in range(1, n):
        if w[ell] < 1e-12:
            continue
        grad[ell:] += w[ell] * (1 - h[:n-ell])
        grad[:n-ell] += -w[ell] * h[ell:]
    return grad

def optimize(h, n, max_time, rng):

    h = project(h, n)
    best = h.copy()
    best_val = evaluate(h, n)
    
    t0 = None
    temp = 0.05
    lr = 0.01
    
    while (t0 is None or time.time() - t0 < max_time):

            t0 = time.time()
        elif time.time() - t0 > max_time:

        grad = compute_grad(h, corr, temp)

        h_new = project(h_new, n)
        
        val = evaluate(h_new, n)
        if val < best_val:
            best = h_new.copy()
            best_val = val
            # Occasionally increase step

            lr *= 0.8

def eval_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    corr = np.correlate(h, 1.0 - h, mode="full")

    m = np.max(half)
    w = np.exp((half - m) / temp)
