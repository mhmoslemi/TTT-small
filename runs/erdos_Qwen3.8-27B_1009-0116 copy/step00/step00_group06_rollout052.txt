                h[i] = np.clip(old + d, 0.0, 1.0)
                h = repair(h, n)
                o = c5_value(h, dx)
                if o < best_o - 1e-13:
                    best_o = o
                    best_hi = h.copy()
            if best_hi is not None:
                h = best_hi
                obj = best_o
                improved = True
            else:
                h[i] = old
                h = repair(h, n)
        if not improved:
            step *= 0.65
            if step < 3e-4:
                break
    nm_deadline = t_start + time_limit * 0.95
    if time.time() < nm_deadline:
        try:
            res = minimize(nm_objective, h.copy(), args=(n, dx),
                          method='Nelder-Mead',
                          options={'maxiter': 4000, 'fatol': 1e-10, 'xatol': 1e-11})
            res_h = repair(res.x, n)
            res_o = c5_value(res_h, dx)
            if res_o < obj:
                h, obj = res_h, res_o
        except Exception:
            pass
    return h, obj

def binary_push(h, n, dx, rng, fractional_thresh=0.5):
    h = h.copy()
    obj = c5_value(h, dx)
    for _ in range(n):
        i = rng.randint(n)
        if h[i] > 0.05 and h[i] < 0.95:
            for target_val in [0.0, 1.0]:
                h[i] = target_val
                h = repair(h, n)
                o = c5_value(h, dx)
                if o < obj - 1e-12:
                    obj = o
                    break
    return h, obj

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    best_h = repair(np.array(initial_h_values, dtype=float), n)
    best_c5 = c5_value(best_h, dx)

    rem = budget_s - (time.time() - t0)
    if rem > 20:
        t_lim = min(100, rem * 0.1)
        h1, c5_1 = local_optimize(best_h, n, dx, rng, t_lim)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    for _ in range(5):
        rem = budget_s - (time.time() - t0)
        if rem < 20:
            break
        h_bp, c5_bp = binary_push(best_h, n, dx, rng)
        if c5_bp < best_c5:
            best_h, best_c5 = h_bp, c5_bp
            t_lim = min(30, rem * 0.05)
            h1, c5_1 = local_optimize(best_h, n, dx, rng, t_lim)
            if c5_1 < best_c5:
                best_h, best_c5 = h1, c5_1

    while (budget_s - (time.time() - t0)) > 50:
        scale = 0.025
        h_pert = repair(best_h + rng.uniform(-scale, scale, n), n)
        rem = budget_s - (time.time() - t0)
        t_lim = min(35, rem * 0.035)
        h_p, c5_p = local_optimize(h_pert, n, dx, rng, t_lim)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p

    rem = budget_s - (time.time() - t0)
    if rem > 15:
        t_lim = rem - 5
        h_f, c5_f = local_optimize(best_h, n, dx, rng, t_lim)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    rem = budget_s - (time.time() - t0)
    if rem > 5:
        h_bp2, c5_bp2 = binary_push(best_h, n, dx, rng)
        if c5_bp2 < best_c5:
            best_h, best_c5 = h_bp2, c5_bp2

    best_h = repair(best_h, n)
    best_c5 = c5_value(best_h, dx)
    return best_h, best_c5, n
