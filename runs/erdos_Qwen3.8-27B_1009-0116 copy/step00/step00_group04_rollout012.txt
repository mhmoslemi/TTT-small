import numpy as np

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_integral(h, n_points):
    target = n_points / 2.0
    h = h.copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h += diff / n_points
        h = np.clip(h, 0.0, 1.0)
    return h


def upgrade_resolution(h, n_old, n_new):
    h_new = np.zeros(n_new)
    for i in range(n_old):
        s = min(int(i * n_new / n_old), n_new - 1)
        e = max(int((i + 1) * n_new / n_old), s + 1)
        h_new[s:e] = h[i]
    return repair_integral(h_new, n_new)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    cur_h = h.copy()
    cur_c5 = best_c5
    T = 0.004
    T_lo = 1e-8
    cool = 0.9996
    t_start = time.time()
    stall = 0
    reheat_T = 0.003

    while time.time() - t_start < max_time:
        if T < T_lo:
            break

        new_h = cur_h.copy()

        if binary_mode:
            ones = np.where(new_h > 0.5)[0]
            zeros = np.where(new_h < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[int(rng.integers(len(ones)))]
                j = zeros[int(rng.integers(len(zeros)))]
                new_h[i] = 0.0
                new_h[j] = 1.0
        else:
            npert = int(rng.integers(1, 4))
            for _ in range(npert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))

                amt = rng.uniform(0.0, min(cur_h[i], 1.0 - cur_h[j])) * 0.7

def directed_perturb(h, n, rng):
    """Perturb guided by the worst shift."""
    new_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    m = k_max - n + 1  # shift amount

    npert = int(rng.integers(1, 3))
