import numpy as np
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def random_binary(n, rng):
    h = rng.random(n)
    h = (h >= 0.5).astype(np.float64)
    s = int(h.sum())
    if s > n // 2:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:s - n // 2]] = 0
    elif s < n // 2:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:n // 2 - s]] = 1
    return h

def block_pattern(n, block_len, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        b = min(block_len, n - pos)
        vals = rng.random(b)
        h[pos:pos + b] = (vals >= 0.5).astype(np.float64)
        pos += b
    s = int(h.sum())
    if s > n // 2:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:s - n // 2]] = 0
    elif s < n // 2:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:n // 2 - s]] = 1
    return h

def greedy_phase(h, dx, max_passes=200, deadline=None):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        cur_c5 = compute_c5(h, dx)
        found = False
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    best_h = h.copy()
                    found = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not found:
            break
    return best_h, best_c5

def simulated_annealing(h, dx, T0, T_min, cooling, deadline, seed=123):

    current = h.copy()
    current_c5 = compute_c5(current, dx)
    best = current.copy()
    best_c5 = current_c5
    T = T0
    while time.time() < deadline:
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        current[i], current[j] = current[j], current[i]
        new_c5 = compute_c5(current, dx)
        delta = new_c5 - current_c5
        if delta < 0 or (delta < T and rng.random() < np.exp(-min(delta, 5.0) / T)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = current.copy()
        else:
            current[i], current[j] = current[j], current[i]
        T = max(T_min, T * cooling)
    return best, best_c5
