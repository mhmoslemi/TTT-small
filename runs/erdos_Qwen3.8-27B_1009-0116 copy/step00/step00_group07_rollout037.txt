    c5_best = min(c5b, c5c)
    h_best = hb.copy() if c5b <= c5c else h_init.copy()

    # Main search
    h = h_best.copy()
    c5_cur = c5_best
    T = 0.008
    stuck = 0

    while time.time() < deadline:
        # Choose move type
        if time.time() < t0 + budget_s * 0.4:
            mix = 0.7  # mostly binary early
        elif time.time() < t0 + budget_s * 0.7:
            mix = 0.5
        else:
            mix = 0.2  # mostly continuous late

        if rng.random() < mix:
            # Binary swap
            hb2 = np.round(h).clip(0, 1)
            no2 = int(hb2.sum())
            if 0 < no2 < n:
                oi = np.where(hb2 == 1)[0]
                zi = np.where(hb2 == 0)[0]
                i = oi[rng.randint(len(oi))]
                j = zi[rng.randint(len(zi))]
                hb2[i] = 0.0
                hb2[j] = 1.0
                nc = compute_c5(hb2)
                d = nc - c5_cur
                if d < 0 or (T > 1e-9 and rng.random() < np.exp(-d / T)):
                    h = hb2.copy()
                    c5_cur = nc
                    if nc < c5_best:
                        c5_best = nc
                        h_best = h.copy()
                        stuck = 0
                    else:
                        stuck += 1
                else:
                    stuck += 1
        else:
            # Continuous perturbation
            hc = h.copy()
            kp = rng.randint(1, 4)
            ix = rng.choice(n, kp, replace=False)
            scale = 0.04 if T > 0.003 else 0.015
            hc[ix] += rng.randn(kp) * scale
            hc = np.clip(hc, 0.0, 1.0)
            hc = _fix_sum(hc, tgt, n)
            nc = compute_c5(hc)
            d = nc - c5_cur
            if d < 0 or (T > 1e-9 and rng.random() < np.exp(-d / T)):
                h = hc.copy()
                c5_cur = nc
                if nc < c5_best:
                    c5_best = nc
                    h_best = h.copy()
                    stuck = 0
                else:
                    stuck += 1
            else:
                stuck += 1

        # Cool
        T *= 0.999996
        if T < 1e-9:
            T = 1e-9

        # Restart if stuck
        if stuck > 1200:
            T = 0.018
            stuck = 0
            r = rng.random()
            if r < 0.35:
                hp = (h_best > 0.5).astype(float)
                nr = max(2, n // 8)
                rix = rng.choice(n, nr, replace=False)
                hp[rix] = 1.0 - hp[rix]
                hp = _rebalance_bin(hp, tgt_i, rng)
                h = hp
                c5_cur = compute_c5(h)
            elif r < 0.65:
                h = np.clip(h_best + rng.randn(n) * 0.02, 0, 1)
                h = _fix_sum(h, tgt, n)
                c5_cur = compute_c5(h)
            else:
                # Bigger restart
                hp = (h_best > 0.5).astype(float)
                nr3 = max(5, n // 5)
                rix3 = rng.choice(n, nr3, replace=False)
                hp[rix3] = rng.randint(0, 2, nr3).astype(float)
                hp = _rebalance_bin(hp, tgt_i, rng)
                h = hp
                c5_cur = compute_c5(h)

    # Final polish: tiny continuous adjustments
    if time.time() < deadline:
        hp = h_best.copy()
        cp = c5_best
        for _ in range(500):
            if time.time() >= deadline:
                break
            idx = rng.randint(n)
            hp2 = hp.copy()
            hp2[idx] = np.clip(hp[idx] + rng.randn() * 0.005, 0, 1)
            hp2 = _fix_sum(hp2, tgt, n)
            nc2 = compute_c5(hp2)
            if nc2 < cp:
                hp = hp2
                cp = nc2
        if cp < c5_best:
            c5_best = cp
            h_best = hp.copy()

    return h_best, c5_best, n
