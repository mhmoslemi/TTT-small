    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-14:
        h[:] = 0.5
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        idx = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if len(idx) == 0:
            idx = np.where((h >= 0.0) & (h <= 1.0))[0]
            if len(idx) == 0:
                break
        step = diff / len(idx)
        h[idx] = np.clip(h[idx] + step, 0.0, 1.0)
    return h

def sum_eq(h, n):
    return np.sum(h) - n / 2.0

def build_constraints(n):
    return [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]

def c5_obj(h, dx, n):
    return compute_c5(h, dx)

def penalized_c5(h, dx, n):
    hc = np.clip(h, 0.0, 1.0)
    pen = 200.0 * (np.sum(hc) - n / 2.0) ** 2
    return compute_c5(hc, dx) + pen

def coordinate_perturb(h, n, rng, num_steps):
    target = n / 2.0
    h = h.copy()
    for _ in range(num_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = 0.01 * rng.randn()
        h[i] += delta
        h[j] -= delta
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= delta
            h[j] += delta
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 70
    dx = 2.0 / n
    constraints = build_constraints(n)

    # Initialize from parent construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(n, 0.5)
    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, dx)

    elapsed = time.time() - t0
    iteration = 0

    while elapsed < budget_s - 15:
        iteration += 1
        # Phase 1: COBYLA local optimization from best
        try:
            res = minimize(c5_obj, h_best.copy(), method='COBYLA',
                           args=(dx, n), constraints=constraints,
                           options={'maxiter': 400, 'rhofac': 1.15, 'catol': 1e-14})
            if res.x is not None:
                hc = repair(res.x, n)
                c5c = compute_c5(hc, dx)
                if c5c < best_c5:
                    h_best = hc
                    best_c5 = c5c
        except Exception:
            pass

        elapsed = time.time() - t0
        if elapsed > budget_s - 20:
            break

        # Phase 2: Nelder-Mead refinement
        try:
            res2 = minimize(penalized_c5, h_best.copy(), method='Nelder-Mead',
                            args=(dx, n),
                            options={'maxiter': 3000, 'xatol': 1e-11, 'fatol': 1e-13, 'adaptive': True})
            hc = repair(res2.x, n)
            c5c = compute_c5(hc, dx)
            if c5c < best_c5:
                h_best = hc
                best_c5 = c5c
        except Exception:
            pass

        elapsed = time.time() - t0
        if elapsed > budget_s - 20:
            break

        # Phase 3: Coordinate perturbation + local search
        h_coord = coordinate_perturb(h_best, n, rng, 8)
        h_coord = repair(h_coord, n)
        try:
            res3 = minimize(penalized_c5, h_coord.copy(), method='Nelder-Mead',
                            args=(dx, n),
                            options={'maxiter': 1500, 'xatol': 1e-11, 'fatol': 1e-13})
            hc = repair(res3.x, n)
            c5c = compute_c5(hc, dx)
            if c5c < best_c5:
                h_best = hc
                best_c5 = c5c
        except Exception:
            pass

        elapsed = time.time() - t0
        if elapsed > budget_s - 15:
            break

        # Phase 4: Gaussian perturbation with varying sigma
        sigma = [0.05, 0.1, 0.15, 0.25, 0.4][iteration % 5]
        h_pert = h_best + rng.normal(0, sigma, n)
        h_pert = repair(h_pert, n)
        try:
            res4 = minimize(c5_obj, h_pert.copy(), method='COBYLA',
                            args=(dx, n), constraints=constraints,
                            options={'maxiter': 200, 'rhofac': 1.15, 'catol': 1e-14})
            hc = repair(res4.x, n)
            c5c = compute_c5(hc, dx)
            if c5c < best_c5:
                h_best = hc
                best_c5 = c5c
        except Exception:
            pass

        elapsed = time.time() - t0
        if elapsed > budget_s - 10:
            break

        # Phase 5: Every 5th iteration, try a larger structural perturbation
        if iteration % 5 == 0 and elapsed < budget_s - 30:
            h_struct = h_best.copy()
            # Randomly shift a block
            block_size = rng.randint(5, 15)
            start = rng.randint(0, n - block_size)
            block = h_struct[start:start+block_size].copy()
            h_struct[start:start+block_size] = h_struct[start+block_size:start+2*block_size]
            h_struct[start+block_size:start+2*block_size] = block
            h_struct = repair(h_struct, n)
            try:
                res5 = minimize(c5_obj, h_struct.copy(), method='COBYLA',
                                args=(dx, n), constraints=constraints,
                                options={'maxiter': 200, 'rhofac': 1.2, 'catol': 1e-14})
                hc = repair(res5.x, n)
                c5c = compute_c5(hc, dx)
                if c5c < best_c5:
                    h_best = hc
                    best_c5 = c5c
            except Exception:
                pass

    # Final verification
    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, dx)
    return (h_best, best_c5, n)
