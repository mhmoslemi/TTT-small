                               options={'maxiter': 200, 'ftol': 1e-14})
                x_r = repair(res.x, best_n)
                c5 = compute_c5(x_r, 2.0 / best_n)
                if c5 < best_c5:
                    best_h = x_r
                    best_c5 = c5
            except Exception:
                pass

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
