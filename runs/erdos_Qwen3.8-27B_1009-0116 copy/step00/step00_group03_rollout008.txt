            if room < 1e-15:
                break
            h = np.clip(h + diff * (1.0 - h) / room, 0.0, 1.0)
        else:
            room = h.sum()
            if room < 1e-15:
                break
            h = np.clip(h + diff * h / room, 0.0, 1.0)
    return h

def penalized_obj(x, dx, n, mu):
    c5v = compute_c5(x, dx)
    sv = np.sum(x) - n * 0.5
    return c5v + mu * sv * sv

def slsqp_optimize(h, n, dx, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(penalized_obj, h.copy(), args=(dx, n, 1e6),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
        return repair_h(res.x, n)
    except Exception:
        return h

def binary_local_search(h_bin, n, dx, max_passes, rng):
    h = h_bin.copy().astype(float)
    best_c5 = compute_c5(h, dx)
    for _ in range(max_passes):
        improved = False
        oi = rng.permutation(np.where(h > 0.5)[0])
        zi = rng.permutation(np.where(h < 0.5)[0])
        for i in oi:
            for j in zi:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def to_binary(h, n, rng):
    hb = np.where(h > 0.5, 1.0, 0.0)
    half = n // 2
    n1 = int(hb.sum())
    if n1 > half:
        oi = np.where(hb == 1)[0]
        idxs = rng.choice(oi, n1 - half, replace=False)
        hb[idxs] = 0.0
    elif n1 < half:
        zi = np.where(hb == 0)[0]
        idxs = rng.choice(zi, half - n1, replace=False)
        hb[idxs] = 1.0
    return hb

def time_remaining(t0, budget_s):
    return budget_s - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(70.0)
        h0[rng.choice(70, 35, replace=False)] = 1.0
    n = len(h0)
    dx = 2.0 / n
    h0 = repair_h(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Continuous optimization from initial
    if time_remaining(t0, budget_s) > 120:
        h1 = slsqp_optimize(best_h, n, dx, maxiter=1000)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # Phase 2: Binary local search from optimized
    if time_remaining(t0, budget_s) > 60 and n % 2 == 0:
        hb = to_binary(best_h, n, rng)
        hb, cb = binary_local_search(hb, n, dx, 300, rng)
        if cb < best_c5:
            best_c5, best_h = cb, hb

    # Phase 3: Iterative perturbation + multi-method optimization
    for it in range(80):
        if time_remaining(t0, budget_s) < 50:
            break
        scale = rng.uniform(0.03, 0.30)
        hp = repair_h(best_h + rng.normal(0.0, scale, n), n)
        if time_remaining(t0, budget_s) > 40:
            ho = slsqp_optimize(hp, n, dx, maxiter=500)
            co = compute_c5(ho, dx)
            if co < best_c5:
                best_c5, best_h = co, ho
        if time_remaining(t0, budget_s) > 30 and n % 2 == 0:
            hb = to_binary(best_h, n, rng)
            hb, cb = binary_local_search(hb, n, dx, 150, rng)
            if cb < best_c5:
                best_c5, best_h = cb, hb

    # Phase 4: Random binary restarts for global exploration
    for it in range(30):
        if time_remaining(t0, budget_s) < 30:
            break
        if n % 2 != 0:
            break
        hb = np.zeros(n)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        hb, cb = binary_local_search(hb, n, dx, 200, rng)
        if cb < best_c5:
            best_c5, best_h = cb, hb

    # Phase 5: Final continuous polish with high iteration count
    if time_remaining(t0, budget_s) > 100:
        ho = slsqp_optimize(best_h, n, dx, maxiter=3000)
        co = compute_c5(ho, dx)
        if co < best_c5:
            best_c5, best_h = co, ho
        # Second round of binary polish on the polished result
        if time_remaining(t0, budget_s) > 40 and n % 2 == 0:
            hb = to_binary(best_h, n, rng)
            hb, cb = binary_local_search(hb, n, dx, 300, rng)
            if cb < best_c5:
                best_c5, best_h = cb, hb

    best_h = repair_h(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
