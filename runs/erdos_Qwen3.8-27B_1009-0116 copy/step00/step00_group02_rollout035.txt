def all_single_swaps_improve(h_bin, n):
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]
    best_key = compute_c5(h_bin)
    best_swap = None
    for oi in ones:
        for zi in zeros:
            h_trial = h_bin.copy()
            h_trial[oi] = 0.0
            h_trial[zi] = 1.0
            c5_t = compute_c5(h_trial)
            if c5_t < best_key - 1e-14:
                best_key = c5_t
                best_swap = (oi, zi)
    return best_swap, best_key

def nelder_objective(x, n):
    h = project_feasible(x, n)
    return compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = 45
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    except Exception:
        pass
    if h_init is None:
        h_init = np.full(n, 0.5)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Subgradient descent
    elapsed = time.time() - t0
    if budget_s - elapsed > 180:
        h1, c5_1 = subgrad_descent(h_init, n, 30000, 0.02, budget_s - elapsed - 150)
        if c5_1 < best_c5:
            best_c5, best_h, n = c5_1, h1, len(h1)

    # Phase 2: Binary swap optimization
    elapsed = time.time() - t0
    if budget_s - elapsed > 120:
        h_bin = binary_round(best_h, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        improved = True
        while improved and time.time() - t0 < budget_s - 80:
            improved = False
            for _ in range(50):
                if time.time() - t0 > budget_s - 80:
                    break
                h_bin = np.asarray(best_h)
                if h_bin.min() < 0.01 or h_bin.max() > 0.99:
                    h_bin = binary_round(h_bin, n)
                swap, c5_new = all_single_swaps_improve(h_bin, n)
                if swap is not None:
                    h_bin[swap[0]] = 0.0
                    h_bin[swap[1]] = 1.0
                    improved = True
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_bin.copy()
            if not improved:
                # Try double swaps
                h_bin = np.asarray(best_h)
                if h_bin.min() < 0.01 or h_bin.max() > 0.99:
                    h_bin = binary_round(h_bin, n)
                ones = np.where(h_bin > 0.5)[0]
                zeros = np.where(h_bin < 0.5)[0]
                c5_cur = compute_c5(h_bin)
                for _ in range(200):
                    if time.time() - t0 > budget_s - 80:
                        break
                    o1, o2 = np.random.choice(len(ones), 2, replace=False)
                    z1, z2 = np.random.choice(len(zeros), 2, replace=False)
                    h_tri = h_bin.copy()
                    h_tri[ones[o1]] = 0.0
                    h_tri[ones[o2]] = 0.0
                    h_tri[zeros[z1]] = 1.0
                    h_tri[zeros[z2]] = 1.0
                    c5_tri = compute_c5(h_tri)
                    if c5_tri < c5_cur - 1e-14:
                        h_bin = h_tri
                        c5_cur = c5_tri
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h_bin.copy()
                        ones = np.where(h_bin > 0.5)[0]
                        zeros = np.where(h_bin < 0.5)[0]

    # Phase 3: Nelder-Mead continuous polish
    elapsed = time.time() - t0
    if budget_s - elapsed > 40:
        n_cur = len(best_h)
        try:
            res = minimize(nelder_objective, best_h.copy(), args=(n_cur,),
                           method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12,
                                    'adaptive': True})
            h_nm = project_feasible(res.x, n_cur)
            c5_nm = compute_c5(h_nm)
            if c5_nm < best_c5:
                best_c5, best_h = c5_nm, h_nm
                n = n_cur
        except Exception:
            pass

    # Phase 4: More subgradient rounds with restarts
    elapsed = time.time() - t0
    for attempt in range(10):
        if time.time() - t0 > budget_s - 20:
            break
        h_start = best_h + np.random.randn(n) * 0.08
        h_start = project_feasible(h_start, n)
        h4, c5_4 = subgrad_descent(h_start, n, 8000, 0.01, min(budget_s - time.time() - t0 - 10, 60))
        if c5_4 < best_c5:
            best_c5, best_h, n = c5_4, h4, len(h4)

    # Phase 5: Final binary polish
    elapsed = time.time() - t0
    if budget_s - elapsed > 20:
        h_bin = binary_round(best_h, n)
        c5_bin = compute_c5(h_bin)
        for _ in range(200):
            if time.time() - t0 > budget_s - 5:
                break
            swap, c5_new = all_single_swaps_improve(h_bin, n)
            if swap is None:
                break
            h_bin[swap[0]] = 0.0
            h_bin[swap[1]] = 1.0
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_bin.copy()
            else:
                break

    return best_h, best_c5, len(best_h)
