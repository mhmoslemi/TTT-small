Let me also think about whether I should use the explicit constraint approach or the smooth max approach. For n=45, the explicit approach has 89 inequality constraints + 1 equality. That's manageable. For n=100, it's 199 inequalities. For n=200, 399. SLSQP might struggle with 399 constraints.

I'll use:
- Smooth max for all n (fast, no constraint explosion)
- Explicit constraints only for small n (n ≤ 80) as refinement

Let me also add a gradient for the smooth max to speed up SLSQP:

d/dh_i [sum_k exp(corr[k]*dx/beta) * beta / sum_k exp(corr[k]*dx/beta)] * ... 

Actually, the gradient of log-sum-exp is complicated for the correlation. Let me just let SLSQP compute numerical gradients. For n ≤ 200, that's 200 gradient evaluations per iteration, each requiring computing correlations. Should be fast enough.

Let me finalize my approach:

1. Start with initial_h_values
2. For each resolution n in [45, 80, 120, 200]:
   a. Upsample best solution to n points
   b. Run SLSQP with smooth max (beta = 0.1, then 0.01, then 0.001)
   c. For n ≤ 80, also run SLSQP with explicit constraints
   d. Track best
3. Return best

Let me also add random perturbations to avoid local optima.

One more thought: the initial_h_values has 45 points. The constraint is sum(h) = 45/2 = 22.5. When I upsample to n points, I need sum = n/2.

```python
import numpy as np
from scipy.optimize import minimize
import time
import functools

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def upsample_linear(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    old_x = np.linspace(0, 2, n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

def enforce_constraints(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = corr * dx
    m = np.max(vals)
    if beta < 1e-12:
        return m
    shifted = (vals - m) / beta
    shifted = np.clip(shifted, -200, 200)
    return m + beta * np.log(np.sum(np.exp(shifted)))

def smooth_max_fn(x, n, beta):
    return smooth_max_obj(x, n, beta)

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def corr_ineq(x, n, k, dx):
    h = x[:n]
    c = np.correlate(h, 1.0 - h, mode="full")
    return x[n] - c[k] * dx

def corr_ineq_jac(x, n, k, dx):
    # Numerical gradient is fine; return None to let SLSQP approximate
    return None

def t_obj(x, n, k, dx):
    return x[n]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h_init = np.asarray(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h_init = np.ones(45) * 0.5

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Resolutions to try
    n_init = len(h_init)
    resolutions = sorted(set([n_init, 80, 120, 200, 300]))

    for n in resolutions:
        if time.time() - t0 > budget_s * 0.82:
            break

        # Multiple trials with perturbation
        n_trials = max(1, int(min(4, budget_s * 0.15 / max(1, (2*n-1) * n * 1e-5))))
        n_trials = max(n_trials, 2)

        for trial in range(n_trials):
            if time.time() - t0 > budget_s * 0.88:
                break

            h0 = upsample_linear(best_h, n)
            if trial > 0:
                scale = 0.15 / trial
                h0 = h0 + rng.randn(n) * scale
            h0 = enforce_constraints(h0, n)
            current_h = h0.copy()

            # Smooth max annealing
            for beta in [0.5, 0.1, 0.01, 0.001]:
                if time.time() - t0 > budget_s * 0.88:
                    break

                cons = [{'type': 'eq',
                         'fun': sum_eq,
                         'jac': sum_eq_jac,
                         'args': (n,)}]
                bounds = [(0.0, 1.0)] * n

                res = minimize(smooth_max_fn, current_h, method='SLSQP',
                              args=(n, beta), bounds=bounds, constraints=cons,
                              options={'maxiter': 150, 'ftol': 1e-7})

                if res.success:
                    current_h = enforce_constraints(res.x, n)
                    c5_curr = compute_c5(current_h)
                    if c5_curr < best_c5:
                        best_h = current_h.copy()
                        best_c5 = c5_curr
                else:
                    break

        # Explicit constraint refinement for small n
        if n <= 100 and time.time() - t0 < budget_s * 0.6:
            n_corr = 2 * n - 1
            dx = 2.0 / n
            h_ref = upsample_linear(best_h, n)
            h_ref = enforce_constraints(h_ref, n)

            t_init = compute_c5(h_ref) + 0.05
            x0 = np.concatenate([h_ref, [t_init]])

            cons = [{'type': 'eq',
                     'fun': lambda x: sum_eq(x[:n], n),
                     'args': (n,)}]

            # Use a wrapper to avoid lambda issues
            def eq_fun(x):
                return sum_eq(x[:n], n)

            def eq_jac(x):
                g = np.zeros(n + 1)
                g[:n] = 1.0
                return g

            cons2 = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

            for k in range(n_corr):
                cons2.append({'type': 'ineq', 'fun': corr_ineq,
                              'args': (n, k, dx)})

            bounds = [(0.0, 1.0)] * n + [(0.05, 1.0)]

            res = minimize(lambda x: x[n], x0, method='SLSQP',
                          bounds=bounds, constraints=cons2,
                          options={'maxiter': 500, 'ftol': 1e-8})

            if res.success:
                h_new = enforce_constraints(res.x[:n], n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new

    # Try to do one more polish at the best resolution found
    n_best = len(best_h)
    if time.time() - t0 < budget_s * 0.85:
        for beta in [0.0005, 0.0001, 0.00001]:
            if time.time() - t0 > budget_s * 0.92:
                break
            cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac,
                     'args': (n_best,)}]
            bounds = [(0.0, 1.0)] * n_best
            res = minimize(smooth_max_fn, best_h, method='SLSQP',
                          args=(n_best, beta), bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_new = enforce_constraints(res.x, n_best)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = enforce_constraints(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
```
