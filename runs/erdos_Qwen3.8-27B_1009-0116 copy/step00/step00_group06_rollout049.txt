    return np.max(corr) * dx

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Smooth max
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m)) / len(corr)) / alpha

def sum_constraint(h):
