def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n_points):
    target = n_points / 2.0
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    # Distribute difference to elements that can absorb it
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if np.sum(mask) > 0:
        h[mask] += diff / np.float64(np.sum(mask))
        h = np.clip(h, 0.0, 1.0)
    # Fix residual
    for _ in range(n_points + 1):
        residual = target - np.sum(h)
        if abs(residual) < 1e-12:
            break
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) == 0:
                break
            i = idx[np.argmax(h < 1.0 - h[mask])]  # most room
            # Actually just pick first one with room
            i = idx[0]
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) == 0:
                break
            i = idx[0]
        h[i] += residual
        h[i] = np.clip(h[i], 0.0, 1.0)
    return h

def objective(h, n_points):
    return compute_c5(h, n_points)
