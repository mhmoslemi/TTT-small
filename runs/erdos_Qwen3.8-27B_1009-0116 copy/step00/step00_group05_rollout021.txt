    return dx * np.max(corr)

    desired = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(20):

        if abs(diff) < 1e-10:
            break
        if diff > 0:

            room = 1.0 - h
            mask = room > 1e-10
            if not np.any(mask):
                break
            step = np.minimum(diff, room[mask].min()) * 0.9
            h[mask] += step
        else:
            # Need to decrease sum; subtract from elements > 0
            room = h
            mask = room > 1e-10
            if not np.any(mask):

            step = np.minimum(-diff, room[mask].min()) * 0.9
            h[mask] -= step

    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:

        n_mask = np.sum(mask)
        if n_mask > 0:

            h[mask] += adj
            h = np.clip(h, 0, 1)

def compute_c5(h):

def repair(h, n):
    desired = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):

        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:

        np.clip(h, 0.0, 1.0, out=h)
    diff = desired - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

    return h
