import numpy as np
import time

def project_feasible(x, target_sum):
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def eval_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def soft_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    w = np.exp(beta * (corr - cmax))
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(corr))
        w[corr.argmax()] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        s = m - (n - 1)
        wm = w[m]
        # (1-h[j-s]) for valid j, j-s
        j1, j2 = max(0, s), min(n, s + n)
        if j1 < j2:
            grad[j1:j2] += wm * (1.0 - h[j1 - s:j2 - s])
        # -h[j+s] for valid j, j+s
        j3, j4 = max(0, -s), min(n, n - s)
        if j3 < j4:
            grad[j3:j4] -= wm * h[j3 + s:j4 + s]
    return grad

def optimize(h0, dx, time_limit, seed):
    n = len(h0)
    target_sum = n / 2.0
    h = project_feasible(h0.astype(float), target_sum)
    best_c5 = eval_c5(h, dx)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    lr = 0.03
    beta = 60.0
    start = time.time()
    stale = 0
    while time.time() - start < time_limit:
        g = soft_grad(h, dx, beta)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-10:
            break
        g = g / gnorm * max(gnorm, 1e-8)
        improved = False
        for _ in range(25):
            h_new = project_feasible(h - lr * g, target_sum)
            c5_new = eval_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                lr = min(lr * 1.15, 0.3)
                stale = 0
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr = 0.03
            stale += 1
            if stale > 30:
                h = best_h + rng.randn(n) * 0.08
                h = project_feasible(h, target_sum)
                stale = 0
            elif stale > 10:
                beta = min(beta * 1.4, 300.0)
                if beta > 200:
                    h = project_feasible(best_h + rng.randn(n) * 0.04, target_sum)
                    beta = 60.0
    return best_h, best_c5
