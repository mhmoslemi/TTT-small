```python
def cutting_plane_step(h, n, max_planes=50):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    A_ub = []
    b_ub = []
    
    # Compute current C values and gradients
    C = np.correlate(h, 1.0 - h, mode="full") * dx
    
    for k_idx in range(n_corr):
        offset = k_idx - (n - 1)
        grad_k = np.zeros(n)
        for i in range(max(0, -offset), min(n, n - offset)):
            ia = i + offset
            ib = i
            grad_k[ia] += (1.0 - h[ib]) * dx
            grad_k[ib] -= h[ia] * dx
        
        # Cutting plane: grad_k^T h - t ≤ C_k - grad_k^T h_current
        row = np.append(grad_k, -1.0)
        rhs = C[k_idx] - grad_k @ h
        A_ub.append(row)
        b_ub.append(rhs)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 ≤ h ≤ 1, t free
    bounds = [(0, 1)] * n + [(None, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None
```

```python
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_smooth_grad(h, tau=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = tau * corr * dx
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-10:
            continue
        offset = k_idx - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        for i in range(i_lo, i_hi):
            ia = i + offset
            ib = i
            grad[ia] += wk * (1.0 - h[ib])
            grad[ib] -= wk * h[ia]
    return grad * dx

def optimize_smooth(h, n, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.08
    tau = 80.0
    stalled = 0
    while time.time() < deadline:
        grad = compute_smooth_grad(h, tau)
        h_new = h - step * grad
        h_new = project_box_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-9:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.05, 0.15)
            stalled = 0
        else:
            step *= 0.92
            tau = max(tau * 0.97, 10.0)
            stalled += 1
            if stalled > 200:
                break
        if step < 1e-9:
            break
    return best_h, best_c5

def cutting_piecewise_grad(h, n, top_k=20):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-top_k:]
    A_ub = []
    b_ub = []
    for k_idx in top_idx:
        offset = k_idx - (n - 1)
        g = np.zeros(n)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        for i in range(i_lo, i_hi):
            ia = i + offset
            ib = i
            g[ia] += (1.0 - h[ib])
            g[ib] -= h[ia]
        g *= dx
        row = np.append(g, -1.0)
        rhs = corr[k_idx] * dx - g @ h
        A_ub.append(row)
        b_ub.append(rhs)
    return np.array(A_ub), np.array(b_ub)

def optimize_cutting_plane(h, n, deadline, max_iter=500):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        A_ub, b_ub = cutting_piecewise_grad(h, n, top_k=25)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        h_new = res.x[:n]
        h_new = project_box_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-8:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = 0.5 * h + 0.5 * h_new
            h = project_box_sum(h, n)
            c5_check = compute_c5(h)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h.copy()
    return best_h, best_c5

def local_search_anneal(h, n, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    temp = 0.005
    while time.time() < deadline and temp > 1e-6:
        n_pert = np.random.randint(1, 6)
        idx = np.random.choice(n, n_pert, replace=False)
        h_new = h.copy()
        h_new[idx] += np.random.normal(0, temp, n_pert)
        h_new = project_box_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            temp = min(temp * 1.02, 0.01)
        else:
            accept = np.random.random() < np.exp(-(c5_new - best_c5) / max(temp, 1e-8))
            if accept:
                h = h_new
        temp *= 0.9997
        if n_pert < 1:
            break
    return best_h, best_c5

def project_to_grid(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_box_sum(h_new, n_new)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    else:
        n = 95
        h = np.full(n, 0.5)
    h = project_box_sum(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Smooth gradient descent on original grid
    d1 = t_start + budget_s * 0.35
    h1, c5_1 = optimize_smooth(h, n, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Cutting plane refinement
    d2 = t_start + budget_s * 0.55
    h2, c5_2 = optimize_cutting_plane(best_h, n, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Local search / annealing
    d3 = t_start + budget_s * 0.75
    h3, c5_3 = local_search_anneal(best_h, n, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Try finer grid
    remaining = deadline - time.time()
    if remaining > 60 and n < 199:
        n2 = 199 if remaining > 150 else 149
        h_fine = project_to_grid(best_h, n, n2)
        d4a = t_start + min(budget_s * 0.9, time.time() + remaining * 0.5)
        hf1, cf1 = optimize_smooth(h_fine, n2, d4a)
        if cf1 < best_c5:
            best_c5 = cf1
            best_h = hf1.copy()
        d4b = t_start + min(budget_s * 0.95, time.time() + (deadline - time.time()) * 0.4)
        hf2, cf2 = optimize_cutting_plane(best_h, n2, d4b)
        if cf2 < best_c5:
            best_c5 = cf2
            best_h = hf2.copy()

    # Phase 5: Final polish
    remaining = deadline - time.time()
    if remaining > 20:
        hp, cp = local_search_anneal(best_h, len(best_h), t_start + budget_s - 5)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()

    # Final feasibility check
    final_n = len(best_h)
    best_h = project_box_sum(best_h, final_n)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, final_n
```
