import numpy as np
from scipy.optimize import minimize
import time

def get_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    lag = k_max - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)
    # -h[j+lag] where 0 <= j+lag < n
    v2 = (j + lag >= 0) & (j + lag < n)
    grad[v2] -= h[j[v2] + lag]
    # +(1-h[j-lag]) where 0 <= j-lag < n
    v1 = (j - lag >= 0) & (j - lag < n)
    grad[v1] += (1.0 - h[j[v1] - lag])
    return dx * grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h) - n / 2.0
        if abs(s) < 1e-9:
            break
        if s > 0:
            mask = h > 1e-6
            if mask.sum() == 0:
                break
            h[mask] -= s / mask.sum()
        else:
            mask = h < 1.0 - 1e-6
            if mask.sum() == 0:
                break
            h[mask] += (-s) / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

            continue
        lag = idx - (n - 1)
        v2 = (j + lag >= 0) & (j + lag < n)
        grad[v2] -= weights[idx] * h[j[v2] + lag]
        v1 = (j - lag >= 0) & (j - lag < n)
        grad[v1] += weights[idx] * (1.0 - h[j[v1] - lag])
    return dx * grad


def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h) - n / 2.0
        if abs(s) < 1e-13:
            break
        if s > 0:
            mask = h > 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] -= s / cnt
        else:
            mask = h < 1.0 - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += (-s) / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _con_fun(x, n):
    return np.sum(x) - n / 2.0


def _con_jac(x, n):
    return np.ones(n)


def _slsqp_opt(h_init, n, maxiter, ftol, use_smooth_grad=False):
    if len(h_init) != n:
        xo = np.linspace(0.0, 2.0, len(h_init), endpoint=False)
        xn = np.linspace(0.0, 2.0, n, endpoint=False)
        h_init = np.interp(xn, xo, h_init, period=2.0)
    h_w = project_feasible(h_init, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    jac_fn = get_c5_grad_smooth if use_smooth_grad else get_c5_grad
    try:
        res = minimize(get_c5, h_w, jac=jac_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        h_out = project_feasible(res.x, n)
        return h_out, get_c5(h_out)
    except Exception:
        return h_w, get_c5(h_w)


def _interpolate(h_from, n_to):
    xo = np.linspace(0.0, 2.0, len(h_from), endpoint=False)
    xn = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return np.interp(xn, xo, h_from, period=2.0)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(45)
        h0[:22] = 1.0
        h0[22] = 0.5

    n0 = len(h0)
    h_best = project_feasible(h0, n0)
    c5_best = get_c5(h_best)

    def time_ok(frac):
        return (time.time() - t0) < budget_s * frac

    # Phase 1: Optimize at native resolution
    for a in range(12):
        if not time_ok(0.45):
            break
        if a == 0:
            h_try = h_best.copy()
            sm = False
        elif a == 1:
            h_try = h_best.copy()
            sm = True
        else:
            scale = max(0.02, 0.2 * (1.0 - a / 15.0))
            h_try = np.clip(h_best + rng.randn(n0) * scale, 0, 1)
            sm = (a % 3 == 0)
        h_r, c5_r = _slsqp_opt(h_try, n0, 500, 1e-12, use_smooth_grad=sm)
        if c5_r < c5_best:
            c5_best = c5_r
            h_best = h_r.copy()

    # Phase 2: Try higher resolutions
    res_list = []
    for mult in [2, 3, 4, 5, 6, 8]:
        nt = n0 * mult
        if n0 < nt <= 600:
            res_list.append(nt)
    for extra in [60, 90, 120, 150, 180, 240, 360, 480]:
        if extra > n0 and extra <= 600 and extra not in res_list:
            res_list.append(extra)
    res_list.sort()

    for n_try in res_list:
        if not time_ok(0.78):
            break
        h_interp = _interpolate(h_best, n_try)
        h_interp = project_feasible(h_interp, n_try)
        n_st = 3 if n_try <= 120 else (2 if n_try <= 200 else 1)
        mi = 400 if n_try <= 90 else (250 if n_try <= 180 else (150 if n_try <= 300 else 80))
        for s in range(n_st):
            if not time_ok(0.82):
                break
            h_try = h_interp.copy()
            if s > 0:
                h_try = np.clip(h_interp + rng.randn(n_try) * 0.07, 0, 1)
            sm = (s == 1)
            h_r, c5_r = _slsqp_opt(h_try, n_try, mi, 1e-10, use_smooth_grad=sm)
            if c5_r < c5_best:
                c5_best = c5_r
                h_best = h_r.copy()

    # Phase 3: Final polish
    nf = len(h_best)
    for _ in range(5):
        if not time_ok(0.96):
            break
        h_try = np.clip(h_best + rng.randn(nf) * 0.015, 0, 1)
        h_r, c5_r = _slsqp_opt(h_try, nf, 600, 1e-13, use_smooth_grad=False)
        if c5_r < c5_best:
            c5_best = c5_r
            h_best = h_r.copy()

    h_best = np.clip(h_best, 0.0, 1.0)
    c5_final = get_c5(h_best)
    return h_best, c5_final, len(h_best)
