def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h


def obj_smooth(h, n, dx, t):

def coordinate_descent(h, n, max_sweeps=50):
    h = h.copy()
    target = n * 0.5

            # Simple: try a few candidates
            candidates = [0.0, 1.0, old, old * 0.5, 0.5 + (old - 0.5) * 0.5]
            best_val = best
            best_cand = old
            for cand in candidates:
                if abs(cand - best_cand) < 1e-10:
                    continue
                h[i] = cand
                # Fix sum by adjusting a neighbor
                diff = target - h.sum()
                j = (i + 1) % n

            break
    return h, best
