            rr = np.random.default_rng(seed + 500000 + r)
            ch += rr.normal(0, 0.02, n)
            ch = fix_sum(ch, ts)
        hn, c5n = sa_cont(ch, seed + 200000 + r * 31, t0, tl)
        if c5n < bc5:
            bc5, bh = c5n, hn

    # Phase 4: Scipy polish
    if time.time() - t0 < tl - 5:
        hn, c5n = scipy_polish(bh, t0, tl)
        if c5n < bc5:
            bc5 = c5n
            bh = fix_sum(hn, ts)

    # Phase 5: One more fine continuous pass
    if time.time() - t0 < tl - 5:
        hn, c5n = sa_cont(bh, seed + 999000, t0, tl)
        if c5n < bc5:
            bc5 = c5n
            bh = hn

    bh = fix_sum(bh, ts)
    return bh, compute_c5(bh), n
