    for _ in range(steps):
        if time.time() - ts > max_time or T < 0.0005:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = ones[random.randint(0, len(ones) - 1)]
        j = zeros[random.randint(0, len(zeros) - 1)]
        cur[i] = 0
        cur[j] = 1
        nc = compute_c5(cur)
        d = nc - cc5
        if d <= 0 or random.random() < np.exp(-d / max(T, 1e-10)):
            cc5 = nc
            if nc < bc5:
                bc5 = nc
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
        T *= 0.99985
    return best, bc5

def sa_2swap(start, T0, steps, max_time):
    cur = start.copy()
    cc5 = compute_c5(cur)
    best = cur.copy()
    bc5 = cc5
    T = T0
    ts = time.time()
    n = len(cur)
    for _ in range(steps):
        if time.time() - ts > max_time or T < 0.0005:
            break
        ones = list(np.where(cur == 1)[0])
        zeros = list(np.where(cur == 0)[0])
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1, i2 = random.sample(ones, 2)
        j1, j2 = random.sample(zeros, 2)
        cur[i1] = 0; cur[i2] = 0; cur[j1] = 1; cur[j2] = 1
        nc = compute_c5(cur)
        d = nc - cc5
        if d <= 0 or random.random() < np.exp(-d / max(T, 1e-10)):
            cc5 = nc
            if nc < bc5:
                bc5 = nc
                best = cur.copy()
        else:
            cur[i1] = 1; cur[i2] = 1; cur[j1] = 0; cur[j2] = 0
        T *= 0.9997
    return best, bc5

def continuous_refine(h_bin, max_time=200.0):
    n = len(h_bin)
    target = n / 2.0
    dx = 2.0 / n
    h = (h_bin + 0.08 * np.random.randn(n)).astype(np.float64)
    h = np.clip(h, 0, 1)
    h += (target - h.sum()) / n
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ts = time.time()
    lr = 0.04
    stall = 0
    while time.time() - ts < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        cv = dx * np.max(corr)
        top = np.argsort(corr)[-4:]
        grad = np.zeros(n)
        for j in top:
            k = j - (n - 1)
            for p in range(n):
                if 0 <= p - k < n:
                    grad[p] += (1.0 - h[p - k])
                if 0 <= p + k < n:
                    grad[p] -= h[p + k]
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        hp = np.clip(h - lr * grad / gn, 0, 1)
        for _ in range(2):
            hp += (target - hp.sum()) / n
            hp = np.clip(hp, 0, 1)
        npv = compute_c5(hp)
        if npv < cv - 1e-14:
            h = hp
            cv = npv
            if cv < best_c5:
                best_c5 = cv
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.01, 0.15)
        else:
            lr *= 0.88
            stall += 1
            if stall > 50:
                p = random.randint(0, n - 1)
                h[p] = np.clip(h[p] + random.gauss(0, 0.12), 0, 1)
                h += (target - h.sum()) / n
                h = np.clip(h, 0, 1)
                lr = 0.04
                stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    t0 = time.time()
    n = len(initial_h_values)
    target = n // 2

    h0 = make_valid_binary(initial_h_values, n, target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Also try original fractional values
    h_frac = initial_h_values.astype(np.float64).copy()
    h_frac = np.clip(h_frac, 0, 1)
    h_frac += (target - h_frac.sum()) / n
    h_frac = np.clip(h_frac, 0, 1)
    cf = compute_c5(h_frac)
    if cf < best_c5:
        best_c5 = cf
        best_h = h_frac.copy()

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SA with single-swap moves
    n_iter = 0
    while rem() > 150:
        if n_iter == 0:
            start = h0.copy()
            T0 = 0.12
        else:
            start = make_valid_binary(best_h, n, target)
            npert = random.randint(6, 30)
            ones = list(np.where(start == 1)[0])
            zeros = list(np.where(start == 0)[0])
            for _ in range(npert):
                a, b = random.choice(ones), random.choice(zeros)
                start[a] = 0
                start[b] = 1
            T0 = 0.07

        sol, c5s = sa_binary(start, T0, 500000, min(rem() - 150, 100))
        if c5s < best_c5:
            best_c5 = c5s
            best_h = sol.copy()
        n_iter += 1
        if best_c5 <= 0.3751:
            break  # hit integer optimum 15/40=0.375

    # Phase 2: 2-swap SA for diversity
    while rem() > 100:
        start = make_valid_binary(best_h, n, target)
        ones = list(np.where(start == 1)[0])
        zeros = list(np.where(start == 0)[0])
        if len(ones) >= 2 and len(zeros) >= 2:
            for _ in range(random.randint(3, 12)):
                i1, i2 = random.sample(ones, 2)
                j1, j2 = random.sample(zeros, 2)
                start[i1] = 0; start[i2] = 0; start[j1] = 1; start[j2] = 1
        sol, c5s = sa_2swap(start, 0.08, 300000, min(rem() - 100, 80))
        if c5s < best_c5:
            best_c5 = c5s
            best_h = sol.copy()
        if best_c5 <= 0.3751:
            break

    # Phase 3: Exhaustive local search on best binary
    if rem() > 20:
        is_binary = np.allclose(best_h, np.round(best_h), atol=1e-6)
        if is_binary:
            ls_h, ls_c5 = full_local_search(best_h, min(rem() - 20, 45))
            if ls_c5 < best_c5:
                best_c5 = ls_c5
                best_h = ls_h.copy()

    # Phase 4: Continuous refinement
    if rem() > 20:
        if np.allclose(best_h, np.round(best_h), atol=1e-6):
            ch, cc = continuous_refine(best_h, min(rem() - 20, 180))
        else:
            ch = best_h.copy()
            ch += 0.03 * np.random.randn(n)
            ch = np.clip(ch, 0, 1)
            ch += (target - ch.sum()) / n
            ch = np.clip(ch, 0, 1)
            cc = compute_c5(ch)
            if cc < best_c5:
                best_h = ch
                best_c5 = cc
            # Short SA on continuous
            ts2 = time.time()
            T2 = 0.02
            while time.time() - ts2 < min(rem() - 20, 60) and T2 > 0.001:
                p = random.randint(0, n - 1)
                old = best_h[p]
                best_h[p] = np.clip(best_h[p] + random.gauss(0, 0.08), 0, 1)
                best_h += (target - best_h.sum()) / n
                best_h = np.clip(best_h, 0, 1)
                nc = compute_c5(best_h)
                if nc < best_c5 or random.random() < np.exp(-(nc - best_c5) / max(T2, 1e-10)):
                    best_c5 = nc
                else:
                    best_h[p] = old
                    best_h += (target - best_h.sum()) / n
                    best_h = np.clip(best_h, 0, 1)
                T2 *= 0.9995

    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    diff = target - best_h.sum()
    if abs(diff) > 1e-9:
        best_h += diff / n
        best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)
