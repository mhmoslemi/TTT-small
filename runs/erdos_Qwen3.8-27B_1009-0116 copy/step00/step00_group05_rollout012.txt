    dx = 2.0 / n

    m = np.max(corr)
    vals = T * (corr - m)
    return (m + np.log(np.sum(np.exp(vals))) / T)

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-10:

    h = np.clip(h, 0.0, 1.0)

        h2[i] += eps
        f1 = smooth_max_c5(h2, T)

import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-10:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-10:
                break
    return h

def smooth_c5(h, T):
    n = len(h)

        else:

            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-10:
                break
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def grad_smooth(h, T, eps=1e-4):
    n = len(h)
    g = np.zeros(n)
    f0 = smooth_c5(h, T)
    for i in range(n):
        hp = h.copy()
        hp[i] = min(1.0, h[i] + eps)
        d = hp[i] - h[i]
        g[i] = (smooth_c5(hp, T) - f0) / d if d > 1e-12 else 0.0
    return g

def continuous_opt(h0, n, tb, rng):
    h = h0.copy()
    best = evaluate_c5(h)
    hb = h.copy()
    lr = 0.02
    T = 15.0
    t0 = time.time()
    for it in range(5000):
        if time.time() - t0 > tb:
            break
        g = grad_smooth(h, T)
        h_new = project_feasible(h - lr * g, n)
        c = evaluate_c5(h_new)
        if c < best:
            best = c
            hb = h_new.copy()
            h = h_new
        else:
            lr *= 0.96
        if it % 300 == 299:
            T *= 1.15
            lr = max(lr, 0.008)
    return hb, best

def binary_cd(h, max_rounds, ncand, tb, rng, sa_temp=0.0):
    n = len(h)
    best = evaluate_c5(h)
    hb = h.copy()
    t0 = time.time()
    for rnd in range(max_rounds):
        if time.time() - t0 > tb:
            break
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for i in order:
            if time.time() - t0 > tb:
                break
            if h[i] > 0.5:
                zeros = np.where(h == 0)[0]
                if len(zeros) == 0:
                    continue
                k = min(ncand, len(zeros))
                perm = rng.choice(zeros, size=k, replace=False)
                for j in perm:
                    h[i] = 0.0
                    h[j] = 1.0
                    c = evaluate_c5(h)
                    accept = c < best or (sa_temp > 0 and c - best < 0.01 and rng.rand() < np.exp(-(c - best) / sa_temp))
                    if accept and c < best:
                        best = c
                        hb = h.copy()
                        improved = True
                    elif not accept:
                        h[i] = 1.0
                        h[j] = 0.0
            else:
                ones = np.where(h == 1)[0]
                if len(ones) == 0:
                    continue
                k = min(ncand, len(ones))
                perm = rng.choice(ones, size=k, replace=False)
                for j in perm:
                    h[i] = 1.0
                    h[j] = 0.0
                    c = evaluate_c5(h)
                    accept = c < best or (sa_temp > 0 and c - best < 0.01 and rng.rand() < np.exp(-(c - best) / sa_temp))
                    if accept and c < best:
                        best = c
                        hb = h.copy()
                        improved = True
                    elif not accept:
                        h[i] = 0.0
                        h[j] = 1.0
        if not improved and sa_temp <= 0:
            break
    return hb, best

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def make_periodic(n, rng):
    p = rng.randint(5, max(6, n // 5))
    h = np.zeros(n)
    half = max(1, p // 2)
    for i in range(n):
        h[i] = 1.0 if (i % p) < half else 0.0
    n1 = int(h.sum())
    tgt = n // 2
    if n1 > tgt:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        for i in range(n1 - tgt):
            h[idx[i]] = 0
    elif n1 < tgt:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        for i in range(tgt - n1):
            h[idx[i]] = 1
    return h

def round_to_binary(hc, n):
    hb = (hc > 0.5).astype(float)
    n1 = int(hb.sum())
    tgt = n // 2
    if n1 != tgt:
        if n1 > tgt:
            idx = np.where(hb == 1)[0]
            ord_ = np.argsort(hc[idx])
            for i in range(n1 - tgt):
                hb[idx[ord_[i]]] = 0
        else:
            idx = np.where(hb == 0)[0]
            ord_ = np.argsort(-hc[idx])
            for i in range(tgt - n1):
                hb[idx[ord_[i]]] = 1
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 70

    par_avail = ('initial_h_values' in globals() and initial_h_values is not None)
    par_n = len(initial_h_values) if par_avail else 0

    ns = [70, 80, 60, 100, 50, 90, 120]
    if par_avail and par_n in ns:
        ns.remove(par_n)
        ns.insert(0, par_n)

    for ni, n in enumerate(ns):
        if time.time() - t0 > budget_s * 0.95:
            break
        if ni == 0 and par_avail and par_n == n:
            tb = budget_s * 0.45
        elif ni == 0:
            tb = budget_s * 0.25
        else:
            tb = budget_s * 0.065

        # Phase 1: continuous optimization
        if par_avail and ni == 0 and par_n == n:
            h0 = project_feasible(np.array(initial_h_values, dtype=float), n)
        else:
            h0 = make_periodic(n, rng)

        hc, c5c = continuous_opt(h0, n, tb * 0.3, rng)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()
            best_n = n

        # Phase 2: binary CD from rounded continuous
        hb = round_to_binary(hc, n)
        hcd, c5cd = binary_cd(hb, 2000, 50, tb * 0.45, rng)
        if c5cd < best_c5:
            best_c5 = c5cd
            best_h = hcd.copy()
            best_n = n

        # Phase 3: SA-style CD to escape local optima
        hcd2, c5cd2 = binary_cd(hcd, 800, 50, tb * 0.15, rng, sa_temp=0.003)
        if c5cd2 < best_c5:
            best_c5 = c5cd2
            best_h = hcd2.copy()
            best_n = n

        # Phase 4: fresh binary starts with CD
        for _ in range(2):
            if time.time() - t0 > budget_s * 0.93:
                break
            h_r = make_binary(n, rng)
            h_cd3, c5_3 = binary_cd(h_r, 500, 40, tb * 0.1, rng)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h_cd3.copy()
                best_n = n

    if best_h is None:
        n = 70
        best_h = make_binary(n, rng)
        best_c5 = evaluate_c5(best_h)
        best_n = n

    # Final validation
    best_c5 = evaluate_c5(best_h)
    s = np.sum(best_h) * (2.0 / len(best_h))
    if abs(s - 1.0) > 0.01:
        best_h = project_feasible(best_h, len(best_h))
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
