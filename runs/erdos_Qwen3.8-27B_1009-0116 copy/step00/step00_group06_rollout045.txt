import numpy as np
import time
from scipy.optimize import minimize
import math

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = float(np.max(c))
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def optimize_slsqp(x0, n, alphas, maxiter=400):
    x_cur = x0.copy()
    for alpha in alphas:
        try:
            res = minimize(
                lambda x, a=alpha: soft_max_obj(x, n, a),
                x_cur, method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq',
                              'fun': lambda x: np.sum(x) - n / 2.0,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': maxiter, 'ftol': 1e-12}
            )
            if res.fun < soft_max_obj(x_cur, n, alpha) + 1e-12:
                x_cur = repair(res.x, n)
        except Exception:
            pass
    return x_cur

def coord_descent(h, n, rng, time_limit):
    t_start = time.time()
    best_c5 = compute_c5(h.copy(), n)
    deltas = [0.5, 0.3, 0.15, 0.05]
    improved = True
    while improved and (time.time() - t_start) < time_limit * 0.8:
        improved = False
        for i in range(n):
            if (time.time() - t_start) > time_limit * 0.8:
                break
            for d in deltas:
                for s in [-1, 1]:
                    nh = h.copy()
                    nh[i] = np.clip(h[i] + s * d, 0.0, 1.0)
                    nh = repair(nh, n)
                    c5_v = compute_c5(nh, n)
                    if c5_v < best_c5 - 1e-10:
                        h = nh
                        best_c5 = c5_v
                        improved = True
                        break
                if improved:
                    break
    return h, best_c5
