def project_box_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}"""
    # Binary search on lambda
    lo, hi = h.min() - target, h.max() + target  # wide bounds
    # Actually: h - lambda clipped to [0,1]
    # sum(clip(h-lambda, 0, 1)) is decreasing in lambda
    # We want sum = target
    
    def sum_at_lambda(lam):
        return np.clip(h - lam, 0, 1).sum()
    
    # Find lambda such that sum_at_lambda(lam) = target
    # sum_at_lambda is continuous and decreasing
    # At lam=0: sum_at_lambda = sum(clip(h,0,1)) >= target (hopefully)
    # At lam=1: sum_at_lambda = 0 <= target (hopefully)
    # If h already in [0,1] and sum > target, we need lam > 0
    # If h already in [0,1] and sum < target, we need lam < 0
    
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = sum_at_lambda(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

    return float(np.max(corr) * dx)

def make_random_binary(n, rng, target):

   _threshold = 1e-12

def pgd_step(h, n, alpha=20.0, step=None):
    """One projected gradient step"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    exp_c = np.exp(alpha * (corr - cmax))
    w = exp_c / np.sum(exp_c)

        wj = w[j]

            continue

            grad[s:n] += wj * (1.0 - h[0:n-s])
            grad[0:n-s] -= wj * h[s:n]
        else:
            sp = -s
            grad[0:n-sp] += wj * (1.0 - h[sp:n])
            grad[sp:n] -= wj * h[0:n-sp]
    grad *= dx

    h_new = h - step * grad

    return h_new

    lo, hi = -1.0, 1.0

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    return np.clip(h - lam, 0.0, 1.0)

            if si == 0 and h0 is not None and n == len(h0):
                h_start = h0.copy()
            elif si == 0:
                h_start = make_random_binary(n, rng, target)
            elif si == 1:
                h_start = make_random_binary(n, rng, target)
            elif si == 2 and time.time() - t0 < budget_s - 60:
                h_start = make_greedy_binary(n, rng, target)
            else:
                h_start = np.zeros(n)
                block = max(2, n // 8)
                offset = rng.randint(0, block) if block > 1 else 0
                for b in range(offset, n, block * 2):
                    h_start[b:min(b + block, n)] = 1.0
                h_start = project_box_sum(h_start, target)

            h_pgd, c5_pgd = pgd_optimize(h_start, n, t0, budget_s)

            h_bin = h_pgd.copy()
            if n % 2 == 0:
                h_bin = (h_bin > 0.5).astype(float)
                if h_bin.sum() != n // 2:
                    if h_bin.sum() > n // 2:
                        idx1 = np.where(h_bin == 1)[0]
                        h_bin[idx1[:int(h_bin.sum() - n // 2)]] = 0
                    else:
                        idx0 = np.where(h_bin == 0)[0]
                        h_bin[idx0[:int(n // 2 - h_bin.sum())]] = 1
            else:
                h_bin = (h_bin > 0.5).astype(float)
                s = h_bin.sum()
                if s > n // 2:
                    idx1 = np.where(h_bin == 1)[0]
                    h_bin[idx1[0]] = 0.5
                elif s < n // 2:
                    idx0 = np.where(h_bin == 0)[0]
                    if len(idx0) > 0:
                        h_bin[idx0[0]] = 0.5

            h_ls, c5_ls = local_search(h_bin, rng, t0, budget_s, max_iter=6000)

            if c5_ls < c5_pgd:
                h_candidate = h_ls
                c5_candidate = c5_ls
            else:
                h_candidate = h_pgd
                c5_candidate = c5_pgd

            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()

    if best_h is not None and time.time() - t0 < budget_s - 5:
        best_h, best_c5 = local_search(best_h, rng, t0, budget_s, max_iter=5000)

    return best_h, best_c5, len(best_h)
