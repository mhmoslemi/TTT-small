def compute_c5(h, dx):
    """Compute C5 = max of correlate(h, 1-h, 'full') * dx"""
    c = np.correlate(h, 1-h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > 0:
        h = h * (n/2.0) / s
    h = np.clip(h, 0, 1)
    # If still not exact, adjust
    diff = n/2.0 - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        idx = np.argsort(h)
        if diff > 0:
            for i in reversed(idx):  # increase from largest
                add = min(diff, 1-h[i])
                h[i] += add
                diff -= add
                if diff <= 0: break
        else:
            for i in idx:  # decrease from smallest
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if diff >= 0: break
    return h

def subgradient_step(h, dx, lr):
    """Compute subgradient of max_k c[k]*dx and take a step"""
    c = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(c)
    lag = max_idx - (len(h) - 1)  # the actual lag
    n = len(h)

    grad = np.zeros(n)

Let me also think about whether the subgradient approach will work well. The objective is the max of many smooth functions, and the subgradient only gives information about one function at a time. This can be inefficient.

Alternative: use a "majorize-minimize" or "barrier" approach where we penalize ALL c[k] values, not just the max:
F(h) = sum_k exp(alpha * c[k] * dx) / n  (softmax approximation of max)

This gives a smooth objective with a full gradient. As alpha → ∞, it approaches the max.

Let me use this approach:

```python
def smooth_c5(h, dx, alpha=50):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax approximation of max
    c_shifted = c - c.max()
    exp_c = np.exp(alpha * c_shifted)
    return c.max() + np.log(exp_c.sum() / len(c)) / alpha
```

d/dh[p] of smooth_c5 = (1/alpha) * sum_k exp(alpha*(c[k]-cmax)) * (dc[k]/dh[p]) / sum_k' exp(alpha*(c[k']-cmax))

This is a weighted average of the individual subgradients. Much better than just one subgradient!

    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s == 0:
        h = np.ones(n) / n * target
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break

            for i in order:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if diff >= -1e-12:
                    break
    return h

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')

Actually wait, I realize I should also try a different approach: since the problem is essentially about the binary pattern (for even n), maybe I should focus more on combinatorial search.

For a binary sequence of length n with n/2 ones:
The overlap at lag m is: count of i where h[i]=1 and h[i+m]=0 (for valid i).

This is the number of "1 followed by (m positions later) 0" patterns.

We want to minimize the MAX over all m.

This is related to the "autocorrelation" of the sequence. Let a[i] = 2*h[i] - 1 ∈ {-1, 1}. Then:
h[i](1-h[i+m]) = (1+a[i])/2 * (1-a[i+m])/2 = (1 - a[i] - a[i+m] + a[i]*a[i+m])/4

Sum over valid i:
= N_valid/4 - sum(a[i])/4 - sum(a[i+m])/4 + sum(a[i]*a[i+m])/4
= N_valid/4 - (sum_a - boundary)/4 - (sum_a - boundary)/4 + AC[m]/4

where AC[m] = sum_i a[i]*a[i+m] is the autocorrelation.

import numpy as np
from scipy.signal import correlate
import time

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        h = np.full(n, 0.5)
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
            if abs(diff) <= 1e-12:
                break
    return h

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def smooth_obj_grad(h, n, dx, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    c_shifted = np.clip((c - c_max) * alpha, -60.0, 0.0)
    exp_c = np.exp(c_shifted)
    exp_sum = exp_c.sum()
    weights = exp_c / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        i_s = max(0, -lag)
        i_e = min(n, n - lag)
        if i_s >= i_e:
            continue
        valid = np.arange(i_s, i_e)
        grad[valid] += w * (1.0 - h[valid + lag])
        grad[valid + lag] += w * (-h[valid])
    grad *= dx
    obj = c_max + np.log(exp_sum) / alpha
    return obj, grad

def subgrad_optimize(h_init, n, dx, rng, t_check, time_left_fn, max_iter=5000):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    lr = 0.03
    alpha = 60.0
    patience = 0
    for it in range(max_iter):
        if time_left_fn() < 5:
            break
        if t_check % 200 == 0:
            alpha = min(alpha * 1.5, 300.0)
        obj, grad = smooth_obj_grad(h, n, dx, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        h_new = project_feasible(h - (lr / gnorm) * grad, n)
        val = compute_c5(h_new, dx)
        if val < best_val - 1e-12:
            best_h = h_new.copy()
            best_val = val
            patience = 0
        else:
            patience += 1
        h = h_new
        if val > obj + 1e-10:
            lr *= 0.92
        else:
            lr = min(lr * 1.03, 0.15)
        if patience > 400:
            break
    return best_h, best_val
