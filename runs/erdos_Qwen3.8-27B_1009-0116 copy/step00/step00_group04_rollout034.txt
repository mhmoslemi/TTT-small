    for n in res_list:
        if elapsed() > budget_s * 0.8:
            break
        if n <= n0:
            continue
        dx = 2.0 / n
        # Interpolate
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n)

        # Also try a random binary init at this resolution
        h_rand = np.zeros(n)
        perm = rng.permutation(n)
        h_rand[perm[:n // 2]] = 1.0
        h_rand = project_feasible(h_rand, n)

        # Optimize both
        for h_init in [h_interp, h_rand]:
            if elapsed() > budget_s * 0.82:
                break
            alphas = [50, 100, 200] if n > 200 else [50, 100, 200, 400]
            mi = 100 if n > 300 else 200
            h_opt, c5_opt = optimize_block(h_init, n, dx, alphas, mi, budget_s * 0.08)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n0 = n

    # Phase 5: Final polish at best resolution
    if elapsed() < budget_s * 0.92:
        n_cur = len(best_h)
        dx_cur = 2.0 / n_cur
        try:
            res = minimize(lambda x: compute_c5(x, dx_cur), best_h,
                          method='Nelder-Mead',
                          options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12})
            h_new = project_feasible(res.x, n_cur)
            c5 = compute_c5(h_new, dx_cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
        except Exception:
            pass

        # One more smooth pass
        if elapsed() < budget_s * 0.95:
            best_h, best_c5 = optimize_block(best_h, n_cur, dx_cur,
                                             [200, 500, 1000], 100, 30)

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h, 2.0 / n_final)

    return (best_h.tolist(), best_c5, n_final)
