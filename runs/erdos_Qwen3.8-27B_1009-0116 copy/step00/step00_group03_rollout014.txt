        return h
    if diff > 0:

    else:

            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        if abs(diff) < 1e-12:
