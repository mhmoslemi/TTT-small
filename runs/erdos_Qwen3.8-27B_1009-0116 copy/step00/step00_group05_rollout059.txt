    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        h[:] = 1.0 / n * target  # uniform
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        idx = np.where((h > 0.01) & (h < 0.99))[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h):
    """Push values towards 0 or 1 while maintaining sum constraint"""
    n = len(h)
    target = n / 2.0
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly target ones
    n_ones = int(h_bin.sum())
    if n_ones > target:
        # Need to turn some 1s to 0s
        # Turn off the ones with lowest original h value
        order = np.argsort(h)  # ascending
        for i in order:
            if n_ones <= target:

            if h_bin[i] == 1:
                h_bin[i] = 0
                n_ones -= 1

        for i in order:

        best_c5 = best_c
    return best, best_c5

def ls_steepest_with_kicks(h, n_kicks, budget_per_kick, deadline, rng):
    n = len(h)
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best_local = h_cur.copy()
    c5_best_local = c5_cur
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        h_new, c5_new = ls_steepest(h_cur, 300, deadline)
        if c5_new < c5_cur - 1e-13:
            h_cur, c5_cur = h_new, c5_new
        if c5_cur < c5_best_local - 1e-13:
            best_local = h_cur.copy()
            c5_best_local = c5_cur
        # Kick: 3-6 random swaps
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_kick = int(rng.integers(3, 7))
        for _ in range(n_kick):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
            m = list(ones)
            m[m == i] = j
            ones = np.array(m)
            m2 = list(zeros)
            m2[m2 == j] = i
            zeros = np.array(m2)
    return best_local, c5_best_local

def continuous_refine(h, deadline):
    n = len(h)
    target = n / 2.0
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        s = hc.sum()
        if s < 1e-10:
            return 1.0
        hc = hc * (target / s)
        hc = np.clip(hc, 0.0, 1.0)
        return compute_c5(hc)
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': 300, 'ftol': 1e-12})
        h_out = project_feasible(res.x)
        c5_out = compute_c5(h_out)
        return h_out, c5_out
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 70
        h0 = np.zeros(n0)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0

    best_h = project_feasible(h0)
    best_c5 = compute_c5(best_h)

    # Binarize initial
    h_bin = binarize(best_h)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin

    # Phase 1: Aggressive binary local search with kicks (60% budget)
    deadline1 = t0 + budget_s * 0.6
    h_cur = binarize(best_h)
    n_kicks = 300
    h_opt, c5_opt = ls_steepest_with_kicks(h_cur, n_kicks, 0, deadline1, rng)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    # Also try starting from a fresh random binary
    deadline1b = t0 + budget_s * 0.7
    for restart in range(10):
        if time.time() > deadline1b:
            break
        h_rand = np.zeros(n0)
        h_rand[rng.choice(n0, n0 // 2, replace=False)] = 1.0
        h_o, c5_o = ls_steepest_with_kicks(h_rand, 30, 0, min(deadline1b, time.time() + 40), rng)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 2: Continuous refinement (10% budget)
    deadline2 = t0 + budget_s * 0.80
    if time.time() < deadline2:
        h_cr, c5_cr = continuous_refine(best_h, deadline2)
        if c5_cr < best_c5:
            best_h, best_c5 = h_cr, c5_cr
        # Re-binarize and search
        h_bin2 = binarize(best_h)
        h_o, c5_o = ls_steepest(h_bin2, 200, deadline2)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 3: Try other n values (15% budget)
    deadline3 = t0 + budget_s * 0.95
    for n_try in [100, 140, 50]:
        if time.time() > deadline3:
            break
        t_sub = min(40.0, deadline3 - time.time())
        h_t = np.zeros(n_try)
        h_t[rng.choice(n_try, n_try // 2, replace=False)] = 1.0
        h_o, c5_o = ls_steepest_with_kicks(h_t, 20, 0, min(deadline3, time.time() + t_sub), rng)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
        # Also try from parent interpolated
        if time.time() < deadline3:
            h_interp = np.interp(
                np.linspace(0, 2, n_try, endpoint=False),
                np.linspace(0, 2, len(best_h), endpoint=False),
                best_h
            )
            h_bid = binarize(project_feasible(h_interp))
            h_o2, c5_o2 = ls_steepest_with_kicks(h_bid, 15, 0, min(deadline3, time.time() + t_sub), rng)
            if c5_o2 < best_c5:
                best_h, best_c5 = h_o2, c5_o2

    # Final safety: one more round of steepest on best
    if time.time() < t0 + budget_s * 0.98:
        h_final = binarize(best_h)
        h_f, c5_f = ls_steepest(h_final, 100, t0 + budget_s * 0.98)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    best_h = project_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
