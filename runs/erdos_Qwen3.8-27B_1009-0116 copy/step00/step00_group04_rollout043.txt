    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    target = n // 2
    while np.sum(h) != target:
        if np.sum(h) > target:
            idx = rng.choice(np.flatnonzero(h == 1))
            h[idx] = 0.0
        else:
            idx = rng.choice(np.flatnonzero(h == 0))
            h[idx] = 1.0
    best_v = compute_c5(h, n)
    best_h = h.copy()
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    for it in range(max_iter):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        v = compute_c5(h, n)
        if v < best_v:
            best_v = v
            best_h = h.copy()
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_v


def upscale(h, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(np.clip(h_new, 0, 1), n_new)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    remaining = lambda: budget_s - (time.time() - t0)

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        n = 42
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0

    n = len(h_init)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, n)
    best_n = n

    # Phase 1: SLSQP at original resolution
    if remaining() > 150:
        try:
            def obj(hf):
                return compute_c5(hf, n)

            def objg(hf):
                _, cores = compute_c5_cores(hf, n)
                return grad_core(hf, n, int(np.argmax(cores)))

            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
                     'jac': lambda h: np.ones(n)}]
            res = minimize(obj, h_init, method='SLSQP', jac=objg,
                           bounds=[(0, 1)] * n, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-12})
            h_r = project_feasible(res.x, n)
            v = compute_c5(h_r, n)
            if v < best_c5:
                best_c5, best_h = v, h_r.copy()
        except Exception:
            pass

    # Phase 2: Subgradient passes with various lr
    for lr0 in [0.04, 0.08, 0.15, 0.025]:
        if remaining() < 80:
            break
        h_curr = best_h.copy()
        iters = int(min(8000, max(500, remaining() * 8)))
        h_curr, v = subgradient_opt(h_curr, n, iters, lr0=lr0)
        if v < best_c5:
            best_c5, best_h = v, h_curr.copy()

    # Phase 3: Binary local search
    if remaining() > 60:
        h_bin = best_h.copy()
        h_bin[h_bin > 0.5] = 1.0
        h_bin[h_bin <= 0.5] = 0.0
        h_bin, v_bin = binary_local_search(h_bin, n, max_iter=5000, rng=rng)
        if v_bin < best_c5:
            best_c5, best_h, best_n = v_bin, h_bin.copy(), n

    # Phase 4: Upscale and refine
    for factor in [2, 4]:
        if remaining() < 120:
            break
        n_new = best_n * factor
        h_up = upscale(best_h, best_n, n_new)
        # Subgradient at higher res
        h_up, v_up = subgradient_opt(h_up, n_new, 3000, lr0=0.03)
        # SLSQP at higher res
        if remaining() > 60:
            try:
                def obj2(hf):
                    return compute_c5(hf, n_new)

                def objg2(hf):
                    _, cores = compute_c5_cores(hf, n_new)
                    return grad_core(hf, n_new, int(np.argmax(cores)))

                cons2 = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_new / 2.0,
                          'jac': lambda h: np.ones(n_new)}]
                res2 = minimize(obj2, h_up, method='SLSQP', jac=objg2,
                                bounds=[(0, 1)] * n_new, constraints=cons2,
                                options={'maxiter': 150, 'ftol': 1e-10})
                h_r2 = project_feasible(res2.x, n_new)
                v2 = compute_c5(h_r2, n_new)
                if v2 < v_up:
                    h_up, v_up = h_r2, v2
            except Exception:
                pass
        if v_up < best_c5:
            best_c5, best_h, best_n = v_up, h_up.copy(), n_new

    # Phase 5: Final refinement
    if remaining() > 40:
        n_b = len(best_h)
        h_curr = best_h.copy()
        h_curr, v = subgradient_opt(h_curr, n_b, 4000, lr0=0.015)
        if v < best_c5:
            best_c5, best_h = v, h_curr.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
