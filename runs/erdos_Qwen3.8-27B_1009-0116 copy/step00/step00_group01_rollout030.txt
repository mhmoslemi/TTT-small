import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp, expit
import time

_state = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(params):
    T = _state['T']
    dx = _state['dx']
    h_loc = np.clip(params, 0.0, 1.0)
    corr = np.correlate(h_loc, 1.0 - h_loc, mode='full') * dx
    m = np.max(corr)
    if T < 1e-8:
        return float(m)
    return float(m + T * logsumexp((corr - m) / T))

def sum_con(params):
    return np.sum(params) - _state['target_sum']

def sum_con_jac(params):
    return np.ones(_state['n'])

def fix_sum(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def do_slsqp(h_start, n, maxiter, ftol):
    constraints = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_max_obj, h_start, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.x is not None:
        hn = fix_sum(res.x, n)
        cn = compute_c5(hn)
        return hn, cn
    return h_start, compute_c5(h_start)
