def compute_subgradient(h, n, dx):
    """Compute subgradient of C5 at h"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    values = corr * dx
    max_val = np.max(values)
    
    # Active set
    tol = 1e-10
    active_mask = values >= max_val - tol
    active_indices = np.where(active_mask)[0]
    
    if len(active_indices) == 0:
        return np.zeros(n), values
    
    lags = active_indices - (n - 1)  # actual shifts
    
    grad = np.zeros(n)
    for lag in lags:
        # For each i, the contribution is:
        # (1-h[i-lag]) * 1_{0<=i-lag<n} - h[i+lag] * 1_{0<=i+lag<n}
        for i in range(n):

            il = i - lag
            if 0 <= il < n:
                val += (1.0 - h[il])
            ip = i + lag
            if 0 <= ip < n:
                val -= h[ip]
            grad[i] += val

    # values[k] = f_k current value
    # Compute a_k for each lag
    lags_all = np.arange(-(n-1), n)

    # a = (1-h[i-l])*1_{valid(i-l)} - h[i+l]*1_{valid(i+l)}
    
    a = np.zeros(2*n-1)
    il = i - lags_all
    ip = i + lags_all
    valid_l = (il >= 0) & (il < n)
    valid_p = (ip >= 0) & (ip < n)
    
    a[valid_l] += (1.0 - h[il[valid_l]])
    a[valid_p] -= h[ip[valid_p]]

    b = values - dx * h[i] * a

This is again a 1D convex minimization, solvable by ternary search!

```python
def optimize_pair(h, i, j, n, dx, values):
    """Find optimal δ to add to h[i] and subtract from h[j]"""
    lags_all = np.arange(-(n-1), n)
    
    def coeff(idx):
        il = idx - lags_all
        ip = idx + lags_all
        valid_l = (il >= 0) & (il < n)
        valid_p = (ip >= 0) & (ip < n)
        a = np.zeros(2*n-1)
        a[valid_l] += (1.0 - h[il[valid_l]])
        a[valid_p] -= h[ip[valid_p]]
        return a
    
    a_i = coeff(i)
    a_j = coeff(j)
    d = dx * (a_i - a_j)
    
    # Minimize max_k (values[k] + d[k] * δ) over δ
    # s.t. -h[i] ≤ δ ≤ 1-h[i] and h[j]-1 ≤ δ ≤ h[j]
    lo = max(-h[i], h[j]-1)
    hi = min(1-h[i], h[j])
    
    if lo >= hi:
        return h, 0.0
    
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(values + d * m1)
        f2 = np.max(values + d * m2)
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    
    δ = (lo + hi) / 2
    return h, δ
```

This is clean and maintains the constraint exactly.

import numpy as np

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 0:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            idx = np.where(h > 0.0)[0]
            for i in idx:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

    return np.correlate(h, 1.0 - h, mode='full') * (2.0/n)

        else:

import numpy as np
import time


def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-15:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            for i in idx:
                if diff <= 1e-12:
                    break
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
        else:
            idx = np.where(h > 0.0)[0]
            for i in idx:
                if diff >= -1e-12:
                    break
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h


def compute_values(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_subgrad(h, n, dx, values):
    max_val = np.max(values)
    tol = 1e-6
    active = values >= max_val - tol
    n_a = np.sum(active)
    if n_a == 0:
        return np.zeros(n)
    active_idx = np.where(active)[0]
    lags = active_idx - (n - 1)
    grad = np.zeros(n)
    for lag in lags:
        if lag >= 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        else:
            grad[:n + lag] += (1.0 - h[-lag:])
            grad[-lag:] -= h[:n + lag]
    grad *= dx / n_a
    return grad


def pair_move(h, i, j, n, dx, values, rng):
    lags_all = np.arange(-(n - 1), n)
    il_i = i - lags_all
    ip_i = i + lags_all
    il_j = j - lags_all
    ip_j = j + lags_all
    a_i = np.zeros(2 * n - 1)
    a_j = np.zeros(2 * n - 1)
    vl_i = (il_i >= 0) & (il_i < n)
    vp_i = (ip_i >= 0) & (ip_i < n)
    vl_j = (il_j >= 0) & (il_j < n)
    vp_j = (ip_j >= 0) & (ip_j < n)
    a_i[vl_i] += (1.0 - h[il_i[vl_i]])
    a_i[vp_i] -= h[ip_i[vp_i]]
    a_j[vl_j] += (1.0 - h[il_j[vl_j]])
    a_j[vp_j] -= h[ip_j[vp_j]]
    d = dx * (a_i - a_j)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return h, 0.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(values + d * m1) <= np.max(values + d * m2):
            hi = m2
        else:
            lo = m1
    delta = (lo + hi) / 2.0
    return delta


def subgradient_phase(h, n, dx, t_start, duration):
    t_end = t_start + duration
    lr = 0.08
    decay = 0.99997
    values = compute_values(h, n)
    best_c5 = np.max(values)
    best_h = h.copy()
    stall = 0
    while time.time() < t_end:
        grad = compute_subgrad(h, n, dx, values)
        h_new = h - lr * grad
        h_new = repair(h_new, n)
        values_new = compute_values(h_new, n)
        c5_new = np.max(values_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                lr *= 0.95
                stall = 0
        h = h_new
        values = values_new
        lr *= decay
    return best_h, best_c5


def pair_phase(h, n, dx, t_start, duration, rng, best_c5, best_h):
    t_end = t_start + duration
    values = compute_values(h, n)
    n_iter = 0
    while time.time() < t_end:
        if n_iter % 5000 == 0:
            grad = compute_subgrad(h, n, dx, values)
        i = rng.randint(n)
        if n_iter % 5000 < 2500:
            j = np.argmax(-np.sign(grad[i]) * grad)
        else:
            j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        delta = pair_move(h, i, j, n, dx, values, rng)
        if abs(delta) > 1e-10:
            h[i] += delta
            h[j] -= delta
            h = np.clip(h, 0.0, 1.0)
            h = repair(h, n)
            values = compute_values(h, n)
            c5 = np.max(values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        n_iter += 1
    return best_h, best_c5, h, values
