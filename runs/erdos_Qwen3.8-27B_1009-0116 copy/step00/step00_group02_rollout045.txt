def sum_diff(x, target):
    return np.sum(x) - target

    return np.ones(len(x))

        else:
            h += diff / len(h)

    return h

    return project_feasible(h, n / 2.0)


def slsqp_optimize(h_start, n, dx, target_sum, max_iter=200, ftol=1e-10):
