import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def resample(h_old, n_new):
    """Resample h from its current resolution to n_new points."""
    n_old = len(h_old)
    # h represents piecewise constant over [0,2]
    # Each of n_old intervals has width 2/n_old
    # We want to interpolate to n_new intervals of width 2/n_new
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old  # centers
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def enforce_sum(h, n, target=None):
    """Ensure sum(h) = n/2 and 0 <= h <= 1."""
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Adjust all values proportionally
        if diff > 0:
            # Can only increase where h < 1
            mask = h < 1.0
            if not np.any(mask):
                break
            cap = 1.0 - h[mask]
            scale = min(1.0, diff / np.sum(cap))
            h[mask] += cap * scale
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            cap = h[mask]
            scale = min(1.0, -diff / np.sum(cap))
            h[mask] -= cap * scale
    return h

    best_c5 = compute_c5(h)
    dx = 2.0 / n
    target = n / 2.0

        idx = np.random.permutation(n)
        improved = False
        
        for i in idx:
            old_val = h[i]
            
            # Try several candidate values
            candidates = [old_val]
            # Temperature-based step
            temp = max(0.1, 1.0 - iteration / max_iters)
            step = 0.5 * temp
            for delta in [-step, step, -0.1, 0.1, -0.5, 0.5]:

                if 0.0 <= new_val <= 1.0:

            candidates.append(0.0)
            candidates.append(1.0)
            
            best_candidate = old_val
            best_score = best_c5

                h[i] = cv
                # Enforce sum constraint

                    best_candidate = cv
                    h[:] = h_adj

                best_c5 = best_score
        
        if not improved and iteration > 20:
