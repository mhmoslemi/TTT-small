    # Phase 2: Smoothed SLSQP (decreasing eps)
    for eps in [0.05, 0.01, 0.002]:
        if elapsed() > budget_s * 0.35:
            break
        res = snoop(best_h, 2000, 1e-10, eps=eps)
        c5_try = compute_c5(fix_sum(res.x, n0))
        if c5_try < best_c5:
            best_h = fix_sum(res.x, n0)
            best_c5 = c5_try

    # Phase 3: Multi-resolution refinement
    for factor in [2, 3]:
        if elapsed() > budget_s * 0.70:
            break
        n_hi = n0 * factor
        h_hi = resample(best_h, n_hi)
        res = snoop(h_hi, 2000, 1e-10)
        h_hi_opt = fix_sum(res.x, n_hi)
        c5_hi = compute_c5(h_hi_opt)
        # Try high-res directly
        if c5_hi < best_c5:
            best_h = h_hi_opt
            best_c5 = c5_hi
        # Try downsampled
        h_lo = resample(h_hi_opt, n0)
        c5_lo = compute_c5(h_lo)
        if c5_lo < best_c5:
            best_h = h_lo
            best_c5 = c5_lo
        # Second pass at high res
        if elapsed() < budget_s * 0.80:
            res2 = snoop(h_hi_opt, 1500, 1e-11)
            h_hi2 = fix_sum(res2.x, n_hi)
            c5_hi2 = compute_c5(h_hi2)
            if c5_hi2 < best_c5:
                best_h = h_hi2
                best_c5 = c5_hi2
            h_lo2 = resample(h_hi2, n0)
            if compute_c5(fix_sum(h_lo2, n0)) < best_c5:
                best_c5 = compute_c5(fix_sum(h_lo2, n0))
                best_h = fix_sum(h_lo2, n0)

    # Phase 4: Random restarts + local search
    n = len(best_h)
    while elapsed() < budget_s - 15:
        sigma = rng.choice([0.01, 0.02, 0.04, 0.06, 0.10])
        h_try = best_h + rng.normal(0, sigma, n)
        h_try = fix_sum(h_try, n)
        res = snoop(h_try, 800, 1e-10)
        h_try = fix_sum(res.x, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            # Local coordinate search
            best_h, best_c5 = local_search(best_h, rng, 100, 0.015)
            best_c5 = compute_c5(best_h)

    # Final polish
    if elapsed() < budget_s - 3:
        res = snoop(best_h, 3000, 1e-13)
        h_final = fix_sum(res.x, n)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, float(best_c5), len(best_h))
