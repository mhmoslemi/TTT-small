import numpy as np

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h += (target - h.sum()) / n
    return np.clip(h, 0.0, 1.0)

def ov(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5(h):
    return ov(h, 2.0 / len(h)).max()

def grad(h, dx, T):
    n = len(h)
    o = ov(h, dx)
    e = np.exp(T * (o - o.max()))
    p = e / e.sum()
    c = np.correlate(h, p, mode='full')
    A = c[2 * n - 2:3 * n - 2][::-1]
    c2 = np.correlate(h, p[::-1], mode='full')
    B = c2[2 * n - 2:3 * n - 2][::-1]
    return o.max(), dx * (1.0 - A - B)
