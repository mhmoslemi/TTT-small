import numpy as np
import time
import sys

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.copy()
    lo = -float(h.max()) - 0.5
    hi = 1.0 - float(h.min()) + 0.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h + mid, 0.0, 1.0).sum())
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, h, mode='full') * dx
    neg = -alpha * r
    mx = float(neg.max())
    ev = np.exp(neg - mx)
    w = ev / float(ev.sum())
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-12:
            continue
        k = idx - (n - 1)
        ak = abs(k)
        if ak == 0:
            grad += 2.0 * wk * h
        elif ak < n:
            grad[:n - ak] += wk * h[ak:]
            grad[ak:] += wk * h[:n - ak]
    grad *= -dx
    return grad, w

def optimize(h_init, budget_s, start_time, max_sets=100):
    n = len(h_init)
    target = n * 0.5
    h = project_feasible(h_init.astype(float), target)
    best = compute_c5(h)
    best_h = h.copy()
    alpha = 4.0
    lr = 0.008
    stagnant = 0
    for it in range(200000):
        if time.time() - start_time > budget_s - 3:
            break
        grad, _ = smooth_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best - 1e-9:

            best_h = h_new.copy()
            h = h_new
