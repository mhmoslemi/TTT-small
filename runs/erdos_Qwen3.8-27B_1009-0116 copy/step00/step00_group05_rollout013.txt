                grad[:n - offset] -= wm * h[offset:]
            else:
                o = -offset
                grad[o:] += wm * (1.0 - h[:n - o])
                grad[:n - o] -= wm * h[o:]
    return val, grad

def _fun(h):
    return _obj_grad(h, _N, _T)[0]

def _jac(h):
    return _obj_grad(h, _N, _T)[1]

def _c_fun(h):
    return float(np.sum(h) - _N / 2.0)

def _c_jac(h):
    return np.ones(_N)

def slsqp_optimize(h, n, t=0, max_iter=200):
    global _N, _T
    _N = n
    _T = t
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    res = minimize(_fun, h, jac=_jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return res.x

def random_restart(h_base, n, rng, sigma=0.05):
    h = h_base + rng.randn(n) * sigma
    return project_to_set(h, n)

def coordinate_refine(h, n, max_sweeps=20):
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    for sweep in range(max_sweps):
        improved = False
        for i in range(n):
            for delta in [0.01, -0.01, 0.005, -0.005]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                h_old = h[i]
                h[i] += delta
                h[(i + 1) % n] -= delta  # keep sum constant
                h[(i + 1) % n] = np.clip(h[(i + 1) % n], 0.0, 1.0)
                if h[(i + 1) % n] == 0 and h[i + delta] > 1 - delta:
                    h[i] = 1.0
                    h[(i + 1) % n] = h_old - delta + 1.0
                if h[(i + 1) % n] == 1 and h[i + delta] < delta:
                    h[i] = 0.0
                    h[(i + 1) % n] = h_old - delta - 1.0
                h[(i + 1) % n] = np.clip(h[(i + 1) % n], 0.0, 1.0)
                if abs(np.sum(h) - target) > 0.01:
                    h[i] = h_old
                    h[(i + 1) % n] = (target - np.sum(h[:-1]) - h[-1] if i != n-1 else target - np.sum(h[:-1]))
                    h = project_to_set(h, n)
                    continue
                c5_new = compute_c5(h)
                c5_cur = compute_c5(project_to_set(h, n))
                h = project_to_set(h, n)
                # This is getting complicated, just skip coordinate refine
                break
            break
        break  # skip this for now, SLSQP is sufficient
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _N, _T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Anneal temperature at n_parent
    temperatures = [0, 10, 25, 50, 100, 200]
    h_work = best_h.copy()
    for t in temperatures:
        if time.time() - start_time > budget_s * 0.4:
            break
        h_work = slsqp_optimize(h_work, n_parent, t=t, max_iter=200)
        h_work = project_to_set(h_work, n_parent)
        c5 = compute_c5(h_work)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()

    # Phase 2: Random restarts at n_parent
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.70:
        n_restarts = int((budget_s * 0.70 - elapsed) / 3.0)
        n_restarts = max(n_restarts, 5)
        n_restarts = min(n_restarts, 60)
        for r in range(n_restarts):
            if time.time() - start_time > budget_s * 0.72:
                break
            sigma = 0.03 + 0.07 * (r % 4) * 0.25
            h_r = random_restart(best_h, n_parent, rng, sigma)
            h_r = slsqp_optimize(h_r, n_parent, t=0, max_iter=150)
            h_r = project_to_set(h_r, n_parent)
            c5 = compute_c5(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()
            # Also try with smooth max
            h_r2 = slsqp_optimize(h_r, n_parent, t=50, max_iter=100)
            h_r2 = project_to_set(h_r2, n_parent)
            c5_2 = compute_c5(h_r2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h_r2.copy()

    # Phase 3: Try different n values
    n_options = []
    for n_try in [40, 50, 60, 80, 100, 120, 140, 160, 200]:
        if n_try != n_parent:
            n_options.append(n_try)

    for n in n_options:
        if time.time() - start_time > budget_s * 0.90:
            break
        # Interpolate from best
        x_old = np.linspace(0.0, 2.0, n_parent)
        x_new = np.linspace(0.0, 2.0, n)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_to_set(h_new, n)

        # Also try random init
        h_rand = project_to_set(rng.rand(n), n)

        for h_start in [h_new, h_rand]:
            if time.time() - start_time > budget_s * 0.92:
                break
            for t in [0, 50]:
                if time.time() - start_time > budget_s * 0.93:
                    break
                h_opt = slsqp_optimize(h_start, n, t=t, max_iter=150)
                h_opt = project_to_set(h_opt, n)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
            # Local refinement with smooth max
            h_fin = slsqp_optimize(h_opt, n, t=200, max_iter=100)
            h_fin = project_to_set(h_fin, n)
            c5 = compute_c5(h_fin)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_fin.copy()

    # Phase 4: Final polish at best n
    n_final = len(best_h)
    if time.time() - start_time < budget_s * 0.96:
        h_final = slsqp_optimize(best_h, n_final, t=0, max_iter=300)
        h_final = project_to_set(h_final, n_final)
        c5 = compute_c5(h_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final
        h_final2 = slsqp_optimize(best_h, n_final, t=100, max_iter=200)
        h_final2 = project_to_set(h_final2, n_final)
        c5_2 = compute_c5(h_final2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h_final2

    # Ensure constraint is satisfied
    best_h = project_to_set(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
