import numpy as np
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = np.round(h).astype(int)
    n1 = h.sum()
    target = n // 2
    if n1 > target:
        # Remove excess ones
        ones = np.where(h == 1)[0]
        excess = n1 - target
        idx = np.random.choice(ones, excess, replace=False)
        h[idx] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        need = target - n1
        idx = np.random.choice(zeros, need, replace=False)
        h[idx] = 1
    return h.astype(float)

def local_search_2flip(h, dx, max_rounds=20, seed=42):
    """Exhaustive 2-flip local search."""
    rng = np.random.default_rng(seed)
    h = h.copy()
    c5 = eval_c5(h, dx)
    n = len(h)
    
    for _ in range(max_rounds):
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        # Shuffle for variety
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0, 1
                new_c5 = eval_c5(h, dx)
                if new_c5 < c5:
                    c5 = new_c5
                    improved = True
                else:
                    h[i], h[j] = 1, 0
        if not improved:
            break
    return h, c5

def simulated_annealing(h, dx, seed=42, T0=0.005, steps=20000):
    """SA with random 2-flips."""
    rng = np.random.default_rng(seed)
    h = h.copy()
    c5 = eval_c5(h, dx)
    best = h.copy()
    best_c5 = c5
    n = len(h)
    
    T = T0
    for _ in range(steps):
        T *= 0.9995
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i], h[j] = 0, 1
        new_c5 = eval_c5(h, dx)
        delta = new_c5 - c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c5 = new_c5
            if c5 < best_c5:
                best = h.copy()
                best_c5 = c5
        else:
            h[i], h[j] = 1, 0

def continuous_optimize(h0, dx, max_iter=5000, lr=0.01, seed=42):
    """Gradient descent with subgradient on max of correlations."""
    rng = np.random.default_rng(seed)
    h = h0.astype(float).copy()
    n = len(h)
    target_sum = n / 2
    
    for it in range(max_iter):
        # Compute all correlations
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_best = np.argmax(corr)
        
        # Gradient w.r.t. h for the best correlation
        # corr[k] = sum_i h[i] * (1 - h[i + k - (n-1)]) for valid indices
        # d(corr[k])/dh[i] = (1 - h[i + k - (n-1)]) + h[i] * (-1) * [i+k-(n-1) == i ? no...]
        # Actually: f = sum_i h[i]*(1-h[i+shift]) where shift = k_best - (n-1)
        # For the raw correlate: corr_raw[k] = sum_i h[i] * g[i + k - (n-1)]
        # where g = 1-h
        # d corr_raw[k] / d h[i] = g[i + k - (n-1)] (from the h[i] term)
        #   + sum_j (-1) * h[j] * [j + k - (n-1) == i]  (from the g term)
        # = g[i + shift] - h[i - shift]  (where shift = k - (n-1))
        #   (terms that go out of bounds are 0)
        
        shift = k_best - (n - 1)
        grad = np.zeros(n)
        # Term 1: g[i + shift] = 1 - h[i + shift] for valid i+shift
        if shift >= 0:
            # i + shift valid for i in [0, n-1-shift]
            grad[:n-shift] += (1 - h[shift:n])
        else:
            # i + shift valid for i in [-shift, n-1]
            grad[-shift:] += (1 - h[:n+shift])
        
        # Term 2: -h[i - shift] for valid i-shift
        if shift >= 0:
            # i - shift valid for i in [shift, n-1]
            grad[shift:] -= h[:n-shift]
        else:
            # i - shift valid for i in [0, n+shift-1]
            grad[:n+shift] -= h[-shift:]
        
        # Add Lagrange multiplier for sum constraint
        # Use a simple penalty
        grad += 0.1 * (h.sum() - target_sum)  # push toward target
        # Project to [0,1] gradient
        grad[h >= 1] = 0
        grad[h <= 0] = 0
        
        # Step
        h -= lr * grad
        h = np.clip(h, 0, 1)
        # Fix sum
        diff = target_sum - h.sum()
        h += diff / n
    
    return h

                h[j] = 0
            h[i] = 1
        if best_iv is not None:
            h[best_iv[0]] = 0
            h[best_iv[1]] = 1
            c5 += best_d
        else:
            break
    return h, c5

def sa_search(h, dx, seed=42, steps=40000, T0=0.005):
    """Simulated annealing with 2-flip, 4-flip, and reversal moves."""
    rng = np.random.default_rng(seed)
    h = h.copy()
    n = len(h)
    c5 = eval_c5(h, dx)
    best = h.copy()
    best_c5 = c5
    for s in range(steps):
        T = T0 * (0.005) ** (s / steps)
        r = rng.random()
        if r < 0.55:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0; h[j] = 1
            nc = eval_c5(h, dx)
            d = nc - c5
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
                c5 = nc
                if c5 < best_c5:
                    best = h.copy(); best_c5 = c5
            else:
                h[i] = 1; h[j] = 0
        elif r < 0.75:
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            i1, i2 = rng.choice(ones, 2, replace=False)
            j1, j2 = rng.choice(zeros, 2, replace=False)
            h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
            nc = eval_c5(h, dx)
            d = nc - c5
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
                c5 = nc
                if c5 < best_c5:
                    best = h.copy(); best_c5 = c5
            else:
                h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
        else:
            a = rng.integers(0, n - 1)
            b = rng.integers(a + 1, n)
            h[a:b+1] = h[a:b+1][::-1]
            nc = eval_c5(h, dx)
            d = nc - c5
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
                c5 = nc
                if c5 < best_c5:
                    best = h.copy(); best_c5 = c5
            else:
                h[a:b+1] = h[a:b+1][::-1]
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 70

    try:
        init = np.asarray(initial_h_values, dtype=float)
    except Exception:
        init = None

    # Candidate sizes (even, reasonable range)
    ns = [70]
    if init is not None:
        ns.insert(0, len(init))
    for nn in [62, 64, 66, 68, 72, 74, 76, 78, 80, 84]:
        if nn % 2 == 0 and nn not in ns:
            ns.append(nn)
    ns = sorted(set(ns))

    def remaining():
        return budget_s - (time.time() - t0)

    for n in ns:
        if remaining() < 8:
            break
        dx = 2.0 / n
        rng = np.random.default_rng(seed + n * 7)

        # Build starting points
        starts = []
        if init is not None and len(init) == n:
            starts.append(init.copy())
        for _ in range(2):
            hv = np.zeros(n)
            hv[rng.choice(n, n // 2, replace=False)] = 1.0
            starts.append(hv)
        # Block pattern 1100
        if n % 4 == 0:
            hv = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
            starts.append(hv)
        # Alternating 1010
        hv = np.zeros(n)
        hv[::2] = 1.0
        if int(hv.sum()) == n // 2:
            starts.append(hv)

        for si, h0 in enumerate(starts):
            if remaining() < 6:
                break
            tag = f"n={n} s={si}"

            # Phase 1: local search
            h_ls, c5_ls = ls_2flip(h0, dx, max_rounds=150, seed=seed + n * 13 + si)
            if c5_ls < best_c5:
                best_c5 = c5_ls; best_h = h_ls.copy(); best_n = n
                print(f"  LS {tag}: {c5_ls:.6f}")

            # Phase 2: SA from local optimum
            if remaining() > 4:
                h_sa, c5_sa = sa_search(h_ls, dx, seed=seed + 5000 + n * 3 + si,
                                        steps=30000, T0=0.005)
                if c5_sa < best_c5:
                    best_c5 = c5_sa; best_h = h_sa.copy(); best_n = n
                    print(f"  SA {tag}: {c5_sa:.6f}")

                # Phase 3: final LS on SA result
                h_f, c5_f = ls_2flip(h_sa, dx, max_rounds=120, seed=seed + 9000 + n)
                if c5_f < best_c5:
                    best_c5 = c5_f; best_h = h_f.copy(); best_n = n
                    print(f"  LS2 {tag}: {c5_f:.6f}")

    # Long SA on the current best
    if best_h is not None and remaining() > 15:
        dx = 2.0 / best_n
        h_long, c5_long = sa_search(best_h, dx, seed=seed + 77777,
                                    steps=min(80000, int(remaining() * 8000)), T0=0.006)
        c5_long = eval_c5(h_long, dx)
        if c5_long < best_c5:
            best_c5 = c5_long; best_h = h_long.copy()
            print(f"  LongSA: {c5_long:.6f}")
        # Final polish
        h_f, c5_f = ls_2flip(best_h, dx, max_rounds=200, seed=seed + 88888)
        if c5_f < best_c5:
            best_c5 = c5_f; best_h = h_f.copy()
            print(f"  FinalLS: {c5_f:.6f}")

    # Verify
    dx = 2.0 / best_n
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure exact sum
    s = best_h.sum()
    target = best_n // 2
    if abs(s - target) > 1e-9:
        idx = np.argsort(-best_h)
        best_h = np.zeros(best_n)
        best_h[idx[:target]] = 1.0
    best_c5 = eval_c5(best_h, dx)

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev < best_c5:
            best_c5 = ev
    except Exception:
        pass

    print(f"FINAL: n={best_n}, C5={best_c5:.6f}")
    return best_h, best_c5, best_n
