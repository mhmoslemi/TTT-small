            bh, bv = h, v
    return bh, bv

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rs = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(47, 0.5)
    n0 = len(h0)

    h0 = _proj(h0, n0)
    best_h, best_c5 = h0.copy(), _c5(h0)

    # Phase 1: Optimize base resolution
    tl = t0 + budget_s * 0.25
    h, cv = _optimize(h0, n0, tl)
    if cv < best_c5:
        best_h, best_c5 = h, cv

    # Coordinate descent polish
    h2, cv2 = _coord_desc(best_h, n0, sweeps=3)
    if cv2 < best_c5:
        best_h, best_c5 = h2, cv2

    # Phase 2: Random restarts
    n_restarts = 20
    for i in range(n_restarts):
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        scale = 0.12 * (1 + 0.3 * np.sin(i))
        hp = _proj(best_h + rs.uniform(-scale, scale, n0), n0)
        tl2 = time.time() + min(30, rem * 0.2)
        h, cv = _optimize(hp, n0, tl2)
        if cv < best_c5:
            best_h, best_c5 = h, cv
            h2, cv2 = _coord_desc(best_h, n0, sweeps=2)
            if cv2 < best_c5:
                best_h, best_c5 = h2, cv2

    # Phase 3: Upscale
    for mult in [2, 3]:
        rem = budget_s - (time.time() - t0)
        if rem < 100:
            break
        nn = n0 * mult
        hi = _proj(_interp(best_h, nn), nn)
        tl3 = time.time() + min(120, rem * 0.45)
        h, cv = _optimize(hi, nn, tl3)
        if cv < best_c5:
            best_h, best_c5 = h, cv

    # Final coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h2, cv2 = _coord_desc(best_h, len(best_h), sweeps=4)
        if cv2 < best_c5:
            best_h, best_c5 = h2, cv2

    best_c5 = _c5(best_h)
    return best_h, best_c5, len(best_h)
