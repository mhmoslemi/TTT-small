import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            idx = np.argmin(h) if diff2 > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def smooth_objective(h, T):
    """T*log(sum exp(corr_k*dx/T)) - smooth max."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    max_c = np.max(corr)
    log_sum = max_c + T * np.log(np.sum(np.exp((corr - max_c) / T)))
    return log_sum

def smooth_gradient(h, T):
    """Gradient of smooth objective."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    # Softmax weights
    max_c = np.max(corr)
    weights = np.exp((corr - max_c) / T)
    weights /= np.sum(weights)
    # grad = dx * sum_k weights[k] * dcorr_k/dh
    # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # s_k = k - (n-1)
    # dcorr_k/dh[j] = (1-h[j+s_k])*[0<=j+s_k<n] + (-h[j-s_k])*[0<=j-s_k<n] (for s_k != 0)
    #                = (1-2*h[j]) (for s_k == 0)
    grads = np.zeros(n)
    for k in range(len(corr)):
        s = k - (n - 1)
        w = weights[k] * dx
        if s == 0:
            grads += w * (1.0 - 2.0 * h)
        else:
            # First term: (1-h[j+s]) for j where 0<=j+s<n → j in [max(0,-s), min(n-1, n-1-s)]
            # Second term: -h[j-s] for j where 0<=j-s<n → j in [max(0,s), min(n-1, n-1+s)]
            j_start1 = max(0, -s)
            j_end1 = min(n - 1, n - 1 - s)
            if j_start1 <= j_end1:
                idx1 = np.arange(j_start1, j_end1 + 1)
                grads[idx1] += w * (1.0 - h[idx1 + s])
            j_start2 = max(0, s)
            j_end2 = min(n - 1, n - 1 + s)
            if j_start2 <= j_end2:
                idx2 = np.arange(j_start2, j_end2 + 1)
                grads[idx2] += w * (-h[idx2 - s])
    return grads

def optimize_smooth(h, T_start=0.1, T_end=0.01, n_iter=5000):
    """Optimize with decreasing temperature."""
    n = len(h)
    best_h = h.copy()
    best_f = compute_c5(h)
    lr_start = 0.1
    lr_end = 0.001
    
    for i in range(n_iter):
        frac = i / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        f = smooth_objective(h, T)
        g = smooth_gradient(h, T)
        
        # Gradient step
        h_new = h - lr * g
        h_new = project(h_new)
        
        # Accept if it improves the true objective
        true_f = compute_c5(h_new)
        if true_f < best_f:
            best_h = h_new.copy()
            best_f = true_f
        h = h_new
    
    return best_h, best_f

def coordinate_polish(h, max_iter=500):
    """Polish with coordinate descent."""
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        improved = False
        for j in range(n):
            # Try to adjust h[j] to improve
            # Current contribution: h[j]*(1-h[j]) at lag 0, and cross terms at other lags
            # Simple approach: try a few values
            for delta in [0.05, -0.05, 0.02, -0.02, 0.1, -0.1]:
                h_new = h.copy()
                h_new[j] = np.clip(h_new[j] + delta, 0.0, 1.0)
                # Adjust to maintain sum
                diff = target - np.sum(h_new)
                if abs(diff) < 0.01:
                    h_new = h_new + diff / n
                    h_new = np.clip(h_new, 0.0, 1.0)
                else:
                    continue
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-9:
                    h = h_new
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    # Get initial

    # Phase 2: Coordinate polish

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            idx = np.argmin(h) if diff2 > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    w = np.exp((corr - mc) / T)
    w /= np.sum(w)
    grads = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-12:
            continue
        if s == 0:
            grads += wk * (1.0 - 2.0 * h)
        else:
            j1 = max(0, -s)
            j2 = min(n - 1, n - 1 - s)
            if j1 <= j2:
                idx1 = np.arange(j1, j2 + 1)
                grads[idx1] += wk * (1.0 - h[idx1 + s])
            j3 = max(0, s)
            j4 = min(n - 1, n - 1 + s)
            if j3 <= j4:
                idx2 = np.arange(j3, j4 + 1)
                grads[idx2] += wk * (-h[idx2 - s])
    return grads

def optimize_smooth(h, T0=0.06, T1=0.003, n_iter=8000, lr0=0.15, lr1=0.002):
    n = len(h)
    best_h = h.copy()
    best_f = compute_c5(h)
    for i in range(n_iter):
        fr = i / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** fr
        lr = lr0 * (lr1 / lr0) ** fr
        g = smooth_grad(h, T)
        h = project(h - lr * g)
        f = compute_c5(h)
        if f < best_f:
            best_h = h.copy()
            best_f = f
    return best_h, best_f

def coord_polish(h, max_iter=400, time_limit=None, t_start=None):
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    ts = t_start if t_start is not None else time.time()
    steps = [0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
    for it in range(max_iter):
        if time_limit is not None and time.time() - ts > time_limit:
            break
        improved = False
        rng_local = np.random.default_rng(1000 + it)
        order = rng_local.permutation(n)
        for j in order:
            for d in steps:
                h_new = h.copy()
                h_new[j] = np.clip(h_new[j] + d, 0.0, 1.0)
                diff = target - np.sum(h_new)
                if abs(diff) > 0.5:
                    continue
                h_new = project(h_new)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-9:
                    h = h_new
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def random_init(n, rng):
    h = rng.uniform(0.15, 0.85, size=n)
    return project(h)

def structured_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.integers(3, max(4, n // 8))
    block_sizes = rng.multinomial(n, np.ones(n_blocks) / n_blocks)
    start = 0
    for bs in block_sizes:
        val = rng.uniform(0.1, 0.9)
        h[start:start + bs] = val
        start += bs
    return project(h)
