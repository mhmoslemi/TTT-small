import numpy as np
import time

def _corr_max(h):
    """Compute max of full correlation with complement, times dx."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * corr.max()

def _init_binary(h, n, rng):
    """Convert to binary with sum = n/2."""
    target = n // 2
    h_b = (h > 0.5).astype(np.int8)
    s = h_b.sum()
    if s > target:
        idx = np.where(h_b == 1)[0]
        sel = rng.choice(idx, size=s-target, replace=False)
        h_b[sel] = 0
    elif s < target:
        idx = np.where(h_b == 0)[0]
        sel = rng.choice(idx, size=target-s, replace=False)
        h_b[sel] = 1
    return h_b

def _sa_optimize(h, rng, t_start, budget_s, patience=20000):
    """Simulated annealing on binary sequence."""
    n = len(h)
    best = h.copy()
    best_c5 = _corr_max(best.astype(float))
    T = 0.04
    cool = 0.99998
    min_T = 0.003
    no_improve = 0
    
    while time.time() - t_start < budget_s:
        if T < min_T:
            T = min_T
        # Single swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0, 1
        c5 = _corr_max(h.astype(float))
        d = c5 - best_c5
        if d < 0 or rng.random() < np.exp(-d / T):
            if c5 < best_c5:
                best = h.copy()
                best_c5 = c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = 1, 0
            no_improve += 1
        T *= cool
        if no_improve > patience:
            h = best.copy()
            T = min(0.03, T * 1.5)
            no_improve = 0
    
    return best, best_c5

    """Try k random swaps and keep the best."""
    n = len(h)
    best_local = h.copy()
    best_local_c5 = _c5(h.astype(np.float64))
    for _ in range(k):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        h[i] = 0; h[j] = 1
        c5 = _c5(h.astype(np.float64))
        if c5 < best_local_c5:
            best_local_c5 = c5
        else:
            h[i] = 1; h[j] = 0
    return best_local, best_local_c5

def _continuous_refine(h_bin, n, deadline):
    """Try to improve with continuous values."""
    h = h_bin.astype(np.float64)
    best_c5 = _c5(h)
    best = h.copy()
    dx = 2.0 / n
    for _ in range(200):
        if time.time() > deadline:
            break
        # Find the lag with max correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        # Try to reduce by adjusting values near the peak
        # Nudge a random value slightly
        idx = int(np.random.default_rng().integers(n))
        old = h[idx]
        h[idx] = np.clip(old + 0.01 if old < 0.5 else old - 0.01, 0, 1)
        # Rebalance sum
        diff = h.sum() - n / 2
        if abs(diff) > 1e-9:
            # Distribute correction
            others = np.where(np.arange(n) != idx)[0]
            oidx = others[np.random.default_rng().integers(len(others))]
            h[oidx] = np.clip(h[oidx] - diff, 0, 1)
            h = np.clip(h, 0, 1)
            # Final fix
            h_sum = h.sum()
            h[idx] = np.clip(h[idx] + (n/2 - h.sum()), 0, 1)
        c5 = _c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        else:
            h = best.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.95

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Optimize from initial_h_values (n=80)
    h0 = None
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        pass

    if h0 is not None and len(h0) > 0:
        n = len(h0)
        hb = _to_binary(h0, n, rng)
        c5 = _c5(hb)
        best_h = hb.astype(np.float64)
        best_c5 = c5
        # SA for 40% of budget
        d1 = t0 + budget_s * 0.40
        hb_opt, c5_opt = _sa(hb.copy(), rng, d1)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = hb_opt.astype(np.float64)

    # Phase 2: Try structured constructions for various n
    n_candidates = [64, 80, 96, 112, 128, 160, 192, 256]
    for n_try in n_candidates:
        if time.time() > deadline:
            break
        # Golden ratio spread
        alpha = (1.0 + math.sqrt(5.0)) / 2.0
        hc = np.zeros(n_try, dtype=np.int8)
        positions = set()
        for k in range(n_try // 2):
            pos = int((k + 1) * alpha * n_try) % n_try
            positions.add(pos)
        hc[list(positions)] = 1
        hc = _to_binary(hc, n_try, rng)
        c5c = _c5(hc)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.astype(np.float64)
        # Short SA from this
        if time.time() + budget_s * 0.08 < deadline:
            d = min(time.time() + budget_s * 0.08, deadline)
            hc_opt, c5_opt = _sa(hc.copy(), rng, d)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = hc_opt.astype(np.float64)

    # Phase 3: Random restarts for promising n
    n_best = len(best_h) if best_h is not None else 80
    for _ in range(5):
        if time.time() > deadline * 0.85:
            break
        for n_try in [n_best, min(n_best + 32, 512), max(n_best - 16, 48)]:
            if time.time() > deadline * 0.85:
                break
            hb_r = _to_binary(None, n_try, rng)
            c5_r = _c5(hb_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = hb_r.astype(np.float64)
            if time.time() + budget_s * 0.06 < deadline:
                d = min(time.time() + budget_s * 0.06, deadline)
                hb_opt, c5_opt = _sa(hb_r, rng, d)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = hb_opt.astype(np.float64)

    # Phase 4: Final intensive SA from best
    if best_h is not None and time.time() < deadline:
        n = len(best_h)
        hb = _to_binary(best_h, n, rng)
        hb_opt, c5_opt = _sa(hb, rng, deadline, max_patience=120000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = hb_opt.astype(np.float64)

    # Phase 5: Try continuous refinement if time
    if best_h is not None and time.time() < deadline * 0.98:
        n = len(best_h)
        cr, c5_cr = _continuous_refine(best_h, n, deadline)
        if c5_cr < best_c5:
            best_c5 = c5_cr
            best_h = cr

    # Final validation
    n = len(best_h)
    h_out = np.clip(best_h, 0.0, 1.0)
    integral = np.sum(h_out) * (2.0 / n)
    if abs(integral - 1.0) > 1e-6:
        h_out = _to_binary(h_out, n, rng).astype(np.float64)
        best_c5 = _c5(h_out)
    final_c5 = _c5(h_out)
    return h_out, final_c5, n
