    h = np.clip(h, 0, 1)
    if abs(h.sum() - target_sum) < 1e-12:
        return h
    # Find tau such that sum(clamp(h - tau, 0, 1)) = target_sum
    # If h.sum() > target_sum, tau > 0 (we subtract)
    # If h.sum() < target_sum, tau < 0 (we add)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine-tune to meet constraint exactly
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-10:

        mask = (h_proj > 1e-8) & (h_proj < 1 - 1e-8)
        if mask.any():
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0, 1)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, target_sum):

    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)

def projected_gradient_descent(h0, T_schedule, target_sum, max_iter=5000, tol=1e-10):

    n = len(h)
    best_h = h0.copy()
    best_obj = compute_c5(h0)
    lr = 0.01
    for it in range(max_iter):

        if T < 1e-6:

        h_new = project_to_feasible(h_new, target_sum)
        # Check improvement

        h = h_new
        if it > 100 and abs(actual_obj - best_obj) < tol:

            pass
    return best_h, best_obj

    best_h = h.copy()
    lr = 0.04
    for it in range(max_iter):
        if tcheck():
            break
        g, sobj = smax_grad(h, T)
        alpha = lr
        for _ in range(10):
            h_try = project_feasible(h - alpha * g, target_sum)
            c5 = compute_c5(h_try)
            if c5 < best:
                best = c5
                best_h = h_try.copy()
            if c5 <= sobj + 1e-9:
                h = h_try
                lr = alpha * 1.05
                break
            alpha *= 0.4
        else:
            h = project_feasible(h - alpha * g, target_sum)
            lr = alpha
    return best_h, best

def random_perturb(h, target_sum, strength, rng):
    h_p = h + strength * (rng.random(len(h)) - 0.5)
    return project_feasible(h_p, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h_parent = np.array(initial_h_values, dtype=float)
    n = len(h_parent)
    ts = float(n) / 2.0
    h = project_feasible(h_parent, ts)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def tcheck():
        return (time.time() - t0) > budget_s - 15

    # Phase 1: Multi-temperature gradient descent
    temps = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for T in temps:
        if tcheck():
            break
        h_cur = best_h.copy()
        n_iter = 2000 if T > 0.1 else 3000
        best_h, best_c5 = local_opt(h_cur, ts, T, n_iter, tcheck)

    # Phase 2: Stochastic restarts with local refinement
    n_restarts = 0
    while not tcheck():
        n_restarts += 1
        strength = 0.2 / (1.0 + 0.08 * n_restarts)
        h_try = random_perturb(best_h, ts, strength, rng)
        for T in [0.05, 0.01]:
            if tcheck():
                break
            h_try, c5_local = local_opt(h_try, ts, T, 800, tcheck)
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_try.copy()

    # Phase 3: Very fine local search
    if not tcheck():
        h_cur = best_h.copy()
        for _ in range(10000):
            if tcheck():
                break
            g, sobj = smax_grad(h_cur, 0.0005)
            alpha = 0.003
            for _ in range(5):
                h_n = project_feasible(h_cur - alpha * g, ts)
                c5_n = compute_c5(h_n)
                if c5_n < best_c5:
                    best_c5 = c5_n
                    best_h = h_n.copy()
                    h_cur = h_n
                    break
                alpha *= 0.5
            else:
                h_cur = project_feasible(h_cur - alpha * g, ts)

    # Phase 4: Try binary rounding + local repair
    if not tcheck():
        h_bin = (best_h > 0.5).astype(float)
        # Adjust to meet sum
        deficit = ts - h_bin.sum()
        if deficit > 0:
            idx = np.argsort(best_h)[::-1][:int(deficit)]
            h_bin[idx] = 1.0
        elif deficit < 0:
            idx = np.argsort(best_h)[:int(-deficit)]
            h_bin[idx] = 0.0
        h_bin = project_feasible(h_bin, ts)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Local refine from binary
        if not tcheck():
            h_r, c5_r = local_opt(h_bin, ts, 0.01, 3000, tcheck)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    return best_h, best_c5, n
