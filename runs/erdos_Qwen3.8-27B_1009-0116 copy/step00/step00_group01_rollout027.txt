import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def all_shift_c5(h, dx):
    g = 1.0 - h
    return np.correlate(h, g, mode='full') * dx

def project(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, n, sh):
    g = np.zeros(n)
    # term1: (1-h[j-sh]) for valid j
    a, b = max(0, sh), min(n, n + sh)
    if a < b:
        g[a:b] += (1.0 - h[a-sh:b-sh])
    # term2: -h[j+sh] for valid j  
    a, b = max(0, -sh), min(n, n - sh)
    if a < b:
        g[a:b] -= h[a+sh:b+sh]
    return g

def psolve(h0, n, time_limit=50.0):
    """Projected subgradient solver."""
    dx = 2.0 / n
    target = n / 2.0
    h = project(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    t0 = time.time()
    alpha = 0.08
    dec = 0.9998
    patience = 0
    pat_max = 500
    
    while time.time() - t0 < time_limit:
        shifts = all_shift_c5(h, dx)
        mx = shifts.max()
        # Softmax weighting over top shifts
        w = np.exp((shifts - mx) * 20.0)
        ws = w.sum()
        if ws < 1e-15:
            break
        w /= ws
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for i in range(2*n - 1):
            if w[i] > 1e-7:
                sh = i - (n - 1)
                grad += w[i] * grad_shift(h, n, sh)
        
        h = project(h - alpha * grad, target)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        alpha *= dec
        if patience > pat_max:
            alpha = max(alpha * 0.5, 0.001)
            patience = 0
            if alpha < 0.001:
                break
    
    return best_h, best_c5

def slsqp_polish(h0, n, time_limit=30.0):
    """SLSQP polish with smoothed objective."""
    dx = 2.0 / n
    target = n / 2.0
    T = 50.0  # smoothing parameter
    
    def obj(x):
        corr = all_shift_c5(x, dx)
        c = corr * T
        c -= c.max()
        return np.log(np.sum(np.exp(c))) / T
    
    def obj_grad(x):
        corr = all_shift_c5(x, dx)
        c = corr * T
        c -= c.max()
        e = np.exp(c)
        p = e / e.sum()
        # grad of smooth max wrt corr
        # grad of corr wrt h: same as grad of max at the "weighted" shifts
        g = np.zeros(n)
        for i in range(2*n - 1):
            if p[i] > 1e-10:
                sh = i - (n-1)
                g += p[i] * grad_shift(x, n, sh)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    t0 = time.time()
    res = minimize(obj, h0.copy(), jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 500, 'ftol': 1e-14})
    
    if time.time() - t0 > time_limit:
        return h0, compute_c5(h0, dx)
    
    h = project(res.x, target)
    c5 = compute_c5(h, dx)
    return h, c5

def upsample(h, n_old, n_new):
    """Upsample h to n_new points using linear interpolation."""

        else:

    bounds = [(0, 1)] * n
    t0 = time.time()
    res = minimize(obj, h0.copy(), jac=objg, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 400, 'ftol': 1e-15})
    if time.time() - t0 > tlim:
        return h0, compute_c5(h0, dx)
    h = project(res.x, tgt)
    return h, compute_c5(h, dx)

def binary_local_search(h0, n, tlim=20.0):
    dx = 2.0 / n
    tgt = n / 2.0
    h = project(h0, tgt)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    s = int(h_bin.sum())
    if s > tgt:
        idx = np.argsort(h_bin)[::-1]
        for i in range(s - int(tgt)):
            h_bin[idx[i]] = 0.0
    elif s < tgt:
        idx = np.argsort(h_bin)
        for i in range(int(tgt) - s):
            h_bin[idx[i]] = 1.0
    h_bin = project(h_bin, tgt)
    bh, bc5 = h_bin.copy(), compute_c5(h_bin, dx)
    t0 = time.time()
    while time.time() - t0 < tlim:
        improved = False
        ones = list(np.where(h_bin > 0.5)[0])
        zeros = list(np.where(h_bin < 0.5)[0])
        for i in ones:
            for j in zeros:
                h_bin[i], h_bin[j] = 0.0, 1.0
                c5 = compute_c5(h_bin, dx)
                if c5 < bc5 - 1e-10:
                    bc5, bh, improved = c5, h_bin.copy(), True
                else:
                    h_bin[i], h_bin[j] = 1.0, 0.0
                if time.time() - t0 > tlim:
                    return bh, bc5
        if not improved:
            break
    return bh, bc5

def upsample(h, no, nn):
    xo = np.arange(no) / no * 2.0 + 1.0 / no
    xn = np.arange(nn) / nn * 2.0 + 1.0 / nn
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    ts = time.time()

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(47, 0.5)
    n0 = len(h0)
    h0 = project(h0, n0 / 2.0)

    bh, bn = h0.copy(), n0
    bc5 = compute_c5(bh, 2.0 / bn)

    # Phase 1: PS at initial resolution
    rem = budget_s - (time.time() - ts) - 120
    if rem > 30:
        h1, c1 = psolve(h0, n0, min(rem, 90))
        if c1 < bc5:
            bh, bc5, bn = h1, c1, n0

    # Phase 2: SLSQP polish
    rem = budget_s - (time.time() - ts) - 60
    if rem > 20:
        h2, c2 = slsqp_polish(bh, bn, min(rem, 25))
        if c2 < bc5:
            bh, bc5 = h2, c2

    # Phase 3: Binary local search
    rem = budget_s - (time.time() - ts) - 50
    if rem > 10:
        h3, c3 = binary_local_search(bh, bn, min(rem, 20))
        if c3 < bc5:
            bh, bc5 = h3, c3

    # Phase 4: Upsample and optimize
    for mult in [2, 4]:
        nn = n0 * mult
        if nn >= 800:
            break
        rem = budget_s - (time.time() - ts) - 60
        if rem < 40:
            break
        hu = project(upsample(bh, bn, nn), nn / 2.0)
        h4, c4 = psolve(hu, nn, min(rem - 30, 80))
        if c4 < bc5:
            bh, bc5, bn = h4, c4, nn
        rem = budget_s - (time.time() - ts) - 40
        if rem > 20:
            h5, c5 = slsqp_polish(bh, bn, min(rem, 20))
            if c5 < bc5:
                bh, bc5 = h5, c5
        rem = budget_s - (time.time() - ts) - 30
        if rem > 10:
            h5b, c5b = binary_local_search(bh, bn, min(rem, 15))
            if c5b < bc5:
                bh, bc5 = h5b, c5b

    # Phase 5: Additional PS + SLSQP at best resolution
    rem = budget_s - (time.time() - ts) - 10
    if rem > 20:
        h6, c6 = psolve(bh, bn, min(rem - 10, 50))
        if c6 < bc5:
            bh, bc5 = h6, c6
    rem = budget_s - (time.time() - ts) - 5
    if rem > 15:
        h7, c7 = slsqp_polish(bh, bn, min(rem, 20))
        if c7 < bc5:
            bh, bc5 = h7, c7

    # Final: ensure validity
    bh = project(bh, bn / 2.0)
    fc5 = compute_c5(bh, 2.0 / bn)
    return bh, fc5, bn
