import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-12:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_val(h, n, dx, t):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = np.max(c)
    shifted = t * (c - cmax)
    shifted = np.clip(shifted, -400, 400)
    return cmax + np.log(np.sum(np.exp(shifted))) / t

def grad_soft_max(h, n, dx, t):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = np.max(c)
    shifted = t * (c - cmax)
    shifted = np.clip(shifted, -400, 400)
    neg_log_sum = -np.log(np.sum(np.exp(shifted)))
    log_w = shifted + neg_log_sum
    indices = np.arange(2 * n - 1)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        lw = log_w[m]
        if lw < -30:
            continue
        k = m - (n - 1)
        term1 = np.zeros(n)
        v1_s = max(0, k)
        v1_e = min(n, n + k)
        if v1_e > v1_s:
            term1[v1_s:v1_e] = 1.0 - h[v1_s - k:v1_e - k]
        term2 = np.zeros(n)
        v2_s = max(0, -k)
        v2_e = min(n, n - k)
        if v2_e > v2_s:
            term2[v2_s:v2_e] = -h[v2_s + k:v2_e + k]
        grad += (np.exp(lw)) * dx * (term1 + term2)
    return grad

def gd_soft_max(h, n, dx, t=80.0, max_iter=1500, lr_init=0.003, time_limit=None, t0=None):

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        grad = grad_soft_max(h, n, dx, t)
        h_new = h - lr * grad
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.95
        h = h_new
        if lr < 1e-9:
            break
    return best_h, best_c5

def binary_round(h, n):
    h = h.copy().astype(float)
    srt = np.argsort(-h)
    h_new = np.zeros(n)
    count_full = n // 2
    for i in range(count_full):
        h_new[srt[i]] = 1.0
    if n % 2 == 1:
        h_new[srt[count_full]] = 0.5
    return h_new

def local_search(h, n, dx, n_steps, rng, time_limit, t0):

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

            best_c5 = c
            best_h = h.copy()
        else:
            h[i] = old_i
            h[j] = old_j
            h = repair(h, n)
    return best_h, best_c5
