def fix_sum(h, target):
    h = h.copy()
    for _ in range(500):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        # Distribute correction proportionally
        room_up = 1.0 - h
        room_down = h
        if s > target:
            excess = s - target
            total_room = np.sum(room_down)
            if total_room < excess:
                h[:] = 0
                break
            h -= room_down * (excess / total_room)
        else:

            total_room = np.sum(room_up)
            if total_room < deficit:
                h[:] = 1
                break

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, target):
    h = h.astype(float).copy()
    for _ in range(5000):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h = h + room * (d / tr)
        else:
            room = h
            tr = room.sum()
            if tr < 1e-14:
                break
            h = h - room * (-d / tr)
