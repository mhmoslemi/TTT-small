import numpy as np
import time

def project_to_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection on offset."""
    target = n / 2.0
    # Find t such that sum(clip(h+t, 0, 1)) = target
    lo, hi = -max(h.max(), 0) - 1, 1 - min(h.min(), 0) + 1
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c, np.max(c)

def compute_gradient(h, k_star, n):
    """Gradient of c_{k_star} w.r.t. h."""
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        # c = sum_{i=0}^{n-1-shift} h[i]*(1-h[i+shift])
        # d/dh[k] = (1-h[k+shift]) if k+shift < n - ... 
        # For i in [0, n-1-shift]: contributes (1-h[i+shift]) to grad[i], -h[i] to grad[i+shift]
        m = n - shift
        grad[:m] += (1.0 - h[shift:])
        grad[shift:] += -h[:m]
    else:
        s = -shift
        m = n - s
        # c = sum_{i=s}^{n-1} h[i]*(1-h[i-s])
        # For i in [s, n-1]: contributes (1-h[i-s]) to grad[i], -h[i] to grad[i-s]
        grad[s:] += (1.0 - h[:m])
        grad[:m] += -h[s:]
    return grad

def optimize(h_init, n, budget_s=1100, seed=42):
    rng = np.random.RandomState(seed)
    h = project_to_feasible(h_init.astype(float).copy(), n)
    best_c5, best_h = np.max(np.correlate(h, 1-h, mode='full') * 2.0/n), h.copy()
    
    start = time.time()
    lr = 0.01
    momentum = np.zeros(n)
    mu = 0.9
    
    while time.time() - start < budget_s - 30:
        c, c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5, best_h = c5_val, h.copy()
        
        # Soft-max weights
        alpha = 50.0
        weights = np.exp(alpha * c - alpha * np.max(c))
        weights /= weights.sum()
        
        # Weighted gradient
        # Only compute gradient for shifts with significant weight
        k_star = np.argmax(c)
        g = weights[k_star] * compute_gradient(h, k_star, n)
        # Add a few more if needed for stability
        top_k = np.argsort(weights)[-5:]
        for k in top_k:
            if k != k_star and weights[k] > 0.01:
                g += weights[k] * compute_gradient(h, k, n)
        
        # Momentum update
        momentum = mu * momentum - lr * g
        h = project_to_feasible(h + momentum, n)
        
        # Occasionally try different learning rates
        if c5_val > best_c5 + 0.01:
            lr = lr * 0.5
    
    return best_h, best_c5

    if shift > 0:
        m = n - shift
        g[:m] += 1.0 - h[shift:]
        g[shift:] -= h[:m]
    else:
        s = -shift
        m = n - s
        g[s:] += 1.0 - h[:m]

    return g

    """Compute smoothed gradient using log-sum-exp type soft max."""

    # Only use shifts with significant weight
    threshold = 0.01 * w.max()
    for k in range(len(w)):

            shift = k - (n - 1)
            g += w[k] * grad_shift(h, shift, n)
    return g

import numpy as np
import time

def project_h(h, n):
    target = n / 2.0
    lo = -float(h.max()) - 0.2
    hi = 1.0 - float(h.min()) + 0.2
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.sum(np.clip(h + mid, 0.0, 1.0))) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def get_overlap(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def grad_shift(h, s, n):
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        m = n - s
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    else:
        s2 = -s
        m = n - s2
        g[s2:] += 1.0 - h[:m]
        g[:m] -= h[s2:]
    return g

def optimize(h0, n, ttl, seed):

            best = cur

            stall = 0
        else:

        frac = min(1.0, (time.time() - t0) / max(ttl, 1.0))
        alpha = 25.0 + 500.0 * frac

        g = np.zeros(n)

    return np.interp(xn, xo, h)

def try_binarize(h, n):
