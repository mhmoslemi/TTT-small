import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_constraint(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        m = h < 1.0
        if m.sum() > 0:
            h[m] = np.clip(h[m] + diff / m.sum(), 0.0, 1.0)
    else:
        m = h > 0.0
        if m.sum() > 0:
            h[m] = np.clip(h[m] + diff / m.sum(), 0.0, 1.0)
    diff2 = n / 2.0 - h.sum()
    if abs(diff2) > 1e-12:
        h += diff2 / n
        h = np.clip(h, 0.0, 1.0)
    return h

def elapsed(t0):
    return time.time() - t0

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:

    target = n / 2.0
    dx = 2.0 / n
    h0 = project_constraint(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Differential evolution (global search)
    if elapsed(t0) < budget_s * 0.30:
        bounds = [(0.0, 1.0)] * n
        def pen_obj(h):
            return compute_c5(h, dx) + abs(h.sum() - target) * 5.0
        try:
            de = differential_evolution(pen_obj, bounds, seed=seed,
                                        maxiter=70, tol=1e-10,
                                        popsize=10, mutation=(0.4, 1.0),
                                        recombination=0.8, init='sobol',
                                        polish=False)
            cand = project_constraint(de.x, n)
            c = compute_c5(cand, dx)
            if c < best_c5:
                best_h, best_c5 = cand, c
        except Exception:
            pass

    # Phase 2: SLSQP local refinement from multiple starts
    if elapsed(t0) < budget_s * 0.50:
        for _ in range(8):
            if elapsed(t0) > budget_s * 0.48:

            h_start = best_h + rng.randn(n) * 0.04
            h_start = project_constraint(h_start, n)
            try:
                res = minimize(lambda x: compute_c5(x, dx), h_start,
                             method='SLSQP', bounds=[(0.0, 1.0)] * n,
                             constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                             options={'maxiter': 250, 'ftol': 1e-14})
                cand = project_constraint(res.x, n)
                c = compute_c5(cand, dx)
                if c < best_c5:

            except Exception:
                pass

    # Phase 3: Simulated annealing (fine-grained)
    if elapsed(t0) < budget_s * 0.72:
        h = best_h.copy()
        c = best_c5
        T = 0.004

        for step in range(max_steps):
