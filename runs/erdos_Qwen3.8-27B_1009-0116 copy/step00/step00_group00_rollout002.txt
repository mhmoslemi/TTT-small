The correlation at shift s is:
C(s) = Σ_i h[i] · (1 - h[i+s])  (where i and i+s must be valid indices, 0 to n-1)

In numpy, correlate(h, 1-h, mode='full')[k] corresponds to shift s = k - (n-1).

For the subgradient at the maximizing shift s*:
∂C(s*)/∂h[j] = Σ_i ∂[h[i](1-h[i+s*])]/∂h[j]

h[j] appears as:
- h[i] when i=j: contributes (1-h[j+s*]) if j+s* is valid
- h[i+s*] when i+s*=j, i.e., i=j-s*: contributes -h[j-s*] if j-s* is valid

So ∂C(s*)/∂h[j] = (1-h[j+s*])·[0≤j+s*<n] - h[j-s*]·[0≤j-s*<n]

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    corr = correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def smooth_max_val(h, dx, alpha):
    corr = correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp((corr - m) * alpha))) / alpha

def smooth_max_grad(h, dx, alpha):
    n = len(h)
    corr = correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    w = np.exp((corr - m) * alpha)
    w /= w.sum()
    jac = np.zeros(n)
    ncorr = len(corr)
    for k in range(ncorr):
        if w[k] < 1e-12:
            continue
        shift = k - (n - 1)
        wk = w[k] * dx
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                jac[i] += wk * (1.0 - h[j])
            j2 = i - shift
            if 0 <= j2 < n:
                jac[i] -= wk * h[j2]
    return jac

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 1e-5) & (h < 1.0 - 1e-5)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-10:
            h[np.argmin(h) if diff2 > 0 else np.argmax(h)] += diff2
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h, n, time_budget, seed_val):
    dx = 2.0 / n
    np.random.seed(seed_val)
    start = project_feasible(h + 0.005 * np.random.randn(n), n)
    if compute_c5(start, dx) < compute_c5(h, dx):
        h_used = start
    else:
        h_used = h.copy()
    alpha = 200.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n / 2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    res = minimize(lambda x: smooth_max_val(x, dx, alpha), h_used,
                   jac=lambda x: smooth_max_grad(x, dx, alpha),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out, dx)

def coord_descent(h, n, max_time):
    dx = 2.0 / n
    t0 = time.time()
    for _ in range(80):
        if time.time() - t0 > max_time:
            break
        improved = False
        for i in range(n):
            best_v = h[i]
            best_c = None
            for v in np.concatenate([[0.0, 1.0], np.linspace(0, 1, 7)[1:-1]]):
                h[i] = v
                diff = n / 2.0 - h.sum()
                if abs(diff) > 1e-8:
                    free = (h > 1e-5) & (h < 1.0 - 1e-5) & (np.arange(n) != i)
                    if free.sum() > 0:

                        h = np.clip(h, 0, 1)
                c = compute_c5(h, dx)

                    best_c = c
                    best_v = v
            h[i] = best_v
            if best_v != h[i]:
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return project_feasible(h, n)
