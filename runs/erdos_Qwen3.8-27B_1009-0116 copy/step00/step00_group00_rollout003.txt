import numpy as np
from scipy.optimize import minimize, brentq

def project_feasible(x, S):
    """Project x onto {h: 0<=h<=1, sum(h)=S}"""
    # Water-filling: find alpha s.t. sum(clip(x_i - alpha, 0, 1)) = S
    def f(alpha):
        return np.sum(np.clip(x - alpha, 0, 1)) - S
    # f is monotonically decreasing in alpha
    # f(-inf) = sum(x) - S (large positive if sum(x) > S)
    # f(+inf) = -S (negative)
    lo, hi = -10, 10
    if f(lo) < 0:
        lo = -50
    if f(hi) > 0:
        hi = 50
    alpha = brentq(f, lo, hi, xtol=1e-12)
    return np.clip(x - alpha, 0, 1)

import numpy as np
from scipy.optimize import minimize, brentq
import time

def project_feasible(x, S):
    def f(a):
        return np.sum(np.clip(x - a, 0.0, 1.0)) - S
    lo, hi = -10.0, 10.0
    if f(lo) < 0:
        lo = -50.0
    if f(hi) > 0:
        hi = 50.0
    alpha = brentq(f, lo, hi, xtol=1e-14)
    return np.clip(x - alpha, 0.0, 1.0)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_corr_shift(h, n, shift):
    g = np.zeros(n)
    if shift >= 0:
        for i in range(shift, n):
            g[i] += 1.0 - h[i - shift]
        for i in range(0, n - shift):
            g[i] -= h[i + shift]
    else:
        s = -shift
        for i in range(0, n - s):
            g[i] += 1.0 - h[i + s]
        for i in range(s, n):
            g[i] -= h[i - s]
    return g

def lsq_smooth_obj(h, n, tau):
    corr = compute_corr(h, n)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau

def lsq_smooth_grad(h, n, tau):
    corr = compute_corr(h, n)
    m = np.max(corr)
    w = np.exp(tau * (corr - m))
    w /= np.sum(w)
    g = np.zeros(n)
    for idx in range(len(corr)):
        shift = idx - (n - 1)
        g += w[idx] * grad_corr_shift(h, n, shift)
    return g * (2.0 / n) * tau * 0 + g * (2.0 / n)

def slsqp_refine(h, n, tau=80.0, max_iter=200):
    S = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - S,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(lsq_smooth_obj, h, jac=lambda x: lsq_smooth_grad(x, n, tau),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})
    h2 = project_feasible(res.x, S)
    return h2, np.max(compute_corr(h2, n))

        k_idx = np.argmax(corr)

        if c5_now < c5_best - 1e-12:
            c5_best = c5_now
            h_best = h.copy()
        shift = k_idx - (n - 1)
        g = grad_corr_shift(h, n, shift) * dx
        # Also add small contribution from near-max to avoid degeneracy
        top_k = np.argsort(corr)[-3:]
        for tk in top_k:
            s2 = tk - (n - 1)
            g += 0.3 * grad_corr_shift(h, n, s2) * dx
        h_new = h - lr * g
        h_new = project_feasible(h_new, S)
