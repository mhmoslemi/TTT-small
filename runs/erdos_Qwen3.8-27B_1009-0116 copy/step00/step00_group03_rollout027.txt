import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def repair(h, n_points):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h


def coord_search(h, n_points, step=0.05, n_grid=20):
    improved = True
    n_passes = 0
    while improved and n_passes < 15:
        improved = False
        n_passes += 1
        for i in range(n_points):
            best_val = h[i]
            best_c5 = compute_c5(h, n_points)
            for _g in range(n_grid):
                t = _g / (n_grid - 1) * 2.0 - 1.0
                cand_val = best_val + t * step
                if cand_val < 0 or cand_val > 1:
                    continue
                test = h.copy()
                test[i] = cand_val
                test = repair(test, n_points)
                c5t = compute_c5(test, n_points)
                if c5t < best_c5 - 1e-13:
                    best_val = cand_val
                    best_c5 = c5t
                    h = test.copy()
                    improved = True
    return h

        return compute_c5(x, n_points)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points

        nonlocal best_h, best_c5
        try:
