        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 4: Random restarts with local optimization
    attempt = 0
    while time.time() - t0 < budget_s - 3:
        attempt += 1
        if attempt % 4 == 0:
            h0 = best_h + 0.15 * np.random.randn(n)
        elif attempt % 4 == 1:
            h0 = best_h + 0.05 * np.random.randn(n)
        elif attempt % 4 == 2:
            h0 = best_h + 0.02 * np.random.randn(n)
        else:
            h0 = np.random.rand(n)
        h0 = project_feasible(h0, n)
        h, c = slsqp_opt(h0, n, dx, maxiter=400, ftol=1e-12)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
