import numpy as np
from scipy.optimize import minimize
import time

_opt = {}

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(vals, alpha):
    m = np.max(vals)
    v = vals - m
    return m + np.log(np.sum(np.exp(alpha * v))) / alpha

def _eval_obj(h):
    vals = np.correlate(h, 1.0 - h, mode='full') * _opt['dx']
    return smooth_max_obj(vals, _opt['alpha'])

def _eval_grad(h):
    n = len(h)
    vals = np.correlate(h, 1.0 - h, mode='full') * _opt['dx']
    m = np.max(vals)
    v = vals - m
    ev = np.exp(_opt['alpha'] * v)
    w = ev / np.sum(ev)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            jr1 = np.arange(0, n - s)
            grad[jr1] += wk * _opt['dx'] * (1.0 - h[jr1 + s])
            jr2 = np.arange(s, n)
            grad[jr2] -= wk * _opt['dx'] * h[jr2 - s]
        else:
            sa = -s
            jr1 = np.arange(sa, n)
            grad[jr1] += wk * _opt['dx'] * (1.0 - h[jr1 - sa])
            jr2 = np.arange(0, n - sa)
            grad[jr2] -= wk * _opt['dx'] * h[jr2 + sa]
    return grad

def _con_fun(h):
    return float(np.sum(h)) - _opt['target']

def _con_jac(h):
    return np.ones(_opt['n'])

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                amt = min(diff, 1.0 - h[i])
                if amt > 1e-15:
                    h[i] += amt
                    diff -= amt
                if diff <= 1e-14:
                    break
        else:
            diff = -diff
            order = np.argsort(h)[::-1]
            for i in order:
                amt = min(diff, h[i])
                if amt > 1e-15:
                    h[i] -= amt
                    diff -= amt
                if diff <= 1e-14:
                    break
    return h

def upsample(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old, dtype=float)
    xn = np.linspace(0.0, n_old - 1.0, n_new)
    return np.interp(xn, xo, h)

def _slsqp_optimize(h, alpha, dx, n, target, maxiter=300, tol=1e-10):
    _opt['dx'] = dx
    _opt['alpha'] = alpha
    _opt['n'] = n
    _opt['target'] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_eval_obj, h, jac=_eval_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None:
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    best_c5 = np.inf
    best_h = None

    # Get initial values
    h_init = None

        h_init = np.full(n_t, 0.5)
        mid = n_t // 2
        h_init += 0.2 * np.exp(-0.5 * ((np.arange(n_t) - mid) / (n_t * 0.15)) ** 2)

    n_init = len(h_init)
    dx_init = 2.0 / n_init
    h_init = project_feasible(h_init, n_init / 2.0)
    c5_init = compute_c5(h_init, dx_init)
    best_c5 = c5_init
    best_h = h_init.copy()

    resolutions = [n_init, 60, 80, 100, 120, 150, 200]
    resolutions = sorted(set(r for r in resolutions if r >= 30))

    for n in resolutions:
        if time.time() - t0 > budget_s * 0.72:
            break
        dx = 2.0 / n
        target = n / 2.0

        # Initialize at this resolution
        if len(best_h) == n:
            h0 = best_h.copy()
        else:
            h0 = upsample(best_h, n)
        h0 = project_feasible(h0, target)
        c5_cur = compute_c5(h0, dx)

        # SLSQP optimization with increasing alpha
        for alpha in [20, 50, 150, 400, 800]:
            if time.time() - t0 > budget_s * 0.75:
                break
            h_new = _slsqp_optimize(h0, alpha, dx, n, target, maxiter=200)
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5_cur - 1e-8:
                c5_cur = c5_new
                h0 = h_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()

        # Random restarts (2 tries)
        for _r in range(2):
            if time.time() - t0 > budget_s * 0.78:
                break
            h_r = h0 + 0.04 * rng.standard_normal(n)
            h_r = project_feasible(h_r, target)
            h_r = _slsqp_optimize(h_r, 100, dx, n, target, maxiter=150)
            h_r = project_feasible(h_r, target)
            c5_r = compute_c5(h_r, dx)
            if c5_r < best_c5:

                best_h = h_r.copy()

    # Final refinement at best resolution
    if best_h is not None:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0
        for alpha in [500, 1000]:
            if time.time() - t0 > budget_s * 0.92:

            h_f = _slsqp_optimize(best_h, alpha, dx_f, n_f, target_f, maxiter=200)

            if compute_c5(h_f, dx_f) < best_c5:
                best_c5 = compute_c5(h_f, dx_f)
                best_h = h_f

    dx_final = 2.0 / n_final
    best_h = project_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h, dx_final)

    return best_h, final_c5, n_final
