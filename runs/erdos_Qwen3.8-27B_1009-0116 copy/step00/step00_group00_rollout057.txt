def corr_grad(h, k):

    grad = np.zeros(n)
    if k < n:
        # corr[k] = sum_{j=0}^{n-1-k} h[j+k] * (1-h[j])
        # d/dh[i] = (1-h[i-k]) * I(0<=i-k<=n-1-k) - h[i+k] * I(0<=i<=n-1-k)
        # Term 1: i-k in [0, n-1-k] => i in [k, n-1]
        idx1 = np.arange(k, n)
        grad[idx1] += (1 - h[idx1 - k])
        # Term 2: i in [0, n-1-k]
        idx2 = np.arange(0, n - k)
        grad[idx2] -= h[idx2 + k]
    else:
        kp = k - n + 1
        # corr[k] = sum_{j=kp}^{n-1} h[j] * (1-h[j-kp])

        grad[idx1] += (1 - h[idx1 - kp])
        idx2 = np.arange(0, n - kp)
        grad[idx2] -= h[idx2 + kp]

    target = new_n / 2.0
    h_new += (target - np.sum(h_new)) / new_n
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new += (target - np.sum(h_new)) / new_n
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def lse_and_grad(x, tau, penalty=100.0):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(tau * shifted)
    exp_sum = np.sum(exps)
    lse_val = m + np.log(exp_sum) / tau
    eps = 1e-7
    grad = np.zeros(n)
    for i in range(n):
        xp = x.copy()
        xm = x.copy()
        xp[i] = min(1.0, x[i] + eps)
        xm[i] = max(0.0, x[i] - eps)
        ae = xp[i] - xm[i]
        if ae < 1e-14:
            grad[i] = 0.0
            continue
        cp = np.correlate(xp, 1.0 - xp, mode='full') * dx
        cm = np.correlate(xm, 1.0 - xm, mode='full') * dx
        mp = np.max(cp)
        mm = np.max(cm)
        lp = mp + np.log(np.sum(np.exp(tau * (cp - mp)))) / tau
        lm = mm + np.log(np.sum(np.exp(tau * (cm - mm)))) / tau
        grad[i] = (lp - lm) / ae
    s = np.sum(x) - n / 2.0
    penalty_val = penalty * s * s
    penalty_grad = 2.0 * penalty * s
    return lse_val + penalty_val, grad + penalty_grad

def optimize_lse(h0, time_limit=60.0):
    n = len(h0)
    t0 = time.time()
    x = h0.copy()
    tau_schedule = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 400.0]
    for tau in tau_schedule:
        if time.time() - t0 > time_limit - 5:
            break
        penalty = 100.0 if tau < 50.0 else 200.0
        res = minimize(lse_and_grad, x, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       args=(tau, penalty),
                       options={'maxiter': 150, 'ftol': 1e-14, 'gtol': 1e-10})
        x = np.clip(res.x, 0.0, 1.0)
    target = n / 2.0
    x += (target - np.sum(x)) / n
    x = np.clip(x, 0.0, 1.0)
    return x

def polish_local(h0, n_trials=50):
    n = len(h0)
    best = h0.copy()
    best_c5 = compute_c5(best)
    rng = np.random.default_rng(12345)
    for _ in range(n_trials):
        trial = best + 0.02 * rng.standard_normal(n)
        trial = np.clip(trial, 0.0, 1.0)
        trial += (n / 2.0 - np.sum(trial)) / n
        trial = np.clip(trial, 0.0, 1.0)
        c = compute_c5(trial)
        if c < best_c5:
            best, best_c5 = trial, c
    return best, best_c5

def slsqp_refine(h0, time_limit=60.0):
    n = len(h0)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    c5_init = compute_c5(h0)
    
    def slsqp_obj(x):
        return x[-1]
    
    def make_ineq(kk):
        def fun(x):
            h = x[:-1]
            c = np.correlate(h, 1.0 - h, mode='full')[kk] * dx
            return x[-1] - c
        return fun
    
    def make_ineq_grad(kk):
        def fun(x):
            h = x[:-1]
            g = np.zeros(n)
            if kk < n:
                idx1 = np.arange(kk, n)
                g[idx1] += (1.0 - h[idx1 - kk])
                idx2 = np.arange(0, n - kk)
                g[idx2] -= h[idx2 + kk]
            else:
                kp = kk - n + 1
                idx1 = np.arange(kp, n)
                g[idx1] += (1.0 - h[idx1 - kp])
                idx2 = np.arange(0, n - kp)
                g[idx2] -= h[idx2 + kp]
            g *= -dx
            g = np.concatenate([g, [1.0]])
            return g
        return fun
    
    cons = []
    for k in range(n_corr):
        cons.append({'type': 'ineq', 'fun': make_ineq(k), 'grad': make_ineq_grad(k)})
    cons.append({'type': 'eq', 'fun': lambda x: np.sum(x[:-1]) - n / 2.0})
    
    x0 = np.concatenate([h0, [c5_init * 1.001]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    t0 = time.time()
    
    res = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
    if time.time() - t0 > time_limit:
        return h0
    if res.x is not None:
        h_opt = np.clip(res.x[:-1], 0.0, 1.0)
        h_opt += (n / 2.0 - np.sum(h_opt)) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
        return h_opt
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(45, 0.5)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Optimize at initial resolution
    time_left = budget_s - (time.time() - start_time)
    if time_left > 40:
        h_opt = optimize_lse(h_init, time_limit=min(100, time_left - 20))
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
        # Local polish
        if time_left > 60:
            h_opt, c5 = polish_local(best_h, n_trials=200)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5
    
    # Phase 2: Try multiple resolutions
    resolutions = [60, 80, 100, 120, 160, 200]
    for n in resolutions:
        time_left = budget_s - (time.time() - start_time)
        if time_left < 80:
            break
        
        h_start = upsample(best_h, n)
        time_per = max(30, (time_left - 40) // 3)
        
        # Trial 1: direct
        h_opt = optimize_lse(h_start, time_limit=time_per)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
        
        time_left = budget_s - (time.time() - start_time)
        if time_left < 50:
            continue
        
        # Trial 2: perturbed
        h_pert = h_start + 0.05 * rng.standard_normal(n)
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert += (n / 2.0 - np.sum(h_pert)) / n
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_opt = optimize_lse(h_pert, time_limit=time_per)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
        
        # Trial 3: from original init at this resolution
        h_from_init = upsample(h_init, n)
        time_left = budget_s - (time.time() - start_time)
        if time_left > 60:
            h_opt = optimize_lse(h_from_init, time_limit=min(50, time_left - 20))
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5
    
    # Phase 3: SLSQP refinement at best resolution (if small enough)
    time_left = budget_s - (time.time() - start_time)
    n_best = len(best_h)
    if time_left > 50 and n_best <= 120:
        h_slsqp = slsqp_refine(best_h, time_limit=min(90, time_left - 20))
        c5 = compute_c5(h_slsqp)
        if c5 < best_c5:
            best_h, best_c5 = h_slsqp, c5
    
    # Phase 4: Final local polish
    time_left = budget_s - (time.time() - start_time)
    if time_left > 20:
        h_polished, c5 = polish_local(best_h, n_trials=min(500, int(time_left * 20)))
        if c5 < best_c5:
            best_h, best_c5 = h_polished, c5
    
    # Ensure constraints
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h += (n_final / 2.0 - np.sum(best_h)) / n_final
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_final
