import numpy as np
import time

def c5_values(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1 - h, mode="full") * dx

def c5_max(h):
    return float(np.max(c5_values(h)))

def smooth_grad(h, beta):
    n = len(h)
    v = c5_values(h)
    m = np.max(v)
    w = np.exp(beta * (v - m))
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        wt = w[k] * dx
        if s >= 0:
            if s == 0:
                g += wt * (1.0 - 2.0 * h)
            else:
                lhs = np.zeros(n)
                rhs = np.zeros(n)
                lhs[s:] = 1.0 - h[:n - s]
                rhs[:n - s] = -h[s:]
                g += wt * (lhs + rhs)
        else:
            m2 = -s
            if m2 == 0:
                g += wt * (1.0 - 2.0 * h)
            else:
                lhs = np.zeros(n)
                rhs = np.zeros(n)
                lhs[m2:] = 1.0 - h[:n - m2]
                rhs[:n - m2] = -h[m2:]
                g += wt * (lhs + rhs)
    return g

def project(h):
    n = len(h)
    tgt = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0, 1).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    r = np.clip(h - (lo + hi) / 2.0, 0, 1)
    d = tgt - r.sum()
    if abs(d) > 1e-12:
        free = (r > 1e-10) & (r < 1.0 - 1e-10)
        if free.any():
            r[free] += d / free.sum()
        else:
            r[0] = np.clip(r[0] + d, 0, 1)
    return np.clip(r, 0, 1)

def grad_desc(h, beta, max_iter, time_fn, budget_frac):
    n = len(h)
    lr = 2.0 / (beta + 1.0)
    cur_c5 = c5_max(h)
    for _ in range(max_iter):
        if time_fn() < budget_frac:
            break
        g = smooth_grad(h, beta)
        h_new = project(h - lr * g)
        c5_new = c5_max(h_new)
        if c5_new < cur_c5:
            h, cur_c5 = h_new, c5_new
        else:
            lr *= 0.7
        if lr < 1e-6:
            break
    return h, cur_c5

def coord_refine(h, best_c5, time_fn):
    n = len(h)
    deltas = [0.05, 0.02, 0.01, 0.005, -0.05, -0.02, -0.01, -0.005]
    for _ in range(200):
        if time_fn() < 3.0:
            break
        improved = False
        for j in range(n):
            for d in deltas:
                h2 = h.copy()
                h2[j] = min(1.0, max(0.0, h2[j] + d))
                h2 = project(h2)
                c2 = c5_max(h2)
                if c2 < best_c5 - 1e-12:
                    h, best_c5 = h2, c2
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def make_start(n, rng):
    h = np.zeros(n)
    k = n // 2
    h[:k] = 1.0
    if n % 2 == 1:
        h[k] = 0.5
    return project(h + rng.normal(0, 0.4, n))

def binary_snap(h, rng, n):
    b = (h > 0.5).astype(float)
    tgt = n / 2.0
    ns = b.sum()
    if ns > int(tgt):
        idx1 = np.where(b == 1)[0]
        idx1 = idx1[np.argsort(h[idx1])[:int(ns - tgt)]]
        b[idx1] = 0.0
    elif ns < int(tgt):
        idx0 = np.where(b == 0)[0]
        idx0 = idx0[np.argsort(1.0 - h[idx0])[:int(tgt - ns)]]
        b[idx0] = 1.0
    if abs(b.sum() - tgt) > 1e-10:
        fr = tgt - b.sum()
        free = np.where((b == 0) | (b == 1))[0]
        b[free[rng.randint(len(free))]] = fr
    return project(b)

    rng = np.random.RandomState(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    best_h, best_c5 = h_init.copy(), c5_max(h_init)

    # Phase 1: Multi-scale projected gradient
    for phase in range(10):

        beta = 15.0 * (phase + 1) ** 1.3

    # Phase 2: Coordinate refinement from best
    if tl() > 10:

        if tl() < 20:

        sigma = 0.2 + 0.3 * (r / max(1, n_restarts))
        h_r = project(best_h + rng.normal(0, sigma, n))
        h_r, c5_r = grad_desc(h_r, 60 + 20 * r, 200, tl, 10)
        if c5_r < best_c5:

            h_r, best_c5 = coord_refine(h_r.copy(), best_c5, tl)

    best_c5 = c5_max(best_h)
    return best_h, best_c5, len(best_h)
