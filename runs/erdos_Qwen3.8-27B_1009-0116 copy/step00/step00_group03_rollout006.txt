    t_start = time.time()
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(70, 0.5)
    n = len(h0)
    target = n / 2.0
    h0 = project_feasible(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    # Main optimization from initial
    rem = budget_s - (time.time() - t_start)
    t_main = min(300.0, rem * 0.45)
    best_h, best_c5 = opt_combined(best_h, target, t_main)
    # Additional refinement passes with decreasing perturbation
    passes = [
        (0.03, 40.0, 40.0),
        (0.02, 30.0, 30.0),
        (0.01, 25.0, 25.0),
        (0.05, 40.0, 30.0),
        (0.015, 30.0, 25.0),
        (0.04, 30.0, 25.0),
        (0.008, 20.0, 20.0),
        (0.025, 30.0, 20.0),
    ]
    for scale, t_sg, t_sm in passes:
        rem = budget_s - (time.time() - t_start)
        if rem < 20:
            break
        h_r = perturb(best_h, target, rng, scale)
        # Also try a different perturbation direction
        h_r2 = perturb(h0, target, rng, scale * 2)
        t_each = min(30.0, rem / 4.0)
        hr1, cr1 = opt_combined(h_r, target, t_each)
        rem = budget_s - (time.time() - t_start)
        if rem > 15:
            hr2, cr2 = opt_combined(h_r2, target, min(25.0, rem / 2.0))
        else:
            cr2, hr2 = 999.0, h_r
        if cr1 < best_c5:
            best_c5 = cr1
            best_h = hr1
        if cr2 < best_c5:
            best_c5 = cr2
            best_h = hr2
    # Final polish on best
    rem = budget_s - (time.time() - t_start)
    if rem > 10:
        best_h, best_c5 = opt_combined(best_h, target, min(60.0, rem * 0.8))
    # Verify and finalize
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
