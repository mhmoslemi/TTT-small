import numpy as np
from scipy import optimize
import time

def compute_corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h, dx):
    return np.max(compute_corr(h, dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        nf = np.sum(free)
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_softmax(h, dx, lam=20.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    scaled = lam * corr
    scaled -= np.max(scaled)
    weights = np.exp(scaled)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = lam * weights[m]
        if w < 1e-10:
            continue
        k = m - (n - 1)
        i_min = max(0, -k)
        i_max = min(n, n - k)
        for i in range(i_min, i_max):
            grad[i] += w * (1.0 - h[i + k]) * dx
            grad[i + k] += w * (-h[i]) * dx
    return grad

def projected_grad_desc(h, dx, n, max_iter=5000, time_limit=100):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    step = 0.01
    lam = 20.0
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        g = grad_softmax(h, dx, lam)
        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # Increase step slightly

        else:

        h = h_new
        if step < 1e-6:

            lam = min(lam * 1.5, 200.0)  # sharpen

    return np.interp(x_new, x_old, h)

def project_feasible(h, n):
    target = n / 2.0
    # Binary search for lambda: sum(clip(h+lambda, 0, 1)) = target
    lo = -h.max() - 0.1
    hi = 1.0 - h.min() + 0.1
    for _ in range(200):
        mid = 0.5 * (lo + hi)

        if s > target:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-13:
            break
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def grad_softmax(h, dx, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    sc = lam * corr
    sc -= np.max(sc)
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = lam * w[m]
        if wm < 1e-12:
            continue
        k = m - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        for i in range(i_lo, i_hi):
            grad[i] += wm * (1.0 - h[i + k]) * dx
            grad[i + k] -= wm * h[i] * dx
    return grad

def pgd(h_in, dx, n, max_iter, t_limit, lam0=20.0):

    h = h_in.copy()
    best_h = h.copy()

        h_new = project_feasible(h - step * g, n)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            stall = 0
            step = min(step * 1.03, 0.15)
        else:

            step *= 0.92

                step = max(step, 0.01)
                if stall > 300:
                    h_new = project_feasible(h + np.random.randn(n) * 0.04, n)
                    stall = 0
        h = h_new
    return best_h, best_c5

def slsqp_refine(h_in, dx, n, t_limit=120.0):
    lam = 40.0
    def f(x):
        c = compute_corr(x, dx)
        s = lam * c
        s -= s.max()
        return np.log(np.sum(np.exp(s))) / lam
    x0 = h_in.copy()
    try:
        res = minimize(f, x0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq',
                                     'fun': lambda x: x.sum() - n / 2.0,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': 300, 'ftol': 1e-12})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5 = compute_c5(h_new, dx)
            if c5 < compute_c5(x0, dx):
                return h_new, c5
    except Exception:
        pass
    return x0, compute_c5(x0, dx)
