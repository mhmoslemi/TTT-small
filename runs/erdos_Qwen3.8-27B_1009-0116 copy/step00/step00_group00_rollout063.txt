import numpy as np
from scipy.interpolate import interp1d
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def subgradient(h, n):
    """Compute subgradient of C5 = max_k corr_k * dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Active lags: within tolerance of max
    eps = max(1e-6, 0.01 * max_val)
    active = corr >= max_val - eps
    
    # Compute gradient for active lags
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if not active[idx]:
            continue
        k = idx - (n - 1)  # actual lag
        # dc_k/dh[i] = (1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]
        # Vectorized:
        i = np.arange(n)
        # First term: (1-h[i-k]) where 0<=i-k<n
        mask1 = (i - k >= 0) & (i - k < n)
        if np.any(mask1):
            grad[mask1] += (1.0 - h[i[mask1] - k])
        # Second term: -h[i+k] where 0<=i+k<n
        mask2 = (i + k >= 0) & (i + k < n)
        if np.any(mask2):
            grad[mask2] -= h[i[mask2] + k]
    
    return grad * dx / np.sum(active)

def project(h, n):
    """Project h onto {sum(h)=n/2, 0<=h<=1}."""
    target = n / 2.0
    # Bisection on tau
    lo, hi = -h.max() - 1, -h.min() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        h_p = np.clip(h - mid, 0, 1)
        s = h_p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune to get exact sum
    diff = target - h_p.sum()
    # Distribute adjustment to elements not at boundary
    interior = (h_p > 1e-8) & (h_p < 1 - 1e-8)
    if np.any(interior):
        h_p[interior] += diff / np.sum(interior)
        h_p = np.clip(h_p, 0, 1)
    return h_p

def optimize_subgrad(h_init, n, steps=50000, t_end=None):
    """Projected subgradient with decreasing step size."""
    h = h_init.copy().astype(float)
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    h = project(h, n)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    step0 = 0.1
    for step in range(steps):
        if t_end is not None and time.time() > t_end:
            break
        
        g = subgradient(h, n)
        lr = step0 / (1 + 0.01 * step)
        h_new = h - lr * g
        h_new = project(h_new, n)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def upsample(h, n_new):
    """Upsample h to n_new points."""
    x_old = np.arange(len(h))
    x_new = np.arange(n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0, 1)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

    t_start = time.time()
    deadline = t_start + budget_s * 0.92

    if c5 < best_c5:

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_c5(h, alpha=50.0):
    """Smooth approximation of C5 using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def make_h_from_pattern(p, n):

    period = len(p)
    q = n // period
    rem = n % period
    h = np.tile(p, q + 1)[:n]
    return h

def optimize_periodic(period, n, n_restart=20, t_end=None, seed=42):

    """Projected subgradient optimization for general h."""

    h = np.clip(h, 0, 1)

    h = _project(h, n)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    dx = 2.0 / n
    step0 = 0.5
    patience = 0
    target = best_c5
    
    for it in range(steps):
        if t_end and time.time() > t_end:
            break
        if it > 0 and it % 5000 == 0:
            if best_c5 >= target - 1e-9:
                patience += 1
                if patience > 6:
                    break
            else:
                target = best_c5
                patience = 0
        
        # Compute subgradient
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        eps = max(1e-8, 0.005 * max_val)
        active = corr >= max_val - eps
        
        grad = np.zeros(n)
        n_active = np.sum(active)
        for idx in range(2 * n - 1):
            if not active[idx]:
                continue
            k = idx - (n - 1)
            i = np.arange(n)
            m1 = (i - k >= 0) & (i - k < n)
            m2 = (i + k >= 0) & (i + k < n)
            grad[m1] += (1.0 - h[i[m1] - k])
            grad[m2] -= h[i[m2] + k]
        
        grad = grad * dx / n_active
        
        # Step

        h_new = h - lr * grad
        h_new = _project(h_new, n)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def _project(h, n):

    target = n / 2.0

    for _ in range(80):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))

        else:

        if np.sum(interior) > 0:
            h_p[interior] += diff / np.sum(interior)
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

        p_f = p_f + (half_period - p_f.sum()) / period
        p_f = np.clip(p_f, 0.0, 1.0)
        h = np.tile(p_f, q)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_p = p_f.copy()
    return best_p, best_c5

def optimize_subgrad(h_init, n, steps, t_end, step0=0.4, thresh=0.003):
    h = _project(h_init.astype(float).copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(steps):
        if t_end is not None and time.time() > t_end:
            break
        g = _subgrad_step(h, n, thresh)
        lr = step0 / (1.0 + 0.02 * np.sqrt(float(it + 1)))
        h = _project(h - lr * g, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _constraint_target
    t_start = time.time()
    deadline = t_start + budget_s * 0.93

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 1: Periodic constructions (most promising for C5)
    configs = [(5, 45), (5, 50), (5, 60), (5, 55), (3, 45), (3, 48),
               (7, 42), (7, 49), (6, 42), (6, 48), (6, 54), (6, 60),
               (4, 48), (10, 50), (10, 60), (9, 45), (15, 45), (2, 46)]
    for period, n in configs:
        if time.time() > deadline:
            break
        p_opt, c5 = optimize_periodic(period, n, 8, deadline, seed)
        if p_opt is not None and c5 < best_c5:
            best_c5 = c5
            best_n = n
            best_h = np.tile(p_opt, n // period)

    # Phase 2: General subgradient optimization
    starts = []
    try:
        starts.append(np.array(initial_h_values, dtype=float))
    except:
        pass
    if best_h is not None:
        starts.append(best_h)

    for hs in starts[:3]:
        if time.time() > deadline:
            break
        n_g = len(hs)
        h_opt, c5 = optimize_subgrad(hs, n_g, 120000, deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_n = n_g
            best_h = h_opt

    # Phase 2b: Second subgradient pass with different params
    if best_h is not None and time.time() < deadline:
        n_g = best_n
        h_opt2, c5_2 = optimize_subgrad(best_h, n_g, 80000, deadline,
                                         step0=0.2, thresh=0.001)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h_opt2
            best_n = n_g

    # Phase 3: SLSQP polish with smooth max
    if best_h is not None and time.time() < deadline - 60:
        n_b = best_n
        _constraint_target = n_b / 2.0
        bounds = [(0.0, 1.0)] * n_b
        cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
        res = minimize(smooth_c5_obj, best_h, method='SLSQP', args=(100.0,),
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 600, 'ftol': 1e-15})
        h_pol = _project(np.clip(res.x, 0.0, 1.0), n_b)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
            best_n = n_b

    # Phase 4: One more subgradient refinement
    if best_h is not None and time.time() < deadline - 10:
        n_g = best_n
        h_ref, c5_ref = optimize_subgrad(best_h, n_g, 60000, deadline,
                                          step0=0.1, thresh=0.001)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
            best_n = n_g

    if best_h is None:
        best_h = np.full(45, 2.0 / 45.0)
        best_n = 45
        best_c5 = compute_c5(best_h)

    best_h = _project(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
