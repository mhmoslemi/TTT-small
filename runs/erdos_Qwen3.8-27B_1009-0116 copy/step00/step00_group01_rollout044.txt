import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target_sum) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h_new)
    free = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
    if np.sum(free) > 0 and abs(diff) > 1e-14:
        h_new[free] += diff / np.sum(free)
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def compute_c5(h, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def interpolate_up(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def compute_corr_and_grad(h, n, tau):
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    m = np.max(corr)
    exp_c = np.exp((corr - m) / tau)
    weights = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        offset = k - (n - 1)
        lo1 = max(0, offset)
        hi1 = min(n - 1, n - 1 + offset)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += w * (1.0 - h[lo1 - offset:hi1 + 1 - offset])
        lo2 = max(0, -offset)
        hi2 = min(n - 1, n - 1 - offset)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= w * h[lo2 + offset:hi2 + 1 + offset]
    return corr, grad

def projected_grad_descend(h, n, t_start, budget_s):
    target_sum = n / 2.0
    h_best = h.copy()
    c5_best = compute_c5(h, n)
    taus = np.array([0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001])
    n_stages = len(taus)
    for si in range(n_stages):
        tau = taus[si]
        iters = 800
        step = 0.15 * tau / (1.0 + 0.1 * si)
        for it in range(iters):
            if time.time() - t_start > budget_s - 30:
                return h_best, c5_best
            corr, grad = compute_corr_and_grad(h, n, tau)
            c5_val = np.max(corr) * (2.0 / n)
            if c5_val < c5_best - 1e-12:
                c5_best = c5_val
                h_best = h.copy()
            h_new = h - step * grad
            h_new = project_feasible(h_new, target_sum)
            new_c5 = compute_c5(h_new, n)
            if new_c5 <= c5_val + 1e-10:
                h = h_new
            else:
                step *= 0.8
                if step < 1e-6:
                    break
    return h_best, c5_best

def random_init(n, rng):
    h = rng.random(n)
    h = project_feasible(h, n / 2.0)
    return h
