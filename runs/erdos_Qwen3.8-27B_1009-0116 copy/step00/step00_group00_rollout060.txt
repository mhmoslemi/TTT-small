```python
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if diff > 0:

        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            if diff <= 1e-10:

    elif diff < 0:

        order = np.argsort(-h)

            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem

                break
```

```python
def lse_optimize(h0, n, dx, alpha, maxiter=2000, ftol=1e-10):
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        shifted = alpha * (c - m)
        # numerically stable logsumexp
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        shifted = alpha * (c - m)
        weights = np.exp(shifted)
        weights /= weights.sum()
        # grad of c w.r.t. x, weighted average
        # c[k] = sum_i x[i]*(1-x[i+k]) * dx (with boundary handling in correlate)
        # This is complex; use numerical gradient
        return None
    
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 50})
        # L-BFGS-B doesn't support equality constraints well
        # Let's use SLSQP instead
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
        h_out = make_feasible(res.x, n)
        return h_out, compute_c5(h_out, dx)
    except:
        return h0, compute_c5(h0, dx)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_sum_n = 0

def sum_con_fun(x):
    return np.sum(x) - _sum_n / 2.0

def obj_max_corr(x, dx):
    return float(np.max(np.correlate(x, 1.0 - x, mode='full')) * dx)

def obj_lse(x, dx, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            idx = np.where(room > 1e-10)[0]
            if len(idx) == 0:
                break
            h[idx] = np.minimum(1.0, h[idx] + diff / len(idx))
        else:
            idx = np.where(h > 1e-10)[0]
            if len(idx) == 0:
                break
            h[idx] = np.maximum(0.0, h[idx] + diff / len(idx))
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h0, n, dx, maxiter=2000, alpha=None):
    global _sum_n
    _sum_n = n
    cons = [{'type': 'eq', 'fun': sum_con_fun}]
    bnds = [(0.0, 1.0)] * n
    try:
        if alpha is None:
            res = minimize(obj_max_corr, h0, args=(dx,), method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-9})
        else:
            res = minimize(obj_lse, h0, args=(dx, alpha), method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-9})
        h_out = make_feasible(res.x, n)
        return h_out, compute_c5(h_out, dx)
    except Exception:
        h_out = make_feasible(h0, n)
        return h_out, compute_c5(h_out, dx)

def nelder_optimize(h0, n, dx, maxiter=300):
    h0 = make_feasible(h0, n)
    def obj_pen(x):
        c = np.max(np.correlate(x, 1.0 - x, mode='full')) * dx
        pen = 500.0 * abs(np.sum(x) - n / 2.0)
        pen += 500.0 * (np.sum(np.maximum(0.0, x - 1.0)) + np.sum(np.maximum(0.0, -x)))
        return c + pen
    try:
        res = minimize(obj_pen, h0, method='Nelder-Mead',
                      options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-9})
        h_out = make_feasible(res.x, n)
        return h_out, compute_c5(h_out, dx)
    except Exception:
        return h0, compute_c5(h0, dx)
```
