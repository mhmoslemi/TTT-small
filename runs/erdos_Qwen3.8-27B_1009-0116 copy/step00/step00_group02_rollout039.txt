import numpy as np
from scipy.optimize import minimize
import time

# Global params for scipy callback
_P = {}

def _obj(x):
    comp = compute_overlaps(x)
    m = float(comp.max())
    b = _P['beta']
    return m + float(np.log(np.sum(np.exp(b*(comp-m)))))/b

    comp = compute_overlaps(x)
    m = float(comp.max())
    b = _P['beta']
    w = np.exp(b*(comp-m))
    w /= float(w.sum())
    n = len(x)
    g = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for k in nz:
        g += w[k] * grad_overlap(x, k-(n-1))
    return g

def _con(x):
    return float(x.sum()) - _P['target']

def project_feasible(x, n):
    target = n/2.0
    lo = float(x.min())-1.0
    hi = float(x.max())+1.0
    for _ in range(100):
        mid = (lo+hi)*0.5
        h = np.clip(x+mid, 0, 1)
        if float(h.sum()) > target:
            hi = mid
        else:
            lo = mid

    return np.correlate(h, 1.0-h, mode='full')*(2.0/len(h))

def compute_c5(h):
    return float(compute_overlaps(h).max())

    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[:n-s] -= h[s:]
            g[s:] += 1.0-h[:n-s]
    else:
        t = -s
        if t < n:
            g[:n-t] += 1.0-h[t:]
            g[t:] -= h[:n-t]
    return g
