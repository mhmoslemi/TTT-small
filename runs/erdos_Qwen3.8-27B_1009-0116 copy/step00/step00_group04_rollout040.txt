import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        nf = np.sum(free)
        if nf > 0:
            h[free] += diff2 / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n, maxiter=500, ftol=1e-12):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target}]

    def obj(h):
        return compute_c5(h, n)

    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            hx = repair_h(res.x, n)
            vx = compute_c5(hx, n)
            if vx < compute_c5(h0, n):
                return hx, vx
    except Exception:
        pass
    return h0.copy(), compute_c5(h0, n)

def local_search(h, n, n_iters, t0, budget_s, stall_limit=20000):
    best = h.copy()
    best_val = compute_c5(best, n)
    rng = np.random.RandomState(777)
    deltas = [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    stall = 0
    temp = 0.001

    for it in range(n_iters):
        if time.time() - t0 > budget_s - 30:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = deltas[rng.randint(len(deltas))]
        found = False
        for sign in (1, -1):
            nh = best.copy()
            ni = best[i] + sign * delta
            nj = best[j] - sign * delta
            if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                continue
            nh[i] = min(max(ni, 0.0), 1.0)
            nh[j] = min(max(nj, 0.0), 1.0)
            v = compute_c5(nh, n)
            if v < best_val - 1e-14:
                best_val = v
                best = nh
                stall = 0
                temp = min(temp * 0.999, 1e-6)
                found = True
                break
        if not found:
            stall += 1
            if stall > stall_limit:
                temp = min(temp * 1.01, 0.01)
        if stall > stall_limit * 3 and temp <= 1e-6:
            break
    return best, best_val

def upscale(h, n):
    n2 = n * 2
    h2 = np.empty(n2)
    h2[0::2] = h
    h2[1::2] = h
    return repair_h(h2, n2)

def random_binary(n, rng):
    ones = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.ones(42) * 0.5
    n0 = len(h0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(best_h, best_n)

    elapsed = lambda: time.time() - t0
    rem = lambda: budget_s - elapsed()

    # Phase 1: SLSQP
    if rem() > 120:
        h1, v1 = slsqp_optimize(best_h, best_n, maxiter=400)
        if v1 < best_val:
            best_h, best_val = h1, v1

    # Phase 2: Local search at base resolution
    if rem() > 80:
        n_iters = int(min(rem() * 30000, 2000000))
        h2, v2 = local_search(best_h, best_n, n_iters, t0, budget_s)
        if v2 < best_val:
            best_h, best_val = h2, v2

    # Phase 3: Upscale to 2x and optimize
    if rem() > 200:
        n2 = best_n * 2
        h_up = upscale(best_h, best_n)
        h3, v3 = slsqp_optimize(h_up, n2, maxiter=300)
        if v3 < best_val:
            best_h, best_n, best_val = h3, n2, v3
        if rem() > 100:
            n_iters = int(min(rem() * 15000, 800000))
            h3, v3 = local_search(best_h, best_n, n_iters, t0, budget_s)
            if v3 < best_val:
                best_h, best_val = h3, v3

    # Phase 4: Try random restarts at base resolution
    if rem() > 150:
        rng = np.random.RandomState(seed + 1)
        for attempt in range(5):
            if rem() < 30:
                break
            h_rand = random_binary(n0, rng)
            # Smooth slightly
            h_rand = np.clip(h_rand + rng.randint(0, 2, n0) * 0.0 - 0.0, 0, 1)
            # Actually keep binary, then SLSQP will smooth
            hr, vr = slsqp_optimize(h_rand, n0, maxiter=200)
            if vr < best_val:
                n_iters = int(min(rem() * 20000, 300000))
                hr, vr = local_search(hr, n0, n_iters, t0, budget_s)
                if vr < best_val:
                    best_h, best_n, best_val = hr, n0, vr

    # Phase 5: Final SLSQP polish
    if rem() > 60:
        h_fin, v_fin = slsqp_optimize(best_h, best_n, maxiter=200, ftol=1e-14)
        if v_fin < best_val:
            best_h, best_val = h_fin, v_fin

    best_h = repair_h(best_h, best_n)
    best_val = compute_c5(best_h, best_n)
    return best_h, best_val, best_n
