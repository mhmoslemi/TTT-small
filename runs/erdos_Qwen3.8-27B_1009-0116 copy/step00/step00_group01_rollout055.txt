        else:
            h0 = step_interpolate(best_h, n_res)
        h0 = repair(h0, n_res)

        local_best = h0.copy()
        local_c5 = compute_c5(local_best, dx)

        # Phase 1: SLSQP (handles constraints natively)
        if time.time() < t_end - 20:
            for trial in range(3):
                if time.time() > t_end - 10:
                    break
                if trial == 0:
                    x0 = local_best.copy()
                else:
                    x0 = local_best + np.random.randn(n_res) * 0.03
                    x0 = repair(x0, n_res)
                try:
                    res = slsqp_optimize(x0, n_res, dx, t0, t_end)
                    h_new = repair(res.x, n_res)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < local_c5:
                        local_c5 = c5_new
                        local_best = h_new
                except Exception:
                    pass

        # Phase 2: Nelder-Mead with penalty
        if time.time() < t_end - 10:
            n_restarts = max(1, int((t_end - time.time()) / 30))
            for ri in range(min(n_restarts, 4)):
                if time.time() > t_end - 5:
                    break
                if ri == 0:
                    x0 = local_best.copy()
                else:
                    x0 = local_best + np.random.randn(n_res) * 0.015
                    x0 = repair(x0, n_res)
                max_iter = max(50, min(400, 30000 // n_res))
                try:
                    res = minimize(penalized_obj, x0, args=(n_res, dx),
                                   method='Nelder-Mead',
                                   options={'maxiter': max_iter, 'xatol': 1e-9,
                                            'fatol': 1e-12, 'adaptive': True,
                                            'maxfev': max_iter * (n_res + 1)})
                    h_new = repair(res.x, n_res)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < local_c5:
                        local_c5 = c5_new
                        local_best = h_new
                except Exception:
                    pass

        # Phase 3: Local coordinate search
        if time.time() < t_end - 3:
            local_best, local_c5 = local_search(local_best, n_res, dx,
                                                max_rounds=60, t_limit=t_end, t0=t0)

        # Update global best
        if local_c5 < best_c5:
            best_h = local_best
            best_c5 = local_c5

    # Final repair and evaluation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = repair(best_h, n_final)
    final_c5 = compute_c5(best_h, dx_final)

    return (best_h, final_c5, n_final)
