from scipy import optimize
from scipy.interpolate import PchipInterpolator
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return float((corr * dx).max())

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h_src, n_new):
    n_src = len(h_src)
    if n_src == n_new:
        return project_feasible(h_src, n_new)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    f = PchipInterpolator(x_src, h_src)
    return project_feasible(f(x_new), n_new)

def grad_active(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
    else:
        m = -k
        if m < n:
            g[:n - m] += (1.0 - h[m:])
            g[m:] -= h[:n - m]
    return g * dx
