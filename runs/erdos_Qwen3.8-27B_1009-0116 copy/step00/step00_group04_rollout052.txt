def grad_corr(h, n):
    """Compute gradient of all correlation values w.r.t. h"""
    # Returns (2n-1, n) array: grad_c[k, m] = d c[k] / d h[m]
    grad = np.zeros((2*n-1, n))
    for k in range(2*n-1):

        # Term 1: (1-h[m-j]) for 0<=m-j<n
        idx1 = np.arange(n) - j  # m-j for each m
        mask1 = (idx1 >= 0) & (idx1 < n)

        idx2 = np.arange(n) + j  # m+j for each m

        t2 = np.zeros(n)

    return grad

        g2[mask2] = -h[idx2[mask2]]

    """Gradient of soft-max objective. cw: (2n-1,) weights summing to 1."""
    dx = 2.0 / n
