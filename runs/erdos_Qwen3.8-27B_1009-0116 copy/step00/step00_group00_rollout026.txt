def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(500):
        deficit = target - np.sum(h)
        if abs(deficit) < 1e-13:
            break
        if deficit > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = min(deficit / count, (1.0 - h[mask]).min() * 0.5)
            h[mask] += step
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = min((-deficit) / count, h[mask].min() * 0.5)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(np.clip(h_new, 0, 1), n_new)

def objective_func(h, n, alpha=80.0):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    m = np.max(corr)
    if m < 1e-10:
        return m
    terms = np.exp(alpha * (corr - m))
    return m + np.log(terms.sum()) / alpha

def run_slsqp(h0, n, maxiter=300, alpha=80.0):
    h0 = fix_sum(h0.copy(), n)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(objective_func, h0, args=(n, alpha), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
    h_opt = fix_sum(res.x, n)
    c5 = compute_c5(h_opt)
    return h_opt, c5

import numpy as np
from scipy.optimize import minimize, fmin_slsqp
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0: break
            s = min(d/c, (1.0 - h[m]).min() * 0.8)
            h[m] += s
        else:
            m = h > 0.0
            c = m.sum()
            if c == 0: break
            s = min(-d/c, h[m].min() * 0.8)
            h[m] -= s
    np.clip(h, 0, 1, out=h)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return fix_sum(h.copy(), n_new)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(np.clip(h_new, 0, 1), n_new)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    if m < 1e-15:
        return m
    e = np.exp(alpha * (c - m))
    return m + np.log(e.sum()) / alpha

def slsqp_optimize(h0, n, alpha=60, maxiter=400):
    h0 = fix_sum(h0.copy(), n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, args=(n, alpha), method='SLSQP',

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(2000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
            c = int(m.sum())
            if c == 0:
                break
            step = min(d / c, (1.0 - h[m]).min() * 0.9)
            h[m] += step
        else:
            m = h > 1e-14
            c = int(m.sum())
            if c == 0:
                break
            step = min(-d / c, h[m].min() * 0.9)
            h[m] -= step
    np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return fix_sum(h.astype(float).copy(), n_new)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(np.clip(h_new, 0.0, 1.0), n_new)

    return m + np.log(np.sum(e)) / alpha
