    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    t0 = time.time()
    try:
        res = minimize(opt_func, h_init, jac=opt_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       maxiter=maxiter, options={'ftol': ftol, 'disp': False})
    except Exception:
        return None
    if res is None:
        return None
    h_out = repair(res.x, n)
    return h_out

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 47
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0
        h0 = repair(h0, n0)

    n0 = len(h0)
    h0 = repair(h0, n0)
    best_c5 = c5_value(h0)
    best_h = h0.copy()

    # Phase 1: optimize at initial resolution
    if remaining() > 20:
        h_opt = do_optimize(h0, max_time=15, maxiter=100)
        if h_opt is not None:
            c5t = c5_value(h_opt)
            if c5t < best_c5:
                best_c5, best_h = c5t, h_opt

    # Progressive resolution increases
    plan = [(94, 20), (188, 80), (376, 250), (752, 500)]
    for n_target, max_t in plan:
        if remaining() < max_t + 20:
            break
        h_interp = interp_h(best_h, n_target)
        c5_interp = c5_value(h_interp)
        if c5_interp < best_c5:
            best_c5, best_h = c5_interp, h_interp

        # Try optimizing at this resolution
        actual_t = min(max_t, remaining() - 30)
        if actual_t > 15:
            h_opt = do_optimize(h_interp, max_time=actual_t, maxiter=200)
            if h_opt is not None:
                c5_opt = c5_value(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt

        # Try a perturbed restart at this resolution
        if remaining() > 30:
            h_pert = h_interp + rng.normal(0, 0.05, n_target)
            h_pert = repair(h_pert, n_target)
            h_opt2 = do_optimize(h_pert, max_time=min(30, remaining() - 20), maxiter=80)
            if h_opt2 is not None:
                c5_opt2 = c5_value(h_opt2)
                if c5_opt2 < best_c5:
                    best_c5, best_h = c5_opt2, h_opt2

    # Final: try a few random restarts at best resolution if time permits
    n_best = len(best_h)
    if remaining() > 40:
        for trial in range(3):
            if remaining() < 20:
                break
            h_try = best_h + rng.normal(0, 0.1, n_best)
            h_try = repair(h_try, n_best)
            h_opt = do_optimize(h_try, max_time=min(25, remaining() - 10), maxiter=60)
            if h_opt is not None:
                c5t = c5_value(h_opt)
                if c5t < best_c5:
                    best_c5, best_h = c5t, h_opt

    # Final repair and evaluation
    best_h = repair(best_h, len(best_h))
    final_c5 = c5_value(best_h)

    # Use pre-imported evaluator if available for final check
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 > final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, len(best_h))
