import numpy as np
from scipy.optimize import minimize
import time

_g_target = 0.0
_g_dx = 0.0
_g_t = 1.0


def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)


def _lse_val(x):
    dx, t = _g_dx, _g_t
    corr = np.correlate(x, 1 - x, mode='full')
    cmax = corr.max()
    return (cmax + np.log(np.exp(t * (corr - cmax)).sum()) / t) * dx


def _lse_grad(x):
    dx, t = _g_dx, _g_t
    n = len(x)
    corr = np.correlate(x, 1 - x, mode='full')
    cmax = corr.max()
    w = np.exp(t * (corr - cmax))
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            if w[m] < 1e-14:
                continue
            k = m - (n - 1)
            jp = j + k
            if 0 <= jp < n:
                s += w[m] * (1.0 - x[jp])
            jm = j - k
            if 0 <= jm < n:
                s += w[m] * (-x[jm])
        grad[j] = s * dx
    return grad


def _eq_fun(x):
    return x.sum() - _g_target


def _eq_jac(x):
    return np.ones_like(x)


def repair(x, n):
    target = n / 2.0
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if s > 1e-15:
        x *= target / s
    x = np.clip(x, 0.0, 1.0)

            break
        if diff > 0:

        else:
            mask = x > 1e-15
        if not mask.any():
            break
        adj = diff / mask.sum()
        x[mask] = np.clip(x[mask] + adj, 0.0, 1.0)
    return x

    global _g_target, _g_dx, _g_t
    _g_target = n / 2.0
    _g_dx = dx
    _g_t = t
    x0 = repair(x0, n)
    bounds = list(zip(np.zeros(n), np.ones(n)))

    try:
        res = minimize(_lse_val, x0, method='SLSQP', jac=_lse_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(res.x, n)
    except Exception:
        return x0


def coord_descent(h, n, dx, t_max):
    current_c5 = compute_c5(h, dx)
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    improved = True
    while improved and time.time() < t_max:
        improved = False
        for j in range(n):
            if time.time() > t_max:
                break
            for step in steps:
                for sign in [1, -1]:
                    x = h.copy()
                    x[j] += sign * step
                    x = repair(x, n)
                    c5 = compute_c5(x, dx)
                    if c5 < current_c5 - 1e-12:
                        h = x
                        current_c5 = c5
                        improved = True
                        break
                if improved:
                    break
    return h, current_c5
