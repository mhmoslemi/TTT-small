def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_soft(h, temperature):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Soft max
    max_corr = np.max(corr)
    shifted = corr - max_corr
    soft_max = max_corr + temperature * np.log(np.mean(np.exp(shifted / temperature)))
    return soft_max * dx

def repair(h, n_points):
    """Project h to feasible set: [0,1]^n with sum = n/2"""
    target = n_points / 2.0
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute difference
    if diff > 0:
        # Need to increase sum - add to elements < 1
        room = 1.0 - h
        if room.sum() < diff:
            return np.clip(h + diff / n_points, 0, 1)
        # Proportionally add to elements with room
        h = h + diff * room / room.sum()
    else:
        # Need to decrease sum - subtract from elements > 0
        space = h
        if space.sum() < -diff:
            return np.clip(h + diff / n_points, 0, 1)
        h = h + diff * space / space.sum()
    return np.clip(h, 0, 1)

    """Local optimization using L-BFGS-B with soft max objective"""
    target = n_points / 2.0
    
    def objective(x):
        h = repair(x, n_points)
        return compute_c5_soft(h, temperature=0.01)

        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_curr = project_feasible(h_interp, n_try)
        h_opt, c5_opt = continuous_local_search(h_curr, n_try, time_limit=40.0)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final polish
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = continuous_local_search(best_h, len(best_h), time_limit=30.0)

    # Validate
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
