        # Anneal learning rate
        lr = max(0.001, lr * 0.999)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            j = order[i]
            h[j] = np.clip(h[j] + diff, 0.0, 1.0)
            diff = target - np.sum(h)
    return h

def c5_obj(x, dx):
    return compute_c5(x, dx)

def sum_con(x, target):
    return np.sum(x) - target

def subgrad_of_corr(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        l = -lag
        if l < n:
            grad[:n - l] += (1.0 - h[l:])
            grad[l:] -= h[:n - l]
    return grad

def subgradient_descent(h, dx, n, steps, lr0):
    h = h.copy()
    lr = lr0
    target = n / 2.0
    for it in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_idx = int(np.argmax(corr))
        lag = k_idx - (n - 1)
        grad = subgrad_of_corr(h, n, lag)
        h = h - lr * dx * grad
        h = np.clip(h, 0.0, 1.0)
        h = repair_h(h, n)
        decay = 0.9995 if it < steps * 3 // 4 else 0.998
        lr = max(0.001, lr * decay)
    return h

def binarize(h, n):
    h = h.copy()
    if n % 2 == 0:
        k = n // 2
        idx = np.argsort(h)[::-1]
        h_b = np.zeros(n)
        h_b[idx[:k]] = 1.0
    else:
        k = (n - 1) // 2
        idx = np.argsort(h)[::-1]
        h_b = np.zeros(n)
        h_b[idx[:k]] = 1.0
        if k < n:
            h_b[idx[k]] = 0.5
    return h_b

def interpolate_init(n_init, h_init, n):
    if n == n_init:
        return h_init.copy()
    x_old = np.linspace(0.0, 2.0, n_init)
    x_new = np.linspace(0.0, 2.0, n)
    return np.interp(x_new, x_old, h_init)
