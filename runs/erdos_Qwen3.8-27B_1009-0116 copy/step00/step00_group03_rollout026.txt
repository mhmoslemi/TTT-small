    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

    return h.sum() - _GS


def _repair(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(len(h), ts / len(h))
    h *= ts / s

        c = m.sum()
        if c > 0:
            h[m] += d / c
            h = np.clip(h, 0.0, 1.0)
