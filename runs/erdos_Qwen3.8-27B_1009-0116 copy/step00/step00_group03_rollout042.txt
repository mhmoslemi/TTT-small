    n = len(h)
    m = j - (n - 1)  # the lag
    grad = np.zeros(n)
    for k in range(n):
        # Check if k is in I(j): 0 ≤ k < n AND 0 ≤ k+m < n
        if 0 <= k + m < n:
            grad[k] += (1.0 - h[k + m])
        # Check if k-m is in I(j): 0 ≤ k-m < n AND 0 ≤ (k-m)+m = k < n
        if 0 <= k - m < n:
            # (k-m)+m = k, which we already know is in [0,n)
            # But we also need 0 ≤ k < n (true) and 0 ≤ (k-m) < n
            grad[k] -= h[k - m]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def repair(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-14
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_component(h, j):
    n = len(h)
    m = j - (n - 1)
    grad = np.zeros(n)
    km_hi = min(n - 1, n - 1 - m)
    km_lo = max(0, -m)
    for k in range(km_lo, km_hi + 1):
        grad[k] = (1.0 - h[k + m])
    km2_hi = min(n - 1, n - 1 + m)
    km2_lo = max(0, 1 + m) if m < 0 else max(0, 1 + m)
    for k in range(n):
        if 0 <= k - m < n:
            grad[k] -= h[k - m]
    return grad

def subgradient_minimize(h, dx, target_sum, max_iters, lr_init):
    n = len(h)
    lr = lr_init
    best_h = h.copy()
    best_c5, _ = compute_c5(h, dx)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        grad = grad_corr_component(h, k_max)
        grad += 0.5 * (np.sum(h) - target_sum) / n
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = repair(h, target_sum)
        c5, _ = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 200 == 199:
            lr *= 0.97
    return best_h, best_c5

def slsqp_minimize(h, dx, target_sum, maxiter=300):
    n = len(h)
    def obj(x):
        return float(np.max(np.correlate(x, 1.0 - x, mode='full')))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = repair(res.x, target_sum)
        return h_new
    except Exception:
        return h

def coordinate_search(h, dx, target_sum, n_sweeps=5):
    n = len(h)
    best_c5, _ = compute_c5(h, dx)
    for sweep in range(n_sweeps):
        for i in range(n):
            orig = h[i]
            best_c5_i = best_c5
            best_val = orig
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                nv = orig + delta

                    best_val = nv
            h[i] = best_val
            h = repair(h, target_sum)
            best_c5 = best_c5_i
