from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return dx * np.max(np.correlate(h, 1.0 - h, mode='full'))


def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(200):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0:
                break
            h[m] = np.minimum(1.0, h[m] + d / c)
        else:
            m = h > 0.0
            c = m.sum()
            if c == 0:
                break
            h[m] = np.maximum(0.0, h[m] + d / c)
    return h


def _soft_max_obj(h, n, beta, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if beta < 1:
        return m
    return m + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - m))))


def _slsqp_optimize(h_start, n, beta, maxiter, target, bounds):
    def obj(h):
        return _soft_max_obj(h, n, beta, target)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return repair(res.x if res.x is not None else h_start, n)
