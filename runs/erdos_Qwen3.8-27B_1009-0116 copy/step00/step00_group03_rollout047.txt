                                if cn < cc - 1e-12:
                                    cc = cn
                                    improved = True
                                else:
                                    hc[i] += d
                                    hc[j] -= d
                            elif hc[j] >= d and hc[i] <= 1.0 - d:
                                hc[j] -= d
                                hc[i] += d
                                cn = compute_c5(hc, dx)
                                if cn < cc - 1e-12:
                                    cc = cn
                                    improved = True
                                else:
                                    hc[j] += d
                                    hc[i] -= d
                if not improved:
                    break
            if cc < best_c5:
                h_best, best_c5 = hc, cc

        if best_c5 < c5_prev - 1e-10:
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 15:
            h_best = repair_h(h_best + 0.3 * rng.randn(n), n)
            best_c5 = compute_c5(h_best, dx)
            stagnation = 0

    if time.time() - t0 < budget_s - 5:
        try:
            res = minimize(obj_func, h_best, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 1000, 'ftol': 1e-14})
            ht = repair_h(res.x, n)
            ct = compute_c5(ht, dx)
            if ct < best_c5:
                h_best, best_c5 = ht, ct
        except Exception:
            pass

    h_best = repair_h(h_best, n)
    best_c5 = compute_c5(h_best, dx)
    return h_best, best_c5, n
