import numpy as np
from scipy.optimize import minimize
import time

_PARAMS = {'n': 42, 'alpha': 100.0}

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _objective(h):
    n = _PARAMS['n']
    alpha = _PARAMS['alpha']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if alpha <= 1.0:
        return m
    scaled = alpha * (corr - m)
    scaled = np.clip(scaled, -100, 100)
    return m + np.log(np.sum(np.exp(scaled))) / alpha

def _eq_fun(h):
    return np.sum(h) - _PARAMS['n'] / 2.0

def _eq_jac(h):
    return np.ones(_PARAMS['n'])

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return repair(h_new, n_new)

def _run_slsqp(h0, n, alpha, maxiter, ftol):
    _PARAMS['n'] = n
    _PARAMS['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_objective, h0.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return repair(res.x, n)
    except Exception:
        return h0.copy()

def _run_nelder(h0, n, maxiter):
    _PARAMS['n'] = n
    _PARAMS['alpha'] = 1.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'ineq', 'fun': _ineq_low, 'args': (),
             'args': None, 'jac': None}]
    # Use penalty for bounds and equality in NM
    try:
        res = minimize(_nm_objective, h0.copy(), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'fatol': 1e-10,
                                'xatol': 1e-10, 'adaptive': True})
        return repair(res.x, n)
    except Exception:
        return h0.copy()

def _nm_objective(h):
    val = compute_c5(h, len(h))
    # Penalty for bounds violation
    bp = np.sum(np.clip(-h, 0, None)**2) + np.sum(np.clip(h-1, 0, None)**2)
    # Penalty for equality violation
    ep = (np.sum(h) - len(h)/2.0)**2 * 10.0
    return val + bp + ep

def _ineq_low(h):
    return np.min(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    # Get initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(42, 0.5)

    n = len(h_init)
    h_init = repair(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, n)
    best_n = n

    # Phase 1: SLSQP with increasing alpha (smooth max -> true max)
    for alpha in [5.0, 20.0, 50.0, 150.0, 500.0, 1000.0]:
        if time.time() - t0 > budget_s * 0.45:
            break
        h_sol = _run_slsqp(best_h, n, alpha, 800, 1e-14)
        c5_sol = compute_c5(h_sol, n)
        if c5_sol < best_c5 - 1e-12:
            best_c5 = c5_sol
            best_h = h_sol.copy()
            best_n = n

    # Phase 2: Multiple perturbed restarts
    if time.time() - t0 < budget_s * 0.65:
        for trial in range(8):
            if time.time() - t0 > budget_s * 0.65:
                break
            scale = 0.05 / (trial + 1)
            h_try = best_h + np.random.normal(0, scale, n)
            h_try = repair(h_try, n)
            h_sol = _run_slsqp(h_try, n, 200.0, 400, 1e-13)
            c5_sol = compute_c5(h_sol, n)
            if c5_sol < best_c5 - 1e-12:
                best_c5 = c5_sol
                best_h = h_sol.copy()
                best_n = n

    # Phase 3: Nelder-Mead for potential local-minimum escape
    if time.time() - t0 < budget_s * 0.75:
        h_try = best_h + np.random.normal(0, 0.02, n)
        h_try = repair(h_try, n)
        h_sol = _run_nelder(h_try, n, 3000)
        c5_sol = compute_c5(h_sol, n)
        if c5_sol < best_c5 - 1e-12:
            best_c5 = c5_sol
            best_h = h_sol.copy()
            best_n = n
            # Refine with SLSQP
            h_sol = _run_slsqp(best_h, n, 500.0, 500, 1e-14)
            c5_sol = compute_c5(h_sol, n)
            if c5_sol < best_c5 - 1e-12:
                best_c5 = c5_sol
                best_h = h_sol.copy()

    # Phase 4: Upsample to higher resolution and optimize
    if time.time() - t0 < budget_s * 0.80:
        n2 = n * 2
        h_up = upsample(best_h, n2)
        c5_up = compute_c5(h_up, n2)
        if c5_up < best_c5 + 0.02:
            for alpha in [50.0, 200.0, 500.0]:
                if time.time() - t0 > budget_s * 0.90:
                    break
                h_sol = _run_slsqp(h_up, n2, alpha, 500, 1e-13)
                c5_sol = compute_c5(h_sol, n2)
                if c5_sol < best_c5 - 1e-12:
                    best_c5 = c5_sol
                    best_h = h_sol.copy()
                    best_n = n2
                    h_up = best_h.copy()

    # Phase 5: Final polish with direct max at original resolution
    if time.time() - t0 < budget_s * 0.95:
        _PARAMS['n'] = best_n
        _PARAMS['alpha'] = 1.0
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        try:
            res = minimize(_objective, best_h.copy(), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-15})
            h_sol = repair(res.x, best_n)
            c5_sol = compute_c5(h_sol, best_n)
            if c5_sol < best_c5 - 1e-12:
                best_c5 = c5_sol
                best_h = h_sol.copy()
        except Exception:
            pass

    # Final validation and repair
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    # Optional: evaluate with provided evaluator
    try:
        if 'evaluate_erdos_solution' in globals():
            eval_result = evaluate_erdos_solution(best_h)
            if eval_result is not None and np.isfinite(eval_result):
                if float(eval_result) < best_c5:
                    best_c5 = float(eval_result)
    except Exception:
        pass

    return best_h, best_c5, best_n
