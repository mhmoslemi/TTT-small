import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_exact(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-15
            if np.sum(room[mask]) > 1e-15:
                h[mask] += diff / np.sum(room[mask])
        else:
            mask = h > 1e-15
            if np.sum(h[mask]) > 1e-15:
                h[mask] += diff / np.sum(h[mask])
        np.clip(h, 0.0, 1.0, out=h)
    return h

    return compute_c5(project_exact(x, n))
