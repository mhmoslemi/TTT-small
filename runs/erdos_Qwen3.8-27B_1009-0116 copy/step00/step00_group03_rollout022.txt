    """Smooth optimization using L-BFGS-B."""
    n = len(h0)
    dx = 2.0 / n
    
    def objective(params):
        h = 1.0 / (1.0 + np.exp(-params))  # sigmoid

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / n)

def local_search_best(h, time_limit=30):
    """Best-improvement local search for binary h."""
    n = len(h)
    h = (h > 0.5).astype(float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = float(np.max(corr))
    dx = 2.0 / n
    
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improve = 0
        best_a, best_b = -1, -1
        best_corr = None
        
        for a in ones:
            for b in zeros:
                h[a] = 0.0
                h[b] = 1.0
                new_corr = np.correlate(h, 1.0 - h, mode="full")
                nm = float(np.max(new_corr))
                improve = cur_max - nm
                if improve > best_improve + 1e-12:
                    best_improve = improve
                    best_a, best_b = a, b
                    best_corr = new_corr
                h[a] = 1.0
                h[b] = 0.0
        
        if best_a < 0:
            break  # local optimum
        
        h[best_a] = 0.0
        h[best_b] = 1.0
        corr = best_corr
        cur_max = float(np.max(best_corr))
    
    return h, cur_max * dx

def simulated_annealing(h, time_limit=60, seed=42):
    """Simulated annealing for binary h."""
    rng = np.random.RandomState(seed)
    n = len(h)
    h = (h > 0.5).astype(float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    
    start = time.time()
    T = 0.1
    cool = 0.9995
    
    while time.time() - start < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(np.max(new_corr))
        
        if new_max < cur_max or rng.rand() < np.exp((cur_max - new_max) / T):
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
        
        T *= cool
        if T < 0.005:
            T = 0.1
            # Perturb

                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a], h[b] = 0.0, 1.0
    
    return best_h, best_max * dx

def smooth_optimize(h0, time_limit=30):
    """Smooth optimization on [0,1] variables, then round."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy().astype(float)
    
    def obj_and_grad(x):
        h = 1.0 / (1.0 + np.exp(-np.clip(x, -30, 30)))
        # Project to sum = n/2
        s = h.sum()
        h = h * (n / 2.0) / s
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Log-sum-exp
        beta = 20.0
        m = np.max(corr)
        val = (m + np.log(np.sum(np.exp(beta * corr - beta * m))) / beta) * dx
        return val
    
    x0 = np.log(np.clip(h, 0.01, 0.99) / (1 - np.clip(h, 0.01, 0.99)))
    
    result = minimize(obj_and_grad, x0, method='L-BFGS-B',
                      options={'maxiter': 500, 'gtol': 1e-6})
    
    h_opt = 1.0 / (1.0 + np.exp(-np.clip(result.x, -30, 30)))
    h_opt = h_opt * (n / 2.0) / h_opt.sum()
    h_opt = np.clip(h_opt, 0, 1)
    
    # Round to binary preserving sum
    h_bin = (h_opt > 0.5).astype(float)
    diff = n / 2.0 - h_bin.sum()
    if diff > 0:
        # Need more 1s: flip largest 0→1
        order = np.argsort(-h_opt[np.where(h_bin < 0.5)[0]])
        idx = np.where(h_bin < 0.5)[0][order[:int(diff)]]
        h_bin[idx] = 1.0
    elif diff < 0:
        order = np.argsort(h_opt[np.where(h_bin > 0.5)[0]])
        idx = np.where(h_bin > 0.5)[0][order[:abs(int(diff))]]
        h_bin[idx] = 0.0
    
    return h_bin
