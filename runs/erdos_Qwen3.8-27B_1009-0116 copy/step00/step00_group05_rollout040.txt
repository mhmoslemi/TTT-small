import numpy as np

import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-15)[0]
        else:
            free = np.where(h > 1e-15)[0]
        if len(free) == 0:
            break
        adj = diff / len(free)
        h[free] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_objective(h, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    S = cmax + mu * np.log(np.exp(np.sum((corr - cmax) / mu) - 1.0) + 1.0)
    S = cmax + mu * np.log(np.sum(np.exp((corr - cmax) / mu)))
