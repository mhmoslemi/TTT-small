import numpy as np
from scipy.optimize import minimize
import time

_g_target = 35.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def constraint_fn(x):
    return float(np.sum(x) - _g_target)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * (n / 2.0) / s
    h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h_bin, n, deadline, rng):
    target = n / 2.0
    for _ in range(500):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        current_c5 = compute_c5(h_bin, n)
        best_delta = 0.0
        best_pair = None
        # Sample swaps if too many
        pairs = [(i, j) for i in ones for j in zeros]
        if len(pairs) > 800:
            idx = rng.choice(len(pairs), 800, replace=False)
            pairs = [pairs[k] for k in idx]
        for i, j in pairs:
            h_bin[i], h_bin[j] = 0.0, 1.0
            c5_new = compute_c5(h_bin, n)
            h_bin[i], h_bin[j] = 1.0, 0.0
            delta = c5_new - current_c5
            if delta < best_delta - 1e-12:
                best_delta = delta
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h_bin[i], h_bin[j] = 0.0, 1.0
    return h_bin

def to_binary(h, n):
    order = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[order[-int(n // 2):]] = 1.0
    return h_bin

    rng = np.random.default_rng(seed)
    start_time = time.time()
