        corr = np.correlate(h, omh, mode="full") * dx
        max_val = np.max(corr)
        # Find active lags (within tolerance)
        active = np.where(corr >= max_val - 1e-8)[0]
        # Average gradient over active lags
        grad = np.zeros(n)
        for a in active:
            grad += lag_grad(h, n, a)
        grad /= len(active)
        # Adaptive step size
        lr = lr0 / (1.0 + it * 0.01)
        h_new = proj(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
        else:
            h = proj(h + 0.5 * lr * grad, n)  # restore
    return best_h, best_c5
