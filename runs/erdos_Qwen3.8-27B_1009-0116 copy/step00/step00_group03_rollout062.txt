    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.random.permutation(len(h))
        for i in idx:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                d = min(1.0 - h[i], diff)
            else:
                d = max(-h[i], diff)
            h[i] += d
            diff -= d
    return h
