    return float(np.max(corr) * dx)

    return float(np.sum(np.abs(corr)**p) ** (1.0/p) * dx)

        return time.time() - t0

        h = h.copy()
        c5 = compute_c5(h, dx)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def binarize(h, rng):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        to_rm = rng.choice(idx, n_ones - target, replace=False)
        h_bin[to_rm] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        to_add = rng.choice(idx, target - n_ones, replace=False)
        h_bin[to_add] = 1
    return h_bin

def local_search_binary(h, dx, rng, max_iter=200000, dt=None):
    h = h.astype(float).copy()
    n = len(h)
    c5 = compute_c5(h, dx)
    stale = 0
    for it in range(max_iter):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, dx)
        if new_c5 <= c5:
            c5 = new_c5
            stale = 0
        else:
            delta = new_c5 - c5
            if dt is not None and np.log(rng.random()) < -delta / dt:
                c5 = new_c5
                stale = 0
            else:
                h[i] = 1.0
                h[j] = 0.0
                stale += 1
        if stale > 30000:
            break
    return h, c5

def systematic_search(h, dx, rng, max_passes=50):
    h = h.astype(float).copy()
    n = len(h)
    c5 = compute_c5(h, dx)
    for p in range(max_passes):
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            if h[i] != 1:
                continue
            for j in zeros:
                if h[j] != 0:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h, dx)
                if new_c5 < c5:
                    c5 = new_c5
                    improved = True
                    zeros.append(i)
                    ones.remove(j)
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            break
    return h, c5

def continuous_opt(h_start, n, dx, rng, max_time=45):
    h0 = h_start.astype(float).copy()
    h0 = np.clip(h0, 0.01, 0.99)
    s = np.sum(h0)
    h0 = h0 * (n / 2.0) / s
    h0 = np.clip(h0, 0.01, 0.99)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return float(np.max(corr) * dx)
    
    best_h = h0.copy()
    best_val = obj(h0)
    t_start = time.time()
    
    for p in [2, 4, 8, 16, 32, 64]:
        if time.time() - t_start > max_time:
            break
        def obj_p(h):
            corr = np.correlate(h, 1 - h, mode="full")
            return float(np.sum(corr**p) ** (1.0 / p) * dx)
        
        for attempt in range(3):
            if time.time() - t_start > max_time:
                break
            res = minimize(obj_p, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-10})
            if res.x is not None:
                res_x = np.clip(res.x, 0, 1)
                s_val = np.sum(res_x)
                if s_val > 0:
                    res_x = res_x * (n / 2.0) / s_val
                    res_x = np.clip(res_x, 0, 1)
                    v = obj(res_x)
                    if v < best_val:
                        best_val = v
                        best_h = res_x.copy()
    
    return best_h, best_val

def make_structured(n, rng, seed_offset=0):
    """Create various structured sequences as starting points."""
    s = rng.seed(seed_offset)
    patterns = []
    
    # Quadratic residue-like
    h_qr = np.zeros(n)
    for i in range(n):
        h_qr[i] = 1.0 if (int(i * 0.5) ** 2) % 7 < 3 else 0.0
    patterns.append(binarize(h_qr, rng))
    
    # Block structure with varying block sizes
    h_blk = np.zeros(n)
    pos = 0
    while pos < n:
        blen = rng.randint(3, 8)
        h_blk[pos:pos + blen] = 1.0
        pos += blen
        blen2 = rng.randint(3, 8)
        pos += blen2
    patterns.append(binarize(h_blk, rng))
    
    # Low-discrepancy (Halton-like)
    h_ld = np.zeros(n)
    for i in range(n):
        val = (i * 0.6180339887) % 1.0

    h_2s = np.zeros(n)
    for i in range(n):
        h_2s[i] = 1.0 if ((i * 3) % 7) < 3 else 0.0
    patterns.append(binarize(h_2s, rng))
    
    return patterns
