def compute_gradient(h, n, T):
    dx = 2.0 / n
    # Compute all correlations
    # C[k] for k = -(n-1), ..., 0, ..., n-1
    # C[k] = dx * sum_i h[i] * (1-h[i+k]) for valid i
    
    # Using np.correlate
    corr = np.correlate(h, 1-h, mode="full") * dx  # length 2n-1
    # corr[n-1+k] = C[k] for shift k
    
    # Softmax weights
    weights = np.exp(corr / T)
    weights /= weights.sum()
    
    # Gradient: weighted sum of individual gradients
    # dC[k]/dh[j] = dx * [(1-h[j+k]) if 0<=j+k<n] - dx * [h[j-k] if 0<=j-k<n]
    # 
    # For shift k, the gradient vector is:
    # grad_k[j] = dx * [(1-h[j+k]) * 1_{0<=j+k<n} - h[j-k] * 1_{0<=j-k<n}]
    #
    # The j+k constraint: for shift k, valid j are those with 0<=j<n and 0<=j+k<n
    # The j-k constraint: valid j are those with 0<=j<n and 0<=j-k<n (= k<=j<n)
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        if weights[n-1+k] < 1e-15:
            continue
        w = weights[n-1+k] * dx
        
        # First term: (1-h[j+k]) for j where 0<=j+k<n
        if k >= 0:
            # j ranges from 0 to n-1-k
            js = np.arange(0, n-k)
            grad[js] += w * (1 - h[js+k])
            # Second term: -h[j-k] for j where 0<=j-k<n, i.e., j from k to n-1
            js2 = np.arange(k, n)
            grad[js2] -= w * h[js2-k]
        else:
            # k < 0, let k_neg = -k
            k_neg = -k
            # j+k = j-k_neg, valid when 0<=j-k_neg<n, i.e., j from k_neg to n-1
            js = np.arange(k_neg, n)
            grad[js] += w * (1 - h[js-k_neg])
            # Second term: -h[j-k] = -h[j+k_neg], valid when 0<=j+k_neg<n, i.e., j from 0 to n-1-k_neg
            js2 = np.arange(0, n-k_neg)
            grad[js2] -= w * h[js2+k_neg]
    
    return grad, corr, np.max(corr)

import numpy as np

def project_h(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_new = np.clip(h + mid, 0.0, 1.0)
        if h_new.sum() < target_sum:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    diff = target_sum - h_new.sum()
    if abs(diff) > 1e-10:
        h_new += diff / len(h_new)
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def compute_grad_smoothed(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Softmax weights
    shifted = corr - np.max(corr)
    weights = np.exp(shifted / T)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k_shift in range(-(n-1), n):
        w = weights[n-1+k_shift]
        if w < 1e-15:
            continue
        wd = w * dx
        kk = k_shift
        # First term: (1-h[j+kk]) for valid j
        if kk >= 0:
            js = slice(0, n - kk)
            grad[:n-kk] += wd * (1.0 - h[kk:])
            # Second term: -h[j-kk] for j from kk to n-1
            grad[kk:] -= wd * h[:n-kk]
        else:
            kk_neg = -kk
            js = slice(kk_neg, n)
            grad[kk_neg:] += wd * (1.0 - h[:n-kk_neg])
            grad[:n-kk_neg] -= wd * h[kk_neg:]
    
    return grad, corr

    s = h.sum()

        return h
    lo, hi = -1.0, 1.0

        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def grad_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    w = np.exp((corr - max_c) / T)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):

            grad[:m] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:m]
        else:
            kn = -k
            m = n - kn
            grad[kn:] += wk * (1.0 - h[:m])
            grad[:m] -= wk * h[kn:]
    return grad, max_c

        T_init = max(0.015, 0.06 / (fac ** 0.5))
        lr_init = max(0.02, 0.12 / (fac ** 0.5))

        h_trial, c5_trial = optimize_nest(h_interp, n_new * 0.5, pb, rng, T_init, lr_init)

        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial
            n_cur = n_new

        remaining = budget_s - (time.time() - t0)

    # Final polish at best resolution found
    t_left = budget_s - (time.time() - t0)
    if t_left > 10.0:
        hf = best_h.copy()
        target_f = len(hf) * 0.5
        hf, cf = optimize_nest(hf, target_f, t_left * 0.6, rng, 0.01, 0.03, mu_init=0.5)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    # Ensure final constraints
    target_final = len(best_h) * 0.5
    best_h = project_h(best_h, target_final)
    best_c5 = eval_c5(best_h)

    try:
        v = evaluate_erdos_solution(best_h)
        if v is not None:
            best_c5 = min(best_c5, float(v))
    except Exception:
        pass

    return (best_h, best_c5, len(best_h))
