import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_feasible(h, n):
    """Project h onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection for lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fine-tune
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        # Distribute difference among non-bound variables
        mask = (h_proj > 1e-8) & (h_proj < 1 - 1e-8)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

    c_max = np.max(c)

        val, grad = obj_and_grad(h_clipped, dx, t)
        diff = h_clipped.sum() - target
        val += 0.5 * penalty * diff * diff
        grad += penalty * diff * np.ones(n)
        return val, grad

    h_opt = np.clip(res.x, 0.0, 1.0)

    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    if abs(diff) > 1e-12:
        mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_p[mask] += diff / mask.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

    diff = h.sum() - n / 2.0
    val += 0.5 * penalty * diff * diff
    grad += penalty * diff
