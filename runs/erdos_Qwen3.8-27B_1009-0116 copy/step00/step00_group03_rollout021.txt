For n=70: O(35*35) = 1225 evaluations per pass, each taking ~10μs, so ~12ms per pass. Can do thousands of passes per second.

Let me implement both: random walk when stuck, steepest descent for refinement.

```python
def steepest_descent_pass(h, dx):
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_c5 = compute_c5(h, dx)
    best_i, best_j = None, None
    best_new_c5 = best_c5
    
    for i in ones:
        for j in zeros:
            h[i] = 0
            h[j] = 1
            c5 = compute_c5(h, dx)
            if c5 < best_new_c5:
                best_new_c5 = c5
                best_i, best_j = i, j
            h[i] = 1
            h[j] = 0
    
    if best_i is not None:
        h[best_i] = 0
        h[best_j] = 1
        return True, best_new_c5
    return False, best_c5
```

This is O(n² * correlate_cost) per pass. For n=70: 1225 * 10μs = 12ms. Very fast.

import numpy as np
import time

def compute_c5(h, dx):
    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cor)) * dx

def steepest_pass(h, dx):
    n = len(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    best_c5 = compute_c5(h, dx)
    best_improve = None
    best_c5_new = best_c5
    
    for i in ones:
        for j in zeros:
            h[i] = 0
            h[j] = 1
            c5 = compute_c5(h, dx)
            if c5 < best_c5_new - 1e-12:
                best_c5_new = c5
                best_improve = (i, j)
            h[i] = 1
            h[j] = 0
    
    if best_improve is not None:
        h[best_improve[0]] = 0
        h[best_improve[1]] = 1
    return best_improve is not None, best_c5_new

def local_search(h, dx, time_limit, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    t0 = time.time()
    no_improve = 0
    
    while time.time() - t0 < time_limit:
        # Try steepest descent pass
        improved, c5 = steepest_pass(h, dx)
        if improved:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
