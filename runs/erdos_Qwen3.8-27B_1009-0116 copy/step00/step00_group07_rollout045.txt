import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def fix_sum_binary(h, rng):
    n = len(h)
    target = n // 2
    s = int(h.sum())
    while s > target:
        ones = np.where(h == 1)[0]
        idx = rng.randint(len(ones))
        h[ones[idx]] = 0.0
        s -= 1
    while s < target:
        zeros = np.where(h == 0)[0]
        idx = rng.randint(len(zeros))
        h[zeros[idx]] = 1.0
        s += 1
    return h

def hill_climb(h, max_iter=200, max_time=30):
    n = len(h)
    t_start = time.time()
    c5 = compute_c5(h)
    for _ in range(max_iter):
        if time.time() - t_start > max_time:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_val = c5
        best_pair = None
        for i in ones:
            for j in zeros:
                h[j] = 1.0
                h[i] = 0.0
                val = compute_c5(h)
                if val < best_val:
                    best_val = val
                    best_pair = (i, j)
                h[j] = 0.0
                h[i] = 1.0
        if best_pair is None:
            break
        i, j = best_pair
        h[j] = 1.0
        h[i] = 0.0
        c5 = best_val
    return h, c5

def _sum_con(x):
    return np.sum(x) - x.size / 2.0

def _obj(x):
    return compute_c5(x)

    t_start = time.time()
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_con}]
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    for attempt in range(8):

            x0[mask] += diff / mask.sum()
            x0 = np.clip(x0, 0, 1)
        res = minimize(_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': 300, 'ftol': 1e-11})
        xr = np.clip(res.x, 0, 1)

                xr = np.clip(xr, 0, 1)
        c5r = compute_c5(xr)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = xr.copy()
    return best_h, best_c5
