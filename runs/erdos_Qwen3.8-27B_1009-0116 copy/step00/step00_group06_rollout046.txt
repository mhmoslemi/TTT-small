import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def c5_obj(x):
    return eval_c5(x)


def c5_smooth_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    exp_arg = np.exp(corr - m)
    return m + np.log(np.sum(exp_arg)) / 50.0


def sum_eq(x):
    return np.sum(x) - len(x) / 2.0


def sum_eq_jac(x):
    return np.ones(len(x))


def make_bounds(n):
    return [(0.0, 1.0)] * n


def make_constraints():
    return [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]


def make_smooth_constraints():
    return [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]


def repair(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    if s > 1e-12:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Final nudge for rounding
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h


def slsqp_solve(x0, n, maxiter):
    bounds = make_bounds(n)
    constraints = make_constraints()
    res = minimize(c5_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    return res.x, res.fun


def slsqp_smooth(x0, n, maxiter):
    bounds = make_bounds(n)
    constraints = make_smooth_constraints()
    res = minimize(c5_smooth_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    return res.x, eval_c5(res.x)


def coordinate_refine(h, n, max_passes=5):
    """Simple coordinate descent maintaining sum constraint."""
    dx = 2.0 / n
    best = h.copy()
    best_val = eval_c5(best)
    for _ in range(max_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n, max(1, n // 40)):
                if i == j:
                    continue
                si = best[i]
                sj = best[j]
                for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
                    ni = si + delta
                    nj = sj - delta
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        best[i], best[j] = ni, nj
                        v = eval_c5(best)
                        if v < best_val - 1e-12:
                            best_val = v
                            improved = True
                        else:
                            best[i], best[j] = si, sj
        if not improved:
            break
    return best, best_val
