import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum_binary(h, n, rng):
    h_b = (h >= 0.5).astype(float)
    target = n // 2
    num_ones = int(np.sum(h_b))
    if num_ones > target:
        ones = np.where(h_b == 1)[0]
        extra = num_ones - target
        if extra > 0:
            to_flip = rng.choice(ones, extra, replace=False)
            h_b[to_flip] = 0.0
    elif num_ones < target:
        zeros = np.where(h_b == 0)[0]
        need = target - num_ones
        if need > 0:
            to_flip = rng.choice(zeros, need, replace=False)
            h_b[to_flip] = 1.0
    return h_b

def binary_steepest(h, n, max_iters, t0, budget_frac):
    best_c5 = compute_c5(h)
    h = h.copy()
    for it in range(max_iters):
        if time.time() - t0 > budget_frac * 1100:
            break
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-9
        best_move = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h)
                delta = best_c5 - c
                if delta > best_delta:
                    best_delta = delta
                    best_move = (int(i), int(j))
                h[i] = 1.0
                h[j] = 0.0
        if best_move is None:
            break
        i, j = best_move
        h[i] = 0.0
        h[j] = 1.0
        best_c5 = compute_c5(h)
    return h, best_c5

def continuous_sa(h, n, t0, time_limit, rng):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    h_curr = h.copy()
    c_curr = best_c5
    T = 0.015
    T_min = 0.001
    step = 0.12
    no_improve = 0
    while time.time() < time_limit:
        h_new = h_curr.copy()
        if rng.random() < 0.35:
            ones = np.where(h_new >= 0.7)[0]
            zeros = np.where(h_new <= 0.3)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                h_new[i] = 0.0
                h_new[j] = 1.0
            else:
                i, j = rng.choice(n, 2, replace=False)
                d = step * rng.randn()
                h_new[i] += d
                h_new[j] -= d
        else:
            i, j = rng.choice(n, 2, replace=False)
            d = step * rng.randn()
            h_new[i] += d
            h_new[j] -= d
        h_new = np.clip(h_new, 0.0, 1.0)
        resid = n / 2.0 - np.sum(h_new)
        if abs(resid) > 1e-12:
            h_new -= resid / n
            h_new = np.clip(h_new, 0.0, 1.0)
        c_new = compute_c5(h_new)
        if c_new < c_curr or rng.random() < np.exp(-(c_new - c_curr) / T):
            h_curr = h_new
            c_curr = c_new
        if c_curr < best_c5:
            best_c5 = c_curr
            best_h = h_curr.copy()
            no_improve = 0
        else:
            no_improve += 1
        T = max(T * 0.9997, T_min)
        if no_improve > 8000:
            step = max(0.05, step * 0.95)
            if step < 0.02:
                step = 0.12
            no_improve = 0
    return best_h, best_c5

    # Phase 1: Binary steepest descent from parent
    h_bin = fix_sum_binary(start_h, n, rng)
    c_bin = compute_c5(h_bin)
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = h_bin.copy()

    h_opt, c_opt = binary_steepest(best_h, n, max_iters=100, t0=t0, budget_frac=0.35)
    if c_opt < best_c5:
        best_c5 = c_opt
        best_h = h_opt.copy()

    # Try complement
    comp_h = 1.0 - h_opt
    c_comp = compute_c5(comp_h)
    if c_comp < best_c5:
        best_c5 = c_comp
        best_h = comp_h.copy()

    # Phase 2: Continuous SA refinement (long phase)
    time_limit = t0 + budget_s * 0.93
    h_ref, c_ref = continuous_sa(best_h, n, t0, time_limit, rng)
    if c_ref < best_c5:
        best_c5 = c_ref
        best_h = h_ref.copy()

    # Phase 3: Another round of binary + continuous from best
    if time.time() < t0 + budget_s * 0.95:
        h_bin2 = fix_sum_binary(best_h, n, rng)
        c_bin2 = compute_c5(h_bin2)
        if c_bin2 < best_c5:
            best_c5 = c_bin2
            best_h = h_bin2.copy()
        h_opt2, c_opt2 = binary_steepest(best_h, n, max_iters=50, t0=t0, budget_frac=0.85)
        if c_opt2 < best_c5:
            best_c5 = c_opt2
            best_h = h_opt2.copy()
        final_limit = min(t0 + budget_s * 0.97, time.time() + 0.5)
        h_ref2, c_ref2 = continuous_sa(best_h, n, t0, final_limit, rng)
        if c_ref2 < best_c5:
            best_c5 = c_ref2
            best_h = h_ref2.copy()

    # Final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    resid = target_sum - np.sum(best_h)
    if abs(resid) > 1e-10:
        best_h -= resid / n
        best_h = np.clip(best_h, 0.0, 1.0)

    c5_final = compute_c5(best_h)
    return (best_h, c5_final, n)
