def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    top_m = np.argsort(c)[-K:]
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    def _af(x):
        cc = _cv(x)
        return float(np.mean(cc[top_m]))
    def _ag(x):
        wts = np.zeros(2*n-1)
        wts[top_m] = 1.0 / K
        return _grad_from_w(x, wts, n)
    try:
        res = minimize(_af, h, jac=_ag, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    def _el():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP p-norm with increasing p
    for p in [10, 30, 100, 200, 400]:
        if _el() > budget_s - 350:
            break
        h_new = _slsqp(best_h, p, 'p', 4000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 2: SLSQP soft-max with increasing alpha
    for alpha in [50000, 200000, 800000, 3000000]:
        if _el() > budget_s - 250:
            break
        h_new = _slsqp(best_h, alpha, 's', 4000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 3: Active-set refinement
    if _el() < budget_s - 180:
        for _rd in range(40):
            if _el() > budget_s - 180:
                break
            for K in [3, 5, 8]:
                h_new = _active_refine(best_h, n, K, 1500)
                c5n = _eval(h_new)
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h_new

    # Phase 4: Directed random pair moves targeting max lag
    if _el() < budget_s - 100:
        h = best_h.copy()
        c5c = _eval(h)
        for step in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]:
            if _el() > budget_s - 100:
                break
            for _it in range(30000):
                if _el() > budget_s - 100:
                    break
                c = _cv(h)
                kmax = int(np.argmax(c))
                L = kmax - (n - 1)
                # Bias towards indices in the max-lag support
                if L >= 0:
                    support = list(range(max(0, L-3), min(n, n-L+3)))
                else:
                    support = list(range(max(0, -L-3), min(n, n+L+3)))
                if rng.rand() < 0.7 and len(support) > 1:
                    i = support[rng.randint(len(support))]
                    j = support[rng.randint(len(support))]
                else:
                    i, j = rng.randint(n, 2)
                if i == j:
                    continue
                d = step * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c:
                        c5c = c5n
                        if c5n < best_c5:
                            best_c5 = c5n
                            best_h = h.copy()
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d

    # Phase 5: Multi-start exploration
    if _el() < budget_s - 50:
        for _r in range(25):
            if _el() > budget_s - 50:
                break
            sc = 0.001 + 0.012 * rng.rand()
            hp = _repair(best_h + sc * rng.randn(n), n)
            h_new = _slsqp(hp, 200, 'p', 2500, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

    # Phase 6: Final SLSQP polish
    if _el() < budget_s - 15:
        h_new = _slsqp(best_h, 5000, 'p', 6000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
        h_new = _slsqp(best_h, 5000000, 's', 3000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 7: Ultra-fine random moves
    if _el() < budget_s - 3:
        h = best_h.copy()
        c5c = _eval(h)
        scale = 2e-5
        stall = 0
        for _it in range(200000):
            if _el() > budget_s - 3:
                break
            i, j = rng.randint(n, 2)
            if i == j:
                continue
            d = rng.choice([1.0, -1.0]) * abs(rng.randn()) * scale
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    stall = 0
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
                    stall += 1
                    if stall > 3000:
                        scale *= 0.5
                        stall = 0
                        if scale < 1e-8:
                            break

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
