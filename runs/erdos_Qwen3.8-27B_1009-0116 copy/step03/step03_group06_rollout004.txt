    hb = np.where(h >= 0.5, 1.0, 0.0)
    hb = _repair(hb, n)
    best_c5 = _c5(hb, dx)
    best_h = hb.copy()
    cur = hb.copy()
    cur_c5 = best_c5
    T = T0
    # Pre-compute indices for speed
    while time.time() < deadline and T > 0.00003:
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Sometimes do multi-bit swap
        if rng.random() < 0.15 and len(ones) >= 3 and len(zeros) >= 3:
            k = rng.randint(2, 5)
            if k > len(ones) or k > len(zeros):
                k = 2
            oi = rng.choice(ones, size=k, replace=False)
            zi = rng.choice(zeros, size=k, replace=False)
            cur[oi] = 0.0
            cur[zi] = 1.0
            nc = _c5(cur, dx)
            d = nc - cur_c5
            if d < 0 or rng.random() < np.exp(-d / T):
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = cur.copy()
            else:
                cur[oi] = 1.0
                cur[zi] = 0.0
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            cur[i], cur[j] = 0.0, 1.0
            nc = _c5(cur, dx)
            d = nc - cur_c5
            if d < 0 or rng.random() < np.exp(-d / T):
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = cur.copy()
            else:
                cur[i], cur[j] = 1.0, 0.0
        T *= mult
    return best_h, best_c5

def _fine_search(h, n, dx, rng, deadline, total_iters=300000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    per = max(500, total_iters // len(scales))
    for idx_s, s in enumerate(scales):
        improved = False
        iters = 0
        while iters < per and time.time() < deadline:
            iters += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = s * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                continue
            h[i], h[j] = ni, nj
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not improved and idx_s >= 5:
            break
    return h, best_c5

def _block_search(h, n, dx, rng, deadline, n_iters=10000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for _ in range(n_iters):
        if time.time() >= deadline:
            break
        bs = rng.randint(2, 4)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.003, 0.02) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        seg_i = h[i:i+bs]
        seg_j = h[j:j+bs]
        if np.any(seg_i + eps < -1e-10) or np.any(seg_i + eps > 1.0 + 1e-10):
            ok = False
        if np.any(seg_j - eps < -1e-10) or np.any(seg_j - eps > 1.0 + 1e-10):
            ok = False
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5 = _c5(h, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best_c5

def _make_binary_start(n, rng, mode):
    target = n // 2
    h = np.zeros(n)
    if mode == 0:
        idx = rng.choice(n, size=target, replace=False)
        h[idx] = 1.0
    elif mode == 1:
        # Quasi-periodic: place ones at specific intervals
        period = n // target
        for k in range(target):
            pos = int(k * period + k * 0.5 + rng.uniform(-0.3, 0.3))
            pos = pos % n
            h[pos] = 1.0
        # Fix if duplicates
        h = (h > 0).astype(float)
        if h.sum() != target:
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            excess = int(h.sum() - target)
            if excess > 0:
                for idx in rng.choice(ones, size=excess, replace=False):
                    h[idx] = 0.0
            elif excess < 0:
                for idx in rng.choice(zeros, size=-excess, replace=False):
                    h[idx] = 1.0
    elif mode == 2:
        # Block structure: alternating blocks
        block = n // (2 * target)
        for k in range(target):
            start = k * block + k
            for b in range(min(block, n - start)):
                h[start + b] = 1.0
        h = (h > 0).astype(float)
        if h.sum() != target:
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            diff = target - int(h.sum())
            if diff > 0:
                for idx in rng.choice(zeros, size=diff, replace=False):
                    h[idx] = 1.0
            elif diff < 0:
                for idx in rng.choice(ones, size=-diff, replace=False):
                    h[idx] = 0.0
    else:
        # Random walk structure
        h[0] = 1.0
        for i in range(1, n):
            if rng.random() < 0.5:
                h[i] = 1.0 - h[i-1]
            else:
                h[i] = h[i-1]
        h = (h > 0.5).astype(float)
        if h.sum() != target:
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            diff = target - int(h.sum())
            if diff > 0:
                for idx in rng.choice(zeros, size=diff, replace=False):
                    h[idx] = 1.0
            elif diff < 0:
                for idx in rng.choice(ones, size=-diff, replace=False):
                    h[idx] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Gradient warm-up (60s)
    tE = t0 + 60
    for a in [200, 1000, 5000]:
        if time.time() >= tE or rem() < 900:
            break
        h, c5 = _grad_opt(best_h, n, dx, a, 250)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: Multi-restart binary SA (350s)
    if rem() > 650:
        tEnd = time.time() + min(350, rem() - 600)
        n_restarts = 12
        t_per = (tEnd - time.time()) / n_restarts
        for r in range(n_restarts):
            if time.time() >= tEnd:
                break
            t_sub = time.time() + min(t_per, 40)
            if r == 0:
                start = best_h
            else:
                start = _make_binary_start(n, rng, r % 4)
            h, c5 = _binary_sa(start, n, dx, rng, t_sub, T0=0.006 - 0.0004*r)
            if c5 < best_c5:
                best_h, best_c5 = h, c5
                # Quick gradient from improved binary
                h, c5 = _grad_opt(h, n, dx, 3000, 150)
                if c5 < best_c5:
                    best_h, best_c5 = h, c5

    # Phase 3: Gradient refinement (150s)
    if rem() > 450:
        tEnd = time.time() + min(150, rem() - 400)
        for a in [3000, 10000, 30000]:
            if time.time() >= tEnd:
                break
            h, c5 = _grad_opt(best_h, n, dx, a, 250)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Phase 4: Block + fine local search (250s)
    if rem() > 300:
        tEnd = time.time() + min(120, rem() - 250)
        h, c5 = _block_search(best_h, n, dx, rng, tEnd, 8000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    if rem() > 200:
        tEnd = time.time() + min(130, rem() - 170)
        h, c5 = _fine_search(best_h, n, dx, rng, tEnd, 200000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 5: Another SA round from current best (100s)
    if rem() > 200:
        tEnd = time.time() + min(100, rem() - 180)
        h, c5 = _binary_sa(best_h, n, dx, rng, tEnd, T0=0.002, mult=0.99999)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h, c5 = _grad_opt(best_h, n, dx, 20000, 200)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 6: Final intensive fine search (80s)
    if rem() > 60:
        tEnd = time.time() + min(80, rem() - 40)
        h, c5 = _fine_search(best_h, n, dx, rng, tEnd, 200000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: Ultra-fine polish
    if rem() > 30:
        h, c5 = _grad_opt(best_h, n, dx, 80000, 150)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    if rem() > 15:
        tEnd = time.time() + max(3, rem() - 5)
        h, c5 = _fine_search(best_h, n, dx, rng, tEnd, 60000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
