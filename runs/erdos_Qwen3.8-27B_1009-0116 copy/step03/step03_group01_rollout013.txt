def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _sf(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sf_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        wk = w[k]
        lag = k - (n - 1)
        if lag >= 0:
            idx1 = np.arange(0, n - lag)
            g[idx1] += wk * (1.0 - h[idx1 + lag])
            idx2 = np.arange(lag, n)
            g[idx2] -= wk * h[idx2 - lag]
        else:
            L = -lag
            idx1 = np.arange(L, n)
            g[idx1] += wk * (1.0 - h[idx1 - L])
            idx2 = np.arange(0, n - L)
            g[idx2] -= wk * h[idx2 + L]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_sf, h.copy(), jac=_sf_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _swap_segments(h, rng, n):
    hp = h.copy()
    L = rng.randint(2, max(3, n // 8))
    a = rng.randint(0, n - L)
    offset = rng.randint(1, max(2, n - L - a))
    b = a + offset
    if b + L > n:
        b = n - L
    if b == a or b + L > n:
        return hp
    hp[a:a+L], hp[b:b+L] = hp[b:b+L].copy(), hp[a:a+L].copy()
    return hp

def _binary_perturb(h, rng, n, kmin=2, kmax=8):
    hp = h.copy()
    k = rng.randint(kmin, kmax + 1)
    idxs = rng.choice(n, size=k, replace=False)
    hp[idxs] = 1.0 - hp[idxs]
    return _repair(hp, n)

def _gauss_perturb(h, rng, n, sc_base=0.02):
    sc = sc_base * (1.0 + 2.0 * rng.rand())
    return _repair(h + sc * rng.randn(n), n)
