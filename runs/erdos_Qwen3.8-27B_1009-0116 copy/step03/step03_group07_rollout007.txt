def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    # Bisection for sum constraint
    for _ in range(80):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = (target - s) / n
        h = np.clip(h + diff, 0, 1)
    # Exact adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                inc = min(diff, 1.0 - h[idx])
                h[idx] += inc
                diff -= inc
            else:
                dec = min(-diff, h[idx])
                h[idx] -= dec
                diff += dec
    return h

def weighted_grad(h, n, alpha, topk):
    """Compute weighted gradient of soft-max C5."""
    corr = compute_corr(h)
    m = corr.max()
    # Exp weights, clipped for numerical stability
    exps = np.exp(alpha * (corr - m))
    exps = np.maximum(exps, 0)
    w = exps / exps.sum()
    
    # Gradient: for each shift s, d(corr[s])/d(h[j])
    # corr stored as full (2n-1 entries), index mi corresponds to shift s = mi - (n-1)
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        wi = w[mi]
        if wi < 1e-12:
            continue
        s = mi - (n - 1)
        # j as i: j in [max(0,-s), min(n-1, n-1-s)], contrib (1-h[j+s])
        jlo = max(0, -s)
        jhi = min(n - 1, n - 1 - s)
        if jlo <= jhi:
            idx = np.arange(jlo, jhi + 1)
            grad[idx] += wi * (1.0 - h[idx + s])
        # j as i+s: j in [max(0,s), min(n-1, n-1+s)], contrib -h[j-s]  
        jlo2 = max(0, s)

        if jlo2 <= jhi2:
            idx2 = np.arange(jlo2, jhi2 + 1)
            grad[idx2] += wi * (-h[idx2 - s])
    return grad

    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for r in range(n_rounds):

        improved = False
        n_tries = 6 * n
        for _ in range(n_tries):

            j = rng.randint(n)
            if i == j:
                continue

            for d in steps:

    except Exception:

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        h = np.clip(h + (target - s) / n, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                inc = min(diff, 1.0 - h[idx]); h[idx] += inc; diff -= inc
            else:
                dec = min(-diff, h[idx]); h[idx] -= dec; diff += dec
    return h

def analytic_grad(h, n, alpha):
    corr = compute_corr(h)
    m = corr.max()
    exps = np.exp(np.clip(alpha * (corr - m), -600, 600))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        wi = w[mi]
        if wi < 1e-14:
            continue
        s = mi - (n - 1)
        jlo1 = max(0, s); jhi1 = min(n - 1, n - 1 + s)
        if jlo1 <= jhi1:
            idx = np.arange(jlo1, jhi1 + 1)
            grad[idx] += wi * (1.0 - h[idx - s])
        jlo2 = max(0, -s); jhi2 = min(n - 1, n - 1 - s)
        if jlo2 <= jhi2:
            idx2 = np.arange(jlo2, jhi2 + 1)
            grad[idx2] += wi * (-h[idx2 + s])
    return grad

def subgrad_step(h, n, lr, alpha):
    g = analytic_grad(h, n, alpha)
    h_new = repair(h - lr * g, n)
    return h_new

def coord_descent(h, n, t_end, rng, n_rounds=200):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(8 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    ni, nj = hi + sign * d, hj - sign * d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5; improved = True; break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved and r > 40:
            break
    return h, best

def slsqp_run(h, n, alpha, maxiter):
    def obj(x):
        corr = compute_corr(x)
        m = corr.max()
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < obj(h) + 1e-10:
            return repair(res.x, n)
    except Exception:
        pass
    return h

def make_start(n, seed_val, kind):
    rng = np.random.RandomState(seed_val)
    x = np.arange(n) / n
    if kind == 0:
        h = 0.5 + 0.35 * np.cos(4 * np.pi * x + rng.uniform(0, 3))
    elif kind == 1:
        c = rng.uniform(0.2, 0.8); w = rng.uniform(0.12, 0.3)
        h = np.exp(-((x - c) ** 2) / (2 * w ** 2))
        h = 0.3 + 0.7 * h / h.max()
    elif kind == 2:
        blk = max(2, n // 6)
        h = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h[i:i + blk] = 1.0
        h = 0.6 * h + 0.25 + 0.1 * rng.randn(n)
    elif kind == 3:
        h = 0.5 + 0.3 * rng.randn(n)
    else:
        f = rng.randint(2, 7); ph = rng.uniform(0, 6.28)
        h = 0.5 + 0.38 * np.sin(2 * np.pi * f * x + ph)
    return repair(h, n)

    rng = np.random.RandomState(seed + 999)
    best_h = None; best_c5 = np.inf; best_n = 70

    n = best_n; h = best_h.copy()
