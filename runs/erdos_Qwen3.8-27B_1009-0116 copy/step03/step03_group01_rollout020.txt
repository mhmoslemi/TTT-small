def _proj(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    n = len(v)
    # Bisection on lambda
    lo, hi = v.min() - 1, v.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
import time

def _proj(v, target):
    n = len(v)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _repair(h):
    return _proj(np.clip(h, 0.0, 1.0), len(h) * 0.5)

def _eval(h):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / len(h))

def _c_all(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sf(h, alpha):
    c = _c_all(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha + s.max() / alpha

def _sj(h, alpha):
    n = len(h)
    c = _c_all(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        L = abs(k - (n - 1))
        if L == 0:
            g += w[k] * (1.0 - 2.0 * h)
        else:
            g[L:] += w[k] * (1.0 - h[:n - L])
            g[:n - L] -= w[k] * h[L:]
    return g * 2.0 / n

def _slsqp(h, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sf, h, jac=_sj, args=(alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-30})
        return _repair(res.x)
    except Exception:
        return _repair(h)

def _grad_lag(h, L):
    """Exact gradient of overlap at lag L (0-indexed)."""
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    return g * 2.0 / n

def _max_lag(h):
    c = _c_all(h)
    idx = int(np.argmax(c))
    L = abs(idx - (len(h) - 1))
    return c.max() * 2.0 / len(h), L, idx, c

def _direct_reduce(h, n, t0, deadline):
    """Iteratively reduce the maximum lag via projected gradient steps."""
    best = h.copy()
    best_c5 = _eval(h)
    h = h.copy()
    lr = 0.01
    stall = 0
    while time.time() < deadline:
        c5, L, idx, c = _max_lag(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
            stall = 0
        g = _grad_lag(h, L)
        improved = False
        for f in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_new = _proj(h - f * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 * (1 - 1e-12):
                h = h_new
                improved = True
                break
        if not improved:
            lr *= 0.5
            stall += 1
            if stall > 30 or lr < 1e-10:
                break
        else:
            lr = min(lr * 1.1, 0.02)
    return best, best_c5

def _anneal(h, n, rng, t0, deadline):
    """Simulated annealing with pair moves."""
    best = h.copy()
    best_c5 = _eval(h)
    h = h.copy()
    c5c = best_c5
    T = 0.005
    for it in range(200000):
        if time.time() > deadline:
            break
        if it % 5000 == 0 and it > 0:
            T *= 0.7
            if T < 1e-7:
                break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.05 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / T):
            c5c = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best = h.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return best, best_c5

    c5b = _eval(hb)

    # Combinatorial local search on binary

                    if c5n < best_c5 - 1e-14:
