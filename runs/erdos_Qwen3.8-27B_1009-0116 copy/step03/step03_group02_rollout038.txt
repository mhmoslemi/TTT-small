import numpy as np
from scipy.optimize import minimize
import time

def _fc(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5v(h, dx):
    return float(np.max(_fc(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            step = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(abs(diff), h[idx])
            h[idx] += step if diff > 0 else -step
            diff += -step if diff > 0 else step
            if abs(diff) < 1e-13:
                break
    return h

def _smax_grad(h, alpha, n):
    corr = _fc(h)
    m = corr.max()
    s = np.clip(alpha * (corr - m), -400, 400)
    e = np.exp(s)
    p = e / e.sum()
    val = m + np.log(e.sum()) / alpha
    grad = np.zeros(n)
    shifts = np.arange(-(n-1), n)
    for k in range(2*n-1):
        pk = p[k]
        if pk < 1e-10:
            continue
        sh = shifts[k]
        lo1, hi1 = max(0, sh), min(n, n + sh)
        if lo1 < hi1:
            grad[lo1:hi1] += pk * (1.0 - h[lo1-sh:hi1-sh])
        lo2, hi2 = max(0, -sh), min(n, n - sh)
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2+sh:hi2+sh]
    return val, grad, corr, p

def _effects(h, n):
    E = np.zeros((n, 2*n-1))
    shifts = np.arange(-(n-1), n)
    for i in range(n):
        for k, sh in enumerate(shifts):
            e = 0.0
            if 0 <= i - sh < n:
                e += 1.0 - h[i-sh]
            if 0 <= i + sh < n:
                e -= h[i+sh]
            E[i, k] = e
    return E

def _run_pairs(h, n, corr, time_end, n_coarse=21, n_bisect=25):
    cur_max = corr.max()
    improved = False
    shifts = np.arange(0, 1, n_coarse)
    while time.time() < time_end:
        E = _effects(h, n)
        order = np.random.permutation(n)
        round_improved = False
        for ai in range(n):
            i = order[ai]
            for aj in range(ai+1, n):
                j = order[aj]
                if time.time() > time_end:
                    return h, corr, cur_max, improved
                hi, hj = h[i], h[j]
                dlo = max(-hi, hj - 1.0)
                dhi = min(1.0 - hi, hj)
                if dhi - dlo < 1e-12:
                    continue
                g = E[i] - E[j]
                ds = dlo + shifts * (dhi - dlo)
                vals = np.max(corr[None,:] + ds[:,None]*g[None,:], axis=1)
                bi = int(vals.argmin())
                if vals[bi] >= cur_max - 1e-12:
                    continue
                lo = dlo + shifts[max(0,bi-1)]*(dhi-dlo)
                hh = dlo + shifts[min(n_coarse-1,bi+1)]*(dhi-dlo)
                for _ in range(n_bisect):
                    mid = 0.5*(lo+hh)
                    if np.max(corr + mid*g) <= np.max(corr + hh*g):
                        hh = mid
                    else:
                        lo = mid
                d = 0.5*(lo+hh)
                h[i] = hi + d
                h[j] = hj - d
                corr = corr + d*g
                cur_max = corr.max()
                round_improved = True
        if round_improved:
            improved = True
        else:
            break
    return h, corr, cur_max, improved

import numpy as np
from scipy.optimize import minimize
import time

def _fc(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5v(h, dx):
    return float(np.max(_fc(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                step = min(diff, 1.0 - h[idx])
                h[idx] += step
                diff -= step
            else:
                step = min(-diff, h[idx])
                h[idx] -= step
                diff += step
            if abs(diff) < 1e-13:
                break
    return h

def _smax_grad(h, alpha, n):
    corr = _fc(h)
    m = float(corr.max())
    s = np.clip(alpha * (corr - m), -400, 400)
    e = np.exp(s)
    p = e / e.sum()
    val = m + float(np.log(e.sum())) / alpha
    grad = np.zeros(n)
    shifts = np.arange(-(n - 1), n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-10:
            continue
        sh = shifts[k]
        lo1, hi1 = max(0, sh), min(n, n + sh)
        if lo1 < hi1:
            grad[lo1:hi1] += pk * (1.0 - h[lo1 - sh:hi1 - sh])
        lo2, hi2 = max(0, -sh), min(n, n - sh)
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2 + sh:hi2 + sh]
    return val, grad, corr, p

def _effects(h, n):
    E = np.zeros((n, 2 * n - 1))
    shifts = np.arange(-(n - 1), n)
    for i in range(n):
        for k, sh in enumerate(shifts):
            e = 0.0
            if 0 <= i - sh < n:
                e += 1.0 - h[i - sh]
            if 0 <= i + sh < n:
                e -= h[i + sh]
            E[i, k] = e
    return E

def _bisect_convex(corr, g, lo, hi, n_iter=28):
    for _ in range(n_iter):
        mid = 0.5 * (lo + hi)
        k_at = int(np.argmax(corr + mid * g))
        slope = g[k_at]
        if abs(slope) < 1e-15:
            return mid
        if slope < 0:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def _run_pairs(h, n, corr, time_end, n_coarse=21, n_bisect=28, max_passes=8):
    cur_max = float(corr.max())
    improved = False
    ts = np.linspace(0, 1, n_coarse)
    for _pass in range(max_passes):
        if time.time() > time_end:
            break
        E = _effects(h, n)
        order = np.random.permutation(n)
        pass_imp = False
        for ai in range(n):
            i = order[ai]
            for aj in range(ai + 1, n):
                j = order[aj]
                if time.time() > time_end:
                    return h, corr, cur_max, improved
                hi, hj = h[i], h[j]
                dlo = max(-hi, hj - 1.0)
                dhi = min(1.0 - hi, hj)
                if dhi - dlo < 1e-12:
                    continue
                g = E[i] - E[j]
                ds = dlo + ts * (dhi - dlo)
                vals = np.max(corr[None, :] + ds[:, None] * g[None, :], axis=1)
                bi = int(vals.argmin())
                if vals[bi] >= cur_max - 1e-12:
                    continue
                lo = dlo + ts[max(0, bi - 1)] * (dhi - dlo)
                hh = dlo + ts[min(n_coarse - 1, bi + 1)] * (dhi - dlo)
                d = _bisect_convex(corr, g, lo, hh, n_bisect)
                new_corr = corr + d * g
                if new_corr.max() < cur_max - 1e-13:
                    h[i] = hi + d
                    h[j] = hj - d
                    corr = new_corr
                    cur_max = float(corr.max())
                    pass_imp = True
                    E[i] = _effects(h, n)[i]
                    E[j] = _effects(h, n)[j]
        if pass_imp:
            improved = True
        else:
            break
    return h, corr, cur_max, improved

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n * 0.5
    bounds = [(0.0, 1.0)] * n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5v(h, dx)

    def _rem():
        return budget_s - 8 - (time.time() - t0)

    # Phase 1: SLSQP warm-up
    for alpha in [50, 200, 600]:
        if _rem() < 30:
            break
        def _obj(x, a=alpha):
            c = _fc(x)
            m = float(c.max())
            s = np.clip(a * (c - m), -400, 400)
            return m + float(np.log(np.sum(np.exp(s)))) / a
        try:
            r = minimize(_obj, h, method='SLSQP', bounds=bounds,
                        constraints=[{'type': 'eq', 'fun': lambda x: float(x.sum()) - tgt}],
                        options={'maxiter': 250, 'ftol': 1e-14})
            hc = _repair(r.x, n)
            c5 = _c5v(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
                h = hc.copy()
        except Exception:
            pass

    # Phase 2: Adam with analytic gradient
    if _rem() > 40:
        configs = [(100, 0.02, 400), (400, 0.01, 500), (1500, 0.005, 600), (5000, 0.002, 600)]
        for alpha, lr, nsteps in configs:
            if _rem() < 10:
                break
            b1, b2, eps = 0.85, 0.999, 1e-8
            m_ = np.zeros(n)
            v_ = np.zeros(n)
            t_a = 0
            h = best_h.copy()
            local_best = best_c5
            local_best_h = h.copy()
            stall = 0
            for step in range(nsteps):
                if _rem() < 5:
                    break
                _, grad, _, _ = _smax_grad(h, alpha, n)
                t_a += 1
                m_ = b1 * m_ + (1 - b1) * grad
                v_ = b2 * v_ + (1 - b2) * grad * grad
                h = _repair(h - lr * (m_ / (1 - b1**t_a)) / (np.sqrt(v_ / (1 - b2**t_a)) + eps), n)
                if step % 40 == 39 or step == nsteps - 1:
                    c5 = _c5v(h, dx)
                    if c5 < local_best - 1e-14:
                        local_best = c5
                        local_best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                if stall > 5:
                    break
            if local_best < best_c5:
                best_c5 = local_best
                best_h = local_best_h.copy()

    # Phase 3: Pair sweeps with perturbation (main workhorse)
    stall_count = 0
    while _rem() > 30:
        h = best_h.copy()
        corr = _fc(h)
        chunk = min(50.0, _rem() - 20)
        time_end = time.time() + chunk
        h, corr, cm, imp = _run_pairs(h, n, corr, time_end)
        if imp:
            c5 = cm * dx
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1
        if stall_count >= 3 and _rem() > 60:
            scale = 0.015 + 0.015 * min(stall_count, 5)
            hp = best_h + scale * np.random.randn(n)
            hp = _repair(hp, n)
            corr_p = _fc(hp)
            time_end_p = min(time.time() + 30, t0 + budget_s - 25)
            hp, corr_p, cm_p, _ = _run_pairs(hp, n, corr_p, time_end_p)
            if cm_p * dx < best_c5 - 1e-14:
                best_c5 = cm_p * dx
                best_h = hp.copy()
            stall_count = 0

    # Phase 4: Final high-precision pair polish
    if _rem() > 8:
        h = best_h.copy()
        corr = _fc(h)
        h, corr, cm, _ = _run_pairs(h, n, corr, min(time.time() + 15, t0 + budget_s - 5), 31, 32, 5)
        if cm * dx < best_c5:
            best_c5 = cm * dx
            best_h = h.copy()

    # Phase 5: Final SLSQP
    if _rem() > 3:
        try:
            def _objf(x):
                c = _fc(x)
                m = float(c.max())
                s = np.clip(8000.0 * (c - m), -400, 400)
                return m + float(np.log(np.sum(np.exp(s)))) / 8000.0
            r = minimize(_objf, best_h, method='SLSQP', bounds=bounds,
                        constraints=[{'type': 'eq', 'fun': lambda x: float(x.sum()) - tgt}],
                        options={'maxiter': 400, 'ftol': 1e-16, 'maxls': 20})
            hc = _repair(r.x, n)
            if _c5v(hc, dx) < best_c5:
                best_c5 = _c5v(hc, dx)
                best_h = hc.copy()
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5v(best_h, dx)
    return (best_h, best_c5, n)
