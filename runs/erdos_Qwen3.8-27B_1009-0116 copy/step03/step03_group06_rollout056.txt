import numpy as np
import time
from scipy.optimize import minimize

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _max_corr(h):
    return int(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _make_binary(h, n, n_ones):
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:n_ones]] = 1.0
    return out

def _smooth_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_obj_pen(x, alpha, target, pen):
    return _smooth_obj(x, alpha, target) + pen * (np.sum(x) - target)**2

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _sa_binary(h0, n, seed, time_budget):
    """SA on binary configs. Returns best binary array and its max_corr."""
    rng = np.random.RandomState(seed)
    n_ones = n // 2
    h = _make_binary(h0, n, n_ones)
    best = h.copy()
    best_mc = _max_corr(h)
    
    T = 2.0
    T_floor = 0.05
    deadline = time.time() + time_budget
    last_improve = time.time()
    
    while time.time() < deadline:
        # Check for impatience - restart if no improvement for a while
        if time.time() - last_improve > max(2.0, time_budget * 0.05):
            # Perturb and reheat
            T = 1.5
            for _ in range(rng.randint(3, 10)):
                i = np.where(h == 1)[0]
                j = np.where(h == 0)[0]
                ii, jj = i[rng.randint(len(i))], j[rng.randint(len(j))]
                h[ii], h[jj] = h[jj], h[ii]
            last_improve = time.time()
        
        # Choose move
        r = rng.random()
        if r < 0.6:
            i_arr = np.where(h == 1)[0]
            j_arr = np.where(h == 0)[0]
            i, j = i_arr[rng.randint(len(i_arr))], j_arr[rng.randint(len(j_arr))]
            h[i], h[j] = h[j], h[i]
            new_mc = _max_corr(h)
            if new_mc < best_mc:
                best_mc = new_mc
                best = h.copy()
                last_improve = time.time()
            elif new_mc > best_mc and rng.random() >= np.exp(-(new_mc - best_mc) / T):
                h[i], h[j] = h[j], h[i]
        elif r < 0.85:
            # Double swap
            i_arr = np.where(h == 1)[0]
            j_arr = np.where(h == 0)[0]
            i1, j1 = i_arr[rng.randint(len(i_arr))], j_arr[rng.randint(len(j_arr))]
            i2, j2 = i_arr[rng.randint(len(i_arr))], j_arr[rng.randint(len(j_arr))]
            if len({i1, j1, i2, j2}) < 4:
                continue
            h[i1], h[j1] = h[j1], h[i1]
            h[i2], h[j2] = h[j2], h[i2]
            new_mc = _max_corr(h)
            if new_mc < best_mc:

            elif new_mc > best_mc and rng.random() >= np.exp(-(new_mc - best_mc) / T):
                h[i1], h[j1] = h[j1], h[i1]
                h[i2], h[j2] = h[j2], h[i2]

def _cont_refine(h, n, dx, time_budget, seed):
    """Continuous refinement using L-BFGS-B with softmax objective."""
    rng = np.random.RandomState(seed)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h = _repair(h.copy(), n)
    best_c5 = _c5(best_h, dx)
    deadline = time.time() + time_budget
    
    alpha = 200
    while time.time() < deadline:
        for pen in [10.0, 100.0, 500.0]:
            if time.time() >= deadline:
                break
            try:
                res = minimize(
                    lambda x, a=alpha, p=pen: _smooth_obj_pen(x, a, target, p),
                    best_h, method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-8})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_h, best_c5 = hc, c5
            except:
                pass
        alpha *= 3
        if alpha > 50000:
            break
    
    return best_h, best_c5

def _fine_local(h, n, dx, seed, time_budget):

    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    deadline = time.time() + time_budget
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005]
    per_scale = max(2000, int(time_budget * 50 / len(scales)))
    
    for scale in scales:
        if time.time() >= deadline:
            break
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)

            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = best_h[i] + d, best_h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            best_h[i], best_h[j] = ni, nj
            c5 = _c5(best_h, dx)
            if c5 < best_c5:
                best_c5 = c5
                improved = True
            else:
                best_h[i], best_h[j] = best_h[i] - d, best_h[j] + d
        if not improved:
            break
    return _repair(best_h, n), best_c5

def _struct_start(n, kind, seed):
    rng = np.random.RandomState(seed)
    n_ones = n // 2
    h = np.zeros(n)
    if kind == 0:
        pos = rng.choice(n, n_ones, replace=False)
        h[pos] = 1.0
    elif kind == 1:
        golden = (1.0 + np.sqrt(5.0)) / 2.0
        for k in range(n_ones):
            h[int(k * golden) % n] = 1.0
    elif kind == 2:
        period = n // n_ones + rng.randint(0, 2)
        for k in range(n_ones):
            h[int(k * period) % n] = 1.0
    else:
        h[n_ones:] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    n_ones = n // 2
    h0 = np.array(initial_h_values, dtype=float)
    h_bin = _make_binary(h0, n, n_ones)
    best_h = h_bin.copy()
    best_c5 = _max_corr(h_bin) * dx

    # Phase 1: Binary SA from initial (35%)
    budget1 = min(budget_s * 0.30, budget_s - 200)
    if budget1 > 5:
        hb, mc = _sa_binary(h_bin, n, seed, budget1)
        c5v = mc * dx
        if c5v < best_c5:
            best_h, best_c5 = hb.copy(), c5v

    # Phase 2: Binary SA from structured starts (25%)
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.70:
        rem = budget_s - elapsed - 150
        if rem > 10:
            per_start = min(rem / 3.0, 80.0)
            for k in range(3):
                if time.time() - t_start > budget_s * 0.68:
                    break
                hstart = _struct_start(n, k, seed + 100 + k)
                hb, mc = _sa_binary(hstart, n, seed + 200 + k, per_start)
                c5v = mc * dx
                if c5v < best_c5:
                    best_h, best_c5 = hb.copy(), c5v

    # Phase 3: More SA refinement from current best (20%)
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.80:
        rem = budget_s - elapsed - 100
        if rem > 5:
            hb, mc = _sa_binary(best_h, n, seed + 500, min(rem * 0.8, 120))
            c5v = mc * dx
            if c5v < best_c5:
                best_h, best_c5 = hb.copy(), c5v

    # Phase 4: Continuous refinement (20%)
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.90:
        rem = budget_s - elapsed - 60
        if rem > 5:
            hc, c5v = _cont_refine(best_h, n, dx, min(rem * 0.8, 120))
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v

    # Phase 5: Fine local search
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.96:
        rem = budget_s - elapsed - 5
        if rem > 2:
            hf, c5v = _fine_local(best_h, n, dx, seed + 900, min(rem * 0.9, 40))
            if c5v < best_c5:
                best_h, best_c5 = hf, c5v

    # Final: one more SLSQP polish if time
    elapsed = time.time() - t_start
    if elapsed < budget_s - 3:
        rem = budget_s - elapsed - 1
        if rem > 2:
            hc = _lbfgs_step(best_h, n, 5000, n / 2.0, 100.0, min(200, int(rem * 50)))
            c5v = _c5(hc, dx)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
