from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5v(c, dx):
    return float(np.max(c)) * dx

def _proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        if s < 1e-15:
            h[:] = target / len(h)
            break
        h = np.clip(h * (target / s), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _wgrad(h, c, pw, topK):
    """Weighted gradient of top-K correlations."""
    n = len(h)
    top = np.argsort(pw)[::-1][:topK]
    g = np.zeros(n)
    for ki in top:
        w = pw[ki]
        if w < 1e-12:
            break
        L = ki - (n - 1)
        if L == 0:
            g += w * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n-L] += w * (1.0 - h[L:])
            g[L:] -= w * h[:n-L]
        else:
            M = -L
            g[M:] += w * (1.0 - h[:n-M])
            g[:n-M] -= w * h[M:]
    return g

def _fista(h0, n, dx, target, iters, a0, a1, topK, time_fn, remaining_fn):

    h = h0.copy()

        if remaining_fn() < 30:

        a = a0 * (a1 / a0) ** (it / max(iters - 1, 1))
        c = _corr(y)
        m = c.max()
        s = np.clip(a * (c - m), -500, 500)
        ew = np.exp(s)
        pw = ew / ew.sum()
        g = _wgrad(y, c, pw, topK)
        # Step size: 1/Lipschitz, roughly 2*topK*a for our problem

def _lbfgs(h0, n, dx, target, alpha, lam, maxiter, time_fn, remaining_fn):
    """L-BFGS-B with soft-max + penalty."""
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        pen = lam * (x.sum() - target) ** 2
        return m + np.log(np.sum(np.exp(s))) / alpha + pen
    def jac(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        ew = np.exp(s)
        pw = ew / ew.sum()
        g = _wgrad(x, c, pw, len(c))
        g *= 1.0 / alpha
        g += 2.0 * lam * (x.sum() - target)
        return g
    try:
        res = minimize(obj, h0, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _proj(res.x, target)
    except:
        return h0

            else:

    if rem() > 100:
        for alpha in [100000, 500000]:
            if rem() < 80:
                break
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(_sm_obj, best_h.copy(),
                               jac=lambda x: _sm_jac(x, alpha, 300.0, 8),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-19})
                hc = _proj(res.x, target)
                c5 = _c5v(_corr(hc), dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    # Phase 4: Coordinate descent
    if rem() > 40:
        hc, c5c = _coord_cd(best_h, dx, target, rem, 80)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc

    # Phase 5: Multi-restart with L-BFGS + CD
    if rem() > 60:
        for _ in range(10):
            if rem() < 50:
                break
            h_r = _proj(best_h + 0.006 * rng.randn(n), target)
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(_sm_obj, h_r,
                               jac=lambda x: _sm_jac(x, 200000, 300.0, 8),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-19})
                h_r = _proj(res.x, target)
            except Exception:
                pass
            c5 = _c5v(_corr(h_r), dx)
            if c5 < best_c5:
                best_c5, best_h = c5, h_r
            h_r2, c5r2 = _coord_cd(h_r, dx, target, rem, 20)
            if c5r2 < best_c5:
                best_c5, best_h = c5r2, h_r2
            # Also try FISTA from this point
            if rem() > 55:
                hf2 = _fista(h_r, dx, target, 500, 50000, 300000, 4, rem)
                c5f = _c5v(_corr(hf2), dx)
                if c5f < best_c5:
                    best_c5, best_h = c5f, hf2

    # Phase 6: Ultra-fine random polish
    if rem() > 5:
        h = best_h.copy()
        c5_cur = _c5v(_corr(h), dx)
        deltas = np.array([0.0002, 0.0005, 0.001, 0.002, 0.004,
                          -0.0002, -0.0005, -0.001, -0.002, -0.004])
        for _ in range(8000):
            if rem() < 3:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = deltas[rng.randint(len(deltas))]
            ni, nj = h[i]+d, h[j]-d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                cn = _c5v(_corr(h), dx)
                if cn < c5_cur:
                    c5_cur = cn
                    if cn < best_c5:
                        best_c5, best_h = cn, h.copy()
                else:
                    h[i], h[j] = ni-d, nj+d

    # Phase 7: Final targeted active-set polish
    if rem() > 8:
        h = best_h.copy()
        c5_cur = _c5v(_corr(h), dx)
        for _ in range(500):
            if rem() < 5:
                break
            c = _corr(h)
            top3 = np.argsort(c)[::-1][:3]
            g = np.zeros(n)
            for ki in top3:
                L = ki - (n-1)
                if L == 0:
                    g += (1.0 - 2.0*h)
                elif L > 0:
                    g[:n-L] += (1.0-h[L:]); g[L:] -= h[:n-L]
                else:
                    M = -L
                    g[M:] += (1.0-h[:n-M]); g[:n-M] -= h[M:]
            g /= 3.0
            g -= g.mean()
            for step in [0.03, 0.015, 0.007, 0.003]:
                h_try = _proj(h - step*g, target)
                cn = _c5v(_corr(h_try), dx)
                if cn < c5_cur:
                    h = h_try; c5_cur = cn
                    if cn < best_c5:
                        best_c5, best_h = cn, h.copy()
                    break
            else:
                break

    best_h = _proj(best_h, target)
    best_c5 = _c5v(_corr(best_h), dx)
    return (best_h, best_c5, n)
