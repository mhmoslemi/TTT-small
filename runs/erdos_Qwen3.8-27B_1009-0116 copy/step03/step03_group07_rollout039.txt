def _corr_at_lag(h, n, lag_offset):
    # lag_offset is the index in the 'full' correlation output (0 to 2n-2)
    # corr[k] = sum_i h[i + (k - n + 1)] * (1 - h[i])
    # For k = n-1 (middle): sum_i h[i] * (1-h[i])
    # For k = n-1+d: sum_i h[i+d] * (1-h[i]) for valid i
    shift = lag_offset - (n - 1)
    if shift >= 0:
        # sum_{i=0}^{n-1-shift} h[i+shift] * (1-h[i])
        return np.sum(h[shift:] * (1.0 - h[:n-shift]))
    else:
        # sum_{i=-shift}^{n-1} h[i] * (1-h[i-shift])
        s = -shift
        return np.sum(h[:n-s] * (1.0 - h[s:]))

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        d = target - s
        if abs(d) < 1e-15:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-15:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

        if n_alt == best_n:
            continue
        for kk in range(2):
            if time.time() > t5:
                break
            h_alt = make_start(n_alt, seed + n_alt * 31 + kk, kk % 6)
            h_alt = do_slsqp(h_alt, n_alt, 50000, 3, 0.1, 400)
            h_alt, c5_alt = coord_descent(h_alt, n_alt, min(t5, time.time() + 8), rng)
            if c5_alt < best_c5 - 1e-15:
                best_h = h_alt.copy()
                best_c5 = c5_alt
                best_n = n_alt

    # Phase 6: Final polish (92-99.5%)
    t6 = t0 + budget_s * 0.995
    if best_h is not None:
        h_fin = do_slsqp(best_h, best_n, 500000, 7, 0.5, 600)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-15:
            best_h = h_fin
            best_c5 = c5_fin
        best_h, best_c5 = coord_descent(best_h, best_n, t6, rng)

    if best_h is None:
        best_n = 70
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
