    if s < 1e-12:

                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:

    # Minimize g(d) = max_k (c[k] + d*f[k]) for d in [d_lo, d_hi]
    # Use ternary search (works for convex functions)
    lo, hi = d_lo, d_hi
    cur_max = np.max(c)  # at d=0
    
    for _ in range(40):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        g1 = np.max(c + m1 * f)
        g2 = np.max(c + m2 * f)
        if g1 < g2:
            hi = m2
        else:
            lo = m1
    
    d_star = (lo + hi) / 2
    g_star = np.max(c + d_star * f)
    return d_star, g_star

import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:

        for idx in order:

                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h


def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400.0, 400.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

    return np.sum(x) - target
