    return c[np.argsort(c)[::-1][:k]].sum()

def _grad_tk(x, n, k):
    dx = 2.0 / n
    c = _c5arr(x, n)
    idx = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for mk in idx:
        lag = mk - (n - 1)
        lo, hi = max(0, -lag), min(n, n - lag)
        for i in range(lo, hi):
            g[i + lag] += (1.0 - x[i])
            g[i] -= x[i + lag]
    return g * dx / k

def _slsqp_run(h0, n, fn, jn, maxiter=1200):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    return minimize(fn, h0, jac=jn, method='SLSQP',
                    bounds=bnds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-18})

def _pair_search(h, n, t_end, rng, n_rounds=200, n_steps=None):
    c5_cur = _c5(h, n)
    if n_steps is None:
        steps = np.array([0.06, 0.03, 0.015, 0.006, 0.002, 0.0008,
                          -0.06, -0.03, -0.015, -0.006, -0.002, -0.0008])
    else:
        half = n_steps // 2 + 1
        base = np.array([0.06, 0.03, 0.015, 0.006, 0.002, 0.0008][:6])
        steps = np.concatenate([base[:half - n_steps // 2], -base[:half - n_steps // 2]])
    for _ in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return h, c5_cur
            hs = h[i]
            for j in range(i + 1, n):
                hj = h[j]
                for d in steps:
                    ni, nj = hs + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5_cur - 1e-14:
                        c5_cur = c
                        hs, hj = ni, nj
                        improved = True
                    else:
                        h[i], h[j] = hs, hj
                h[j] = hj
            h[i] = hs
        if not improved:
            break
    return h, c5_cur

def _fine_random(h, n, t_end, rng):
    c5_cur = _c5(h, n)
    tiny = np.array([0.0003, 0.001, 0.003, -0.0003, -0.001, -0.003])
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, n)
            if c < c5_cur:
                c5_cur = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5_cur

def _upsample(h, n_old, n_new):
    xo = np.arange(n_old, dtype=float) / n_old
    xn = np.arange(n_new, dtype=float) / n_new
    return np.interp(xn, xo, np.concatenate([h, [h[0]]]))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: SLSQP smooth-max optimization
    alphas = [2000, 6000, 15000, 35000, 70000, 150000]
    for alpha in alphas:
        for r in range(5):
            if time.time() > t0 + budget_s * 0.28:
                break
            if r == 0:
                h_init = best_h.copy()
            elif r == 1:
                h_init = _repair(best_h + 0.012 * rng.randn(n), n)
            elif r <= 3:
                h_init = _repair(best_h + 0.05 * rng.randn(n), n)
            else:
                h_init = _repair(best_h + 0.12 * rng.randn(n), n)
            try:
                a = alpha
                res = _slsqp_run(h_init, n,
                    lambda x, a=a: _obj_s(x, n, a),
                    lambda x, a=a: _grad_s(x, n, a), maxiter=1500)
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
            except Exception:
                pass

    # Phase 2: Top-k focused SLSQP
    for kval in [3, 5, 8, 12]:
        for r in range(4):
            if time.time() > t0 + budget_s * 0.42:
                break
            if r == 0:
                h_init = best_h.copy()
            else:
                h_init = _repair(best_h + 0.02 * rng.randn(n), n)
            try:
                kv = kval
                res = _slsqp_run(h_init, n,
                    lambda x, kv=kv: _obj_tk(x, n, kv),
                    lambda x, kv=kv: _grad_tk(x, n, kv), maxiter=1000)
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
            except Exception:
                pass

    # Phase 3: Systematic pair local search at n
    t_end3 = t0 + budget_s * 0.72
    if time.time() < t_end3:
        best_h, best_c5 = _pair_search(best_h, n, t_end3, rng, n_rounds=500)

    # Phase 4: Multi-resolution (n -> 2n)
    n2 = 2 * n
    best2_h, best2_c5 = None, 999.0
    t_end4 = t0 + budget_s * 0.86
    if time.time() < t_end4 - 40:
        h2 = _repair(_upsample(best_h, n, n2), n2)
        best2_h = h2.copy()
        best2_c5 = _c5(h2, n2)
        for alpha in [8000, 25000, 60000]:
            for r in range(3):
                if time.time() > t_end4:
                    break
                h_init = best2_h.copy() if r == 0 else _repair(best2_h + 0.02 * rng.randn(n2), n2)
                try:
                    a = alpha
                    res = _slsqp_run(h_init, n2,
                        lambda x, a=a: _obj_s(x, n2, a),
                        lambda x, a=a: _grad_s(x, n2, a), maxiter=800)
                    hc = _repair(res.x, n2)
                    c5 = _c5(hc, n2)
                    if c5 < best2_c5:
                        best2_c5 = c5
                        best2_h = hc.copy()
                except Exception:
                    pass
        if best2_c5 < best_c5 and time.time() < t0 + budget_s - 30:
            best2_h, best2_c5 = _pair_search(
                best2_h, n2, min(t0 + budget_s - 15, t_end4 + 50), rng, n_rounds=25)
        if best2_c5 < best_c5:
            best_h, best_c5, n = best2_h, best2_c5, n2

    # Phase 5: Fine random coordinate search
    t_end5 = t0 + budget_s - 8
    if time.time() < t_end5:
        h5 = best_h.copy()
        best_h, best_c5 = _fine_random(h5, n, t_end5, rng)

    # Phase 6: Final tiny pair polish
    t_end6 = t0 + budget_s - 2
    if time.time() < t_end6:
        tiny_steps = np.array([0.0005, 0.0002, 0.0001, -0.0005, -0.0002, -0.0001])
        h6 = best_h.copy()
        c6 = _c5(h6, n)
        for _ in range(200):
            if time.time() > t_end6:
                break
            for i in range(n):
                if time.time() > t_end6:
                    break
                hs = h6[i]
                for j in range(i + 1, n):
                    hj = h6[j]
                    for d in tiny_steps:
                        ni, nj = hs + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h6[i], h6[j] = ni, nj
                        c = _c5(h6, n)
                        if c < c6 - 1e-15:
                            c6 = c
                            hs, hj = ni, nj
                        else:
                            h6[i], h6[j] = hs, hj
                    h6[j] = hj
                h6[i] = hs
        if c6 < best_c5:
            best_h, best_c5 = h6, c6

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
