                        c5c = c5n; hi, hj = ni, nj; imp = True
                    else:
                        h[i], h[j] = hi, hj
                h[j] = hj
            h[i] = hi
        if not imp:
            break
    return h, c5c

def _rand_search(h, n, rng, n_iter, t_end, scale0):
    h = h.copy()
    c5c = _eval(h)
    scale = scale0
    noimp = 0
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n; noimp = 0
            scale = min(scale*1.12, 0.015)
        else:
            h[i], h[j] = h[i]-d, h[j]+d
            noimp += 1
            if noimp > 400:
                scale *= 0.75; noimp = 0
                if scale < 1e-7:
                    break
    return h, c5c

def _binary_ls(h, n, t_end):
    h = h.copy()
    c5c = _eval(h)
    for _ in range(300):
        if time.time() > t_end:
            break
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for i in ones:
            if time.time() > t_end:
                break
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                c5n = _eval(h)
                if c5n < c5c - 1e-15:
                    c5c = c5n; improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, c5c

def _ward_search(h, n, t_end, scale0):
    """Warped coordinate search with adaptive 1D line search per coordinate pair."""
    h = h.copy()
    c5c = _eval(h)
    scale = scale0
    rng_l = np.random.RandomState(int(time.time()*1000) % 10000)
    for _ in range(80000):
        if time.time() > t_end:
            break
        i, j = rng_l.randint(n), rng_l.randint(n)
        if i == j:
            continue
        # Try a set of candidate directions
        best_d = 0.0
        best_c = c5c
        for d in [scale, -scale, scale*0.3, -scale*0.3, scale*2, -scale*2]:
            ni, nj = h[i]+d, h[j]-d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c = _eval(h)
            h[i], h[j] = h[i]-d, h[j]+d
            if c < best_c - 1e-15:
                best_c = c; best_d = d
        if best_d != 0.0:
            h[i] += best_d; h[j] -= best_d
            c5c = best_c
            scale = min(scale*1.05, 0.02)
        else:
            scale *= 0.98
            if scale < 1e-7:
                break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP with increasing smoothness
    for alpha in [200, 2000, 20000, 200000, 500000]:
        if time.time() > t_end - 400:
            break
        h = _slsqp(best_h, n, alpha, 2500, 1e-19)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Structured pairwise local search
    dsets = [
        np.array([0.04, 0.1, 0.2, 0.3, -0.04, -0.1, -0.2, -0.3]),
        np.array([0.008, 0.02, 0.06, 0.12, -0.008, -0.02, -0.06, -0.12]),
        np.array([0.001, 0.004, 0.012, 0.03, -0.001, -0.004, -0.012, -0.03]),
        np.array([0.0002, 0.0008, 0.003, 0.008, -0.0002, -0.0008, -0.003, -0.008]),
    ]
    for ds in dsets:
        if time.time() > t_end - 150:
            break
        best_h, best_c5 = _pairwise(best_h, n, rng, 300, t_end-150, ds)

    # Phase 3: Warped adaptive search
    if time.time() < t_end - 120:
        best_h, best_c5 = _ward_search(best_h, n, t_end-120, 0.005)

    # Phase 4: Binary local search
    if time.time() < t_end - 100:
        hb = (best_h > 0.5).astype(float)
        hb = _repair(hb, n)
        hb, cb = _binary_ls(hb, n, t_end-100)
        # Blend binary back with continuous
        hblend = 0.5*best_h + 0.5*hb
        hblend = _repair(hblend, n)
        cb2 = _eval(hblend)
        if cb2 < best_c5:
            best_c5, best_h = cb2, hblend
        if cb < best_c5:
            best_c5, best_h = cb, hb.copy()

    # Phase 5: Multi-start SLSQP
    if time.time() < t_end - 100:
        for _ in range(12):
            if time.time() > t_end - 80:
                break
            sc = 0.005 + 0.04*rng.rand()
            hp = _repair(best_h + sc*rng.randn(n), n)
            h = _slsqp(hp, n, 100000, 1500, 1e-19)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
                best_h, best_c5 = _pairwise(best_h, n, rng, 50, t_end-80, dsets[2])

    # Phase 6: Different n values
    if time.time() < t_end - 80:
        for n2 in [60, 80, 96]:
            if time.time() > t_end - 40:
                break
            # Downsample/upsample
            if n2 <= n:
                idx = np.linspace(0, n-1, n2).astype(int)
                h2 = best_h[idx]
            else:
                fr = np.linspace(0, n-1, n2)
                h2 = np.interp(fr, np.arange(n), best_h)
            h2 = _repair(h2, n2)
            c5_2 = _eval(h2)
            if time.time() < t_end - 30:
                h2 = _slsqp(h2, n2, 50000, 1500, 1e-19)
                h2, c5_2 = _pairwise(h2, n2, rng, 100, t_end-30, dsets[1])
            if c5_2 < best_c5:
                best_c5, best_h, n = c5_2, h2, n2

    # Phase 7: Final SLSQP polish
    if time.time() < t_end - 10:
        h = _slsqp(_repair(best_h, n), n, 2000000, 3000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 8: Ultra-fine random
    if time.time() < t_end - 3:
        h = best_h.copy()
        c5c = _eval(h)
        scale = 5e-5
        for _ in range(100000):
            if time.time() > t_end - 3:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i]+d, h[j]-d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = ni-d, nj+d
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
