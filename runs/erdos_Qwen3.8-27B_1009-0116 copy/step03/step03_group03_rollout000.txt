```python
def _best_pair_move(h, target, c_arr, grad_matrix, n):
    """Find the best pair move (i, j, delta) to reduce max correlation"""
    max_c = c_arr.max()
    # Focus on correlations near the max
    active = np.where(c_arr > 0.95 * max_c)[0]
    
    best_improve = 1e-15
    best_i, best_j, best_d = -1, -1, 0
    best_new_max = max_c
    
    for i in range(n):
        if h[i] >= 1.0:
            continue
        for j in range(n):
            if j == i or h[j] <= 0.0:
                continue
            # Gradient difference
            dg = np.mean(grad_matrix[active, i]) - np.mean(grad_matrix[active, j])
            if dg <= 0:
                continue
            # Max possible delta
            d_max = min(1.0 - h[i], h[j])
            # Tentative improvement (linear approx)
            improve = dg * d_max * 0.1  # conservative
            if improve > best_improve:
                # Evaluate more precisely
                d = d_max * 0.1
                h[i] += d
                h[j] -= d
                c_new = _corr(h)
                new_max = c_new.max()
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_i, best_j, best_d = i, j, d
                    best_improve = improve
                h[i] -= d
                h[j] += d
    return best_i, best_j, best_d
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_lse(h, alpha):
    n = len(h)
    c = _corr(h)
    ac = alpha * c
    ac -= ac.max()
    p = np.exp(ac)
    p /= p.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        k = abs(m - (n - 1))
        w = p[m]
        if k < n:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] -= w * h[k:]
    return g

def _grad_lag(h, n, m):
    k = abs(m - (n - 1))
    g = np.zeros(n)
    if k < n:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-13:
                break
    return h

def _nesterov(h, target, dx, budget, rng):
    n = len(h)
    x = h.copy()
    y = h.copy()
    best = _c5(x, dx)
    best_x = x.copy()
    t0 = time.time()
    momentum = 0.9
    alphas_list = []
    alphas_list.extend(np.linspace(50, 300, 20))
    alphas_list.extend(np.linspace(300, 3000, 25))
    alphas_list.extend(np.linspace(3000, 30000, 30))
    alphas_list.extend(np.linspace(30000, 100000, 40))
    alphas_list.extend(np.linspace(100000, 300000, 50))
    total_it = 0
    for ai, alpha in enumerate(alphas_list):
        if time.time() - t0 > budget:
            break
        lr = 0.5 / (1.0 + 0.01 * (total_it / len(alphas_list)))
        for _it in range(50):
            if time.time() - t0 > budget:
                break
            g = _grad_lse(y, alpha)
            y_new = momentum * y + (1.0 - momentum) * x
            x_new = _project(y_new - lr * g, target)
            x, y = x_new, y_new
            total_it += 1
            c5 = _c5(x, dx)
            if c5 < best:
                best = c5
                best_x = x.copy()
    return best_x, best

def _pair_scan(h, target, dx, budget, rng):
    n = len(h)
    best = _c5(h, dx)
    best_x = h.copy()
    t0 = time.time()
    while time.time() - t0 < budget:
        c_arr = _corr(best_x)
        max_c = c_arr.max()
        thresh = 0.85 * max_c
        active = np.where(c_arr > thresh)[0]
        G = np.array([_grad_lag(best_x, n, m) for m in active])
        avg_g = G.mean(axis=0)
        scores = []
        for i in range(n):
            if best_x[i] >= 0.999:
                continue
            for j in range(n):
                if j == i or best_x[j] <= 0.001:
                    continue
                gain = avg_g[i] - avg_g[j]
                if gain > 1e-10:
                    d_max = min(1.0 - best_x[i], best_x[j])
                    scores.append((gain * d_max, i, j, d_max))
        if not scores:
            break
        scores.sort(key=lambda x: -x[0])
        moved = False
        for score, i, j, d_max in scores[:15]:
            for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
                d = d_max * frac
                best_x[i] += d
                best_x[j] -= d
                best_x = _repair(best_x, n)
                c5_new = _c5(best_x, dx)
                if c5_new < best - 1e-14:
                    best = c5_new
                    moved = True
                    break
                else:
                    best_x[i] -= d
                    best_x[j] += d
                    best_x = _repair(best_x, n)
            if moved:
                break
        if not moved:
            # Random perturbation to escape
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d = 0.02 * rng.rand() * (1 if rng.rand() < 0.5 else -1)
            d *= min(1.0 - best_x[min(i,j)], best_x[max(i,j)]) * 3
            ni, nj = min(i,j), max(i,j)
            if 0 < best_x[ni] + d < 1 and 0 < best_x[nj] - d < 1:
                best_x[ni] += d
                best_x[nj] -= d
                best_x = _repair(best_x, n)
    return best_x, best

def _sim_anneal(h, target, dx, budget, rng):
    n = len(h)
    x = h.copy()
    best = _c5(x, dx)
    best_x = x.copy()
    t0 = time.time()
    t_dur = budget
    it = 0
    while time.time() - t0 < budget:
        frac = 1.0 - (time.time() - t0) / t_dur
        temp = 0.01 * frac + 0.0001
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        d = rng.uniform(0.001, 0.05) * (1 if rng.rand() < 0.5 else -1)
        d *= min(1.0 - x[min(i,j)], x[max(i,j)]) * 10
        d = max(-0.1, min(0.1, d))
        ni, nj = min(i,j), max(i,j)
        if x[ni] + d < 0 or x[ni] + d > 1 or x[nj] - d < 0 or x[nj] - d > 1:
            continue
        x[ni] += d
        x[nj] -= d
        x = _repair(x, n)
        c5_new = _c5(x, dx)
        delta = c5_new - best
        if delta < 0 or rng.rand() < np.exp(-delta / temp):
            if c5_new < best:
                best = c5_new
                best_x = x.copy()
        else:
            x[ni] -= d
            x[nj] += d
            x = _repair(x, n)
        it += 1
    return best_x, best

def _slsqp_refine(h, target, dx, budget, rng):
    n = len(h)
    best_x = h.copy()
    best = _c5(h, dx)
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(30000.0 * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / 30000.0
    def cons(x):
        return x.sum() - target
    for r in range(12):
        if time.time() - t0 > budget:
            break
        h_p = best_x + 0.015 * rng.randn(n)
        h_p = _repair(h_p, n)
        try:
            res = minimize(obj, h_p, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons}],
                          options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best:
                best = c5
                best_x = hc
        except Exception:
            pass
    return best_x, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(h, dx)
    best_h = h.copy()

    def tleft():
        return budget_s - (time.time() - t0)

    # Phase 1: Nesterov projected subgradient
    if tleft() > 350:
        b1 = min(300, tleft() - 200)
        best_h, best_c5 = _nesterov(best_h, target, dx, b1, rng)

    # Phase 2: Pair scan with active-set gradient
    if tleft() > 250:
        b2 = min(280, tleft() - 200)
        best_h, best_c5 = _pair_scan(best_h, target, dx, b2, rng)

    # Phase 3: Nesterov again (re-start with sharpened alpha)
    if tleft() > 200:
        b3 = min(180, tleft() - 150)
        best_h, best_c5 = _nesterov(best_h, target, dx, b3, rng)

    # Phase 4: Simulated annealing
    if tleft() > 150:
        b4 = min(130, tleft() - 120)
        h_sa, c5_sa = _sim_anneal(best_h, target, dx, b4, rng)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa

    # Phase 5: Pair scan again (finer)
    if tleft() > 100:
        b5 = min(90, tleft() - 80)
        best_h, best_c5 = _pair_scan(best_h, target, dx, b5, rng)

    # Phase 6: SLSQP multi-restart
    if tleft() > 80:
        b6 = min(70, tleft() - 50)
        best_h, best_c5 = _slsqp_refine(best_h, target, dx, b6, rng)

    # Phase 7: Final pair polish
    if tleft() > 30:
        b7 = min(25, tleft() - 10)
        best_h, best_c5 = _pair_scan(best_h, target, dx, b7, rng)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
