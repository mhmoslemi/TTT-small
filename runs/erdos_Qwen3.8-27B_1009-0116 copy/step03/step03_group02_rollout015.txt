def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _g_vec(h, p, n):
    kk = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _upscale(h, factor):
    n_new = len(h) * factor
    h_new = np.repeat(h, factor)
    return h_new

    corr = _corr(h)
    cur_max = np.max(corr)
    improved = False
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for j_idx in range(ai+1, n):
            j = order[j_idx]
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-14:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            # Quick check: is moving in either direction helpful?
            eps = (d_max - d_min) * 0.01
            if np.max(corr + eps*gd) >= cur_max and np.max(corr - eps*gd) >= cur_max:
                continue
            # Coarse scan
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.array([np.max(corr + d*gd) for d in ds])
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-13:
                continue
            # Bisection
            lo = ds[max(0, bi-1)]
            hi_b = ds[min(n_coarse-1, bi+1)]
            for _ in range(n_bisect):
                mid = 0.5*(lo + hi_b)
                if np.max(corr + mid*gd) <= np.max(corr + hi_b*gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5*(lo + hi_b)
            h[i] += d_best
            h[j] -= d_best
            corr += d_best * gd
            cur_max = np.max(corr)
            improved = True
    return h, corr, improved

def _g_vec(h, p, n):
    kk = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _upscale(h, factor):
    return np.repeat(h, factor)

def _sweep(h, n, time_limit, n_coarse=20, n_bisect=50):
    corr = _corr(h)
    cur_max_val = float(np.max(corr))
    improved = False
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for j_idx in range(ai+1, n):
            j = order[j_idx]
            if time.time() > time_limit:
                return h, corr, improved
            hi_val, hj_val = h[i], h[j]
            d_min = max(-hi_val, hj_val - 1.0)
            d_max = min(1.0 - hi_val, hj_val)
            if d_max - d_min < 1e-14:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            d_t = 0.5 * (d_min + d_max)
            if np.max(corr + d_t*gd) >= cur_max_val - 1e-14:
                continue
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.array([np.max(corr + d*gd) for d in ds])
            bi = int(np.argmin(vals))
            if vals[bi] >= cur_max_val - 1e-13:
                continue
            lo = float(ds[max(0, bi-1)])
            hi_b = float(ds[min(n_coarse-1, bi+1)])
            for _ in range(n_bisect):
                mid = 0.5*(lo + hi_b)
                if np.max(corr + mid*gd) <= np.max(corr + hi_b*gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5*(lo + hi_b)
            h[i] += d_best
            h[j] -= d_best
            corr += d_best * gd
            nm = float(np.max(corr))
            if nm < cur_max_val:
                cur_max_val = nm
            improved = True
    return h, corr, improved

def _smax_obj(x, alpha):
    c = _corr(x)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _eq_con_jac(x, target):
    return np.ones(len(x))

def _slsqp_polish(h, n, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_con_jac, 'args': (target,)}]
    try:
        res = minimize(_smax_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 100})
        return _repair(res.x, n)
    except Exception:
        return h

def _optimize_block(h, n, t_start, t_end, label):
    """Run sweep rounds until time expires, return best."""
    best_c5 = _c5(h, 2.0/n)
    best_h = h.copy()
    stall = 0
    while time.time() < t_end:
        chunk = min(35.0, t_end - time.time())
        h, _, imp = _sweep(h, n, time.time() + chunk, 20, 50)
        c5 = _c5(h, 2.0/n)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                h_esc = _slsqp_polish(h, n, 400 + 100*stall, 350)
                h_esc, _, _ = _sweep(h_esc, n, time.time() + 20, 18, 45)
                c5_esc = _c5(h_esc, 2.0/n)
                if c5_esc < best_c5:
                    best_c5 = c5_esc
                    best_h = h_esc.copy()
                    h = h_esc
                if stall >= 8:
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def elapsed():
        return time.time() - t0

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _c5(best_h, 2.0/n0)
    best_n = n0

    # Phase 1: Optimize at original resolution
    t1_end = t0 + budget_s * 0.15
    if elapsed() < t1_end:
        h1, c5_1 = _optimize_block(h0, n0, t0, t1_end, "n0")
        if c5_1 < best_c5:
            best_c5, best_h, best_n = c5_1, h1, n0

    # Phase 2: Try higher resolutions by upscaling
    n_candidates = sorted(set([n0*f for f in [2, 3, 4, 5] if n0*f <= 400]))
    for n_try in n_candidates:
        if elapsed() > budget_s * 0.65:
            break
        factor = n_try // n0 if n_try % n0 == 0 else n_try // n0
        if factor < 2:
            continue
        h_try = _upscale(best_h, factor)
        h_try = _repair(h_try, n_try)
        t_phase_end = t0 + budget_s * 0.65
        t_sweep = t_phase_end - 30
        h_try, c5_try = _optimize_block(h_try, n_try, time.time(), t_sweep, "upscaled")
        c5_try = _c5(h_try, 2.0/n_try)
        if c5_try < best_c5:
            best_c5, best_h, best_n = c5_try, h_try, n_try

    # Phase 3: Restrictive high-resolution try (n*5 or 350)
    n_high = min(350, n0 * 5)
    if n_high > best_n and elapsed() < budget_s * 0.50:
        factor = n_high // n0 if n_high % n0 == 0 else int(n_high / n0)
        if factor >= 2:
            h_hi = _upscale(best_h, factor)
            if len(h_hi) != n_high:
                h_hi = np.interp(np.linspace(0, 1, n_high, endpoint=False),
                                  np.linspace(0, 1, len(h_hi), endpoint=False), h_hi)
                h_hi = _repair(h_hi, n_high)
            else:
                h_hi = _repair(h_hi, n_high)
            t_hi_end = t0 + budget_s * 0.60
            h_hi, c5_hi = _optimize_block(h_hi, n_high, time.time(), t_hi_end, "high")
            c5_hi = _c5(h_hi, 2.0/n_high)
            if c5_hi < best_c5:
                best_c5, best_h, best_n = c5_hi, h_hi, n_high

    # Phase 4: Intensive optimization
    n = best_n
    h = best_h.copy()
    t4_end = t0 + budget_s * 0.88
    stall = 0
    while elapsed() < t4_end:
        chunk = min(45.0, t4_end - time.time())
        h, _, imp = _sweep(h, n, time.time() + chunk, 25, 55)
        c5 = _c5(h, 2.0/n)
        if c5 < best_c5 - 1e-15:
            best_c5, best_h = c5, h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 5 and elapsed() < t4_end - 50:
                scale = 0.008 + 0.004 * (stall - 5)
                h_p = h + scale * np.random.randn(n)
                h_p = _repair(h_p, n)
                h_p, _, _ = _sweep(h_p, n, time.time() + 25, 20, 45)
                c5_p = _c5(h_p, 2.0/n)
                if c5_p < best_c5:
                    best_c5, best_h = c5_p, h_p.copy()
                    h = h_p
                stall = 0
            elif stall > 15:
                break

    # Phase 5: SLSQP polish with increasing alpha
    if elapsed() < budget_s - 15:
        for alpha in [3000, 8000]:
            if elapsed() > budget_s - 10:
                break
            h_f = _slsqp_polish(best_h, best_n, alpha, 500)
            c5_f = _c5(h_f, 2.0/best_n)
            if c5_f < best_c5:
                best_c5, best_h = c5_f, h_f.copy()

    # Phase 6: Final high-precision sweep
    if elapsed() < budget_s - 4:
        h_f, _, _ = _sweep(best_h, best_n, t0 + budget_s - 1, 30, 65)
        c5_f = _c5(h_f, 2.0/best_n)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0/best_n)
    return (best_h, best_c5, best_n)
