```python
def precompute_A(h, n):

    ncorr = 2 * n - 1
    # For each i, A[i, k] = g[i+k-(n-1)] - h[i-k+(n-1)]

    # Let's use a convolution-like approach
    
    # Actually, let's just build it with broadcasting
    i_idx = np.arange(n)[:, None]  # (n, 1)
    k_idx = np.arange(ncorr)[None, :]  # (1, ncorr)
    
    pos_idx = i_idx + k_idx - (n - 1)  # (n, ncorr)
    neg_idx = i_idx - k_idx + (n - 1)  # (n, ncorr)
    
    # Safe indexing
    pos_safe = np.where((pos_idx >= 0) & (pos_idx < n), pos_idx, 0)
    neg_safe = np.where((neg_idx >= 0) & (neg_idx < n), neg_idx, 0)
    
    A = np.where((pos_idx >= 0) & (pos_idx < n), g[pos_safe], 0.0) \
      - np.where((neg_idx >= 0) & (neg_idx < n), h[neg_safe], 0.0)
    return A
```

  - Subject to: 0 ≤ h[i]+δ ≤ 1, 0 ≤ h[j]-δ ≤ 1
  
The 1D optimization: min_δ max_k (c[k] + A_k[k] * δ)

This is a min-max of linear functions. The optimum is at a breakpoint or boundary.

Simple approach: 
- The current max is m = max(c), achieved at set S = {k: c[k] ≈ m}
- We want to find δ such that the max of (c[k] + A[k]*δ) is minimized
- For small δ, the max is determined by the elements in S that have the most positive A[k]
- We want to make those decrease: choose δ > 0 if A[k] < 0 for the maximizing k, etc.

Practical approach: ternary search on δ in the valid range.
- The function f(δ) = max_k (c[k] + A[k]*δ) is convex in δ
- Ternary search converges in O(log(range/epsilon)) iterations
- Each evaluation is O(ncorr) = O(n)

```python
def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * 2.0 / n)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def precompute_A(h, n):
    g = 1.0 - h
    ncorr = 2 * n - 1
    i_idx = np.arange(n)[:, None].astype(float)
    k_idx = np.arange(ncorr)[None, :].astype(float)
    pos_idx = (i_idx + k_idx - (n - 1)).astype(int)
    neg_idx = (i_idx - k_idx + (n - 1)).astype(int)
    pos_ok = (pos_idx >= 0) & (pos_idx < n)
    neg_ok = (neg_idx >= 0) & (neg_idx < n)
    ps = np.where(pos_ok, pos_idx, 0)
    ns = np.where(neg_ok, neg_idx, 0)
    A = np.where(pos_ok, g[ps], 0.0) - np.where(neg_ok, h[ns], 0.0)
    return A

def find_best_delta(c, A_dir, dmin, dmax, n_ternary=30):
    """Find δ ∈ [dmin, dmax] minimizing max(c + A_dir * δ)"""
    a, b = dmin, dmax
    if a >= b:
        return 0.0
    # Ternary search on convex function
    for _ in range(n_ternary):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(c + A_dir * m1)
        f2 = np.max(c + A_dir * m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def coord_sweep(h, n, A, max_sweeps=100, t_end=None):
    """Vectorized coordinate descent with exact line search"""
    c = compute_corr(h)
    ncorr = len(c)
    improved_any = False
    for sweep in range(max_sweeps):
        if t_end and time.time() > t_end:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                dmin = max(-hi + 1e-12, hj - 1.0 + 1e-12)
                dmax = min(1.0 - hi - 1e-12, hj - 1e-12)
                if dmin >= dmax:
                    continue
                A_dir = A[i] - A[j]
                delta = find_best_delta(c, A_dir, dmin, dmax)
                if abs(delta) < 1e-14:
                    continue
                # Apply update
                h[i] += delta
                h[j] -= delta
                # Update correlation
                c = c + delta * A_dir  # linear approx
                # Recompute exact (for accuracy)

                improved = True
            if not improved and sweep > 2:
                break
        # Recompute exact correlation and A periodically
        if sweep % 5 == 4:
            c = compute_corr(h)
            A = precompute_A(h, n)
            improved_any = True
    return h

def obj_smooth(h, alpha, n):
    corr = compute_corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.sum(np.exp(s)) / alpha

def do_slsqp(h, n, alpha, maxiter=500):

        corr = compute_corr(h_)
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -500, 500)
        return m + np.sum(np.exp(s)) / alpha
    
    def eq_con(h_):

    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)

        return h
```

```python
def find_best_delta_bp(c, A_dir, dmin, dmax, top_m=10):
    """Find δ minimizing max(c + A_dir*δ) using breakpoint method on top-m"""
    # Pick the indices that are "active" (near the max)
    m_val = np.max(c)
    # Get top m indices by c + A_dir*0 = c
    top_idx = np.argsort(c)[::-1][:top_m]
    
    # Also include indices that might become active
    # For δ > 0: indices with high A_dir become more important
    # For δ < 0: indices with low A_dir become more important
    top_pos = np.argsort(c + A_dir * dmax)[::-1][:top_m]
    top_neg = np.argsort(c + A_dir * dmin)[::-1][:top_m]
    
    candidates = np.unique(np.concatenate([top_idx, top_pos, top_neg]))
    
    # Collect candidate δ values
    delta_cands = set([dmin, dmax, 0.0])
    for idx_i in range(len(candidates)):
        for idx_j in range(idx_i+1, len(candidates)):
            ii, jj = candidates[idx_i], candidates[idx_j]
            if A_dir[ii] != A_dir[jj]:
                dp = (c[jj] - c[ii]) / (A_dir[ii] - A_dir[jj])
                if dmin <= dp <= dmax:
                    delta_cands.add(dp)
    
    # Evaluate at each candidate
    best_d = 0.0
    best_f = np.max(c)
    for dp in delta_cands:
        f_val = np.max(c + A_dir * dp)
        if f_val < best_f - 1e-15:
            best_f = f_val
            best_d = dp
    return best_d
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * 2.0 / n)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def precompute_A(h, n):
    g = 1.0 - h
    ncorr = 2 * n - 1
    ii = np.arange(n)[:, None]
    kk = np.arange(ncorr)[None, :]
    pos = ii + kk - (n - 1)
    neg = ii - kk + (n - 1)
    po = (pos >= 0) & (pos < n)
    no = (neg >= 0) & (neg < n)
    ps = np.where(po, pos, 0)
    ns = np.where(no, neg, 0)
    A = np.where(po, g[ps], 0.0) - np.where(no, h[ns], 0.0)
    return A

def find_delta_bp(c, Ad, lo, hi, topm=6):
    if lo >= hi:
        return 0.0
    ti = np.argsort(c)[::-1][:topm]
    tp = np.argsort(c + Ad * hi)[::-1][:topm]
    tn = np.argsort(c + Ad * lo)[::-1][:topm]
    idxs = np.unique(np.concatenate([ti, tp, tn]))
    cands = [lo, hi, 0.0]
    for a in range(len(idxs)):
        for b in range(a + 1, len(idxs)):
            ia, ib = idxs[a], idxs[b]
            den = Ad[ia] - Ad[ib]
            if abs(den) > 1e-15:
                dp = (c[ib] - c[ia]) / den
                if lo < dp < hi:
                    cands.append(dp)
    best_d = 0.0
    best_f = float(np.max(c))
    for dp in cands:
        f = float(np.max(c + Ad * dp))
        if f < best_f - 1e-14:
            best_f = f
            best_d = dp
    return best_d

def coord_desc(h, n, t_end):
    A = precompute_A(h, n)
    c = compute_corr(h)
    ncorr = len(c)
    sweep = 0
    while time.time() < t_end:
        sweep += 1
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                lo = max(-hi + 1e-14, hj - 1.0 + 1e-14)
                hi_ = min(1.0 - hi - 1e-14, hj - 1e-14)
                if lo >= hi_:
                    continue
                Ad = A[i] - A[j]
                dp = find_delta_bp(c, Ad, lo, hi_)
                if abs(dp) < 1e-14:
                    continue
                h[i] += dp
                h[j] -= dp
                c = c + dp * Ad
        if sweep % 8 == 0:
            c = compute_corr(h)
            A = precompute_A(h, n)
    h = repair(h, n)
    return h

def sliqp_obj(h_, alpha, n):
    corr = compute_corr(h_)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -400, 400)
    return m + np.sum(np.exp(s)) / alpha

def do_slsqp(h, n, alpha, maxiter):
    def obj(h_):
        return sliqp_obj(h_, alpha, n)
    def eqc(h_):
        return h_.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eqc}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        return repair(res.x, n)
    except Exception:
        return h

def do_lbfgs(h, n, alpha, pen, maxiter):
    def objp(h_):
        return sliqp_obj(h_, alpha, n) + pen * (h_.sum() - n / 2.0) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objp, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-14})
        return repair(res.x, n)
    except Exception:
        return h
```
