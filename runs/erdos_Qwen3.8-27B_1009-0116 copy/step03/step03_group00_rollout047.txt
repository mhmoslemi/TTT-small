    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5_val(best_h, n)
    best_n = n

    # Phase 1: SLSQP with increasing alpha
    alphas = [3000, 10000, 30000, 60000]
    for alpha in alphas:
        if elapsed() > budget_s - 300:
            break
        for r in range(4):
            if elapsed() > budget_s - 300:
                break
            if r == 0:
                h_init = best_h.copy()
            else:
                h_init = _repair(best_h + 0.015 * (1 + 0.5 * r) * rng.randn(n), n)
            try:
                res = _slsqp_smooth(h_init, n, alpha, 600)
                hc = _repair(res.x, n)
                c5 = _c5_val(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 2: Sub-gradient descent on active lags
    if elapsed() < budget_s - 250:
        best_h, best_c5 = _subgrad_phase(best_h, best_n, budget_s - 250, rng)

    # Phase 3: Pair moves
    if elapsed() < budget_s - 200:
        best_h, best_c5 = _pair_phase(best_h, best_n, budget_s - 200, rng)

    # Phase 4: SLSQP refine
    if elapsed() < budget_s - 120:
        for alpha in [20000, 50000]:
            if elapsed() > budget_s - 120:
                break
            try:
                res = _slsqp_smooth(best_h, best_n, alpha, 1000)
                hc = _repair(res.x, best_n)
                c5 = _c5_val(hc, best_n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 5: Ultra-fine coordinate descent
    if elapsed() < budget_s - 40:
        best_h, best_c5 = _ultra_fine(best_h, best_n, budget_s - 40, rng)

    # Phase 6: Try interpolated n=100
    if elapsed() < budget_s - 80:
        n2 = 100
        x_old = np.linspace(0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = _repair(h2, n2)
        for alpha in [15000, 40000]:
            if elapsed() > budget_s - 50:
                break
            try:
                res = _slsqp_smooth(h2, n2, alpha, 500)
                hc = _repair(res.x, n2)
                c5 = _c5_val(hc, n2)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
                    best_n = n2
            except Exception:
                pass
        if best_n == n2 and elapsed() < budget_s - 25:
            best_h, best_c5 = _pair_phase(best_h, n2, budget_s - 25, rng)
            best_h, best_c5 = _ultra_fine(best_h, n2, budget_s - 15, rng)

    # Phase 7: Final polish on original n
    if elapsed() < budget_s - 10 and best_n != n:
        x_old = np.linspace(0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0, 2.0, n, endpoint=False)
        hn = np.interp(x_new, x_old, best_h)
        hn = _repair(hn, n)
        try:
            res = _slsqp_smooth(hn, n, 30000, 400)
            hc = _repair(res.x, n)
            c5 = _c5_val(hc, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
                best_n = n
        except Exception:
            pass

    # Final ultra-fine
    if elapsed() < budget_s - 2:
        best_h, best_c5 = _ultra_fine(best_h, best_n, budget_s - 2, rng)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5_val(best_h, best_n)
    return (best_h, best_c5, best_n)
