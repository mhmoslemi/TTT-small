import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_round(h, n, corr, time_limit, n_coarse=30, n_bisect=50, order=None):
    improved = False
    n_pairs = 0
    n_improved = 0
    if order is None:
        order = np.random.permutation(n)
    for ai, i in enumerate(order):
        for j in order[ai + 1:]:
            n_pairs += 1
            if time.time() > time_limit:
                return h, corr, improved, n_pairs, n_improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            range_d = d_max - d_min
            if range_d < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            # Quick check: does moving in either direction help?
            cur_max = np.max(corr)
            # Evaluate at boundaries and midpoint
            test_ds = np.array([d_min, 0.0, d_max, 0.5*(d_min+d_max)])
            vals = np.array([np.max(corr + d*gd) for d in test_ds])
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-13:
                continue
            # Coarse grid search
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
            n_improved += 1
    return h, corr, improved, n_pairs, n_improved

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _grad_smax(x, alpha):
    c = _corr(x)
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    w /= w.sum()
    # grad of corr[m] w.r.t. x[p]
    # corr[m] = sum_i x[i]*(1-x[i+m-(n-1)])
    # d/dx[p] = (1-x[p+m-(n-1)]) - x[p-m+(n-1)] (with bounds)
    n = len(x)
    g = np.zeros(n)
    for p in range(n):
        for m in range(2*n-1):
            shift = m - (n-1)
            j1 = p + shift
            j2 = p - shift
            val = 0.0
            if 0 <= j1 < n:
                val += (1.0 - x[j1])
            if 0 <= j2 < n and j2 != p:
                val -= x[j2]
            elif 0 <= j2 < n and j2 == p:
                val -= x[j2]
            g[p] += w[m] * val
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _pair_sweep(h, n, corr, time_limit, n_bisect=50, use_fixed_order=False):
    improved = False
    if use_fixed_order:
        order = np.arange(n)
    else:
        order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for ai2 in range(ai+1, n):
            j = order[ai2]
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            cur_max = float(np.max(corr))
            # Quick check at endpoints and center
            mid_d = 0.5 * (d_min + d_max)
            v0 = cur_max
            v1 = float(np.max(corr + d_min * gd))
            v2 = float(np.max(corr + d_max * gd))
            v3 = float(np.max(corr + mid_d * gd))
            vmin = min(v1, v2, v3)
            if vmin >= cur_max - 1e-12:
                continue
            # Bisection
            lo, hi_b = d_min, d_max
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            if abs(d_best) < 1e-15:
                continue
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

def _slsqp(h, n, target, bounds, alpha, maxiter, tol):

    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad_step(h, n, dx, target, lr, n_iters):
    """Project subgradient steps on the smooth max."""
    h = h.copy()
    for _ in range(n_iters):
        corr = _corr(h)
        M = float(np.max(corr))
        active = corr > M - 0.01
        if not np.any(active):
            break
        # Weighted gradient
        w = (corr[active] - M + 0.01)
        w = np.clip(w, 0, None)
        w /= w.sum()
        # Compute gradient
        g = np.zeros(n)
        active_indices = np.where(active)[0]
        for m in active_indices:
            shift = m - (n - 1)
            for p in range(n):
                j1 = p + shift
                j2 = p - shift
                val = 0.0
                if 0 <= j1 < n:
                    val += (1.0 - h[j1])
                if 0 <= j2 < n:
                    val -= h[j2]
                g[p] += w[np.where(active_indices == m)[0][0]] * val
        # Projected step
        h_new = h - lr * g
        h_new = h_new - (h_new.sum() - target) / n
        h_new = np.clip(h_new, 0, 1)
        h = _repair(h_new, n)
    return h

    corr = _corr(h)
    m = float(np.max(corr))
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    w /= w.sum()

    # g[p] += w[m] * ((1-h[j1] if valid) - h[j2] if valid)

    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _pair_sweep(h, n, corr, deadline, n_bisect, fixed_order):
    improved = False
    if fixed_order:
        order = np.arange(n)
    else:
        order = np.random.permutation(n)
    for ai in range(n):
        i = int(order[ai])
        for bi in range(ai + 1, n):
            j = int(order[bi])
            if time.time() > deadline:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            cur_max = float(np.max(corr))
            mid_d = 0.5 * (d_min + d_max)
            v0 = float(np.max(corr + mid_d * gd))
            if v0 >= cur_max - 1e-12:
                v1 = float(np.max(corr + d_min * gd))
                v2 = float(np.max(corr + d_max * gd))
                if max(v1, v2, v0) >= cur_max - 1e-12:
                    continue
            lo, hi_b = d_min, d_max
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if float(np.max(corr + mid * gd)) <= float(np.max(corr + hi_b * gd)):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            if abs(d_best) < 1e-14:
                continue
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 100})

    except Exception:

At each step:

            hc = _slsqp_step(best_h, n, target, bounds, alpha, 1200)
            c5 = _c5(hc, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = hc.copy()
        # Final projected gradient
        if time.time() - t0 < budget_s - 3:
            hg = _proj_grad(best_h, n, 500, 30)
            c5_g = _c5(hg, dx)
            if c5_g < best_c5 - 1e-15:
                best_c5 = c5_g
                best_h = hg.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
