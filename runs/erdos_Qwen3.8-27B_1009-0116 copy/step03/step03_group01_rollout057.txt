import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        M = abs(lag)
        wk = w[k]
        if M == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag >= 0:
            g[M:] += wk * (1.0 - h[:n-M])
            g[:n-M] -= wk * h[M:]
        else:
            g[:n-M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n-M]
    return g

def _project(h):
    """Project h onto {x: 0<=x_i<=1, sum(x)=n/2}."""
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
    return h

def _pgd(h, alpha, max_iter=500, tol=1e-15):
    """Projected gradient descent with backtracking line search."""
    n = len(h)
    h = _project(h.copy())
    f = _smooth_obj(h, alpha)
    
    for it in range(max_iter):
        g = _smooth_grad(h, alpha)
        # Project gradient: for sum constraint, subtract mean of gradient
        # Then for bounds, handle in projection
        g_proj = g - np.mean(g)  # project onto sum=const
        
        # Backtracking line search
        step = 1.0 / (alpha * n)  # initial step (Lipschitz bound)
        best_h, best_f = h.copy(), f
        for _bs in range(30):
            new_h = _project(h - step * g_proj)
            new_f = _smooth_obj(new_h, alpha)
            if new_f < f - 1e-4 * step * np.dot(g_proj, h - new_h):
                best_h, best_f = new_h, new_f
                break
            step *= 0.5
        
        if np.max(np.abs(best_h - h)) < tol:
            break
        h = best_h
        f = best_f
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    c = _corr(h)
    return float(c.max() * 2.0 / n)

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _corr_grad(h, w)

def _corr_grad(h, w):
    """Gradient of weighted sum of correlations."""
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        M = abs(lag)
        wk = w[k]
        if M == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag >= 0:
            g[M:] += wk * (1.0 - h[:n - M])
            g[:n - M] -= wk * h[M:]
        else:
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _project(h):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(-(h < 1.0 - 1e-12) - (h > 1e-12)) if diff > 0 else np.argsort(h < 1.0 - 1e-12)
        # simpler: just spread remaining
        for i in range(n):
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff -= a  # diff is negative, a is positive, so diff becomes more negative... 
                # wait: diff -= a means diff = diff - a, but a = min(-diff, h[i]), so a <= -diff
                # new diff = diff - a <= diff - (-diff) = 0... no that's wrong
                # let me redo: diff < 0, a = min(-diff, h[i]) > 0
                # h[i] -= a (decrease h[i])
                # new sum = old_sum - a, so new diff = target - (old_sum - a) = diff + a
                diff += a
    return h

def _slsqp(h0, alpha, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def fun(x):
        return _smooth_obj(x, alpha)
    def jac(x):
        return _smooth_grad(x, alpha)
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < _smooth_obj(h0, alpha):
            return _project(res.x)
    except Exception:
        pass
    return _project(h0)

def _subgrad_descent(h0, max_iter=5000, t0=0.01, tmin=1e-7):
    """Subgradient method for exact max."""
    n = len(h0)
    h = _project(h0.copy())
    c5 = _eval(h)
    for it in range(max_iter):
        t = t0 * (tmin / t0) ** (it / max_iter)
        c = _corr(h)
        idx = np.argmax(c)
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
        g = _corr_grad(h, w)
        g = g - np.mean(g)
        trial = _project(h - t * g)
        if _eval(trial) < c5 - 1e-16:
            h = trial
            c5 = _eval(h)
    return h, c5

def _coord_descent(h0, n_moves=20000, scale0=0.01):
    """Random pairwise coordinate moves."""
    n = len(h0)
    rng = np.random.RandomState(42)
    h = _project(h0.copy())
    c5 = _eval(h)
    scale = scale0
    stuck = 0
    for it in range(n_moves):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-15:
            c5 = c5n
            stuck = 0
            scale = min(scale * 1.1, 0.05)
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            stuck += 1
            if stuck > 1000:
                scale *= 0.5
                stuck = 0
                if scale < 1e-8:
                    break
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(_corr(h).max() * 2.0 / n)

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _corr_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        M = abs(lag)
        wk = w[k]
        if M == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag >= 0:
            g[M:] += wk * (1.0 - h[:n - M])
            g[:n - M] -= wk * h[M:]
        else:
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _corr_grad(h, w)

def _project(h):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for i in range(n):
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
    return h

def _con_fun(x):
    return x.sum() - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def _slsqp(h0, alpha, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    def _fo(x):
        return _smooth_obj(x, alpha)

    def _jo(x):
        return _smooth_grad(x, alpha)

    try:
        res = minimize(_fo, h0, jac=_jo, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h2 = _project(res.x)
            if _eval(h2) < _eval(h0) + 1e-12:
                return h2
    except Exception:
        pass
    return _project(h0)

def _subgrad(h0, max_iter, t0=0.005, tmin=1e-7, topk=1):
    n = len(h0)
    h = _project(h0.copy())
    c5 = _eval(h)
    for it in range(max_iter):
        t = t0 * (tmin / t0) ** (it / float(max_iter))
        c = _corr(h)
        if topk >= len(c):
            idx = np.argmax(c)
            w = np.zeros(len(c))
            w[idx] = 1.0
        else:
            top_idx = np.argsort(c)[-topk:]
            w = np.zeros(len(c))
            w[top_idx] = 1.0 / topk
        g = _corr_grad(h, w)
        g = g - g.mean()
        trial = _project(h - t * g)
        c5n = _eval(trial)
        if c5n < c5 - 1e-16:
            h = trial
            c5 = c5n
    return h, c5

def _coord(h0, n_moves, scale0, rng):
    n = len(h0)
    h = _project(h0.copy())
    c5 = _eval(h)
    scale = scale0
    stuck = 0
    for it in range(n_moves):
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-15:
            c5 = c5n
            stuck = 0
            scale = min(scale * 1.08, 0.05)
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            stuck += 1
            if stuck > 800:
                scale *= 0.6
                stuck = 0
                if scale < 5e-7:
                    break
    return h, c5

def _slsqp_direct(h0, alpha, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    def _fo2(x):
        return _smooth_obj(x, alpha)

    def _jo2(x):
        return _smooth_grad(x, alpha)

    try:
        res = minimize(_fo2, h0, jac=_jo2, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return _project(res.x)
    except Exception:
        pass
    return _project(h0)
