                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _lse_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _lse_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s = s - s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(n - 1, 2 * n - 1):
        L = k - (n - 1)
        if w[k] < 1e-14:
            continue
        g[L:] += w[k] * (1.0 - x[:n - L])
        g[:n - L] -= w[k] * x[L:]
    for k in range(0, n - 1):
        M = n - 1 - k
        if w[k] < 1e-14:
            continue
        g[:n - M] += w[k] * (1.0 - x[M:])
        g[M:] -= w[k] * x[:n - M]
    return g

def _slsqp(h, n, alpha, iters):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pairwise_cd(h, n, durations, rng):
    c5c = _eval(h)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    best = h.copy()
    for ds in durations:
        rng.shuffle(pairs)
        improved = False
        for (i, j) in pairs:
            hi, hj = h[i], h[j]
            bd, bc = 0.0, c5c
            for d in [ds, -ds, 0.5 * ds, -0.5 * ds]:
                ni, nj = hi + d, hj - d
                if ni < -1e-10 or ni > 1 + 1e-10 or nj < -1e-10 or nj > 1 + 1e-10:
                    continue
                h[i], h[j] = ni, nj
                cn = _eval(h)
                if cn < bc:
                    bc, bd = cn, d
            if bd:
                h[i], h[j] = hi + bd, hj - bd
                c5c = bc
                improved = True
                if c5c < _eval(best):
                    best = h.copy()
            else:
                h[i], h[j] = hi, hj
        if not improved:
            break
    return h, c5c

def _random_tune(h, n, t_end, rng):
    c5c = _eval(h)
    best = h.copy()
    best_c = c5c
    scale = 0.003
    no_imp = 0
    while time.time() < t_end:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c5c - 1e-15:
            c5c = cn
            no_imp = 0
            scale = min(scale * 1.12, 0.01)
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            no_imp += 1
            if no_imp > 200:
                scale *= 0.7
                no_imp = 0
                if scale < 1e-8:
                    break
    return best, best_c

def _optimize_n(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = _eval(h)
    # Phase 1: SLSQP with geometric alpha schedule
    alphas = [50, 200, 800, 3000, 12000, 50000, 200000, 800000]
    iters_per = max(400, int(800000 / n))
    for a in alphas:
        if time.time() > t_end - 120:
            break
        h = _slsqp(best_h, n, a, iters_per)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    # Phase 2: Multi-start at high alpha
    if time.time() < t_end - 80:
        for _ in range(12):
            if time.time() > t_end - 80:
                break
            sc = 0.004 + 0.02 * rng.rand()
            hp = _repair(best_h + sc * rng.randn(n), n)
            h = _slsqp(hp, n, 200000, 600)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    # Phase 3: Paired coordinate descent
    if time.time() < t_end - 30:
        t_cd = min(time.time() + 40, t_end - 30)
        h2 = best_h.copy()
        h2, c5c = _pairwise_cd(h2, n, [0.1, 0.04, 0.015, 0.006, 0.002, 0.0008], rng)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h2.copy()
    # Phase 4: Random fine-tuning
    if time.time() < t_end - 5:
        h3, c5c3 = _random_tune(best_h, n, t_end - 5, rng)
        if c5c3 < best_c5:
            best_c5 = c5c3
            best_h = h3.copy()
    # Phase 5: Final SLSQP polish
    if time.time() < t_end - 2:
        h4 = _slsqp(best_h, n, 500000, 1500)
        c5 = _eval(h4)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h4.copy()
    return best_h, best_c5

def _make_start(n, seed):
    if n == 70:
        return _repair(np.array(initial_h_values, dtype=float).copy(), n)
    rng = np.random.RandomState(seed)
    x = np.arange(n)
    h = 0.5 + 0.25 * np.sin(2 * np.pi * 2 * x / n + 0.3)
    h = np.clip(h, 0, 1)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Primary: n=70 with initial_h_values
    n70 = np.array(initial_h_values, dtype=float)
    n_pts = len(n70)
    t_end_70 = t0 + budget_s * 0.6
    h70 = _repair(n70, n_pts)
    best70, c5_70 = _optimize_n(h70, n_pts, t_end_70, rng)
    best_h, best_c5, best_n = best70, c5_70, n_pts

    # Secondary: try other n values
    remaining = budget_s - (_el() + 20)
    if remaining > 30:
        other_ns = [50, 60, 90]
        per_n = max(15, remaining / len(other_ns))
        for n_try in other_ns:
            if _el() > budget_s - 20:
                break
            t_end_n = t0 + _el() + per_n * (budget_s * 0.4 / remaining)
            t_end_n = max(t_end_n, t0 + _el() + 15)
            h_init = _make_start(n_try, seed + n_try)
            h_r, c5_r = _optimize_n(h_init, n_try, min(t_end_n, t0 + budget_s - 20), rng)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Final ultra-fine polish on the best
    if _el() < budget_s - 10:
        h_fin = _slsqp(best_h, best_n, 1000000, 2000)
        c5_fin = _eval(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        h_fin2, c5_fin2 = _random_tune(best_h, best_n, t0 + budget_s - 3, rng)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
