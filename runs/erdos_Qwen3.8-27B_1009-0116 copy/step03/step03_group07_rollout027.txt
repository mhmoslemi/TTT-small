```python
def compute_grad_max(h, n, topk=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top-k correlation indices
    top_idx = np.argsort(corr)[::-1][:topk]
    max_val = corr[top_idx[0]]
    
    # Weight by proximity to max (softmax-like)
    weights = np.exp(-10 * (max_val - corr[top_idx]))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for wi, k_idx in zip(weights, top_idx):
        k = k_idx - (n - 1)  # actual shift
        for j in range(n):
            dj = 0.0
            # h[j] appears as h[n+k] when n = j-k
            jp = j - k
            if 0 <= jp < n:
                dj += (1.0 - h[jp])
            # h[j] appears in (1-h[n]) when n = j
            jm = j + k
            if 0 <= jm < n:
                dj -= h[jm]
            grad[j] += wi * dj
    
    return grad, corr
```

def proj_grad_descent(h, n, max_iter=5000, t_end=None):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = 0.08
    for it in range(max_iter):
        if t_end is not None and time.time() > t_end:
            break
        grad, corr = grad_max_corr(h, n, topk=5, softness=35.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = lr
        improved = False
        for _ in range(15):
            h_try = h - step * grad / gnorm
            h_try = repair(h_try, n)
            c5_try = compute_c5(h_try, n)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True
                lr = min(step * 1.1, 0.5)
                break
            step *= 0.5
        if not improved:
            lr *= 0.85
            if lr < 1e-9:
                break
    return h, best_c5

def grad_local_search(h, n, n_rounds=200, t_end=None):
    h = h.copy()
    best = compute_c5(h, n)
    for r in range(n_rounds):
        if t_end is not None and time.time() > t_end:
            break
        grad, corr = grad_max_corr(h, n, topk=3, softness=25.0)
        order = np.argsort(np.abs(grad))[::-1]
        improved = False
        for i in order[:20]:
            if t_end is not None and time.time() > t_end:
                break
            direction = -1.0 if grad[i] > 0 else 1.0
            for j in order[:20]:
                if j == i:
                    continue
                for d in [0.03, 0.015, 0.007, 0.003]:
                    ni = h[i] + direction * d
                    nj = h[j] - direction * d
                    if ni < -0.05 or ni > 1.05 or nj < -0.05 or nj > 1.05:
                        continue
                    oi, oj = h[i], h[j]
                    h[i] = max(0.0, min(1.0, ni))
                    h[j] = max(0.0, min(1.0, nj))
                    h = repair(h, n)
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = oi, oj
                    h = repair(h, n)
                if improved:
                    break
        if not improved and r > 40:
            break
    return h, best

def make_start(n, seed_val, kind):
    rng = np.random.RandomState(seed_val)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.1 * rng.randn(n)
    elif kind == 2:
        f = rng.randint(2, 8)
        ph = rng.uniform(0, 6.283)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 3:
        block = max(2, n // 7)
        h = np.zeros(n)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        h = 0.5 * h + 0.2 * rng.randn(n)
    else:
        h = rng.uniform(0, 1, n)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 777)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    init_h = None
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        if len(init_h) > 0:
            init_n = len(init_h)
            init_h = repair(init_h, init_n)
            best_h = init_h.copy()
            best_c5 = compute_c5(init_h, init_n)
            best_n = init_n
    except NameError:
        pass
    if best_h is None:
        best_h = make_start(70, seed, 0)
        best_c5 = compute_c5(best_h, 70)
        best_n = 70

    h_work = best_h.copy()
    n_p = best_n

    # Phase 1: SLSQP with escalating smoothness (28% budget)
    t1_end = t0 + budget_s * 0.28
    for alpha in [5000, 20000, 80000, 250000]:
        if time.time() > t1_end:
            break
        for topk, fw in [(1, 0.0), (3, 0.005), (5, 0.02)]:
            if time.time() > t1_end:
                break
            h_new = do_slsqp(h_work.copy(), n_p, alpha, topk, fw, 800)
            c5_new = compute_c5(h_new, n_p)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new.copy()
                best_c5 = c5_new
                h_work = h_new
            elif c5_new < compute_c5(h_work, n_p) - 1e-14:
                h_work = h_new

    # Phase 2: L-BFGS-B refinement (8% budget)
    t2_end = t0 + budget_s * 0.36
    for alpha, pen in [(50000, 1000), (200000, 5000)]:
        if time.time() > t2_end:
            break
        h_new = do_lbfgs(h_work.copy(), n_p, alpha, 3, 0.01, pen, 3000)
        c5_new = compute_c5(h_new, n_p)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new

    # Phase 3: Projected gradient descent (22% budget)
    t3_end = t0 + budget_s * 0.58
    h_pg, c5_pg = proj_grad_descent(h_work, n_p, max_iter=10000, t_end=t3_end)
    if c5_pg < best_c5 - 1e-14:
        best_h = h_pg.copy()
        best_c5 = c5_pg
        h_work = h_pg

    # Phase 4: Gradient-guided local search (15% budget)
    t4_end = t0 + budget_s * 0.73
    h_gl, c5_gl = grad_local_search(h_work, n_p, n_rounds=250, t_end=t4_end)
    if c5_gl < best_c5 - 1e-14:
        best_h = h_gl.copy()
        best_c5 = c5_gl
        h_work = h_gl

    # Phase 5: Second gradient pass on best (7% budget)
    t5_end = t0 + budget_s * 0.80
    h_pg2, c5_pg2 = proj_grad_descent(best_h.copy(), best_n, max_iter=5000, t_end=t5_end)
    if c5_pg2 < best_c5 - 1e-14:
        best_h = h_pg2.copy()
        best_c5 = c5_pg2

    # Phase 6: Try alternative n values (8% budget)
    t6_end = t0 + budget_s * 0.88
    for n_alt in [65, 80, 100]:
        if time.time() > t6_end:
            break
        if n_alt == best_n:
            continue
        h_alt = make_start(n_alt, seed + n_alt * 17, 3)
        h_alt = do_slsqp(h_alt, n_alt, 50000, 3, 0.01, 600)
        h_alt, c5_alt = proj_grad_descent(h_alt, n_alt, max_iter=3000, t_end=t6_end)
        if c5_alt < best_c5 - 1e-14:
            best_h = h_alt.copy()
            best_c5 = c5_alt
            best_n = n_alt

    # Phase 7: Final polish (10% budget)
    t7_end = t0 + budget_s * 0.99
    if best_h is not None:
        h_fin = do_slsqp(best_h.copy(), best_n, 300000, 1, 0.0, 1000)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-14:
            best_h = h_fin.copy()
            best_c5 = c5_fin
        h_fin2, c5_fin2 = proj_grad_descent(best_h, best_n, max_iter=4000, t_end=t7_end)
        if c5_fin2 < best_c5 - 1e-14:
            best_h = h_fin2.copy()
            best_c5 = c5_fin2
        h_fin3, c5_fin3 = grad_local_search(best_h, best_n, n_rounds=120, t_end=t0 + budget_s * 0.995)
        if c5_fin3 < best_c5 - 1e-14:
            best_h = h_fin3.copy()
            best_c5 = c5_fin3

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
