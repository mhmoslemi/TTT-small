        h = np.clip(h, 0.0, 1.0)

Actually, I want to add one more idea: **mirror descent / Fenchel dual**.

The problem min max_k g_k(h) s.t. constraints can be reformulated as:
min_h max_{p >= 0, sum p = 1} sum_k p_k * g_k(h) s.t. constraints

The inner max is over a simplex. If we fix p, the inner problem is:
min_h sum_k p_k * g_k(h) s.t. constraints

where g_k(h) = ∑_i h[i](1-h[i-s_k]) is quadratic (non-convex) in h.

This is a non-convex QP, so it's not easier. But we can use SLSQP to solve it for a fixed p, and then update p via gradient ascent on the dual.

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def _sf(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sj(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        so = k - (n - 1)
        if so >= 0:
            g[so:] += wk * (1.0 - x[:n - so])
            g[:n - so] -= wk * x[so:]
        else:
            t = -so
            g[:n - t] += wk * (1.0 - x[t:])
            g[t:] -= wk * x[:n - t]
    return g

def _grad_k(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        t = -s
        g[:n - t] += (1.0 - h[t:])
        g[t:] -= h[:n - t]
    return g

def _slsqp(h, n, alpha, iters):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sf, h, jac=_sj, args=(alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _nm_block(h, n, indices, maxfev):
    """Nelder-Mead on subset of variables."""
    def obj(x_block):
        x = h.copy()
        x[indices] = x_block
        return _eval(_project(x, n))
    x0 = h[indices].copy()
    res = fmin(obj, x0, maxfev=maxfev, full_output=0)
    h[indices] = res
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def _sf(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sj(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14: continue
        so = k - (n - 1)
        if so >= 0:
            g[so:] += wk * (1.0 - x[:n-so])
            g[:n-so] -= wk * x[so:]
        else:
            t = -so
            g[:n-t] += wk * (1.0 - x[t:])
            g[t:] -= wk * x[:n-t]
    return g

def _grad_k(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n-s]); g[:n-s] -= h[s:]
    else:
        t = -s
        g[:n-t] += (1.0 - h[t:]); g[t:] -= h[:n-t]
    return g

def _slsqp(h, n, alpha, iters):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_sf, h, jac=_sj, args=(alpha,), method='SLSQP',
                     bounds=bnds, constraints=cons, options={'maxiter': iters, 'ftol': 1e-19})
        return _repair(r.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _nm_sub(h, n, idx, alpha, maxfev):
    def obj(xb):
        x = h.copy(); x[idx] = np.clip(xb, 0, 1)
        return _sf(_repair(x, n), alpha)
    try:
        res = minimize(obj, h[idx].copy(), method='Nelder-Mead',
                       options={'maxiter': maxfev, 'xatol': 1e-10, 'fatol': 1e-10})
        h[idx] = res.x
    except Exception:
        pass
    return h
