    # Project gradient onto null space of sum constraint
    grad_ns = grad - np.mean(grad)  # remove mean component
    h_new = h - step_size * grad_ns
    return repair(h_new, n)

def compute_grad(h, n, alpha):
    corr = np.correlate(h, 1.0-h, mode='full')
    w = np.exp(alpha * (corr - corr.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for l in range(-(n-1), n):
        m = l + n - 1
        wm = w[m]
        if wm < 1e-15:
            continue
        # term1: (1-h[i-l]) for valid i
        if l >= 0:
            # i-l in [0,n-1] means i in [l, n-1]
            grad[l:] += wm * (1.0 - h[:-l] if l > 0 else 1.0 - h)
            # i+l in [0,n-1] means i in [0, n-1-l]
            if l < n:

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        rem = abs(d)
        for idx in order:
            if rem < 1e-15:
                break
            if d > 0:
                a = min(rem, 1.0 - h[idx])
                h[idx] += a
                rem -= a
            else:
                a = min(rem, h[idx])
                h[idx] -= a
                rem -= a
    return h

def smooth_max_grad(h, n, alpha):
    corr = compute_corr(h)
    w = np.exp(alpha * (corr - np.max(corr)))
    w /= w.sum()
    grad = np.zeros(n)
    for l in range(-(n-1), n):
        m = l + n - 1
        wm = w[m]
        if wm < 1e-12:
            continue
        if l >= 0:
            nl = n - l
            if l > 0:
                grad[l:] += wm * (1.0 - h[:nl])
            else:
                grad[:] += wm * (1.0 - h)
            if nl > 0:
                grad[:nl] -= wm * h[l:]
        else:
            ll = -l
            nl = n - ll
            grad[:nl] += wm * (1.0 - h[ll:])
            grad[ll:] -= wm * h[:nl]
    return grad

def gradient_descent(h, n, alpha, lr_init, max_iters, t_end):
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    h_cur = h.copy()
    lr = lr_init
    stall = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        g = smooth_max_grad(h_cur, n, alpha)
        g_ns = g - np.mean(g)
        h_new = repair(h_cur - lr * g_ns, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            stall = 0
            if it % 200 == 0 and it > 0:
                lr = min(lr * 1.05, 0.3)
        else:
            lr *= 0.5
            stall += 1
            if stall > 50:
                break
    return best_h, best_c5

def targeted_cd(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(n * 8):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sgn in (1, -1):
                    ni = hi + sgn * d
                    nj = hj - sgn * d

                        continue

                    h[j] = min(max(nj, 0), 1)
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:

                        improved = True
                        break
                    h[i], h[j] = hi, hj

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        rem = abs(d)
        for idx in order:
            if rem < 1e-15:
                break
            if d > 0:
                a = min(rem, 1.0 - h[idx])
                h[idx] += a
                rem -= a
            else:
                a = min(rem, h[idx])
                h[idx] -= a
                rem -= a
    return h

def smooth_max_grad(h, n, alpha):
    corr = compute_corr(h)
    w = np.exp(alpha * (corr - np.max(corr)))
    w /= w.sum()
    grad = np.zeros(n)
    for l in range(-(n - 1), n):
        m = l + n - 1
        wm = w[m]
        if wm < 1e-12:
            continue
        if l >= 0:
            nl = n - l
            grad[l:] += wm * (1.0 - h[:nl])
            grad[:nl] -= wm * h[l:]
        else:
            ll = -l
            nl = n - ll
            grad[:nl] += wm * (1.0 - h[ll:])
            grad[ll:] -= wm * h[:nl]
    return grad

                continue
