import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(-diff, h[idx])
            h[idx] += a if diff > 0 else -a
            diff -= a if diff > 0 else -a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_smmax(h, n, alpha):
    c = _corr(h)
    cm = c.max()
    w = np.exp(np.clip(alpha * (c - cm), -500, 500))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        kk = k - (n - 1)
        if kk >= 0:
            grad[kk:] += wk * (1.0 - h[:n - kk])
            grad[:n - kk] -= wk * h[kk:]
        else:
            grad[:n + kk] += wk * (1.0 - h[-kk:])
            grad[-kk:] -= wk * h[:n + kk]
    return grad

def _proj_grad(h, n, alpha, lr, niters, mom):
    v = np.zeros(n)
    cur_obj = _smooth_max(_corr(h), alpha)
    for _ in range(niters):
        g = _grad_smmax(h, n, alpha)
        v = mom * v - lr * g
        h2 = _repair(np.clip(h + v, 0.0, 1.0), n)
        obj2 = _smooth_max(_corr(h2), alpha)
        if obj2 < cur_obj - 1e-14:
            h = h2
            cur_obj = obj2
        else:
            v *= 0.7
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    i1 = p - kk
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += 1.0 - h[i1[m1]]
    i2 = p + kk
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _sweep(h, n, tlimit, nc, nb):
    corr = _corr(h)
    imp = False
    order = np.random.permutation(n)
    nmax = float(np.max(corr))
    for ai in range(n):
        i = order[ai]
        for aj in range(ai + 1, n):
            if time.time() > tlimit:
                return h, imp
            j = order[aj]
            dmin = max(-h[i], h[j] - 1.0)
            dmax = min(1.0 - h[i], h[j])
            if dmax - dmin < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(dmin, dmax, nc)
            vals = np.max(corr[None, :] + ds[:, None] * gd, axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= nmax - 1e-13:
                continue
            lo = max(dmin, ds[max(0, bi - 1)])
            hi = min(dmax, ds[min(nc - 1, bi + 1)])
            for _ in range(nb):
                mid = 0.5 * (lo + hi)
                if np.max(corr + mid * gd) <= np.max(corr + hi * gd):
                    hi = mid
                else:
                    lo = mid
            dB = 0.5 * (lo + hi)
            h[i] += dB
            h[j] -= dB
            corr = corr + dB * gd
            nmax = float(np.max(corr))
            imp = True
    return h, imp

def _slsqp(h, n, target, bounds, alpha, mx):
    try:
        r = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                     bounds=bounds,
                     constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (target,)}],
                     options={'maxiter': mx, 'ftol': 1e-15, 'maxls': 50})
        return _repair(r.x, n)
    except Exception:
        return h

def _trust_constr(h, n, target, bounds, alpha, mx):
    try:
        r = minimize(_obj_smax, h, args=(alpha,), method='trust-constr',
                     bounds=bounds,
                     constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (target,)}],
                     options={'maxiter': mx, 'gtol': 1e-10, 'xtol': 1e-12})
        return _repair(r.x, n)
    except Exception:
        return h

def _optimize(h, n, tlimit):

    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h = _repair(h.copy(), n)
    c5 = _c5(h, dx)

    # Phase A: Projected gradient with annealing alpha
    sched = [(8, 0.02, 400), (20, 0.012, 400), (60, 0.006, 350),
             (150, 0.003, 300), (400, 0.0015, 250), (1000, 0.0008, 200)]
    for alpha, lr, nit in sched:
        if time.time() > tlimit - 50:
            break
        h = _proj_grad(h, n, alpha, lr, nit, 0.85)
        cn = _c5(h, dx)
        if cn < c5:
            c5 = cn

    # Phase B: SLSQP
    if time.time() < tlimit - 30:
        h2 = _slsqp(h, n, target, bounds, 250, 350)
        cn = _c5(h2, dx)
        if cn < c5:
            h, c5 = h2, cn

    # Phase C: Coordinate sweeps
    stall = 0
    while time.time() < tlimit - 12:
        chunk = min(22.0, tlimit - 12 - time.time())
        if chunk < 2:
            break
        h, imp = _sweep(h, n, time.time() + chunk, 22, 32)
        cn = _c5(h, dx)
        if cn < c5 - 1e-14:
            c5 = cn
            stall = 0
        else:
            stall += 1
        if stall >= 3 and time.time() < tlimit - 20:

            cn = _c5(h, dx)
            if cn < c5:
                c5 = cn
            stall = 0

    # Phase D: Final polish
    if time.time() < tlimit - 3:
        h2 = _slsqp(h, n, target, bounds, 3000, 500)
        cn = _c5(h2, dx)
        if cn < c5:

        h2 = _trust_constr(h, n, target, bounds, 2000, 200)
        cn = _c5(h2, dx)
        if cn < c5:
            h, c5 = h2, cn

    return _repair(h, n), c5
