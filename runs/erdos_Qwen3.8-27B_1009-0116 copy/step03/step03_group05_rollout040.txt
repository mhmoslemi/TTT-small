import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] = np.minimum(h[mask] + diff / mask.sum(), 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] = np.maximum(h[mask] + diff / mask.sum(), 0.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            for i in np.where(h < 1.0)[0]:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
                if abs(diff) < 1e-15:
                    break
        else:
            for i in np.where(h > 0.0)[0]:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
                if abs(diff) < 1e-15:
                    break
    return h

def _grad_lag(h, L, n):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += 1.0 - h[M:]
        g[M:] -= h[:n - M]
    return g

def _topk_grad(h, k):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for ki in idx:
        g += _grad_lag(h, ki - (n - 1), n)
    return g / k, idx

def _make_pattern(n, nc, gap_frac):
    h = np.zeros(n)
    period = 2.0 / nc
    for c in range(nc):
        s = c * period + period * gap_frac * 0.5
        e = (c + 1) * period - period * gap_frac * 0.5
        i0 = max(0, int(s * n / 2.0))
        i1 = min(n, int(e * n / 2.0) + 1)
        h[i0:i1] = 1.0
    return _repair(h, n)

def _soft_obj(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return m + np.sum(np.exp(s)) / (alpha * len(c)) + penalty

def _soft_grad(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    ew = np.exp(s)
    pw = ew / (ew.sum() + 1e-30)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= x[:n - M]
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

    target = n / 2.0

    best = h.copy()

            break
        try:
            res = minimize(
                _soft_obj, best.copy(),

                jac=_soft_grad,
