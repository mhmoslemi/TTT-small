import numpy as np
from scipy.optimize import minimize
import time

def compute_overlap(h, L, dx):
    n = len(h)
    if L >= 0:
        m = n - L
        if m <= 0: return 0.0
        return np.dot(h[:m], 1.0 - h[L:]) * dx
    else:
        M = -L
        m = n - M
        if m <= 0: return 0.0
        return np.dot(h[M:], 1.0 - h[:m]) * dx

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_overlap(h, L, dx):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        m = n - L
        if m <= 0: return g
        # h[j] * (1-h[j+L]) for j in [0, m-1]
        g[:m] += (1.0 - h[L:L+m])  # (1-h[j+L]) for j in [0,m-1]
        g[L:] -= h[:m]  # -h[j-L] for j in [L, n-1], j-L in [0, m-1]
    else:
        M = -L
        m = n - M
        if m <= 0: return g
        # h[j] * (1-h[j-M]) for j in [M, n-1]
        g[M:] += (1.0 - h[:m])  # (1-h[j-M]) for j in [M, n-1], j-M in [0, m-1]
        g[:m] -= h[M:]  # -h[j+M] for j in [0, m-1], j+M in [M, n-1]
    return g * dx

def repair(h, n, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    # Proportional scaling then clip, then fix
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

        return float(np.dot(h[L:L+m], 1.0 - h[:m]) * dx) if m > 0 else 0.0
    M = -L

    return float(np.dot(h[:m], 1.0 - h[M:M+m]) * dx) if m > 0 else 0.0

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        m = n - L
        if m <= 0:
            return g
        g[L:L+m] += 1.0 - h[:m]
        g[:m] -= h[L:L+m]
    else:
        M = -L
        m = n - M
        if m <= 0:
            return g
        g[:m] += 1.0 - h[M:M+m]
        g[M:M+m] -= h[:m]

    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
