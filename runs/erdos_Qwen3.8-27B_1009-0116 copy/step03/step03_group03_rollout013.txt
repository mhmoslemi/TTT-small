def _to_binary(h, n):
    """Convert to binary pattern maintaining sum constraint"""
    target = n / 2.0
    # Sort by h value, set top target to 1, rest to 0
    thresh = np.sort(h)[int(target)]
    hb = (h >= thresh).astype(float)
    # Fix sum
    s = hb.sum()
    if s > target:
        # Flip some 1s to 0s (those closest to threshold)
        ones = np.where(hb == 1)[0]
        ones_sorted = ones[np.argsort(h[ones])]
        for i in ones_sorted:

                hb[i] = 0
                s -= 1

                s += 1
    return hb

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_obj(x, alpha, target):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    pen = 800.0 * (np.sum(x) - target) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + pen

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _lag_grad(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += (1.0 - h[-k:])
        g[-k:] -= h[:n + k]
    return g

def _binary_round(h, n):
    target = int(n / 2)
    idx = np.argsort(-h)
    hb = np.zeros(n)
    for i in range(target):
        hb[idx[i]] = 1.0
    return hb
