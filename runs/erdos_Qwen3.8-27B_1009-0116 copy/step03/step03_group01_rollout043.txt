import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _correlate_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    center = n - 1
    if k >= center:
        L = k
        m = n - L
        if m <= 0:
            return g
        g[:m] += (1.0 - h[L:])
        g[L:] += (-h[:m])
    else:
        L = center - k
        m = n - L
        if m <= 0:
            return g
        g[L:] += (1.0 - h[:m])
        g[:m] += (-h[L:])
    return g

def _subgrad_opt(h, n, lr0, steps, topk=3):
    h = h.copy()
    c5 = _eval(h)
    lr = lr0
    for s in range(steps):
        c = _correlate_full(h)
        # Find top-k lags
        idx = np.argsort(c)[::-1][:topk]
        # Average gradient of top-k
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(h, int(k))
        g /= topk
        # Center the gradient to preserve sum
        g -= g.mean()
        # Step
        h_new = h - lr * g
        h_new = _repair(h_new, n)
        c5_new = _eval(h_new)
        if c5_new < c5 - 1e-18:
            h = h_new
            c5 = c5_new
            lr = min(lr * 1.05, lr0 * 10)
        else:
            lr *= 0.8
            if lr < 1e-8:
                break
    return h, c5

def _sf(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sj(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_lag(x, k)
    return g

def _slsqp(h, n, alpha, iters):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sf, h, jac=_sj, args=(alpha,),

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _correlate_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    center = n - 1
    if k >= center:
        L = k
        m = n - L
        if m <= 0:
            return g
        g[:m] += (1.0 - h[L:])
        g[L:] += (-h[:m])
    else:
        L = center - k
        m = n - L
        if m <= 0:
            return g
        g[L:] += (1.0 - h[:m])
        g[:m] += (-h[L:])
    return g

def _subgrad(h, n, steps, topk, lr_init):
    h = h.copy()
    c5 = _eval(h)
    lr = lr_init
    plateau = 0
    for s in range(steps):
        c = _correlate_full(h)
        top_idx = np.argsort(c)[::-1][:topk]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(h, int(k))
        g /= topk
        g -= g.mean()
        h_new = _repair(h - lr * g, n)
        c5_new = _eval(h_new)
        if c5_new < c5 - 1e-17:
            h = h_new
            c5 = c5_new
            lr = min(lr * 1.03, 0.8)
            plateau = 0
        else:
            lr *= 0.85
            plateau += 1
            if plateau > 200:
                break
    return h, c5

def _sf(x, a):
    c = _correlate_full(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sj(x, a):
    n = len(x)
    c = _correlate_full(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_lag(x, k)
    return g

def _slsqp(h, n, alpha, iters):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sf, h, jac=_sj, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _coord_fine(h, n, iters, seed, scale_init):
    h = h.copy()
    c5c = _eval(h)
    rng = np.random.RandomState(seed)
    scale = scale_init
    stuck = 0
    for _ in range(iters):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
            stuck = 0
            scale = min(scale * 1.15, 0.02)
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            stuck += 1
            if stuck > 300:
                scale *= 0.6
                stuck = 0
                if scale < 1e-7:
                    break
    return h, c5c

def _gen_binary(n, rng):
    h = rng.rand(n)
    h = (h > 0.5).astype(float)
    excess = h.sum() - n / 2
    if excess > 0:

    elif excess < 0:
        zeros = np.where(h == 0.0)[0]
        h[rng.choice(zeros, int(-excess), replace=False)] = 1.0
    return h
