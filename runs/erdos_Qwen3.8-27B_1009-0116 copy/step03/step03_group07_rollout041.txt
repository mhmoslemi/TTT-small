    stall = 0
    for sw in range(n_sweeps):
        if time.time() > t_limit:
            break
        imp = False
        ntrials = 4 * n
        for _ in range(ntrials):
            if time.time() > t_limit:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h2, c2 = ls_exchange(h, n, i, j, t_limit)
            if c2 < best - 1e-15:
                h, best, imp, stall = h2, c2, True, 0
            else:
                stall += 1
        if not imp:
            stall += 10
            if stall > 40:
                break
    return h, best

def perturb_h(h, n, rng, scale):
    h2 = h.copy()
    a = rng.randint(n)
    b = rng.randint(n)
    while b == a:
        b = rng.randint(n)
    h2[a] += scale
    h2[b] -= scale
    return repair(h2, n)

def make_start(n, seed_val, kind):
    rng = np.random.RandomState(seed_val)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.1 * rng.randn(n)
    elif kind == 2:
        f = rng.randint(2, 8)
        ph = rng.uniform(0, 6.283)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * f * np.arange(n) / float(n) + ph)
    elif kind == 3:
        f1, f2 = rng.randint(2, 6), rng.randint(5, 12)
        p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
        x = np.arange(n) / float(n)
        h = 0.5 + 0.25 * np.sin(2*np.pi*f1*x + p1) + 0.15 * np.sin(2*np.pi*f2*x + p2)
    elif kind == 4:
        h = 0.5 + 0.3 * np.cos(np.pi * np.arange(n) / float(n)) ** 3
    else:
        h = rng.uniform(0, 1, n)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.RandomState(seed + 12345)
    best_h, best_c5, best_n = None, np.inf, 70

    init_h = None
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        pass
    if init_h is not None and len(init_h) > 10:
        best_n = len(init_h)
        best_h = repair(init_h, best_n)
        best_c5 = compute_c5(best_h, best_n)
    else:
        best_h = make_start(70, seed, 2)
        best_c5 = compute_c5(best_h, 70)
        best_n = 70

    n_p = best_n
    h_w = best_h.copy()

    # Phase 1: SLSQP refinement
    for alpha in [1000, 10000, 80000, 300000]:
        if elapsed() > budget_s * 0.22:
            break
        h_new = slsqp_opt(h_w.copy(), n_p, alpha, 700, 1e-18)
        c5_new = compute_c5(h_new, n_p)
        if c5_new < best_c5 - 1e-14:
            best_h, best_c5 = h_new.copy(), c5_new
        if c5_new < compute_c5(h_w, n_p) - 1e-14:
            h_w = h_new

    # Phase 2: Intensive CD with line search
    t_cd2 = t0 + budget_s * 0.55
    h_2, c5_2 = cd_search(h_w, n_p, 800, t_cd2, rng)
    if c5_2 < best_c5 - 1e-14:
        best_h, best_c5 = h_2.copy(), c5_2

    # Phase 3: SLSQP + CD on best, with perturbation restarts
    t_p3 = t0 + budget_s * 0.78
    for trial in range(6):
        if elapsed() > t_p3:
            break
        if trial == 0:
            h_cur = best_h.copy()
        else:
            h_cur = perturb_h(best_h, best_n, rng, 0.015 + 0.01 * trial)
        h_s = slsqp_opt(h_cur, best_n, 150000, 500, 1e-19)
        c5_s = compute_c5(h_s, best_n)
        h_cd, c5_cd = cd_search(h_s, best_n, 150, t_p3, rng)
        c5_pick = min(c5_s, c5_cd)
        if c5_pick < best_c5 - 1e-14:
            best_h = h_cd if c5_cd <= c5_s else h_s
            best_c5 = c5_pick

    # Phase 4: Try nearby n values
    t_p4 = t0 + budget_s * 0.91
    for dn in [-4, -2, 2, 4, -8, 8]:
        n_alt = best_n + dn
        if n_alt < 30 or n_alt > 200 or elapsed() > t_p4:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_alt)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = repair(h_interp, n_alt)
        h_a = slsqp_opt(h_interp, n_alt, 100000, 500, 1e-18)
        c5_a = compute_c5(h_a, n_alt)
        h_a2, c5_a2 = cd_search(h_a, n_alt, 80, t_p4, rng)
        if c5_a2 < best_c5 - 1e-14:
            best_h, best_c5, best_n = h_a2.copy(), c5_a2, n_alt
        elif c5_a < best_c5 - 1e-14:
            best_h, best_c5, best_n = h_a.copy(), c5_a, n_alt

    # Phase 5: Final CD + SLSQP polish
    t_end = t0 + budget_s * 0.985
    h_fin, c5_fin = cd_search(best_h, best_n, 500, t_end, rng)
    if c5_fin < best_c5 - 1e-14:
        best_h, best_c5 = h_fin.copy(), c5_fin
    h_pol = slsqp_opt(best_h, best_n, 500000, 600, 1e-20)
    c5_pol = compute_c5(h_pol, best_n)
    if c5_pol < best_c5 - 1e-14:
        best_h, best_c5 = h_pol.copy(), c5_pol
    h_last, c5_last = cd_search(best_h, best_n, 150, t0 + budget_s * 0.995, rng)
    if c5_last < best_c5 - 1e-14:
        best_h, best_c5 = h_last.copy(), c5_last

    if best_h is None:
        best_n = 70
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
