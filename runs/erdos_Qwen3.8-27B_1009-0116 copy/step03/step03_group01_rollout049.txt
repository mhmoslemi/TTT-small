    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * 2.0 / len(x)

    c = _correl(x)
    m = c.max()

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': 1e-19})
