def _repair(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = (1.0 - h)
            if room.max() < 1e-14:
                break
            h += diff / np.sum(room > 1e-14) * np.where(room > 1e-14, 1.0, 0.0)
        else:
            room = h
            if room.max() < 1e-14:
                break
            h += diff / np.sum(room > 1e-14) * np.where(room > 1e-14, 1.0, 0.0)
        h = np.clip(h, 0, 1)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-12:
                break

    return (best_h, best_c5, best_n)
