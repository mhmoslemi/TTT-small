    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h.copy(), _compute_c5(h, dx)
    try:
        res = minimize(
            lambda x: _topk_obj(x, target, k, pen), h,
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8}
        )
        hc = _repair(res.x, n)
        c5 = _compute_c5(hc, dx)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5
    except Exception:
        pass
    return best_h, best_c5

def _cobyla_opt(h, n, dx, maxiter=2000):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [lambda x: target - np.sum(x), lambda x: np.sum(x) - target]
    try:
        res = minimize(
            lambda x: _true_max_obj(x, target, 500.0), h,
            method='COBYLA', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'rhobeg': 0.01, 'catol': 1e-12}
        )
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

def _perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        block = rng.randint(3, 10)
        i = rng.randint(0, n - block)
        hp[i:i+block] = np.roll(hp[i:i+block], rng.randint(1, block))
    elif kind == 1:
        f = rng.randint(1, 5)
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.03, 0.08)
        hp += amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.02, n)
    elif kind == 3:
        j = rng.randint(n // 3, 2 * n // 3)
        hp = np.concatenate([h[j:], h[:j]])
    else:
        f1, f2 = rng.randint(1, 4), rng.randint(1, 4)
        amp = rng.uniform(0.02, 0.06)
        t = np.arange(n) / n
        hp += amp * np.sin(2*np.pi*f1*t + rng.uniform(0,6)) * np.sin(2*np.pi*f2*t + rng.uniform(0,6))
    return _repair(hp, n)

def _fine_local(h, n, dx, rng, deadline, n_tries=60000):
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per_scale = max(400, n_tries // len(scales))
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            h[i], h[j] = ni, nj
            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = ni - d, nj + d
        if not improved and scale < 0.001:
            break
    return h, best_c5

def _block_local(h, n, dx, rng, deadline, n_iters=5000):
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    for bs in [2, 3]:
        for _ in range(n_iters // 2):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(0, n - bs)
            j = rng.randint(0, n - bs)
            if abs(i - j) < bs:
                continue
            eps = rng.uniform(0.002, 0.01) * (1.0 if rng.random() < 0.5 else -1.0)
            ok = True
            for di in range(bs):
                if h[i+di] + eps < 0 or h[i+di] + eps > 1:
                    ok = False
                if h[j+di] - eps < 0 or h[j+di] - eps > 1:
                    ok = False
            if not ok:
                continue
            h[i:i+bs] += eps
            h[j:j+bs] -= eps
            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
            else:
                h[i:i+bs] -= eps
                h[j:j+bs] += eps
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remain():
        return budget_s - (time.time() - t_start)

    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    # Phase 1: L-BFGS smooth max warmup (50s)
    tEnd = min(t_start + 50, t_start + budget_s - 700)
    for alpha in [100, 500, 2000]:
        if time.time() >= tEnd:
            break
        for pen in [50, 500]:
            h_t, c5_t = _lbfgs_smooth(best_h, n, dx, alpha, pen, maxiter=250)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 2: SLSQP true max (200s)
    tEnd = min(time.time() + 200, t_start + budget_s - 400)
    for _ in range(4):
        if time.time() >= tEnd or remain() < 400:
            break
        h_t, c5_t = _slsqp_true(best_h, n, dx, maxiter=600)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 3: Top-k SLSQP (150s)
    if remain() > 400:
        tEnd = time.time() + min(150, remain() - 350)
        for k in [5, 10, 20, 50]:
            if time.time() >= tEnd:
                break
            h_t, c5_t = _slsqp_topk(best_h, n, dx, k, maxiter=400)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 4: COBYLA on true max (120s)
    if remain() > 350:
        tEnd = time.time() + min(120, remain() - 300)
        h_t, c5_t = _cobyla_opt(best_h, n, dx, maxiter=1500)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 5: Multi-restart (150s)
    if remain() > 300:
        tEnd = time.time() + min(150, remain() - 250)
        for r in range(12):
            if time.time() >= tEnd or remain() < 250:
                break
            start = _perturb(best_h, n, rng, r % 5)
            h_t, c5_t = _lbfgs_topk(start, n, dx, 10, 100, maxiter=200)
            h_t, c5_t = _slsqp_topk(h_t, n, dx, 8, maxiter=250)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 6: Block local (100s)
    if remain() > 200:
        tEnd = time.time() + min(100, remain() - 150)
        h_t, c5_t = _block_local(best_h, n, dx, rng, tEnd, 4000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 7: SLSQP true max polish (80s)
    if remain() > 120:
        tEnd = time.time() + min(80, remain() - 80)
        for _ in range(3):
            if time.time() >= tEnd:
                break
            h_t, c5_t = _slsqp_true(best_h, n, dx, maxiter=500)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 8: Fine local search (70s)
    if remain() > 80:
        tEnd = time.time() + min(70, remain() - 40)
        h_t, c5_t = _fine_local(best_h, n, dx, rng, tEnd, 50000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 9: Final SLSQP + fine (20s)
    if remain() > 25:
        h_t, c5_t = _slsqp_true(best_h, n, dx, maxiter=300)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
    if remain() > 10:
        tEnd = time.time() + max(2, min(8, remain() - 2))
        h_t, c5_t = _fine_local(best_h, n, dx, rng, tEnd, 15000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
