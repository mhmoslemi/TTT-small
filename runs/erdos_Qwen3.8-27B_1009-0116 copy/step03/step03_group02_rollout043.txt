    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _g_vec(h, p, n):
    kk = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)

        else:

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _optimal_d(G, c, lo, hi):
    """Minimize max_k(c[k] + d*G[k]) for d in [lo, hi]."""
    v_lo = float(np.max(c + lo * G))
    v_hi = float(np.max(c + hi * G))

        dc = c[None, :] - c[:, None]

            bi = int(np.argmin(vals))
            if vals[bi] < best_v - 1e-15:
                best_v = float(vals[bi])
                best_d = float(tv[bi])
    return best_d, best_v

def _sweep_round(h, n, corr, time_limit, order):

    improved = False
    n_imp = 0
    for ai in range(len(order)):

            break

            j = order[bj]
            hi_p, hj_p = h[i], h[j]

            v_lo = float(np.max(corr + d_min * G))

                h[j] = hj_p - best_d
                corr = corr + best_d * G
                improved = True
                n_imp += 1
    return h, corr, improved, n_imp

                    # Quick sweep at new resolution
                    corr_t = _corr(best_h)
                    ord_t = np.random.permutation(n_try)
                    best_h, corr_t, _, _ = _sweep_round(best_h, n_try, corr_t, time.time() + 40, ord_t)
                    best_h = _repair(best_h, n_try)
                    best_c5 = _c5(best_h, best_dx)

        # Perturbation when stalled
        if stall >= 4 and time.time() - t0 < budget_s - 180:
            pscale = min(0.06, 0.015 + 0.01 * stall)
            h_pert = best_h + pscale * np.random.randn(best_n)
            h_pert = _repair(h_pert, best_n)
            hc = _slsqp_step(h_pert, best_n, best_n / 2.0, bounds0, 400, 200)
            hc_corr = _corr(hc)
            ord2 = np.random.permutation(best_n)
            hc, hc_corr, _, _ = _sweep_round(hc, best_n, hc_corr, time.time() + 35, ord2)
            hc = _repair(hc, best_n)
            c5_t = _c5(hc, best_dx)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                best_h = hc.copy()
            stall = 0

        # Fourier restart every 25 rounds when very stalled
        if stall >= 10 and total_rounds % 25 == 0 and time.time() - t0 < budget_s - 200:
            h_f = _fourier_init(best_n)
            hc = _slsqp_step(h_f, best_n, best_n / 2.0, bounds0, 300, 300)
            hc_corr = _corr(hc)
            ord3 = np.random.permutation(best_n)
            hc, hc_corr, _, _ = _sweep_round(hc, best_n, hc_corr, time.time() + 40, ord3)
            hc = _repair(hc, best_n)
            c5_t = _c5(hc, best_dx)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                best_h = hc.copy()
            stall = 0

    # Phase 3: Final multi-round sweep
    if time.time() - t0 < budget_s - 30:
        for _ in range(4):
            if time.time() - t0 > budget_s - 15:
                break
            h = best_h.copy()
            corr = _corr(h)
            order = np.random.permutation(best_n)
            h, corr, _, _ = _sweep_round(h, best_n, corr, time.time() + 25, order)
            h = _repair(h, best_n)
            c5_t = _c5(h, best_dx)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                best_h = h.copy()

    # Phase 4: SLSQP polish
    if time.time() - t0 < budget_s - 5:
        for alpha in [800, 2000]:
            if time.time() - t0 > budget_s - 3:
                break
            hc = _slsqp_step(best_h, best_n, best_n / 2.0, bounds0, alpha, 500)
            c5_t = _c5(hc, best_dx)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                best_h = hc.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_dx)
    return (best_h, best_c5, best_n)
