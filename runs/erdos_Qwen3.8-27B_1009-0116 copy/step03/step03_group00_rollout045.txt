            best_c5 = c5_f
            best_h = h_f
        # Ultra-fine
        if elapsed() < budget_s - 8:
            h_u = best_h.copy()
            cu = best_c5
            tiny = np.array([5e-5, 1e-4, 2e-4, -5e-5, -1e-4, -2e-4])
            while time.time() < t0 + budget_s - 5:
                i = rng.randint(n_f)
                j = rng.randint(n_f)
                if i == j:
                    continue
                d = tiny[rng.randint(len(tiny))]
                hi, hj = h_u[i], h_u[j]
                ni, nj = hi + d, hj - d
                if 0 < ni < 1 and 0 < nj < 1:
                    h_u[i], h_u[j] = ni, nj
                    c = _c5(h_u)
                    if c < cu:
                        cu = c
                    else:
                        h_u[i], h_u[j] = hi, hj
            if cu < best_c5:
                best_c5 = cu
                best_h = h_u

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5(best_h)
    return (best_h, best_c5, len(best_h))
