import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

    s = np.clip(alpha * (c - m), -400, 400)

             'jac': lambda x: np.ones(n)}]
    return minimize(_smooth_obj, h_init,
                    jac=lambda x: _smooth_grad(x, n, alpha),
                    args=(n, alpha),
                    bounds=bounds, constraints=cons,
                    method='SLSQP',
                    options={'maxiter': maxiter, 'ftol': 1e-18})

def _grad_step(h, n, lr):
    """One step of projected gradient descent on the true max objective."""
    c = _corr(h) * (2.0 / n)
    # Subgradient of max: gradient of the argmax component
    k_max = np.argmax(c)
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0 - h[:n-lag]
        g[:n-lag] -= h[lag:]
    else:
        L = -lag
        g[:n-L] += 1.0 - h[L:]
        g[L:] -= h[:n-L]
    g *= 2.0 / n
    # Projected gradient step
    h_new = h - lr * g
    # Project to [0,1] and fix sum
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _repair(h_new, n)
    return h_new

    h_new = np.interp(x_new, x_old, h_ext)
    return _repair(h_new, n_new)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 350)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad(x, n, alpha):
    dxv = 2.0 / n
    c = _corr(x) * dxv
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 350)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-12:
            continue

            g += wi * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wi * (1.0 - x[:n - lag])
            g[:n - lag] -= wi * x[lag:]
        else:
            L = -lag
            g[:n - L] += wi * (1.0 - x[L:])
            g[L:] -= wi * x[:n - L]
