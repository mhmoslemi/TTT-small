def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def smooth_max(corr, alpha):
    max_c = np.max(corr)
    s = alpha * (corr - max_c)
    s = np.clip(s, -500, 0)
    return max_c + np.log(np.sum(np.exp(s))) / alpha

def obj_slsqp(h, n, alpha, topk, fw):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = smooth_max(corr, alpha)
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        m += fw * np.sum((sc - sc.mean())**2)
    return m

def do_slsqp(h, n, alpha, topk, fw, maxiter, t_end):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def obj(x):
        return obj_slsqp(x, n, alpha, topk, fw)
    def con(x):
        return x.sum() - target
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.fun < obj_slsqp(h, n, alpha, topk, fw):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

    target = n / 2.0

            h_p[i] = min(1.0, h_p[i] + eps)

            h_m[i] = max(0.0, h_m[i] - eps)

            if f_new < best_f - 1e-15:
                h = h_new
                best_h = h.copy()
                best_f = f_new
                break
    return best_h

    steps = [0.05, 0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

        if w[idx] < 1e-14:
            continue
        jo = jp - off
        jp_off = jp + off
        # dcorr/dh[j] = (1-h[j-off])*[valid] - h[j+off]*[valid]
        contrib = np.zeros(n)
        mask_o = (jo >= 0) & (jo < n)
        mask_p = (jp_off >= 0) & (jp_off < n)
        contrib[mask_o] += 1 - h[jo[mask_o]]
        contrib[mask_p] -= h[jp_off[mask_p]]
        grad += w[idx] * contrib
    grad /= alpha
    return val, grad, corr

    target = n / 2.0

    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)

            c5n = compute_c5(hn, np_)
            if c5n < best_c5 - 1e-15:
                best_h, best_c5, hw = hn, c5n, hn

    # Phase 2: L-BFGS + Projected gradient (18%)
    t2 = t0 + budget_s * 0.38
    if time.time() < t2:
        for a in [5000, 20000, 60000]:
            if time.time() > t2:
                break
            for pen in [2000, 20000]:
                if time.time() > t2:
                    break
                hn = lbfgs_step(hw, np_, a, 3, 0.01, pen, 5000, t2)
                c5n = compute_c5(hn, np_)
                if c5n < best_c5 - 1e-15:
                    best_h, best_c5, hw = hn, c5n, hn
        if time.time() < t2:
            for a in [5000, 30000, 100000]:
                if time.time() > t2:
                    break
                hn = proj_grad(hw, np_, a, 400, t2)
                c5n = compute_c5(hn, np_)
                if c5n < best_c5 - 1e-15:
                    best_h, best_c5, hw = hn, c5n, hn

    # Phase 3: Long coordinate descent (30%)
    t3 = t0 + budget_s * 0.68
    hc, c5c = coord_desce(hw, np_, 900, t3, rng)
    if c5c < best_c5 - 1e-15:
        best_h, best_c5 = hc, c5c

    # Phase 4: SLSQP polish + coord (15%)
    t4 = t0 + budget_s * 0.83
    if time.time() < t4:
        for a in [60000, 150000]:
            if time.time() > t4:
                break
            hf = slsqp_step(best_h, best_n, a, 5, 0.015, 1500, t4)
            c5f = compute_c5(hf, best_n)
            if c5f < best_c5 - 1e-15:
                best_h, best_c5 = hf, c5f
        if time.time() < t4:
            hf2, c5f2 = coord_desce(best_h, best_n, 500, t4, rng)
            if c5f2 < best_c5 - 1e-1
            :
                best_h, best_c5 = hf2, c5f2

    # Phase 5: Alternative n values (10%)
    t5 = t0 + budget_s * 0.93
    if time.time() < t5:
        for na in [64, 72, 80, 90, 100]:
            if time.time() > t5 or na == best_n:
                continue
            for kk in range(2):
                if time.time() > t5:
                    break
                ha = gen_start(na, seed + na * 7 + kk, kk % 4)
                ha = slsqp_step(ha, na, 15000, 3, 0.01, 600, t5)
                ha, c5a = coord_desce(ha, na, 200, t5, rng)
                if c5a < best_c5 - 1e-15:
                    best_h, best_c5, best_n = ha, c5a, na

    # Phase 6: Final polish (5%)
    t6 = t0 + budget_s * 0.998
    if best_h is not None and time.time() < t6:
        hp = proj_grad(best_h, best_n, 80000, 200, t6)
        c5p = compute_c5(hp, best_n)
        if c5p < best_c5 - 1e-15:
            best_h, best_c5 = hp, c5p
        if time.time() < t6:
            hp2, c5p2 = coord_desce(best_h, best_n, 200, t6, rng)
            if c5p2 < best_c5 - 1e-15:
                best_h, best_c5 = hp2, c5p2

    if best_h is None:
        best_n = 70
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
