```python
import numpy as np
from scipy.optimize import minimize
import time

_state = {}

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def lse_obj(h):
    corr = compute_corr(h)
    a = _state['alpha']
    m = np.max(corr)
    shifted = a * (corr - m)
    shifted = np.clip(shifted, -600, 600)
    val = m + np.log(np.sum(np.exp(shifted))) / a
    topk = _state['topk']
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        val += _state['flat_w'] * np.sum((sc - sc.mean())**2)
    return val

def lse_grad(h):
    n = len(h)
    corr = compute_corr(h)
    a = _state['alpha']
    m = np.max(corr)
    shifted = np.clip(a * (corr - m), -600, 600)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # Gradient of log-sum-exp part: sum_k w_k * dc[k]/dh[j]
    # dc[k]/dh[j] for lag k = idx - (n-1):
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)  # actual lag
        if k >= 0:
            # dc/dh[j] = (1-h[j-k]) for j>=k - h[j+k] for j+k<n
            valid_j = np.arange(k, n)
            grad[valid_j] += w * (1.0 - h[valid_j - k])
            valid_j2 = np.arange(0, n - k)
            grad[valid_j2] -= w * h[valid_j2 + k]
        else:
            kp = -k
            valid_j = np.arange(0, n - kp)
            grad[valid_j] += w * (1.0 - h[valid_j + kp])
            valid_j2 = np.arange(kp, n)
            grad[valid_j2] -= w * h[valid_j2 - kp]
    return grad

def subgradient_step(h, n, alpha, topk, lr, t_end):
    h = h.copy()
    _state['alpha'] = alpha
    _state['topk'] = topk
    _state['flat_w'] = 0.0
    best_c5 = compute_c5(h, n)
    for it in range(2000):
        if time.time() > t_end:
            break
        g = lse_grad(h)
        # Add constraint gradient for sum
        g += 0.01 * np.ones(n) * (h.sum() - n/2.0)
        h_new = h - lr * g
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.8
    return repair(h, n), best_c5

def coord_descent(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    # Use gradient to pick directions
    steps = [0.02, 0.01, 0.005, 0.001, 0.0005, 0.0002]
    improved_count = 0
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        round_improved = False
        for _ in range(6 * n):
            if time.time() > t_end:
                break
            # Pick i based on gradient (more likely to pick high-gradient elements)
            corr = compute_corr(h)
            m = np.max(corr)
            top_lags = np.argsort(corr)[::-1][:5]
            # Compute which indices are "active"
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hi + delta, hj - delta
                    if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        round_improved = True
                        improved_count += 1
                        break
                    h[i], h[j] = hi, hj
                if round_improved:
                    break
        if not round_improved and r > 30:
            break
    return h, best

def do_slsqp(h, n, alpha, topk, flat_w, maxiter):
    _state['alpha'] = alpha
    _state['topk'] = topk
    _state['flat_w'] = flat_w
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(lse_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def do_lbfgs(h, n, alpha, topk, flat_w, pen, maxiter):
    _state['alpha'] = alpha
    _state['topk'] = topk
    _state['flat_w'] = flat_w
    _state['pen'] = pen
    def pen_obj(x):
        return lse_obj(x) + pen * (x.sum() - n/2.0)**2
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * 2.0 / n)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def lse_objective(h, alpha, topk, flat_w):
    corr = compute_corr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -600, 600)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        val += flat_w * np.sum((sc - sc.mean())**2)
    return val

def do_slsqp(h, n, alpha, topk, flat_w, maxiter):
    def obj(x):
        return lse_objective(x, alpha, topk, flat_w)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def do_lbfgs(h, n, alpha, topk, flat_w, pen, maxiter):
    def obj(x):
        return lse_objective(x, alpha, topk, flat_w) + pen * (x.sum() - n / 2.0) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
        return repair(res.x, n)
    except Exception:

def inner_qp(h0, n, p, maxiter=100):
    """Solve min_h p·corr(h) via projected gradient"""
    h = h0.copy()
    for it in range(maxiter):

        corr = compute_corr(h)
        # grad of corr[k] w.r.t. h
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            w = p[idx]
            if w < 1e-12:
                continue
            k = idx - (n - 1)
            if k >= 0:

                grad[j_start:j_end] += w * (1.0 - h[max(0, j_start - k):j_end - k])
                # Wait, this isn't right. Let me redo.
                # For lag k>=0: dc/dh[j] = (1-h[j-k]) for j>=k, minus h[j+k] for j+k<n

                kp = -k
                for j in range(0, n - kp):
                    grad[j] += w * (1.0 - h[j + kp])
                for j in range(kp, n):
                    grad[j] -= w * h[j - kp]

        h_new = h_new + 0.1 * (n / 2.0 - h_new.sum()) / n  # pull toward sum constraint

        else:
            p = p_new / s

def coord_descent(h, n, t_end, rng, max_rounds=300):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.03, 0.01, 0.005, 0.002, 0.001]
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        n_inner = min(3 * n, 200)
        for _ in range(n_inner):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hi + delta, hj - delta
                    if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved and r > 20:
            break
    return h, best
