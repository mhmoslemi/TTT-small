Actually, for efficiency, let me precompute the effect of a swap on the correlation. When we swap h[i] and h[j] (assuming h[i]=1, h[j]=0), the new values are h[i]=0, h[j]=1. The change in correlation at lag k is:

For the term involving position i:
- Before: h[i]*(1-h[i+k]) = 1*(1-h[i+k])
- After: h[i]*(1-h[i+k]) = 0*(1-h[i+k]) = 0
- Change at lag k (i as first index): -(1-h[i+k]) = h[i+k]-1

Similarly for position j:
- Before: h[j]*(1-h[j+k]) = 0*(1-h[j+k]) = 0
- After: h[j]*(1-h[j+k]) = 1*(1-h[j+k]) = 1-h[j+k]
- Change at lag k (j as first index): +(1-h[j+k])

And we also need to consider positions where i or j is the SECOND index:
For position i as second index (i = m+k, so m = i-k):
- Before: h[i-k]*(1-h[i]) = h[i-k]*(1-1) = 0
- After: h[i-k]*(1-0) = h[i-k]
- Change: +h[i-k]

For position j as second index (j = m+k, so m = j-k):
- Before: h[j-k]*(1-h[j]) = h[j-k]*(1-0) = h[j-k]
- After: h[j-k]*(1-1) = 0
- Change: -h[j-k]

So the total change at lag k is:
Δk = h[i+k] - 1 + (1-h[j+k]) + h[i-k] - h[j-k] (with appropriate bounds checking)

This is O(1) per lag per swap, so O(n) per swap to evaluate all lags. Much faster than recomputing the full correlation!

                h[i], h[j] = h[j], h[i]
        elif r < 0.8:
            # 2-swap move
            i1, j1, i2, j2 = rng.randint(n, 4)
            if len(set([i1, j1, i2, j2])) < 4:
                continue
            if h[i1] == h[j1] or h[i2] == h[j2]:
                continue
            h[i1], h[j1] = h[j1], h[i1]
            h[i2], h[j2] = h[j2], h[i2]
            c5_new = _c5(h, n)
            d = c5_new - c5
            if d < 0 or (T > t_min and rng.random() < np.exp(-d / (2.0 * T))):
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    stagnant = 0
                stagnant += 1
            else:
                h[i1], h[j1] = h[j1], h[i1]
                h[i2], h[j2] = h[j2], h[i2]
        else:
            # Block shift: move a block of size 2-5 to a new location
            blen = rng.randint(2, 6)
            start = rng.randint(0, n - blen)
            # Find a target gap to place: pick a random position
            dst = rng.randint(0, n - blen)
            if abs(start - dst) < blen:
                continue
            # Simple: swap element at start+j with element at dst+j for j in range(blen)
            ok = True
            for j in range(blen):
                if start + j == dst + j:
                    ok = False
                    break
            if not ok:
                continue
            for j in range(blen):
                h[start + j], h[dst + j] = h[dst + j], h[start + j]
            c5_new = _c5(h, n)
            d = c5_new - c5
            if d < 0 or (T > t_min and rng.random() < np.exp(-d / (3.0 * T))):
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    stagnant = 0
                stagnant += 1
            else:
                for j in range(blen):
                    h[start + j], h[dst + j] = h[dst + j], h[start + j]
        if stagnant > 3000:
            T = max(T * 0.5, t_min)
            stagnant = 0
        if stagnant > 15000:
            # Kick: randomize 5 positions
            for _ in range(5):
                p, q = rng.randint(n, 2)
                if p != q and h[p] != h[q]:
                    h[p], h[q] = h[q], h[p]
            T = max(T * 2.0, t0_init * 0.3)
            stagnant = 0
    return best_h, best_c5

def _focused_local(h, n, rng, deadline):
    h = h.copy()
    c5 = _c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    no_improve = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        active = np.where(corr > 0.95 * max_val)[0]
        improved = False
        for am in active:
            k = am - (n - 1)
            # Find contributing positions for this lag
            if k >= 0:
                for i in range(n - k):
                    if h[i] == 1 and h[i + k] == 0:
                        # Try swapping h[i] with a random 0 elsewhere
                        zeros = np.where(h == 0)[0]
                        zeros = zeros[(zeros != i) & (zeros != i + k)]
                        if len(zeros) == 0:
                            continue
                        trial = rng.choice(zeros, size=min(15, len(zeros)), replace=False)
                        for c in trial:
                            h[i], h[c] = h[c], h[i]
                            c5_new = _c5(h, n)
                            if c5_new < c5:
                                c5 = c5_new
                                if c5 < best_c5:
                                    best_c5 = c5
                                    best_h = h.copy()
                                improved = True
                            else:
                                h[i], h[c] = h[c], h[i]
                            if improved:
                                break
                        if improved:
                            break
            else:
                for i in range(-k):
                    if h[i - k] == 1 and h[i] == 0:
                        zeros = np.where(h == 0)[0]
                        zeros = zeros[(zeros != i) & (zeros != i - k)]
                        if len(zeros) == 0:
                            continue
                        trial = rng.choice(zeros, size=min(15, len(zeros)), replace=False)
                        for c in trial:
                            h[i - k], h[c] = h[c], h[i - k]
                            c5_new = _c5(h, n)
                            if c5_new < c5:
                                c5 = c5_new
                                if c5 < best_c5:
                                    best_c5 = c5
                                    best_h = h.copy()
                                improved = True
                            else:
                                h[i - k], h[c] = h[c], h[i - k]
                            if improved:
                                break
                        if improved:
                            break
        if not improved:
            no_improve += 1
            if no_improve > 50:
                for _ in range(3):
                    p, q = rng.randint(n, 2)
                    if p != q and h[p] != h[q]:
                        h[p], h[q] = h[q], h[p]
                c5 = _c5(h, n)
                no_improve = 0
        else:
            no_improve = 0
    return best_h, best_c5

def _nb_refine(h, n, deadline):
    h = h.copy()
    target = n / 2.0
    c5 = _c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return _c5(np.clip(x, 0, 1), n)

    def _eq(x):
        return x.sum() - target

    cons = ({'type': 'eq', 'fun': _eq},)
    while time.time() < deadline:
        try:
            res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
            hc = _repair(res.x, n)
            c5_new = _c5(hc, n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = hc.copy()
                h = hc.copy()
            else:
                break
        except Exception:
            break
    return best_h, best_c5

def _pair_polish(h, n, rng, deadline):
    h = h.copy()
    c5 = _c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    steps = [0.001, 0.003, 0.01, 0.03, -0.001, -0.003, -0.01, -0.03]
    while time.time() < deadline:
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        for d in steps:
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5_new = _c5(h, n)
                if c5_new < c5:
                    c5 = c5_new
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    break
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    h_bin = _to_binary(h0, n)
    c5_bin = _c5(h_bin, n)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 1: SA on binary (55% budget)
    sa_deadline = t0 + 0.55 * budget_s
    h_sa, c5_sa = _sa_binary(h_bin, n, rng, sa_deadline)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 2: Second SA with different start (15% budget)
    if time.time() < t0 + 0.70 * budget_s:
        h_bin2 = np.zeros(n)
        ones_pos = rng.permutation(n)[:n // 2]
        h_bin2[ones_pos] = 1.0
        # Blend with best
        h_bin2 = 0.5 * h_bin2 + 0.5 * best_h
        h_bin2 = _to_binary(h_bin2, n)
        sa_deadline2 = t0 + 0.70 * budget_s
        h_sa2, c5_sa2 = _sa_binary(h_bin2, n, rng, sa_deadline2, t0_init=best_c5 * 0.08)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 3: Focused local search (15% budget)
    ls_deadline = t0 + 0.85 * budget_s
    h_ls, c5_ls = _focused_local(best_h, n, rng, ls_deadline)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    # Phase 4: Non-binary SLSQP refinement (7% budget)
    nb_deadline = t0 + 0.92 * budget_s
    h_nb, c5_nb = _nb_refine(best_h, n, nb_deadline)
    if c5_nb < best_c5:
        best_c5 = c5_nb
        best_h = h_nb.copy()

    # Phase 5: Pair polish (8% budget)
    pp_deadline = t0 + budget_s - 2
    h_pp, c5_pp = _pair_polish(best_h, n, rng, pp_deadline)
    if c5_pp < best_c5:
        best_c5 = c5_pp
        best_h = h_pp.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
