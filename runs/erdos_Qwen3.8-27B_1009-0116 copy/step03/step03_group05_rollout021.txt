from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _pair_a(h, i, j, n):
    a = np.zeros(2*n-1)
    for k in range(2*n-1):
        L = k - (n-1)
        al = 0.0
        if L >= 0:
            if i <= n-1-L: al += 1.0 - h[i+L]
            if i >= L: al -= h[i-L]
            if j <= n-1-L: al -= 1.0 - h[j+L]
            if j >= L: al += h[j-L]
        else:
            M = -L
            if i >= M: al += 1.0 - h[i-M]
            if i <= n-1-M: al -= h[i+M]
            if j >= M: al -= 1.0 - h[j-M]
            if j <= n-1-M: al += h[j+M]
        a[k] = al
    return a

def _min_pair(c, a, hi, hj):
    """Minimize max(c + a*d) over d in (-hj, 1-hi). Returns optimal d."""
    lo = -hj + 1e-15
    hi_d = 1.0 - hi - 1e-15
    span = hi_d - lo
    if span < 1e-12:
        return 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = lo + (1.0 - gr) * span
    c2 = lo + gr * span
    f1 = np.max(c + a * c1)
    f2 = np.max(c + a * c2)
    for _ in range(50):
        if f1 < f2:
            lo = c1
        else:
            hi_d = c2
        span = hi_d - lo
        if span < 1e-13:
            break
        c1 = lo + (1.0 - gr) * span
        c2 = lo + gr * span
        f1 = np.max(c + a * c1)
        f2 = np.max(c + a * c2)
    d = (lo + hi_d) / 2.0
    if np.max(c + a * d) < np.max(c) - 1e-14:
        return d
    return 0.0

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

        elif sweep_count % 3 == 1:

        else:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _pair_a_b(h, i, j, n):
    a = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        al = 0.0
        if L >= 0:
            if i <= n - 1 - L:
                al += 1.0 - h[i + L]
            if i >= L:
                al -= h[i - L]
            if j <= n - 1 - L:
                al -= 1.0 - h[j + L]
            if j >= L:
                al += h[j - L]
        else:
            M = -L
            if i >= M:
                al += 1.0 - h[i - M]
            if i <= n - 1 - M:
                al -= h[i + M]
            if j >= M:
                al -= 1.0 - h[j - M]
            if j <= n - 1 - M:
                al += h[j + M]
        a[k] = al
    b = np.zeros(2 * n - 1)
    d_idx = abs(i - j)
    if d_idx > 0 and d_idx <= n - 1:
        b[n - 1 + d_idx] = 1.0
        b[n - 1 - d_idx] = 1.0
    return a, b

def _min_pair_d(c, a, b, hi, hj):
    lo = -hj + 1e-15
    hi_d = 1.0 - hi - 1e-15
    span = hi_d - lo
    if span < 1e-13:
        return 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = lo + (1.0 - gr) * span
    c2 = lo + gr * span
    v1 = c + a * c1 + b * c1 * c1
    v2 = c + a * c2 + b * c2 * c2
    f1 = np.max(v1)
    f2 = np.max(v2)
    for _ in range(55):
        if f1 < f2:
            lo = c1
        else:
            hi_d = c2
        span = hi_d - lo
        if span < 1e-14:
            break
        c1 = lo + (1.0 - gr) * span
        c2 = lo + gr * span
        v1 = c + a * c1 + b * c1 * c1
        f1 = np.max(v1)
        v2 = c + a * c2 + b * c2 * c2
        f2 = np.max(v2)
    d = (lo + hi_d) / 2.0
    vstar = c + a * d + b * d * d
    if np.max(vstar) < np.max(c) - 1e-14:
        return d
    return 0.0

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    best_h = _repair(np.array(initial_h_values, dtype=float), n)
    best_c5 = _c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    # Phase 1: Cyclic pair optimization with exact 1D minimization
    max_sweeps = 200
    for sweep in range(max_sweeps):
        if elapsed() > budget_s - 350:
            break
        h = best_h.copy()
        c = _corr(h)
        any_improve = False
        # Build pair list with rotation for diversity
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        if sweep % 3 == 0:
            rng.shuffle(pairs)
        elif sweep % 3 == 1:
            pairs.sort(key=lambda p: -(h[p[0]] - h[p[1]]) ** 2)
        else:
            pairs.sort(key=lambda p: h[p[0]] + h[p[1]])
        for (i, j) in pairs:
            if elapsed() > budget_s - 350:
                break
            a, b = _pair_a_b(h, i, j, n)
            d = _min_pair_d(c, a, b, h[i], h[j])
            if d != 0.0:
                h[i] += d
                h[j] -= d
                h = _repair(h, n)
                c = _corr(h)
                cm = float(np.max(c)) * dx
                if cm < best_c5 - 1e-15:
                    best_c5 = cm
                    best_h = h.copy()
                any_improve = True
        if not any_improve:
            break

    # Phase 2: High-alpha soft-max L-BFGS-B
    for a_val in [30000, 100000, 500000]:
        if elapsed() > budget_s - 150:
            break
        try:
            res = minimize(
                lambda x: _softobj(x, a_val, 80.0), best_h.copy(),
                jac=lambda x: _softgrad(x, a_val, 80.0),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 400, 'ftol': 1e-18})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    # Phase 3: More pair sweeps (targeted on high-|h_i - h_j| pairs)
    for _sw in range(50):
        if elapsed() > budget_s - 100:
            break
        h = best_h.copy()
        c = _corr(h)
        any_imp = False
        # Focus on pairs where one is high and one is low
        order = np.argsort(h)
        for rank_i in range(n // 2):
            i = order[rank_i]
            for rank_j in range(n - 1 - n // 2, n):
                j = order[rank_j]
                a, b = _pair_a_b(h, i, j, n)
                d = _min_pair_d(c, a, b, h[i], h[j])
                if d != 0.0:
                    h[i] += d
                    h[j] -= d
                    h = _repair(h, n)
                    c = _corr(h)
                    cm = float(np.max(c)) * dx
                    if cm < best_c5 - 1e-15:
                        best_c5 = cm
                        best_h = h.copy()
                    any_imp = True
        if not any_imp:
            break

    # Phase 4: Active-set gradient refinement
    h = best_h.copy()
    stall = 0
    while elapsed() < budget_s - 50:
        c = _corr(h)
        cur_c5 = float(np.max(c)) * dx
        top_m = 8
        top_idx = np.argsort(c)[::-1][:top_m]
        g = np.zeros(n)
        for ki in top_idx:
            L = ki - (n - 1)
            if L == 0:
                g += (1.0 - 2.0 * h)
            elif L > 0:
                g[L:] += (1.0 - h[:n - L])
                g[:n - L] -= h[L:]
            else:
                M = -L
                g[:n - M] += (1.0 - h[M:])
                g[M:] -= h[:n - M]
        g_c = g - g.mean()
        moved = False
        for step in [0.01, 0.005, 0.002, 0.001, 0.0004]:
            h_new = _repair(h - step * g_c, n)
            c5_new = _c5(h_new, dx)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                moved = True
                break
        if not moved:
            # Random micro-perturbations
            for _ in range(100):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                for d in [0.0005, 0.001, 0.002, -0.0005, -0.001, -0.002]:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        h = _repair(h, n)
                        c5_t = _c5(h, dx)
                        if c5_t < best_c5:
                            best_c5 = c5_t
                            best_h = h.copy()
                            moved = True
                        else:
                            h = best_h.copy()
                        break
        if not moved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0

    # Phase 5: Final ultra-fine pair polish
    if elapsed() < budget_s - 5:
        h = best_h.copy()
        c = _corr(h)
        for i in range(n):
            for j in range(i + 1, n):
                a, b = _pair_a_b(h, i, j, n)
                d = _min_pair_d(c, a, b, h[i], h[j])
                if d != 0.0:
                    h[i] += d
                    h[j] -= d
                    h = _repair(h, n)
                    c = _corr(h)
                    cm = float(np.max(c)) * dx
                    if cm < best_c5:
                        best_c5 = cm
                        best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
