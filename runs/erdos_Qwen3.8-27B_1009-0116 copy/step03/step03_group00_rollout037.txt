def _pair_optimize(h, i, j, n, max_iter=50):
    """Find optimal (h[i], h[j]) given h[i]+h[j] = const, using ternary search."""
    c = h[i] + h[j]
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if lo >= hi:
        return h[i], h[j]
    
    best_a = (lo + hi) / 2
    best_val = _c5(h, n)
    
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        h[i] = m1; h[j] = c - m1
        v1 = _c5(h, n)
        h[i] = m2; h[j] = c - m2
        v2 = _c5(h, n)
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        h[i] = (lo + hi) / 2; h[j] = c - h[i]
        v = _c5(h, n)
        if v < best_val:
            best_val = v
            best_a = h[i]

    h[j] = c - best_a

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    total = np.sum(e)
    weights = e / total
    g = np.zeros(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += w * (1.0 - x[i])
            g[i] -= w * x[jp]
    return g * dx_val

def _pair_sweep(h, n, rng, max_sweeps=50, samples=30):
    """Coordinate descent: for each pair, find best (a, c-a) via sampling."""
    c5_cur = _c5(h, n)
    nsamp = samples
    for sweep in range(max_sweeps):
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for i in order:
            for j in range(i+1, n):
                c = h[i] + h[j]
                lo = max(0.0, c - 1.0)
                hi = min(1.0, c)
                if hi - lo < 1e-10:
                    continue
                # Sample points
                best_a = h[i]
                best_v = c5_cur
                a_grid = np.linspace(lo, hi, nsamp)
                old_i, old_j = h[i], h[j]
                for a in a_grid:
                    h[i] = a
                    h[j] = c - a
                    v = _c5(h, n)
                    if v < best_v:
                        best_v = v
                        best_a = a
                h[i] = best_a
                h[j] = c - best_a
                if best_v < c5_cur - 1e-15:
                    c5_cur = best_v
                    improved = True
                elif best_a != old_i:
                    h[i] = best_a
                    h[j] = c - best_a
        if not improved:
            break
    return h, c5_cur

def _slsqp_run(h_init, n, alpha, maxiter, bounds_list, cons):
    return minimize(
        lambda x: _smooth_obj(x, n, alpha), h_init,
        jac=lambda x: _smooth_grad(x, n, alpha),
        method='SLSQP', bounds=bounds_list, constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-15})

def _random_init(n, rng):
    """Generate a random valid h."""

def _structured_init(n, rng, type):
    """Generate structured initializations."""
    x = np.linspace(0, 2.0, n, endpoint=False)
    if type == 0:  # Smooth bump
        center = rng.uniform(0.2, 1.8)
        width = rng.uniform(0.3, 0.8)
        h = np.exp(-((x - center)**2) / (2 * width**2))
        h += 0.1 * rng.randn(n)
    elif type == 1:  # Periodic
        freq = rng.randint(2, 5)
        phase = rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.3 * np.sin(freq * np.pi * x + phase)
        h += 0.05 * rng.randn(n)
    elif type == 2:  # Step function
        thresh = rng.uniform(0.3, 0.7)
        h = (x < thresh * 2.0).astype(float)
        h += 0.1 * rng.randn(n)
    elif type == 3:  # Two bumps
        c1, c2 = sorted(rng.uniform(0.1, 1.9, 2))
        h = np.exp(-((x - c1)**2) / 0.1) + np.exp(-((x - c2)**2) / 0.1)
        h /= h.max()
        h += 0.05 * rng.randn(n)
    else:
        h = rng.rand(n)
    return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    total = np.sum(e)
    weights = e / total
    g = np.zeros(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += w * (1.0 - x[i])
            g[i] -= w * x[jp]
    return g * dx_val

def _con_fun(x, n):
    return np.sum(x) - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def _slsqp(h_init, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(_con_fun, n=n),
             'jac': partial(_con_jac, n=n)}]
    try:
        res = minimize(partial(_smooth_obj, n=n, alpha=alpha), h_init,
                       jac=partial(_smooth_grad, n=n, alpha=alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h_init

def _pair_sweep(h, n, max_sweeps, samples, t_end):
    prev_c5 = _c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        for i in range(n):
            for j in range(i + 1, n):
                c = h[i] + h[j]
                lo = max(0.0, c - 1.0)
                hi = min(1.0, c)
                if hi - lo < 1e-12:
                    continue
                old_i = h[i]
                best_a = old_i
                best_v = _c5(h, n)
                a_grid = np.linspace(lo, hi, samples)
                for a in a_grid:
                    h[i] = a
                    h[j] = c - a
                    v = _c5(h, n)
                    if v < best_v:
                        best_v = v
                        best_a = a
                h[i] = best_a
                h[j] = c - best_a
        cur_c5 = _c5(h, n)
        if cur_c5 >= prev_c5 - 1e-14:
            break
        prev_c5 = cur_c5
    return h, prev_c5

def _sa_optimize(h, n, t_end, rng):
    c5_cur = _c5(h, n)
    best_h = h.copy()
    best_c5 = c5_cur
    T = 0.0004
    min_T = 1e-10
    cool = 0.9999
    steps = np.array([0.001, 0.003, 0.008, 0.02, 0.05])
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = steps[rng.randint(len(steps))] * (1.0 if rng.rand() < 0.5 else -1.0)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < -1e-14 or ni > 1 + 1e-14 or nj < -1e-14 or nj > 1 + 1e-14:
            continue
        h[i] = max(0.0, min(1.0, ni))
        h[j] = max(0.0, min(1.0, nj))
        c5_new = _c5(h, n)
        if c5_new < c5_cur or (T > 1e-10 and rng.rand() < np.exp(-(c5_new - c5_cur) / T)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i], h[j] = hi, hj
        T = max(T * cool, min_T)
    return best_h, best_c5

def _init_bump(n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    center = rng.uniform(0.3, 1.7)
    width = rng.uniform(0.2, 0.7)
    h = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    h += 0.05 * rng.randn(n)
    return _repair(h, n)

def _init_periodic(n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    freq = rng.randint(2, 5)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.35 * np.sin(freq * np.pi * x + phase)
    h += 0.05 * rng.randn(n)
    return _repair(h, n)

def _init_two_bump(n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    c1, c2 = sorted(rng.uniform(0.2, 1.8, 2))
    w1, w2 = rng.uniform(0.15, 0.4, 2)
    h = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
    h /= h.max()
    h += 0.03 * rng.randn(n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n = len(initial_h_values)
        h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    else:

        h0 = _init_bump(n, rng)

    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: SLSQP from parent and structured inits
    t_phase1_end = t0 + budget_s * 0.25

        inits.append(_init_bump(n, rng))

            h_try = _slsqp(_repair(h_init.copy(), n), n, alpha, 600)
            c5 = _c5(h_try, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try.copy()

    # Phase 2: Pairwise coordinate descent (coarse)

    h = best_h.copy()
    h, c5 = _pair_sweep(h, n, max_sweeps=30, samples=20, t_end=t_p2_end)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: SLSQP refinement
    t_p3_end = t0 + budget_s * 0.65
    for r in range(3):
        if time.time() > t_p3_end:
            break
        h_init = best_h + 0.01 * rng.randn(n)
        h_init = _repair(h_init, n)
        h_try = _slsqp(h_init, n, 50000, 800)
        c5 = _c5(h_try, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()
    h_try = _slsqp(best_h.copy(), n, 100000, 600)
    c5 = _c5(h_try, n)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_try.copy()

    # Phase 4: Pairwise coordinate descent (fine)
    t_p4_end = t0 + budget_s * 0.8
    h = best_h.copy()
    h, c5 = _pair_sweep(h, n, max_sweeps=40, samples=35, t_end=t_p4_end)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: SA
    t_p5_end = t0 + budget_s * 0.92
    h_best, c5_best = _sa_optimize(best_h.copy(), n, t_p5_end, rng)
    if c5_best < best_c5:
        best_c5 = c5_best
        best_h = h_best.copy()

    # Phase 6: Final pairwise + SLSQP
    t_end = t0 + budget_s - 5
    if time.time() < t_end - 30:
        h = best_h.copy()
        h, c5 = _pair_sweep(h, n, max_sweeps=20, samples=30, t_end=t_end - 10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if time.time() < t_end - 5:
        h_try = _slsqp(best_h.copy(), n, 80000, 400)
        c5 = _c5(h_try, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    total = np.sum(e)
    weights = e / total
    g = np.zeros(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += w * (1.0 - x[i])
            g[i] -= w * x[jp]
    return g * dx_val

def _con_fun(x, n):
    return np.sum(x) - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def _slsqp(h_init, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(_con_fun, n=n),
             'jac': partial(_con_jac, n=n)}]
    try:
        res = minimize(partial(_smooth_obj, n=n, alpha=alpha), h_init,
                       jac=partial(_smooth_grad, n=n, alpha=alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h_init

def _pair_sweep(h, n, max_sweeps, samples, t_end):
    prev_c5 = _c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        any_improved = False
        for i in range(n):
            for j in range(i + 1, n):
                c = h[i] + h[j]
                lo = max(0.0, c - 1.0)
                hi = min(1.0, c)
                if hi - lo < 1e-12:
                    continue
                old_i = h[i]
                best_a = old_i
                best_v = _c5(h, n)
                a_grid = np.linspace(lo, hi, samples)
                for a in a_grid:
                    h[i] = a
                    h[j] = c - a
                    v = _c5(h, n)
                    if v < best_v:
                        best_v = v
                        best_a = a
                h[i] = best_a
                h[j] = c - best_a
                if best_v < best_v - 1e-15 and best_a != old_i:
                    any_improved = True
        cur_c5 = _c5(h, n)
        if cur_c5 >= prev_c5 - 1e-14:
            break
        prev_c5 = cur_c5
    return h, prev_c5

def _sa_optimize(h, n, t_end, rng):
    c5_cur = _c5(h, n)
    best_h = h.copy()
    best_c5 = c5_cur
    T = 0.0004
    min_T = 1e-10
    cool = 0.9999
    steps = np.array([0.001, 0.003, 0.008, 0.02, 0.05])
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = steps[rng.randint(len(steps))] * (1.0 if rng.rand() < 0.5 else -1.0)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < -1e-14 or ni > 1 + 1e-14 or nj < -1e-14 or nj > 1 + 1e-14:
            continue
        h[i] = max(0.0, min(1.0, ni))
        h[j] = max(0.0, min(1.0, nj))
        c5_new = _c5(h, n)
        if c5_new < c5_cur or (T > 1e-10 and rng.rand() < np.exp(-(c5_new - c5_cur) / T)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i], h[j] = hi, hj
        T = max(T * cool, min_T)
    return best_h, best_c5

        if time.time() > t_end1 - 15:
            break
        inits.append(_init_bump(n, rng))
        inits.append(_init_periodic(n, rng))
        inits.append(_init_two_bump(n, rng))
    for alpha in [3000, 10000, 30000]:
        for h_init in inits[:9]:
            if time.time() > t_end1:
                break
            h_try = _slsqp(_repair(h_init.copy(), n), n, alpha, 500)
            c5 = _c5(h_try, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try.copy()

    # Phase 2: Pairwise coordinate descent - coarse (25% time)
    t_end2 = t0 + total * 0.48
    h = best_h.copy()
    h, c5 = _pair_sweep(h, n, max_sweeps=40, samples=20, t_end=t_end2)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: SLSQP high-alpha refinement (10% time)
    t_end3 = t0 + total * 0.58
    for r in range(4):
        if time.time() > t_end3:
            break
        h_init = best_h + 0.01 * rng.randn(n)
        h_init = _repair(h_init, n)
        h_try = _slsqp(h_init, n, 50000, 700)
        c5 = _c5(h_try, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()
    if time.time() < t_end3 - 5:
        h_try = _slsqp(best_h.copy(), n, 100000, 500)
        c5 = _c5(h_try, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 4: Pairwise coordinate descent - fine (22% time)
    t_end4 = t0 + total * 0.80
    h = best_h.copy()
    h, c5 = _pair_sweep(h, n, max_sweeps=50, samples=35, t_end=t_end4)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Simulated annealing (10% time)
    t_end5 = t0 + total * 0.91
    h_best, c5_best = _sa_optimize(best_h.copy(), n, t_end5, rng)
    if c5_best < best_c5:
        best_c5 = c5_best
        best_h = h_best.copy()

    # Phase 6: Final pairwise + SLSQP polish (remaining time)
    t_final = t0 + total - 5
    if time.time() < t_final - 30:
        h = best_h.copy()
        h, c5 = _pair_sweep(h, n, max_sweeps=25, samples=30, t_end=t_final - 8)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if time.time() < t_final - 5:
        h_try = _slsqp(best_h.copy(), n, 80000, 400)
        c5 = _c5(h_try, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
