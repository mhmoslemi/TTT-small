    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_h(h)) * dx)

def _smooth_max(vals, alpha):
    m = np.max(vals)
    s = np.clip(alpha * (vals - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _obj_pen(x, alpha, target):
    c = _corr_h(x)
    sm = _smooth_max(c, alpha)
    return sm + 0.05 * (np.sum(x) - target) ** 2

def _obj_sls(x, alpha):
    return _smooth_max(_corr_h(x), alpha)

def _con_eq(x, target):
    return np.sum(x) - target

def _run_slq(h, n, dx, alpha, maxiter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_sls, h, args=(alpha,), method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _con_eq, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _run_lbfgs(h, n, dx, alpha, maxiter=200):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_pen, h, args=(alpha, target), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _coord_sweep(h, n, dx, deadline, max_sweeps=2000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    found = True
    sweep = 0
    stall = 0
    while found or stall < 30:
        found = False
        sweep += 1
        if sweep > max_sweeps:
            break
        if time.time() >= deadline:
            break
        for i in range(n):
            if time.time() >= deadline:
                return h, best_c5
            for j in range(n):
                if j == i:
                    continue
                for eps in (0.003, 0.008, -0.003, -0.008):
                    hi = h[i]
                    hj = h[j]
                    ni = hi + eps
                    nj = hj - eps
                    if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c = _c5(h, dx)
                    if c < best_c5 - 1e-15:
                        best_c5 = c
                        found = True
                        stall = 0
                    else:
                        h[i] = hi
                        h[j] = hj
        if not found:
            stall += 1
            if stall > 30:
                break
        else:
            stall = 0
    return h, best_c5

def _random_fine(h, n, dx, rng, deadline, n_tries=200000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per = max(500, n_tries // len(scales))
    for scale in scales:
        if time.time() >= deadline:
            break
        bench = best_c5
        for _ in range(per):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h[i] = ni
            h[j] = nj
            c = _c5(h, dx)
            if c < best_c5 - 1e-15:
                best_c5 = c
            else:
                h[i] = ni - d
                h[j] = nj + d
        if best_c5 == bench and scale <= 0.001:
            break
    return h, best_c5

    if kind == 0:
        hp += rng.normal(0, 0.04, n)

    elif kind == 2:

        hp = np.roll(hp, sh)
    elif kind == 3:

        i = rng.randint(0, max(1, n - b))
        j = rng.randint(0, max(1, n - b))
        if abs(i - j) >= b:
            tmp = hp[i:i+b].copy()
            hp[i:i+b] = hp[j:j+b]
            hp[j:j+b] = tmp

        hp *= 1.0 + rng.uniform(-0.03, 0.03)
    else:
        p = rng.uniform(-0.05, 0.05, n)
        p -= p.mean()
        hp += p
    return _repair(hp, n)
