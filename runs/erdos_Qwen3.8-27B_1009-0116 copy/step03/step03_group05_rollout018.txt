            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _waterfill(h, n, dx, M, iters, t_end):
    step = 0.008
    mom = np.zeros(n)
    best_local_c5 = _c5(h, dx)
    for _ in range(iters):
        if time.time() > t_end:
            break
        c = _corr(h)
        top = np.argsort(c)[::-1][:M]
        g = np.zeros(n)
        w = np.ones(M)
        w[0] = 2.5
        for i in range(1, M):
            w[i] = 1.0 - 0.4 * (c[top[i]] - c[top[M-1]]) / (c[top[0]] - c[top[M-1]] + 1e-10)
        for i, ki in enumerate(top):
            g += w[i] * _grad_k(h, ki)
        g -= g.mean()
        mom = 0.7 * mom + 0.3 * g
        h_new = _repair(h - step * mom, n)
        c5_new = _c5(h_new, dx)
        if c5_new < best_local_c5:
            h = h_new
            best_local_c5 = c5_new
        else:
            step *= 0.85
        if step < 1e-7:
            break
    return h, best_local_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def elapsed():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: L-BFGS-B soft-max sweep
    for a in [500, 2000, 8000, 30000, 100000, 300000]:
        if elapsed() > budget_s - 350:
            break
        for _attempt in range(2):
            if elapsed() > budget_s - 350:
                break
            try:
                res = minimize(
                    lambda x: _softobj(x, a, 100.0), best_h.copy(),
                    jac=lambda x: _softgrad(x, a, 100.0),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 800, 'ftol': 1e-18})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 2: Water-filling on active set
    for M in [3, 5, 8, 12, 15]:
        if elapsed() > budget_s - 250:
            break
        h, c5w = _waterfill(best_h.copy(), n, dx, M, 4000, budget_s - 250 + t0)
        if c5w < best_c5:
            best_c5 = c5w
            best_h = h

    # Phase 3: SLSQP with exact constraint
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    for a in [50000, 200000]:
        if elapsed() > budget_s - 150:
            break
        try:
            res = minimize(
                lambda x: _softobj(x, a, 0.0), best_h.copy(),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-18})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    # Phase 4: Gradient-guided pair coordinate descent
    if elapsed() < budget_s - 100:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        dlist = np.array([0.0005, 0.001, 0.002, 0.004, 0.008, 0.015, 0.03,
                         -0.0005, -0.001, -0.002, -0.004, -0.008, -0.015, -0.03])
        for _sweep in range(120):
            if elapsed() > budget_s - 100:
                break
            c = _corr(h)
            top = np.argsort(c)[::-1][:6]
            g = np.zeros(n)
            for ki in top:
                g += _grad_k(h, ki)
            order = np.argsort(np.abs(g))[::-1][:30]
            improved = False
            for ii in range(len(order)):
                i = order[ii]
                for jj in range(ii + 1, min(ii + 25, len(order))):
                    j = order[jj]
                    best_d = 0.0
                    best_c = c5_cur
                    for d in dlist:
                        ni, nj = h[i] + d, h[j] - d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c_val = _c5(h, dx)
                            if c_val < best_c:
                                best_c = c_val
                                best_d = d
                            h[i], h[j] = ni - d, nj + d
                    if best_d != 0.0:
                        h[i] += best_d
                        h[j] -= best_d
                        h = _repair(h, n)
                        c5_cur = _c5(h, dx)
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        break
            if not improved:
                break

    # Phase 5: Simulated annealing
    if elapsed() < budget_s - 50:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        T = 2.5e-5
        for _it in range(60000):
            if elapsed() > budget_s - 50:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.025, 0.025)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            h = _repair(h, n)
            c_new = _c5(h, dx)
            if c_new < c5_cur or (T > 1e-11 and rng.rand() < np.exp(-(c_new - c5_cur) / T)):
                c5_cur = c_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
                h = _repair(h, n)
            T *= 0.99985

    # Phase 6: Multi-restart perturbation + waterfill
    if elapsed() < budget_s - 30:
        for _r in range(8):
            if elapsed() > budget_s - 30:
                break
            hp = best_h + 0.005 * rng.randn(n)
            hp = _repair(hp, n)
            try:
                res = minimize(
                    lambda x: _softobj(x, 200000, 200.0), hp,
                    jac=lambda x: _softgrad(x, 200000, 200.0),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 400, 'ftol': 1e-18})
                hc = _repair(res.x, n)
            except Exception:
                hc = hp
            hc, c5w = _waterfill(hc, n, dx, 5, 1500, t0 + budget_s - 30)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

    # Phase 7: Ultra-fine coordinate polish
    if elapsed() < budget_s - 3:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        tiny = np.array([0.0001, 0.0003, 0.0007, 0.0015, 0.003,
                        -0.0001, -0.0003, -0.0007, -0.0015, -0.003])
        for _ in range(25000):
            if elapsed() > budget_s - 3:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
