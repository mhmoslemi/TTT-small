import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def smooth_val(x, alpha, target, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (x.sum() - target) ** 2

def smooth_grad(x, alpha, target, pen):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        r = k - n + 1
        if r >= 0:
            lo, hi = 0, n - r
            for j in range(lo, hi):
                g[j] += wk * (1.0 - x[j + r])
            for j in range(r, n):
                g[j] -= wk * x[j - r]
        else:
            rr = -r
            for j in range(rr, n):
                g[j] += wk * (1.0 - x[j - rr])
            for j in range(n - rr):
                g[j] -= wk * x[j + rr]
    g += 2.0 * pen * (x.sum() - target)
    return g

def local_search(h, n, rng, max_time):
    c5c = eval_c5(h)
    best = h.copy()
    best_c5 = c5c
    t_end = time.time() + max_time
    scales = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    scale_i = 0
    no_imp = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scales[scale_i] * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            # Try smaller step
            d *= 0.3
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
        h[i], h[j] = ni, nj
        c5n = eval_c5(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
            no_imp = 0
            if c5c < best_c5:
                best_c5 = c5c
                best = h.copy()
            # Occasionally increase scale
            if no_imp == 0 and rng.rand() < 0.05 and scale_i > 0:
                scale_i -= 1
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            no_imp += 1
            if no_imp > 200:
                no_imp = 0
                if scale_i < len(scales) - 1:
                    scale_i += 1
    return best, best_c5

def simulated_annealing(h, n, rng, t_start, t_end, t0, t1):
    best = h.copy()
    best_c5 = eval_c5(h)
    c5c = best_c5
    while time.time() < t1:
        T = t0 * (t1 / t_start) ** ((time.time() - t_start) / (t1 - t_start))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.05 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = eval_c5(h)
        if c5n < c5c:
            c5c = c5n
            if c5c < best_c5:
                best_c5 = c5c
                best = h.copy()
        else:
            if rng.rand() < np.exp(-(c5n - c5c) / max(T, 1e-10)):
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best, best_c5

def snap_binary(h, n):
    hs = h.copy()
    hs[hs > 0.7] = 1.0
    hs[hs < 0.3] = 0.0
    # Repair to maintain sum
    hs = repair(hs, n)
    return hs

import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def smooth_val(x, alpha, target, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (x.sum() - target) ** 2

def smooth_grad(x, alpha, target, pen):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        r = k - n + 1
        if r >= 0:
            for j in range(n - r):
                g[j] += wk * (1.0 - x[j + r])
            for j in range(r, n):
                g[j] -= wk * x[j - r]
        else:
            rr = -r
            for j in range(rr, n):
                g[j] += wk * (1.0 - x[j - rr])
            for j in range(n - rr):
                g[j] -= wk * x[j + rr]
    g += 2.0 * pen * (x.sum() - target)
    return g

    best = h.copy()

        if time.time() > dt_end:
            break
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() > dt_end:
                    break
                hi = h[i]
                for d in step_sizes:
                    ni, nj = hi + d, h[j] - d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c5n = eval_c5(h)

                        hi = ni

                            best = h.copy()
                    else:
                        h[i], h[j] = hi, h[j]
                h[i] = hi
            if time.time() > dt_end:
                break
        if not imp:
            break
