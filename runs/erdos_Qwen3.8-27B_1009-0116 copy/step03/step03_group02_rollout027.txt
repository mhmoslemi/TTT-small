def _sweep_pair(h, j, n, corr, time_limit, n_coarse=25, n_bisect=35):

    best_d = 0
    best_i = -1
    best_new_max = np.max(corr)
    cur_max = best_new_max

        d_max = min(1.0 - hi, hj)

            continue
        gd = _g_vec(h, i, n) - _g_vec(h, j, n)

        # Bisection

    s = h.sum()

        return np.full(n, 0.5)

                h[idx] += add
                diff -= add

        gi[m2] -= x[idx2[m2]]
        grad[i] = np.dot(weights, gi)
    grad += 2.0 * mu * (np.sum(x) - target)
    return val, grad

def _lbfgsb_step(h0, n, target, alpha, mu, maxiter=500):
    def obj(x):
        v, _ = _obj_grad(x, alpha, n, target, mu)
        return v
    def gr(x):
        _, g = _obj_grad(x, alpha, n, target, mu)
        return g
    try:
        res = minimize(obj, h0, jac=gr, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return h0

def _slsqp_step(h, n, target, alpha, maxiter=300):
    def obj(x):
        corr = _corr(x)
        m = float(np.max(corr))
        s = alpha * (corr - m)
        s = np.clip(s, -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def eqc(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eqc}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _sweep_round(h, n, corr, time_limit, n_coarse=25, n_bisect=40):
    improved = False
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for bj in range(ai + 1, n):
            j = order[bj]
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            cur_max = np.max(corr)
            if vals[bi] >= cur_max - 1e-14:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

def _fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.arange(n) * 2.0 / n
    h0 = 0.5 * np.ones(n)
    for m in range(1, 15):
        freq = m * np.pi / 2.0
        amp = 0.12 / np.sqrt(m)
        h0 += amp * np.cos(freq * x) * rng.choice([-1, 1])
    return _repair(h0, n)

def _optimize_nh(h0, n, dx, target, t0, budget_s, phase_frac):
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)
    t_phase = t0 + budget_s * phase_frac
    # L-BFGS-B warmup
    for alpha, mu in [(50, 1.0), (150, 5.0), (400, 50.0), (1000, 500.0)]:
        if time.time() > t_phase - 60:
            break
        hc = _lbfgsb_step(best_h, n, target, alpha, mu, 400)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5, best_h = c5, hc.copy()
    # SLSQP step
    if time.time() < t_phase - 40:
        hc = _slsqp_step(best_h, n, target, 300, 300)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5, best_h = c5, hc.copy()
    # Coordinate sweeps loop
    stall = 0
    while time.time() < t_phase - 20:
        h = best_h.copy()
        corr = _corr(h)
        chunk = min(15.0, t_phase - 25 - time.time())
        if chunk < 1:
            break
        h, corr, imp = _sweep_round(h, n, corr, time.time() + chunk, 25, 40)
        if imp:
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5, best_h = c5, h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall >= 5 and time.time() < t_phase - 50:
            scale = 0.02 + 0.01 * min(stall, 10)
            hp = _repair(best_h + scale * np.random.randn(n), n)
            hc = _lbfgsb_step(hp, n, target, 500, 200.0, 300)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _sweep_round(hc, n, hc_corr, time.time() + 10, 25, 35)
            c5t = _c5(hc, dx)
            if c5t < best_c5:
                best_c5, best_h = c5t, hc.copy()
            stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 0
    # Phase A: optimize n=70 from initial_h_values
    n70 = len(initial_h_values)
    dx70 = 2.0 / n70
    tgt70 = n70 / 2.0
    h70 = _repair(np.array(initial_h_values, dtype=float).copy(), n70)
    h70, c5_70 = _optimize_nh(h70, n70, dx70, tgt70, t0, budget_s, 0.45)
    best_h, best_c5, best_n = h70, c5_70, n70
    # Phase B: try n=200 from Fourier
    if time.time() - t0 < budget_s * 0.55:
        n200 = 200
        dx200 = 2.0 / n200
        tgt200 = n200 / 2.0
        h200 = _fourier_init(n200, seed)
        h200, c5_200 = _optimize_nh(h200, n200, dx200, tgt200, t0, budget_s, 0.80)
        if c5_200 < best_c5:
            best_h, best_c5, best_n = h200, c5_200, n200
    # Phase C: try n=350
    if time.time() - t0 < budget_s * 0.78:
        n350 = 350
        dx350 = 2.0 / n350
        tgt350 = n350 / 2.0
        h350 = _fourier_init(n350, seed + 7)
        h350, c5_350 = _optimize_nh(h350, n350, dx350, tgt350, t0, budget_s, 0.93)
        if c5_350 < best_c5:
            best_h, best_c5, best_n = h350, c5_350, n350
    # Final polish on best
    if time.time() - t0 < budget_s - 10:
        dxb = 2.0 / best_n
        tgtb = best_n / 2.0
        hc = _slsqp_step(best_h, best_n, tgtb, 2000, 400)
        cb = _c5(hc, dxb)
        if cb < best_c5:
            best_c5, best_h = cb, hc.copy()
        hc, _, _ = _sweep_round(best_h.copy(), best_n, _corr(best_h), time.time() + 8, 31, 45)
        cb = _c5(hc, dxb)
        if cb < best_c5:
            best_c5, best_h = cb, hc.copy()
    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
