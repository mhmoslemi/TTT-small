```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                ad = min(diff, 1.0 - h[idx])
                h[idx] += ad
                diff -= ad
            else:
                sd = min(-diff, h[idx])
                h[idx] -= sd
                diff += sd
            if abs(diff) < 1e-12:
                break
    return h

def _gradient_smooth(h, alpha):
    """Compute gradient of smooth_max(corr(h)) w.r.t. h."""
    c = _corr(h)
    m = np.max(c)
    w = np.exp(np.clip(alpha * (c - m), -500, 500))
    w /= w.sum()
    n = len(h)
    grad = np.zeros(n)
    k_vals = np.arange(-(n-1), n)
    for p in range(2*n-1):
        k = k_vals[p]
        # grad_k = shift(1-h, -k) - shift(h, k) (with bounds)
        # term1: (1-h[m+k]) for valid m
        lo1 = max(0, -k)
        hi1 = min(n, n-k)
        if lo1 < hi1:
            grad[lo1:hi1] += w[p] * (1.0 - h[lo1+k:hi1+k])
        # term2: -h[m-k] for valid m
        lo2 = max(0, k)
        hi2 = min(n, n+k)
        if lo2 < hi2:
            grad[lo2:hi2] -= w[p] * h[lo2-k:hi2-k]
    return grad

def _proj_gradient_step(h, grad, lr, n):
    """Take a projected gradient step with sum constraint."""
    h_new = h - lr * grad
    # Repair to satisfy constraints
    h_new = _repair(h_new, n)
    return h_new

def _pair_sweep(h, n, time_limit, n_coarse=20, n_bisect=30):
    """Coordinate pair sweep with bisection (similar to previous but slightly different)."""
    improved = False
    corr = _corr(h)
    cur_max = np.max(corr)
    order = np.random.permutation(n)
    
    for ai in range(n):
        i = order[ai]
        for aj in range(ai+1, n):
            j = order[aj]
            if time.time() > time_limit:
                return h, corr, improved
            if cur_max <= 0:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            # Compute gradient of corr w.r.t. (d_i = +d, d_j = -d)
            # g[k] = ∂c[k]/∂h[i] - ∂c[k]/∂h[j]
            k_vals = np.arange(-(n-1), n)
            g = np.zeros(2*n-1)
            for p in range(2*n-1):
                k = k_vals[p]
                gp = 0.0
                # ∂c[p]/∂h[i]: (1-h[i+k])*valid - h[i-k]*valid
                if 0 <= i+k < n:
                    gp += (1.0 - h[i+k])
                if 0 <= i-k < n:
                    gp -= h[i-k]
                # ∂c[p]/∂h[j]:
                if 0 <= j+k < n:
                    gp -= (1.0 - h[j+k])
                if 0 <= j-k < n:
                    gp += h[j-k]
                g[p] = gp
            
            # 1D search: minimize max(corr + d*g) over d in [d_min, d_max]
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None,:] + ds[:,None]*g[None,:], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-14:
                continue
            lo = max(d_min, ds[max(0, bi-1)])
            hi_b = min(d_max, ds[min(n_coarse-1, bi+1)])
            for _ in range(n_bisect):
                mid = 0.5*(lo + hi_b)
                if np.max(corr + mid*g) <= np.max(corr + hi_b*g):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5*(lo + hi_b)
            h[i] += d_best
            h[j] -= d_best
            corr = corr + d_best * g
            new_max = np.max(corr)
            if new_max < cur_max:
                cur_max = new_max
                improved = True
    return h, corr, improved

def _binary_init(n, rng):
    """Random binary initialization with exactly n/2 ones."""
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h

def _slsqp(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 30})
        return _repair(res.x, n)
    except:
        return h
```

Wait, one more idea: **use CVXPY for the continuous optimization**. The problem:
minimize t
s.t. c_k ≤ t/dx for all k (where c_k = corr(h, 1-h)[k])
     sum(h) = n/2
     0 ≤ h ≤ 1

This is NOT convex because c_k is quadratic in h. But if I linearize the quadratic terms (use the current h as linearization point), I get a sequence of convex problems. This is the "successive convex approximation" (SCA) method.

At iteration t:
- Linearize c_k(h) ≈ c_k(h_t) + ∇c_k(h_t)^T (h - h_t)
- Solve: min t s.t. c_k(h_t) + g_k^T(h-h_t) ≤ t/dx, sum(h)=n/2, 0≤h≤1
- This is a linear program (LP)! Can be solved very fast.

This might be much more effective than SLSQP because it handles the constraints exactly.

Let me implement SCA:
1. Start from current h
2. Compute c_k and gradients g_k for all k
3. Solve LP: min t s.t. c_k + g_k^T(h-h_t) ≤ t/dx, sum(h)=n/2, 0≤h≤1
4. Update h
5. Repeat

For the LP, I can use scipy.optimize.linprog or cvxpy. With n=70 variables + 1 (t) = 71 variables, and 2n-1 + 1 = 140 constraints + bound constraints, it's a small LP that solves in milliseconds.

```python
def _sca_step(h, dx, n, target):
    """One SCA iteration: linearize and solve LP."""
    from scipy.optimize import linprog
    c = _corr(h)  # shape (2n-1,)
    # Gradient of c_k w.r.t. h
    # g[k, m] = ∂c_k/∂h[m]
    k_vals = np.arange(-(n-1), n)
    # Build gradient matrix (2n-1, n)
    G = np.zeros((2*n-1, n))
    for p in range(2*n-1):
        k = k_vals[p]
        lo1 = max(0, -k)
        hi1 = min(n, n-k)
        if lo1 < hi1:
            G[p, lo1:hi1] += (1.0 - h[lo1+k:hi1+k])
        lo2 = max(0, k)
        hi2 = min(n, n+k)
        if lo2 < hi2:
            G[p, lo2:hi2] -= h[lo2-k:hi2-k]
    
    # LP: min t
    # vars: [h_0, ..., h_{n-1}, t]
    # objective: [0, ..., 0, 1]
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    # constraints: c_k + g_k^T(h - h_t) <= t/dx
    # => c_k - g_k^T h_t + g_k^T h - t/dx <= 0
    # => g_k^T h - t/dx <= -(c_k - g_k^T h_t)
    # => g_k^T h - t/dx <= g_k^T h_t - c_k
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    for p in range(2*n-1):
        A_ub[p, :n] = G[p]
        A_ub[p, n] = -1.0/dx
        b_ub[p] = G[p] @ h - c[p]
    
    # equality: sum(h) = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0,1)]*n + [(0, None)]  # h in [0,1], t >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n].copy()
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                ad = min(diff, 1.0 - h[idx])
                h[idx] += ad
                diff -= ad
            else:
                sd = min(-diff, h[idx])
                h[idx] -= sd
                diff += sd
            if abs(diff) < 1e-12:
                break
    return h

def _grad_matrix(h, n):
    k_vals = np.arange(-(n - 1), n)
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        k = k_vals[p]
        lo1, hi1 = max(0, -k), min(n, n - k)
        if lo1 < hi1:
            G[p, lo1:hi1] += (1.0 - h[lo1 + k:hi1 + k])
        lo2, hi2 = max(0, k), min(n, n + k)
        if lo2 < hi2:
            G[p, lo2:hi2] -= h[lo2 - k:hi2 - k]
    return G

def _sca_block(h, dx, n, target, max_iter=40, tol=1e-12):
    for _ in range(max_iter):
        c = _corr(h)
        G = _grad_matrix(h, n)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        b_ub = np.zeros(2 * n - 1)
        for p in range(2 * n - 1):
            A_ub[p, :n] = G[p]
            A_ub[p, n] = -1.0 / dx
            b_ub[p] = G[p] @ h - c[p]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds = [(0, 1)] * n + [(0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        h_new = _repair(res.x[:n], n)
        c5_new = _c5(h_new, dx)
        c5_old = _c5(h, dx)
        if c5_old - c5_new < tol:
            break
        h = h_new
    return h

def _pair_sweep(h, n, time_limit, n_coarse=20, n_bisect=30):
    improved = False
    corr = _corr(h)
    cur_max = np.max(corr)
    order = np.random.permutation(n)
    k_vals = np.arange(-(n - 1), n)
    for ai in range(n):
        i = order[ai]
        for aj in range(ai + 1, n):
            j = order[aj]
            if time.time() > time_limit:
                return h, corr, improved
            if cur_max <= 1e-15:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            g = np.zeros(2 * n - 1)
            for p in range(2 * n - 1):
                k = k_vals[p]
                gp = 0.0
                if 0 <= i + k < n:
                    gp += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    gp -= h[i - k]
                if 0 <= j + k < n:
                    gp -= (1.0 - h[j + k])
                if 0 <= j - k < n:
                    gp += h[j - k]
                g[p] = gp
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * g[None, :], axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= cur_max - 1e-14:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * g) <= np.max(corr + hi_b * g):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] += d_best
            h[j] -= d_best
            corr = corr + d_best * g
            nm = float(np.max(corr))
            if nm < cur_max:
                cur_max = nm
                improved = True
    return h, corr, improved

            c5 = _c5(h2, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, h2.copy()

    # Phase 4: SLSQP + pair sweep finish
    if time.time() < t0 + budget_s - 3:
        hc = _slsqp(best_h, n, target, bounds, 3000, 500)
        hc, _, _ = _pair_sweep(hc, n, time.time() + 5, 25, 35)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
