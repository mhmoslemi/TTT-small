def _lp_step(h_old, n, dx):
    # Solve LP to improve h
    from scipy.optimize import linprog
    # Variables: h[0],...,h[n-1], t (total n+1 variables)
    # Minimize: t
    # c = [0,...,0, 1]
    # 
    # Constraints (inequality): A @ [h,t] <= b
    # For each shift k (there are 2n-1 shifts):
    #   sum_i (a[i,k] - b[i,k]) * h[i] - t/dx <= 0
    #   where a[i,k] = 1 - h_old[i+k] (if valid) else 0
    #         b[i,k] = h_old[i-k] (if valid) else 0
    #
    # Equality: sum(h) = n/2
    # Bounds: 0 <= h[i] <= 1, t >= 0
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = []
    b_ub = []
    m = 2 * n - 1
    for t_idx in range(m):
        shift = t_idx - (n - 1)
        row = np.zeros(n)
        for i in range(n):
            val = 0.0
            # a[i] = 1 - h_old[i+shift] if valid
            if 0 <= i + shift < n:
                val += 1.0 - h_old[i + shift]
            # b[i] = h_old[i-shift] if valid
            if 0 <= i - shift < n:
                val -= h_old[i - shift]
            row[i] = val
        row = np.append(row, -1.0 / dx)
        A_ub.append(row)
        b_ub.append(0.0)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h_old

corr[t_idx] ≈ sum_i valid [h[i]*(1-h_old[i+shift]) - h_old[i]*h[i+shift] + h_old[i]*h_old[i+shift]]

The linear terms in h are:
h[i]*(1-h_old[i+shift]) - h_old[i]*h[i+shift]

For the second term, h[i+shift] appears. Let j = i+shift, so h[j] appears with coefficient -h_old[j-shift].

So the total coefficient of h[j] in corr[t_idx] is:
(1-h_old[j+shift]) * [j valid for first term] - h_old[j-shift] * [j-shift valid for second term]

Where:
- First term: j is in range [0, n-1] and j+shift is in [0, n-1]
- Second term: j is in range [0, n-1] and j-shift is in [0, n-1]

So coefficient of h[j] in linearized corr[t_idx]:
= (1 - h_old[j+shift]) * [0≤j+shift<n] - h_old[j-shift] * [0≤j-shift<n]

And the constant term is:
sum_i valid h_old[i]*h_old[i+shift]

The LP constraint:
sum_j coeff[j] * h[j] + const ≤ t/dx

Or:
sum_j coeff[j] * h[j] - t/dx ≤ -const

In matrix form: [coeff[0], ..., coeff[n-1], -1/dx] @ [h, t] ≤ -const

This is correct.

Let me code this up properly:

```python
def _linearize_lp(h_old, n, dx):
    m = 2 * n - 1
    A_rows = []
    b_vals = []
    for t_idx in range(m):
        shift = t_idx - (n - 1)
        coeff = np.zeros(n)
        const = 0.0
        for j in range(n):
            c_j = 0.0
            if 0 <= j + shift < n:
                c_j += 1.0 - h_old[j + shift]
                const += h_old[j] * h_old[j + shift]
            if 0 <= j - shift < n:
                c_j -= h_old[j - shift]
            coeff[j] = c_j
        A_row = np.append(coeff, -1.0 / dx)
        A_rows.append(A_row)
        b_vals.append(-const)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h_old.copy()
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_round(h, n, corr, t_limit, n_coarse, n_bisect):
    improved = False
    order = np.random.permutation(n)
    for ai in range(n - 1):
        i = order[ai]
        for j in order[ai + 1:]:
            if time.time() > t_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.array([np.max(corr + d * gd) for d in ds])
            bi = np.argmin(vals)
            if vals[bi] >= np.max(corr) - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr += d_best * gd
            improved = True
    return h, corr, improved

def _slsqp(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _lp_step(h_old, n, dx):
    m = 2 * n - 1
    A_rows = []
    b_vals = []
    for t_idx in range(m):
        shift = t_idx - (n - 1)
        coeff = np.zeros(n)
        const = 0.0
        for j in range(n):
            c_j = 0.0
            if 0 <= j + shift < n:
                c_j += 1.0 - h_old[j + shift]
                const += h_old[j] * h_old[j + shift]
            if 0 <= j - shift < n:
                c_j -= h_old[j - shift]
            coeff[j] = c_j
        A_row = np.append(coeff, -1.0 / dx)
        A_rows.append(A_row)
        b_vals.append(-const)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return _repair(res.x[:n], n)
    return h_old.copy()

def _lp_loop(h, n, dx, t_limit, n_iter=200):
    for _ in range(n_iter):
        if time.time() > t_limit:
            break
        h_new = _lp_step(h, n, dx)
        c5_new = _c5(h_new, dx)
        c5_old = _c5(h, dx)
        if c5_new <= c5_old + 1e-12:
            h = h_new
        else:
            # Damped
            h_try = 0.5 * h + 0.5 * h_new
            h_try = _repair(h_try, n)
            if _c5(h_try, dx) <= c5_old:
                h = h_try
        c5_old = _c5(h, dx)
    return h
