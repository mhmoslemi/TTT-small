    if abs(s - target) > 1e-9:

                    h[idx] += a
                    diff -= a

                    diff += a

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(h, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sj(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        shift = k - (n - 1)
        m_lo = max(0, -shift)
        m_hi = min(n, n - shift)
        wl = w[k]
        for mi in range(m_lo, m_hi):
            g[mi] += wl * (1.0 - h[mi + shift])
            g[mi + shift] -= wl * h[mi]
    return g

def _grad_k(h, k):
    n = len(h)
    shift = k - (n - 1)
    m_lo = max(0, -shift)
    m_hi = min(n, n - shift)
    g = np.zeros(n)
    for mi in range(m_lo, m_hi):
        g[mi] += (1.0 - h[mi + shift])
        g[mi + shift] -= h[mi]
    return g

def _slp(h, n, max_iters=300, topk=10, use_tol=0.01):
    best_h = h.copy()
    best_val = _eval(h)
    alpha = 0.7
    stall = 0
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = corr.max()
        thresh = cmax - use_tol * cmax
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) < topk:
            extra = np.argsort(corr)[-topk:]
            active = np.unique(np.concatenate([active, extra]))
        active = np.sort(active[:topk])
        m = len(active)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((m + 2 * n, n + 1))
        b_ub = np.zeros(m + 2 * n)
        for i, k in enumerate(active):
            g = _grad_k(h, k)
            A_ub[i, :n] = g
            A_ub[i, n] = -1.0
            b_ub[i] = np.dot(g, h) - corr[k]
        A_ub[m:m+n, :n] = -np.eye(n)
        A_ub[m+n:m+2*n, :n] = np.eye(n)
        b_ub[m+n:m+2*n] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=[(0.0, 1.0)] * n + [(0.0, 2.0 * n)],
                          method='highs')
        except Exception:
            break
        if not res.success:
            break
        h_lp = res.x[:n]
        h_new = _repair(h + alpha * (h_lp - h), n)
        val = _eval(h_new)
        if val < best_val - 1e-15:
            h = h_new
            best_val = val
            best_h = h.copy()
            alpha = min(alpha * 1.15, 1.0)
            stall = 0
        else:
            alpha *= 0.45
            stall += 1
            if stall > 40 or alpha < 1e-9:
                break
    return best_h, best_val
