            if abs(h_new.sum() - target) > 1e-8:
                h_new *= target / h_new.sum()
                h_new = np.clip(h_new, 0, 1)
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new
        except Exception:
            pass

    # Phase 6: Try from complement
    if _el() < 3:
        try:
            hf = np.clip(1.0 - best_h, 0, 1)
            if abs(hf.sum() - target) > 1e-10:
                hf *= target / hf.sum()
                hf = np.clip(hf, 0, 1)
            res = minimize(_full_obj, hf, args=(100000,), method='SLSQP',
                           bounds=bounds_full,
                           constraints=[{'type': 'eq', 'fun': _sum_eq, 'args': (target,)}],
                           options={'maxiter': 1000, 'ftol': 1e-16})
            h_new = np.clip(res.x, 0, 1)
            if abs(h_new.sum() - target) > 1e-8:
                h_new *= target / h_new.sum()
                h_new = np.clip(h_new, 0, 1)
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new
        except Exception:
            pass

    best_h = np.clip(best_h, 0, 1)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
