                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep(h, n, corr, tlimit, nc=30, nb=40):
    imp = False
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for j in order[ai + 1:]:
            if time.time() > tlimit:
                return h, corr, imp
            hi, hj = h[i], h[j]
            dlo = max(-hi, hj - 1.0)
            dhi = min(1.0 - hi, hj)
            if dhi - dlo < 1e-13:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            cur = np.max(corr)
            ds = np.linspace(dlo, dhi, nc)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cur - 1e-14:
                continue
            lo = ds[max(0, bi - 1)]
            hi_b = ds[min(nc - 1, bi + 1)]
            for _ in range(nb):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] += d_best
            h[j] -= d_best
            corr = corr + d_best * gd
            imp = True
    return h, corr, imp

def _grad_phase(h, n, dx, tlimit, alpha0=30, alpha_max=3000):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    eta = 0.008
    alpha = alpha0
    stagnant = 0
    while time.time() < tlimit:
        c = _corr(h)
        g = _sm_grad2(h, alpha, n)
        f0 = _sm_val(c, alpha)
        accepted = False
        for _ls in range(45):
            h_try = _project_feasible(h - eta * g, n)
            if np.max(np.abs(h_try - h)) < 1e-15:
                break
            c_try = _corr(h_try)
            f_try = _sm_val(c_try, alpha)
            if f_try <= f0 - 0.4 * eta * np.dot(g, h - h_try):
                h = h_try
                accepted = True
                break
            eta *= 0.5
        if not accepted:
            eta *= 0.5
            if eta < 1e-8:
                eta = 0.008
        c5 = _c5(h, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stagnant = 0
            eta = min(eta * 1.08, 0.06)
            alpha = min(alpha + 2.0, alpha_max)
        else:
            stagnant += 1
            if stagnant > 30:
                alpha = min(alpha * 1.5, alpha_max)
                eta = 0.004
                stagnant = 0
        if alpha > alpha_max and stagnant > 60:
            break
    return best_h, best_c5

def _slsqp(h, n, dx, alpha, maxiter):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _sm_val(_corr(x), alpha)
    def eqfun(x):
        return float(np.sum(x)) - target
    cons = [{'type': 'eq', 'fun': eqfun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _fourier_init(n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 12):
        amp = rng.uniform(0.02, 0.12) / m
        ph = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * m * x + ph)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: SLSQP warmup (quick)
    p1_end = t0 + min(50.0, budget_s * 0.06)
    for al in [40, 150, 400]:
        if time.time() > p1_end:
            break
        h2 = _slsqp(best_h, n, dx, al, 250)
        c5 = _c5(h2, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()

    # Phase 2: Analytical gradient descent
    p2_end = time.time() + (budget_s - 180) * 0.35
    h, c5 = _grad_phase(best_h, n, dx, p2_end, alpha0=40, alpha_max=2500)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Coordinate sweeps
    p3_end = time.time() + (budget_s - 120) * 0.35
    stall = 0
    while time.time() < p3_end:
        h = best_h.copy()
        corr = _corr(h)
        chunk = min(35.0, p3_end - time.time())
        h, corr, imp = _sweep(h, n, corr, time.time() + chunk, 30, 45)
        c5 = _c5(h, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 4 and time.time() < p3_end - 40:
                h_p = best_h + 0.04 * np.random.randn(n)
                h_p = _repair(h_p, n)
                h_p, c5_p = _grad_phase(h_p, n, dx, time.time() + 25, 200, 1500)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()
                stall = 0

    # Phase 4: Alternate gradient + sweep
    p4_end = time.time() + (budget_s - 80) * 0.25
    while time.time() < p4_end:
        h, c5 = _grad_phase(best_h, n, dx, time.time() + 20, 200, 2500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h = best_h.copy()
        corr = _corr(h)
        h, corr, _ = _sweep(h, n, corr, time.time() + 20, 25, 40)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: High-precision final sweeps
    rem = budget_s - (time.time() - t0) - 25
    if rem > 15:
        h = best_h.copy()
        corr = _corr(h)
        h, corr, _ = _sweep(h, n, corr, time.time() + min(rem * 0.5, 40), 35, 55)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Final SLSQP polish
    rem = budget_s - (time.time() - t0) - 3
    if rem > 10:
        h2 = _slsqp(best_h, n, dx, 2000, 400)
        c5 = _c5(h2, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()
        h = best_h.copy()
        corr = _corr(h)
        h, corr, _ = _sweep(h, n, corr, time.time() + min(rem * 0.4, 15), 30, 50)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
