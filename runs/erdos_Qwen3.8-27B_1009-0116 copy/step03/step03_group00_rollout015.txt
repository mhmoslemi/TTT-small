    """Gradient descent focusing on top lags"""
    h = h.copy()
    c5c = _c5(h, n)
    dx = 2.0 / n
    for _ in range(n_iter):
        c = np.correlate(h, 1.0-h, mode='full') * dx
        # Get top max_lags
        top_idx = np.argsort(c)[::-1][:max_lags]
        # Compute gradient of sum of top lags
        g = np.zeros(n)
        for k in top_idx:
            lag = k - (n-1)
            lo = max(0, -lag)
            hi = min(n, n-lag)
            g[lo:hi] -= h[lo+lag:hi+lag]
            g[lo+lag:hi+lag] += (1.0 - h[lo:hi])
        g *= dx / len(top_idx)
        # Project: move in -g direction, project to [0,1], fix sum
        h_new = h - step_size * g

        if c5_new < c5c:
            h, c5c = h_new, c5_new
        else:
            step_size *= 0.8
            if step_size < 1e-8:
                break

        if time.time() > end_t:

        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for k in top:
            lag = k - (n - 1)
            lo = max(0, -lag)
            hi = min(n, n - lag)
            g[lo:hi] -= h[lo+lag:hi+lag]
            g[lo+lag:hi+lag] += (1.0 - h[lo:hi])
        g *= dx / top_k
        step = min(ss, 0.05)
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _repair(h_new, n)
        c5n = _c5(h_new, n)
        if c5n < c5c - 1e-15:
            h, c5c = h_new, c5n
            ss = min(ss * 1.1, 0.05)
        else:
            ss *= 0.85
            if ss < 1e-10:
                ss = 0.001
    return h, c5c

def _pair_search(h, n, end_t, rng):
    c5c = _c5(h, n)
    steps = np.array([0.002, 0.005, 0.012, 0.03])
    stride = max(1, n // 80)
    for _pass in range(500):
        if time.time() > end_t:
            break
        imp = False
        for i in range(0, n, stride):
            if time.time() > end_t:
                break
            j_start = max(i + 1, 0)
            for j in range(j_start, min(i + stride * 3, n)):
                hi0, hj0 = h[i], h[j]
                for d in steps:
                    for sgn in (1, -1):
                        dd = sgn * d
                        if not (0 <= hi0 + dd <= 1 and 0 <= hj0 - dd <= 1):
                            continue
                        h[i], h[j] = hi0 + dd, hj0 - dd
                        c = _c5(h, n)
                        if c < c5c - 1e-14:
                            c5c = c
                            hi0, hj0 = h[i], h[j]
                            imp = True
                        else:
                            h[i], h[j] = hi0, hj0
                h[i] = hi0
        if not imp:
            break
    return h, c5c

def _interp(h, ns, nd):
    xs = np.arange(ns) * (2.0 / ns)
    xd = np.arange(nd) * (2.0 / nd)
    pos = xd * ns / 2.0
    i0 = pos.astype(int) % ns
    i1 = (i0 + 1) % ns
    f = pos - pos.astype(int)
    return (1.0 - f) * h[i0] + f * h[i1]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def eta():
        return time.time() - t0

    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5, best_n = h0.copy(), _c5(h0, n), n

    # Phase 1: SLSQP at n with diverse alphas and restarts
    alpha_schedule = [2000, 5000, 12000, 30000, 80000, 200000]
    for alpha in alpha_schedule:
        nr = 12 if alpha < 12000 else 7
        for r in range(nr):
            if eta() > budget_s - 300:
                break
            if r == 0:
                hi = best_h.copy()
            elif r == 1:
                hi = _repair(best_h + 0.03 * rng.randn(n), n)
            elif r == 2:
                hi = _repair(0.5 * best_h + 0.5 * rng.rand(n), n)
            else:
                hi = _repair(best_h + (0.01 + 0.008 * r) * rng.randn(n), n)
            hc = _slsqp(hi, n, alpha, 700)
            c = _c5(hc, n)
            if c < best_c5:
                best_c5, best_h = c, hc
        if eta() > budget_s - 300:
            break

    # Phase 2: Fast gradient descent at n
    if eta() < budget_s - 250:
        best_h, best_c5 = _grad_descent(best_h, n, t0 + budget_s - 250, top_k=5)

    # Phase 3: Pair search at n
    if eta() < budget_s - 180:
        best_h, best_c5 = _pair_search(best_h, n, t0 + budget_s - 180, rng)

    # Phase 4: Higher resolutions
    for n2 in [100, 150, 200]:
        if eta() > budget_s - 80:
            break
        h2 = _repair(_interp(best_h, best_n, n2), n2)
        c5_2 = _c5(h2, n2)
        improved = False
        for alpha in [5000, 20000]:
            if eta() > budget_s - 50:
                break
            if n2 <= 120:
                hc = _slsqp(h2, n2, alpha, 500)
            else:
                hc = _lbfgs_pen(h2, n2, alpha, 350)
            c = _c5(hc, n2)
            if c < best_c5:
                best_c5, best_h, best_n, improved = c, hc, n2, True
        if best_n == n2 and eta() < budget_s - 40:
            best_h, best_c5 = _grad_descent(best_h, n2, t0 + budget_s - 60, top_k=5)
            best_h, best_c5 = _pair_search(best_h, n2, t0 + budget_s - 40, rng)

    # Phase 5: Final intensive pair search
    if eta() < t0 + budget_s - 20:
        best_h, best_c5 = _pair_search(best_h, best_n, t0 + budget_s - 20, rng)

    # Phase 6: Ultra-fine random coordinate descent
    if eta() < t0 + budget_s - 3:
        h = best_h.copy()
        cc = _c5(h, best_n)
        tiny = np.array([1e-4, 3e-4, 7e-4, 1.5e-3, -1e-4, -3e-4, -7e-4, -1.5e-3])
        while time.time() < t0 + budget_s - 3:
            i = rng.randint(best_n)
            j = rng.randint(best_n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, best_n)
                if c < cc:
                    cc = c
                    if c < best_c5:
                        best_c5, best_h = c, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
