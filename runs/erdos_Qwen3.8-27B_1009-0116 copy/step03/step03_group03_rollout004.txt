    # Phase 3: SLSQP with log-sum-exp
    if _tl() > 80:
        for alpha in [30000, 150000, 500000]:
            if _tl() < 80:
                break
            try:
                x_opt = _slsqp_smooth(best_h, target, alpha, 400)
                hc = _repair(x_opt, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except:
                pass

    # Phase 4: Multi-restart with structured perturbations
    if _tl() > 50:
        for r in range(8):
            if _tl() < 50:
                break
            if r < 4:
                scale = 0.01 * (1.0 + 0.4 * r)
                h_p = _repair(best_h + scale * rng.randn(n), n)
            else:
                # Structured: add sinusoidal perturbation
                freq = (r - 3 + 1) * 2.0 / n
                h_p = best_h + 0.02 * np.sin(2.0 * np.pi * freq * np.arange(n))
                h_p = _repair(h_p, n)
            try:
                x_opt = _slsqp_smooth(h_p, target, 200000, 300)
                hc = _repair(x_opt, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except:
                pass

    # Phase 5: Fine random pair search (accept-only)
    if _tl() > 15:
        h = best_h.copy()
        c_cur = best_c5
        steps = [0.001, 0.004, 0.01, 0.03, -0.001, -0.004, -0.01, -0.03]
        for _ in range(200000):
            if _tl() < 15:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            for d in steps:
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < c_cur:
                        c_cur = c
                        best_c5 = c
                        best_h = h.copy()
                        break
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d

    # Phase 6: Final directed pair polish
    if _tl() > 10:
        h2, best_c5 = _pair_swap(best_h, n, dx, best_c5, t0, budget_s, rng, 20000)
        if _c5(h2, dx) < best_c5:
            best_c5 = _c5(h2, dx)
            best_h = h2.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
