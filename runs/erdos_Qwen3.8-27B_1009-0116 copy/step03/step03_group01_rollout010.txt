    # numpy.correlate(a, b, 'full')[k] = sum_i a[i] * b[i-(k-(len(b)-1))]
    # Actually: c[k] = sum_i a[i] b[i + (len(b)-1-k)]  for appropriate range
    # Let me just use the definition: for k >= 0, c[k] = sum_{i=k-(n-1)}^{n-1} a[i] b[i-(k-(n-1))]
    # This is getting confusing. Let me just compute gradient directly.
    
    # For the full correlate: c[k] = sum_j a[j] * b[j - (k - (n-1))]
    # where the sum is over j such that 0 <= j < n and 0 <= j-(k-(n-1)) < n
    # Let m = k - (n-1), then c[k] = sum_j a[j] * b[j-m] for valid j
    # dc[k]/da[j] = b[j-m] (if valid)
    # dc[k]/db[j] = a[j+m] (if valid)
    # But b = 1-h, so db[j] = -dh[j]
    # dc[k]/dh[j] from a part: b[j-m] = 1-h[j-m]
    # dc[k]/dh[j] from b part: -a[j+m] = -h[j+m]
    
    g = np.zeros(n)
    m = kmax - (n - 1)  # the actual lag offset
    # a part: for j where 0<=j<n and 0<=j-m<n: dc/da[j] = b[j-m]
    j_start = max(0, m)
    j_end = min(n, n + m)
    for j in range(j_start, j_end):
        g[j] += (1.0 - h[j - m])  # b[j-m]
    # b part: for j where 0<=j<n and 0<=j+m<n: dc/db[j] = a[j+m], dc/dh[j] = -a[j+m]
    j_start2 = max(0, -m)
    j_end2 = min(n, n - m)
    for j in range(j_start2, j_end2):
        g[j] -= h[j + m]  # -a[j+m]
    
    g *= dx
    return c5, g

def _run_single(h0, n, budget_remaining, rng):
    """Optimize from h0, return best h and c5."""
    h = _repair(h0.copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    t0 = time.time()
    
    def _el():
        return time.time() - t0
    
    # Phase A: SLSQP with soft-max at multiple alphas
    for alpha in [500, 5000, 50000, 500000]:
        if _el() > budget_remaining - 50:
            break
        h = _opt_slsqp(h, n, alpha, 2000, _el, budget_remaining - 50)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase B: Subgradient descent
    if _el() < budget_remaining - 30:
        h = best_h.copy()
        lr = 0.01
        for it in range(5000):
            if _el() > budget_remaining - 30:
                break
            c5, g = _eval_grad(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            # Projected subgradient step (maintain sum constraint)
            g_proj = g - g.mean()  # project onto sum-preserving subspace
            h_new = h - lr * g_proj
            h_new = _repair(h_new, n)
            # Accept if better or always (for subgradient)
            if _eval(h_new) <= c5 + 1e-10:
                h = h_new
            else:
                lr *= 0.95
            if lr < 1e-8:
                lr = 0.01
    
    # Phase C: Intensive pairwise local search
    if _el() < budget_remaining - 10:
        h = best_h.copy()
        c5c = _eval(h)
        # Try many small random perturbations
        step = 0.005
        for it in range(200000):
            if _el() > budget_remaining - 10:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = step * rng.randn()
            ni = h[i] + d
            nj = h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-15:
                    c5c = c5n
                    step = min(step * 1.05, 0.02)
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
                    step *= 0.999
                    if step < 1e-7:
                        step = 0.003
                        it = max(0, it - 5000)  # effectively restart step decay
    
    return best_h, best_c5

def _opt_slsqp(h, n, alpha, maxiter, _el, deadline):
    """SLSQP with soft-max objective."""
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.sum(np.exp(s - s.max())) / alpha  # shifted for numerical stability

    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        sm = s - s.max()
        w = np.exp(sm)
        w /= w.sum()
        g = np.zeros(n)
        # Gradient of c[k] w.r.t. h for each k, weighted by w
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            wk = w[k]
            m_off = k - (n - 1)
            # a part
            j_start = max(0, m_off)
            j_end = min(n, n + m_off)
            for j in range(j_start, j_end):
                g[j] += wk * (1.0 - x[j - m_off])
            # b part  
            j_start2 = max(0, -m_off)
            j_end2 = min(n, n - m_off)
            for j in range(j_start2, j_end2):
                g[j] -= wk * x[j + m_off]
        # Additional gradient from the log-sum-exp and max
        # Actually for the soft-max: f = m + log(sum(exp(s))/alpha) = m + log(sum(exp(alpha*(c-m))))/alpha
        # df/dx = (1/alpha) * sum_k alpha * exp(alpha*(c[k]-m)) / sum_k' exp(alpha*(c[k']-m)) * dc[k]/dx
        #        = sum_k w[k] * dc[k]/dx
        # where w[k] = exp(alpha*(c[k]-m)) / sum exp(alpha*(c[k']-m))
        # But we also have the -m in the exponent, so:
        # Actually f = m + (1/alpha)*log(sum(exp(alpha*(c-m))))
        # df/dx = dm/dx + (1/alpha) * sum_k alpha*exp(alpha*(c[k]-m))*dc[k]/dx / sum exp(alpha*(c'-m))
        #        = dm/dx + sum_k w[k] dc[k]/dx
        # where w[k] = exp(alpha*(c[k]-m)) / sum exp(alpha*(c'-m))
        # And dm/dx is the gradient of c[kmax] (the max component)
        # But this is getting complicated. The gradient I computed above (sum w[k] dc[k]/dx) is the gradient of the "log-sum-exp part".
        # The full gradient also includes dm/dx. But for large alpha, m dominates, so dm/dx ≈ gradient of c[kmax].
        # For simplicity, let's add it:
        kmax = np.argmax(c)
        m_off = kmax - (n - 1)
        j_start = max(0, m_off)
        j_end = min(n, n + m_off)
        for j in range(j_start, j_end):
            g[j] += (1.0 - x[j - m_off])
        j_start2 = max(0, -m_off)
        j_end2 = min(n, n - m_off)
        for j in range(j_start2, j_end2):
            g[j] -= x[j + m_off]
        
        return g  # Note: missing the dx factor, but it's a constant multiplier

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _sum_fun(x, n):
    return np.sum(x) - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)
