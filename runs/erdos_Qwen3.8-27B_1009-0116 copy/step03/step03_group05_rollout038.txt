    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:

                diff -= a

                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break

def _grad_one_lag(h, k, n):
    """Gradient of correlation at lag k (where full correlation index = k + n - 1)."""
    L = k  # actual lag in the full correlation
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        # d c[L]/d h[j] = (1-h[j-L])*I[j>=L] - h[j+L]*I[j<n-L]
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        M = -L
        # d c[-M]/d h[j] = (1-h[j+M])*I[j<n-M] - h[j-M]*I[j>=M]
        g[:n-M] += 1.0 - h[M:]
        g[M:] -= h[:n-M]
    return g

def _grad_top_k(h, n, top_k):
    """Average gradient of top-k correlations."""
    c = _corr(h)
    idx = np.argsort(c)[::-1][:top_k]
    G = np.zeros(n)
    for ki in idx:
        L = ki - (n - 1)
        G += _grad_one_lag(h, L, n)
    return G / len(idx), c, idx

def _softmax_obj(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softmax_jac(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)

        pk = pw[k]
        if pk < 1e-14:

        G += pk * _grad_one_lag(x, L, n)
    G *= 1.0 / alpha
    G += 2.0 * lam * (np.sum(x) - n / 2.0)
    return G

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _grad_lag(h, L, n):
    g = np.zeros(n)
    if L == 0:
        return 1.0 - 2.0 * h
    if L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += 1.0 - h[M:]
        g[M:] -= h[:n - M]
    return g

def _sm_obj(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(x) - n / 2.0) ** 2

def _sm_jac(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    G = np.zeros(n)
    for k in range(2 * n - 1):
        pk = pw[k]
        if pk < 1e-14:
            continue
        G += pk * _grad_lag(x, k - (n - 1), n)
    G /= alpha
    G += 2.0 * lam * (np.sum(x) - n / 2.0)
    return G

def _antiper_obj(az, half, alpha, dx):
    h = np.empty(2 * half)
    h[:half] = az
    h[half:] = 1.0 - az
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m * dx + np.log(np.sum(np.exp(s))) / alpha

def _pg_minimax(h, n, dx, t0, budget, reserve):
    h_cur = h.copy()
    c5_cur = _c5_val(h_cur, dx)
    step0 = 0.04
    for _ in range(800):
        if time.time() - t0 > budget - reserve:
            break
        c = _corr(h_cur)
        top3 = np.argsort(c)[::-1][:3]
        G = np.zeros(n)
        for ki in top3:
            G += _grad_lag(h_cur, ki - (n - 1), n)
        G /= 3.0
        G -= G.mean()
        moved = False
        sp = step0
        for _sh in range(8):
            h_try = _repair(h_cur - sp * G, n)
            c5_try = _c5_val(h_try, dx)
            if c5_try < c5_cur - 1e-16:
                h_cur = h_try
                c5_cur = c5_try
                moved = True
                break
            sp *= 0.4
        if moved:
            step0 = min(step0 * 1.03, 0.1)
        else:
            step0 *= 0.65
            if step0 < 1e-7:
                break
    return h_cur, c5_cur

def _pair_ternary(h, n, dx, t0, budget, reserve):
    h_cur = h.copy()
    c5_cur = _c5_val(h_cur, dx)
    best_local = c5_cur
    for sweep in range(80):
        if time.time() - t0 > budget - reserve:
            break
        c = _corr(h_cur)
        top5 = np.argsort(c)[::-1][:5]
        G = np.zeros(n)
        for ki in top5:
            G += _grad_lag(h_cur, ki - (n - 1), n)
        order = np.argsort(np.abs(G))[::-1][:25]
        found = False
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + 12, len(order))):
                j = order[jj]
                lo = -min(h_cur[i], 1.0 - h_cur[j])
                hi = min(1.0 - h_cur[i], h_cur[j])
                if hi - lo < 1e-8:
                    continue
                for _ts in range(14):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    h_cur[i], h_cur[j] = h_cur[i] + m1, h_cur[j] - m1
                    f1 = _c5_val(h_cur, dx)
                    h_cur[i], h_cur[j] = h_cur[i] + m2, h_cur[j] - m2
                    f2 = _c5_val(h_cur, dx)
                    h_cur[i], h_cur[j] = h_cur[i] - m2, h_cur[j] + m2
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                d = (lo + hi) / 2.0
                if abs(d) > 1e-8:
                    h_cur[i] += d
                    h_cur[j] -= d
                    h_cur = _repair(h_cur, n)
                    c5_new = _c5_val(h_cur, dx)
                    if c5_new < c5_cur - 1e-14:
                        c5_cur = c5_new
                        found = True
                    break
        if not found:
            break
    return h_cur, c5_cur

def _anneal(h, n, dx, rng, t0, budget, reserve):
    h_cur = h.copy()
    c5_cur = _c5_val(h_cur, dx)
    T = 4e-5
    for _ in range(80000):
        if time.time() - t0 > budget - reserve:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.025, 0.025)
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        h_cur = _repair(h_cur, n)
        c_new = _c5_val(h_cur, dx)
        if c_new < c5_cur or (T > 1e-10 and rng.rand() < np.exp(-(c_new - c5_cur) / T)):
            c5_cur = c_new
        else:
            h_cur[i] -= d
            h_cur[j] += d
            h_cur = _repair(h_cur, n)
        T *= 0.9997
        if T < 1e-10:
            T = 1e-10
    return h_cur, c5_cur

def _fine_polish(h, n, dx, rng, t0, budget, reserve):
    h_cur = h.copy()
    c5_cur = _c5_val(h_cur, dx)
    deltas = np.array([0.0004, 0.001, 0.002, 0.004, -0.0004, -0.001, -0.002, -0.004])
    for _ in range(30000):
        if time.time() - t0 > budget - reserve:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = deltas[rng.randint(len(deltas))]
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h_cur[i], h_cur[j] = ni, nj
            c = _c5_val(h_cur, dx)
            if c < c5_cur:
                c5_cur = c
            else:
                h_cur[i], h_cur[j] = ni - d, nj + d
    return h_cur, c5_cur

def _try_lbfgs(h_start, n, dx, alphas, lams, t0, budget, reserve, maxiter=400):
    bounds_list = [(0.0, 1.0)] * n
    best = h_start.copy()
    best_c5 = _c5_val(best, dx)
    for a, lam in zip(alphas, lams):
        if time.time() - t0 > budget - reserve:
            break
        try:
            res = minimize(partial(_sm_obj, alpha=a, lam=lam, n=n), best.copy(),
                           jac=partial(_sm_jac, alpha=a, lam=lam, n=n),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            hc = _repair(res.x, n)
            c5 = _c5_val(hc, dx)
            if c5 < best_c5:
                best_c5, best = c5, hc
        except Exception:
            pass
    return best, best_c5
