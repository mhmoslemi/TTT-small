def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * 2.0 / n)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def slsqp_opt(h, n, alpha, maxiter=400):
    target = n / 2.0
    def obj(v):
        corr = compute_corr(v)
        m = np.max(corr)

        val = m + np.log1p(np.sum(np.exp(s)) - 1.0) / alpha

                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15})
