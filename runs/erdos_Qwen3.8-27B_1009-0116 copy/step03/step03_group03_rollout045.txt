When h[i] → h[i]+t and h[j] → h[j]-t:
- Terms where p = i: (h[i]+t)(1-h[i+m-n+1]) if i+m≠j, or (h[i]+t)(1-h[j]+t) if i+m=j
- Terms where p+m-n+1 = i, i.e., p = i-m+n-1: h[i-m+n-1](1-(h[i]+t)) if i-m+n-1 ≠ j
- Terms where p = j: (h[j]-t)(1-h[j+m-n+1])
- Terms where p+m-n+1 = j: h[j-m+n-1](1-(h[j]-t))

def _project(h, n):

        if interior.sum() > 0:
            h[interior] += d / interior.sum()
        else:
            h += d / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            add = min(d, 1.0 - h[idx]) if d > 0 else max(d, h[idx] - 1.0)
            h[idx] += add
            d -= add
            if abs(d) < 1e-15:

def _grad_lag(h, n, m):
    k = m - (n - 1)

    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += (1.0 - h[-k:])
        g[-k:] -= h[:n + k]
    return g

    c = _corr(x)

    s = np.clip(alpha * (c - m), -300, 300)
