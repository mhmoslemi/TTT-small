from scipy.optimize import minimize, differential_evolution
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _project_sum(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max()
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def _subgrad_active(h, n, k_active):
    """Subgradient of max using active lags k_active (indices in correlation array)"""
    dx = 2.0 / n
    g = np.zeros(n)
    for m in k_active:
        lag = m - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += (1.0 - h[i + lag])
                g[i + lag] -= h[i]
        else:
            d = -lag
            for i in range(d, n):
                g[i] += (1.0 - h[i - d])
                g[i - d] -= h[i]
    g /= len(k_active)
    return g * dx

def _optimize_subgrad(h0, n, max_iter, eta0=0.01, momentum=0.9):
    """Projected subgradient with momentum"""
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5(h, n)
    eta = eta0
    
    for t in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        c5_cur = np.max(corr)
        # Active set: lags within 5% of max
        threshold = c5_cur - 0.05 * c5_cur
        active = list(np.where(corr >= threshold)[0])
        
        g = _subgrad_active(h, n, active)
        v = momentum * v - eta * g
        h_new = h + v
        h_new = _project_sum(h_new, n)
        
        c5_new = _c5(h_new, n)
        if c5_new < c5_cur:
            h = h_new
            eta *= 1.01
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            eta *= 0.9
            h = h  # don't update
        
        if eta < 1e-6:
            break
    
    return best_h, best_c5

from scipy.optimize import minimize
import time

def _c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _project(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # Fix any residual
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - p)
        else:
            order = np.argsort(p)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - p[idx])
                p[idx] += add
                diff -= add
            else:
                sub = min(-diff, p[idx])
                p[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return p

def _topk_obj_grad(x, n, k, lam):
    """Objective: max + lam * mean of top-k. Returns (val, grad)."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    idx = np.argsort(corr)[::-1][:k]
    obj = m + lam * np.mean(corr[idx])
    # Gradient: subgradient of max + lam/k * mean gradient of top-k
    g_max = np.zeros(n)
    # For the max, use the argmax subgradient
    # For the top-k mean, average gradients over top-k
    g = np.zeros(n)
    for mk in idx:
        lag = mk - (n - 1)
        if lag >= 0:
            i0, i1 = 0, n - lag
        else:
            i0, i1 = -lag, n
        for i in range(i0, i1):
            j = i + lag
            g[i] += (1.0 - x[j])
            g[j] -= x[i]
    g *= dx / k
    return obj, g

def _slsqp(h0, n, k, lam, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}
    ]
    res = minimize(
        lambda x: _topk_obj_grad(x, n, k, lam)[0],
        h0, jac=lambda x: _topk_obj_grad(x, n, k, lam)[1],
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-18}
    )
    return res.x

def _coord_search(h, n, deadline, rng):
    """Coordinate descent: try moving mass from one position to another."""
    c5_cur = _c5(h, n)
    steps = np.array([0.001, 0.003, 0.008, 0.02, 0.05])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                if time.time() > deadline:
                    break
                for s in steps:
                    if h[i] < s or h[j] < s:
                        continue
                    h[i] -= s
                    h[j] += s
                    c = _c5(h, n)
                    if c < c5_cur - 1e-14:
                        c5_cur = c
                        improved = True
                    else:
                        h[i] += s
                        h[j] -= s
                    # Also try reverse
                    if h[j] < s or h[i] < s:
                        continue
                    h[i] += s
                    h[j] -= s
                    c = _c5(h, n)
                    if c < c5_cur - 1e-14:
                        c5_cur = c
                        improved = True
                    else:
                        h[i] -= s
                        h[j] += s
    return h, c5_cur

def _sca_step(h0, n):
    """One SCA iteration: solve linearized LP."""
    dx = 2.0 / n
    c = np.correlate(h0, 1.0 - h0, mode='full') * dx
    nc = len(c)  # 2n-1
    
    # Build LP: minimize t
    # Variables: [h[0],...,h[n-1], t]
    # Objective: minimize t → c_lin = [0,...,0,1]
    c_lin = np.zeros(n+1)
    c_lin[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: -∇g_k(h0) @ h + t >= -g_k(h0) + ∇g_k(h0) @ h0
    # i.e., ∇g_k(h0) @ h - t <= g_k(h0) - ∇g_k(h0) @ h0
    # Equivalently: -∇g_k(h0) @ h + t >= g_k(h0) - ∇g_k(h0) @ h0
    # In standard form (<=): ∇g_k(h0) @ h - t <= g_k(h0) - ∇g_k(h0) @ h0
    
    A_ub = np.zeros((nc, n+1))
    b_ub = np.zeros(nc)
    
    for mk in range(nc):

        gk = np.zeros(n)
        if lag >= 0:
            for i in range(n-lag):
                gk[i] += (1.0 - h0[i+lag])
                gk[i+lag] -= h0[i]

                gk[i] += (1.0 - h0[i-d])
                gk[i-d] -= h0[i]
        gk *= dx
        
        A_ub[mk, :n] = gk
        A_ub[mk, n] = -1.0
        b_ub[mk] = c[mk] - gk @ h0
    
    # Box constraints: 0 <= h <= 1 → -h <= 0 and h <= 1
    # h <= 1: A @ x <= 1
    # 0 <= h: -h <= 0 → included in bounds
    
    # Use bounds instead of inequality constraints for box
    bounds = [(0.0, 1.0)] * n + [(None, None)]  # t is free
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    from scipy.optimize import linprog
    res = linprog(c_lin, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    return res.x[:n]

But there's a subtlety: the SCA direction. In SCA for non-convex problems, we solve:
minimize g_k(h₀) + ∇g_k(h₀)ᵀ(h - h₀) for all k ≥ t

But we need to ensure the linearization is an UNDER-estimator (for a minimization problem). For a concave function, the tangent is an over-estimate; for a convex function, it's an under-estimate.

g_k(h) = Σ h[i](1-h[i+lag]) is NOT convex or concave in general. It's a quadratic with an indefinite Hessian (the cross terms make it indefinite).

So the linearization is neither an over- nor under-estimate globally. The SCA might not monotonically decrease the objective.

- The LP solution is close to h₀ (locally, the linearization is accurate)
