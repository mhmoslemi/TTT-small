                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            if abs(d_best) < 1e-10:
                continue
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

def _obj_factory(h, alpha):
    def obj(x):
        c = _corr(x)
        m = float(np.max(c))
        s = alpha * (c - m)
        s = np.clip(s, -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    return obj

def _slsqp_step(h, n, dx, target, bounds, alpha, maxiter):
    def eq_con(x):
        return float(np.sum(x)) - target
    cons = [{'type': 'eq', 'fun': eq_con}]
    obj = _obj_factory(h, alpha)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _binary_projection(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    for k in range(target):
        h_b[idx[k]] = 1.0
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-up with increasing sharpness
    alphas = [40, 150, 400, 1000, 2500]
    for alpha in alphas:
        if time.time() - t0 > budget_s * 0.12:
            break
        hc = _slsqp_step(best_h, n, dx, target, bounds, alpha, 500)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 2: Main loop - alternating single-coord and pair sweeps
    stall = 0
    main_iter = 0
    while time.time() - t0 < budget_s - 120:
        main_iter += 1
        h = best_h.copy()
        corr = _corr(h)
        time_left = budget_s - 120 - (time.time() - t0)

        # Single coordinate sweep (fast, many passes)
        chunk_s = max(5.0, min(25.0, time_left * 0.15))
        h, corr, imp1 = _sweep_single(h, n, corr, time.time() + chunk_s, max_sweeps=15)

        # Pair sweep (slower, more global)
        chunk_p = max(10.0, min(40.0, time_left * 0.25))
        h, corr, imp2 = _pair_sweep(h, n, corr, time.time() + chunk_p, 32, 40)

        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        # Periodic SLSQP with increasing alpha
        if main_iter % 3 == 0 and time.time() - t0 < budget_s - 140:
            alpha = 300 + 200 * min(stall, 15)
            hc = _slsqp_step(best_h, n, dx, target, bounds, min(alpha, 8000), 400)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _sweep_single(hc, n, hc_corr, time.time() + 10, 8)
            hc, hc_corr, _ = _pair_sweep(hc, n, hc_corr, time.time() + 15, 24, 30)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc.copy()
                stall = 0

        # Perturbation / restart when stalled
        if stall >= 4 and time.time() - t0 < budget_s - 160:
            pert = min(0.06, 0.015 + 0.01 * stall)
            h_pert = best_h + pert * np.random.randn(n)
            h_pert = _repair(h_pert, n)
            hc = _slsqp_step(h_pert, n, dx, target, bounds, 600, 350)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _sweep_single(hc, n, hc_corr, time.time() + 8, 10)
            hc, hc_corr, _ = _pair_sweep(hc, n, hc_corr, time.time() + 15, 28, 35)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc.copy()
            stall = 0

        # Binary projection heuristic (every 8 iterations when stalled)
        if stall >= 7 and main_iter % 8 == 0 and time.time() - t0 < budget_s - 160:
            h_bin = _binary_projection(best_h, n)
            hc = _slsqp_step(h_bin, n, dx, target, bounds, 200, 400)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _sweep_single(hc, n, hc_corr, time.time() + 10, 10)
            hc, hc_corr, _ = _pair_sweep(hc, n, hc_corr, time.time() + 15, 28, 35)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc.copy()
            stall = 0

    # Phase 3: Final high-precision refinement
    if time.time() - t0 < budget_s - 25:
        for _ in range(6):
            if time.time() - t0 > budget_s - 12:
                break
            h = best_h.copy()
            corr = _corr(h)
            h, corr, _ = _sweep_single(h, n, corr, time.time() + 12, 20, 160)
            h, corr, _ = _pair_sweep(h, n, corr, time.time() + 18, 48, 50)
            c5 = _c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: Final SLSQP polish at very high alpha
    if time.time() - t0 < budget_s - 4:
        for alpha in [3000, 10000]:
            if time.time() - t0 > budget_s - 3:
                break
            hc = _slsqp_step(best_h, n, dx, target, bounds, alpha, 800)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
