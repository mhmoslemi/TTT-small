def _grad_corr_lag(h, lag, n):
    """Gradient of corr_lag w.r.t. h."""
    g = np.zeros(n)
    m = abs(lag)
    if lag >= 0:
        # corr = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
        for i in range(n - m):
            g[i + m] += (1.0 - h[i])  # from h[i+m]
            g[i] -= h[i + m]  # from (1-h[i])
    else:
        # corr = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        for i in range(n - m):
            g[i] += (1.0 - h[i + m])  # from h[i]
            g[i + m] -= h[i]  # from (1-h[i+m])
    return g
