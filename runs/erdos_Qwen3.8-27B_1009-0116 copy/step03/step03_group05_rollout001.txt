t ≥ sum_i h[i] - h^T B_k h
h^T B_k h - sum_i h[i] + t ≥ 0

If B_k is positive semidefinite, this is a convex constraint. But B_k is generally not PSD (it's related to a shift matrix).

So it's a general quadratic constraint, making this a QCP (quadratically constrained program). SCIP or another solver might handle this.

def _equalize_dir(h, top_idx, n):
    """Compute the direction that equalizes the top correlations' decrease rate."""
    k = len(top_idx)
    # Compute gradients for each active correlation
    grads = np.zeros((k, n))
    for i, ki in enumerate(top_idx):
        L = ki - (n - 1)
        g = np.zeros(n)
        if L == 0:
            g[:] = (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] = (1.0 - h[:n - L])
            g[:n - L] = -h[L:]
        else:
            M = -L
            g[:n - M] = (1.0 - h[M:])
            g[M:] = -h[:n - M]
        grads[i] = g
    # G matrix: G[i,j] = sum(grad_i * grad_j)
    G = grads @ grads.T  # k x k
    # w = G^{-1} * 1
    try:
        w = np.linalg.solve(G, np.ones(k))
    except:
        w = np.ones(k) / k
    w = np.clip(w, 0, None)  # ensure positive weights
    if w.sum() < 1e-10:
        w = np.ones(k) / k
    else:
        w = w / w.sum()
    # Direction: sum w_i * grad_i (negated for descent)
    d = grads.T @ w
    # Project onto feasible set (make mean zero to preserve sum)
    d -= d.mean()
    return d, w

                    best_c = c5_cur
                    for d in deltas:
                        ni, nj = h[i] + d, h[j] - d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx)
                            if c < best_c:
                                best_c = c
                                best_d = d
                            h[i], h[j] = ni - d, nj + d
                    if best_d != 0.0:
                        h[i] += best_d
                        h[j] -= best_d
                        h = _repair(h, n)
                        c5_cur = _c5(h, dx)
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        break
            if not improved:
                break
            g = _sm_grad(h, 2000000, n, 0.0)
            order = np.argsort(np.abs(g))[::-1][:35]

    # Phase 6: Ultra-fine random polish
    if _el() < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        tiny = np.array([1e-4, 3e-4, 7e-4, 1.5e-3, -1e-4, -3e-4, -7e-4, -1.5e-3])
        for _ in range(20000):
            if _el() > budget_s - 5:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i] -= d
                    h[j] += d

    # Phase 7: Final SLSQP polish
    if _el() < budget_s - 2:
        cons = [{'type': 'eq', 'fun': _sum_con, 'args': (n,)}]
        try:
            res = minimize(_sm_obj, best_h.copy(), args=(5000000, n, 0.0),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-18})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
