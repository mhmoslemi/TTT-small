```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def repair_binary(h, n, target):
    h = np.round(h).astype(float)
    h = np.clip(h, 0, 1)
    s = h.sum()
    diff = target - s
    if diff > 0:
        zeros = np.where(h == 0)[0]
        if len(zeros) >= int(diff):
            picks = np.random.choice(zeros, int(diff), replace=False)
            h[picks] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        if len(ones) >= int(-diff):
            picks = np.random.choice(ones, int(-diff), replace=False)
            h[picks] = 0.0
    # Final fix
    while h.sum() < target - 0.01:
        z = np.where(h == 0)[0]
        if len(z) == 0: break
        h[np.random.choice(z)] = 1.0
    while h.sum() > target + 0.01:
        o = np.where(h == 1)[0]
        if len(o) == 0: break
        h[np.random.choice(o)] = 0.0
    return h

def sa_binary(rng, n, T0, T1, cool, deadline, c5_thresh):
    dx = 2.0 / n
    target = n // 2
    h = (rng.rand(n) < 0.5).astype(float)
    h = repair_binary(h, n, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = T0
    one_pos = list(np.where(h == 1)[0])
    zero_pos = list(np.where(h == 0)[0])
    while time.time() < deadline and T > T1:
        # Swap a 1 and a 0
        i = one_pos[rng.randint(len(one_pos))]
        j = zero_pos[rng.randint(len(zero_pos))]
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best_c5 or rng.random() < np.exp(-(c5 - best_c5) / T):
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if c5 < c5_thresh:
                return best_h, best_c5
            one_pos.remove(i); one_pos.append(j)
            zero_pos.remove(j); zero_pos.append(i)
        else:
            h[i], h[j] = 1.0, 0.0
        T *= cool
    return best_h, best_c5
```

The constraints are bilinear (h[i] * h[i+k]), so this is a QCQP (quadratically constrained QP), not an LP or SOCP.

But we can linearize: for a fixed shift m, the constraint is:
sum_i h[i] - sum_i h[i]*h[i+m] ≤ t/dx

The term sum_i h[i]*h[i+m] is the problem. We can't linearize it directly.

However, we can use the fact that for 0≤h≤1:
h[i]*h[i+m] ≥ h[i] + h[i+m] - 1 (this is a valid lower bound)
so sum_i h[i]*h[i+m] ≥ sum_i (h[i] + h[i+m] - 1) 
And:
h[i]*h[i+m] ≤ min(h[i], h[i+m]) ≤ (h[i]+h[i+m])/2

Using the upper bound: sum_i h[i]*h[i+m] ≤ (1/2) sum_i (h[i] + h[i+m])
So: sum_i h[i]*(1-h[i+m]) ≥ sum_i h[i] - (1/2) sum_i (h[i] + h[i+m])
= (1/2) sum_i h[i] - (1/2) sum_i h[i+m]
= (1/2) * (sum_i h[i] - sum_i h[i+m])

This gives a lower bound on the overlap, not useful for an upper bound.

For an upper bound on the overlap:
sum_i h[i]*(1-h[i+m]) ≤ sum_i h[i] - sum_i h[i]*h[i+m]
≤ sum_i h[i] - (max over valid i of... )

```python
def repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

    if s < 1e-12:
        return np.full(len(h), target / len(h))
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h
```
