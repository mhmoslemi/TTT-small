    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

                h[idx] -= sub
                diff += sub

    return h

    kk = np.arange(-(n - 1), n)

    idx1 = p - kk

    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

    """Single-pass coordinate pair sweep in fixed order."""
    improved = False
    # Use 0 to n-1, and for each i, sweep j = i+1 to n-1

        if time.time() > time_limit:
            break
        hi = h[i]

            if time.time() > time_limit:
                break
            hj = h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = gd_i - _g_vec(h, j, n)
            # Quick check: does any shift help?
            cur_max = np.max(corr)
            # Evaluate at a few points

            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-13:
                continue
            # Bisection
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            hi = h[i]
            gd_i = _g_vec(h, i, n)  # update since h[i] changed
            improved = True
    return h, corr, improved

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _g_all(h, n):
    """Precompute all gradient vectors. Returns (2n-1, n) array."""
    kk = np.arange(-(n - 1), n)
    G = np.zeros((2 * n - 1, n))
    for p in range(n):
        idx1 = p - kk
        m1 = (idx1 >= 0) & (idx1 < n)
        G[m1, p] += (1.0 - h[idx1[m1]])
        idx2 = p + kk
        m2 = (idx2 >= 0) & (idx2 < n)
        G[m2, p] -= h[idx2[m2]]
    return G

def _sweep(h, n, corr, G, t_limit, n_coarse=20, n_bisect=30, order=None):
    """Coordinate pair sweep using precomputed gradients (updated incrementally)."""
    improved = False
    if order is None:
        order = np.arange(n)
    for ai, i in enumerate(order):
        if time.time() > t_limit:
            break
        hi = h[i]
        for j in order[ai + 1:]:
            if time.time() > t_limit:
                break
            hj = h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = G[:, i] - G[:, j]
            cur_max = np.max(corr)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-14:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hib = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hib)
                if np.max(corr + mid * gd) <= np.max(corr + hib * gd):
                    hib = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hib)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            # Update G for changed coordinates
            for c in (i, j):
                G[:, c] = _g_vec(h, c, n)
            improved = True
    return h, corr, improved

def _slsqp(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _fourier_init(n, seed=0):
    rng = np.random.RandomState(seed)
    x = np.arange(n) * 2.0 / n
    h = np.full(n, 0.5)
    for m in range(1, 26):
        a = rng.uniform(-1, 1) / (m ** 0.7)
        b = rng.uniform(-1, 1) / (m ** 0.7)
        h += 0.3 * (a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x))
    return _repair(h, n)

def _block_init(n, seed=0):
    rng = np.random.RandomState(seed)
    nb = max(2, n // 8)
    bs = max(1, n // nb)
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i // bs) % 2 == 0 else 0.0
    h += rng.uniform(-0.05, 0.05, n)
    return _repair(h, n)

def _try_candidate(h_init, n, t_limit):
    """Run a quick optimization on a candidate and return (h, c5)."""
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h = _repair(h_init.copy(), n)
    c5 = _c5(h, dx)
    # SLSQP
    for alpha in [100, 400]:
        if time.time() > t_limit:
            break
        hc = _slsqp(h, n, target, bounds, alpha, 200)
        c5_t = _c5(hc, dx)
        if c5_t < c5:
            h, c5 = hc, c5_t
    # One sweep
    if time.time() < t_limit:
        corr = _corr(h)
        G = _g_all(h, n)
        h, corr, _ = _sweep(h, n, corr, G, time.time() + 10, 18, 28)
        c5_t = _c5(h, dx)
        if c5_t < c5:
            c5 = c5_t
    return h, c5
