        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    return float(_corr(h).max() * 2.0 / len(h))

def _sf(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sj(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            g += w[k] * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += w[k] * (1.0 - h[:n - L])
            g[:n - L] -= w[k] * h[L:]
        else:
            M = -L
            g[:n - M] += w[k] * (1.0 - h[M:])
            g[M:] -= w[k] * h[:n - M]
    return g

def _slsqp(h, n, alpha, iters, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sf, h, jac=_sj, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_grad(h):
    c = _corr(h)
    mb = c.max()
    active = np.where(c >= mb - 1e-6)[0]
    n = len(h)
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += 1.0 - h[:n - L]
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += 1.0 - h[M:]
            g[M:] -= h[:n - M]
    return g / max(len(active), 1)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP with aggressive alpha schedule
    for alpha in [500, 5000, 50000, 500000, 5000000, 50000000]:
        if _el() > budget_s * 0.35:
            break
        h = _slsqp(best_h, n, alpha, 4000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Active-set projected gradient descent
    if _el() < budget_s * 0.55:
        h = best_h.copy()
        c5c = _eval(h)
        step = 0.05
        stall = 0
        while _el() < budget_s * 0.55 and stall < 200:
            g = _active_grad(h)
            g -= g.mean()  # project off sum constraint
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            g = g / gn * step
            improved = False
            for f in [1.0, 0.5, 0.25, 0.1, 0.05]:
                h_new = _repair(h - f * g * gn, n)
                c5_new = _eval(h_new)
                if c5_new < c5c - 1e-18:
                    h = h_new
                    c5c = c5_new
                    improved = True
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_h = h.copy()
                    break
            if not improved:
                step *= 0.7
                stall += 1
            else:
                step = min(step * 1.1, 0.1)
                stall = 0
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h.copy()

    # Phase 3: Binary bounce + re-optimize
    if _el() < budget_s * 0.7:
        # Project to binary
        order = np.argsort(best_h)
        h_bin = np.zeros(n)
        h_bin[order[n // 2:]] = 1.0
        c5_bin = _eval(h_bin)
        # Slightly perturb binary and optimize
        for pert_scale in [0.05, 0.02, 0.01]:
            if _el() > budget_s * 0.7:
                break
            h_p = _repair(h_bin + pert_scale * rng.randn(n), n)
            h_o = _slsqp(h_p, n, 200000, 2000, 1e-19)
            c5 = _eval(h_o)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_o.copy()

    # Phase 4: Multi-start with structured patterns
    if _el() < budget_s * 0.85:
        for r in range(15):
            if _el() > budget_s * 0.85:
                break
            rtype = r % 4
            if rtype == 0:
                pert = 0.005 * rng.randn(n)
            elif rtype == 1:
                freq = rng.randint(2, 10)
                pert = 0.015 * np.sin(2 * np.pi * freq * np.arange(n) / n)
            elif rtype == 2:
                pert = np.zeros(n)
                bs = rng.randint(7, 25)
                st = rng.randint(0, n - bs)
                pert[st:st + bs] = rng.choice([-1, 1]) * 0.008
            else:
                pert = 0.003 * rng.randn(n)
                kernel = np.array([1., 2., 3., 2., 1.]); kernel /= kernel.sum()
                pert = np.convolve(pert, kernel, mode='same')
            hp = _repair(best_h + pert, n)
            h_o = _slsqp(hp, n, 300000, 1500, 1e-19)
            c5 = _eval(h_o)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_o.copy()

    # Phase 5: Ultra-fine pair coordinate descent
    if _el() < budget_s * 0.95:
        h = best_h.copy()
        c5c = _eval(h)
        deltas = np.array([5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
                           -5e-4, -2e-4, -1e-4, -5e-5, -2e-5, -1e-5])
        for it in range(200000):
            if _el() > budget_s * 0.95:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                ni, nj = h[i] + d, h[j] - d
                if -1e-12 <= ni <= 1 + 1e-12 and -1e-12 <= nj <= 1 + 1e-12:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-17:
                        c5c = c5n
                        if c5n < best_c5:
                            best_c5 = c5n
                            best_h = h.copy()
                        break
                    h[i], h[j] = h[i] - d, h[j] + d
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h.copy()

    # Phase 6: Final SLSQP polish
    if _el() < budget_s - 20:
        for alpha in [100000000, 500000000]:
            if _el() > budget_s - 10:
                break
            h = _slsqp(best_h, n, alpha, 5000, 1e-20)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 7: Last micro-moves
    if _el() < budget_s - 3:
        h = best_h.copy()
        c5c = _eval(h)
        for it in range(50000):
            if _el() > budget_s - 3:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.choice([1., -1.]) * rng.choice([1e-5, 2e-5, 5e-5, 1e-6])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
