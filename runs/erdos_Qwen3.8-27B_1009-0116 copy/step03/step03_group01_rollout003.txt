from scipy.optimize import linprog

def _linearized_step(h, n, delta=0.05):
    """Solve LP for one active-set step."""
    c = _corr(h)
    m = c.max()
    active = np.where(c >= m - 1e-8)[0]
    
    # Build Jacobian of c_k at h
    J = np.zeros((2*n-1, n))
    for k in active:
        l = k - (n-1)
        if l >= 0:
            J[k, :n-l] += 1.0 - h[l:]
            J[k, l:] += -h[:n-l]
        else:
            al = -l
            J[k, al:] += 1.0 - h[:n-al]
            J[k, :n-al] += -h[al:]
    
    # LP: min t s.t. J_active @ delta_h <= t - c_active(h)
    #     sum(delta_h) = 0
    #     -delta <= delta_h + (h - 0.5) - ... 
    
    # Variables: [delta_h (n), t (1)]
    # Minimize: t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: J_active @ delta_h - t <= c(h_active) - m  (i.e., c_lin <= t)
    # => J_active @ delta_h - t <= 0 (since we want c_k + J_k·dh <= t, i.e., J_k·dh <= t - c_k)
    # Since c_k = m for active: J_k·dh <= t - m
    A_ub = np.hstack([J[active], -np.eye(len(active))])
    b_ub = np.zeros(len(active))  # t >= m (but t is free to be < m)
    
    # Actually: J_k·dh <= t - c_k(h) = t - m for active k
    # So: J_k·dh - t <= -m
    A_ub = np.hstack([J[active], -np.ones((len(active), 1))])
    b_ub = -m + c[active]  # This should be about 0 for active lags
    
    # Equality: sum(delta_h) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: 0 <= h + dh <= 1, |dh| <= delta
    bounds = [(max(-h[i], -delta), min(1-h[i], delta)) for i in range(n)]
    bounds.append((None, None))  # t is free
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return h
    return _repair(h + res.x[:n], n)

    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(len(h))
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        l = k - (len(h) - 1)
        if l >= 0:
            g[:len(h)-l] += w[k] * (1.0 - h[l:])
            g[l:] -= w[k] * h[:len(h)-l]
        else:
            al = -l
            g[al:] += w[k] * (1.0 - h[:len(h)-al])
            g[:len(h)-al] -= w[k] * h[al:]
    return g

def _slsqp(h, n, alpha, iters, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sf, h, jac=_sf_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_pg(h, n, steps, lr0):
    h = h.copy()
    c5 = _eval(h)
    lr = lr0
    stall = 0
    for _ in range(steps):
        c = _corr(h)
        m = c.max()
        active = np.where(c >= m - 1e-10)[0]
        g = _grad_lags(h, active) / len(active) * (2.0 / n)
        g -= g.mean()
        for _ls in range(40):
            h_new = _repair(h - lr * g, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h = h_new
                c5 = c5n
                stall = 0
                lr = min(lr * 1.05, 0.1)
                break
            lr *= 0.5
        else:
            stall += 1
            lr = max(lr * 0.3, 1e-8)
            if stall > 100:
                break
    return h, c5

def _coord_desc(h, n, rng, max_iter, time_limit_fn, base_scale):
    h = h.copy()
    c5c = _eval(h)
    scale = base_scale
    no_impr = 0
    for _ in range(max_iter):
        if time_limit_fn():
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-12 or ni > 1+1e-12 or nj < -1e-12 or nj > 1+1e-12:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
            no_impr = 0
            scale = min(scale * 1.15, 0.02)
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            no_impr += 1
            if no_impr > 800:
                scale *= 0.6
                no_impr = 0
                if scale < 1e-7:
                    break
    return h, c5c

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    xs = np.linspace(0, 1, n_src)
    xt = np.linspace(0, 1, n_dst)
    return _repair(np.interp(xt, xs, h_src), n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    n_p = len(initial_h_values)
    h_p = _repair(np.array(initial_h_values, dtype=float).copy(), n_p)
    best_h = h_p.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP on parent with increasing alpha
    h = h_p.copy()
    for alpha in [100, 500, 2000, 10000, 50000, 200000]:
        if _el() > budget_s - 300:
            break
        h = _slsqp(h, n_p, alpha, 3000, 1e-19)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Active-set projected gradient
    if _el() < budget_s - 200:
        h2, c5 = _active_pg(best_h, n_p, 2000, 0.02)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()

    # Phase 3: Coordinate descent
    if _el() < budget_s - 150:
        h3, c5 = _coord_desc(best_h, n_p, rng, 80000, lambda: _el() > budget_s - 150, 0.005)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h3.copy()

    # Phase 4: Try other n values
    n_trials = [n for n in [n_p-10, n_p+10, n_p+20, n_p-20, 50, 60, 80, 100, 140, 200]
                if 40 <= n <= 400 and n != n_p]
    for n in n_trials:
        if _el() > budget_s - 120:
            break
        h = _interp(best_h, n)
        for alpha in [500, 10000, 100000]:
            if _el() > budget_s - 120:
                break
            h = _slsqp(h, n, alpha, 2000, 1e-18)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        h2, c5 = _active_pg(h, n, 500, 0.01)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h2.copy()

    # Phase 5: Multi-start on best n
    n_b = len(best_h)
    if _el() < budget_s - 100:
        for _r in range(15):
            if _el() > budget_s - 100:
                break
            sc = 0.002 + 0.02 * rng.rand()
            hp = _repair(best_h + sc * rng.randn(n_b), n_b)
            h = _slsqp(hp, n_b, 50000, 1500, 1e-18)
            h2, c5 = _active_pg(h, n_b, 500, 0.01)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h2.copy()
            c5 = _eval(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
            # Try 1-h symmetry
            h_sym = _repair(1.0 - h, n_b)
            h2s, c5s = _active_pg(h_sym, n_b, 300, 0.01)
            if c5s < best_c5 - 1e-12:
                best_c5 = c5s
                best_h = h2s.copy()

    # Phase 6: Final SLSQP polish
    if _el() < budget_s - 30:
        h = _slsqp(best_h, n_b, 1000000, 5000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 7: Active PG final
    if _el() < budget_s - 15:
        h2, c5 = _active_pg(best_h, n_b, 3000, 0.005)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()

    # Phase 8: Ultra-fine coordinate moves
    if _el() < budget_s - 5:
        h = best_h.copy()
        c5c = _eval(h)
        for _it in range(100000):
            if _el() > budget_s - 5:
                break
            i, j = rng.randint(n_b), rng.randint(n_b)
            if i == j:
                continue
            d = rng.choice([1, -1]) * rng.choice([1, 2, 3, 5]) * 1e-5
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    # Phase 9: One more SLSQP at the end
    if _el() < budget_s - 3:
        h = _slsqp(best_h, n_b, 500000, 2000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = _repair(best_h, n_b)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n_b)
