        if c5x < best_c5:
            best_c5, best_h = c5x, hx

    # Phase 3: Binary local search (exploit binary structure)
    if ok(150):
        hb, cb = _bin_local(best_h, n, dx, elapsed, budget_s - 150)
        if cb < best_c5:
            best_c5, best_h = cb, hb
        # Also try from 1-best
        hb2, cb2 = _bin_local(1.0 - best_h, n, dx, elapsed, budget_s - 150)
        if cb2 < best_c5:
            best_c5, best_h = cb2, hb2

    # Phase 4: PG on symmetric constraint
    if ok(150):
        hs = best_h.copy()
        for i in range(half):
            hs[i + half] = 1.0 - hs[i]
        hs = _repair(hs, n)
        hb, cb = _pg_solve(hs, n, dx, [5000, 20000], 300, 0.004, True)
        if cb < best_c5:
            best_c5, best_h = cb, hb

    # Phase 5: SLSQP again
    if ok(100):
        hx = _slsqp_polish(best_h, n, dx, 80000, 2000)
        c5x = _c5_val(hx, dx)
        if c5x < best_c5:
            best_c5, best_h = c5x, hx

    # Phase 6: Continuous local search (fine)
    if ok(40):
        hb, cb = _cont_local(best_h, n, dx, elapsed, budget_s - 40, n_pass=25)
        if cb < best_c5:
            best_c5, best_h = cb, hb

    # Phase 7: Ultra-fine random perturbations
    if ok(15):
        h = best_h.copy()
        c5_cur = best_c5
        while ok(5):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.choice([2e-5, 5e-5, -2e-5, -5e-5, 1e-5, -1e-5])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5_val(h, dx)
                if c < c5_cur - 1e-16:
                    c5_cur = c
                else:
                    h[i], h[j] = ni - d, nj + d
        if c5_cur < best_c5:
            best_c5, best_h = c5_cur, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
