import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                s2 = min(-diff, h[idx])
                h[idx] -= s2
                diff += s2
            if abs(diff) < 1e-15:
                break
    return h

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _project(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(90):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _ls_exp_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    total = e.sum()
    obj = m + np.log(total) / alpha
    w = e / total
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        i0 = max(0, -lag)
        i1 = min(n, n - lag)
        for i in range(i0, i1):
            j = i + lag
            g[j] += wk * (1.0 - h[i])
            g[i] -= wk * h[j]
    return obj, g * dx

def _topk_obj_grad(h, n, k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(corr)[::-1][:k]
    obj = corr[idx].sum() / k
    g = np.zeros(n)
    for mk in idx:
        lag = mk - (n - 1)
        i0 = max(0, -lag)
        i1 = min(n, n - lag)
        for i in range(i0, i1):
            j = i + lag
            g[j] += (1.0 - h[i])
            g[i] -= h[j]
    return obj, g * dx / k

def _nesterov(h_init, n, alpha, n_iter, t_end, use_topk=False, k_top=3):
    h = h_init.copy()
    v = np.zeros(n)
    beta = 0.9
    lr_base = 1.0 / max(alpha, 10) if not use_topk else 0.3
    momentum = 0.9
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if use_topk:
            obj, grad = _topk_obj_grad(h, n, k_top)
            lr = lr_base / (1.0 + 0.0001 * it)
        else:
            obj, grad = _ls_exp_obj_grad(h, n, alpha)
            lr = lr_base / (1.0 + 0.00005 * it)
        v = momentum * v - lr * grad
        y = h + v
        h = _project(y, n)
    return h

            for d in deltas:
                ni, nj = hi + d, hj - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5_cur - 1e-16:
                        c5_cur = c
                        hi, hj = ni, nj
                        imp = True
                    else:
                        h[i], h[j] = hi, hj
        if not imp:
            break
    return h, c5_cur

def _stem_search(h, n, t_end, rng):
    c5_cur = _c5(h, n)
    tiny = [0.0003, 0.0008, 0.002, -0.0003, -0.0008, -0.002]
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = _c5(h, n)
            if c < c5_cur:
                c5_cur = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    t_end = t0 + budget_s - 20

    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: Multi-scale Nesterov (coarse to fine)
    scales = [(50, 15000, 1.0), (200, 12000, 1.0), (800, 12000, 1.0),
              (3000, 15000, 1.0), (10000, 15000, 1.0), (50000, 10000, 1.0)]
    for alpha, n_it, lm in scales:
        if time.time() > t_end - 200:
            break
        h_amt = _nesterov_opt(best_h, n, alpha, n_it, t_end - 200, lr_mult=lm)
        c = _c5(h_amt, n)
        if c < best_c5:
            best_c5 = c
            best_h = h_amt.copy()

    # Phase 2: Top-k focused optimization
    for k_val in [2, 3, 4, 6]:
        if time.time() > t_end - 150:
            break
        h_amt = _nesterov_opt(best_h, n, 0, 8000, t_end - 150, use_topk=True, k_top=k_val)
        c = _c5(h_amt, n)
        if c < best_c5:
            best_c5 = c
            best_h = h_amt.copy()

    # Phase 3: Multi-restart with perturbations
    for r in range(20):
        if time.time() > t_end - 100:
            break
        scale = 0.02 + 0.01 * (r % 4)
        h_try = _repair(best_h + scale * rng.randn(n), n)
        for alpha, n_it in [(5000, 4000), (30000, 4000)]:
            if time.time() > t_end - 100:
                break
            h_amt = _nesterov_opt(h_try, n, alpha, n_it, t_end - 100)
            c = _c5(h_amt, n)
            if c < best_c5:
                best_c5 = c
                best_h = h_amt.copy()
                h_try = h_amt.copy()
        # Also try top-k from perturbed start
        if time.time() < t_end - 100:
            h_amt = _nesterov_opt(h_try, n, 0, 3000, t_end - 100, use_topk=True, k_top=2)
            c = _c5(h_amt, n)
            if c < best_c5:
                best_c5 = c
                best_h = h_amt.copy()

    # Phase 4: Pairwise local search (random pairs)
    if time.time() < t_end - 30:
        h_loc = best_h.copy()
        h_loc, c5_loc = _pairwise_search(h_loc, n, t_end - 30, rng)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc.copy()

    # Phase 5: Very fine random coordinate search
    if time.time() < t_end:
        h_fine = best_h.copy()
        h_fine, c5_fine = _stem_search(h_fine, n, t_end, rng)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
