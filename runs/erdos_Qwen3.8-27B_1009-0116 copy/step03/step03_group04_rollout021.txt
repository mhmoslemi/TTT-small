import numpy as np
from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _soft_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = np.max(c)
    w = np.exp(np.clip(alpha * (c - m), -50.0, 50.0))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-12:
            continue
        jo = k - (n - 1)
        i_lo = max(0, -jo)
        i_hi = min(n - 1, n - 1 - jo)
        ia = np.arange(i_lo, i_hi + 1)
        ja = ia + jo
        g[ia] += wk * (1.0 - h[ja])
        g[ja] -= wk * h[ia]
    return g

def _project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-12:
            break
    return h

def _pgd(h, n, dx, t0, deadline, max_iter, alpha0, alpha1, mu0):
    h = h.copy()
    c5 = _c5(h, dx)
    best_c5 = c5
    best_h = h.copy()

    v = np.zeros(n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max(1, max_iter // 4))
        alpha = alpha0 + (alpha1 - alpha0) * frac
        mu = mu0 * (1.0 - 0.3 * frac)
        g = _soft_grad(h, alpha)
        g -= g.mean()
        v = mu * v + g
        step = 0.02 * (0.95 ** (it // 200))
        for _bt in range(10):
            h_new = _project_sum(h - step * v, target)
            c5_new = _c5(h_new, dx)
            if c5_new <= c5 + 1e-16:
                h = h_new
                c5 = c5_new
                break
            step *= 0.5
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _coord_descent(h, n, dx, t0, deadline, n_sweps):
    h = h.copy()
    c5 = _c5(h, dx)
    target = n / 2.0
    deltas = np.array([5e-3, 2e-3, 8e-4, 3e-4, -5e-3, -2e-3, -8e-4, -3e-4])

            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(n):
                if i == j:
                    continue
                for d in deltas:
                    ni, nj = h[i] + d, h[j] - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                    h[i], h[j] = ni, nj
                    c5_new = _c5(h, dx)
                    if c5_new < c5 - 1e-15:
                        c5 = c5_new
                    else:
                        h[i], h[j] = ni - d, nj + d
