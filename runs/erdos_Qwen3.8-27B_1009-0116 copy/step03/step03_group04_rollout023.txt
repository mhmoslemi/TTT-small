import numpy as np

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-10:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if d > 1e-12:
            o = np.argsort(h)
            for k in o:
                a = min(d, 1.0 - h[k])
                h[k] += a; d -= a
                if d < 1e-12: break
        elif d < -1e-12:
            d = -d
            o = np.argsort(h)[::-1]
            for k in o:
                a = min(d, h[k])
                h[k] -= a; d -= a
                if d < 1e-12: break
    return h

def _smooth_max(c, a):
    m = float(np.max(c))
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / a

def _grad(h, lags, n):
    g = np.zeros(n)
    nm1 = n - 1
    for m in lags:
        for i in range(n):
            j = i + m - nm1
            if 0 <= j < n:
                g[i] += 1.0 - h[j]
                g[j] -= h[i]
    return g / max(len(lags), 1)

    return _smooth_max(_corr(x), a)

    return x.sum() - t

            rx = _slsqp(bh, bnd, tgt, 200000, 2000)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < bc5:
                bc5, bh = c5, hc
        except Exception:
            pass

    # P8: One more informed PS pass if time remains
    if time.time() - t0 < budget_s - 3:
        hp, c5p = _ps(bh, n, dx, 20, t0 + budget_s - 2, True)
        if c5p < bc5:
            bc5, bh = c5p, hp.copy()

    bh = _repair(bh, n)
    bc5 = _c5(bh, dx)
    return (bh, bc5, n)
