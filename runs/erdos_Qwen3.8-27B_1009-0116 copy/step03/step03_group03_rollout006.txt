def find_best_pair_move(h, n, G, c_arr, M):
    # G is (2n-1) x n gradient matrix
    # c_arr is correlation array
    # M is current max
    # Find best (i, j, delta) that reduces M
    best_improve = 0
    best_i, best_j, best_d = -1, -1, 0
    
    active = np.where(c_arr >= M - 0.01)[0]  # near-active lags
    
    for i in range(n):
        if h[i] >= 1.0:
            continue
        gi = G[:, i]
        for j in range(n):
            if j == i or h[j] <= 0.0:
                continue
            gj = G[:, j]
            a = gi - gj  # change per unit δ
            # For δ > 0: need a[m] < 0 for active m
            # Max reduction: δ_max_pos = min(1-h[i], h[j])
            # New max ≈ max(c_arr + a * δ)
            # For small δ: new max ≈ M + max(a[active]) * δ (if active dominates)
            # We want max(a[active]) < 0
            a_max_active = np.max(a[active])
            if a_max_active < 0:
                δ = min(1 - h[i], h[j])
                # Check all lags
                new_vals = c_arr + a * δ
                new_max = np.max(new_vals)
                improve = M - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i, best_j, best_d = i, j, δ
            # For δ < 0 (transfer from i to j):
            δ_neg_max = min(h[i], 1 - h[j])
            if δ_neg_max > 0:
                a_max_active_neg = np.max(a[active])  # same a, but δ is negative
                # new_vals = c_arr + a * (-δ_neg)
                # = c_arr - a * δ_neg
                # Need max(c_arr - a * δ_neg) < M
                # = max(c_arr + (-a) * δ_neg) < M
                # Need max((-a)[active]) < 0, i.e., min(a[active]) > 0

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _grad_matrix(h, n):
    """Compute (2n-1) x n matrix where G[m, i] = ∂c[m]/∂h[i]"""
    G = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            G[m, k:] += (1.0 - h[:n - k])
            G[m, :n - k] -= h[k:]
        else:
            G[m, :n + k] += (1.0 - h[-k:])
            G[m, -k:] -= h[:n + k]
    return G

    obj = m + np.log(np.sum(np.exp(s))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            G[m, k:] += (1.0 - h[:n - k])
            G[m, :n - k] -= h[k:]
        else:
            G[m, :n + k] += (1.0 - h[-k:])
            G[m, -k:] -= h[:n + k]
    return G

def _smooth_obj(x, alpha, target):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    obj = m + np.log(np.sum(np.exp(s))) / alpha
    obj += 1000.0 * (np.sum(x) - target) ** 2
    return obj

def _slsqp_step(h, n, target, alpha, maxit, tleft_fn):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _smooth_obj, h, args=(alpha, target),
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t}],
            options={'maxiter': maxit, 'ftol': 1e-18, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return None

def _pair_sweep(h, n, c_arr, G, M, tol_frac=0.005, max_passes=15000):
    best_h, best_M = h.copy(), M
    for _ in range(max_passes):
        if tleft() < 80:
            break
        M = c_arr.max()
        active_mask = c_arr >= M - tol_frac
        best_imp, bi, bj, bd = 1e-14, -1, -1, 0.0
        for i in range(n):
            if h[i] <= 1e-12 and h[n-1-i] <= 1e-12:
                pass
            hi, hj_max = h[i], 1.0
            for j in range(n):
                if j == i:
                    continue
                hj = h[j]
                a = G[:, i] - G[:, j]
                d_pos = min(1.0 - h[i], hj)
                if d_pos > 1e-10:
                    amax = np.max(a[active_mask])
                    if amax < -1e-10:
                        nv = c_arr + a * d_pos
                        imp = M - np.max(nv)
                        if imp > best_imp:
                            best_imp, bi, bj, bd = imp, i, j, d_pos
                d_neg = min(h[i], 1.0 - hj)
                if d_neg > 1e-10:
                    amin = np.min(a[active_mask])
                    if amin > 1e-10:
                        nv = c_arr - a * d_neg
                        imp = M - np.max(nv)
                        if imp > best_imp:

        if bi < 0:
            break
        h[bi] += bd
        h[bj] -= bd
        h = _repair(h, n)
        c_arr = _corr(h)
        G = _grad_matrix(h, n)
        new_M = c_arr.max()
        if new_M < best_M - 1e-13:
            best_M = new_M
            best_h = h.copy()
        M = new_M
    return best_h, best_M
