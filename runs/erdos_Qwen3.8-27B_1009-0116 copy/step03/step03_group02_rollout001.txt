def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _pair_sweep(h, n, corr, time_limit, n_coarse, n_bisect):
    """Coordinate-pair sweep with bisection."""
    improved = False
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for j in order[ai+1:]:
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            # Check if moving helps
            # At d=0, max is current. At d=d_min and d=d_max, check.
            cur_max = np.max(corr)
            # Quick check at endpoints and midpoint
            test_ds = [d_min, 0.0, 0.5*(d_min+d_max), d_max]
            best_val = cur_max
            best_d = 0.0
            for d in test_ds:
                if d == 0.0:
                    continue
                v = np.max(corr + d * gd)
                if v < best_val:
                    best_val = v
                    best_d = d
            if best_d == 0.0:
                continue
            # Bisection around best_d
            lo = max(d_min, best_d - 0.1*(d_max-d_min))
            hi_ = min(d_max, best_d + 0.1*(d_max-d_min))
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_)
                if np.max(corr + mid * gd) <= np.max(corr + hi_ * gd):
                    hi_ = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_)
            # Only apply if it improves
            if np.max(corr + d_best * gd) < cur_max - 1e-14:
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

def _subgrad_step(h, n, corr, step_size):
    """Take one subgradient step to reduce max correlation."""
    cmax = np.max(corr)
    active = np.where(corr >= cmax - 1e-8)[0]
    
    # Average gradient over active lags
    g_avg = np.zeros(n)
    for k_idx in active:
        lag = k_idx - (n - 1)
        for i in range(n):
            # ∂c_lag/∂h[i]
            # From our formula: (1-h[i-lag])*[i-lag valid] - h[i+lag]*[i+lag valid]
            il = i - lag
            if 0 <= il < n:
                g_avg[i] += (1.0 - h[il])
            ipl = i + lag
            if 0 <= ipl < n:
                g_avg[i] -= h[ipl]
    g_avg /= len(active)
    
    # Projected gradient step
    h_new = h - step_size * g_avg
    return _repair(h_new, n)
