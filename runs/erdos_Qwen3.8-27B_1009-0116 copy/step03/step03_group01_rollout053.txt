                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _pair_d(h, i, j):
    """Golden-section: optimal d transferring mass from j to i."""
    d_lo = max(-h[i] + 1e-15, h[j] - 1.0 - 1e-15)
    d_hi = min(1.0 - h[i] - 1e-15, h[j] - 1e-15)
    if d_lo >= d_hi:
        return 0.0
    gr = 0.6180339887498949
    a, b = d_lo, d_hi
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)

    def _f(d):
        h[i] += d; h[j] -= d
        m = np.correlate(h, 1.0 - h, mode='full').max()
        h[i] -= d; h[j] += d
        return m

    fc = _f(c_); fd = _f(d_)
    for _ in range(50):
        if b - a < 1e-14:
            break
        if fc < fd:
            b = d_; d_ = c_; fd = fc
            c_ = b - gr * (b - a); fc = _f(c_)
        else:
            a = c_; c_ = d_; fc = fd
            d_ = a + gr * (b - a); fd = _f(d_)
    return (a + b) / 2.0

def _pair_pass_ordered(h, n):
    improved = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = _pair_d(h, i, j)
            if abs(d) > 1e-13:
                h[i] += d; h[j] -= d
                h[i] = min(max(h[i], 0.0), 1.0)
                h[j] = min(max(h[j], 0.0), 1.0)
                improved += 1
    return improved

def _pair_pass_random(h, n, rng):
    improved = 0
    idx = np.arange(n)
    rng.shuffle(idx)
    for a in range(n):
        for b in range(a + 1, n):
            i, j = int(idx[a]), int(idx[b])
            d = _pair_d(h, i, j)
            if abs(d) > 1e-13:
                h[i] += d; h[j] -= d
                h[i] = min(max(h[i], 0.0), 1.0)
                h[j] = min(max(h[j], 0.0), 1.0)
                improved += 1
    return improved

def _slsqp_polish(h, n, budget_remaining):
    alpha = 200000

    def _sf(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha

    try:
        res = minimize(_sf, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': 1500, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Ordered pair optimization (fast convergence)
    for _ in range(300):
        if _el() > budget_s * 0.12:
            break
        prev = _eval(h)
        _pair_pass_ordered(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
        if prev - c5 < 1e-14:
            break

    # Phase 2: Random-order pair optimization with restarts
    stall = 0
    for _ in range(5000):
        if _el() > budget_s - 120:
            break
        prev = _eval(h)
        _pair_pass_random(h, n, rng)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy(); stall = 0
        elif prev - c5 < 1e-13:
            stall += 1
        else:
            stall = 0
        if stall >= 4:
            h = best_h + 0.02 * rng.randn(n)
            h = _repair(h, n)
            stall = 0

    # Phase 3: Try a few diverse random starts
    for _trial in range(5):
        if _el() > budget_s - 80:
            break
        h2 = np.clip(0.5 + 0.3 * rng.randn(n), 0, 1)
        h2 = _repair(h2, n)
        for _ in range(150):
            if _el() > budget_s - 80:
                break
            _pair_pass_ordered(h2, n)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2.copy()

    # Phase 4: SLSQP polish from best
    if _el() < budget_s - 40:
        h = _slsqp_polish(best_h.copy(), n, budget_s - _el())
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()

    # Phase 5: Final pair optimization sweep
    h = best_h.copy()
    for _ in range(400):
        if _el() > budget_s - 5:
            break
        prev = _eval(h)
        _pair_pass_ordered(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
        if prev - c5 < 1e-14:
            break

    # Phase 6: One more SLSQP + pair polish
    if _el() < budget_s - 10:
        h = _slsqp_polish(best_h.copy(), n, 10)
        for _ in range(100):
            if _el() > budget_s - 3:
                break
            _pair_pass_ordered(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
