    """Compute new C5 after h[i]+=d, h[j]-=d, given previous correlation c_prev."""
    # For each lag, compute the change
    # c_{n-1+k} = sum_m h[m+k]*(1-h[m])
    # After swap, only terms involving h[i] or h[j] change
    c_new = c_prev.copy()
    for k in range(-(n-1), n):
        # Valid range for m: 0 <= m <= n-1, 0 <= m+k <= n-1
        m_min = max(0, -k)
        m_max = min(n-1, n-1-k)
        if m_min > m_max:
            continue
        delta = 0
        for m in range(m_min, m_max+1):
            mk = m + k
            # Old term: h[mk]*(1-h[m])
            # New term: h'[mk]*(1-h'[m])
            hi_old = h[i]

                h[idx] += add; diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub; diff += sub
            if abs(diff) < 1e-14: break
    return h

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-10: continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i + lag] += wk * (1.0 - x[i])
                g[i] -= wk * x[i + lag]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    return g * dx

    return minimize(_smooth_obj, h0, args=(n, alpha),
                    jac=_smooth_grad, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-18})

            if 0 <= j + k < n: a += d * h[j + k]

            if 0 <= j - k < n and j - k != i: a -= d * (1.0 - h[j - k])

            if abs(a) > 1e-18:

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i + lag] += wk * (1.0 - x[i])
                g[i] -= wk * x[i + lag]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _cons_fun(x):
    return np.sum(x) - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i + lag] += wk * (1.0 - x[i])
                g[i] -= wk * x[i + lag]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    return g * dx

def _slsqp_opt(h0, n, alpha, maxiter, bounds, cons):
    return minimize(_smooth_obj, h0, args=(n, alpha),
                    jac=_smooth_grad, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-18})

def _local_fast(h, n, t_end, rng, deltas):
    h = h.copy()
    c5_cur = _c5(h, n)
    best_c5 = c5_cur
    best_h = h.copy()
    stale = 0
    while time.time() < t_end and stale < 4:
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for ai in range(n - 1):
            i = int(idx[ai])
            hi = h[i]
            if hi < 1e-14 and hi > 1 - 1e-14:
                h[i] = hi
                continue
            for aj in range(ai + 1, n):
                if time.time() > t_end:
                    break
                j = int(idx[aj])
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    mc = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
                    if mc < c5_cur - 1e-15:
                        c5_cur = mc
                        hi, hj = ni, nj
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                    else:
                        h[i], h[j] = hi, hj
            h[i] = hi
        if not improved:
            stale += 1
        else:
            stale = 0
    return best_h, best_c5

def _local_best(h, n, t_end, rng, deltas):

    best_c5 = c5_cur
    best_h = h.copy()

        best_imp = 0.0
        best_i = best_j = -1
        best_d = 0.0
        for i in range(n):
            hi = h[i]
            for j in range(i + 1, n):
                if time.time() > t_end:
                    break
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                    h[i], h[j] = ni, nj
                    mc = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
                    h[i], h[j] = hi, hj
                    imp = c5_cur - mc
                    if imp > best_imp + 1e-15:
                        best_imp = imp
                        best_i, best_j, best_d = i, j, d
        if best_i < 0:
            stale += 1
            continue
        h[best_i] += best_d
        h[best_j] -= best_d
        c5_cur = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
        if c5_cur < best_c5:

        stale = 0
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return best_h, best_c5

def _fourier_perturb(h, n, rng, amplitude):

    freq = rng.randint(1, max(2, n // 2))
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    h2 += amplitude * np.sin(np.pi * freq * x / n + phase)
