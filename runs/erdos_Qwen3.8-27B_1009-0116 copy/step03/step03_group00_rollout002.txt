            for j in range(i + 1, n):
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    cv = _c5(h, n)
                    if cv < c5c - 1e-16:
                        c5c = cv
                        hi, hj = ni, nj
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
                    for md in micro:
                        ni2, nj2 = hi + md, hj - md
                        if ni2 < 0 or ni2 > 1 or nj2 < 0 or nj2 > 1:
                            continue
                        h[i], h[j] = ni2, nj2
                        cv2 = _c5(h, n)
                        if cv2 < c5c - 1e-16:
                            c5c = cv2
                            hi, hj = ni2, nj2
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
                h[j] = hj
            h[i] = hi
        if not improved:
            break
    return h, c5c

def _sa_search(h, n, t_end, rng):
    c5c = _c5(h, n)
    T = 0.0002
    minT = 1e-10
    cool = 0.9998
    steps_pool = np.array([0.002, 0.005, 0.01, 0.02, 0.05])
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + rng.randint(1, n)) % n
        d = steps_pool[rng.randint(5)] * (1 if rng.rand() < 0.5 else -1)
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
            continue
        h[i] = max(0.0, min(1.0, ni))
        h[j] = max(0.0, min(1.0, nj))
        cv = _c5(h, n)
        if cv < c5c or (T > 1e-11 and rng.rand() < np.exp(-(cv - c5c) / T)):
            c5c = cv
        else:
            h[i] = hi if 'hi' in dir() else h[i]
            h[i], h[j] = max(0.0, min(1.0, h[i])), max(0.0, min(1.0, h[j]))
            h[i] = h[i] - d + d  # no-op, restore below
        T = max(T * cool, minT)
    return h, c5c

def _slsqp(h_init, n, alpha, maxit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    return minimize(_obj_sm, h_init, args=(n, alpha),
                    jac=lambda x: _grad_sm(x, n, alpha),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': maxit, 'ftol': 1e-17})

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def el():
        return time.time() - t0
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: SLSQP increasing alpha
    for alpha in [1500, 4000, 10000, 25000, 60000]:
        nrep = 5 if alpha <= 4000 else 3
        for r in range(nrep):
            if el() > budget_s - 250:
                break
            x0 = best_h.copy() if r == 0 else _repair(best_h + 0.015 * rng.randn(n), n)
            try:
                res = _slsqp(x0, n, alpha, 300)
                hc = _repair(res.x, n)
                cv = _c5(hc, n)
                if cv < best_c5:
                    best_c5, best_h = cv, hc
            except:
                pass

    # Phase 2: Pair local search (main refinement)
    t_end2 = t0 + budget_s - 200
    best_h, best_c5 = _pair_search(best_h, n, t_end2, rng)

    # Phase 3: More SLSQP from refined solution
    for alpha in [10000, 30000]:
        if el() > budget_s - 100:
            break
        try:
            res = _slsqp(best_h.copy(), n, alpha, 200)
            hc = _repair(res.x, n)
            cv = _c5(hc, n)
            if cv < best_c5:
                best_c5, best_h = cv, hc
        except:
            pass

    # Phase 4: Another pair search round
    t_end4 = t0 + budget_s - 80
    best_h, best_c5 = _pair_search(best_h, n, t_end4, rng)

    # Phase 5: Simulated annealing to escape
    if el() < budget_s - 40:
        h_sa = best_h.copy()
        c5c = best_c5
        T = 0.00015
        while time.time() < t0 + budget_s - 40:
            i = rng.randint(n)
            j = (i + rng.randint(1, n)) % n
            d = np.array([0.002, 0.005, 0.01, 0.03])[rng.randint(4)] * (1 if rng.rand() < 0.5 else -1)
            ni, nj = h_sa[i] + d, h_sa[j] - d
            if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                continue
            old_i, old_j = h_sa[i], h_sa[j]
            h_sa[i] = max(0.0, min(1.0, ni))
            h_sa[j] = max(0.0, min(1.0, nj))
            cv = _c5(h_sa, n)
            if cv < c5c or (T > 1e-11 and rng.rand() < np.exp(-(cv - c5c) / T)):
                c5c = cv
                if cv < best_c5:
                    best_c5, best_h = cv, h_sa.copy()
            else:
                h_sa[i], h_sa[j] = old_i, old_j
            T = max(T * 0.9999, 1e-11)

    # Phase 6: Final ultra-fine polish
    if time.time() < t0 + budget_s - 5:
        h_f = best_h.copy()
        c5f = _c5(h_f, n)
        tiny = np.array([0.0003, 0.0007, 0.0015, 0.004,
                         -0.0003, -0.0007, -0.0015, -0.004])
        for _pp in range(2000):
            if time.time() > t0 + budget_s - 5:
                break
            imp = False
            for i in range(n):
                hi = h_f[i]
                for j in range(i + 1, min(i + 15, n)):
                    hj = h_f[j]
                    for d in tiny:
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h_f[i], h_f[j] = ni, nj
                        cv = _c5(h_f, n)
                        if cv < c5f - 1e-17:
                            c5f = cv
                            hi, hj = ni, nj
                            imp = True
                        else:
                            h_f[i], h_f[j] = hi, hj
                    h_f[j] = hj
                h_f[i] = hi
            if c5f < best_c5:
                best_c5, best_h = c5f, h_f.copy()
            if not imp:
                break

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
