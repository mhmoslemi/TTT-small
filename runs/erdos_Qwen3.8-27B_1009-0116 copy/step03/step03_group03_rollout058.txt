```python
def _coord_opt_fast(h, j, corr, n):
    """Fast 1D minimax using precomputed correlation."""
    a = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k == 0:
            a[m] = 1.0 - 2.0 * h[j]
        elif k > 0:
            if j + k < n:
                a[m] += 1.0 - h[j + k]
            if j >= k:
                a[m] -= h[j - k]
        else:
            mk = -k
            if j >= mk:
                a[m] += 1.0 - h[j - mk]
            if j + mk < n:
                a[m] -= h[j + mk]
    
    b = corr - a * h[j]
    v0 = np.max(b)
    v1 = np.max(a + b)
    p0, p1 = np.argmax(b), np.argmax(a + b)
    best_p = 0.0 if v0 <= v1 else 1.0
    best_v = min(v0, v1)
    if abs(a[p0] - a[p1]) > 1e-14:
        pi = (b[p1] - b[p0]) / (a[p0] - a[p1])
        if 0.0 < pi < 1.0:
            vi = a[p0] * pi + b[p0]
            if vi < best_v:
                best_p, best_v = pi, vi
    return best_p, a
```

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h
