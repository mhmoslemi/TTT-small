def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _smooth_max(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _grad_smooth_max(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()

    else:

                break

def _nesterov(h, n, alpha, iters, base_lr, t_end):
    """Nesterov accelerated projected gradient descent"""
    best = h.copy()
    best_c = _eval(h)
    x = h.copy()
    x_prev = h.copy()
    m = 1.0  # momentum init (Nesterov: (1+m_t)/(2+m_t))

        if time.time() > t_end:
            break
        # Nesterov step
        m = (1.0 + m) / (2.0 + m)
        y = x + m * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)

            z = y - step * g
            z = _project(z, n)

        x = z
        c = _eval(x)
        if c < best_c:
            best_c = c
            best = x.copy()
    
    return best, best_c

            break
        lr = lr0 / (1.0 + 0.005 * it)
        fy = _smooth_max_val(np.correlate(y, 1.0 - y, mode='full'), alpha)
        z = y.copy()
        for _bt in range(15):
            z = y - lr * (lr0 / lr) * g
            z = _project(z, n)
            fz = _smooth_max_val(np.correlate(z, 1.0 - z, mode='full'), alpha)
            if fz < fy - 1e-4 * lr * gn * gn:
                break
            lr *= 0.5
        xp = x
        x = z
        c = _eval(x)
        if c < best_c:
            best_c = c
            best = x.copy()
    return best, best_c

def _coord_opt(h, n, t_end):
    best = h.copy()
    best_c = _eval(h)
    x = h.copy()
    fracs = np.array([0.05, 0.15, 0.3, 0.5, 0.7, 0.9,
                      -0.05, -0.15, -0.3, -0.5, -0.7, -0.9])
    for epoch in range(60):
        if time.time() > t_end:
            break
        imp = 0
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(i + 1, n):
                hi, hj = x[i], x[j]
                pos_d = min(hi, 1.0 - hj)
                neg_d = min(hj, 1.0 - hi)
                if pos_d < 1e-9 and neg_d < 1e-9:
                    continue
                for f in fracs:
                    d = f * max(pos_d, neg_d)
                    if d > 0:
                        if d > pos_d:
                            continue
                    else:
                        if abs(d) > neg_d:
                            continue
                    x[i], x[j] = hi + d, hj - d
                    cn = _eval(x)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        best = x.copy()
                        imp += 1
                    else:
                        x[i], x[j] = hi, hj
        if imp == 0:
            break
    return best, best_c

def _fine_polish(h, n, t_end):
    best = h.copy()
    best_c = _eval(h)
    x = h.copy()
    rng = np.random.RandomState(12345)
    ascale = 0.01
    for it in range(200000):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = ascale * rng.randn()
        if d > 0:
            if h_max_pos(x[i], x[j]) < d:
                continue
        else:
            if h_max_neg(x[i], x[j]) < -d:
                continue
        x[i], x[j] = x[i] + d, x[j] - d
        cn = _eval(x)
        if cn < best_c - 1e-15:
            best_c = cn
            best = x.copy()
            ascale = min(ascale * 1.1, 0.05)
        else:
            x[i], x[j] = x[i] - d, x[j] + d
            ascale = max(ascale * 0.95, 1e-5)
    return best, best_c

def h_max_pos(a, b):
    return min(a, 1.0 - b)

def h_max_neg(a, b):
    return min(b, 1.0 - a)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP with increasing smoothness
    for alpha in [300, 2000, 15000, 80000]:
        if time.time() - t0 > budget_s * 0.22:
            break
        h = _slsqp_opt(best_h, n, alpha, 4000, t0 + budget_s * 0.22)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Nesterov accelerated gradient
    for alpha in [3000, 10000, 30000]:
        if time.time() - t0 > budget_s * 0.45:
            break
        h2, c2 = _nesterov_opt(best_h, n, alpha, 50000, 0.03, t0 + budget_s * 0.45)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 3: Coordinate descent
    if time.time() - t0 < budget_s * 0.65:
        h3, c3 = _coord_opt(best_h, n, t0 + budget_s * 0.65)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()

    # Phase 4: Multi-start with diverse perturbations
    if time.time() - t0 < budget_s * 0.82:
        for r in range(20):
            if time.time() - t0 > budget_s * 0.82:
                break
            sc = 0.003 + 0.02 * rng.rand()
            hp = _repair(best_h + sc * rng.randn(n), n)
            h4 = _slsqp_opt(hp, n, 8000, 600, t0 + budget_s * 0.82)
            c4 = _eval(h4)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()

    # Phase 5: Unfreeze degenerate variables and re-optimize
    if time.time() - t0 < budget_s * 0.88:
        h5 = best_h.copy()
        deg_mask = (h5 < 0.05) | (h5 > 0.95)
        if deg_mask.sum() > 3:
            for idx in np.where(deg_mask)[0]:
                h5[idx] = 0.5
            h5 = _repair(h5, n)
            h5 = _slsqp_opt(h5, n, 20000, 2000, t0 + budget_s * 0.88)
            c5_v = _eval(h5)
            if c5_v < best_c5:
                best_c5 = c5_v
                best_h = h5.copy()

    # Phase 6: Fine polynomial search
    if time.time() - t0 < budget_s * 0.94:
        h6, c6 = _fine_polish(best_h, n, t0 + budget_s * 0.94)
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6.copy()

    # Phase 7: Final SLSQP + coord polish
    if time.time() - t0 < budget_s - 10:
        h7 = _slsqp_opt(best_h, n, 50000, 3000, t0 + budget_s - 10)
        c7 = _eval(h7)
        if c7 < best_c5:
            best_c5 = c7
            best_h = h7.copy()
        if time.time() - t0 < budget_s - 5:
            h7, c7 = _coord_opt(best_h, n, t0 + budget_s - 5)
            if c7 < best_c5:
                best_c5 = c7
                best_h = h7.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
