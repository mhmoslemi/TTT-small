def _smooth_max(c, alpha):

    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _eq_con(x, target):

def _repair(h, n):

    if abs(diff) > 1e-12:

            order = np.argsort(1.0 - h)
        else:

                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 30})

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _precompute_G(h, n):
    G = np.zeros((n, 2 * n - 1))
    for i in range(n):
        m1_lo, m1_hi = i, min(i + n - 1, 2 * n - 2)
        if m1_lo <= m1_hi:
            idx = (i - m1_lo + n - 1) - np.arange(m1_hi - m1_lo + 1)
            G[i, m1_lo:m1_hi + 1] += (1.0 - h[idx])
        m2_lo, m2_hi = max(0, n - 1 - i), min(2 * n - 2 - i, 2 * n - 2)
        if m2_lo <= m2_hi:
            idx2 = (i + m2_lo - (n - 1)) + np.arange(m2_hi - m2_lo + 1)
            G[i, m2_lo:m2_hi + 1] -= h[idx2]
    return G

def _sweep_fast(h, n, corr, G, time_limit, n_coarse=22, n_bisect=28):
    improved = False
    order = list(range(n))
    np.random.shuffle(order)
    for ai, i in enumerate(order):
        for j in order[ai + 1:]:
            if time.time() > time_limit:
                return h, corr, G, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            g = G[i] - G[j]
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * g[None, :], axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= float(np.max(corr)) - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * g) <= np.max(corr + hi_b * g):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] += d_best
            h[j] -= d_best
            corr += d_best * g
            G = _precompute_G(h, n)
            improved = True
    return h, corr, G, improved

def _slsqp(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 30})
        if res.success or res.fun < 1e10:
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _subgrad_step(h, n, lr, n_active=3):
    corr = _corr(h)
    top = np.argsort(corr)[::-1][:n_active]
    w = np.ones(len(top)) / len(top)
    grad = np.zeros(n)
    for wi, m in zip(w, top):
        k = m - (n - 1)
        j1_lo, j1_hi = max(0, k), min(n - 1, n - 1 + k)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi + 1] += wi * (1.0 - h[(j1_lo - k):(j1_hi - k + 1)])
        j2_lo, j2_hi = max(0, -k), min(n - 1, n - 1 - k)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] -= wi * h[(j2_lo + k):(j2_hi + k + 1)]
    h_new = h - lr * grad
    return _repair(h_new, n)

def _try_restart(h_best, n, dx, target, bounds, c5_best, time_budget, rng):
    hc = h_best + 0.03 * rng.standard_normal(n)
    hc = _repair(hc, n)
    hc = _slsqp(hc, n, target, bounds, 400, 200)
    corr = _corr(hc)
    G = _precompute_G(hc, n)
    hc, corr, G, _ = _sweep_fast(hc, n, corr, G, time.time() + min(15, time_budget), 20, 25)
    c5 = float(np.max(corr) * dx)
    if c5 < c5_best - 1e-14:
        return hc, c5
    hc = h_best + 0.06 * rng.standard_normal(n)
    hc = _repair(hc, n)
    hc = _slsqp(hc, n, target, bounds, 600, 250)
    corr = _corr(hc)
    G = _precompute_G(hc, n)
    hc, corr, G, _ = _sweep_fast(hc, n, corr, G, time.time() + min(15, time_budget), 20, 25)
    c5 = float(np.max(corr) * dx)
    if c5 < c5_best - 1e-14:
        return hc, c5
    return None, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-up
    for alpha in [40, 120, 350, 800, 1500]:
        if time.time() - t0 > budget_s * 0.12:
            break

        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = hc.copy()

        if imp:

            else:
                stall += 1
        else:

            Gc = _precompute_G(hc, n)
