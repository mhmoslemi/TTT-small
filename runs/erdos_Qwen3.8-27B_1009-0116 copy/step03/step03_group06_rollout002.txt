```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)

    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

    return h
```

```python
        # If all b_k for active k are positive, increasing t increases max -> decrease t
        # If all b_k for active k are negative, increasing t decreases max -> increase t
        active = np.where(vals >= m - 1e-10)[0]
        if np.all(b[active] > 0):
            hi = mid
        elif np.all(b[active] < 0):
            lo = mid
        else:
            # Mixed: the max is minimized here (or very close)
            lo = hi = mid
            break
    t_opt = (lo + hi) / 2
    h[j] = t_opt
    return t_opt
```

This is a great idea! Let me refine it. The binary search works because the max of affine functions is convex, and in 1D, a convex function's minimum can be found by checking the sign of the subgradient.

Actually, even better: the minimum of max_k (a_k + t*b_k) over t ∈ [0,1] occurs at a point where two (or more) of the linear functions are equal and are the maximum. We can find this by:
1. Computing b_k for each k
2. At t=0, the max is at k* = argmax(a_k + 0*b_k) = argmax(a_k)
3. If b_{k*} < 0, we can increase t to reduce the max
4. Find the t where another line crosses the max

But the binary search approach is simpler and fast enough.

Let me also think about a **pairwise coordinate descent**: optimize (h[i], h[j]) jointly. For two coordinates, corr[k] is bilinear (product of two variables), so the max is not jointly convex. But we can do inner-outer: fix j, optimize i, fix i, optimize j.

Actually, single coordinate optimization is already quite powerful. Let me implement a full cyclic coordinate descent with this exact 1D optimization.

The key advantage: each step is O(n^2) for computing correlations twice, but we can do it faster. Actually, changing one element h[j] changes corr[k] for all k by a known amount. Let me precompute.

Actually, for n=70, the correlation array has 139 elements. Computing np.correlate is O(n^2) = O(4900). With 70 coordinates in a cycle, each full pass is 70 * 2 * 4900 ≈ 686,000 operations. That's very fast.

We can do thousands of full passes in a second. This should converge very quickly to a local optimum.

```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _lse_obj(x, alpha, target, pen):
    corr = _corr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def _optimize_lbfgs(h, n, dx, alpha, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h.copy(), _c5(h, dx)
    for pen in [50.0, 200.0, 1000.0]:
        try:
            res = minimize(lambda x, a=alpha, p=pen: _lse_obj(x, a, target, p),
                best_h, method='L-BFGS-B', bounds=bounds,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except:
            pass
    return best_h, best_c5

def _optimize_slsqp(h, n, dx, alpha, maxiter=400):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x, a=alpha: _lse_obj(x, a, target, 0.0),
            h, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
            options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)

def _coord_opt(h, j, n):
    """Exact 1D optimization of h[j] to minimize max correlation."""
    hs = h[j]
    h[j] = 0.0
    c0 = _corr(h).copy()
    h[j] = 1.0
    c1 = _corr(h).copy()
    h[j] = hs
    b = c1 - c0
    a = c0
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = a + mid * b
        m = np.max(vals)
        active = vals >= m - 1e-12
        if np.sum(b[active] > 1e-12) > 0 and np.sum(b[active] < -1e-12) == 0:
            hi = mid
        elif np.sum(b[active] < -1e-12) > 0 and np.sum(b[active] > 1e-12) == 0:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def _cyclic_cd(h, n, dx, n_passes=200, deadline=None):
    """Cyclic coordinate descent with exact 1D subproblem."""
    h = h.copy()
    best_c5 = _c5(h, dx)
    improved = True
    pass_idx = 0
    while pass_idx < n_passes:
        if deadline is not None and time.time() >= deadline:
            break
        any_change = False
        for j in range(n):
            old = h[j]
            new = _coord_opt(h, j, n)
            if abs(new - old) > 1e-14:
                h[j] = new
                any_change = True
        h = _repair(h, n)
        c5 = _c5(h, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            improved = True
        elif not any_change:
            break
        pass_idx += 1
    return h, best_c5

def _worst_lag_opt(h, n, dx, max_iter=60):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h.copy(), _c5(h, dx)
    for _ in range(max_iter):
        corr = _corr(best_h)
        kw = np.argmax(corr)
        def wlo(x):
            return _corr(x)[kw] + 1000.0 * (np.sum(x) - target) ** 2
        try:
            res = minimize(wlo, best_h, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                options={'maxiter': 200, 'ftol': 1e-14})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5 - 1e-14:
                best_h, best_c5 = hc, c5
            else:
                break
        except:
            break
    return best_h, best_c5

def _diverse_starts(n, rng, parent):
    starts = [parent.copy()]
    starts.append(_repair(1.0 - parent, n))
    for i in range(4):
        f = rng.randint(1, 6)
        amp = rng.uniform(0.2, 0.4)
        ph = rng.uniform(0, 2 * np.pi)
        x = 0.5 + amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        starts.append(_repair(x, n))
    for i in range(3):
        x = np.clip(0.5 + rng.randn(n) * 0.3, 0, 1)
        x = np.convolve(x, np.ones(5) / 5, mode='same')
        starts.append(_repair(x, n))
    return starts

def _anneal(h, n, dx, deadline, T0=0.004, Tmin=1e-5, cool=0.9998):
    best_c5 = _c5(h, dx)
    h_cur, c5_cur = h.copy(), best_c5
    h_best = h_cur.copy()
    T = T0
    rng = np.random.RandomState(12345)
    while time.time() < deadline and T > Tmin:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 0.01 * (2.0 * rng.random() - 1.0)
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _c5(h_cur, dx)
        if c5n < c5_cur or rng.random() < np.exp(-(c5n - c5_cur) / T):
            c5_cur = c5n
            if c5n < best_c5:
                best_c5, h_best = c5n, h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
        T *= cool
    return h_best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    best_h, best_c5 = parent.copy(), _c5(parent, dx)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Exact cyclic coordinate descent on parent (60s)
    if tl() > 70:
```
