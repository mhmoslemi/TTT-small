    s = h.sum()

        if s <= 1e-14:
            h = np.full(n, 0.5)
            break
        factor = target / s
        h *= factor
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    # Exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(500):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        d = target - s
        if abs(d) < 1e-15:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-15:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def h_from_fourier(coeffs, n):
    i = np.arange(n)
    h = np.full(n, 0.5)
    m = len(coeffs) // 2
    for k in range(m):
        freq = k + 1
        h += coeffs[2*k] * np.cos(2.0 * np.pi * freq * i / n)
        h += coeffs[2*k+1] * np.sin(2.0 * np.pi * freq * i / n)
    return repair(h, n)

def fourier_obj_wrap(c, n):
    h = h_from_fourier(c, n)
    return compute_c5(h, n)

def slsqp_opt(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def objective(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -500, 500)
        return m * dx + dx * np.log(np.sum(np.exp(s))) / alpha
    def constraint(hv):
        return hv.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constraint}]
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
    for r in range(rounds):
        if time.time() > t_end:
            break
        improved = False
        ntries = 8 * n
        for _ in range(ntries):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    ni, nj = hi + sign * d, hj - sign * d
                    if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved and r > 40:
            break
    return h, best

def make_harmonic_start(n, seed_val, nharm=4):
    rng = np.random.RandomState(seed_val)
    coeffs = []
    for k in range(nharm):
        amp = 0.45 / (k + 1.5)
        coeffs.extend([rng.uniform(-amp, amp), rng.uniform(-amp, amp)])
    return h_from_fourier(np.array(coeffs), n)

def make_quarter_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    i = np.arange(n)
    h = 0.5 + 0.45 * np.sin(np.pi * i / n + rng.uniform(0, np.pi))
    h += 0.1 * rng.randn(n)
    return repair(h, n)

def make_block_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    nb = rng.randint(3, 7)
    bs = n // (2 * nb)
    if bs < 1:

    h = np.zeros(n)

        start = 2 * b * bs + rng.randint(0, max(1, bs // 2))
        end = min(start + bs, n)
        h[start:end] = 1.0
    h += 0.05 * rng.randn(n)
    return repair(h, n)
