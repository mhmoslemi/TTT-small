import numpy as np
from scipy.optimize import minimize

def _corr_fft(h, n):
    """Compute full cross-correlation of h with 1-h using FFT."""
    N = 2 * n  # power of 2 >= 2n-1
    a = np.zeros(N)
    b = np.zeros(N)
    a[:n] = h
    b[:n] = 1.0 - h
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    # cross-correlation: ifft(A * conj(B))
    c = np.fft.irfft(A * np.conj(B), N)[:2*n-1]
    return c

def _c5(h, n):

    c = _corr_fft(h, n) * dx

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h, n))) * dx

def _project(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    c = _corr(h, n) * dx_val
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    total = e.sum()
    w = e / total
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-14:

        i_min = max(0, -lag)
        i_max = min(n, n - lag)

            jp = i + lag
            g[jp] += wk * (1.0 - h[i])
            g[i] -= wk * h[jp]
    return g * dx_val, float(m + np.log(total) / alpha)

def _best_pair_d(c, delta, h, i, j):
    """Find optimal d for pair (i,j). Returns (d, new_c5)."""
    d_lo = max(-h[i], 1.0 - 1.0 - (1.0-h[j]))  # h[i]+d >= 0, h[j]-d <= 1
    # h[i]+d >= 0 => d >= -h[i]
    # h[i]+d <= 1 => d <= 1-h[i]
    # h[j]-d >= 0 => d <= h[j]
    # h[j]-d <= 1 => d >= h[j]-1
    d_lo = max(-h[i], h[j]-1.0)
    d_hi = min(1.0-h[i], h[j])
    if d_hi - d_lo < 1e-12:
        return 0.0, float(np.max(c))
    
    # Evaluate at critical points
    candidates = [0.0, d_lo, d_hi]
    # Top 2 crossing
    sorted_idx = np.argsort(c)[::-1]
    k1, k2 = sorted_idx[0], sorted_idx[1]
    dd = delta[k1] - delta[k2]
    if abs(dd) > 1e-14:
        d_cross = (c[k2] - c[k1]) / dd
        if d_lo < d_cross < d_hi:
            candidates.append(d_cross)
    
    best_d = 0.0
    best_val = float('inf')
    for d in candidates:
        d = np.clip(d, d_lo, d_hi)
        val = float(np.max(c + d * delta))
        if val < best_val:
            best_val = val
            best_d = d
    return best_d, best_val

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h, n))) * dx

def _project(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj_grad(h, n, alpha):
    dx_val = 2.0 / n
    c = _corr(h, n) * dx_val
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    total = e.sum()
    p = e / total
    obj = m + np.log(total) / alpha
    g = dx_val * (p @ _A(h, n))
    return obj, g

def _A(h, n):
    """A[k,i] = dc_k/dh_i (without dx). Shape (2n-1, n)."""
    n_c = 2 * n - 1
    lags = np.arange(n_c) - (n - 1)
    A = np.zeros((n_c, n))
    for k in range(n_c):
        lag = lags[k]
        for i in range(n):
            ip = i + lag
            im = i - lag
            v = 0.0
            if 0 <= ip < n:
                v += (1.0 - h[ip])
            if 0 <= im < n:
                v -= h[im]
            A[k, i] = v
    return A

def _slsqp_smooth(h0, n, alpha, maxiter, end_time):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def fun(x):
        return _smooth_obj_grad(x, n, alpha)[0]
    def jac(x):
        return _smooth_obj_grad(x, n, alpha)[1]
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() < end_time and res.success:
            return _project(res.x, n / 2.0)
    except Exception:
        pass
    return None

def _proj_grad(h0, n, alpha, end_time):
    target = n / 2.0
    x = h0.copy()
    best = x.copy()
    best_c5 = _c5(x, n)
    v = np.zeros(n)
    lr = 0.005
    mu = 0.85
    stuck = 0
    for step in range(50000):
        if time.time() > end_time:
            break
        g, _ = _smooth_obj_grad(x, n, alpha)
        v = mu * v + lr * g
        x_new = _project(x - v, target)
        c5_new = _c5(x_new, n)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best = x_new.copy()
            x = x_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                lr *= 0.5
                v *= 0.5
                stuck = 0
                if lr < 1e-7:
                    break
            x = x_new
    return best, best_c5

def _pair_search(h0, n, max_passes, end_time):
    h = h0.copy()
    target = n / 2.0
    c5_cur = _c5(h, n)
    best_h = h.copy()
    best_c5 = c5_cur
    dx_val = 2.0 / n
    for p in range(max_passes):
        if time.time() > end_time:
            break
        A_mat = _A(h, n)
        c = _corr(h, n) * dx_val
        improved = False
        total_gain = 0.0
        for i in range(n):
            if time.time() > end_time:
                break
            for j in range(i + 1, n):
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-13:
                    continue
                delta = (A_mat[:, i] - A_mat[:, j]) * dx_val
                candidates = [0.0, d_lo, d_hi]
                idx_sorted = np.argsort(c)[::-1][:4]
                for a in range(len(idx_sorted)):
                    for b in range(a + 1, len(idx_sorted)):
                        ka, kb = idx_sorted[a], idx_sorted[b]
                        dd = delta[ka] - delta[kb]
                        if abs(dd) > 1e-15:
                            dc = (c[kb] - c[ka]) / dd
                            if d_lo + 1e-13 < dc < d_hi - 1e-13:
                                candidates.append(dc)
                best_d = 0.0
                best_val = float(np.max(c))
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))
                    val = float(np.max(c + d * delta))
                    if val < best_val - 1e-16:
                        best_val = val
                        best_d = d
                if best_d != 0.0 and best_val < c5_cur - 1e-15:
                    h[i] += best_d
                    h[j] -= best_d
                    c += best_d * delta
                    c5_cur = best_val
                    improved = True
                    total_gain += (best_val - c5_cur) * -1
        h = _project(h, target)
        c5_check = _c5(h, n)
        if c5_check < best_c5 - 1e-16:
            best_c5 = c5_check
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def _topk_slsqp(h0, n, k, end_time):
    dx_val = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def fun(x):
        c = _corr(x, n) * dx_val
        idx = np.argsort(c)[::-1][:k]
        return float(np.sum(c[idx]))
    def jac(x):
        c = _corr(x, n) * dx_val
        idx = np.argsort(c)[::-1][:k]
        A_mat = _A(x, n)
        g = dx_val * (np.zeros(2*n-1) @ 0.0)
        g = np.zeros(n)
        for ki in idx:
            g += x[:0].sum() * 0 + A_mat[ki] * dx_val
        return g / k
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-14})
        if res.success:
            return _project(res.x, n / 2.0)
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float).copy()

            h_init = best_h.copy() if r == 0 else _project(best_h + 0.02 * (1 + 0.3*r) * rng.randn(n), n/2.0)

    if time.time() < t0 + budget_s - 200:

        if c5_pg < best_c5:

    if time.time() < t0 + budget_s - 180:

            if time.time() > t0 + budget_s - 150:

            res = _topk_slsqp(best_h, n, k, t0 + budget_s - 150)
            if res is not None:
                c5_try = _c5(res, n)
                if c5_try < best_c5:
                    best_c5 = c5_try

            best_h = h_pg2

            if c5_ps2 < best_c5:
                best_c5 = c5_ps2

    best_c5 = _c5(best_h, n)
