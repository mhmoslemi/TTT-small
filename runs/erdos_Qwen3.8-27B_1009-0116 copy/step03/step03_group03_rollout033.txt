import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _project(x, target, n):
    """Euclidean projection onto {0<=h<=1, sum(h)=target}"""
    lo, hi = -x.max() - 1.0, 1.0 - x.min() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + 0.5*(lo+hi), 0.0, 1.0)

def _power_sum(h, p):
    c = _corr(h)
    return (np.sum(c**p))**(1.0/p)

def _ps_grad(h, p):
    n = len(h)
    c = _corr(h)
    ss = np.sum(c**p)
    if ss < 1e-30:
        return np.zeros(n)
    w = c**(p-1) / ss**(1.0/p)
    grad = np.zeros(n)
    for m in range(2*n-1):
        k = m - (n-1)
        wm = w[m]
        if abs(wm) < 1e-18:
            continue
        if k >= 0:
            grad[:n-k] += wm * (1.0 - h[k:n])
            grad[k:n] -= wm * h[:n-k]
        else:
            kp = -k
            grad[kp:n] += wm * (1.0 - h[:n-kp])
            grad[:n-kp] -= wm * h[kp:n]
    return grad

def _pg_solve(h, n, target, budget_fn, iters=5000):
    """Proximal gradient with power-sum, adaptive p."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    p = 4
    step = 1.0
    for it in range(iters):
        if budget_fn() < 30:
            break
        # Increase p periodically
        p = min(4 * (1 << (it // 500)), 512)
        g = _ps_grad(h, p)
        # Backtracking
        f_cur = _power_sum(h, p)
        for _bt in range(30):
            h_new = _project(h - step * g, target, n)
            f_new = _power_sum(h_new, p)
            if f_new <= f_cur - 0.5 * step * np.dot(g, h_new - h):
                break
            step *= 0.5
        else:
            step *= 0.5
            h_new = _project(h - step * g, target, n)
        h = h_new
        step = min(step * 1.5, 10.0)  # increase step if it worked
        c5_now = _c5(h, dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(x, target, n):
    lo, hi = -float(x.max()) - 1.0, 1.0 - float(x.min()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

def _power_sum(h, p):
    c = _corr(h)
    return float((np.sum(c ** p)) ** (1.0 / p))

def _ps_grad(h, p):
    n = len(h)
    c = _corr(h)
    ss = float(np.sum(c ** p))
    if ss < 1e-30:
        return np.zeros(n)
    w = c ** (p - 1) / ss ** (1.0 / p)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = n - 1 - m
        wm = w[m]
        if abs(wm) < 1e-18:
            continue
        if k >= 0:
            grad[:n-k] += wm * (1.0 - h[k:n])
            grad[k:n] -= wm * h[:n-k]
        else:
            kp = -k
            grad[kp:n] += wm * (1.0 - h[:n-kp])
            grad[:n-kp] -= wm * h[kp:n]
    return grad

def _lag_grad(h, n, m):
    k = n - 1 - m
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:n])
        g[k:n] -= h[:n-k]
    else:
        kp = -k
        g[kp:n] += (1.0 - h[:n-kp])
        g[:n-kp] -= h[kp:n]
    return g

def _sum_con(x, target):
    return np.sum(x) - target

def _obj_p64(x):

def _obj_p256(x):

def _coord_descent(h, n, target, t0, budget_s, rng, iterations=200000):

    best_h = h.copy()
    best_c5 = _c5(h, dx)
    deltas = (0.004, 0.002, 0.001, -0.004, -0.002, -0.001, 0.012, -0.012, 0.0005, -0.0005, 0.006, -0.006)
    for it in range(iterations):
        if budget_s - (time.time() - t0) < 8:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        for d in deltas:
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

                    best_c5 = c5_try
                    best_h = h.copy()
                    break
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c5

        h_opt, c5_opt = _coord_descent(best_h, n, target, t0, budget_s, rng, 100000)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 9: Final SLSQP polish
    if budget_s - (time.time() - t0) > 3:
        try:
            res = minimize(_obj_p256, best_h, method='SLSQP', bounds=bounds,
                         constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (target,)}],
                         options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
            hc = _project(res.x, target, n)
            c5_try = _c5(hc, dx)
            if c5_try < best_c5:
                best_h, best_c5 = hc, c5_try
        except Exception:
            pass

    best_h = _project(best_h, target, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
