import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _project(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2} using bisection."""
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def _eval_correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval_c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _softmax_grad(h, n, alpha, top_k):
    """Compute gradient of softmax-smoothed max overlap."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Select top_k lags
    idx = np.argsort(c)[-top_k:]
    m = c[idx].max()
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    
    g = np.zeros(n)
    n_minus_1 = n - 1
    for wj, k_idx in zip(w, idx):
        lag = k_idx - n_minus_1
        if lag >= 0:
            # d/dh[j] g_lag = (1-h[j-lag])*1[j>=lag] - h[j+lag]*1[j<=n-1-lag]
            g[lag:] += wj * (1.0 - h[:n - lag])
            g[:n - lag] -= wj * h[lag:]
        else:
            m2 = -lag
            # d/dh[j] g_{-m} = (1-h[j+m])*1[j<=n-1-m] - h[j-m]*1[j>=m]
            g[:n - m2] += wj * (1.0 - h[m2:])
            g[m2:] -= wj * h[:n - m2]
    return g

def _pgd(h, n, alpha, top_k, steps, lr0, momentum):
    """Projected gradient descent with momentum."""
    h = h.copy()
    v = np.zeros(n)
    best = h.copy()
    best_c5 = _eval_c5(h)
    lr = lr0
    
    for it in range(steps):
        g = _softmax_grad(h, n, alpha, top_k)
        v = momentum * v - lr * g
        h_new = _project(h + v, n)
        
        c5_new = _eval_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        
        # Simple adaptive: if no improvement, reduce lr
        if c5_new > _eval_c5(h) + 1e-10:
            lr *= 0.9
            v = 0.9 * v
        h = h_new
