import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
                if diff < 1e-13:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
                if diff > -1e-13:
                    break
    return h

def _softmax_obj(h, alpha, lam, beta):
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    pen = lam * (h.sum() - len(h) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + pen - beta * np.sum(h * (1.0 - h))

def _softmax_grad(h, alpha, lam, beta):
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(h)
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        L = ki - (n - 1)
        pk = pw[ki]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += pk * (1.0 - h[:n - L])
            g[:n - L] -= pk * h[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (h.sum() - n / 2.0)
    g += -2.0 * beta * (1.0 - 2.0 * h)
    return g

def _active_set_linesearch(h, top_lags, n, dx, max_ls=40):
    c = _corr(h)
    n_correct = len(c)
    g = np.zeros(n)
    for ki in top_lags:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g_c = g - g.mean()
    lo, hi = 0.0, 1.0
    best_h = h.copy()
    best_c = _c5(h, dx)
    for _ in range(max_ls):
        mid = (lo + hi) / 2.0
        h_try = _repair(h - mid * g_c, n)
        c_try = _c5(h_try, dx)
        if c_try < best_c:
            best_h = h_try
            best_c = c_try
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-8:
            break
    return best_h, best_c
