```python
                    corr_d = corr_c
                    val_d = val_c
                    c_ = b - gr * (b - a)
                    corr_c = _eval_at(c_)
                    val_c = _smooth_max(corr_c, alpha)
                else:
                    a = c_
                    c_ = d_
                    corr_c = corr_d
                    val_c = val_d
                    d_ = a + gr * (b - a)
                    corr_d = _eval_at(d_)
                    val_d = _smooth_max(corr_d, alpha)
            
            # Pick the best of a, c, d, b
            candidates = [lo, hi, 0.5*(lo+hi), c_, d_]
            for cv in candidates:
                if lo - 1e-12 <= cv <= hi + 1e-12:
                    corr_v = _eval_at(cv)
                    val_v = _smooth_max(corr_v, alpha)
                    if val_v < best_val:
                        best_val = val_v
                        best_h_i = cv
                        best_corr = corr_v
            
            h[i] = best_h_i
            corr = best_corr
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _slsqp_step(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h.copy(), args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _repair(res.x, n)
        return h
    except Exception:
        return h

def _coord_descent(h, n, target, alpha, n_passes, t_limit):
    corr = _corr(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _pass in range(n_passes):
        if time.time() > t_limit:
            break
        for i in range(n):
            if time.time() > t_limit:
                break
            os = h.sum() - h[i]
            lo = max(0.0, target - os)
            hi = min(1.0, target - os)
            span = hi - lo
            if span < 1e-10:
                continue
            a, b = lo, hi
            c_ = b - gr * (b - a)
            d_ = a + gr * (b - a)
            hc = h.copy(); hc[i] = c_
            hd = h.copy(); hd[i] = d_
            fc = _smooth_max(_corr(hc), alpha)
            fd = _smooth_max(_corr(hd), alpha)
            for _it in range(22):
                if fc < fd:
                    b, d_, fd = d_, c_, fc
                    c_ = b - gr * (b - a)
                    htmp = h.copy(); htmp[i] = c_
                    fc = _smooth_max(_corr(htmp), alpha)
                else:
                    a, c_, fc = c_, d_, fd
                    d_ = a + gr * (b - a)
                    htmp = h.copy(); htmp[i] = d_
                    fd = _smooth_max(_corr(htmp), alpha)
            best_v, best_hv = min(fc, fd), c_ if fc < fd else d_
            cur_v = _smooth_max(corr, alpha)
            if best_v < cur_v - 1e-14:
                h[i] = best_hv
                corr = _corr(h)
    return h

def _sweep_round(h, n, corr, time_limit, n_coarse=25, n_bisect=30):
    improved = False
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for j_idx in range(ai + 1, n):
            j = order[j_idx]
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = int(np.argmin(vals))
            cur_max = float(np.max(corr))
            if vals[bi] >= cur_max - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if float(np.max(corr + mid * gd)) <= float(np.max(corr + hi_b * gd)):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

def _structured_perturb(h, n, target, rng):
    mode = rng.randint(4)
    x = np.linspace(0, 2, n)
    if mode == 0:
        freq = rng.choice([1, 2, 3, 4, 5])
        phase = rng.uniform(0, 2 * np.pi)
        h_p = h + 0.03 * np.sin(np.pi * freq * x + phase)
    elif mode == 1:
        h_p = h + 0.04 * np.random.randn(n)
    elif mode == 2:
        # shift a block
        bsz = rng.randint(3, 10)
        start = rng.randint(0, n - bsz)
        shift = rng.choice([-1, 1]) * rng.randint(1, 4)
        h_p = h.copy()
        seg = h[start:start + bsz].copy()
        s2 = np.clip(start + shift, 0, n - bsz)
        h_p[s2:s2 + bsz] = seg
        h_p[start:start + bsz] = 1.0 - seg
    else:
        # random binary-like perturbation
        mask = rng.random(n) < 0.1
        h_p = h.copy()
        h_p[mask] = rng.random(mask.sum())
    return _repair(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-up
    for alpha in [40, 120, 350, 800]:
        if time.time() - t0 > budget_s * 0.1:
            break
        hc = _slsqp_step(best_h, n, target, bounds, alpha, 300)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5, best_h = c5, hc.copy()

    # Phase 2: Coordinate descent + pair sweeps
    stall = 0
    total_iter = 0
    main_deadline = budget_s - 90
    while time.time() - t0 < main_deadline:
        total_iter += 1
        # Coordinate descent
        h_cd = _coord_descent(best_h.copy(), n, target, 200,
                              n_passes=3, t_limit=time.time() + 30)
        c5 = _c5(h_cd, dx)
        if c5 < best_c5 - 1e-14:
            best_c5, best_h = c5, h_cd.copy()
            stall = 0
        else:
            stall += 1

        # Pair sweep
        h_sw = best_h.copy()
        corr = _corr(h_sw)
        chunk_end = min(time.time() + 40, main_deadline)
        h_sw, corr, imp = _sweep_round(h_sw, n, corr, chunk_end, 25, 30)
        if imp:
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5 - 1e-14:
                best_c5, best_h = c5, h_sw.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # Periodic SLSQP to escape local minima
        if total_iter % 5 == 0 and time.time() - t0 < main_deadline - 30:
            alpha = 200 + 80 * (total_iter // 5)
            hc = _slsqp_step(best_h, n, target, bounds, alpha, 350)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _sweep_round(hc, n, hc_corr, time.time() + 15, 20, 25)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-15:
                best_c5, best_h = c5_t, hc.copy()
                stall = 0

        # Structured perturbation when stalled
        if stall >= 3 and time.time() - t0 < main_deadline - 40:
            h_p = _structured_perturb(best_h, n, target, rng)
            hc = _slsqp_step(h_p, n, target, bounds, 500, 250)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _sweep_round(hc, n, hc_corr, time.time() + 20, 25, 30)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-15:
                best_c5, best_h = c5_t, hc.copy()
            # Also try simpler coordinate descent from perturbation
            hc2 = _coord_descent(h_p, n, target, 200, 5, time.time() + 20)
            c5_t2 = _c5(hc2, dx)
            if c5_t2 < best_c5 - 1e-15:
                best_c5, best_h = c5_t2, hc2.copy()
            stall = 0

    # Phase 3: Alternating high-precision coordinate descent and sweeps
    remaining = budget_s - 15 - (time.time() - t0)
    n_final = max(1, int(remaining // 25))
    for _ in range(n_final):
        if time.time() - t0 > budget_s - 12:
            break
        h_f = _coord_descent(best_h.copy(), n, target, 500, 5, time.time() + 20)
        c5 = _c5(h_f, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f.copy()
        corr_f = _corr(best_h)
        best_h, corr_f, _ = _sweep_round(best_h, n, corr_f, time.time() + 15, 31, 38)
        c5 = float(np.max(corr_f) * dx)
        if c5 < best_c5:
            best_c5 = c5

    # Phase 4: Final SLSQP polish
    if time.time() - t0 < budget_s - 3:
        for alpha in [1500, 3000]:
            if time.time() - t0 > budget_s - 2:
                break
            hc = _slsqp_step(best_h, n, target, bounds, alpha, 500)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
