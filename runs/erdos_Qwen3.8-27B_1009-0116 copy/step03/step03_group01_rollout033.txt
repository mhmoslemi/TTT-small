                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    return float(np.max(_corr(h)) * 2.0 / len(h))

def _sf(x, a, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sj(x, a, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        j = k - (n - 1)
        if j >= 0:
            i = np.arange(n - j)
            g[i + j] += wk * (1.0 - x[i])
            g[i] -= wk * x[i + j]
        else:
            m2 = -j
            i = np.arange(m2, n)
            g[i] += wk * (1.0 - x[i - m2])
            g[i - m2] -= wk * x[i]
    return g

def _sum_fun(x):
    return x.sum() - len(x) / 2.0

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_opt(h, n, alpha, iters=2000, ftol=1e-19):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_sf, h, jac=_sj, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_grad(h, n, lags):
    g = np.zeros(n)
    for kk in lags:
        j = kk - (n - 1)
        if j >= 0:
            i = np.arange(n - j)
            g[i + j] += (1.0 - h[i])
            g[i] -= h[i + j]
        else:
            m2 = -j
            i = np.arange(m2, n)
            g[i] += (1.0 - h[i - m2])
            g[i - m2] -= h[i]
    return g / len(lags)

def _proj_grad(h, n, topk, lr0, niter, tol=1e-6):
    h = h.copy()
    lr = lr0
    best = _eval(h)
    stagnant = 0
    for it in range(niter):
        c = _corr(h)
        cmax = c.max()
        active = np.where(c > cmax - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        active = active[:topk]
        g = _active_grad(h, n, active)
        h_new = _repair(h - lr * g, n)
        c_new = _eval(h_new)
        if c_new < best - 1e-18:
            h = h_new
            best = c_new
            stagnant = 0
            lr = min(lr * 1.01, 0.05)
        else:
            improved = False
            for f in [0.5, 0.25, 0.1]:
                h_try = _repair(h - lr * f * g, n)
                c_try = _eval(h_try)
                if c_try < best - 1e-18:
                    h = h_try
                    best = c_try
                    lr = lr * f
                    improved = True
                    break
            if not improved:
                lr *= 0.7
                stagnant += 1
            else:
                stagnant = 0
        if stagnant > 50:
            break
    return h, best

def _sa_optimize(h, n, rng, n_iter, T0, T_min, cool, deadline_fn):
    h = h.copy()
    c5_cur = _eval(h)
    T = T0
    best_local = c5_cur
    best_local_h = h.copy()
    no_improve = 0
    for it in range(n_iter):
        if deadline_fn():
            break
        r = rng.random()
        if r < 0.45:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = (T * 8 + 0.002) * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if not (0 <= ni <= 1 and 0 <= nj <= 1):
                continue
            h[i], h[j] = ni, nj
            c5_new = _eval(h)
            if c5_new <= c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / T):
                c5_cur = c5_new
                if c5_cur < best_local - 1e-15:
                    best_local = c5_cur
                    best_local_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                no_improve += 1
        elif r < 0.70:
            i = rng.randint(n - 1)
            d = (T * 5 + 0.001) * rng.randn()
            ni, nj = h[i] + d, h[i + 1] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[i + 1] = ni, nj
                c5_new = _eval(h)
                if c5_new <= c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / T):
                    c5_cur = c5_new
                    if c5_cur < best_local - 1e-15:
                        best_local = c5_cur
                        best_local_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    h[i], h[i + 1] = h[i] - d, h[i + 1] + d
        elif r < 0.85:
            i = rng.randint(max(1, n - 5), min(n - 1, 5))
            i = rng.randint(n - 1)
            d = (T * 4 + 0.001) * rng.randn()
            ni, nj = h[i] + d, h[i + 1] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[i + 1] = ni, nj
                c5_new = _eval(h)
                if c5_new <= c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / T):
                    c5_cur = c5_new
                    if c5_cur < best_local - 1e-15:
                        best_local = c5_cur
                        best_local_h = h.copy()
                        no_improve = 0
                else:
                    h[i], h[i + 1] = h[i] - d, h[i + 1] + d
        else:
            freq = rng.randint(1, max(2, n // 4))
            amp = (T * 3 + 0.001) * rng.randn()
            phase = rng.uniform(0, 2 * np.pi)
            x_idx = np.arange(n)
            pert = amp * np.sin(2.0 * np.pi * freq * x_idx / n + phase)
            h_new = _repair(h + pert, n)
            c5_new = _eval(h_new)
            if c5_new <= c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / T):
                h = h_new
                c5_cur = c5_new
                if c5_cur < best_local - 1e-15:
                    best_local = c5_cur
                    best_local_h = h.copy()
                    no_improve = 0
        T = max(T * cool, T_min)
        if no_improve > 5000:
            T = min(T * 2.0, T0 * 0.3)
            no_improve = 0
    return best_local_h, best_local

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    def _el():
        return time.time() - t0

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP with increasing smoothness
    for alpha in [300, 3000, 30000, 300000]:
        if _el() > budget_s - 350:
            break
        h = _slsqp_opt(best_h, n, alpha, 2000, 1e-19)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Active-set projected gradient (main workhorse)
    if _el() < budget_s - 250:
        h = best_h.copy()
        lr_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        for lr0 in lr_schedule:
            if _el() > budget_s - 250:
                break
            h, c5 = _proj_grad(h, n, topk=5, lr0=lr0, niter=300, tol=1e-5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 3: Simulated annealing with structured moves
    if _el() < budget_s - 150:
        h, c5 = _sa_optimize(best_h, n, rng, 300000, 0.003, 1e-8, 0.9998,
                             lambda: _el() > budget_s - 150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Second round active-set + SA
    if _el() < budget_s - 80:
        h = best_h.copy()
        h, c5 = _proj_grad(h, n, topk=3, lr0=0.0005, niter=500, tol=1e-6)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h, c5 = _sa_optimize(best_h, n, rng, 200000, 0.001, 1e-9, 0.9999,
                             lambda: _el() > budget_s - 80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Final SLSQP polish
    if _el() < budget_s - 40:
        h = _slsqp_opt(best_h, n, 1000000, 3000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Ultra-fine coordinate descent
    if _el() < budget_s - 10:
        h = best_h.copy()
        c5c = _eval(h)
        steps = np.array([1e-5, 2e-5, 5e-5, 1e-4, 2e-4])
        for _it in range(80000):
            if _el() > budget_s - 10:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.choice(steps) * rng.choice([1.0, -1.0])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
