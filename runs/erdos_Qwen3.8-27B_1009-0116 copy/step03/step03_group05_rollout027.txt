    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def c5_val(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_corr(h, k, n):
    """Gradient of c_k = sum_i h[i]*(1-h[i+k]) w.r.t. h"""
    g = np.zeros(n)
    L = k - (n - 1)  # actual lag
    if L == 0:
        g = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n-L])
        g[:n-L] -= h[L:]
    else:
        M = -L
        g[:n-M] += (1.0 - h[M:])
        g[M:] -= h[:n-M]
    return g

def active_grad(h, idx_list, n):
    """Average gradient of correlations at given indices"""
    g = np.zeros(n)
    for ki in idx_list:
        g += grad_corr(h, ki, n)
    g /= len(idx_list)
    return g

def proj_onto(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} approximately"""
    return repair(h, n)

def softmax_obj(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    pen = lam * (np.sum(x) - n/2.0)**2
    return m + np.log(np.sum(np.exp(s))) / alpha + pen

def softmax_grad(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2*n - 1):
        pk = pw[k]
        if pk < 1e-14:
            continue
        g += pk * grad_corr(x, k, n)
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - n/2.0)
    return g

from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = max(diff, h[idx] - 1.0)
            h[idx] += a
            diff -= a
    return h

def c5_of(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_one(h, lag_idx, n):
    g = np.zeros(n)
    L = lag_idx - (n - 1)
    if L == 0:
        g = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        M = -L
        g[:n-M] += 1.0 - h[M:]
        g[M:] -= h[:n-M]
    return g

    g = np.zeros(n)
    for ki in idx:
        g += grad_one(h, ki, n)
    return g / len(idx)

def softmax_obj(x, a, lam, n):
    c = np.correlate(x, 1.0-x, mode='full')
    m = np.max(c)
    s = np.clip(a*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s)))/a + lam*(x.sum()-n/2.0)**2

def softmax_jac(x, a, lam, n):
    c = np.correlate(x, 1.0-x, mode='full')
    m = np.max(c)
    s = np.clip(a*(c-m), -500, 500)

    g += 2.0*lam*(x.sum()-n/2.0)

    """Spectral shaping: flatten autocorrelation in frequency domain"""
    H = np.fft.rfft(h)
    mag = np.abs(H)
    phase = np.angle(H)
    R0 = np.dot(h, h)  # power
    target_mag = np.sqrt(R0 / n) * np.ones_like(mag)
    target_mag[0] = H[0]  # keep DC
    new_mag = (1-alpha)*mag + alpha*target_mag
    H_new = new_mag * np.exp(1j*phase)
    h_new = np.fft.irfft(H_new, n=n)
    return h_new

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            ord_ = np.argsort(1.0 - h)
        else:
            ord_ = np.argsort(h)
        for i in ord_:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def grad_sm(h, alpha, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)

    pw = ew / ew.sum()
    g = np.zeros(n)

            g += pw[k] * grad1(h, k, n)

    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
        else:
            o = np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break

            else:
                a = max(d, -h[i])

    for mt in [2, 3, 5, 8, 15]:
        h = bh.copy()
        step = 0.08
        stalled = 0
        for _ in range(600):
            if time.time() - t0 > lim:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            cm = c.max()
            thr = cm - 0.003 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            g = act_grad(h, idx, n)
            g -= g.mean()
            ok = False
            for _b in range(30):
                hn = repair(h - step * g, n)
                cn = c5v(hn, dx)
                if cn < c5v(h, dx) - 1e-18:
                    h = hn
                    ok = True
                    if cn < bc:
                        bc, bh = cn, h.copy()
                    step = min(step * 1.12, 1.5)
                    stalled = 0
                    break
                step *= 0.72
            if not ok:
                stalled += 1
                if stalled > 8:
                    break
    return bh, bc

def spec_phase(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.15, 0.25, 0.4, 0.55, 0.4, 0.25, 0.15, 0.1]:
        h = bh.copy()
        for _ in range(250):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a) * mag + a * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-18:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def cd2d(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    halts = 0
    for _ in range(200):
        if time.time() - t0 > lim:
            break
        imp = False
        for _ in range(500):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            for d in (0.01, 0.004, 0.0015, 0.0008, -0.01, -0.004, -0.0015, -0.0008):
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = c5v(h, dx)
                    if c < cc - 1e-15:
                        h = repair(h, n)
                        cc = c5v(h, dx)
                        imp = True
                        if cc < bc:
                            bc, bh = cc, h.copy()
                        break
                    h[i], h[j] = ni - d, nj + d
        halts = halts + 1 if not imp else 0
        if halts >= 4:
            break
    return bh, bc

def polish(h, n, dx, rng, cnt, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    dt = np.array([4e-4, 8e-4, 1.5e-3, -4e-4, -8e-4, -1.5e-3])
    for _ in range(cnt):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = dt[rng.randint(6)]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def opt_h(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = lbfgs_phase(bh, n, dx, t0, lim - 250)
    if c < bc:
        bh, bc = r, c
    r, c = slsqp_phase(bh, n, dx, t0, lim - 180)
    if c < bc:
        bh, bc = r, c
    r, c = act_phase(bh, n, dx, t0, lim - 130)
    if c < bc:
        bh, bc = r, c
    r, c = spec_phase(bh, n, dx, t0, lim - 90)
    if c < bc:
        bh, bc = r, c
    r, c = lbfgs_phase(bh, n, dx, t0, lim - 70)
    if c < bc:
        bh, bc = r, c
    r, c = cd2d(bh, n, dx, rng, t0, lim - 30)
    if c < bc:
        bh, bc = r, c
    r, c = spec_phase(bh, n, dx, t0, lim - 15)
    if c < bc:
        bh, bc = r, c
    r, c = polish(bh, n, dx, rng, 25000, t0, lim - 3)
    if c < bc:
        bh, bc = r, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    r, c = opt_h(h0, nl, dx0, seed, t0, budget_s - 80)
    if c < bc:
        bh, bc, bn = r, c, nl

    n2 = nl * 2
    if n2 < 1000 and time.time() - t0 < budget_s - 100:
        xo = np.linspace(0, 2, nl, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, h0, period=2.0), n2)
        d2 = 2.0 / n2
        r, c = opt_h(h2, n2, d2, seed + 1, t0, budget_s - 15)
        if c < bc:
            bh, bc, bn = r, c, n2

    n3 = nl * 3
    if n3 < 1000 and time.time() - t0 < budget_s - 15:
        xo = np.linspace(0, 2, nl, endpoint=False)
        xn = np.linspace(0, 2, n3, endpoint=False)
        h3 = repair(np.interp(xn, xo, h0, period=2.0), n3)
        d3 = 2.0 / n3
        r, c = opt_h(h3, n3, d3, seed + 2, t0, budget_s - 3)
        if c < bc:
            bh, bc, bn = r, c, n3

    if time.time() - t0 < budget_s - 8:
        rng2 = np.random.RandomState(seed + 500)
        hp = repair(bh + 0.015 * rng2.randn(bn), bn)
        r, c = opt_h(hp, bn, 2.0 / bn, seed + 900, t0, budget_s - 2)
        if c < bc:
            bh, bc, bn = r, c, bn

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
