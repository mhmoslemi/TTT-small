def _equalize_dir(h, c, top, n):
    """Compute direction to equalize top lags."""
    K = len(top)
    if K < 2:
        return np.zeros(n)
    # Gradients for each top lag
    Js = [_grad_c(h, m, n) for m in top]
    # Want: c[top[0]] + J0·d = c[top[i]] + Ji·d for i=1..K-1
    # => (J0 - Ji)·d = c[top[i]] - c[top[0]]
    A = np.zeros((K-1, n))
    b = np.zeros(K-1)
    for i in range(1, K):
        A[i-1] = Js[0] - Js[i]
        b[i-1] = c[top[i]] - c[top[0]]
    # Min norm solution: d = A^T (A A^T)^{-1} b
    AtA = A @ A.T + 1e-12 * np.eye(K-1)
    try:
        y = np.linalg.solve(AtA, b)
    except:
        y = np.linalg.lstsq(AtA, b, rcond=None)[0]
    d = A.T @ y
    return d

    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
