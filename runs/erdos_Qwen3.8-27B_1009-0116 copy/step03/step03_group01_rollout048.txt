def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sJ_lse(x, a):
    n = len(x)

    m = c.max()
    s = np.clip(a * (c - m), -500, 500)

            continue

        wk = w[k_idx]
        if lag >= 0:
            cnt = n - lag
            g[:cnt] += wk * (1.0 - x[lag:lag+cnt])
            g[lag:lag+cnt] -= wk * x[:cnt]
        else:

            cnt = n - ll
            g[ll:ll+cnt] += wk * (1.0 - x[:cnt])

        cnt = n - ll
        g[ll:ll+cnt] += (1.0 - h[:cnt])
        g[:cnt] -= h[ll:ll+cnt]
