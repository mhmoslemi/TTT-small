from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth(corr, alpha):
    m = corr.max()
    s = np.clip(alpha * (corr - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj(x, alpha, target, pen):
    c = _corr(x)
    val = _smooth(c, alpha)
    if pen > 0:
        val += pen * (x.sum() - target) ** 2
    return val

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        idx = np.argsort(1.0 - h if d > 0 else h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-14:
                break
    return h

def _lbfgs(h, n, dx, alpha, dt=40):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best, bc5 = h, _c5(h, dx)
    t_s = time.time()
    for pen in [10.0, 100.0, 1000.0, 10000.0]:
        if time.time() - t_s > dt:
            break
        res = minimize(lambda x, a=alpha, p=pen: _obj(x, a, target, p),
                       best, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-10})
        hc = _repair(res.x, n)
        c5 = _c5(hc, dx)
        if c5 < bc5:
            best, bc5 = hc, c5
    return best, bc5

def _slsqp(h, n, dx, alpha, dt=30, maxiter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    t_s = time.time()
    try:
        res = minimize(lambda x: _obj(x, alpha, target, 0), h,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)

def _grad_local(h, n, dx, deadline, lr=0.001):

    h = h.copy()
    target = n / 2.0

        ki = corr.argmax()
        m = ki - n + 1
        g = np.zeros(n)

            g[m:] -= h[:n-m]
        else:

            g[m2:] += (1.0 - h[:n-m2])
            g[:n-m2] -= h[m2:]
        h_new = _repair(h - lr * g, n)
        c5 = _c5(h_new, dx)
        if c5 < best - 1e-15:
            h, best = h_new, c5
        else:
            lr *= 0.8
            if lr < 1e-7:
                break
    return h, best

def _pair_local(h, n, dx, rng, deadline, max_it=300000):
    best = _c5(h, dx)
    h = h.copy()
    scale = 0.01
    stall = 0
    for _ in range(max_it):

            break
        if stall > 3000:

            stall = 0

    return h, best

def _anneal_local(h, n, dx, rng, deadline):
    best = _c5(h, dx)
    hc = h.copy()
    cc = best
    T = 0.002
    T_min = 1e-5
    cool = 0.9998
    while time.time() < deadline and T > T_min:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 0.005 * (2.0 * rng.random() - 1.0)
        ni, nj = hc[i] + d, hc[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:

        cn = _c5(hc, dx)
        if cn < cc or rng.random() < np.exp(-(cn - cc) / T):
            cc = cn
            if cn < best:
                best = cn
        else:
            hc[i], hc[j] = ni - d, nj + d
        T *= cool
    return hc, best

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth(corr, alpha):
    m = corr.max()
    s = np.clip(alpha * (corr - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj(x, alpha, target, pen):
    c = _corr(x)
    val = _smooth(c, alpha)
    if pen > 0:
        val += pen * (x.sum() - target) ** 2
    return val

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        idx = np.argsort(1.0 - h if d > 0 else h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-14:
                break
    return h

def _lbfgs(h, n, dx, alpha, dt=15):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best, bc5 = h, _c5(h, dx)
    t_s = time.time()
    for pen in [10.0, 200.0, 2000.0, 20000.0]:
        if time.time() - t_s > dt:
            break
        res = minimize(lambda x, a=alpha, p=pen: _obj(x, a, target, p),
                       best, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-10})
        hc = _repair(res.x, n)
        c5 = _c5(hc, dx)
        if c5 < bc5:
            best, bc5 = hc, c5
    return best, bc5

def _slsqp(h, n, dx, alpha, dt=25, maxiter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    t_s = time.time()
    try:
        res = minimize(lambda x: _obj(x, alpha, target, 0), h,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _grad_local(h, n, dx, deadline, lr=0.002):
    best = _c5(h, dx)
    h = h.copy()
    while time.time() < deadline and lr > 1e-8:
        corr = _corr(h)
        ki = int(corr.argmax())
        m = ki - n + 1
        g = np.zeros(n)
        if m >= 0:
            if m < n:
                g[:n - m] += (1.0 - h[m:])
                g[m:] -= h[:n - m]
        else:
            m2 = -m
            if m2 < n:
                g[m2:] += (1.0 - h[:n - m2])
                g[:n - m2] -= h[m2:]
        h_new = _repair(h - lr * g, n)
        c5 = _c5(h_new, dx)
        if c5 < best - 1e-15:
            h, best = h_new, c5
            lr = min(lr * 1.05, 0.01)
        else:
            lr *= 0.75
    return h, best

def _pair_local(h, n, dx, rng, deadline, max_it=500000):
    best = _c5(h, dx)
    h = h.copy()
    scale = 0.01
    stall = 0
    for _ in range(max_it):
        if time.time() >= deadline:
            break
        if stall > 4000:
            scale *= 0.55
            stall = 0
            if scale < 1e-7:
                break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * (2.0 * rng.random() - 1.0)
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
            continue
        h[i], h[j] = ni, nj
        c5 = _c5(h, dx)
        if c5 < best - 1e-15:
            best, stall = c5, 0
        else:
            h[i], h[j] = ni - d, nj + d
            stall += 1
    return h, best

def _anneal_local(h, n, dx, rng, deadline):
    best = _c5(h, dx)
    hc = h.copy()
    cc = best
    T = 0.003
    T_min = 5e-6
    cool = 0.99985
    while time.time() < deadline and T > T_min:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 0.006 * (2.0 * rng.random() - 1.0)
        ni, nj = hc[i] + d, hc[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        hc[i], hc[j] = ni, nj
        cn = _c5(hc, dx)
        if cn < cc or rng.random() < np.exp(-(cn - cc) / T):
            cc = cn
            if cn < best:
                best = cn
        else:
            hc[i], hc[j] = ni - d, nj + d
        T *= cool
    return hc, best

def _perturb(h, n, rng, kind):
    if kind == 0:
        hb = (h > 0.5).astype(float)
        ones, zeros = np.where(hb == 1)[0], np.where(hb == 0)[0]
        for _ in range(rng.randint(2, 7)):
            if len(ones) and len(zeros):
                hb[rng.choice(ones)] = 0.0
                hb[rng.choice(zeros)] = 1.0
                ones, zeros = np.where(hb == 1)[0], np.where(hb == 0)[0]
        return hb
    elif kind == 1:
        amp = rng.uniform(0.04, 0.12)
        f = rng.randint(1, 5)
        ph = rng.uniform(0, 6.28)
        t = np.arange(n) / n
        return _repair(h + amp * np.sin(2 * np.pi * f * t + ph), n)
    elif kind == 2:
        k = rng.randint(1, n // 3)
        return _repair(np.roll(h, k), n)
    elif kind == 3:
        f1, f2 = rng.randint(1, 4), rng.randint(1, 4)
        p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
        t = np.arange(n) / n
        amp = rng.uniform(0.03, 0.10)
        return _repair(h + amp * np.sin(2 * np.pi * f1 * t + p1) * np.sin(2 * np.pi * f2 * t + p2), n)
    else:
        return _repair(h + rng.uniform(-0.06, 0.06, n), n)

def _block_local(h, n, dx, rng, deadline, max_it=30000):
    best = _c5(h, dx)
    h = h.copy()
    eps0 = 0.008
    stall = 0
    for _ in range(max_it):
        if time.time() >= deadline:
            break
        if stall > 3000:
            eps0 *= 0.6
            stall = 0
            if eps0 < 5e-5:
                break
        bs = rng.choice([2, 3])
        if bs >= n:
            continue
        i = rng.randint(0, n - bs)
        j = rng.randint(0, n - bs)
        if abs(i - j) < bs + 1:
            continue
        eps = eps0 * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if h[i + di] + eps < -1e-12 or h[i + di] + eps > 1 + 1e-12:
                ok = False
                break
            if h[j + di] - eps < -1e-12 or h[j + di] - eps > 1 + 1e-12:
                ok = False
                break
        if not ok:
            continue
        h[i:i + bs] += eps
        h[j:j + bs] -= eps
        c5 = _c5(h, dx)
        if c5 < best - 1e-15:
            best, stall = c5, 0
        else:
            h[i:i + bs] -= eps
            h[j:j + bs] += eps
            stall += 1
    return h, best
