    # Set h[j] = 0 to compute b_k, then set h[j] = 1 to compute a_k + b_k
    h[j] = 0.0
    corr0 = _corr_full(h)
    h[j] = 1.0
    corr1 = _corr_full(h)
    h[j] = h_save
    
    a = corr1 - corr0  # slope
    b = corr0  # intercept
    
    # f(t) = max_k (a_k * t + b_k) for t in [0,1]
    # Convex piecewise linear
    # Evaluate at key points
    candidates = [0.0, 1.0, h_save]
    # Add intersection points of top lines
    # Simple: just do a fine search
    grid = np.linspace(0, 1, 50)
    vals = np.max(a[:, None] * grid[None, :] + b[:, None], axis=0)
    best_t = grid[np.argmin(vals)]
    
    # Refine around best_t
    if 0 < best_t < 1:
        fine = np.linspace(max(0, best_t - 0.02), min(1, best_t + 0.02), 100)
        vals_f = np.max(a[:, None] * fine[None, :] + b[:, None], axis=0)
        best_t = fine[np.argmin(vals_f)]
    
    h[j] = best_t
    return best_t

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _smooth_c5_grad(h, alpha):
    n = len(h)
    corr = _corr_full(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)
    f = m + np.log(np.sum(exp_s)) / alpha
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wk = w[d + n - 1]
        if wk < 1e-12:
            continue
        if d >= 0:
            idx1 = np.arange(n - d)
            grad[idx1] += wk * (1.0 - h[idx1 + d])
            idx2 = np.arange(d, n)
            grad[idx2] -= wk * h[idx2 - d]
        else:
            dd = -d
            idx1 = np.arange(dd, n)
            grad[idx1] += wk * (1.0 - h[idx1 - dd])
            idx2 = np.arange(n - dd)
            grad[idx2] -= wk * h[idx2 + dd]
    return f, grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _lbfgs_analytic(h, n, dx, alpha, maxiter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _smooth_c5_grad(x, alpha)[0]

    def grad(x):
        return _smooth_c5_grad(x, alpha)[1]

    best_h, best_c5 = h.copy(), _c5(h, dx)
    for pen in [50.0, 500.0, 5000.0]:
        def obj_p(x):
            return _smooth_c5_grad(x, alpha)[0] + pen * (x.sum() - target) ** 2

        def grad_p(x):
            return _smooth_c5_grad(x, alpha)[1] + 2.0 * pen * (x.sum() - target)

        try:
            res = minimize(obj_p, best_h, jac=grad_p, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
    return best_h, best_c5

def _coord_descent(h, n, dx, deadline, max_sweeps=200):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() >= deadline:
            break
        improved = False
        for j in range(n):
            h_save = h[j]
            h[j] = 0.0
            corr0 = _corr_full(h)
            h[j] = 1.0
            corr1 = _corr_full(h)
            h[j] = h_save
            a = corr1 - corr0
            b = corr0
            grid = np.linspace(0, 1, 80)
            vals = np.max(a[:, None] * grid[None, :] + b[:, None], axis=0) * dx
            best_t = grid[np.argmin(vals)]
            if abs(best_t - h_save) < 1e-6:
                continue
            h[j] = best_t
            h = _repair(h, n)
            c5_new = _c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                h = _repair(np.concatenate([h[:j], [h_save], h[j+1:]]), n)
        if not improved:
            break
    return h, best_c5

def _pair_local(h, n, dx, rng, deadline, n_tries=80000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    per = max(400, n_tries // len(scales))
    for scale in scales:
        improved = False
        for _ in range(per):
            if time.time() >= deadline:
                return h, best_c5
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i], h[j] = ni, nj
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not improved and scale < 0.003:
            break
    return h, best_c5

def _slsqp_polish(h, n, dx, alpha, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}
    try:
        res = minimize(lambda x: _smooth_c5_grad(x, alpha)[0], h,
                       jac=lambda x: _smooth_c5_grad(x, alpha)[1],
                       method='SLSQP', bounds=bounds, constraints=[con],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)
