        h[j] = 1.0
        c5_new = _c5(h, dx)
        d = c5_new - c5_cur
        accept = (d < 0) or (T > 1e-10 and rng.random() < np.exp(min(0, -d / T)))
        if accept:
            c5_cur = c5_new
            ones_list.remove(i)
            zeros_list.remove(j)
            ones_list.append(j)
            zeros_list.append(i)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return h_best, c5_best

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _to_binary(h, n):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    target = n // 2
    s = int(h.sum())
    if s != target:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if s < target:
            for k in range(target - s):
                h[zeros[k]] = 1.0
                h[ones[-1-k]] = 0.0
        else:
            for k in range(s - target):
                h[ones[k]] = 0.0
                h[zeros[-1-k]] = 1.0
    return h

def _sa_binary(h, n, dx, rng, deadline, T0, T_min, cool, two_swap_frac=0.2):
    h = h.copy()
    c5_cur = _c5(h, dx)
    c5_best = c5_cur
    h_best = h.copy()
    T = T0
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    steps = 0
    while time.time() < deadline and T > T_min:
        if steps % 5000 == 0 and steps > 0 and rng.random() < two_swap_frac:
            # 2-swap move
            if len(ones_list) >= 2 and len(zeros_list) >= 2:
                i1 = ones_list[rng.randint(len(ones_list))]
                i2 = ones_list[rng.randint(len(ones_list)-1)]
                if i1 == i2:
                    i2 = ones_list[(ones_list.index(i1)+1) % len(ones_list)]
                j1 = zeros_list[rng.randint(len(zeros_list))]
                j2 = zeros_list[rng.randint(len(zeros_list)-1)]
                if j1 == j2:
                    j2 = zeros_list[(zeros_list.index(j1)+1) % len(zeros_list)]
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                c5_new = _c5(h, dx)
                d = c5_new - c5_cur
                if d < 0 or (T > 1e-10 and rng.random() < min(1.0, np.exp(-d/T))):
                    c5_cur = c5_new
                    ones_list.remove(i1); ones_list.remove(i2)
                    zeros_list.remove(j1); zeros_list.remove(j2)
                    ones_list.extend([j1, j2]); zeros_list.extend([i1, i2])
                    if c5_new < c5_best:
                        c5_best = c5_new; h_best = h.copy()
                else:
                    h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
        else:
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            h[i] = 0.0; h[j] = 1.0
            c5_new = _c5(h, dx)
            d = c5_new - c5_cur
            if d < 0 or (T > 1e-10 and rng.random() < min(1.0, np.exp(-d/T))):
                c5_cur = c5_new
                ones_list.remove(i); zeros_list.remove(j)
                ones_list.append(j); zeros_list.append(i)
                if c5_new < c5_best:
                    c5_best = c5_new; h_best = h.copy()
            else:
                h[i] = 1.0; h[j] = 0.0
        T *= cool
        steps += 1
    return h_best, c5_best

def _cont_refine(h, n, dx, rng, deadline):
    h = h.copy()
    best_c5 = _c5(h, dx)
    for scale in [0.08, 0.04, 0.02, 0.008, 0.003]:
        improved = False
        n_iter = 30000
        for _ in range(n_iter):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n); j = rng.randint(n)
            if i == j: continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i]+d, h[j]-d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
            h[i], h[j] = ni, nj
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5; improved = True
            else:
                h[i], h[j] = ni-d, nj+d
        if not improved: break
    return h, best_c5

def _slsqp_polish(h, n, dx, deadline):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': lambda x: float(np.sum(x) - target)}
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(corr))
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=[constraint],
                       options={'maxiter': 200, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)

    t_start = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t_start)

    h0 = np.array(initial_h_values, dtype=float).copy()

    best_dx = dx0

    c5_b = _c5(h_bin, dx0)

    # Phase 1: Main binary SA (350s)
    t_end = t_start + 350

    if c5_s < best_c5:
        best_h, best_c5 = h_s, c5_s

    # Phase 2: SA restarts from best (200s)
    t_end = t_start + 550
    for r in range(12):

        h_r = best_h.copy()
        k = rng.randint(3, 8)

            zeros = np.where(h_r == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                a = ones[rng.randint(len(ones))]; b = zeros[rng.randint(len(zeros))]
                h_r[a], h_r[b] = h_r[b], h_r[a]
        t_sub = min(time.time() + 12, t_end)

        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 3: Try smaller n values
    for n_try in [36, 40, 48, 56]:
        if rem() < 150: break
        dx_t = 2.0 / n_try
        tgt = n_try // 2
        # Generate diverse starting patterns

                rng.shuffle(h_t)
                h_t[:tgt] = 1.0
            h_t = h_t.astype(float)
            h_t[h_t > 0.5] = 1.0; h_t[h_t <= 0.5] = 0.0
            s = int(h_t.sum())
            if s != tgt:
                if s < tgt:
                    idx = np.where(h_t == 0)[0][:tgt-s]
                    h_t[idx] = 1.0
                    idx = np.where(h_t == 1)[0][-(tgt-s):]
                    h_t[idx] = 0.0
                else:

    return (best_h, best_c5, best_n)
