```python
import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _max_lag_grad(h, n):
    """Compute subgradient of max correlation at the maximizing lag"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)  # index in full output
    # k_star corresponds to lag = k_star - (n-1)
    lag = k_star - (n - 1)
    # Gradient of corr at this lag w.r.t. h
    grad = np.zeros(n)
    for i in range(n):
        # d/dh[i] sum_j h[j]*(1-h[j+lag]) = (1-h[i+lag]) - h[i-lag]
        g = 0.0
        if 0 <= i + lag < n:
            g += 1.0 - h[i + lag]
        if 0 <= i - lag < n:
            g -= h[i - lag]
        grad[i] = g
    return grad, int(corr[k_star] * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def _lbfgs_fast(h, n, dx, alpha, maxiter=200):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h.copy(), _compute_c5(h, dx)
    for pen in [50.0, 200.0, 1000.0]:
        try:
            res = minimize(
                _smooth_obj, best_h, args=(alpha, target, pen),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-6}
            )
            hc = _repair(res.x, n)
            c5 = _compute_c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
    return best_h, best_c5

def _subgrad_step(h, n, dx, lr, n_steps=5000):
    """Subgradient method on the active max lag"""
    h = h.copy()
    best_c5 = _compute_c5(h, dx)
    target = n / 2.0
    for step in range(n_steps):
        grad, _ = _max_lag_grad(h, n)
        # Projected gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # Repair sum constraint
        s = h_new.sum()
        if s > 1e-12:
            h_new = h_new * (target / s)
            h_new = np.clip(h_new, 0.0, 1.0)
        # Simple repair
        diff = target - h_new.sum()
        if abs(diff) > 1e-10:
            # Add/subtract to the element with most room
            if diff > 0:
                idx = np.argmax(1.0 - h_new)
                h_new[idx] = min(1.0, h_new[idx] + diff)
            else:
                idx = np.argmax(h_new)
                h_new[idx] = max(0.0, h_new[idx] + diff)
        c5 = _compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            h = h_new
        else:
            lr *= 0.95
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_lbfgs(x, alpha, target, pen):
    c = _corr(x)
    return _smooth_max(c, alpha) + pen * (x.sum() - target) ** 2

def _lbfgs(h, n, dx, alpha, maxiter=200):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h.copy(), _c5(h, dx)
    for pen in [50.0, 500.0, 2000.0]:
        try:
            res = minimize(_obj_lbfgs, best_h, args=(alpha, target, pen),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-6})
            hc = _repair(res.x, n)
            c = _c5(hc, dx)
            if c < best_c5:
                best_h, best_c5 = hc, c
        except Exception:
            pass
    return best_h, best_c5

def _slsqp(h, n, dx, alpha, maxiter=400):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: _smooth_max(_corr(x), alpha), h,
                      method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _local_search(h, n, dx, rng, deadline, n_tries=2000000):
    """Aggressive pairwise local search"""
    h = h.copy()
    best = _c5(h, dx)
    target = n / 2.0
    # Scale schedule
    scales = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    per = max(1000, n_tries // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() >= deadline:
                return h, best
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            hi, hj = h[i] + d, h[j] - d
            if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                continue
            h[i], h[j] = hi, hj
            c = _c5(h, dx)
            if c < best - 1e-14:
                best = c
            else:
                h[i], h[j] = hi - d, hj + d
    return h, best

def _block_search(h, n, dx, rng, deadline, n_iters=500000):
    """Block moves: add/subtract eps to contiguous blocks"""
    h = h.copy()
    best = _c5(h, dx)
    for _ in range(n_iters):
        if time.time() >= deadline:
            return h, best
        bs = rng.choice([2, 3, 4, 5])
        i = rng.randint(0, n - bs)
        j = rng.randint(0, n - bs)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.001, 0.01) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for d in range(bs):
            if 0 > h[i+d] + eps or h[i+d] + eps > 1:
                ok = False
            if 0 > h[j+d] - eps or h[j+d] - eps > 1:
                ok = False
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c = _c5(h, dx)
        if c < best - 1e-14:
            best = c
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best

def _anneal(h, n, dx, rng, deadline, T0=0.005, Tmin=1e-5, cool=0.9998):
    h = h.copy()
    best = _c5(h, dx)
    best_h = h.copy()
    T = T0
    while time.time() < deadline and T > Tmin:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.01 * (2.0 * rng.random() - 1.0)
        hi, hj = h[i] + d, h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i], h[j] = hi, hj
        c = _c5(h, dx)
        if c < best:
            best = c
            best_h = h.copy()
        if c <= _c5(best_h, dx) or rng.random() < np.exp(-(c - best) / T):
            # accept
            pass
        else:
            h[i], h[j] = hi - d, hj + d
        T *= cool
    return best_h, best

def _anneal_v2(h, n, dx, rng, deadline):
    """Better annealing: track current C5 to avoid recomputing"""
    h = h.copy()
    c_cur = _c5(h, dx)
    best = c_cur
    best_h = h.copy()
    T = 0.004
    Tmin = 5e-5
    cool = 0.99975
    while time.time() < deadline and T > Tmin:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.012 * (2.0 * rng.random() - 1.0)
        hi, hj = h[i] + d, h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i], h[j] = hi, hj
        c_new = _c5(h, dx)
        if c_new < c_cur:
            c_cur = c_new
            if c_new < best:
                best = c_new
                best_h = h.copy()
        elif rng.random() < np.exp(-(c_new - c_cur) / T):
            c_cur = c_new
        else:
            h[i], h[j] = h[i] - d, h[j] + d
        T *= cool
    return best_h, best

def _perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        a = rng.randint(n // 3, 2 * n // 3)
        hp = np.concatenate([hp[a:], hp[:a]])
    elif kind == 1:
        f = rng.randint(1, 6)
        ph = rng.uniform(0, 2 * np.pi)
        hp += 0.08 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.uniform(-0.04, 0.04, n)
    elif kind == 3:
        i, j = rng.randint(n), rng.randint(n)
        while i == j:
            j = rng.randint(n)
        hi, hj = hp[i], hp[j]
        hp[i] = (hi + hj) / 2 + 0.1
        hp[j] = (hi + hj) / 2 - 0.1
    else:
        block = rng.randint(5, 15)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _obj_smooth(x, alpha, target, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (x.sum() - target) ** 2

def _eq_con(x, target):
    return x.sum() - target

def _obj_sl(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lbfgs(h, n, dx, alpha, maxiter=250):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h.copy(), _c5(h, dx)
    for pen in [50.0, 500.0, 2000.0]:
        try:
            res = minimize(_obj_smooth, best_h, args=(alpha, target, pen),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-6})
            hc = _repair(res.x, n)
            c = _c5(hc, dx)
            if c < best_c5:
                best_h, best_c5 = hc, c
        except Exception:
            pass
    return best_h, best_c5

def _slsqp(h, n, dx, alpha, maxiter=400):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_sl, h, args=(alpha,), method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _block_search(h, n, dx, rng, deadline, n_iters=200000):

    best = _c5(h, dx)
    for _ in range(n_iters):
        if time.time() >= deadline:

        i = rng.randint(0, n - bs + 1)

            continue
        eps = rng.uniform(0.002, 0.012) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if h[i+di] + eps < -1e-10 or h[i+di] + eps > 1.0 + 1e-10:

            if h[j+di] - eps < -1e-10 or h[j+di] - eps > 1.0 + 1e-10:

        h[j:j+bs] -= eps

        if c < best - 1e-14:

        else:
            h[i:i+bs] -= eps

    h = h.copy()
    c_cur = _c5(h, dx)
    best = c_cur
    best_h = h.copy()

    while time.time() < deadline and T > Tmin:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        hi, hj = h[i] + d, h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:

        h[i], h[j] = hi, hj
        c_new = _c5(h, dx)
        if c_new < best:
            best = c_new
            best_h = h.copy()
        if c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / T):
            c_cur = c_new
        else:
            h[i], h[j] = hi - d, hj + d
        T *= cool
    return best_h, best

            if restart > 0:
                start = _perturb(best_h, n, rng, restart - 1)
            else:
                start = best_h.copy()
            ht, ct = _anneal(start, n, dx, rng, tEnd,
                            T0=0.005 - restart * 0.001, Tmin=2e-5, cool=0.9996)
            if ct < best_c5:
                best_h, best_c5 = ht, ct

    # Phase 6: Multi-restart gradient + local
    if _rem() > 150:
        for r in range(5):
            if _rem() < 100:
                break
            start = _perturb(best_h, n, rng, (r + 1) % 5)
            ht, ct = _lbfgs(start, n, dx, 800, maxiter=120)
            tEnd2 = time.time() + min(10, max(2, _rem() - 80))
            ht, ct = _multi_local(ht, n, dx, rng, tEnd2)
            if ct < best_c5:
                best_h, best_c5 = ht, ct

    # Phase 7: SLSQP polish
    if _rem() > 80:
        for alpha in [3000, 15000]:
            if _rem() < 40:
                break
            ht, ct = _slsqp(best_h, n, dx, alpha, maxiter=300)
            if ct < best_c5:
                best_h, best_c5 = ht, ct

    # Phase 8: Final ultra-fine local search
    if _rem() > 30:
        tEnd = time.time() + max(2, min(25, _rem() - 5))
        ht, ct = _multi_local(best_h, n, dx, rng, tEnd)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Final: tiny gradient polish
    if _rem() > 15:
        ht, ct = _grad_local(best_h, n, dx, rng, time.time() + max(2, min(10, _rem() - 3)),
                            n_iters=30000)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
