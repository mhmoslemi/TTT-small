from scipy.optimize import linprog, minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

    cv = _corr(h)
    cmax = cv.max()
    threshold = cmax * (1.0 - relax)
    active = np.where(cv >= threshold)[0]
    if len(active) > max_lags:
        active = np.argsort(cv)[::-1][:max_lags]
    if len(active) == 0:
        active = np.array([np.argmax(cv)])
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []

    for idx in active:
        k = idx - n1
        if k >= 0:
            nn = n - k
            if nn <= 0:
                continue
            coeff = np.zeros(n)
            coeff[:nn] += 1.0 - h[k:k+nn]  # 1 - h[m+k] for m in [0,nn-1]
            coeff[k:] -= h[:nn]  # -h[m-k] for m in [k, n-1]

            kp = -k
            nn = n - kp

            coeff[kp:] += 1.0 - h[:nn]  # 1 - h[m-kp] for m in [kp, n-1]
            coeff[:nn] -= h[kp:kp+nn]  # -h[m+kp] for m in [0, nn-1]
            const = np.dot(h[:nn], h[kp:])
        
        row = np.zeros(n + 1)
        row[:n] = coeff
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-const)

    b_ub = np.array(b_rows)

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options=opts)

            if damp < 1.0:

            else:
