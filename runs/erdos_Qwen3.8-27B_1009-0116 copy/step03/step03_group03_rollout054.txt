from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _grad_all(h, n):
    """Compute gradient matrix: G[m, i] = d corr[m] / d h[i]"""
    G = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            G[m, :n-k] += (1.0 - h[k:])
            G[m, k:] -= h[:n-k]
        else:
            j = -k
            G[m, j:] += (1.0 - h[:n-j])
            G[m, :n-j] -= h[j:]
    return G

def _pair_optimize(h, c, G, i, j, n):
    """Optimize the pair (h[i], h[j]) maintaining sum constraint.
    Find delta that minimizes max_m (c[m] + delta * (G[m,i] - G[m,j]))
    subject to 0 <= h[i]+delta <= 1 and 0 <= h[j]-delta <= 1
    """
    a = G[:, i] - G[:, j]  # coefficient for delta
    b = c.copy()  # constant term
    lo = max(-h[i], -(1.0 - h[j]))
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return 0.0
    
    # f(delta) = max_m (a[m]*delta + b[m]) - this is convex in delta
    # Use ternary search
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    delta = (lo + hi) / 2.0
    # Check if it's actually better than delta=0
    f0 = np.max(b)
    fd = np.max(a * delta + b)
    if fd < f0 - 1e-14:
        return delta
    return 0.0

def _repair(h, n):

            else:

            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:

def _grad_all(h, n):

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _grad_all(h, n):
    G = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            G[m, :n-k] += (1.0 - h[k:])
            G[m, k:] -= h[:n-k]
        else:
            j = -k
            G[m, j:] += (1.0 - h[:n-j])
            G[m, :n-j] -= h[j:]
    return G
