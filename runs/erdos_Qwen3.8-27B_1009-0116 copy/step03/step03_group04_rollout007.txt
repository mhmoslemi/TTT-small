        c5_cur = _c5(h, dx)
        deltas = np.array([0.003, 0.001, 0.0004, -0.003, -0.001, -0.0004])
        for _pass in range(40):
            if elapsed() > budget_s - 60:
                break
            imp = False
            for i in range(n):
                if elapsed() > budget_s - 60:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    hj = h[j]
                    for d in deltas:
                        ni, nj = hi + d, hj - d
                        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c5_cur - 1e-15:
                            c5_cur, hi, hj = c, ni, nj
                            imp = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
            if not imp:
                break

    # Phase 5: Ultra-fine random perturbations
    if elapsed() < budget_s - 15:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        while elapsed() < budget_s - 15:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.choice([5e-5, 1e-4, 2e-4, 5e-4, -5e-5, -1e-4, -2e-4, -5e-4])
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                else:
                    h[i] = ni - d
                    h[j] = nj + d

    # Phase 6: Final SLSQP polish
    if elapsed() < budget_s - 3:
        try:
            res = minimize(_obj_full, best_h, args=(300000,), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _eq_fun,
                                         'args': (target,)}],
                           options={'maxiter': 2000, 'ftol': 1e-16})
            hx = _repair(res.x, n)
            c5 = _c5(hx, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hx
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
