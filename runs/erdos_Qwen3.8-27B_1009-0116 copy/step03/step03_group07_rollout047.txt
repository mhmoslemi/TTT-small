def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def grad_at_lag(h, n, m):
    grad = np.zeros(n)
    if m >= 0:
        if m < n:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
    else:
        mm = -m
        if mm < n:
            grad[:n - mm] += (1.0 - h[mm:])
            grad[mm:] -= h[:n - mm]
    return grad

def subgrad_descent(h, n, t_end, rng, initial_step=0.008):

        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        topk = 5
        sidx = np.argsort(corr)[::-1][:topk]
        tau = max(0.005 * mx, 1e-8)
        wts = np.exp((corr[sidx] - mx) / tau)
        wts /= wts.sum()
        grad = np.zeros(n)
        for ii in range(topk):
            m = sidx[ii] - (n - 1)
            grad += wts[ii] * grad_at_lag(h, n, m)
        grad -= grad.mean()
        h_new = np.clip(h + step * grad, 0.0, 1.0)
        h_new = repair(h_new, n)

def coord_descent(h, n, n_rounds, t_end, rng):

    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for r in range(n_rounds):

            j = rng.randint(n)

                continue

            for d in steps:
                found = False
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hi + delta, hj - delta
                    if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:

                        found = True
                        break
