            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _grad_lag(h, n, m):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        g[:n+k] += (1.0 - h[-k:])
        g[-k:] -= h[:n+k]
    return g

def _slsqp_obj(x, alpha, target):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + 80.0 * (np.sum(x) - target)**2

def _subgrad_step(h, n, it):
    alpha = min(600000, 8000 * (1 + it // 800))
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for idx in active:
        g += w[idx] * _grad_lag(h, n, idx)
    g += 0.08 * (h.sum() - n / 2.0) / (n / 2.0) * np.ones(n)
    return g, m

def _pair_search(h, n, dx, max_iters, t_deadline):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    c5_cur = best_c5
    deltas = [0.0005, 0.002, 0.005, 0.01, 0.02, 0.04,
              -0.0005, -0.002, -0.005, -0.01, -0.02, -0.04]
    for _it in range(max_iters):
        if time.time() > t_deadline:
            break
        improve = False
        for i in range(n):
            if time.time() > t_deadline:
                break
            for j in range(i + 1, n):
                if time.time() > t_deadline:
                    break
                hi_base, hj_base = h[i], h[j]
                for d in deltas:
                    ni, nj = hi_base + d, hj_base - d
                    if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                        continue
                    h[i], h[j] = ni, nj
                    h = _repair(h, n)
                    c5_try = _c5(h, dx)
                    h[i], h[j] = hi_base, hj_base
                    h = _repair(h, n)
                    if c5_try < c5_cur - 1e-12:
                        h[i] = ni
                        h[j] = nj
                        h = _repair(h, n)
                        c5_cur = c5_try
                        improve = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        break
        if not improve:
            break
    return best_h, best_c5, h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def tleft():
        return budget_s - (time.time() - t0)

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP with increasing smoothness
    for alpha in [15000, 40000, 100000, 250000, 500000]:
        if tleft() < 220:
            break
        for restart in range(2):
            if tleft() < 200:
                break
            start = best_h.copy()
            if restart == 1:
                start += 0.008 * rng.randn(n)
                start = _repair(start, n)
            try:
                res = minimize(_slsqp_obj, start, args=(alpha, target),
                              method='SLSQP', bounds=bounds,
                              options={'maxiter': 600, 'ftol': 1e-18, 'iprint': 0})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
            except Exception:
                pass

    # Phase 2: Projected subgradient with momentum
    if tleft() > 250:
        h = best_h.copy()
        step = 0.04
        v = np.zeros(n)
        mom = 0.82
        for it in range(60000):
            if tleft() < 250:
                break
            g, m = _subgrad_step(h, n, it)
            g_n = np.linalg.norm(g)
            if g_n < 1e-10:
                break
            g = g / g_n
            v = mom * v - step * g
            h_new = _repair(h + v, n)
            c5_new = _c5(h_new, dx)
            c5_old = _c5(h, dx)
            if c5_new <= c5_old * 1.003 + 1e-8:
                h = h_new
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, h.copy()
                step = min(0.06, step * 1.01)
            else:
                v *= 0.4
                step *= 0.85
            if it % 3000 == 0:
                mom = min(0.92, mom * 1.005)

    # Phase 3: Thorough pair search (multiple sweeps)
    if tleft() > 120:
        h = best_h.copy()
        best_h, best_c5, h = _pair_search(h, n, dx, 50, t0 + budget_s - 120)

    # Phase 4: Simulated annealing on pairs
    if tleft() > 80:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        temp = 0.002
        for it in range(100000):
            if tleft() < 80:
                break
            temp = max(0.0001, 0.002 * 0.9995**it)
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d = rng.uniform(-0.02, 0.02)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            h = _repair(h, n)
            c5_try = _c5(h, dx)
            delta_c = c5_try - c5_cur
            if delta_c < 0 or rng.random() < np.exp(-delta_c / temp):
                c5_cur = c5_try
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:
                h[i] -= d
                h[j] += d
                h = _repair(h, n)

    # Phase 5: Final SLSQP polish
    if tleft() > 40:
        for _r in range(5):
            if tleft() < 45:
                break
            start = best_h.copy()
            if _r > 0:
                start += 0.005 * rng.randn(n)
                start = _repair(start, n)
            try:
                res = minimize(_slsqp_obj, start, args=(500000, target),
                              method='SLSQP', bounds=bounds,
                              options={'maxiter': 800, 'ftol': 1e-18, 'iprint': 0})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
            except Exception:
                pass

    # Phase 6: Ultra-fine local search
    if tleft() > 20:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        fine_deltas = [0.0002, 0.0005, 0.001, 0.002, -0.0002, -0.0005, -0.001, -0.002]
        for _ in range(200000):
            if tleft() < 20:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            best_d = 0
            best_c = c5_cur
            for d in fine_deltas:
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                h = _repair(h, n)
                c5_try = _c5(h, dx)
                h[i], h[j] = ni - d, nj + d
                h = _repair(h, n)
                if c5_try < best_c:
                    best_c = c5_try
                    best_d = d
            if best_d != 0:
                h[i] += best_d
                h[j] -= best_d
                h = _repair(h, n)
                c5_cur = best_c
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
