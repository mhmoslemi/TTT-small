import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

class _ObjFunc:
    def __init__(self, alpha, target):
        self.alpha = alpha
        self.target = target
    def __call__(self, x):
        corr = _corr(x)
        v = _smooth_max(corr, self.alpha)
        v += 80.0 * (np.sum(x) - self.target) ** 2
        return v

def _lbfgs(h, n, dx, alpha, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_ObjFunc(alpha, target), h, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _binary_sa(h, n, dx, rng, deadline, T_init=0.8, T_min=0.001, cool=0.99999):
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n // 2
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    if len(ones) > target:
        excess = list(ones)[len(ones)-target:]
        for i in excess:
            h[i] = 0; ones.discard(i); zeros.add(i)
    elif len(ones) < target:
        needed = target - len(ones)
        for i in list(zeros)[:needed]:
            h[i] = 1; zeros.discard(i); ones.add(i)
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    T = T_init
    no_improve = 0
    stall_limit = 80000
    while time.time() < deadline and T > T_min:
        o_list = list(ones)
        z_list = list(zeros)
        if len(o_list) < 2 or len(z_list) < 2:
            break
        if no_improve > stall_limit:
            n_swap = rng.randint(3, 8)
            oi = rng.choice(o_list, min(n_swap, len(o_list)), replace=False)
            zi = rng.choice(z_list, min(n_swap, len(z_list), len(oi)), replace=False)
            h[oi] = 0; h[zi] = 1
            for i in oi: ones.discard(i); zeros.add(i)
            for j in zi: zeros.discard(j); ones.add(j)
            c5_k = _c5(h, dx)
            if c5_k > best_c5 + 0.05:
                h[oi] = 1; h[zi] = 0
                for i in oi: zeros.discard(i); ones.add(i)
                for j in zi: ones.discard(j); zeros.add(j)
            no_improve = 0
            T = max(T, T_init * 0.4)
            continue
        i = o_list[rng.randint(len(o_list))]
        j = z_list[rng.randint(len(z_list))]
        h[i], h[j] = 0.0, 1.0
        c5_new = _c5(h, dx)
        delta = c5_new - best_c5
        if c5_new < best_c5:
            best_c5 = c5_new; best_h = h.copy(); no_improve = 0
        elif rng.random() < np.exp(-max(0.0, delta) / T):
            no_improve += 1
        else:
            h[i], h[j] = 1.0, 0.0
            no_improve += 1
        ones.add(j); ones.discard(i); zeros.add(i); zeros.discard(j)
        T *= cool
    return best_h, best_c5

def _fine_tune(h, n, dx, rng, deadline):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for scale in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
        improved = False
        iters = 0
        limit = 8000
        while time.time() < deadline and iters < limit:

            i = rng.randint(n)
            j = rng.randint(n)

            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d

            h[i], h[j] = ni, nj

            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                h[i], h[j] = oi, oj
        if not improved:
            break
    return h, best_c5
