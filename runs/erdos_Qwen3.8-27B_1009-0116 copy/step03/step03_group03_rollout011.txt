def _repair(h, n):

        else:

            diff -= add

        g[:n - k] -= h[k:]
    else:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-10:
                break
    return h

def _grad_single(h, n, m_idx):
    k = m_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += (1.0 - h[-k:])
        g[-k:] -= h[:n + k]
    return g

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sum_con(x, target):
    return np.sum(x) - target

def run(seed=42, budget_s=1100, **kwargs):

    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

                options={'maxiter': 600, 'ftol': 1e-15, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:

            h = _repair(h, n)

                c5 = _c5(hc, dx)
                if c5 < best_c5:

            except Exception:
                pass

        c_val = _c5(h, dx)
        c_arr = _corr(h)
        max_c = float(c_arr.max())
        active = np.argsort(c_arr)[-3:]
        all_grads = {m: _grad_single(h, n, m) for m in active}
