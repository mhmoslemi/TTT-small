            diff -= add

    """Coefficient of h[i] in each c_m."""
    n_corr = 2 * n - 1

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _coord_coeff(h, i, n):
    nc = 2 * n - 1
    a = np.zeros(nc)
    for m in range(nc):
        k = m - (n - 1)
        if k >= 0:
            if k <= i:
                a[m] += (1.0 - h[i - k])
            if i + k <= n - 1:
                a[m] -= h[i + k]
        else:
            if i >= -k:
                a[m] += (1.0 - h[i + k])
            if i <= n - 1 + k:
                a[m] -= h[i - k]
    return a

def _coord_opt(c_arr, a, h_i):
    b = c_arr - a * h_i
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a * m1 + b) < np.max(a * m2 + b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _grad_active(h, n, m_list):
    g = np.zeros(n)
    for m in m_list:
        k = m - (n - 1)
        if k >= 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        else:
            g[:n+k] += (1.0 - h[-k:])
            g[-k:] -= h[:n+k]
    return g / len(m_list)

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _coord_descent(h, n, c_arr, max_sweeps, rng, stop_fn, best_c5_ref):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = c_arr.max() * dx
    no_imp = 0
    for sw in range(max_sweeps):
        if stop_fn():
            break
        order = list(range(n))
        rng.shuffle(order)
        imp = False
        for i in order:
            a = _coord_coeff(h, i, n)
            t_star = _coord_opt(c_arr, a, h[i])
            d = t_star - h[i]
            if abs(d) > 1e-11:
                c_arr += a * d
                h[i] = t_star
                imp = True
        if imp:
            no_imp = 0
            c5_new = c_arr.max() * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:

            if no_imp >= 8:

                    h = _repair(h, n)
                    c_arr = _corr(h)
                    no_imp = 0
            if no_imp >= 30:
                break
    return h, best_h, best_c5
