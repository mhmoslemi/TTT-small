def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, n):
    return float(np.max(_corr_full(h))) * (2.0 / n)

def _project(p, n):
    target = n * 0.5
    lo = p.min() - 1.0
    hi = p.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(p - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_lag(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        kp = -k
        g[:n - kp] += (1.0 - h[kp:])
        g[kp:] -= h[:n - kp]
    return g

def _sum_eq(x):
    return x.sum() - x.size * 0.5

def _sum_jac(x):
    return np.ones(x.size)

def _lse_obj(x, n, alpha):
    c = _corr_full(x) * (2.0 / n)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -300.0, 300.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr_full(x) * dx_val
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -300.0, 300.0)
    e = np.exp(s)
    total = np.sum(e)
    weights = e / total
    g = np.zeros(n)
    for idx in range(len(c)):
        w = weights[idx]
        if w < 1e-14:
            continue
        g += w * _grad_lag(x, n, idx - (n - 1))
    return g * dx_val

def _pg_optimize(h, n, end_time, rng, top_k, lr_init, momentum, noise_period):
    best_c5 = _c5_val(h, n)
    h_cur = h.copy()
    v = np.zeros(n)
    lr = lr_init
    it = 0
    stagnation = 0
    while time.time() < end_time:
        c = _corr_full(h_cur)
        c5_now = float(np.max(c)) * (2.0 / n)
        if c5_now < best_c5:
            best_c5 = c5_now
            stagnation = 0
        else:
            stagnation += 1

        top_idx = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for idx in top_idx:
            g += _grad_lag(h_cur, n, idx - (n - 1))
        g /= float(top_k)

        v = momentum * v - lr * g
        h_new = _project(h_cur + v, n)
        if np.allclose(h_new, h_cur, atol=1e-15):
            lr *= 0.5
            v *= 0.5
            continue

        c5_new = _c5_val(h_new, n)
        if c5_new <= c5_now:
            h_cur = h_new
            lr = min(lr * 1.02, 0.4)
        else:
            lr *= 0.6
            v *= 0.3

        it += 1
        if it % noise_period == 0 and (lr < 0.02 or stagnation > 200):
            scale = 0.005 + 0.005 * (stagnation / 100.0)
            h_cur = _project(h_cur + min(scale, 0.05) * rng.randn(n), n)

    c5_final = _c5_val(h_cur, n)
    if c5_final < best_c5:
        best_c5 = c5_final
        return h_cur, best_c5
    return h_cur, best_c5

def _pair_search(h, n, end_time, rng):
    best_c5 = _c5_val(h, n)
    h_cur = h.copy()
    deltas = np.array([0.004, 0.002, 0.001, 0.0005, -0.004, -0.002, -0.001, -0.0005])
    no_improve = 0
    while time.time() < end_time:
        c = _corr_full(h_cur)
        top_idx = np.argsort(c)[::-1][:3]
        g = np.zeros(n)
        for idx in top_idx:
            g += _grad_lag(h_cur, n, idx - (n - 1))
        g /= 3.0

        top_g = np.argsort(np.abs(g))[::-1][:min(12, n)]
        improved = False
        for _ in range(100):
            if time.time() > end_time:
                return h_cur, best_c5
            i = top_g[rng.randint(len(top_g))]
            j = rng.randint(n)
            if i == j:
                continue
            d = deltas[rng.randint(len(deltas))]
            ni = h_cur[i] + d
            nj = h_cur[j] - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            old_hi, old_hj = h_cur[i], h_cur[j]
            h_cur[i], h_cur[j] = ni, nj
            c5_new = _c5_val(h_cur, n)
            if c5_new < best_c5 - 1e-16:
                best_c5 = c5_new
                improved = True
                no_improve = 0
            else:
                h_cur[i], h_cur[j] = old_hi, old_hj
        if not improved:
            no_improve += 1
            if no_improve > 50:
                scale = 0.002 * np.sqrt(no_improve / 50.0)
                h_cur = _project(h_cur + min(scale, 0.03) * rng.randn(n), n)
            else:
                h_cur = _project(h_cur + 0.003 * rng.randn(n), n)
    return h_cur, best_c5

def _interp_to(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h_old, h_old])
    x_ext = np.concatenate([x_old, x_old + 2.0])
    return np.interp(x_new, x_ext, h_ext)

def _coord_descent(h, n, end_time, rng):
    best_c5 = _c5_val(h, n)
    h_cur = h.copy()
    no_improve = 0
    while time.time() < end_time:
        c = _corr_full(h_cur)
        improved = False
        for j in range(n):
            if time.time() > end_time:
                return h_cur, best_c5
            # Compute A_k for all k
            A = np.zeros(2 * n - 1)
            for idx in range(2 * n - 1):
                lag = idx - (n - 1)
                A[idx] = _grad_lag(h_cur, n, lag)[j]
            B = c - A * h_cur[j]
            
            # Find breakpoints
            candidates = [0.0, 1.0]
            for idx1 in range(2 * n - 1):
                for idx2 in range(idx1 + 1, 2 * n - 1):
                    denom = A[idx1] - A[idx2]

                        if 0.0 <= h_star <= 1.0:

    """A_k for all k = -(n-1) to (n-1), at coordinate j."""
    A = np.zeros(2 * n - 1)
    # k >= 0: k = 0 to n-1, idx = n-1+k

        idx = n - 1 + k
        if j >= k:
            A[idx] += 1.0 - h[j - k]
        if j + k < n:
            A[idx] -= h[j + k]

        if j + kp < n:
            A[idx] += 1.0 - h[j + kp]
        if j >= kp:
            A[idx] -= h[j - kp]
    return A

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, n):
    return float(np.max(_corr_full(h))) * (2.0 / n)

def _project(p, n):
    target = n * 0.5
    lo = p.min() - 1.0
    hi = p.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(p - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_lag(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        kp = -k
        g[:n - kp] += (1.0 - h[kp:])
        g[kp:] -= h[:n - kp]
    return g

def _sum_eq(x):
    return x.sum() - x.size * 0.5

def _sum_jac(x):
    return np.ones(x.size)

def _lse_obj(x, n, alpha):
    c = _corr_full(x) * (2.0 / n)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -300.0, 300.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr_full(x) * dx_val
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -300.0, 300.0)
    e = np.exp(s)
    total = np.sum(e)
    weights = e / total
    g = np.zeros(x.size)
    for idx in range(len(c)):
        w = weights[idx]
        if w < 1e-14:
            continue
        g += w * _grad_lag(x, x.size, idx - (x.size - 1))
    return g * dx_val

def _pg_opt(h, n, end_t, rng, top_k, lr0, mom, nper):
    bc5 = _c5_val(h, n)
    hc = h.copy()
    v = np.zeros(n)
    lr = lr0
    it = 0
    stg = 0
    while time.time() < end_t:
        c = _corr_full(hc)
        c5n = float(np.max(c)) * (2.0 / n)
        if c5n < bc5:
            bc5 = c5n
            stg = 0
        else:
            stg += 1
        ti = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for idx in ti:
            g += _grad_lag(hc, n, idx - (n - 1))
        g /= float(top_k)
        v = mom * v - lr * g
        hn = _project(hc + v, n)
        c5n2 = _c5_val(hn, n)
        if c5n2 <= c5n + 1e-15:
            hc = hn
            lr = min(lr * 1.01, 0.4)
        else:
            lr *= 0.65
            v *= 0.4
        it += 1

            hc = _project(hc + sc * rng.randn(n), n)
    c5f = _c5_val(hc, n)
    if c5f < bc5:

    return hc, bc5

    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, n):
    return float(np.max(_corr_full(h))) * (2.0 / n)

def _project(p, n):
    target = n * 0.5
    lo = p.min() - 1.0
    hi = p.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(p - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid

def _grad_lag(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        kp = -k
        g[:n - kp] += 1.0 - h[kp:]
        g[kp:] -= h[:n - kp]
    return g

def _sum_eq(x):
    return x.sum() - x.size * 0.5

def _sum_jac(x):
    return np.ones(x.size)

def _lse_obj(x, n, alpha):
    c = _corr_full(x) * (2.0 / n)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -300.0, 300.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, n, alpha):
    dxv = 2.0 / n
    c = _corr_full(x) * dxv
    m = float(np.max(c))
    w = np.exp(np.clip(alpha * (c - m), -300.0, 300.0))
    w /= w.sum()
    g = np.zeros(n)
    for i in range(len(c)):
        if w[i] < 1e-14:
            continue
        g += w[i] * _grad_lag(x, n, i - (n - 1))
    return g * dxv

    c = _corr_full(x) * (2.0 / n)
    m = float(np.max(c))
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -300.0, 300.0)))) / alpha

def _lse_grad(x, n, alpha):
    dxv = 2.0 / n
    c = _corr_full(x) * dxv
    m = float(np.max(c))
    w = np.exp(np.clip(alpha * (c - m), -300.0, 300.0))
    w /= w.sum()
    g = np.zeros(n)
    for i in range(len(c)):
        if w[i] >= 1e-14:
            g += w[i] * _grad_lag(x, n, i - (n - 1))
    return g * dxv

def _pg_opt(h, n, et, rng, tk, lr0, mom, npr):
    bc = _c5_val(h, n)
    hc, v, lr = h.copy(), np.zeros(n), lr0
    it = stg = 0
    while time.time() < et:
        c = _corr_full(hc)
        c5n = float(np.max(c)) * (2.0 / n)
        if c5n < bc:
            bc, stg = c5n, 0
        else:
            stg += 1
        g = np.zeros(n)
        for idx in np.argsort(c)[::-1][:tk]:
            g += _grad_lag(hc, n, idx - (n - 1))
        g /= tk
        v = mom * v - lr * g
        hn = _project(hc + v, n)
        if _c5_val(hn, n) <= c5n + 1e-15:
            hc, lr = hn, min(lr * 1.01, 0.4)
        else:
            lr, v *= 0.65, 0.4
        it += 1
        if it % npr == 0 and (lr < 0.015 or stg > 150):
            hc = _project(hc + min(0.005 + 4e-4 * stg, 0.04) * rng.randn(n), n)
    cf = _c5_val(hc, n)
    return hc, (cf if cf < bc else bc)

def _A_at_j(h, n, j):
    A = np.zeros(2 * n - 1)
    for k in range(n):
        if j >= k:
            A[n - 1 + k] += 1.0 - h[j - k]
        if j + k < n:
            A[n - 1 + k] -= h[j + k]
    for kp in range(1, n):
        if j + kp < n:
            A[n - 1 - kp] += 1.0 - h[j + kp]
        if j >= kp:
            A[n - 1 - kp] -= h[j - kp]
    return A

def _coord_desc(h, n, et, rng, rsts=3):
    bc, hc = _c5_val(h, n), h.copy()
    for _ in range(rsts):
        ni = 0
        while time.time() < et and ni < 15:
            imp, c = False, _corr_full(hc)
            for j in range(n):
                A = _A_at_j(hc, n, j)
                B = c - A * hc[j]
                dM = A[:, None] - A[None, :]
                nM = B[None, :] - B[:, None]
                mk = np.abs(dM) > 1e-10
                hs = np.full_like(dM, np.nan)
                hs[mk] = nM[mk] / dM[mk]
                hs = hs.ravel()
                hs = hs[np.isfinite(hs) & (hs >= 0.0) & (hs <= 1.0)]
                cands = np.concatenate([[0.0, 1.0, hc[j]], hs])
                vals = np.max(A[None, :] * cands[:, None] + B[None, :], axis=1)
                bj = float(cands[np.argmin(vals)])
                if abs(bj - hc[j]) > 1e-8:
                    hc[j] = bj
                    imp = True
                    c = _corr_full(hc)
            c5n = _c5_val(hc, n)
            if c5n < bc:
                bc, ni = c5n, 0
            elif not imp:
                ni += 1
                hc = _project(hc + 0.003 * rng.randn(n), n)
    return hc, bc

def _pair_search(h, n, et, rng):
    bc, hc = _c5_val(h, n), h.copy()
    dl = np.array([3e-3, 1e-3, 5e-4, 2e-3, -3e-3, -1e-3, -5e-4, -2e-3])
    while time.time() < et:
        c = _corr_full(hc)
        ti = np.argsort(c)[::-1][:3]
        g = np.zeros(n)
        for idx in ti:
            g += _grad_lag(hc, n, idx - (n - 1))
        tg = np.argsort(np.abs(g))[::-1][:min(10, n)]
        imp = False
        for _ in range(60):
            if time.time() > et:
                break
            i, j = tg[rng.randint(len(tg))], rng.randint(n)
            if i == j:
                continue
            d = dl[rng.randint(len(dl))]
            ni, nj = hc[i] + d, hc[j] - d
            if not (0 <= ni <= 1 and 0 <= nj <= 1):
                continue
            oi, oj = hc[i], hc[j]
            hc[i], hc[j] = ni, nj
            c5n = _c5_val(hc, n)
            if c5n < bc - 1e-16:
                bc = c5n
                imp = True
            else:
                hc[i], hc[j] = oi, oj
        if not imp:
            hc = _project(hc + 0.002 * rng.randn(n), n)
    return hc, bc

def _interp(h, no, nn):

    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    return np.interp(xn, np.concatenate([xo, xo + 2.0]), np.concatenate([h, h]))
