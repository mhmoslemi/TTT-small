import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _grad_row(h, n, m):
    g = np.zeros(n)
    k = m - (n - 1)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += (1.0 - h[-k:])
        g[-k:] -= h[:n + k]
    return g

def _grad_matrix(h, n):
    return np.array([_grad_row(h, n, m) for m in range(2 * n - 1)])

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _sum_con(x, target):
    return float(np.sum(x) - target)

def _best_pair(h, g, n, top_k=15):
    """Find (i,j) maximizing g[j]-g[i] subject to feasibility. Returns (i,j,diff) or None."""
    valid_src = np.where(h > 1e-10)[0]
    valid_dst = np.where(h < 1.0 - 1e-10)[0]
    if len(valid_src) == 0 or len(valid_dst) == 0:
        return None
    gs = g[valid_src]
    gd = g[valid_dst]
    top_s = valid_src[np.argsort(gs)[::-1][:top_k]]
    top_d = valid_dst[np.argsort(gd)[:top_k]]
    best_diff = -1e12
    best_i, best_j = -1, -1
    for j in top_s:
        for i in top_d:
            if i == j:
                continue
            d = g[j] - g[i]
            if d > best_diff:
                best_diff = d
                best_i, best_j = i, j
    if best_i < 0 or best_diff <= 1e-12:
        return None
    return best_i, best_j, best_diff

def _line_search(h, i, j, d_max, n, dx, cur_max, n_evals=7):
    fracs = np.array([1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04])
    best_d = 0.0
    best_mc = cur_max
    for f in fracs[:n_evals]:
        d = d_max * f
        if d < 1e-10:
            break
        h[i] += d
        h[j] -= d
        nm = float(np.max(_corr(h)))
        if nm < best_mc:
            best_mc = nm
            best_d = d
        h[i] -= d
        h[j] += d
    return best_d, best_mc
