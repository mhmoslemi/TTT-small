def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _pair_grad(h, n, i, j):
    a = np.zeros(2 * n - 1)
    for pos in range(2 * n - 1):
        s = pos - (n - 1)
        if 0 <= i + s < n:
            a[pos] -= h[i + s]
        if 0 <= i - s < n:
            a[pos] -= h[i - s]
        if 0 <= j + s < n:
            a[pos] += h[j + s]
        if 0 <= j - s < n:
            a[pos] += h[j - s]
    return a

def _smooth_grad(h, n, alpha):
    """Gradient of smooth max of correlations."""
    corr = _corr_full(h)
    s_shifts = np.arange(2 * n - 1) - (n - 1)
    # Gradient of each corr_s w.r.t. h_j: 1 - h[j+s] - h[j-s]
    # Shape: (m, n) where m = 2n-1, n = len(h)
    grad_matrix = np.ones((2 * n - 1, n))
    for jj in range(n):
        for pos in range(2 * n - 1):
            s = s_shifts[pos]
            if 0 <= jj + s < n:
                grad_matrix[pos, jj] -= h[jj + s]
            if 0 <= jj - s < n:
                grad_matrix[pos, jj] -= h[jj - s]
    
    # Softmax weights
    shifted = corr - np.max(corr)
    weights = np.exp(alpha * shifted)
    weights /= weights.sum()
    
    # Gradient = weighted sum of individual gradients
    return weights @ grad_matrix  # shape (n,)

def _project(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}."""
    return _repair(h, n)

def _grad_descent(h, n, dx, deadline, n_steps=5000):
    """Projected gradient descent on smooth max."""
    h = h.copy()
    best_c5 = _compute_c5(h, dx)
    best_h = h.copy()
    
    alpha = 1000.0
    eta = 0.01
    
    for step in range(n_steps):
        if time.time() >= deadline:
            break
        
        g = _smooth_grad(h, n, alpha)
        h_new = h - eta * g
        h_new = _project(h_new, n)
        
        c5 = _compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            #Anneal: slightly increase step
            eta = min(eta * 1.01, 0.05)
        else:
            h = best_h.copy()
            eta *= 0.8
            alpha *= 1.2
    
    return best_h, best_c5

from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_obj(x, n, alpha, target):
    corr = _corr_full(x)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + 100.0 * (np.sum(x) - target) ** 2

def _lbfgs_opt(h, n, dx, alpha=500, maxiter=200):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: _smooth_obj(x, n, alpha, target), h,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except:
        return h, _compute_c5(h, dx)

def _slsqp_opt(h, n, dx, alpha=3000, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]

        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except:
        return h, _compute_c5(h, dx)

def _pair_round(h, n, dx, deadline, rng):
    """One full round: find best pair and optimal d, apply if improving."""
    h = h.copy()
    corr = _corr_full(h)
    max_corr = float(np.max(corr))
    m = len(corr)
    eps = 1e-9
    
    best_improve = 1e-14
    best_i, best_j, best_d = -1, -1, 0.0
    best_new_max = max_corr
    
    # Try all pairs
    for i in range(n):
        if deadline and time.time() >= deadline:
            break
        for j in range(i + 1, n):
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-12:
                continue
            
            # Numerical gradient for this pair
            h[i] += eps
            h[j] -= eps
            corr1 = _corr_full(h)
            h[i] -= eps
            h[j] += eps
            a = (corr1 - corr) / eps
            
            # Find optimal d
            c_diff = corr[None, :] - corr[:, None]
            a_diff = a[:, None] - a[None, :]
            mask = np.abs(a_diff) > 1e-12
            with np.errstate(divide='ignore', invalid='ignore'):
                all_br = np.where(mask, c_diff / a_diff, np.inf)
            flat = all_br.ravel()
            vm = np.isfinite(flat) & (flat > d_lo + 1e-10) & (flat < d_hi - 1e-10)
            cands = np.unique(np.concatenate(([d_lo, d_hi, 0.0], flat[vm])))
            
            vals = np.max(corr[None, :] + a[None, :] * cands[:, None], axis=1)
            idx = np.argmin(vals)
            improve = max_corr - vals[idx]
            
            if improve > best_improve:
                best_improve = improve
                best_i, best_j, best_d = i, j, cands[idx]
                best_new_max = float(vals[idx])

        h[best_i] += best_d
        h[best_j] -= best_d
        h = np.clip(h, 0.0, 1.0)
        # Verify sum constraint
        diff = h.sum() - n / 2.0
        if abs(diff) > 1e-10:
            h = _repair(h, n)
    
    return h, float(best_new_max * dx)

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_obj(x, n, alpha, pen):
    corr = _corr_full(x)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - n / 2.0) ** 2

def _lbfgs_opt(h, n, dx, alpha=500, maxiter=200, pen=100.0):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: _smooth_obj(x, n, alpha, pen), h,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

def _slsqp_opt(h, n, dx, alpha=3000, maxiter=250):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(lambda x: _smooth_obj(x, n, alpha, 0.0), h,
                       method='SLSQP', bounds=bounds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

def _pair_round(h, n, dx, deadline):
    h = h.copy()
    corr = _corr_full(h)
    max_corr = float(np.max(corr))
    m = len(corr)
    eps = 1e-9
    c_diff = corr[None, :] - corr[:, None]
    best_improve = 1e-14
    best_i, best_j, best_d = -1, -1, 0.0
    best_new_max = max_corr
    for i in range(n):
        if deadline and time.time() >= deadline:
            break
        for j in range(i + 1, n):
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-12:
                continue
            si, sj = h[i], h[j]
            h[i] = si + eps
            h[j] = sj - eps
            corr1 = _corr_full(h)
            h[i], h[j] = si, sj
            a = (corr1 - corr) / eps
            a_diff = a[:, None] - a[None, :]
            mask = np.abs(a_diff) > 1e-12
            with np.errstate(divide='ignore', invalid='ignore'):
                all_br = np.where(mask, c_diff / a_diff, np.inf)
            flat = all_br.ravel()
            vm = np.isfinite(flat) & (flat > d_lo + 1e-10) & (flat < d_hi - 1e-10)
            extra = flat[vm]
            cands = np.unique(np.concatenate(([d_lo, d_hi, 0.0], extra)))
            vals = np.max(corr[None, :] + a[None, :] * cands[:, None], axis=1)
            idx = int(np.argmin(vals))
            improve = max_corr - float(vals[idx])
            if improve > best_improve:
                best_improve = improve
                best_i, best_j, best_d = i, j, float(cands[idx])
                best_new_max = float(vals[idx])
    if best_improve > 1e-14:
        h[best_i] += best_d
        h[best_j] -= best_d
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - n / 2.0) > 1e-10:
            h = _repair(h, n)
    return h, float(best_new_max * dx)

def _pair_descent(h, n, dx, deadline, max_rounds=200):
    h = h.copy()
    c5 = _compute_c5(h, dx)
    no_improve = 0
    for _ in range(max_rounds):
        if time.time() >= deadline:
            break
        h_t, c5_t = _pair_round(h, n, dx, deadline)
        if c5_t < c5 - 1e-13:
            h, c5 = h_t, c5_t
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                break
    return h, c5

def _perturb_restart(h, n, rng, amp=0.03):
    hp = h + rng.normal(0, amp, n)
    return _repair(hp, n)
