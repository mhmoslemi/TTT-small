import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _sf(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sj(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        L = k_idx - (n - 1)
        i_min = max(0, -L)
        i_max = min(n - 1, n - 1 - L)
        if i_min > i_max:
            continue
        idx1 = np.arange(i_min, i_max + 1)
        idx2 = idx1 + L
        g[idx1] += wk * (1.0 - x[idx2])
        g[idx2] -= wk * x[idx1]
    return g

def _opt_slsqp(h, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    res = minimize(_sf, h, jac=_sj, args=(alpha,),
                   method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return _repair(res.x, n)

For minimizing max_k (a_k + d * b_k) over d in [d_lo, d_hi]:
- This is a convex function of d
- The minimum occurs where the subgradient contains 0
- The subgradient at d is the convex hull of {b_k : k active at d}
- Active at d: k such that a_k + d*b_k = max_j (a_j + d*b_j)

An exact approach:
- The breakpoints of the max function occur where a_k + d*b_k = a_m + d*b_m, i.e., d = (a_m - a_k)/(b_k - b_m)
- At each breakpoint, evaluate the max
- The minimum is at the breakpoint where the max is smallest, or at a boundary

For the top K lags (say K=10), there are at most K(K-1)/2 = 45 breakpoints. I can evaluate all of them.

                            best_c5, best_h = c5c, h.copy()
                    else:
                        h[i] -= d
                        h[j] += d
            if not imp:
                no_imp += 1
                if no_imp >= 8:
                    # Perturb to escape plateau
                    for _tr in range(3):
                        h[i0] = np.clip(h[int(rng.randint(n))] + 0.005 * rng.randn(), 0, 1)
                        h = _repair(h, n)
                        c5t = _eval(h)
                        if c5t < c5c:
                            c5c = c5t
                            c = np.correlate(h, 1.0 - h, mode='full')
                            best_c5, best_h = c5c, h.copy()
                            no_imp = 0
                            break
                    if no_imp >= 15:
                        break
            else:
                no_imp = 0
        if _el() > budget_s - 80:
            break

    # Phase 3: Multi-start basin hopping
    if _el() < budget_s - 60:
        for _r in range(8):
            if _el() > budget_s - 60:
                break
            sc = 0.005 + 0.015 * rng.rand()
            hp = _repair(best_h + sc * rng.randn(n), n)
            h2 = _opt_slsqp(hp, n, 200000, 1000)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h = c5, h2.copy()

    # Phase 4: Final SLSQP polish
    if _el() < budget_s - 20:
        h = _opt_slsqp(best_h, n, 500000, 2500)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 5: Ultra-fine random moves
    if _el() < budget_s - 5:
        h = best_h.copy()
        c5c = _eval(h)
        steps = [1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 1e-6]
        for sv in steps:
            if _el() > budget_s - 5:
                break
            for _it in range(20000):
                if _el() > budget_s - 5:
                    break
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = rng.choice([1.0, -1.0]) * sv
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-16:
                        c5c = c5n
                        if c5c < best_c5:
                            best_c5, best_h = c5c, h.copy()
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
