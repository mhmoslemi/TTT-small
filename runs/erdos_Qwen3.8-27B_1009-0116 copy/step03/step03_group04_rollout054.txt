def exact_pair_delta(c, g, delta_min, delta_max, top_k_idx, K_eval):
    # c: current correlation values (all lags)
    # g: gradient difference for this pair (all lags)
    # top_k_idx: indices of top lags to consider for intersection
    # K_eval: number of top lags to evaluate at each candidate
    
    if delta_max - delta_min < 1e-14:
        return 0.0
    
    # Evaluate at boundaries
    best_val = max(c[top_k_idx] + delta_min * g[top_k_idx])
    best_delta = delta_min
    val_max = max(c[top_k_idx] + delta_max * g[top_k_idx])
    if val_max < best_val:
        best_val = val_max
        best_delta = delta_max
    
    # Check intersections of top lags
    idx = top_k_idx
    for a in range(len(idx)):
        for b in range(a+1, len(idx)):
            ga, gb = g[idx[a]], g[idx[b]]
            if abs(ga - gb) < 1e-14:
                continue
            d_star = (c[idx[b]] - c[idx[a]]) / (ga - gb)
            d_star = max(delta_min, min(delta_max, d_star))
            val = max(c[top_k_idx] + d_star * g[top_k_idx])
            if val < best_val:
                best_val = val
                best_delta = d_star
    
    return best_delta

def _repair(h, n):

        return np.full(n, 0.5)

        if abs(diff) > 1e-12:

import numpy as np
from scipy.optimize import minimize
import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _esm_eq(x, target):
    return np.sum(x) - target


def _do_slsqp(x0, bounds, target, alpha, maxiter):
    res = minimize(_smooth_max, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _esm_eq, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x


def _compute_M(h, n):
    nm1 = n - 1
    M = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_lo = max(0, nm1 - i)
        k_hi = min(2 * n - 1, 2 * nm1 - i + 1)
        for k in range(k_lo, k_hi):
            q = i + k - nm1
            M[k, i] += (1.0 - h[q])
        k_lo2 = max(0, i - nm1)
        k_hi2 = min(2 * n - 1, i + 1)
        for k in range(k_lo2, k_hi2):
            p = i - k + nm1
            M[k, i] -= h[p]
    return M


def _exact_pair_delta(c, g, dmin, dmax, top_idx, n_top, c_all):
    if dmax - dmin < 1e-14:
        return 0.0
    top = top_idx[:n_top]
    ct = c_all[top]
    gt = g[top]
    best_val = float(np.max(ct + dmin * gt))
    best_d = dmin
    cand = float(np.max(ct + dmax * gt))
    if cand < best_val:
        best_val, best_d = cand, dmax
    nt = n_top
    for a in range(nt):
        for b in range(a + 1, nt):
            da, db = gt[a], gt[b]
            denom = da - db
            if abs(denom) < 1e-14:
                continue
            d_star = (ct[b] - ct[a]) / denom
            if d_star <= dmin or d_star >= dmax:
                continue
            val = float(np.max(c_all + d_star * g))
            if val < best_val:
                best_val, best_d = val, d_star
    return best_d


def _pair_sweeps(h, n, dx, n_sweeps, n_top, t0, budget_s, reserve):
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    for sw in range(n_sweeps):
        if time.time() > t0 + budget_s - reserve:
            break
        M = _compute_M(h, n)
        c = _corr(h)
        top_idx = np.argsort(c)[::-1]
        imp_total = 0.0
        order = np.random.permutation(n * (n - 1) // 2)
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        for pi in order:
            if time.time() > t0 + budget_s - reserve:
                break
            i, j = pairs[pi]
            dmin = max(-h[i], h[j] - 1.0)
            dmax = min(1.0 - h[i], h[j])
            if dmax - dmin < 1e-12:
                continue
            g = M[:, i] - M[:, j]
            delta = _exact_pair_delta(c, g, dmin, dmax, top_idx, n_top, c)
            if abs(delta) < 1e-14:
                continue
            h[i] += delta
            h[j] -= delta
            h = np.clip(h, 0.0, 1.0)
            c = _corr(h)
            c5_new = float(np.max(c)) * dx
            if c5_new < best_c5 - 1e-15:
                imp_total += best_c5 - c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                h[i] -= delta
                h[j] += delta
                c = _corr(h)
        if imp_total < 1e-12:
            break
    return best_h, best_c5


def _subgrad_phases(h, n, dx, t0, budget_s, reserve):
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    target = n / 2.0
    for ph in range(12):
        if time.time() > t0 + budget_s - reserve:
            break
        step = 0.08 * (0.5 ** ph)
        c5_cur = best_c5
        c = _corr(h)
        for it in range(300):
            if time.time() > t0 + budget_s - reserve:
                break
            top_k = np.argsort(c)[-5:]
            g = np.zeros(n)
            nm1 = n - 1
            for m in top_k:
                for i in range(n):
                    q = i + m - nm1
                    if 0 <= q < n:
                        g[i] += (1.0 - h[q])
                    p = i - m + nm1
                    if 0 <= p < n:
                        g[i] -= h[p]
            g /= len(top_k)
            h_new = _repair(h - step * g, n)
            c5_new = _c5(h_new, dx)
            if c5_new < c5_cur - 1e-15:
                h, c5_cur = h_new, c5_new
                c = _corr(h)
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:
                step *= 0.5
    return best_h, best_c5
