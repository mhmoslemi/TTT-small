        h01[zeros_idx[order]] = 1
    if h01.sum() != target:
        return None, np.inf
    curr = compute_c5(h01, n)
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        ones = np.where(h01 == 1)[0]
        zeros = np.where(h01 == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h01[i], h01[j] = 0, 1
        c5_new = compute_c5(h01, n)
        if c5_new < curr - 1e-15:
            curr = c5_new
        else:
            h01[i], h01[j] = 1, 0
    return h01, curr
