def _bimodal_snap(h, n, threshold=0.85):
    """Push values near 0 or 1 to exactly 0 or 1, redistribute mass."""
    h = h.copy()
    n_zeros = np.sum(h < (1.0 - threshold))  # values that would go to 0
    n_ones = np.sum(h > threshold)  # values that would go to 1
    # If we snap: new_sum = n_ones * 1 + n_zeros * 0 + sum(middle values)
    # We need total = n/2
    middle = [(i, h[i]) for i in range(n) if (1.0-threshold) <= h[i] <= threshold]
    if not middle:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _pair_golden(h, n, i, j):
    s = h[i] + h[j]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-12:
        return _c5(h, n)
    si, sj = h[i], h[j]
    gr = (np.sqrt(5) - 1) / 2
    lo, hi2 = t_lo, t_hi
    c1 = hi2 - gr * (hi2 - lo)
    c2 = lo + gr * (hi2 - lo)

    def ev(t):
        h[i] = t
        h[j] = s - t
        return float(np.max(_corr(h)))

    f1 = ev(c1)
    f2 = ev(c2)
    for _ in range(38):
        if f1 < f2:
            hi2 = c2; c2 = c1; f2 = f1
            c1 = hi2 - gr * (hi2 - lo); f1 = ev(c1)
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi2 - lo); f2 = ev(c2)
    t_opt = (lo + hi2) / 2
    h[i], h[j] = t_opt, s - t_opt
    c5_after = _c5(h, n)
    if c5_after > _c5(h, n) + 1e-15:
        h[i], h[j] = si, sj
    return c5_after

def _pair_sweep(h, n, max_passes=40, time_limit=None):
    c5_cur = _c5(h, n)
    for p in range(max_passes):
        if time_limit and time.time() > time_limit:
            break
        improved = False
        for i in range(n - 1):
            if time_limit and time.time() > time_limit:
                return h, _c5(h, n)
            for j in range(i + 1, n):
                s = h[i] + h[j]
                if s < 1e-12 or s - 1.0 < -1e-12:
                    continue
                c5_cur = _pair_golden(h, n, i, j)
                if p > 0 and not improved:
                    pass
        # Check convergence
        c5_check = _c5(h, n)
        if abs(c5_check - c5_cur) < 1e-15:
            break
        c5_cur = c5_check
    return h, _c5(h, n)

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    e = np.exp(s)
    total = np.sum(e)
    weights = e / total
    g = np.zeros(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += w * (1.0 - x[i])
            g[i] -= w * x[jp]
    return g * dx_val

def _slsqp_polish(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(
        lambda x: _smooth_obj(x, n, alpha), h,
        jac=lambda x: _smooth_grad(x, n, alpha),
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-15})
    return _repair(res.x, n)

def _triple_subopt(h, n, indices, alpha=15000):
    idx = list(indices)
    m = len(idx)
    if m < 2:
        return
    s = sum(h[i] for i in idx)
    bounds = [(0.0, 1.0)] * m
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - s,
             'jac': lambda x: np.ones(m)}]
    def obj(z):
        hh = h.copy()
        for k, ii in enumerate(idx):
            hh[ii] = z[k]
        return _smooth_obj(hh, n, alpha)
    def grd(z):
        hh = h.copy()
        for k, ii in enumerate(idx):
            hh[ii] = z[k]
        g = _smooth_grad(hh, n, alpha)
        return np.array([g[ii] for ii in idx])
    z0 = np.array([h[i] for i in idx])
    try:
        res = minimize(obj, z0, jac=grd, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 150, 'ftol': 1e-14})
        if res.x is not None:
            for k, ii in enumerate(idx):
                h[ii] = np.clip(res.x[k], 0, 1)
    except Exception:
        pass
