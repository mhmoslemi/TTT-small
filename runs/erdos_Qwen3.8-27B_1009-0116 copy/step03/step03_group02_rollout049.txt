import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_round(h, n, corr, time_limit, n_coarse=25, n_bisect=35):
    improved = False
    order = np.random.permutation(n)
    n_pairs = 0
    n_improved = 0
    for ai, i in enumerate(order):
        for j in order[ai + 1:]:
            n_pairs += 1
            if time.time() > time_limit:
                return h, corr, improved, n_pairs, n_improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            range_d = d_max - d_min
            if range_d < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            cur_max = np.max(corr)
            if vals[bi] >= cur_max - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
            n_improved += 1
    return h, corr, improved, n_pairs, n_improved

def _slsqp_step(h, n, dx, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgradient_step(h, n, target, lr):
    """One subgradient step for max corr."""
    corr = _corr(h)
    c_max = np.max(corr)
    # Subgradient: average of gradients of active (near-max) correlations
    active = np.where(corr > c_max - 0.01 * c_max)[0]
    n = len(h)
    grad = np.zeros(n)
    for k_idx in active:
        p = k_idx - (n - 1)
        idx1 = np.arange(max(0, p), min(n, n + p))
        if len(idx1) > 0:
            grad[idx1] += (1.0 - h[idx1 - p])
        idx2 = np.arange(max(0, -p), min(n, n - p))
        if len(idx2) > 0:
            grad[idx2] -= h[idx2 + p]
    grad /= len(active)
    h_new = h - lr * grad
    return _repair(h_new, n)

Where corr has 2n-1 elements. The "middle" element (k=n-1, which corresponds to lag 0) is:
corr[n-1] = sum_i h[i]*(1-h[i]) = sum h[i] - sum h[i]² = n/2 - sum h[i]²

To minimize this, we want to maximize sum h[i]², which means making h as "peaked" as possible (0 or 1). But the other lags will increase.

So there's a trade-off: making h more peaked reduces the self-overlap but increases the cross-overlap at other lags.

This suggests the optimal h might be close to a step function (0 or 1) with a specific pattern.

`_smooth_max`: m + log(sum(exp(alpha*(c-m))))/alpha

If c-m ranges from -0.1 to 0, then alpha*(c-m) ranges from -5000 to 0. We clip to [-500, 500], so exp(s) ranges from e^-500 to 1. The sum is at most 139. log(139) ≈ 5. This is fine numerically.

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_round(h, n, corr, time_limit, n_coarse, n_bisect, temperature=0.0):
    improved = False
    order = np.random.permutation(n)
    for ai, i in enumerate(order):
        for j in order[ai + 1:]:
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            cm = np.max(corr)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cm - temperature:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

def _slsqp(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 100})
        return _repair(res.x, n)
    except Exception:
        return h

def _perturb_structured(h, n, scale, rng):
    x = np.arange(n)
    freq = rng.choice([1, 2, 3, 4, 5, 6, 8])
    phase = rng.uniform(0, 2 * np.pi)
    pert = scale * np.sin(2 * np.pi * freq * x / n + phase)
    return _repair(h + pert, n)

def _perturb_binary(h, n, rng):
    h_b = h.copy()
    k = rng.integers(3, n // 4)
    idx = rng.choice(n, k, replace=False)
    h_b[idx] = rng.uniform(0.0, 1.0, k)
    return _repair(h_b, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 1)

    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-up with increasing alpha
    for alpha in [50, 200, 800, 3000]:
        if time.time() - t0 > budget_s * 0.13:
            break
        hc = _slsqp(best_h, n, target, bounds, alpha, 500)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 2: Main loop - sweeps with SA + SLSQP + perturbations
    stall = 0
    sa_temp = 0.003
    it = 0
    while time.time() - t0 < budget_s - 160:
        it += 1
        # Coordinate sweep (with slight SA for exploration)
        h = best_h.copy()
        corr = _corr(h)
        tl = min(35.0, budget_s - 160 - (time.time() - t0))
        h, corr, imp = _sweep_round(h, n, corr, time.time() + tl, 30, 40, sa_temp)
        c5 = _c5(h, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
            sa_temp = max(0.0002, sa_temp * 0.92)
        else:
            stall += 1
            # Occasionally accept SA-worse solution for exploration
            if c5 < best_c5 + sa_temp * 2 and stall < 8:
                best_h = h.copy()

        # SLSQP periodically with increasing alpha
        if it % 3 == 0 and time.time() - t0 < budget_s - 190:
            alpha = min(15000, 800 + 800 * it)
            hc = _slsqp(best_h, n, target, bounds, alpha, 400)
            c5s = _c5(hc, dx)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = hc.copy()
                stall = 0

        # Multi-direction perturbation when stalled
        if stall >= 4 and time.time() - t0 < budget_s - 190:
            scale = min(0.12, 0.03 + 0.015 * stall)
            for pi in range(4):
                if time.time() - t0 > budget_s - 190:
                    break
                if pi < 2:
                    h_p = _perturb_structured(best_h, n, scale, rng)
                elif pi == 2:
                    h_p = _perturb_binary(best_h, n, rng)
                else:
                    h_p = _repair(best_h + 0.05 * rng.randn(n), n)
                hc = _slsqp(h_p, n, target, bounds, min(10000, 1500 + 1000 * pi), 350)
                corr_p = _corr(hc)
                hc, _, _ = _sweep_round(hc, n, corr_p, time.time() + 12, 25, 35, 0.0)
                c5p = _c5(hc, dx)
                if c5p < best_c5:
                    best_c5 = c5p
                    best_h = hc.copy()
                    stall = 0
                    break
            sa_temp = max(sa_temp, 0.0015)

    # Phase 3: High-precision sweeps (no temperature)
    for _ in range(10):
        if time.time() - t0 > budget_s - 20:
            break
        corr = _corr(best_h)
        h_s, _, _ = _sweep_round(best_h.copy(), n, corr, time.time() + 12, 40, 50, 0.0)
        c5 = _c5(h_s, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_s.copy()

    # Phase 4: Very high-alpha SLSQP polish
    if time.time() - t0 < budget_s - 5:
        for alpha in [10000, 30000]:
            if time.time() - t0 > budget_s - 3:
                break
            hc = _slsqp(best_h, n, target, bounds, alpha, 800)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
