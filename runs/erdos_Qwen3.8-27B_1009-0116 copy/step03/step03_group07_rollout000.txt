import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def soft_obj(v, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_opt(h, n, alpha, maxiter=600):
    def obj(v):
        return soft_obj(v, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return h

def lbfgs_opt(h, n, alpha, pen, maxiter=4000):
    def obj(v):
        return soft_obj(v, alpha) + pen * (v.sum() - n / 2.0) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return repair(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(8 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hid, hjd = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hid + delta, hjd - delta
                    if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hid, hjd
                if improved:
                    break
        if not improved and r > 40:
            break
    return h, best

def swap_descent(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(4 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hi, hj = h[i], h[j]
            h[i], h[j] = hj, hi
            c5 = compute_c5(h, n)
            if c5 < best - 1e-15:
                best = c5
                improved = True
            else:
                h[i], h[j] = hi, hj
        if not improved and r > 20:
            break
    return h, best

def perturbed_restart(h, n, rng, sigma):
    h2 = h + sigma * rng.randn(n)
    return repair(h2, n)
