    target = n / 2.0
    s = h.sum()
    if s < 1e-14:

            h[idx] += a
            diff -= a
            if abs(diff) < 1e-12:
                break
    return h

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, h[idx] - 1.0)
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-12:
                break
    return h

def _slope(h, n, j):
    sl = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            if k < n:
                if k <= j:
                    sl[m] += 1.0 - h[j - k]
                if j <= n - 1 - k:
                    sl[m] -= h[j + k]
        else:
            kk = -k
            if kk < n:
                if j <= n - 1 - kk:
                    sl[m] += 1.0 - h[j + kk]
                if j >= kk:
                    sl[m] -= h[j - kk]
    return sl

def _best_d(lo, hi, c, sl):
    if hi - lo < 1e-15:
        return 0.0
    candidates = [lo, hi, 0.0]
    candidates.extend(np.linspace(lo, hi, 41).tolist())
    nm = len(c)
    top = np.argsort(c)[-12:]
    for a in range(len(top)):
        for b in range(a + 1, len(top)):
            dg = sl[top[a]] - sl[top[b]]
            if abs(dg) > 1e-14:
                d = (c[top[b]] - c[top[a]]) / dg
                if lo < d < hi:
                    candidates.append(d)
    best_d, best_v = 0.0, c.max()
    for d in candidates:
        v = (c + sl * d).max()
        if v < best_v:
            best_v, best_d = v, d
    return best_d

def _coord_pass(h, n, c, rng, n_rounds, t_lim):
    for rnd in range(n_rounds):
        if time.time() > t_lim:
            break
        order = rng.permutation(n)
        any_move = False
        for j in order:
            sl = _slope(h, n, j)
            lo, hi = -h[j], 1.0 - h[j]
            if hi - lo < 1e-14:
                continue
            d = _best_d(lo, hi, c, sl)
            if abs(d) > 1e-14:
                cv = (c + sl * d).max()
                if cv < c.max() - 1e-15:
                    h[j] += d
                    c = c + sl * d
                    c = np.maximum(c, 0.0)
                    any_move = True
        if not any_move:
            break
    return h, c

def _grad_pass(h, n, c, rng, n_steps, t_lim):
    momentum = np.zeros(n)
    mu = 0.85
    step = 0.02
    for it in range(n_steps):
        if time.time() > t_lim:
            break
        q = min(5 + it // 2000, 25)
        top = np.argsort(c)[-q:]
        g = np.zeros(n)
        for m in top:
            g += _slope(h, n, 0)  # placeholder, need per-m gradient
        # Compute combined gradient properly
        g = np.zeros(n)
        for m in top:
            k = m - (n - 1)
            for j in range(n):
                if k >= 0:
                    if k < n:
                        if k <= j:
                            g[j] += 1.0 - h[j - k]
                        if j <= n - 1 - k:
                            g[j] -= h[j + k]
                else:
                    kk = -k
                    if kk < n:
                        if j <= n - 1 - kk:
                            g[j] += 1.0 - h[j + kk]
                        if j >= kk:
                            g[j] -= h[j - kk]
        g /= len(top)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        g /= gn
        momentum = mu * momentum + step * g
        h_new = h - momentum
        h_new = _repair(h_new, n)
        c_new = _corr(h_new)
        C_new = c_new.max()
        C_old = c.max()
        if C_new < C_old - 1e-14:
            h, c = h_new, c_new
            step = min(step * 1.1, 0.15)
        else:
            step *= 0.75
            if step < 1e-5:
                h_new = _repair(h + 0.03 * rng.randn(n), n)
                c = _corr(h_new)
                h = h_new
                momentum = np.zeros(n)
                step = 0.01
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    c = _corr(h)
    best_h = h.copy()
    best_c5 = c.max() * dx

    def tleft():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent (many passes)
    h, c = _coord_pass(h, n, c, rng, 2000, t0 + budget_s * 0.35)
    cv = c.max() * dx
    if cv < best_c5:
        best_c5, best_h = cv, h.copy()

    # Phase 2: Gradient descent with active set
    h, c = _grad_pass(h, n, c, rng, 30000, t0 + budget_s * 0.55)
    cv = c.max() * dx
    if cv < best_c5:
        best_c5, best_h = cv, h.copy()

    # Phase 3: More coordinate descent
    h, c = _coord_pass(h, n, c, rng, 3000, t0 + budget_s * 0.70)
    cv = c.max() * dx
    if cv < best_c5:

    # Phase 4: Multi-restart with perturbations
    for r in range(12):
        if tleft() < 40:

        h_p = _repair(best_h + 0.04 * rng.randn(n), n)
        c_p = _corr(h_p)
        h_p, c_p = _coord_pass(h_p, n, c_p, rng, 800, t0 + budget_s * 0.85)
        cv = c_p.max() * dx
        if cv < best_c5:
            best_c5, best_h = cv, h_p.copy()

        cv = c_p.max() * dx
        if cv < best_c5:
            best_c5, best_h = cv, h_p.copy()

    # Phase 5: Final ultra-fine coordinate descent
    h, c = _coord_pass(best_h.copy(), n, _corr(best_h), rng, 5000, t0 + budget_s - 5)
    cv = c.max() * dx
    if cv < best_c5:
        best_c5, best_h = cv, h.copy()

    # Phase 6: Tiny random moves
    h = best_h.copy()
    c = _corr(h)
    C = c.max()
    for _ in range(200000):
        if tleft() < 3:
            break

        for d in (0.0001, 0.0005, 0.002, -0.0001, -0.0005, -0.002):

                cv = (c + sl * d).max()
                if cv < C - 1e-15:
                    h[j] += d
                    c = c + sl * d
                    C = cv

    cv = C * dx
    if cv < best_c5:

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
