import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _sm_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for d in range(n):
        k = n - 1 + d
        if w[k] > 1e-13:
            g[d:] += w[k] * (1.0 - h[:n-d])
            g[:n-d] -= w[k] * h[d:]
    for s2 in range(1, n):
        k = n - 1 - s2
        if w[k] > 1e-13:
            g[:n-s2] += w[k] * (1.0 - h[s2:])
            g[s2:] -= w[k] * h[:n-s2]

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _active_lag_step(h, n, topk, lr, t0, deadline):
    h = h.copy()
    c5 = _eval(h)
    for _ in range(300):
        if time.time() > deadline: break

        idx = np.argsort(c)[-topk:]
        base = c.min()

            if shift >= 0:

                g[d:] += wt[ii] * (1.0 - h[:n-d])
                g[:n-d] -= wt[ii] * h[d:]
            else:
                s = -shift
