import numpy as np

import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0-h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _topk_grad(h, top_idx, n):
    g = np.zeros(n)
    for ki in top_idx:
        L = ki - (n-1)
        if L == 0:
            g += (1.0 - 2.0*h)
        elif L > 0:
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
        else:
            M = -L
            g[:n-M] += (1.0 - h[M:])
            g[M:] -= h[:n-M]
    return g

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)

            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h
