    """Compute weighted sum of correlations and its gradient.
    w: weights for each lag (length 2n-1), summing to 1.
    Returns (value, gradient).
    """
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.dot(w, c))
    
    g = np.zeros(n)
    for p in range(2*n-1):
        if w[p] < 1e-15:
            continue
        k = p - (n-1)
        if k == 0:
            g += w[p] * (1.0 - 2.0 * h)
        elif k > 0:

            g[:n-k] += w[p] * (1.0 - h[k:])
            # Term 2: j in [k, n): contribution -h[j-k]
            g[k:] += w[p] * (-h[:n-k])

def _project(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}."
    # Use bisection on the shift t
    # We want: sum(clip(h_i - t, 0, 1)) = target
    # clip(h_i - t, 0, 1) = max(0, min(1, h_i - t))
    
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h - t, 0.0, 1.0)

    val, w = _smooth_max_grad(c, alpha)

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max_grad(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    return m + float(np.log(e.sum())) / alpha, w

def _corr_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-15:
            continue
        k = p - (n - 1)
        if k == 0:
            g += wp * (1.0 - 2.0 * h)
        elif k > 0:
            g[:n-k] += wp * (1.0 - h[k:])
            g[k:] += -wp * h[:n-k]
        else:
            kp = -k
            g[kp:] += wp * (1.0 - h[:n-kp])
            g[:n-kp] += -wp * h[kp:]
    return g

def _obj_grad(x, alpha):
    c = _corr(x)
    val, w = _smooth_max_grad(c, alpha)
    return val, _corr_grad(x, w)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _slsqp(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _obj_grad(x, alpha)[0]
    def jac(x):
        return _obj_grad(x, alpha)[1]
    def eq(x):
        return float(np.sum(x)) - target
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h

def _sweep(h, n, t_end, n_coarse=22, n_bisect=32):
    corr = _corr(h)
    improved = False
    order = np.random.permutation(n)
    n1 = 2 * n - 1
    for ai in range(n):
        i = order[ai]
        for ai2 in range(ai + 1, n):
            j = order[ai2]
            if time.time() > t_end:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            ei = np.zeros(n1)
            ej = np.zeros(n1)
            for p in range(n1):
                k = p - (n - 1)
                if k == 0:
                    ei[p] = 1 - 2*hi; ej[p] = 1 - 2*hj
                elif k > 0:
                    ei[p] = (1-h[i+k] if i+k<n else 0) - (1-h[i-k] if i>=k else 0)
                    ej[p] = (1-h[j+k] if j+k<n else 0) - (1-h[j-k] if j>=k else 0)
                else:
                    kp = -k
                    ei[p] = (1-h[i-kp] if i>=kp else 0) - (1-h[i+kp] if i+kp<n else 0)
                    ej[p] = (1-h[j-kp] if j>=kp else 0) - (1-h[j+kp] if j+kp<n else 0)
            gd = ei - ej
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= np.max(corr) - 1e-14:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hib = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hib)
                if np.max(corr + mid * gd) <= np.max(corr + hib * gd):
                    hib = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hib)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

def _binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def _block_swap(h, n, seed):
    rng = np.random.RandomState(seed)
    a = rng.randint(0, n - 10)
    b = a + rng.randint(10, n - a)
    blen = rng.randint(3, n - b - a + 1)
    h2 = h.copy()
    temp = h2[a:a+blen].copy()
    h2[a:a+blen] = h2[b:b+blen]
    h2[b:b+blen] = temp
    return _repair(h2, n)

                best_c5 = c5_f
                best_h = h_f.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
