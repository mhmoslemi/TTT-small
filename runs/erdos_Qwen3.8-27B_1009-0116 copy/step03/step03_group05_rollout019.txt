        d = rng.uniform(-0.03, 0.03)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        h = _repair(h, n)
        c_new = _c5(h, dx)
        if c_new < c5_cur or (T > 1e-10 and rng.rand() < np.exp(-(c_new - c5_cur) / T)):
            c5_cur = c_new
        else:
            h[i] -= d
            h[j] += d
            h = _repair(h, n)
        T *= dec
    return h, c5_cur

def _polish(h, n, dx, rng, Tmax):
    t_start = time.time()
    h = h.copy()
    c5_cur = _c5(h, dx)
    tiny = np.array([0.0004, 0.001, 0.002, -0.0004, -0.001, -0.002])
    for _ in range(50000):
        if (time.time() - t_start) > Tmax:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5_cur - 1e-16:
                c5_cur = c
                h = _repair(h, n)
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5_cur

def _multi_gauss(h, n, dx, rng, scale):
    hp = h + scale * rng.randn(n)
    return _repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: L-BFGS-B soft-max
    rem = budget_s - (time.time() - t0)
    t1 = min(280, rem - 500)
    if t1 > 30:
        h1, c1 = _lbfgsb_phase(h0, n, dx, t1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # Phase 2: Pair coordinate descent
    rem = budget_s - (time.time() - t0)
    t2 = min(300, rem - 300)
    if t2 > 20:
        h2, c2 = _pair_cd(best_h, n, dx, rng, t2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Another L-BFGS-B from CD result
    rem = budget_s - (time.time() - t0)
    t3 = min(150, rem - 200)
    if t3 > 20:
        h3, c3 = _lbfgsb_phase(best_h, n, dx, t3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: More pair CD
    rem = budget_s - (time.time() - t0)
    t4 = min(200, rem - 100)
    if t4 > 20:
        h4, c4 = _pair_cd(best_h, n, dx, rng, t4)
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    # Phase 5: Simulated annealing
    rem = budget_s - (time.time() - t0)
    t5 = min(150, rem - 50)
    if t5 > 15:
        h5, c5 = _sa_phase(best_h, n, dx, rng, t5)
        if c5 < best_c5:
            best_c5, best_h = c5, h5

    # Phase 6: Final pair CD + polish
    rem = budget_s - (time.time() - t0)
    t6 = min(100, rem - 30)
    if t6 > 10:
        h6, c6 = _pair_cd(best_h, n, dx, rng, t6)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    rem = budget_s - (time.time() - t0)
    t7 = max(10, rem - 5)
    if t7 > 5:
        h7, c7 = _polish(best_h, n, dx, rng, t7)
        if c7 < best_c5:
            best_c5, best_h = c7, h7

    # Multi-restart: try a few perturbations
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        for scale in [0.015, 0.03, 0.05]:
            if (time.time() - t0) > budget_s - 15:
                break
            hr = _multi_gauss(best_h, n, dx, rng, scale)
            hr, cr = _pair_cd(hr, n, dx, rng, min(10, rem - 15))
            if cr < best_c5:
                best_c5, best_h = cr, hr

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
