    else:
        x = np.linspace(0, 2, n, endpoint=False)
        m1 = np.random.randint(2, 12)
        m2 = np.random.randint(2, 12)
        if m2 == m1:
            m2 += 1
        amp = np.random.uniform(0.02, 0.05)
        p1 = np.random.uniform(0, 2 * np.pi)
        p2 = np.random.uniform(0, 2 * np.pi)
        h2 = h + amp * np.cos(np.pi * m1 * x + p1) + 0.5 * amp * np.cos(np.pi * m2 * x + p2)
        h2 -= 0.5 * amp * (np.mean(np.cos(np.pi * m1 * x + p1)) + np.mean(np.cos(np.pi * m2 * x + p2)))
        return _repair(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-up
    for alpha in [30, 100, 300, 800, 1500]:
        if time.time() - t0 > 75:
            break
        hc = _slsqp(best_h, n, target, bounds, alpha, 300)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 2: Main optimization loop
    stall = 0
    rnd = 0
    while time.time() - t0 < budget_s - 55:
        rnd += 1
        h = best_h.copy()
        corr = _corr(h)
        G = _all_g(h, n)
        h, corr, imp, ni = _sweep_round(h, n, corr, G, time.time() + 25, 35, 50)
        if imp:
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # Periodic SLSQP refresh
        if rnd % 3 == 0 and time.time() - t0 < budget_s - 70:
            alpha = min(4000, 300 + 150 * rnd)
            hc = _slsqp(best_h, n, target, bounds, alpha, 250)
            hc_corr = _corr(hc)
            hc_G = _all_g(hc, n)
            hc, hc_corr, _, _ = _sweep_round(hc, n, hc_corr, hc_G, time.time() + 15, 30, 45)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc.copy()
                stall = 0

        # Perturbation when stalled
        if stall >= 3 and time.time() - t0 < budget_s - 70:
            h_p = _perturb(best_h, n, stall)
            hc = _slsqp(h_p, n, target, bounds, 500, 200)
            hc_corr = _corr(hc)
            hc_G = _all_g(hc, n)
            hc, hc_corr, _, _ = _sweep_round(hc, n, hc_corr, hc_G, time.time() + 18, 30, 45)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc.copy()
            stall = 0

    # Phase 3: Final precision sweeps (higher resolution)
    if time.time() - t0 < budget_s - 15:
        for _ in range(8):
            if time.time() - t0 > budget_s - 10:
                break
            h = best_h.copy()
            corr = _corr(h)
            G = _all_g(h, n)
            h, corr, _, _ = _sweep_round(h, n, corr, G, time.time() + 15, 50, 60)
            c5 = _c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: Final SLSQP polish
    if time.time() - t0 < budget_s - 3:
        hc = _slsqp(best_h, n, target, bounds, 5000, 500, 1e-18)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
