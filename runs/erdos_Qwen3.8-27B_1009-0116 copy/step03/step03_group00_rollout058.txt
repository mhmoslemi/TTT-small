    if s < 1e-12:
        return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

    if total < 1e-300:

            jp = i + lag
            g[jp] += w * (1.0 - x[i])

def _corr(h):

    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.ones(h) 
        # Better: sort by available room
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:

    return h

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    c = _corr(x) * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    t = e.sum()
    w = e / t

    for k in range(len(c)):

            continue

            g[j] += w[k] * (1.0 - x[i])

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _softmax_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softmax_grad(x, n, alpha):
    dx = 2.0 / n
    c = _corr(x) * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    t = e.sum()
    w = e / max(t, 1e-300)
    g = np.zeros(len(x))
    np_ = len(x)
    for k in range(len(c)):
        if w[k] < 1e-12:
            continue
        lag = k - (np_ - 1)
        lo = max(0, -lag)
        hi = min(np_, np_ - lag)
        for i in range(lo, hi):
            j = i + lag
            g[j] += w[k] * (1.0 - x[i])
            g[i] -= w[k] * x[j]
    return g * dx

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def _slsqp(h0, n, alpha, maxiter):

    r = minimize(_softmax_obj, h0, jac=_softmax_grad, method='SLSQP',
                 bounds=bnd, constraints=con, args=(n, alpha),
                 options={'maxiter': maxiter, 'ftol': 1e-18})
    return r.x

def _proj_grad(h, n, steps, lr):
    for _ in range(steps):
        c = _corr(h) * (2.0 / n)
        km = c.argmax()
        lag = km - (n - 1)
        g = np.zeros(n)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        for i in range(lo, hi):
            j = i + lag
            g[j] += (1.0 - h[i])
            g[i] -= h[j]
        g *= (2.0 / n)
        g -= g.mean()
        h -= lr * g
        h = np.clip(h, 0.0, 1.0)
        h = _repair(h, n)
    return h

                            h[i], h[j] = hi, hj
        if imp:

        else:
            stagnant += 1

    bh, bc5 = h.copy(), c5
    T = 4e-4
    cool = 0.9998
    while time.time() < t_end:

            c5 = cn
            if cn < bc5:
                bc5 = cn
                bh = h.copy()
        else:
            h[i], h[j] = hi, hj
        T = max(T * cool, 1e-10)
    return bh, bc5
