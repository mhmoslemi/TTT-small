    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for idx in order:
            a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            if diff < 1e-14: break
    else:
        order = np.argsort(h)
        for idx in order:
            a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if diff > -1e-14: break
    return h

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _smooth_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w_all = e / np.sum(e)
    g = np.zeros(n)
    for k in range(len(c)):
        w = w_all[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += w * (1.0 - x[i])
            g[i] -= w * x[jp]
    return g * dx_val

def _active_subgrad(h, n, topk_ids):
    dx_val = 2.0 / n
    g = np.zeros(n)
    cnt = len(topk_ids)
    for k in topk_ids:
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += (1.0 - h[i])
            g[i] -= h[jp]
    return g * dx_val / cnt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(best_h, n)

    # Phase 1: Adam continuation (smooth -> sharp)
    if tl() > 250:
        h = best_h.copy()
        ma = np.zeros(n); va = np.zeros(n); t_a = 0
        for alpha in [80, 300, 1000, 4000, 15000, 60000, 200000]:
            n_it = 3000 if alpha < 1000 else 1500
            for it in range(n_it):
                if tl() < 250: break
                t_a += 1
                g = _smooth_grad(h, n, alpha)
                ma = 0.85 * ma + 0.15 * g
                va = 0.999 * va + 0.001 * g * g
                mh = ma / (1 - 0.85 ** t_a)
                vh = va / (1 - 0.999 ** t_a)
                lr = 0.003 / (1.0 + 0.005 * (t_a % 2000))
                h = _project(h - lr * mh / (np.sqrt(vh) + 1e-8), n)
                c5v = _c5(h, n)
                if c5v < best_c5:
                    best_c5 = c5v; best_h = h.copy()

    # Phase 2: Subgradient on top-k lags with momentum
    if tl() > 200:
        h = best_h.copy(); c5c = _c5(h, n)
        mom = np.zeros(n)
        for it in range(4000):
            if tl() < 200: break
            c = _c5_vec(h, n)
            topk = np.argsort(c)[::-1][:min(8, len(c))]
            g = _active_subgrad(h, n, topk)
            mom = 0.7 * mom + g
            for lr in [0.008, 0.004, 0.002, 0.001, 0.0005]:
                ht = _project(h - lr * mom, n)
                c5t = _c5(ht, n)
                if c5t < c5c:
                    h = ht; c5c = c5t
                    if c5c < best_c5:
                        best_c5 = c5c; best_h = h.copy()
                    break

    # Phase 3: SLSQP refinement
    if tl() > 100:
        for alpha in [30000, 120000]:
            if tl() < 50: break
            try:
                res = minimize(_smooth_obj, best_h, args=(n, alpha),
                               jac=_smooth_grad, method='SLSQP',
                               bounds=[(0, 1)] * n,
                               constraints=[{'type': 'eq',
                                             'fun': lambda x: np.sum(x) - n / 2.0,
                                             'jac': lambda x: np.ones(n)}],
                               options={'maxiter': 400, 'ftol': 1e-18})
                hc = _repair(res.x, n)
                c5v = _c5(hc, n)
                if c5v < best_c5:
                    best_c5 = c5v; best_h = hc
            except Exception:
                pass

    # Phase 4: Focused pair search on active indices
    if tl() > 60:
        h = best_h.copy(); c5c = _c5(h, n)
        c = _c5_vec(h, n); m = np.max(c)
        active = np.where(c > m * 0.985)[0]
        aff = set()
        for k in active:
            lag = k - (n - 1)
            a0 = max(0, -lag); a1 = min(n, n - lag)
            for i in range(a0, a1):
                aff.add(i); aff.add(i + lag)
        aff = sorted(aff)
        if len(aff) > 30:
            aff = aff[:30]
        dets = np.array([0.001, 0.003, 0.007, 0.015, 0.03,
                        -0.001, -0.003, -0.007, -0.015, -0.03])
        for sw in range(40):
            if tl() < 60: break
            imp = False
            for i in aff:
                if tl() < 60: break
                hi = h[i]
                for j in aff:
                    if j <= i: continue
                    hj = h[j]
                    for d in dets:
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
                        h[i], h[j] = ni, nj
                        cv = _c5(h, n)
                        if cv < c5c - 1e-15:
                            c5c = cv; hi, hj = ni, nj; imp = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i] = hi
            if not imp: break
        if c5c < best_c5:
            best_c5 = c5c; best_h = h.copy()

    # Phase 5: Try n=100
    if tl() > 50 and n != 100:
        n2 = 100
        x_old = np.linspace(0, 2.0, n, endpoint=False)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = _repair(h2, n2)
        ma2 = np.zeros(n2); va2 = np.zeros(n2); t2 = 0
        for alpha in [5000, 30000]:
            for it in range(1000):
                if tl() < 50: break
                t2 += 1
                g = _smooth_grad(h2, n2, alpha)
                ma2 = 0.85 * ma2 + 0.15 * g
                va2 = 0.999 * va2 + 0.001 * g * g
                mh = ma2 / (1 - 0.85 ** t2)
                vh = va2 / (1 - 0.999 ** t2)
                lr = 0.002 / (1.0 + 0.005 * (t2 % 1000))
                h2 = _project(h2 - lr * mh / (np.sqrt(vh) + 1e-8), n2)
                c5v = _c5(h2, n2)
                if c5v < best_c5:
                    best_c5 = c5v; best_h = h2.copy(); n = n2
        if n == n2 and tl() > 30:
            try:
                res = minimize(_smooth_obj, h2, args=(n2, 80000),
                               jac=_smooth_grad, method='SLSQP',
                               bounds=[(0, 1)] * n2,

                                             'fun': lambda x: np.sum(x) - n2 / 2.0,
                                             'jac': lambda x: np.ones(n2)}],

                c5v = _c5(hc, n2)
                if c5v < best_c5:

            except Exception:
                pass

    # Phase 6: Ultra-fine final search
    if tl() > 15:
        h = best_h.copy(); c5c = _c5(h, n)
        sd = np.array([0.0003, 0.0008, 0.002, -0.0003, -0.0008, -0.002])
        for sw in range(200):
            if tl() < 15: break
            imp = False
            for i in range(0, n, max(1, n // 25)):
                if tl() < 15: break
                hi = h[i]
                for j in range(i + 1, n, max(1, n // 25)):
                    hj = h[j]
                    for d in sd:
                        ni, nj = hi + d, hj - d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            cv = _c5(h, n)
                            if cv < c5c - 1e-15:
                                c5c = cv; hi, hj = ni, nj; imp = True
                            else:
                                h[i], h[j] = hi, hj
                    h[i] = hi
            if not imp: break
        if c5c < best_c5:
            best_c5 = c5c; best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
