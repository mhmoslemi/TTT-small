Let me try a different approach: **gradient-based line search**

Instead of trying all pairs, use the gradient of the max function to determine which direction to move:
1. Compute corr, find argmax lag
2. Compute gradient of that specific correlation w.r.t. h
3. Project gradient onto feasible set (sum h = n/2, 0 ≤ h ≤ 1)
4. Take a line search step in that direction

This is essentially **subgradient method with projection**.

Actually, since the objective is max of bilinear functions, the subgradient of the max is the gradient of the active (maximum) function. So:
- g = gradient of f_{k*}(h) where k* = argmax corr
- Direction: -g projected onto {sum = 0} (to maintain equality constraint)
- Then project h + step * direction back to [0,1]^n and sum = n/2

Let me implement this:

```python
def _subgradient_step(h, n, corr, step_size):
    """Take one subgradient descent step."""
    k_star = np.argmax(corr)  # index in full correlation
    # gradient of corr[k_star] w.r.t. h
    # corr[k] = sum_i h[i] * (1-h[i+k]) where k = k_star - (n-1)
    # d/d(h[p]) corr[k_star] = (1-h[p+k])_valid - h[p-k]_valid
    # where k = k_star - (n-1)
    k = k_star - (n - 1)
    grad = np.zeros(n)
    # (1-h[p+k]) for valid p+k
    valid1 = (np.arange(n) + k >= 0) & (np.arange(n) + k < n)
    grad[valid1] += (1.0 - h[np.arange(n)[valid1] + k])
    # -h[p-k] for valid p-k
    valid2 = (np.arange(n) - k >= 0) & (np.arange(n) - k < n)
    grad[valid2] -= h[np.arange(n)[valid2] - k]
    
    # Project gradient to sum=0 subspace
    grad -= np.mean(grad)
    
    # Line search
    best_h = h.copy()
    best_val = float(np.max(corr))
    for frac in [step_size, step_size*0.5, step_size*0.25, step_size*0.125]:
        h_try = h - frac * grad
        h_try = np.clip(h_try, 0, 1)
        h_try = _repair(h_try, n)
        c_new = float(np.max(_corr(h_try)))
        if c_new < best_val:
            best_val = c_new
            best_h = h_try.copy()
    return best_h
```

            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            if abs(d_best) < 1e-14:
                continue
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            cur_max = float(np.max(corr))
            improved = True
    return h, corr, improved

def _anneal(h, n, corr, T, rng, n_steps=3000):
    improved = False
    for _ in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        d_min = max(-hi, hj - 1.0)
        d_max = min(1.0 - hi, hj)
        if d_max - d_min < 1e-12:
            continue
        gd = _g_vec(h, i, n) - _g_vec(h, j, n)
        best_d = 0.0
        best_val = float(np.max(corr))
        for _ in range(6):
            d = d_min + rng.random() * (d_max - d_min)
            val = float(np.max(corr + d * gd))
            if val < best_val:
                best_val = val
                best_d = d
        if best_d != 0.0:
            h[i] += best_d
            h[j] -= best_d
            corr = corr + best_d * gd
            improved = True
            continue
        d = d_min + rng.random() * (d_max - d_min)
        val = float(np.max(corr + d * gd))
        delta = val - best_val
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h[i] += d
            h[j] -= d
            corr = corr + d * gd
            improved = True
    return h, corr, improved

def _subgrad_step(h, n, corr, step):
    k_star = int(np.argmax(corr))
    k = k_star - (n - 1)
    idx = np.arange(n)
    grad = np.zeros(n)
    v1 = (idx + k >= 0) & (idx + k < n)
    grad[v1] += (1.0 - h[idx[v1] + k])
    v2 = (idx - k >= 0) & (idx - k < n)
    grad[v2] -= h[idx[v2] - k]
    grad -= np.mean(grad)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    for frac in [step, step * 0.5, step * 0.25, step * 0.1]:
        ht = np.clip(h - frac * grad, 0.0, 1.0)
        ht = _repair(ht, n)
        cm = float(np.max(_corr(ht)))
        if cm < cur_max:
            cur_max = cm
            best_h = ht.copy()
    return best_h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-up (12% time)
    p1_end = t0 + budget_s * 0.12
    for alpha in [40, 100, 300, 800, 2000]:
        if time.time() > p1_end:
            break
        hc = _slsqp(best_h, n, target, bounds, alpha, 500)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 2: Main optimization with annealing + sweeps (63% time)
    p2_end = t0 + budget_s * 0.75
    T = 0.03
    stall = 0
    iters = 0
    while time.time() < p2_end:
        iters += 1
        chunk = min(25.0, p2_end - time.time())
        if chunk < 4:
            break

        # Coordinate sweep
        h = best_h.copy()
        corr = _corr(h)
        h, corr, imp = _sweep(h, n, corr, time.time() + chunk * 0.55, 16, 22)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        # Annealing (skip if very close to end)
        if time.time() < p2_end - 8:
            h2 = best_h.copy()
            corr2 = _corr(h2)
            T = max(T * (0.97 if stall < 3 else 1.15), 0.0005)
            T = min(T, 0.06)
            h2, corr2, _ = _anneal(h2, n, corr2, T, rng, 2500)
            c5a = float(np.max(corr2) * dx)
            if c5a < best_c5 - 1e-14:
                best_c5 = c5a
                best_h = h2.copy()
                stall = 0

        # Subgradient kick when stalled
        if stall >= 3 and time.time() < p2_end - 15:
            h3 = best_h.copy()
            corr3 = _corr(h3)
            step = 0.02 + 0.01 * stall
            for _ in range(min(stall, 5)):
                h3 = _subgrad_step(h3, n, corr3, step)
                corr3 = _corr(h3)
            c5s = _c5(h3, dx)
            if c5s < best_c5 - 1e-14:
                best_c5 = c5s
                best_h = h3.copy()
            stall = 0

        # Periodic SLSQP
        if iters % 4 == 0 and time.time() < p2_end - 25:
            alpha = 150 + 80 * iters
            hc = _slsqp(best_h, n, target, bounds, min(alpha, 3000), 350)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _sweep(hc, n, hc_corr, time.time() + 12, 12, 18)
            c5t = float(np.max(hc_corr) * dx)
            if c5t < best_c5 - 1e-14:
                best_c5 = c5t
                best_h = hc.copy()
                stall = 0

    # Phase 3: Intensive local polish (18% time)
    p3_end = t0 + budget_s * 0.93
    while time.time() < p3_end:
        h = best_h.copy()
        corr = _corr(h)
        h, corr, imp = _sweep(h, n, corr, time.time() + 18, 20, 28)
        if imp:
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
        if time.time() < p3_end - 8:
            hc = _slsqp(best_h, n, target, bounds, 3000, 500)
            c5 = _c5(hc, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = hc.copy()
        if time.time() > p3_end - 3:
            break

    # Phase 4: Final high-alpha SLSQP
    if time.time() < budget_s - 5:
        for a in [5000, 8000]:
            if time.time() > budget_s - 3:
                break
            hc = _slsqp(best_h, n, target, bounds, a, 800)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
