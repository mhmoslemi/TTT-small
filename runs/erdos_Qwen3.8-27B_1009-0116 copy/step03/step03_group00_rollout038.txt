        return np.full(n, 0.5)

    dx_v = 2.0 / n
    g = np.zeros(n)
    c = _corr(h, n)  # not needed if we pass wts

                g[i] -= w * h[i + ll]

    """Objective and gradient for SLSQP with penalty."""
    c = _corr(h, n)
    m = np.max(c)

                break
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a

```python
import numpy as np
from scipy.optimize import minimize
import time


def _cd(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def _c5(h, n):
    return float(np.max(_cd(h, n)))


def _rp(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = np.clip(h * (t / s), 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
    return h


def _lwvg(h, n, alpha):
    c = _cd(h, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    tot = float(np.sum(e))
    w = e / tot
    val = m + np.log(tot) / alpha
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - h[i + lag])
                g[i + lag] -= wk * h[i]
        else:
            ll = -lag
            for i in range(n - ll):
                g[i + ll] += wk * (1.0 - h[i])
                g[i] -= wk * h[i + ll]
    return val, g * (2.0 / n)


def _scf(x, n):
    return np.sum(x) - n / 2.0


def _scj(x, n):
    return np.ones(n)


def _snf(x, n, alpha, wp):
    v, g = _lwvg(x, n, alpha)
    d = np.sum(x) - n / 2.0
    return v + wp * d * d


def _snj(x, n, alpha, wp):
    v, g = _lwvg(x, n, alpha)
    d = np.sum(x) - n / 2.0
    return g + 2.0 * wp * d


def _slo(h0, n, alpha, mit, wp):
    cons = [{'type': 'eq', 'fun': _scf, 'jac': _scj, 'args': (n,)}]
    return minimize(_snf, h0, args=(n, alpha, wp), jac=_snj,
                    method='SLSQP', bounds=[(0.0, 1.0)] * n,
                    constraints=cons,
                    options={'maxiter': mit, 'ftol': 1e-18}).x


def _ps(h, n, t_end, dts):
    c5c = _c5(h, n)
    for _ in range(5000):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return h, c5c
            hi0 = h[i]
            for j in range(i + 1, n):
                hj0 = h[j]
                for d in dts:
                    ni, nj = hi0 + d, hj0 - d
                    if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                        continue
                    h[i], h[j] = ni, nj
                    cn = _c5(h, n)
                    if cn < c5c - 1e-16:
                        c5c, imp = cn, True
                        hi0, hj0 = ni, nj
                    else:
                        h[i], h[j] = hi0, hj0
            h[i] = hi0
        if not imp:
            break
    return h, c5c


def _gs(h, n, t_end, rng):
    c5c = _c5(h, n)
    sv = [0.03, 0.01, 0.004, 0.001, 0.0003]
    stuck = 0
    while time.time() < t_end:
        c = _cd(h, n)
        top3 = np.argsort(c)[::-1][:3]
        km = int(top3[int(rng.randint(3))])
        lag = km - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += 1.0 - h[i + lag]
                g[i + lag] -= h[i]
        else:
            ll = -lag
            for i in range(n - ll):
                g[i + ll] += 1.0 - h[i]
                g[i] -= h[i + ll]
        g *= 2.0 / n
        ju, jd = int(np.argmin(g)), int(np.argmax(g))
        ok = False
        if ju != jd:
            for s in sv:
                nu, nd = h[ju] + s, h[jd] - s
                if nu > 1.0 or nd < 0.0:
                    continue
                sa, sb = h[ju], h[jd]
                h[ju], h[jd] = nu, nd
                cn = _c5(h, n)
                if cn < c5c - 1e-16:
                    c5c, ok = cn, True
                    break
                h[ju], h[jd] = sa, sb
        if not ok:
            for _ in range(5):
                a, b = int(rng.randint(n)), int(rng.randint(n))
                if a == b:
                    continue
                s = sv[int(rng.randint(len(sv)))]
                sg = 1.0 if rng.rand() < 0.5 else -1.0
                na, nb = h[a] + sg * s, h[b] - sg * s
                if 0.0 <= na <= 1.0 and 0.0 <= nb <= 1.0:
                    sa, sb = h[a], h[b]
                    h[a], h[b] = na, nb
                    cn = _c5(h, n)
                    if cn < c5c - 1e-16:
                        c5c = cn
                        ok = True
                        break
                    h[a], h[b] = sa, sb
        stuck += 0 if ok else 1
        if stuck > 500:
            stuck = 0
            for _ in range(10):
                a, b = int(rng.randint(n)), int(rng.randint(n))
                if a == b:

                if 0.0 <= na <= 1.0 and 0.0 <= nb <= 1.0:

                        c5c = cn
                        ok = True
                        stuck = 0
                        break
                    h[a], h[b] = sa, sb
    return h, c5c
