from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_obj(h, alpha):
    corr = _corr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    corr = _corr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    w = np.exp(s)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        j = np.arange(n)
        v1 = (j + lag >= 0) & (j + lag < n)
        v2 = (j - lag >= 0) & (j - lag < n)
        t = np.zeros(n)
        t[v1] += 1.0 - h[j[v1] + lag]
        t[v2] += -h[j[v2] - lag]
        grad += wk * t
    return grad

def _smooth_obj_pen(h, alpha, target, pen):
    return _smooth_obj(h, alpha) + pen * (np.sum(h) - target) ** 2

def _smooth_obj_pen_grad(h, alpha, target, pen):
    f = _smooth_obj(h, alpha) + pen * (np.sum(h) - target) ** 2
    g = _smooth_grad(h, alpha) + 2 * pen * (np.sum(h) - target)
    return f, g

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

    target = n / 2.0

        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5
    except Exception:
        pass
    return best_h, best_c5

def _smooth_grad(h, alpha):
    n = len(h)
    corr = _corr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    w = np.exp(s)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        j = np.arange(n)
        v1 = (j + lag >= 0) & (j + lag < n)
        v2 = (j - lag >= 0) & (j - lag < n)
        t = np.zeros(n)
        t[v1] += 1.0 - h[j[v1] + lag]
        t[v2] -= h[j[v2] - lag]
        grad += wk * t
    return grad

    g = _smooth_grad(h, alpha) + 2 * pen * (np.sum(h) - target)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_obj(h, alpha):
    corr = _corr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    corr = _corr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    w = np.exp(s)
    w /= np.sum(w)
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        jp = j + lag
        jm = j - lag
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        t = np.zeros(n)
        if np.any(v1):
            t[v1] = 1.0 - h[jp[v1]]
        if np.any(v2):
            t[v2] += -h[jm[v2]]
        grad += wk * t
    return grad

def _obj_fn(x, alpha, target, pen):
    f = _smooth_obj(x, alpha) + pen * (np.sum(x) - target) ** 2
    g = _smooth_grad(x, alpha) + 2.0 * pen * (np.sum(x) - target)
    return f, g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _lbfgs(h, n, dx, alpha, maxiter=600, pen=2000.0):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h.copy(), _c5(h, dx)
    for p in [pen, pen * 20]:
        try:
            res = minimize(_obj_fn, best_h, args=(alpha, target, p),
                          method='L-BFGS-B', bounds=bounds, jac=True,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
    return best_h, best_c5

def _slsqp(h, n, dx, alpha, maxiter=400):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h.copy(), _c5(h, dx)
    for pen in [10000.0, 100000.0]:
        try:
            res = minimize(_obj_fn, best_h, args=(alpha, target, pen),
                          method='SLSQP', bounds=bounds, jac=True,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
    return best_h, best_c5

def _proj_grad(h, n, dx, alpha, n_iters=20000, deadline=None):
    best_c5 = _c5(h, dx)
    h_cur = h.copy()
    target = n / 2.0
    step = 0.01
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        grad = _smooth_grad(h_cur, alpha)
        grad -= np.mean(grad)
        h_new = h_cur - step * grad
        h_new = _repair(h_new, n)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            h_cur = h_new
            step = min(step * 1.1, 0.03)
        elif c5_new < _c5(h_cur, dx) + 1e-15:
            h_cur = h_new
        else:
            step *= 0.85
            if step < 1e-7:
                step = 0.015
    return h_cur, min(best_c5, _c5(h_cur, dx))

def _smart_local(h, n, dx, rng, deadline):
    best_c5 = _c5(h, dx)
    h = h.copy()
    alpha = 8000
    it = 0
    while time.time() < deadline:
        it += 1
        corr = _corr(h)
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        w = np.exp(s)
        w /= np.sum(w)
        grad = np.zeros(n)
        j = np.arange(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            lag = k - (n - 1)
            jp = j + lag
            jm = j - lag
            v1 = (jp >= 0) & (jp < n)
            v2 = (jm >= 0) & (jm < n)
            t = np.zeros(n)
            if np.any(v1):
                t[v1] = 1.0 - h[jp[v1]]
            if np.any(v2):
                t[v2] += -h[jm[v2]]
            grad += wk * t
        grad -= np.mean(grad)
        if it % 5 == 0:
            i = rng.randint(n)
            jx = rng.randint(n)
        else:
            i = np.argmax(grad)
            jx = np.argmin(grad)
        if i == jx:
            i = (i + 1) % n
        for ds in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
            ni = h[i] - ds
            nj = h[jx] + ds
            if ni < 0 or nj > 1:
                ni = h[i] + ds
                nj = h[jx] - ds
                if ni > 1 or nj < 0:
                    break
            h[i], h[jx] = ni, nj
            c5n = _c5(h, dx)
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
                break
            else:
                h[i], h[jx] = ni + (ds if ni == h[i] - ds else -ds), h[jx] - (ds if nj == h[jx] + ds else -ds)
                h[i] = min(max(h[i], 0), 1)
                h[jx] = min(max(h[jx], 0), 1)
                break
        if it % 200 == 0:

def _fine_local(h, n, dx, rng, deadline):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per_scale = 20000
    for scale in scales:
        improved = 0
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            jx = rng.randint(n)
            if i == jx:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[jx] - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            h[i], h[jx] = ni, nj
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved += 1
            else:
                h[i], h[jx] = ni - d, nj + d
        if improved == 0 and scale < 0.003:
            break
    return h, best_c5

def _triple_local(h, n, dx, rng, deadline):
    best_c5 = _c5(h, dx)
    h = h.copy()
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        k = rng.randint(n)
        if len(set([i, j, k])) < 3:

        d1 = 0.003 * (2.0 * rng.random() - 1.0)

        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if min(ni, nj, nk) < -1e-10 or max(ni, nj, nk) > 1.0 + 1e-10:

        h[i], h[j], h[k] = ni, nj, nk

        else:
