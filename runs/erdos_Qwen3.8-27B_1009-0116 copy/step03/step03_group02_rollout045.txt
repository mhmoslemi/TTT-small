import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_round(h, n, corr, time_limit, n_coarse=31, n_bisect=48):
    improved = False
    order = np.random.permutation(n)
    for ai, i in enumerate(order):
        for j in order[ai + 1:]:
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            cur_max = np.max(corr)
            if vals[bi] >= cur_max - 1e-14:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

def _subgrad_steps(h, n, dx, target, n_steps=200, lr0=0.04, lr1=0.0003):
    h = h.copy()
    momentum = np.zeros(n)
    decay = 0.85
    best_c5 = float(np.max(_corr(h)) * dx)
    for t in range(n_steps):
        lr = lr0 * (lr1 / lr0) ** (t / max(1, n_steps))
        corr = _corr(h)
        # Use subgradient from top-3 lags for robustness
        top3 = np.argsort(corr)[-3:]
        grad = np.zeros(n)

                if 0 <= p + s < n:

                       bounds=bounds, constraints=cons,

    except Exception:

def _block_perturb(h, n):
    h = h.copy()
    bsz = np.random.choice([3, 5, 7, 10])
    start = np.random.randint(0, max(1, n - bsz))
    h[start:start+bsz] = np.random.choice([0.0, 1.0], size=bsz)

def _swap_perturb(h, n):
    h = h.copy()
    i, j = np.random.choice(n, 2, replace=False)
    h[i], h[j] = h[j], h[i]
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-up
    for alpha in [40, 150, 500, 1500]:
        if time.time() - t0 > budget_s * 0.08:
            break
        hc = _slsqp_step(best_h, n, dx, target, bounds, alpha, 500)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 2: Main diverse loop
    stall = 0
    it = 0
    while time.time() - t0 < budget_s - 110:
        it += 1
        h = best_h.copy()
        corr = _corr(h)
        time_left = budget_s - 110 - (time.time() - t0)
        chunk = min(30.0, time_left * 0.3)
        h, corr, imp = _sweep_round(h, n, corr, time.time() + chunk, 31, 48)
        if imp:
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # Subgradient every 2 iterations
        if it % 2 == 0 and time.time() - t0 < budget_s - 130:
            h_sg = _subgrad_steps(best_h, n, dx, target, 120, 0.03, 0.0002)
            hc2 = _sweep_round(h_sg, n, _corr(h_sg), time.time() + 8, 25, 35)[0]
            c5_t = _c5(hc2, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc2.copy()
                stall = 0

        # SLSQP every 3 iterations
        if it % 3 == 0 and time.time() - t0 < budget_s - 130:
            alpha = 400 + 200 * it
            hc = _slsqp_step(best_h, n, dx, target, bounds, alpha, 400)
            hc2 = _sweep_round(hc, n, _corr(hc), time.time() + 10, 25, 35)[0]
            c5_t = _c5(hc2, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc2.copy()
                stall = 0

        # Diverse perturbation when stalled
        if stall >= 3 and time.time() - t0 < budget_s - 130:
            mode = stall % 3
            if mode == 0:
                h_p = _block_perturb(best_h, n)
            elif mode == 1:
                h_p = _swap_perturb(best_h, n)
                for _ in range(3):
                    h_p = _swap_perturb(h_p, n)
            else:
                h_p = best_h + min(0.08, 0.015 * stall) * np.random.randn(n)
                h_p = _repair(h_p, n)
            hc = _slsqp_step(h_p, n, dx, target, bounds, 800, 300)
            hc2 = _sweep_round(hc, n, _corr(hc), time.time() + 12, 28, 40)[0]
            c5_t = _c5(hc2, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc2.copy()
            stall = 0

    # Phase 3: Intensive final sweeps
    if time.time() - t0 < budget_s - 18:
        for _ in range(8):
            if time.time() - t0 > budget_s - 5:
                break
            h = best_h.copy()
            corr = _corr(h)
            h, corr, _ = _sweep_round(h, n, corr, time.time() + 8, 35, 55)
            c5 = _c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: Final high-alpha SLSQP polish
    if time.time() - t0 < budget_s - 2:
        try:
            hc = _slsqp_step(best_h, n, dx, target, bounds, 10000, 1200)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
