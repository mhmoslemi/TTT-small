def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)

        else:

                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:

                h[idx] -= sub
                diff += sub

            # Also try slight perturbation of scaled init
            if n_alt != n:
                h_scaled = np.interp(np.linspace(0, n - 1, n_alt), np.arange(n), h_init)
                h_try2 = _repair(h_scaled, n_alt)
                c5_try2 = _c5(h_try2, 2.0 / n_alt)
                if c5_try2 < best_c5 + 0.01 and time.time() < t_alt_end - 20:
                    h_t3, c5_t3 = _optimize_from(h_try2, n_alt, min(time.time() + 20, t_alt_end), rng)
                    if c5_t3 < best_c5:
                        best_c5 = c5_t3
                        best_h = h_t3.copy()
                        best_n = n_alt
    # Phase 3: Final polish on best
    if time.time() < budget_s - 10:
        t_pol = min(time.time() + (budget_s - time.time() - 10), budget_s - 5)
        h_pol, c5_pol = _optimize_from(best_h, best_n, t_pol, rng)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
