            if c_minus >= 0.0:
                h[j] = c_minus
                h[(j+1) % n] = h[(j+1) % n] + eps
                h = np.clip(h, 0.0, 1.0)
                cm = _c5(h, dx)
            else:
                cm = best_c5 + 1
            h[j] = orig
            h[(j+1) % n] = np.clip(h[(j+1) % n], 0.0, 1.0)
            grad = (cp - cm) / (2.0 * eps)
            step = -0.3 * grad * 0.01
            step = max(-0.02, min(0.02, step))
            new_j = orig + step
            new_k = h[(j+1) % n] - step
            if new_j < 0.0 or new_j > 1.0 or new_k < 0.0 or new_k > 1.0:
                continue
            h[j] = new_j
            h[(j+1) % n] = new_k
            c = _c5(h, dx)
            h_r = _repair(h, n)
            c_r = _c5(h_r, dx)
            h = h_r
            if c_r < best_c5 - 1e-14:
                best_c5 = c_r
                improved = True
        passed += 1
        stagnant = stagnant + 1 if not improved else 0
    return h, best_c5

def _basin_hop(h, n, dx, rng, deadline, n_kicks=60):
    best_h, best_c5 = h.copy(), _c5(h, dx)
    amp = 0.04
    for k in range(n_kicks):
        if time.time() >= deadline:
            break
        hp = best_h.copy()
        mask = rng.random(n) < 0.3
        hp[mask] = hp[mask] + amp * (2.0 * rng.random(mask.sum()) - 1.0)
        if k % 10 == 9:
            shift = rng.randint(1, 7)
            hp = np.roll(hp, shift)
        hp = _repair(hp, n)
        ht, ct = _lbfgs(hp, n, dx, 800, maxiter=200)
        ht, ct = _slsqp_topk(ht, n, dx, 3, 3000, maxiter=300)
        if ct < best_c5:
            best_h, best_c5 = ht, ct
            amp = max(0.015, amp * 0.85)
        else:
            if k % 5 == 4:
                amp = min(0.2, amp * 1.15)
    return best_h, best_c5

def _fine_shuffle(h, n, dx, rng, deadline, n_tries=80000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per = max(400, n_tries // len(scales))
    for sc in scales:
        imp = False
        for _ in range(per):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = (i + rng.randint(1, n)) % n
            d = sc * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-10 or ni > 1.0+1e-10 or nj < -1e-10 or nj > 1.0+1e-10:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < best_c5 - 1e-14:
                best_c5 = c
                imp = True
            else:
                h[i], h[j] = ni - d, nj + d
        if not imp and sc < 0.001:
            break
    return h, best_c5

def _mirror_opt(h, n, dx):
    hm = 1.0 - h
    hm = _repair(hm, n)
    cm = _c5(hm, dx)
    if cm < _c5(h, dx):
        return hm, cm
    return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    t1 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h0.copy(), _c5(h0, dx)

    # Phase 1: L-BFGS-B warmup
    tEnd = t0 + 35
    for alpha in [100, 500, 1500, 5000]:
        if time.time() >= tEnd:
            break
        ht, ct = _lbfgs(best_h, n, dx, alpha, maxiter=250)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 2: SLSQP + topk refinement
    tEnd = t0 + 100
    for alpha in [2000, 8000, 30000]:
        if time.time() >= tEnd:
            break
        ht, ct = _slsqp(best_h, n, dx, alpha, maxiter=500)
        if ct < best_c5:
            best_h, best_c5 = ht, ct
    for tk in [2, 3, 5]:
        if time.time() >= tEnd:
            break
        ht, ct = _slsqp_topk(best_h, n, dx, tk, 5000, maxiter=400)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Mirror check
    hm, cm = _mirror_opt(best_h, n, dx)
    if cm < best_c5:
        best_h, best_c5 = hm, cm

    # Phase 3: Basin hopping (main global search)
    if time.time() < t0 + budget_s - 300:
        tEnd = min(t0 + budget_s - 300, time.time() + 250)
        best_h, best_c5 = _basin_hop(best_h, n, dx, rng, tEnd, n_kicks=50)

    # Phase 4: Coordinate descent
    if time.time() < t0 + budget_s - 200:
        tEnd = min(t0 + budget_s - 200, time.time() + 180)
        ht, ct = _coord_descent(best_h, n, dx, rng, tEnd, n_passes=15000)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 5: Second basin hop from refined point
    if time.time() < t0 + budget_s - 150:
        tEnd = min(t0 + budget_s - 150, time.time() + 100)
        ht, ct = _basin_hop(best_h, n, dx, rng, tEnd, n_kicks=25)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 6: SLSQP final polish with topk
    if time.time() < t0 + budget_s - 80:
        for tk in [2, 3, 4]:
            if time.time() >= t0 + budget_s - 80:
                break
            ht, ct = _slsqp_topk(best_h, n, dx, tk, 20000, maxiter=500)
            if ct < best_c5:
                best_h, best_c5 = ht, ct

    # Phase 7: Fine shuffle
    if time.time() < t0 + budget_s - 40:
        tEnd = min(t0 + budget_s - 40, time.time() + 120)
        ht, ct = _fine_shuffle(best_h, n, dx, rng, tEnd, n_tries=60000)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 8: SLSQP ultra-high alpha
    if time.time() < t0 + budget_s - 20:
        ht, ct = _slsqp(best_h, n, dx, 50000, maxiter=400)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 9: Final micro-shuffle
    if time.time() < t0 + budget_s - 5:
        tEnd = min(t0 + budget_s - 3, time.time() + 15)
        ht, ct = _fine_shuffle(best_h, n, dx, rng, tEnd, n_tries=15000)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
