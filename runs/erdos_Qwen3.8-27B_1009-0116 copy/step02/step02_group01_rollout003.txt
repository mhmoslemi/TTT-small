    if abs(s - target) > 1e-10:
        for _ in range(30):
            h *= target / s
            h = np.clip(h, 0.0, 1.0)
            s = h.sum()

                break
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)

                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                else:
                    add = max(diff, -h[idx])
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-13:

        j2 = i + k - n + 1
        if 0 <= j2 < n:
            g[i] += (1.0 - h[j2])
            g[j2] -= h[i]

    m = np.max(c)
    if beta > 0:
        c = c + beta * (c - c.mean())**2
        m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha
