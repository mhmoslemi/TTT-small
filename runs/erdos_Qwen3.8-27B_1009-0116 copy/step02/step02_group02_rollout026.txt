    if abs(diff) > 1e-13:

                a = min(diff, 1.0 - x[k]); x[k] += a; diff -= a
            else:
                a = min(-diff, x[k]); x[k] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _binary_cd(h, n, rng, max_evals=100000):

    dx = 2.0 / n
    h = h.copy()
    best_c = _c5(h, dx)

    zeros = np.where(h == 0.0)[0].tolist()

def _binary_sa(h_init, n, rng, T0, Tmax_evals, t0, deadline):
    """Simulated annealing on binary sequence."""
    dx = 2.0 / n
    h = h_init.copy()
    cur_c = _c5(h, dx)
    best_c = cur_c
    best_h = h.copy()
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    
    for it in range(Tmax_evals):
        if time.time() > deadline:
            break
        T = T0 * (0.9995 ** it)
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a], h[b] = 0.0, 1.0
        c_new = _c5(h, dx)
        delta = c_new - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-8)):
            cur_c = c_new
            ones.remove(a); ones.append(b)
            zeros.remove(b); zeros.append(a)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[a], h[b] = 1.0, 0.0
    return best_h, best_c

def _fista(h0, alpha, target, n, iters=3000):
    """FISTA with log-sum-exp smoothing."""
    dx = 2.0 / n
    x = h0.astype(float).copy()
    z = x.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    
    for it in range(iters):
        # Compute gradient of log-sum-exp
        c = np.correlate(z, 1.0 - z, mode='full')
        m = float(c.max())
        s = np.clip(alpha * (c - m), -500.0, 500.0)
        e = np.exp(s)
        w = e / float(e.sum())
        
        # Gradient
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                g[:n-lag] += wk * (1.0 - z[lag:])
                g[lag:] -= wk * z[:n-lag]
            else:
                L = -lag
                g[L:] += wk * (1.0 - z[:-L])
                g[:-L] -= wk * z[L:]
        
        y = z - step * g
        xn = _proj(y, target)
        
        # Backtracking
        cnt = 0
        while cnt < 8:
            c_obj = np.correlate(xn, 1.0 - xn, mode='full')
            m_obj = float(c_obj.max())
            sn = np.clip(alpha * (c_obj - m_obj), -500.0, 500.0)
            fn = m_obj + np.log(float(np.exp(sn).sum())) / alpha
            quad = (m + np.log(float(np.exp(s).sum())) / alpha +
                    float(np.dot(g, xn - z)) + 0.5/step * float(np.dot(xn-z, xn-z)))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * g
            xn = _proj(y, target)
            cnt += 1
        
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0)/tn * (xn - x), target)
        x = xn
        t = tn
        
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    
    return bx, bc

def _frac_cd(h, dx, n, rng, passes=200):
    """Fractional coordinate descent for near-binary h."""
    best_c = _c5(h, dx)
    h = h.copy()
    dvals = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05,
             -0.002, -0.005, -0.01, -0.02, -0.03, -0.05]
    
    for p in range(passes):
        imp = False
        for _ in range(n * 5):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            for d in dvals:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best_c - 1e-13:
                    best_c = c
                    imp = True
                    break
                h[i], h[j] = hi0, hj0
            else:
                h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, best_c

        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                g[:n-lag] += wk * (1.0 - z[lag:])
                g[lag:] -= wk * z[:n-lag]
            else:
                L = -lag
                g[L:] += wk * (1.0 - z[:-L])
                g[:-L] -= wk * z[L:]
        y = z - step * g
        xn = _proj(y, target)
        cnt = 0
        ok = False
        while cnt < 10:
            c_n = np.correlate(xn, 1.0 - xn, mode='full')
            m_n = float(c_n.max())
            s_n = np.clip(alpha * (c_n - m_n), -500.0, 500.0)
            fn = m_n + np.log(float(np.exp(s_n).sum())) / alpha
            quad = (m_z + np.log(float(np.exp(s_z).sum())) / alpha +
                    float(np.dot(g, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z)))
            if fn <= quad + 1e-10:
                ok = True
                break
            step *= 0.5
            y = z - step * g
            xn = _proj(y, target)
            cnt += 1
        if not ok:
            break
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _frac_cd(h, dx, n, rng, passes=300, t0=None, budget_s=None, min_remain=10):
    best_c = _c5(h, dx)
    h = h.copy()
    dvals = [0.001, 0.003, 0.005, 0.01, 0.015, 0.025, 0.04,
             -0.001, -0.003, -0.005, -0.01, -0.015, -0.025, -0.04]
    for p in range(passes):
        if t0 is not None and _rem_time(t0, budget_s) < min_remain:
            break
        imp = False
        for _ in range(n * 4):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            for d in dvals:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best_c - 1e-13:
                    best_c = c
                    imp = True
                    break
                h[i], h[j] = hi0, hj0
            else:
                h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, best_c

def _tri_cd(h, dx, n, rng, passes=100, t0=None, budget_s=None, min_remain=10):
    best_c = _c5(h, dx)
    h = h.copy()
    for p in range(passes):
        if t0 is not None and _rem_time(t0, budget_s) < min_remain:
            break
        imp = False
        for _ in range(n * 3):
            i = rng.randint(n)
            j = rng.randint(n)
            k = rng.randint(n)
            if len(set([i, j, k])) < 3:
                continue
            d = rng.choice([0.005, 0.01, 0.02, 0.03])
            d = d * (1.0 if rng.rand() > 0.5 else -1.0)
            d2 = -0.5 * d * (1.0 if rng.rand() > 0.5 else -1.0)
            d3 = -d - d2
            hi, hj, hk = h[i], h[j], h[k]
            ni, nj, nk = hi + d, hj + d2, hk + d3
            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                h[i], h[j], h[k] = ni, nj, nk
                c = _c5(h, dx)
                if c < best_c - 1e-13:
                    best_c = c
                    imp = True
                    continue
                h[i], h[j], h[k] = hi, hj, hk
        if not imp:
            break
    return h, best_c

def _slsqp_con(x, target):
    return float(np.sum(x)) - target

def _make_binary(h, n):
    hb = (h > 0.5).astype(float)
    cnt = int(hb.sum())
    if cnt > n // 2:
        idx = np.where(hb == 1)[0]
        rng_local = np.random.RandomState(0)
        rng_local.shuffle(idx)
        for i in range(cnt - n // 2):
            hb[idx[i]] = 0.0
    elif cnt < n // 2:
        idx = np.where(hb == 0)[0]
        rng_local = np.random.RandomState(0)
        rng_local.shuffle(idx)
        for i in range(n // 2 - cnt):
            hb[idx[i]] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n = len(initial_h_values)
        h_start = np.array(initial_h_values, dtype=float)
    else:
        n = 70
        h_start = np.zeros(n)
        h_start[:n // 2] = 1.0

    dx = 2.0 / n
    tgt = n / 2.0
    h0 = _proj(h_start, tgt)
    bh = h0.copy()
    bc = _c5(bh, dx)

    # Phase 1: Quick FISTA warm-up
    for a in [100, 400, 1500]:
        if _rem_time(t0, budget_s) < 80:
            break
        h, c = _fista(bh, a, tgt, n, iters=2000)
        if c < bc:
            bc = c
            bh = h

    # Phase 2: Fractional CD initial pass
    if _rem_time(t0, budget_s) > 40:
        h, c = _frac_cd(bh, dx, n, rng, passes=200, t0=t0, budget_s=budget_s, min_remain=40)
        if c < bc:
            bc = c
            bh = h

    # Phase 3: Binary SA - multiple restarts
    h_bin = _make_binary(bh, n)
    for trial in range(12):
        if _rem_time(t0, budget_s) < 100:
            break
        if trial == 0:
            h_try = h_bin
        elif trial == 1:
            h_try = np.zeros(n)
            h_try[rng.choice(n, n // 2, replace=False)] = 1.0
        else:
            h_try = np.zeros(n)
            positions = rng.choice(n, n // 2, replace=False)
            h_try[positions] = 1.0
        h_g, c_g = _binary_cd_fast(h_try, n, rng, max_rounds=500, evals_per_round=20)
        deadline = time.time() + min(_rem_time(t0, budget_s) - 100, 50)
        h_sa, c_sa = _binary_sa(h_g, n, rng, t0=t0, deadline=deadline, max_evals=60000)
        if c_sa < bc:
            bc = c_sa
            bh = h_sa
            h_bin = bh.copy()

    # Phase 4: FISTA polish on best
    for a in [500, 2000, 8000]:
        if _rem_time(t0, budget_s) < 50:
            break
        h, c = _fista(bh, a, tgt, n, iters=3000)
        if c < bc:
            bc = c
            bh = h

    # Phase 5: Intensive fractional CD + triple CD
    if _rem_time(t0, budget_s) > 30:
        h, c = _frac_cd(bh, dx, n, rng, passes=400, t0=t0, budget_s=budget_s, min_remain=30)
        if c < bc:
            bc = c
            bh = h
    if _rem_time(t0, budget_s) > 20:
        h, c = _tri_cd(bh, dx, n, rng, passes=80, t0=t0, budget_s=budget_s, min_remain=20)
        if c < bc:
            bc = c
            bh = h

    # Phase 6: More binary exploration with 2-swap moves
    if _rem_time(t0, budget_s) > 60:
        h_bin2 = _make_binary(bh, n)
        h_g2, _ = _binary_cd_fast(h_bin2, n, rng, max_rounds=800, evals_per_round=25)
        for trial in range(4):
            if _rem_time(t0, budget_s) < 60:
                break
            deadline2 = time.time() + min(_rem_time(t0, budget_s) - 60, 40)
            h_sa2, c_sa2 = _binary_sa(h_g2, n, rng, t0=t0, deadline=deadline2, max_evals=40000)
            if c_sa2 < bc:
                bc = c_sa2
                bh = h_sa2
        # FISTA on new best
        if _rem_time(t0, budget_s) > 40:
            h, c = _fista(bh, 3000, tgt, n, iters=3000)
            if c < bc:
                bc = c
                bh = h
        if _rem_time(t0, budget_s) > 20:
            h, c = _frac_cd(bh, dx, n, rng, passes=200, t0=t0, budget_s=budget_s, min_remain=20)
            if c < bc:
                bc = c
                bh = h

    # Phase 7: SLSQP final polish
    if _rem_time(t0, budget_s) > 30:
        bnds = [(0.0, 1.0)] * n
        try:
            r = minimize(_sobj, bh, args=(5000,), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (tgt,)}],
                         options={'maxiter': 600, 'ftol': 1e-15})
            hf = _proj(r.x, tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc = cf
                bh = hf
        except Exception:
            pass

    # Phase 8: Final CD sweep
    if _rem_time(t0, budget_s) > 15:
        h, c = _frac_cd(bh, dx, n, rng, passes=300, t0=t0, budget_s=budget_s, min_remain=15)
        if c < bc:
            bc = c
            bh = h
    if _rem_time(t0, budget_s) > 10:
        h, c = _tri_cd(bh, dx, n, rng, passes=50, t0=t0, budget_s=budget_s, min_remain=10)
        if c < bc:
            bc = c
            bh = h

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
