import time

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

    # d/dh[i] = (1-h[i+lag]) if 0<=i+lag<n, minus h[i-lag] if 0<=i-lag<n

    grad[valid1] += (1 - h[np.arange(n)[valid1] + lag])
    grad[valid2] -= h[np.arange(n)[valid2] - lag]

def _slsqp(h, n, dx, alpha, maxiter, deadline):
    def _obj(x):
        c = _corr(x)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -400, 400)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

    def _cons(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() > deadline:
            return h
        return _repair(res.x, n)

        return h
