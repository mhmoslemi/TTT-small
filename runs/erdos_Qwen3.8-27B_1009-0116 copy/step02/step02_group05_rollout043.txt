```python
def _corr_grad(h, n):
    """Compute gradient of full correlation c = correlate(h, 1-h) w.r.t. h.
    Returns grad of shape (2n-1, n)."""
    grad = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        k = j - (n-1)  # lag
        # valid i: max(0,-k) <= i <= min(n-1, n-1-k)
        i_lo = max(0, -k)
        i_hi = min(n-1, n-1-k)
        for i in range(i_lo, i_hi+1):
            # c[j] includes h[i]*(1-h[i+k])
            # d/dh[i]: (1-h[i+k])
            grad[j, i] += (1.0 - h[i+k])
            # d/dh[i+k]: -h[i]
            grad[j, i+k] -= h[i]
    return grad
```

import time

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break

    grad = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        j = k + (n-1)
        i_lo = max(0, -k)
        i_hi = min(n-1, n-1-k)
        i_range = np.arange(i_lo, i_hi+1)
        grad[j, i_range] += (1.0 - h[i_range + k])
        grad[j, i_range + k] -= h[i_range]
    return grad

    lr = lr_init
    for it in range(max_iter):

            break
        corr = _corr(h)
        s = np.clip(alpha*(corr - corr.max()), -500, 500)
        w = np.exp(s)
        w /= w.sum()
        G = _corr_grad_vec(h, n)
        grad = (w @ G) * dx
        grad -= grad.mean()  # project to maintain sum constraint
        # Backtracking line search
        best_h = h.copy()
        best_f = _c5(h, dx)
        for _ls in range(15):
            h_try = np.clip(h - lr*grad, 0.0, 1.0)
            h_try = _repair(h_try, n)
            f_try = _c5(h_try, dx)
            if f_try < best_f:
                best_f = f_try
                best_h = h_try
            lr *= 0.7
        h = best_h
        lr = max(lr * 1.2, lr_init * 0.1)  # restore learning rate somewhat
    return h

    stall = 0

    """Return gradient of _corr(h) w.r.t. h. Shape: (2n-1, n)."""
    grad = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        j = k + (n-1)  # output index
        n_lo = max(0, -k)
        n_hi = min(n-1, n-1-k)
        # For each m in 0..n-1:
        # d/dh[m] = (1-h[m-k]) if n_lo <= m-k <= n_hi
        #           - h[m+k] if n_lo <= m <= n_hi
        # First term: m-k in [n_lo, n_hi], so m in [n_lo+k, n_hi+k]
        m1_lo = max(0, n_lo + k)
        m1_hi = min(n-1, n_hi + k)
        if m1_lo <= m1_hi:
            m1 = np.arange(m1_lo, m1_hi+1)
            grad[j, m1] += (1.0 - h[m1 - k])
        # Second term: m in [n_lo, n_hi]
        if n_lo <= n_hi:
            m2 = np.arange(n_lo, n_hi+1)
            grad[j, m2] -= h[m2 + k]
    return grad
