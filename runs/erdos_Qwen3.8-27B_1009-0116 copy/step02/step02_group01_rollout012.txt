def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    e = np.exp(s)
    total = np.sum(e)
    weights = e / total
    g = np.zeros(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += w * (1.0 - x[i])
            g[i] -= w * x[jp]
    return g * dx_val

def _topk_obj(x, n, k):
    c = _corr(x) * (2.0 / n)
    idx = np.argsort(c)[::-1][:k]
    return np.sum(c[idx])

def _topk_grad(x, n, k):
    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    idx = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for mk in idx:
        lag = mk - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += (1.0 - x[i])
            g[i] -= x[jp]
    return g * dx_val / k
