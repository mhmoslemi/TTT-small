import numpy as np
from scipy.optimize import minimize
import time

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return np.max(_corr_arr(h)) * dx

def _smooth_obj(x, alpha):
    corr = _corr_arr(x)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp(h, n, dx, bounds, eq_con, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _compute_c5(h, dx)
    try:
        res = minimize(lambda x, a=alpha: _smooth_obj(x, a), h,
                       method='SLSQP', bounds=bounds, constraints=[eq_con],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

def _det_sweep(h, n, dx, deadline):
    best = h.copy()
    best_c5 = _compute_c5(best, dx)
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05]
    for scale in scales:
        if time.time() >= deadline:
            break
        for _round in range(200):
            if time.time() >= deadline:
                return best, best_c5
            improved = False
            deltas = [scale, -scale, scale*0.6, -scale*0.6, scale*0.25, -scale*0.25]
            for i in range(n):
                for j in range(i + 1, n):
                    hi = best[i]
                    if hi <= 0 or hi >= 1:
                        continue
                    for d in deltas:
                        ni, nj = hi + d, best[j] - d
                        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                            continue
                        best[i], best[j] = ni, nj
                        c5 = _compute_c5(best, dx)
                        if c5 < best_c5 - 1e-14:
                            best_c5 = c5
                            improved = True
                        else:
                            best[i], best[j] = ni - d, nj + d
                        if improved:
                            break
                    if best[i] != hi:
                        hi = best[i]
            if not improved:
                break
    return best, best_c5

def _random_local(h, n, dx, rng, deadline, n_iter=5000):
    best = h.copy()
    best_c5 = _compute_c5(best, dx)
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for scale in scales:
        if time.time() >= deadline:
            break
        stall = 0
        for _ in range(n_iter):
            if time.time() >= deadline:
                return best, best_c5
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = best[i] + d, best[j] - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            best[i], best[j] = ni, nj
            c5 = _compute_c5(best, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                stall = 0
            else:
                best[i], best[j] = ni - d, nj + d
                stall += 1
            if stall > 200:
                break
    return best, best_c5
