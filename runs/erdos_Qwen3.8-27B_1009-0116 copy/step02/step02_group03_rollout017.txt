import numpy as np

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(500):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        if diff > 0:
            order = np.argsort(-h)  # positions closest to 1 first (least room to add)
            # Actually we want to add to positions with most room
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _smooth_obj(h, n, alpha):
    c = _corr(h)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _con_fun(h, n):
    return np.sum(h) - n / 2.0

def _slsqp(h0, n, alpha, maxiter, t_limit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _con_fun(x, n)}]
    
    def obj(x):
        return _smooth_obj(x, n, alpha)
    
    def obj_grad(x):
        # Gradient of smoothed max
        c = _corr(x)
        m = np.max(c)
        s = alpha * (c - m)
        s = np.clip(s, -500, 500)
        w = np.exp(s) / np.sum(np.exp(s))
        # grad of c[k] w.r.t. x
        grad = np.zeros(n)
        for k_idx in range(len(c)):
            if w[k_idx] < 1e-10:
                continue
            shift = k_idx - (n - 1)
            if shift >= 0:
                for j in range(n):
                    if j >= shift:
                        grad[j] += w[k_idx] * (1.0 - x[j - shift])
                    else:
                        grad[j] -= w[k_idx] * x[j + shift]
            else:
                s2 = -shift
                for j in range(n):
                    if j < n - s2:
                        grad[j] += w[k_idx] * (1.0 - x[j + s2])
                    else:
                        grad[j] -= w[k_idx] * x[j - s2]
        return grad
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h0

Wait, the gradient computation is O(n * 2n) = O(n^2) per call, and SLSQP calls the gradient many times. For n=70, that's 70*140 = 9800 operations per gradient call. With maybe 1000 gradient calls, that's 10M operations. Should be fine.

But actually, I can vectorize the gradient computation:

For shift >= 0:
c[k] = sum_{i=0}^{n-1-k} x[i+k]*(1-x[i])
dc[k]/dx[j] = (1-x[j-k]) for j >= k, and -x[j+k] for j < k, else 0

For j in 0..n-1:
  dc[k]/dx[j] = (1-x[j-k]) * 1[j>=k] - x[j+k] * 1[j+k<n, j<k]

Hmm, this is still a loop over k. Let me just do it as a loop over the active shifts (those with significant weight w[k_idx]).

Actually, for high alpha, only the top 1-2 shifts will have significant weight. So the gradient computation is fast in practice.

from scipy.optimize import minimize
import time

def _c5(h, dx):

def _repair(h, n):

    target = n / 2.0
    s = h.sum()

        return np.full(n, 0.5)
    for _ in range(500):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    c = _corr(h)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    w /= np.sum(w)
    n = len(h)
    grad = np.zeros(n)
    active = w > 1e-8
    for ki in np.where(active)[0]:
        sh = ki - (n - 1)
        if sh >= 0:
            for j in range(n):
                if j >= sh:
                    grad[j] += w[ki] * (1.0 - h[j - sh])
                else:
                    grad[j] -= w[ki] * h[j + sh]
        else:
            s2 = -sh
            for j in range(n):
                if j < n - s2:
                    grad[j] += w[ki] * (1.0 - h[j + s2])
                else:
                    grad[j] -= w[ki] * h[j - s2]
    return grad

def _slsqp_opt(h0, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha), h0,
                       jac=lambda x: _smooth_grad(x, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h0

def _lbfgs_opt(h0, n, alpha, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_obj(x, alpha) + pen * (np.sum(x) - n / 2.0) ** 2
    def grad(x):
        return _smooth_grad(x, alpha) + 2 * pen * (np.sum(x) - n / 2.0)
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return _repair(res.x, n)
    except Exception:
        return h0

def _targeted_cd(h, n, dx, n_iters, t_end, rng):
    h = h.copy()
    best = _c5(h, dx)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        c = _corr(h)
        kmax = np.argmax(c)
        shift = abs(kmax - (n - 1))
        if shift == 0:
            i = int(rng.randint(n))
            for d in [0.02, -0.02, 0.05, -0.05, 0.01, -0.01]:
                ni = h[i] + d
                if 0 <= ni <= 1:
                    h[i] = ni
                    h = _repair(h, n)
                    v = _c5(h, dx)
                    if v < best:
                        best = v
                    else:

                        h = _repair(h, n)
        else:

            # Find pair (i, i+shift) or (i, i-shift) with high contribution
            if kmax >= n - 1:

                for i in range(n - shift):
                    contrib = h[i + shift] * (1 - h[i])
                    candidates.append((contrib, i, i + shift))
                candidates.sort(reverse=True)
                moved = False
                for contrib, i, j in candidates[:20]:
                    if moved:
                        break
                    for d in [0.02, 0.03, 0.01, -0.01, -0.02, -0.03]:
                        hi, hj = h[i], h[j]

def _pair_cd(h, n, dx, n_iters, t_end, rng):
    h = h.copy()
    best = _c5(h, dx)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        improved = False
        for _attempt in range(30):
