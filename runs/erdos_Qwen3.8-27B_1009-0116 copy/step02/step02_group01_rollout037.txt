```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    """Compute correlation array h * (1-h) for all lags."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    """Compute C5 value."""
    return np.max(compute_corr(h)) * dx

def compute_smooth_obj(x, alpha):
    """Log-sum-exp smooth approximation to max."""
    c = compute_corr(x)
    m = np.max(c)
    # Shift for numerical stability
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def compute_smooth_grad(x, alpha):
    """Gradient of log-sum-exp smooth objective."""
    n = len(x)
    c = compute_corr(x)  # length 2n-1
    m = np.max(c)
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)  # weights for each lag, length 2n-1
    
    # Gradient: for each j, sum over lags of w[lag] * dc[lag]/dh[j]
    # Lag index k in [0, 2n-2] corresponds to shift = k - (n-1)
    # dc[k]/dh[j] = (1-h[j+shift]) * I(0≤j+shift<n) - h[j-shift] * I(0≤j-shift<n)
    # where shift = k - (n-1)
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)  # from -(n-1) to (n-1)
    
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        s = shifts[k]  # lag
        if s >= 0:
            # h[j] in first factor: j from 0 to n-1-s, contributes (1-h[j+s])
            for j in range(n - s):
                grad[j] += w[k] * (1.0 - x[j + s])
            # h[j] in second factor: j from s to n-1, contributes -h[j-s]
            for j in range(s, n):
                grad[j] -= w[k] * x[j - s]
        else:
            s2 = -s  # positive
            # h[j] in first factor: j from s2 to n-1, contributes (1-h[j-s2]) = (1-h[j+s])
            for j in range(s2, n):
                grad[j] += w[k] * (1.0 - x[j + s])
            # h[j] in second factor: j from 0 to n-1-s2, contributes -h[j+s2] = -h[j-s]
            for j in range(n - s2):
                grad[j] -= w[k] * x[j - s]
    
    return grad

def repair(h, n):
    """Repair h to satisfy constraints."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h
```

```python
                                best_c5, best_h = c5_cur, h.copy()
                        else:
                            h[i], h[j] = hi, hj
                h[i] = hi
            if not any_imp:
                break

    # Phase 5: Try larger n (80) (80s)
    if elapsed() < budget_s - 50:
        n2 = 80
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
        h2 = np.interp(x_new, x_old, best_h)
        h2 = _repair(h2, n2)
        for alpha in [200, 1000, 5000]:
            if elapsed() > budget_s - 50:
                break
            h2_new = _slsqp(h2, n2, alpha, 300)
            c5_2 = _c5(h2_new, dx2)
            if c5_2 < best_c5:
                best_c5, best_h, n = c5_2, h2_new, n2
                best_h = h2_new.copy()

    # Phase 6: Final fine polish (40s)
    if elapsed() < budget_s - 10:
        h = best_h.copy()
        c5_cur = best_c5
        T = 0.0002
        while elapsed() < budget_s - 10:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.choice([0.0005, 0.001, 0.003, 0.01,
                            -0.0005, -0.001, -0.003, -0.01])
            hi, hj = h[i], h[j]
            h[i] = max(0.0, min(1.0, hi + d))
            h[j] = max(0.0, min(1.0, hj - (h[i] - hi)))
            h = _repair(h, n)
            c = _c5(h, dx)
            if c < c5_cur or (T > 1e-9 and rng.rand() < np.exp(-(c - c5_cur) / T)):
                c5_cur = c
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:
                h[i], h[j] = hi, hj
            T = max(T * 0.9999, 1e-10)

    # Phase 7: One final SLSQP polish
    if elapsed() < budget_s - 5:
        for alpha in [5000, 20000]:
            if elapsed() > budget_s - 5:
                break
            h_new = _slsqp(best_h, n, alpha, 500)
            c5 = _c5(h_new, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
