                break
    return h


def _softmax_value(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha


def _smooth_obj(h, alpha, n, target):
    corr = _corr(h)
    sm = _softmax_value(corr, alpha)
    pen = 80.0 * (float(np.sum(h)) - target) ** 2
    return sm + pen


def _smooth_grad(h, alpha, n, target):
    corr = _corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    n_corr = len(corr)
    shifts = np.arange(n_corr) - (n - 1)
    grad = np.zeros(n)
    for k in range(n_corr):
        sk = shifts[k]
        wk = w[k]
        if sk >= 0:
            grad[:n - sk] += wk * (1.0 - h[sk:])
            grad[sk:] += -wk * h[:n - sk]
        else:
            ss = -sk
            grad[ss:] += wk * (1.0 - h[:n - ss])
            grad[:n - ss] += -wk * h[ss:]
    grad += 160.0 * (float(np.sum(h)) - target)
    return grad


def _lbfgs_opt(h0, n, dx, alpha, bounds, maxiter, ftol):
    target = n / 2.0
    res = minimize(_smooth_obj, h0, args=(alpha, n, target),
                   jac=_smooth_grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-9,
                            'maxfun': maxiter * 3})
    hc = _repair(res.x, n)
    return hc, _c5(hc, dx)


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _eq_con_jac(x, target):
    return np.ones(len(x))


def _final_slsqp(h0, n, dx, bounds, cons, target):
    alpha = 10000
    res = minimize(_smooth_obj, h0, args=(alpha, n, target),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 600, 'ftol': 1e-17, 'iprint': 0})
    hc = _repair(res.x, n)
    return hc, _c5(hc, dx)


def _coord_descent(h, n, dx, t_end, steps_per_sweep=60):
    local_best = _c5(h, dx)
    deltas = np.array([0.001, 0.002, 0.004, 0.008, 0.015, 0.03])
    all_d = np.concatenate([deltas, -deltas])
    no_improve = 0
    swaps = 0
    while time.time() < t_end and no_improve < 5 and swaps < steps_per_sweep * 20:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            vi = h[i]
            for d in all_d:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                if need > 0:
                    order = np.argsort(h)
                else:
                    order = np.argsort(-h)
                for j in order[:30]:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0.0 or nj > 1.0:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5_try = _c5(h, dx)
                    if c5_try < local_best - 1e-14:
                        local_best = c5_try
                        improved = True
                        swaps += 1
                        break
                    else:
                        h[i] = vi
                        h[j] = nj - need
            if improved and swaps >= steps_per_sweep:
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    return h, local_best


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Multi-scale L-BFGS-B from initial
    p1_end = t_start + min(budget_s * 0.20, 200)
    alphas = [30, 60, 100, 200, 400, 800, 1500, 3000]
    for alpha in alphas:
        if time.time() > p1_end:
            break
        hc, c5 = _lbfgs_opt(best_h, n, dx, alpha, bounds, 400, 1e-17)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 2: Multi-start from perturbed copies
    p2_end = t_start + min(budget_s * 0.60, 550)
    n_restarts = 10
    scales = [0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 1.0]
    for r in range(n_restarts):
        if time.time() > p2_end:
            break
        sc = scales[r % len(scales)]
        pert = best_h + rng.randn(n) * sc
        pert = _repair(pert, n)
        for alpha in [100, 300, 800]:
            if time.time() > p2_end:
                break
            hc, c5 = _lbfgs_opt(pert, n, dx, alpha, bounds, 250, 1e-16)
            if c5 < best_c5:
                best_c5, best_h = c5, hc

    # Phase 3: High-alpha refinement (approaching true max)
    p3_end = t_start + min(budget_s * 0.80, 800)
    for alpha in [3000, 6000, 12000, 25000]:
        if time.time() > p3_end:
            break
        hc, c5 = _lbfgs_opt(best_h, n, dx, alpha, bounds, 500, 1e-18)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 4: Coordinate descent with constraint-preserving two-point moves
    p4_end = t_start + budget_s - 15
    if time.time() < p4_end:
        h_cd = best_h.copy()
        h_cd, cd_best = _coord_descent(h_cd, n, dx, p4_end, steps_per_sweep=40)
        if cd_best < best_c5:
            best_c5, best_h = cd_best, h_cd

    # Phase 5: Final SLSQP polish
    if time.time() < t_start + budget_s - 3:
        cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
        try:
            hc, c5 = _final_slsqp(best_h, n, dx, bounds, cons, target)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 6: Tiny final nudge
    if time.time() < t_start + budget_s - 1:
        h_nud = best_h.copy()
        fine_d = np.array([0.0003, 0.0007, 0.0015, 0.003, -0.0003, -0.0007,
                           -0.0015, -0.003])
        for i in range(n):
            if time.time() > t_start + budget_s - 1:
                break
            vi = h_nud[i]
            for d in fine_d:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                if need > 0:
                    order = np.argsort(h_nud)
                else:
                    order = np.argsort(-h_nud)
                for j in order[:15]:
                    if j == i:
                        continue
                    nj = h_nud[j] + need
                    if nj < 0.0 or nj > 1.0:
                        continue
                    h_nud[i] = ni
                    h_nud[j] = nj
                    c5_try = _c5(h_nud, dx)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_nud.copy()
                        break
                    else:
                        h_nud[i] = vi
                        h_nud[j] = nj - need

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
