import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h)) * 2.0 / n)

def _smooth_max_grad(h, alpha):
    n = len(h)
    corr = _corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    exp_s = np.exp(s)
    sum_exp = exp_s.sum()
    obj = m + np.log(sum_exp) / alpha
    w = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            if shift == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[:n - shift] += wk * (1.0 - h[shift:])
                grad[shift:] -= wk * h[:n - shift]
        else:
            s2 = -shift
            if s2 == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[s2:] += wk * (1.0 - h[:n - s2])
                grad[:n - s2] -= wk * h[s2:]
    return obj, grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if s < 1e-12:
            return np.full(n, 0.5)
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _proj_grad_descent(h, n, alpha, n_iter, lr_init, budget_end, rng):
    h = h.copy()
    lr = lr_init
    best_obj = _c5(h, n)
    for it in range(n_iter):
        if time.time() > budget_end:
            break
        obj, grad = _smooth_max_grad(h, alpha)
        norm_g = np.sqrt(np.sum(grad ** 2))
        if norm_g < 1e-12:
            break
        grad_n = grad / norm_g
        for _bt in range(30):
            h_new = _repair(h - lr * grad, n)
            obj_new, _ = _smooth_max_grad(h_new, alpha)
            if obj_new < obj + 1e-14:
                h = h_new
                obj = obj_new
                break
            lr *= 0.5
        c = _c5(h, n)
        if c < best_obj:
            best_obj = c
        if lr < 1e-10:
            break
        lr = min(lr_init, lr * 1.05)
    return h, best_obj

def _slsqp(h, n, alpha, maxiter):
    def obj(x):
        o, _ = _smooth_max_grad(x, alpha)
        return o
    con = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
           'jac': lambda x: np.ones(len(x))}
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _pair_search(h, n, dx, n_passes, budget_end, rng):
    h = h.copy()
    best = _c5(h, n)
    deltas = [0.002, 0.005, 0.01, 0.02, 0.04, -0.002, -0.005, -0.01, -0.02, -0.04]
    for _p in range(n_passes):
        if time.time() > budget_end:
            break
        improved = False
        temp = 0.005 / (1 + _p * 0.2)
        for _a in range(8):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in deltas:
                ni, nj = hi + d, hj - d
                if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, n)
                if c < best or rng.rand() < np.exp(-(c - best) / max(temp, 1e-15)):
                    if c < best:
                        best = c
                        improved = True
                else:
                    h[i], h[j] = hi, hj
        if not improved and _p > 40:
            break
    return h, best

    else:

def _optimize_n(n, h_start, seed, budget_end):

    best_h = h_start.copy()
