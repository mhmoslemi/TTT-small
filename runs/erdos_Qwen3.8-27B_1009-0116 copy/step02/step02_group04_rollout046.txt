def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _softmax_max(vals, beta):
    m = np.max(vals)
    s = beta * (vals - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / beta

def _obj_smooth(x, dx, beta):
    c = _corr(x)
    return np.max(c) * dx + (np.max(_softmax_max(c * dx, beta)) - np.max(c) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

        return h, _c5(h, dx)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, dx, alpha):
    c = _corr(x) * dx
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha
