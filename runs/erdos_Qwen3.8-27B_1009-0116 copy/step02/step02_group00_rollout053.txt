                h[idx] += add
                diff -= add
            else:

                h[idx] -= sub
                diff += sub

        for i in range(n):
            if time.time() > p5_end:
                break
            vi = h[i]
            for d in fine_d:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                if need > 0:
                    order = np.argsort(h)
                else:
                    order = np.argsort(-h)
                for j in order[:12]:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0.0 or nj > 1.0:
                        continue
                    h[i], h[j] = ni, nj
                    c5_try = _c5(h, dx)
                    if c5_try < local_best - 1e-15:
                        local_best = c5_try
                        break
                    else:
                        h[i], h[j] = vi, nj
                else:
                    continue
                break
        if local_best < best_c5:
            best_c5, best_h = local_best, h

    if time.time() < t_start + budget_s - 3:
        try:
            res = minimize(_obj_lse, best_h, args=(10000, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-17})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
