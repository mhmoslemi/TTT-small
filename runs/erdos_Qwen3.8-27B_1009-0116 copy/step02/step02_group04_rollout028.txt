            for idx in np.argsort(1.0 - h):
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                if diff < 1e-9: break
        else:
            for idx in np.argsort(h):
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if diff > -1e-9: break
    return h

def _sobj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, dx, alpha, maxit, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5}]
    try:
        r = minimize(_sobj, h, args=(alpha,), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        h2 = _repair(r.x, n)
        return h2, _c5(h2, dx)
    except:
        return h, _c5(h, dx)

def _local_sa(h, n, dx, rng, deadline, T0=0.01, T_end=1e-8):
    """Simulated annealing local search on pair exchanges."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    c5 = best_c5
    total_steps = 50000
    for step in range(total_steps):
        if time.time() >= deadline:
            break
        T = T0 * (T_end / T0) ** (step / total_steps)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = (2.0 * rng.random() - 1.0) * 0.01
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        new_c5 = _c5(h, dx)
        delta = new_c5 - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            for idx in np.argsort(1.0 - h):
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                if diff < 1e-9: break
        else:
