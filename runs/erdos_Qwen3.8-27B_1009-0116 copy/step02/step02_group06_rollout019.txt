import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx), c

def _smm(c, a):
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _smm_obj(x, a):
    return _smm(np.correlate(x, 1.0 - x, mode='full'), a)

def _eq(x, t):
    return np.sum(x) - t

def _slsqp(h, n, a, mi):
    b = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq, 'args': (n / 2.0,)}]
    try:
        r = minimize(_smm_obj, h, args=(a,), method='SLSQP', bounds=b,
                     constraints=cons, options={'maxiter': mi, 'ftol': 1e-14})
        return _repair(r.x, n)
    except:
        return h

def _gdiff(h, i, j, n):
    ks = np.arange(-(n - 1), n)
    ncc = 2 * n - 1
    def _g1(idx):
        g = np.zeros(ncc)
        ik = idx + ks
        v1 = (ik >= 0) & (ik < n) & (ks != 0)
        g[v1] = 1.0 - h[ik[v1]]
        g[ks == 0] = 1.0 - 2.0 * h[idx]
        ik2 = idx - ks
        v2 = (ik2 >= 0) & (ik2 < n) & (ks != 0)
        g[v2] -= h[ik2[v2]]
        return g
    return _g1(i) - _g1(j)

def _sweep(h, n, dx, corr, nco, nfi, fw):
