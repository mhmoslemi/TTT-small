- If j = i+L: term is h[i](1-h[j]) → (h[i]+d)(1-h[j]+d) = h[i] - h[i]h[j] + d(1-h[j]) + d*h[i] + d^2*0... 
  Wait: (h[i]+d)(1-h[j]+d) = h[i](1-h[j]) + d(1-h[j]) + h[i]*d + d^2
  Change: d(1-h[j]) + d*h[i] + d^2
  But we already counted d(1-h[j]) from i's first-index change and +d*h[i] from j's second-index change.
  So the ADDITIONAL cross term is: d^2

- If i = j+L: term is h[j](1-h[i]) → (h[j]-d)(1-h[i]-d) = h[j](1-h[i]) - d(1-h[i]) - d*h[j] + d^2
  Change: -d(1-h[i]) - d*h[j] + d^2
  We already counted -d(1-h[i]) from j's first-index change and -d*h[j] from i's second-index change.
  So ADDITIONAL cross term is: d^2

So in both linked cases, there's an additional +d^2.

Therefore:
O_L(d) = O_L(0) + d * g_L + d^2 * s_L

where:
g_L = (1-h[i+L]) if i+L valid else 0) - h[i-L] if i-L valid else 0) - (1-h[j+L]) if j+L valid else 0) + h[j-L] if j-L valid else 0)
s_L = 1 if (j==i+L and 0<=i<n) or (i==j+L and 0<=j<n) [need to check validity], 0 otherwise

Wait, s_L = 1 only if the pair (i,j) actually appears together in the sum for lag L. That means:
- j = i+L AND 0<=i<n AND 0<=j<n (both are in [0,n))
- OR i = j+L AND 0<=j<n AND 0<=i<n

Since 0<=i<n and 0<=j<n are always true (they're our indices), the condition is just j==i+L or i==j+L.

But we also need the pair to be a valid term: for lag L, the valid first indices p satisfy 0<=p<n and 0<=p+L<n.
- If j=i+L: the term is h[i](1-h[i+L]) = h[i](1-h[j]). Valid if 0<=i<n and 0<=i+L<n. Since 0<=i<n always and i+L=j which is in [0,n), this is valid.
- If i=j+L: the term is h[j](1-h[j+L]) = h[j](1-h[i]). Valid if 0<=j<n and 0<=j+L<n. Since 0<=j<n always and j+L=i which is in [0,n), this is valid.

So s_L = 1 if j==i+L or i==j+L, else 0. At most 2 lags have s_L=1.

Great! So for each pair (i,j):

2. Compute s (vector of 139 values, mostly 0, at most 2 ones)

                hi, hj = h[i], h[j]
                if hi <= 0 and hj >= 1:
                    continue
                if hi >= 1 and hj <= 0:

                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_max - d_min < 1e-10:
                    continue
                # Compute g_L for all lags
                # g_L = (1-h[i+L]) if valid - h[i-L] if valid - (1-h[j+L]) if valid + h[j-L] if valid
                g = np.zeros(2 * n - 1)
                # i as first at lag L: i+L valid, L from -(i) to n-1-i, index in correlate: L + n - 1
                # i as second at lag L: i-L valid, L from i-n+1 to i
                # j as first: j+L valid
                # j as second: j-L valid

                Lags = np.arange(-(n - 1), n)  # lag values

                g[mask_i1] += (1.0 - h[i + Lags[mask_i1]])
                
                # i as second: 0 <= i-L < n => i-n < L <= i => i-n+1 <= L <= i
                mask_i2 = (Lags >= i - n + 1) & (Lags <= i)
                g[mask_i2] += -h[i - Lags[mask_i2]]
                
                # j as first: 0 <= j+L < n
                mask_j1 = (Lags >= -j) & (Lags < n - j)
                g[mask_j1] -= (1.0 - h[j + Lags[mask_j1]])
                
                # j as second: 0 <= j-L < n
                mask_j2 = (Lags >= j - n + 1) & (Lags <= j)
                g[mask_j2] += h[j - Lags[mask_j2]]
                
                # s_L: 1 if j==i+L or i==j+L
                s = np.zeros(2 * n - 1)
                # j = i + L => L = j - i
                L1 = j - i
                if abs(L1) <= n - 1:
                    s[L1 + n - 1] = 1.0
                # i = j + L => L = i - j

                if abs(L2) <= n - 1:
                    s[L2 + n - 1] = 1.0
                # (L1 and L2 are negatives of each other, and they correspond to different lags
                # unless i==j which we excluded)
                
                # Now find d* minimizing max_L(O_L + d*g_L + d^2*s_L)
                # Ternary search on [d_min, d_max]
                a, b = d_min, d_max
                for _ in range(40):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    f1 = float(np.max(O + m1 * g + m1*m1 * s))
                    f2 = float(np.max(O + m2 * g + m2*m2 * s))
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                d_star = 0.5 * (a + b)
                
                # Evaluate improvement
                c_new = float(np.max(O + d_star * g + d_star*d_star * s))
                if c_new < bc - 1e-12:
                    h[i] += d_star
                    h[j] -= d_star
                    O = O + d_star * g + d_star * d_star * s
                    # Recompute to avoid drift
                    O = _overlap_vec(h)
                    bc = float(O.max()) * dx
                    improved = True
        
        if not improved and sweep > 10:
            break
    
    return h, _c5(h, dx)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.clip(y - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            a = min(diff, 1.0 - x[k]) if diff > 0 else max(diff, -x[k])
            x[k] += a
            diff -= a
            if abs(diff) < 1e-14:
                break
    return x

def _overlap(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def _compute_inf(h, n):
    inf = np.zeros((n, 2 * n - 1))
    L = np.arange(-(n - 1), n)
    for p in range(n):
        m1 = (L >= -p) & (L < n - p)
        inf[p, m1] += (1.0 - h[p + L[m1]])
        m2 = (L >= p - n + 1) & (L <= p)
        inf[p, m2] += -h[p - L[m2]]
    return inf

def _pair_cd(h, n, rng, n_sweeps, t0, deadline):
    dx = 2.0 / n
    O = _overlap(h)
    bc = float(O.max()) * dx
    L = np.arange(-(n - 1), n)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        inf = _compute_inf(h, n)
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_max - d_min < 1e-10:
                    continue
                g = inf[i] - inf[j]
                # s vector: positions where i,j are linked
                s = np.zeros(2 * n - 1)
                L1 = j - i  # lag where j = i + L
                if abs(L1) <= n - 1:
                    s[L1 + n - 1] = 1.0
                L2 = i - j  # lag where i = j + L
                if abs(L2) <= n - 1:
                    s[L2 + n - 1] = 1.0
                
                # Ternary search for optimal d
                a, b = d_min, d_max
                for _ in range(25):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    f1 = float(np.max(O + m1 * g + m1 * m1 * s))
                    f2 = float(np.max(O + m2 * g + m2 * m2 * s))
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                d = 0.5 * (a + b)
                c_new = float(np.max(O + d * g + d * d * s))
                if c_new < bc / dx - 1e-10:  # compare in un-scaled units
                    h[i] += d
                    h[j] -= d
                    O = _overlap(h)
                    bc = float(O.max()) * dx
                    improved = True
        if not improved and sweep > 5:
            break
    return h, bc

def _softmax_fista(h0, alpha, target, n_iters, t0, deadline):
    """Simple softmax-smoothed FISTA."""
    n = len(h0)
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bc = _c5(x, 2.0 / n)
    bx = x.copy()
    
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        # Gradient of softmax-smoothed max
        c = _overlap(z)
        m = float(c.max())
        s = np.clip(alpha * (c - m), -300.0, 300.0)
        w = np.exp(s)
        w /= w.sum()
        # Gradient w.r.t. h
        g = np.zeros(n)
        L = np.arange(-(n - 1), n)
        for p in range(n):
            m1 = (L >= -p) & (L < n - p)
            g[p] += float(w[m1].dot(1.0 - z[p + L[m1]]))
            m2 = (L >= p - n + 1) & (L <= p)
            g[p] += float(w[m2].dot(-z[p - L[m2]]))
        
        # FISTA step
        step = 1.0 / (2.0 * alpha * n + 4.0)
        y = z - step * g
        xn = _proj(y, target)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, 2.0 / n)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _scon(x, target):
    return float(np.sum(x)) - target

    # Phase 4: Random perturbations + pair CD
