from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - x[i + lag])
                g[i + lag] -= wk * x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    return g

def _fista(h0, alpha, target, iters, t_end):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    best_time = time.time()
    for it in range(iters):
        if time.time() > t_end:
            break
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 6:
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _cd_linesearch(h, dx, n, n_sweeps, t_end):

    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):

                # Golden section search

                h_save = np.array([h[i], h[j]])
                h[i] += a; h[j] -= a
                fa = _c5(h, dx)
                h[i] += (b - a); h[j] -= (b - a)
                fb = _c5(h, dx)

                        fb = _c5(h, dx)

    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    one_minus_x = 1.0 - x
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:

            g[:n - lag] -= wk * x[lag:]
        else:
            L = -lag
            g[:n - L] += wk * one_minus_x[L:]
            g[L:] -= wk * x[:n - L]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    omx = 1.0 - x
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[lag:] += wk * omx[:n - lag]
            g[:n - lag] -= wk * x[lag:]
        else:
            L = -lag
            g[:n - L] += wk * omx[L:]
            g[L:] -= wk * x[:n - L]
    return g

def _fista(h0, alpha, target, iters, t_end):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        if time.time() > t_end:
            break
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 6:
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _cd_ls(h, dx, n, n_sweeps, t_end):
    bc = _c5(h, dx)
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = -h[i]
                hi = 1.0 - h[j]
                if hi - lo < 1e-10:
                    continue
                hi_arr = [h[i], h[j]]
                a = lo + (1.0 - gr) * (hi - lo)
                b = lo + gr * (hi - lo)
                h[i] = hi_arr[0] + a
                h[j] = hi_arr[1] - a
                fa = _c5(h, dx)
                h[i] = hi_arr[0] + b
                h[j] = hi_arr[1] - b
                fb = _c5(h, dx)
                for _ in range(22):
                    if hi - lo < 1e-8:
                        break
                    if fa < fb:
                        hi = b
                        b = a
                        fb = fa
                        a = lo + (1.0 - gr) * (hi - lo)
                        h[i] = hi_arr[0] + a
                        h[j] = hi_arr[1] - a
                        fa = _c5(h, dx)
                    else:
                        lo = a
                        a = b
                        fa = fb
                        b = lo + gr * (hi - lo)
                        h[i] = hi_arr[0] + b
                        h[j] = hi_arr[1] - b
                        fb = _c5(h, dx)
                d_best = 0.5 * (lo + hi)
                h[i] = hi_arr[0] + d_best
                h[j] = hi_arr[1] - d_best
                fc = _c5(h, dx)
                if fc < bc - 1e-12:
                    bc = fc
                    improved = True
                else:
                    h[i] = hi_arr[0]
                    h[j] = hi_arr[1]
        if not improved:
            break
    return h, bc

    h = np.zeros(n)
    idx = np.zeros(n // 2, dtype=int)
    if pat == 0:
        idx = np.arange(0, n, 2)[:n // 2]

        idx = np.sort(rng.choice(n, n // 2, replace=False))
    else:

    h[idx] = 1.0
    return h
