def _subgrad(h, n, tol=1e-4):
    c = _corr(h)
    amax = np.max(c)
    active = np.where(c >= amax - tol)[0]
    nact = len(active)
    if nact == 0:
        active = [np.argmax(c)]
        nact = 1
    grad = np.zeros(n)
    for k in active:
        # dc[k]/dh[i] = 1-2*h[k-i] if 0<=k-i<n else 0
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        for i in range(lo, hi+1):
            grad[i] += 1.0 - 2.0 * h[k-i]
    grad /= nact
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_obj(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx

def _slsqp_con(x, target):
    return np.sum(x) - target

def _slsqp(h, n, dx, alpha, maxiter=300, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'args': (target,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _coord_decent(h, n, dx, deadline, max_sweeps=500):
    h = h.copy()
    best_c5 = _c5(h, dx)
    gr = (np.sqrt(5) + 1) / 2
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_x = h[i]
            c_cur = _corr(h)
            # Build coefficients for 1D subproblem
            # c[k](x) = a_k*x^2 + b_k*x + c_k
            a = np.zeros(2*n-1)
            b = np.zeros(2*n-1)
            c = c_cur.copy()
            for k in range(2*n-1):
                ki = k - i
                if 0 <= ki < n:
                    if ki != i:
                        b[k] = 1.0 - 2.0 * h[ki]
                        c[k] -= b[k] * old_x
                    else:  # k == 2i
                        a[k] = -1.0
                        b[k] = 1.0
                        c[k] -= (old_x - old_x*old_x)
            # Golden section to find x minimizing max_k (a_k*x^2+b_k*x+c_k)
            lo, hi = 0.0, 1.0
            x1 = hi - (hi-lo)/gr
            x2 = lo + (hi-lo)/gr
            f1 = np.max(a*x1*x1 + b*x1 + c)
            f2 = np.max(a*x2*x2 + b*x2 + c)
            for _ in range(40):
                if f1 < f2:
                    hi, x2, f2 = x2, x1, f1
                    x1 = hi - (hi-lo)/gr
                    f1 = np.max(a*x1*x1 + b*x1 + c)
                else:
                    lo, x1, f1 = x1, x2, f2
                    x2 = lo + (hi-lo)/gr
                    f2 = np.max(a*x2*x2 + b*x2 + c)
            x_best = (lo + hi) / 2.0
            if abs(x_best - old_x) > 1e-7:
                h[i] = x_best
                nc = _c5(h, dx)
                if nc < best_c5:
                    best_c5 = nc

                # Note: we keep h[i]=x_best even if c5 didn't improve,

        if not improved_any and sweep > 5:

    best_c5 = _c5(h, dx)
    best_h = h.copy()

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
