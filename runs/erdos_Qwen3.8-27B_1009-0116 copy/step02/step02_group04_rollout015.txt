        return h

def _powell_refine(h, n, dx):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def obj(x):
        if np.any(x < -1e-9) or np.any(x > 1.0 + 1e-9):
            return 10.0
        return _c5(x, dx)
    try:
        res = minimize(obj, h, method='Powell', bounds=bounds,
                       options={'maxiter': 400, 'xtol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _de_explore(n, dx, rng, deadline, pop_size=30):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def obj(x):
        if abs(np.sum(x) - target) > 0.5:
            return 10.0 + abs(np.sum(x) - target)
        hc = _repair(x, n)
        return _c5(hc, dx)
    try:
        res = differential_evolution(obj, bounds, seed=rng.randint(100000),
                                     maxiter=80, popsize=pop_size, tol=1e-10,
                                     mutation=(0.3, 1.2), recombination=0.9,
                                     polish=False, update_interval=1,
                                     seed=rng.randint(100000))
        return _repair(res.x, n)
    except Exception:
        return None

def _targeted_local(h, n, dx, rng, deadline):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for scale in scales:
        stalls = 0
        max_iters = 3000
        for it in range(max_iters):
            if time.time() >= deadline:
                return h, best_c5
            corr = _corr(h)
            m = np.max(corr)
            top_k = np.argsort(corr)[-6:]
            for lag in top_k:
                shift = lag - (n - 1)
                i_min = max(0, -shift)
                i_max = min(n, n - shift)
                affected = list(range(i_min, i_max))
                if len(affected) < 2:
                    continue
                i = affected[rng.randint(len(affected))]
                j = affected[rng.randint(len(affected))]
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                hi, hj = h[i], h[j]
                ni, nj = hi + d, hj - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5_new = _c5(h, dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    stalls = 0
                else:
                    h[i], h[j] = hi, hj
                    stalls += 1
                if stalls > 600:
                    break
            if stalls > 600:
                break
        # Also try swaps
        for _ in range(200):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c5_new = _c5(h, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
            else:
                h[i], h[j] = h[j], h[i]
    return h, best_c5

def _multi_start_local(h, n, dx, rng, deadline):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    while time.time() < deadline:
        # Generate perturbation
        norm = rng.randn(n)
        norm = norm / np.linalg.norm(norm)
        for scale in [0.03, 0.08, 0.15]:
            cand = _repair(best_h + scale * norm, n)
            c5 = _c5(cand, dx)
            if c5 < best_c5:
                # Refine from here
                t_sub = min(time.time() + 15, deadline)
                cand, c5 = _targeted_local(cand, n, dx, rng, t_sub)
                if c5 < best_c5:
                    best_h = cand.copy()
                    best_c5 = c5
                break
            if time.time() >= deadline:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-start (120s)
    t_end = t_start + 120
    for alpha in [30, 100, 300, 800]:
        if time.time() > t_end or budget_s - (time.time() - t_start) < 700:
            break
        hc = _slsqp_refine(best_h, n, dx, alpha, 600)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 2: Differential evolution global exploration (250s)
    if budget_s - (time.time() - t_start) > 500:
        t_end = t_start + 370
        tdeadline = t_end - 10
        cand = _de_explore(n, dx, rng, tdeadline)
        if cand is not None:
            c5 = _c5(cand, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = cand

    # Phase 3: Powell local optimization (150s)
    t_end = t_start + 520
    for _ in range(4):
        if time.time() > t_end or budget_s - (time.time() - t_start) < 350:
            break
        hc = _powell_refine(best_h, n, dx)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 4: Multi-start targeted local search (250s)
    if budget_s - (time.time() - t_start) > 300:
        t_end = t_start + 770
        best_h, best_c5 = _multi_start_local(best_h, n, dx, rng, t_end)

    # Phase 5: SLSQP high-alpha refinement (150s)
    if budget_s - (time.time() - t_start) > 150:
        t_end = t_start + 920
        for alpha in [1000, 3000, 8000]:
            if time.time() > t_end or budget_s - (time.time() - t_start) < 60:
                break
            hc = _slsqp_refine(best_h, n, dx, alpha, 1000)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

    # Phase 6: Final fine-grained targeted local (100s)
    if budget_s - (time.time() - t_start) > 50:
        t_end = t_start + 1020
        best_h, best_c5 = _targeted_local(best_h, n, dx, rng, t_end)

    # Phase 7: Ultimate SLSQP polish
    if budget_s - (time.time() - t_start) > 20:
        hc = _slsqp_refine(best_h, n, dx, 20000, 1500)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
