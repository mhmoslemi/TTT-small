```python
    if s < 1e-12:
```

This is a much more structured local search. Let me refine the algorithm:

**Algorithm: Structured Local Search (SLS)**
1. Compute correlations c and interaction matrix M
2. For each pair (i,j):
   a. Compute g = M[i] - M[j]
   b. Find optimal δ in [−h[j], 1−h[i]] that minimizes max(c + δg)
   c. Track the best (i,j,δ)
3. Apply the best move
4. Repeat until no improvement found or time expires

The 1D optimization in step 2b: minimize f(δ) = max_r(c[r] + δ·g[r]) over δ ∈ [δ_min, δ_max].
This is the minimum of a convex piecewise-linear function. The minimum occurs at:
- An endpoint, or
- A point where two "active" lines cross

To find it efficiently:
- The active set is the set of r where c[r] + δ·g[r] is near the maximum
- We can find the minimum by: among the lines c[r] + δ·g[r], find where the upper envelope has its minimum

A simple but effective approach:
- Compute the maximum of c (call it c_max)
- The "active" correlations are those with c[r] > c_max - ε for some small ε
- For each pair of active correlations (r1, r2), compute δ* = (c[r1]-c[r2])/(g[r2]-g[r1])
- Evaluate f at all such δ* and at the endpoints
- Pick the best

With only a few active correlations (say top 5-10), this is O(k²) per pair where k is the number of active correlations. Very fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _cv(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_cv(h)) * dx)

def _sobj(x, n, a):
    c = _cv(x)
    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(a*(c-m),-500,500))))/a

def _sg(x, n, a):
    c = _cv(x)
    m = np.max(c)
    p = np.exp(np.clip(a*(c-m),-500,500))
    p /= p.sum()
    g = np.zeros(n)
    for r in range(2*n-1):
        if p[r] < 1e-14:
            continue
        s = r-(n-1)
        lo1,hi1 = max(0,-s),min(n,n-s)
        if lo1<hi1:
            g[lo1:hi1] += p[r]*(1.0-x[lo1+s:hi1+s])
        lo2,hi2 = max(0,s),min(n,n+s)
        if lo2<hi2:
            g[lo2:hi2] -= p[r]*x[lo2-s:hi2-s]
    return g

def _sog(x, n, a):
    return _sobj(x,n,a), _sg(x,n,a)

def _rep(h, n):
    h = np.array(h,dtype=float).copy()
    h = np.clip(h,0.0,1.0)
    t = n/2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n,0.5)
    h *= t/s
    h = np.clip(h,0.0,1.0)
    d = t - h.sum()
    if abs(d) > 1e-10:
        o = np.argsort(1.0-h) if d>0 else np.argsort(h)
        for idx in o:
            if d>0:
                a=min(d,1.0-h[idx]); h[idx]+=a; d-=a
            else:
                a=min(-d,h[idx]); h[idx]-=a; d+=a
            if abs(d)<1e-10:
                break
    return h

        s = r-(n-1)
        lo,hi = max(0,-s),min(n,n-s)

        lo,hi = max(0,s),min(n,n+s)
        if lo<hi:

    return M

def _od(c, g, lo, hi, ti):
    cands = [lo,hi,0.0]
    for a in range(len(ti)):
        for b in range(a+1,len(ti)):

            dg = g[r1]-g[r2]

                if lo<d<hi:

    ca = np.clip(np.array(cands),lo,hi)
    v = (c[None,:]+ca[:,None]*g[None,:]).max(axis=1)
    bi = np.argmin(v)
    return ca[bi], v[bi]

def _slp(h, n, dx, mx, rng):
    h = h.copy()
    t0 = time.time()
```
```python
    while time.time()-t0 < mx:
        c = _cv(h)
        cm = np.max(c)
        M = _om(h,n)
        ti = np.argpartition(-c,10)[:10]
        bv,bm = cm,None
        for i in range(n):
            for j in range(i+1,n):
                lo = max(-h[i],h[j]-1.0)
                hi = min(1.0-h[i],h[j])
                if hi-lo<1e-12:
                    continue
                d,v = _od(c,M[i]-M[j],lo,hi,ti)
                if v<bv-1e-12:
                    bv,bm = v,(i,j,d)
        if bm is None:
            break  # Convergence
        h[bm[0]]+=bm[2]; h[bm[1]]-=bm[2]
    h=_rep(h,n)
    return h,_c5(h,dx)
```

- No improvement found (convergence)
