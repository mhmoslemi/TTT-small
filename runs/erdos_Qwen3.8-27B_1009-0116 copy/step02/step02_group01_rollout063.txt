import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_obj_grad(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    exps = np.exp(s)
    w = exps / (exps.sum() + 1e-30)
    n = len(x)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-8:
            continue
        s = k - (n - 1)
        if s >= 0:
            for i in range(n - s):
                grad[i + s] += wk * (1.0 - x[i])
                grad[i] += -wk * x[i + s]
        else:
            d = -s
            for i in range(d, n):
                grad[i] += -wk * x[i - d]
                grad[i - d] += wk * (1.0 - x[i])
    return m + np.log(np.sum(exps)) / alpha, grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    b = min(-diff, h[idx])
                    h[idx] -= b
                    diff += b
                if abs(diff) < 1e-12:
                    break
    return h

def _pair_sweep(h, c5_cur, dx, n, deltas, max_iter, t0, budget_s, margin):
    improved = True
    while improved:
        improved = False
        for i in range(n):
            if time.time() - t0 > budget_s - margin:
                return h, c5_cur
            hi = h[i]
            for j in range(i + 1, n):
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < c5_cur - 1e-15:
                        c5_cur = c
                        hi, hj = ni, nj
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
                h[j] = hj
            h[i] = hi
    return h, c5_cur

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_obj_grad(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    exps = np.exp(s)
    w = exps / (exps.sum() + 1e-30)
    n = len(x)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-8:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            for i in range(n - sh):
                grad[i + sh] += wk * (1.0 - x[i])
                grad[i] += -wk * x[i + sh]
        else:
            d = -sh
            for i in range(d, n):
                grad[i] += -wk * x[i - d]
                grad[i - d] += wk * (1.0 - x[i])
    return m + np.log(np.sum(exps)) / alpha, grad

def _cons_fun(x, target):
    return np.sum(x) - target

def _cons_jac(x, target):
    return np.ones(len(x))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    b = min(-diff, h[idx])
                    h[idx] -= b
                    diff += b
                if abs(diff) < 1e-12:
                    break
    return h

def _pair_sweep(h, c5_cur, dx, n, deltas, t0, deadline):
    improved = True
    while improved:

        for i in range(n):

                return h, c5_cur
            hi = h[i]
            for j in range(i + 1, n):

                    c = _c5(h, dx)
                    if c < c5_cur - 1e-15:
                        c5_cur = c
                        hi, hj = ni, nj
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
                h[j] = hj
            h[i] = hi
    return h, c5_cur
