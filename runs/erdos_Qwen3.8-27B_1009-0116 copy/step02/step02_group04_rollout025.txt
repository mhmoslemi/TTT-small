    """Compute coef[k, j] = coefficient of h[j] in corr[k]."""
    # corr[k] = sum_i h[i] * (1-h[i+shift]) where shift = k-(n-1)
    # coef of h[j] in corr[k]:
    #   from term i=j: (1 - h[j+shift]) if 0 <= j+shift < n
    #   from term i=j-shift: -h[j-shift] if 0 <= j-shift < n
    shifts = np.arange(2*n-1) - (n-1)  # -（n-1) to n-1
    # coef[k,j] = (1-h[j+shifts[k]]) * (0<=j+shifts[k]<n) - h[j-shifts[k]] * (0<=j-shifts[k]<n)

    # j ranges 0..n-1, shift ranges -(n-1)..n-1
    j_idx = np.arange(n)[:, None]  # (n, 1)
    sh_idx = shifts[None, :]  # (1, 2n-1)
    
    jp = j_idx + sh_idx  # (n, 2n-1): j + shift
    jm = j_idx - sh_idx  # (n, 2n-1): j - shift

    valid_m = (jm >= 0) & (jm < n)

    # coef[j, k] = coefficient of h[j] in corr[k]
    return coef.T  # (2n-1, n)

def _correlate_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_correlate_full(h)) * dx)

def _compute_coef(h, n):
    shifts = np.arange(2 * n - 1) - (n - 1)
    j_idx = np.arange(n)[:, None]
    sh_idx = shifts[None, :]
    jp = j_idx + sh_idx
    jm = j_idx - sh_idx
    valid_p = (jp >= 0) & (jp < n)
    valid_m = (jm >= 0) & (jm < n)
    hjp = np.where(valid_p, h[np.clip(jp, 0, n - 1)], 0.0)
    hjm = np.where(valid_m, h[np.clip(jm, 0, n - 1)], 0.0)
    return ((valid_p * (1.0 - hjp)) - hjm).T

def _optimal_delta(corr, b, lo, hi, n_iter=35):
    if hi - lo < 1e-14:
        return 0.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr + m1 * b) <= np.max(corr + m2 * b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

            break

        for idx in range(len(pairs)):
            if time.time() >= deadline:
