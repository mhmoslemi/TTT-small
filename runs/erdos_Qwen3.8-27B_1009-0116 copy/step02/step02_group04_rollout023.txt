from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr_full(h)) * dx

def _smooth_obj(x, alpha):
    corr = _corr_full(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _corr_gradient(h, n, s):
    """Gradient of c(s) = sum_i h[i]*(1-h[i-s]) w.r.t. h"""
    g = np.zeros(n)
    for k in range(n):
        val = 0.0
        js = k - s
        if 0 <= js < n:
            val += (1.0 - h[js])
        jt = k + s
        if 0 <= jt < n:
            val -= h[jt]
        g[k] = val
    return g

def _targeted_local_search(h, n, dx, rng, deadline):
    """Local search targeting top correlation lags using gradient info."""
    best_c5 = _c5(h, dx)
    h = h.copy()
    
    while time.time() < deadline:
        corr = _corr_full(h)
        # Find top 5 lags
        topk = 5
        top_idx = np.argsort(corr)[-topk:][::-1]
        top_s = top_idx - (n - 1)  # convert to shift
        
        # Compute gradients for top lags
        grads = np.zeros((topk, n))
        for li, s in enumerate(top_s):
            grads[li] = _corr_gradient(h, n, int(s))
        
        # Find best move: pick k, l, d to minimize max of top correlations
        best_move_val = best_c5 + 1e-10
        best_move = None
        
        n_trials = 50
        for _ in range(n_trials):
            k = rng.randint(n)
            l = rng.randint(n)
            if k == l:
                continue
            # Direction: d for h[k], -d for h[l]
            dir_vec = np.zeros(n)
            dir_vec[k] = 1.0
            dir_vec[l] = -1.0
            
            # Gradient of each top correlation along this direction
            # dc[li]/dd = grads[li] @ dir_vec
            dc = grads @ dir_vec  # shape (topk,)
            
            # We want to find d that minimizes max(corr[li] + d*dc[li])
            # Subject to 0 <= h[k]+d <= 1 and 0 <= h[l]-d <= 1
            d_max = min(1.0 - h[k], h[l])
            d_min = max(-(h[k]), -(1.0 - h[l]))
            
            if d_max - d_min < 1e-10:
                continue
            
            # Optimal d: minimize max over li of (corr_top[li] + d * dc[li])
            # This is a 1D minimax problem
            # Binary search or golden section
            c_top = corr[top_idx]
            
            def f(d):
                return np.max(c_top + d * dc)
            
            # Sample a few candidate d values
            candidates = np.linspace(d_min * 0.99, d_max * 0.99, 20)
            if d_max > 0:
                candidates = np.concatenate([candidates, [d_max * 0.5, d_max * 0.9]])
            if d_min < 0:
                candidates = np.concatenate([candidates, [d_min * 0.5, d_min * 0.9]])
            
            best_d = min(candidates, key=lambda d: f(d))
            best_f = f(best_d)
            
            if best_f * dx < best_move_val:
                best_move_val = best_f * dx
                best_move = (k, l, best_d)
        
        if best_move is None:
            break
        
        k, l, d = best_move
        h[k] += d
        h[l] -= d
        h = np.clip(h, 0.0, 1.0)
        h = _repair(h, n)
        
        new_c5 = _c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            break  # No improvement, stop

    """Gradient of c(s)=sum_i h[i]*(1-h[i-s]) wrt h, with shift s (can be negative)."""

            v += 1.0 - h[js]
        jt = k + s
        if 0 <= jt < n:

        g[k] = v
    return g

    h = h.copy()

        if time.time() >= deadline:

        found = False
        n_trials = 30

            dc = G[:, k] - G[:, l]  # gradient of top corr values along this direction

            d_lo = max(-h[k], h[l] - 1.0)
            d_hi = min(1.0 - h[k], h[l])
            if d_hi - d_lo < 1e-12:

            # The optimal d balances the top two active components

                return np.max(c_top + d * dc)

            a, b = d_lo, d_hi
            for _gs in range(30):
                m1 = a + 0.382 * (b - a)
                m2 = a + 0.618 * (b - a)
                if ff(m1) < ff(m2):
                    b = m2
                else:
                    a = m1
            d_opt = 0.5 * (a + b)

            improved_c = c_top + d_opt * dc

        else:
