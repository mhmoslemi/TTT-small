def _corr_max(h, dx):
    """Compute C5 value for a binary or continuous sequence."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _binary_from_real(reals, n):
    """Convert real-valued sequence to binary with exactly n/2 ones."""
    h = np.zeros(n)
    # Get the top n/2 positions
    k = n // 2
    indices = np.argsort(reals)[-k:]
    h[indices] = 1.0
    return h

def _swap_eval(h, i, j, dx):
    """Evaluate C5 after swapping h[i] and h[j] (binary: 0<->1)."""
    # For binary: h[i]=1, h[j]=0 -> h[i]=0, h[j]=1
    # or h[i]=0, h[j]=1 -> h[i]=1, h[j]=0
    h[i], h[j] = h[j], h[i]
    c5 = _corr_max(h, dx)
    h[i], h[j] = h[j], h[i]  # undo
    return c5

def _tabu_search(h, dx, n, t_end, rng, tabu_len=50):

    n_tabu = 69  # lags
    best_h = h.copy()
    best_c5 = _corr_max(h, dx)
    tabu = {}  # (i,j) -> time when it becomes untabu
    it = 0
    no_improve = 0
    while time.time() < t_end:
        it += 1
        if it % 500 == 0 and time.time() > t_end:
            break
        # Generate candidate moves
        moves = []
        # Focus on positions that contribute to worst correlations
        # Simple: random pair
        for _ in range(20):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                continue
            key = (min(i,j), max(i,j))
            if key in tabu and tabu[key] > it:
                continue
            moves.append((i, j))
        
        if not moves:
            # Clear tabu
            tabu = {}
            continue
        
        # Evaluate all moves
        best_move_c5 = best_c5 + 1  # threshold
        best_move = None
        for i, j in moves:

import numpy as np
import time

def _c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _binary_init(reals, n, rng):
    k = n // 2
    idx = np.argsort(reals)[-k:]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _random_binary(n, rng):
    h = np.zeros(n)
    k = n // 2
    idx = rng.choice(n, k, replace=False)
    h[idx] = 1.0
    return h

def _local_descent(h, dx, n, t_end):
    c5_cur = _c5(h, dx)
    for _ in range(200):
        if time.time() > t_end:
            break
        improved = False
        best_ij = None
        best_val = c5_cur
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if time.time() > t_end:
                return h, c5_cur
            for j in zeros:
                h[i] = 0; h[j] = 1
                c5_try = _c5(h, dx)
                h[i] = 1; h[j] = 0
                if c5_try < best_val - 1e-14:
                    best_val = c5_try
                    best_ij = (i, j)
        if best_ij is not None:
            i, j = best_ij
            h[i] = 0; h[j] = 1
            c5_cur = best_val
            improved = True
        if not improved:
            break
    return h, c5_cur

def _tabu_search(h, dx, n, t_end, rng, tabu_len=80, n_cand=30):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    cur_c5 = best_c5
    tabu_until = {}
    it = 0
    stall = 0
    while time.time() < t_end:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        cands = []
        for _ in range(n_cand):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            key = min(i, j) * n + max(i, j)
            if tabu_until.get(key, 0) > it:
                continue
            cands.append((i, j, key))
        if not cands:
            tabu_until.clear()
            continue
        best_move_c5 = cur_c5 + 0.01
        best_move = cands[0]
        for i, j, key in cands:
            h[i] = 0; h[j] = 1
            c5_try = _c5(h, dx)
            h[i] = 1; h[j] = 0
            if c5_try < best_move_c5:
                best_move_c5 = c5_try
                best_move = (i, j, key)
        i, j, key = best_move
        h[i] = 0; h[j] = 1
        cur_c5 = best_move_c5
        tabu_until[key] = it + tabu_len
        if len(tabu_until) > 5000:
            oldest = min(tabu_until, key=tabu_until.get)
            del tabu_until[oldest]
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                break
    return best_h, best_c5

def _sa_search(h, dx, n, t_end, rng, T0=0.3, Tend=0.005):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    cur_c5 = best_c5
    total_it = 0
    while time.time() < t_end:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0; h[j] = 1
        c5_try = _c5(h, dx)
        total_it += 1
        frac = min(total_it / 300000.0, 1.0)
        T = T0 * (Tend / T0) ** frac
        delta = c5_try - cur_c5
        if c5_try <= cur_c5 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = c5_try
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
    return best_h, best_c5

def _structured_binary(n, rng):
    """Create structured binary sequences for diversity."""
    k = n // 2
    # Quadratic residue pattern
    h_qr = np.zeros(n)
    for i in range(n):
        if (i * i) % n < k * 2.0 / n * n * 0.5:
            h_qr[i] = 1.0
    # Fix to exactly k ones
    if h_qr.sum() != k:
        h_qr = _binary_init(np.where(h_qr == 1, rng.rand(n), rng.rand(n)), n)
        h_qr = _binary_init(reals=h_qr + 0.1, n=n, rng=rng)
    # Golden ratio
    h_gr = np.zeros(n)
    vals = sorted([(int((i * 0.618033988749895) % 1.0 * n), i) for i in range(n)])
    h_gr[[v[0] for v in vals[:k]]] = 1.0
    return h_qr, h_gr
