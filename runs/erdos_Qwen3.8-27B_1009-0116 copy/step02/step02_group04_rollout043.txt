import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(z, target):
    n = len(z)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a; diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _grad_active(h, tol=1e-5):
    n = len(h)
    corr = _corr(h)
    mx = np.max(corr)
    active = np.where(corr >= mx - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - n + 1
        if lag == 0:
            grad += 1.0 - 2.0 * h
        elif lag > 0:
            grad[:n-lag] += 1.0 - h[lag:]
            grad[lag:] -= h[:n-lag]
        else:
            L = -lag
            grad[L:] += 1.0 - h[:n-L]
            grad[:n-L] -= h[L:]
    grad /= len(active)
    return grad, mx, active

def _subgrad_descend(h0, n, dx, budget_frac, deadline, seed):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    h = h0.copy()
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    alpha = 0.1
    patience = 0
    t0 = time.time()
    total_time = budget_frac  # in seconds
    while time.time() < deadline:
        frac = (time.time() - t0) / total_time if total_time > 0 else 1.0
        # Adaptive step
        grad, mx, active = _grad_active(h)
        z = h - alpha * grad
        h_new = _project(z, target)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
            alpha = min(alpha * 1.05, 0.5)
        else:
            h = h_new  # still update h (momentum-like)
            patience += 1
            if patience > 30:
                # Restart with perturbation
                alpha = max(0.01, alpha * 0.5)
                noise = 0.02 * rng.randn(n)
                h = _project(h + noise, target)
                patience = 0
        # Occasionally try larger perturbations
        if int(frac * 20) % 20 == 1 and patience > 10:
            noise = 0.1 * rng.randn(n)
            h = _project(h + noise, target)
            patience = 0
    return best_h, best_c5

    if c5 < best_c5:
        best_c5 = c5; best_h = h

    # Phase 4: SLSQP polish with high alpha (120s)
    t_end = min(deadline, t0 + 630)
    for alpha in [1000, 3000, 6000]:
        if time.time() > t_end:
            break
        h, c5 = _slsqp(best_h, n0, dx0, alpha, 800, t_end)
        if c5 < best_c5:
            best_c5 = c5; best_h = h
        h2 = _project(best_h + 0.02 * rng.randn(n0), target0)
        h2, c52 = _slsqp(h2, n0, dx0, alpha, 400, t_end)
        if c52 < best_c5:
            best_c5 = c52; best_h = h2

    # Phase 5: Local search (120s)
    t_end = min(deadline, t0 + 750)
    best_h, best_c5 = _local_search(best_h, n0, dx0, t_end, seed + 50)

    # Phase 6: Try other n values (150s)
    t_end = min(deadline, t0 + 900)
    for n_try in [80, 60, 100, 50]:
        if time.time() > t_end or deadline - time.time() < 50:
            break
        dx_try = 2.0 / n_try
        t_sub = min(t_end, time.time() + 45)
        h_t, c5_t = _optimize_n(best_h, n_try, dx_try, t_sub, seed + n_try)
        if c5_t < best_c5:
            best_c5 = c5_t; best_h = h_t

    # Phase 7: Final polish
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    if deadline - time.time() > 25:
        best_h, best_c5 = _slsqp(best_h, n_f, dx_f, 5000, 1500,
                                  min(deadline, time.time() + 25))
    if deadline - time.time() > 12:
        best_h, best_c5 = _local_search(best_h, n_f, dx_f,
                                         min(deadline, time.time() + 12), seed + 200)

    best_h = _project(best_h, n_f / 2.0)
    best_c5 = _c5(best_h, dx_f)
    return (best_h, best_c5, n_f)
