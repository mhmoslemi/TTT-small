import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

    dx = 2.0 / n
    c = _corr(x) * dx

    return m + np.log(np.sum(np.exp(s))) / alpha

    dx = 2.0 / n
    c = _corr(x) * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)

    pdx = p * dx

    for m_idx in range(n):

    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _smooth_obj(x, alpha, mu):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x) * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def _smooth_grad(x, alpha, mu):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x) * dx
    m = c.max()

    e = np.exp(s)
    p = e / e.sum()

    grad = np.zeros(n)

            grad[i] += (1.0 - x[sec])

        if k2lo <= k_star <= k2hi:
