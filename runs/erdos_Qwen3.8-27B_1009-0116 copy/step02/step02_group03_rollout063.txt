def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(500):
        if s > 1e-12:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

        return np.max(c * x + rest)

    g0 = g(lo)
    g1 = g(hi)

        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    def update(h, n):
        nonlocal best_h, best_c5, best_n
        c5 = _compute_c5(h, 2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    init_h = None
    init_n = 70
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass
    if init_h is not None:
        init_h = _repair(init_h, init_n)
        update(init_h, init_n)

    n_list = [50, 60, 70, 80, 100]
    if init_n not in n_list:
        n_list.append(init_n)
    n_list = sorted(set(n_list))
    if init_h is not None:
        n_list.remove(init_n)
        n_list = [init_n] + n_list

    time_per_n = (budget_s - 150) / len(n_list)

    for n in n_list:
        if time.time() - t0 > budget_s - 100:
            break
        time_limit = min(t0 + budget_s - 70, time.time() + time_per_n)
        starts = []
        if init_h is not None and init_n == n:
            starts.append(init_h.copy())
        for kind in range(5):
            starts.append(_make_feasible(n, seed + kind * 137 + n * 7, kind))

        for h0 in starts:
            if time.time() > time_limit:
                break
            current_h, _ = _exact_coord_descent(h0, n, max_passes=400)
            update(current_h, n)
            cur_c5 = _compute_c5(current_h, 2.0 / n)

            iters = 0
            no_improve = 0
            while time.time() < time_limit and iters < 200:
                strength = 0.04 + 0.12 / (1.0 + no_improve * 0.5)
                h_pert = _perturb(current_h, n, rng, strength)
                h_cd, _ = _exact_coord_descent(h_pert, n, max_passes=250)
                c5_pert = _compute_c5(h_cd, 2.0 / n)
                update(h_cd, n)
                if c5_pert < cur_c5 - 1e-9:
                    current_h = h_cd
                    cur_c5 = c5_pert
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 30:
                        break
                iters += 1

            if time.time() < time_limit:
                h_s = _slsqp_polish(current_h, n, 500.0, 150)
                update(h_s, n)
                c5_s = _compute_c5(h_s, 2.0 / n)
                if c5_s < cur_c5:
                    current_h = h_s
                    cur_c5 = c5_s
                h_f, _ = _exact_coord_descent(current_h, n, max_passes=300)
                update(h_f, n)

    if best_h is not None and time.time() - t0 < budget_s - 40:
        h_fin, _ = _exact_coord_descent(best_h, best_n, max_passes=400)
        update(h_fin, best_n)
        for _ in range(30):
            if time.time() - t0 > budget_s - 10:
                break
            h_pert = _perturb(best_h, best_n, rng, 0.03)
            h_cd2, _ = _exact_coord_descent(h_pert, best_n, max_passes=200)
            update(h_cd2, best_n)
            if time.time() - t0 > budget_s - 15:
                break
            h_sz = _slsqp_polish(best_h, best_n, 800.0, 100)
            update(h_sz, best_n)

    if best_h is None:
        best_n = 70
        best_h = _repair(np.full(best_n, 0.5), best_n)
        best_c5 = _compute_c5(best_h, 2.0 / best_n)
    best_h = _repair(best_h, best_n)
    best_c5 = _compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
