import numpy as np
from scipy.optimize import minimize
import time

def compute_g(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return float(np.max(compute_g(h)) * dx)

def smooth_obj(x, beta, lam):
    g = compute_g(x)
    bg = beta * g
    bg_max = bg.max()
    log_sum = bg_max + np.log(np.sum(np.exp(bg - bg_max)))
    obj = log_sum / beta
    n = len(x)
    obj += lam * (x.sum() - n/2.0)**2
    return obj

def smooth_grad(x, beta, lam):
    n = len(x)
    h = x
    g = compute_g(h)
    bg = beta * g
    bg_max = bg.max()
    exp_bg = np.exp(bg - bg_max)
    w = exp_bg / exp_bg.sum()
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        lo1 = max(0, -m)
        hi1 = min(n - 1, n - 1 - m)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wk * (1.0 - h[idx + m])
        lo2 = max(0, m)
        hi2 = min(n - 1, m + n - 1)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] += wk * (-h[idx - m])
    
    grad += 2.0 * lam * (x.sum() - n / 2.0)
    return grad

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def cd_pass(h, n, dx, n_grid, interior_only=False):
    hi_c5 = compute_c5(h, dx)
    improved = False
    pairs = []
    if interior_only:
        for i in range(n):
            for j in range(i + 1, n):
                if 0.05 < h[i] < 0.95 or 0.05 < h[j] < 0.95:
                    pairs.append((i, j))
    else:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for i, j in pairs:
        hsum = h[i] + h[j]
        lo = max(0.0, hsum - 1.0)
        hi = min(1.0, hsum)
        if hi - lo < 1e-10:
            continue
        orig_i = h[i]
        best_c = hi_c5
        best_t = orig_i
        for t in np.linspace(lo, hi, n_grid):
            if abs(t - orig_i) < 1e-10:
                continue
            h[i] = t
            h[j] = hsum - t
            c = compute_c5(h, dx)
            if c < best_c:
                best_c = c
                best_t = t
        if best_c < hi_c5 - 1e-13:
            h[i] = best_t
            h[j] = hsum - best_t
            hi_c5 = best_c
            improved = True
        else:
            h[i] = orig_i
    return h, hi_c5, improved

import numpy as np
from scipy.optimize import minimize
import time

def compute_g(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return float(np.max(compute_g(h)) * dx)

def smooth_obj(x, beta, lam):
    g = compute_g(x)
    bg = beta * g
    bg_max = bg.max()
    log_sum = bg_max + np.log(np.sum(np.exp(bg - bg_max)))
    n = len(x)
    return log_sum / beta + lam * (x.sum() - n / 2.0) ** 2

def smooth_grad(x, beta, lam):
    n = len(x)
    h = x
    g = compute_g(h)
    bg = beta * g
    bg_max = bg.max()
    ebg = np.exp(bg - bg_max)
    w = ebg / ebg.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        lo1 = max(0, -m)
        hi1 = min(n - 1, n - 1 - m)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wk * (1.0 - h[idx + m])
        lo2 = max(0, m)
        hi2 = min(n - 1, m + n - 1)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] += wk * (-h[idx - m])
    grad += 2.0 * lam * (x.sum() - n / 2.0)
    return grad

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

    hi_c5 = compute_c5(h, dx)
    improved = False
    for i in range(n):
        if interior_only and h[i] <= 0.02 and (i == 0 or True):
            pass
        for j in range(i + 1, n):
            if interior_only and h[i] <= 0.02 and h[j] <= 0.02:
                continue
            if interior_only and h[i] >= 0.98 and h[j] >= 0.98:
                continue
            hsum = h[i] + h[j]
            lo = max(0.0, hsum - 1.0)
            hi = min(1.0, hsum)
            if hi - lo < 1e-10:
                continue
            orig_i = h[i]
            best_c = hi_c5
            best_t = orig_i
            for t in np.linspace(lo, hi, n_grid):
                if abs(t - orig_i) < 1e-10:
                    continue
                h[i] = t
                h[j] = hsum - t
                c = compute_c5(h, dx)
                if c < best_c:
                    best_c = c
                    best_t = t
            if best_c < hi_c5 - 1e-13:
                h[i] = best_t
                h[j] = hsum - best_t
                hi_c5 = best_c
                improved = True
            else:
                h[i] = orig_i
    return h, hi_c5, improved

def cd_loop(h, n, dx, t_limit, t_start, n_grid, max_passes, interior=False):
    for _ in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        h, _, imp = cd_pass(h, n, dx, n_grid, interior_only=interior)
        if not imp:
            break
        h = repair(h, n)

        return repair(res.x, n)
    except Exception:
        return h.copy()

    mx = g.max()
    active = np.where(g >= mx - 1e-10)[0]

        lo2 = max(0, m)

        if lo2 <= hi2:

            grad_sum[idx] += (-h[idx - m])
    grad_sum /= len(active)
    h_new = h - step_size * grad_sum
    return repair(h_new, n)
