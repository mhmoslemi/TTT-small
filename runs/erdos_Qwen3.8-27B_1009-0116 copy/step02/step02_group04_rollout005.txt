```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _obj(h, alpha):
    c = _corr(h)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = float(h.sum())
    if s < 1e-14:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - float(h[idx]))
                h[idx] = float(h[idx]) + a
                diff -= a
            else:
                a = min(-diff, float(h[idx]))
                h[idx] = float(h[idx]) - a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    target = float(n) / 2.0
    cons = ({'type': 'eq', 'fun': lambda x: float(np.sum(x) - target)},)
    try:
        res = minimize(_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        if _c5(hc, dx) < _c5(h, dx):
            return hc, _c5(hc, dx)
    except:
        pass
    return h, _c5(h, dx)

def _sweep(h, n, dx, rng, deadline):
    """Systematic pairwise sweep with adaptive step sizes."""
    best = _c5(h, dx)
    h = h.copy()
    base_deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d_scale in base_deltas:
        if time.time() >= deadline:
            break
        improved = False
        # Shuffle indices for diversity
        idx = rng.permutation(n)
        for ii in range(n):
            i = idx[ii]
            if time.time() >= deadline:
                break
            for jj in range(ii + 1, n):
                j = idx[jj]
                for sign in (1.0, -1.0):
                    d = sign * d_scale
                    ni, nj = float(h[i]) + d, float(h[j]) - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < best - 1e-13:
                        best = c
                        improved = True
                    else:
                        h[i], h[j] = ni - d, nj + d
    return h, best
