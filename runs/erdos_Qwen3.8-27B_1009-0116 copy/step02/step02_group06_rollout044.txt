from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha, dx):
    c = _corr(x)
    return _smooth_max(c, alpha) * dx

def _eq_con(x, target):
    return np.sum(x) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)

    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _line_search(corr, g, d_min, d_max, iters=50):
    if d_max - d_min < 1e-14:
        return 0.5 * (d_min + d_max)

            d_max = d_mid
        elif gk < -1e-12:
            d_min = d_mid
        else:
            break
    return 0.5 * (d_min + d_max)

def _sweep(h, n, dx, corr, max_rounds, bs_iters):
    for rnd in range(max_rounds):
        any_change = False
        cur_max = float(np.max(corr))
        k_star = int(np.argmax(corr))
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_min > d_max - 1e-10 or d_max - d_min < 1e-12:
                    continue
                gd = _g_vec(h, i, n) - _g_vec(h, j, n)
                # Quick rejection: if gradient at current max has wrong sign both ways
                gk = gd[k_star]
                if gk > 1e-10 and d_min >= 0:
                    continue
                if gk < -1e-10 and d_max <= 0:
                    continue
                d_opt = _line_search(corr, gd, d_min, d_max, bs_iters)

                    continue

                    corr = new_corr
                    cur_max = new_max
                    k_star = int(np.argmax(corr))
                    any_change = True
