import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _top_k_corr(h, k=5):
    c = _corr(h)
    idx = np.argsort(c)[-k:]
    return idx, c[idx]

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    if m < -500/alpha:
        return m
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    return _smooth_max(_corr(x), alpha) * dx

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx

def _cons_fun(x, target):
    return np.sum(x) - target

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _cons_fun, 'args': (target,)}]
    try:
        res = minimize(_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _pair_search(h, n, dx, rng, deadline, max_rounds=2000):
    best = h.copy()
    best_c5 = _c5(best, dx)
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ntrials = 60 + 40 * min(stall // 8, 4)
        for _ in range(ntrials):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            d = (0.005 + 0.15 * rng.random()) * (1 if rng.random() < 0.5 else -1)
            ni, nj = best[i] + d, best[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            oi, oj = best[i], best[j]
            best[i], best[j] = ni, nj
            nc = _c5(best, dx)
            if nc < best_c5 - 1e-14:
                best_c5 = nc
                improved = True
            else:
                best[i], best[j] = oi, oj
        if not improved:
            stall += 1
            # Try fine steps
            for d in [0.002, -0.002, 0.005, -0.005, 0.008, -0.008, 0.012, -0.012]:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                ni, nj = best[i] + d, best[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    oi, oj = best[i], best[j]
                    best[i], best[j] = ni, nj
                    nc = _c5(best, dx)
                    if nc < best_c5 - 1e-14:
                        best_c5 = nc
                        improved = True
                        stall = 0
                        break
                    else:
                        best[i], best[j] = oi, oj
        else:
            stall = 0
        if stall > 40:
            break
    return best, best_c5

def _simulated_annealing(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    curr = best.copy()
    curr_c5 = best_c5
    T0, Tend = 0.02, 0.0005
    n_total = 0
    max_iter = int((deadline - time.time()) * 3000)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T0 * (Tend / T0) ** (it / max(1, max_iter))
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        d = (0.01 + 0.08 * rng.random()) * (1 if rng.random() < 0.5 else -1)
        ni, nj = curr[i] + d, curr[j] - d
        if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
            continue
        ni, nj = np.clip(ni, 0, 1), np.clip(nj, 0, 1)
        oi, oj = curr[i], curr[j]
        curr[i], curr[j] = ni, nj
        nc = _c5(curr, dx)
        if nc < curr_c5:
            curr_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best = curr.copy()
        else:
            delta = (nc - curr_c5) / T
            if delta < 50 and rng.random() < np.exp(-delta):
                curr_c5 = nc
            else:
                curr[i], curr[j] = oi, oj
        n_total += 1
    return best, best_c5

def _structured_init(n, seed_val):
    rng2 = np.random.RandomState(seed_val)
    t = np.linspace(0, 2.0, n, endpoint=False)
    freqs = [1.0, 1.5, 2.0, 2.5, 3.0]
    f = int(seed_val % len(freqs))
    base = 0.5 + 0.25 * np.sin(2 * np.pi * freqs[f] * t) + 0.1 * rnd_noise(rng2, n)
    return _repair(base, n)

def rnd_noise(rng2, n):
    return rng2.randn(n)

def _binary_init(n, rng):
    h = np.zeros(n)
    pos = rng.choice(n, n // 2, replace=False)
    h[pos] = 1.0
    return h
