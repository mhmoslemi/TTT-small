                    break

            else:
                h[i], h[j] = hi0, hj0

    return h, bc

def _proj(y, target):

        for k in idx:

            x[k] += av if diff > 0 else -av

                         constraints=[{'type': 'eq', 'fun': _con_fun, 'args': (tgt,)}],
                         options={'maxiter': 400, 'ftol': 1e-15})
            hf = _proj(r.x, tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc, bh = cf, hf
        except Exception:
            pass

    # Phase 11: Final micro-CD
    if rem > 12:
        micro_d = [0.0005, 0.001, 0.002, 0.003, -0.0005, -0.001, -0.002, -0.003]
        bh, bc = _random_cd(bh, dx, n, 300, rng, dvals=micro_d)

    # Final: project and evaluate
    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
