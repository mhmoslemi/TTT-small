import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-10:
        idx = np.argsort(1 - h) if d > 0 else np.argsort(h)
        for i in idx:
            a = min(abs(d), (1 - h[i]) if d > 0 else h[i])
            h[i] += a if d > 0 else -a
            d = t - h.sum()
            if abs(d) < 1e-12:

    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-14})
        return _repair(res.x, n)
    except:
        return h

            if 0 <= il < n:
                g += 1.0 - h[il]
            if 0 <= ir < n:
                g -= h[ir]
            A[i, idx] = g
    return A
