import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                s = min(diff, 1.0 - h[i])
                h[i] += s; diff -= s
            else:
                s = min(-diff, h[i])
                h[i] -= s; diff += s
            if abs(diff) < 1e-13:
                break
    return h

def _grad_smooth(h, dx, t):
    n = len(h)
    c = _corr(h) * dx
    cs = c - c.max()
    w = np.exp(t * cs)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = w[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[lag:] += pk * dx * (1.0 - h[:n - lag])
            g[:n - lag] -= pk * dx * h[lag:]
        else:
            L = -lag
            g[:n - L] += pk * dx * (1.0 - h[L:])
            g[L:] -= pk * dx * h[:n - L]
    return g

def _grad_max(h, dx):
    """Gradient of the max correlation (subgradient)"""
    n = len(h)
    c = _corr(h) * dx
    mx = c.max()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if c[k] < mx - 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[lag:] += dx * (1.0 - h[:n - lag])
            g[:n - lag] -= dx * h[lag:]
        else:
            L = -lag
            g[:n - L] += dx * (1.0 - h[L:])
            g[L:] -= dx * h[:n - L]
    g /= (2 * n - 1)  # normalize
    return g

def _pair_search(h, dx, target, budget_fn, max_iter=50000):
    """Local pair-swap search"""
    n = len(h)
    c5_cur = _c5(h, dx)
    # Try all pairs with small transfers
    deltas = np.array([0.01, 0.02, 0.05, 0.1])
    for d in deltas:
        for _ in range(max_iter):
            if budget_fn():
                break
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    for sign in [1, -1]:
                        ni = h[i] + sign * d
                        nj = h[j] - sign * d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx)
                            if c < c5_cur - 1e-15:
                                c5_cur = c
                                improved = True
                            else:
                                h[i], h[j] = ni - sign*d, nj + sign*d
                                h[i], h[j] = h[i] - 0, h[j] - 0  # no-op
                    # restore
                    pass
            if not improved:
                break
    return h, c5_cur
