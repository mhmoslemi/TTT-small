    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return np.sum(x) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    k = np.arange(-(n - 1), n)

    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

    for i in range(n):

            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds1 = np.linspace(d_min, d_max, n_coarse)
            nc1 = corr[None, :] + ds1[:, None] * gd[None, :]
            c5s1 = np.max(nc1, axis=1) * dx
            bi1 = np.argmin(c5s1)
            cur_max = np.max(corr) * dx

                    lo = max(d_min, d_best - fine_wid)
                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c5s2 = np.max(nc2, axis=1) * dx
                    bi2 = np.argmin(c5s2)
                    d_best = ds2[bi2]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

def _slsqp(h, n, dx, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except:
        return h

def _binary_project(h, n, rng):
    """Project to nearest binary with exactly n/2 ones"""
    target = n // 2
    b = (h > 0.5).astype(float)
    s = b.sum()
    if s != target:
        # Adjust
        if s > target:
            # Need to remove (s - target) ones: pick smallest h among b==1
            ones = np.where(b == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(s - target)]:
                b[ones[idx]] = 0
        else:
            # Need to add (target - s) ones: pick largest h among b==0
            zeros = np.where(b == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(target - s)]:
                b[zeros[idx]] = 1
    return b

def _flip_bits(b, n, k, rng):
    """Flip k bits while maintaining sum constraint"""
    b = b.copy()
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]

        if rng.rand() < 0.5:

            j = zeros[rng.randint(len(zeros))]

            j = ones[rng.randint(len(ones))]
        b[i] = 1 - b[i]
        b[j] = 1 - b[j]
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
    return b

def _binary_local_search(b, n, dx, n_iter, rng, T_start, T_end):
    """Simulated annealing on binary sequences"""
    b = b.copy()
    corr = _corr(b)
    best_c5 = float(np.max(corr) * dx)
    best_b = b.copy()
    T = T_start
    cooling = (T_end / T_start) ** (1.0 / max(n_iter, 1))
    
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    for it in range(n_iter):

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = 0.0, 1.0
        corr_new = _corr(b)
        c5_new = float(np.max(corr_new) * dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_b = b.copy()

            b[i], b[j] = 1.0, 0.0

                best_c5 = c5
                best_h = h.copy()
            if not imp:
                break

    # Phase 7: Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        hc = _slsqp(best_h, n, dx, target, bounds, 5000, 1200)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 8: Final ultra-fine sweep
    if time.time() - t0 < budget_s - 1:
        h = best_h.copy()
        corr = _corr(h)
        h, corr, _ = _sweep(h, n, dx, corr, 41, 81, 0.004)
        corr = _corr(h)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
