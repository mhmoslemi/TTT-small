                diff -= add

                continue
            k = idx - (n-1)
            if k == 0:
                g += pi * (1.0 - 2.0*h)
            elif k > 0:
                g[:n-k] += pi * (1.0 - h[k:])
                g[k:] -= pi * h[:n-k]
            else:
                kk = -k
                g[kk:] += pi * (1.0 - h[:n-kk])
                g[:n-kk] -= pi * h[kk:]
        if prev_g is not None:
            st = h - prev_h
            yt = g - prev_g
            sy = np.dot(st, yt)
            yy = np.dot(yt, yt)
            if yy > 1e-16 and sy > 0:
                lr = np.clip(sy/yy, 0.05*lr0, 20.0*lr0)
        prev_h = h.copy()
        prev_g = g.copy()
        v = beta * v + (1.0 - beta) * g
        h = _repair(h - lr * v, n, target)
        beta = min(0.95, beta * 1.001)
    return h

def _slsqp(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n, target)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n, target)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    def elapsed():
        return time.time() - t0

    # Phase 1: Subgradient with momentum
    for alpha, n_it, lr0 in [(50, 4000, 0.02), (150, 4000, 0.01), (400, 3000, 0.005)]:
        if elapsed() > budget_s - 600:
            break
        h2 = _subgrad(best_h, n, target, alpha, n_it, lr0)
        c5 = _c5(h2, dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2

    # Phase 2: Coarse sweeps
    h = best_h.copy()
    for _ in range(150):
        if elapsed() > budget_s - 500:
            break
        h, corr, imp = _sweep_pass(h, n, 15, 0, 0)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
        if not imp:
            break

    # Phase 3: SLSQP
    h = best_h.copy()
    for alpha in [300, 800, 1500]:
        if elapsed() > budget_s - 350:
            break
        h2 = _slsqp(h, n, target, bounds, alpha, 600)
        c5 = _c5(h2, dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2
        h = best_h

    # Phase 4: Fine sweeps
    h = best_h.copy()
    for _ in range(250):
        if elapsed() > budget_s - 250:
            break
        h, corr, imp = _sweep_pass(h, n, 21, 51, 0.015)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
        if not imp:
            break

    # Phase 5: Restart + sweep (escape local optima)
    for sr in range(12):
        if elapsed() > budget_s - 120:
            break
        h_p = best_h + 0.05 * rng.randn(n)
        h_p = _repair(h_p, n, target)
        h_p, corr_p, _ = _sweep_pass(h_p, n, 21, 41, 0.01)
        c5_p = float(np.max(corr_p) * dx)
        if c5_p < best_c5:
            best_c5 = c5_p; best_h = h_p.copy()

    # Phase 6: SLSQP high alpha
    if elapsed() < budget_s - 40:
        h2 = _slsqp(best_h, n, target, bounds, 4000, 1500)
        c5 = _c5(h2, dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2

    # Phase 7: Ultra-fine sweeps
    h = best_h.copy()
    for _ in range(150):
        if elapsed() > budget_s - 8:
            break
        h, corr, imp = _sweep_pass(h, n, 31, 81, 0.004)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
        if not imp:
            break

    # Phase 8: Final SLSQP
    if elapsed() < budget_s - 3:
        h2 = _slsqp(best_h, n, target, bounds, 8000, 2000)
        c5 = _c5(h2, dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2

    best_h = _repair(best_h, n, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
