import numpy as np

import time

_g_best_h = None
_g_best_c5 = np.inf
_g_best_n = 70

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(500):
        if s > 1e-12:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                add = min(diff, max(0.0, 1.0 - h[idx]))
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, max(0.0, h[idx]))
                h[idx] -= sub
                diff += sub
    return h

def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _try_update(h, n):

def _make_binary_start(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def _make_periodic_start(n, seed, period):
    rng = np.random.RandomState(seed)
    ones_in_period = period // 2
    pattern = np.zeros(period)
    ones = rng.choice(period, size=ones_in_period, replace=False)
    pattern[ones] = 1.0
    h = np.tile(pattern, n // period + 1)[:n]
    return _repair(h, n)

def _lbfgs_refine(h, n, dx, time_limit, alpha, penalty):
    t_start = time.time()
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        val = _smooth_max(corr, alpha)
        val += penalty * (x.sum() - target) ** 2
        return val

    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-10})
        if time.time() - t_start < time_limit * 0.5:
            res2 = minimize(obj, res.x, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12})
            res = res2
        return _repair(res.x, n)
    except Exception:
        return h

def _slsqp_refine(h, n, dx, time_limit, alpha):
    t_start = time.time()
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    ones_vec = np.ones(n)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return _smooth_max(corr, alpha)

    def con_fun(x):
        return x.sum() - target

    def con_jac(x):
        return ones_vec

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return h
