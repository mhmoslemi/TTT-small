def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)

                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

                h[idx] -= sub
                diff += sub

    return h

def _eval(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _ls_best(h, n, dx, max_rounds=300, t_deadline=None):
    """Local search: best-improvement swap"""
    h = h.copy()
    best_val = _eval(h, dx)
    for r in range(max_rounds):
        if t_deadline is not None and time.time() > t_deadline:
            break
        best_delta = 0.0
        best_pair = None
        for i in range(n):
            for j in range(i + 1, n):
                old_i, old_j = h[i], h[j]
                h[i], h[j] = old_j, old_i
                v = _eval(h, dx)
                d = best_val - v
                h[i], h[j] = old_i, old_j
                if d > best_delta:
                    best_delta = d
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = h[j], h[i]
        best_val -= best_delta
    return h, best_val

import numpy as np
from scipy.optimize import minimize
import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h


def _eval(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _ls_best(h, n, dx, max_rounds, t_dead):
    h = h.copy()
    best_val = _eval(h, dx)
    for r in range(max_rounds):
        if time.time() > t_dead:
            break
        best_d = 1e-15
        best_pair = None
        for i in range(n):
            for j in range(i + 1, n):
                oi, oj = h[i], h[j]
                h[i], h[j] = oj, oi
                v = _eval(h, dx)
                d = best_val - v
                h[i], h[j] = oi, oj
                if d > best_d:
                    best_d = d
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = h[j], h[i]
        best_val -= best_d
    return h, best_val


def _perturb_swap(h, n, rng, k):
    h = h.copy()
    for _ in range(k):
        i, j = rng.choice(n, 2, replace=False)
        h[i], h[j] = h[j], h[i]
    return h


def _perturb_noise(h, n, rng, scale):
    h = h.copy()
    h += rng.randn(n) * scale

def _obj_lse(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + float(np.sum(s)) / alpha) * (2.0 / n)


def _obj_pnorm(x, p, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.sum(corr ** p)) ** (1.0 / p) * (2.0 / n)


def _eq_con(x, target):
    return float(np.sum(x) - target)
