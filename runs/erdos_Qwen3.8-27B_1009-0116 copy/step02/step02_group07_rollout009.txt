    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

    return m + np.log(np.sum(np.exp(s))) / alpha

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

    hi_c5 = _c5(h, dx)
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hsum = h[i] + h[j]
            lo = max(0.0, hsum - 1.0)
            hi = min(1.0, hsum)
            if hi - lo < 1e-10:
                continue
            orig_i, orig_j = h[i], h[j]
            best_t, best_c = orig_i, hi_c5
            for t in np.linspace(lo + 1e-10, hi - 1e-10, n_grid):
                h[i], h[j] = t, hsum - t
                c = _c5(h, dx)
                if c < best_c:

                hi_c5 = best_c
                improved = True
            else:

            cnt ^= v & 1
            v >>= 1
        t[i] = cnt
    return t

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _constraint_fun}]

    def elapsed():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)

    starts = [_to_01(h0, n)]
    tm = _thue_morse(n)
    starts.append(_to_01(tm, n))
    for _ in range(3):
        h_r = np.zeros(n)
        h_r[rng.permutation(n)[:n // 2]] = 1
        starts.append(h_r)
    for blk in [2, 3, 4, 5, 7]:
        pat = np.tile(np.concatenate([np.ones(blk), np.zeros(blk)]), n // (2 * blk) + 1)[:n]
        starts.append(_to_01(pat, n))
    starts = list(dict.fromkeys([x.tobytes() for x in starts]))
    starts_real = [None] * len(starts)
    for ki, k in enumerate(starts):
        starts_real[ki] = np.frombuffer(k, dtype=float).copy()

    for h_s in starts_real:
        if elapsed() > budget_s - 350:
            break
        h_try = h_s.copy()
        for rep in range(3):
            if elapsed() > budget_s - 340:
                break
            h_try, c_try = _greedy_01(h_try, n, dx, max_iter=3000, t_start=t0, t_limit=budget_s - 340)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()
            if elapsed() > budget_s - 340:
                break
            h_try = h_try.copy()
            for _ in range(3):
                oi = rng.choice(np.where(h_try == 1)[0])
                oz = rng.choice(np.where(h_try == 0)[0])
                h_try[oi], h_try[oz] = h_try[oz], h_try[oi]

    for a in [300, 800, 2000, 5000]:
        if elapsed() > budget_s - 250:
            break
        h = _repair(best_h.copy(), n)
        h = _slsqp(h, n, bounds, cons, a, 500, 1e-14)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    if elapsed() < budget_s - 180:
        h, c = _cd_loop(best_h, n, dx, budget_s - 220, t0, n_grid=15, max_passes=1000)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    if elapsed() < budget_s - 100:
        h = _slsqp(best_h.copy(), n, bounds, cons, 8000, 600, 1e-15)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    if elapsed() < budget_s - 50:
        h, c = _cd_loop(best_h, n, dx, budget_s - 60, t0, n_grid=25, max_passes=600)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    for sig in [0.05, 0.1, 0.15, 0.08, 0.03]:
        if elapsed() > budget_s - 30:
            break
        h_p = _repair(best_h + sig * rng.randn(n), n)
        h = _slsqp(h_p, n, bounds, cons, 2000, 300, 1e-13)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
            if elapsed() < budget_s - 40:
                h2, c2 = _cd_loop(h, n, dx, min(budget_s - 50, elapsed() + 50), t0, n_grid=20, max_passes=200)
                if c2 < best_c5:
                    best_c5, best_h = c2, h2.copy()

    if elapsed() < budget_s - 15:
        h, c = _cd_loop(best_h, n, dx, budget_s - 20, t0, n_grid=31, max_passes=400)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
