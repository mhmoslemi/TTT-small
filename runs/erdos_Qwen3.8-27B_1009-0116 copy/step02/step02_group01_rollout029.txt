def _grad_lag(h, m, n):
    k = m - (n - 1)
    g = np.zeros(n)
    j = np.arange(n)

    g[mask_m] += 1.0 - h[jm[mask_m]]
    g[mask_p] -= h[jp[mask_p]]

This should be much more efficient than blind pair search.

Let me code this up:

```python
def _pair_search(h, n, dx, K=5, max_iter=1000):
    """Efficient pair search using gradient information."""
    best_c5 = _c5of(h, dx)
    for it in range(max_iter):
        c_vals = np.correlate(h, 1.0 - h, mode='full')
        active = np.argsort(c_vals)[-K:]
        # Weight lags by their proximity to max
        weights = np.zeros(2*n-1)
        for m in active:
            weights[m] = 1.0  # or could use softmax weights
        
        g = np.zeros(n)
        for m in active:
            g += _grad_lag(h, m, n)
        g -= np.mean(g)
        
        # Find best pair (i,j) to move δ from j to i
        best_gain = 0
        best_i, best_j = -1, -1
        for i in range(n):
            if h[i] >= 1.0:
                continue
            for j in range(n):
                if i == j or h[j] <= 0.0:
                    continue
                gain = g[i] - g[j]
                if gain < best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
        
        if best_i < 0 or best_gain > -1e-10:
            break
        
        # Line search: how much to move
        h_new = h.copy()
        h_new[best_i] += 0.01
        h_new[best_j] -= 0.01
        h_new = _repair(h_new, n)
        c5_new = _c5of(h_new, dx)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
        else:
            # Try smaller steps
            found = False
            for delta in [0.005, 0.002, 0.001, 0.0005]:
                h_new = h.copy()
                h_new[best_i] += delta
                h_new[best_j] -= delta
                h_new = _repair(h_new, n)
                c5_new = _c5of(h_new, dx)
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
                    found = True
                    break
            if not found:
                break
    
    return h, best_c5
```

This is O(n²) per iteration for finding the best pair, and O(1) iterations of line search. Much better than the brute force approach.

But wait, the gain calculation using a single gradient might not be accurate when c5 is the max of many terms. Let me use the actual c5 change for the top candidates.

Actually, for the min-max problem, the subgradient of max(c) is in the convex hull of the gradients of the active (maximizing) components. Using the average gradient of the top K lags is a reasonable approximation.

Let me also add a "triple" move: adjust 3 variables. This expands the search space.

def _c5of(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, m, n):
    k = m - (n - 1)
    g = np.zeros(n)
    j = np.arange(n)
    jm = j - k
    jp = j + k
    mask_m = (jm >= 0) & (jm < n)
    mask_p = (jp >= 0) & (jp < n)
    g[mask_m] += 1.0 - h[jm[mask_m]]
    g[mask_p] -= h[jp[mask_p]]
    return g

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    for _ in range(200):
        h *= target / (s + 1e-30)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:

        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)

                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
