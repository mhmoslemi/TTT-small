import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _project(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_new = np.clip(h - mid, 0.0, 1.0)
        if h_new.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fine adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _subgrad_step(h, n, dx, lr, target):
    """One subgradient step on F(h) = max_k c_k(h)"""
    corr = _corr(h)
    # Find max and near-max elements
    max_val = float(np.max(corr))
    # Subgradient is average of gradients at near-max positions
    thresh = max_val - 0.5  # within 0.5 of max
    active = np.where(corr >= thresh)[0]
    
    # Compute gradient
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)  # shift
        # d(c_k)/d(h[j]) = (1-h[j+k]) * 1[0<=j+k<n] - h[j-k] * 1[0<=j-k<n]
        j = np.arange(n)
        term1 = np.zeros(n)
        valid1 = (j + k >= 0) & (j + k < n)
        term1[valid1] = 1.0 - h[j[valid1] + k]
        term2 = np.zeros(n)
        valid2 = (j - k >= 0) & (j - k < n)
        term2[valid2] = h[j[valid2] - k]
        grad += (term1 - term2)
    
    grad /= len(active)
    h_new = _project(h - lr * grad, target)
    return h_new, max_val

def _slsqp_step(h, n, dx, target, bounds, alpha, maxiter):
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    
    def eq_con(x):
        return np.sum(x) - target
    
    cons = [{'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _project(res.x, target)
    except Exception:
        return h
