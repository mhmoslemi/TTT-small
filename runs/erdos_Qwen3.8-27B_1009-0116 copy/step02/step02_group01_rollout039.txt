```python
import time

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_val(h, alpha):
    c = _corr(h)
    m = float(np.max(c))
    z = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(z)))) / alpha

def _grad_smooth(h, alpha):
    n = len(h)
    c = _corr(h)
    m = float(np.max(c))
    z = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(z)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        # First term: (1-h[j-s]) for j in [max(0,s), min(n-1, n-1+s)]
        lo1 = max(0, s)
        hi1 = min(n - 1, n - 1 + s)

        hi2 = min(n - 1, n - 1 - s)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= wk * h[lo2+s:hi2+1+s]
    return grad

                    diff += a

        c = (x.sum() - target) / target
        g += 2.0 * pen * c / target
        return g

                   bounds=[(0.0, 1.0)] * n,

    return _repair(res.x, target)

def _opt_slsqp(h0, n, target, alpha, max_iter, time_limit):
    t_start = time.time()
    def obj(x):
        return _smooth_val(x, alpha) * (2.0 / n)

        return _grad_smooth(x, alpha) * (2.0 / n)

def _coord_descent(h, n, target, dx, time_limit, t0):

    deltas = np.array([0.001, 0.003, 0.01, -0.001, -0.003, -0.01,
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h)) * 2.0 / n)

def _sv(h, alpha):
    c = _corr(h)
    m = float(np.max(c))
    z = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(z)))) / alpha

def _gs(h, alpha):
    n = len(h)
    c = _corr(h)
    m = float(np.max(c))
    z = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(z)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        lo = max(0, s)
        hi = min(n - 1, n - 1 + s)
        if lo <= hi:
            g[lo:hi + 1] += wk * (1.0 - h[lo - s:hi + 1 - s])
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        if lo <= hi:
            g[lo:hi + 1] -= wk * h[lo + s:hi + 1 + s]
    return g

def _rp(h, tgt):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-10:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i])
                    h[i] += a
                    d -= a
                else:
                    a = min(-d, h[i])
                    h[i] -= a
                    d += a
                if abs(d) < 1e-13:
                    break
    return h

def _lf(x, alpha, pen):
    n = len(x)
    t = n * 0.5
    return _sv(x, alpha) * 2.0 / n + pen * ((x.sum() - t) / t) ** 2

def _lfg(x, alpha, pen):
    n = len(x)
    t = n * 0.5
    g = _gs(x, alpha) * 2.0 / n
    g += 2.0 * pen * (x.sum() - t) / t / t
    return g

def _sf(x, alpha):
    n = len(x)
    return _sv(x, alpha) * 2.0 / n

def _sg(x, alpha):
    n = len(x)
    return _gs(x, alpha) * 2.0 / n

def _cf(x):
    return x.sum() - len(x) * 0.5

def _cj(x):
    return np.ones(len(x))

                     jac=_lfg, bounds=[(0.0, 1.0)] * n,

        return _rp(r.x, n * 0.5)

            bc, bh = c, hn.copy()

    # Phase 5: Coordinate descent
    if bh is not None and budget_s - (time.time() - t0) > 25:
        hp, cp = _cs(bh.copy(), bn, budget_s - (time.time() - t0) - 10, t0)
        if cp < bc:
            bc, bh = cp, hp.copy()

    # Phase 6: Final SLSQP polish
    if bh is not None and budget_s - (time.time() - t0) > 8:
        hn = _sq(bh, 100000, 150)
        c = _c5(hn, bn)
        if c < bc:
            bc, bh = c, hn.copy()

    bh = _rp(bh, bn * 0.5)
    bc = _c5(bh, bn)
    return (bh, bc, bn)
```
