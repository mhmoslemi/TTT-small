def _grad_corr(h, n):
    """Compute gradient of each correlation lag w.r.t. h.
    Returns array of shape (2n-1, n) where grad[k, i] = d(corr[k])/d(h[i])."""

    # d/dh[i0]: (1-h[i0+k])*I(valid) - h[i0-k]*I(valid)
    grad = np.zeros((2*n-1, n))
    for i0 in range(n):
        for k in range(-(n-1), n):
            # Term 1: h[i0] * (1-h[i0+k]) -> derivative (1-h[i0+k]) if i0+k in [0,n-1]
            if 0 <= i0 + k < n:

            if 0 <= i0 - k < n:

    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:

        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a

                break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _smooth(h, dx, alpha):
    c = _corr(h)
    m = c.max()
    z = np.clip(alpha * (c - m), -300, 300)
    return (m + np.log(np.sum(np.exp(z))) / alpha) * dx

def _slsqp(h, n, dx, alpha, maxiter, deadline):
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cns = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
            'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: _smooth(x, dx, alpha), h, method='SLSQP',
                       bounds=bnds, constraints=cns,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 100})
        return _repair(res.x, n)
    except Exception:
        return h

def _pair_opt(h, i, j, dx):
    dmin = max(-h[i], -(1.0 - h[j])) + 1e-9
    dmax = min(1.0 - h[i], h[j]) - 1e-9
    if dmax - dmin < 1e-8:
        return _c5(h, dx), 0.0
    hi, hj = h[i], h[j]
    base = _c5(h, dx)

    vp = _c5(h, dx)

    vm = _c5(h, dx)
    h[i], h[j] = hi, hj
    if vp >= base - 1e-14 and vm >= base - 1e-14:
        return base, 0.0
    best_d, best_v = 0.0, base
    ds = np.linspace(dmin, dmax, 20)
    for d in ds:
        if abs(d) < 0.5 * probe:
            continue
        h[i], h[j] = hi + d, hj - d
        v = _c5(h, dx)
        if v < best_v:
            best_v, best_d = v, d
    if best_d != 0.0:
        sp = (dmax - dmin) / 19.0
        for d in np.linspace(best_d - sp, best_d + sp, 14):
            d = float(np.clip(d, dmin, dmax))
            h[i], h[j] = hi + d, hj - d
            v = _c5(h, dx)
            if v < best_v:

    h[i], h[j] = hi + best_d, hj - best_d

def _full_cd(h, n, dx, rng, deadline):
    best = _c5(h, dx)
    h = h.copy()
    stall = 0
    while time.time() < deadline:
        sweep_improved = False
        order = list(range(n))
        rng.shuffle(order)
        for a in range(n):

                    continue
