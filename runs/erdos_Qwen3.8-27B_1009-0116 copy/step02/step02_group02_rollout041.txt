def _grad_smooth(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - h[i + lag])
                g[i + lag] -= wk * h[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - h[i - L])
                g[i - L] -= wk * h[i]
    return g

def _fista(h0, alpha, target, iters, dx):
    n = len(h0)
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    best = _c5(x, dx)
    best_x = x.copy()
    step = 2.0 / (3.0 * alpha * n)
    for _ in range(iters):
        g = _grad_smooth(z, alpha)
        y = z - step * g
        xn = _proj(y, target)
        for _ in range(6):
            if _c5(xn, dx) <= _c5(z, dx) + 1e-11:
                break
            step *= 0.5
            y = z - step * g
            xn = _proj(y, target)
        tnew = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tnew * (xn - x), target)
        x = xn
        t = tnew
        c = _c5(x, dx)
        if c < best:
            best = c
            best_x = x.copy()
    return best_x, best

def _cd(h, dx, target, n_passes, rng):
    n = len(h)
    h = h.copy()
    bc = _c5(h, dx)
    dpos = [0.001, 0.003, 0.005, 0.008, 0.012, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18]
    for _ in range(n_passes):
        imp = False
        for _ in range(n * 10):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            for d in dpos:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < bc - 1e-13:
                    bc = c
                    imp = True
                    break
                h[i], h[j] = hi0, hj0
            else:
                h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, bc

def _slsqp_obj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum() / len(s)) / a

def _slsqp_con(x, target):
    return float(np.sum(x)) - target

def _rem(t0, budget):
    return budget - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = None

    # Build candidate list of (n, initial_h)
    candidates = []
    if h_parent is not None:
        candidates.append((len(h_parent), _proj(h_parent, len(h_parent) / 2.0)))

    for n_try in [60, 64, 70, 72, 76, 80, 84, 90, 100, 120]:
        if h_parent is not None and n_try == len(h_parent):
            continue
        x = np.arange(n_try)
        h_raw = 0.5 + 0.25 * np.sin(2 * np.pi * x / n_try) + 0.15 * np.sin(4 * np.pi * x / n_try + 0.7)
        candidates.append((n_try, _proj(h_raw, n_try / 2.0)))

    candidates.sort(key=lambda t: abs(t[0] - 70))

    best_h = None
    best_c = np.inf
    best_n = 70

    for n, h0 in candidates:
        if _rem(t0, budget_s) < 40:
            break
        dx = 2.0 / n
        target = n / 2.0
        h = h0.copy()
        c = _c5(h, dx)

        # FISTA sweep
        for alpha in [15, 40, 100, 300, 800, 2000, 5000]:
            if _rem(t0, budget_s) < 30:
                break
            h, c = _fista(h, alpha, target, 2500, dx)

        # CD round 1
        if _rem(t0, budget_s) < 20:
            break
        h, c = _cd(h, dx, target, 400, rng)

        # SLSQP
        if _rem(t0, budget_s) < 15:
            break
        bnds = [(0.0, 1.0)] * n
        for a_pol in [3000, 8000]:
            if _rem(t0, budget_s) < 12:
                break
            try:
                r = minimize(_slsqp_obj, h, args=(a_pol,), method='SLSQP',
                             bounds=bnds,
                             constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (target,)}],
                             options={'maxiter': 500, 'ftol': 1e-15})
                hf = _proj(r.x, target)
                cf = _c5(hf, dx)
                if cf < c:
                    c = cf
                    h = hf
            except Exception:
                pass

        # CD round 2
        if _rem(t0, budget_s) < 15:
            break
        h, c = _cd(h, dx, target, 300, rng)

        # Random restarts from current best
        for s in [0.05, 0.1, 0.15, 0.25, 0.4]:
            if _rem(t0, budget_s) < 20:
                break
            hp = _proj(h + s * rng.randn(n), target)
            h2, c2 = _fista(hp, 500, target, 2000, dx)
            if c2 < c:
                h, c = h2, c2
                h2, c2 = _cd(h, dx, target, 150, rng)
                if c2 < c:
                    h, c = h2, c2

        if c < best_c:
            best_c = c
            best_h = h
            best_n = n

    # Final intensive polish on best
    if best_h is not None and _rem(t0, budget_s) > 15:
        dx = 2.0 / best_n
        target = best_n / 2.0
        best_h, best_c = _cd(best_h, dx, target, 800, rng)
        if _rem(t0, budget_s) > 12:
            try:
                bnds = [(0.0, 1.0)] * best_n
                r = minimize(_slsqp_obj, best_h, args=(20000,), method='SLSQP',
                             bounds=bnds,
                             constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (target,)}],
                             options={'maxiter': 1000, 'ftol': 1e-16})
                hf = _proj(r.x, target)
                cf = _c5(hf, dx)
                if cf < best_c:
                    best_c = cf
                    best_h = hf
            except Exception:
                pass
        if _rem(t0, budget_s) > 8:
            best_h, best_c = _cd(best_h, dx, target, 500, rng)

    best_h = _proj(best_h, best_n / 2.0)
    best_c = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c, best_n)
