import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = float(np.clip(y-mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0-x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _smo_c5(h, dx, alpha):
    c = np.correlate(h, 1.0-h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha*(c-m), -400.0, 400.0)
    e = np.exp(s)
    return m + np.log(float(e.sum()))/alpha

def _smo_grad(h, dx, alpha):

    s = np.clip(alpha*(c-m), -400.0, 400.0)

        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n-1)

                g[i+lag] -= wk*h[i]*dx

                g[i] += wk*(1.0-h[i-L])*dx
                g[i-L] -= wk*h[i]*dx
    return g

    z = h0.copy()

    bx = x.copy()

            if fn <= quad + 1e-10:

        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _full_cd(h, dx, n, bc, passes, rng):

                        imp += 1
                        break
                    h[i], h[j] = hi0, hj0
                else:
                    h[i], h[j] = hi0, hj0
        if imp == 0:

        else:

import time

    c = np.correlate(h, 1.0 - h, mode='full')

def _proj(y, target):

    lo = float(y.min()) - 2.0

    for _ in range(60):

        if s > target:

        else:

    return x

def _smo_obj(x, a, dx):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(a * (c - m), -400.0, 400.0)
    e = np.exp(s)
    return m + np.log(float(e.sum())) / a

def _smo_grad(x, a, dx):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(a * (c - m), -400.0, 400.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - x[i + lag]) * dx
                g[i + lag] -= wk * x[i] * dx
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L]) * dx
                g[i - L] -= wk * x[i] * dx
    return g

def _fista(h0, alpha, tgt, dx, iters):
    n = len(h0)
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (2.0 * alpha * dx * n + 1.0)
    for _ in range(iters):
        gz = _smo_grad(z, alpha, dx)
        y = z - step * gz
        xn = _proj(y, tgt)
        for _ in range(10):
            fn = _smo_obj(xn, alpha, dx)
            quad = _smo_obj(z, alpha, dx) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-9:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, tgt)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), tgt)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _cd(h, dx, n, bc, passes, dvals, t0, budget):
    h = h.copy()
    dv = list(dvals)
    for _ in range(passes):
        if time.time() - t0 > budget - 10:
            break
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                hi0, hj0 = h[i], h[j]
                found = False
                for d in dv:
                    ni, nj = hi0 + d, hj0 - d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    ni = max(0.0, min(1.0, ni))
                    nj = max(0.0, min(1.0, nj))
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < bc - 1e-13:
                        bc = c
                        imp = True
                        found = True
                        break
                if not found:
                    h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, bc

def _slsqp_f(x, a, dx):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(a * (c - m), -400.0, 400.0)
    e = np.exp(s)
    return m + np.log(float(e.sum())) / a

def _slsqp_g(x, a, dx):
    return _smo_grad(x, a, dx)

def _slsqp_c(x, tgt):
    return float(x.sum()) - tgt

def _try_n(n_t, alpha, iters, t0, budget, rng):
