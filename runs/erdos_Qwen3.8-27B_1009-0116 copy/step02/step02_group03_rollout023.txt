                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _project(res.x, n)
    except Exception:
        return h

def _make_starts(n, seed, count):
    rng = np.random.RandomState(seed)
    starts = []
    h1 = 0.5 + 0.25 * rng.randn(n)
    starts.append(_project(h1, n))
    h2 = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.12 * rng.randn(n)
    starts.append(_project(h2, n))
    h3 = np.zeros(n)
    a, b = n // 3, 2 * n // 3
    h3[:a] = 1.0
    h3[b:] = 1.0
    h3 += 0.08 * rng.randn(n)
    starts.append(_project(h3, n))
    for k in range(count - 3):
        freq = rng.randint(2, 8)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        h += 0.06 * rng.randn(n)
        starts.append(_project(h, n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    def try_update(h, n):
        nonlocal best_h, best_c5, best_n
        dx = 2.0 / n
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    init_h = None
    init_n = 0
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass

    n_list = [70]
    if init_n > 0 and init_n != 70:
        n_list.insert(0, init_n)
    n_list += [60, 80, 90, 100]
    n_list = sorted(set(n_list))

    time_per_n = (budget_s - 100) / len(n_list)

    for n in n_list:
        if time.time() - t0 > budget_s - 80:
            break
        dx = 2.0 / n
        n_starts = 5
        starts = []
        if init_h is not None and init_n == n:
            starts.append(_project(init_h, n))
        starts += _make_starts(n, seed + n * 131, n_starts)

        for si, h0 in enumerate(starts):
            if time.time() - t0 > budget_s - 60:
                break
            h = h0.copy()
            try_update(h, n)

            # Phase A: Subgradient momentum
            h_sub = _subgrad_momentum(h, n, 200, 0.15, 0.995)
            try_update(h_sub, n)
            if _c5(h_sub, dx) < _c5(h, dx):
                h = h_sub

            # Phase B: L-BFGS-B with smooth max
            for alpha in [80, 200, 500]:
                if time.time() - t0 > budget_s - 60:
                    break
                for tk in [3, 5]:
                    h_new = _lbfgs_run(h, n, alpha, tk, 200)
                    try_update(h_new, n)
                    if _c5(h_new, dx) < _c5(h, dx):
                        h = h_new

            # Phase C: More subgradient with smaller lr
            h_sub2 = _subgrad_momentum(h, n, 300, 0.05, 0.999)
            try_update(h_sub2, n)
            if _c5(h_sub2, dx) < _c5(h, dx):
                h = h_sub2

            # Phase D: SLSQP refinement
            for alpha in [500, 1500, 3000]:
                if time.time() - t0 > budget_s - 60:
                    break
                h_new = _slsqp_run(h, n, alpha, 4, 200)
                try_update(h_new, n)
                if _c5(h_new, dx) < _c5(h, dx):
                    h = h_new

            # Phase E: Coordinate search
            h_cd, c5_cd = _coord_search(h, n, dx, 80, t0 + budget_s - 40, rng)
            try_update(h_cd, n)
            if c5_cd < _c5(h, dx):
                h = h_cd

        # Extra polish for initial construction
        if init_h is not None and init_n == n:
            h_extra = _subgrad_momentum(h, n, 500, 0.02, 0.9995)
            try_update(h_extra, n)
            if _c5(h_extra, dx) < _c5(h, dx):
                h = h_extra
            for alpha in [5000, 10000]:
                if time.time() - t0 > budget_s - 40:
                    break
                h_new = _slsqp_run(h, n, alpha, 3, 300)
                try_update(h_new, n)
                if _c5(h_new, dx) < _c5(h, dx):
                    h = h_new
            h_cd, _ = _coord_search(h, n, dx, 100, t0 + budget_s - 20, rng)
            try_update(h_cd, n)

    # Final polish
    if best_h is not None:
        dx_b = 2.0 / best_n
        h_fin, _ = _coord_search(best_h, best_n, dx_b, 150, t0 + budget_s - 10, rng)
        try_update(h_fin, best_n)
        if time.time() - t0 < budget_s - 15:
            h_fin2 = _slsqp_run(best_h, best_n, 8000, 3, 200)
            try_update(h_fin2, best_n)

    if best_h is None:
        n = 70
        best_h = _project(np.full(n, 0.5), n)
        best_c5 = _c5(best_h, 2.0 / n)
        best_n = n

    best_h = _project(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
