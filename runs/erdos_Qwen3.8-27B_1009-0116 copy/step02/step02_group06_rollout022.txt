        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _slsqp_soft(h0, alpha, n, bounds, cons, maxit):
    try:
        res = minimize(_sm_obj, h0, args=(alpha, n), jac=_sm_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5_val(best_h, dx)

    # Phase 1: Coordinate pair descent (fast, incremental G)
    G = _all_G(h, n)
    c = _corr(h)
    for _ in range(6):
        if time.time() - t0 > budget_s - 550:
            break
        h, c, G = _sweep(h, n, G, c, max_ro=200)
        cv = float(np.max(c) * dx)
        if cv < best_c5:
            best_c5 = cv
            best_h = h.copy()

    # Phase 2: SLSQP with softmax (increasing alpha)
    for alpha in [30, 60, 120, 250, 500]:
        if time.time() - t0 > budget_s - 380:
            break
        hc = _slsqp_soft(best_h, alpha, n, bounds, cons, 800)
        c5 = _c5_val(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 3: More coordinate descent after SLSQP
    if time.time() - t0 < budget_s - 250:
        h = best_h.copy()
        G = _all_G(h, n)
        c = _corr(h)
        for _ in range(6):
            if time.time() - t0 > budget_s - 250:
                break
            h, c, G = _sweep(h, n, G, c, max_ro=300)
            cv = float(np.max(c) * dx)
            if cv < best_c5:
                best_c5 = cv
                best_h = h.copy()

    # Phase 4: High-alpha SLSQP for tightness
    for alpha in [800, 1500, 3000, 6000]:
        if time.time() - t0 > budget_s - 120:
            break
        hc = _slsqp_soft(best_h, alpha, n, bounds, cons, 1200)
        c5 = _c5_val(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 5: Perturbation restarts
    for pert in [0.08, 0.05, 0.03, 0.06, 0.04]:
        if time.time() - t0 > budget_s - 50:
            break
        h_p = best_h + pert * rng.randn(n)
        h_p = _repair(h_p, n)
        G_p = _all_G(h_p, n)
        c_p = _corr(h_p)
        h_p, c_p, G_p = _sweep(h_p, n, G_p, c_p, max_ro=150)
        hp2 = _slsqp_soft(h_p, 400, n, bounds, cons, 500)
        G_p2 = _all_G(hp2, n)
        c_p2 = _corr(hp2)
        hp2, c_p2, _ = _sweep(hp2, n, G_p2, c_p2, max_ro=200)
        c5 = float(np.max(c_p2) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hp2

    # Phase 6: Final fine polish
    if time.time() - t0 < budget_s - 10:
        h = best_h.copy()
        G = _all_G(h, n)
        c = _corr(h)
        h, c, G = _sweep(h, n, G, c, max_ro=500)
        cv = float(np.max(c) * dx)
        if cv < best_c5:
            best_c5 = cv
            best_h = h
        try:
            hc = _slsqp_soft(best_h, 5000, n, bounds, cons, 1500)
            c5 = _c5_val(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
