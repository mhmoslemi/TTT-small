    golden = (np.sqrt(5)-1)/2
    a, b_ = d_lo, d_hi
    c1 = a + (1-golden)*(b_ - a)
    c2 = a + golden*(b_ - a)
    f1 = float(np.max(c + c1*b))
    f2 = float(np.max(c + c2*b))

            c1 = a + (1-golden)*(b_ - a)

            c2 = a + golden*(b_ - a)
            f2 = float(np.max(c + c2*b))

    for _ in range(50):
        mid = 0.5*(lo+hi)
        if float(np.clip(y-mid, 0.0, 1.0).sum()) > target: lo = mid
        else: hi = mid

            else:

```python
def _golden_max(c, b, q, lo, hi, iters=35):
    if hi - lo < 1e-14: return 0.5*(lo+hi)
    gr = (np.sqrt(5.0)-1.0)/2.0
    c1 = lo + (1.0-gr)*(hi-lo)
    c2 = lo + gr*(hi-lo)
    f1 = float(np.max(c + c1*b + c1*c1*q))
    f2 = float(np.max(c + c2*b + c2*c2*q))
    for _ in range(iters):
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = lo + (1.0-gr)*(hi-lo)
            f1 = float(np.max(c + c1*b + c1*c1*q))
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr*(hi-lo)
            f2 = float(np.max(c + c2*b + c2*c2*q))
    return 0.5*(lo+hi)
```

And compute q:
```python
def _pair_q(n, i, j):
    q = np.zeros(2*n-1)
    k1 = n-1 + j - i
    k2 = n-1 + i - j
    if 0 <= k1 < 2*n-1: q[k1] = 1.0
    if 0 <= k2 < 2*n-1: q[k2] = 1.0
    return q
```
