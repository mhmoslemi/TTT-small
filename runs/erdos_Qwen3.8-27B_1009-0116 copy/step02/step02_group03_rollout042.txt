For a step function with integral 1 over [0,2], the "average" value is 0.5. The overlap integral ∫h(x)(1-h(x+k))dx measures how much h is "high" where h shifted by k is "low".

If h were a perfect 50/50 step function (1 on [0,1], 0 on [1,2]), then:
- For k=0: ∫h(1-h) = 0 (since h is 0 or 1)

- Actually for k=1: ∫₀¹ 1*(1-0) dx = 1, and ∫₁² 0*(1-1) dx = 0, so total = 1

So a step function gives C5 ≈ 1, which is terrible. We need a more "oscillatory" function.

The optimal structure should have h oscillating rapidly enough that for any shift k, the overlap h(x)(1-h(x+k)) is small on average. This suggests high-frequency oscillations.

For a sinusoidal h(x) = 0.5 + A*sin(2πf x), the overlap for shift k would involve cos terms. The max would be reduced by having high frequency.

But we're constrained to [0,1], so A ≤ 0.5. With A=0.5 (h is 0 or 1), we get a square wave.

The optimal might be a combination of multiple frequencies to "spread out" the correlations.

This matches the approach: use multi-frequency sinusoidal initializations and optimize.

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_n = 0
_g_pen = 0.0

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(600):
        s = h.sum()
        if s < 1e-12:
            return np.full(n, 0.5)
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-10:
            break
        diff = target - h.sum()
        idxs = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in idxs:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = -min(-diff, h[idx])
            h[idx] += a
    return h

def _init_h(n, seed, kind):
    rng = np.random.RandomState(seed)
    x = np.arange(n) / n
    if kind == 0:
        h = 0.5 + 0.35 * rng.randn(n)
    elif kind == 1:
        h = np.where(x < 0.5, 1.0, 0.0) + 0.2 * rng.randn(n)
    elif kind == 2:
        h = np.zeros(n)
        h[:n//3] = 1.0
        h[2*n//3:] = 1.0
        h += 0.12 * rng.randn(n)
    elif kind == 3:
        f = rng.randint(2, 8)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28))
    elif kind == 4:
        blk = max(1, n // 8)
        h = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h[i:i+blk] = 1.0
        h += 0.15 * rng.randn(n)
    elif kind == 5:
        h = 0.5 + 0.28*np.sin(2*np.pi*3*x) + 0.18*np.sin(2*np.pi*7*x) + 0.06*rng.randn(n)
    elif kind == 6:
        h = 0.5 + 0.2*np.sin(2*np.pi*5*x) + 0.2*np.sin(2*np.pi*11*x + 0.7) + 0.05*rng.randn(n)
    else:
        h = rng.uniform(0, 1, n)
    return _repair(h, n)

def _obj_smooth(x):
    return _smooth_max(_corr(x), _g_alpha)

def _obj_pen(x):
    return _smooth_max(_corr(x), _g_alpha) + _g_pen * (np.sum(x) - _g_n * 0.5) ** 2

def _con_sum(x):
    return np.sum(x) - _g_n * 0.5

def _de_gen(pop, n, alpha, f, cr, rng):
    size = len(pop)
    fits = np.array([_smooth_max(_corr(h), alpha) for h in pop])
    new_pop = [h.copy() for h in pop]
    for i in range(size):
        others = [j for j in range(size) if j != i]
        a, b, c = rng.choice(others, 3, replace=False)
        mutant = pop[a] + f * (pop[b] - pop[c])
        mask = rng.rand(n) >= cr
        mutant[mask] = pop[i][mask]
        mutant = _repair(mutant, n)
        ft = _smooth_max(_corr(mutant), alpha)
        if ft < fits[i]:
            new_pop[i] = mutant
            fits[i] = ft
    return new_pop

def _coord_target(h, n, dx, n_iter, t_end, rng):
    h = h.copy()
    best = _c5(h, dx)
    dele = [0.002, 0.005, 0.01, 0.02, 0.04, 0.07]
    no_imp = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        improved = False
        corr = _corr(h)
        k_max = int(np.argmax(corr))
        for _t in range(25):
            i = rng.randint(n)
            if rng.rand() < 0.7:
                j = rng.randint(n)
            else:
                j = (i + k_max) % n
            if i == j:
                continue
            for d in dele:
                for sgn in [1, -1]:
                    dd = d * sgn
                    ni, nj = h[i] + dd, h[j] - dd
                    if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < best:
                        best = c
                        improved = True
                        break
                    h[i], h[j] = ni - dd, nj + dd
                if improved:
                    break
        no_imp = no_imp + 1 if not improved else 0
        if no_imp > 300:
            h = _repair(h + 0.04 * rng.randn(n), n)
            no_imp = 0
            best = _c5(h, dx)
    return h, best

def _slsqp_refine(h, n, alpha, maxiter):
    global _g_alpha, _g_n
    _g_alpha = alpha
    _g_n = n
    try:
        res = minimize(_obj_smooth, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _con_sum}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs_refine(h, n, alpha, pen, maxiter):
    global _g_alpha, _g_n, _g_pen
    _g_alpha = alpha
    _g_n = n
    _g_pen = pen
    try:
        res = minimize(_obj_pen, h, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 70

    def _upd(h, n):

        c = _c5(h, 2.0 / n)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
            best_n = n

    n_list = [60, 70, 80, 90]
    has_init = False
    init_n = 0
    try:
        init_n = len(initial_h_values)

        if init_n not in n_list:

    except NameError:

    n_main = init_n if has_init else 70

    # Phase 1: DE global search
    t_end1 = t0 + budget_s * 0.30
    pop_size = 14
    pop = []
    if has_init and init_n == n_main:

    f_de, cr_de = 0.85, 0.75

        if gen % 40 == 39 and gen > 0:

    # Phase 2: Local refinement on best
    h_cur = best_h.copy() if best_h is not None else np.full(n_main, 0.5)
    n_cur = best_n
