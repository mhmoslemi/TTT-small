    if 0 <= idx_j1 < n:

    if 0 <= idx_j2 < n:
        term += h[idx_j2]

    if k >= len(corr):
        return float(corr.sum())
    idx = np.argpartition(corr, -k)[-k:]
    return float(corr[idx].sum())

    s = np.sum(np.exp(np.clip(alpha * (corr - m), -50, 50)))
