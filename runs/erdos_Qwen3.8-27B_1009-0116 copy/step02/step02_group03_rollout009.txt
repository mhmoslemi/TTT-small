def _local_search_bin(positions, n, max_iters=50):
    """Try all single-swap mutations and keep improvements"""
    pos_set = frozenset(positions)
    zeros = [i for i in range(n) if i not in pos_set]
    best_set = set(positions)
    best_c = _c5_from_set(best_set, n)
    for _ in range(max_iters):
        improved = False
        best_swap = None
        best_swap_c = best_c
        for one in list(best_set):
            for zero in zeros:
                if zero in best_set:
                    continue

                new_set.remove(one)

            best_set.remove(best_swap[0])
            best_set.add(best_swap[1])
            best_c = best_swap_c
            improved = True
        if not improved:
            break
    return list(best_set), best_c

One key insight: for a binary sequence, the correlation at lag k (for positive k) is:
sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])

This counts the number of "1→0" transitions at distance k. If we think in terms of the positions of 1s, say p_1 < p_2 < ... < p_m (m = n/2), then:
- At lag k, the count is the number of pairs (p_i, p_j) where p_j - p_i = k... no that's not right either.

Actually, sum_{i} h[i]*(1-h[i+k]) counts positions i where h[i]=1 and h[i+k]=0. For a binary sequence, this is:
- Total 1s minus the number of i where h[i]=1 AND h[i+k]=1

    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_con(x, target):
    return np.sum(x) - target

    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

                       constraints=cons, args=(alpha, target),
                       options={'maxiter': maxiter, 'ftol': 1e-14})

    if init_n > 0:
        if init_n not in n_candidates:
            n_candidates.append(init_n)
        n_candidates.remove(init_n)
        n_candidates = [init_n] + n_candidates

    # Phase 1: Binary local search from initial
    if init_h is not None and init_n % 2 == 0:
        h_bin = _round_binary(init_h, init_n)
        h_bin, c_bin = _bin_local_search(h_bin, init_n, t0 + min(60, budget_s * 0.15))
        _try_update(h_bin, init_n)
        # SLSQP continuous refinement
        for alpha in [50, 200, 800]:
            if time.time() > t0 + budget_s * 0.3:
                break
            h_r = _slsqp_refine(h_bin, init_n, alpha, 200, t0 + budget_s * 0.3)
            _try_update(h_r, init_n)
            if _c5_val(h_r, init_n) < _c5_val(h_bin, init_n):
                h_bin = h_r
        # Coordinate descent
        h_cd, c_cd = _coord_descend(h_bin, init_n, t0 + budget_s * 0.4, 100)
        _try_update(h_cd, init_n)

    # Phase 2: Binary GA for various n
    for n in n_candidates:
        if time.time() > t0 + budget_s * 0.75:
            break
        if n % 2 != 0:
            continue
        phase_end = t0 + budget_s * 0.75
        # Binary GA
        h_ga, c_ga = _ga_binary(n, pop_size=120, max_gens=400, seed=seed + n * 13, t_end=phase_end - 60)
        _try_update(h_ga, n)
        # Binary local search on GA result
        h_ls, c_ls = _bin_local_search(h_ga, n, phase_end - 40)
        _try_update(h_ls, n)
        # Brief anneal for diversity
        if time.time() < phase_end - 30:
            h_an, c_an = _bin_anneal(h_ls, n, phase_end - 30, seed + n)
            _try_update(h_an, n)
            if c_an < c_ls:
                h_ls = h_an
            h_ls, c_ls = _bin_local_search(h_ls, n, phase_end - 20)
            _try_update(h_ls, n)
        # Continuous refinement
        if time.time() < phase_end:
            h_ref = _slsqp_refine(h_ls, n, 100, 150, phase_end)
            _try_update(h_ref, n)
            h_cd, _ = _coord_descend(h_ref, n, t0 + budget_s - 30, 80)
            _try_update(h_cd, n)

    # Phase 3: Final polish on global best
    if best_h is not None:
        polish_end = t0 + budget_s - 5
        h_cd, _ = _coord_descend(best_h, best_n, polish_end, 150)
        _try_update(h_cd, best_n)
        if time.time() < polish_end:
            h_s = _slsqp_refine(best_h, best_n, 200, 100, polish_end)
            _try_update(h_s, best_n)
            h_cd2, _ = _coord_descend(h_s, best_n, polish_end, 50)
            _try_update(h_cd2, best_n)

    if best_h is None:
        n = 70
        best_h = _repair(np.full(n, 0.5), n)
        best_c5 = _c5_val(best_h, n)
        best_n = n

    best_h = _repair(best_h, best_n)
    best_c5 = _c5_val(best_h, best_n)
    return (best_h, best_c5, best_n)
