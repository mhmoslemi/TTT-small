    h = h.copy()
    best_c5 = _c5_of(h, dx)
    for _ in range(max_passes):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_gain = 1e-10
        best_i = best_j = -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = _c5_of(h, dx)
                h[i] = 1.0
                h[j] = 0.0
                gain = best_c5 - c
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        best_c5 -= best_gain
    return h, best_c5

def _perturb_bin(h, n, rng, n_flips):
    h = h.copy()
    for _ in range(n_flips):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
    return h

def _make_bin(n, seed, kind):
    rng = np.random.RandomState(seed)
    k = n // 2
    h = np.zeros(n)
    if kind == 0:
        h[rng.choice(n, k, replace=False)] = 1.0
    elif kind == 1:
        h[:k] = 1.0
    elif kind == 2:
        p = max(2, n // 10)
        h[::p] = 1.0
        cnt = int(h.sum())
        if cnt > k:
            rm = rng.choice(np.where(h == 1)[0], cnt - k, replace=False)
            h[rm] = 0
        elif cnt < k:
            ad = rng.choice(np.where(h == 0)[0], k - cnt, replace=False)
            h[ad] = 1
    elif kind == 3:
        p = rng.randint(7, 17)
        ph = rng.randint(0, p)
        h = ((np.arange(n) + ph) % p < p // 2).astype(float)
        cnt = int(h.sum())
        if cnt > k:
            rm = rng.choice(np.where(h == 1)[0], cnt - k, replace=False)
            h[rm] = 0
        elif cnt < k:
            ad = rng.choice(np.where(h == 0)[0], k - cnt, replace=False)
            h[ad] = 1
    elif kind == 4:
        for i in range(n):
            s, v = 0, i
            while v:
                s = (s + (v & 1) * (v >> 1 & 1)) & 1
                v >>= 1
            h[i] = 1.0 if s == 0 else 0.0
        cnt = int(h.sum())
        if cnt > k:
            rm = rng.choice(np.where(h == 1)[0], cnt - k, replace=False)
            h[rm] = 0
        elif cnt < k:
            ad = rng.choice(np.where(h == 0)[0], k - cnt, replace=False)
            h[ad] = 1
    elif kind == 5:
        gr = (1.0 + 5.0 ** 0.5) / 2.0
        for i in range(n):
            h[i] = 1.0 if (i * gr) % 1.0 < 0.5 else 0.0
        cnt = int(h.sum())
        if cnt > k:
            rm = rng.choice(np.where(h == 1)[0], cnt - k, replace=False)
            h[rm] = 0
        elif cnt < k:
            ad = rng.choice(np.where(h == 0)[0], k - cnt, replace=False)
            h[ad] = 1
    else:
        h[rng.choice(n, k, replace=False)] = 1.0
    return h

def _obj_smooth(x):
    corr = _corr_full(x)
    m = np.max(corr)
    s = _g_alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _g_alpha + 5000.0 * (np.sum(x) - _g_n / 2.0) ** 2

def _repair_cont(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s < 1e-12:
            h = np.full(n, 0.5)
            break
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _cont_refine(h, n, time_limit, t0):
    global _g_alpha, _g_n
    best_h = h.copy()
    best_c5 = _c5_of(h, 2.0 / n)
    n_orig = n
    _g_n = n_orig
    for alpha in [100, 300, 800, 2000]:
        if time.time() - t0 > time_limit:
            break
        _g_alpha = alpha
        try:
            bounds = [(0.0, 1.0)] * n_orig
            res = minimize(_obj_smooth, best_h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_new = _repair_cont(res.x, n_orig)
            c5_new = _c5_of(h_new, 2.0 / n_orig)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 0
    time_limit = t0 + budget_s - 10

    n_list = [n for n in range(50, 150, 2)
              if n in [50,54,56,58,60,62,64,66,68,70,72,74,76,78,80,82,84,86,88,90,92,94,96,98,100,104,108,112,116,120,124,128,132,136,140,144]]

    has_init = False
    init_n = 0
    try:
        init_n = len(initial_h_values)
        has_init = True
    except NameError:
        pass

    if has_init:
        if init_n in n_list:
            n_list.remove(init_n)
        n_list = [init_n if init_n % 2 == 0 else init_n - 1] + n_list

    per_n_budget = (budget_s - 80) / len(n_list)

    for n in n_list:
        if time.time() > time_limit:
            break
        dx = 2.0 / n
        start_t = time.time()
        n_time_budget = min(per_n_budget, 80.0)

        starts = []
        if has_init and n == init_n:
            h_init = np.array(initial_h_values, dtype=float).copy()
            starts.append(h_init)
            starts.append(_to_binary(h_init, n))
        elif has_init and n != init_n:
            h_init = np.array(initial_h_values, dtype=float).copy()
            x_new = np.linspace(0, 2, n)
            x_old = np.linspace(0, 2, init_n)
            h_interp = np.interp(x_new, x_old, h_init)
            starts.append(_to_binary(h_interp, n))
        for kind in range(7):
            starts.append(_make_bin(n, seed + kind * 991 + n * 13, kind))
        for extra in range(3):
            starts.append(_make_bin(n, seed + rng.randint(99999) + n * 7, 0))

        local_best = None
        local_c5 = np.inf

        for h0 in starts:
            if time.time() > start_t + n_time_budget:
                break
            h_opt, c5_opt = _swap_search(h0, n, dx, 400, time_limit, t0)
            if c5_opt < local_c5:
                local_c5 = c5_opt
                local_best = h_opt.copy()

        if local_best is None:
            continue

        # ILS phase
        il_c5 = local_c5
        il_h = local_best.copy()
        il_stagnant = 0
        max_il_iters = max(50, int((n_time_budget * 0.5) / 0.05))

        for _ in range(max_il_iters):
            if time.time() > start_t + n_time_budget:
                break
            n_flips = rng.randint(1, 4)
            h_pert = _perturb_bin(il_h, n, rng, n_flips)
            h_opt, c5_opt = _swap_search(h_pert, n, dx, 300, time_limit, t0)
            if c5_opt < il_c5:
                il_c5 = c5_opt
                il_h = h_opt.copy()
                il_stagnant = 0
            else:
                il_stagnant += 1
            if il_stagnant > 80:
                break

        local_c5 = min(local_c5, il_c5)
        if il_c5 < local_c5:
            local_best = il_h.copy()

        # Continuous refinement
        if time.time() < start_t + n_time_budget:
            h_cont, c5_cont = _cont_refine(local_best, n, time_limit, t0)
            if c5_cont < local_c5:
                local_c5 = c5_cont
                local_best = h_cont.copy()
            # Round back to binary and re-optimize
            h_b = _to_binary(local_best, n)
            h_b, c5_b = _swap_search(h_b, n, dx, 400, time_limit, t0)
            if c5_b < local_c5:
                local_c5 = c5_b
                local_best = h_b.copy()

        if local_best is not None:
            c5_final = _c5_of(local_best, dx)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = local_best.copy()
                best_n = n

    # Final polish on global best
    if best_h is not None and time.time() < time_limit:
        dx_b = 2.0 / best_n
        h_pol, c5_pol = _swap_search(best_h, best_n, dx_b, 500, time_limit, t0)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        if time.time() < time_limit:
            h_c, c5_c = _cont_refine(best_h, best_n, time_limit, t0)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    if best_h is None:
        n = 70
        best_h = _to_binary(np.full(n, 0.5), n)
        best_c5 = _c5_of(best_h, 2.0 / n)
        best_n = n

    return (best_h, best_c5, best_n)
