import time

def _correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_correlation(h)) * dx)

def _smooth_max(vals, alpha):
    m = np.max(vals)
    if m > 100:
        return m  # overflow protection
    s = np.sum(np.exp(alpha * (vals - m)))
    return m + np.log(s) / alpha

def _smooth_obj_and_grad(h, alpha):
    c = _correlation(h)
    m = np.max(c)
    ex = np.exp(alpha * (c - m))
    weights = ex / np.sum(ex)  # softmax weights
    # Smooth objective
    fval = m + np.log(np.sum(ex)) / alpha
    # Gradient: sum_k weights[k] * grad_c[k]
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        # d c[k]/d h[j] = (1-h[j-lag]) if valid - h[j+lag] if valid

            jl = j - lag
            if 0 <= jl < n:
                gj += 1.0 - h[jl]
            jr = j + lag
            if 0 <= jr < n:
                gj -= h[jr]
            grad[j] += w * gj
    return fval, grad

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):

def _grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag > 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    elif lag < 0:
        L = -lag
        g[:n - L] += 1.0 - h[L:]
        g[L:] -= h[:n - L]
    else:
        g = 1.0 - 2.0 * h
    return g

    c = _corr(h)
    m = np.max(c)

    ex = np.exp(sh)
    s = np.sum(ex)
    w = ex / s

    t_ultra = np.array([0.003, -0.003, 0.005, -0.005, 0.008, -0.008, 0.012, -0.012])
    h5, c5_5 = _pair_search(best_h.copy(), target, dx, t_end, t_ultra, 150)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Final repair
    best_h = _project(best_h, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
