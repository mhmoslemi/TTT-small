def _minimax_1d(corr, g, d_min, d_max):
    # Find d in [d_min, d_max] that minimizes max_k (corr[k] + d * g[k])
    n_lines = len(corr)
    # Check boundaries
    best_d = d_min
    best_val = max(corr + d_min * g)
    val_max = max(corr + d_max * g)
    if val_max < best_val:
        best_d = d_max
        best_val = val_max
    
    # Check all pairwise intersections
    for a in range(n_lines):
        for b in range(a+1, n_lines):
            dg = g[a] - g[b]
            if abs(dg) < 1e-15:
                continue
            d = (corr[b] - corr[a]) / dg
            if d < d_min - 1e-15 or d > d_max + 1e-15:

def _grad_smax(x, alpha):
    c = _corr(x)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    # Gradient of smooth_max w.r.t. x
    # smooth_max = m + log(sum(exp(alpha*(c-m))))/alpha
    # dc/dx is the correlation gradient
    # grad = alpha * w * dc/dx / alpha = w * dc/dx
    # But m depends on x too... 
    # Actually, the standard log-sum-exp trick:
    # f(x) = m + (1/alpha) * log(sum(exp(alpha*(c_k - m))))
    # = m + (1/alpha) * log(sum(exp(alpha*c_k)) / exp(alpha*m))
    # = m + (1/alpha) * log(sum(exp(alpha*c_k))) - m
    # = (1/alpha) * log(sum(exp(alpha*c_k)))
    # So f(x) = (1/alpha) * log(sum(exp(alpha*c_k)))
    # df/dx_k = (1/alpha) * sum_j alpha*exp(alpha*c_j) * dc_j/dx_k / sum_j exp(alpha*c_j)
    #          = sum_j exp(alpha*c_j)/sum_j exp(alpha*c_j) * dc_j/dx_k
    #          = sum_j w_j * dc_j/dx_k
    # where w_j = exp(alpha*(c_j - m)) / sum(exp(alpha*(c_l - m)))
    
    # dc_j/dx for correlate(h, 1-h):
    # c_j = sum_i h[i]*(1-h[i+k]) where k = j-(n-1)
    # dc_j/dx_p = (1-h[p-k]) - h[p+k] (with valid index checks)
    # This is _g_vec but with a different sign convention...
    # Actually _g_vec gives the gradient for increasing h[p], which is:
    # dc_j/dh_p = (1-h[p-k]) - h[p+k] (where k=j-(n-1))
    # Same as _g_vec(h, p, n)[j]
    
    n = len(x)
    grad = np.zeros(n)
    for p in range(n):
        gp = _g_vec(x, p, n)
        grad[p] = float(np.dot(w, gp))
    return grad
