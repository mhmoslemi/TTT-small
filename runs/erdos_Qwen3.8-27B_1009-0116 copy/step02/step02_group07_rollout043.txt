import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _lbfgs(h, n, alpha, maxiter=3000, ftol=1e-15):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _subgrad_opt(h, n, t_limit, n_iter=200000):
    dx = 2.0 / n
    offset = n - 1
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k = int(np.argmax(c)) - offset
        grad = np.zeros(n)
        if k >= 0:
            for i in range(n - k):
                grad[i] += 1.0 - h[i + k]
                grad[i + k] -= h[i]
        else:
            kk = -k
            for i in range(kk, n):
                grad[i] += 1.0 - h[i - kk]
                grad[i - kk] -= h[i]
        step = 1.0 / (5.0 + 0.02 * it)
        h_new = np.clip(h - step * grad, 0.0, 1.0)
        target = n / 2.0
        diff = target - h_new.sum()
        if abs(diff) > 1e-10:
            order = np.argsort(1.0 - h_new) if diff > 0 else np.argsort(h_new)
            for idx in order:

                    a = min(diff, 1.0 - h_new[idx])
                    h_new[idx] += a
                    diff -= a

                    a = min(-diff, h_new[idx])
                    h_new[idx] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
        h = h_new
        c5 = _c5(h, dx)
        if c5 < best_c5:

            best_h = h.copy()
    return best_h, best_c5
