```python
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _minimax_1d(A, B, lo, hi, n_grid=4001):
    """Minimize max_m(A[m] + B[m]*t) over t in [lo, hi].
    A, B are 1D arrays of same length. Returns optimal t."""
    if lo >= hi:
        return lo
    t_grid = np.linspace(lo, hi, n_grid)
    vals = np.max(A[None, :] + t_grid[:, None] * B[None, :], axis=1)
    t_best = t_grid[np.argmin(vals)]
    # Local refinement
    eps = (hi - lo) / n_grid
    t_fine = np.linspace(max(lo, t_best - 2*eps), min(hi, t_best + 2*eps), 4001)
    vals_f = np.max(A[None, :] + t_fine[:, None] * B[None, :], axis=1)
    t_best = t_fine[np.argmin(vals_f)]
    return t_best

def _coord_sweep_exact(h, n, dx):
    """One full pass of exact single-coordinate optimization (with sum repair)."""
    target = n / 2.0
    improved = False
    for i in range(n):
        # Compute B[m] = coefficient of h[i] in corr[m]
        B = np.zeros(2 * n - 1)
        for m in range(2 * n - 1):
            s = m - (n - 1)
            # h[i] as first factor h[j] with j=i: need 0<=i+s<n
            if 0 <= i + s < n:
                B[m] += (1.0 - h[i + s])
            # h[i] as second factor h[j+s] with j+s=i, j=i-s: need 0<=i-s<n
            if 0 <= i - s < n:
                B[m] -= h[i - s]
        
        h_i = h[i]
        corr_cur = _corr(h)
        A = corr_cur - h_i * B
        
        # Minimize max(A + t*B) over t in [0, 1]
        t_opt = _minimax_1d(A, B, 0.0, 1.0)
        
        if abs(t_opt - h_i) > 1e-10:
            h[i] = t_opt
            # Repair sum
            h = _repair(h, n)
            improved = True
    return h, improved
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _make_s_arr(n):
    return np.arange(-(n-1), n)

def _compute_B(h, i, n, s_arr):
    B = np.zeros(2 * n - 1)
    mask1 = (i + s_arr >= 0) & (i + s_arr < n)
    mask2 = (i - s_arr >= 0) & (i - s_arr < n)
    B[mask1] += (1.0 - h[i + s_arr[mask1]])
    B[mask2] -= h[i - s_arr[mask2]]
    return B

def _minimax_1d(A, B, lo, hi, n_grid=4001):
    if hi - lo < 1e-14:
        return lo
    t_grid = np.linspace(lo, hi, n_grid)
    vals = np.max(A[None, :] + t_grid[:, None] * B[None, :], axis=1)
    t_best = t_grid[np.argmin(vals)]
    eps = (hi - lo) / n_grid
    lo2 = max(lo, t_best - 3 * eps)
    hi2 = min(hi, t_best + 3 * eps)
    if hi2 - lo2 > 1e-14:
        t_fine = np.linspace(lo2, hi2, n_grid)
        vals_f = np.max(A[None, :] + t_fine[:, None] * B[None, :], axis=1)
        t_best = t_fine[np.argmin(vals_f)]
    return t_best

def _coord_pass(h, n, s_arr, rng=None):
    order = np.arange(n)
    if rng is not None:
        order = rng.permutation(n)
    improved = 0
    for i in order:
        B = _compute_B(h, i, n, s_arr)
        corr_cur = _corr(h)
        A = corr_cur - h[i] * B
        t_opt = _minimax_1d(A, B, 0.0, 1.0)
        if abs(t_opt - h[i]) > 1e-10:
            h[i] = t_opt
            h = _repair(h, n)
            improved += 1
    return h, improved

def _pair_pass(h, n, dx, s_arr, rng, n_pairs=300):
    corr = _corr(h)
    improved = False
```

```python
def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    m2 = (i - s_arr >= 0) & (i - s_arr < n)
    B[m1] += (1.0 - h[i + s_arr[m1]])
    B[m2] -= h[i - s_arr[m2]]
    return B

def _minimax_1d(A, B, lo, hi, n_grid=4001):
    if hi - lo < 1e-14:
        return lo
    tg = np.linspace(lo, hi, n_grid)
    vals = np.max(A[None, :] + tg[:, None] * B[None, :], axis=1)

    eps = (hi - lo) / n_grid
```
