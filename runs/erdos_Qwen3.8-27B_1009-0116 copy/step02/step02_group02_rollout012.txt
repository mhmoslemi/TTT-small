            if g[i] < g[j]:
                i, j = j, i
            hi0, hj0 = h[i], h[j]
            for d in dvals:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                    continue
                h[i] = ni
                h[j] = nj
                c = _c5(h, dx)
                if c < bc - 1e-12:
                    bc = c
                    imp = True
                    break
                h[i] = hi0
                h[j] = hj0
            else:
                h[i] = hi0
                h[j] = hj0
        if not imp:
            break
    return h, bc

def _fista_n(h0, n, dx, target, alphas, iters_per):
    """Run FISTA with multiple alphas."""
    bh = h0.copy()
    bc = _c5(bh, dx)
    for a in alphas:
        h, c = _fista(bh, a, target, iters_per, dx)
        if c < bc:
            bc = c
            bh = h
    return bh, bc

def _rem_time(t0, budget):
    return budget - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    tgt0 = n0 / 2.0
    h_init = _proj(np.array(initial_h_values, dtype=float), tgt0)
    bh = h_init.copy()
    bc = _c5(bh, dx0)

    # Phase 1: FISTA on initial with increasing alpha
    alphas_main = [50, 100, 200, 400, 800, 1500, 3000]
    for a in alphas_main:
        if _rem_time(t0, budget_s) < 40:
            break
        h, c = _fista(bh, a, tgt0, 5000, dx0)
        if c < bc:
            bc = c
            bh = h

    # Phase 2: Intensive CD
    if _rem_time(t0, budget_s) > 30:
        h, c = _cd(bh, dx0, n0, 800, rng)
        if c < bc:
            bc = c
            bh = h
        h, c = _grad_cd(bh, dx0, n0, 400, rng)
        if c < bc:
            bc = c
            bh = h

    # Phase 3: Random restarts with FISTA+CD
    sigmas = [0.1, 0.2, 0.35, 0.6, 0.15, 0.4, 0.7, 0.25, 0.5, 0.8,
              0.3, 0.45, 0.12, 0.28, 0.55, 0.4, 0.65, 0.18, 0.38, 0.22]
    for s in sigmas:
        if _rem_time(t0, budget_s) < 50:
            break
        hp = _proj(bh + s * rng.randn(n0), tgt0)
        h, c = _fista(hp, 300, tgt0, 4000, dx0)
        if c < bc:
            bc = c
            bh = h
            if _rem_time(t0, budget_s) > 30:
                h2, c2 = _cd(bh, dx0, n0, 300, rng)
                if c2 < bc:
                    bc = c2
                    bh = h2

    # Phase 4: Fourier structured restarts
    if _rem_time(t0, budget_s) > 80:
        for freq in range(1, 6):
            for amp in [0.15, 0.25, 0.35, 0.5]:
                if _rem_time(t0, budget_s) < 40:
                    break
                xg = np.arange(n0)
                phase = rng.rand() * 2.0 * np.pi
                hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * xg / n0 + phase)
                hs = _proj(hs, tgt0)
                h, c = _fista(hs, 200, tgt0, 3000, dx0)
                if c < bc:
                    bc = c
                    bh = h
                    h2, c2 = _cd(bh, dx0, n0, 200, rng)
                    if c2 < bc:
                        bc = c2
                        bh = h2

    # Phase 5: Try different n values (n=90, n=100)
    if _rem_time(t0, budget_s) > 120:
        for n_alt in [90, 100]:
            if _rem_time(t0, budget_s) < 80:
                break
            dx_a = 2.0 / n_alt
            tgt_a = n_alt / 2.0
            # Interpolate initial to new n
            x_new = np.linspace(0, 2, n_alt)
            x_old = np.linspace(0, 2, n0)
            h_interp = np.interp(x_new, x_old, bh)
            h_alt = _proj(h_interp, tgt_a)
            alphas_alt = [100, 300, 800]
            h_a, c_a = _fista(h_alt, 100, tgt_a, 4000, dx_a)
            if c_a < bc:
                bc = c_a
                bh = _proj(np.interp(x_old, x_new, h_a), tgt0)
            h_a, c_a = _fista(h_a, 300, tgt_a, 4000, dx_a)
            if c_a < bc:
                bc = c_a
                bh = _proj(np.interp(x_old, x_new, h_a), tgt0)
            h_a, c_a = _fista(h_a, 800, tgt_a, 3000, dx_a)
            if c_a < bc:
                bc = c_a
                bh = _proj(np.interp(x_old, x_new, h_a), tgt0)
            if _rem_time(t0, budget_s) > 30:
                h_a, c_a = _cd(h_a, dx_a, n_alt, 300, rng)
                if c_a < bc:
                    bc = c_a
                    bh = _proj(np.interp(x_old, x_new, h_a), tgt0)

    # Phase 6: SLSQP polish
    if _rem_time(t0, budget_s) > 25:
        bnds = [(0.0, 1.0)] * n0
        for a_pol in [3000, 8000]:
            if _rem_time(t0, budget_s) < 20:
                break
            try:
                def _obj(x, a=a_pol):
                    return _sobj(x, a)
                def _con(x):
                    return float(np.sum(x)) - tgt0
                r = minimize(_obj, bh, method='SLSQP', bounds=bnds,
                             constraints=[{'type': 'eq', 'fun': _con}],
                             options={'maxiter': 500, 'ftol': 1e-15})
                hf = _proj(r.x, tgt0)
                cf = _c5(hf, dx0)
                if cf < bc:
                    bc = cf
                    bh = hf
            except Exception:
                pass

    # Phase 7: Fine-tuning restarts
    if _rem_time(t0, budget_s) > 60:
        for s in [0.03, 0.05, 0.07, 0.04, 0.06, 0.08, 0.03, 0.05]:
            if _rem_time(t0, budget_s) < 40:
                break
            hp = _proj(bh + s * rng.randn(n0), tgt0)
            h, c = _fista(hp, 800, tgt0, 4000, dx0)
            if c < bc:
                bc = c
                bh = h
                h2, c2 = _cd(bh, dx0, n0, 200, rng)
                if c2 < bc:
                    bc = c2
                    bh = h2

    # Phase 8: Final intensive CD
    if _rem_time(t0, budget_s) > 20:
        h, c = _cd(bh, dx0, n0, 2000, rng)
        if c < bc:
            bc = c
            bh = h
        h, c = _grad_cd(bh, dx0, n0, 500, rng)
        if c < bc:
            bc = c
            bh = h

    # Final SLSQP
    if _rem_time(t0, budget_s) > 15:
        try:
            bnds = [(0.0, 1.0)] * n0
            r = minimize(_sobj, bh, args=(10000,), method='SLSQP', bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (tgt0,)}],
                         options={'maxiter': 800, 'ftol': 1e-16})
            hf = _proj(r.x, tgt0)
            cf = _c5(hf, dx0)
            if cf < bc:
                bc = cf
                bh = hf
        except Exception:
            pass

    if _rem_time(t0, budget_s) > 10:
        h, c = _cd(bh, dx0, n0, 500, rng)
        if c < bc:
            bc = c
            bh = h

    bh = _proj(bh, tgt0)
    bc = _c5(bh, dx0)
    return (bh, bc, n0)

def _slsqp_con(x, target):
    return float(np.sum(x)) - target
