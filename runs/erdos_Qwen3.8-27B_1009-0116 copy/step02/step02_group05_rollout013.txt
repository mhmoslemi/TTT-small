def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        remaining = diff
        for idx in order:
            if remaining > 0:
                add = min(remaining, 1.0 - h[idx])
                h[idx] += add
                remaining -= add
            else:
                sub = min(-remaining, h[idx])
                h[idx] -= sub
                remaining += sub
            if abs(remaining) < 1e-13:
                break
    return h

def _smooth_grad_obj(h, n, alpha):
    corr = _corr(h)
    mx = np.max(corr)
    s = np.clip(alpha * (corr - mx), -500, 500)
    exp_s = np.exp(s)
    wsum = exp_s.sum()
    weights = exp_s / wsum
    obj = mx + np.log(wsum) / alpha
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-8:
            continue
        m = k - (n - 1)
        w = weights[k]
        if m >= 0:
            grad[:n - m] += w * (1.0 - h[m:n])
            grad[m:] -= w * h[:n - m]
        else:
            grad[-m:] += w * (1.0 - h[:n + m])
            grad[:n + m] -= w * h[-m:]
    return obj, grad, mx

    def rem(margin):
        return time.time() < t0 + budget_s - margin

    # Phase 1: Gradient descent
    if rem(500):
        best_h, best_c5 = _grad_descent(best_h, n, dx, t0 + budget_s - 480)

    # Phase 2: Aggressive local search
    if rem(400):
        best_h, best_c5 = _local_search(best_h, n, dx, rng, t0 + budget_s - 380, 10000)

    # Phase 3: Simulated annealing
    if rem(350):
        best_h, best_c5 = _sim_anneal(best_h, n, dx, rng, t0 + budget_s - 330)

    # Phase 4: Local search after SA
    if rem(280):
        best_h, best_c5 = _local_search(best_h, n, dx, rng, t0 + budget_s - 260, 6000)

    # Phase 5: SLSQP polish at multiple alphas
    if rem(220):
        for alpha in [80, 200, 500, 1200, 2500]:
            if not rem(50):
                break
            hc = _slsqp_run(best_h, n, dx, alpha, 500)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        if rem(100):
            best_h, best_c5 = _local_search(best_h, n, dx, rng, t0 + budget_s - 80, 8000)

    # Phase 6: Binary nudge search
    if rem(120):
        best_h, best_c5 = _binary_nudge_search(best_h, n, dx, rng, t0 + budget_s - 100)

    # Phase 7: Restarts with perturbation
    sigmas = [0.05, 0.1, 0.15, 0.08, 0.12, 0.06, 0.2, 0.04]
    it = 0
    while rem(80):
        sigma = sigmas[it % len(sigmas)]
        hp = _project(best_h + sigma * rng.randn(n), n)
        t_end = t0 + budget_s - 30
        hg, cg = _grad_descent(hp, n, dx, t_end)
        if cg < best_c5:
            best_c5, best_h = cg, hg
        elif cg < best_c5 + 0.003 and rem(40):
            hl, cl = _local_search(hg, n, dx, rng, t0 + budget_s - 20, 3000)
            if cl < best_c5:
                best_c5, best_h = cl, hl
        it += 1

    # Phase 8: Final fine local search
    if rem(8):
        best_h, best_c5 = _local_search(best_h, n, dx, rng, t0 + budget_s - 2, 5000)

    best_h = _project(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
