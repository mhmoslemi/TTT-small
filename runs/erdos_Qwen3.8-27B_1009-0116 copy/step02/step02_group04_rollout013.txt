def _corr_grad(h):
    """Return correlation array and its Jacobian (d corr[j] / dh[i] for all i,j)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Jacobian shape: (2n-1, n)
    # corr[j] = sum_valid h[n + lag_j] * (1 - h[n])
    # d corr[j] / dh[m] = (1-h[m-lag_j])*[valid] - h[m+lag_j]*[valid]
    J = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        lag = j - (n-1)
        if lag >= 0:
            # n ranges 0 to n-1-lag
            # d/dh[m]: m = n+lag (first term) or m = n (second term)
            # For m in [lag, n-1]: (1-h[m-lag])
            # For m in [0, n-1-lag]: -h[m+lag]
            J[j, lag:] += 1.0 - h[:n-lag]

        else:

            # m = n+lag = n-L: first term (1-h[m-lag]) = (1-h[m+L]) for m in [0, n-1-L]

            L = -lag
            J[j, :n-L] += 1.0 - h[L:]
            J[j, L:] -= h[:n-L]
    return corr, J

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _smooth_obj(h, alpha):
    corr = _corr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * 0  # just the max part, dx applied outside

def _project(h, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def _corr_jacobian(h):
    n = len(h)
    corr = _corr(h)
    J = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        lag = j - (n-1)
        if lag >= 0:
            if n - lag > 0:
                J[j, lag:] += 1.0 - h[:n-lag]
                J[j, :n-lag] -= h[lag:]
        else:
            L = -lag
            if n - L > 0:
                J[j, :n-L] += 1.0 - h[L:]
                J[j, L:] -= h[:n-L]
    return corr, J

def _smooth_grad(h, alpha):
    corr, J = _corr_jacobian(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    return J.T @ w, corr, w

def _pgd_step(h, lr, alpha, n):
    grad, corr, w = _smooth_grad(h, alpha)
    h_new = _project(h - lr * grad, n)
    return h_new, grad

def _slsqp(h, n, dx, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = _corr(x)
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def obj_grad(x):
        grad, _, _ = _smooth_grad(x, alpha)
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

            c5 = _c5(hc, dx)
            if c5 < np.inf:

    except Exception:
        pass
    return h, _c5(h, dx)

                    break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, n):
    t = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def _corr_jac(h):
    n = len(h)
    c = _corr(h)
    J = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        lag = j - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                J[j, lag:] += 1.0 - h[:m]
                J[j, :m] -= h[lag:]
        else:
            L = -lag
            m = n - L
            if m > 0:
                J[j, :m] += 1.0 - h[L:]
                J[j, L:] -= h[:m]
    return c, J

def _sgrad(h, alpha):
    c, J = _corr_jac(h)
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    return J.T @ w, c, w

def _smooth_f(h, alpha):
    c = _corr(h)
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500, 500)
    return mx + np.log(s.sum()) / alpha + np.log(np.exp(0))  # just mx + logsumexp/alpha

def _slsqp(h, n, dx, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    tgt = n / 2.0
    bnd = [(0.0, 1.0)] * n

    def f(x):
        c = _corr(x)
        mx = c.max()
        s = np.clip(alpha * (c - mx), -500, 500)
        return mx + np.log(s.sum()) / alpha

    def g(x):
        gr, _, _ = _sgrad(x, alpha)
        return gr

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(f, h, jac=g, method='SLSQP', bounds=bnd, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        if r.success or (hasattr(r, 'x') and r.x is not None):
            hc = _project(r.x, n)
            c5 = _c5(hc, dx)
            if np.isfinite(c5):
                return hc, c5
    except Exception:
        pass
    return h, _c5(h, dx)

def _pgd(h, n, dx, deadline):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    alphas = [30, 80, 200, 500, 1000, 2000]
    for a in alphas:
        d = 2.0 / (n)
        lr = 0.1 * d / (1 + 0.01 * a)
        mom = np.zeros(n)
        beta = 0.85
        for it in range(2000):
            if time.time() >= deadline:
                return bh, bc5
            gr, _, _ = _sgrad(bh, a)
            mom = beta * mom + gr
            h2 = _project(bh - lr * mom, n)
            c52 = _c5(h2, dx)
            if c52 < bc5:
                bc5 = c52
                bh = h2
                lr = min(lr * 1.01, 0.5 * d)
            else:
                lr *= 0.8
                mom *= 0.5
            if lr < 1e-8:
                break
        # SLSQP polish after each alpha
        if time.time() < deadline - 20:
            bh, bc5 = _slsqp(bh, n, dx, a, 200, min(time.time() + 20, deadline))
    return bh, bc5

def _local(h, n, dx, rng, deadline):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for sc in scales:

    y = np.linspace(0, 2, n_new, endpoint=False)
    return _project(np.interp(y, x, h), n_new)
