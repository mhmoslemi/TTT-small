        else:
            r = -m
            for j in range(n):
                g = 0.0
                if j <= n - 1 - r:
                    g += 1.0 - h[j + r]
                if j >= r:
                    g -= h[j - r]
                grad[j] += w * g
    grad *= dx
    return obj, grad

def _proj_grad_descent(h, target_sum, n, dx, deadline, alpha_schedule):
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)
    for alpha in alpha_schedule:
        if time.time() >= deadline:
            break
        step = 0.5 / (alpha + 1.0)
        min_step = 1e-8
        for _it in range(500):
            if time.time() >= deadline:
                break
            obj, grad = _softmax_obj_grad(best_h, alpha, dx)
            # Project gradient onto nullspace of constraint (subtract mean to ensure divergence-free)
            grad_pc = grad - np.mean(grad)
            h_new = best_h - step * grad_pc
            h_new = _project_feasible(h_new, target_sum, n)
            new_c5 = _compute_c5(h_new, dx)
            new_obj, _ = _softmax_obj_grad(h_new, alpha, dx)
            if new_c5 <= best_c5 + 1e-15 or new_obj < obj:
                best_h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                step = min(step * 1.2, 2.0)
            else:
                step *= 0.6
                if step < min_step:
                    break
    return best_h, best_c5

def _focused_local(h, n, dx, rng, deadline, target_sum):
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)
    scales = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.00015]
    for scale in scales:
        stall = 0
        for _pass in range(300):
            if time.time() >= deadline:
                break
            improved = False
            for _ in range(1500):
                if time.time() >= deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                ni, nj = best_h[i] + d, best_h[j] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                best_h[i], best_h[j] = ni, nj
                c5 = _compute_c5(best_h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    best_h[i], best_h[j] = ni - d, nj + d
            if improved:
                stall = 0
            else:
                stall += 1
                if stall >= 3:
                    break
    return best_h, best_c5

def _coordinate_descent(h, n, dx, target_sum, deadline, max_rounds=50):
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    rng = np.random.RandomState(12345)
    for step in step_sizes:
        if time.time() >= deadline:
            break
        for _rnd in range(max_rounds):
            if time.time() >= deadline:
                break
            order = rng.permutation(n)
            improved = False
            for j in order:
                if time.time() >= deadline:
                    break
                # Try moving h[j] up and down
                for delta in [step, -step, step*0.5, -step*0.5]:
                    if best_h[j] + delta < -1e-12 or best_h[j] + delta > 1.0 + 1e-12:
                        continue
                    trial = best_h.copy()
                    trial[j] += delta
                    # Redistribute to maintain sum
                    trial = _project_feasible(trial, target_sum, n)
                    c5 = _compute_c5(trial, dx)
                    if c5 < best_c5 - 1e-13:
                        best_h = trial
                        best_c5 = c5
                        improved = True
                        break
            if not improved:
                break
    return best_h, best_c5

def _slsqp_polish(h, n, dx, target_sum, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _compute_c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    eq_con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    def _obj(x, a=alpha):
        o, _ = _softmax_obj_grad(x, a, dx)
        return o
    def _obj_grad(x, a=alpha):
        _, g = _softmax_obj_grad(x, a, dx)
        return g
    try:
        from scipy.optimize import minimize
        res = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=[eq_con],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _project_feasible(res.x, target_sum, n)
        c5 = _compute_c5(hc, dx)
        return hc, c5
    except Exception:
        return h, _compute_c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target_sum = n / 2.0

    def remaining():
        return budget_s - (time.time() - t_start)

    h0 = _project_feasible(np.array(initial_h_values, dtype=float).copy(), target_sum, n)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    # Phase 1: Projected gradient with annealing (coarse to fine) - ~300s
    t_end = t_start + 300
    best_h, best_c5 = _proj_grad_descent(
        best_h, target_sum, n, dx, t_end,
        alpha_schedule=[10, 30, 80, 200, 500, 1200, 2500, 5000]
    )

    # Phase 2: SLSQP with gradient (300s)
    t_end = t_start + 600
    for alpha in [200, 500, 1500, 4000, 10000]:
        if time.time() > t_end or remaining() < 150:
            break
        best_h, best_c5 = _slsqp_polish(best_h, n, dx, target_sum, alpha, 600, t_end)

    # Phase 3: Diversified restarts with projected gradient (200s)
    t_end = t_start + 800
    for restart in range(6):
        if time.time() > t_end or remaining() < 150:
            break
        h_r = _project_feasible(best_h + 0.06 * rng.randn(n), target_sum, n)
        h_r, c5_r = _proj_grad_descent(
            h_r, target_sum, n, dx, t_end,
            alpha_schedule=[100, 400, 1000, 3000]
        )
        if c5_r < best_c5:
            best_h = h_r
            best_c5 = c5_r

    # Phase 4: Coordinate descent + focused local search (100s)
    if remaining() > 80:
        t_end = min(t_start + 950, time.time() + 50)
        best_h, best_c5 = _coordinate_descent(best_h, n, dx, target_sum, t_end, 30)

    # Phase 5: Intensive pair local search (60s)
    if remaining() > 50:
        t_end = min(time.time() + remaining() - 20, t_start + budget_s - 20)
        best_h, best_c5 = _focused_local(best_h, n, dx, rng, t_end, target_sum)

    # Phase 6: Final SLSQP + tiny local search
    if remaining() > 30:
        t_end = time.time() + min(25, remaining() - 10)
        best_h, best_c5 = _slsqp_polish(best_h, n, dx, target_sum, 8000, 800, t_end)
    if remaining() > 15:
        t_end = time.time() + min(10, remaining() - 5)
        best_h, best_c5 = _focused_local(best_h, n, dx, rng, t_end, target_sum)

    best_h = _project_feasible(best_h, target_sum, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
