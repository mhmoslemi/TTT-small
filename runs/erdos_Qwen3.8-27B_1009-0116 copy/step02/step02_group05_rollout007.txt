    grad = np.zeros(n)
    for m in range(2 * n - 1):

        if wm < 1e-8:

        if k >= 0:

            for j in range(k, n):

            for j in range(l, n):
                grad[j] -= wm * h[j - l]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _smact(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    val = m + np.log(e.sum()) / alpha
    return w, val

def _grad_corr_w(h, n, w):
    grad = np.zeros(n)
    nm1 = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-8:
            continue
        k = m - nm1
        if k >= 0:
            for j in range(k, n):
                grad[j] += wm * (1.0 - h[j - k])
            for j in range(n - k):
                grad[j] -= wm * h[j + k]
        else:
            l = -k
            for j in range(n - l):
                grad[j] += wm * (1.0 - h[j + l])
            for j in range(l, n):
                grad[j] -= wm * h[j - l]
    return grad

def _pgd(h, n, dx, alpha, n_iter, deadline, lr0=0.02):
    best = h.copy()
    best_val = _c5(best, dx)
    lr = lr0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = _corr(best)
        wm, _ = _smact(c, alpha)
        g = _grad_corr_w(best, n, wm)
        new_h = _repair(best - lr * g, n)
        new_val = _c5(new_h, dx)
        if new_val < best_val - 1e-14:
            best_val = new_val
            best = new_h
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.85
        if lr < 1e-8:
            break
    return best, best_val

def _local_search(h, n, dx, rng, deadline, max_rounds=8000):
    best = h.copy()
    best_c5 = _c5(best, dx)
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ntrials = 25 + 20 * min(stall // 8, 6)
        for _ in range(ntrials):
            mt = rng.randint(3)
            if mt == 0:
                i, j = int(rng.randint(n)), int(rng.randint(n))
                if i == j:
                    continue
                d = rng.choice([0.005, 0.01, 0.02, 0.04, 0.06, 0.09]) * rng.choice([1, -1])
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                    continue
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-14:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = ni - d, nj + d
            elif mt == 1:
                idx = rng.choice(n, 3, replace=False)
                d1, d2 = rng.uniform(-0.06, 0.06, 2)
                d3 = -(d1 + d2)
                nv = best[idx] + np.array([d1, d2, d3])
                if nv.min() < -1e-9 or nv.max() > 1 + 1e-9:
                    continue
                old = best[idx].copy()
                best[idx] = nv
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-14:
                    best_c5 = nc
                    improved = True
                else:
                    best[idx] = old
            else:
                i, j = int(rng.randint(n)), int(rng.randint(n))
                if i == j:
                    continue
                d = rng.uniform(-0.12, 0.12)
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                    continue
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-14:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = ni - d, nj + d
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return best, best_c5

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    target = n / 2.0
    def obj(x):
        _, v = _smact(_corr(x), alpha)
        return v * dx
    def cfun(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'args': (target,)}]
    try:
        res = minimize(obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _binary_init(n, rng):
    h = np.zeros(n)
    pos = rng.choice(n, n // 2, replace=False)
    h[pos] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _c5(best_h, dx)

    def el():
        return time.time() - t0

    # Phase 1: PGD with multi-alpha on parent
    d1 = t0 + budget_s * 0.2
    for alpha in [15, 40, 80, 160, 350, 700]:
        if time.time() > d1:
            break
        hn, vn = _pgd(best_h, n, dx, alpha, 800, d1, lr0=0.025)
        if vn < best_c5:
            best_c5, best_h = vn, hn

    # Phase 2: SLSQP polish
    d2 = t0 + budget_s * 0.38
    for alpha in [150, 400, 800]:
        if time.time() > d2:
            break
        hn = _slsqp(best_h, n, dx, alpha, 250, 1e-13)
        vn = _c5(hn, dx)
        if vn < best_c5:
            best_c5, best_h = vn, hn.copy()

    # Phase 3: Local search
    d3 = t0 + budget_s * 0.62
    best_h, best_c5 = _local_search(best_h, n, dx, rng, d3)

    # Phase 4: Multi-start PGD + LS
    sigmas = [0.08, 0.15, 0.25, 0.4, 0.12, 0.3, 0.5, 0.2, 0.06, 0.35]
    it = 0
    d4 = t0 + budget_s * 0.88
    while el() < budget_s * 0.85:
        sigma = sigmas[it % len(sigmas)]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        hn, vn = _pgd(hp, n, dx, 250, 400, d4, lr0=0.015)
        if vn < best_c5:
            best_c5, best_h = vn, hn
        elif vn < best_c5 + 0.004 and el() < budget_s * 0.82:
            hl, vl = _local_search(hn, n, dx, rng, t0 + budget_s * 0.82)
            if vl < best_c5:
                best_c5, best_h = vl, hl
        # Also try from binary init occasionally
        if it % 5 == 4:
            hb = _repair(_binary_init(n, rng), n)
            hn2, vn2 = _pgd(hb, n, dx, 80, 300, d4, lr0=0.03)
            if vn2 < best_c5:
                best_c5, best_h = vn2, hn2
        it += 1

    # Phase 5: Final PGD + SLSQP + LS
    if el() < budget_s - 20:
        hn, vn = _pgd(best_h, n, dx, 600, 500, t0 + budget_s - 15, lr0=0.01)
        if vn < best_c5:
            best_c5, best_h = vn, hn
    if el() < budget_s - 8:
        hn = _slsqp(best_h, n, dx, 500, 150, 1e-14)
        vn = _c5(hn, dx)
        if vn < best_c5:
            best_c5, best_h = vn, hn.copy()
    if el() < budget_s - 2:
        best_h, best_c5 = _local_search(best_h, n, dx, rng, t0 + budget_s - 0.5)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
