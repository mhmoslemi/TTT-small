    lo, hi = -10.0, 10.0

        mid = (lo + hi) * 0.5
        pr = np.clip(h_new - mid, 0.0, 1.0)
        if pr.sum() > target:
            lo = mid
        else:
            hi = mid
    pr = np.clip(h_new - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - pr.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - pr) if diff > 0 else np.argsort(pr)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - pr[idx])
                pr[idx] += a
                diff -= a
            else:
                a = min(-diff, pr[idx])
                pr[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return pr

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _gvec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    i1 = p - k
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = p + k
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _sweep(h, n, dx, corr, nc, nf, fw):
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hi, hj = h[i], h[j]
            dlo = max(-hi, hj - 1.0)
            dhi = min(1.0 - hi, hj)
            if dlo > dhi - 1e-13 or dhi - dlo < 1e-13:
                continue
            gd = _gvec(h, i, n) - _gvec(h, j, n)
            ds = np.linspace(dlo, dhi, nc)
            nc1 = corr[None, :] + ds[:, None] * gd[None, :]
            c5s = np.max(nc1, axis=1) * dx
            bi = np.argmin(c5s)
            cur = np.max(corr) * dx
            if c5s[bi] < cur - 1e-14:
                db = ds[bi]
                if nf > 2:
                    w = fw if fw > 0 else (dhi - dlo) / nc
                    lo = max(dlo, db - w)
                    hh = min(dhi, db + w)
                    if hh > lo:
                        ds2 = np.linspace(lo, hh, nf)
                        nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                        c52 = np.max(nc2, axis=1) * dx
                        db = ds2[np.argmin(c52)]
                h[i] = hi + db
                h[j] = hj - db
                corr = corr + db * gd
                improved = True
    return h, corr, improved

def _slsqp(h, n, dx, target, bounds, alpha, mx):
    def obj(x):
        return _smooth_max_val(_corr(x), alpha)
    def jac(x):
        _, g = _smooth_max_grad(x, alpha)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': mx, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def el():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP with analytical gradient, increasing alpha
    for alpha in [10, 25, 50, 100, 200, 400, 800, 1600, 3200, 6400]:
        if el() > budget_s - 550:
            break
        hc = _slsqp(best_h, n, dx, target, bounds, alpha, 600)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = hc

    # Phase 2: Subgradient + projection (exploration)
    if el() < budget_s - 450:
        h = best_h.copy()
        eta = 0.8
        for it in range(300):
            if el() > budget_s - 450:
                break
            h = _subgrad_proj(h, eta)
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
            eta *= 0.997
        # Polish after subgradient
        hc = _slsqp(best_h, n, dx, target, bounds, 400, 400)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = hc

    # Phase 3: Coarse coordinate sweeps
    h = best_h.copy()
    corr = _corr(h)
    for _ in range(50):
        if el() > budget_s - 350:
            break
        h, corr, imp = _sweep(h, n, dx, corr, 15, 31, 0.03)
        corr = _corr(h)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        if not imp:
            break

    # Phase 4: Medium sweeps
    h = best_h.copy()
    corr = _corr(h)
    for _ in range(80):
        if el() > budget_s - 250:
            break
        h, corr, imp = _sweep(h, n, dx, corr, 21, 51, 0.015)
        corr = _corr(h)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        if not imp:
            break

    # Phase 5: Fine sweeps
    h = best_h.copy()
    corr = _corr(h)
    for _ in range(100):
        if el() > budget_s - 150:
            break
        h, corr, imp = _sweep(h, n, dx, corr, 31, 80, 0.008)
        corr = _corr(h)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        if not imp:
            break

    # Phase 6: Structured perturbations + SLSQP
    for sr in range(5):
        if el() > budget_s - 60:
            break
        x = np.linspace(0, 2, n)
        k1 = sr + 2
        k2 = sr + 3
        pert = 0.04 * np.sin(2 * np.pi * k1 * x / 2) + 0.03 * np.cos(2 * np.pi * k2 * x / 2)
        h_p = _repair(best_h + pert, n)
        hc = _slsqp(h_p, n, dx, target, bounds, 300, 400)
        cr = _corr(hc)
        hc, cr, _ = _sweep(hc, n, dx, cr, 21, 51, 0.01)
        c5 = float(np.max(cr) * dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = hc

    # Phase 7: Utra-fine final sweeps
    if el() < budget_s - 20:
        h = best_h.copy()
        corr = _corr(h)
        for _ in range(120):
            if el() > budget_s - 10:
                break
            h, corr, imp = _sweep(h, n, dx, corr, 40, 120, 0.004)
            corr = _corr(h)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
            if not imp:
                break

    # Phase 8: Final SLSQP polish
    if el() < budget_s - 5:
        hc = _slsqp(best_h, n, dx, target, bounds, 8000, 2000)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = hc
        # One more ultra-fine sweep
        if el() < budget_s - 2:
            cf = _corr(best_h)
            best_h, cf, _ = _sweep(best_h, n, dx, cf, 50, 150, 0.002)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
