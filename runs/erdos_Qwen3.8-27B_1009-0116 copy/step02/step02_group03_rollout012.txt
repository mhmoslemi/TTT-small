import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _soft(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > 1e-15:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for i in idx:
        if abs(diff) < 1e-14:
            break
        d = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += d
        diff -= d
    return h

def _slsqp(h, n, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _soft(_corr(x), alpha)
    def con(x):
        return x.sum() - target
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs(h, n, alpha, pen, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _soft(_corr(x), alpha) + pen * (x.sum() - target) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h

def _pair_cd(h, n, dx, n_passes, t_end, rng, step_scale=1.0):
    h = h.copy()
    best = _c5(h, dx)
    steps = np.array([0.002, 0.005, 0.01, 0.02, 0.04, -0.002, -0.005, -0.01, -0.02, -0.04]) * step_scale
    for p in range(n_passes):
        if time.time() > t_end:
            break
        imp = False
        ss = max(0.001, 0.02 / (1 + p * 0.05))
        steps = np.array([0.002, 0.005, 0.01, 0.02, 0.04, -0.002, -0.005, -0.01, -0.02, -0.04]) * max(0.3, min(1.0, ss / 0.02))
        # Systematic sweep (every 3rd pass to save time)
        if p % 3 == 0 and n <= 100:
            for i in range(n):
                if time.time() > t_end:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    if time.time() > t_end:
                        break
                    hj = h[j]
                    for d in steps:
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best - 1e-14:
                            best = c
                            imp = True
                        else:
                            h[i], h[j] = hi, hj
        # Random pairs
        n_rand = 300 if n <= 100 else 150
        for _ in range(n_rand):
            if time.time() > t_end:
                break
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            hi, hj = h[i], h[j]

                ni, nj = hi + d, hj - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best - 1e-14:
                    best = c
                    imp = True
                else:
                    h[i], h[j] = hi, hj
        if not imp and p > 40:
            break

def _interp(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    return _repair(np.interp(x_new, x_old, h_old, period=2.0), n_new)

def _make_start(n, seed, kind):
    rng = np.random.RandomState(seed)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.1 * rng.randn(n)
    elif kind == 2:
        a, b = n // 4, 3 * n // 4
        h = np.where((np.arange(n) < a) | (np.arange(n) >= b), 1.0, 0.0) + 0.1 * rng.randn(n)
    elif kind == 3:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * np.arange(n) / n + rng.uniform(0, 6.28)) + 0.05 * rng.randn(n)
    elif kind == 4:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 5 * np.arange(n) / n) * np.cos(2 * np.pi * 2 * np.arange(n) / n) + 0.05 * rng.randn(n)
    else:
        h = rng.uniform(0, 1, n)
    return _repair(h, n)
