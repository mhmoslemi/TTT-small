    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _jacobian(h, n):
    """Compute Jacobian of corr_full w.r.t. h."""
    n_lags = 2 * n - 1
    eps = 1e-8
    J = np.zeros((n_lags, n))
    for i in range(n):
        hp = h.copy(); hp[i] += eps
        hm = h.copy(); hm[i] -= eps
        J[:, i] = (_corr_full(hp) - _corr_full(hm)) / (2 * eps)
    return J

def _lp_refine(h, n, dx, deadline, max_rounds=500):
    """LP outer approximation with line search."""
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    target = n / 2.0
    
    for rnd in range(max_rounds):
        if time.time() >= deadline:
            break
        corr = _corr_full(best_h)
        J = _jacobian(best_h, n)
        n_lags = len(corr)
        
        # LP: min t s.t. J@h_new - t <= corr - J@h_best, sum(h_new)=target, 0<=h_new<=1
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.hstack([J, -np.ones((n_lags, 1))])
        b_ub = corr - J @ best_h
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        
        h_lp = res.x[:n]
        
        # Line search
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02]:
            h_try = (1 - alpha) * best_h + alpha * h_lp
            h_try = _repair(h_try, n)
            c5_try = _compute_c5(h_try, dx)
            if c5_try < best_c5 - 1e-12:
                best_h = h_try
                best_c5 = c5_try
                improved = True
                break
        
        if not improved:
            # Try a small random perturbation to escape
            h_try = best_h + 0.02 * np.random.rand(n) * 2 - 0.02
            h_try = _repair(h_try, n)
            c5_try = _compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
            else:
                # Check if we're at a true local optimum
                break
    
    return best_h, best_c5

def _smooth_obj(x, alpha):

    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

    if time.time() >= deadline:
        return h, _compute_c5(h, dx)
    target = n / 2.0

        hc = _repair(res.x, n)
        c5 = _compute_c5(hc, dx)
        return hc, c5
    except Exception:
        return h, _compute_c5(h, dx)

                    h[j_best] += d  # revert
            else:
                # Reset i to original if no improvement
                h[i] = orig
        
        if not improved and n_pass > 5:
            break

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
