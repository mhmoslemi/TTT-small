def _c5_val(h, dx):

def _repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            s = max(h.sum(), 1e-15)
            h *= target / s
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0.0:

def _grad_lag(h, n, L):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h

        g[:n - L] += (1.0 - h[L:])
        g[L:] -= h[:n - L]
    else:
        L2 = -L
        g[L2:] += (1.0 - h[:n - L2])
        g[:n - L2] -= h[L2:]
    return g
