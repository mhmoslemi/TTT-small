```python
def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:

                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

    For each h[i], the derivative of corr[k] w.r.t. h[i] is:
    corr[k] = sum_j h[j]*(1-h[j+k"]) at various lags

    active = (corr >= max_val - 1e-8)
    # For each i, compute d(corr[k])/dh[i] for active k

    #   If j + lag = i, i.e., j = i - lag: contributes -h[i-lag] for valid i-lag
    grad = np.zeros(n)
```

```python
    weights /= np.sum(weights)
    
    # Gradient of corr w.r.t. h
    # corr[k] = sum_j h[j] * (1 - h[j + lag]) where lag = k - (n-1)
    # d corr[k] / dh[i] = (1-h[i+lag]) if i+lag valid + (-h[i-lag]) if i-lag valid
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        w = weights[k_idx]
        # For each i: contribution from h[i] term
        # (1-h[i+lag]) weighted
        for i in range(n):
            j2 = i + lag
            if 0 <= j2 < n:
                grad[i] += w * (1.0 - h[j2])
            j1 = i - lag
            if 0 <= j1 < n:
                grad[i] += w * (-h[j1])
    
    grad *= dx
    # Gradient descent step
    h_new = h - lr * grad
    h_new = _repair(h_new, n)
    return h_new
```

For a given lag `lag`, the gradient contribution to each i is:
- (1 - h[i+lag]) if 0 <= i+lag < n
- -h[i-lag] if 0 <= i-lag < n

This can be computed with numpy slicing:
```python
if lag >= 0:
    grad[:n-lag] += w * (1 - h[lag:])  # (1-h[i+lag]) part
    grad[lag:] += w * (-h[:n-lag])     # -h[i-lag] part
else:
    grad[-lag:] += w * (1 - h[:n+lag])  # (1-h[i+lag]) part  
    grad[:n+lag] += w * (-h[-lag:])     # -h[i-lag] part
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_c5(h, n, dx, alpha):
    corr = _corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w[k] * (1.0 - h[lag:])
            grad[lag:] += w[k] * (-h[:n - lag])
        else:
            grad[-lag:] += w[k] * (1.0 - h[:n + lag])
            grad[:n + lag] += w[k] * (-h[-lag:])
    return grad * dx

def _obj_slsqp(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx

def _sum_fn(x, target):
    return np.sum(x) - target

def _slsqp(h0, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _sum_fn, 'args': (target,)}]
    try:
        res = minimize(_obj_slsqp, h0, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h0.copy()

def _gradient_descent(h, n, dx, rng, deadline, alpha=100.0):
    best = h.copy()
    best_c = _c5(best, dx)
    lr = 0.002
    for it in range(50000):
        if time.time() > deadline:
            break
        g = _grad_c5(best, n, dx, alpha)
        h_new = best - lr * g
        h_new = _repair(h_new, n)
        c_new = _c5(h_new, dx)
        if c_new < best_c - 1e-14:
            best = h_new
            best_c = c_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-5:
                if rng.random() < 0.1:
                    h_new = _repair(best + rng.randn(n) * 0.02, n)
                    c_new = _c5(h_new, dx)
                    if c_new < best_c:
                        best, best_c = h_new, c_new
                        lr = 0.002
                break
    return best, best_c

def _pair_local_search(h, n, dx, rng, deadline, max_rounds=3000):
    best = h.copy()
    best_c = _c5(best, dx)
    steps = np.array([0.003, 0.005, 0.008, 0.012, 0.02, 0.035,
                      -0.003, -0.005, -0.008, -0.012, -0.02, -0.035])
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ntrials = 40 + 30 * min(stall // 5, 4)
        for _ in range(ntrials):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for d in steps:
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12:
                    continue
                if nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                oi, oj = best[i], best[j]
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c - 1e-13:
                    best_c = nc
                    improved = True
                    break
                else:
                    best[i], best[j] = oi, oj
            if improved:
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 25:
                break
    return best, best_c

def _sa_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c = _c5(best, dx)
    curr = h.copy()
    curr_c = best_c
    temp = 0.005
    while time.time() < deadline and temp > 1e-5:
        temp *= 0.997
        for _ in range(8):
            if time.time() > deadline:
                break
            h_new = curr.copy()
            mt = int(rng.randint(3))
            if mt == 0:
                i = int(rng.randint(n))
                h_new[i] = np.clip(h_new[i] + rng.uniform(-0.08, 0.08), 0, 1)
            elif mt == 1:
                i, j = rng.choice(n, 2, replace=False)
                d = rng.uniform(-0.06, 0.06)
                h_new[i] = np.clip(h_new[i] + d, 0, 1)
                h_new[j] = np.clip(h_new[j] - d, 0, 1)
            else:
                i = int(rng.randint(n - 2))
                d = rng.uniform(-0.04, 0.04) * rng.randn(3)
                h_new[i:i+3] = np.clip(h_new[i:i+3] + d, 0, 1)
            h_new = _repair(h_new, n)
            c_new = _c5(h_new, dx)
            acc = c_new < curr_c or (temp > 0 and rng.random() < np.exp(-(c_new - curr_c) / temp))
            if acc:
                curr, curr_c = h_new, c_new
                if c_new < best_c:
                    best, best_c = h_new.copy(), c_new
    return best, best_c

def _make_starts(n, seed):
    rng = np.random.RandomState(seed)
    starts = []
    h1 = np.zeros(n)
    h1[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append(h1)
    t = np.linspace(0, 2.0, n, endpoint=False)
    starts.append(_repair(0.5 + 0.3 * np.sin(4 * np.pi * t + 0.7), n))
    h3 = np.zeros(n)
    for b in range(0, n, 8):
        h3[b:b+4] = 1.0
    starts.append(_repair(h3, n))
    h4 = np.zeros(n)
    tm = 0
    for i in range(n):
        tm ^= (tm >> 1)
        h4[i] = float(tm & 1)
    starts.append(_repair(h4, n))
    return [_repair(s, n) for s in starts]

    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
```
