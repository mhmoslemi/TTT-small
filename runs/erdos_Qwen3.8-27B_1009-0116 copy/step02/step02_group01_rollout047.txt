    """Minimize max_k (base[k] + t*slope[k]) over t in [t_low, t_high]."""

    # Check if min is at boundary
    f_low = np.max(base + t_low * slope)
    f_high = np.max(base + t_high * slope)

    # At optimum, the subgradient contains 0
    # i.e., there exist k1, k2 with base[k1]+t*slope[k1] = base[k2]+t*slope[k2] = max

import numpy as np
from scipy.optimize import minimize
import time

def _c5_val(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

def _solve_1d(base, slope):
    """Minimize max_k(base[k] + t*slope[k]) for t in [0,1]."""
    lo, hi = 0.0, 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(base + m1 * slope)
        f2 = np.max(base + m2 * slope)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, 2.0 / n)
    return (best_h, best_c5, n)
