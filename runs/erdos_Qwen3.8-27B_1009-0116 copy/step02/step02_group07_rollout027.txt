def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            a = min(abs(diff), 1.0 if diff > 0 else h[idx]) if diff > 0 else min(-diff, h[idx])
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _sum_constraint(x, target):
    return x.sum() - target

def _smooth_obj(x, alpha, target):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, bounds, target, alpha, maxiter, ftol):
    try:
        cons = ({'type': 'eq', 'fun': _sum_constraint, 'args': (target,)})
        res = minimize(_smooth_obj, h, args=(alpha, target), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _cd_pass(h, n, dx, n_grid, rng, refine=True):
    hi_c5 = _c5(h, dx)
    n_pairs = n * (n - 1) // 2
    rows = np.repeat(np.arange(n), n - np.arange(n) - 1)
    cols = []
    for i in range(n):
        cols.extend(range(i + 1, n))
    pairs = np.array(list(zip(rows, cols)))
    perm = rng.permutation(n_pairs)
    improved = False
    for pi in range(n_pairs):
        i, j = int(pairs[perm[pi], 0]), int(pairs[perm[pi], 1])
        hsum = h[i] + h[j]
        lo = max(0.0, hsum - 1.0)
        hi = min(1.0, hsum)
        span = hi - lo
        if span < 1e-10:
            continue
        orig_i, orig_j = h[i], h[j]
        best_t, best_c = orig_i, hi_c5
        for t in np.linspace(lo + span * 1e-7, hi - span * 1e-7, n_grid):
            h[i], h[j] = t, hsum - t
            c = _c5(h, dx)
            if c < best_c:
                best_c, best_t = c, t
        if refine and best_t != orig_i and span > 1e-8:
            eps = span / max(n_grid - 1, 1)
            lo2, hi2 = max(lo, best_t - eps), min(hi, best_t + eps)
            if hi2 - lo2 > 1e-10:
                for t in np.linspace(lo2, hi2, 9):
                    h[i], h[j] = t, hsum - t
                    c = _c5(h, dx)
                    if c < best_c:
                        best_c, best_t = c, t
        if best_c < hi_c5 - 1e-14:
            h[i], h[j] = best_t, hsum - best_t
            hi_c5 = best_c
            improved = True
        else:
            h[i], h[j] = orig_i, orig_j
    return h, hi_c5, improved

def _block3_pass(h, n, dx, n_grid, n_triples, rng):
    hi_c5 = _c5(h, dx)
    improved = False
    for _ in range(n_triples):
        idx = rng.choice(n, 3, replace=False)
        i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
        s = h[i] + h[j] + h[k]
        if s < 1e-10 or s > 3.0 - 1e-10:
            continue
        orig = (h[i], h[j], h[k])
        best, best_c = orig, hi_c5
        t1lo, t1hi = max(0.0, s - 2.0), min(1.0, s)
        if t1hi - t1lo < 1e-10:
            continue
        for t1 in np.linspace(t1lo + 1e-8, t1hi - 1e-8, n_grid):
            t2lo = max(0.0, s - t1 - 1.0)
            t2hi = min(1.0, s - t1)
            if t2hi - t2lo < 1e-10:
                continue
            for t2 in np.linspace(t2lo + 1e-8, t2hi - 1e-8, n_grid):
                t3 = s - t1 - t2
                if t3 < -1e-9 or t3 > 1.0 + 1e-9:
                    continue
                h[i], h[j], h[k] = t1, t2, t3
                c = _c5(h, dx)
                if c < best_c:
                    best_c, best = c, (t1, t2, t3)
        if best_c < hi_c5 - 1e-14:
            h[i], h[j], h[k] = best
            hi_c5 = best_c
            improved = True
        else:
            h[i], h[j], h[k] = orig
    return h, hi_c5, improved

def _binary_round(h, n, rng):
    hb = (h > 0.5).astype(float)
    diff = n / 2.0 - hb.sum()
    if diff > 0:
        order = np.argsort(h)
        for idx in order[:int(diff)]:
            hb[idx] = 1.0
    elif diff < 0:
        order = np.argsort(h)
        for idx in order[:int(-diff)]:
            hb[idx] = 0.0
    return hb
