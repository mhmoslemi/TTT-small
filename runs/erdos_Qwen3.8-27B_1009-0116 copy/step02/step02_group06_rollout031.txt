import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _jac_all(h, n):
    c = 2*n - 1
    ks = np.arange(c)
    js = np.arange(n)
    m1 = js[:, None] + ks[None, :] - (n-1)
    m2 = js[:, None] - ks[None, :] + (n-1)
    ok1 = (m1 >= 0) & (m1 < n)
    ok2 = (m2 >= 0) & (m2 < n)
    t1 = np.where(ok1, 1.0 - h[np.clip(m1, 0, n-1)], 0.0)
    t2 = np.where(ok2, h[np.clip(m2, 0, n-1)], 0.0)
    return (t1 - t2).T

def _solve_1d(corr, g, d_lo, d_hi):
    if d_hi - d_lo < 1e-14:
        return 0.5*(d_lo + d_hi)
    npts = 21
    ds = np.linspace(d_lo, d_hi, npts)
    vals = np.max(corr[None, :] + ds[:, None] * g[None, :], axis=1)
    bi = np.argmin(vals)
    lo = ds[max(0, bi-1)]
    hi = ds[min(npts-1, bi+1)]
    for _ in range(40):
        mid = 0.5*(lo+hi)
        eps = 1e-10*(hi-lo)+1e-15
        fL = np.max(corr + (mid-eps)*g)
        fR = np.max(corr + (mid+eps)*g)
        if fL <= fR:
            hi = mid
        else:
            lo = mid
    return 0.5*(lo+hi)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _sweep_pass(h, J, corr, n, dx):
    """One full pass over all pairs using precomputed Jacobian"""
    improved = False
    c5_cur = float(np.max(corr)) * dx
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            hi, hj = h[i], h[j]
            d_lo = max(-hi, hj - 1.0)
            d_hi = min(1.0 - hi, hj)
            if d_hi - d_lo < 1e-12:
                continue
            g = J[:, i] - J[:, j]
            d_best = _solve_1d(corr, g, d_lo, d_hi)
            if abs(d_best) < 1e-14:
                continue
            new_corr = corr + d_best * g
            new_c5 = float(np.max(new_corr)) * dx
            if new_c5 < c5_cur - 1e-15:
                h[i] += d_best
                h[j] -= d_best
                corr = new_corr
                J = _jac_all(h, n)  # Update Jacobian
                c5_cur = new_c5
                improved = True
    return h, corr, J, improved

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _slsqp(h, n, target, bounds, alpha, maxiter):
    dx = 2.0/n
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    def obj_grad(x):
        c = _corr(x)
        J = _jac_all(x, n)
        m = float(np.max(c))
        s = alpha * (c - m)
        s = np.clip(s, -500, 500)
        w = np.exp(s) / np.sum(np.exp(s))
        return (J.T @ w) / alpha * alpha  # = J.T @ w
        # Wait: d/dx [m + log(sum(exp(alpha*(c-m))))/alpha]
        # = (1/alpha) * (1/sum(exp)) * sum(exp * alpha * dc/dx)
        # = sum(exp/sum(exp)) * dc/dx
        # = w^T * dc/dx
        # dc/dx = J (rows are k, cols are j)
        # So grad = J.T @ w
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):

def _jac_all(h, n):
    c = 2 * n - 1
    ks = np.arange(c)
    js = np.arange(n)
    m1 = js[:, None] + ks[None, :] - (n - 1)
    m2 = js[:, None] - ks[None, :] + (n - 1)
    ok1 = (m1 >= 0) & (m1 < n)
    ok2 = (m2 >= 0) & (m2 < n)
    t1 = np.where(ok1, 1.0 - h[np.clip(m1, 0, n - 1)], 0.0)
    t2 = np.where(ok2, h[np.clip(m2, 0, n - 1)], 0.0)
    return (t1 - t2).T

def _solve_1d(corr, g, d_lo, d_hi):
    if d_hi - d_lo < 1e-13:
        return 0.5 * (d_lo + d_hi)
    npts = 11
    ds = np.linspace(d_lo, d_hi, npts)
    vals = np.max(corr[None, :] + ds[:, None] * g[None, :], axis=1)
    bi = int(np.argmin(vals))
    lo = ds[max(0, bi - 1)]
    hi = ds[min(npts - 1, bi + 1)]

        fR = float(np.max(corr + (mid + eps) * g))
        if fL <= fR:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)

def _sweep_pass(h, J, corr, n, order):
    c5_cur = float(np.max(corr))
    imp_count = 0
    for i, j in order:
        if i == j:
            continue
        hi, hj = h[i], h[j]
        d_lo = max(-hi, hj - 1.0)
        d_hi = min(1.0 - hi, hj)
        if d_hi - d_lo < 1e-12:
            continue
        g = J[:, i] - J[:, j]
        if np.all(np.abs(g) < 1e-15):
            continue
        d_best = _solve_1d(corr, g, d_lo, d_hi)
        if abs(d_best) < 1e-14:
            continue
        new_max = float(np.max(corr + d_best * g))
        if new_max < c5_cur - 1e-15:
            h[i] += d_best
            h[j] -= d_best
            corr = corr + d_best * g
            c5_cur = new_max
            imp_count += 1
    if imp_count > 0:
        J = _jac_all(h, n)
    return h, corr, J, imp_count

        m = float(np.max(c))
        s = alpha * (c - m)

        w = e / e.sum()
        f = m + float(np.log(e.sum())) / alpha
        J = _jac_all(x, n)
        grad = J.T @ w
        return f, grad
    def cons_fun(x):
        return np.sum(x) - target

        return np.ones(n)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        out = _repair(res.x, n)
        return out
    except Exception:
        return h
