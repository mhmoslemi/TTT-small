import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _proj(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h2 = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target - h2.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h2)
        else:
            order = np.argsort(h2)

def _grad_smoothed(h, alpha):

    m = c.max()

    shifts = np.arange(2 * n - 1) - (n - 1)
    js = np.arange(n)[:, None]
    ss = shifts[None, :]
    jm = js - ss
    jp = js + ss
    vm = (jm >= 0) & (jm < n)
    vp = (jp >= 0) & (jp < n)
    jm_c = np.clip(jm, 0, n - 1)
    jp_c = np.clip(jp, 0, n - 1)
    mat = (1.0 - h[jm_c]) * vm - h[jp_c] * vp
    grad = mat.dot(w)
    return grad, c, m

def _pgd(h, n, dx, alphas, t_limit, t_start):

    for alpha in alphas:

            break

                break

                        best_c5 = c_new
                        best_h = h.copy()
                    break
                lr *= 0.5
            if lr < 1e-8:

    return best_h, best_c5

def _pair_cd(h, n, dx, t_limit, t_start, max_passes=5000, n_trials=257):
    c = _corr(h)
    best_c5 = float(c.max() * dx)
    rng = np.random.RandomState(0)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for _pass in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        improved = False
        for _ in range(n_trials):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            A = np.zeros(2 * n - 1)
            vm_i = (i - shifts >= 0) & (i - shifts < n)
            vp_i = (i + shifts >= 0) & (i + shifts < n)
            A[vm_i] += (1.0 - h[i - shifts[vm_i]])
            A[vp_i] += -h[i + shifts[vp_i]]
            vm_j = (j - shifts >= 0) & (j - shifts < n)
            vp_j = (j + shifts >= 0) & (j + shifts < n)
            A[vm_j] -= (1.0 - h[j - shifts[vm_j]])
            A[vp_j] += h[j + shifts[vp_j]]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-10:
                continue
            top_idx = np.argsort(c)[-10:]
            cands = {lo, hi}
            for k1 in top_idx:
                for k2 in top_idx:
                    if k1 != k2 and abs(A[k1] - A[k2]) > 1e-12:
                        d = (c[k2] - c[k1]) / (A[k1] - A[k2])
                        if lo + 1e-10 < d < hi - 1e-10:
                            cands.add(d)
            for d in np.linspace(lo, hi, 5):
                cands.add(d)
            best_d = 0.0
            best_val = best_c5
            for d in cands:
                d = float(np.clip(d, lo, hi))
                v = float((c + d * A).max() * dx)
                if v < best_val - 1e-14:
                    best_val = v
                    best_d = d

                h[j] -= best_d
                c = c + best_d * A
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5, c
