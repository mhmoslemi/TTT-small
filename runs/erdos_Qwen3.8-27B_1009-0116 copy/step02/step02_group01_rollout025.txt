                    best_local = c_try

    """Pair coordinate descent.”"""
    best_local = _c5_val(h, dx)
    deltas = [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001]

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, 'full')

def _c5_val(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _smooth_obj(h, alpha):
    c = _corr_full(h)
    m = c.max()
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr_full(h)
    m = c.max()
    logs = alpha * (c - m)
    w = np.exp(logs)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        wk = w[k]
        if wk < 1e-13:
            continue
        if L >= 0:
            if L < n:
                grad[L:] += wk * (1.0 - h[:n - L])
                grad[:n - L] -= wk * h[L:]
        else:
            Lp = -L
            if 2 * Lp < n:
                grad[Lp:n - Lp] += wk * (1.0 - h[2 * Lp:])
                grad[2 * Lp:] -= wk * h[Lp:n - Lp]
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                room = (1.0 - h[idx]) if diff > 0 else h[idx]
                adj = min(abs(diff), room)
                h[idx] += adj if diff > 0 else -adj
                diff += adj if diff < 0 else -adj
                if abs(diff) < 1e-14:
                    break
    return h

def _slsqp_step(h, alpha, bounds, cons, maxiter, ftol):
    res = minimize(
        _smooth_obj, h, args=(alpha,),
        jac=lambda x, a=alpha: _smooth_grad(x, a),
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def _coord_desc(h, n, dx, rng, budget_fn, reserve, n_passes, delta_scale):
    best_local = _c5_val(h, dx)
    deltas = np.array([0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001]) * delta_scale
    stagnant = 0
    for _ in range(n_passes):
        if budget_fn() > reserve:
            break
        any_imp = False
        for _ in range(800):
            if budget_fn() > reserve:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = deltas[rng.randint(len(deltas))]
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                continue
            hi, hj = h[i], h[j]
            h[i] = max(0.0, min(1.0, ni))
            h[j] = max(0.0, min(1.0, nj))
            c_new = _c5_val(h, dx)
            if c_new < best_local - 1e-15:
                best_local = c_new
                any_imp = True
                stagnant = 0
            else:
                h[i], h[j] = hi, hj
        if not any_imp:
            stagnant += 1
            if stagnant >= 4:
                break
    return h, best_local

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5_val(best_h, dx)

    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: ones}]

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP with increasing alpha (main optimization)
    alphas = [200, 1000, 5000, 20000, 80000, 300000]
    for alpha in alphas:
        if elapsed() > budget_s - 250:
            break
        for attempt in range(2):
            if elapsed() > budget_s - 250:

                h_try = _repair(h_try, n)
                c5 = _c5_val(h_try, dx)
                if c5 < best_c5 - 1e-15:
                    best_c5, best_h = c5, h_try.copy()
                    h = h_try
            except Exception:

    # Phase 2: Second round of SLSQP from best (polish)
    for alpha in [50000, 200000, 1000000]:

        except Exception:
            pass

    # Phase 3: Coarse coordinate descent
    if elapsed() < budget_s - 100:

        if c < best_c5:

    if elapsed() < budget_s - 80:

            if elapsed() > budget_s - 80:

            try:

        h = best_h.copy()

        for _ in range(50000):

                break
            i = rng.randint(n)
            j = rng.randint(n)

                continue

            h[i], h[j] = ni, nj
            c_new = _c5_val(h, dx)

                if c_cur < best_c5:

                h[i], h[j] = hi, hj

    # Final repair and validation
    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
