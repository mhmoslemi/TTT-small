    # Upper bound: current max

    hi_t = float(np.max(corr))  # current max (d=0)
    # Actually a better lower bound
    lo_t = float(np.min(corr) - np.max(np.abs(gd)) * max(abs(d_min), abs(d_max)))
    
    for _ in range(60):
        mid_t = (lo_t + hi_t) / 2
        # Check feasibility: ∃ d ∈ [d_min, d_max] s.t. corr[k] + d*gd[k] ≤ mid_t ∀k
        lo_d = d_min
        hi_d = d_max
        feasible = True
        for k in range(n_c):
            if gd[k] > 1e-14:
                # d ≤ (mid_t - corr[k]) / gd[k]
                ub = (mid_t - corr[k]) / gd[k]

                    hi_d = ub
                if hi_d < lo_d:
                    feasible = False
                    break
            elif gd[k] < -1e-14:
                # d ≥ (mid_t - corr[k]) / gd[k]
                lb = (mid_t - corr[k]) / gd[k]
                if lb > lo_d:
                    lo_d = lb
                if lo_d > hi_d:
                    feasible = False
                    break
            else:
                if corr[k] > mid_t:
                    feasible = False
                    break
        if feasible:
            hi_t = mid_t
        else:
            lo_t = mid_t
    
    t_star = hi_t
    # Find d at t_star
    lo_d = d_min
    hi_d = d_max
    for k in range(n_c):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _gd(h, i, j, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    # gradient for i
    ik = i + k
    m1 = (ik >= 0) & (ik < n)
    g[m1] += (1.0 - h[ik[m1]])
    ik2 = i - k
    m2 = (ik2 >= 0) & (ik2 < n)

    # subtract gradient for j

    jk2 = j - k
    m4 = (jk2 >= 0) & (jk2 < n)

    return g

                g = _gd(h, i, j, n)

                # Binary search for optimal d
                t_lo = float(np.min(corr))
                t_hi = cur_max
                for _ in range(50):

                    lo_d = d_min
                    hi_d = d_max
                    ok = True
                    pos_mask = g > 1e-14
                    neg_mask = g < -1e-14
                    if np.any(pos_mask):
                        ub = np.min((t_mid - corr[pos_mask]) / g[pos_mask])
                        hi_d = min(hi_d, float(ub))
                    if np.any(neg_mask):

                        lo_d = max(lo_d, float(lb))
                    if lo_d > hi_d + 1e-14:

                    if ok:
                        t_hi = t_mid
                    else:
                        t_lo = t_mid
                if t_hi < cur_max - 1e-13:

                    lo_d = d_min
                    hi_d = d_max
                    pos_mask = g > 1e-14

                    if np.any(neg_mask):

    m1 = (ik >= 0) & (ik < n)
    g[m1] += (1.0 - h[ik[m1]])
    ik2 = i - k
    m2 = (ik2 >= 0) & (ik2 < n)
    g[m2] -= h[ik2[m2]]
    jk = j + k
    m3 = (jk >= 0) & (jk < n)
    g[m3] -= (1.0 - h[jk[m3]])
    jk2 = j - k
    m4 = (jk2 >= 0) & (jk2 < n)
    g[m4] += h[jk2[m4]]
    return g

def _exact_sweep_pass(h, n, dx, corr):
    improved = False
    for i in range(n - 1, -1, -1):
        for j in range(i):
            if j == i:
                continue
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            g = _gd(h, i, j, n)
            cur_max = float(np.max(corr))
            t_lo = float(np.min(corr) - np.max(np.abs(g)) * max(abs(d_min), abs(d_max)))
            t_hi = cur_max
            pos_mask = g > 1e-14
            neg_mask = g < -1e-14
            for _ in range(45):
                t_mid = (t_lo + t_hi) / 2.0
                lo_d = d_min
                hi_d = d_max
                if np.any(pos_mask):
                    ub = np.min((t_mid - corr[pos_mask]) / g[pos_mask])
                    if ub < hi_d:
                        hi_d = float(ub)
                if np.any(neg_mask):
                    lb = np.max((t_mid - corr[neg_mask]) / g[neg_mask])
                    if lb > lo_d:
                        lo_d = float(lb)
                if lo_d > hi_d:
                    t_lo = t_mid
                else:
                    t_hi = t_mid
            if t_hi < cur_max - 1e-13:
                lo_d = d_min
                hi_d = d_max
                if np.any(pos_mask):
                    hi_d = min(hi_d, float(np.min((t_hi + 1e-14 - corr[pos_mask]) / g[pos_mask])))
                if np.any(neg_mask):
                    lo_d = max(lo_d, float(np.max((t_hi + 1e-14 - corr[neg_mask]) / g[neg_mask])))
                if lo_d > hi_d:
                    d_star = (lo_d + hi_d) / 2.0
                else:
                    d_star = (lo_d + hi_d) / 2.0
                d_star = max(d_min, min(d_max, d_star))
                if abs(d_star) > 1e-14:
                    h[i] = hi + d_star
                    h[j] = hj - d_star
                    corr = corr + d_star * g
                    improved = True
    return h, corr, improved

def _exact_sweep(h, n, dx, corr, max_passes=50):
    for _ in range(max_passes):
        h, corr, imp = _exact_sweep_pass(h, n, dx, corr)
        if not imp:
            break
    return h, corr

def _obj_smax(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _eq_con(x, target):
    return np.sum(x) - target

def _slsqp_solve(h, n, dx, alpha, maxiter=600):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and not np.any(np.isnan(res.x)):
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()

def _de_obj(x, n):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    diff = np.sum(h) - n / 2.0
    penalty = 1e5 * diff * diff
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx + penalty

def _de_search(h_best, n, seed, maxiter=200, popsize=20):
    rng = np.random.RandomState(seed)
    bounds = [(0.0, 1.0)] * n
    x0 = _repair(h_best + 0.1 * rng.randn(n), n)
    try:
        res = differential_evolution(_de_obj, bounds, args=(n,),
                                     seed=seed, maxiter=maxiter,
                                     popsize=popsize, tol=1e-10,
                                     mutation=(0.3, 1.0), recombination=0.8,
                                     polish=True, init='latinhypercube')
        if res.x is not None and not np.any(np.isnan(res.x)):
            return _repair(res.x, n)
    except Exception:
        pass
    return h_best.copy()

def _binary_snap(h, n):
    """Push values toward 0 or 1, maintaining sum constraint"""
    target = n / 2.0
    idx_one = np.argsort(-h)[:int(round(target))]
    h_bin = np.zeros(n)
    h_bin[idx_one] = 1.0
    return h_bin

def _score(h, dx):
    return float(np.max(_corr(h)) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Exact coordinate pair sweeps
    h = best_h.copy()
    corr = _corr(h)
    while time.time() - t0 < budget_s - 500:
        h, corr = _exact_sweep(h, n, dx, corr, max_passes=10)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
        else:
            break

    # Phase 2: SLSQP with increasing smoothness
    for alpha in [30, 80, 200, 500, 1200, 3000, 8000]:
        if time.time() - t0 > budget_s - 350:
            break
        hc = _slsqp_solve(best_h, n, dx, alpha, 500)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = hc
            corr = _corr(best_h)

    # Phase 3: Sweets from SLSQP
    if time.time() - t0 < budget_s - 300:
        h = best_h.copy()
        corr = _corr(h)
        for _ in range(30):
            if time.time() - t0 > budget_s - 300:
                break
            h, corr = _exact_sweep(h, n, dx, corr, max_passes=5)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: Differential evolution (global)
    if time.time() - t0 < budget_s - 200:
        de_time_left = min(budget_s - 200 - (time.time() - t0), 250)
        maxiter_de = max(30, int(de_time_left / 2.0))
        hc = _de_search(best_h, n, seed + 1, maxiter=maxiter_de, popsize=15)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 5: Random perturbations + sweep
    for trial in range(15):
        if time.time() - t0 > budget_s - 80:
            break
        scale = 0.03 + 0.07 * rng.rand()
        h_pert = best_h + scale * rng.randn(n)
        h_pert = _repair(h_pert, n)
        corr_p = _corr(h_pert)
        h_pert, corr_p = _exact_sweep(h_pert, n, dx, corr_p, max_passes=15)
        c5 = float(np.max(corr_p) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pert.copy()

    # Phase 6: Binary snap + re-optimize
    if time.time() - t0 < budget_s - 40:
        h_bin = _binary_snap(best_h, n)
        corr_b = _corr(h_bin)
        h_bin, corr_b = _exact_sweep(h_bin, n, dx, corr_b, max_passes=20)
        c5 = float(np.max(corr_b) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_bin.copy()

    # Phase 7: Final SLSQP polish + sweep
    if time.time() - t0 < budget_s - 10:
        hc = _slsqp_solve(best_h, n, dx, 5000, 1000)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    if time.time() - t0 < budget_s - 3:
        h = best_h.copy()
        corr = _corr(h)
        h, corr = _exact_sweep(h, n, dx, corr, max_passes=20)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
