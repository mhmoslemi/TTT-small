from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(c, alpha):
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _all_grads(h, n):
    cl = 2*n - 1
    grads = np.zeros((n, cl))
    for i in range(n):
        for m in range(cl):
            s = m - (n-1)
            k1 = i - s
            if 0 <= k1 < n:
                grads[i, m] += 1.0 - h[k1]
            k2 = i + s
            if 0 <= k2 < n:
                grads[i, m] -= h[k2]
    return grads

def _pair_sweep(h, n, dx, n_grid, n_bisect):
    """One full sweep over all pairs (i,j) with i<j"""
    corr = _corr(h)
    grads = _all_grads(h, n)
    improved = False
    for i in range(n):
        for j in range(i+1, n):
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-14:
                continue
            g = grads[i] - grads[j]
            # min_d max_k (corr[k] + d*g[k]) for d in [d_min, d_max]
            # Convex function of d. Use grid + bisection on subgradient.
            
            # Grid
            ds = np.linspace(d_min, d_max, n_grid)
            vals = np.max(corr[None,:] + ds[:,None]*g[None,:], axis=1)
            bi = np.argmin(vals)
            
            # Refine: bisection on subgradient
            # subgradient of f(d) = max_k (c0[k]+d*g[k]) is a convex combination of active g[k]
            # At optimum, 0 is in subgradient
            lo = ds[max(0, bi-1)]
            hi_b = ds[min(len(ds)-1, bi+1)]
            for _ in range(n_bisect):
                mid = (lo + hi_b) / 2
                # f'(mid-) = min active g, f'(mid+) = max active g
                # If all g[k] where c0[k]+mid*g[k] = max are >= 0, f is increasing at mid
                # If all are <= 0, f is decreasing
                active = corr + mid * g
                max_val = np.max(active)
                # subgradient: values of g[k] where active[k] is close to max
                close = active > max_val - 1e-10
                if np.all(g[close] >= 0):
                    lo = mid  # f is non-decreasing after mid
                elif np.all(g[close] <= 0):
                    hi_b = mid  # f is non-increasing before mid
                else:
                    break  # mid is optimal
            d_opt = (lo + hi_b) / 2
            
            # Check if it improves
            new_corr = corr + d_opt * g
            new_max = np.max(new_corr)
            if new_max < np.max(corr) - 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                corr = new_corr
                # Update grads (they depend on h)
                # Actually we need to recompute grads since h changed
                grads = _all_grads(h, n)
                improved = True
    return h, improved

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smax(c, a):
    m = float(np.max(c))
    s = a * (c - m)
    s = np.clip(s, -400, 400)
    return m + float(np.log(np.sum(np.exp(s)))) / a

def _obj(x, a):
    return _smax(_corr(x), a)

def _all_grads(h, n):
    cl = 2 * n - 1
    G = np.zeros((n, cl))
    for i in range(n):
        for m in range(cl):
            s = m - n + 1
            k1 = i - s
            if 0 <= k1 < n:
                G[i, m] += 1.0 - h[k1]
            k2 = i + s
            if 0 <= k2 < n:
                G[i, m] -= h[k2]
    return G

def _pair_sweep(h, n, dx, ng, nb, tick=None, t0=None, budget=0, reserve=60):

    G = _all_grads(h, n)
    improved = False

    for i in range(n):
        for j in range(i + 1, n):
            if tick and time.time() - t0 > budget - reserve:
                return h, improved
            hi, hj = h[i], h[j]
            d_lo = max(-hi, hj - 1.0)
            d_hi = min(1.0 - hi, hj)
            if d_hi - d_lo < 1e-14:
                continue
            g = G[i] - G[j]
            # max at current
            cm = float(np.max(corr))
            # Grid
            ds = np.linspace(d_lo, d_hi, ng)
            vals = np.max(corr[None, :] + ds[:, None] * g[None, :], axis=1)
            bi = int(np.argmin(vals))
            # Refine with subgradient bisection
            lo = ds[max(0, bi - 1)]
            hi_ = ds[min(ng - 1, bi + 1)]
            for _ in range(nb):

                act = corr + mid * g
                mv = float(np.max(act))
                close = act > mv - 1e-9
                if not np.any(close):

                gmax = float(np.max(g[close]))
                if gmin > 1e-12:

                elif gmax < -1e-12:
                    hi_ = mid
                else:

            d_opt = 0.5 * (lo + hi_)

                h[j] -= d_opt
                h[i] = min(max(h[i], 0.0), 1.0)

                updates += 1
                if updates % 30 == 0:

                improved = True

    return h, improved

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smax(c, a):
    m = float(np.max(c))
    s = a * (c - m)
    s = np.clip(s, -400, 400)
    return m + float(np.log(np.sum(np.exp(s)))) / a

def _obj(x, a):
    return _smax(_corr(x), a)

def _constraint_eq(x, target):
    return np.sum(x) - target

def _all_grads(h, n):
    cl = 2 * n - 1
    G = np.zeros((n, cl))
    for i in range(n):
        lo1 = i
        hi1 = min(cl - 1, i + n - 1)
        if lo1 <= hi1:
            ms = np.arange(lo1, hi1 + 1)
            G[i, lo1:hi1 + 1] += 1.0 - h[i + n - 1 - ms]
        lo2 = max(0, n - 1 - i)
        hi2 = min(cl - 1, 2 * n - 2 - i)
        if lo2 <= hi2:
            ms = np.arange(lo2, hi2 + 1)
            G[i, lo2:hi2 + 1] -= h[i + ms - n + 1]
    return G

def _pair_sweep(h, n, dx, ng, nb, t0, budget, reserve):
    corr = _corr(h)
    G = _all_grads(h, n)
    improved = False
    updates = 0
    for i in range(n):
        for j in range(i + 1, n):
            if time.time() - t0 > budget - reserve:
                return h, improved
            hi, hj = h[i], h[j]
            d_lo = max(-hi, hj - 1.0)
            d_hi = min(1.0 - hi, hj)
            if d_hi - d_lo < 1e-14:
                continue
            g = G[i] - G[j]
            cm = float(np.max(corr))
            ds = np.linspace(d_lo, d_hi, ng)
            vals = np.max(corr[None, :] + ds[:, None] * g[None, :], axis=1)
            bi = int(np.argmin(vals))
            lo = ds[max(0, bi - 1)]
            hi_b = ds[min(ng - 1, bi + 1)]
            for _ in range(nb):
                mid = 0.5 * (lo + hi_b)
                act = corr + mid * g
                mv = float(np.max(act))
                close = act > mv - 1e-9
                if not np.any(close):
                    break
                gmin = float(np.min(g[close]))
                gmax = float(np.max(g[close]))
                if gmin > 1e-12:
                    lo = mid
                elif gmax < -1e-12:
                    hi_b = mid
                else:
                    break
            d_opt = 0.5 * (lo + hi_b)
            new_cm = float(np.max(corr + d_opt * g))
            if new_cm < cm - 1e-13:
                h[i] = min(max(h[i] + d_opt, 0.0), 1.0)
                h[j] = min(max(h[j] - d_opt, 0.0), 1.0)
                corr = _corr(h)
                G = _all_grads(h, n)
                updates += 1
                improved = True
    return h, improved

def _slsqp_step(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _constraint_eq, 'args': (target,)}]
    try:
        res = minimize(_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    # Phase 1: Progressive SLSQP

    # Phase 2: Pair sweeps coarse→fine
