from scipy.optimize import minimize, linprog

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx, c * dx

def _crit_vals(h, dx):
    """Return the correlation values (not scaled by dx) and the max."""
    c = np.correlate(h, 1.0 - h, mode='full')
    return c

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    exp_s = np.exp(s)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp  # softmax weights over lags

    # c[k] = sum_{valid i} h[i] * (1 - h[i+r])
    # d c[k] / d h[j] = (1-h[j+r])*I[j valid for i] - h[j-r]*I[j-r valid for i]
    g = np.zeros(2*n-1)  # gradient per lag

        # valid i range: max(0,-r) <= i <= min(n-1, n-1-r)

                diff -= add
            else:

                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def _c5_with_arr(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx, c

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / np.sum(e)
    n = len(x)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        r = k - (n - 1)
        if r >= 0:
            i_lo, i_hi = 0, n - 1 - r
            j_lo, j_hi = r, n - 1
        else:
            rp = -r
            i_lo, i_hi = rp, n - 1
            j_lo, j_hi = 0, n - 1 - rp
        # Term 1: (1-h[j+r]) for j in [i_lo, i_hi]
        if r >= 0:
            grad[i_lo:i_hi+1] += w[k] * (1.0 - x[i_lo+r:i_hi+r+1])
        else:
            grad[i_lo:i_hi+1] += w[k] * (1.0 - x[i_lo-r:i_hi-r+1])  # h[j+r] where r<0 means h[j-rp]
        # Term 2: -h[j-r] for j in [j_lo, j_hi]
        if r >= 0:
            grad[j_lo:j_hi+1] -= w[k] * x[j_lo-r:j_hi-r+1]
        else:
            grad[j_lo:j_hi+1] -= w[k] * x[j_lo-r:j_hi-r+1]  # j-r = j+rp
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp(h, n, dx, alpha_list, maxiter, budget_check):
    best_h, best_c5 = h.copy(), _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    for alpha in alpha_list:
        if budget_check():
            break
        try:
            res = minimize(
                lambda x: _smooth_obj(x, alpha), best_h,
                jac=lambda x: _smooth_grad(x, alpha),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass
    return best_h, best_c5

def _proj_grad(h, n, dx, iters, budget_check):
    h = h.copy()
    c5_cur = _c5(h, dx)
    lr = 0.01
    for it in range(iters):
        if budget_check():
            break
        c, cdx = _c5_with_arr(h, dx)
        # Top 3 lags
        top = np.argsort(c)[-3:][::-1]
        g = np.zeros(n)
        for k in top:
            w = 1.0 if k == top[0] else 0.5
            r = k - (n - 1)
            if r >= 0:
                i_lo, i_hi = 0, n - 1 - r
                j_lo, j_hi = r, n - 1
            else:
                rp = -r; i_lo, i_hi = rp, n - 1; j_lo, j_hi = 0, n - 1 - rp
            if r >= 0:

                g[j_lo:j_hi+1] -= w * h[j_lo-r:j_hi-r+1]
            else:
                g[i_lo:i_hi+1] += w * (1.0 - h[i_lo-r:i_hi-r+1])
                g[j_lo:j_hi+1] -= w * h[j_lo-r:j_hi-r+1]
        h_new = _repair(h - lr * g, n)
        c5_new = _c5(h_new, dx)
        if c5_new < c5_cur:
            h = h_new; c5_cur = c5_new; lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.8
    return h, c5_cur

def _sa_fine(h, n, dx, rng, iters, T0, T1, budget_check):
    h = h.copy()
    c5_cur = _c5(h, dx)
    T = T0
    for it in range(iters):
        if budget_check():
            break
        i, j = rng.randint(n), rng.randint(n)

            c5_cur = c5_new
        else:
            h[i], h[j] = hi, hj
        T = max(T1, T * (T1 / T0) ** (1.0 / max(iters, 1)))
    return h, c5_cur

    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def _c5_arr(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx, c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _lag_grad(x, k, n):
    r = k - (n - 1)
    g = np.zeros(n)
    if r >= 0:
        i_lo, i_hi = 0, n - 1 - r
        j_lo, j_hi = r, n - 1
    else:
        i_lo, i_hi = -r, n - 1
        j_lo, j_hi = 0, n - 1 + r
    g[i_lo:i_hi + 1] += (1.0 - x[i_lo + r:i_hi + r + 1])
    g[j_lo:j_hi + 1] -= x[j_lo - r:j_hi - r + 1]
    return g

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_gradient(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        grad += w[k] * _lag_grad(x, k, n)
    return grad

def _slsqp_opt(h, n, dx, alphas, maxiter, stop_fn):
    best_h, best_c5 = h.copy(), _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    for a in alphas:
        if stop_fn():
            break
        try:
            res = minimize(
                lambda x, al=a: _smooth_obj(x, al), best_h,
                jac=lambda x, al=a: _smooth_gradient(x, al),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass
    return best_h, best_c5

def _proj_grad(h, n, dx, iters, stop_fn):
    h = h.copy()
    c5_cur = _c5(h, dx)
    lr = 0.008
    for _ in range(iters):
        if stop_fn():
            break
        _, c = _c5_arr(h, dx)
        top = np.argsort(c)[-4:][::-1]
        g = np.zeros(n)
        for idx, k in enumerate(top):
            w = 1.0 if idx == 0 else 0.4
            g += w * _lag_grad(h, k, n)
        h_new = _repair(h - lr * g, n)
        c5_new = _c5(h_new, dx)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.85
            if lr < 1e-5:
                break
    return h, c5_cur

def _sa_refine(h, n, dx, rng, iters, T0, stop_fn):
    h = h.copy()
    c5_cur = _c5(h, dx)
    T = T0
    cool = (1e-7 / T0) ** (1.0 / max(iters, 1))
    steps = np.array([0.002, 0.005, 0.01, 0.02])
    for _ in range(iters):
        if stop_fn():
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = steps[rng.randint(len(steps))] * (1 if rng.rand() < 0.5 else -1)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < -1e-14 or ni > 1 + 1e-14 or nj < -1e-14 or nj > 1 + 1e-14:
            continue
        h[i], h[j] = max(0.0, min(1.0, ni)), max(0.0, min(1.0, nj))
        c5_new = _c5(h, dx)
        if c5_new < c5_cur or (T > 1e-8 and rng.rand() < np.exp(-(c5_new - c5_cur) / T)):
            c5_cur = c5_new
        else:
            h[i], h[j] = hi, hj
        T = max(1e-7, T * cool)
    return h, c5_cur

def _upsample(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new, n_new)
