from scipy.optimize import minimize
import time

def _c5_val(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(c.max())

def _c5_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_smooth_max(h, n, c, alpha):
    """Gradient of (1/alpha)*log(sum(exp(alpha*c))) w.r.t. h."""
    cm = c.max()
    w = np.exp(alpha * (c - cm))
    w /= w.sum()
    # G has shape (2n-1, n)
    G = np.zeros((2*n-1, n))
    for idx in range(2*n-1):
        k = idx - (n-1)
        for m in range(n):
            v = 0.0
            mk = m - k
            if 0 <= mk < n:
                v += (1.0 - h[mk])
            mk2 = m + k
            if 0 <= mk2 < n:
                v -= h[mk2]
            G[idx, m] = v
    return w @ G

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _smooth_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    if cm < 1e-15:
        return cm
    s = np.exp(alpha * (c - cm))
    return cm + np.log(s.sum()) / alpha

        w = np.exp(alpha * (c - cm))

    # c[j] = sum_i h[i+k] * (1-h[i]) where k = j-(n-1)

import time

def _c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def _grad_ovl(h, n):
    G = np.zeros((2*n-1, n))
    ka = np.arange(-(n-1), n)
    for m in range(n):
        v1 = (m+ka >= 0) & (m+ka < n)
        v2 = (m-ka >= 0) & (m-ka < n)
        G[v1, m] = 1.0 - h[m+ka[v1]]
        G[v2, m] += -h[m-ka[v2]]
    return G

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-15: return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0-h) if d > 0 else np.argsort(h)
        for i in idx:
            if abs(d) < 1e-14: break
            a = min(abs(d), h[i] if d < 0 else 1.0-h[i])
            h[i] += a if d > 0 else -a
            d += a if d > 0 else -a
    return h

def _nest(h, n, dx, t_end):

    stall = 0

        c = np.correlate(y, 1.0-y, mode='full')
        cm = c.max()
        w = np.exp(np.clip(alpha*(c-cm), -500, 500)); w /= w.sum()
        g = _grad_ovl(y, n).T @ w
        h_n = _repair(h_c - lr*g, n)

            alpha = max(alpha * 0.9, 5.0)
        h_p, h_c = h_c, h_n
        alpha = min(alpha * 1.003, 3000.0)
        if alpha > 200: mu = max(mu*0.999, 0.3)
    return best, bv

        for i0 in range(n):
            if time.time() > t_end: break
            i = order[i0] if focused else i0
            for j in range(i+1, n):
                if time.time() > t_end: break
                hs = h[i] + h[j]
                lo, hi = max(0.0, hs-1.0), min(1.0, hs)
                if hi - lo < 1e-10: continue

                for t in np.linspace(lo+1e-8, hi-1e-8, ng):
                    h[i] = t; h[j] = hs - t
                    v = _c5v(h, dx)
                    if v < bc: bc = v

def _slsqp(h, n, dx, alpha, maxit=400):
    t = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        cm = c.max()
        return cm + np.log(np.sum(np.exp(np.clip(alpha*(c-cm),-400,400))))/alpha
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0,1)]*n,

import numpy as np
from scipy.optimize import minimize
import time

def _c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _grad_ovl(h, n):
    G = np.zeros((2 * n - 1, n))
    ka = np.arange(-(n - 1), n)
    for m in range(n):
        v1 = (m + ka >= 0) & (m + ka < n)
        v2 = (m - ka >= 0) & (m - ka < n)
        G[v1, m] = 1.0 - h[m + ka[v1]]
        G[v2, m] += -h[m - ka[v2]]
    return G

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if abs(d) < 1e-14:
                break
            a = min(abs(d), h[i] if d < 0 else 1.0 - h[i])
            h[i] += a if d > 0 else -a
            d += a if d > 0 else -a
    return h

def _nest(h, n, dx, t_end, alpha0=60.0, lr0=0.05):
    h_c, h_p = h.copy(), h.copy()
    best, bv = h.copy(), _c5v(h, dx)
    alpha, mu, lr = alpha0, 0.8, lr0
    stall = 0
    while time.time() < t_end:
        y = h_c + mu * (h_c - h_p)
        c = np.correlate(y, 1.0 - y, mode='full')
        cm = c.max()
        w = np.exp(np.clip(alpha * (c - cm), -500, 500))
        ws = w.sum()
        if ws < 1e-300:
            break
        w /= ws
        g = _grad_ovl(y, n).T @ w
        h_n = _repair(h_c - lr * g, n)
        v = _c5v(h_n, dx)
        if v < bv - 1e-14:

        h_p, h_c = h_c, h_n
        alpha = min(alpha * 1.004, 4000.0)
        if alpha > 300:
            mu = max(mu * 0.9985, 0.25)
    return best, bv

    while time.time() < t_end:
        improved_any = False

            sens = np.abs(_grad_ovl(h, n).T @ w)
            order = np.argsort(-sens)
        else:
            order = np.arange(n)
        for i0 in range(n):
            if time.time() > t_end:
                break
            i = int(order[i0]) if focused else i0
            for j in range(i + 1, n):
                if time.time() > t_end:

                if focused and j > i:

                    # skip: just use sequential j for simplicity

def _slsqp(h, n, dx, alpha, maxit=400):

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cm = c.max()
        return cm + np.log(np.sum(np.exp(np.clip(alpha * (c - cm), -400, 400)))) / alpha
    try:
        r = minimize(obj, h.copy(), method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - t}],
                     options={'maxiter': maxit, 'ftol': 1e-15})
        x = _repair(r.x, n)
        return x, _c5v(x, dx)
    except Exception:
        return h.copy(), _c5v(h, dx)
