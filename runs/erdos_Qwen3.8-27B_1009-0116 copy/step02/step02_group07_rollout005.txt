    """Compute smoothed gradient of max correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights over top values
    m = c.max()
    weights = np.exp(alpha * (c - m))
    weights /= weights.sum()
    
    g = np.zeros(n)
    for k in range(2*n - 1):

        if w < 1e-10:

        for j in range(n):

            if 0 <= jk < n:

            if 0 <= jk2 < n:
                g[j] -= w * h[jk2]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

        wk = w[k]
        if wk < 1e-12:
            continue
        offset = k - offset_max
        for j in range(n):
            jm = j - offset
            jp = j + offset
            if 0 <= jm < n:
                g[j] += wk * (1.0 - h[jm])
            if 0 <= jp < n:
                g[j] -= wk * h[jp]
    return g

def _proj_subgrad(h, grad, lr, n):
    """One step of projected subgradient: h - lr*grad, projected to {0<=h<=1, sum=n/2}."""
    target = n / 2.0
    # Bisection on lambda for projection
    lo, hi = -lr * grad.max(), lr * (-grad.min())
    # g = h - lr*grad + lam, clipped to [0,1], need sum = target
    def sum_clipped(lam):
        g = np.clip(h - lr * grad + lam, 0.0, 1.0)
        return g.sum()
    for _ in range(60):
        mid = (lo + hi) / 2
        if sum_clipped(mid) < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lr * grad + lam, 0.0, 1.0)

def _subgrad_loop(h, n, dx, n_iter, t_start, t_limit):
    """Projected subgradient method."""
    alpha = 50
    best_c = _c5(h, dx)
    for it in range(n_iter):
        if time.time() - t_start > t_limit:
            break
        alpha = min(alpha * 1.02, 500)
        g = _smooth_grad(h, n, alpha)
        lr = 0.5 / (np.linalg.norm(g) + 1e-10)
        lr = min(lr, 0.3)
        h_new = _proj_subgrad(h, g, lr, n)
        c = _c5(h_new, dx)
        if c < best_c:
            best_c = c
        else:
            lr *= 0.5
            h_new = _proj_subgrad(h, g, lr, n)
            c = _c5(h_new, dx)
            if c < best_c:
                best_c = c
            else:
                h_new = h.copy()
        h = h_new
    return h, _c5(h, dx)

def _pair_cd(h, n, dx, n_grid, max_passes, t_start, t_limit):
    """Pairwise coordinate descent with 1D search."""
    hi_c5 = _c5(h, dx)
    improved = False
    for _ in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hsum = h[i] + h[j]
                lo = max(0.0, hsum - 1.0)
                hi = min(1.0, hsum)
                if hi - lo < 1e-10:
                    continue
                orig_i = h[i]
                orig_j = h[j]
                # Coarse grid search
                best_t = orig_i
                best_c = hi_c5
                ts = np.linspace(lo, hi, n_grid)
                for t in ts:
                    h[i] = t
                    h[j] = hsum - t
                    c = _c5(h, dx)
                    if c < best_c:
                        best_c = c
                        best_t = t
                # Fine search around best
                if best_t != orig_i:
                    span = (hi - lo) / max(n_grid - 1, 1)
                    lo2 = max(lo, best_t - span)
                    hi2 = min(hi, best_t + span)
                    ts2 = np.linspace(lo2, hi2, 21)
                    for t in ts2:
                        h[i] = t
                        h[j] = hsum - t
                        c = _c5(h, dx)
                        if c < best_c:
                            best_c = c
                            best_t = t
                if best_c < hi_c5 - 1e-13:
                    h[i] = best_t
                    h[j] = hsum - best_t
                    hi_c5 = best_c
                    improved = True
                else:
                    h[i] = orig_i
                    h[j] = orig_j
    return h, hi_c5, improved

def _nm_optimize(h, n, dx, maxiter=5000):
    """Nelder-Mead optimization."""
    def obj(x):
        x = _repair(x, n)
        return _c5(x, dx)
    try:
        res = minimize(obj, h, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-7, 'fatol': 1e-9})
        return _repair(res.x, n), _c5(_repair(res.x, n), dx)
    except Exception:
        return h.copy(), _c5(h, dx)

def _slsqp_smooth(h, n, bounds, cons, alpha, maxiter, ftol):
    def _obj(x):
        x = np.clip(x, 0.0, 1.0)
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    try:
        res = minimize(_obj, h, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _periodic_init(n, period, rng):
    """Create a periodic-like initial condition."""
    base = rng.uniform(0.3, 0.7, period)
    h = np.tile(base, n // period + 1)[:n]
    return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_grad(h, n, alpha):
    c = _corr(h)
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    g = np.zeros(n)
    off_max = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        off = k - off_max
        for j in range(n):
            jm = j - off
            jp = j + off
            if 0 <= jm < n:
                g[j] += wk * (1.0 - h[jm])
            if 0 <= jp < n:
                g[j] -= wk * h[jp]
    return g

def _proj_step(h, g, lr, n):
    target = n / 2.0
    # Find lambda such that clip(h - lr*g + lambda) sums to target
    base = h - lr * g
    lo, hi = -lr * g.max() - 1, lr * (-g.min()) + 1
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.clip(base + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(base + lam, 0.0, 1.0)

def _subgrad_loop(h, n, dx, iters, t0, t_end, alpha_start=50, alpha_end=300):
    best_c = _c5(h, dx)
    for it in range(iters):
        if time.time() > t_end:
            break
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(iters-1, 1))
        g = _smooth_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        lr = min(0.5 / (gn + 1e-10), 0.2)
        h_new = _proj_step(h, g, lr, n)
        c = _c5(h_new, dx)
        if c < best_c:
            best_c = c
            h = h_new
        else:
            # Halve step
            h_new2 = _proj_step(h, g, lr * 0.5, n)
            c2 = _c5(h_new2, dx)
            if c2 < best_c:
                best_c = c2
                h = h_new2
            elif c2 < c:
                h = h_new2
    return h, _c5(h, dx)

def _cd_pass(h, n, dx, grid):
    hi_c5 = _c5(h, dx)
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hs = h[i] + h[j]
            lo = max(0.0, hs - 1.0)
            hi_v = min(1.0, hs)
            if hi_v - lo < 1e-10:
                continue
            oi = h[i]
            oj = h[j]
            best_t = oi
            best_c = hi_c5
            for t in np.linspace(lo, hi_v, grid):
                h[i] = t
                h[j] = hs - t
                c = _c5(h, dx)
                if c < best_c:
                    best_c = c
                    best_t = t
            if best_t != oi and best_c < hi_c5 - 1e-13:
                # Refine
                span = (hi_v - lo) / max(grid - 1, 1)
                for t in np.linspace(max(lo, best_t - span), min(hi_v, best_t + span), 15):
                    h[i] = t
                    h[j] = hs - t
                    c = _c5(h, dx)
                    if c < best_c:
                        best_c = c
                        best_t = t
                h[i] = best_t
                h[j] = hs - best_t
                hi_c5 = best_c
                improved = True
            else:
                h[i] = oi
                h[j] = oj
    return h, hi_c5, improved

def _cd_loop(h, n, dx, grid, passes, t0, t_end):
    for _ in range(passes):
        if time.time() > t_end:
            break
        h, c5, imp = _cd_pass(h, n, dx, grid)
        if not imp:
            break
    return h, _c5(h, dx)

def _slsqp(h, n, alpha, maxiter, t0, t_end):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    def obj(x):
        c = _corr(np.clip(x, 0, 1))
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(s))) / alpha
    time_left = t_end - time.time()
    if time_left < 5:
        return h
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': min(maxiter, int(time_left * 5)), 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

    if c < best_c5:
        best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
