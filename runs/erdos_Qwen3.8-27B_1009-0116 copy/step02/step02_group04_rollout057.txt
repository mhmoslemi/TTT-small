    # Try boundaries
    best_x = lo
    best_f = f(lo)
    if f(hi) < best_f:

    # Subgradient = conv{A[m] : m in active}
    # If min active A > 0, we need to go left
    # If max active A < 0, we need to go right
    # Otherwise, we're at a minimum

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -100, 100)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _pnorm_obj(x, p):

    return np.sum(np.abs(c) ** p) ** (1.0/p)

def _project(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(50):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            cnt = mask.sum()
            if cnt == 0: break

            h[mask] += step
            h[mask] = np.minimum(h[mask], 1.0)

            h[mask] -= step
            h[mask] = np.maximum(h[mask], 0.0)
        diff = target - h.sum()
    return h

        h = new_h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _ls_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -80, 80)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _ls_grad(x, alpha, eps=1e-7):
    n = len(x)
    f0 = _ls_obj(x, alpha)
    g = np.zeros(n)
    for i in range(n):
        x[i] += eps
        f1 = _ls_obj(x, alpha)
        x[i] -= 2*eps
        f2 = _ls_obj(x, alpha)
        x[i] += eps
        g[i] = (f1 - f2) / (2*eps)
    return g

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    for _ in range(80):
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = 1.0 - h > 1e-12
            c = m.sum()
            if c == 0: break
            h[m] = np.minimum(h[m] + d/c, 1.0)
        else:
            m = h > 1e-12
            c = m.sum()
            if c == 0: break
            h[m] = np.maximum(h[m] - d/c, 0.0)
        d = t - h.sum()
    return h

def _slsqp(h, n, alpha, maxiter, deadline):
    t = n / 2.0
    bd = [(0.0, 1.0)] * n
    co = [{'type': 'eq',
           'fun': lambda x: np.sum(x) - t,
           'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_ls_obj, h, args=(alpha,), method='SLSQP',
                     bounds=bd, constraints=co,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return _proj(r.x, n)
    except Exception:
        return h

def _pgd(h, n, alpha, deadline, lr0=0.005):
    h = h.copy()
    lr = lr0
    for _ in range(20000):
        if time.time() > deadline:
            break
        g = _ls_grad(h, alpha)
        ng = np.linalg.norm(g)
        if ng < 1e-14:

        for _bt in range(30):

            if _ls_obj(nh, alpha) < _ls_obj(h, alpha):
                break
            lr *= 0.5
        else:
            break
        h = nh
        lr = min(lr * 1.05, lr0)
    return h

    h = np.where((x * alpha_kron + rng.rand() * 0.01) % 1.0 < 0.5, 1.0, 0.0)
    return _proj(h, n)

    best = _c5(h, dx)
    h = h.copy()
