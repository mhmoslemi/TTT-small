```python
def _corr_grad(h, n, lags):
    """Compute gradient of max corr contribution for given lags."""
    g = np.zeros(n)
    for lag in lags:
        # d/dh[j] of sum_i h[i](1-h[i+lag])
        # = (1-h[j+lag]) if valid - h[j-lag] if valid
        # First term: (1 - h[j+lag]) for j where 0<=j+lag<n
        # This is (1-h) shifted right by -lag, zero-padded
        term1 = np.zeros(n)
        if lag >= 0:
            term1[:n-lag] = 1.0 - h[lag:]
        else:
            term1[-lag:] = 1.0 - h[:n+lag]
        # Second term: -h[j-lag] for j where 0<=j-lag<n
        term2 = np.zeros(n)
        if lag >= 0:
            term2[lag:] = -h[:n-lag]
        else:
            term2[:n+lag] = -h[-lag:]
        g += term1 + term2
    return g
```

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_obj(x, a):
    c = _corr(x)
    m = float(np.max(c))
    s = a * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.sum(np.exp(s))) / a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, tgt / n)
    h = (tgt / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
            if abs(d) < 1e-14:
                break
    return h

def _slsqp(h, n, a, maxit, deadline):
    dx = 2.0 / n
    if time.time() >= deadline:
        return h, _c5(h, dx)
    bnds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}
    try:
        r = minimize(lambda x: _smooth_obj(x, a), h, method='SLSQP',
                     bounds=bnds, constraints=[con],
                     options={'maxiter': maxit, 'ftol': 1e-18})
        hc = _repair(r.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _corr_grad_vec(h, n, lags):
    g = np.zeros(n)
    for lag in lags:
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if lag >= 0:
            t1[:n - lag] = 1.0 - h[lag:]
            t2[lag:] = -h[:n - lag]
        else:
            t1[-lag:] = 1.0 - h[:n + lag]
            t2[:n + lag] = -h[-lag:]
        g += t1 + t2
    return g

def _proj_grad(h, n, dx, deadline, max_iter=20000):
    best = _c5(h, dx)
    h = h.copy()
    lr = 0.005
    mom = np.zeros(n)
    beta = 0.7
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        c = _corr(h)
        top = np.argsort(c)[-8:]
        g = _corr_grad_vec(h, n, top)
        mom = beta * mom + (1 - beta) * g
        h_new = _repair(h - lr * mom, n)
        c_new = _c5(h_new, dx)
        if c_new < best - 1e-14:
            best = c_new
            h = h_new
            lr = min(lr * 1.005, 0.02)
        else:
            lr *= 0.97
            if lr < 1e-9:
                lr = 0.005
                mom *= 0.3
    return h, best

def _local_search(h, n, dx, rng, deadline):
    best = _c5(h, dx)
    h = h.copy()
    for scale in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        stalled = 0
        while time.time() < deadline and stalled < 10:
            if time.time() >= deadline:
                break
            imp = False
            for _ in range(4000):
                if time.time() >= deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                dv = scale * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + dv, h[j] - dv
                if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best - 1e-13:
                    best = c
                    imp = True
                else:
                    h[i], h[j] = ni - dv, nj + dv
            stalled += 0 if imp else 1
    for _sw in range(3):
        if time.time() >= deadline:
            break
        for i in range(n):
            if time.time() >= deadline:
                break
            for j in range(i + 1, n):
                if time.time() >= deadline:
                    break
                for f in [0.003, 0.001, 0.0005]:
                    if h[i] > f + 1e-10 and h[j] < 1.0 - f - 1e-10:
                        h[i] -= f
                        h[j] += f
                        c = _c5(h, dx)
                        if c < best - 1e-13:
                            best = c
                        else:
                            h[i] += f
                            h[j] -= f
    return h, best
