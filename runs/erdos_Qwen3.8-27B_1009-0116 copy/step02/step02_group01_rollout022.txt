    """Project x onto {0<=h<=1, sum(h)=c}."""
    # h_i = clip(x_i - mu, 0, 1), find mu such that sum = c
    # sum(clip(x_i - mu, 0, 1)) is monotone decreasing in mu
    # When mu -> -inf: sum -> n
    # When mu -> +inf: sum -> 0
    
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mu = (lo + hi) / 2
        h = np.clip(x - mu, 0, 1)
        s = h.sum()
        if s > c:
            lo = mu
        else:
            hi = mu
    h = np.clip(x - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = c - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = (h > 0) & (h < 1)
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
            else:
                # All at bounds, distribute to smallest/highest

def repair(h, n):

        h *= target / s

            order = np.argsort(h) if diff > 0 else np.argsort(-h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
