                d_max = min(1.0 - hi, hj)
                if d_max - d_min < 1e-12:

                gd = grad[i] - grad[j]
                d_best, new_val = _max_of_lines(corr, gd, d_min, d_max, n_coarse, n_fine, fine_w)
                if new_val < cur_max - 1e-14:
                    h[i] += d_best
                    h[j] -= d_best
                    corr = corr + d_best * gd
                    # Update grad (approximate - recompute would be expensive)
                    grad[i] = grad[i]  # not updated exactly, but ok for few rounds
                    grad[j] = grad[j]
                    cur_max = new_val
                    improved = True
        # Recompute grads each round for accuracy
        grad = _compute_grads(h, n)
        if not improved:
            break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _compute_grads(h, n):
    lags = np.arange(-(n - 1), n)
    grad = np.zeros((n, 2 * n - 1))
    for p in range(n):
        for mi, k in enumerate(lags):
            i1 = p + k
            if 0 <= i1 < n:
                grad[p, mi] += (1.0 - h[i1])
            i2 = p - k
            if 0 <= i2 < n:
                grad[p, mi] -= h[i2]
    return grad

def _opt_d(corr, gd, lo, hi, nc, nf, fw):
    ds = np.linspace(lo, hi, nc)
    vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
    bi = np.argmin(vals)
    db = ds[bi]
    if nf > 2 and fw > 0:
        flo = max(lo, db - fw)
        fhi = min(hi, db + fw)
        ds2 = np.linspace(flo, fhi, nf)
        vals2 = np.max(corr[None, :] + ds2[:, None] * gd[None, :], axis=1)
        bi2 = np.argmin(vals2)
        db = ds2[bi2]
        return db, float(vals2[bi2])
    return db, float(vals[bi])

def _pair_sweep(h, n, dx, max_rounds, nc, nf, fw):
    corr = _corr(h)
    for r in range(max_rounds):
        grad = _compute_grads(h, n)
        improved = False
        cur_max = np.max(corr) * dx
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                d_lo = max(-hi, hj - 1.0)
                d_hi = min(1.0 - hi, hj)
                if d_hi - d_lo < 1e-12:
                    continue
                gd = grad[i] - grad[j]
                db, nv = _opt_d(corr, gd, d_lo, d_hi, nc, nf, fw)
                if nv < cur_max - 1e-15:
                    h[i] += db
                    h[j] -= db
                    corr = corr + db * gd
                    cur_max = nv
                    improved = True
        if not improved:
            break
    return h

def _slsqp(h, n, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj_f(x):
        return _smooth_max(_corr(x), alpha)
    def con_f(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': con_f}]
    try:
        res = minimize(obj_f, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _topk_obj(x, k):
    c = _corr(x)
    idx = np.argpartition(c, -k)[-k:]
    return float(np.mean(c[idx]))

def _slsqp_topk(h, n, k, maxiter):
