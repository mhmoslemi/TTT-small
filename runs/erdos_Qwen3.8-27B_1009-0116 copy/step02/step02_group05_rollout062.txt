            else:
                v = min(-diff, h[k]); h[k] -= v; diff += v
            if abs(diff) < 1e-10:

    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(a * (c - m), -500, 500)))) / a

    return _smax(_corr(x), a) * dx

def _obj_pen(x, a, dx, pen, tgt):
    return _smax(_corr(x), a) * dx + pen * (np.sum(x) - tgt) ** 2

def _con_eq(x, tgt):
    return np.sum(x) - tgt

def _slsqp(h, n, dx, a, mi=500, ft=1e-15):
    tgt = 0.5 * n
    try:
        r = minimize(_obj_sm, h, args=(a, dx), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con_eq, 'args': (tgt,)}],
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return h.copy()

def _lbfgs(h, n, dx, a, pen, mi=2000):
    tgt = 0.5 * n
    try:
        r = fmin_l_bfgs_b(_obj_pen, h, args=(a, dx, pen, tgt),
                          disp=0, pgtol=1e-16, gtol=1e-14,
                          maxiter=mi, iprint=0, factr=1e14)
        return _repair(r, n)
    except Exception:
        return h.copy()

def _eval_sh(h, i, j, d, dx):
    oi, oj = h[i], h[j]
    h[i] = oi + d
    h[j] = oj - d
    c = _c5(h, dx)
    h[i] = oi
    h[j] = oj
    return c

def _gs_pair(h, i, j, dx, nits=12):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return 0.0
    gr = 0.6180339887498949
    a, b = lo, hi
    c1 = b - gr * (b - a)
    d1 = a + gr * (b - a)
    fc = _eval_sh(h, i, j, c1, dx)
    fd = _eval_sh(h, i, j, d1, dx)
    for _ in range(nits):
        if fc < fd:
            b = d1; d1 = c1; fd = fc
            c1 = b - gr * (b - a)
            fc = _eval_sh(h, i, j, c1, dx)
        else:
            a = c1; c1 = d1; fc = fd
            d1 = a + gr * (b - a)
            fd = _eval_sh(h, i, j, d1, dx)
    return 0.5 * (a + b)

        return h
    h *= tgt / s
    np.clip(h, 0.0, 1.0, out=h)
    diff = tgt - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for k in order:
            if diff > 0:
                v = min(diff, 1.0 - h[k]); h[k] += v; diff -= v
            else:
                v = min(-diff, h[k]); h[k] -= v; diff += v
            if abs(diff) < 1e-10:
                break
    return h

def _smax(c, a):
    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(a * (c - m), -500, 500)))) / a

def _obj_sm(x, a, dx):
    return _smax(_corr(x), a) * dx

def _obj_pen(x, a, dx, pen, tgt):
    return _smax(_corr(x), a) * dx + pen * (np.sum(x) - tgt) ** 2

def _con_eq(x, tgt):
    return np.sum(x) - tgt

def _slsqp(h, n, dx, a, mi=500, ft=1e-15):
    tgt = 0.5 * n
    try:
        r = minimize(_obj_sm, h, args=(a, dx), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con_eq, 'args': (tgt,)}],
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return h.copy()

def _lbfgs(h, n, dx, a, pen, mi=2000):
    tgt = 0.5 * n
    try:
        r = fmin_l_bfgs_b(_obj_pen, h, args=(a, dx, pen, tgt),
                          disp=0, pgtol=1e-16, gtol=1e-14,
                          maxiter=mi, iprint=0, factr=1e14)
        return _repair(r, n)
    except Exception:
        return h.copy()

def _eval_sh(h, i, j, d, dx):
    oi, oj = h[i], h[j]
    h[i] = oi + d
    h[j] = oj - d
    c = _c5(h, dx)
    h[i] = oi
    h[j] = oj
    return c

def _gs_pair(h, i, j, dx, nits=12):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return 0.0
    gr = 0.6180339887498949
    a, b = lo, hi
    c1 = b - gr * (b - a)
    d1 = a + gr * (b - a)
    fc = _eval_sh(h, i, j, c1, dx)
    fd = _eval_sh(h, i, j, d1, dx)
    for _ in range(nits):
        if fc < fd:
            b = d1; d1 = c1; fd = fc
            c1 = b - gr * (b - a)
            fc = _eval_sh(h, i, j, c1, dx)
        else:
            a = c1; c1 = d1; fc = fd
            d1 = a + gr * (b - a)
            fd = _eval_sh(h, i, j, d1, dx)
    return 0.5 * (a + b)

def _sweep(h, n, dx, dl, gi=12):
    best = h.copy()
    bc5 = _c5(best, dx)
    chg = True
    while chg and time.time() < dl:
        chg = False
        for i in range(n):
            for j in range(i + 1, n):
                d = _gs_pair(best, i, j, dx, gi)
                if abs(d) > 1e-10:
                    ni = best[i] + d
                    nj = best[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        best[i] = ni
                        best[j] = nj
                        nc = _c5(best, dx)
                        if nc < bc5 - 1e-14:
                            bc5 = nc
                            chg = True
                        else:
                            best[i] = ni - d
                            best[j] = nj + d
    return best, bc5

def _anneal(h, n, dx, rng, dl, T0=0.02):
    best = h.copy()
    bc5 = _c5(best, dx)
    cur = h.copy()
    cc5 = bc5
    T = T0
    while time.time() < dl:
        T = max(T * 0.9995, 0.0003)
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        d = (rng.random() * 2.0 - 1.0) * 0.1
        ni = cur[i] + d
        nj = cur[j] - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        oi, oj = cur[i], cur[j]
        cur[i] = ni
        cur[j] = nj
        nc = _c5(cur, dx)
        if nc < cc5 or rng.random() < np.exp(-(nc - cc5) / T):
            cc5 = nc
            if nc < bc5:
                bc5 = nc
                best[:] = cur
        else:
            cur[i] = oi
            cur[j] = oj
    return best, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    bh = h0.copy()
    bc5 = _c5(bh, dx)

    # Phase 1: SLSQP from initial with increasing smoothness
    for a in [50, 150, 400]:
        if time.time() > dl - 550:
            break
        hc = _slsqp(h0, n, dx, a, 600, 1e-15)
        c5 = _c5(hc, dx)
        if c5 < bc5:
            bc5 = c5
            bh = hc.copy()

    # Phase 2: L-BFGS with penalty
    if time.time() < dl - 450:
        for p in [50, 500, 2000]:
            if time.time() > dl - 450:
                break
            hc = _lbfgs(bh, n, dx, 200, p, 1500)
            c5 = _c5(hc, dx)
            if c5 < bc5:
                bc5 = c5
                bh = hc.copy()

    # Phase 3: Simulated annealing to explore landscape
    if time.time() < dl - 300:
        ac, ac5 = _anneal(bh, n, dx, rng, dl - 300, 0.02)
        if ac5 < bc5:
            bc5 = ac5
            bh = ac

    # Phase 4: Golden-section pair sweep (key optimization)
    if time.time() < dl - 150:
        bh, bc5 = _sweep(bh, n, dx, dl - 150, 12)

    # Phase 5: Multi-start perturbations + sweep
    sigs = [0.3, 0.5, 0.2, 0.4, 0.6, 0.15, 0.35, 0.7, 0.25, 0.1]
    it = 0
    while time.time() < dl - 60:
        s = sigs[it % len(sigs)]
        hp = _repair(bh + s * rng.randn(n), n)
        if it < 5:
            hc = _slsqp(hp, n, dx, 300, 300, 1e-14)
        else:
            hc = hp
        hl, cl = _sweep(hc, n, dx, dl - 60, 10)
        if cl < bc5:
            bc5 = cl
            bh = hl
        it += 1

    # Phase 6: Final SLSQP polish
    if time.time() < dl - 15:
        for a in [800, 2000]:
            if time.time() > dl - 15:
                break
            hc = _slsqp(bh, n, dx, a, 600, 1e-16)
            c5 = _c5(hc, dx)
            if c5 < bc5:
                bc5 = c5
                bh = hc.copy()

    # Phase 7: Final fine sweep
    if time.time() < dl - 5:
        bh, bc5 = _sweep(bh, n, dx, dl - 2, 10)

    bh = _repair(bh, n)
    bc5 = _c5(bh, dx)
    return (bh, bc5, n)
