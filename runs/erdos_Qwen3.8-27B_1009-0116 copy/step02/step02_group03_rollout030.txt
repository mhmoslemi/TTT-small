                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _fine_pair(h, n, dx, n_iters, rng, budget_end, min_step=1e-4, max_step=0.05):
    """Very fine pair-move search."""
    h = h.copy()
    best = _c5(h, dx)
    for _ in range(n_iters):
        if time.time() > budget_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        step = rng.uniform(min_step, max_step)
        for sign in [1, -1]:
            d = sign * step
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < best - 1e-14:
                best = c
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, best

def _targeted_search(h, n, dx, n_iters, rng, budget_end):
    """Search targeting the shifts that give max correlation."""
    h = h.copy()
    best = _c5(h, dx)
    for _ in range(n_iters):

        c = _corr(h)

        top_k = np.argsort(c)[::-1][:5]

        # k_idx is index in full correlation
        # Determine the shift
        shift = k_idx - (n - 1)

        # The correlation at shift s involves positions i and i+s
        # Try swapping between a position with h=1 and one with h=0
        # near the problematic region
        if shift >= 0:
            # Positive shift: h[i+shift]*(1-h[i])
            # Reduce by making h[i+shift] smaller or h[i] larger
            lo = 0
            hi = n - shift
        else:
            lo = -shift
            hi = n
        if hi - lo < 2:
            continue
        # Pick pairs
        for attempt in range(3):
            a = lo + rng.randint(hi - lo)
            b = lo + rng.randint(hi - lo)
            if a == b:
                continue
            ia = a if shift >= 0 else a - shift  # position
            ib = b if shift >= 0 else b - shift
            if ia < 0 or ia >= n or ib < 0 or ib >= n:
                continue
            ia, ib = ia % n, ib % n  # wrap? No, don't wrap
            if ia >= n or ib >= n:
                continue
            step = rng.uniform(0.001, 0.02)
            d = rng.choice([-step, step])
            ni, nj = h[ia] + d, h[ib] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:

            h[ia], h[ib] = ni, nj
            c_new = _c5(h, dx)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-15:
                break
            h += room * diff / rs
        else:
            av = h.copy()
            as_ = av.sum()
            if as_ < 1e-15:
                break
            h += av * diff / as_
        h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _slsqp_smooth(h, n, alpha, maxiter):
    target = n / 2.0
    def obj(x):
        c = _corr(x)
        m = np.max(c)
        s = alpha * (c - m)
        s = np.clip(s, -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha + 1e6 * (np.sum(x) - target) ** 2
    return _slsqp_run(h, n, obj, maxiter)

def _slsqp_true(h, n, maxiter):
    target = n / 2.0
    def obj(x):
        c = _corr(x)
        m = np.max(c)
        return m + 1e6 * (np.sum(x) - target) ** 2
    return _slsqp_run(h, n, obj, maxiter)

def _slsqp_run(h, n, obj, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lbfgs_smooth(h, n, alpha, maxiter):
    target = n / 2.0
    def obj(x):
        c = _corr(x)
        m = np.max(c)
        s = alpha * (c - m)
        s = np.clip(s, -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha + 1e4 * (np.sum(x) - target) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pair_search(h, n, dx, n_iters, rng, budget_end, lo=1e-4, hi=0.02):
    h = h.copy()
    best = _c5(h, dx)
    counts = [0] * 8
    for _ in range(n_iters):
        if time.time() > budget_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        step = rng.uniform(lo, hi)
        for sgn in (1, -1):
            d = sgn * step
            ni = h[i] + d
            nj = h[j] - d

                continue
            h[i] = ni
            h[j] = nj
            c = _c5(h, dx)
