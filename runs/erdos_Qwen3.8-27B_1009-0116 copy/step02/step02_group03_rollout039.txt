import time

    return float(np.max(corr) * dx)

def repair_cont(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > 1e-15:
            h *= target / s
        h = np.clip(h, 0.0, 1.0)

def _repair_exact(h, n):
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for idx in order:

            break
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a

    s = alpha * (corr - m)
    s = np.clip(s, -400, 400)
    val = m + np.log(np.sum(np.exp(s))) / alpha

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > 1e-15:
            h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

        h = np.zeros(n)
        for i in range(0, n, 2*blk):
            h[i:i+blk] = 1.0
        h += 0.2 * rng.randn(n)

    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

    target = n * 0.5
    def obj(x):
        return _smooth(x, alpha)
    def con(x):
        return x.sum() - target
    def jcn(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=Bounds(np.zeros(n), np.ones(n)),
                       constraints=[{'type':'eq','fun':con,'jac':jcn}],
                       options={'maxiter':maxit,'ftol':1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _bin_sa(h, n, dx, ns, rng, tend, t0=0.004, t1=5e-5):
    h = h.copy()
    bc = _c5(h, dx)
    bh = h.copy()
    ol = list(np.where(h==1)[0])
    zl = list(np.where(h==0)[0])
    for it in range(ns):
        if time.time() > tend:
            break
        f = it / max(ns, 1)
        tp = t0 * (t1/t0)**f
        ii = rng.randint(len(ol))
        jj = rng.randint(len(zl))
        i, j = ol[ii], zl[jj]
        h[i], h[j] = 0.0, 1.0
        c5 = _c5(h, dx)
        if c5 < bc:
            bc = c5
            bh = h.copy()
            ol[ii], zl[jj] = j, i
        elif rng.random() < np.exp(-(c5-bc)/max(tp,1e-15)):
            ol[ii], zl[jj] = j, i
        else:
            h[i], h[j] = 1.0, 0.0
    return bh, bc

def _coord(h, n, dx, npass, rng, tend):
    h = h.copy()
    best = _c5(h, dx)
    dts = [0.001, 0.003, 0.007, 0.015, 0.03]
    for _p in range(npass):
        if time.time() > tend:

        imp = False

                    imp = True
                    break
                h[i], h[j] = hi, hj
                h = _repair(h, n)
        if not imp:

So the correlation includes ALL 2n-1 values. For the center value (lag 0 in circular terms), corr[n-1] = sum h[i]*(1-h[i]) which is 0 for binary but positive for continuous.

For continuous h, the center value is sum h[i]*(1-h[i]). To minimize this, we want h[i] to be as close to 0 or 1 as possible (binary-like). This supports the binary approach.

The maximum is likely at a non-center lag for binary, and at a nearby lag for continuous.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_g_a = 200.0
_g_t = 35.0

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tg = n * 0.5
    for _ in range(300):
        s = h.sum()
        if abs(s - tg) < 1e-14:
            break
        if s > 1e-15:
            h *= tg / s
        h = np.clip(h, 0.0, 1.0)
    d = tg - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for ix in o:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[ix]); h[ix] += a; d -= a
            else:
                a = min(-d, h[ix]); h[ix] -= a; d += a
    return h

def _rbin(h, n, rng):
    h = np.round(h).astype(float)
    h = np.clip(h, 0.0, 1.0)
    t = n // 2
    s = int(h.sum())
    if s > t:
        ix = rng.choice(np.where(h == 1)[0], s - t, replace=False)
        h[ix] = 0.0
    elif s < t:
        ix = rng.choice(np.where(h == 0)[0], t - s, replace=False)
        h[ix] = 1.0
    return h

def _start(n, seed, kind):
    rng = np.random.RandomState(seed)
    if kind == 0:
        h = rng.uniform(0, 1, n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.2 * rng.randn(n)
    elif kind == 2:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
        h += 0.15 * rng.randn(n)
    elif kind == 3:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * np.arange(n) / n) + 0.1 * rng.randn(n)
    elif kind == 4:
        h = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
        h += 0.2 * rng.randn(n)
    elif kind == 5:
        b = max(1, n // 8)
        h = np.zeros(n)
        for i in range(0, n, 2 * b):
            h[i:i + b] = 1.0
        h += 0.25 * rng.randn(n)
    else:
        h = 0.5 + 0.3 * rng.randn(n)
    return _repair(h, n)

def _smobj(h, a):
    c = _corr(h)
    m = np.max(c)
    s = np.clip(a * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / a

def _oobj(x):
    return _smobj(x, _g_a)

def _ocon(x):
    return x.sum() - _g_t

def _oconi(x):
    return np.ones(len(x))

def _osls(h0, n, a, mi):
    global _g_a, _g_t
    _g_a = a
    _g_t = n * 0.5
    try:
        r = minimize(_oobj, h0, method='SLSQP',
                     bounds=Bounds(np.zeros(n), np.ones(n)),
                     constraints=[{'type': 'eq', 'fun': _ocon, 'jac': _oconi}],
                     options={'maxiter': mi, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h0

def _bsa(h0, n, dx, ns, rng, te, ti=0.005, tf=3e-5):
    h = h0.copy()
    best = _c5(h, dx)
    bh = h.copy()
    ol = list(np.where(h == 1)[0])
    zl = list(np.where(h == 0)[0])
    for it in range(ns):
        if time.time() > te:
            break
        f = it / max(ns, 1)
        tp = ti * (tf / ti) ** f
        ii, jj = rng.randint(len(ol)), rng.randint(len(zl))
        i, j = ol[ii], zl[jj]
        h[i], h[j] = 0.0, 1.0
        c5 = _c5(h, dx)
        if c5 < best:
            best = c5
            bh = h.copy()
            ol[ii], zl[jj] = j, i
        elif rng.random() < np.exp(-(c5 - best) / max(tp, 1e-15)):
            ol[ii], zl[jj] = j, i
        else:
            h[i], h[j] = 1.0, 0.0
    return bh, best

def _cd(h0, n, dx, npass, rng, te):
    h = h0.copy()
    best = _c5(h, dx)
    dts = [0.0005, 0.002, 0.005, 0.015, 0.04]
    for _p in range(npass):
        if time.time() > te:
            break
        imp = False
        for _ in range(200):
            if time.time() > te:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in dts:
                if hi + d > 1.0 + 1e-10 or hj - d < -1e-10:
                    continue
                h[i], h[j] = hi + d, hj - d
                h = _repair(h, n)
                c5 = _c5(h, dx)
                if c5 < best:
                    best = c5
                    imp = True
                    break
                h[i], h[j] = hi, hj
                h = _repair(h, n)
        if not imp:
            break
    return h, best

    rng = np.random.RandomState(seed)
    bh, bc5, bn = None, np.inf, 70

        dx = 2.0 / n

        h0 = ih.copy() if (ih is not None and n == in_) else None
        if h0 is None:

        hb = _rbin(h0, n, rng)
        hb, cb = _bsa(hb, n, dx, 500000, rng, te)
        if cb < bc5:

        hc = hb + 0.01 * rng.randn(n)

            hn = _osls(hc, n, a, 200)
            cn = _c5(hn, dx)
            if cn < _c5(hc, dx):

            t_bsa2 = min(time.time() + 120, t0 + budget_s - 5)

                hn2 = _osls(hc2, n, 20000, 150)

                if cn2 < bc5:
