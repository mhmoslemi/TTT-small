def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_lag(h, m_idx):
    n = len(h)
    lag = m_idx - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        k = n - lag
        g[lag:lag + k] += (1.0 - h[:k])
        g[:k] -= h[lag:lag + k]
    else:
        la = -lag
        k = n - la
        g[:k] += (1.0 - h[la:la + k])
        g[la:la + k] -= h[:k]
    return g

    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha
