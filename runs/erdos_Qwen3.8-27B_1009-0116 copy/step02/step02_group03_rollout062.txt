    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)

def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    for _ in range(200):
        h = h * (target / max(s, 1e-15))
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break

                diff -= add

            best_c5, best_h = c5_7, h7.copy()

    best_h = _fix(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
