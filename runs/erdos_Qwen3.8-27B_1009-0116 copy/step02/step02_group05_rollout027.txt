def _smgrad(h, alpha, dx):

    m = np.max(c)

    e = np.exp(s)
    w = e / e.sum()

    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-14:
            continue
        shift = ki - (n - 1)
        if shift == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif shift > 0:
            # Term 1: grad[i] += (1-h[i-shift]) for i in [shift, n-1]
            grad[shift:] += wk * (1.0 - h[:-shift])
            # Term 2: grad[i] -= h[i+shift] for i in [0, n-1-shift]
            grad[:n - shift] -= wk * h[shift:]

            # Term 2: grad[i] -= h[i-s2] for i in [s2, n-1]
            grad[s2:] -= wk * h[:n - s2]
    grad *= dx
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smobj(h, alpha, dx):
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha) * dx

def _smgrad(h, alpha, dx):
    n = len(h)
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-14:
            continue
        shift = ki - (n - 1)
        if shift == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif shift > 0:
            grad[shift:] += wk * (1.0 - h[:-shift])
            grad[:n - shift] -= wk * h[shift:]
        else:
            s2 = -shift
            grad[:n - s2] += wk * (1.0 - h[s2:])
            grad[s2:] -= wk * h[:n - s2]
    grad *= dx
    return grad

def _obj_func(x, alpha, dx):
    return _smobj(x, alpha, dx)

def _grad_func(x, alpha, dx):
    return _smgrad(x, alpha, dx)

def _con_func(x, target):
    return np.sum(x) - target

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    try:
        res = minimize(_obj_func, h, args=(alpha, dx), jac=_grad_func,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _con_func, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _grad_corr_at(h, ki, n):
    shift = ki - (n - 1)
    grad = np.zeros(n)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        grad[shift:] += (1.0 - h[:-shift])
        grad[:n - shift] -= h[shift:]
    else:
        s2 = -shift
        grad[:n - s2] += (1.0 - h[s2:])
        grad[s2:] -= h[:n - s2]
    return grad
