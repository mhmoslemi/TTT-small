import time

def _c5(h, dx):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _pair_descent(h, n, dx, rng, max_passes, deadline):
    """Systematic pair descent with 1D evaluation."""
    best = h.copy()
    best_c5 = _c5(best, dx)
    target = n / 2.0
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Random permutation of pairs
        pairs = []
        for i in range(n):
            for j in range(i+1, n):
                pairs.append((i, j))
        rng.shuffle(pairs)
        
        n_evals = 0
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            # Feasible range for δ
            delta_min = max(-best[i], -(1.0 - best[j]))  # wait
            # best[i] + δ >= 0 => δ >= -best[i]
            # best[i] + δ <= 1 => δ <= 1 - best[i]
            # best[j] - δ >= 0 => δ <= best[j]
            # best[j] - δ <= 1 => δ >= best[j] - 1
            d_lo = max(-best[i], best[j] - 1.0)
            d_hi = min(1.0 - best[i], best[j])
            if d_hi - d_lo < 1e-12:

            # Sample δ values
            n_samples = 16
            d_cands = np.linspace(d_lo, d_hi, n_samples)
            # Add 0

            best_d = 0.0
            best_val = best_c5

                best[j] -= d
                v = _c5(best, dx)
                if v < best_val:

                    best_d = d
                best[i] -= d
                best[j] += d

    gr = (np.sqrt(5) - 1) / 2  # 0.618...

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_obj(x, alpha, dx):
    return _smooth_max(np.correlate(x, 1.0 - x, mode='full'), alpha) * dx

def _sum_con(x, target):
    return np.sum(x) - target

def _do_slsqp(h, n, dx, alpha, maxiter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'args': (target,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()
