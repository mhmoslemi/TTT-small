    # Phase 1: SLSQP from initial with increasing alpha
    for a in [40, 80, 160, 320, 600, 1000]:
        if not _rem(350):
            break
        hc = _slsqp(h_init, n, dx, a, 300, 1e-14)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 2: Nelder-Mead refinement
    if _rem(300):
        hc = _nm_optimize(best_h, n, dx)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 3: Differential evolution (global)
    if _rem(350):
        try:
            bounds = [(0.0, 1.0)] * n
            de_res = differential_evolution(
                _de_obj, bounds=bounds, args=(n, dx),
                seed=seed, maxiter=60, popsize=14,
                tol=1e-5, mut=(0.4, 0.9), rebinfra=0.5,
                polish=False, init='sobol')
            de_h = _repair(de_res.x, n)
            de_c5 = _c5(de_h, dx)
            if de_c5 < best_c5:
                best_c5, best_h = de_c5, de_h.copy()
                # Refine DE result
                if _rem(250):
                    hc = _slsqp(de_h, n, dx, 300, 300, 1e-14)
                    c5 = _c5(hc, dx)
                    if c5 < best_c5:
                        best_c5, best_h = c5, hc.copy()
        except Exception:
            pass

    # Phase 4: Pair local search
    if _rem(200):
        deadline = t0 + budget_s - 180
        best_h, best_c5 = _pair_local(best_h, n, dx, rng, deadline)

    # Phase 5: SLSQP polish
    if _rem(150):
        for a in [500, 1000, 2000]:
            if not _rem(130):
                break
            hc = _slsqp(best_h, n, dx, a, 500, 1e-15)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()

    # Phase 6: Simulated annealing
    if _rem(120):
        deadline = t0 + budget_s - 100
        best_h, best_c5 = _sa_local(best_h, n, dx, rng, deadline,
                                      t_start=0.3, t_end=0.005, max_iter=6000)

    # Phase 7: Restart + optimize
    sigmas = [0.12, 0.2, 0.3, 0.15, 0.25, 0.35, 0.5, 0.08, 0.18, 0.4]
    it = 0
    while _rem(80):
        sigma = sigmas[it % len(sigmas)]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        hc = _slsqp(hp, n, dx, 250, 200, 1e-13)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
        it += 1

    # Phase 8: Final pair local search
    if _rem(30):
        deadline = t0 + budget_s - 10
        best_h, best_c5 = _pair_local(best_h, n, dx, rng, deadline)

    # Phase 9: Final Nelder-Mead
    if _rem(15):
        hc = _nm_optimize(best_h, n, dx)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 10: Final fine local search
    if _rem(5):
        deadline = t0 + budget_s - 1
        best_h, best_c5 = _pair_local(best_h, n, dx, rng, deadline)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
