import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort((1.0 - h) if diff > 0 else h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _sym_unfold(s, n):
    half = n // 2
    h = np.zeros(n)
    h[:half] = s
    h[half:] = s[::-1]
    return h

def _sym_fold(h, n):
    return h[:n // 2].copy()

def _enforce_symmetry(h, n):
    h2 = (h + h[::-1]) / 2.0
    return h2

def _smooth_max(c, alpha):
    m = float(np.max(c))
    sc = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(sc)))) / alpha

def _obj_smax_full(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _obj_smax_sym(s, alpha, n):
    h = _sym_unfold(s, n)
    return _smooth_max(_corr(h), alpha)

def _eq_con_full(x, target):
    return np.sum(x) - target

def _eq_con_sym(s, target_sym, n):
    h = _sym_unfold(s, n)
    return np.sum(h) - n / 2.0

def _slsqp_full(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_full, 'args': (n / 2.0,)}]
    try:
        res = minimize(_obj_smax_full, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _slsqp_sym(s, n, alpha, maxiter):
    half = n // 2
    bounds = [(0.0, 1.0)] * half
    cons = [{'type': 'eq', 'fun': _eq_con_sym, 'args': (n / 4.0, n)}]
    try:
        res = minimize(_obj_smax_sym, s, args=(alpha, n), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h = _sym_unfold(res.x, n)
        return _repair(h, n)
    except Exception:
        return _sym_unfold(s, n)

def _sweep_pass(h, n, dx, rng, n_coarse, n_fine, fine_frac):
    corr = _corr(h)
    tau = np.arange(-(n - 1), n)
    n_pairs = n * (n - 1) // 2
    pairs = np.array([(i, j) for i in range(n) for j in range(i + 1, n)])
    perm = rng.permutation(n_pairs)
    improved = False
    n_imp = 0
    for idx in perm:
        i, j = pairs[idx]
        hi, hj = h[i], h[j]
        d_min = max(-hi, hj - 1.0)
        d_max = min(1.0 - hi, hj)
        if d_max - d_min < 1e-12:
            continue
        gd = np.zeros(2 * n - 1)
        for p, sgn in ((i, 1.0), (j, -1.0)):
            v1 = (p + tau >= 0) & (p + tau < n)
            v2 = (p - tau >= 0) & (p - tau < n)
            gd[v1] += sgn * (1.0 - h[p + tau[v1]])
            gd[v2] += sgn * (-h[p - tau[v2]])
        ds = np.linspace(d_min, d_max, n_coarse)
        nc = corr[None, :] + ds[:, None] * gd[None, :]
        c5s = np.max(nc, axis=1) * dx
        bi = int(np.argmin(c5s))
        cur_max = float(np.max(corr)) * dx
        if c5s[bi] >= cur_max - 1e-14:
            continue
        d_best = ds[bi]
        best_val = c5s[bi]
        if n_fine > 2 and fine_frac > 0:
            sp = (d_max - d_min) / max(n_coarse - 1, 1)
            win = sp * fine_frac
            lo = max(d_min, d_best - win)
            hi2 = min(d_max, d_best + win)
            if hi2 - lo > 1e-14:
                ds2 = np.linspace(lo, hi2, n_fine)
                nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                c5s2 = np.max(nc2, axis=1) * dx
                bi2 = int(np.argmin(c5s2))
                if c5s2[bi2] < best_val:
                    d_best = ds2[bi2]
                    best_val = c5s2[bi2]
        h[i] = hi + d_best
        h[j] = hj - d_best
        corr = corr + d_best * gd
        improved = True
        n_imp += 1
    return h, corr, improved, n_imp
