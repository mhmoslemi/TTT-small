def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += w[k] * (1.0 - x[i + lag])
                g[i + lag] -= w[k] * x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += w[k] * (1.0 - x[i - L])
                g[i - L] -= w[k] * x[i]
    return g

def _fista(h0, alpha, target, iters, t0, budget_s, deadline=80.0):

    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()

    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        if time.time() - t0 > budget_s - deadline:
            break
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        fn = _sobj(xn, alpha)
        quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
        cnt = 0
        while fn > quad + 1e-10 and cnt < 8:
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()

def _subgrad(h, dx, target, n_iters, t0, budget_s, rng):
    n = len(h)
    bh = h.copy()
    bc = _c5(bh, dx)
    lrs = np.linspace(0.02, 0.002, n_iters)
    for it in range(n_iters):
        if time.time() - t0 > budget_s - 50:
            break
        c = np.correlate(bh, 1.0 - bh, mode='full')
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = np.zeros(n)
        if lag == 0:
            g[:] = 1.0 - 2.0 * bh
        else:
            L = abs(lag)
            if lag > 0:
                fwd = np.arange(n) - lag
                bwd = np.arange(n) + lag
            else:
                fwd = np.arange(n) + L
                bwd = np.arange(n) - L
            fv = (fwd >= 0) & (fwd < n)
            bv = (bwd >= 0) & (bwd < n)
            g[fv] += (1.0 - bh[np.clip(fwd, 0, n - 1)][fv] if fv.sum() > 0 else 0)
            g[fv] = np.where(fv, g + (1.0 - bh[np.clip(fwd, 0, n - 1)]), g)
            g[bv] = np.where(bv, g - bh[np.clip(bwd, 0, n - 1)], g)
        lr = lrs[min(it, len(lrs) - 1)]
        bh = _proj(bh - lr * g, target)
        c_new = _c5(bh, dx)
        if c_new < bc:
            bc = c_new
    return bh, bc

    return bs - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = _proj(np.array(initial_h_values, dtype=float), tgt)
    bh = h0.copy()
    bc = _c5(bh, dx)

    # Phase 1: Aggressive coordinate descent
    if _rem(t0, budget_s) > 60:
        bh, bc2 = _coord_sweep(bh, dx, tgt, 300, rng)
        if bc2 < bc:
            bc = bc2
            bh = bh

    # Phase 2: FISTA with smooth max
    for a in [100, 300, 800, 2000, 5000]:
        if _rem(t0, budget_s) < 80:
            break
        h, c = _fista(bh, a, tgt, 3000, t0, budget_s)
        if c < bc:
            bc = c
            bh = h
        if _rem(t0, budget_s) > 30:
            h2, c2 = _coord_sweep(bh, dx, tgt, 100, rng)
            if c2 < bc:
                bc = c2
                bh = h2

    # Phase 3: More coordinate descent (focused)
    if _rem(t0, budget_s) > 40:
        bh, bc2 = _coord_sweep(bh, dx, tgt, 500, rng)
        if bc2 < bc:
            bc = bc2
            bh = bh

    # Phase 4: Subgradient on worst lag
    if _rem(t0, budget_s) > 30:
        bh, bc2 = _subgrad(bh, dx, tgt, 200, t0, budget_s, rng)
        if bc2 < bc:
            bc = bc2
            bh = bh
        if _rem(t0, budget_s) > 20:
            h2, c2 = _coord_sweep(bh, dx, tgt, 100, rng)
            if c2 < bc:
                bc = c2
                bh = h2

    # Phase 5: Random restarts with structured perturbations
    for s in [0.08, 0.15, 0.05, 0.12, 0.2, 0.1, 0.06, 0.18]:
        if _rem(t0, budget_s) < 60:
            break
        hp = _proj(bh + s * rng.randn(n), tgt)
        h, c = _fista(hp, 500, tgt, 2000, t0, budget_s, deadline=50)
        if _rem(t0, budget_s) < 30:
            break
        h2, c2 = _coord_sweep(h, dx, tgt, 200, rng)
        if c2 < bc:
            bc = c2
            bh = h2

    # Phase 6: Try different n with structured starts
    for nn in [50, 60, 80, 40]:
        if _rem(t0, budget_s) < 120:
            break
        dxn = 2.0 / nn
        tgn = nn / 2.0
        # Smooth periodic start
        xs = np.arange(nn)
        hs = 0.5 + 0.3 * np.sin(2 * np.pi * xs / nn * 2.0) + 0.15 * np.sin(2 * np.pi * xs / nn * 5.0)
        hs = _proj(hs, tgn)
        h, c = _fista(hs, 300, tgn, 1500, t0, budget_s, deadline=80)
        if _rem(t0, budget_s) < 60:
            h2, c2 = _coord_sweep(h, dxn, tgn, 400, rng)
        else:
            h2, c2 = _coord_sweep(h, dxn, tgn, 80, rng)
        if c2 < bc:
            bc = c2
            bh = _proj(h2, tgn)

    # Phase 7: Binary-like constructions
    if _rem(t0, budget_s) > 80:
        for trial in range(3):
            if _rem(t0, budget_s) < 40:
                break
            # Start from random binary-like
            hb = (rng.rand(n) > 0.5).astype(float)
            hb = _proj(hb, tgt)
            # Smooth slightly
            hb = _proj(0.7 * hb + 0.3 * 0.5, tgt)
            h, c = _fista(hb, 200, tgt, 1500, t0, budget_s, deadline=50)
            if _rem(t0, budget_s) < 30:
                break
            h2, c2 = _coord_sweep(h, dx, tgt, 300, rng)
            if c2 < bc:
                bc = c2
                bh = h2

    # Phase 8: Final intensive coordinate descent
    if _rem(t0, budget_s) > 20:
        nfinal = len(bh)
        dxf = 2.0 / nfinal
        tgff = nfinal / 2.0
        bh, bc2 = _coord_sweep(bh, dxf, tgff, 800, rng)
        if bc2 < bc:
            bc = bc2

    # Final SLSQP polish
    if _rem(t0, budget_s) > 25:
        nf = len(bh)
        dxf = 2.0 / nf
        tgf = nf / 2.0
        bnds = [(0.0, 1.0)] * nf
        try:
            r = minimize(_sobj, bh, args=(3000,), method='L-BFGS-B',
                         bounds=bnds, options={'maxiter': 300, 'ftol': 1e-16})
            hf = _proj(r.x, tgf)
            cf = _c5(hf, dxf)
            if cf < bc:
                bc = cf
                bh = hf
        except Exception:
            pass
        if _rem(t0, budget_s) > 10:
            h2, c2 = _coord_sweep(bh, dxf, tgf, 200, rng)
            if c2 < bc:
                bc = c2
                bh = h2

    bh = _proj(bh, float(len(bh)) / 2.0)
    bc = _c5(bh, 2.0 / len(bh))
    return (bh, bc, len(bh))
