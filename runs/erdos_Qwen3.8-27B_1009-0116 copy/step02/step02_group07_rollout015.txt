def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

        obj, grad = _smooth_max_grad(h, alpha)
        v = 0.85 * v - lr * grad
        h_new = _repair(h + v, n)
        actual = _c5(h_new, dx)
        if actual < best_obj:
            best_obj = actual
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.5)
        else:
            v *= 0.5
            lr = max(lr * 0.85, 0.005)
            h_trial = _repair(h - 0.3 * lr * grad, n)
            at = _c5(h_trial, dx)
            if at < best_obj:
                best_obj = at
                best_h = h_trial.copy()
                h = h_trial
    return best_h, best_obj

def _cd_pass(h, n, dx, n_grid=21, max_pairs=None):
    hi_c5 = _c5(h, dx)
    improved = 0
    pairs_done = 0
    for i in range(n):
        if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
            continue
        for j in range(i + 1, n):
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            pairs_done += 1
            if max_pairs and pairs_done > max_pairs:
                break
            hsum = h[i] + h[j]
            lo = max(0.0, hsum - 1.0)
            hi = min(1.0, hsum)
            if hi - lo < 1e-10:
                continue
            orig_i, orig_j = h[i], h[j]
            best_t, best_c = orig_i, hi_c5
            span = hi - lo
            for t in np.linspace(lo + 1e-9, hi - 1e-9, n_grid):
                h[i] = t
                h[j] = hsum - t
                cc = np.max(_corr(h)) * dx
                if cc < best_c:
                    best_c = cc
                    best_t = t
            if best_c < hi_c5 - 1e-13:
                h[i] = best_t
                h[j] = hsum - best_t
                hi_c5 = best_c
                improved += 1
            else:
                h[i], h[j] = orig_i, orig_j
    return h, hi_c5, improved

def _cd_loop(h, n, dx, t_limit, t_start, n_grid=25, max_passes=3000):
    for _ in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        h, c5, imp = _cd_pass(h, n, dx, n_grid)
        if imp == 0:
            break
    return h, _c5(h, dx)

def _slsqp_quick(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def obj_fn(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha

    def obj_grad(x):
        _, g = _smooth_max_grad(x, alpha)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _interp_parent(n_target, n_src):
    parent = np.array(initial_h_values, dtype=float)
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_tgt = (np.arange(n_target) + 0.5) / n_target * 2.0
    return np.interp(x_tgt, x_src, parent)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    n_values = [70, 50, 60, 80, 90, 55, 65]

    for n in n_values:
        if elapsed() > budget_s - 50:
            break
        dx = 2.0 / n

        if n == 70:
            h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
        else:
            h = _repair(_interp_parent(n, 70), n)

        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h, best_n = c, h.copy(), n

        # Gradient phase 1: low-to-mid alpha
        if elapsed() < budget_s - 100:
            alphas = [15, 40, 100, 250, 600]
            h, c = _proj_grad_opt(h, n, alphas, budget_s - 110, t0, max_iter=60000)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n

        # SLSQP polish
        if elapsed() < budget_s - 60:
            h = _slsqp_quick(h, n, 400, 400)
            c = _c5(h, 2.0 / n)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n

        # CD pass
        if elapsed() < budget_s - 30:
            h, c = _cd_loop(h, n, dx, budget_s - 35, t0, n_grid=21, max_passes=1500)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n

        # Gradient phase 2: high alpha
        if elapsed() < budget_s - 15:
            alphas = [600, 1200, 2500]
            h, c = _proj_grad_opt(h, n, alphas, budget_s - 20, t0, max_iter=40000)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n

    # Final CD on best
    if elapsed() < budget_s - 10:
        h, c = _cd_loop(best_h, best_n, 2.0 / best_n, budget_s - 12, t0,
                        n_grid=29, max_passes=2000)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Random restarts
    sigmas = [0.06, 0.1, 0.04, 0.15, 0.08]
    for si in range(len(sigmas)):
        if elapsed() > budget_s - 8:
            break
        n = best_n
        dx = 2.0 / n
        h_p = _repair(best_h + sigmas[si] * rng.randn(n), n)
        alphas = [200, 600, 1500]
        h, c = _proj_grad_opt(h_p, n, alphas, min(budget_s - 5, elapsed() + 40), t0)
        if c < best_c5:
            best_c5, best_h, best_n = c, h.copy(), n
            h2, c2 = _cd_loop(h, best_n, dx, min(budget_s - 5, elapsed() + 20), t0,
                              n_grid=25, max_passes=500)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
