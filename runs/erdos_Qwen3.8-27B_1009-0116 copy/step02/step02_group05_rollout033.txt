def _slsqp(h, n, dx, alpha, maxiter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def confun(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': confun}]
    try:
        res = minimize(_smooth_max_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _sa_optimize(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    cur = h.copy()
    cur_c5 = best_c5
    step = 0.04
    base_temp = 0.003
    stall = 0
    while time.time() < deadline:
        T = base_temp * max(0.005, 1.0 - stall / 800.0)
        r = rng.random()
        if r < 0.30:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            d = rng.uniform(-step, step)
            ni, nj = cur[i] + d, cur[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            trial = cur.copy()
            trial[i], trial[j] = ni, nj
        elif r < 0.50:
            idx = rng.choice(n, 3, replace=False)
            deltas = rng.uniform(-step, step, 3)
            deltas -= deltas.mean()
            trial = cur.copy()
            trial[idx] += deltas
            trial = np.clip(trial, 0.0, 1.0)
            trial = _repair(trial, n)
        elif r < 0.70:
            bl = int(rng.randint(2, min(6, n)))
            st = int(rng.randint(0, max(1, n - bl)))
            idx = np.arange(st, st + bl)
            deltas = rng.uniform(-step, step, bl)
            deltas -= deltas.mean()
            trial = cur.copy()
            trial[idx] += deltas
            trial = np.clip(trial, 0.0, 1.0)
            trial = _repair(trial, n)
        elif r < 0.85:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            trial = cur.copy()
            trial[i], trial[j] = cur[j], cur[i]
        else:
            idx = rng.choice(n, int(rng.randint(2, 5)), replace=False)
            deltas = rng.uniform(-step, step, len(idx))
            deltas -= deltas.mean()
            trial = cur.copy()
            trial[idx] += deltas
            trial = np.clip(trial, 0.0, 1.0)
            trial = _repair(trial, n)
        tc = _c5(trial, dx)
        delta = tc - cur_c5
        if tc < cur_c5 or (T > 1e-10 and rng.random() < np.exp(-max(0.0, delta) / T)):
            cur = trial
            cur_c5 = tc
            if tc < best_c5:
                best_c5 = tc
                best = cur.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 400:
            step *= 0.55
            base_temp *= 0.3
            stall = 0
            if step < 0.0003:
                break
    return best, best_c5

def _crit_hill(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    step = 0.008
    stall = 0
    while time.time() < deadline:
        c = _corr(best)
        k_star = int(np.argmax(c))
        improved = False
        ntries = 200
        for _ in range(ntries):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            d = rng.uniform(-step, step)
            ni, nj = best[i] + d, best[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            trial = best.copy()
            trial[i], trial[j] = ni, nj
            tc = _c5(trial, dx)
            if tc < best_c5 - 1e-12:
                best = trial
                best_c5 = tc
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 60:
                step *= 0.75
            if step < 0.0002:
                break
    return best, best_c5

def _sinusoid_init(n, rng, nharmonics=5, amp=0.3):
    t = np.arange(n) / n
    phases = rng.uniform(0, 2 * np.pi, nharmonics)
    amps = amp * rng.uniform(0.5, 1.0, nharmonics)
    h = np.zeros(n)
    for m in range(1, nharmonics + 1):
        h += amps[m - 1] * np.sin(2 * np.pi * m * t + phases[m - 1])
    h += 0.5
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _c5(best_h, dx)

    def rem(margin=10):
        return time.time() < t0 + budget_s - margin

    # Phase 1: SLSQP from initial
    h_work = best_h.copy()
    for alpha in [40, 100, 250, 600, 1500]:
        if not rem(250):
            break
        h_work = _slsqp(h_work, n, dx, alpha, 300, 1e-14)
        c = _c5(h_work, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_work.copy()

    # Phase 2: Long SA
    if rem(150):
        dl = t0 + budget_s - 150
        best_h, best_c5 = _sa_optimize(best_h, n, dx, rng, dl)

    # Phase 3: Critical lag hill climb
    if rem(80):
        dl = t0 + budget_s - 80
        best_h, best_c5 = _crit_hill(best_h, n, dx, rng, dl)

    # Phase 4: SLSQP polish
    if rem(60):
        for alpha in [800, 2000, 5000]:
            if not rem(55):
                break
            h_work = _slsqp(best_h, n, dx, alpha, 350, 1e-15)
            c = _c5(h_work, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h_work.copy()

    # Phase 5: More SA
    if rem(60):
        dl = t0 + budget_s - 60
        best_h, best_c5 = _sa_optimize(best_h, n, dx, rng, dl)

    # Phase 6: Restart with diverse starts
    it = 0
    sigmas = [0.15, 0.3, 0.5, 0.08, 0.25, 0.4, 0.12, 0.2]
    while rem(30):
        if it % 3 == 2:
            hp = _sinusoid_init(n, np.random.RandomState(seed + 9997 + it), 4, 0.25)
        else:
            hp = _repair(best_h + sigmas[it % len(sigmas)] * rng.randn(n), n)
        hc = _slsqp(hp, n, dx, 400, 250, 1e-14)
        c = _c5(hc, dx)
        if c < best_c5:
            best_c5 = c
            best_h = hc.copy()
        elif c < best_c5 + 0.003 and rem(35):
            dl2 = t0 + budget_s - 35
            hl, cl = _sa_optimize(hc, n, dx, rng, dl2)
            if cl < best_c5:
                best_c5 = cl
                best_h = hl.copy()
        it += 1

    # Phase 7: Final crit hill + fine SA
    if rem(20):
        dl = t0 + budget_s - 12
        best_h, best_c5 = _crit_hill(best_h, n, dx, rng, dl)
    if rem(10):
        dl = t0 + budget_s - 3
        best_h, best_c5 = _sa_optimize(best_h, n, dx, rng, dl)

    # Final polish
    if rem(5):
        h_work = _slsqp(best_h, n, dx, 3000, 500, 1e-16)
        c = _c5(h_work, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_work.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
