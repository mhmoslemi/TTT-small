        else:

                proj[idx] += a
                diff -= a
            else:

                proj[idx] -= a
                diff += a

import numpy as np

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _project(h, target):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - p.sum()
    if abs(d) > 1e-10:
        ord_ = np.argsort(1.0 - p) if d > 0 else np.argsort(p)
        for i in ord_:
            if d > 0:
                a = min(d, 1.0 - p[i]); p[i] += a; d -= a
            else:
                a = min(-d, p[i]); p[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return p

def _subgrad(h, n, top):
    corr = _corr(h)
    act = np.argsort(corr)[-top:]
    g = np.zeros(n)
    for k in act:

        for j in range(n):

            gm = j - s
            if 0 <= gp < n:
                g[j] += 1.0 - h[gp]
            if 0 <= gm < n:
                g[j] -= h[gm]
    g /= len(act)
    return g, act
