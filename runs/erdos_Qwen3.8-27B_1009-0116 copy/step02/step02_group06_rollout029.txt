    if s < 1e-12:
        return np.full(n, 0.5)

                h[idx] -= sub; diff += sub

    g = np.zeros(2*n-1)
    i1 = p - k; m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = p + k; m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]

    s = np.clip(alpha * (c - m), -500, 500)

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * t / s
    h = np.clip(h, 0.0, 1.0)

                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])

                d += a

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12: break
    return h


def _gvec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    i1 = p - k
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = p + k
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g


def _smax(h, alpha, K):
    c = _corr(h)
    if K is not None and 0 < K < len(c):
        c = c[np.argsort(c)[-K:]]
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha


def _eqfun(x, target):
    return np.sum(x) - target


def _slsqp(h, n, alpha, K, maxiter):
    t = n / 2.0
    cons = [{'type': 'eq', 'fun': _eqfun, 'args': (t,)}]
    b = [(0.0, 1.0)] * n
    try:
        r = minimize(_smax, h, args=(alpha, K), method='SLSQP',
                     bounds=b, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return h


def _sweep(h, n, dx, nc, nco, nfi, fw):
