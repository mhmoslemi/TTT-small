def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.sum(np.exp(s)) / alpha

def _corr_grad(k_max, h, n):
    off = k_max - (n - 1)
    grad = np.zeros(n)
    if off >= 0:
        for m in range(n - off):
            grad[m] += (1.0 - h[m + off])
            grad[m + off] += (-h[m])
    else:
        d = -off
        for m in range(d, n):
            grad[m] += (1.0 - h[m - d])
            grad[m - d] += (-h[m])
    return grad

def _momentum_descend(h, n, dx, t_limit, steps=5000):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    v = np.zeros(n)
    base_lr = 0.015
    for s in range(steps):
        if time.time() > t_limit:
            break
        c = _corr(h)
        k_max = np.argmax(c)
        g = _corr_grad(k_max, h, n)
        lr = base_lr / (1.0 + 0.001 * s)
        v = 0.85 * v - lr * g
        h_new = h + v
        h_new = _repair(h_new, n)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            v *= 0.4
    return best_h, best_c5

def _cd_focused(h, n, dx, grid_pts, focus_n, rng):
    hi_c5 = _c5(h, dx)
    c = _corr(h)
    top_lags = np.argsort(c)[-focus_n:]
    fp = set()
    for kk in top_lags:
        off = kk - (n - 1)
        if off >= 0:
            for i in range(n - off):
                fp.add((i, i + off))
        else:
            for i in range(-off, n):
                fp.add((i, i + off))
    focus_list = [(min(a,b), max(a,b)) for a,b in fp]
    other = [(i,j) for i in range(n) for j in range(i+1,n) if (i,j) not in set(focus_list)]
    rng.shuffle(other)
    all_pairs = focus_list + other[:30]
    improved = False
    for (i, j) in all_pairs:
        hs = h[i] + h[j]
        lo = max(0.0, hs - 1.0)
        hi = min(1.0, hs)
        if hi - lo < 1e-10:
            continue
        oi, oj = h[i], h[j]
        bt, bc = oi, hi_c5
        for t in np.linspace(lo + 1e-9, hi - 1e-9, grid_pts):
            h[i], h[j] = t, hs - t
            cv = _c5(h, dx)
            if cv < bc:
                bc, bt = cv, t
        if bc < hi_c5 - 1e-14:
            h[i], h[j] = bt, hs - bt
            hi_c5 = bc
            improved = True
        else:
            h[i], h[j] = oi, oj
    return h, hi_c5, improved

def _slsqp(h, n, bounds, cons, alpha, maxiter, ftol):
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _binarize_snap(h, n, threshold=0.02):
    h2 = h.copy()
    for i in range(n):
        if h2[i] < threshold:
            h2[i] = 0.0
        elif h2[i] > 1.0 - threshold:
            h2[i] = 1.0
    h2 = _repair(h2, n)
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    global _sum_target
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    _sum_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun}]

    def elapsed():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: SLSQP warmup with increasing smoothness
    for a in [60, 150, 400, 900]:
        if elapsed() > budget_s - 450:
            break
        h = _slsqp(h, n, bounds, cons, a, 350, 1e-13)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 2: Analytical subgradient with momentum
    if elapsed() < budget_s - 380:
        h2, c2 = _momentum_descend(best_h, n, dx, budget_s - 380, steps=4000)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Focused CD coarse
    if elapsed() < budget_s - 250:
        h = best_h.copy()
        for _ in range(600):
            if elapsed() > budget_s - 260:
                break
            h, c, imp = _cd_focused(h, n, dx, 17, 3, rng)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
            if not imp:
                break

    # Phase 4: SLSQP medium polish
    if elapsed() < budget_s - 180:
        h = _slsqp(best_h, n, bounds, cons, 700, 500, 1e-14)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 5: Focused CD fine
    if elapsed() < budget_s - 100:
        h = best_h.copy()
        for _ in range(1000):
            if elapsed() > budget_s - 105:
                break
            h, c, imp = _cd_focused(h, n, dx, 25, 5, rng)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
            if not imp:
                break

    # Phase 6: SLSQP high alpha
    if elapsed() < budget_s - 80:
        h = _slsqp(best_h, n, bounds, cons, 1500, 500, 1e-15)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 7: Multi-restart with diverse perturbations
    sigmas = [0.04, 0.06, 0.1, 0.03, 0.12, 0.05]
    for si, sig in enumerate(sigmas):
        if elapsed() > budget_s - 60:
            break
        h_p = best_h + sig * rng.randn(n)
        h_p = _repair(h_p, n)
        a = 300 + 200 * si
        h = _slsqp(h_p, n, bounds, cons, a, 350, 1e-12)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
        for _ in range(40):
            if elapsed() > budget_s - 55:
                break
            h, c2, imp = _cd_focused(h, n, dx, 19, 3, rng)
            if c2 < best_c5:
                best_c5, best_h = c2, h.copy()
            if not imp:
                break

    # Phase 8: Binarize and re-optimize (exploit discrete structure)
    if elapsed() < budget_s - 50:
        h_bin = _binarize_snap(best_h, n, threshold=0.015)
        c_bin = _c5(h_bin, dx)
        if c_bin < best_c5:
            best_c5, best_h = c_bin, h_bin.copy()
        else:
            h_mix = 0.7 * best_h + 0.3 * h_bin
            h_mix = _repair(h_mix, n)
            c_mix = _c5(h_mix, dx)
            if c_mix < best_c5:
                best_c5, best_h = c_mix, h_mix.copy()

    # Phase 9: Final ultra-fine CD
    if elapsed() < budget_s - 25:
        h = best_h.copy()
        hi_c5 = best_c5
        for _ in range(2000):
            if elapsed() > budget_s - 30:
                break
            h, cv, imp = _cd_focused(h, n, dx, 31, 4, rng)
            if cv < hi_c5:
                hi_c5 = cv
            if cv < best_c5:
                best_c5, best_h = cv, h.copy()
            if not imp:
                break

    # Phase 10: Final SLSQP
    if elapsed() < budget_s - 3:
        h = _slsqp(best_h, n, bounds, cons, 2500, 400, 1e-15)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
