    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + np.sum(s) / alpha) * (2.0 / n)


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _make_starts(h0, n, rng, target):
    """Create multiple starting points for multi-start SA."""
    starts = [h0.copy()]
    # Binary-rounded start
    h_bin = np.round(h0)
    h_bin = _repair(h_bin, n)
    starts.append(h_bin.copy())
    # Randomly perturbed start
    h_rand = h0.copy()
    for _ in range(20):
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            d = (rng.rand() - 0.5) * 0.3
            ni, nj = h_rand[i] + d, h_rand[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_rand[i], h_rand[j] = ni, nj
    h_rand = _repair(h_rand, n)
    starts.append(h_rand.copy())
    # Anti-pattern start: try inverting high/low regions
    h_inv = h0.copy()
    thresh = np.median(h0)
    h_inv[h0 > thresh] = np.maximum(0, 2*thresh - h0[h0 > thresh])
    h_inv[h0 <= thresh] = np.minimum(1, 2*thresh - h0[h0 <= thresh])
    h_inv = _repair(h_inv, n)
    starts.append(h_inv.copy())
    return starts


def _sa_single(h_start, n, dx, t_end, rng, target):
    """Single SA run with targeted perturbation and adaptive steps."""
    h = h_start.copy()
    cur_c5 = _c5(h, dx)
    sa_best = cur_c5
    sa_best_h = h.copy()
    
    T = 0.015
    Tend = 5e-7
    cooldown = 0.99985
    improved_its = 0
    total_its = 0
    stuck_its = 0
    
    top_lags = np.arange(n - 1)  # default
    corr = _corr(h)
    top_lags = np.argpartition(corr, -3)[-3:]
    
    while time.time() < t_end:
        total_its += 1
        
        # Update top lags periodically
        if total_its % 500 == 0:
            corr = _corr(h)
            top_lags = np.argpartition(corr, -3)[-3:]
        
        # Adaptive step size
        if stuck_its > 2000:
            T = max(T * 1.5, 0.005)
            stuck_its = 0
        elif improved_its > 0 and total_its % 5000 == 0:
            T *= 0.95
            improved_its = 0
        
        # Choose move: targeted (70%) or random (30%)
        if rng.rand() < 0.7 and len(top_lags) > 0:
            lag = top_lags[rng.randint(len(top_lags))]
            offset = int(lag) - (n - 1)
            if offset > 0 and offset < n:
                # Find high-product pairs along this lag
                if offset > 0:
                    vi = h[:n - offset]
                    vj = h[offset:]
                else:
                    vi = h[-offset:]
                    vj = h[:n + offset]
                prods = vi * vj
                total_p = prods.sum()
                if total_p > 1e-10:
                    idx = rng.rand() * total_p
                    cumsum = np.cumsum(prods)
                    i = int(np.searchsorted(cumsum, idx))
                    i = min(i, n - offset - 1)
                    j = i + offset
                else:
                    i = rng.randint(n)
                    j = (i + offset) % n
            else:
                i = rng.randint(n)
                j = rng.randint(n)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        
        if i == j:
            continue
        
        dmax = 0.04 + 0.06 * (T / 0.015)
        delta = (rng.rand() - 0.5) * 2.0 * dmax
        ni = h[i] + delta
        nj = h[j] - delta
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        
        h[i], h[j] = ni, nj
        new_c5 = _c5(h, dx)
        
        if new_c5 <= cur_c5 + T:
            cur_c5 = new_c5
            if new_c5 < sa_best - 1e-15:
                sa_best = new_c5
                sa_best_h = h.copy()
                improved_its += 1
        else:
            h[i], h[j] = h[i] - delta, h[j] + delta
            stuck_its += 1
        
        T = max(T * cooldown, Tend)
    
    return sa_best_h, sa_best


def _coord_descent(h, n, dx, t_end):
    """Fine coordinate descent with pair-wise line search."""
    h = h.copy()
    local_best = _c5(h, dx)
    deltas = np.array([0.0005, 0.001, 0.002, 0.005, 0.01,
                       -0.0005, -0.001, -0.002, -0.005, -0.01])
    no_improve = 0
    
    while time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            vi = h[i]
            for d in deltas:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                if need > 0:
                    order = np.argsort(h)
                else:
                    order = np.argsort(-h)
                found = False
                for j in order:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0.0 or nj > 1.0:
                        continue
                    h[i], h[j] = ni, nj
                    c5_try = _c5(h, dx)
                    if c5_try < local_best - 1e-15:
                        local_best = c5_try
                        improved = True
                        found = True
                        break
                    else:
                        h[i], h[j] = vi, h[j] - need
                if found:
                    break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                break
    
    return h, local_best


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)
    
    # Phase 1: SLSQP polish (12% budget)
    p1_end = t_start + min(budget_s * 0.12, 120)
    for alpha in [800, 2000, 5000]:
        if time.time() > p1_end:
            break
        try:
            res = minimize(_obj_lse, best_h, args=(alpha, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-15})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass
    
    # Phase 2: Multi-start SA (58% budget)
    p2_end = t_start + min(budget_s * 0.70, 750)
    if time.time() < p2_end:
        starts = _make_starts(best_h, n, rng, target)
        # Sort by quality, try best first
        starts.sort(key=lambda x: _c5(x, dx))
        
        for si, hs in enumerate(starts):
            per_start_time = max(50, (p2_end - time.time()) / max(1, len(starts) - si))
            st = time.time()
            hsa, c5a = _sa_single(hs, n, dx, st + per_start_time, rng, target)
            if c5a < best_c5:
                best_c5, best_h = c5a, hsa.copy()
            if time.time() > p2_end - 30:
                break
    
    # Phase 3: Coordinate descent (18% budget)
    p3_end = t_start + min(budget_s * 0.90, 960)
    if time.time() < p3_end:
        h_cd, c5_cd = _coord_descent(best_h, n, dx, p3_end)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd
    
    # Phase 4: Final SA refinement on best (8% budget)
    p4_end = t_start + budget_s - 15
    if time.time() < p4_end - 10:
        h_fin, c5_fin = _sa_single(best_h, n, dx, p4_end, rng, target)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin
    
    # Phase 5: Final SLSQP polish
    if time.time() < t_start + budget_s - 5:
        try:
            res = minimize(_obj_lse, best_h, args=(8000, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass
    
    # Final repair and evaluation
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
