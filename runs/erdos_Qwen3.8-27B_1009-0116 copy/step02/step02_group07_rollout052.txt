import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

                sl -= h[i - s]
        if 0 <= j + s < n:
            if j + s == i:
                sl -= 1.0 - h[i] + h[j]; q += 1.0
            else:
                sl -= 1.0 - h[j + s]
        if 0 <= j - s < n:
            if j - s == i:
                sl += 1.0 - h[j] + h[i]; q += 1.0
            else:
                sl += h[j - s]
        s_vec[k] = sl
        q_vec[k] = q
    return s_vec, q_vec

def _ternary(C, s_vec, q_vec, lo, hi, n_it=20):
    a, b = lo, hi
    for _ in range(n_it):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(C + s_vec * m1 + q_vec * m1 * m1)
        f2 = np.max(C + s_vec * m2 + q_vec * m2 * m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def _pair_cd(h, n, dx, t_limit, t_start, max_sweeps=5000, n_iter=20, order='seq'):
    best_c5 = _c5(h, dx)
    idx = np.arange(n)
    for sw in range(max_sweeps):
        if time.time() - t_start > t_limit:
            break
        C = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if order == 'rand':
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng2 = np.random.RandomState(sw)
            rng2.shuffle(pairs)
        else:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        for i, j in pairs:
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-13:
                continue
            s_vec, q_vec = _slopes(h, i, j, n)
            k_star = int(np.argmax(C))
            if abs(s_vec[k_star]) < 1e-6:
                continue
            d = _ternary(C, s_vec, q_vec, lo, hi, n_iter)
            if abs(d) < 1e-14:
                continue
            h[i] += d; h[j] -= d
            C += s_vec * d + q_vec * d * d
            new_max = float(C.max()) * dx
            if new_max < best_c5 - 1e-14:
                best_c5 = new_max
                improved = True
            else:
                h[i] -= d; h[j] += d
                C -= s_vec * d + q_vec * d * d
        if not improved:
            break
    h = _repair(h, n)
    return h, _c5(h, dx)

def _slsqp_polish(h, n, bounds, cons, alpha, maxiter, ftol):
    def _obj(x, a):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(a * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / a
    try:
        res = minimize(_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    cfg_rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    def elapsed():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: Exact pair CD (sequential order)
    if elapsed() < budget_s - 300:
        h, c = _pair_cd(best_h, n, dx, budget_s - 300, t0, max_sweeps=5000, n_iter=20, order='seq')
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 2: SLSQP with log-sum-exp
    for a in [200, 500, 1000]:
        if elapsed() > budget_s - 250:
            break
        h = _slsqp_polish(best_h, n, bounds, cons, a, 400, 1e-14)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 3: Pair CD (random order for better mixing)
    if elapsed() < budget_s - 200:
        h, c = _pair_cd(best_h, n, dx, budget_s - 200, t0, max_sweeps=5000, n_iter=22, order='rand')
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 4: SLSQP + pair CD refinement loop
    for it in range(5):
        if elapsed() > budget_s - 100:
            break
        h = _slsqp_polish(best_h, n, bounds, cons, 2000, 300, 1e-15)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
        if elapsed() > budget_s - 80:
            break
        h2, c2 = _pair_cd(best_h, n, dx, min(budget_s - 80, elapsed() + 60), t0, max_sweeps=3000, n_iter=25, order='seq')
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 5: Try alternate n (80, 90) with interpolation
    for n_alt in [80, 90, 64]:
        if elapsed() > budget_s - 40:
            break
        dx_alt = 2.0 / n_alt
        # Interpolate best_h to new grid
        x_old = np.linspace(0, 2, n, endpoint=False) + dx / 2
        x_new = np.linspace(0, 2, n_alt, endpoint=False) + dx_alt / 2
        h_alt = np.interp(x_new, x_old, best_h, period=2.0)
        h_alt = _repair(h_alt, n_alt)
        b_alt = [(0.0, 1.0)] * n_alt
        c_alt = [{'type': 'eq', 'fun': lambda x: x.sum() - n_alt / 2.0}]
        c_now = _c5(h_alt, dx_alt)
        h_alt = _slsqp_polish(h_alt, n_alt, b_alt, c_alt, 500, 300, 1e-14)
        c_now = _c5(h_alt, dx_alt)
        h_alt, c_now = _pair_cd(h_alt, n_alt, dx_alt, min(budget_s - 30, elapsed() + 50), t0, max_sweeps=2000, n_iter=20, order='seq')
        if c_now < best_c5:
            best_c5 = c_now
            best_h = h_alt.copy()
            n = n_alt
            dx = dx_alt

    # Phase 6: Final polish with pair CD
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    if elapsed() < budget_s - 10:
        h, c = _pair_cd(best_h, n_final, dx_final, budget_s - 15, t0, max_sweeps=5000, n_iter=25, order='seq')
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
