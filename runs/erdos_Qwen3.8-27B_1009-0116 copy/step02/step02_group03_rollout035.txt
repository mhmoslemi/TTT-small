                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved and r > 50:
            break
    return h, best

def make_start(n, seed_val, kind):
    rng = np.random.RandomState(seed_val)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.1 * rng.randn(n)
    elif kind == 2:
        f = rng.randint(2, 8)
        ph = rng.uniform(0, 6.283)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 3:
        block = max(2, n // 7)
        h = np.zeros(n)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        h = 0.5 * h + 0.2 * rng.randn(n)
    elif kind == 4:
        h = 0.5 + 0.3 * np.cos(np.pi * np.arange(n) / n) ** 3
    else:
        h = rng.uniform(0, 1, n)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 777)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    init_h = None
    init_n = 70
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass
    if init_h is not None:
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, init_n)
        best_n = init_n

    n_p = best_n
    h_work = best_h.copy()

    # Phase 1: SLSQP with escalating alpha and top-k flattening
    for alpha in [500, 2000, 8000, 20000, 50000]:
        if time.time() - t0 > budget_s * 0.30:
            break
        for topk in [1, 3, 5, 7]:
            if time.time() - t0 > budget_s * 0.30:
                break
            for fw in [0.0, 0.003, 0.01, 0.05]:
                if time.time() - t0 > budget_s * 0.30:
                    break
                h_new = do_slsqp(h_work.copy(), n_p, alpha, topk, fw, 500)
                c5_new = compute_c5(h_new, n_p)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h_work = h_new
                elif c5_new < compute_c5(h_work, n_p) - 1e-14:
                    h_work = h_new

    # Phase 2: L-BFGS-B refinement
    if time.time() - t0 < budget_s * 0.55:
        for alpha in [8000, 20000, 50000]:
            if time.time() - t0 > budget_s * 0.55:
                break
            for pen in [200, 2000, 20000]:
                if time.time() - t0 > budget_s * 0.55:
                    break
                h_new = do_lbfgs(h_work.copy(), n_p, alpha, 3, 0.01, pen, 3000)
                c5_new = compute_c5(h_new, n_p)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h_work = h_new

    # Phase 3: Long coordinate descent
    if time.time() - t0 < budget_s * 0.82:
        h_cd, c5_cd = coord_descent(h_work, n_p, 400, t0 + budget_s * 0.82, rng)
        if c5_cd < best_c5 - 1e-14:
            best_h = h_cd.copy()
            best_c5 = c5_cd

    # Phase 4: SLSQP on best + coordinate on best
    if best_h is not None and time.time() - t0 < budget_s * 0.92:
        h_fin = do_slsqp(best_h.copy(), best_n, 50000, 3, 0.01, 800)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-14:
            best_h = h_fin.copy()
            best_c5 = c5_fin
        h_fin2, c5_fin2 = coord_descent(best_h, best_n, 200, t0 + budget_s * 0.92, rng)
        if c5_fin2 < best_c5 - 1e-14:
            best_h = h_fin2.copy()
            best_c5 = c5_fin2

    # Phase 5: Try alternative n values (short budget)
    if time.time() - t0 < budget_s * 0.97:
        for n_alt in [64, 72, 80, 100]:
            if time.time() - t0 > budget_s * 0.97:
                break
            if n_alt == best_n:
                continue
            for kk in range(2):
                if time.time() - t0 > budget_s * 0.97:
                    break
                h_alt = make_start(n_alt, seed + n_alt * 13 + kk, kk % 5)
                h_alt = do_slsqp(h_alt, n_alt, 10000, 3, 0.01, 400)
                c5_alt = compute_c5(h_alt, n_alt)
                if c5_alt < best_c5 - 1e-14:
                    best_h = h_alt.copy()
                    best_c5 = c5_alt
                    best_n = n_alt

    # Phase 6: Final polish
    if best_h is not None and time.time() - t0 < budget_s * 0.99:
        h_p = do_slsqp(best_h.copy(), best_n, 100000, 5, 0.02, 500)
        c5_p = compute_c5(h_p, best_n)
        if c5_p < best_c5 - 1e-14:
            best_h = h_p.copy()
            best_c5 = c5_p
        h_p2, c5_p2 = coord_descent(best_h, best_n, 80, t0 + budget_s * 0.995, rng)
        if c5_p2 < best_c5 - 1e-14:
            best_h = h_p2.copy()
            best_c5 = c5_p2

    if best_h is None:
        best_n = 70
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
