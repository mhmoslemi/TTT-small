    cons.append({'type': 'eq', 'fun': lambda x: x[:-1].sum() - n/2.0})
    
    x0 = np.append(h, c_full.max())
    bounds = [(0.0, 1.0)] * n + [(c_full.max() * 0.9, 2.0)]
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x[:-1], n)
    except:
        return h.copy()

def _block_cd(h, n, dx, block_size, t_limit, t_start, max_passes=200):

    improved = True
    n_pass = 0
    while improved and n_pass < max_passes:
        if time.time() - t_start > t_limit:
            break
        improved = False
        n_pass += 1
        # Shuffle block order for diversity
        starts = list(range(0, n - block_size + 1, block_size // 2))
        np.random.shuffle(starts)
        for start in starts:
            end = min(start + block_size, n)
            bs = end - start
            if bs < 3:
                continue
            orig_block = h[start:end].copy()
            orig_sum = orig_block.sum()
            
            def obj_b(x):
                h_tmp = h.copy()
                h_tmp[start:end] = x
                c, _ = _c5(h_tmp, dx)
                return c
            
            bounds_b = [(0.0, 1.0)] * bs
            cons_b = [{'type': 'eq', 'fun': lambda x: x.sum() - orig_sum}]
            
            try:
                res = minimize(obj_b, orig_block, method='SLSQP',
                              bounds=bounds_b, constraints=cons_b,
                              options={'maxiter': 200, 'ftol': 1e-12})
                new_block = np.clip(res.x, 0.0, 1.0)
                # Fix sum
                diff = orig_sum - new_block.sum()
                if abs(diff) > 1e-10:
                    new_block += diff / bs
                    new_block = np.clip(new_block, 0.0, 1.0)
                    # Re-fix
                    diff2 = orig_sum - new_block.sum()
                    if abs(diff2) > 1e-12:
                        new_block[np.argmax(1.0 - new_block if diff2 > 0 else new_block)] += diff2
            
            h_new = h.copy()
            h_new[start:end] = new_block
            c_new, _ = _c5(h_new, dx)
            if c_new < hi_c5 - 1e-13:
                h[start:end] = new_block
                hi_c5 = c_new
                improved = True
    return h, hi_c5

def _pair_cd(h, n, dx, t_limit, t_start, n_grid=20, max_passes=500):

    improved = True
    n_pass = 0

            break

        n_pass += 1

                    hi_c5 = best_c
                    improved = True
                else:
                    h[i], h[j] = orig_i, orig_j
    return h, hi_c5

def _proj_grad(h, n, dx, n_steps=1000, lr_init=0.01):
    """Projected subgradient descent on max correlation."""
    h = h.copy()
    c_full = np.correlate(h, 1.0 - h, mode='full')
    m = c_full.max()
    # Active set: lags within 1% of max
    active = c_full >= m * 0.999
    for step in range(n_steps):
        # Compute gradient for active lags
        grad = np.zeros(n)
        for k_idx in np.where(active)[0]:
            k = k_idx - (n - 1)  # actual lag
            for j in range(n):
                g = 0.0
                if 0 <= j - k < n:
                    g += (1 - h[j - k])
                if 0 <= j + k < n:
                    g -= h[j + k]
                grad[j] += g / active.sum()
        # Projected gradient step
        lr = lr_init * (1.0 / (1.0 + 0.01 * step))
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum
        diff = n/2.0 - h_new.sum()
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        # Update
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        m_new = c_new.max()
        if m_new < m:
            h = h_new
            m = m_new
            active = c_new >= m * 0.999
            c_full = c_new
        # else keep h
    return h, m * dx

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx), c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _slsqp_smooth(h, n, alpha, maxiter=500):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _slsqp_minimax(h, n, n_top=12, maxiter=300):
    c_full = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c_full)[-n_top:]
    t0 = c_full.max()
    
    def obj(x):
        return x[-1]
    
    def cons_func(x, k):
        return x[-1] - np.correlate(x[:-1], 1.0 - x[:-1], mode='full')[k]
    
    cons = [{'type': 'ineq', 'fun': lambda x, k=k: cons_func(x, k)} for k in top_idx]
    cons.append({'type': 'eq', 'fun': lambda x: x[:-1].sum() - n/2.0})
    
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(t0 * 0.95, t0 * 1.05)]
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x[:-1], n)
    except Exception:
        return h.copy()

def _block_cd(h, n, dx, block_size, t_limit, t_start, max_passes=200):
    hi_c5, _ = _c5(h, dx)
    n_pass = 0
    improved = True
    while improved and n_pass < max_passes:
        if time.time() > t_limit:
            break
        improved = False
        n_pass += 1
        step = max(1, block_size // 3)
        starts = list(range(0, n - block_size + 1, step))
        np.random.shuffle(starts)
        for start in starts:
            end = min(start + block_size, n)
            bs = end - start
            if bs < 3:
                continue
            orig = h[start:end].copy()
            osum = orig.sum()
            
            def obj_b(x):
                h2 = h.copy()
                h2[start:end] = x
                return float(np.max(np.correlate(h2, 1.0 - h2, mode='full')) * dx)
            
            bounds_b = [(0.0, 1.0)] * bs
            cons_b = [{'type': 'eq', 'fun': lambda x: x.sum() - osum}]
            try:
                res = minimize(obj_b, orig, method='SLSQP', bounds=bounds_b,
                              constraints=cons_b, options={'maxiter': 150, 'ftol': 1e-11})
                nb = np.clip(res.x, 0.0, 1.0)
                d = osum - nb.sum()
                nb += d / bs
                nb = np.clip(nb, 0.0, 1.0)
                h2 = h.copy()
                h2[start:end] = nb
                c2, _ = _c5(h2, dx)
                if c2 < hi_c5 - 1e-13:
                    h[start:end] = nb
                    hi_c5 = c2
                    improved = True
            except Exception:
                pass
    return h, hi_c5

def _pair_cd(h, n, dx, t_limit, t_start, n_grid=20, max_passes=100):
    hi_c5, _ = _c5(h, dx)
    n_pass = 0
    improved = True

            break

                    hi_c5 = bc

                else:

def _proj_grad_step(h, n, n_steps=500, lr=0.02):
    h = h.copy()
    c_full = np.correlate(h, 1.0 - h, mode='full')
    m = c_full.max()
    for s in range(n_steps):
        tol = 0.01 * m + 1e-6
        active = c_full >= m - tol
        na = active.sum()
        if na == 0:
            na = 1
            active = np.zeros(len(c_full), dtype=bool)
            active[np.argmax(c_full)] = True
        grad = np.zeros(n)
        active_lags = np.where(active)[0]
        for idx in active_lags:
            k = idx - (n - 1)
            for j in range(n):
                g = 0.0
                ik = j - k
                if 0 <= ik < n:
                    g += 1.0 - h[ik]
                ik2 = j + k
                if 0 <= ik2 < n:
                    g -= h[ik2]
                grad[j] += g
        grad /= na
        h2 = np.clip(h - lr * grad, 0.0, 1.0)
        d = n / 2.0 - h2.sum()
        h2 += d / n
        h2 = np.clip(h2, 0.0, 1.0)
        c2 = np.correlate(h2, 1.0 - h2, mode='full')
        m2 = c2.max()
        if m2 < m:
            h, m, c_full = h2, m2, c2
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return h, m * dx

from scipy.optimize import minimize
import time

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx), c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _sum_eq_n(x, n):
    return x.sum() - n / 2.0

def _mm_sum_eq(x, n):
    return x[:-1].sum() - n / 2.0

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _mm_obj(x):
    return x[-1]

def _mm_ineq(x, k):
    return x[-1] - np.correlate(x[:-1], 1.0 - x[:-1], mode='full')[k]

def _block_obj(x, h_full, start, end, dx):
    h2 = h_full.copy()
    h2[start:end] = x
    return float(np.max(np.correlate(h2, 1.0 - h2, mode='full')) * dx)

def _slsqp_smooth(h, n, alpha, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq_n, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _slsqp_minimax(h, n, n_top=12, maxiter=300):
    c_full = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c_full)[-n_top:]
    t0 = c_full.max()
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(t0 * 0.95, t0 * 1.05)]
    cons = []
    for k in top_idx:
        cons.append({'type': 'ineq', 'fun': _mm_ineq, 'args': (k,)})
    cons.append({'type': 'eq', 'fun': _mm_sum_eq, 'args': (n,)})
    try:
        res = minimize(_mm_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x[:-1], n)
    except Exception:
        return h.copy()

def _block_cd(h, n, dx, block_size, t_limit, t_start, max_passes=200):
    hi_c5, _ = _c5(h, dx)
    n_pass = 0
    improved = True
    while improved and n_pass < max_passes:
        if time.time() > t_limit:
            break
        improved = False
        n_pass += 1
        step = max(1, block_size // 3)
        starts = list(range(0, n - block_size + 1, step))
        np.random.shuffle(starts)
        for start in starts:

            bs = end - start
            if bs < 3:
                continue
            orig = h[start:end].copy()
            osum = orig.sum()
            bounds_b = [(0.0, 1.0)] * bs
            cons_b = [{'type': 'eq', 'fun': _sum_eq_n, 'args': (int(bs * 2),)}]

def _blk_sum(x, osum):

def _pair_cd(h, n, dx, t_limit, t_start, n_grid=20, max_passes=100):

    n_pass = 0
    improved = True

        if time.time() > t_limit:
            break
        improved = False
        n_pass += 1
        for i in range(n):
            for j in range(i + 1, n):
                hs = h[i] + h[j]
                lo, hi = max(0.0, hs - 1.0), min(1.0, hs)
                if hi - lo < 1e-10:
                    continue

                else:

def _proj_grad(h, n, n_steps=500, lr0=0.02):
    h = h.copy()
    c_full = np.correlate(h, 1.0 - h, mode='full')
    m = c_full.max()
    lr = lr0
    for s in range(n_steps):
        tol = 0.01 * m + 1e-7
        active = c_full >= m - tol
        na = active.sum()
        if na == 0:
            active = np.zeros(len(c_full), dtype=bool)
            active[np.argmax(c_full)] = True
            na = 1
        grad = np.zeros(n)
        active_lags = np.where(active)[0]
        for idx in active_lags:
            k = idx - (n - 1)
            for j in range(n):
                g = 0.0
                ik = j - k
                if 0 <= ik < n:
                    g += 1.0 - h[ik]
                ik2 = j + k
                if 0 <= ik2 < n:
                    g -= h[ik2]
                grad[j] += g
        grad /= na
        h2 = np.clip(h - lr * grad, 0.0, 1.0)
        d = n / 2.0 - h2.sum()
        h2 += d / n
        h2 = np.clip(h2, 0.0, 1.0)
        c2 = np.correlate(h2, 1.0 - h2, mode='full')
        m2 = c2.max()
        if m2 < m - 1e-14:
            h, m, c_full = h2, m2, c2
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h, m * dx

            best_c5, best_h = c, h2.copy()

    # Phase 11: Final minimax polish
    if elapsed() < budget_s - 3:
        h2 = _slsqp_minimax(best_h, n, n_top=20, maxiter=200)
        c, _ = _c5(h2, dx)
        if c < best_c5:
            best_c5, best_h = c, h2.copy()

    best_h = _repair(best_h, n)
    best_c5, _ = _c5(best_h, dx)
    return (best_h, best_c5, n)
