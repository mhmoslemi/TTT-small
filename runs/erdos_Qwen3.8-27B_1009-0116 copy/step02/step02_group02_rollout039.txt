def _proj(y, target):
    # Project onto {x: 0<=x<=1, sum(x)=target}
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    x = np.clip(y - mid, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

    n = len(x)

    s = np.clip(a * (c - m), -500.0, 500.0)

def _lag_grad(h, lag, n):
    """Gradient of overlap at specific lag"""
    g = np.zeros(n)
    c = np.correlate(h, 1-h, mode='full')
    idx = lag + n - 1  # index in full correlation
    # For positive lag L: c[idx] = sum_i h[i] * (1-h[i+L])
