import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / x.size if False else m + np.log(np.sum(np.exp(s))) / alpha

def _obj(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx

def _con_fun(x, target):
    return np.sum(x) - target

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _con_fun, 'args': (target,)}]
    try:
        res = minimize(_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _project_constrained(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgradient_step(h, n, dx, lr):
    corr = _corr(h)
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        if 0 <= i - s < n:
            grad[i] += (1.0 - h[i - s])
        if 0 <= i + s < n:
            grad[i] -= h[i + s]
    h_new = h - lr * grad
    return _project_constrained(h_new, n / 2.0)

def _subgradient(h, n, dx, t0, budget_s, margin, rng):
    best = h.copy()
    best_c5 = _c5(best, dx)
    target = n / 2.0
    lr_start = 0.02
    deadline = t0 + budget_s - margin
    n_iter = 0
    while time.time() < deadline:
        batch = 5
        h_avg = np.zeros(n)
        for _ in range(batch):
            h = _subgradient_step(best, n, dx, lr_start / (1.0 + 0.01 * n_iter))
            h_avg += h
        h_avg /= batch
        c5_new = _c5(h_avg, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best = h_avg.copy()
        n_iter += 1
        if n_iter % 200 == 0:
            lr_start *= 0.95
    return best, best_c5

def _fine_local_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    steps = np.array([0.001, 0.002, 0.005, 0.01, 0.02, 0.04,
