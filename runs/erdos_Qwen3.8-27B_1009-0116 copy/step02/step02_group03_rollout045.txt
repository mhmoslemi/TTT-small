        if abs(diff) < 1e-13:

        if diff > 0:

            if not mask.any(): break
            h[mask] += diff / mask.sum()
        else:

            if not mask.any(): break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _multi_subgrad(h, n):
    """Subgradient using top-k active lags."""
    c = _corr(h)
    m = c.max()
    # Top 5 lags near the max
    topk = np.argsort(c)[::-1][:min(5, len(c))]
    active = topk[c[topk] > m - 0.5]  # within 0.5 of max
    if len(active) == 0:
        active = topk[:1]
    g = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)
        for i in range(n):
            if 0 <= i - lag < n:
                g[i] += 1.0 - h[i - lag]
            if 0 <= i + lag < n:
                g[i] -= h[i + lag]
    g /= len(active)
    return g

def _slsqp_opt(h, n, target, alpha, maxiter=200):
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def con(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter':maxiter, 'ftol':1e-14})
        return _project(res.x, target)
    except:
        return h

def _lbfgs_opt(h, n, target, alpha, pen, maxiter=500):

                       options={'maxiter':maxiter, 'ftol':1e-16})

    except:

            h = h2; best = _c5(h, dx)
    if time.time() < budget_end - 5:
        h2, _ = _grad_steps(h, n, tgt, 150, budget_end - 3)
        if _c5(h2, dx) < best:
            h = h2; best = _c5(h, dx)
    if time.time() < budget_end - 2:
        h2, _ = _pair_cd(h, n, dx, tgt, budget_end - 1, 30)
        if _c5(h2, dx) < best:
            h = h2; best = _c5(h, dx)
    return _project(h, tgt), best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    best_h, best_c5, best_n = None, np.inf, 70

    def _upd(h, n):
        nonlocal best_h, best_c5, best_n
        c = _c5(h, 2.0 / n)
        if c < best_c5:
            best_h, best_c5, best_n = h.copy(), c, n

    deadline = t0 + budget_s - 5.0

    init_h, init_n = None, 70
    try:
        init_n = len(initial_h_values)
        init_h = _project(np.array(initial_h_values, dtype=float).copy(), init_n / 2.0)
    except NameError:
        pass

    # Main optimization on initial solution
    if init_h is not None:
        n_m = init_n
        tgt_m = n_m / 2.0
        dx_m = 2.0 / n_m
        _upd(init_h, n_m)
        h = init_h.copy()
        t_phase = t0 + budget_s * 0.55

        # Stage 1: Gradient-based
        for pen in [200, 1000, 5000]:
            if time.time() > t_phase - 80:
                break
            h2 = _lbfgs_opt(h, n_m, tgt_m, 200, pen, 400)
            if _c5(h2, dx_m) < _c5(h, dx_m):
                h = h2
            _upd(h, n_m)

        for al in [200, 500, 1000, 2000]:
            if time.time() > t_phase - 40:
                break
            h2 = _slsqp_opt(h, n_m, tgt_m, al, 250)
            if _c5(h2, dx_m) < _c5(h, dx_m):
                h = h2
            _upd(h, n_m)

        if time.time() < t_phase - 20:
            h2, _ = _grad_steps(h, n_m, tgt_m, 300, t_phase - 5)
            if _c5(h2, dx_m) < _c5(h, dx_m):
                h = h2
            _upd(h, n_m)

        t_cd = t0 + budget_s * 0.75
        if time.time() < t_cd - 10:
            h2, _ = _pair_cd(h, n_m, dx_m, tgt_m, t_cd - 5, 60)
            if _c5(h2, dx_m) < _c5(h, dx_m):
                h = h2
            _upd(h, n_m)

        # Stage 2: Annealing for escape
        t_anneal = t0 + budget_s * 0.90
        if time.time() < t_anneal - 30:
            h2, c2 = _rand_cd(h, n_m, dx_m, tgt_m, t_anneal - 10, 250000, seed)
            if c2 < _c5(h, dx_m):
                h = h2
            _upd(h, n_m)

        # Stage 3: Final tight optimization
        t_final = deadline
        h2 = _slsqp_opt(h, n_m, tgt_m, 3000, 300)
        if time.time() < t_final - 30 and _c5(h2, dx_m) < _c5(h, dx_m):
            h = h2
        _upd(h, n_m)
        if time.time() < t_final - 10:
            h2, _ = _grad_steps(h, n_m, tgt_m, 200, t_final - 5)
            if _c5(h2, dx_m) < _c5(h, dx_m):
                h = h2
            _upd(h, n_m)
        if time.time() < t_final - 3:
            h2, _ = _pair_cd(h, n_m, dx_m, tgt_m, t_final - 1, 40)
            if _c5(h2, dx_m) < _c5(h, dx_m):
                h = h2
            _upd(h, n_m)
        if time.time() < t_final - 1:
            h2 = _slsqp_opt(h, n_m, tgt_m, 5000, 200)
            _upd(h2, n_m)
        if time.time() < t_final - 0.5:
            h2, _ = _pair_cd(best_h, n_m, dx_m, tgt_m, t_final, 20)
            _upd(h2, n_m)

    # Alternative n and starts if time remains
    if time.time() < deadline - 40:
        try_ns = [70, 60, 80, 50, 90]
        if init_n not in try_ns:
            try_ns.insert(0, init_n)
        for n_a in try_ns:
            if time.time() > deadline - 20:
                break
            if init_h is not None and n_a == init_n:
                continue
            tgt_a = n_a / 2.0
            dx_a = 2.0 / n_a
            t_a_end = min(time.time() + 40, deadline - 5)
            for kind in range(4):
                if time.time() > t_a_end - 10:
                    break
                h0 = _make_start(n_a, seed + kind * 300 + n_a * 7, tgt_a, kind)
                _upd(h0, n_a)
                h2, c2 = _refine(h0, n_a, t_a_end - 5)
                _upd(h2, n_a)

    # Perturb best and re-optimize
    if best_h is not None and time.time() < deadline - 15:
        n_b = best_n
        rng2 = np.random.RandomState(seed + 999)
        for trial in range(3):
            if time.time() > deadline - 5:
                break
            h_pert = best_h + 0.02 * rng2.randn(n_b)
            h_pert = _project(h_pert, n_b / 2.0)
            h2, c2 = _refine(h_pert, n_b, deadline - 2)
            _upd(h2, n_b)

    if best_h is None:
        best_h = _project(np.full(70, 0.5), 35.0)
        best_c5 = _c5(best_h, 2.0 / 70)
        best_n = 70

    best_h = _project(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
