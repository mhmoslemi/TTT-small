import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def _smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    F = m + np.log(e.sum()) / alpha
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    j = np.arange(n)
    for ki in range(2 * n - 1):
        if w[ki] < 1e-12:
            continue
        lag = lags[ki]
        jm = j - lag
        jp = j + lag
        mask_m = (jm >= 0) & (jm < n)
        mask_p = (jp >= 0) & (jp < n)
        g = np.zeros(n)
        g[mask_m] = (1.0 - h[jm[mask_m]])
        g[mask_p] -= h[jp[mask_p]]
        grad += w[ki] * g
    return F, grad

def _grad_desc(h, n, deadline, top_k=3, lr_init=0.005):
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    lr = lr_init
    stall = 0
    j = np.arange(n)
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[::-1][:top_k]
        lags = idx - (n - 1)
        grad = np.zeros(n)
        for lag in lags:
            jm = j - lag
            jp = j + lag
            mask_m = (jm >= 0) & (jm < n)
            mask_p = (jp >= 0) & (jp < n)
            g = np.zeros(n)
            g[mask_m] = (1.0 - h[jm[mask_m]])
            g[mask_p] -= h[jp[mask_p]]
            grad += g
        grad /= top_k
        h_new = _repair(h - lr * grad, n)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.6
                stall = 0
                if lr < 1e-7:
                    break
    return best_h, best_c5

def _slsqp_step(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    def obj(x):
        return _smooth_obj_grad(x, n, alpha)[0]
    def objg(x):
        return _smooth_obj_grad(x, n, alpha)[1]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=[eq],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc)
    except Exception:
        return h, _compute_c5(h)

def _guided_local(h, n, dx, rng, deadline):
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    j_arr = np.arange(n)
    for scale in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        improved = False
        for _ in range(5000):
            if time.time() >= deadline:
                return best_h, best_c5
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            worst = np.argmax(corr) - (n - 1)
            if worst > 0:
                ii = np.arange(0, n - worst)
            elif worst < 0:
                ii = np.arange(-worst, n)
            else:
                ii = np.arange(n - 1)
            jj = ii + worst
            if len(ii) == 0:
                break
            wts = np.maximum(best_h[ii] * (1.0 - best_h[jj]), 0.001)
            wts /= wts.sum()
            pi = rng.choice(len(ii), p=wts)
            i, j = ii[pi], jj[pi]
            d = scale * (2.0 * rng.random() - 1.0)
            if (1.0 - best_h[j]) > best_h[i]:
                d = abs(d)
            else:
                d = -abs(d)
            ni, nj = best_h[i] + d, best_h[j] - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            best_h[i], best_h[j] = ni, nj
            c5n = _compute_c5(best_h)
            if c5n < best_c5 - 1e-13:
                best_c5 = c5n
                improved = True
            else:
                best_h[i], best_h[j] = ni - d, nj + d
        if not improved:
            break
    return best_h, best_c5

def _multi_start_local(h, n, dx, rng, deadline, n_starts=4):
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    time_per = (deadline - time.time()) / max(n_starts, 1)
    for s in range(n_starts):
        if time.time() >= deadline:
            break
        dl = time.time() + max(time_per, 5)
        if s == 0:
            start = best_h.copy()
        else:
            start = _repair(best_h + 0.03 * rng.randn(n), n)
        hh, c5h = _guided_local(start, n, dx, rng, dl)
        if c5h < best_c5:
            best_h, best_c5 = hh, c5h
    return best_h, best_c5
