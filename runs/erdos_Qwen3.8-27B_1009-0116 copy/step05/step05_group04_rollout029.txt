from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def smooth_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s_clipped = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s_clipped))) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s_clipped = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s_clipped - s_clipped.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)

        if sh >= 0:

            g[i1] += wk * (1.0 - h[i1 + sh])

        if s > target:
            lo = mid
        else:
            hi = mid

        return np.array([x.sum() - n * 0.5])

        return np.array([np.ones(n)])

    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return repair(res.x, n)

def binary_push(h, n, n_iter, rng):
    best = h.copy()
    best_c = eval_c5(best)
    for _ in range(n_iter):
        i = rng.randint(n)

        c = eval_c5(h)
        if c < best_c:

            best = h.copy()
        else:
            h = best.copy()

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, n_iter, step0, rng):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(n_iter):
        c = np.correlate(best, 1.0 - best, mode='full')
        cm = c.max()
        tol = 0.01 * cm
        top = c > cm - tol
        g = np.zeros(n)
        for k in np.where(top)[0]:
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - best[i1 + sh])
                if sh > 0:
                    g[sh:] -= best[:n-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - best[:n-sp])
                g[:n-sp] -= best[sp:]
        g *= 2.0 / n
        v = mu * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c:
                best = h_new; best_c = c_new
                step = min(step * 1.3, 0.5)
                improved = True
                break
        if not improved:
            step *= 0.5
    return best, best_c

def _fine_search(h, n_tries, rng):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy(); best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        d = 3e-5 * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni-d, nj+d
    return best, best_c

    t0 = time.time()
    def _el(): return time.time() - t0
    rng = np.random.RandomState(seed)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _top_grad(h, n, top_m):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_m:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * 2.0 / n

def _subgrad(h, n_iter, step0):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    v = np.zeros(n); mu = 0.85; step = step0
    stuck = 0
    for _ in range(n_iter):
        g = _top_grad(best, n, 3)
        v = mu * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best = h_new; best_c = c_new; step = min(step * 1.2, 0.5)
                improved = True; stuck = 0
                break
        if not improved:
            step *= 0.6; stuck += 1
            if stuck > 100:
                best = _repair(best + 0.01 * np.random.randn(n), n)
                best_c = _eval(best); step = step0; stuck = 0
                v = np.zeros(n)
    return best, best_c

def _fine_search(h, n_tries, rng):
    n = len(h); h = h.copy()
    c_cur = _eval(h); best = h.copy(); best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = 3e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c
