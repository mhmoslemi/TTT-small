import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            adj = min(abs(diff), room)
            h[idx] += adj if diff > 0 else -adj
            diff += -adj if diff > 0 else adj
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _obj_lse(h, alpha):
    c = _raw_corr(h)
    dx = 2.0 / len(h)
    cd = c * dx
    m = cd.max()
    s = np.clip(alpha * (cd - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_lse(h, alpha):
    n = len(h)
    c = _raw_corr(h)
    dx = 2.0 / n
    cd = c * dx
    m = cd.max()
    s = np.clip(alpha * (cd - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wk = w[idx]
        if k >= 0:
            j1 = np.arange(k, n)
            g[j1] += wk * (1.0 - h[j1 - k])
            j2 = np.arange(0, n - k)
            g[j2] -= wk * h[j2 + k]
        else:
            p = -k
            j1 = np.arange(0, n - p)
            g[j1] += wk * (1.0 - h[j1 + p])
            j2 = np.arange(p, n)
            g[j2] -= wk * h[j2 - p]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_obj_lse, h, jac=_grad_lse, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(np.clip(res.x, 0, 1), n)
    except Exception:
        return _repair(h, n)

def _subgrad_step(h, step_list):
    """Projected subgradient step targeting the max correlation."""
    n = len(h)
    dx = 2.0 / n
    c = _raw_corr(h)
    k_max = c.argmax()
    k = k_max - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        j1 = np.arange(k, n)
        g[j1] += (1.0 - h[j1 - k])
        j2 = np.arange(0, n - k)
        g[j2] -= h[j2 + k]
    else:
        p = -k
        j1 = np.arange(0, n - p)
        g[j1] += (1.0 - h[j1 + p])
        j2 = np.arange(p, n)
        g[j2] -= h[j2 - p]
    g *= dx
    c5c = _eval(h)
    for step in step_list:
        h_try = _proj_sum(h - step * g, n * 0.5)
        c5_try = _eval(h_try)
        if c5_try < c5c - 1e-14:
            return h_try, c5_try
    return h, c5c

def _topk_balance(h, topk, n_iter, rng):
    """Balance the top-k correlations by alternating subgradient steps."""
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    best = h.copy()
    for _ in range(n_iter):
        c = _raw_corr(h)
        top_idx = np.argsort(c)[-topk:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        # Average gradient over top-k
        g = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            if k >= 0:
                j1 = np.arange(k, n)
                g[j1] += (1.0 - h[j1 - k])
                j2 = np.arange(0, n - k)
                g[j2] -= h[j2 + k]
            else:
                p = -k
                j1 = np.arange(0, n - p)
                g[j1] += (1.0 - h[j1 + p])
                j2 = np.arange(p, n)
                g[j2] -= h[j2 - p]

def _coordinate_hill(h, n_iter, scale, rng):
    """Pairwise coordinate descent."""
    n = len(h)
    c5c = _eval(h)
    best = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:

                c5c = c5n
                if c5n < _eval(best):
                    best = h.copy()
            else:

    return best, c5c

    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
