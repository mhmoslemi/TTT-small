    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, j, d, n):
    """O(n) vectorized computation of correlation change when h[j] += d."""
    nn = 2 * n - 1
    if d == 0.0:
        return np.zeros(nn)
    dc = np.zeros(nn)
    # h[j] as first factor: m from n-1-j to 2n-2-j
    lo1 = n - 1 - j
    hi1 = 2 * n - 2 - j
    vals = j + np.arange(lo1, hi1 + 1) - (n - 1)
    dc[lo1:hi1 + 1] += d * (1.0 - h[vals])
    # h[j] as second factor (inside 1-h[...]): m from j to j+n-1
    lo2 = j
    hi2 = j + n - 1
    vals2 = j - np.arange(lo2, hi2 + 1) + (n - 1)
    dc[lo2:hi2 + 1] -= d * h[vals2]
    return dc

def _pair_swap(h, c, i, j, d, n):
    """Apply swap and return new correlation."""
    dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
    return c + dc

def _sa_optimize(h, c, rng, n_iter, T_start, T_end, scale, n):
    """Simulated annealing with pairwise swaps."""
    dx = 2.0 / n
    c5 = c.max() * dx
    best_c5 = c5
    best_h = h.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        delta = new_c5 - c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            h[i] = ni
            h[j] = nj
            c = new_c
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, _corr(best_h), best_c5

def _greedy_optimize(h, c, rng, n_iter, scale, n):
    """Pure greedy pairwise swaps."""
    dx = 2.0 / n
    c5 = c.max() * dx
    best_c5 = c5
    best_h = h.copy()
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c5 = (c + dc).max() * dx
        if new_c5 < c5 - 1e-15:
            h[i] = ni
            h[j] = nj
            c = c + dc
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, _corr(best_h), best_c5

def _slsqp_smooth(h, alpha, maxiter):
    """SLSQP with log-sum-exp smoothing."""
    n = len(h)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        # Vectorized gradient computation
        # c[k] = sum_i x[i] * (1-x[i+k-(n-1)])
        # dc[k]/dx[j] = (1-x[j+k-(n-1)]) - x[j-(k-(n-1))] (for valid indices)
        # This is exactly _pos_change(x, j, 1, n) (the unit change)
        # But we need to weight by w[k]
        # g[j] = sum_k w[k] * dc[k]/dx[j]
        # = sum_k w[k] * (1-x[j+k-(n-1)]) - sum_k w[k] * x[j-(k-(n-1))]
        # Let's just use the pos_change approach
        for j in range(n):
            dc_j = _pos_change(x, j, 1.0, n)
            g[j] = np.dot(w, dc_j)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
