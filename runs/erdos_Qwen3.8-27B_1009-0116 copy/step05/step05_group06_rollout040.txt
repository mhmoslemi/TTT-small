def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    m = c.max()

    w = np.exp(s - s.max())

    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _sfun(x, target):
    return x.sum() - target

def _sjac(x, target):
    return np.ones(len(x))

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _nesterov(h, alpha, nsteps, step0, mu):
    n = len(h)
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    step = step0
    for _ in range(nsteps):
        g = _smooth_grad(h, alpha)
        v = mu * v - step * g
        h_try = _proj_to_sum(h + v, n * 0.5)
        fn = _smooth_obj(h_try, alpha)
        if fn < f:
            h, f = h_try, fn
            step = min(step * 1.03, step0 * 4.0)
        else:
            step *= 0.6
            v *= 0.3
    return h

def _coord_descending(h, nsteps, rng, scale):
    n = len(h)
    c5c = _eval(h)
    for _ in range(nsteps):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _fourier_perturb(h, rng, n_modes, amp):
    n = len(h)
    hp = h.copy()
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.randint(1, n // 2 + 1)
        phase = rng.rand() * 2.0 * np.pi
        hp += amp * rng.randn() * np.sin(2.0 * np.pi * freq * t / n + phase)
    return _repair(hp, n)

def _topk_grad(h, topk):
    n = len(h)
    c = _corr(h)
    dx = 2.0 / n
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / topk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: L-BFGS-B with penalty, increasing alpha
    for alpha in [100, 500, 2000, 10000, 50000, 200000]:
        if _el() > 150:
            break
        h_new = _lbfgs_opt(best_h, alpha, 4000, lam=5000.0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 2: Nesterov projected gradient
    if _el() < budget_s - 500:
        h_new = _nesterov(best_h, 80000, 8000, 0.004, 0.88)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: SLSQP fine-tune at base resolution
    if _el() < budget_s - 400:
        h_new = _slsqp_opt(best_h, 300000, 2000, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 4: Upsample to 2n and optimize
    if _el() < budget_s - 400:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [2000, 10000, 50000, 200000]:
            if _el() > budget_s - 300:
                break
            h_hi = _lbfgs_opt(h_hi, alpha, 3000, lam=5000.0)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if _el() < budget_s - 250:
            h_hi = _nesterov(h_hi, 300000, 5000, 0.002, 0.9)
            h_hi = _repair(h_hi, n1)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if _el() < budget_s - 200:
            h_hi = _slsqp_opt(h_hi, 500000, 1500, 1e-18)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1

    # Phase 5: Multi-start Fourier perturbations at best resolution
    nc = best_n
    if _el() < budget_s - 250:
        for _r in range(35):
            if _el() > budget_s - 250:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = _fourier_perturb(best_h, rng, rng.randint(3, 10), 0.015)
            elif strat == 1:
                hp = _repair(best_h + 0.01 * rng.randn(nc), nc)
            elif strat == 2:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(nc), nc)
            elif strat == 3:
                shift = rng.randint(1, max(2, nc // 10))
                hp = _repair(np.roll(best_h, shift), nc)
            else:
                hp = best_h.copy()
                mask = rng.rand(nc) < 0.2
                hp[mask] += 0.03 * rng.randn(mask.sum())
                hp = _repair(hp, nc)
            alpha_pick = [10000, 50000, 200000][rng.randint(3)]
            h_new = _lbfgs_opt(hp, alpha_pick, 1200, lam=3000.0)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), nc

    # Phase 6: Top-k subgradient for active lags
    if _el() < budget_s - 150:
        nc = best_n
        h_sub = best_h.copy()
        c5c = _eval(h_sub)
        for _it in range(600):
            if _el() > budget_s - 150:
                break
            g = _topk_grad(h_sub, 5)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            g = g / gn
            for _bt in range(30):
                h_try = _proj_to_sum(h_sub - 0.005 * g, nc * 0.5)
                c5n = _eval(h_try)
                if c5n < c5c:
                    h_sub, c5c = h_try, c5n
                    break
            if c5c < best_c5:
                best_c5, best_h, best_n = c5c, h_sub.copy(), nc

    # Phase 7: Fine coordinate descent (annealing)
    if _el() < budget_s - 40:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for scale in [0.003, 0.001, 0.0004, 0.0001]:
            if _el() > budget_s - 40:
                break
            h, c5c = _coord_descending(h, 10000, rng, scale)
            if c5c < best_c5:
                best_c5, best_h, best_n = c5c, h.copy(), nc

    # Phase 8: Final SLSQP + L-BFGS polish
    if _el() < budget_s - 10:
        h = _slsqp_opt(best_h, 500000, 1500, 1e-19)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)
    if _el() < budget_s - 2:
        h = _lbfgs_opt(best_h, 1000000, 2000, lam=10000.0)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
