            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    lo1 = max(0, i)
    hi1 = min(2 * n - 2, i + n - 1)
    if lo1 <= hi1:
        hidx = n - 1 + i - np.arange(lo1, hi1 + 1)
        dc[lo1:hi1 + 1] += di * (1.0 - h[hidx])
    lo2 = max(0, n - 1 - i)
    hi2 = min(2 * n - 2, 2 * n - 2 - i)
    if lo2 <= hi2:
        hidx = i + np.arange(lo2, hi2 + 1) - n + 1
        dc[lo2:hi2 + 1] -= di * h[hidx]
    return dc

def _ternary_max(c, d, lo, hi, iters=45):
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c + m1 * d)
        f2 = np.max(c + m2 * d)
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def _sm_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _greedy_pair_sweep(h, c, n, n_passes, rng, improve_thresh=1e-14):
    """Multiple passes of greedy pair descent"""
    c_max = c.max()
    for p in range(n_passes):
        improved = 0
        for _ in range(n):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            di_max = 1.0 - h[i]
            dj_max = h[j]
            if di_max < 1e-12 and dj_max < 1e-12:
                continue
            dc_i = _pos_change(h, i, 1.0, n)
            dc_j = _pos_change(h, j, 1.0, n)
            d = dc_i - dc_j
            delta = _ternary_max(c, d, -dj_max, di_max)
            new_max = np.max(c + delta * d)
            if new_max < c_max - improve_thresh:
                h[i] += delta
                h[j] -= delta
                c += delta * d
                c_max = new_max
                improved += 1
        if improved == 0:
            break
    return h, c, c_max

def _sa_pair(h, c, n, n_iter, T0, Tcool, rng, improve_thresh=1e-15):
    """Simulated annealing with pair moves"""
    c_max = c.max()
    T = T0
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        di_max = 1.0 - h[i]
        dj_max = h[j]
        if di_max < 1e-12 and dj_max < 1e-12:
            continue
        dc_i = _pos_change(h, i, 1.0, n)
        dc_j = _pos_change(h, j, 1.0, n)
        d = dc_i - dc_j
        delta = _ternary_max(c, d, -dj_max, di_max)
        new_max = np.max(c + delta * d)
        delta_f = new_max - c_max
        accept = False
        if delta_f < -improve_thresh:
            accept = True
        elif T > 1e-10 and rng.random() < np.exp(-delta_f / T):
            accept = True
        if accept:
            h[i] += delta
            h[j] -= delta
            c += delta * d
            c_max = new_max
        T *= Tcool
    return h, c, c_max

def _block_sweep(h, c, n, n_passes, rng, block_size=3):
    """Try moving a block of consecutive elements"""
    c_max = c.max()
    for p in range(n_passes):
        improved = 0
        for _ in range(n // 2):
            i = rng.randint(n - block_size)
            j = rng.randint(n - block_size)
            if j - i < block_size:
                continue
            # Try shifting mass from block j to block i
            di_max = 1.0 - h[i:i+block_size].max()
            dj_max = h[j:j+block_size].min()
            scale = min(di_max, dj_max)
            if scale < 1e-10:
                continue
            d_total = np.zeros(2 * n - 1)
            amp = scale * 0.5
            for b in range(block_size):
                d_total += _pos_change(h, i+b, amp/block_size, n)
                d_total -= _pos_change(h, j+b, amp/block_size, n)
            # ternary over how much of max move to take
            delta = _ternary_max(c, d_total, 0.0, 1.0, iters=30)
            new_max = np.max(c + delta * d_total)
            if new_max < c_max - 1e-14:
                for b in range(block_size):
                    h[i+b] += delta * amp / block_size
                    h[j+b] -= delta * amp / block_size
                c += delta * d_total
                c_max = new_max
                improved += 1
        if improved == 0:
            break
    return h, c, c_max

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP polish at initial resolution
    for alpha in [5000, 50000, 200000]:
        if _el() > 50:
            break
        h_new = _slsqp(best_h, alpha, 2000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Greedy pair descent at n0
    if _el() < budget_s - 500:
        h_work = best_h.copy()
        c_work = _corr(h_work)
        n_w = len(h_work)
        target_t = budget_s - 450
        while _el() < target_t:
            h_work, c_work, _ = _greedy_pair_sweep(h_work, c_work, n_w, 200, rng)
            h_work = _repair(h_work, n_w)
            c_work = _corr(h_work)
            c5 = _eval(h_work)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_work.copy(), n_w
            elif c5 > best_c5 + 1e-6:
                h_work = best_h.copy()
                c_work = _corr(h_work)

    # Phase 3: SA pair moves at n0
    if _el() < budget_s - 350:
        h_work = best_h.copy()
        c_work = _corr(h_work)
        n_w = len(h_work)
        n_iter_sa = 200000
        for T0 in [0.01, 0.005, 0.002, 0.001]:
            if _el() > budget_s - 350:
                break
            h_work, c_work, _ = _sa_pair(h_work, c_work, n_w, n_iter_sa, T0, 0.9999, rng)
            h_work = _repair(h_work, n_w)
            c_work = _corr(h_work)
            c5 = _eval(h_work)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_work.copy(), n_w

    # Phase 4: Block moves at n0
    if _el() < budget_s - 250:
        h_work = best_h.copy()
        c_work = _corr(h_work)
        n_w = len(h_work)
        h_work, c_work, _ = _block_sweep(h_work, c_work, n_w, 50, rng, block_size=2)
        h_work = _repair(h_work, n_w)
        c5 = _eval(h_work)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_work.copy(), n_w
        h_work, c_work, _ = _block_sweep(h_work, c_work, n_w, 30, rng, block_size=3)
        h_work = _repair(h_work, n_w)
        c5 = _eval(h_work)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_work.copy(), n_w

    # Phase 5: Upsample to 2*n0 and optimize
    if _el() < budget_s - 200:
        n1 = n0 * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        c_hi = _corr(h_hi)
        # SLSQP at higher res
        if _el() < budget_s - 150:
            h_hi = _slsqp(h_hi, 100000, 1500, 1e-20)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        # Greedy pairs at higher res
        if _el() < budget_s - 80:
            h_hi = best_h.copy()
            c_hi = _corr(h_hi)
            n_hi = len(h_hi)
            target_t = budget_s - 80
            while _el() < target_t:
                h_hi, c_hi, _ = _greedy_pair_sweep(h_hi, c_hi, n_hi, 100, rng)
                h_hi = _repair(h_hi, n_hi)
                c_hi = _corr(h_hi)
                c5 = _eval(h_hi)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h_hi.copy(), n_hi
                elif c5 > best_c5 + 1e-5:
                    h_hi = best_h.copy()
                    c_hi = _corr(h_hi)
                    break

    # Phase 6: Fine grinding
    if _el() < budget_s - 30:
        h_fine = best_h.copy()
        c_fine = _corr(h_fine)
        n_fine = len(h_fine)
        t_rem = budget_s - 30 - _el()
        n_iter = int(t_rem * 8000)
        if n_iter > 1000:
            h_fine, c_fine, _ = _sa_pair(h_fine, c_fine, n_fine, n_iter, 0.0005, 0.99999, rng)
        h_fine = _repair(h_fine, n_fine)
        c5 = _eval(h_fine)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fine.copy(), n_fine

    # Phase 7: Final polish
    if _el() < budget_s - 10:
        h_pol = _slsqp(best_h, 500000, 1500, 1e-20)
        c5 = _eval(h_pol)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_pol.copy(), len(h_pol)

    # Final verification
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
