import numpy as np
from scipy.optimize import minimize
import time

def _project(h, n):
    target = n * 0.5
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(hp - 0.5))
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - hp[idx])
                hp[idx] += a
                diff -= a
            else:
                a = min(-diff, hp[idx])
                hp[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return hp

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)
