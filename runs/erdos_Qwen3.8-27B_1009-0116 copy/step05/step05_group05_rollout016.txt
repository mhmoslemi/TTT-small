import numpy as np

def _linearized_subproblem(h_t, n):
    # Compute current correlations
    c = np.correlate(h_t, 1.0 - h_t, mode='full')
    dx = 2.0 / n
    c_vals = c * dx
    
    # Build LP:
    # Variables: h[0..n-1], t
    # Minimize: t
    # Constraints:
    #   c_k + grad_k^T (h - h_t) <= t  for all k  =>  grad_k^T h - t <= c_k - grad_k^T h_t
    #   sum(h) = n/2
    #   0 <= h[i] <= 1
    #   t >= 0 (implicit from constraints)
    
    # Actually, let's use the form: A_ub @ x <= b_ub
    # x = [h, t]
    # For each k: grad_k^T h - t <= c_k^val - grad_k^T h_t (where c_k^val = c_k * dx)
    # This is: [grad_k, -1] @ [h, t] <= c_k^val - grad_k^T h_t
    
    c_full = np.correlate(h_t, 1.0 - h_t, mode='full')  # length 2n-1
    m = len(c_full)
    
    # Gradient of c_k with respect to h
    # c_k = sum_i h[i] * (1 - h[(i+k) mod appropriate])
    # For the 'full' correlation, the k-th element (0-indexed) corresponds to lag k-(n-1)
    
    # Let me compute gradients properly
    A_rows = []
    b_rhs = []
    
    for k in range(m):
        sh = (n - 1) - k  # shift
        if sh >= 0:
            # c_k = sum_{i=0}^{n-1-sh} h[i] * (1 - h[i+sh])
            # d/dh[i] = (1 - h[i+sh]) for i = 0..n-1-sh
            # d/dh[i+sh] = -h[i] for i = 0..n-1-sh, i.e., for j=sh..n-1: -h[j-sh]
            grad = np.zeros(n)
            idx1 = np.arange(n - sh)
            if sh == 0:
                grad[idx1] = 1.0 - h_t[idx1] - h_t[idx1]  # d/dh[i] of h[i](1-h[i]) = 1-2h[i]
            else:
                grad[idx1] += (1.0 - h_t[idx1 + sh])
                idx2 = np.arange(sh, n)
                grad[idx2] -= h_t[idx2 - sh]
        else:
            sp = -sh
            # c_k = sum_{i=sp}^{n-1} h[i] * (1 - h[i-sp])
            grad = np.zeros(n)
            idx1 = np.arange(sp, n)
            grad[idx1] += (1.0 - h_t[idx1 - sp])
            idx2 = np.arange(n - sp)
            grad[idx2] -= h_t[idx2 + sp]
        
        # Linearized: c_k^val + grad^T (h - h_t) <= t
        # => grad^T h - t <= c_k^val - grad^T h_t
        c_val = c_full[k] * dx
        rhs = c_val - np.dot(grad, h_t)
        
        row = np.zeros(n + 1)
        row[:n] = grad
        row[n] = -1.0
        A_rows.append(row)
        b_rhs.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rhs)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]
    return h_t, c_vals.max()

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

            return res.x[:n]
    except Exception:
        pass
    return None

from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _binary_snap(h, n):
    """Snap h to nearest binary with correct number of 1s."""
    n_ones = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[sorted_idx[:n_ones]] = 1.0
    return h_b

def _topk_subgrad_dir(h, n, k_top):
    """Compute weighted subgradient direction for top-k correlations."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:k_top]
    cvals = c[idx]
    cmax = cvals[0]
    # Exponential weights emphasizing the top
    w = np.exp(5.0 * (cvals - cmax))
    w /= w.sum()
    
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        sh = (n - 1) - ki
        if sh >= 0:
            i1 = np.arange(n - sh)
            if sh == 0:
                g[i1] += wi * (1.0 - 2.0 * h[i1])
            else:
                g[i1] += wi * (1.0 - h[i1 + sh])
                i2 = np.arange(sh, n)
                g[i2] -= wi * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wi * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wi * h[i2 + sp]
    return g * dx, c * dx

def _solve_lp(h, n):
    """Solve the linearized min-max LP. Returns solution or None."""
    m = 2 * n - 1
    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode='full') * dx
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for k in range(m):
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            if sh == 0:
                A_ub[k, i1] = 1.0 - 2.0 * h[i1]
            else:
                A_ub[k, i1] += 1.0 - h[i1 + sh]
                i2 = np.arange(sh, n)
                A_ub[k, i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            A_ub[k, i1] += 1.0 - h[i1 - sp]
            i2 = np.arange(n - sp)
            A_ub[k, i2] -= h[i2 + sp]
        A_ub[k, n] = -1.0
        b_ub[k] = c_full[k] - np.dot(A_ub[k, :n], h)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, 1)] * n + [(0, None)]

    except Exception:
        pass
    return None

def _subgrad_refine(h, n, n_iters, k_top, alpha0, t0, budget):

def _pairwise_refine(h, n, n_iters, t0, budget):
    """Random pairwise moves."""
    best_h = h.copy()
    best_c5 = _eval(h)
    cur = h.copy()

        ni, nj = cur[i] + d, cur[j] - d

        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = cur.copy()
        else:

    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        if sh == 0:
            g[i1] = 1.0 - 2.0 * h[i1]
        else:
            g[i1] += 1.0 - h[i1 + sh]
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sp = -(-sh)
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += 1.0 - h[i1 - sp]
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _subgrad_dir(h, n, k_top, sharpness):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:k_top]
    cvals = c[idx]
    cmax = cvals[0]
    w = np.exp(sharpness * (cvals - cmax))
    w /= w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        g += wi * _grad_lag(h, ki, n)
    return g * dx

def _solve_lp(h, n, k_cons):
    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c_full)[::-1][:k_cons]
    m = len(top_idx)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for ii, k in enumerate(top_idx):
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            if sh == 0:
                A_ub[ii, i1] = 1.0 - 2.0 * h[i1]
            else:
                A_ub[ii, i1] += 1.0 - h[i1 + sh]
                i2 = np.arange(sh, n)
                A_ub[ii, i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            A_ub[ii, i1] += 1.0 - h[i1 - sp]
            i2 = np.arange(n - sp)
            A_ub[ii, i2] -= h[i2 + sp]
        A_ub[ii, n] = -1.0
        b_ub[ii] = c_full[k] - np.dot(A_ub[ii, :n], h)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_refine(h, n, t0, budget, k_cons=80):
    best_h = h.copy()
    best_c5 = _eval(h)
    cur = h.copy()
    stall = 0
    while time.time() - t0 < budget:
        lp_sol = _solve_lp(cur, n, k_cons)
        if lp_sol is None:
            break
        if np.max(np.abs(lp_sol - cur)) < 1e-7:
            stall += 1
            if stall > 5:
                break
            continue
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            cand = _repair(alpha * lp_sol + (1 - alpha) * cur, n)
            c5 = _eval(cand)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5; best_h = cand.copy(); cur = cand.copy()
                stall = 0; improved = True
                break
        if not improved:
            stall += 1
            if stall > 10:
                break
    return best_h, best_c5

def _subgrad_refine(h, n, t0, budget, k_top, alpha0, sharp):
    best_h = h.copy()
    best_c5 = _eval(h)
    cur = h.copy()
    vel = np.zeros(n)
    stall = 0
    it = 0
    while time.time() - t0 < budget:
        it += 1
        g = _subgrad_dir(cur, n, k_top, sharp)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        alpha = alpha0 / (1.0 + 0.003 * it)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            a = alpha * mult
            vel = 0.85 * vel - a * g
            cand = _proj_to_sum(cur + vel, n * 0.5)
            c5 = _eval(cand)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5; best_h = cand.copy(); cur = cand.copy()
                stall = 0; improved = True
                break
        if not improved:
            vel = np.zeros(n)
            stall += 1
            if stall > 200:
                break
    return best_h, best_c5

def _pairwise(h, n, t0, budget, n_iter, scale):
    best_h = h.copy()
    best_c5 = _eval(h)
    cur = h.copy()
    rng = np.random.RandomState(7)
    for _ in range(n_iter):
        if time.time() - t0 > budget:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = cur[i] + d, cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        cur[i], cur[j] = ni, nj
        c5 = _eval(cur)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5; best_h = cur.copy()
        else:
            cur[i], cur[j] = ni - d, nj + d
    return best_h, best_c5

def _binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n_ones]] = 1.0
    return hb

def _upsample(h, factor):
    return np.repeat(h, factor)
