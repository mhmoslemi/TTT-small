import numpy as np
from scipy.optimize import minimize
import time

def _correlate(h):
    n = len(h)
    m = 2*n - 1
    N = 1
    while N < m:
        N *= 2
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(np.flip(1.0 - h), N)
    return np.fft.irfft(H * V, N)[:m]

def _eval(h):
    n = len(h)
    c = _correlate(h)
    return float(np.max(c) * 2.0 / n)

def _eval_c(h):
    n = len(h)
    c = _correlate(h) * (2.0 / n)
    return c

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    return _proj(h, n * 0.5)

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _correlate(h) * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for sh in range(n):
        k = n - 1 - sh
        wk = w[k]
        if wk > 1e-15:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
    for sp in range(1, n):
        k = n - 1 + sp
        wk = w[k]
        if wk > 1e-15:
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    g *= dx
    return g

def _slsqp_opt(h, alpha, maxiter, t_limit):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = _correlate(x) * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def jac(x):
        return _grad_smooth(x, alpha)
    
    def con_fun(x):
        return x.sum() - n * 0.5
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _grad_descent(h, alpha, n_iter, t_limit):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    c5 = _eval(h)
    lr = 0.1
    t_start = time.time()
    
    for it in range(n_iter):
        if time.time() - t_start > t_limit:
            break
        g = _grad_smooth(h, alpha)
        # Project gradient onto tangent space
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = g / gn  # Normalize
        
        # Backtracking line search
        for factor in range(10):
            step = lr * (0.5 ** factor)
            h_new = _proj(h - step * g * gn, target)
            c5_new = _eval(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                break
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h, c5

def _exact_pair(h, i, j, n_iter=60):
    """Exact 1D minimization for pair move h[i]+d, h[j]-d."""
    n = len(h)
    dx = 2.0 / n
    c0 = _correlate(h) * dx
    # Compute a[k] = ∂c_k/∂h[i] - ∂c_k/∂h[j] (times dx, already included in c0)
    a = np.zeros(2*n-1)
    for k in range(2*n-1):
        sh = (n-1) - k
        ai = 0.0
        aj = 0.0
        if sh >= 0:
            if i < n - sh:
                ai += (1.0 - h[i+sh])
            if sh == 0 or i >= sh:
                ai -= h[i-sh] if i >= sh else 0
            if j < n - sh:
                aj += (1.0 - h[j+sh])
            if sh == 0 or j >= sh:
                aj -= h[j-sh] if j >= sh else 0
        else:
            sp = -sh
            if i >= sp:
                ai += (1.0 - h[i-sp])
            if i + sp < n:
                ai -= h[i+sp]
            if j >= sp:
                aj += (1.0 - h[j-sp])
            if j + sp < n:
                aj -= h[j+sp]
        a[k] = ai - aj
    
    dlo = max(-h[i], h[j] - 1.0)
    dhi = min(1.0 - h[i], h[j])
    if dlo >= dhi:
        return h, _eval(h)
    
    # Ternary search for min of max_k(c0[k] + d*a[k])
    lo, hi = dlo, dhi
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c0 + m1 * a)
        f2 = np.max(c0 + m2 * a)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    best_d = (lo + hi) / 2.0
    
    h[i] += best_d
    h[j] -= best_d
    h = np.clip(h, 0, 1)
    return h, _eval(h)

    return h, c5

def _upsample(h, factor):
    n = len(h)
    h_hi = np.repeat(h, factor)
    return _repair(h_hi, n * factor)

    m = 2*n - 1
    a = np.zeros(m)
    # For each k, ∂c_k/∂h[i]:
    # sh = (n-1) - k
    # if sh >= 0: (1-h[i+sh])*1_{i+sh<n} - h[i-sh]*1_{i>=sh}
    # if sh < 0: (1-h[i-sp])*1_{i>=sp} - h[i+sp]*1_{i+sp<n} where sp=-sh
    
    # Vectorized: for i:
    # Positive shifts (k = 0 to n-1, sh = n-1 to 0):
    #   (1-h[i+sh]) for sh = 0 to n-1-i-1: this is (1-h[i:min(n, i+n)])
    #   -h[i-sh] for sh = 1 to i: this is -(h[max(0,i-n):i])
    
    # Let me just do it element-wise for clarity, it's O(n) per k and O(n²) total
    # But with numpy vectorization over k:
    
    shifts = np.arange(2*n-1) - (n-1)  # ranges from -(n-1) to (n-1)
    # For shift s:
    # ∂c/∂h[p] = (1-h[p+s])*1_{0≤p+s<n} - h[p-s]*1_{0≤p-s<n}
    
    # For p = i:
    # term1: (1-h[i+s]) for s where 0 ≤ i+s < n, i.e., -i ≤ s ≤ n-1-i
    # term2: -h[i-s] for s where 0 ≤ i-s < n, i.e., i-n+1 ≤ s ≤ i
    
    s = shifts  # shape (2n-1,)
    # term1 for p=i:
    mask1 = (i + s >= 0) & (i + s < n)
    term1_i = np.where(mask1, 1.0 - h[np.clip(i+s, 0, n-1)], 0.0)
    # term2 for p=i:
    mask2 = (i - s >= 0) & (i - s < n)
    term2_i = np.where(mask2, -h[np.clip(i-s, 0, n-1)], 0.0)
    
    # Same for p=j:
    mask1j = (j + s >= 0) & (j + s < n)
    term1_j = np.where(mask1j, 1.0 - h[np.clip(j+s, 0, n-1)], 0.0)
    mask2j = (j - s >= 0) & (j - s < n)
    term2_j = np.where(mask2j, -h[np.clip(j-s, 0, n-1)], 0.0)
    
    a = term1_i + term2_i - term1_j - term2_j
    return a

import numpy as np
from scipy.optimize import minimize
import time

def _correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_correlate(h)) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    return _proj(h, n * 0.5)

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _correlate(h) * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for sh in range(n):
        k = n - 1 - sh
        wk = w[k]
        if wk > 1e-14:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
    for sp in range(1, n):
        k = n - 1 + sp
        wk = w[k]
        if wk > 1e-14:
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g * dx

def _slsqp(h, alpha, maxiter):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n

    def obj(x):
        c = _correlate(x) * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha

    def jac(x):
        return _grad_smooth(x, alpha)

    def cf(x):
        return x.sum() - n * 0.5

    def cj(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_desc(h, alpha, n_iter, t_lim):
    n = len(h)
    target = n * 0.5
    c5 = _eval(h)
    lr = 0.05
    t0 = time.time()
    for _ in range(n_iter):
        if time.time() - t0 > t_lim:
            break
        g = _grad_smooth(h, alpha)
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        improved = False
        for fac in range(8):
            step = lr * (0.5 ** fac)
            h_new = _proj(h - step * g, target)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h, c5

def _pairvec_a(h, i, j):
    n = len(h)
    s = np.arange(2 * n - 1) - (n - 1)
    i_s = i + s
    i_ns = i - s
    j_s = j + s
    j_ns = j - s
    m1 = (i_s >= 0) & (i_s < n)
    m2 = (i_ns >= 0) & (i_ns < n)
    m3 = (j_s >= 0) & (j_s < n)
    m4 = (j_ns >= 0) & (j_ns < n)
    a = np.zeros(2 * n - 1)
    a[m1] += 1.0 - h[i_s[m1]]
    a[m2] -= h[i_ns[m2]]
    a[m3] -= (1.0 - h[j_s[m3]])
    a[m4] += h[j_ns[m4]]
    return a

def _pairwise(h, n_rounds, t_lim, rng):
