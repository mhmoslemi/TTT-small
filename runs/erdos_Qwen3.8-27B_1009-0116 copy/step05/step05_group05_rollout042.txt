            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-16:
            c5 = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5

def _block_opt(h, n_trials, rng):
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_trials):
        bs = rng.randint(3, 11)
        st = rng.randint(0, max(1, n - bs))
        for _t in range(15):
            h_new = h.copy()
            h_new[st:st+bs] += 0.015 * rng.randn(bs)
            h_new = _repair(h_new, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-14:
                h = h_new; c5 = c5n; break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: LP cutting-plane with various topk
    for topk in [50, 100, 200, 559]:
        if _el() > 250:
            break
        h_new, c5 = _lp_search(best_h.copy(), best_n, topk, max_iter=60)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    # Phase 2: Active set descent
    if _el() < budget_s - 350:
        h_new, c5 = _active_descent(best_h.copy(), 3000, 5)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    # Phase 3: More LP with focused shifts
    if _el() < budget_s - 300:
        for topk in [30, 60, 120]:
            if _el() > budget_s - 250:
                break
            h_new, c5 = _lp_search(best_h.copy(), best_n, topk, max_iter=50)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()

    # Phase 4: Active set with fewer top shifts
    if _el() < budget_s - 250:
        h_new, c5 = _active_descent(best_h.copy(), 4000, 3)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    # Phase 5: Block optimization
    if _el() < budget_s - 180:
        h_new, c5 = _block_opt(best_h.copy(), 4000, rng)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    # Phase 6: Multi-scale pairwise
    if _el() < budget_s - 130:
        for scale in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if _el() > budget_s - 100:
                break
            h_new, c5 = _pairwise(best_h.copy(), 5000, scale, rng)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()

    # Phase 7: LP + active set alternation
    if _el() < budget_s - 80:
        for _rnd in range(3):
            if _el() > budget_s - 60:
                break
            h_new, c5 = _lp_search(best_h.copy(), best_n, 50, max_iter=30)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()
            h_new, c5 = _active_descent(best_h.copy(), 2000, 3)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()

    # Phase 8: Ultra-fine pairwise
    if _el() < budget_s - 15:
        for scale in [1e-4, 5e-5, 2e-5, 1e-5]:
            if _el() > budget_s - 5:
                break
            h_new, c5 = _pairwise(best_h.copy(), 4000, scale, rng)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()

    # Phase 9: Final LP polish
    if _el() < budget_s - 3:
        h_new, c5 = _lp_search(best_h.copy(), best_n, 30, max_iter=20)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
