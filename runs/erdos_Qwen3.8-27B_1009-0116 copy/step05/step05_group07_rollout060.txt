    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

        return h
    h *= target / s

                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent(h, topm, max_iter, step0=0.005):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    stale = 0
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - 15:
            break
        c = _corrs(h)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(h, k)
        g /= topm
        found = False
        for _bt in range(45):
            h_new = _proj_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-12 * step:
                h = h_new
                f = f_new
                found = True
                stale = 0
                break
            step *= 0.5
        if not found:
            stale += 1
            step *= 0.7
            if step < 1e-11 or stale > 30:
                break
            continue
        step = min(step * 1.06, step0)
    return h, f

def _round_robin(h, n_lags, steps_per_lag):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    for _ in range(n_lags):
        if time.time() - _t0 > _budget - 15:
            break
        c = _corrs(h)
        k_star = c.argmax()
        for _s in range(steps_per_lag):
            g = _grad_lag(h, k_star)
            found = False
            for step in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                h_new = _proj_sum(h - step * g, n * 0.5)
                f_new = _eval(h_new)
                if f_new < f - 1e-14:
                    h = h_new
                    f = f_new
                    found = True
                    break
            if not found:
                break
    return h, f

def _pairwise_fine(h, n_iter, scale, seed_val):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 5:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, f

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    fi = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(fi(x_new), 0, 1), n * factor)

def _level_top(h, topm, n_steps=200):
    """Try to equalize top lags by cooperative gradient steps"""
    n = len(h)
    h = h.copy()
    f = _eval(h)
    for _ in range(n_steps):
        if time.time() - _t0 > _budget - 10:
            break
        c = _corrs(h)
        top_idx = np.argsort(c)[::-1][:topm]
        cvals = c[top_idx]
        spread = cvals.max() - cvals.min()
        if spread < 1e-9:
            break
        # Weight toward the higher lags more
        w = (cvals - cvals.min() + 1e-12) ** 2
        w /= w.sum()
        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            g += w[i] * _grad_lag(h, k)
        step = 0.002
        found = False
        for _bt in range(25):
            h_new = _proj_sum(h - step * g, n * 0.5)
            f_new = _eval(h_new)
            if f_new < f - 1e-13:
                h = h_new
                f = f_new
                found = True
                break
            step *= 0.5
        if not found:
            break
    return h, f

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def _el():
        return time.time() - _t0

    # Phase 1: Top-K SLSQP at base resolution, iterate active set
    c = _corrs(best_h)
    for K in [3, 5, 8, 5]:
        if _el() > 50:
            break
        top_idx = np.argsort(c)[::-1][:K]
        hn = _slsqp_topk(best_h, top_idx, 3500, 1e-20, alpha=8000)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()
        c = _corrs(best_h)

    # Phase 2: Active descent + leveling at n0
    if _el() < budget_s - 250:
        for topm in [3, 5, 7]:
            if _el() > budget_s - 230:
                break
            hn, fc = _active_descent(best_h, topm, 6000, 0.004)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()
        hn, fc = _level_top(best_h, 4, 300)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase 3: Round-robin single-lag optimization
    if _el() < budget_s - 200:
        for _rr in range(6):
            if _el() > budget_s - 180:
                break
            hn, fc = _round_robin(best_h, 50, 10)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 4: Upsample to 2x, optimize at higher resolution
    if _el() < budget_s - 250:
        n1 = n0 * 2
        h2 = _upsample(best_h, 2)
        c2 = _corrs(h2)
        for K in [3, 5]:
            if _el() > budget_s - 200:
                break
            top_idx = np.argsort(c2)[::-1][:K]
            hn = _slsqp_topk(h2, top_idx, 2500, 1e-20, alpha=10000)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()
                best_n = n1
            c2 = _corrs(best_h)
        if _el() < budget_s - 150:
            for topm in [3, 5]:
                if _el() > budget_s - 130:
                    break
                hn, fc = _active_descent(best_h, topm, 4000, 0.003)
                if fc < best_c5:
                    best_c5 = fc
                    best_h = hn.copy()
            hn, fc = _level_top(best_h, 3, 200)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 5: Iterate SLSQP with re-identified active set
    for _it in range(5):
        if _el() > budget_s - 120:
            break
        nc = len(best_h)
        c = _corrs(best_h)
        top_idx = np.argsort(c)[::-1][:5]
        hn = _slsqp_topk(best_h, top_idx, 2000, 1e-21, alpha=12000)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()
        else:
            break

    # Phase 6: Refined active descent + round-robin
    if _el() < budget_s - 80:
        for topm in [3, 5]:
            if _el() > budget_s - 60:
                break
            hn, fc = _active_descent(best_h, topm, 5000, 0.002)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()
        if _el() < budget_s - 50:
            hn, fc = _round_robin(best_h, 30, 8)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 7: Fine pairwise moves (multiple scales)
    for scale, nit in [(0.003, 8000), (0.001, 12000), (0.0003, 15000), (0.00005, 15000)]:
        if _el() > budget_s - 15:
            break
        hn, fc = _pairwise_fine(best_h, nit, scale, seed + int(scale * 1e8))
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase 8: Multi-start from perturbed best
    nc = len(best_h)
    if _el() < budget_s - 40:
        for _r in range(10):
            if _el() > budget_s - 15:
                break
            hp = best_h + 0.004 * rng.randn(nc)
            hp = _repair(hp, nc)
            c = _corrs(hp)
            top_idx = np.argsort(c)[::-1][:5]
            hn = _slsqp_topk(hp, top_idx, 1200, 1e-20, alpha=10000)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase 9: Final polish
    if _el() < budget_s - 8:
        c = _corrs(best_h)
        top_idx = np.argsort(c)[::-1][:3]
        hn = _slsqp_topk(best_h, top_idx, 2000, 1e-21, alpha=15000)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()
        hn, fc = _active_descent(best_h, 3, 3000, 0.001)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
