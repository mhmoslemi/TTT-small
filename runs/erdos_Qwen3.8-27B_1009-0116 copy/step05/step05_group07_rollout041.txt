    if _rem() > 200:
        for tm in [5, 8, 12]:
            if _rem() < 180:
                break
            hn, fc = _active_descent(best_h, n0, _t0 + _rem() - 170, topm=tm, max_iter=4000)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase D: Simulated annealing for exploration
    if _rem() > 250:
        hn, fc = _sa_optimize(best_h, n0, _t0 + _rem() - 230,
                              t_init=0.03, t_final=0.0005, n_iter=1500000)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()
        # Post-SA polish
        if _rem() > 100:
            hn, fc = _momentum_descend(best_h, n0, _t0 + _rem() - 80, alpha=5000, top_k=10, max_iter=3000)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase E: Upsample to 2x and optimize
    n1 = n0 * 2
    if _rem() > 350:
        h2 = _upsample(best_h, 2)
        if _rem() > 300:
            hn = _slsqp_fast(h2, n1, 8000, 500, _t0 + _rem() - 280)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5; best_h = hn.copy(); best_n = n1
        if _rem() > 250:
            hn, fc = _momentum_descend(best_h, best_n, _t0 + _rem() - 230, alpha=3000, top_k=12, max_iter=5000)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()
        if _rem() > 200:
            hn, fc = _active_descent(best_h, best_n, _t0 + _rem() - 180, topm=10, max_iter=4000)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase F: Pairwise fine-tuning
    for scale, nit in [(0.01, 100000), (0.003, 150000), (0.001, 200000), (0.0003, 150000)]:
        if _rem() < 30:
            break
        hn, fc = _pairwise(best_h, best_n, _t0 + _rem() - 20, n_iter=nit, scale=scale)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    # Phase G: Multi-start exploration
    nc = len(best_h)
    if _rem() > 120:
        for r in range(6):
            if _rem() < 80:
                break
            hp = best_h + 0.03 * rng.randn(nc)
            hp = _repair(hp, nc)
            hn, fc = _momentum_descend(hp, nc, _t0 + _rem() - 60, alpha=3000, top_k=10, max_iter=2000)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase H: Final active descent + pairwise
    if _rem() > 30:
        hn, fc = _active_descent(best_h, best_n, _t0 + _rem() - 15, topm=8, max_iter=3000)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()
        hn, fc = _pairwise(best_h, best_n, _t0 + 5, n_iter=50000, scale=0.0005)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
