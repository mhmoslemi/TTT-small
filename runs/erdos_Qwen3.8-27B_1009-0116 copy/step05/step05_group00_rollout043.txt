def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, top_k, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(c, -top_k)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += dx * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= dx * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += dx * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= dx * h[i2 + sp]
    g /= top_k
    return g

def _proj_grad(h, tk, lr, max_it, t_lim):
    n = len(h)
    dx = 2.0 / n
    t0 = time.time()
    c5 = _eval(h)
    for _ in range(max_it):
        if time.time() - t0 > t_lim:
            break
        g = _topk_grad(h, tk, dx)
        step = lr
        moved = False
        for _ in range(6):
            h2 = _repair(h - step * g, n)
            c2 = _eval(h2)
            if c2 < c5 - 1e-15:
                h, c5 = h2, c2
                moved = True
                break
            step *= 0.5
        if not moved:
            break
    return h, c5

def _full_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _full_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_full_obj, h, jac=_full_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return _repair(np.interp(xn, xo, h_old), n_new)

def _binary_search(h, max_rounds, t_lim):
    n = len(h)
    h = (h > 0.3).astype(float)
    h = _repair(h, n)
    c5 = _eval(h)
    t0 = time.time()
    for _ in range(max_rounds):
        if time.time() - t0 > t_lim:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > t_lim:
                break
            old = h[i]
            h[i] = 1.0 - old
            h2 = _repair(h, n)
            c2 = _eval(h2)
            if c2 < c5 - 1e-13:
                c5 = c2
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    return h, c5

def _swap_search(h, max_rounds, t_lim):
    n = len(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    h = _repair(h, n)
    c5 = _eval(h)
    t0 = time.time()
    rng_local = np.random.RandomState(123)
    for _ in range(max_rounds):
        if time.time() - t0 > t_lim:
            break
        improved = False
        for _ in range(3 * n):
            if time.time() - t0 > t_lim:
                break
            i = ones[rng_local.randint(len(ones))]
            j = zeros[rng_local.randint(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            h2 = _repair(h, n)
            c2 = _eval(h2)
            if c2 < c5 - 1e-13:
                ones = np.where(h2 > 0.5)[0]
                zeros = np.where(h2 < 0.5)[0]
                h = h2
                c5 = c2
                improved = True
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def check():
        return time.time() - t0 < budget_s - 10

    # Phase 1: Fast projected gradient descent
    for tk in [1, 2, 3, 5]:
        if not check():
            break
        for lr in [0.08, 0.03, 0.01]:
            if time.time() - t0 > budget_s - 500:
                break
            h_pg, c5 = _proj_grad(best_h.copy(), tk, lr, 150, 25)
            if c5 < best_c5:
                best_c5, best_h = c5, h_pg.copy()

    # Phase 2: SLSQP with increasing sharpness
    for alpha in [3000, 20000, 100000]:
        if time.time() - t0 > budget_s - 350:
            break
        h_s = _slsqp(best_h, alpha, 2000, 1e-18)
        c5 = _eval(h_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h_s.copy()

    # Phase 3: Binary local search
    if check():
        h_b, c5 = _binary_search(best_h, 40, 60)
        if c5 < best_c5:
            best_c5, best_h = c5, h_b.copy()
        h_s = _slsqp(best_h, 50000, 1500, 1e-18)
        c5 = _eval(h_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h_s.copy()

    # Phase 4: Binary swap search
    if time.time() - t0 < budget_s - 250:
        h_sw, c5 = _swap_search(best_h, 25, 80)
        if c5 < best_c5:
            best_c5, best_h = c5, h_sw.copy()

    # Phase 5: Try different n values
    if time.time() - t0 < budget_s - 200:
        candidates = [100, 120, 150, 200, 240, 280, best_n + 20]
        for n_try in candidates:
            if time.time() - t0 > budget_s - 200:
                break
            if n_try < 50 or abs(n_try - best_n) < 5:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5 = _proj_grad(h_r, 3, 0.03, 80, 15)
            h_r = _slsqp(h_r, 50000, 800, 1e-17)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 6: Complement trick + re-optimize
    if time.time() - t0 < budget_s - 120:
        h_c = _repair(1.0 - best_h, best_n)
        h_c, c5 = _proj_grad(h_c, 3, 0.03, 100, 30)
        h_c = _slsqp(h_c, 100000, 1500, 1e-18)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5, best_h = c5, h_c.copy()

    # Phase 7: Periodic structure exploitation
    if time.time() - t0 < budget_s - 80:
        n = best_n
        for T in [5, 7, 10, 14, 20, 28, 35, 70]:
            if time.time() - t0 > budget_s - 80:
                break
            if n % T != 0:
                continue
            m = n // T
            p = np.zeros(T)
            for i in range(m):
                p += best_h[i * T:(i + 1) * T]
            p /= m
            hp = _repair(np.tile(p, m), n)
            hp = _slsqp(hp, 80000, 1000, 1e-18)
            c5 = _eval(hp)
            if c5 < best_c5:
                best_c5, best_h = c5, hp.copy()

    # Phase 8: Final SLSQP with high alpha
    if time.time() - t0 < budget_s - 30:
        h_f = _slsqp(best_h, 500000, 3000, 1e-20)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f.copy()

    # Phase 9: Fine coordinate descent
    if time.time() - t0 < budget_s - 5:
        h_cd = best_h.copy()
        nc = len(h_cd)
        c5c = _eval(h_cd)
        for scale in [0.004, 0.001, 0.0003]:
            if time.time() - t0 > budget_s - 5:
                break
            for _it in range(4000):
                if time.time() - t0 > budget_s - 5:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h_cd[i] + d, h_cd[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h_cd[i], h_cd[j] = ni, nj
                c5n = _eval(h_cd)
                if c5n < c5c - 1e-15:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h_cd.copy()
                else:
                    h_cd[i], h_cd[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
