import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

                    diff -= a

                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_corr(x, weights):

    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:

            g[i1] += wk * (1.0 - x[i1 + sh])

                g[i2] -= wk * x[i2 - sh]
        else:

            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g * (2.0 / n)  # dx factor

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha * (2.0/len(x))

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _grad_corr(x, w)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_w(x, w):
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _w_smooth(c, temp):
    c_norm = c - c.max()
    w = np.exp(c_norm / max(temp, 1e-8))
    s = w.sum()
    if s < 1e-30:
        return np.ones(len(c)) / len(c)
    return w / s

def _w_top(c, K, p):
    top_k = np.argsort(c)[-K:]
    w = np.zeros(len(c))
    vals = c[top_k]
    wmax = vals.max()
    if wmax > 1e-15:
        w[top_k] = (vals / wmax) ** p
    else:
        w[top_k] = 1.0
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[top_k] = 1.0 / K
    return w

def _fista(x0, w, n_iter, lr):
    n = len(x0)
    x = x0.copy()
    y = x0.copy()
    t = 1.0
    best = _eval(x)
    best_x = x.copy()
    for it in range(n_iter):
        g = _grad_w(y, w) * (2.0 / n)
        y_new = _repair(y - lr * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = y_new + ((t - 1.0) / t_new) * (y_new - x)
        x = y_new
        t = t_new
        if (it + 1) % 80 == 0:
            c5 = _eval(x)
            if c5 < best:
                best = c5
                best_x = x.copy()
    c5 = _eval(x)
    if c5 < best:
        best = c5
        best_x = x.copy()
    return best_x, best

def _lse_obj(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(float(np.sum(np.exp(s)))) / alpha) * (2.0 / n)

def _lse_grad(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _grad_w(x, w) * (2.0 / n)

def _wsum_obj(x, w, n):
    return float(np.dot(w, _corr(x))) * (2.0 / n)

def _wsum_grad(x, w, n):
    return _grad_w(x, w) * (2.0 / n)

def _eq_fun(x):
    return x.sum() - len(x) * 0.5

def _eq_jac(x):
    return np.ones(len(x))

def _do_slsqp(h0, fun, jac, args, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(fun, h0, jac=jac, args=args, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
