def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        if s > target:

        else:
            hi = mid

    sh = (n - 1) - k_idx

def _nesterov(h, max_iters, top_k, step_init):
    """Nesterov accelerated projected subgradient."""
    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()
    momentum = 0.0
    best_h = h.copy()
    best_c5 = _eval(h_curr)
    
    for it in range(max_iters):
        # Nesterov extrapolation
        momentum_new = (1 + momentum) / 2
        h_head = h_curr + (momentum - 1) / momentum_new * (h_curr - h_prev)
        h_head = np.clip(h_head, 0, 1)
        
        g = _subgrad_max(h_head, top_k)
        step = step_init / (1.0 + 0.1 * it)
        h_new = _proj_to_sum(h_head - step * g, n * 0.5)
        
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h_prev = h_curr
        h_curr = h_new
        momentum = momentum_new
    
    return best_h, best_c5

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pair_delta(i, j, h):
    n = len(h)
    lag = np.arange(-(n - 1), n)
    m1 = (i + lag >= 0) & (i + lag < n)
    m2 = (i - lag >= 0) & (i - lag < n)
    ci = np.zeros(2 * n - 1)
    ci[m1] = 1.0 - h[i + lag[m1]]
    ci[m2] -= h[i - lag[m2]]
    m3 = (j + lag >= 0) & (j + lag < n)
    m4 = (j - lag >= 0) & (j - lag < n)
    cj = np.zeros(2 * n - 1)
    cj[m3] = 1.0 - h[j + lag[m3]]
    cj[m4] -= h[j - lag[m4]]
    return ci - cj

def _pairwise(h, n_iters, rng):
    n = len(h)
    dx = 2.0 / n
    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
    c5c = float(c0.max())
    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi - 1e-12:
            continue
        a = _pair_delta(i, j, h)
        dg = dx * a
        grid = np.linspace(dlo, dhi, 64)
        vals = np.max(c0[None, :] + grid[:, None] * dg[None, :], axis=1)
        am = vals.argmin()
        lo_d = dlo if am == 0 else grid[max(0, am - 1)]
        hi_d = dhi if am == len(grid) - 1 else grid[min(len(grid) - 1, am + 1)]
        for _ in range(30):
            m1 = lo_d + (hi_d - lo_d) / 3.0
            m2 = hi_d - (hi_d - lo_d) / 3.0
            if np.max(c0 + m1 * dg) < np.max(c0 + m2 * dg):
                hi_d = m2
            else:
                lo_d = m1
        d = (lo_d + hi_d) * 0.5
        c5n = float(np.max(c0 + d * dg))
        if c5n < c5c - 1e-16:
            h[i] += d; h[j] -= d
            h = np.clip(h, 0, 1)
            c0 = c0 + d * dg
            c5c = c5n
    return h, c5c

def _slsqp_obj(x, n, dx, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = c.max()
    s = np.clip(alpha * (c - cm), -400, 400)
    s -= s.max()
    return cm + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_grad(x, n, dx, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = c.max()
    s = np.clip(alpha * (c - cm), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    n_thr = max(1, len(w) * 0.02)
    top_idx = np.argsort(w)[::-1][:n_thr]
    for k in top_idx:
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            ii = np.arange(n - sh)
            g[ii] += wk * (1.0 - x[ii + sh])
            if sh > 0:
                jj = np.arange(sh, n)
                g[jj] -= wk * x[jj - sh]
        else:
            sp = -sh
            ii = np.arange(sp, n)
            g[ii] += wk * (1.0 - x[ii - sp])
            jj = np.arange(n - sp)
            g[jj] -= wk * x[jj + sp]
    return g

def _slsqp(c, alpha, maxiter):
    n = len(c)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: _slsqp_obj(x, n, dx, alpha), c,
                       jac=lambda x: _slsqp_grad(x, n, dx, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(c, n)

def _active_descent(h, max_iters, n_top):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cv = c[idx]
        w = np.maximum(cv - cv.mean(), 0.0)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(n_top) / n_top
        else:
            w /= ws
        g = np.zeros(n)
        for wv, k in zip(w, idx):
            sh = (n - 1) - k
            if sh >= 0:
                ii = np.arange(n - sh)
                g[ii] += wv * (1.0 - h[ii + sh])
                if sh > 0:
                    jj = np.arange(sh, n)
                    g[jj] -= wv * h[jj - sh]
            else:
                sp = -sh
                ii = np.arange(sp, n)
                g[ii] += wv * (1.0 - h[ii - sp])
                jj = np.arange(n - sp)
                g[jj] -= wv * h[jj + sp]
        g *= dx
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
            hn = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(hn)
            if c5n < c5c - 1e-15:
                h = hn; c5c = c5n; improved = True; break
        if not improved:
            break
    return h, c5c

def _multi_start_slsqp(h, n_slsqp, rng, budget_end):
    t0 = time.time()
    n = len(h)
    best = h.copy(); bc = _eval(best)
    for _ in range(n_slsqp):

        h_new = _slsqp(hp, 300000, 600)
        c5 = _eval(h_new)
        if c5 < bc:
            bc = c5; best = h_new.copy()
    return best, bc
