def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _topm_grad(h, m, alpha_avg=10.0):

    n = len(h)

    top_idx = idx[:m]
    top_c = c[top_idx]
    # Softmax among top m

    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)

        wk = w[i] * dx
        sh = (n - 1) - k
        if sh >= 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g, c.max() * 2.0 / n

            best_h = h.copy()
    
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Top-m PG at n=280 (fast exploration)
    for m_val in [3, 5, 8, 12, 20, 32]:
        if time.time() - t0 > 150:
            break
        hh, c5 = _pg_solve(best_h, m_val, 30000, t0, budget_s, 0.15 / (m_val / 3 + 1), 0.004, 0.82)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    # Phase 2: SLSQP polish at n=280
    if time.time() - t0 < 250:
        hh = _slsqp_polish(best_h, 3000, 600, 1e-15)
        c5 = _eval(hh)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    # Phase 3: Upsample to 560 and PG
    if time.time() - t0 < 350:
        n1 = n0 * 2
        hh = _upsample(best_h, 2)
        hh = _repair(hh, n1)
        for m_val in [4, 8, 16, 32, 48]:
            if time.time() - t0 > 650:
                break
            hh, c5 = _pg_solve(hh, m_val, 20000, t0, budget_s, 0.14, 0.003, 0.82)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hh.copy()
                best_n = n1

    # Phase 4: SLSQP at higher resolution
    if time.time() - t0 < 800:
        hh = _slsqp_polish(best_h, 8000, 1000, 1e-15)
        c5 = _eval(hh)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    # Phase 5: Multi-start PG with perturbations
    nc = best_n
    if time.time() - t0 < 920:
        for _r in range(20):
            if time.time() - t0 > 950:
                break
            hp = best_h + 0.015 * rng.randn(nc)
            hp = _repair(hp, nc)
            hh, c5 = _pg_solve(hp, 16, 8000, t0, budget_s, 0.95, 0.004, 0.85)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hh.copy()

    # Phase 6: Fine pairwise local search
    if time.time() - t0 < 1060:
        hh = best_h.copy()
        nc = len(hh)
        c5c = _eval(hh)
        for scale in [0.003, 0.001, 0.0003]:
            for _it in range(8000):
                if time.time() - t0 > 1080:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = hh[i] + d, hh[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                hh[i], hh[j] = ni, nj
                c5n = _eval(hh)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = hh.copy()
                else:
                    hh[i], hh[j] = ni - d, nj + d

    # Phase 7: Final SLSQP polish
    if time.time() - t0 < 1095:
        hh = _slsqp_polish(best_h, 15000, 1500, 1e-16)
        c5 = _eval(hh)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
