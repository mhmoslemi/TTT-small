    best = (a + b) / 2.0
    bf = _f(best)
    for cand in (lo, hi, 0.0):
        if lo <= cand <= hi:
            fc = _f(cand)
            if fc < bf:
                bf = fc
                best = cand
    return best

def _pair_descent(h, n, t_budget, rng):
    t0 = time.time()
    best_h = h.copy()
    best_c, _ = _eval(best_h)
    n_pairs_per_sweep = 3
    recon_interval = 20
    updates_since_rebuild = 0
    G = _build_G(h, n)
    for sweep in range(10000):
        if time.time() - t0 > t_budget:
            break
        if updates_since_rebuild >= recon_interval:
            G = _build_G(h, n)
            updates_since_rebuild = 0
        _, c_full = _eval(h)
        order = rng.permutation(n)
        improved = False
        for j in order:
            if time.time() - t0 > t_budget:
                break
            for _try in range(n_pairs_per_sweep):
                m = rng.randint(n)
                if m == j:
                    continue
                d = G[j, :] - G[m, :]
                lo = max(-h[j], h[m] - 1.0)
                hi = min(1.0 - h[j], h[m])
                if lo >= hi:
                    continue
                delta = _opt_delta(c_full, d, lo, hi)
                if abs(delta) < 1e-12:
                    continue
                h[j] = max(0.0, min(1.0, h[j] + delta))
                h[m] = max(0.0, min(1.0, h[m] - delta))
                updates_since_rebuild += 1
                f_new, c_new = _eval(h)
                if f_new < best_c - 1e-14:
                    best_c = f_new
                    best_h = h.copy()
                    improved = True
                    c_full = c_new
                elif f_new > c_full.max() + 0.001:
                    # Too bad, revert
                    h[j] = max(0.0, min(1.0, h[j] - delta))
                    h[m] = max(0.0, min(1.0, h[m] + delta))
                    updates_since_rebuild -= 1
        if not improved:
            break
    return best_h, best_c

def _subgrad(h, n, t_budget, rng):
    t0 = time.time()
    best_h = h.copy()
    best_c, _ = _eval(best_h)
    step = 0.01
    decay = 0.999
    avg_h = h.copy()
    for it in range(50000):
        if time.time() - t0 > t_budget:
            break
        _, c = _eval(avg_h)
        idx = np.argwhere(c == c.max()).flatten()
        # Average gradient over active lags
        g = np.zeros(n)
        for k in idx:
            g += _coord_grad_single(avg_h, k, n)
        g /= len(idx)
        # Projected subgradient step
        avg_h = avg_h - step * g
        avg_h = _repair(avg_h, n)
        step *= decay
        f_new, _ = _eval(avg_h)
        if f_new < best_c:
            best_c = f_new
            best_h = avg_h.copy()
    return best_h, best_c
