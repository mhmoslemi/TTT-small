from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_active(h, K=20):
    n = len(h)
    c = _corr(h)
    mid = n - 1
    top = np.argsort(c)[::-1][:K]
    w = np.exp(80.0 * (c[top] - c[top].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top):
        s = int(ki) - mid
        if s == 0:
            continue
        if s > 0:
            m = n - s
            grad[s:n] += (1.0 - h[:m])
            grad[:m] -= h[s:n]
        else:
            s = -s
            m = n - s
            grad[:m] += (1.0 - h[s:n])
            grad[s:n] -= h[:m]
    grad -= grad.mean()
    return grad, c

def _grad_descent(h, dx, t_end, K=20):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr = 0.003
    patience = 0
    while time.time() < t_end:
        g, c = _grad_active(h, K)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        for _ in range(8):
            h_new = h - lr * g / gn
            h_new = _repair(h_new, tgt)
            cn = _c5(h_new, dx)
            if cn < best - 1e-12:
                h = h_new
                best = cn
                lr = min(lr * 1.02, 0.02)
                patience = 0
                break
            else:
                lr *= 0.75
        else:
            patience += 1
            if patience > 40:
                break
    return h, best

def _slsqp(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def obj_grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        p = np.exp(s)
        p /= p.sum()
        # gradient of log-sum-exp part
        # d/dc_k [lse] = p_k
        # d c_k / d x_j (chain rule)
        g = np.zeros(n)
        for ki in range(len(c)):
            s_ki = int(ki) - (n - 1)
            if s_ki == 0 or p[ki] < 1e-10:
                continue
            if s_ki > 0:
                m2 = n - s_ki
                g[s_ki:n] += p[ki] * (1.0 - x[:m2])
                g[:m2] -= p[ki] * x[s_ki:n]
            else:
                s2 = -s_ki
                m2 = n - s2
                g[:m2] += p[ki] * (1.0 - x[s2:n])
                g[s2:n] -= p[ki] * x[:m2]
        return g
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq, 'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

        improved = False
        for _ in range(12):
            h_new = _repair(np.clip(h - lr * g / gn, 0, 1), tgt)
            cn = _c5(h_new, dx)
            if cn < best - 1e-14:
                h, best = h_new, cn
                lr = min(lr * 1.015, 0.025)
                improved = True
                break
            lr *= 0.72
        if not improved:
            stall += 1
            if stall > 60:
                break
    return h, best

def _pair_search(h, dx, rng, t_end, K=8):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    step = 0.015
    while time.time() < t_end:
        c = _corr(h)
        mid = n - 1
        top = np.argsort(c)[::-1][:K]
        found = False
        for ki in top:
            s = int(ki) - mid
            if s == 0:
                continue
            ss = abs(s)
            m = n - ss
            if s > 0:
                scores = h[ss:n] * (1.0 - h[:m])
            else:
                scores = h[:m] * (1.0 - h[ss:n])
            order = np.argsort(scores)[::-1][:25]
            rng.shuffle(order)
            for a in order:
                if s > 0:
                    i1, i2 = a + ss, a
                else:
                    i1, i2 = a, a + ss
                for d in [step, -step, step * 0.5]:
                    if s > 0:
                        h[i1] -= d; h[i2] += d
                    else:
                        h[i1] += d; h[i2] -= d
                    if 0 <= h[i1] and h[i1] <= 1 and 0 <= h[i2] and h[i2] <= 1:
                        cn = _c5(h, dx)
                        if cn < best - 1e-14:
                            best = cn
                            found = True
                        else:
                            if s > 0:
                                h[i1] += d; h[i2] -= d
                            else:
                                h[i1] -= d; h[i2] += d
                            break
                    else:
                        if s > 0:
                            h[i1] += d; h[i2] -= d
                        else:
                            h[i1] -= d; h[i2] += d
                        break
                if found:
                    break
            if found:
                step = min(step * 1.03, 0.04)
    return h, best

def _slsqp(h, dx, alpha, maxiter=250):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        se = np.exp(s)
        return m + float(np.log(np.sum(se))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq, 'jac': lambda x: ones}],
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.05 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(4, n // 4), replace=False)
        hp[idx] += 0.1 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        f = rng.uniform(1, 6)
        ph = rng.uniform(0, 6.28)
        hp += 0.04 * np.sin(2 * np.pi * f * x + ph)
    else:
        idx = rng.choice(n, max(3, n // 6), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def _fine(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.0015, 0.0008, 4e-4, 2e-4, 8e-5, 3e-5, 1e-5]
    for sc in scales:
        while time.time() < t_end:
            for _ in range(3):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = sc * (2.0 * rng.random() - 1.0)
                h[i] += d; h[j] -= d
                if 0 <= h[i] and h[i] <= 1 and 0 <= h[j] and h[j] <= 1:
                    cn = _c5(h, dx)
                    if cn < best - 1e-15:
                        best = cn
                    else:
                        h[i] -= d; h[j] += d
                else:
                    h[i] -= d; h[j] += d
    return h, best

def _interp_h(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
                len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    # Phase 1: Gradient descent (fast, broad)
    h, c = _gdescent(best_h, dx, t0 + 120, K=20)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 2: SLSQP multi-alpha
    for a in [500, 2000, 8000]:
        if time.time() > t0 + 250:
            break
        h, c = _slsqp(best_h, dx, a, 250)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: Targeted pair search
    h, c = _pair_search(best_h, dx, rng, t0 + 450, K=10)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 4: Multi-start
    t4_end = t0 + 650
    for r in range(12):
        if time.time() > t4_end or budget_s - (time.time() - t0) < 280:
            break
        hp = _perturb(best_h, rng, r % 4)
        hp, c = _gdescent(hp, dx, min(t4_end - 3, time.time() + 10), K=15)
        hp, c = _slsqp(hp, dx, 5000, 200)
        if c < best_c5:
            best_h, best_c5 = hp, c
            hp, c = _pair_search(hp, dx, rng, min(t4_end - 1, time.time() + 8), K=6)
            if c < best_c5:
                best_h, best_c5 = hp, c

    # Phase 5: Higher resolution
    t5_end = t0 + 850
    if budget_s - (time.time() - t0) > 270:
        n2, dx2 = 2 * n, 1.0 / n
        h2 = _repair(_interp_h(best_h, n2), n2 / 2.0)
        h2, c2 = _gdescent(h2, dx2, min(t5_end - 80, time.time() + 80), K=15)
        h2, c2 = _slsqp(h2, dx2, 3000, 200)
        h2, c2 = _pair_search(h2, dx2, rng, t5_end, K=8)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # Phase 6: Final targeted refinement
    nb, dxb = best_n, 2.0 / best_n
    best_h, best_c5 = _pair_search(best_h, dxb, rng, t0 + 1000, K=6)

    # Phase 7: Fine search
    best_h, best_c5 = _fine(best_h, dxb, rng, t0 + 1070)

    # Phase 8: Final SLSQP polish
    if budget_s - (time.time() - t0) > 15:
        h, c = _slsqp(best_h, dxb, 20000, 150)
        if c < best_c5:
            best_h, best_c5 = h, c
    if budget_s - (time.time() - t0) > 4:
        best_h, best_c5 = _fine(best_h, dxb, rng, t0 + budget_s - 1)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
