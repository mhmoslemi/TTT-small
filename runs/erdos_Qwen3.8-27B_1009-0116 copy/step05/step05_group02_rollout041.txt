def _proj(h, n):

def _grad_comp(h, k):

    g = np.zeros(n)
    sh = (n - 1) - k

        if sh == 0:
            g[:] += (1.0 - h[:])
        else:

            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _lse_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    cs = c * dx
    m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    cs = c * dx
    m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for k in nz:
        g += w[k] * _grad_comp(h, k)
    return g * dx

def _lp_obj(h, p):

    dx = 2.0 / len(h)
    cs = np.maximum(c * dx, 1e-30)
    return (np.sum(cs**p))**(1.0/p)

def _lp_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cs = np.maximum(c * dx, 1e-30)
    sp = np.sum(cs**p)
    norm_val = sp**(1.0/p)
    # d/dh_j of (sum cs^p)^(1/p) = (1/p) * (sum cs^p)^(1/p - 1) * sum p * cs_k^(p-1) * dc_k/dh_j
    # = (sum cs^p)^(1/p-1) * sum cs_k^(p-1) * dc_k/dh_j
    coeff = sp**(1.0/p - 1.0)
    weights = cs**(p-1)
    g = np.zeros(n)
    nz = np.where(weights > 1e-12)[0]
    for k in nz:
        g += weights[k] * _grad_comp(h, k)
    return g * dx * coeff

    except Exception:

    dx = 2.0 / n
    h_curr = h.copy()
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    best = _eval(h_curr)
    best_h = h_curr.copy()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for t in range(n_iter):
        c = np.correlate(h_curr, 1.0 - h_curr, mode='full') * dx
        # Top-k selection
        if top_k >= len(c):
            idx = np.arange(len(c))
            weights = np.ones(len(c)) / len(c)
        else:
            idx = np.argsort(c)[-top_k:]
            vals = c[idx]
            # Softmax weighting within top-k
            w = np.exp(50 * (vals - vals.max()))
            weights = w / w.sum()
        
        g = np.zeros(n)
        for i, k in enumerate(idx):
            g += weights[i] * _grad_comp(h_curr, k)
        g *= dx
        
        lr = base_lr / (1 + t * 0.001)
        m_t = beta1 * m_t + (1 - beta1) * g
        v_t = beta2 * v_t + (1 - beta2) * g**2
        m_hat = m_t / (1 - beta1**(t+1))
        v_hat = v_t / (1 - beta2**(t+1))
        
        h_new = _proj(h_curr - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        c5 = _eval(h_new)

            best = c5
            best_h = h_new.copy()
            h_curr = h_new

            c5_try = _eval(h_try)
            if c5_try < best:
                best = c5_try
                best_h = h_try.copy()
                h_curr = h_try
    
    return best_h, best

            if time.time() - t0 > budget_s - 160:
                break
            fn_o = partial(_lp_obj, p=p)
            fn_g = partial(_lp_grad, p=p)
            h_new = _slsqp(best_h, fn_o, fn_g, 1500, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 5: Multi-start with diverse strategies
    nc = len(best_h)
    if time.time() - t0 < budget_s - 150:
        for r in range(35):
            if time.time() - t0 > budget_s - 100:
                break
            strat = r % 6
            if strat == 0:
                hp = _repair(best_h + 0.01 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.005 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                n_blocks = rng.randint(2, 5)
                for _ in range(n_blocks):
                    bs = rng.randint(10, max(11, nc // 5))
                    st = rng.randint(0, nc - bs)
                    hp[st:st+bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 3:
                f = rng.randint(2, nc // 4 + 1)
                t_arr = np.arange(nc)
                amp = 0.012 * rng.rand() * (1.0 if rng.rand() < 0.5 else -1.0)
                hp = best_h + amp * np.sin(2.0*np.pi*f*t_arr/nc + rng.rand()*2.0*np.pi)
                hp = _repair(hp, nc)
            elif strat == 4:
                hp = best_h.copy()
                n_flip = max(5, nc // 15)
                idx = rng.choice(nc, n_flip, replace=False)
                hp[idx] = 1.0 - hp[idx]
                hp = _repair(hp, nc)
            else:
                hp = _repair(best_h * rng.uniform(0.85, 1.15) + 0.008 * rng.randn(nc), nc)

            fn_o = partial(_lse_obj, alpha=20000)
            fn_g = partial(_lse_grad, alpha=20000)
            h_new = _slsqp(hp, fn_o, fn_g, 800, 1e-17)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 6: ADAM fine-tuning with smaller steps
    if time.time() - t0 < budget_s - 80:
        for lr_try in [0.0015, 0.0008, 0.0004]:
            if time.time() - t0 > budget_s - 30:
                break
            dl = lambda: (time.time() - t0 > budget_s - 25)
            h_a, c5a = _adam_subgrad(best_h, 4000, lr_try, 5, dl)
            if c5a < best_c5:
                best_c5 = c5a
                best_h = h_a.copy()

    # Phase 7: Coordinate descent fine-tuning
    if time.time() - t0 < budget_s - 15:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for scale in [0.0008, 0.0002, 5e-5, 1e-5]:
            if time.time() - t0 > budget_s - 8:
                break
            for _it in range(10000):
                if time.time() - t0 > budget_s - 8:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 8: Final SLSQP polish
    if time.time() - t0 < budget_s - 3:
        fn_o = partial(_lse_obj, alpha=150000)
        fn_g = partial(_lse_grad, alpha=150000)
        h_new = _slsqp(best_h, fn_o, fn_g, 2500, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
