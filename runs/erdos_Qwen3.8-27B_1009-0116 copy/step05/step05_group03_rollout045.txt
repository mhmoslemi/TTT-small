import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp_sm(h, dx, alpha, maxiter=300):
    """SLSQP with soft-max (log-sum-exp) objective."""
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    
    def eq(x):
        return x.sum() - tgt
    
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _subgrad_dir(h, top_k=5, sharp=80.0):
    """Compute weighted subgradient focusing on top lags."""
    n = len(h)
    c = _corr(h)
    top = np.argsort(c)[::-1][:top_k]
    w = np.exp(sharp * (c[top] - c[top].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top):
        s = abs(int(ki - (n - 1)))
        if s == 0:
            continue
        m = n - s
        g = np.zeros(n)
        g[s:n] += 1.0 - h[:m]
        g[:m] -= h[s:n]
        grad += wi * g
    grad -= grad.mean()  # ensure zero sum for constraint
    return grad

def _sgd_momentum(h, dx, t_end, lr0=0.003, momentum=0.9):
    """Subgradient descent with momentum."""
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    v = np.zeros(n)
    lr = lr0
    for _ in range(5000000):
        if time.time() > t_end:
            break
        g = _subgrad_dir(h)
        v = momentum * v - lr * g
        h_new = _repair(np.clip(h + v, 0, 1), tgt)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new
            h = h_new
            lr = min(lr * 1.005, 0.02)
        else:
            v *= 0.5
            lr *= 0.95
            if lr < 1e-7:
                break
    return h, best

def _block_search(h, dx, rng, t_end, block_sizes=None):
    """Targeted search with block moves on worst lags."""
    if block_sizes is None:
        block_sizes = [1, 2, 3, 5]
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    tgt = n / 2.0
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:4]
        # Try block moves for each top lag
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            for bs in block_sizes:
                if bs > m // 2:
                    continue
                for _ in range(4):
                    a = rng.randint(0, m - bs + 1)
                    d = rng.uniform(0.003, 0.015)
                    # Move mass from block at [a+bs, a+2bs] to [a, a+bs]
                    # or vice versa, within the lag's affected region
                    i1_start = a + bs
                    i2_start = a
                    if i1_start + bs > n or i2_start + bs > n:
                        continue
                    valid = True
                    for b2 in range(bs):
                        if h[i1_start + b2] < d / bs or h[i2_start + b2] > 1 - d / bs:
                            valid = False
                            break
                    if not valid:
                        continue
                    for b2 in range(bs):
                        h[i1_start + b2] -= d / bs
                        h[i2_start + b2] += d / bs
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                    else:
                        for b2 in range(bs):
                            h[i1_start + b2] += d / bs
                            h[i2_start + b2] -= d / bs
        # Random point moves
        for _ in range(6):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.008, 0.008)
            h[i] += d
            h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
    return h, best

def _directional_fine(h, dx, t_end, top_k=3):
    """Fine search along subgradient directions."""
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    tgt = n / 2.0
    scales = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]
    for sc in scales:

                cn = _c5(h_new, dx)
                if cn < best:
                    best = cn
                    h = h_new
                    break
    return h, best
