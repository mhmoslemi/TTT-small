```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return float(x.sum() - 0.5 * len(x))

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _tile_obj(x, T, n, alpha):
    h = np.tile(x, n // T)
    return _obj(h, alpha)

def _tile_grad(x, T, n, alpha):
    h = np.tile(x, n // T)
    gf = _grad(h, alpha)
    m = n // T
    g = gf[:T].copy()
    for p in range(1, m):
        g += gf[p * T:(p + 1) * T]
    return g / m

def _slsqp_tile(h_full, T, alpha, maxiter, ftol):
    n = len(h_full)
    if n % T != 0:
        return h_full.copy()
    hp = h_full[:T].copy()
    bnds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_tile_obj, hp, jac=_tile_grad, args=(T, n, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.tile(_repair(res.x, T), n // T)
    except Exception:
        return h_full.copy()

def _binary_search(h, n, rng, max_rounds, time_limit_fn):
    h_b = (h > 0.5).astype(float)
    h_b = _repair(h_b, n)
    c5c = _eval(h_b)
    ones = np.where(h_b > 0.5)[0]
    zeros = np.where(h_b < 0.5)[0]
    for rnd in range(max_rounds):
        if time_limit_fn():
            break
        improved = False
        for _ in range(2000):
            if time_limit_fn():
                break
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h_b[i] = 1.0
            h_b[j] = 0.0
            c2 = _eval(h_b)
            if c2 < c5c - 1e-13:
                c5c = c2
                improved = True
                ones = np.where(h_b > 0.5)[0]
                zeros = np.where(h_b < 0.5)[0]
            else:
                h_b[i] = 0.0
                h_b[j] = 1.0
        if not improved:
            break
    return h_b, c5c
