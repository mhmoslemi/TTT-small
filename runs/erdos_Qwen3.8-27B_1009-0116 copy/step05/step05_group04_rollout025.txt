import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def _obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha

    def _grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g

    def _c_fun(x):
        return x.sum() - n * 0.5

    def _c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topk_descent(h, K, n_rounds):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for _r in range(n_rounds):
        c = _corr(h)
        topk = np.argsort(c)[-K:]
        g = np.zeros(n)
        for k in topk:
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= h[i2 + sp]
        g *= dx / K
        improved = False
        for step in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5

def _pairwise_search(h, n_trials, rng, time_limit=None, t_start=None):
    n = len(h)
    c5 = _eval(h)
    step = 0.015
    for _ in range(n_trials):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5:
            c5 = c5n
            step = min(step * 1.05, 0.04)
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            step = max(step * 0.97, 0.003)
    return h, c5

def _coord_descent(h, n_passes, rng):
    n = len(h)
    c5 = _eval(h)
    for _p in range(n_passes):
        idx = np.random.permutation(n)
        for i in idx:
            old = h[i]
            best_v = old
            best_c = c5
            for frac in [0.98, 1.02, 0.95, 1.05, 0.9, 1.1, 0.85, 1.15]:
                v = old * frac
                if v < 0 or v > 1:
                    continue
                h[i] = v
                h = _repair(h, n)
                c5n = _eval(h)
                if c5n < best_c:
                    best_c = c5n
                    best_v = v
            h[i] = best_v
            c5 = best_c
    return h, c5

def _fft_restart(h, n_modes, rng):
    n = len(h)
    H = np.fft.rfft(h)
    for k in range(1, min(n_modes, len(H) - 1)):
        H[k] *= np.exp(1j * rng.rand() * 2 * np.pi)
        H[n - k] = np.conj(H[k])
    h_new = np.fft.irfft(H, n)
    return _repair(h_new, n)

        c5 = _eval(h_c)

    for it in range(8):

        # SLSQP polish

            c5 = _eval(h_c)
            if c5 < best_c5:
