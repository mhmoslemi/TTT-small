import numpy as np

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _repair(h, n):
    # clip and adjust sum
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smoothed_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    cx = c - c.max()
    return c.max() + np.log(np.sum(np.exp(alpha * cx))) / alpha

def _grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    s = alpha * (c - c.max())
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:

    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    prev_g = None
    prev_h = None
    lr_bb = lr

        g = _grad(h, alpha)
        # Barzilai-Borwein step size
        if prev_g is not None and prev_h is not None:
            s = h - prev_h
            y = g - prev_g
            sy = s @ y
            yy = y @ y
            ss = s @ s
            if yy > 1e-16:
                if sy > 0:
                    lr_bb = sy / yy  # BB1
                else:
                    lr_bb = ss / sy if sy != 0 else lr_bb  # BB2 (fallback)
                lr_bb = np.clip(lr_bb, 1e-8, 1.0)
        prev_g = g.copy()
        prev_h = h.copy()
        # Gradient step
        h_new = _repair(h - lr_bb * g, n)
        c5_new = _eval(h_new)
        if c5_new < c5 - 1e-15:
            h = h_new
            c5 = c5_new
            if np.linalg.norm(g, np.inf) < tol:
                break
        else:
            # Backtrack
            lr_bb *= 0.5
            h_new = _repair(h - lr_bb * g, n)
            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-15:
                h = h_new
                c5 = c5_new
            else:
                lr_bb = max(lr_bb, 1e-10)
    return h, c5

def _lbfgs_smoothed(h, alpha, n_iter):
    n = len(h)
    def obj(hh):
        return _smoothed_obj(hh, alpha)
    def grad(hh):
        return _grad(hh, alpha)
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': n_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    return _repair(res.x, n), _eval(_repair(res.x, n))

def _bang_bang_local(h, n_rounds, rng):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    for _ in range(n_rounds):
        i = rng.randint(n)
        old = h[i]
        h[i] = 1.0 if old > 0.5 else 0.0
        h = _repair(h, n)
        c2 = _eval(h)
        if c2 < c5 - 1e-14:
            c5 = c2
        else:
            h[i] = old
            h = _repair(h, n)
    return h, c5

    xn = np.linspace(0, 2.0, n_new, endpoint=False)
