```python
def _compute_grad_matrix(h):
    """Compute G[k,i] = dc_k/dh_i for all k, i. Shape: (2n-1, n)."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    # k = 0 corresponds to shift n-1
    # k = n-1 corresponds to shift 0
    # k = 2n-2 corresponds to shift -(n-1)
    for s in range(0, n):
        # Shift s >= 0, k = n-1-s
        k = n - 1 - s
        G[k, 0:n-s] += (1.0 - h[s:n])
        if s > 0:
            G[k, s:n] -= h[0:n-s]
        # Shift -s, k = n-1+s (only if s > 0)
        if s > 0:
            k2 = n - 1 + s
            G[k2, s:n] += (1.0 - h[0:n-s])
            G[k2, 0:n-s] -= h[s:n]
    return G
```

def _compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    # Lag 0: output index n-1
    # c_0 = sum_i h[i](1-h[i])
    # dc_0/dh[i] = (1-h[i]) - h[i] = 1 - 2*h[i]
    G[n-1, :] = 1.0 - 2.0 * h
    
    for s in range(1, n):
        # Lag s (positive): output index n-1+s
        # c_s = sum_{n=0}^{n-1-s} h[n+s](1-h[n])
        # dc_s/dh[i] = (1-h[i-s])*1[s<=i<=n-1] - h[i+s]*1[0<=i<=n-1-s]
        k_pos = n - 1 + s
        G[k_pos, s:n] += (1.0 - h[0:n-s])  # (1-h[i-s]) for i from s to n-1
        G[k_pos, 0:n-s] -= h[s:n]           # -h[i+s] for i from 0 to n-1-s
        
        # Lag -s: output index n-1-s
        # c_{-s} = sum_{n=s}^{n-1} h[n-s](1-h[n])
        # dc_{-s}/dh[i] = (1-h[i+s])*1[0<=i<=n-1-s] - h[i-s]*1[s<=i<=n-1]
        k_neg = n - 1 - s
        G[k_neg, 0:n-s] += (1.0 - h[s:n])   # (1-h[i+s]) for i from 0 to n-1-s
        G[k_neg, s:n] -= h[0:n-s]           # -h[i-s] for i from s to n-1
    
    return G

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    G[n - 1, :] = 1.0 - 2.0 * h
    for s in range(1, n):
        kp = n - 1 + s
        G[kp, s:n] += (1.0 - h[0:n - s])
        G[kp, 0:n - s] -= h[s:n]
        kn = n - 1 - s
        G[kn, 0:n - s] += (1.0 - h[s:n])
        G[kn, s:n] -= h[0:n - s]
    return G

def _coord_descent(h, max_sweeps, grid_size):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    tg = np.linspace(0.0, 1.0, grid_size)
    stall = 0
    for sw in range(max_sweeps):
        n_imp = 0
        c_base = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        for i in range(n):
            old = h[i]
            gi = G[:, i]
            b = c_base - old * gi
            vals = b[None, :] + tg[:, None] * gi[None, :]
            mv = vals.max(axis=1)
            bi = np.argmin(mv)
            bt = tg[bi]
            if abs(bt - old) > 1e-10:
                h[i] = bt
                n_imp += 1
        if n_imp > 0:
            h = _proj_to_sum(h, n * 0.5)
            c5n = _eval(h)
            if c5n < c5 - 1e-14:
                c5 = c5n
                stall = 0
            else:
                stall += 1
        else:
            break
        if stall > 20:
            break
    return h, c5

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, lr, max_iters):
    n = len(h)
    y = h.copy()
    v = np.zeros(n)
    c5 = _eval(h)
    best_y = h.copy()
    best_c5 = c5
    mu = 0.9
    for it in range(max_iters):
        g = _smooth_grad(y, alpha)
        y = y + mu * v - lr * g
        y = _proj_to_sum(y, n * 0.5)
        v = y - best_y + v * 0  # Nesterov: y_new = y + mu*v - lr*g, v_new = y_new - y
        if it % 200 == 0:
            c5n = _eval(y)
            if c5n < best_c5:
                best_c5 = c5n
                best_y = y.copy()
                c5 = best_c5
    return best_y, best_c5

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent(h, max_iters, n_top):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += w * (1.0 - h[sh:n])
                if sh > 0:
                    g[sh:] -= w * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += w * (1.0 - h[:n - sp])
                g[:n - sp] -= w * h[sp:]
        g *= dx
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            hn = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(hn)
            if c5n < c5c - 1e-15:
                h = hn
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def _update(hn, cn):
        nonlocal best_h, best_c5, best_n
        if cn < best_c5:
            best_c5 = cn
            best_h = hn.copy()
            best_n = len(hn)

    # Phase 1: Coordinate descent (fast, many sweeps)
    if _el() < budget_s - 400:
        h1, c1 = _coord_descent(best_h.copy(), 20000, 65)
        _update(h1, c1)

    # Phase 2: Nesterov on smooth approximation
    if _el() < budget_s - 350:
        for alpha in [2000, 10000, 50000, 200000]:
            if _el() > budget_s - 320:
                break
            lr = 0.5 / (alpha * n0 * 0.05)
            h2, c2 = _nesterov(best_h.copy(), alpha, lr, 3000)
            _update(h2, c2)

    # Phase 3: Coarse-to-fine coordinate descent
    if _el() < budget_s - 300:
        h3, c3 = _coord_descent(best_h.copy(), 15000, 129)
        _update(h3, c3)
    if _el() < budget_s - 200:
        h3, c3 = _coord_descent(best_h.copy(), 10000, 257)
        _update(h3, c3)

    # Phase 4: SLSQP polish
    if _el() < budget_s - 200:
        for alpha in [50000, 200000, 1000000]:
            if _el() > budget_s - 170:

            h4 = _slsqp(best_h.copy(), alpha, 1500, 1e-20)
            c4 = _eval(h4)
            _update(h4, c4)

    # Phase 5: Active set descent

        _update(h5, c5)
    if _el() < budget_s - 60:

            if strat == 0:

            elif strat == 1:

        _update(h8, c8)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
