from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _subgrad_topk(h, K=5, temp=50.0):

def _proj_subgrad(h, dx, t_end, K=5, lr0=0.01):
    n = len(h)
    tgt = n / 2.0
    best_c = _c5(h, dx)
    h = h.copy()

def _slsqp_smooth(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def jac(x):
        c = _corr(x)
        m = c.max()
        s = alpha * (c - m)
        s = np.clip(s, -400, 400)
        e = np.exp(s)
        Z = e.sum()
        # grad of log-sum-exp part: alpha * e / Z * (_dc/dx - dmax/dx)
        # For simplicity, use numerical or just objective (SLSQP can handle)
        return None
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt}],

def _targeted_pairs(h, dx, rng, t_end, n_iter_per_round=100):

            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            # Try moving mass to reduce this lag
            for _ in range(n_iter_per_round // 6):
                if time.time() >= t_end:
                    break
                a = rng.randint(m)
                i1 = a + s  # h[i1] is the "source" in the product

def _block_move(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    block_sizes = [2, 3, 4, 5, 8]
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:4]
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0 or s >= n:
                continue
            for bs in block_sizes:
                if s + bs > n:
                    continue
                a = rng.randint(n - s - bs + 1)
                # Move block of size bs from position a to position a+s (or vice versa)
                d = rng.uniform(0.01, 0.05)
                src = slice(a, a + bs)
                dst = slice(a + s, a + s + bs)
                h[src] -= d
                h[dst] += d
                if h.min() >= 0 and h.max() <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                    else:
                        h[src] += d
                        h[dst] -= d
                else:
                    h[src] += d
                    h[dst] -= d
    return h, best

Also, I want to add a key improvement: **conjugate move**. For the single worst lag s, the optimal move (within the constraint manifold) is to shift mass in a way that most reduces O_s. The steepest descent direction for O_s (projected onto the constraint sum(h)=constant) is:

- The bounds constraint gives α ∈ [α_min, α_max]

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-15:
                break
    return h

def _grad_lag(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        return 1.0 - 2.0 * h
    m = n - s
    g[s:] += 1.0 - h[:m]
    g[:m] -= h[s:]
    return g

def _subgrad(h, K, temp):
    n = len(h)
    c = _corr(h)
    top = np.argsort(c)[::-1][:K]
    tc = c[top]
    w = np.exp(temp * (tc - tc.max()))
    w /= w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, top):
        s = abs(int(ki - (n - 1)))
        g += wi * _grad_lag(h, s)
    g -= g.mean()
    return g

def _subgrad_ls(h, dx, t_end, K=5, temp=50.0, max_steps=50000):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    for step in range(max_steps):
        if time.time() > t_end:
            break
        g = _subgrad(h, K, temp)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        # Line search: binary search for best alpha in [0, alpha_max]
        pos = d > 0
        neg = d < 0
        a_max = 1.0
        if pos.any():
            a_max = min(a_max, ((1.0 - h[pos]) / d[pos]).min())
        if neg.any():
            a_max = min(a_max, (h[neg] / (-d[neg])).min())
        a_max = min(a_max, 0.3)
        # Try a few alphas
        best_a = 0.0
        best_c_local = best
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a = a_max * frac
            if a < 1e-10:
                continue
            h_try = _repair(h + a * d, tgt)
            c_try = _c5(h_try, dx)
            if c_try < best_c_local:
                best_c_local = c_try
                best_a = a
        if best_a > 0 and best_c_local < best:
            h = _repair(h + best_a * d, tgt)
            best = best_c_local
        else:
            # Reduce K or break
            if K > 1:
                K = max(1, K - 1)
            else:
                break
    return h, best

def _slsqp(h, dx, alpha, maxit):
    n = len(h)
    tgt = n / 2.0
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -350, 350)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt}],
                     options={'maxiter': maxit, 'ftol': 1e-14})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _pairs(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:5]
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            for _ in range(8):
                a = rng.randint(m)
                i1, i2 = a + s, a
                d = rng.uniform(0.002, 0.02)
                h[i1] -= d; h[i2] += d
                if 0 <= h[i1] and h[i1] <= 1 and 0 <= h[i2] and h[i2] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                    else:
                        h[i1] += d; h[i2] -= d
                else:
                    h[i1] += d; h[i2] -= d
                if time.time() > t_end:
                    break
        # Random pair
        for _ in range(4):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.01, 0.01)
            h[i] += d; h[j] -= d
            if 0 <= h[i] and h[i] <= 1 and 0 <= h[j] and h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best
