        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _lag_grad(h, k_lag, n):
    dx = 2.0 / n
    sh = (n - 1) - k_lag
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):

                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:

def _targeted_descent(h, n_rounds, time_limit):
    """Multi-scale lag-targeted descent."""
    n = len(h)
    c5 = _eval(h)
    c = _corr(h)
    for _r in range(n_rounds):
        if time.time() > time_limit:
            break
        c = _corr(h)
        # Get top 5 lags
        top5 = np.argsort(c)[-5:]
        # Try single lag descent
        for k_w in [top5[-1], top5[-2], top5[-3]]:
            g = _lag_grad(h, k_w, n)
            for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
                h_new = _proj_to_sum(h - step * g, n * 0.5)
                c5n = _eval(h_new)
                if c5n < c5 - 1e-13:
                    h = h_new
                    c5 = c5n
                    break
        # Try multi-lag average
        g_multi = np.zeros(n)
        for k_w in top5:
            g_multi += _lag_grad(h, k_w, n)
        g_multi /= len(top5)
        for step in [0.03, 0.015, 0.007, 0.003, 0.001]:
            h_new = _proj_to_sum(h - step * g_multi, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-13:
                h = h_new
                c5 = c5n
                break
    return h, c5

    for i in range(n_iter):

            break

            c = _corr(h)
        t = t0 * (t1 / t0) ** (i / max(1, n_iter - 1))
        c_max = c.max()

        w /= w.sum()
        if rng.rand() < 0.7:

        else:
            k_pick = rng.randint(len(c))

            h = h_new
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c

def _nesterov(h, alpha, steps, step0, time_limit):

    v = np.zeros(n)

    step = step0

    for _s in range(steps):
        if time.time() > time_limit:

        else:
            step *= 0.4
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _smooth_obj(h, alpha)
    return h

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    h[idx] += a
                    diff -= a

                    break

            g_count += 1
            if g_count % 50 == 1 or g_cache is None:
                c = _corr(h_f)
                k_w = c.argmax()
                g_cache = _lag_grad(h_f, k_w, nc)
                g_abs = np.abs(g_cache)
                if g_abs.sum() > 1e-15:
                    g_probs = g_abs / g_abs.sum()
                else:
                    g_probs = np.full(nc, 1.0 / nc)
            if rng.rand() < 0.7:
                i = rng.choice(nc, p=g_probs)
                j = rng.choice(nc, p=g_probs)
                if i == j:
                    j = (j + 1) % nc
                d = 4e-5 if g_cache[i] >= g_cache[j] else -4e-5
            else:
                i, j = rng.randint(nc), rng.randint(nc)
                if i == j:
                    continue
                d = 3e-5 * rng.randn()
            ni, nj = h_f[i] + d, h_f[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_f[i], h_f[j] = ni, nj
                c5n = _eval(h_f)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h_f.copy()
                        best_n = nc
                else:
                    h_f[i], h_f[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
