def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

        if s > target:
            lo = mid
        else:

            g += _grad_for_lag(s, h)
        g /= topm

import numpy as np
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(s, h):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        sp = -s
        g[:n - sp] += 1.0 - h[sp:]
        g[sp:] -= h[:n - sp]
    return g * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(85):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

    n = len(h)
    tgt = n * 0.5
    h = h.copy().astype(float)
    v = np.zeros(n)

            break
        c = _corrs(h)
        idx = np.argsort(c)[::-1][:topm]

            g += _grad_lag(k - n + 1, h)
        g /= topm
        v = mom * v - step * g
        hn = _proj_sum(h + v, tgt)

        else:
            stall += 1
            step *= 0.5
            v *= 0.4
            if step < 1e-10 or stall > 300:
                break
    return h, f

def _lp_jump(h, nlags):
    try:
        import cvxpy as cp
        n = len(h)
        c = _corrs(h)
        act = np.argsort(c)[::-1][:nlags]
        hv = cp.Variable(n)
        tv = cp.Variable()
        cs = [hv >= 0, hv <= 1, cp.sum(hv) == n * 0.5]
        for k in act:
            gk = _grad_lag(k - n + 1, h)
            cs.append(cp.sum(gk * hv) <= tv + np.dot(gk, h) - c[k])
        p = cp.Problem(cp.Minimize(tv), cs)
        p.solve(solver=cp.ECOS, max_iter=10000, eps_abs=1e-10)
        if hv.value is not None:
            return _repair(np.array(hv.value, dtype=float), n)
    except Exception:
        pass
    return h

    return _repair(np.clip(f(x1), 0, 1), n_new)
