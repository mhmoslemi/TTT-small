    n = len(h)

    g_prev = _smax_grad(h, alpha)
    f_prev = _smax_obj(h, alpha)

            break
        g = _smax_grad(h, alpha)
        v = mu * v + (1 - mu) * g
        # BB step size

        dg = g - g_prev
        num = np.dot(dh, dg)
        den = np.dot(dg, dg) + 1e-15
        step = max(0.01 * abs(num / den), 1e-5) if abs(num) > 1e-15 else step0

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(h))

def _smax_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    return g

def _lag_grad(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _proj_grad(h, alpha, n_steps, t_end):
    n = len(h)
    f_prev = _smax_obj(h, alpha)
    h_arr = h.copy()
    g_prev = _smax_grad(h, alpha)
    h_pool = h_arr.copy()
    v = np.zeros(n)
    mu = 0.8
    step = 0.01
    for _s in range(n_steps):
        if time.time() > t_end:
            break
        g = _smax_grad(h_arr, alpha)
        v = mu * v + (1.0 - mu) * g
        dh = h_arr - h_pool
        dg = g - g_prev
        den = np.dot(dg, dg) + 1e-15
        step_bb = max(0.005, min(0.3, 0.5 * abs(np.dot(dh, dg) / den)))
        step = 0.6 * step + 0.4 * step_bb
        accepted = False
        for _bt in range(8):
            h_new = _proj(h_arr - step * v, n * 0.5)
            f_new = _smax_obj(h_new, alpha)
            if f_new <= f_prev + 1e-14:
                h_pool = h_arr.copy()
                h_arr = h_new
                g_prev = g.copy()
                f_prev = f_new
                accepted = True
                step = min(step * 1.05, 0.5)
                break
            step *= 0.5
        if not accepted:
            step = 0.3 * step
        h_pool = h_arr.copy()
        g_prev = _smax_grad(h_arr, alpha)
    return h_arr

def _multi_lag_descent(h, n_rounds, t_end):

    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def c_fun(x):
        return x.sum() - n * 0.5
    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    f_start = _smax_obj(h, alpha)
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        if time.time() > t_end:
            pass
        if res.fun < f_start - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _sa_pairs(h, n, t_end, n_iter, rng):
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    t0v, t1v = 0.004, 3e-6
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0v * (t1v / t0v) ** (i / max(1, n_iter - 1))
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.003 + 0.02 * t)
        old_i1, old_i2 = h[i1], h[i2]
        h[i1], h[i2] = old_i1 + d, old_i2 - d
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        else:
            h[i1], h[i2] = old_i1, old_i2
            h = _repair(h, n)
    return best_h, best_c
