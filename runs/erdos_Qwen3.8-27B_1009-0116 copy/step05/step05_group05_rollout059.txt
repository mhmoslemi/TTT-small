def _coord_optimize(h, n, n_inner=30):
    """Optimize one coordinate exactly (or near-exactly via ternary search)."""
    # Precompute c at current h
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    dx = 2.0 / n
    
    for i in range(n):
        # For each k, compute a_k and b_k such that c_k ≈ a_k + b_k * t (t = h[i])
        # c_k = sum_j h[j]*(1-h[j+sh]) where the term involving h[i] is:
        # h[i]*(1-h[i+sh]) - h[i-sh]*h[i] (for sh >= 0, with bounds)
        # = h[i] * [(1-h[i+sh]) - h[i-sh]]
        # So a_k = c_k - h[i]*b_k where b_k = [(1-h[i+sh_k]) - h[i-sh_k]]
        
        # Actually, let's just compute the gradient directly
        # g_k[i] as defined in the previous code
        b = np.zeros(2*n-1)
        for k in range(2*n-1):
            sh = (n-1) - k
            if sh >= 0:
                gi = 0.0
                if 0 <= i+sh < n:
                    gi += (1.0 - h[i+sh])
                if 0 <= i-sh < n:
                    gi -= h[i-sh]
                b[k] = gi
            else:
                sp = -sh
                gi = 0.0
                if 0 <= i-sp < n:
                    gi += (1.0 - h[i-sp])
                if 0 <= i+sp < n:
                    gi -= h[i+sp]
                b[k] = gi
        
        a = c - h[i] * b
        
        # Now minimize max_k (a[k] + b[k]*t) over t in [0,1]
        # Ternary search
        lo, hi = 0.0, 1.0
        for _ in range(n_inner):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(a + b * m1)
            f2 = np.max(a + b * m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        t_new = (lo + hi) / 2.0
        h[i] = t_new
    
    return h

            sp = -sh
            i1 = np.arange(sp, n); g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp); g[i2] -= wk * h[i2 + sp]
    return g

def _cons_fun(x): return x.sum() - len(x) * 0.5
def _cons_jac(x): return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type':'eq','fun':_cons_fun,'jac':_cons_jac}],
                       options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _precompute(n):
    shifts = np.arange(-(n-1), n)
    k_idx = (n-1) - shifts
    return shifts, k_idx

def _pair_grad_vec(h, n, i, j, shifts, k_idx):
    idx1 = i + shifts; v1 = (idx1 >= 0) & (idx1 < n)
    idx2 = i - shifts; v2 = (idx2 >= 0) & (idx2 < n)
    b_i = np.zeros(2*n-1)
    if v1.any(): b_i[k_idx[v1]] += (1.0 - h[idx1[v1]])
    if v2.any(): b_i[k_idx[v2]] -= h[idx2[v2]]
    idx1 = j + shifts; v1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - shifts; v2 = (idx2 >= 0) & (idx2 < n)
    b_j = np.zeros(2*n-1)
    if v1.any(): b_j[k_idx[v1]] += (1.0 - h[idx1[v1]])
    if v2.any(): b_j[k_idx[v2]] -= h[idx2[v2]]
    return b_i, b_j

def _pair_optimize(h, n, n_sweeps, n_inner=20, rng=None, shifts=None, k_idx=None):
    if shifts is None:
        shifts, k_idx = _precompute(n)
    n_pairs_str = n
    n_pairs_rand = n // 2
    for _sw in range(n_sweeps):
        for i in range(n):
            j = (i + 1) % n
            C = h[i] + h[j]
            lo_t = max(0.0, C - 1.0); hi_t = min(1.0, C)
            if lo_t >= hi_t: continue
            c = np.correlate(h, 1.0 - h, mode='full')
            b_i, b_j = _pair_grad_vec(h, n, i, j, shifts, k_idx)
            d = b_i - b_j
            lo_d = lo_t - h[i]; hi_d = hi_t - h[i]
            lo, hi = lo_d, hi_d
            for _ in range(n_inner):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(c + d * m1) < np.max(c + d * m2): hi = m2
                else: lo = m1
            cand = np.array([lo_d, hi_d, (lo+hi)*0.5])
            vals = np.array([np.max(c + d*t) for t in cand])
            h[i] = np.clip(h[i] + cand[np.argmin(vals)], 0, 1)
            h[j] = np.clip(C - h[i], 0, 1)
        for _ in range(n_pairs_rand):
            i = rng.randint(n); j = rng.randint(n)
            if i == j: continue
            C = h[i] + h[j]
            lo_t = max(0.0, C - 1.0); hi_t = min(1.0, C)
            if lo_t >= hi_t: continue
            c = np.correlate(h, 1.0 - h, mode='full')
            b_i, b_j = _pair_grad_vec(h, n, i, j, shifts, k_idx)
            d = b_i - b_j
            lo_d = lo_t - h[i]; hi_d = hi_t - h[i]
            lo, hi = lo_d, hi_d
            for _ in range(n_inner):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(c + d * m1) < np.max(c + d * m2): hi = m2
                else: lo = m1
            cand = np.array([lo_d, hi_d, (lo+hi)*0.5])
            vals = np.array([np.max(c + d*t) for t in cand])
            h[i] = np.clip(h[i] + cand[np.argmin(vals)], 0, 1)
            h[j] = np.clip(C - h[i], 0, 1)
    return h

def _active_descent(h, max_iters, n_top):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15: weights = np.ones(n_top) / n_top
        else: weights /= wsum
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh); g[i1] += w * (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n); g[i2] -= w * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n); g[i1] += w * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp); g[i2] -= w * h[i2 + sp]
        g *= dx
        improved = False
        for step in [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new; c5c = c5n; improved = True; break
        if not improved: break
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h); best_n = n0
    shifts0, k_idx0 = _precompute(n0)

    # Phase 1: SLSQP at n0
    for alpha in [2000, 20000, 200000, 2000000]:
        if _el() > 120: break
        h_new = _slsqp(best_h, alpha, 1500, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new

    # Phase 2: Pair exact optimization at n0 (main improvement)
    if _el() < budget_s - 400:
        time_left = min(budget_s - _el() - 400, 200)
        n_sweeps = int(time_left * 30)  # ~30 sweeps/sec at n=280
        n_sweeps = min(n_sweeps, 4000)
        h_new = _pair_optimize(best_h.copy(), n0, n_sweeps, 22, rng, shifts0, k_idx0)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new

    # Phase 3: Active descent at n0
    if _el() < budget_s - 350:
        h_new, c5 = _active_descent(best_h.copy(), 2000, 5)
        if c5 < best_c5: best_c5, best_h = c5, h_new

    # Phase 4: More pair optimization
    if _el() < budget_s - 300:
        time_left = min(budget_s - _el() - 300, 150)
        n_sweeps = min(int(time_left * 30), 3000)
        h_new = _pair_optimize(best_h.copy(), n0, n_sweeps, 25, rng, shifts0, k_idx0)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new

    # Phase 5: SLSQP polish at n0
    if _el() < budget_s - 250:
        for alpha in [500000, 2000000]:
            if _el() > budget_s - 230: break
            h_new = _slsqp(best_h, alpha, 1000, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5: best_c5, best_h = c5, h_new

    # Phase 6: Upsample to 2*n0
    if _el() < budget_s - 350:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        h_hi = _slsqp(h_hi, 100000, 800, 1e-18)
        c5 = _eval(h_hi)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_hi, n1
        shifts1, k_idx1 = _precompute(n1)
        if _el() < budget_s - 200:
            time_left = min(budget_s - _el() - 200, 120)
            n_sweeps = min(int(time_left * 8), 2000)
            h_new = _pair_optimize(best_h.copy(), n1, n_sweeps, 20, rng, shifts1, k_idx1)
            c5 = _eval(h_new)
            if c5 < best_c5: best_c5, best_h, best_n = c5, h_new, n1
        if _el() < budget_s - 120:
            h_hi2 = _slsqp(best_h, 500000, 800, 1e-18)
            c5 = _eval(h_hi2)
            if c5 < best_c5: best_c5, best_h, best_n = c5, h_hi2, n1
        if _el() < budget_s - 60:
            time_left = min(budget_s - _el() - 60, 60)
            n_sweeps = min(int(time_left * 8), 1000)
            h_new = _pair_optimize(best_h.copy(), best_n, n_sweeps, 25, rng)
            c5 = _eval(h_new)
            if c5 < best_c5: best_c5, best_h, best_n = c5, h_new, len(h_new)

    # Phase 7: Multi-start at best_n
    n_cur = best_n
    if _el() < budget_s - 200:
        shifts_c, k_idx_c = _precompute(n_cur)
        for _r in range(8):
            if _el() > budget_s - 180: break
            noise_scale = 0.02
            hp = _repair(best_h + noise_scale * rng.randn(n_cur), n_cur)
            h_new = _slsqp(hp, 300000, 500, 1e-18)
            h_new = _pair_optimize(h_new, n_cur, 300, 20, rng, shifts_c, k_idx_c)
            c5 = _eval(h_new)
            if c5 < best_c5: best_c5, best_h, best_n = c5, h_new, n_cur

    # Phase 8: Final long pair optimization
    if _el() < budget_s - 30:
        nf = len(best_h)
        shifts_f, k_idx_f = _precompute(nf)
        time_left = max(0, budget_s - _el() - 30)
        rate = 30 if nf <= 300 else 8
        n_sweeps = min(int(time_left * rate), 5000)
        h_new = _pair_optimize(best_h.copy(), nf, n_sweeps, 30, rng, shifts_f, k_idx_f)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h, best_n = c5, h_new, nf

    # Final SLSQP polish
    if _el() < budget_s - 10:
        h_new = _slsqp(best_h, 1000000, 1000, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h, best_n = c5, h_new, len(h_new)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
