    return h

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _alt_lp(h, dx, n_outer=30, max_time_s=60):
    n = len(h)
    tgt = n / 2.0
    best_h = h.copy()
    best_c = _c5(h, dx)
    hw = h.copy()
    t0 = time.time()
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.hstack([np.ones((1, n)), [[0.0]]])
    b_eq = np.array([tgt])
    bounds = [(0, 1)] * n + [(0, None)]
    b_ub = np.zeros(2 * n - 1)
    for it in range(n_outer * 2):
        if time.time() - t0 > max_time_s:
            break
        if it % 2 == 0:
            A = np.zeros((2 * n - 1, n))
            omh = 1.0 - hw
            for k in range(2 * n - 1):
                s = k - (n - 1)
                if s >= 0:
                    A[k, s:] = omh[:n - s]
                else:
                    A[k, -s:] = omh[-s:]
            var = 'h'
        else:
            A = np.zeros((2 * n - 1, n))
            for k in range(2 * n - 1):
                s = k - (n - 1)
                if s >= 0:
                    A[k, :n - s] = hw[s:]
                else:
                    A[k, -s:] = hw[:-s]
            var = 'z'
        A_ub = np.hstack([dx * A, -np.ones((2 * n - 1, 1))])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            continue
        x = res.x[:n]
        if var == 'h':
            h_new = _repair(x, tgt)
        else:
            h_new = _repair(1.0 - x, tgt)
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        hw = 0.5 * hw + 0.5 * h_new
        hw = _repair(hw, tgt)
    return best_h, best_c

def _swap_search(h, dx, rng, max_time_s):
    n = len(h)
    best_c = _c5(h, dx)
    best_h = h.copy()
    h = h.copy()
    t0 = time.time()
    phases = [(0.03, 15), (0.015, 15), (0.008, 20), (0.004, 25), (0.002, 30), (0.001, 35)]
    for step, max_iters in phases:
        for _ in range(max_iters):
            if time.time() - t0 > max_time_s:
                break
            batch = 2000
            for _ in range(batch):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = step * (1.0 if rng.random() < 0.5 else -1.0)
                hi, hj = h[i] + d, h[j] - d
                if 0 <= hi <= 1 and 0 <= hj <= 1:
                    h[i], h[j] = hi, hj
                    c_new = _c5(h, dx)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d
            c = np.correlate(h, 1.0 - h, mode='full')
            top3 = np.argsort(c)[::-1][:3]
            for ki in top3:
                s = abs(int(ki - (n - 1)))
                if s == 0 or s >= n:
                    continue
                m = n - s
                for _ in range(8):
                    a = rng.randint(m)
                    i1, i2 = a + s, a
                    d2 = step * 0.5 * (1.0 if h[a] > h[a + s] else -1.0)
                    hi, hj = h[i1] + d2, h[i2] - d2
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        h[i1], h[i2] = hi, hj
                        c_new = _c5(h, dx)
                        if c_new < best_c:
                            best_c = c_new
                            best_h = h.copy()
                        else:
                            h[i1], h[i2] = h[i1] - d2, h[i2] + d2
    return best_h, best_c

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.05 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 6), replace=False)
        hp[idx] += 0.1 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.04 * np.sin(2 * np.pi * rng.uniform(1, 6) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(2, n // 10), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    # Phase 1: Alternating LP (fast, exact subproblems)
    h1, c1 = _alt_lp(best_h, dx, n_outer=40, max_time_s=80)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: Swap search on best
    t_end2 = t0 + 300
    h2, c2 = _swap_search(best_h, dx, rng, t_end2 - time.time())
    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # Phase 3: Alternating LP again from improved point
    if time.time() < t0 + 380:
        h3, c3 = _alt_lp(best_h, dx, n_outer=30, max_time_s=70)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3
        h3, c3 = _swap_search(best_h, dx, rng, min(30, 1100 - (time.time() - t0)))
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Multi-start with diverse perturbations
    t_end4 = t0 + 700
    for r in range(16):
        rem = t_end4 - time.time()
        if rem < 30:
            break
        hp = _perturb(best_h, rng, r % 4)
        hp, cp = _alt_lp(hp, dx, n_outer=15, max_time_s=min(20, rem - 15))
        hp, cp = _swap_search(hp, dx, rng, min(12, rem - 20))
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    # Phase 5: Try 2x resolution
    if budget_s - (time.time() - t0) > 250:
        n2, dx2 = 2 * n, 1.0 / n
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        h2, c2 = _alt_lp(h2, dx2, n_outer=20, max_time_s=40)
        h2, c2 = _swap_search(h2, dx2, rng, 60)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # Phase 6: Final intensive optimization on best
    nb, dxb = best_n, 2.0 / best_n
    t_end6 = t0 + budget_s - 60
    if time.time() < t_end6:
        h6, c6 = _alt_lp(best_h, dxb, n_outer=20, max_time_s=min(40, t_end6 - time.time()))
        if c6 < best_c5:
            best_h, best_c5 = h6, c6
        best_h, best_c5 = _swap_search(best_h, dxb, rng, max(1, t_end6 - time.time()))

    # Final fine-tuning
    if budget_s - (time.time() - t0) > 10:
        nb2, dxb2 = best_n, 2.0 / best_n
        best_h, best_c5 = _swap_search(best_h, dxb2, rng, budget_s - (time.time() - t0) - 5)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
