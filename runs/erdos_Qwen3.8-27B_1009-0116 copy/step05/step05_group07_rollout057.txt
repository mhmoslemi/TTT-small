For the coordinate descent: given all h_j for j≠i, the objective as a function of h_i is:
f(h_i) = max_k (stuff not involving h_i + contribution of h_i to c_k)

The contribution of h_i to c_k is linear in h_i (since c_k is bilinear and we're fixing everything else):
- For lags where h_i appears as the first factor: adds h_i * (1-h_{i+s}) or h_i * (1-h_{i-s})
- For lags where h_i appears as (1-h_i): adds -h_i * h_{i±s}

So f(h_i) = max_k (a_k + b_k * h_i) where a_k and b_k are constants depending on i and the other h's.

This is a 1D problem: minimize max_k (a_k + b_k * h_i) over h_i ∈ [0,1].

The optimal h_i makes the top two (a_k + b_k * h_i) equal (if possible), or is at a boundary.

This is a nice subproblem to solve exactly!

```python
import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _coord_optimize(i, h):
    n = len(h)
    h_save = h[i]
    h[i] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_save
    a = c0 * (2.0 / n)
    b = (c1 - c0) * (2.0 / n)
    active = np.nonzero(np.abs(b) > 1e-15)[0]
    if len(active) == 0:
        return h_save
    # Grid + refine
    tg = np.linspace(0, 1, 100)
    vals = np.maximum.reduce([a[k] + b[k]*tg for k in active]) if len(active) <= 50 else None
    if vals is None:
        M = a[active][:,None] + b[active][:,None]*tg[None,:]
        vals = M.max(axis=0)
    else:
        pass
    idx = np.argmin(vals)
    t_best = tg[idx]
    if 0 < idx < 99:
        lo, hi = tg[idx-1], tg[idx+1]
        gr = (np.sqrt(5)-1)/2
        for _ in range(40):
            t1 = hi - gr*(hi-lo)
            t2 = lo + gr*(hi-lo)
            v1 = np.max(a[active] + b[active]*t1)
            v2 = np.max(a[active] + b[active]*t2)
            if v1 < v2: hi = t2
            else: lo = t1
        t_best = (lo+hi)/2
    return float(np.clip(t_best, 0, 1))

def _coord_descent(h, n_sweeps=5):
    n = len(h)
    target = n / 2.0
    f = _eval(h)
    h = h.copy()
    for sweep in range(n_sweeps):
        if time.time() - _t0 > _budget - 30:
            break
        improved = False
        for i in range(n):
            if time.time() - _t0 > _budget - 30:
                break
            new_val = _coord_optimize(i, h)
            # Adjust to maintain sum constraint
            h_save = h[i]
            h[i] = new_val
            diff = target - h.sum()
            if abs(diff) > 1e-9:
                # Distribute diff to other coordinates
                # Simple: add to a random other coordinate
                j = (i + 1) % n
                h[j] += diff
                h[j] = np.clip(h[j], 0, 1)
                # If clipping caused issue, fix
                if not (0 <= h[j] <= 1):
                    h[j] = 0 if h[j] < 0 else 1
                    # Redistribute remaining
                    diff2 = target - h.sum()
                    # Spread to all other coordinates
                    others = np.delete(np.arange(n), i)
                    per = diff2 / len(others)
                    h[others] += per
                    h[others] = np.clip(h[others], 0, 1)
                    # Final fix
                    h = _repair(h, n)
            f_new = _eval(h)
            if f_new < f - 1e-14:
                f = f_new
                improved = True
            else:
                h[i] = h_save
                    # Re-correct sum
                h[j] -= diff
                h = _repair(h, n)
        if not improved and sweep > 0:
            break
    return h, f
```

        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_lag(k, h)
    return g

def _slsqp_run(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def cf(x):
        return x.sum() - n * 0.5
    def cj(x):
        return np.ones(len(x))
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _active_descent_v2(h, max_iter=5000, m_init=3, m_max=25):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = 0.003
    m = m_init
    stall = 0
    for it in range(max_iter):
        if time.time() - _t0 > _budget - 20:
            break
        c = _corrs_raw(h) * (2.0 / n)
        # Adaptive M: increase if top-M are close in value
        top_vals = np.sort(c)[::-1][:m_max]
        m_eff = m
        if m < m_max:
            # Increase M if (m+1)-th is close to m-th
            if top_vals[m] > 0.95 * top_vals[m-1]:
                m_eff = min(m + 2, m_max)
        top_idx = np.argsort(c)[::-1][:m_eff]
        # Compute gradient
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k, h)
        g /= m_eff
        # Project gradient onto sum-zero subspace
        g -= g.mean()
        # Line search with backtracking
        found = False
        for _bt in range(40):
            h_new = _proj_sum(h - step * g, target)
            if np.allclose(h_new, h, atol=1e-10):
                break
            f_new = _eval(h_new)
            if f_new < f - 1e-7 * step * np.dot(g, h_new - h):
                h = h_new
                f = f_new
                found = True
                stall = 0
                break
            step *= 0.5
        if not found:
            stall += 1
            step = max(step * 0.7, 0.0001)
            if stall > 200:
                # Try increasing M
                m = min(m + 3, m_max)
                step = 0.002
                stall = 0
        else:
            step = min(step * 1.05, 0.005)
            if stall == 0 and it % 500 == 0:
                m = min(m + 1, m_max)
    return h, f

def _pairwise_targeted(h, n_iter=20000, seed_val=42):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    c = _corrs_raw(h) * (2.0 / n)
    for it in range(n_iter):
        if time.time() - _t0 > _budget - 8:
            break
        if it % 500 == 0:
            c = _corrs_raw(h) * (2.0 / n)
        # Pick a lag that's active
        k_star = int(c.argmax())
        s = k_star - (n - 1)
        # Find coordinates that contribute most to this lag
        if s >= 0:
            # c = sum_{j=0}^{n-1-s} h[j]*(1-h[j+s])
            contrib = np.zeros(n)
            if s == 0:
                contrib[:] = h * (1 - h)
            else:
                contrib[:n-s] += h[:n-s] * (1 - h[s:])
                contrib[s:] += h[s:] * (1 - h[:n-s])
        else:
            sp = -s
            contrib = np.zeros(n)
            contrib[sp:] += h[sp:] * (1 - h[:n-sp])
            contrib[:n-sp] += h[:n-sp] * (1 - h[sp:])
        
        # Pick a high-contribution coordinate to decrease
        i = int(np.argmax(contrib + 0.001 * rng.rand(n)))
        # Pick another to increase (to maintain sum)
        j = int(np.argmin(contrib + 0.001 * rng.rand(n)))
        if i == j:
            continue
        # Try a small transfer
        for d in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            ni, nj = h[i] + d, h[j] - d  # decrease i (reduces overlap), increase j
            # Wait, I want to REDUCE the contribution. 
            # If h[i] is high and h[i+s] is high, decreasing h[i] helps.
            # Or if h[i] is low and h[i+s] is high, increasing h[i] helps (makes 1-h[i] smaller... no)
            # Actually h[i]*(1-h[j+s]) - to reduce, either decrease h[i] or increase h[i+s]
            # Let's just try both directions
            pass
        # Simpler: just try decreasing h[i] and increasing h[j] by d
        d = 0.003 * (1 + 0.5*rng.randn())
        d = np.clip(d, 0.0005, 0.01)
        ni, nj = h[i] - d, h[j] + d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni + d, nj - d
                # Try reverse
                ni2, nj2 = h[j] - d, h[i] + d
                if 0 <= ni2 <= 1 and 0 <= nj2 <= 1:
                    h[j], h[i] = ni2, nj2
                    fn = _eval(h)
                    if fn < f - 1e-15:
                        f = fn
                    else:
                        h[j], h[i] = ni2 + d, nj2 - d
    return h, f
