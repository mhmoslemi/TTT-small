    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

    if abs(s - target) < 1e-9:

    h *= target / s
    h = np.clip(h, 0.0, 1.0)

- The lags at the maximum are the "bottleneck"
- We can try to specifically reduce those lags by adjusting h values that contribute most to them

For lag k (in the full correlation), c[k] = sum_j h[j] * (1-h[j+offset]) where offset = k-(n-1).
If we increase h at a position where (1-h[j+offset]) is large, we increase c[k].
If we decrease h at a position where (1-h[j+offset]) is large, we decrease c[k].
Similarly, increasing h[j+offset] where h[j] is large decreases c[k].

So the "focal" move is: for the lag k at the maximum, find the j that contributes most to c[k], and move mass from h[j] to h[j+offset] (or vice versa) to reduce c[k].

def _focal_step(h, c, rng, n_iter, n):
    """Targeted reduction of maximum correlation lags."""
    dx = 2.0 / n
    c5c = c.max() * dx
    best_c5 = c5c
    best_h = h.copy()
    
    for _ in range(n_iter):
        kmax = np.argmax(c)
        offset = kmax - (n - 1)
        
        # Determine which pairs (j, j+offset) contribute to c[kmax]
        if offset >= 0:
            valid = np.arange(0, n - offset)
            contribs = h[valid] * (1 - h[valid + offset])
        else:
            valid = np.arange(-offset, n)
            contribs = h[valid + offset] * (1 - h[valid])
        
        # Find the pair with maximum contribution
        j_idx = np.argmax(contribs)
        j = valid[j_idx]
        
        if offset >= 0:
            j2 = j + offset
            # Move mass from j to j2 (reduces h[j]*(1-h[j2]))
            d = 0.01 * rng.random()
        else:
            j2 = j + offset  # j2 < j
            # Move mass from j to j2 (reduces h[j2]*(1-h[j]))
            d = 0.01 * rng.random()

        nj2 = h[j2] + d
