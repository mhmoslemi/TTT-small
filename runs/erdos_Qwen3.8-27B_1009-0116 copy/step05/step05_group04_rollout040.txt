                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, sh):
    n = len(h)
    g = np.zeros(n)
    if sh >= 0:
        idx = np.arange(n - sh)
        g[idx] += 1.0 - h[idx + sh]
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        g += w[k] * _grad_lag(h, sh)
    return g * dx

def _nesterov(h, alpha, steps, lr0, n):
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    mu = 0.82
    step = lr0
    for _ in range(steps):
        g = _smooth_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1 - mu) * g
        h_new = _project_sum(h - step * v, n * 0.5)
        f_new = _smooth_obj(h_new, alpha)
        if f_new < f:
            h, f, step = h_new, f_new, min(step * 1.04, lr0 * 3)
        else:
            step *= 0.5
            h = _project_sum(h_prev - step * v, n * 0.5)
            f = _smooth_obj(h, alpha)
    return h

def _slsqp(h, alpha, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _cyclic_subgrad(h, n, n_iter, top_k, lr0):
    h = h.copy()
    dx = 2.0 / n
    lr = lr0
    mom = np.zeros(n)
    mu = 0.6
    for it in range(n_iter):
        c = _corr(h)
        top = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in top:
            g += _grad_lag(h, (n - 1) - k)
        g *= dx / top_k
        mom = mu * mom + g
        c5_old = _eval(h)
        moved = False
        for tlr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05, lr * 0.02]:
            h_new = _project_sum(h - tlr * mom, n * 0.5)
            if _eval(h_new) < c5_old - 1e-15:
                h, lr, moved = h_new, min(tlr * 1.08, lr0 * 2), True
                break
        if not moved:
            lr *= 0.4
        if it % 100 == 0:
            lr = max(lr0 * 0.85 ** (it // 100), lr0 * 0.02)
    return h

def _grad_guided_search(h, n, n_iter, rng):
    h = h.copy()
    c5 = _eval(h)
    dx = 2.0 / n
    for _ in range(n_iter):
        c = _corr(h)
        top = np.argsort(c)[-3:]
        g = np.zeros(n)
        for k in top:
            g += _grad_lag(h, (n - 1) - k)
        g *= dx / len(top)
        pos_mask = g > 0.01
        neg_mask = g < -0.01
        if not any(pos_mask) or not any(neg_mask):
            break
        pidx = np.where(pos_mask)[0]
        nidx = np.where(neg_mask)[0]
        i = pidx[rng.randint(len(pidx))]
        j = nidx[rng.randint(len(nidx))]
        best_d = None
        for d in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            ni, nj = h[i] - d, h[j] + d
            if 0 <= ni and ni <= 1 and 0 <= nj and nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5 - 1e-15:
                    c5 = c5n
                    best_d = d
                    break
                h[i], h[j] = ni + d, nj - d
        if best_d is None:
            for d in [0.002, 0.001, 0.0005]:
                ni, nj = h[i] - d, h[j] + d
                if 0 <= ni and ni <= 1 and 0 <= nj and nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5 - 1e-16:
                        c5 = c5n
                        break
                    h[i], h[j] = ni + d, nj - d
    return h

def _fine_pairwise(h, n, n_iter, rng, amp0=2e-4):
    h = h.copy()
    c5 = _eval(h)
    amp = amp0
    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = amp * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            hi_h, hj_h = h[i], h[j]
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-16:
                c5 = c5n
                amp = min(amp * 1.03, amp0 * 5)
            else:
                h[i], h[j] = hi_h, hj_h
                amp *= 0.97
    return h

def _sa_block(h, n, t0, t1, n_iter, rng):
    c_cur = _eval(h)
    best_h, best_c = h.copy(), c_cur
    for i in range(n_iter):
        t = t0 * (t1 / t0) ** (i / max(1, n_iter - 1))
        blk = rng.randint(2, max(3, n // 10))
        st = rng.randint(0, n - blk)
        h_new = h.copy()
        h_new[st:st + blk] += rng.randn(blk) * (0.005 + 0.04 * t)
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-8)):
            h, c_cur = h_new, c_new
            if c_cur < best_c:
                best_c, best_h = c_cur, h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Nesterov at original resolution
    for alpha in [3000, 30000, 150000, 800000]:
        if _el() > 80:
            break
        hc = _nesterov(best_h, alpha, 400, 0.012, best_n)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 2: SLSQP progressive
    for alpha in [10000, 100000, 500000, 2000000]:
        if _el() > 200:
            break
        hc = _slsqp(best_h, alpha, 3000, best_n)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 3: Cyclic subgradient
    if _el() < budget_s - 250:
        hc = _cyclic_subgrad(best_h, best_n, 2000, 5, 0.03)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 4: Gradient-guided search
    if _el() < budget_s - 200:
        hc = _grad_guided_search(best_h, best_n, 5000, rng)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 5: Upsample 2x and optimize
    if _el() < budget_s - 350:
        n1 = best_n * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha in [5000, 50000, 300000]:
            if _el() > budget_s - 200:
                break
            h_hi = _nesterov(h_hi, alpha, 300, 0.008, n1)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if _el() < budget_s - 150:
            h_hi = _cyclic_subgrad(best_h, best_n, 1500, 5, 0.02)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h = c5, h_hi.copy()
        if _el() < budget_s - 100:
            h_hi = _grad_guided_search(best_h, best_n, 3000, rng)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h = c5, h_hi.copy()

    # Phase 6: SA for escape
    if _el() < budget_s - 150:
        t_rem = budget_s - 150 - _el()
        n_iter = min(int(t_rem / 0.04), 15000)
        hc, c5s = _sa_block(best_h, best_n, 0.004, 1e-6, n_iter, rng)
        if c5s < best_c5:
            best_c5, best_h = c5s, hc.copy()

    # Phase 7: Re-optimize after SA
    if _el() < budget_s - 80:
        hc = _nesterov(best_h, 500000, 300, 0.005, best_n)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()
        hc = _grad_guided_search(best_h, best_n, 3000, rng)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 8: SLSQP final polish
    if _el() < budget_s - 30:
        hc = _slsqp(best_h, 5000000, 5000, best_n)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 9: Ultra-fine pairwise
    if _el() < budget_s - 2:
        t_rem = budget_s - 2 - _el()
        n_iter = min(int(t_rem / 0.002), 80000)
        best_h = _fine_pairwise(best_h, best_n, n_iter, rng)
        best_c5 = _eval(best_h)

    # Phase 10: Final gradient-guided + pairwise
    if _el() < budget_s - 1:
        best_h = _grad_guided_search(best_h, best_n, 2000, rng)
        best_h = _fine_pairwise(best_h, best_n, 5000, rng, 1e-4)
        best_c5 = _eval(best_h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
