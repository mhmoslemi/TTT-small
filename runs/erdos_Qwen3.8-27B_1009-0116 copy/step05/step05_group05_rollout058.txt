    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    """Project h to {0≤h≤1, sum h = target_sum}"""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, k_top):
    """Compute subgradient of max of top-k correlation values"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:k_top]
    cvals = c[idx]
    # Soft weights: proportional to (cvals - mean) + epsilon
    cmin = cvals.min()
    cmax = cvals.max()
    if cmax - cmin < 1e-12:
        w = np.ones(k_top) / k_top
    else:
        w = cvals - cmin
        w[w < 0] = 0
        wsum = w.sum()
        if wsum < 1e-12:

        else:
            w /= wsum

        if sh >= 0:

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_and_corr(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return c.max(), c

def _project(h, target):
    lo, hi = -0.5, 0.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        t = -s
        g[t:] += (1.0 - h[:n-t])
        g[:n-t] -= h[t:]
    return g

def _subgrad_step(h, k_top, alpha, t_end):
    """One batch of subgradient steps"""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _eval(h)
    step = 0.01
    stagnation = 0
    
    while time.time() < t_end:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = c.max()
        
        # Get top k_top indices
        if k_top >= 2*n - 1:
            top_idx = np.arange(2*n-1)
        else:
            top_idx = np.argpartition(c, -k_top)[-k_top:]
        
        top_vals = c[top_idx]
        # Softmax weights
        tv = top_vals - top_vals.max()
        w = np.exp(alpha * tv)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w /= wsum
        
        # Compute gradient
        g = np.zeros(n)
        for wk, k in zip(w, top_idx):
            s = k - (n - 1)
            g += wk * _grad_shift(h, s)
        
        # Normalized gradient
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        
        # Line search with backtracking
        accepted = False
        for _ in range(10):
            h_new = _project(h - step * g, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < cmax - 1e-14:
                h = h_new
                cmax = c5_new
                accepted = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stagnation = 0
                break
            step *= 0.5
        if not accepted:
            step *= 0.5
            stagnation += 1
            if step < 1e-8:
                break
        else:
            step = min(step * 1.05, 0.05)
    
    return best_h, best_c5

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _fast_subgrad(h, k_top, alpha, t_end):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _eval(h)
    step = 0.005
    it = 0
    while time.time() < t_end - 1:
        it += 1
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = c.max()
        # Top-k
        if k_top >= 2 * n - 1:
            top_idx = np.arange(2 * n - 1)
        else:
            top_idx = np.argpartition(c, -k_top)[-k_top:]
        top_vals = c[top_idx]
        tv = top_vals - top_vals.max()
        w = np.exp(alpha * tv)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w /= ws
        # Gradient
        g = np.zeros(n)
        nk1 = n - 1
        for wk, k in zip(w, top_idx):
            s = k - nk1
            if s >= 0:
                g[:n-s] += wk * (1.0 - h[s:])
                g[s:] -= wk * h[:n-s]
            else:
                t = -s
                g[t:] += wk * (1.0 - h[:n-t])
                g[:n-t] -= wk * h[t:]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        # Backtracking line search
        imp = 0
        for _ in range(8):
            h_new = _project(h - step * g, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < cmax - 1e-13:
                h = h_new
                cmax = c5_new
                imp = 1
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            step *= 0.5
        if not imp:
            step *= 0.6
            if step < 1e-9:
                break
        else:
            step = min(step * 1.03, 0.02)
    return best_h, best_c5, it

def _pairwise(h, n_iter, t_end, rng):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for _ in range(n_iter):

        if c5n < c5 - 1e-15:
            c5 = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5
