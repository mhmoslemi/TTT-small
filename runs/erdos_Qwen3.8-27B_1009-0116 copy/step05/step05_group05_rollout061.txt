import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _col_grad(h, j):
    n = len(h)
    g = np.zeros(2*n-1)
    for k in range(2*n-1):
        sh = k - (n-1)
        if sh >= 0:
            v = 0.0
            if j + sh < n:
                v += 1.0 - h[j+sh]
            if sh > 0 and j - sh >= 0:
                v -= h[j-sh]
            g[k] = v
        else:
            sp = -sh
            v = 0.0
            if j - sp >= 0:
                v += 1.0 - h[j-sp]
            if j + sp < n:
                v -= h[j+sp]
            g[k] = v
    return g

def _pairwise_step(h, i, j, n_grid=300):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    gi = _col_grad(h, i)
    gj = _col_grad(h, j)
    a = gi - gj
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_hi - d_lo < 1e-12:
        return h, 0
    dg = np.linspace(d_lo, d_hi, n_grid)
    vals = np.max(c[None,:] + dg[:,None]*a[None,:], axis=1)
    bi = np.argmin(vals)
    if vals[bi] >= c.max() - 1e-10:
        return h, 0
    # Refine
    lo = dg[max(0, bi-2)]
    hi = dg[min(n_grid-1, bi+2)]
    dg2 = np.linspace(lo, hi, 100)
    vals2 = np.max(c[None,:] + dg2[:,None]*a[None,:], axis=1)
    bi2 = np.argmin(vals2)
    if vals2[bi2] >= c.max() - 1e-10:
        return h, 0
    d = dg2[bi2]
    h[i] += d
    h[j] -= d
    return h, 1

def _pairwise_pass(h, n_moves, rng, n_grid=300):
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_moves):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h, improved = _pairwise_step(h, i, j, n_grid)
        if improved:

    return h, c5

def _nesterov_subgrad(h, n_iter, m_top, lr0, mu):
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    h_prev = h.copy()
    v = np.zeros(n)  # momentum
    t = 1.0
    c5 = _eval(h_curr)
    for it in range(n_iter):
        c = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        # Top m active
        idx = np.argpartition(c, -m_top)[-m_top:]
        top_c = c[idx]
        m = top_c.max()
        w = np.exp((top_c - m) / 0.01)  # soft weighting
        w /= w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, idx):
            sh = (n-1) - ki
            if sh >= 0:
                g[:n-sh] += wi * (1.0 - h_curr[sh:])
                g[sh:] -= wi * h_curr[:n-sh]
            else:
                sp = -sh
                g[sp:] += wi * (1.0 - h_curr[:n-sp])
                g[:n-sp] -= wi * h_curr[sp:]
        g *= dx
        # Nesterov
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        beta = (t - 1) / t_new
        h_y = h_curr + beta * (h_curr - h_prev)
        lr = lr0 / (1 + it * 0.01)
        h_new = _proj_to_sum(h_y - lr * g, n * 0.5)
        h_prev = h_curr
        h_curr = h_new
        t = t_new
        c5n = _eval(h_curr)
        if c5n < c5:
            c5 = c5n
    return h_curr, c5

    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i*factor:(i+1)*factor] = h[i]
    return h_hi

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _col_grad_vec(h, j):
    n = len(h)
    g = np.zeros(2 * n - 1)
    sps = np.arange(n - 1, 0, -1)
    idx1 = j - sps
    idx2 = j + sps
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    g[:n-1] = np.where(v1, 1.0 - h[idx1[v1]], 0.0) - np.where(v2, h[idx2[v2]], 0.0)
    g[n-1] = 1.0 - 2.0 * h[j]
    shs = np.arange(1, n)
    idx1b = j + shs
    idx2b = j - shs
    v1b = (idx1b >= 0) & (idx1b < n)
    v2b = (idx2b >= 0) & (idx2b < n)
    g[n:] = np.where(v1b, 1.0 - h[idx1b[v1b]], 0.0) - np.where(v2b, h[idx2b[v2b]], 0.0)
    return g

def _pairwise_pass(h, n_moves, rng, n_grid, t_end):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c5_val = c.max()
    t0 = time.time()
    for _ in range(n_moves):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        gi = _col_grad_vec(h, i)
        gj = _col_grad_vec(h, j)
        a = gi - gj
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-12:
            continue
        dg = np.linspace(d_lo, d_hi, n_grid)
        vals = (c + dg[:, None] * a[None, :]).max(axis=1)
        bi = np.argmin(vals)
        if vals[bi] >= c5_val - 1e-11:
            continue
        lo2 = dg[max(0, bi - 2)]
        hi2 = dg[min(n_grid - 1, bi + 2)]
        dg2 = np.linspace(lo2, hi2, 200)
        vals2 = (c + dg2[:, None] * a[None, :]).max(axis=1)
        bi2 = np.argmin(vals2)
        if vals2[bi2] >= c5_val - 1e-11:
            continue
        d = dg2[bi2]
        h[i] += d
        h[j] -= d
        c = c + d * a
        c5_val = c.max()
    return h, c5_val * 2.0 / n

def _nesterov_subgrad(h, n_iter, m_top, lr0, t_end):
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    h_prev = h.copy()
    t_nest = 1.0
    c5 = _eval(h_curr)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        idx = np.argpartition(c, -m_top)[-m_top:]
        top_c = c[idx]
        m = top_c.max()
        w = np.exp((top_c - m) * 20.0)
        w /= w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, idx):
            sh = (n - 1) - ki
            if sh >= 0:
                g[:n-sh] += wi * (1.0 - h_curr[sh:])
                g[sh:] -= wi * h_curr[:n-sh]
            else:
                sp = -sh
                g[sp:] += wi * (1.0 - h_curr[:n-sp])
                g[:n-sp] -= wi * h_curr[sp:]
        g *= dx
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        h_y = h_curr + beta * (h_curr - h_prev)
        lr = lr0 / (1.0 + 0.005 * it)
        h_new = _proj_to_sum(h_y - lr * g, n * 0.5)
        h_prev = h_curr
        h_curr = h_new
        t_nest = t_new
        c5n = _eval(h_curr)
        if c5n < c5:
            c5 = c5n
    return h_curr, c5

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _smooth_slsqp(h, alpha, maxiter, t_end):
    from scipy.optimize import minimize
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * dx
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - x[sh:])
                g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g
    def cons_fun(x):
        return x.sum() - n * 0.5
    def cons_jac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=grad, args=(), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
