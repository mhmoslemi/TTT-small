def _repair(h, n):
    # Ensure h in [0,1] and sum(h) = n/2
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

def _fft_corr(h):
    n = len(h)
    b = 1.0 - h
    Na = 2 ** int(np.ceil(np.log2(2 * n)))
    A = np.fft.rfft(h, Na)
    B = np.fft.rfft(b, Na)
    C = np.fft.irfft(A * np.conj(B), Na)
    return np.concatenate([C[n-1::-1], C[Na-n+1:Na]])

def _eval(h):
    n = len(h)
    c = _fft_corr(h) * (2.0 / n)
    return float(c.max())

def _grad_corr(k, h):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        # f_k = sum_i h[i] * (1-h[i+s])
        # d/dh[j] = (1-h[j+s]) if j+s < n, minus h[j-s] if j-s >= 0
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n-s] += (1.0 - h[s:n])
            g[s:] -= h[:n-s]
    else:
        sp = -s
        if sp == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[sp:n] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:n]
    return g

def _proj_sum(h, target):
    # Project h onto {x in [0,1]^n : sum(x) = target}
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _active_descent_v2(h, topm, max_iter, step0=0.01):

def _slsqp_smooth(h, alpha, maxiter=2000):
    n = len(h)
    def obj(x):
        c = _fft_corr(x) * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -100, 100)
        sm = s - s.max()
        return m + np.exp(np.logaddexp.reduce(sm)) / alpha  # log-sum-exp
    
    def grad(x):
        c = _fft_corr(x) * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -100, 100)
        sm = s - s.max()
        w = np.exp(sm)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-12:
                continue
            g += w[k] * _grad_corr(k, x)
        g *= (2.0 / n)
        return g
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 
             'jac': lambda x: np.ones(n)}]
    try:

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _pairwise_local(h, n_iter, scale, seed_val):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 10:
            break
        i, j = rng.randint(n, size=2)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)
