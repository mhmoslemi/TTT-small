def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_c(h, k, n):
    """Gradient of c[k] wrt h"""
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:n])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _smooth_max_grad(h, alpha, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_c(h, k, n)
    return g * dx

def _nesterov_subgrad(h, n, t_end, top_k=3, step0=0.003):
    """Nesterov accelerated projected subgradient on top-k lags"""
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h.copy()
    best_c = _eval(h_cur)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    
    while time.time() < t_end:
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = c.max()
        
        # Active set: top-k lags
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_c(h_cur, k, n)
        g *= dx / top_k
        
        # Nesterov
        v = mu * v + (1 - mu) * g
        h_prev = h_cur.copy()
        h_new = _proj(h_cur - step * v, n * 0.5)
        
        c_new = _eval(h_new)
        if c_new < c_max:
            h_cur = h_new
            step = min(step * 1.01, step0 * 10)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        else:
            step *= 0.6
            h_cur = _proj(h_prev - step * v, n * 0.5)
    
    return best_h, best_c

def _slsqp_smooth(h, alpha, maxiter, t_end):
    dx = 2.0 / n_global
    n = len(h)
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
    if time.time() < t_end:
        return _repair(res.x, n)
    return _repair(h, n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
