```python
def _optimal_pair_step(h, i, j, dx, n):
    """Find optimal δ to move h[i]+=δ, h[j]-=δ minimizing max correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Compute d_k = ∂c_k/∂h[i] - ∂c_k/∂h[j]
    d = np.zeros(2*n-1)
    for k in range(2*n-1):
        sh = (n-1) - k
        di = 0.0
        dj = 0.0
        if sh >= 0:
            # c_k involves h[m] for m in [0, n-1-sh] (first pos) and h[m+sh] for m in [0, n-1-sh] (second pos)
            # ∂c_k/∂h[i]: if 0<=i<=n-1-sh: (1-h[i+sh]); if sh<=i<=n-1: -h[i-sh]
            if 0 <= i <= n-1-sh:
                di += (1.0 - h[i+sh])
            if sh <= i <= n-1:
                di += -h[i-sh]
            if 0 <= j <= n-1-sh:
                dj += (1.0 - h[j+sh])
            if sh <= j <= n-1:
                dj += -h[j-sh]
        else:
            sp = -sh
            # c_k involves h[m+sp] for m in [0, n-1-sp] (first pos) and h[m] for m in [0, n-1-sp] (second pos)
            if sp <= i <= n-1:
                di += (1.0 - h[i-sp])
            if 0 <= i <= n-1-sp:
                di += -h[i+sp]
            if sp <= j <= n-1:
                dj += (1.0 - h[j-sp])
            if 0 <= j <= n-1-sp:
                dj += -h[j+sp]
        d[k] = di - dj
    # Now f(δ) = max_k (c[k] + δ*d[k])
    # f is convex. Minimize over valid range of δ.
    dlo = max(-h[i], h[j]-1.0)
    dhi = min(1.0-h[i], h[j])
    # Ternary search
    for _ in range(50):
        m1 = dlo + (dhi-dlo)/3
        m2 = dhi - (dhi-dlo)/3
        f1 = np.max(c + m1*d)
        f2 = np.max(c + m2*d)
        if f1 < f2:
            dhi = m2
        else:
            dlo = m1
    delta = (dlo + dhi) / 2
    new_val = np.max(c + delta*d)
    return delta, new_val
```

```python
def _exact_pair_move(h, i, j, n):
    """Find optimal δ: h[i]+=δ, h[j]-=δ. Returns (h_new, improved)."""
    c = np.correlate(h, 1.0 - h, mode='full')
    d = _pair_grad_diff(h, i, j, n)
    dlo = max(-h[i], h[j] - 1.0)
    dhi = min(1.0 - h[i], h[j])
    if dlo >= dhi - 1e-15:
        return h, False
    # f(δ) = max_k(c[k] + δ*d[k]), convex in δ
    # Ternary search
    a, b = dlo, dhi
    for _ in range(40):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(c + m1 * d) < np.max(c + m2 * d):
            b = m2
        else:
            a = m1
    delta = (a + b) / 2.0
    if abs(delta) < 1e-15:
        return h, False
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, True
```

```python
def _active_descent(h, max_iters, n_top, momentum):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    v = np.zeros(n)
    for it in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmax = cvals[0]
        # Soft weights based on distance from max
        weights = np.exp(-50 * (cmax - cvals))
        weights /= weights.sum()
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += w * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= w * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += w * (1.0 - h[:n-sp])
                g[:n-sp] -= w * h[sp:]
        g *= dx
        # Momentum
        v = momentum * v + (1 - momentum) * g
        # Project step
        improved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj_to_sum(h - step * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            v *= 0.5
            if momentum > 0.1:
                momentum *= 0.9
    return h, c5c
```

```python
def _pair_exact_step(c, d, h_i, h_j):
    """Given current correlations c, gradient difference d, and bounds, find optimal step."""
    dlo = max(-h_i, h_j - 1.0)
    dhi = min(1.0 - h_i, h_j)
    if dlo >= dhi - 1e-14:
        return 0.0
    # f(δ) = max(c + δ*d), convex
    # Use ternary search
    a, b = dlo, dhi
    for _ in range(45):
        t1 = a + (b - a) / 3.0
        t2 = b - (b - a) / 3.0
        if np.max(c + t1 * d) < np.max(c + t2 * d):
            b = t2
        else:
            a = t1
    return (a + b) / 2.0

def _pairwise_refine_exact(h, n_pairs, rng, c5c):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = _pair_grad_diff(h, i, j, n)
        delta = _pair_exact_step(c, d, h[i], h[j])
        if abs(delta) < 1e-14:
            continue
        # Apply
        h[i] += delta
        h[j] -= delta
        # Update c
        c += delta * d
        c5n = np.max(c) * dx
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            # Revert
            h[i] -= delta
            h[j] += delta
            c -= delta * d
    return h, c5c
```

```python
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n-sp])
            g[:n-sp] -= w[k] * h[sp:]
    return g * dx

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:

def _pair_grad_diff(h, i, j, n):
    d = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        sh = (n - 1) - k
        di = 0.0
        dj = 0.0
        if sh >= 0:
            if i <= n - 1 - sh:
                di += (1.0 - h[i + sh])
            if i >= sh:
                di -= h[i - sh]
            if j <= n - 1 - sh:
                dj += (1.0 - h[j + sh])
            if j >= sh:
                dj -= h[j - sh]
        else:
            sp = -sh
            if i >= sp:
                di += (1.0 - h[i - sp])
            if i <= n - 1 - sp:
                di -= h[i + sp]
            if j >= sp:
                dj += (1.0 - h[j - sp])
            if j <= n - 1 - sp:
                dj -= h[j + sp]
        d[k] = di - dj
    return d

def _pairwise_exact(h, n_moves, rng, c5c, recompute_every=500):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    moves = 0
    for it in range(n_moves):
        if it > 0 and it % recompute_every == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = _pair_grad_diff(h, i, j, n)
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi - 1e-14:
            continue
        a, b = dlo, dhi
        for _ in range(40):
            t1 = a + (b - a) / 3.0
            t2 = b - (b - a) / 3.0
            if np.max(c + t1 * d) < np.max(c + t2 * d):
                b = t2
            else:
                a = t1
        delta = (a + b) / 2.0
        if abs(delta) < 1e-14:
            continue
        h[i] += delta
        h[j] -= delta
        c += delta * d
        c5n = np.max(c) * dx
        if c5n < c5c - 1e-15:
            c5c = c5n
            moves += 1
        else:
            h[i] -= delta
            h[j] += delta
            c -= delta * d
    c = np.correlate(h, 1.0 - h, mode='full')
    c5c = np.max(c) * dx
    return h, c5c

def _active_set_descent(h, max_iters, n_top):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    v = np.zeros(n)
    mom = 0.3
    for it in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmax = cvals[0]
        weights = np.exp(-100 * (cmax - cvals))
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += w * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= w * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += w * (1.0 - h[:n-sp])
                g[:n-sp] -= w * h[sp:]
        g *= dx
        v = mom * v + (1 - mom) * g
        improved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = _proj_to_sum(h - step * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            v *= 0.5
            if mom > 0.05:
                mom *= 0.9
    return h, c5c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += w[k] * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w[k] * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n-sp])
            g[:n-sp] -= w[k] * h[sp:]
    return g * dx

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    d = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        sh = (n - 1) - k
        di = 0.0
        dj = 0.0
        ip = i + sh
        if 0 <= ip < n:
            di += 1.0 - h[ip]
        im = i - sh
        if 0 <= im < n:
            di -= h[im]
        jp = j + sh
        if 0 <= jp < n:
            dj += 1.0 - h[jp]
        jm = j - sh
        if 0 <= jm < n:
            dj -= h[jm]
        d[k] = di - dj
    return d

def _pairwise_exact(h, n_moves, rng, c5c, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(n_moves):
        if time.time() > deadline:
            break
        if it > 0 and it % 300 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = _pair_grad_diff(h, i, j, n)
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi - 1e-14:
            continue
        a, b = dlo, dhi
        for _ in range(40):
            t1 = a + (b - a) / 3.0
            t2 = b - (b - a) / 3.0
            if np.max(c + t1 * d) < np.max(c + t2 * d):
                b = t2
            else:
                a = t1
        delta = (a + b) / 2.0
        if abs(delta) < 1e-13:
            continue
        h[i] += delta
        h[j] -= delta
        c += delta * d
        c5n = np.max(c) * dx
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h[i] -= delta
            h[j] += delta
            c -= delta * d
    c = np.correlate(h, 1.0 - h, mode='full')
    c5c = float(np.max(c) * dx)
    return h, c5c

def _active_set_descent(h, max_iters, n_top, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    v = np.zeros(n)
    mom = 0.3
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmax = cvals[0]
        weights = np.exp(-80.0 * (cmax - cvals))
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += w * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= w * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += w * (1.0 - h[:n-sp])
                g[:n-sp] -= w * h[sp:]
        g *= dx
        v = mom * v + (1.0 - mom) * g
        improved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = _proj_to_sum(h - step * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            v *= 0.5
            if mom > 0.05:
                mom *= 0.85
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP
    for alpha in [5000, 50000, 300000]:
        if time.time() - t0 > 55:
            break
        h_new = _slsqp(best_h, alpha, 2000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Active set descent
    if time.time() - t0 < budget_s - 750:
        h_new, c5 = _active_set_descent(best_h.copy(), 2500, 5, t0, budget_s - 700 + t0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 3: Exact pairwise (main workhorse)
    if time.time() - t0 < budget_s - 400:
        h_new, c5 = _pairwise_exact(best_h.copy(), 500000, rng, best_c5, t0, t0 + budget_s - 380)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 4: Cycle active set + pairwise
    for _rnd in range(2):
        if time.time() - t0 > budget_s - 180:
            break
        h_new, c5 = _active_set_descent(best_h.copy(), 1500, 3, t0, t0 + 60)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        if time.time() - t0 > budget_s - 160:
            break
        h_new, c5 = _pairwise_exact(best_h.copy(), 300000, rng, best_c5, t0, t0 + budget_s - 150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 5: SLSQP polish
    if time.time() - t0 < budget_s - 70:
        h_new = _slsqp(best_h, 800000, 3000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 6: Final exact pairwise
    if time.time() - t0 < budget_s - 10:
        h_new, c5 = _pairwise_exact(best_h.copy(), 400000, rng, best_c5, t0, t0 + budget_s - 8)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
