import time

    lo, hi = -2.0, 2.0
    for _ in range(60):

        if np.clip(h - mid, 0, 1).sum() > target:

            hi = mid

def _sm_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)

            g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])

    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,

    out = np.zeros(n * factor)

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sm_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh == 0:
            g[:] += wk * (1.0 - h)
        elif sh > 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha,),
                       method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _worst_lag_dir(h, n):
    """Return gradient direction for the worst lag (scaled to unit norm)."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    k_w = c.argmax()
    sh = (n - 1) - k_w
    g = np.zeros(n)
    if sh == 0:
        g[:] = 1.0 - h
    elif sh > 0:
        g[:n-sh] += (1.0 - h[sh:])
        g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    g *= dx
    norm = np.linalg.norm(g)
    if norm < 1e-15:
        return None, k_w
    return g / norm, k_w

def _alt_worst_lag(h, n, n_rounds, t_end):
    """Iteratively minimize the worst lag via projected gradient."""
    c5 = _eval(h)
    for _ in range(n_rounds):
        if time.time() > t_end:
            break
        g, k_w = _worst_lag_dir(h, n)
        if g is None:
            break
        improved = False
        for lr in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4]:
            h_new = _proj_to_sum(h - lr * g, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-15:
                h = h_new
                c5 = c5_new
                improved = True
                break
        if not improved:

    return h, c5

def _block_sa(h, n, t_end, rng):
    c5 = _eval(h)
    best_c = c5
    best_h = h.copy()
    n_iter = 5000
    t0, t1 = 0.004, 5e-6

def _upsample(h, factor):
    n = len(h)
    out = np.zeros(n * factor)
    for i in range(n):
        out[i*factor:(i+1)*factor] = h[i]
    return out
