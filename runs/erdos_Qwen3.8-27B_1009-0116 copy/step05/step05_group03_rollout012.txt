    h = _project(np.array(initial_h_values, dtype=float), n / 2.0)
    dx = 2.0 / n
    best_h, best_c5 = h.copy(), _c5(h, dx)
    tgt = n / 2.0

    # Phase 1: Nesterov gradient (130s)
    t1 = t0 + 130
    h, c = _nesterov(best_h, dx, t1, top_k=15, temp0=35.0)
    if c < best_c5: best_h, best_c5 = h, c

    # Phase 2: SLSQP LSE multi-alpha (100s)
    t2 = t0 + 230
    for a in [200, 800, 3000, 8000]:
        if time.time() > t2 - 5: break
        hc = _slsqp_lse(best_h, dx, a, 400)
        hc = _project(hc, tgt)
        c2 = _c5(hc, dx)
        if c2 < best_c5: best_h, best_c5 = hc, c2

    # Phase 3: Pair swap (180s)
    t3 = t0 + 410
    h, c = _pair_swap(best_h, dx, rng, t3)
    if c < best_c5: best_h, best_c5 = h, c

    # Phase 4: Nesterov round 2 (80s)
    if time.time() < t0 + 500:
        h, c = _nesterov(best_h, dx, t0 + 495, top_k=20, temp0=60.0)
        if c < best_c5: best_h, best_c5 = h, c

    # Phase 5: SLSQP high alpha (60s)
    if time.time() < t0 + 560:
        for a in [15000, 50000]:
            if time.time() > t0 + 555: break
            hc = _slsqp_lse(best_h, dx, a, 350)
            hc = _project(hc, tgt)
            c2 = _c5(hc, dx)
            if c2 < best_c5: best_h, best_c5 = hc, c2

    # Phase 6: Multi-start (200s)
    t6 = t0 + 760
    for r in range(12):
        rem = budget_s - (time.time() - t0)
        if time.time() > t6 or rem < 320: break
        hp = _perturb_dns(best_h, rng, r % 4)
        hp, c = _nesterov(hp, dx, min(t6 - 3, time.time() + 15), top_k=12, temp0=30.0)
        if c < best_c5:
            best_h, best_c5 = hp, c
            hc = _slsqp_lse(best_h, dx, 10000, 250)
            hc = _project(hc, tgt)
            c2 = _c5(hc, dx)
            if c2 < best_c5: best_h, best_c5 = hc, c2
            hp, c = _pair_swap(best_h, dx, rng, min(t6 - 1, time.time() + 10))
            if c < best_c5: best_h, best_c5 = hp, c

    # Phase 7: Pair swap round 2 (100s)
    t7 = t0 + 860
    h, c = _pair_swap(best_h, dx, rng, t7)
    if c < best_c5: best_h, best_c5 = h, c

    # Phase 8: Fine tune (120s)
    t8 = t0 + 980
    h, c = _fine_tune(best_h, dx, rng, t8)
    if c < best_c5: best_h, best_c5 = h, c

    # Phase 9: Final SLSQP
    if time.time() < t0 + budget_s - 40:
        for a in [30000, 100000, 300000]:
            if time.time() > t0 + budget_s - 45: break
            hc = _slsqp_lse(best_h, dx, a, 250)
            hc = _project(hc, tgt)
            c2 = _c5(hc, dx)
            if c2 < best_c5: best_h, best_c5 = hc, c2

    # Phase 10: Final fine tune
    if time.time() < t0 + budget_s - 12:
        best_h, best_c5 = _fine_tune(best_h, dx, rng, t0 + budget_s - 12)

    best_h = _project(best_h, tgt)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
