def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

                    break
    return h

            f[sp:] += 1.0 - h[:-sp]
            f[:-sp] -= h[sp:]

            if cn < best - 1e-14:

                best = cn
                ok = True
                break
        if not ok:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)

def _proj(h, target):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _shift_grad(h, sh):
    n = len(h)
    f = np.zeros(n)
    if sh >= 0:
        if sh < n:
            f[:n-sh] += 1.0 - h[sh:]
            f[sh:] -= h[:n-sh]
    else:
        sp = -sh
        if sp < n:
            f[sp:] += 1.0 - h[:-sp]
            f[:-sp] -= h[sp:]
    return f

def _topm_descend(h, max_iters, m, decay, base_step):
    n = len(h)
    dx = 2.0 / n
    best = _eval(h)
    for _ in range(max_iters):

        top = np.argsort(c)[::-1][:m]
        tv = c[top]
        w = np.exp(-decay * (tv - tv.min()))

        g = np.zeros(n)
        for wk, k in zip(w, top):
            sh = (n - 1) - k
            g += wk * _shift_grad(h, sh)
        g *= dx
        ok = False
        for s in base_step * np.array([1., .5, .25, .1, .05, .02, .01, .005]):
            hn = _proj(h - s * g, n * 0.5)
            cn = _eval(hn)
            if cn < best - 1e-14:
                h, best, ok = hn, cn, True
                break
        if not ok:
            break
    return h, best

def _pair_grad(h, i, j):
    n = len(h)
    g = np.zeros(2*n-1)
    # Forward for i
    ks = max(i, 0); ke = min(i+n-1, 2*n-2)
    hidx = i+n-1 - np.arange(ks, ke+1)
    g[ks:ke+1] += 1.0 - h[hidx]
    # Backward for i

    if ks <= ke:
        hidx = i-n+1 + np.arange(ks, ke+1)
        g[ks:ke+1] -= h[hidx]
    # Forward for j (negative)
    ks = max(j, 0); ke = min(j+n-1, 2*n-2)
    hidx = j+n-1 - np.arange(ks, ke+1)
    g[ks:ke+1] -= 1.0 - h[hidx]
    # Backward for j (negative)
    ks = max(n-1-j, 0); ke = min(n-2+j, 2*n-2)
    if ks <= ke:
        hidx = j-n+1 + np.arange(ks, ke+1)
        g[ks:ke+1] += h[hidx]

def _fast_pair(h, n_rounds, rng, n_grid=25):

            h[j] = min(max(h[j]-dgrid[bi], 0), 1)
            c = c + dx * dgrid[bi] * g
            best = vals[bi]
    return h, best

                                   'jac':lambda x: np.ones(n)}],
                     options={'maxiter':maxit,'ftol':1e-18})
        return _repair(r.x, n)
    except: return _repair(h, n)
