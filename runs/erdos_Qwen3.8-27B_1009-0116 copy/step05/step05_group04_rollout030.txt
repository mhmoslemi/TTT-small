def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_single_lag(h, k, n):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        idx = np.arange(n - sh)
        g[idx] += (1.0 - h[idx + sh])
        if sh > 0:
            idx2 = np.arange(sh, n)
            g[idx2] -= h[idx2 - sh]
    else:
        sp = -sh
        idx = np.arange(sp, n)
        g[idx] += (1.0 - h[idx - sp])
        idx2 = np.arange(n - sp)
        g[idx2] -= h[idx2 + sp]
    return g

def _slsqp_opt(h, alpha, maxiter, ftol, n):
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_opt(h, alpha, steps, eta, n, deadline_fn):

            step = min(step * 1.03, eta * 3.0)
        else:

            h_try = _proj(h - step * v, target)
            h = h_try
    return h
