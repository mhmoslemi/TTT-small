                step = min(step * 1.05, 1.0)
                break

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target, iters=60):
    lo, hi = -5.0, 5.0
    for _ in range(iters):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-10:
        return h
    h = _proj_sum(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            adj = min(abs(diff), room)
            if adj < 1e-15:
                continue
            h[idx] += adj if diff > 0 else -adj
            diff -= adj if diff > 0 else -adj
            if abs(diff) < 1e-12:
                break
    return h

def _lag_grad_vec(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(n - s)
        g[i1] += (1.0 - h[i1 + s])
        if s > 0:
            i2 = np.arange(s, n)
            g[i2] -= h[i2 - s]
    else:
        sp = -s
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _topk_descent(h, n_iter, K, t_end, rng, step0=0.1):
    n = len(h)
    h_cur = h.copy()
    c_cur = _eval(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    step = step0
    v = np.zeros(n)
    mu = 0.75
    stall = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(c)[::-1][:K]
        p75 = np.percentile(c, 70)
        w = np.maximum(c[top_idx] - p75, 0.0) ** 1.5
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.ones(K)
        else:
            w /= wsum
        g = np.zeros(n)
        for j, k in enumerate(top_idx):
            g += w[j] * _lag_grad_vec(h_cur, k)
        g -= g.mean()
        v = mu * v + (1.0 - mu) * g
        accepted = False
        for trial in range(12):
            h_new = _proj_sum(h_cur - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < c_cur - 1e-14:
                h_cur = h_new
                c_cur = c_new
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = h_cur.copy()
                    stall = 0
                step = min(step * 1.04, step0 * 3.0)
                accepted = True
                break
            step *= 0.55
        if not accepted:
            stall += 1
            step = max(step * 2.0, step0 * 0.1)
            if stall > 150:
                v = np.zeros(n)
                h_cur = _repair(h_cur + 0.01 * rng.randn(n), n)
                c_cur = _eval(h_cur)
                stall = 0
    return best_h, best_c

def _slsqp_smooth(h, alpha, maxiter, ftol):
    n = len(h)
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        dxv = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dxv
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pair_local(h, n_tries, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 4e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-17:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _upsample(h, factor):
    return np.repeat(h, factor)

def _smooth_filter(h, size=5):
    pad = size // 2
    hc = np.concatenate([h[:pad][::-1], h, h[-pad:][::-1]])
    w = np.ones(size) / size
    out = np.convolve(hc, w, mode='valid')[:len(h)]
    return out
