    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sg(h, alpha):
    n = len(h); dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s); w = e / e.sum()
    val = m + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += w[k] * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w[k] * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n-sp])
            g[:n-sp] -= w[k] * h[sp:]
    return val, dx * g

def _pg_step(h, alpha, max_iter, t_end):
    n = len(h); dx = 2.0 / n
    best_c = _eval(h)
    val, g = _sg(h, alpha)
    lr = 0.5 / (np.max(np.abs(g)) + 1e-12)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        h_new = _proj_to_sum(h - lr * g, n * 0.5)
        val_n, _ = _sg(h_new, alpha)
        if val_n < val:
            h = h_new; val = val_n; lr = min(lr * 1.5, 1.0)
            c5 = _eval(h)
            if c5 < best_c:
                best_c = c5
            g = _sg(h, alpha)[1]
            if it % 50 == 0 and it > 0:
                alpha = min(alpha * 1.8, 2e5)
        else:
            lr *= 0.5
            g = _sg(h, alpha)[1]
        if lr < 1e-12:
            break
    return h, best_c

def _active(h, k_top, max_iter, t_end):
    n = len(h); dx = 2.0 / n
    best_c = _eval(h)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:k_top]
        cvals = c[idx]
        base = cvals.mean()
        wts = np.maximum(cvals - base, 0.0)
        ws = wts.sum()
        if ws < 1e-14:
            wts = np.ones(k_top) / k_top
        else:
            wts /= ws
        g = np.zeros(n)
        for wt, k in zip(wts, idx):
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += wt * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wt * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += wt * (1.0 - h[:n-sp])
                g[:n-sp] -= wt * h[sp:]
        g *= dx
        improved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < best_c - 1e-16:
                h = h_new; best_c = c5n; improved = True
                break
        if not improved:
            break
    return h, best_c

def _golden_pair(h, n_trials, t_end, rng):
    n = len(h); dx = 2.0 / n
    best_c = _eval(h)
    for _ in range(n_trials):
        if time.time() > t_end:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dhi - dlo < 1e-10:
            continue
        c0 = np.correlate(h, 1.0 - h, mode='full')
        a = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            sh = (n - 1) - k
            ci = 0.0
            if sh >= 0:
                if i < n - sh:
                    ci += (1.0 - h[i + sh])
                if 0 <= i - sh < n - sh:
                    ci -= h[i - sh]
                if j < n - sh:
                    ci += (1.0 - h[j + sh])
                if 0 <= j - sh < n - sh:
                    ci -= h[j - sh]
            else:
                sp = -sh
                if sp <= i:
                    ci += (1.0 - h[i - sp])
                if i + sp < n - sp:
                    ci -= h[i + sp]
                if sp <= j:
                    ci += (1.0 - h[j - sp])
                if j + sp < n - sp:
                    ci -= h[j + sp]
            a[k] = ci
        gr = np.max(c0 + dx * d * a)
        lo_d, hi_d = dlo, dhi
        g = (np.sqrt(5) - 1) / 2
        c1 = lo_d + g * (hi_d - lo_d)
        c2 = lo_d + (1 - g) * (hi_d - lo_d)
        f1 = np.max(c0 + dx * c1 * a); f2 = np.max(c0 + dx * c2 * a)
        for _gs in range(30):
            if f1 < f2:
                hi_d = c2; c2 = c1; f2 = f1
                c1 = lo_d + g * (hi_d - lo_d); f1 = np.max(c0 + dx * c1 * a)
            else:
                lo_d = c1; c1 = c2; f1 = f2
                c2 = lo_d + (1 - g) * (hi_d - lo_d); f2 = np.max(c0 + dx * c2 * a)
        best_d = (lo_d + hi_d) * 0.5
        if abs(best_d) > 1e-12:
            h[i] += best_d; h[j] -= best_d
            h = np.clip(h, 0, 1)
            c5n = _eval(h)
            if c5n < best_c - 1e-16:
                best_c = c5n
            else:
                h[i] -= best_d; h[j] += best_d
    return h, best_c

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    def obj(x):
        return _sg(x, alpha)[0]
    def jac(x):
        return _sg(x, alpha)[1]
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h); best_n = n0

    # Phase 1: Projected gradient with annealing on n0
    end = t0 + 180
    h, c5 = _pg_step(h.copy(), 50, 20000, end)
    if c5 < best_c5:
        best_c5 = c5; best_h = h.copy()

    # Phase 2: SLSQP with increasing alpha on n0
    end = t0 + 400
    for alpha in [500, 3000, 20000, 100000, 500000]:
        if _el() > budget_s - 400:
            break
        h_new = _slsqp_opt(best_h.copy(), alpha, 3000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new

    # Phase 3: Active descent on n0
    end = t0 + 600
    if _el() < budget_s - 300:
        h, c5 = _active(best_h.copy(), 5, 5000, min(end, t0 + budget_s - 250))
        if c5 < best_c5:
            best_c5 = c5; best_h = h

    # Phase 4: Golden-section pair on n0
    if _el() < budget_s - 200:
        h, c5 = _golden_pair(best_h.copy(), 8000, t0 + budget_s - 180, rng)
        if c5 < best_c5:
            best_c5 = c5; best_h = h

    # Phase 5: Upsample to 2x and optimize
    if _el() < budget_s - 350:
        n1 = n0 * 2
        h_hi = np.zeros(n1)
        for i in range(n0):
            h_hi[i*2:i*2+2] = best_h[i]
        h_hi = _repair(h_hi, n1)
        end5 = t0 + budget_s - 300
        h_hi, c5 = _pg_step(h_hi, 100, 15000, end5)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_hi; best_n = n1
        if _el() < budget_s - 250:
            h_hi = _slsqp_opt(best_h.copy(), 50000, 2000)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_hi; best_n = n1
        if _el() < budget_s - 150:
            h2, c5 = _active(best_h.copy(), 4, 4000, t0 + budget_s - 130)
            if c5 < best_c5:
                best_c5 = c5; best_h = h2; best_n = len(h2)
        if _el() < budget_s - 80:
            h2, c5 = _golden_pair(best_h.copy(), 6000, t0 + budget_s - 70, rng)
            if c5 < best_c5:
                best_c5 = c5; best_h = h2; best_n = len(h2)

    # Phase 6: Multi-start on best_n
    n_cur = best_n
    if _el() < budget_s - 200:
        for _r in range(15):
            if _el() > budget_s - 150:
                break
            s = rng.randint(4)
            if s == 0:
                hp = _repair(best_h + 0.008 * rng.randn(n_cur), n_cur)
            elif s == 1:
                hp = best_h.copy()
                bs = rng.randint(4, n_cur // 8)
                st = rng.randint(0, n_cur - bs)
                hp[st:st+bs] = 1.0 - hp[st:st+bs]
                hp = _repair(hp, n_cur)
            elif s == 2:
                hp = best_h.copy()
                t = np.arange(n_cur)
                f = rng.randint(2, n_cur // 3 + 1)
                hp += 0.003 * np.sin(2 * np.pi * f * t / n_cur)
                hp = _repair(np.clip(hp, 0, 1), n_cur)
            else:
                hp = _repair(np.clip(best_h + 0.03 * (rng.rand(n_cur) - 0.5), 0, 1), n_cur)
            h_new, c5 = _pg_step(hp, 200, 5000, t0 + budget_s - 140 - _r * 8)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new; best_n = n_cur

    # Phase 7: Final long active + golden
    if _el() < budget_s - 100:
        h, c5 = _active(best_h.copy(), 3, 8000, t0 + budget_s - 80)
        if c5 < best_c5:
            best_c5 = c5; best_h = h
    if _el() < budget_s - 40:
        h, c5 = _golden_pair(best_h.copy(), 10000, t0 + budget_s - 30, rng)
        if c5 < best_c5:
            best_c5 = c5; best_h = h

    # Phase 8: Final SLSQP polish
    if _el() < budget_s - 10:
        h = _slsqp_opt(best_h.copy(), 1e6, 3000)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
