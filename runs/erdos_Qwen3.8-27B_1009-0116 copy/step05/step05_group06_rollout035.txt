def _minmax_lines(d, e, lo=0.0, hi=1.0):
    """Minimize max_k (d[k] + e[k]*t) over t in [lo, hi]."""
    # Evaluate at endpoints
    f_lo = np.max(d + e * lo)
    f_hi = np.max(d + e * hi)
    
    if f_lo <= f_hi:
        best_t = lo
    else:
        best_t = hi
    
    # Find the k that achieves max at lo and hi
    k_lo = np.argmax(d + e * lo)
    k_hi = np.argmax(d + e * hi)
    
    if k_lo != k_hi:
        # Try intersection
        de = e[k_hi] - e[k_lo]
        if abs(de) > 1e-15:
            t_int = -(d[k_hi] - d[k_lo]) / de
            if lo < t_int < hi:
                f_int = np.max(d + e * t_int)
                if f_int < min(f_lo, f_hi):
                    best_t = t_int
    
    # Verify with a few evaluations (handle case where intermediate line is max)
    for t_check in np.linspace(lo, hi, 7):
        f_check = np.max(d + e * t_check)
        f_best = np.max(d + e * best_t)
        if f_check < f_best:
            best_t = t_check
    
    # Final refinement with golden section
    a_t, b_t = lo, hi
    gr = (np.sqrt(5) - 1) / 2
    fm = a_t + (1-gr)*(b_t-a_t)
    fn = a_t + gr*(b_t-a_t)
    fm_val = np.max(d + e*fm)
    fn_val = np.max(d + e*fn)
    for _ in range(20):

            b_t = fn
            fn = fm
            fn_val = fm_val
            fm = a_t + (1-gr)*(b_t-a_t)
            fm_val = np.max(d + e*fm)
        else:
            a_t = fm
            fm = fn
            fm_val = fn_val
            fn = a_t + gr*(b_t-a_t)
            fn_val = np.max(d + e*fn)
    
    best_t = (a_t + b_t) / 2.0
    return best_t

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)

        else:

def _coord_sweep(h, c, n, rng, end_time, omega=1.0):
    perm = rng.permutation(n)
    for i in perm:
        if time.time() > end_time:
            break
        B = _compute_B(h, i, n)
        delta = _minmax_1d(c, B, -h[i], 1.0 - h[i])
        delta *= omega
        new_val = h[i] + delta
        if 0.0 <= new_val <= 1.0:
            c += B * delta
            h[i] = new_val

def _upsample(h, factor):
