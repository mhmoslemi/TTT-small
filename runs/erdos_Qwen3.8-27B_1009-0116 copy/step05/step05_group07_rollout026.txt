If feasible, we can take a step in direction d. If infeasible (i.e., the active set is "too large"), we need to reduce the active set size or use a different approach.

Actually, a simpler approach: minimize the SUM of the top-M correlations (not the max). This is a smooth objective when M is fixed, and its gradient is the average of the top-M gradients. Minimizing the sum pushes down all top correlations, and the max will decrease.

Let me implement this:

```python
def _minimize_top_sum(h, M, n_steps, lr):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best = _eval(h)
    for step in range(n_steps):
        c = _all_corrs(h)
        top_M = np.argpartition(c, -M)[-M:]
        g = np.zeros(n)
        for k in top_M:
            g += _grad_lag(k, h)
        g /= M
        # Line search
        step_size = lr
        for _ in range(20):
            h_new = _repair(h - step_size * g, n)
            f_new = _eval(h_new)
            if f_new < best:
                h = h_new
                best = f_new
                break
            step_size *= 0.5
        if step_size < 1e-10:
            break
    return h, best
```

This is clean and fast. Each iteration takes O(n^2) for the correlation + O(M*n) for the gradient + O(M*n^2) for the line search evaluations. For n=280, M=20: about 20 * 78000 = 1.5M ops per line search step. Very fast.

Hmm, the Nesterov implementation above is a bit rough. Let me think about it more carefully.

Standard Nesterov momentum:
- y = h - beta * v (lookahead)
- h_new = proj(y - lr * grad(y))  [gradient at lookahead point]
- v = h_new - h
- h = h_new

But computing grad at y requires an extra correlation evaluation. For simplicity, let me use:
- y = h - beta * v
- h_new = proj(h - lr * grad(h))  [gradient at current point, but use y for projection]

Actually, the standard accelerated gradient method is:
- y_t = h_t + (β_{t-1}/β_t) * (h_t - h_{t-1})
- h_{t+1} = proj(y_t - α_t * ∇f(y_t))

Where α_t = 2/(t+2) for convex problems. But our problem isn't convex, so let me just use a fixed momentum:
- h_{t+1} = proj(h_t - α_t * (∇f(h_t) + μ * (h_t - h_{t-1})))

This is heavy-ball momentum. Simpler and works well in practice.

Let me also add: when we detect that we're stuck (no improvement for 1000 iterations), do a "restart" with a different M or a perturbation.

import numpy as np
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _proj(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(k, h):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += 1.0 - h[s:]
        if s > 0:
            g[s:] -= h[:n - s]
    else:
        sp = -s
        g[sp:] += 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g * (2.0 / n)

def _subgrad(h, M, n_iter, lr, mu, seed, kick_ev=3000, kick_sz=0.004):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    hp = h.copy()
    f = _eval(h)
    best, bh = f, h.copy()
    stuck = 0
    for it in range(n_iter):
        if time.time() - _t0 > _budget - 15:
            break
        c = _corrs(h)
        idx = np.argpartition(c, -M)[-M:] if M < len(c) else np.arange(len(c))
        g = np.zeros(n)
        for k in idx:
            g += _grad_k(k, h)
        g /= len(idx)
        h_new = _proj(h - lr * (g + mu * (h - hp)), n)
        hp = h.copy()
        h = h_new
        f = _eval(h)
        if f < best - 1e-13:
            best, bh, stuck = f, h.copy(), 0
        else:
            stuck += 1
        if stuck > kick_ev:
            h = _proj(h + kick_sz * rng.randn(n), n)
            hp = h.copy()
            stuck = 0
            f = _eval(h)
            if f < best:
                best, bh = f, h.copy()
    return bh, best

def _subgrad_smooth(h, n_iter, lr, mu, alpha, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    hp = h.copy()
    f = _eval(h)
    best, bh = f, h.copy()
    stuck = 0
    for it in range(n_iter):
        if time.time() - _t0 > _budget - 15:
            break
        c = _corrs(h)
        s = np.clip(alpha * (c - c.max()), -200, 200)
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] > 1e-12:
                g += w[k] * _grad_k(k, h)
        h_new = _proj(h - lr * (g + mu * (h - hp)), n)
        hp = h.copy()
        h = h_new
        f = _eval(h)
        if f < best - 1e-13:
            best, bh, stuck = f, h.copy(), 0
        else:
            stuck += 1
        if stuck > 3000:
            h = _proj(h + 0.004 * rng.randn(n), n)
            hp = h.copy()
            stuck = 0
    return bh, best

def _pairwise(h, n_pairs, scale, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    f = _eval(h)
    bh = h.copy()
    for _ in range(n_pairs):
        if time.time() - _t0 > _budget - 8:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            fn = _eval(h)
            if fn < f - 1e-15:
                f, bh = fn, h.copy()
            else:
                h[i], h[j] = hi - d, hj + d
    return bh, f

def _upsample(h, factor):
    n = len(h)
    x0 = np.linspace(0, 2, n, endpoint=False)
    x1 = np.linspace(0, 2, n * factor, endpoint=False)
    fi = interp1d(x0, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _proj(np.clip(fi(x1), 0, 1), n * factor)

def _make_tm(n):
    t = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)
    return _proj(t, n)

def _make_sine(n, freq, amp):
    x = np.linspace(0, 2, n, endpoint=False)
    return _proj(0.5 + amp * np.sin(np.pi * freq * x), n)

def _symm(h):
    return _proj((h + h[::-1]) * 0.5, len(h))
