def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):

def _obj_logsumexp(x, alpha):

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)

        return h.copy()

    """Barzilai-Borwein step"""
    g = _grad_logsumexp(h, alpha)
    if prev_h is not None and prev_g is not None:
        s = h - prev_h
        y = g - prev_g
        sy = s @ y
        yy = y @ y
        ss = s @ s
        if yy > 1e-16 and ss > 1e-16:

    """Pairwise coordinate descent (mass transfer)"""
    rng = np.random.RandomState(12345)
    cur_c = _eval(h)
    for scale in scales:

            ni, nj = h[i] + d, h[j] - d

            if c_new < cur_c - 1e-15:

        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(max(0, n - sh))
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _reweight(h, n_iter, top_m, t_end):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-top_m:]
        cmin, cmax = c[idx].min(), c[idx].max()
        w = np.zeros(2 * n - 1)
        if cmax - cmin > 1e-12:
            w[idx] = np.exp(5.0 * (c[idx] - cmin) / (cmax - cmin))
        else:
            w[idx] = 1.0
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(max(0, n - sh))
                g[i1] += wk * (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * h[i2 + sp]
        moved = False
        for step in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
            ht = _repair(h - step * g, n)
            ct = _eval(ht)
            if ct < cur - 1e-14:
                h = ht; cur = ct; moved = True
                break
        if not moved:
            break
    return h, cur

def _binary_swap(h, n, max_rounds, t_end):
    h = h.copy()
    h[h > 0.5] = 1.0; h[h <= 0.5] = 0.0
    h = _repair(h, n)
    cur = _eval(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in ones.copy():
            for j in zeros.copy():
                h[i] = 0.0; h[j] = 1.0
                c2 = _eval(h)
                if c2 < cur - 1e-13:
                    cur = c2
                    improved = True
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    break
                else:
                    h[i] = 1.0; h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, cur

def _pair_cd(h, n, n_iter, scales, t_end):
    rng = np.random.RandomState(777)
    cur = _eval(h)
    for scale in scales:
        for _ in range(n_iter):
            if time.time() > t_end:
                return h, cur
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur - 1e-15:
                cur = cn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, cur

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def rem():
        return budget_s - elapsed()

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP sweep at original resolution
    for alpha in [5000, 20000, 100000, 500000]:
        if rem() < 150:
            break
        hn = _slsqp(best_h, alpha, 4000, 1e-20)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5; best_h = hn.copy(); best_n = len(hn)

    # Phase 2: Reweiting descent on worst shifts
    if rem() > 120:
        h2 = best_h.copy()
        h2, c5 = _reweight(h2, 5000, 6, t0 + rem() - 30)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2.copy(); best_n = len(h2)
        # Second pass with fewer top shifts
        if rem() > 60:
            h2b = best_h.copy()
            h2b, c5 = _reweight(h2b, 3000, 3, t0 + rem() - 20)
            if c5 < best_c5:
                best_c5 = c5; best_h = h2b.copy(); best_n = len(h2b)

    # Phase 3: Multi-resolution
    if rem() > 200:
        for n_try in [280, 320, 340, 400, 500]:
            if rem() < 180:
                break
            hr = _repair(_resample(best_h, n_try), n_try)
            for alpha in [50000, 300000]:
                if rem() < 170:
                    break
                hr = _slsqp(hr, alpha, 3000, 1e-19)
            c5 = _eval(hr)
            if c5 < best_c5:
                best_c5 = c5; best_h = hr.copy(); best_n = n_try

    # Phase 4: Binary swap optimization
    if rem() > 80:
        for nb in [best_n, best_n - 20, best_n + 20]:
            if rem() < 70:
                break
            if nb < 60:
                continue
            hb = _repair(_resample(best_h, nb), nb)
            hb, c5 = _binary_swap(hb, nb, 800, t0 + rem() - 10)
            if c5 < best_c5:
                best_c5 = c5; best_h = hb.copy(); best_n = nb

    # Phase 5: High-alpha SLSQP polish
    if rem() > 60:
        h5 = _slsqp(best_h, 1000000, 6000, 1e-21)
        c5 = _eval(h5)
        if c5 < best_c5:
            best_c5 = c5; best_h = h5.copy(); best_n = len(h5)

    # Phase 6: Final reweight + pairwise
    if rem() > 40:
        h6 = best_h.copy()
        h6, c5 = _reweight(h6, 2000, 2, t0 + rem() - 15)
        if c5 < best_c5:
            best_c5 = c5; best_h = h6.copy(); best_n = len(h6)

    if rem() > 15:
        h7 = best_h.copy()
        h7, c5 = _pair_cd(h7, len(h7), 5000, [0.004, 0.001, 0.0003], t0 + rem() - 5)
        if c5 < best_c5:
            best_c5 = c5; best_h = h7.copy()

    # Phase 7: Final SLSQP
    if rem() > 10:
        hf = _slsqp(best_h, 2000000, 8000, 1e-20)
        c5 = _eval(hf)
        if c5 < best_c5:
            best_c5 = c5; best_h = hf.copy(); best_n = len(hf)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
