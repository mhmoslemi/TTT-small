            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _topk_subgrad(h, topk, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        g += _grad_shift(h, sh, n)
    return g * dx / topk, c

def _momentum_descent(h0, topk, max_iter, step0=0.05, decay=0.999):
    n = len(h0)
    target = n * 0.5
    dx = 2.0 / n
    h = _proj(h0.copy(), target)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    m_t, v_t = 0.0, 0.0
    best = _eval_c5(h)
    step = step0
    for it in range(max_iter):
        g, c = _topk_subgrad(h, topk, dx)
        m_t = beta1 * m_t + (1 - beta1) * g
        v_t = beta2 * v_t + (1 - beta2) * g * g
        m_hat = m_t / (1 - beta1 ** (it + 1))
        v_hat = v_t / (1 - beta2 ** (it + 1))
        h_new = h - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = _proj(h_new, target)
        c5 = _eval_c5(h_new)
        if c5 < best - 1e-15:
            h = h_new
            best = c5
            step = min(step * 1.05, 0.2)
        else:
            step *= 0.8
            if step < 1e-6:
                break
    return h, best

def _fista(h0, topk, max_iter):
    n = len(h0)
    target = n * 0.5
    dx = 2.0 / n
    h = _proj(h0.copy(), target)
    h_prev = h.copy()
    y = h.copy()
    t = 1.0
    step = 0.1
    best = _eval_c5(h)
    for it in range(max_iter):
        g, c = _topk_subgrad(y, topk, dx)
        # backtracking
        h_trial = _proj(y - step * g, target)
        c5_trial = _eval_c5(h_trial)
        while c5_trial > best and step > 1e-8:
            step *= 0.5
            h_trial = _proj(y - step * g, target)
            c5_trial = _eval_c5(h_trial)
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y_new = h_trial + (t - 1) / t_new * (h_trial - h)
        h, h_prev, t, y = h_trial, h, t_new, y_new
        if c5_trial < best:
            best = c5_trial
    return h, best

def _pairwise(h, n_iter, rng, scales=None):

    c5c = _eval_c5(h)
    if scales is None:

            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _slsqp_polish(h, maxiter):
    from scipy.optimize import minimize
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    alpha = 200000
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * dx
    
    def cons():
        return x.sum() - target
    
    bnds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _proj(res.x, target)
    except:
        return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

    g = np.zeros(n)

        nm = n - sh
        g[:nm] += (1.0 - h[sh:n])
        if sh > 0:
            g[sh:] -= h[:nm]

        nm = n - sp
