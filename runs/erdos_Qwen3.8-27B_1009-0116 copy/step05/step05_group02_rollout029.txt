```python
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:

        for idx in order:

                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _eval_all(h):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lse_obj_grad(h, alpha):
    n = len(h)
    c = _eval_all(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -50, 50)
    w = np.exp(s - s.max())
    w /= w.sum()
    obj = (m + np.log(np.sum(np.exp(s))) / alpha)
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return obj, g

def _nesterov_lse(h0, max_iter, alpha, time_limit):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    mu = 0.85
    best_h = h.copy()
    best_c = _eval(h)
    t_start = time.time()

        obj, g = _lse_obj_grad(y, alpha)
```

Let me also add a key innovation: **gradient-guided pairwise search**. Instead of random pairwise perturbations, use the gradient to determine the best pair to perturb.

For the max-lag subgradient g, the best single-coordinate move is: increase h[j] where g[j] < 0 (reduces the max lag), or decrease h[j] where g[j] > 0. The best pairwise move preserves the sum constraint: increase h[i] (where g[i]<0) and decrease h[j] (where g[j]>0).

This is much more directed than random pairwise search!

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _eval_all(h):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _lag_grad(h, sh):
    n = len(h)
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        if m > 0:
            g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh
        m = n - sp
        g[sp:] += (1.0 - h[:m])
        g[:m] -= h[sp:]
    return g

def _worst_k_grad(h, k):
    c = _eval_all(h)
    idx = np.argpartition(c, -k)[-k:]
    n = len(h)
    g = np.zeros(n)
    for ki in idx:
        g += _lag_grad(h, (n - 1) - ki)
    return g / k, c, idx

def _grad_descent(h0, max_iter, time_limit, k=1, lr0=0.01):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = _eval(h)
    t_s = time.time()
    lr = lr0
    for it in range(max_iter):
        if time.time() - t_s > time_limit:
            break
        g, c, _ = _worst_k_grad(h, k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        h_new = _proj_sum(h - lr * g / gn, n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
            lr *= 0.9
        if lr < 1e-8:
            lr = lr0 * 0.1
    return best_h, best_c

def _nesterov_lse(h0, max_iter, alpha, time_limit):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    mu = 0.8
    best_h = h.copy()
    best_c = _eval(h)
    t_s = time.time()
    for it in range(max_iter):
        if time.time() - t_s > time_limit:
            break
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0, 1)
        c = _eval_all(y)
        m = c.max()
        s = np.clip(alpha * (c - m), -50, 50)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for ki in range(2 * n - 1):
            if w[ki] < 1e-13:
                continue
            g += w[ki] * _lag_grad(y, (n - 1) - ki)
        for lr in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = _proj_sum(y - lr * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new <= best_c + 1e-12:
                break
        h_prev = h
        h = h_new
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
    return best_h, best_c

def _pairwise_grad_search(h0, max_iter, time_limit, top_k=1):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = _eval(h)
    t_s = time.time()
    for it in range(max_iter):
        if time.time() - t_s > time_limit:
            break
        g, c, _ = _worst_k_grad(h, top_k)
        # Find best (i,j) pair: increase i, decrease j
        pos = np.argsort(-g)  # most negative g first (want to increase)
        neg = np.argsort(g)   # most positive g first (want to decrease)
        found = False
        for a in range(min(10, n)):
            for b in range(min(10, n)):
                i, j = pos[a], neg[b]
                if i == j:
                    continue
                for d in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                    if h[i] + d > 1 or h[j] - d < 0:
                        continue
                    h[i] += d
                    h[j] -= d
                    c_new = _eval(h)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        found = True
                        break
                    h[i] -= d
                    h[j] += d
                if found:
                    break
            if found:
                break
        if not found:
            # Random perturbation to escape
            i, j = np.random.randint(n), np.random.randint(n)
            if i != j:
                d = 0.002 * np.random.randn()
                if 0 <= h[i]+d <= 1 and 0 <= h[j]-d <= 1:
                    h[i] += d
                    h[j] -= d
                    if _eval(h) < best_c + 0.001:
                        pass  # accept even if not improving (annealing-like)
    return best_h, best_c

def _downsample(h, new_n):
    n = len(h)
    return np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, n), h)

def _upsample(h, new_n):
    n = len(h)
    return np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, n), h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _eval_all(h):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _lag_grad(h, sh):
    n = len(h)
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        if m > 0:
            g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh; m = n - sp
        g[sp:] += (1.0 - h[:m])
        g[:m] -= h[sp:]
    return g

def _worst_k_grad(h, k):
    c = _eval_all(h)
    idx = np.argpartition(c, -k)[-k:]
    n = len(h)
    g = np.zeros(n)
    for ki in idx:
        g += _lag_grad(h, (n - 1) - ki)
    return g / k, c

def _grad_descent(h0, max_iter, time_limit, k=1, lr0=0.02):
    n = len(h0)
    h = h0.copy(); best_h = h.copy(); best_c = _eval(h)
    t_s = time.time(); lr = lr0
    for it in range(max_iter):
        if time.time() - t_s > time_limit:
            break
        g, _ = _worst_k_grad(h, k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        h_new = _proj_sum(h - lr * g / gn, n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new; best_h = h_new.copy(); h = h_new
        else:
            h = h_new; lr *= 0.9
        if lr < 1e-9:
            lr = lr0 * 0.05
    return best_h, best_c

def _nesterov_lse(h0, max_iter, alpha, time_limit):
    n = len(h0)
    h = h0.copy(); h_prev = h0.copy(); mu = 0.8
    best_h = h.copy(); best_c = _eval(h)
    t_s = time.time()
    for it in range(max_iter):
        if time.time() - t_s > time_limit:
            break
        y = np.clip(h + mu * (h - h_prev), 0, 1)
        c = _eval_all(y); m = c.max()
        s = np.clip(alpha * (c - m), -50, 50)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n)

            h_new = _proj_sum(y - lr * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new <= best_c + 1e-12:
                break
        h_prev = h; h = h_new
        if c_new < best_c:

    n = len(h)
    return np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, n), h)

    c = _eval_all(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

    s = np.clip(alpha * (c - m), -500, 500)

    return np.ones(len(x))

    return x.sum() - len(x) * 0.5
```
