def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    m = c.max()

    return x.sum() - len(x) * 0.5

But there are O(n²) breakpoints. However, only the ones corresponding to the "upper envelope" matter. The upper envelope of n lines has at most n breakpoints.

To find the minimum of the upper envelope:
1. Sort the lines by slope (a_k)
2. Compute the upper envelope (convex hull in 1D)
3. The minimum of the upper envelope is at the "valley" point

This is the "minimize the maximum of linear functions" problem, which can be solved in O(n log n) time.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, n):
    dc = np.zeros(2 * n - 1)
    sh1 = np.arange(0, n - i)
    dc[(n - 1) - sh1] += (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    dc[(n - 1) - sh2] -= h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= h[i + sp[m2]]
    return dc

def _compute_E(h, n):
    return np.array([_pos_change(h, i, n) for i in range(n)])

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pair_search(h, c, n, max_sweeps, elapsed_fn, limit_fn):
    total_changes = 0
    for sweep in range(max_sweeps):
        if limit_fn():
            break
        E = _compute_E(h, n)
        n_changes = 0
        for i in range(n):
            for j in range(i + 1, n):
                a = E[i] - E[j]
                d_min = max(-h[i], h[j] - 1.0)
                d_max = min(1.0 - h[i], h[j])
                if d_min >= d_max - 1e-12:
                    continue
                k_star = c.argmax()
                a_k = a[k_star]
                candidates = [d_min, d_max]
                if abs(a_k) > 1e-12:
                    scale = 0.5 / max(np.abs(a).max(), 1e-10)
                    for mult in (0.25, 0.7, 2.0, 6.0):
                        candidates.append(float(np.clip(-a_k * scale * mult, d_min, d_max)))
                best_d = 0.0
                best_val = c.max()
                for d in candidates:
                    if abs(d) < 1e-13:
                        continue
                    val = (c + d * a).max()
                    if val < best_val - 1e-11:
                        best_val = val
                        best_d = d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    c += best_d * a
                    n_changes += 1
                    total_changes += 1
        if n_changes == 0:
            break
    return h, c, total_changes

def _incr_pair(h, c, rng, n_iter, scale, n):
    dx = 2.0 / n
    c5c = c.max() * dx
    accepted = 0
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        Ei = _pos_change(h, i, n)
        Ej = _pos_change(h, j, n)
        new_c = c + d * (Ei - Ej)
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-15:
            h[i] = ni
            h[j] = nj
            c = new_c
            c5c = new_c5
            accepted += 1
    return h, c, c5c
