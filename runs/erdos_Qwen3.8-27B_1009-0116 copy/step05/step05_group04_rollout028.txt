def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Adjust: redistribute
    diff = target - s
    if abs(diff) > 1e-12:
        # Proportional adjustment
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        # Fine-tune
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) <= 1e-12:
        return h
    h = np.clip(h + diff / n, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m * 2.0 / n + np.log(np.sum(np.exp(s))) / alpha
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - x[sh:n])
                if sh > 0:
                    g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g

        return x.sum() - n * 0.5
    def _cjac(x):
        return np.ones(n)
    try:

                       bounds=bnds, constraints=[{'type':'eq','fun':_cfun,'jac':_cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

Actually, one more idea: **iterative reweighting on the correlation**. 

At the optimum, the worst lags should all have approximately the same value C₅. If one lag k* has c[k*] > C₅ and another k** has c[k**] < C₅, we can potentially reduce k* while not increasing k**.

Algorithm:
1. Compute c = correlate(h, 1-h) * dx
2. Find the top K lags (K around 3-10)
3. Compute the sum of their gradients (equal weight for each active lag)
4. Take a step in the negative direction
5. Project to feasible set
6. Repeat

This is a "balanced subgradient" method. The key insight vs. the previous approach: instead of using a smooth max surrogate (which weights all lags by softmax), I explicitly target only the top K lags with equal weight.

def _balanced_subgrad(h, n, tlim, K=5, lr0=0.01):
    t0 = time.time()
    c = _eval(h)
    best_h, best_c = h.copy(), c
    lr = lr0
    while time.time() - t0 < tlim:
        cc = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        # Top K lags
        top_k = np.argsort(cc)[-K:]
        # Combined gradient
        g = np.zeros(n)
        for k in top_k:
            sh = (n-1) - k
            if sh >= 0:
                g[:n-sh] += (1.0 - h[sh:n])
                if sh > 0:
                    g[sh:] -= h[:n-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n-sp])
                g[:n-sp] -= h[sp:]
        g *= (2.0/n) / K
        # Backtracking line search
        for lr_try in [lr, lr*0.5, lr*0.25, lr*0.1]:

            cn = _eval(h_new)

                h = h_new; c = cn; lr = min(lr_try * 1.05, lr0)
                if cn < best_c:
                    best_c = cn; best_h = h.copy()
                break
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) <= 1e-12:
        return h
    h = np.clip(h + diff / n, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m * 2.0 / n + 1e-3 / alpha * np.log(np.sum(np.exp(s)))

def _smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:n])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _slsqp(h, alpha, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(len(x))
    def _f(x):
        return _smooth_obj(x, n, alpha)
    def _g(x):
        return _smooth_grad(x, n, alpha)
    try:
        res = minimize(_f, h, jac=_g, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_search(h0, n, tlim, rng):
    t0 = time.time()
    h_b = np.where(h0 > 0.5, 1.0, 0.0).copy()
    n1 = int(round(h_b.sum()))
    if n1 > n // 2:

        h_b[idx[:n1 - n // 2]] = 0
    elif n1 < n // 2:
        idx = np.where(h_b == 0)[0]; rng.shuffle(idx)
        h_b[idx[:n // 2 - n1]] = 1
    c = _eval(h_b)
    best_h, best_c = h_b.copy(), c
    stall = 0
    while time.time() - t0 < tlim:
        h_b2 = h_b.copy()
        oi = np.where(h_b2 == 1)[0].copy()
        zi = np.where(h_b2 == 0)[0].copy()
        rng.shuffle(oi); rng.shuffle(zi)
        imp = False
        for i in oi:
            for j in zi:

                    break
                h_b2[i], h_b2[j] = 0.0, 1.0
                cn = _eval(h_b2)
                if cn < c - 1e-14:
                    c = cn; imp = True
                    if cn < best_c:
                        best_c = cn; best_h = h_b2.copy()
                else:
                    h_b2[i], h_b2[j] = 1.0, 0.0

                break
        h_b = h_b2
        if imp:

        else:
            stall += 1
            if stall >= 3:

                for _ in range(kf):
                    a, b = rng.randint(n), rng.randint(n)
                    if h_b[a] != h_b[b]:
                        h_b[a], h_b[b] = h_b[b], h_b[a]
                c = _eval(h_b)
                if c < best_c:
                    best_c = c; best_h = h_b.copy(); stall = 0
                elif stall >= 10:

    return best_h, best_c

def _balanced_subgrad(h, n, tlim, K=5, lr0=0.02):
    t0 = time.time()
    c = _eval(h)
    best_h, best_c = h.copy(), c
    lr = lr0
    while time.time() - t0 < tlim:
        cc = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_k = np.argsort(cc)[-K:]
        g = np.zeros(n)

            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:n])
                if sh > 0:
                    g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n - sp])
                g[:n - sp] -= h[sp:]
        g *= (2.0 / n) / K

                took = True
                break
        if not took:
            lr *= 0.5
            if lr < 1e-9:
                break
    return best_h, best_c

def _pairwise(h, n, tlim, rng):
    t0 = time.time()
    c = _eval(h)
    best_h, best_c = h.copy(), c
    steps_list = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    while time.time() - t0 < tlim:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for s in steps_list:
            ni, nj = h[i] + s, h[j] - s
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:

            h[i], h[j] = ni, nj
            cn = _eval(h)

                c = cn
                if cn < best_c:
                    best_c = cn; best_h = h.copy()
                break
            h[i], h[j] = ni - s, nj + s
    return best_h, best_c
