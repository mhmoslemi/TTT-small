        # sum_k (w_k / alpha) * dc_k/dx

        # So grad = sum_k (w_k / alpha) * dc_k / dx

        # = (1/S) * sum_k exp(alpha*(c_k-m)) * dc_k/dx
        # = sum_k w_k * dc_k/dx
        # So the gradient is sum_k w_k * dc_k/dx, which is what I have above.
        return val, grad
    
    try:

import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

    if time.time() - t0 > 0:
        iters = min(maxiter, int(maxiter * (t_end - time.time()) / 60))
    try:
        r = minimize(obj, h, method='L-BFGS-B', bounds=bds,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': iters * 3})
        hc = _repair(r.x, tgt)
        return hc, _c5_val(hc, dx)
    except Exception:
        return h, _c5_val(h, dx)

def _slsqp_grad(h, dx, alpha, t_end, maxiter=800):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_val_grad(x, n, alpha, 12)[0]
    def jac(x):
        return _smooth_val_grad(x, n, alpha, 12)[1]
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'disp': False})
        hc = _repair(r.x, tgt)
        return hc, _c5_val(hc, dx)
    except Exception:
        return h, _c5_val(h, dx)

def _targeted_search(h, dx, rng, t_end):
    n = len(h)
    best = _c5_val(h, dx)
    best_h = h.copy()
    h = h.copy()
    while time.time() < t_end:
        c = _corr_full(h)
        top = np.argsort(c)[::-1][:6]
        improved = False
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            ml = n - s
            for _ in range(8):
                a = rng.randint(ml)
                if int(ki - (n - 1)) > 0:
                    i1, i2 = a + s, a
                else:
                    i1, i2 = a, a + s
                d = rng.uniform(0.003, 0.025)
                for sign in (1, -1):
                    h[i1] += sign * d
                    h[i2] -= sign * d
                    if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                        cn = _c5_val(h, dx)
                        if cn < best:
                            best = cn
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i1] -= sign * d
                            h[i2] += sign * d
                    else:
                        h[i1] -= sign * d
                        h[i2] += sign * d
        for _ in range(6):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.015, 0.015)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5_val(h, dx)
                if cn < best:
                    best = cn
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
        if not improved:
            h = best_h.copy()
            d = rng.uniform(0.001, 0.008)
            i, j = rng.randint(n), rng.randint(n)
            h[i] += d; h[j] -= d
            h = _repair(np.clip(h, 0, 1), n / 2.0)
    return best_h, best

def _fine_search(h, dx, rng, t_end):
    n = len(h)
    best = _c5_val(h, dx)
    best_h = h.copy()
    h = h.copy()
    scales = [0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001, 5e-05]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5_val(h, dx)
                if cn < best:
                    best = cn
                    best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return best_h, best

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode % 4 == 0:
        hp += 0.05 * (2.0 * rng.random(n) - 1.0)
    elif mode % 4 == 1:
        idx = rng.choice(n, max(5, n // 4), replace=False)
        hp[idx] += 0.1 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode % 4 == 2:
        x = np.linspace(0, 2, n)
        hp += 0.04 * np.sin(2 * np.pi * rng.uniform(1.5, 6.0) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(3, n // 10), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0 / 2.0)
    best_h, best_c5, best_n = h0.copy(), _c5_val(h0, dx0), n0

    # Phase 1: Upsample to 2x and optimize with L-BFGS-B
    n2 = 2 * n0
    dx2 = 2.0 / n2
    h2 = _repair(_upsample(h0, n2), n2 / 2.0)
    for a, lam in [(500, 100), (2000, 500), (10000, 2000)]:
        if time.time() > t0 + 120:
            break
        h2, c2 = _lbfgs_pen(h2, dx2, a, lam, t0 + 120, 3000)
        if c2 < best_c5 * (n0 / n2) * (n2 / n0):
            pass
    # Check if 2x resolution gives better value (comparing C5 directly)
    c2_check = _c5_val(h2, dx2)
    if c2_check < best_c5:
        best_h, best_c5, best_n = h2.copy(), c2_check, n2

    # Phase 2: SLSQP with gradient at best resolution
    nb, dxb = best_n, 2.0 / best_n
    for a in [5000, 20000, 80000]:
        if time.time() > t0 + 300:
            break
        hc, c = _slsqp_grad(best_h, dxb, a, t0 + 300, 600)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 3: Targeted local search
    if time.time() < t0 + 550:
        best_h, best_c5 = _targeted_search(best_h, dxb, rng, min(t0 + 550, t0 + budget_s - 350))

    # Phase 4: More SLSQP at very high alpha
    for a in [200000, 500000, 1000000]:
        if time.time() > t0 + 700:
            break
        hc, c = _slsqp_grad(best_h, dxb, a, t0 + 700, 500)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 5: Multi-start perturbation + refine
    t_end5 = min(t0 + 850, t0 + budget_s - 200)
    for r in range(12):
        if time.time() > t_end5:
            break
        hp = _perturb(best_h, rng, r)
        hp, c = _lbfgs_pen(hp, dxb, 50000, 5000, min(t_end5 - 30, time.time() + 25), 2000)
        hp, c = _slsqp_grad(hp, dxb, 200000, min(t_end5 - 10, time.time() + 20), 300)
        if c < best_c5:
            best_h, best_c5 = hp, c
            best_h, best_c5 = _targeted_search(best_h, dxb, rng, min(t_end5 - 5, time.time() + 10))

    # Phase 6: Fine local search
    if time.time() < t0 + budget_s - 60:
        best_h, best_c5 = _fine_search(best_h, dxb, rng, min(t0 + budget_s - 30, time.time() + 180))

    # Phase 7: Final SLSQP polish
    if budget_s - (time.time() - t0) > 25:
        hc, c = _slsqp_grad(best_h, dxb, 1000000, t0 + budget_s - 10, 300)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 8: Ultra-fine search
    if budget_s - (time.time() - t0) > 8:
        best_h, best_c5 = _fine_search(best_h, dxb, rng, t0 + budget_s - 2)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5_val(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
