import numpy as np
import time

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_corrs(h).max())

def _smooth_obj_and_grad(h, alpha):
    n = len(h)
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m) - (alpha * c - alpha * m).max(), -500, 500)
    # Actually: s = alpha*(c-m), then shift by max for numerical stability
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w_sum = w.sum()
    w /= w_sum
    obj = m + np.log(w_sum) / alpha
    g = np.zeros(n)
    active = w > 1e-10
    for k in np.where(active)[0]:
        s = k - (n - 1)
        wk = w[k]
        if s >= 0:
            if s == 0:
                g += wk * (1.0 - 2.0 * h)
            else:
                g[:n-s] += wk * (1.0 - h[s:])
                g[s:] -= wk * h[:n-s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return obj, g

def _nesterov(h, alpha, lr0, n_iter, momentum, t0, budget):
    n = len(h)
    target = n * 0.5
    h_prev = h.copy()
    h_curr = h.copy()
    best = h.copy()
    best_f = _eval(best)
    lr = lr0
    for t in range(n_iter):
        if time.time() - t0 > budget - 30:
            break
        # Nesterov extrapolation
        beta = momentum
        y = h_curr + beta * (h_curr - h_prev)
        
        # Gradient at y (don't need to project y for gradient, just use it)
        _, g = _smooth_obj_and_grad(np.clip(y, 0, 1), alpha)
        
        # Take step
        h_new = _proj(y - lr * g, target)
        
        # Evaluate
        f_new = _eval(h_new)
        if f_new < best_f:
            best_f = f_new
            best = h_new.copy()
        
        # Adaptive step size
        f_curr = _eval(h_curr)
        if f_new > f_curr + 1e-10:
            lr *= 0.7
        elif f_new < f_curr - 1e-6:
            lr = min(lr * 1.05, lr0 * 2)
        
        h_prev = h_curr
        h_curr = h_new
    return best, best_f

import numpy as np
import time

_t0 = 0.0
_budget = 1100.0

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_corrs(h).max())

def _grad_topm(h, m):
    """Gradient of max using top-m uniform subgradient"""
    n = len(h)
    c = _corrs(h)
    top = np.argsort(c)[::-1][:m]
    g = np.zeros(n)
    for k in top:
        s = k - (n - 1)
        if s >= 0:
            if s == 0:
                g += (1.0 - 2.0 * h)
            else:
                g[:n-s] += (1.0 - h[s:])
                g[s:] -= h[:n-s]
        else:
            sp = -s
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    g /= m
    return g

def _smooth_og(h, alpha):
    """Smooth objective and gradient via log-sum-exp"""
    n = len(h)
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    ws = w.sum()
    w /= ws
    obj = m + np.log(ws) / alpha
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        s = k - (n - 1)
        wk = w[k]
        if s >= 0:
            if s == 0:
                g += wk * (1.0 - 2.0 * h)
            else:
                g[:n-s] += wk * (1.0 - h[s:])
                g[s:] -= wk * h[:n-s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return obj, g

def _nesterov(h, alpha, lr0, n_iter, mom, t0, budget):
    n = len(h)
    tgt = n * 0.5
    h_prev = h.copy()
    h_cur = h.copy()
    best = h.copy()
    best_f = _eval(best)
    lr = lr0
    stall = 0
    for t in range(n_iter):
        if time.time() - t0 > budget - 25:
            break
        y = h_cur + mom * (h_cur - h_prev)
        yclip = np.clip(y, 0.0, 1.0)
        _, g = _smooth_og(yclip, alpha)
        h_new = _proj(y - lr * g, tgt)
        f_new = _eval(h_new)
        if f_new < best_f - 1e-12:
            best_f = f_new
            best = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                lr *= 0.8
                stall = 0
        f_cur = _eval(h_cur)
        if f_new > f_cur + 0.5 * lr * np.dot(g, h_cur - h_new):
            lr *= 0.6
        else:
            lr = min(lr * 1.02, lr0 * 3.0)
        h_prev = h_cur
        h_cur = h_new
    return best, best_f

def _nesterov_topm(h, m, lr0, n_iter, mom, t0, budget):
    n = len(h)
    tgt = n * 0.5
    h_prev = h.copy()
    h_cur = h.copy()
    best = h.copy()
    best_f = _eval(best)
    lr = lr0
    stall = 0
    for t in range(n_iter):
        if time.time() - t0 > budget - 25:
            break
        y = h_cur + mom * (h_cur - h_prev)
        g = _grad_topm(np.clip(y, 0.0, 1.0), m)
        h_new = _proj(y - lr * g, tgt)
        f_new = _eval(h_new)
        if f_new < best_f - 1e-12:
            best_f = f_new
            best = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr *= 0.8; stall = 0
        f_cur = _eval(h_cur)
        if f_new > f_cur + 0.5 * lr * np.dot(g, h_cur - h_new):
            lr *= 0.6
        else:
            lr = min(lr * 1.03, lr0 * 3.0)
        h_prev = h_cur
        h_cur = h_new
    return best, best_f

def _block_swap(h, n_trials, t0, budget):
    n = len(h)
    tgt = n * 0.5
    best = h.copy()
    best_f = _eval(best)
    rng = np.random.RandomState(12345)
    for _ in range(n_trials):
        if time.time() - t0 > budget - 20:
            break
        blen = rng.randint(4, max(5, n // 30))
        i = rng.randint(n - blen)
        j = rng.randint(n - blen)
        if abs(i - j) < blen:
            continue
        tmp = best[i:i+blen].copy()
        best[i:i+blen] = best[j:j+blen]
        best[j:j+blen] = tmp
        best = np.clip(best, 0.0, 1.0)
        f = _eval(best)
        if f > best_f + 1e-6:
            best[j:j+blen] = tmp
            best[i:i+blen] = best[j:j+blen].copy()
            best[j:j+blen] = h[i:i+blen].copy() if False else best[i:i+blen]
            # revert
            best = _proj(best, tgt)
        else:
            if f < best_f:
                best_f = f
    return _proj(best, tgt), best_f

def _sharpen(h, gamma, t0, budget):
    """Push toward binary, then re-project"""
    n = len(h)
    tgt = n * 0.5
    h_s = np.tanh(gamma * (h - 0.5))
    h_s = (h_s + 1.0) * 0.5
    h_p = _proj(h_s, tgt)
    f_p = _eval(h_p)
    f_o = _eval(h)
    return h_p if f_p < f_o else h

def _pairwise_structured(h, n_iter, scales, t0, budget, seed):
    n = len(h)
    tgt = n * 0.5
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_f = _eval(best)
    for _ in range(n_iter):
        if time.time() - t0 > budget - 15:
            break
        scale = rng.choice(scales)
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = best[i] + d, best[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            best[i], best[j] = ni, nj
            fn = _eval(best)
            if fn < best_f - 1e-14:
                best_f = fn
            else:
                best[i], best[j] = ni - d, nj + d
    return best, best_f

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    diff = tgt - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h
