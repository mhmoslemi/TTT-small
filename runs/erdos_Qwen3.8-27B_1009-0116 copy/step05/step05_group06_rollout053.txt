        if abs(d) < 1e-15:
            continue
        h[i] = np.clip(h[i] + d, 0.0, 1.0)
        h[j] = np.clip(h[j] - d, 0.0, 1.0)
        r += (G[i] - G[j]) * d
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: Quick SLSQP
    for alpha in [5000, 30000]:
        if _el() > 80:
            break
        h_new = _slsqp(best_h, alpha, 1500, 1e-18, n0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Global pair CD
    if _el() < budget_s - 200:
        h_new = _pair_cd(best_h, 100000, 40, rng)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Local pair CD (neighborhood)
    if _el() < budget_s - 200:
        h_new = _pair_cd_local(best_h, 80000, 30, rng, window=6)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 4: SLSQP polish
    if _el() < budget_s - 250:
        for alpha in [80000, 300000]:
            if _el() > budget_s - 250:
                break
            h_new = _slsqp(best_h, alpha, 2000, 1e-19, best_n)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), len(h_new)

    # Phase 5: Active-lag focused search
    if _el() < budget_s - 300:
        h_new = _active_lag_search(best_h, 80000, 30, rng)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 6: More global pair CD
    if _el() < budget_s - 300:
        h_new = _pair_cd(best_h, 80000, 35, rng)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 7: Multi-start
    nc = best_n
    if _el() < budget_s - 400:
        for _r in range(20):
            if _el() > budget_s - 400:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = _repair(best_h + 0.04 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 2:
                hp = _repair(np.roll(best_h, rng.randint(1, nc // 3 + 1)), nc)
            elif strat == 3:
                t_arr = np.arange(nc)
                hp = best_h.copy()
                for _ in range(3):
                    hp += 0.015 * rng.randn() * np.sin(2 * np.pi * rng.randint(2, nc // 3 + 1) * t_arr / nc)
                hp = _repair(hp, nc)
            else:
                hp = best_h.copy()
                mask = rng.rand(nc) < 0.2
                hp[mask] = 0.5 + 0.4 * rng.randn(mask.sum())
                hp = _repair(hp, nc)
            h_new = _pair_cd(hp, 25000, 40, rng)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), nc
                if _el() < budget_s - 100:
                    h_pol = _slsqp(best_h, 100000, 1000, 1e-19, nc)
                    c5p = _eval(h_pol)
                    if c5p < best_c5:
                        best_c5, best_h = c5p, h_pol.copy()

    # Phase 8: Final fine pair CD + local
    if _el() < budget_s - 60:
        h_new = _pair_cd(best_h, 50000, 25, rng)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
    if _el() < budget_s - 30:
        h_new = _pair_cd_local(best_h, 40000, 20, rng, window=4)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 9: Final SLSQP
    if _el() < budget_s - 8:
        h_new = _slsqp(best_h, 500000, 1500, 1e-19, best_n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
