    return np.correlate(h, 1.0 - h, mode='full')

    c = _corr(x)

    return m + float(np.log(np.sum(np.exp(s)))) / _slsqp_alpha

    c = _corr(x)
    m = c.max()
    s = np.exp(25.0 * (c - m))
    w = s / s.sum()
    mv = float((w * c).sum())
    vv = float((w * (c - mv) ** 2).sum())
    return mv + _slsqp_lam * np.sqrt(vv + 1e-15)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()

        for i in idx:

                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a

            g[s:n] += 1.0 - h[:m]
            g[:m] -= h[s:n]

        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # Phase 8: Final targeted on best (100s)
    nb, dxb = best_n, 2.0 / best_n
    t8 = min(t0 + 1010, time.time() + 80)
    best_h, best_c5 = _targeted(best_h, dxb, rng, t8, top_k=8, steps=12)

    # Phase 9: Final SLSQP (50s)
    if rem() > 50:
        for a in [50000, 200000]:
            if rem() < 50:
                break
            h, c = _slsqp_run(best_h, a, 200)
            if c < best_c5:
                best_h, best_c5 = h, c

    # Phase 10: Fine search
    if rem() > 15:
        best_h, best_c5 = _fine(best_h, dxb, rng, t0 + budget_s - 3)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
