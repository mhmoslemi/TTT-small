import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        rem = diff
        for idx in order:
            add = min(rem, 1.0 - h[idx])
            h[idx] += add
            rem -= add
            if rem < 1e-12:
                break
    else:
        order = np.argsort(h)
        rem = -diff
        for idx in order:
            r = min(rem, h[idx])
            h[idx] -= r
            rem -= r
            if rem < 1e-12:
                break
    return h

def _grad_shift(h, d):
    n = len(h)
    g = np.zeros(n)
    if 0 <= d < n:
        g[:n-d] += (1.0 - h[d:])
        g[d:] -= h[:n-d]
    elif -n < d < 0:
        dp = -d
        g[dp:] += (1.0 - h[:n-dp])
        g[:n-dp] -= h[dp:]
    return g

def _active_descent(h, max_iter, top_k=1, lr_init=0.004):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    lr = lr_init
    stalled = 0
    for it in range(max_iter):
        ov = np.correlate(h, 1.0 - h, mode='full')
        mx = ov.max()
        active = np.argsort(ov)[-top_k:]
        g = np.zeros(n)
        for k in active:
            g += _grad_shift(h, k - (n - 1))
        g /= len(active)
        fn = np.sum(g * g)
        if fn < 1e-20:
            break
        found = False
        for _ in range(30):
            h_new = _project(h - lr * g, n)
            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new
                found = True
                lr = min(lr * 1.08, 0.04)
                stalled = 0
                break
            lr *= 0.45
        if not found:
            stalled += 1
            lr = lr_init * 0.05
            if stalled > 30:
                break
    return h, c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _coord_descent(h, max_iters, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5 = _eval(h)
    for it in range(max_iters):

def _slsqp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _slsqp_obj_layers(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _slsqp_polish(h0, alpha, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    def _se(x):
        return x.sum() - n * 0.5
    def _sj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_slsqp_obj_layers, h0, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

            best_h = h_cur.copy()

    # Phase 3: Try different n values
    for dn in [0, -10, 10, -20, 20, -5, 5, 14, -14]:
        if time.time() - t0 > budget_s * 0.55:
            break
        n_try = best_n + dn
        if n_try < 100 or n_try % 2 != 0:
            continue
        h_try = _repair(_resample(best_h, n_try), n_try)
        h_try, c5 = _active_descent(h_try, max_iter=4000, top_k=2, lr_init=0.003)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()
            best_n = n_try

    # Phase 4: SLSQP polish
    if time.time() - t0 < budget_s * 0.75:
        h_pol = _slsqp_polish(best_h, 50000, 2000)
        c5 = _eval(h_pol)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pol.copy()
        h_pol = _slsqp_polish(best_h, 200000, 2000)
        c5 = _eval(h_pol)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pol.copy()

    # Phase 5: Coordinate descent
    if time.time() - t0 < budget_s * 0.92:
        remaining = (budget_s * 0.92) - (time.time() - t0)
        max_cd = int(max(5000, remaining * 5000))
        h_cd = best_h.copy()
        h_cd, c5 = _coord_descent(h_cd, max_cd, seed=seed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd.copy()

    # Phase 6: Final active descent
    if time.time() - t0 < budget_s - 3:
        h_fin = best_h.copy()
        h_fin, c5 = _active_descent(h_fin, max_iter=10000, top_k=1, lr_init=0.002)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fin.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
