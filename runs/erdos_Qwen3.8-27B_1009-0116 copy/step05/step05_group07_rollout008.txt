import numpy as np
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(90):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _grad_lag(shift, h):
    n = len(h)
    if shift == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if shift > 0:
        g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(90):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _adam(h, topm, n_iter, lr0, b1, b2, eps, topm_schedule=None):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    mb = np.zeros(n)
    vb = np.zeros(n)
    f = _eval(h)
    t = 0
    for i in range(n_iter):
        t += 1
        if time.time() - _t0 > _budget - 25:
            break
        tm = topm_schedule(i // max(1, n_iter // 10)) if topm_schedule else topm
        c = _corrs(h)
        top_idx = np.argsort(c)[::-1][:tm]
        g = np.zeros(n)
        for ki in top_idx:
            g += _grad_lag(ki - (n - 1), h)
        g *= 2.0 / (n * tm)
        mb = b1 * mb + (1.0 - b1) * g
        vb = b2 * vb + (1.0 - b2) * g * g
        mh = mb / (1.0 - b1 ** t)
        vh = vb / (1.0 - b2 ** t)
        lr = lr0 / (1.0 + 0.0001 * t)
        step = lr * mh / (np.sqrt(vh) + eps)
        h_new = _proj_sum(h - step, target)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
        else:
            for _b in range(3):
                step *= 0.4
                h_new = _proj_sum(h - step, target)
                f_new = _eval(h_new)
                if f_new < f:

                    break
    return h, f

def _active_descent(h, topm, max_iter, step0):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    for _ in range(max_iter):

        c = _corrs(h)

                found = True
                break
            step *= 0.5
        if not found:

        step = min(step * 1.1, step0)
    return h, f

def _round_robin(h, n_lags, steps_per_lag):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    for _ in range(n_lags):
        if time.time() - _t0 > _budget - 15:
            break
        c = _corrs(h)
        k_star = c.argmax()

            g = _grad_lag(k_star - (n - 1), h) * (2.0 / n)

                f_new = _eval(h_new)
                if f_new < f - 1e-14:
                    h, f = h_new, f_new
                    found = True
                    break
            if not found:
                break
    return h, f

def _pairwise(h, n_iter, scale, seed_val):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 8:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0.0, 1.0), n * factor)
