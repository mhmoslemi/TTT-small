import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _top_lags(h, k=8):
    c = _corr(h)
    n = len(h)
    # Get top k lags (indices in the full correlation)
    top = np.argsort(c)[::-1][:k]
    return top, c

def _lag_gradient(h, lag_idx, n):
    """Gradient of the correlation at a specific lag index."""
    s = abs(int(lag_idx - (n - 1)))
    g = np.zeros(n)
    if s == 0:
        g[:] = 2*h - 1
        return g
    m = n - s
    if lag_idx > (n - 1):  # positive shift: c = sum h[i] * (1-h[i-s]) for i=s..n-1
        # Actually need to be more careful
        g[s:n] += 1.0 - h[:m]
        g[:m] -= h[s:n]
    else:  # negative shift
        g[:m] += 1.0 - h[s:n]
        g[s:n] -= h[:m]
    return g

def _smooth_max_obj(h, alpha, top_k=8):
    """Smooth max using log-sum-exp on top k lags."""
    c = _corr(h)
    n = len(h)
    top = np.argsort(c)[::-1][:top_k]
    # Use only top k for efficiency
    vals = c[top]
    m = vals.max()
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha, c, top

def _projected_sgd(h, dx, t_end, lr_init=0.01, momentum=0.9, top_k=10, alpha_scale=50):
    """Projected SGD with momentum on smooth max."""
    n = len(h)
    tgt = n / 2.0
    best_h, best_c = h.copy(), _c5(h, dx)
    v = np.zeros(n)
    lr = lr_init
    it = 0
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:top_k]
        vals = c[top]
        m = vals.max()
        # Weighted gradient
        w = np.exp(alpha_scale * (vals - m))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top):
            grad += wi * _lag_gradient(h, ki, n)
        grad -= grad.mean()  # maintain sum constraint approximately
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        v = momentum * v - lr * grad / gn
        h_new = _repair(np.clip(h + v, 0, 1), tgt)
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
            h = h_new
            lr = min(lr * 1.005, 0.05)
        else:
            h = h_new  # Still accept for exploration
            lr *= 0.95
            if lr < 1e-7:
                break
        it += 1
        if it > 200000:
            break
    return best_h, best_c

def _simulated_annealing(h, dx, rng, t_end, t_start=1.0, t_end_sa=1e-4, top_k=4):
    """Simulated annealing with targeted moves on worst lags."""
    n = len(h)
    tgt = n / 2.0

    t = t_start

        move = rng.randint(3)

            ki = top[rng.randint(len(top))]

            a = rng.randint(m)

                    if cur_c < best_c:
                        best_h, best_c = cur_h.copy(), cur_c
                else:
                    cur_h[i1] += d; cur_h[i2] -= d
            else:
                cur_h[i1] += d; cur_h[i2] -= d
        elif move == 1:

            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = rng.uniform(-0.02, 0.02)
            cur_h[i] += d; cur_h[j] -= d
            if 0 <= cur_h[i] <= 1 and 0 <= cur_h[j] <= 1:

                else:

            else:
                cur_h[i] -= d; cur_h[j] += d
        else:
            # Block update
            w = rng.randint(3, 10)
            start = rng.randint(0, max(1, n - w))
            block = cur_h[start:start+w].copy()
            block += 0.02 * (2*rng.random(w) - 1)
            block = np.clip(block, 0, 1)

            cur_h = _repair(cur_h, tgt)

                if cur_c < best_c:
                    best_h, best_c = cur_h.copy(), cur_c
            else:
                cur_h[start:start+w] = block = np.clip(block, 0, 1)

        t *= cooling
    return best_h, best_c

def _slsqp_smooth(h, dx, alpha, maxiter=500):
    """SLSQP with smooth max objective."""
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        top = np.argsort(c)[::-1][:10]
        vals = c[top]
        m = vals.max()
        return m + float(np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha)
    def grad_obj(x):
        c = _corr(x)
        top = np.argsort(c)[::-1][:10]
        vals = c[top]
        m = vals.max()
        w = np.exp(alpha * (vals - m))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top):
            grad += wi * _lag_gradient(x, ki, n)
        grad -= grad.mean()
        return grad
    def eq(x):
        return x.sum() - tgt
    def grad_eq(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, jac=grad_obj, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq, 'jac': grad_eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _coordinate_descent(h, dx, rng, t_end, iterations_per_pair=5):
    """Coordinate descent on pairs with binary-like updates."""
    n = len(h)
    tgt = n / 2.0
    best_h, best_c = h.copy(), _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        # Identify worst lags
        c = _corr(h)
        top = np.argsort(c)[::-1][:6]
        improved = False
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0: continue
            m = n - s
            # For each pair (i, i+s), try to reduce the contribution
            # h[i+s]*(1-h[i]) + h[i]*(1-h[i+s]) is the "mismatch"
            for _ in range(itertions_per_pair if False else 3):
                a = rng.randint(m)
                i, j = a, a + s
                # Try moving d from j to i or i to j
                for d in [0.01, 0.005, -0.005, -0.01]:
                    h[i] += d; h[j] -= d
                    if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                        new_c = _c5(h, dx)
                        if new_c < best_c:
                            best_c = new_c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    else:
                        h[i] -= d; h[j] += d
        if not improved:
            # Random perturbation to escape
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                d = rng.uniform(-0.015, 0.015)
                h[i] += d; h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    h = _repair(h, tgt)
                else:
                    h[i] -= d; h[j] += d
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _lag_grad(h, ki, n):
    s = abs(int(ki - (n - 1)))
    g = np.zeros(n)
    if s == 0:
        g[:] = 2.0 * h - 1.0
        return g
    m = n - s
    if ki > (n - 1):
        g[s:n] += 1.0 - h[:m]
        g[:m] -= h[s:n]
    else:
        g[:m] += 1.0 - h[s:n]
        g[s:n] -= h[:m]
    return g

def _smooth_obj(x, alpha, top_k):
    c = _corr(x)
    top = np.argsort(c)[::-1][:top_k]
    vals = c[top]
    m = vals.max()
    return m + float(np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha)

def _smooth_grad(x, alpha, top_k, n):
    c = _corr(x)
    top = np.argsort(c)[::-1][:top_k]
    vals = c[top]
    m = vals.max()
    w = np.exp(alpha * (vals - m))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top):
        grad += wi * _lag_grad(x, ki, n)
    grad -= grad.mean()
    return grad

def _sgd_mom(h, dx, t_end, lr_init, momentum, top_k, alpha_scale):
    n = len(h)
    tgt = n / 2.0
    best_h, best_c = h.copy(), _c5(h, dx)
    v = np.zeros(n)
    lr = lr_init
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:top_k]
        vals = c[top]
        m = vals.max()
        w = np.exp(alpha_scale * (vals - m))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top):
            grad += wi * _lag_grad(h, ki, n)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        v = momentum * v - lr * grad / gn
        h_new = _repair(np.clip(h + v, 0, 1), tgt)
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
            h = h_new
            lr = min(lr * 1.003, 0.06)
        else:
            h = h_new
            lr *= 0.92
            if lr < 1e-8:

    return best_h, best_c

def _slsqp(h, dx, alpha, top_k, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_obj(x, alpha, top_k)
    def jac(x):
        return _smooth_grad(x, alpha, top_k, n)
    def eq(x):
        return x.sum() - tgt
    def eqj(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq, 'jac': eqj}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
