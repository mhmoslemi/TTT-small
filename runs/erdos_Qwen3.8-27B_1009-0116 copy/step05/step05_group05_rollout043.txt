import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _shift_grad(h, sh):
    """Gradient of shift `sh` (non-negative) of correlate(h, 1-h)."""
    n = len(h)
    g = np.zeros(n)
    if sh == 0:
        # c_0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        # d/dh[j] = 1 - 2*h[j]
        g[:] = 1.0 - 2.0 * h
    elif sh > 0:
        # c = sum_{i=0}^{n-1-sh} h[i]*(1-h[i+sh])
        # d/dh[j] = (1-h[j+sh]) if j <= n-1-sh, minus h[j-sh] if j >= sh
        idx1 = np.arange(n - sh)
        g[idx1] = 1.0 - h[idx1 + sh]
        idx2 = np.arange(sh, n)
        g[idx2] -= h[idx2 - sh]
    else:
        sp = -sh
        idx1 = np.arange(sp, n)
        g[idx1] = 1.0 - h[idx1 - sp]
        idx2 = np.arange(0, n - sp)
        g[idx2] -= h[idx2 + sp]
    return g

def _top_shifts_grad(h, m, temp):
    """Weighted gradient from top shifts with softmax weighting."""
    n = len(h)
    c = _corr(h)
    cmax = c.max()
    s = np.clip(-temp * (cmax - c), -500, 500)  # softmax weights on shifts near max
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        g += w[k] * _shift_grad(h, sh)
    return g, cmax

def _smooth_obj(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        g += w[k] * _shift_grad(h, sh)
    return g * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj_sum(h, target):
    """Project h onto {x: sum(x)=target, 0<=x<=1} via bisection on offset."""
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _active_descent(h, max_iters, m, temp):
    """Active set descent focusing on top-m shifts with softmax weighting."""
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for it in range(max_iters):
        c = _corr(h)
        cmax = c.max()
        # Softmax weights on all shifts, temperature controls focus
        s = np.clip(-temp * (cmax - c), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        # Compute weighted gradient
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            sh = (n - 1) - k
            g += w[k] * _shift_grad(h, sh)
        g *= dx
        # Project gradient to satisfy sum constraint
        g_mean = g.mean()
        g_dir = g - g_mean  # direction preserves sum
        # Check if direction actually decreases objective
        if np.linalg.norm(g_dir) < 1e-14:
            break
        # Line search with backtracking
        best_t = 0.0
        best_c5 = c5c
        for t in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = _proj_sum(h - t * g_dir, n * 0.5)
            c5_try = _eval(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_t = t
        if best_t == 0.0:
            break
        h = h - best_t * g_dir
        h = _proj_sum(h, n * 0.5)
        c5c = best_c5
    return h, c5c

def _pairwise(h, n_iter, scale, rng):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _structured_local(h, max_rounds, rng, m=5):
    """Optimize pairs of coordinates by minimizing the max over top shifts."""
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _r in range(max_rounds):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        c0 = _corr(h)
        # For shifting mass from j to i by amount d:
        # c_k changes by d * (delta_i(k) - delta_j(k))
        a = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            sh = (n - 1) - k
            ci = 0.0
            cj = 0.0
            if sh >= 0:
                if sh == 0:
                    ci = 1 - 2*h[i]
                    cj = 1 - 2*h[j]
                else:
                    if i <= n - 1 - sh:
                        ci += (1.0 - h[i + sh])
                    if sh <= i < n:
                        ci -= h[i - sh]
                    if j <= n - 1 - sh:
                        cj += (1.0 - h[j + sh])
                    if sh <= j < n:
                        cj -= h[j - sh]
            else:
                sp = -sh
                if sp == 0:
                    ci = 1 - 2*h[i]
                    cj = 1 - 2*h[j]
                else:
                    if sp <= i < n:
                        ci += (1.0 - h[i - sp])
                    if 0 <= i + sp < n - sp:
                        ci -= h[i + sp]
                    if sp <= j < n:
                        cj += (1.0 - h[j - sp])
                    if 0 <= j + sp < n - sp:
                        cj -= h[j + sp]
            a[k] = ci - cj
        # c_new[k] = c0[k] + d*dx*a[k], find d minimizing max
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi - 1e-12:
            continue
        # Grid search for best d
        best_d = 0.0
        best_v = np.inf
        dgrid = np.linspace(dlo, dhi, 15)
        for d in dgrid:
            v = np.max(c0 + dx * d * a)
            if v < best_v:
                best_v = v
                best_d = d
        if abs(best_d) > 1e-10:
            h[i] += best_d
            h[j] -= best_d
            h = np.clip(h, 0, 1)
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i] -= best_d
                h[j] += best_d
    return h, c5c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _shift_grad(h, sh):
    n = len(h)
    g = np.zeros(n)
    if sh == 0:
        g[:] = 1.0 - 2.0 * h
    elif sh > 0:
        g[:n - sh] = 1.0 - h[sh:n]
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] = 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g

def _weighted_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    w_rev = w[::-1]
    for t in range(2 * n - 1):
        wt = w_rev[t]
        if wt < 1e-12:
            continue
        sh = t - (n - 1)
        if sh >= 0:
            idx = np.arange(n - sh)
            g[idx] += wt * (1.0 - h[idx + sh])
            g[sh:] -= wt * h[:n - sh]
        else:
            sp = -sh
            idx = np.arange(sp, n)
            g[idx] += wt * (1.0 - h[idx - sp])
            g[:n - sp] -= h[sp:]

def _smooth_obj(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),

def _active_descent(h, max_iters, temp0, temp1):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for it in range(max_iters):
        frac = it / max(1, max_iters)
        temp = temp0 + (temp1 - temp0) * frac
        c = _corr(h)
        cmax = c.max()
        s = np.clip(-temp * (cmax - c), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = _weighted_grad(h, w) * dx
        g_dir = g - g.mean()
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:

        tmax = min(0.08, 0.5 * np.abs(1.0 / (np.sign(-g_dir) * (h - 0.0) + 1e-15)).min(),

            c5_try = _eval(h_try)

                best_t = t * gn
        if best_t == 0.0:
            break
        h = _proj_sum(h - best_t * g_dir, n * 0.5)
        c5c = best_c5
    return h, c5c

def _pairwise(h, n_iter, scale, rng):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

            hp += 0.005 * rng.randn() * np.sin(2 * np.pi * f * t / n_cur + rng.rand() * 6.28)
        return _repair(hp, n_cur)
    else:
        hp = h.copy()
        perm = rng.permutation(n_cur)
        block = n_cur // 4
        diff = rng.randn(block) * 0.015
        diff -= diff.mean()
        hp[perm[:block]] += diff
        return _repair(np.clip(hp, 0, 1), n_cur)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: SLSQP at n0
    for alpha in [500, 3000, 15000, 80000, 400000]:
        if _el() > 100:
            break
        h_new = _slsqp(best_h.copy(), alpha, 1500, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase B: Active descent at n0
    if _el() < 140:
        h_new, c5 = _active_descent(best_h.copy(), 1500, 3.0, 8.0)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase C: Structured local at n0
    if _el() < 180:
        h_new, c5 = _structured_local(best_h.copy(), 2500, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase D: Upsample to 2x and optimize
    if _el() < budget_s - 480:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        best_h, best_n = h_hi.copy(), n1
        best_c5 = _eval(best_h)
        for alpha in [2000, 15000, 80000, 400000, 1500000]:
            if _el() > budget_s - 380:
                break
            h_hi = _slsqp(h_hi.copy(), alpha, 1200, 1e-20)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h = c5, h_hi.copy()
        if _el() < budget_s - 280:
            h_new, c5 = _active_descent(best_h.copy(), 1500, 3.0, 10.0)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()
        if _el() < budget_s - 180:
            h_new, c5 = _structured_local(best_h.copy(), 2000, rng)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase E: Multi-start at current best
    n_cur = best_n
    if _el() < budget_s - 220:
        for _r in range(12):
            if _el() > budget_s - 180:
                break
            hp = _multi_start(best_h, n_cur, rng, _r % 5)
            h_new = _slsqp(hp, 150000, 500, 1e-20)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), n_cur

    # Phase F: Intensive active descent at best
    if _el() < budget_s - 120:
        h_new, c5 = _active_descent(best_h.copy(), 3000, 5.0, 15.0)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase G: Structured local
    if _el() < budget_s - 70:
        h_new, c5 = _structured_local(best_h.copy(), 3000, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase H: Fine pairwise
    if _el() < budget_s - 30:
        for scale in [0.004, 0.0015, 0.0005, 0.00015]:
            if _el() > budget_s - 10:
                break
            h_new, c5 = _pairwise(best_h.copy(), 2500, scale, rng)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase I: Final SLSQP polish
    if _el() < budget_s - 5:
        h_new = _slsqp(best_h.copy(), 800000, 1500, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), len(h_new)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
