                g += (1.0 - 2.0 * h)

                g[:n - s] -= h[s:]

            p = -s
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]

    x_new = np.linspace(0, 2, n * factor)
    f_int = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f_int(x_new), 0, 1), n * factor)

def _pairwise_polish(h, n_iter, margin=15):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(12345)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - margin:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.015, 0.015)
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _active_descent(h, topm, max_iter, step0, margin=20):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - margin:
            break
        c = _corrs(h)
        top = np.argsort(c)[::-1][:topm]
        g = _grad_lags(h, top) / topm
        found = False
        for _bt in range(40):
            h_new = _proj_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-12:
                h, f, found = h_new, f_new, True
                break
            step *= 0.5
        if not found:
            break
        step = min(step * 1.06, step0)
    return h, f

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: Subgradient at base resolution
    for mlags in [3, 5, 10, 15, 30, 50]:
        if time.time() - _t0 > _budget - 400:
            break
        hn, fn = _sgd(best_h, mlags, 5000, 350)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase B: Smoothed projected gradient
    for alpha in [3000, 15000, 80000, 400000]:
        if time.time() - _t0 > _budget - 300:
            break
        hn, fn = _proj_grad(best_h, alpha, 3000, 250)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase C: SLSQP with softmax
    for alpha in [20000, 150000, 800000]:
        if time.time() - _t0 > _budget - 250:
            break
        hn, fn = _slsqp_smooth(best_h, alpha, 800, 200)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase D: Active descent refinement
    for topm in [3, 5, 10, 20, 40]:
        if time.time() - _t0 > _budget - 150:
            break
        hn, fn = _active_descent(best_h, topm, 4000, 0.004, 120)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase E: Upsample and optimize
    n1 = n0 * 2
    if time.time() - _t0 < _budget - 180:
        h1 = _upsample(best_h, 2)
        for mlags in [5, 10, 20, 40]:
            if time.time() - _t0 > _budget - 130:
                break
            h1, f1 = _sgd(h1, mlags, 3000, 100)
            if f1 < best_c5:
                best_c5 = f1
                best_h = h1.copy()
                best_n = n1
        for alpha in [50000, 300000]:
            if time.time() - _t0 > _budget - 100:
                break
            hn, fn = _slsqp_smooth(best_h, alpha, 1000, 80)
            if fn < best_c5:
                best_c5 = fn
                best_h = hn.copy()
        for topm in [5, 15, 30]:
            if time.time() - _t0 > _budget - 60:
                break
            hn, fn = _active_descent(best_h, topm, 3000, 0.003, 50)
            if fn < best_c5:
                best_c5 = fn
                best_h = hn.copy()

    # Phase F: Pairwise fine-tuning
    nc = len(best_h)
    for scale, nit in [(8000, 0.012), (10000, 0.005), (8000, 0.002)]:
        if time.time() - _t0 > _budget - 20:
            break
        best_h, best_c5 = _pairwise_polish(best_h, nit, 10)

    # Phase G: Multi-start exploration
    if time.time() - _t0 < _budget - 60:
        for _r in range(8):
            if time.time() - _t0 > _budget - 30:
                break
            hp = best_h + 0.004 * rng.randn(nc)
            hp = _repair(hp, nc)
            hn, fn = _sgd(hp, 10, 2000, 25)
            if fn < best_c5:
                best_c5 = fn
                best_h = hn.copy()
            if time.time() - _t0 > _budget - 25:
                break
            hn, fn = _active_descent(best_h, 10, 2000, 0.002, 15)
            if fn < best_c5:
                best_c5 = fn
                best_h = hn.copy()

    # Phase H: Final polish
    if time.time() - _t0 < _budget - 10:
        hn, fn = _active_descent(best_h, 8, 2000, 0.001, 5)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()
        best_h, best_c5 = _pairwise_polish(best_h, 5000, 3)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
