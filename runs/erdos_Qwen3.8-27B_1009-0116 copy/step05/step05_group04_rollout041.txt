import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k, n, dx):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        i = np.arange(n - sh)
        g[i] += (1.0 - h[i + sh])
        if sh > 0:
            j = np.arange(sh, n)
            g[j] -= h[j - sh]
    else:
        sp = -sh
        i = np.arange(sp, n)
        g[i] += (1.0 - h[i - sp])
        j = np.arange(n - sp)
        g[j] -= h[j + sp]
    return g * dx

def _hess_lag(k, n, dx):
    H = np.zeros((n, n))
    sh = (n - 1) - k
    s = abs(sh)
    if s == 0:
        H[np.arange(n), np.arange(n)] = -dx
    else:
        i = np.arange(n - s)
        H[i, i + s] = -dx
        H[i + s, i] = -dx
    return H

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return x.sum() - n * 0.5
    def _cj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha), h, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _newton_active(h, n, tol=1e-6, max_iter=50, el_func=None, budget=None):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _eval(h)
    for it in range(max_iter):
        if el_func and budget and el_func() > budget:
            break
        c = _corrs(h)
        cm = c.max()
        active = np.where(c > cm - tol)[0]
        wa = 1.0 / len(active)
        g = np.zeros(n)
        H = np.zeros((n, n))
        for k in active:
            g += wa * _grad_lag(h, k, n, dx)
            H += wa * _hess_lag(k, n, dx)
        # Add small regularization for positive semi-definiteness
        mu = 1e-6
        H_reg = H + mu * np.eye(n)
        # KKT: [H 1; 1' 0][d; lam] = [-g; 0]
        A = np.zeros((n + 1, n + 1))
        A[:n, :n] = H_reg
        A[:n, n] = 1.0
        A[n, :n] = 1.0
        b = np.concatenate([-g, [0.0]])
        try:
            sol = np.linalg.solve(A, b)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(A, b, rcond=None)[0]
        d = sol[:n]
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            h_new = h + alpha * d
            if h_new.min() < 0 or h_new.max() > 1:
                h_new = _proj_sum(h_new, n * 0.5)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
            if c_new < _eval(h) - 1e-15:
                h = h_new.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

            h = h_new
            c_cur = c_new
            if c_cur < best_c:

                best_h = h.copy()
    return best_h, best_c
