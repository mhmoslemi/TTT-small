import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_lag(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    ss = abs(s)
    m = n - ss
    if s > 0:
        g[:m] += (1.0 - h[ss:])
        g[ss:n] -= h[:m]
    else:
        g[ss:n] += (1.0 - h[:m])
        g[:m] -= h[ss:]
    return g

def _subgrad_top(h, k=25, temp=60.0):
    n = len(h)
    c = _corr(h)
    top_idx = np.argsort(c)[::-1][:k]
    wc = c[top_idx]
    w = np.exp(temp * (wc - wc.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        s = int(ki - (n - 1))
        grad += wi * _grad_lag(h, s)
    grad -= grad.mean()
    return grad

def _sgd(h0, dx, t_end, k=25, temp=60.0):
    n = len(h0)
    tgt = n / 2.0
    h = h0.copy()
    best = _c5(h, dx)
    lr = 0.012
    stall = 0
    while time.time() < t_end:
        g = _subgrad_top(h, k, temp)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = _repair(h - lr * g / gn, tgt)
        c_new = _c5(h_new, dx)
        if c_new < best - 1e-11:
            best = c_new; h = h_new
            lr = min(lr * 1.015, 0.06)
            stall = 0
        else:
            stall += 1
            lr *= 0.93
            if lr < 1e-7 or stall > 600:
                break
    return h, best

def _slsqp(h0, dx, alpha, maxiter=300):
    n = len(h0)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400.0, 400.0)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h0, _c5(h0, dx)

def _sla(h0, dx, rng, t_end, n_lags=8, n_try=12):
    n = len(h0)
    tgt = n / 2.0
    h = h0.copy()
    best = _c5(h, dx)
    scale = 0.018
    stall = 0
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:n_lags]
        improved = False
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            for _ in range(n_try):
                a = rng.randint(m)
                i1, i2 = a + s, a
                d = scale * rng.uniform(0.2, 1.0)
                h[i1] -= d; h[i2] += d
                if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                    cn = _c5(h, dx)
                    if cn < best - 1e-12:
                        best = cn; improved = True
                    else:
                        h[i1] += d; h[i2] -= d
                else:
                    h[i1] += d; h[i2] -= d
        for _ in range(n_try // 2 + 2):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * rng.uniform(-1.0, 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best - 1e-12:
                    best = cn; improved = True
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
        h = _repair(h, tgt)
        if improved:
            stall = 0
            scale = min(scale * 1.04, 0.05)
        else:
            stall += 1
            scale *= 0.91
            if scale < 5e-6 or stall > 80:
                break
    return h, best

def _sa_explore(h0, dx, rng, t_end, T0=0.012, Tend=0.0004, ac=0.9996):
    n = len(h0)
    tgt = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5(h, dx)
    T = T0
    while time.time() < t_end:
        T = max(T * ac, Tend)
        r = rng.random()
        if r < 0.35:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.02, 0.02)
            h[i] += d; h[j] -= d
        elif r < 0.65:
            c = _corr(h)
            ki = np.argmax(c)
            s = abs(int(ki - (n - 1)))
            if s > 0:
                m = n - s
                a = rng.randint(m)
                d = rng.uniform(0.004, 0.018)
                h[a + s] -= d; h[a] += d
            else:
                i, j = rng.randint(n), rng.randint(n)
                if i != j:
                    d = rng.uniform(-0.01, 0.01)
                    h[i] += d; h[j] -= d
        else:
            bl = rng.randint(2, max(3, n // 25))
            st = rng.randint(0, n - bl)
            fac = rng.uniform(0.5, 0.9)
            h[st:st+bl] *= fac
        h = _repair(h, tgt)
        cn = _c5(h, dx)
        if cn < best_c:
            best_c = cn; best_h = h.copy()
        elif cn > best_c:
            if rng.random() > np.exp(-(cn - best_c) / T):
                h = best_h.copy()
    return best_h, best_c

def _interp(h, nnew):
    if len(h) == nnew:
        return h.copy()
    return np.interp(np.linspace(0, 1, nnew), np.linspace(0, 1, len(h)), h)
