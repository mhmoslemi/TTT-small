```python
                if abs(h_new.sum() - target) > 1e-10:
                    h_new = _repair(h_new, n)
                f_new = _eval(h_new)
                if f_new < f - best_improve:
                    best_improve = f - f_new
                    best_i = i
                    best_val = h_new[i]
        if best_i >= 0:
            h[best_i] = best_val
            h = _repair(h, n)
            f = _eval(h)
    return h, f
```

This is O(n_iter * 50 * 4 * O(n)) per iteration which might be too slow. Let me vectorize the evaluation.

Actually, evaluating _eval(h) requires a full correlation which is O(n²). With n=280, that's 78400 multiplications. With 200 candidate evaluations per outer iteration, that's 15.7M ops per outer iteration. At ~100M ops/sec in numpy, that's 0.16s per outer iteration. Over 1000 iterations, that's 160s. Doable but tight.

Better approach: Instead of full re-evaluation, compute the change in c_k_star analytically when we change h[i] by delta:

Δc_k = (2/n) * [(1-h[j]) * delta - h[i'] * delta] (where j and i' depend on the specific pairing)

This is O(1) per candidate! Much faster.

Let me think about this more carefully. If we change h[i] → h[i] + δ, how does c_k change?

c_k = ∑_p h[p] * (1-h[q(p)]) for the appropriate pairs

The change is:
Δc_k = δ * (1-h[q(i)]) - h[p(i)] * δ (where q(i) is the partner of i in the first slot, and p(i) is the partner in the second slot)

For lag k with shift s = k - (n-1):
- If s ≥ 0:
  - i appears in first slot paired with i+s (if i < n-s): contribution δ*(1-h[i+s])
  - i appears in second slot paired with i-s (if i ≥ s): contribution -h[i-s]*δ
- If s < 0 (sp = -s):
  - i appears in first slot paired with i-sp (if i < n-sp): contribution δ*(1-h[i-sp])
  - i appears in second slot paired with i+sp (if i ≥ sp): contribution -h[i+sp]*δ

So Δc_k = δ * [(1-h[partner1]) - h[partner2]] (with terms only present if the index is valid)

For c_k to decrease, we need Δc_k < 0, i.e., the bracket is negative, meaning δ > 0 (increase h[i]) OR the bracket is positive and δ < 0.

This gives us a very fast way to find improving moves!

But we also need to not increase other c_k's above the current max. This is harder to check in O(1). A practical approach: just check c_k_star and accept if it decreases, then verify the full max at the end.

Let me refine the algorithm:

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / n)

def _grad_lag(k, h):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            return 1.0 - 2.0 * h
        i1 = np.arange(n - s)
        g[i1] += (1.0 - h[i1 + s])
        i2 = np.arange(s, n)
        g[i2] -= h[i2 - s]
    else:
        sp = -s
        i1 = np.arange(n - sp)
        g[i1] += (1.0 - h[i1 + sp])
        i2 = np.arange(sp, n)
        g[i2] -= h[i2 - sp]
    return g

def _topm_penalty_obj(h, m, lam):
    n = len(h)
    c = _corrs(h)
    top = np.partition(c, -m)[-m:]
    return top.mean() + lam * (h.sum() - n * 0.5) ** 2

def _topm_penalty_jac(h, m, lam):
    n = len(h)
    c = _corrs(h)
    top_idx = np.argpartition(c, -m)[-m:]
    g = np.zeros(n)
    for k in top_idx:
        g += _grad_lag(k, h)
    g *= (2.0 / n) / m
    g += 2.0 * lam * (h.sum() - n * 0.5)
    return g

def _funjac(h, m, lam):
    return _topm_penalty_obj(h, m, lam), _topm_penalty_jac(h, m, lam)

def _lbfgs_run(h, m, lam, maxiter=2000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_funjac, h.copy(), jac=True, method='L-BFGS-B',
                       args=(m, lam), bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _proj_descent(h, n_steps, step_init, m=5):
    n = len(h)
    f = _eval(h)
    step = step_init
    for _ in range(n_steps):
        if time.time() - _t0 > _budget - 10:
            break
        c = _corrs(h)
        top_idx = np.argpartition(c, -m)[-m:]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k, h)
        g *= (2.0 / n) / m
        improved = False
        for _ in range(25):
            h_new = _repair(h - step * g, n)
            f_new = _eval(h_new)
            if f_new < f - 1e-15:
                h = h_new
                f = f_new
                improved = True
                break
            step *= 0.5
        if not improved:
            break
        step = min(step * 1.05, step_init)
    return h, f

def _coord_fine(h, n_rounds):
    n = len(h)
    f = _eval(h)
    for r in range(n_rounds):
        if time.time() - _t0 > _budget - 5:
            break
        c = _corrs(h)
        k_star = int(c.argmax())
        g = _grad_lag(k_star, h) * (2.0 / n)
        order = np.argsort(np.abs(g))[::-1][:60]
        improved = False
        for i in order:
            direction = -1.0 if g[i] > 0 else 1.0
            for delta in [0.005, 0.002, 0.001]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + direction * delta, 0, 1)
                h_new = _repair(h_new, n)
                f_new = _eval(h_new)
                if f_new < f - 1e-14:
                    h = h_new
                    f = f_new
                    improved = True
                    break
            if time.time() - _t0 > _budget - 5:
                break
        if not improved:
            break
    return h, f

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic',
                 fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def _elapsed():
        return time.time() - _t0

    # Phase 1: L-BFGS-B with various m and lambda
    for m, lam in [(50, 500), (20, 1000), (10, 2000), (5, 5000), (3, 10000)]:
        if _elapsed() > budget_s - 200:
            break
        hn = _lbfgs_run(best_h, m, lam, 3000)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase 2: Projected gradient descent
    if _elapsed() < budget_s - 150:
        for m in [5, 3, 8]:
            if _elapsed() > budget_s - 120:
                break
            hn, fc = _proj_descent(best_h, 20000, 0.003, m)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 3: Coordinate fine-tuning
    if _elapsed() < budget_s - 100:
        hn, fc = _coord_fine(best_h, 200)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase 4: More L-BFGS with different params
    if _elapsed() < budget_s - 80:
        for m, lam in [(3, 20000), (2, 50000)]:
            if _elapsed() > budget_s - 60:
                break
            hn = _lbfgs_run(best_h, m, lam, 3000)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase 5: Upsample to 2x and optimize
    nc = len(best_h)
    if _elapsed() < budget_s - 200 and nc <= 560:
        h2 = _upsample(best_h, 2)
        n2 = len(h2)
        for m, lam in [(50, 500), (10, 2000), (5, 5000)]:
            if _elapsed() > budget_s - 150:

            hn = _lbfgs_run(h2, m, lam, 2000)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()
                best_n = n2
        nc = len(best_h)
        if _elapsed() < budget_s - 100:
            hn, fc = _proj_descent(best_h, 10000, 0.002, m=5)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()
                best_n = len(best_h)

    # Phase 6: Final coordinate descent + L-BFGS polish
    if _elapsed() < budget_s - 30:
        hn, fc = _coord_fine(best_h, 300)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()
        if _elapsed() < budget_s - 10:
            hn = _lbfgs_run(best_h, 3, 30000, 2000)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
```

Wait, I have a problem. The `_funjac` function takes `h, m, lam` but `minimize` with `jac=True` expects the function to return `(f, g)` where the function signature is `fun(x, *args)`. Let me verify this works with scipy.

One fix: fix the active set at the start of each L-BFGS-B call and only re-identify it periodically. But that complicates the code.

f(h) = (1/α) * log(∑ exp(α*c_k))

This is smooth and its gradient is well-defined. But it requires computing exp(α*c_k) for all 2n-1 = 559 values, which involves a loop for the gradient.

```python
def _lse_obj(h, alpha, lam):

    c = _corrs(h)
    s = alpha * (c - c.max())
    f = c.max() + np.log(np.sum(np.exp(s - s.max()))) / alpha
    pen = lam * (h.sum() - n * 0.5) ** 2
    return f + pen

    c = _corrs(h)
    cm = c.max()
    s = alpha * (c - cm)

    w = np.exp(s)
```

```python
        else:

            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
    else:
        sp = -s
        # g[j] = (1-h[j+sp]) for j <= n-1-sp, minus h[j-sp] for j >= sp
        g[:n-sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n-sp]
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / n)

def _grad_lag(k, h):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
    else:
        sp = -s
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g

def _lse_f_jac(h, alpha, lam):
    n = len(h)
    c = _corrs(h)
    cm = c.max()
    s = alpha * (c - cm)
    s_shift = s - s.max()
    w = np.exp(s_shift)
    w_sum = w.sum()
    f = cm + np.log(w_sum) / alpha + lam * (h.sum() - n * 0.5) ** 2
    g = np.zeros(n)
    w_norm = w / w_sum
    for k in range(2 * n - 1):
        if w_norm[k] < 1e-15:
            continue
        g += w_norm[k] * _grad_lag(k, h)
    g *= (2.0 / n)
    g += 2.0 * lam * (h.sum() - n * 0.5)
    return f, g

def _lbfgs(h, alpha, lam, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_f_jac, h.copy(), jac=True, method='L-BFGS-B',
                       args=(alpha, lam), bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': maxiter * 3})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _proj_desc(h, n_steps, step0, m):
    n = len(h)
    f = _eval(h)
    step = step0
    for _ in range(n_steps):
        if time.time() - _t0 > _budget - 8:
            break
        c = _corrs(h)
        top_idx = np.argpartition(c, -m)[-m:]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k, h)
        g *= (2.0 / n) / m
        ok = False
        for _ in range(30):
            h_new = _repair(h - step * g, n)
            f_new = _eval(h_new)
            if f_new < f - 1e-16:
                h = h_new
                f = f_new
                ok = True
                break
            step *= 0.5
        if not ok:
            break
        step = min(step * 1.06, step0)
    return h, f

def _coord_fine(h, n_rounds):
    n = len(h)
    f = _eval(h)
    for _ in range(n_rounds):
        if time.time() - _t0 > _budget - 5:
            break
        c = _corrs(h)
        k_star = int(c.argmax())
        g = _grad_lag(k_star, h) * (2.0 / n)
        order = np.argsort(np.abs(g))[::-1][:80]
        improved = False
        for i in order:
            if time.time() - _t0 > _budget - 5:
                break
            d = -np.sign(g[i])
            for delta in [0.003, 0.001, 0.0005, 0.0002]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d * delta, 0, 1)
                h_new = _repair(h_new, n)
                f_new = _eval(h_new)
                if f_new < f - 1e-14:
                    h = h_new
                    f = f_new
                    improved = True
                    break
        if not improved:
            break
    return h, f

def _topm_lbfgs(h, m, lam, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _tf(x):
        c = _corrs(x)
        top = np.partition(c, -m)[-m:]
        return top.mean() + lam * (x.sum() - n * 0.5) ** 2
    def _tg(x):
        c = _corrs(x)
        ti = np.argpartition(c, -m)[-m:]
        g = np.zeros(n)
        for k in ti:
            g += _grad_lag(k, x)
        g *= (2.0 / n) / m
        g += 2.0 * lam * (x.sum() - n * 0.5)
        return g
    def _tfj(x):
        return _tf(x), _tg(x)
    try:
        res = minimize(_tfj, h.copy(), jac=True, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    fi = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(fi(x_new), 0, 1), n * factor)

def _perturb_reopt(h, m, alpha, lam, maxiter, scale):
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    n = len(h)
    hp = h + scale * rng.randn(n)
    hp = _repair(hp, n)
    hn = _lbfgs(hp, alpha, lam, maxiter)
    return hn
```
