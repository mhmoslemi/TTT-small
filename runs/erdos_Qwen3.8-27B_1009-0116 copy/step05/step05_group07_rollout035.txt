```python
def delta_c(i, j, h, n):
    # i: 1->0, j: 0->1
    d = np.arange(1, n)  # shifts 1 to n-1
    # h[i+d] for d where i+d < n
    h_i_d = np.zeros(n-1)
    valid = (i + d < n)
    h_i_d[valid] = 1 - h[i + d[valid]]
    h_i_m_d = np.zeros(n-1)
    valid2 = (i - d >= 0)
    h_i_m_d[valid2] = h[i - d[valid2]]
    h_j_d = np.zeros(n-1)
    valid3 = (j + d < n)
    h_j_d[valid3] = 1 - h[j + d[valid3]]
    h_j_m_d = np.zeros(n-1)
    valid4 = (j - d >= 0)
    h_j_m_d[valid4] = h[j - d[valid4]]
    return -h_i_d + h_i_m_d + h_j_d - h_j_m_d
```

This is O(n) per swap but with numpy overhead. For 100 candidates, it's 100 * O(n) = fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_shift(d, h):
    n = len(h)
    g = np.zeros(n)
    ad = abs(d)
    if ad >= n:
        return g
    if ad == 0:
        return (2.0 / n) * (1.0 - 2.0 * h)
    idx1 = np.arange(n - ad)
    g[idx1] += (1.0 - h[idx1 + ad])
    idx2 = np.arange(ad, n)
    g[idx2] -= h[idx2 - ad]
    return (2.0 / n) * g

def _delta_c_binary(i, j, h, n):
    d = np.arange(1, n)
    di_d = np.zeros(n - 1)
    m = (i + d < n)
    di_d[m] = 1.0 - h[i + d[m]]
    di_md = np.zeros(n - 1)
    m2 = (i - d >= 0)
    di_md[m2] = h[i - d[m2]]
    dj_d = np.zeros(n - 1)
    m3 = (j + d < n)
    dj_d[m3] = 1.0 - h[j + d[m3]]
    dj_md = np.zeros(n - 1)
    m4 = (j - d >= 0)
    dj_md[m4] = h[j - d[m4]]
    return -di_d + di_md + dj_d - dj_md

def _binary_search(h, max_iter=200000):

    h = (h >= 0.5).astype(float)
    target = n / 2.0
    if h.sum() != target:
        h = _repair(h, n)
        h = (h >= 0.5).astype(float)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    c = _corr(h)
    best_val = float(np.max(c) * 2.0 / n)
    rng = np.random.RandomState(12345)
    for it in range(max_iter):
        if time.time() - _t0 > _budget - 60:
            break
        if not ones or not zeros:
            break
        c = _corr(h)
        d_star = int(np.argmax(c))
        dd = d_star - (n - 1)
        ad = abs(dd)
        if ad == 0 or ad >= n:
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i], h[j] = 0, 1
            nv = float(max(_corr(h)) * 2.0 / n)
            if nv < best_val:
                best_val = nv
                ones.remove(i); zeros.remove(j); ones.append(j); zeros.append(i)
            else:
                h[i], h[j] = 1, 0
            continue
        if dd > 0:
            bad = [p for p in range(n - dd) if h[p] == 1 and h[p + dd] == 0]
        else:
            bad = [p for p in range(-dd, n) if h[p] == 1 and h[p - ad] == 0]
        if not bad:
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i], h[j] = 0, 1
            nv = float(max(_corr(h)) * 2.0 / n)
            if nv < best_val:
                best_val = nv

            else:

        improved = False

            if dd > 0:

            else:

            h_tp = h[target_pos]

def _psgd(h, topm, max_iter, step0):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    stall = 0
    for it in range(max_iter):
        if time.time() - _t0 > _budget - 30:
            break
        c = _corr(h) * (2.0 / n)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            d = k - (n - 1)
            g += _grad_shift(d, h)
        g /= topm
        found = False
        for _bt in range(50):
            h_new = _proj_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-14:
                h = h_new
                f = f_new
                found = True
                break
            step *= 0.5
        if found:
            stall = 0
            step = min(step * 1.04, step0)
        else:
            stall += 1
            if stall > 200:
                break
            step = step0
    return h, f

def _slsqp_exact(h, topm, maxiter):
    n = len(h)
    target = n * 0.5
    def obj(x):
        x = np.clip(x, 0, 1)
        c = _corr(x) * (2.0 / n)
        m = c.max()
        top = np.argsort(c)[::-1][:topm]
        return float(np.mean(c[top])) * n / 2.0
    def grad(x):
        x = np.clip(x, 0, 1)
        c = _corr(x) * (2.0 / n)
        top = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top:
            d = k - (n - 1)
            g += _grad_shift(d, x)
        g /= topm
        g /= (n / 2.0)
        return g
    bnds = [(0.0, 1.0)] * n
    cn = {'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=[cn], options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pairwise(h, n_iter, scale, seed_val):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 10:
            break
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f
```
