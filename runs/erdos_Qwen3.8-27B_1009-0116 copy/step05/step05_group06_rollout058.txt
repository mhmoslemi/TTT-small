```python
    """Gradient of correlation at lag k w.r.t. h"""
    n = len(h)
    shift = (n - 1) - k
    g = np.zeros(n)
    if shift >= 0:
```

But we can use SLSQP on just this single objective! It's much simpler than the log-sum-exp.

```python
def _min_single_corr(h_init, k, n):
    """Minimize correlation at lag k"""
    shift = (n-1) - k
    def obj(h):
        if shift >= 0:
            return np.sum(h[:n-shift] * (1-h[shift:]))
        else:
            s = -shift
            return np.sum(h[s:] * (1-h[:-s]))
    def grad(h):
        g = np.zeros(n)
        if shift >= 0:
            i1 = np.arange(n - shift)
            g[i1] += (1.0 - h[i1 + shift])
            if shift > 0:
                g[i1 + shift] -= h[i1]
        else:
            s = -shift
            i1 = np.arange(s, n)
            g[i1] += (1.0 - h[i1 - s])
            g[i1 - s] -= h[i1]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': 5000, 'ftol': 1e-18})
    return _repair(res.x, n)
```

This is a much simpler problem than the log-sum-exp! And we can cycle through the top few lags.

**Alternating minimization over lags:**
1. Find top K lags
2. For each lag in turn, minimize that specific correlation
3. Check if the overall max decreased
4. Repeat

This is like a "coordinate descent" over the lags!

This could be very effective because:
- Each subproblem is simpler (one bilinear term vs. max of many)
- SLSQP can solve each one faster
- We naturally balance the correlations

Let me restructure my approach around this idea.

Actually, I realize there's an issue: minimizing one correlation might increase others. So we need to be careful. But if we minimize the worst one first, then the second worst, etc., it should work as a heuristic.

Let me also think about the "dual" perspective. At the optimum, we expect several correlations to be roughly equal (at the max). If we can find a solution where the top 3-5 correlations are all minimized simultaneously, that's ideal.

The log-sum-exp approach does this, but it's a poor approximation of the max when alpha is not huge. The alternating approach might be better.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0, 1).sum()
        lo, hi = (mid, hi) if s > target else (lo, mid)
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

    shift = (n - 1) - k
    if shift >= 0:
        return np.sum(h[:n-shift] * (1.0 - h[shift:]))
    else:
        s = -shift
        return np.sum(h[s:] * (1.0 - h[:-s]))

def _lag_grad(h, k, n):
    shift = (n - 1) - k
    g = np.zeros(n)
    if shift >= 0:
        i1 = np.arange(n - shift)
        g[i1] += (1.0 - h[i1 + shift])
        if shift > 0:

def _slsqp_lag(h, k, n, maxiter=5000):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return _lag_obj(x, k, n)
    def jac(x):
        return _lag_grad(x, k, n)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _slsqp_smooth(h, alpha, n, maxiter=3000):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def jac(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for kk in range(2 * n - 1):
            if w[kk] < 1e-12:
                continue
            g += w[kk] * _lag_grad(x, kk, n)
        return g * dx
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _alternating_lag(h, n, n_rounds=5, topk=3, maxiter=3000):
    """Alternate minimizing top-k correlations"""
    for r in range(n_rounds):
        c = _corr(h)
        top_lags = np.argsort(c)[::-1][:topk]
        for k in top_lags:
            h = _slsqp_lag(h, int(k), n, maxiter)
    return h

def _proj_subgrad(h, n, nsteps, step0, topk):
    target = n * 0.5
    dx = 2.0 / n
    c5 = _eval(h)
    step = step0
    for _ in range(nsteps):
        c = _corr(h)
        m = c.max()
        idx = np.argsort(c)[::-1][:topk]
        w = c[idx]
        w -= m
        w = np.exp(1000 * np.clip(w, -0.05, 0))
        w /= w.sum()
        g = np.zeros(n)
        for j, kk in enumerate(idx):
            g += w[j] * _lag_grad(h, int(kk), n)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        for bt in range(25):
            trial = _proj_sum(h - step * g / gn, target)
            c5n = _eval(trial)
            if c5n < c5:
                h, c5 = trial, c5n
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-14:
                break
        else:
            step = min(step * 1.03, step0 * 3.0)
    return h, c5
```

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _lag_shift(k, n):
    return k - (n - 1)

def _lag_obj(h, k, n):
    sh = _lag_shift(k, n)
    if sh >= 0:
        return float(np.sum(h[sh:] * (1.0 - h[:n - sh])))
    s = -sh
    return float(np.sum(h[s:] * (1.0 - h[:-s])))

def _lag_grad(h, k, n):
    sh = _lag_shift(k, n)
    g = np.zeros(n)
    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh])
        g[:n - sh] -= h[sh:]
    else:
        s = -sh
        g[s:] += (1.0 - h[:-s])
        g[:-s] -= h[s:]
    return g

def _slsqp_single_lag(h, k, n, maxiter=5000):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    res = minimize(_lag_obj, h, jac=lambda x: _lag_grad(x, k, n),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return _repair(res.x, n)

def _slsqp_smooth(h, alpha, n, maxiter=3000):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return float(m + np.log(np.sum(np.exp(s))) / alpha)
    def jac(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for kk in range(2 * n - 1):
            if w[kk] > 1e-12:
                g += w[kk] * _lag_grad(x, kk, n)
        return g * (2.0 / n)
    res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return _repair(res.x, n)

def _alternating_lags(h, n, rounds=5, topk=3, maxiter=3000):
    h_cur = h.copy()
    best_c5 = _eval(h_cur)
    for r in range(rounds):
        c = _corr(h_cur)
        top = np.argsort(c)[::-1][:topk]
        for k in top:
            h_try = _slsqp_single_lag(h_cur, int(k), n, maxiter)
            c5_try = _eval(h_try)
            if c5_try < best_c5:
                h_cur, best_c5 = h_try, c5_try
            elif c5_try < _eval(h_cur):
                h_cur = h_try
    return h_cur, best_c5

def _proj_subgrad(h, n, nsteps, step0, topk):
    target = n * 0.5
    dx = 2.0 / n
    c5 = _eval(h)
    step = step0
