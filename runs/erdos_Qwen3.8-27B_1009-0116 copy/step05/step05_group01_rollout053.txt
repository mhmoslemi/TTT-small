                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _soft_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _soft_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _coord_descent(h, n, n_grid, max_sweeps):
    dx = 2.0 / n
    v_grid = np.linspace(0.0, 1.0, n_grid)
    corr = _corr(h)
    c5 = corr.max() * dx
    for sweep in range(max_sweeps):
        improved = 0
        changes = 0
        for i in range(n):
            old = h[i]
            dc1 = _pos_change(h, i, 1.0, n)
            di_grid = v_grid - old
            new_c5s = (corr + di_grid[:, None] * dc1[None, :]).max(axis=1) * dx
            idx = np.argmin(new_c5s)
            if new_c5s[idx] < c5 - 1e-13:
                h[i] = v_grid[idx]
                corr = corr + (v_grid[idx] - old) * dc1
                c5 = new_c5s[idx]
                improved += 1
                changes += 1
                if changes % 15 == 0:
                    corr = _corr(h)
                    c5 = corr.max() * dx
        if improved == 0:
            break
    corr = _corr(h)
    c5 = corr.max() * dx
    return h, c5

def _incr_pair(h, c, rng, n_iter, scale, n):
    dx = 2.0 / n
    c5c = c.max() * dx
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-15:
            h[i] = ni
            h[j] = nj
            c = new_c
            c5c = new_c5
    return h, c, c5c

def _block_cd(h, n, n_grid, max_sweeps, block_sizes):
    dx = 2.0 / n
    v_grid = np.linspace(0.0, 1.0, n_grid)
    corr = _corr(h)
    c5 = corr.max() * dx
    for sweep in range(max_sweeps):
        improved = 0
        for bs in block_sizes:
            for start in range(0, n - bs + 1, bs):
                idxs = list(range(start, min(start + bs, n)))
                old_vals = [h[i] for i in idxs]
                dc_sum = np.zeros(2 * n - 1)
                for i in idxs:
                    dc_sum += _pos_change(h, i, 1.0, n)
                best_delta = 0.0
                best_c5 = c5
                for dv in v_grid:
                    new_c5 = (corr + dv * dc_sum).max() * dx
                    if new_c5 < best_c5 - 1e-13:
                        best_c5 = new_c5
                        best_delta = dv
                if best_delta != 0.0:
                    for i in idxs:
                        h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                    corr = _corr(h)
                    c5 = corr.max() * dx
                    improved += 1
        if improved == 0:
            break
    corr = _corr(h)
    c5 = corr.max() * dx
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP warm-start
    for alpha in [2000, 10000, 50000, 200000]:
        if _el() > 60:
            break
        h_new = _slsqp(best_h, alpha, 2000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 2: Coordinate descent (main new contribution)
    if _el() < budget_s - 500:
        h_cd, c5_cd = _coord_descent(best_h, n0, 200, 40)
        h_cd = _repair(h_cd, n0)
        c5_cd = _eval(h_cd)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd

    # Phase 3: SLSQP with higher alpha
    for alpha in [500000, 2000000]:
        if _el() > budget_s - 400:
            break
        h_new = _slsqp(best_h, alpha, 3000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 4: Fine coordinate descent
    if _el() < budget_s - 350:
        h_cd, c5_cd = _coord_descent(best_h, n0, 500, 30)
        h_cd = _repair(h_cd, n0)
        c5_cd = _eval(h_cd)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd

    # Phase 5: Incremental pairwise refinement
    if _el() < budget_s - 250:
        h_ps = best_h.copy()
        c_ps = _corr(h_ps)
        for scale in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if _el() > budget_s - 250:
                break
            h_ps, c_ps, _ = _incr_pair(h_ps, c_ps, rng, 30000, scale, n0)
        h_ps = _repair(h_ps, n0)
        c5 = _eval(h_ps)
        if c5 < best_c5:
            best_c5, best_h = c5, h_ps

    # Phase 6: Block coordinate descent (structured moves)
    if _el() < budget_s - 200:
        h_bc, c5_bc = _block_cd(best_h, n0, 100, 15, [2, 3, 4])
        h_bc = _repair(h_bc, n0)
        c5_bc = _eval(h_bc)
        if c5_bc < best_c5:
            best_c5, best_h = c5_bc, h_bc

    # Phase 7: SLSQP polish
    if _el() < budget_s - 100:
        h_pol = _slsqp(best_h, 5000000, 4000, 1e-20)
        c5 = _eval(h_pol)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_pol.copy(), n0

    # Phase 8: Ultra-fine coordinate descent
    if _el() < budget_s - 60:
        h_cd, c5_cd = _coord_descent(best_h, n0, 1000, 20)
        h_cd = _repair(h_cd, n0)
        c5_cd = _eval(h_cd)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd

    # Phase 9: Multi-start exploration
    if _el() < budget_s - 120:
        for _r in range(8):
            if _el() > budget_s - 120:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.015 * rng.randn(n0), n0)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(n0), n0)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 5)):
                    bs = rng.randint(3, max(4, n0 // 10))
                    st = rng.randint(0, n0 - bs)
                    hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, n0)
            else:
                hp = best_h.copy()
                perm = rng.permutation(n0)
                hp[perm[:n0 // 4]] = np.clip(1.0 - hp[perm[:n0 // 4]] + 0.005 * rng.randn(n0 // 4), 0, 1)
                hp = _repair(hp, n0)
            h_ms = _slsqp(hp, 200000, 1500, 1e-19)
            h_ms, c5_ms = _coord_descent(h_ms, n0, 200, 20)
            h_ms = _repair(h_ms, n0)
            c5 = _eval(h_ms)
            if c5 < best_c5:
                best_c5, best_h = c5, h_ms

    # Phase 10: Final fine incremental + coordinate
    if _el() < budget_s - 10:
        h_fin = best_h.copy()
        c_fin = _corr(h_fin)
        h_fin, c_fin, _ = _incr_pair(h_fin, c_fin, rng, 40000, 5e-5, n0)
        h_fin, c5_fin = _coord_descent(h_fin, n0, 500, 15)
        h_fin = _repair(h_fin, n0)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
