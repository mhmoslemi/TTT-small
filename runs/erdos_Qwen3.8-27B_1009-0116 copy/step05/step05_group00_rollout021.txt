def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    # Phase 2: L-BFGS-B with penalty
    if time.time() < t_end - 250:
        h2 = _lbfgs_polish(best_h, 200000, t_end - 250)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2.copy()

    # Phase 3: Complement trick
    if time.time() < t_end - 200:
        hc = _repair(1.0 - best_h, best_n)
        hc, c5c = _pgd(hc, 50000, 1e-5, 5000, t_end - 200, momentum=0.9)
        if c5c < best_c5:
            best_c5 = c5c; best_h = hc.copy()

    # Phase 4: Binary SA
    if time.time() < t_end - 150:
        hb, cb = _binary_sa(best_h, t_end - 150)
        if cb < best_c5:
            best_c5 = cb; best_h = hb.copy()
        # Unfreeze the best binary
        hu = _anneal_unfreeze(hb, t_end - 80)
        cu = _eval(hu)
        if cu < best_c5:
            best_c5 = cu; best_h = hu.copy()

    # Phase 5: High-alpha PGD polish
    if time.time() < t_end - 80:
        h, c5 = _pgd(best_h, 300000, 5e-6, 5000, t_end - 80, momentum=0.85)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()

    # Phase 6: Pairwise coordinate descent (refinement)
    if time.time() < t_end - 40:
        h, c5 = _pair_cd(best_h, t_end - 40, 0.003)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
        h, c5 = _pair_cd(best_h, t_end - 20, 0.0005)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()

    # Phase 7: Try nearby n
    if time.time() < t_end - 20:
        for dn in [10, -10, 20]:
            if time.time() > t_end - 20:
                break
            nt = best_n + dn
            if nt < 80 or nt % 2 != 0:
                continue
            hr = _repair(_resample(best_h, nt), nt)
            hr, cr = _pgd(hr, 100000, 3e-5, 3000, t_end - 20, momentum=0.9)
            if cr < best_c5:
                best_c5 = cr; best_h = hr.copy(); best_n = nt

    # Final: one more very fine CD
    if time.time() < t_end - 3:
        h, c5 = _pair_cd(best_h, t_end - 3, 0.0001)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
