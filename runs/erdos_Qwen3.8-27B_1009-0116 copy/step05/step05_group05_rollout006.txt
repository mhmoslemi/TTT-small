        for w, k in zip(weights, idx):
            sh = k - (n - 1)
            wk = w * dx
            if sh >= 0:
                g[sh:] += wk * (1.0 - h[:n - sh])
                g[:n - sh] -= wk * h[sh:]
            else:
                p = -sh
                g[:n - p] += wk * (1.0 - h[p:])
                g[p:] -= wk * h[:n - p]
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 1e-4]:
            h_new = _proj(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _systematic_pair(h, n_passes, t0, tmax):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _pass in range(n_passes):
        if time.time() - t0 > tmax:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            for j in range(i + 1, n):
                if (i + j) % 17 == 0:
                    dlo = max(-h[i], h[j] - 1.0)
                    dhi = min(1.0 - h[i], h[j])
                    if dhi - dlo < 1e-12:
                        continue
                    a = np.zeros(2 * n - 1)
                    for k in range(2 * n - 1):
                        sh = k - (n - 1)
                        ci = 0.0
                        cj = 0.0
                        if sh >= 0:
                            if sh <= i <= n - 1:
                                ci += (1.0 - h[i - sh])
                            if 0 <= i <= n - 1 - sh:
                                ci -= h[i + sh]
                            if sh <= j <= n - 1:
                                cj += (1.0 - h[j - sh])
                            if 0 <= j <= n - 1 - sh:
                                cj -= h[j + sh]
                        else:
                            p = -sh
                            if 0 <= i <= n - 1 - p:
                                ci += (1.0 - h[i + p])
                            if p <= i <= n - 1:
                                ci -= h[i - p]
                            if 0 <= j <= n - 1 - p:
                                cj += (1.0 - h[j + p])
                            if p <= j <= n - 1:
                                cj -= h[j - p]
                        a[k] = ci - cj
                    best_d = 0.0
                    best_v = c.max()
                    for d in np.linspace(dlo, dhi, 15):
                        v = np.max(c + dx * d * a)
                        if v < best_v:
                            best_v = v
                            best_d = d
                    if abs(best_d) > 1e-12:
                        h[i] += best_d
                        h[j] -= best_d
                        h = np.clip(h, 0, 1)
                        c5n = _eval(h)
                        if c5n < c5c - 1e-16:
                            c5c = c5n
                            c = np.correlate(h, 1.0 - h, mode='full')
    return h, c5c

def _upsample_h(h, factor):
    n = len(h)
    n_new = n * factor
    h_hi = np.zeros(n_new)
    for i in range(n_new):
        pos = i / factor
        i0 = int(pos)
        i1 = min(i0 + 1, n - 1)
        frac = pos - i0
        h_hi[i] = (1 - frac) * h[i0] + frac * h[i1]
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: Projected gradient cleanup (fast)
    for alpha in [2000, 20000, 200000]:
        if time.time() - t0 > 40:
            break
        h_new, c5 = _proj_grad(best_h.copy(), alpha, 2000, t0, t0 + 12)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase B: Heavy structured pairwise with restarts
    for _restart in range(5):
        if time.time() - t0 > budget_s - 400:
            break
        h_new, c5 = _structured_pair(best_h.copy(), 30000, rng, t0, t0 + 40, 25)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        else:
            best_h[best_h > 0.5] += 0.01 * rng.randn(best_h[best_h > 0.5].size)
            best_h[best_h <= 0.5] += 0.01 * rng.randn(best_h[best_h <= 0.5].size)
            best_h = _repair(best_h.copy(), best_n)

    # Phase C: Active set descent
    if time.time() - t0 < budget_s - 300:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 5, t0, t0 + 50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase D: More structured pairwise
    if time.time() - t0 < budget_s - 200:
        h_new, c5 = _structured_pair(best_h.copy(), 40000, rng, t0, t0 + 70, 31)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase E: Upsample to 2x and optimize
    if time.time() - t0 < budget_s - 250:
        n1 = n0 * 2
        h_hi = _repair(_upsample_h(best_h, 2), n1)
        if time.time() - t0 < budget_s - 200:
            h_hi, c5 = _proj_grad(h_hi, 50000, 1500, t0, t0 + 30)
        if time.time() - t0 < budget_s - 130:
            h_hi, c5 = _structured_pair(h_hi, 20000, rng, t0, t0 + 50, 21)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if time.time() - t0 < budget_s - 60:
            h_hi, c5 = _active_descent(best_h.copy(), 3000, 5, t0, t0 + 30)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = len(h_hi)

    # Phase F: Multi-start exploration
    n_cur = best_n
    if time.time() - t0 < budget_s - 100:
        for _r in range(8):
            if time.time() - t0 > budget_s - 70:
                break
            hp = best_h.copy()
            t_arr = np.arange(n_cur)
            for _f in range(4):
                f = rng.randint(1, n_cur // 3 + 1)
                hp += 0.004 * rng.randn() * np.sin(2 * np.pi * f * t_arr / n_cur + rng.rand() * 6.28)
            hp = _repair(hp, n_cur)
            h_new, c5 = _proj_grad(hp, 300000, 500, t0, t0 + 5)
            h_new, c5 = _structured_pair(h_new, 5000, rng, t0, t0 + 8, 21)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_cur

    # Phase G: Final structured pairwise polish
    if time.time() - t0 < budget_s - 30:
        h_new, c5 = _structured_pair(best_h.copy(), 30000, rng, t0, t0 + 20, 31)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase H: Ultra-fine active descent
    if time.time() - t0 < budget_s - 5:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 3, t0, t0 + 4)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
