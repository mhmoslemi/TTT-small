def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if abs(di) < 1e-16:
        return dc
    # a-part: r from i to i+n-1
    r1_lo = i
    r1_hi = min(2*n-2, i+n-1)
    # h[i-r+n-1] for r in [r1_lo, r1_hi]
    # i-r+n-1: from i-r1_hi+n-1 to i-r1_lo+n-1 = n-1
    idx_a = np.arange(i - r1_hi + n - 1, n)  # This is [i-r1_hi+n-1, ..., n-1]
    dc[r1_lo:r1_hi+1] += di * (1.0 - h[idx_a])
    
    # v-part: r from n-1-i to 2n-2-i
    r2_lo = max(0, n-1-i)
    r2_hi = min(2*n-2, 2*n-2-i)
    # h[i+r-n+1] for r in [r2_lo, r2_hi]
    # i+r2_lo-n+1 to i+r2_hi-n+1
    idx_v = np.arange(i + r2_lo - n + 1, i + r2_hi - n + 2)
    dc[r2_lo:r2_hi+1] -= di * h[idx_v]
    
    return dc

1. Compute c = _corr(h)
2. For each iteration, pick a random pair (i,j), compute dc = _pos_change(h,i,d,n) + _pos_change(h,j,-d,n)
3. new_c = c + dc
4. If new_c.max() < c.max() - eps (or with SA, accept with probability), update.

2. Phase 1: SLSQP with smooth max (relatively quick, maybe 30-60s)

For the SA, I'll use:
- Temperature schedule: T = T0 * exp(-cooling * iter)
- Accept if new < old, or with probability exp((old-new)/T)
- Move types:
  - Pair swap: h[i] += d, h[j] -= d (keeps sum)
  - Block shift: move a contiguous block
  - Pin: set h[i] to 0 or 1 (with compensation)

Actually, for keeping the sum constraint exactly, the pair swap is the cleanest. Let me focus on that but with SA.

Let me also add a "gradient-guided" move: instead of random d, use the gradient of the max correlation to guide the direction.

The gradient of c_k with respect to h[i] is:
∂c_k/∂h[i] = (1-h[i]) * [h[i] appears in a-part] - h[i] * [h[i] appears in v-part]

For the smooth max, the gradient is a weighted sum of individual correlation gradients, weighted by their relative magnitude.

But for the local search, I'll keep it simple: random pair moves with SA.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if abs(di) < 1e-16:
        return dc
    r1_lo = i
    r1_hi = min(2 * n - 2, i + n - 1)
    idx_a = np.arange(i - r1_hi + n - 1, n)
    dc[r1_lo:r1_hi + 1] += di * (1.0 - h[idx_a])
    r2_lo = max(0, n - 1 - i)
    r2_hi = min(2 * n - 2, 2 * n - 2 - i)
    idx_v = np.arange(i + r2_lo - n + 1, i + r2_hi - n + 2)
    dc[r2_lo:r2_hi + 1] -= di * h[idx_v]
    return dc

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for i in range(n):
        r1_lo = i
        r1_hi = min(2 * n - 2, i + n - 1)
        idx_a = np.arange(i - r1_hi + n - 1, n)
        g[i] += np.dot(w[r1_lo:r1_hi + 1], 1.0 - h[idx_a])
        r2_lo = max(0, n - 1 - i)
        r2_hi = min(2 * n - 2, 2 * n - 2 - i)
        idx_v = np.arange(i + r2_lo - n + 1, i + r2_hi - n + 2)
        g[i] -= np.dot(w[r2_lo:r2_hi + 1], h[idx_v])
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _sa_search(h, c, rng, n_iter, t0_sa, cooling, dx):
    n = len(h)
    c_max = c.max()
    # Simulated annealing with pair moves
    for it in range(n_iter):
        T = t0_sa * np.exp(-cooling * it)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.randn() * (0.01 + 0.03 * T / t0_sa)
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_max = new_c.max()
        if new_max < c_max or rng.random() < np.exp(-(new_max - c_max) / max(T, 1e-10)):
            h[i] = ni
            h[j] = nj
            c = new_c
            c_max = new_max
    return h, c, c_max

def _deterministic_search(h, c, rng, n_iter, scale, n):
    c_max = c.max()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_max = new_c.max()
        if new_max < c_max - 1e-15:
            h[i] = ni
            h[j] = nj
            c = new_c
            c_max = new_max
    return h, c, c_max

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP at original resolution
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if _el() > 100:
            break
        h_new = _slsqp(best_h, alpha, 2000, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 2: Simulated annealing (main optimization)
    if _el() < budget_s - 400:
        h_sa = best_h.copy()
        c_sa = _corr(h_sa)
        n_sa = len(h_sa)
        dx = 2.0 / n_sa
        # Multiple SA rounds with decreasing temperature
        for round_idx in range(5):
            if _el() > budget_s - 400:
                break
            T0 = 0.005 * (0.5 ** round_idx)
            cool = 5000 / n_sa
            h_sa, c_sa, _ = _sa_search(h_sa, c_sa, rng, 30000, T0, cool, dx)
            h_sa = _repair(h_sa, n_sa)
            c5 = _eval(h_sa)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_sa.copy(), n_sa

    # Phase 3: Deterministic greedy search (fine scale)
    if _el() < budget_s - 200:
        h_g = best_h.copy()
        c_g = _corr(h_g)
        n_g = len(h_g)
        for scale in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            if _el() > budget_s - 200:
                break
            h_g, c_g, _ = _deterministic_search(h_g, c_g, rng, 40000, scale, n_g)
        h_g = _repair(h_g, n_g)
        c5 = _eval(h_g)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_g, n_g

    # Phase 4: SLSQP polish
    if _el() < budget_s - 100:

        c5 = _eval(h_pol)
        if c5 < best_c5:

    # Phase 5: Final fine search
    if _el() < budget_s - 30:
        h_f = best_h.copy()
        c_f = _corr(h_f)
        n_f = len(h_f)
        for scale in [1e-4, 5e-5, 1e-5, 5e-6]:
            if _el() > budget_s - 30:

            h_f, c_f, _ = _deterministic_search(h_f, c_f, rng, 50000, scale, n_f)
        h_f = _repair(h_f, n_f)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_f, n_f

    # Phase 6: Multi-start exploration
    if _el() < budget_s - 150:
        nc = len(best_h)
        for _r in range(8):
            if _el() > budget_s - 150:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.01 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(best_h + 0.005 * rng.randn(nc), nc)
            elif strat == 2:
                hp = _repair(np.clip(best_h + 0.03 * (rng.random(nc) - 0.5) * (best_h < 0.15 | best_h > 0.85), 0, 1), nc)
            else:

        c_fin = _corr(h_fin)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if abs(di) < 1e-16:
        return dc
    r1_lo = i
    r1_hi = min(2 * n - 2, i + n - 1)
    idx_a = np.arange(i - r1_hi + n - 1, n)
    dc[r1_lo:r1_hi + 1] += di * (1.0 - h[idx_a])
    r2_lo = max(0, n - 1 - i)
    r2_hi = min(2 * n - 2, 2 * n - 2 - i)
    idx_v = np.arange(i + r2_lo - n + 1, i + r2_hi - n + 2)
    dc[r2_lo:r2_hi + 1] -= di * h[idx_v]
    return dc

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())

        r1_hi = min(2 * n - 2, i + n - 1)

        g[i] += np.dot(w[r1_lo:r1_hi + 1], 1.0 - h[idx_a])

        g[i] -= np.dot(w[r2_lo:r2_hi + 1], h[idx_v])
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _sa_search(h, c, rng, n_iter, T0, cool_rate, n):
    c_max = float(c.max())
    for it in range(n_iter):
        T = T0 * np.exp(-cool_rate * it)

        i = rng.randint(n)

        ni = h[i] + d

        ni = float(np.clip(ni, 0.0, 1.0))
        nj = float(np.clip(nj, 0.0, 1.0))
        d_eff = ni - h[i]
        d_jef = nj - h[j]
        if abs(d_eff) < 1e-15 and abs(d_jef) < 1e-15:
            continue
        dc = _pos_change(h, i, d_eff, n) + _pos_change(h, j, d_jef, n)
        new_c = c + dc
        new_max = float(new_c.max())
        delta = new_max - c_max
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            h[i] = ni
            h[j] = nj
            c = new_c
            c_max = new_max
    return h, c, c_max
