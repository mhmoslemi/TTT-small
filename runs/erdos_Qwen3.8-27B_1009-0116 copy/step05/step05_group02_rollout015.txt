        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pgd_worst(h, n, deadline, rng):
    best = h.copy()
    best_c5 = _eval(h)
    cur = h.copy()
    cur_c5 = best_c5
    lr = 0.015
    stall = 0
    while time.time() < deadline:
        c = np.correlate(cur, 1.0 - cur, mode='full')
        k_star = int(c.argmax())
        sh = (n - 1) - k_star
        g = _grad_lag(cur, sh, n) * (2.0 / n)
        improved = False
        for _ in range(10):
            new = _proj_to_sum(cur - lr * g, n * 0.5)
            nc5 = _eval(new)
            if nc5 < cur_c5 - 1e-17:
                cur = new
                cur_c5 = nc5
                stall = 0
                if nc5 < best_c5:
                    best_c5 = nc5
                    best = cur.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            stall += 1
            if stall > 300:
                cur = _repair(cur + 5e-5 * rng.randn(n), n)
                cur_c5 = _eval(cur)
                stall = 0
                lr = 0.01
            else:
                d = rng.randn(n)
                d -= d.mean()
                d /= (np.linalg.norm(d) + 1e-30)
                for eps in [2e-3, 1e-3, 5e-4]:
                    new = _proj_to_sum(cur + eps * d, n * 0.5)
                    nc5 = _eval(new)
                    if nc5 < cur_c5 - 1e-17:
                        cur = new
                        cur_c5 = nc5
                        if nc5 < best_c5:
                            best_c5 = nc5
                            best = cur.copy()
                        break
    return best, best_c5

def _multi_lag_pgd(h, n, deadline, rng, top_m=3):
    best = h.copy()
    best_c5 = _eval(h)
    cur = h.copy()
    cur_c5 = best_c5
    lr = 0.008
    stall = 0
    while time.time() < deadline:
        c = np.correlate(cur, 1.0 - cur, mode='full')
        cm = c.max()
        s = np.clip(5000 * (c - cm), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        # Focus on top_m
        idx = np.argsort(w)[-top_m:]
        w2 = np.zeros_like(w)
        for i in idx:
            w2[i] = w[i]
        w2 /= w2.sum()
        g = _grad_weighted(w2, cur, n)
        improved = False
        for _ in range(10):
            new = _proj_to_sum(cur - lr * g, n * 0.5)
            nc5 = _eval(new)
            if nc5 < cur_c5 - 1e-17:
                cur = new
                cur_c5 = nc5
                stall = 0
                if nc5 < best_c5:
                    best_c5 = nc5
                    best = cur.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            stall += 1
            if stall > 200:
                cur = _repair(cur + 3e-5 * rng.randn(n), n)
                cur_c5 = _eval(cur)
                stall = 0
                lr = 0.005
    return best, best_c5

def _pairwise(h, n, deadline, rng):
    best = h.copy()
    best_c5 = _eval(h)
    cur = h.copy()
    cur_c5 = best_c5
    for scale in [5e-4, 2e-4, 8e-5, 3e-5, 1e-5]:
        while time.time() < deadline:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = cur[i] + d, cur[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            cur[i], cur[j] = ni, nj
            c5n = _eval(cur)
            if c5n < cur_c5 - 1e-18:
                cur_c5 = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best = cur.copy()
            else:
                cur[i], cur[j] = ni - d, nj + d
    return best, best_c5

def _upsample(h, factor):
    n = len(h)
    out = np.zeros(n * factor)
    for i in range(n):
        out[i * factor:(i + 1) * factor] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP at n0
    for alpha in [5000, 50000, 200000]:
        if time.time() - t0 > 40:
            break
        hh = _slsqp(best_h, alpha, 1200, 1e-18)
        c5 = _eval(hh)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    # Phase 2: Upsample and SLSQP
    if time.time() - t0 < budget_s * 0.18:
        n1 = best_n * 2
        h1 = _upsample(best_h, 2)
        h1 = _repair(h1, n1)
        for alpha in [20000, 200000, 1000000]:
            if time.time() - t0 > budget_s * 0.28:
                break
            h1 = _slsqp(h1, alpha, 1200, 1e-18)
            c5 = _eval(h1)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h1.copy()
                best_n = n1

    nc = best_n
    # Phase 3: PGD worst-lag
    dl3 = t0 + budget_s * 0.45
    hh, c5 = _pgd_worst(best_h, nc, dl3, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = hh.copy()

    # Phase 4: Multi-lag PGD
    if time.time() < t0 + budget_s * 0.45:
        dl4 = t0 + budget_s * 0.60
        hh, c5 = _multi_lag_pgd(best_h, nc, dl4, rng, top_m=3)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    # Phase 5: Multi-start diverse
    dl5 = t0 + budget_s * 0.78
    for _r in range(20):
        if time.time() > dl5:
            break
        s = _r % 5
        if s == 0:
            hp = _repair(best_h + 0.008 * rng.randn(nc), nc)
        elif s == 1:
            hp = _repair(best_h + 0.015 * np.sin(2*np.pi*rng.randint(1,nc//3+1)*np.arange(nc)/nc), nc)
        elif s == 2:
            hp = best_h.copy()
            mask = rng.rand(nc) < 0.25
            hp[mask] += 0.015 * rng.randn(int(mask.sum()))
            hp = _repair(hp, nc)
        elif s == 3:
            hp = _repair(1.0 - best_h + 0.005 * rng.randn(nc), nc)
        else:
            hp = _repair(best_h * rng.uniform(0.92, 1.08) + 0.005 * rng.randn(nc), nc)
        sub_dl = min(dl5, time.time() + budget_s * 0.04)
        hh, c5 = _pgd_worst(hp, nc, sub_dl, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    # Phase 6: Pairwise fine search
    dl6 = t0 + budget_s * 0.90
    hh, c5 = _pairwise(best_h, nc, dl6, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = hh.copy()

    # Phase 7: Final SLSQP
    if time.time() < t0 + budget_s * 0.93:
        hh = _slsqp(best_h, 2000000, 2000, 1e-20)
        c5 = _eval(hh)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    # Phase 8: Ultra-fine pairwise
    if time.time() < t0 + budget_s * 0.97:
        hh, c5 = _pairwise(best_h, nc, t0 + budget_s * 0.97, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hh.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
