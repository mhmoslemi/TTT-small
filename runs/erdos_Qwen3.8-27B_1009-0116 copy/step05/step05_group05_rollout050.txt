import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    s = np.clip(alpha * c, -500, 500)
    sm = s.max()
    return (np.log(np.sum(np.exp(s - sm))) + sm) / alpha * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    s = np.clip(alpha * c, -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    wsum = w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] / wsum
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        wk2 = wk * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk2 * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk2 * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk2 * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk2 * h[i2 + sp]
    return g

def _active_grad(h, n_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[::-1][:n_top]
    cvals = c[idx]
    w = np.maximum(cvals - cvals.min(), 0.0)
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(n_top) / n_top
    else:
        w /= ws
    g = np.zeros(n)
    for wi, k in zip(w, idx):
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wi * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wi * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wi * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wi * h[i2 + sp]
    return g * dx

def _pgd(h, alphas, iters_each):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    best_c5 = _eval(h)
    for alpha in alphas:
        fcur = _smooth_obj(h, alpha)
        for _ in range(iters_each):
            g = _smooth_grad(h, alpha)
            step = 0.02
            for _ls in range(15):
                h_new = _proj_to_sum(h - step * g, target)
                fnew = _smooth_obj(h_new, alpha)
                if fnew < fcur - 1e-8:
                    break
                step *= 0.5
            h = h_new
            fcur = fnew
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def _active_descent(h, max_iters, n_top):
    n = len(h)
    target = n * 0.5
    c5c = _eval(h)
    for _ in range(max_iters):
        g = _active_grad(h, n_top)
        gnorm = np.dot(g, g)
        if gnorm < 1e-20:
            break
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
            h_new = _proj_to_sum(h - step * g, target)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _structured_pair(h, max_rounds, n_grid, rng):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_rounds):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dhi - dlo < 1e-10:
            continue
        c0 = np.correlate(h, 1.0 - h, mode='full')
        a = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            sh = (n - 1) - k
            ci = 0.0
            cj = 0.0
            if sh >= 0:
                if sh <= i and i + sh < n:
                    ci += 1.0 - h[i + sh]
                if 0 <= i - sh and i - sh < n - sh:
                    ci -= h[i - sh]
                if sh <= j and j + sh < n:
                    cj += 1.0 - h[j + sh]
                if 0 <= j - sh and j - sh < n - sh:
                    cj -= h[j - sh]
            else:
                sp = -sh
                if sp <= i and i - sp < n - sp:
                    ci += 1.0 - h[i - sp]
                if 0 <= i + sp and i + sp < n - sp:
                    ci -= h[i + sp]
                if sp <= j and j - sp < n - sp:
                    cj += 1.0 - h[j - sp]
                if 0 <= j + sp and j + sp < n - sp:
                    cj -= h[j + sp]
            a[k] = ci - cj
        best_d = 0.0
        best_v = np.inf
        dgrid = np.linspace(dlo, dhi, n_grid)
        for d in dgrid:
            v = np.max(c0 + dx * d * a)
            if v < best_v:
                best_v = v
                best_d = d
        if abs(best_d) > 1e-10:
            h[i] += best_d
            h[j] -= best_d
            h = np.clip(h, 0, 1)
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i] -= best_d
                h[j] += best_d
    return h, c5c

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _cons_eq(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _pairwise_refine(h, n_iter, scale, rng):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _gradient_pairs(h, rng, n_pairs):
    n = len(h)
    g = _active_grad(h, 3)
    pos = np.where(g < -0.01 * np.abs(g).max())[0]
    neg = np.where(g > 0.01 * np.abs(g).max())[0]
    if len(pos) == 0 or len.neg == 0:
        return [(rng.randint(n), rng.randint(n)) for _ in range(n_pairs)]
    pairs = []
    for _ in range(n_pairs):
        i = pos[rng.randint(len(pos))]
        j = neg[rng.randint(len(neg))]
        pairs.append((i, j))
    return pairs
