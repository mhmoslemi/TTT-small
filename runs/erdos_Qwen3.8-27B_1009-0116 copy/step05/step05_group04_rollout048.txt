def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

            for idx in order:
                if diff > 0:

                else:

        for stp in [0.04,0.02,0.01,0.005,0.002,0.001,0.0005,0.0002]:
            h_new = _proj(h - stp*g, n*0.5)
            if _eval(h_new) < best_c - 1e-15:
                h, best_c = h_new, _eval(h_new); best_h = h.copy(); break
    return best_h, best_c

def _upsample(h, f):
    return np.repeat(h, f)

def _fourier_refine(h, n, t_budget, rng, M=15):
    """Optimize in truncated Fourier space to escape local optima."""
    t_arr = np.arange(n)
    a = np.zeros(M); b = np.zeros(M)
    for m in range(M):
        f_ = m+1
        a[m] = 2.0/n*(h*np.cos(2*np.pi*f_*t_arr/n)).sum()
        b[m] = 2.0/n*(h*np.sin(2*np.pi*f_*t_arr/n)).sum()
    best_h, best_c = h.copy(), _eval(h)
    t_end = time.time() + t_budget
    while time.time() < t_end:
        h_f = 0.5 + np.zeros(n)
        for m in range(M):
            h_f += a[m]*np.cos(2*np.pi*(m+1)*t_arr/n) + b[m]*np.sin(2*np.pi*(m+1)*t_arr/n)
        h_f = _repair(h_f, n)
        c5 = _eval(h_f)
        if c5 < best_c: best_c, best_h = c5, h_f.copy()
        m_r = rng.randint(M); c_r = rng.randint(2)
        old = a[m_r] if c_r==0 else b[m_r]
        d = 0.003*rng.randn()
        if c_r==0: a[m_r]+=d
        else: b[m_r]+=d
        h2 = 0.5 + np.zeros(n)
        for m in range(M):
            h2 += a[m]*np.cos(2*np.pi*(m+1)*t_arr/n) + b[m]*np.sin(2*np.pi*(m+1)*t_arr/n)
        h2 = _repair(h2, n)
        if _eval(h2) >= c5:
            if c_r==0: a[m_r]=old
            else: b[m_r]=old
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time()-t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: Gradient descent base res
    if _el() < budget_s-550:
        bh, bc = _descend(best_h, n0, min(budget_s-750-_el(), 120), rng, M=5)
        if bc < best_c5: best_h, best_c5, best_n = bh, bc, n0

    # Phase 2: Aggressive annealing base res
    if _el() < budget_s-400:
        bh, bc = _anneal(best_h, n0, min(budget_s-550-_el(), 130), rng)
        if bc < best_c5: best_h, best_c5 = bh, bc

    # Phase 3: Fourier-space escape
    if _el() < budget_s-300:
        bh, bc = _fourier_refine(best_h, n0, min(budget_s-400-_el(), 80), rng, M=12)
        if bc < best_c5: best_h, best_c5, best_n = bh, bc, n0

    # Phase 4: Water-fill base res
    if _el() < budget_s-220:
        bh, bc = _waterfill(best_h, n0, min(budget_s-280-_el(), 60))
        if bc < best_c5: best_h, best_c5, best_n = bh, bc, n0

    # Phase 5: Multi-upsample refinement
    for upf in [2, 4]:
        n_new = n0*upf
        if _el() > budget_s-100: break
        h_up = _repair(_upsample(best_h, upf), n_new)
        if _el() < budget_s-80:
            bh, bc = _descend(h_up, n_new, min(budget_s-160-_el(), 50), rng, M=7)
            if bc < _eval(h_up): h_up = bh
        if _el() < budget_s-50:
            bh, bc = _anneal(h_up, n_new, min(budget_s-70-_el(), 35), rng)
            if bc < _eval(h_up): h_up = bh
        if _el() < budget_s-25:
            bh, bc = _waterfill(h_up, n_new, min(budget_s-35-_el(), 25))
            if bc < _eval(h_up): h_up = bh
        cf = _eval(_repair(h_up, n_new))
        if cf < best_c5: best_h, best_c5, best_n = _repair(h_up,n_new), cf, n_new

    # Phase 6: Final gradient polish
    if _el() < budget_s-20:
        bh, bc = _descend(best_h, best_n, min(budget_s-30-_el(), 30), rng, M=3)
        if bc < best_c5: best_h, best_c5, best_n = bh, bc, best_n

    # Phase 7: Fine pairwise local search
    h_f = best_h.copy(); nc = len(h_f); c5_f = _eval(h_f)
    t_end_f = time.time() + max(0, min(budget_s-10-_el(), 45))
    while time.time() < t_end_f:
        i = rng.randint(nc)
        j = (i+1+rng.randint(0,3)) % nc
        d = 4e-5*rng.randn()
        ni, nj = h_f[i]+d, h_f[j]-d
        if 0<=ni<=1 and 0<=nj<=1:
            h_f[i], h_f[j] = ni, nj
            c5n = _eval(h_f)
            if c5n < c5_f - 1e-17:
                c5_f = c5n
                if c5_f < best_c5: best_c5, best_h = c5_f, h_f.copy()
            else:
                h_f[i], h_f[j] = h_f[i]-d, h_f[j]+d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
