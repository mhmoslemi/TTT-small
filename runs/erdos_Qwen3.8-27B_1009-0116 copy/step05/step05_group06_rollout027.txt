def _focal_grad(h, alpha, topk):
    """Gradient focusing on top-k correlations"""
    n = len(h)
    c = _corr(h)
    dx = 2.0/n
    # Get top-k indices
    idx = np.argsort(c)[::-1][:topk]
    # softmax weights over top-k
    ck = c[idx]
    w = np.exp(alpha * (ck - ck.max()))
    w /= w.sum()
    
    g = np.zeros(n)
    for pos, k in enumerate(idx):
        if w[pos] < 1e-15:
            continue
        sh = (n-1) - k
        wk = w[pos] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _local_search(h, n_iter, step_scale, rng):
    """Fine local search with pairwise adjustments"""
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = step_scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

    n = len(x)
    c = _corr(x)
    if topk >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -topk)[-topk:]
    ck = c[idx]
    m = ck.max()
    s = np.clip(alpha * (ck - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for pos in range(len(idx)):
        wp = w[pos]
        if wp < 1e-15:
            continue
        k = idx[pos]
        sh = k - (n - 1)  # shift
        wk = wp * dx
        # First term: (1 - x[i+sh]) for valid i
        if sh >= 0:
            # i+sh < n => i < n-sh
            i_valid = np.arange(n - sh)
            g[i_valid] += wk * (1.0 - x[sh:])
        else:
            # i+sh >= 0 => i >= -sh
            sp = -sh
            i_valid = np.arange(sp, n)
            g[i_valid] += wk * (1.0 - x[:n-sp])
        # Second term: -x[i-sh] for valid i
        if sh >= 0:
            # i-sh >= 0 => i >= sh
            if sh < n:
                i_valid = np.arange(sh, n)
                g[i_valid] -= wk * x[:n-sh]
        else:
            # i-sh < n => i < n+sh = n-sp
            sp = -sh
            i_valid = np.arange(n - sp)
            g[i_valid] -= wk * x[sp:]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _focal_obj(x, alpha, topk):
    c = _corr(x)
    if topk >= len(c):
        ck = c
    else:
        ck = np.partition(c, -topk)[-topk:]
    m = ck.max()
    s = np.clip(alpha * (ck - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _focal_grad(x, alpha, topk):
    n = len(x)
    c = _corr(x)
    if topk >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -topk)[-topk:]
    ck = c[idx]
    m = ck.max()
    s = np.clip(alpha * (ck - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for pos in range(len(idx)):
        wp = w[pos]
        if wp < 1e-15:
            continue
        k = idx[pos]
        sh = k - (n - 1)
        wk = wp * dx
        if sh >= 0:
            if sh < n:
                cnt = n - sh
                g[:cnt] += wk * (1.0 - x[sh:])
                g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _run_lbfgs(h, alpha, topk, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_focal_obj, h, args=(alpha, topk),
                       jac=_focal_grad, method='L-BFGS-B',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _run_slsqp(h, alpha, topk, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_focal_obj, h, args=(alpha, topk),
                       jac=_focal_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj_grad_descent(h, alpha, topk, nsteps, step0, mu):
    n = len(h)
    target = n * 0.5
    f = _focal_obj(h, alpha, topk)
    v = np.zeros(n)
    step = step0
    for _ in range(nsteps):
        g = _focal_grad(h, alpha, topk)
        v = mu * v - step * g
        lam = 1.0
        improved = False
        for _ in range(25):
            h_try = _proj_to_sum(h - lam * v, target)
            fn = _focal_obj(h_try, alpha, topk)
            if fn < f - 1e-12:
                h, f = h_try, fn
                improved = True
                break
            lam *= 0.5
        if not improved:
            break
        step = min(step * 1.02, step0 * 4.0)
    return _repair(h, n)

def _local_search(h, n_iter, step_scale, rng):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = step_scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _upsample(h, factor):
    return np.repeat(h, factor)
