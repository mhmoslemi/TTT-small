import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp(h, dx, alpha, maxiter=500):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _worst_pair_search(h, dx, rng, t_end, n_top=2, step_scale=0.15):
    n = len(h)
    best = _c5(h, dx)

        c = _corr(h)
        top = np.argsort(c)[::-1][:n_top]

            if s == 0:
                continue
            m = n - s

            else:
                contrib = h[:m] * (1.0 - h[s:])
            n_pairs = min(40, m)

                if time.time() > t_end:
                    break

                    i1, i2 = wi + s, wi
                else:
                    i1, i2 = wi, wi + s
                diff = h[i1] - h[i2]

                if d > 0.05:
                    d = 0.05

                    h[i1] -= d; h[i2] += d
                else:
                    h[i1] += d; h[i2] -= d

                    hr = _repair(h, n / 2.0)

                    else:

                        else:
                            h[i1] -= d; h[i2] += d

def _random_pair_search(h, dx, rng, t_end, scale=0.015):

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.05 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(5, n // 6), replace=False)
        hp[idx] += 0.1 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.04 * np.sin(2 * np.pi * rng.uniform(1, 6) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(3, n // 10), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)
