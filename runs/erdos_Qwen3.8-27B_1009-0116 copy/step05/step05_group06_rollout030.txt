    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        c = _corr(h)
        cmax = c.max()
        improved = False
        for i in range(n):
            j = (i + 1) % n
            if i == j:  # n=1 edge case
                continue
            # a_i: effect at position i
            a_i = np.zeros(2 * n - 1)
            for L in range(-(n - 1), n):
                k = (n - 1) + L
                val = 0.0
                if 0 <= i + L < n:
                    val += 1.0 - h[i + L]
                if 0 <= i - L < n:
                    val -= h[i - L]
                a_i[k] = val
            # a_j: effect at position j
            a_j = np.zeros(2 * n - 1)
            for L in range(-(n - 1), n):
                k = (n - 1) + L
                val = 0.0
                if 0 <= j + L < n:
                    val += 1.0 - h[j + L]
                if 0 <= j - L < n:
                    val -= h[j - L]
                a_j[k] = val
            # Net effect: δ at i, -δ at j
            # c_new = c + δ*(a_i - a_j)
            net = a_i - a_j
            old_i, old_j = h[i], h[j]
            best_d = 0.0
            best_mx = cmax
            for d in [-0.1, -0.05, -0.02, -0.005, 0.005, 0.02, 0.05, 0.1]:
                ni = old_i + d
                nj = old_j - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                cn = c + d * net
                mn = cn.max()
                if mn < best_mx - 1e-12:
                    best_mx = mn
                    best_d = d
            if abs(best_d) > 1e-12:
                h[i] += best_d
                h[j] -= best_d
                c += best_d * net
                cmax = c.max()
                improved = True

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

                best_c5, best_h, best_n = c5, h2.copy(), n1
        if time.time() < dl - 100:
            h2 = _topk_subgrad(h2, 300, t0, dl - 80, topk=30)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h2.copy(), n1

    # Phase E: Upsample to 4x (n1*2 = n0*4)
    n2 = n0 * 4
    if time.time() < dl - 250:
        h4 = _repair(_upsample(best_h, n2 // best_n), n2)
        h4 = _momentum(h4, 200000, 800, 0.0005, 0.90, t0, dl - 180)
        c5 = _eval(h4)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h4.copy(), n2
        if time.time() < dl - 100:
            h4 = _coord_desc(h4, 150, t0, dl - 60)
            c5 = _eval(h4)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h4.copy(), n2
        if time.time() < dl - 40:
            h4 = _topk_subgrad(h4, 200, t0, dl - 20, topk=40)
            c5 = _eval(h4)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h4.copy(), n2

    # Phase F: Multi-start at best_n
    nc = best_n
    if time.time() < dl - 80:
        for _r in range(30):
            if time.time() > dl - 60:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                bs = rng.randint(5, max(6, nc // 6))
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 3:
                t_arr = np.arange(nc)
                hp = best_h.copy()
                for _f in range(3):
                    f = rng.randint(1, nc // 3 + 1)
                    hp += 0.006 * rng.randn() * np.sin(2 * np.pi * f * t_arr / nc)
                hp = _repair(hp, nc)
            else:
                mask = rng.rand(nc) < 0.2
                hp = best_h.copy()
                hp[mask] = rng.rand(int(mask.sum()))
                hp = _repair(hp, nc)
            alpha_pick = [20000, 80000, 300000][rng.randint(3)]
            hnew = _slsqp(hp, alpha_pick, 600, t0, dl - 55)
            c5 = _eval(hnew)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hnew.copy(), nc
                # Quick coord polish
                if time.time() < dl - 50:
                    hpol = _coord_desc(hnew, 30, t0, dl - 45)
                    c5p = _eval(hpol)
                    if c5p < best_c5:
                        best_c5, best_h = c5p, hpol.copy()

    # Phase G: Final SLSQP polish
    if time.time() < dl - 15:
        hfin = _slsqp(best_h, 500000, 1500, t0, dl - 8)
        c5 = _eval(hfin)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hfin.copy(), len(hfin)

    # Phase H: Fine coordinate descent
    if time.time() < dl - 3:
        hfin = _coord_desc(best_h, 80, t0, dl - 1)
        c5 = _eval(hfin)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hfin.copy(), len(hfin)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
