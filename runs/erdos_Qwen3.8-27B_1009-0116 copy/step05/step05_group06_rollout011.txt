    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        wk = w[ii] * dx
        k = int(idx[ii])
        off = k - n + 1
        i_lo = max(0, -off)
        i_hi = min(n - 1, n - 1 - off)
        if i_lo > i_hi:
            continue
        p_lo = max(i_lo, -off)
        p_hi = min(i_hi, n - 1 - off)
        if p_lo <= p_hi:
            g[p_lo:p_hi + 1] += wk * (1.0 - h[p_lo + off:p_hi + off + 1])
        q_lo = max(0, i_lo + off)
        q_hi = min(n - 1, i_hi + off)
        if q_lo <= q_hi:
            g[q_lo:q_hi + 1] -= wk * h[q_lo - off:q_hi - off + 1]
    return g

def _slsqp(h, alpha, topk, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_obj, h, jac=_grad, args=(alpha, topk),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mom_grad(h, alpha, topk, nsteps, step0, mu):
    n = len(h)
    f = _obj(h, alpha, topk)
    v = np.zeros(n)
    step = step0
    for _ in range(nsteps):
        g = _grad(h, alpha, topk)
        v = mu * v - step * g
        lam = 1.0
        for _ in range(20):
            h2 = _proj_sum(h - lam * v, n * 0.5)
            f2 = _obj(h2, alpha, topk)
            if f2 < f:
                h, f = h2, f2
                break
            lam *= 0.5
        else:
            h = _proj_sum(h - lam * v, n * 0.5)
            f = _obj(h, alpha, topk)
        step = min(step * 1.02, step0 * 5.0)
    return _repair(h, n)

def _local_search(h, n_iters, scale, rng):
    n = len(h)
    c5c = _c5(h)
    for _ in range(n_iters):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if not (0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0):
            continue
        h[i], h[j] = ni, nj
        c5n = _c5(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    _repair(h, n)
    return h, _c5(h)

def _anneal_step(h, c5c, scale, temp, rng):
    n = len(h)
    i, j = rng.randint(n), rng.randint(n)
    if i == j:
        return c5c
    d = scale * rng.randn()
    ni, nj = h[i] + d, h[j] - d
    if not (0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0):
        return c5c
    h[i], h[j] = ni, nj
    c5n = _c5(h)
    if c5n < c5c:
        return c5n
    if rng.rand() < np.exp(-(c5n - c5c) / max(temp, 1e-10)):
        return c5n
    h[i], h[j] = h[i] - d, h[j] + d
    return c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _c5(h), n0

    # Phase 1: SLSQP with top-k smooth max, increasing alpha
    configs1 = [(5000, 10), (20000, 8), (60000, 6), (150000, 5), (400000, 4)]
    for alpha, topk in configs1:
        if _el() > 180:
            break
        h2 = _slsqp(best_h, alpha, topk, 4000)
        c5 = _c5(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    # Phase 2: Momentum projected gradient
    if _el() < budget_s - 500:
        h2 = _mom_grad(best_h, 30000, 5, 500, 0.004, 0.88)
        c5 = _c5(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    # Phase 3: More SLSQP with very high alpha
    configs3 = [(500000, 3), (1000000, 3), (2000000, 2)]
    for alpha, topk in configs3:
        if _el() > budget_s - 400:
            break
        h2 = _slsqp(best_h, alpha, topk, 5000)
        c5 = _c5(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    # Phase 4: Multi-start with perturbations
    if _el() < budget_s - 350:
        for r in range(20):
            if _el() > budget_s - 350:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = _repair(best_h + 0.008 * rng.randn(n0), n0)
            elif strat == 1:
                hp = _repair(best_h + 0.025 * rng.randn(n0), n0)
            elif strat == 2:
                hp = best_h.copy()
                bs = rng.randint(4, max(5, n0 // 7))
                st = rng.randint(0, n0 - bs)
                hp[st:st + bs] += 0.035 * rng.randn(bs)
                hp = _repair(hp, n0)
            elif strat == 3:
                sh = rng.randint(1, max(2, n0 // 14))
                hp = _repair(np.roll(best_h, sh), n0)
            else:
                hp = best_h.copy()
                freqs = rng.randint(1, n0 // 3 + 1, size=3)
                ta = np.arange(n0)
                for f in freqs:
                    hp += 0.006 * rng.randn() * np.sin(2 * np.pi * f * ta / n0)
                hp = _repair(hp, n0)
            h2 = _slsqp(hp, 80000, 4, 2000)
            c5 = _c5(h2)
            if c5 < best_c5:
                best_c5, best_h = c5, h2

    # Phase 5: Simulated annealing (short)
    if _el() < budget_s - 200:
        h = best_h.copy()
        c5c = _c5(h)
        temp = 0.001
        for scale in [0.002, 0.0008, 0.0002]:
            if _el() > budget_s - 200:
                break
            for _it in range(5000):
                if _el() > budget_s - 200:
                    break
                c5c = _anneal_step(h, c5c, scale, temp, rng)
            temp *= 0.35
        h = _repair(h, n0)
        c5 = _c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 6: Long local search with decreasing scales
    if _el() < budget_s - 80:
        for scale in [0.002, 0.0008, 0.0003, 0.0001, 3e-5, 1e-5]:
            if _el() > budget_s - 80:
                break
            h2, c5 = _local_search(best_h.copy(), 10000, scale, rng)
            if c5 < best_c5:
                best_c5, best_h = c5, h2

    # Phase 7: Final SLSQP polish
    if _el() < budget_s - 25:
        for alpha, topk in [(500000, 3), (1500000, 2)]:
            if _el() > budget_s - 25:
                break
            h2 = _slsqp(best_h, alpha, topk, 3000)
            c5 = _c5(h2)
            if c5 < best_c5:
                best_c5, best_h = c5, h2

    # Phase 8: Ultra-fine final search
    if _el() < budget_s - 3:
        h2, c5 = _local_search(best_h.copy(), 25000, 5e-6, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h)
    return (best_h, best_c5, best_n)
