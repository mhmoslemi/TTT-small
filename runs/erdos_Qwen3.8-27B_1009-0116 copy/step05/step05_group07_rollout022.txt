def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, 'full').max() * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    wsum = w.sum()
    w /= wsum
    g = np.zeros(n)
    nn1 = n - 1
    for k in range(2 * n - 1):

        sh = k - nn1

def _upsample(h, n_new):

    x_new = np.linspace(0, 2, n_new)
    return _repair(np.interp(x_new, x_old, h), n_new)

import numpy as np
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, 'full').max() * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    wsum = w.sum()
    w /= wsum
    g = np.zeros(n)
    nn1 = n - 1
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = k - nn1
        wk = w[k] * dx
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[n - sp:] -= wk * h[:n - sp]
    return g

def _fista(h0, alpha, n_iter, L_mult=2.0):
    n = len(h0)
    target = n * 0.5
    L = alpha * L_mult + 10.0
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_x = h0.copy()
    best_f = _eval(h0)
    skip_eval = max(1, n_iter // 80)
    for k in range(n_iter):
        if time.time() - _t0 > _budget - 30:
            break
        gy = _smooth_grad(y, alpha)
        y_new = y - (1.0 / L) * gy
        x_new = _proj_sum(y_new, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mom = (t - 1.0) / t_new
        y = x_new + mom * (x_new - x)
        np.clip(y, -0.5, 1.5, out=y)
        x = x_new
        t = t_new
        if k % skip_eval == 0 or k == n_iter - 1:
            f = _eval(x)
            if f < best_f:
                best_f = f
                best_x = x.copy()
    f = _eval(x)
    if f < best_f:
        best_x = x.copy()
        best_f = f
    return best_x, best_f

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return _repair(np.interp(x_new, x_old, h), n_new)

def _coord_descent(h, n_iter, topm=5):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    h = h.copy()
    f = _eval(h)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 30:
            break
        c = np.correlate(h, 1.0 - h, 'full') * dx
        top_idx = np.argsort(c)[::-1][:topm]
        improved = False
        for k in top_idx:
            sh = k - (n - 1)
            g_k = np.zeros(n)
            if sh == 0:
                g_k[:] = 1.0 - 2.0 * h
            elif sh > 0:
                g_k[:n - sh] = 1.0 - h[sh:]
                g_k[sh:] = -h[:n - sh]
            else:
                sp = -sh
                g_k[:n - sp] = 1.0 - h[sp:]
                g_k[n - sp:] = -h[:n - sp]
            for step in [0.01, 0.005, 0.002, 0.0008]:
                h_new = _proj_sum(h - step * g_k * dx, target)
                f_new = _eval(h_new)
                if f_new < f - 1e-14:
                    h = h_new; f = f_new; improved = True; break
        if not improved:
            break
    return h, f

def _subgrad_step(h, n_iter, lr0=0.004):

    target = n * 0.5

        else:
            lr *= 0.75
    return h, f
