    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

    if beta > 0:
        ssum = h.sum()
        val += beta * (ssum - len(h) * 0.5) ** 2
    return val

    """Gradient of smooth max + sum penalty"""
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    wm = s - s.max()
    w = np.exp(wm)
    w /= w.sum()
    dx = 2.0 / n

        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            # positive shift
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    if beta > 0:
        g += 2 * beta * (h.sum() - n * 0.5) * np.ones(n)
    return g

def _lbfgsb(h, alpha, beta, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_max, h, jac=_smooth_grad, args=(alpha, beta),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-12})
        return res.x
    except:
        return h.copy()

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-10:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(1.0 - np.abs(2.0*h - 1.0))  # prefer middle values
            for i in idx:
                if diff > 0:
                    a = min(diff, 1.0 - h[i])
                    h[i] += a
                    diff -= a
                else:
                    a = min(-diff, h[i])
                    h[i] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
    return h

def _smooth_obj(x, alpha, beta):

    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for t in range(1, n_iters + 1):

        M = c.max()

        threshold = M - 0.02 * M
        active = np.where(c >= threshold)[0]

        g = np.zeros(n)

            else:

def _coord_descent(h, n_rounds, rng):

    for _ in range(n_rounds):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    T = n / 2.0
    # If current sum is close enough, just return
    if abs(h.sum() - T) < 1e-12:
        return h
    # Water-filling: find lambda such that sum(clip(h - lambda, 0, 1)) = T
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > T: lo = mid
        else: hi = mid
    lam = (lo + hi) / 2
    h = np.clip(h - lam, 0, 1)
    # Fix tiny residual
    d = T - h.sum()

        order = np.argsort(np.abs(h - 0.5))
        for i in order:
            if abs(d) < 1e-14: break
            if d > 0: a = min(d, 1-h[i]); h[i] += a; d -= a
