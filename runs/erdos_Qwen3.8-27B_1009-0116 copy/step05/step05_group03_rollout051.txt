                d -= a

                h[i] -= a
                d += a

    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp(h, dx, alpha, maxiter=400):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _slsqp_adaptive(h, dx, t_end):
    """Adaptive alpha SLSQP: geometric progression from smooth to sharp."""
    n = len(h)
    best_h, best_c = h.copy(), _c5(h, dx)

    """Gradient-directed pair swap search on worst lags."""
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    step_sizes = [0.012, 0.008, 0.005, 0.003, 0.002, 0.001, 0.0005]
    si = 0
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:4]
        improved = False
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            sc = step_sizes[si % len(step_sizes)]
            # Gradient direction: reduce h[a] where 1-h[a+s] large, increase h[a+s]
            # Equivalently: move mass from high h[i] at positions where shift partner is low

                else:
