import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj_and_grad(h, alpha, penalty):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = np.sum(np.exp(s))
    f = m + np.log(sm) / alpha
    if penalty > 0:
        diff = h.sum() - n * 0.5
        f += 0.5 * penalty * diff * diff
    
    w = np.exp(s - s.max())
    w /= w.sum()
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s_shift = k - (n - 1)
        # d(c[k])/d(h[i]) = (1-h[i+s_shift]) * [valid] - h[i-s_shift] * [valid]
        if s_shift >= 0:
            # i ranges from 0 to n-1-s_shift
            i_range = np.arange(n - s_shift)
            g[i_range] += w[k] * (1.0 - h[i_range + s_shift])
            # h[i - s_shift] valid when i >= s_shift
            if s_shift > 0:
                i_range2 = np.arange(s_shift, n)
                g[i_range2] -= w[k] * h[i_range2 - s_shift]
        else:
            sp = -s_shift  # positive
            # i ranges from sp to n-1
            i_range = np.arange(sp, n)
            g[i_range] += w[k] * (1.0 - h[i_range - sp])
            # h[i + sp] valid when i < n - sp
            i_range2 = np.arange(n - sp)
            g[i_range2] -= w[k] * h[i_range2 + sp]
    
    if penalty > 0:
        diff = h.sum() - n * 0.5
        g += penalty * diff * np.ones(n)
    
    return f, g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = min(-diff, h[idx])
                a = -a
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-12:
                break
    return h

        f, _ = _smooth_obj_and_grad(x, alpha, penalty)
        return f

        _, g = _smooth_obj_and_grad(x, alpha, penalty)
        return g
