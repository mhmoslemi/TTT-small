def _corr_fft(h):
    n = len(h)
    g = 1.0 - h
    # Full correlation via FFT

            best_c5 = c5b
            best_h = hb.copy()
        # Binary local search
        hb2, c5b2 = _binary_local_search(hb, nc, 50000, rng)
        if c5b2 < best_c5:
            best_c5 = c5b2
            best_h = hb2.copy()
        # Reoptimize from binary point
        if _el() < budget_s - 60:
            h_r = _lbfgs_run(best_h, 10000, 200.0, 300, True, 30)
            c5r = _eval(h_r)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = h_r.copy()

    # Phase 6: Multiscale pairwise refinement
    if _el() < budget_s - 30:
        nc = len(best_h)
        h = best_h.copy()
        for scale in [0.005, 0.001, 0.0003, 0.00008]:
            if _el() > budget_s - 30:
                break
            h, bc5, bcopy = _pairwise_refine(h, nc, 5000, scale, rng)
            if bc5 < best_c5:
                best_c5 = bc5
                best_h = bcopy.copy()
                h = bcopy.copy()

    # Phase 7: Final L-BFGS ultra-high alpha
    if _el() < budget_s - 15:
        h_new = _lbfgs_run(best_h, 300000, 5000.0, 200, True, 10)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            best_n = len(h_new)

    # Phase 8: Ultra-fine local search
    if _el() < budget_s - 1:
        nc = len(best_h)
        h = best_h.copy()
        c5c = _eval(h)
        ntries = 30000
        for _it in range(ntries):
            if _el() > budget_s - 1:
                break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = 1.5e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                        best_n = nc
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
