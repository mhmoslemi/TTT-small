        i2 = np.arange(n-sp)

def _binary_local_search(h, n, max_iters=100, rng=None):
    """Exact binary swap local search"""
    if rng is None:
        rng = np.random.RandomState(0)
    best = h.copy()
    best_c = _eval(best)
    ones = set(np.where(best > 0.5)[0])
    zeros = set(np.where(best <= 0.5)[0])
    
    for it in range(max_iters):
        improved = False
        c = np.correlate(best, 1.0-best, mode='full')
        cmax = c.max()
        
        # Try all single swaps
        best_delta = 0
        best_a, best_b = None, None
        for a in sorted(ones):
            # Precompute removal effect
            dr = np.zeros(2*n-1)
            for s in range(-(n-1), n):
                # s is the shift, m = s + n-1
                m = s + n - 1
                if 0 <= a+s < n:
                    dr[m] -= (1.0 - best[a+s])
                if 0 <= a-s < n:
                    dr[m] += best[a-s]
            
            for b in sorted(zeros):
                # Compute insertion effect
                di = np.zeros(2*n-1)
                for s in range(-(n-1), n):
                    m = s + n - 1
                    if 0 <= b+s < n:
                        di[m] += (1.0 - best[b+s])
                    if 0 <= b-s < n:
                        di[m] -= best[b-s]
                
                new_c = c + dr + di
                new_max = new_c.max()
                if new_max < cmax - 1e-15:
                    if new_max < cmax - best_delta:
                        best_delta = cmax - new_max
                        best_a, best_b = a, b
                        improved = True
        
        if best_a is not None:
            best[best_a] = 0.0
            best[best_b] = 1.0
            ones.discard(best_a)
            ones.add(best_b)
            zeros.discard(best_b)
            zeros.add(best_a)
            best_c = _eval(best)
    
    return best, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    h *= target / max(s, 1e-15)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in order:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return np.array([x.sum() - n * 0.5])
    def _cj(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_step(h, n, alpha, step):
    g = _smooth_grad(h, alpha)
    h_new = _proj_sum(h - step * g, n * 0.5)
    return h_new

def _binary_steepest(c, h, n):
    cmax = c.max()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None
    s_arr = np.arange(-(n-1), n)
    best_nm = cmax
    best_a = -1
    best_b = -1
    for a in ones:
        dr = np.zeros(2*n-1)
        v1 = (a + s_arr >= 0) & (a + s_arr < n)
        dr[s_arr[v1] + n - 1] -= (1.0 - h[a + s_arr[v1]])
        v2 = (a - s_arr >= 0) & (a - s_arr < n)
        dr[s_arr[v2] + n - 1] += h[a - s_arr[v2]]
        for b in zeros:
            di = np.zeros(2*n-1)
            v3 = (b + s_arr >= 0) & (b + s_arr < n)
            di[s_arr[v3] + n - 1] += (1.0 - h[b + s_arr[v3]])
            v4 = (b - s_arr >= 0) & (b - s_arr < n)
            di[s_arr[v4] + n - 1] -= h[b - s_arr[v4]]
            nm = (c + dr + di).max()
            if nm < best_nm - 1e-14:
                best_nm = nm
                best_a = a
                best_b = b
    if best_a < 0:
        return None
    return (best_a, best_b, best_nm)

def _binary_ls(h, n, max_iters, rng):
    hb = ((h > 0.5) * 1.0).copy()
    # Ensure exactly n/2 ones
    n_ones = int(hb.sum())
    if n_ones > n//2:
        idx = np.where(hb > 0.5)[0]
        for i in idx[n//2:]:
            hb[i] = 0.0
    elif n_ones < n//2:
        idx = np.where(hb <= 0.5)[0]
        for i in idx[:n//2 - n_ones]:
            hb[i] = 1.0
    hb = _repair(hb, n)
    c = np.correlate(hb, 1.0 - hb, mode='full')
    cmax = c.max()
    for it in range(max_iters):
        res = _binary_steepest(c, hb, n)
        if res is None:
            break
        a, b, nm = res
        hb[a] = 0.0
        hb[b] = 1.0
        c = np.correlate(hb, 1.0 - hb, mode='full')
        cmax = c.max()
    cmax = np.max(c) * 2.0 / n
    return hb, float(cmax)

def _unbin(h, n, alpha, steps):
