    # We need coef[k_idx][i] for the linearized expression

    const = np.zeros(2*n-1)
    
    for k_idx in range(2*n-1):
        sh = (n-1) - k_idx  # shift
        if sh >= 0:
            # pairs (i, i+sh) for i=0..n-1-sh
            for i in range(n - sh):
                const[k_idx] += h_t[i] * (1 - h_t[i+sh])
                coef[k_idx, i] += (1 - h_t[i+sh])
                coef[k_idx, i+sh] -= h_t[i]
        else:

            for i in range(n - s):

                coef[k_idx, i+s] += (1 - h_t[i])
                coef[k_idx, i] -= h_t[i+s]

    # s.t. coef[k] @ h + const[k] - tau ≤ 0 for all k
    #      0 ≤ h ≤ 1
    #      sum h = n/2

    # n+1 variables
    A = np.zeros((2*n-1, n+1))
    b = np.zeros(2*n-1)
    for k in range(2*n-1):

        A[k, n] = -1.0
        b[k] = -const[k]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lp_step(h_t, n):
    coef = np.zeros((2 * n - 1, n))
    const = np.zeros(2 * n - 1)
    for k_idx in range(2 * n - 1):
        sh = (n - 1) - k_idx
        if sh >= 0:
            m = n - sh
            coef[k_idx, :m] += (1.0 - h_t[sh:n])
            coef[k_idx, sh:] -= h_t[:m]
            const[k_idx] = np.dot(h_t[:m], 1.0 - h_t[sh:n])
        else:
            s = -sh
            m = n - s
            coef[k_idx, s:] += (1.0 - h_t[:m])
            coef[k_idx, :m] -= h_t[s:]
            const[k_idx] = np.dot(h_t[s:], 1.0 - h_t[:m])
    nv = n + 1
    A = np.empty((2 * n - 1, nv))
    A[:, :n] = coef
    A[:, n] = -1.0
    c_obj = np.zeros(nv)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    except Exception:
        pass
