def _best_2coord_delta(c, coefs_j, coefs_jp, h_j, h_jp):
    """Find optimal δ = h[j]-h[j'] adjustment.
    new_c = c + δ*(coefs_j - coefs_jp)
    Constraints: 0 ≤ h_j+δ ≤ 1, 0 ≤ h_jp-δ ≤ 1
    """
    d_coefs = coefs_j - coefs_jp
    # δ ∈ [max(-h_j, -(1-h_jp)), min(1-h_j, h_jp)]
    lo = max(-h_j, h_jp - 1.0)
    hi = min(1.0 - h_j, h_jp)
    if lo >= hi:
        return 0.0
    # f(δ) = max_k (c[k] + δ*d_coefs[k]) is convex in δ
    # Minimize on [lo, hi]
    # Subgradient at δ: {d_coefs[k] : c[k] + δ*d_coefs[k] = f(δ)}
    # At optimum, 0 ∈ subgradient, i.e., min d_coefs[active] ≤ 0 ≤ max d_coifs[active]
    # Use bisection on subgradient
    for _ in range(40):
        mid = (lo + hi) / 2
        vals = c + mid * d_coefs
        mv = vals.max()
        active = vals >= mv - 1e-12
        sg_lo = d_coefs[active].min()
        sg_hi = d_coefs[active].max()
        if sg_lo < 0 and sg_hi > 0:
            return mid  # optimal
        elif sg_lo >= 0:
            hi = mid  # f is increasing, go left
        else:
            lo = mid  # f is decreasing, go right
    return (lo + hi) / 2

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0., 1.).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-14:
        for i in (np.argsort(1. - h) if d > 0 else np.argsort(h)):
            a = min(abs(d), (1. - h[i]) if d > 0 else h[i])
            if a < 1e-15:
                break
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else -a
            if abs(d) < 1e-14:
                break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1. - h, mode='full')) * 2. / len(h))

def _sm_obj(h, al):
    c = np.correlate(h, 1. - h, mode='full')
    m = c.max()
    s = np.clip(al * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / al

def _sm_grad(h, al):
    n = len(h)
    c = np.correlate(h, 1. - h, mode='full')
    m = c.max()
    s = np.clip(al * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2. / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1. - h[sh:])
            if sh:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1. - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _slsqp(h, al, mx=2000):
    n = len(h)
    try:
        r = minimize(_sm_obj, h, jac=_sm_grad, args=(al,),
                     method='SLSQP', bounds=[(0., 1.)] * n,
                     constraints=[{'type': 'eq', 'A': np.ones((1, n)),
                                   'b': np.array([n * 0.5])}],
                     options={'maxiter': mx, 'ftol': 1e-17})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _all_coefs(h, n):
    co = np.zeros((n, 2 * n - 1))
    co[:, n-1] = 1. - 2. * h
    for sh in range(1, n):
        co[:n-sh, n-1-sh] += 1. - h[sh:]
        co[sh:, n-1-sh] -= h[:n-sh]
        co[sh:, n-1+sh] += 1. - h[:n-sh]
        co[:n-sh, n-1+sh] -= h[sh:]
    return co

def _bisect(c, d, lo, hi):
    for _ in range(45):
        m = (lo + hi) * 0.5
        v = c + m * d
        mv = v.max()
        a = v >= mv - 1e-14
        if d[a].min() < 0 < d[a].max():
            return m
        if d[a].min() >= 0:
            hi = m
        else:
            lo = m
    return (lo + hi) * 0.5

def _pw_pass(h, n, c, co, rng, ntry):
    c5m = c.max()
    for _ in range(ntry):

        if nm < c5m - 1e-14:
            h[j] += dl; h[jp] -= dl
            c += dl * d; c5m = nm
    return c5m

def _pw_pass(h, n, c, co, rng, ntry):
    c5m = c.max()
    moves = 0
    for _ in range(ntry):
        j = rng.randint(n); jp = rng.randint(n)
        if j == jp: continue
        d = co[j] - co[jp]
        lo = max(-h[j], h[jp]-1.)
        hi = min(1.-h[j], h[jp])
        if lo >= hi-1e-12: continue
        dl = _bisect(c, d, lo, hi)
        nm = (c+dl*d).max()
        if nm < c5m-1e-14:
            h[j] += dl; h[jp] -= dl
            c += dl*d; c5m = nm; moves += 1
    return c5m, moves

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0., 1.).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-14:
        for i in (np.argsort(1. - h) if d > 0 else np.argsort(h)):
            a = min(abs(d), (1. - h[i]) if d > 0 else h[i])
            if a < 1e-15:
                break
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else -a
            if abs(d) < 1e-14:
                break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1. - h, mode='full')) * 2. / len(h))

    c = np.correlate(h, 1. - h, mode='full')
    m = c.max()
    s = np.clip(al * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / al

    n = len(h)
    c = np.correlate(h, 1. - h, mode='full')
    m = c.max()
    s = np.clip(al * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2. / n
    g = np.zeros(n)
    for k in range(2 * n - 1):

            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1. - h[sh:])
            if sh:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1. - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

                     method='SLSQP', bounds=[(0., 1.)] * n,
                     constraints=[{'type': 'eq', 'A': np.ones((1, n)),
                                   'b': np.array([n * 0.5])}],
                     options={'maxiter': mx, 'ftol': 1e-17})

    except Exception:

        co[sh:, n-1+sh] += 1. - h[:n-sh]
        co[:n-sh, n-1+sh] -= h[sh:]
