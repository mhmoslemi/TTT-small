    dx = 2.0 / n
    c = _corr(h) * dx
    # Compute gradient of c[k] w.r.t. h at current point
    # dc[k]/dh[j] = (1-h[j-sh]) - h[j+sh] (with validity)
    # Build the linearized constraint: c[k] + dc[k]^T (h-h0) <= t
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    for k in range(2*n-1):

        grad = np.zeros(n)

            if 0 <= j+sh < n:

    h = np.clip(h, 0.0, 1.0).copy()

                    diff -= a

            j1 = np.arange(sh, n)

            if sh > 0:

                grads[k, j2] -= h[j2 + sh]

                # Already handled by first term, second term is at j2=j
                j2 = np.arange(0, n)

            grads[k, j1] += (1.0 - h[j1 - sp])
            j2 = np.arange(0, n-sp)
            grads[k, j2] -= h[j2 + sp]

def _lp_solve(h, n, cuts):
    """Solve LP with accumulated cuts. Returns new h."""
    dx = 2.0 / n
    c = _corr(h) * dx
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]

    for (h0, c0, g) in cuts:
        constraints.append(c0 + g @ (h_var - h0) <= t_var)
    
    # Add current linearization
    for k in range(2*n-1):

        g = np.zeros(n)
        if sh >= 0:
            if sh == 0:
                g[:] = 1.0 - 2.0*h
            else:
                g[sh:] = 1.0 - h[:-sh]
                g[:-sh] -= h[sh:]
        else:
            sp = -sh
            g[sp:] = 1.0 - h[:-sp]
            g[:n-sp] -= h[sp:]
        constraints.append(c[k] + dx * (g @ (h_var - h)) <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.GLOP)
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            return h
    if h_var.value is not None:
        return _repair(np.array(h_var.value), n)
    return h

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:

        else:
            hi = mid

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _smooth_obj(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)

            g[sh:] += wk * (1.0 - h[:n-sh])

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _smooth_obj(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh == 0:
            g[:] += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lp_refine(h, n, n_iters):
    c5c = _eval(h)
    dx = 2.0 / n
    stagnant = 0
    for it in range(n_iters):
        c = _corr(h) * dx
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
        for k in range(2 * n - 1):
            sh = (n - 1) - k
            g = np.zeros(n)
            if sh == 0:
                g[:] = 1.0 - 2.0 * h
            elif sh > 0:
                g[sh:] = 1.0 - h[:n - sh]
                g[:n - sh] -= h[sh:]
            else:
                sp = -sh
                g[:n - sp] = 1.0 - h[sp:]
                g[sp:] -= h[:n - sp]
            constraints.append(c[k] + dx * (g @ (h_var - h)) <= t_var)
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = _repair(np.array(h_var.value), n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-13:
            h = h_new
            c5c = c5n
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 5:
                break
    return h, c5c

def _active_descent(h, max_iters, alpha_w):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for it in range(max_iters):
        c = _corr(h) * dx
        m = c.max()
        s = np.clip(alpha_w * (c - m), -300, 0)
        w = np.exp(s)
        wsum = w.sum()
        if wsum < 1e-15:
            idx = np.argsort(c)[::-1][:5]
            w = np.zeros(2 * n - 1)
            w[idx] = 1.0 / 5
        else:
            w /= wsum
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh == 0:
                g[:] += wk * (1.0 - 2.0 * h)
            elif sh > 0:
                g[sh:] += wk * (1.0 - h[:n - sh])
                g[:n - sh] -= wk * h[sh:]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n - sp])
                g[:n - sp] -= wk * h[sp:]
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _pairwise_refine(h, n_iter, scale, rng):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _upsample(h, factor):
    return np.repeat(h, factor)
