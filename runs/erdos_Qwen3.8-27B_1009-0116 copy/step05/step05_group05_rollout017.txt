import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _coord_descent(h, max_sweeps):
    """Exact coordinate descent with sum constraint repair."""
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    
    for sweep in range(max_sweeps):
        improved = False
        perm = np.random.permutation(n)
        for p in perm:
            h0 = h[p]
            if h0 <= 0.001 or h0 >= 0.999:
                continue  # skip boundary points
            c = _corr(h)
            # Compute gradient of each c_k w.r.t. h[p]
            g = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                sh = (n - 1) - k
                if sh >= 0:
                    if p <= n - 1 - sh:
                        g[k] += (1.0 - h[p + sh])
                    if sh <= p:
                        g[k] -= h[p - sh]
                else:
                    sp = -sh
                    if sp <= p:
                        g[k] += (1.0 - h[p - sp])
                    if p + sp < n:
                        g[k] -= h[p + sp]
            # c_k(x) ≈ c[k] + (x - h0) * g[k] for x near h0
            # Find x in [0,1] minimizing max_k (c[k] + (x-h0)*g[k])
            # This is minimize max_k (a_k * x + b_k) where a_k = g[k], b_k = c[k] - h0*g[k]
            # Upper envelope of lines, find minimum over [0,1]
            
            # Simple approach: ternary search on [0,1] since max of linear is convex
            lo, hi2 = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi2 - lo) / 3
                m2 = hi2 - (hi2 - lo) / 3
                f1 = np.max(c + (m1 - h0) * g)
                f2 = np.max(c + (m2 - h0) * g)
                if f1 < f2:
                    hi2 = m2
                else:
                    lo = m1
            x_star = (lo + hi2) / 2
            if x_star < h0:
                h[p] = x_star
            else:
                h[p] = x_star
            # Repair sum
            h = _repair(h, n)
            c5n = _eval(h)
            if c5n < c5c - 1e-14:
                c5c = c5n
                improved = True
            else:
                h = _repair(h, n)  # revert? No, repair might have changed things
                # Actually let's just keep it since it's at most equally good
        
        if not improved and sweep > 3:
            break
    
    return h, c5c
