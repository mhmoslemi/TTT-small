    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _eval_c5(h):
    n = len(h)
    return float(np.max(_compute_corr(h)) * 2.0 / n)

def _grad_lag(h, s):
    """Gradient of f_s with respect to h, where s is the shift."""
    n = len(h)
    g = np.zeros(n)
    dx = 2.0 / n
    # Contribution from h[i] where i+s is valid: (1-h[i+s])
    # Contribution from h[i+s] where i is valid: -h[i]
    # For each j in [0,n-1]:
    #   j appears as h[i] when i=j, s valid => (1-h[j+s])
    #   j appears as h[i+s] when i=j-s, valid => -h[j-s]
    # First term: for j where j+s in [0,n-1], add (1-h[j+s])
    # Second term: for j where j-s in [0,n-1], subtract h[j-s]
    if s >= 0:
        # j+s valid when j <= n-1-s
        idx1 = np.arange(0, n - s)
        g[idx1] += (1.0 - h[idx1 + s])
        if s > 0:
            # j-s valid when j >= s
            idx2 = np.arange(s, n)
            g[idx2] -= h[idx2 - s]
    else:
        sp = -s
        # j+s = j-sp valid when j >= sp
        idx1 = np.arange(sp, n)
        g[idx1] += (1.0 - h[idx1 - sp])
        # j-s = j+sp valid when j <= n-1-sp
        idx2 = np.arange(0, n - sp)
        g[idx2] -= h[idx2 + sp]
    return g * dx

        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        gx = np.zeros(n)
        dx = 2.0 / n
        for k in range(2*n - 1):
            if w[k] < 1e-15:
                continue
            sh = (n-1) - k
            wk = w[k] * dx
            if sh >= 0:
                gx[:n-sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    gx[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                gx[sp:] += wk * (1.0 - x[:n-sp])
                gx[:n-sp] -= wk * x[sp:]
        return gx
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        from scipy.optimize import minimize
        res = minimize(_obj, h, jac=_grad, args=(),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t_start

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Subgradient descent at parent resolution
    if _el() < budget_s - 350:
        limit = min(250.0, budget_s - 350)
        h_new, c5_new = _subgrad(h0, 80000, 0.04, 3, rng, limit)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Multi-start with perturbations
    if _el() < budget_s - 250:
        for trial in range(25):
            if _el() > budget_s - 250:
                break
            pert = rng.choice([0.003, 0.008, 0.02, 0.05, 0.1])
            mode = trial % 3
            if mode == 0:
                h_kick = _repair(best_h + pert * rng.randn(best_n), best_n)
            elif mode == 1:
                h_kick = _repair(_synth(best_n, rng, rng.randint(5)), best_n)
            else:
                mask = rng.rand(best_n) < 0.15
                h_kick = best_h.copy()
                h_kick[mask] = rng.rand(mask.sum()).clip(0,1)
                h_kick = _repair(h_kick, best_n)
            limit = min(8.0, budget_s - 250 - _el())
            h_new, c5_new = _subgrad(h_kick, 8000, 0.02, 3, rng, limit)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = len(h_new)

    # Phase 3: SLSQP polish with smooth objective
    if _el() < budget_s - 150:
        for alpha in [100000, 500000, 2000000]:
            if _el() > budget_s - 150:
                break
            h_s = _slsqp_polish(best_h, alpha, 3000)
            c5s = _eval(h_s)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = h_s

    # Phase 4: Other resolutions
    if _el() < budget_s - 100:
        for n_try in [200, 360, 440]:
            if _el() > budget_s - 100:
                break
            if n_try > best_n:
                r = best_n
                t_f = np.linspace(0, 2, n_try, endpoint=False)
                t_c = np.linspace(0, 2, r, endpoint=False)
                h_try = _repair(np.interp(t_f, t_c, best_h), n_try)
            else:
                ratio = best_n // n_try
                if ratio < 1:
                    continue
                h_try = np.zeros(n_try)
                for i in range(n_try):
                    h_try[i] = best_h[i*ratio:(i+1)*ratio].mean()
                h_try = _repair(h_try, n_try)
            limit = min(25.0, budget_s - 100 - _el())
            h_new, c5_new = _subgrad(h_try, 15000, 0.03, 3, rng, limit)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_try

    # Phase 5: Binary restart
    if _el() < budget_s - 60:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(h_bin.sum())
        target_ones = best_n // 2
        if n_ones > target_ones:
            one_idx = np.where(h_bin == 1)[0]
            order = one_idx[np.argsort(best_h[one_idx])]
            h_bin[order[:n_ones - target_ones]] = 0
        elif n_ones < target_ones:
            zero_idx = np.where(h_bin == 0)[0]
            order = zero_idx[np.argsort(-best_h[zero_idx])]
            h_bin[order[:target_ones - n_ones]] = 1
        limit = min(25.0, budget_s - 60 - _el())
        h_bb, c5_bb = _binary_ls(h_bin, best_n, 30000, rng, limit)
        for alpha in [0.0, 0.3, 0.6, 1.0]:
            h_soft = _repair(alpha * h_bb + (1-alpha) * best_h, best_n)
            c5s = _eval(h_soft)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = h_soft
        if c5_bb < best_c5:
            best_c5 = c5_bb
            best_h = h_bb.copy()

    # Phase 6: Fine subgradient polish
    if _el() < budget_s - 25:
        limit = budget_s - 25 - _el()
        h_new, c5_new = _subgrad(best_h, 50000, 0.008, 5, rng, limit)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 7: Ultra-fine pairwise
    if _el() < budget_s - 3:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        remaining = budget_s - 3 - _el()
        n_tries = int(remaining / 0.0015)
        for _it in range(n_tries):
            if _el() > budget_s - 3:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 4e-5 * rng.randn()
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i] = ni
            h[j] = nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i] = h[i] - d
                h[j] = h[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
