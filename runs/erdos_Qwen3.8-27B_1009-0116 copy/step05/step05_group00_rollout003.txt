                break
        if not moved: break
    return h, c5c

def _pair_swap(h, n_iter, rng):
    nc = len(h)
    h = h.copy()
    c5c = _eval(h)
    nb = 0
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j: continue
        if h[i] < 0.5 and h[j] < 0.5: continue
        if h[i] > 0.5 and h[j] > 0.5: continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        h2 = _repair(h, nc)
        c5n = _eval(h2)
        if c5n < c5c - 1e-14:
            h, c5c = h2, c5n
            nb += 1
        else:
            h = h2.copy()
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            h = _repair(h, nc)
    return h, c5c

def _make_binary(n, pattern, rng):
    h = np.zeros(n)
    h[:n//2] = 1.0
    if pattern == 0:
        h = h[::2] if n % 2 == 0 else h
        h = np.zeros(n)
        h[::2] = 1.0
        if n % 2 == 1: h[-1] = 0
        cnt = int(h.sum())
        if cnt > n//2:
            h[rng.choice(np.where(h==1)[0], cnt - n//2, replace=False)] = 0
        elif cnt < n//2:
            h[rng.choice(np.where(h==0)[0], n//2 - cnt, replace=False)] = 1
    elif pattern == 1:
        per = 3
        for i in range(n):
            h[i] = 1.0 if (i % per) < 2 else 0.0
        h = _repair(h, n)
        h = (h > 0.5).astype(float)
        h = _repair(h, n)
    elif pattern == 2:
        per = 5
        for i in range(n):
            h[i] = 1.0 if (i % per) < 3 else 0.0
        h = _repair(h, n)
        h = (h > 0.5).astype(float)
        h = _repair(h, n)
    elif pattern == 3:
        g = int(np.sqrt(n))
        for i in range(n):
            r, c_ = divmod(i, g)
            h[i] = 1.0 if (r + c_) % 2 == 0 else 0.0
        h = _repair(h, n)
        h = (h > 0.5).astype(float)
        h = _repair(h, n)
    else:
        idx = np.arange(n)
        h[idx[::2]] = 1.0
        h = _repair(h, n)
        h = (h > 0.5).astype(float)
        h = _repair(h, n)
    return h

def _anneal_binary(h, t0, t1, n_steps, rng):
    nc = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, nc)
    c5c = _eval(h)
    best = c5c
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for step in range(n_steps):
        t = t0 * (t1/t0) ** (step / max(n_steps-1, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        h2 = _repair(h, nc)
        c5n = _eval(h2)
        if c5n < c5c or rng.rand() < np.exp(-(c5n - c5c)/max(t, 1e-6)):
            c5c = c5n
            ones = np.where(h2 > 0.5)[0]
            zeros = np.where(h2 < 0.5)[0]
            h = h2
            if c5c < best: best = c5c
        else:
            h[i], h[j] = 1.0, 0.0
            h = _repair(h, nc)
    return h, best

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def elapsed(): return time.time() - t0

    # Phase 1: Quick SLSQP polish (30s)
    for alpha in [5000, 50000, 200000]:
        if elapsed() > 30: break
        h_new = _slsqp(best_h, alpha, 2000, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Diverse binary starting points + optimize (180s)
    candidates = [(best_h.copy(), best_c5)]
    for pat in range(5):
        if elapsed() > 180: break
        hb = _make_binary(n0, pat, rng)
        for alpha in [10000, 100000]:
            if elapsed() > 40 * (pat + 1) + 30: break
            hb = _slsqp(hb, alpha, 1500, 1e-18)
            c5 = _eval(hb)
            if c5 < best_c5:
                best_c5, best_h = c5, hb.copy()
        if _eval(hb) < best_c5:
            best_c5, best_h = _eval(hb), hb.copy()
        candidates.append((hb.copy(), _eval(hb)))

    # Also try a few different n values
    for n_try in [250, 300, 350, 400]:
        if elapsed() > 180: break
        h_r = _repair(_resample(best_h, n_try), n_try)
        h_r = _slsqp(h_r, 50000, 1000, 1e-18)
        c5 = _eval(h_r)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 3: Intensive SLSQP on best (250s)
    if elapsed() < budget_s - 250:
        for alpha in [50000, 200000, 500000, 2000000, 5000000]:
            if elapsed() > budget_s - 250: break
            h_new = _slsqp(best_h, alpha, 5000, 1e-20)
            c5 = _eval(h_new)
            if c5 < best_c5 - 1e-12:
                best_c5, best_h = c5, h_new.copy()
            # Also try complement
            h_comp = _repair(1.0 - best_h, len(best_h))
            h_comp = _slsqp(h_comp, alpha, 3000, 1e-19)
            c5c = _eval(h_comp)
            if c5c < best_c5 - 1e-12:
                best_c5, best_h = c5c, h_comp.copy()

    # Phase 4: Worst-shift focused descent (150s)
    if elapsed() < budget_s - 150:
        h4 = best_h.copy()
        c5_4 = best_c5
        for _rep in range(50):
            if elapsed() > budget_s - 150: break
            h4, c5_4 = _worst_grad_step(h4, 20)
            if c5_4 < best_c5 - 1e-14:
                best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Simulated annealing on binary pattern (200s)
    if elapsed() < budget_s - 200:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = _repair(h_bin, len(h_bin))
        t_remaining = max(budget_s - 200 - elapsed(), 5)
        n_steps = max(int(t_remaining * 800), 2000)
        h_a, c5_a = _anneal_binary(h_bin, 0.5, 0.001, n_steps, rng)
        # Polish the annealed result
        h_a = _slsqp(h_a, 100000, 2000, 1e-19)
        c5_a = _eval(h_a)
        if c5_a < best_c5 - 1e-14:
            best_c5, best_h = c5_a, h_a.copy()

    # Phase 6: Pair swap descent (150s)
    if elapsed() < budget_s - 150:
        h6 = best_h.copy()
        c5_6 = best_c5
        n_iter = 8000
        t_rem = max(budget_s - 150 - elapsed(), 5)
        n_iter = max(int(t_rem * 1500), 3000)
        h6, c5_6 = _pair_swap(h6, n_iter, rng)
        if c5_6 < best_c5 - 1e-14:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Another round of worst-shift + SLSQP (80s)
    if elapsed() < budget_s - 30:
        h7 = _worst_grad_step(best_h, 30)[0]
        h7 = _slsqp(h7, 5000000, 3000, 1e-21)
        c5_7 = _eval(h7)
        if c5_7 < best_c5 - 1e-14:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: Final micro-adjustment coordinate descent (20s)
    if elapsed() < budget_s - 3:
        h8 = best_h.copy()
        c5c = _eval(h8)
        nc = len(h8)
        for scale in [0.002, 0.0005, 0.0001]:
            if elapsed() > budget_s - 3: break
            for _it in range(3000):
                if elapsed() > budget_s - 3: break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j: continue
                d = scale * rng.randn()
                ni, nj = h8[i] + d, h8[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
                h8[i], h8[j] = ni, nj
                c5n = _eval(h8)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                else:
                    h8[i], h8[j] = ni - d, nj + d
        if c5c < best_c5 - 1e-14:
            best_c5, best_h = c5c, h8.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
