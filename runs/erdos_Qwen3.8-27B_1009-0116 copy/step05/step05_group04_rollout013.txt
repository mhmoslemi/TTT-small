import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lag_grad(h, shift, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            i1 = np.arange(n - shift)
            g[i1] += (1.0 - h[i1 + shift])
        if shift > 0:
            i2 = np.arange(shift, n)
            g[i2] -= h[i2 - shift]
    else:
        sp = -shift
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _adam_descent(h, n, steps, lr, top_k=3):
    target = n * 0.5
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    for _ in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for kw in top_idx:
            sw = (n - 1) - kw
            g += _lag_grad(h, sw, n)
        g /= top_k
        t += 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        h = _proj_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
    return h

def _slsqp_smooth(h, n, alpha, maxiter=2000):
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        s -= s.max()
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _con(x):
        return x.sum() - n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pairwise_refine(h, n, n_tries, rng, scale_arr=None):
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    n_done = 0
    for _ in range(n_tries):
        if scale_arr is not None and n_done < len(scale_arr):
            sc = scale_arr[n_done]
        else:
            sc = 1e-5
        n_done += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = sc * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c_new = _eval(h)
        if c_new < c_cur - 1e-18:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _sa_block(h, n, n_iter, rng, t0=0.004, t1=1e-7):
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    for i in range(n_iter):

        blk = rng.randint(1, max(2, n // 40))
        st = rng.randint(0, n - blk)
        delta = rng.randn(blk) * (t * 3.0)
        h_new = h.copy()
        h_new[st:st + blk] += delta
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-10)):
            h = h_new
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _lag_sweep(h, n, rounds=100, lr=0.02):
    """Alternate descent on each of the top lags individually"""
    target = n * 0.5
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    for r in range(rounds):
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-3:]
        for kw in top_idx:
            sw = (n - 1) - kw
            g = _lag_grad(h, sw, n)
            t += 1
            m = b1 * m + (1.0 - b1) * g
            v = b2 * v + (1.0 - b2) * g * g
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)
            h = _proj_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
    return h
