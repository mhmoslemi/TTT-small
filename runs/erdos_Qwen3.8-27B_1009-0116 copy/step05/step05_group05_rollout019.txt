import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n; sh_vals = np.arange(-(n-1), n); g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15: continue
        sh = sh_vals[k]; wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp]); g[:n-sp] -= wk * h[sp:]
    return g

def _cf(x): return x.sum() - len(x) * 0.5
def _cj(x): return np.ones(len(x))

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _cd_loop(h, n, sh_vals, rng, deadline, n_tern):
    """Exact 1D coordinate descent with ternary search."""
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        for i in rng.permutation(n):
            vp = (i + sh_vals >= 0) & (i + sh_vals < n)
            vm = (i - sh_vals >= 0) & (i - sh_vals < n)
            slope = np.zeros(2*n-1)
            slope[vp] += 1.0 - h[i + sh_vals[vp]]
            slope[vm] -= h[i - sh_vals[vm]]
            a, b = -h[i], 1.0 - h[i]
            if a >= b - 1e-15: continue
            for _ in range(n_tern):
                m1 = a + (b-a)/3.0; m2 = b - (b-a)/3.0
                if np.max(c + m1*slope) < np.max(c + m2*slope): b = m2
                else: a = m1
            d = (a+b)*0.5
            h[i] = np.clip(h[i] + d, 0.0, 1.0)
            c += d * slope
        h = _repair(h, n)
    return h

def _pair_cd(h, n, sh_vals, rng, deadline):
    """Coordinate descent optimizing pairs simultaneously (2D alternating)."""
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        for _ in range(n):
            i = rng.randint(n); j = rng.randint(n)
            if i == j: continue
            vi_p = (i + sh_vals >= 0) & (i + sh_vals < n)
            vi_m = (i - sh_vals >= 0) & (i - sh_vals < n)
            si = np.zeros(2*n-1)
            si[vi_p] += 1.0 - h[i + sh_vals[vi_p]]; si[vi_m] -= h[i - sh_vals[vi_m]]
            vj_p = (j + sh_vals >= 0) & (j + sh_vals < n)
            vj_m = (j - sh_vals >= 0) & (j - sh_vals < n)
            sj = np.zeros(2*n-1)
            sj[vj_p] += 1.0 - h[j + sh_vals[vj_p]]; sj[vj_m] -= h[j - sh_vals[vj_m]]
            d_i = 0.0; d_j = 0.0
            for _it in range(20):
                a, b = -h[i], 1.0 - h[i]
                if a >= b - 1e-14: continue
                for _ in range(18):
                    m1 = a+(b-a)/3.0; m2 = b-(b-a)/3.0
                    if np.max(c + m1*si + d_j*sj) < np.max(c + m2*si + d_j*sj): b = m2
                    else: a = m1
                d_i = (a+b)*0.5
                a, b = -h[j], 1.0 - h[j]

                    else: a = m1
                d_j = (a+b)*0.5
            if abs(d_i) > 1e-14 or abs(d_j) > 1e-14:
                h[i] = np.clip(h[i] + d_i, 0.0, 1.0)
                h[j] = np.clip(h[j] + d_j, 0.0, 1.0)
                c += d_i * si + d_j * sj
        h = _repair(h, n)
    return h

def _active_descent(h, max_iters, n_top):
    n = len(h); dx = 2.0/n; sh_vals = np.arange(-(n-1), n)
    c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]; cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0); wsum = weights.sum()
        if wsum < 1e-15: weights = np.ones(n_top)/n_top
        else: weights /= wsum
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = sh_vals[k]; wk = w * dx
            if sh >= 0:
                g[:n-sh] += wk*(1.0-h[sh:])
                if sh > 0: g[sh:] -= wk*h[:n-sh]
            else:
                sp = -sh; g[sp:] += wk*(1.0-h[:n-sp]); g[:n-sp] -= wk*h[sp:]
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
            h_new = _proj_to_sum(h - step*g, n*0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new; c5c = c5n; improved = True; break
        if not improved: break
    return h, c5c
