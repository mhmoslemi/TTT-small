    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        # Proportional scaling
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:

            else:

                    h[idx] -= a

def _active_descent(h, max_iters, n_top, mom=0.3):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    v = np.zeros(n)
    for it in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full')
        cdx = c * dx
        idx = np.argsort(cdx)[::-1][:n_top]
        cvals = cdx[idx]
        cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum
        g = np.zeros(n)
        for wk, k in zip(weights * dx, idx):
            sh = (n - 1) - k
            if sh >= 0:
                # ∂c/∂h[j] for j in [sh, n-1]: (1-h[j-sh])
                js = np.arange(sh, n)
                g[js] += wk * (1.0 - h[js - sh])
                # ∂c/∂h[j] for j in [0, n-1-sh]: -h[j+sh]
                if sh > 0:
                    jt = np.arange(n - sh)
                    g[jt] -= wk * h[jt + sh]
            else:

                # ∂c/∂h[j] for j in [sp, n-1]: (1-h[j-sp])
                js = np.arange(sp, n)
                g[js] += wk * (1.0 - h[js - sp])

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sg(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    es = e.sum()
    w = e / es
    obj = m + np.log(es) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            js = np.arange(sh, n)
            g[js] += wk * (1.0 - h[js - sh])
            if sh > 0:
                jt = np.arange(n - sh); g[jt] -= wk * h[jt + sh]
        else:
            sp = -sh
            js = np.arange(sp, n)
            g[js] += wk * (1.0 - h[js - sp])
            jt = np.arange(n - sp); g[jt] -= wk * h[jt + sp]
    return obj, g

def _sg_obj(h, alpha):
    return _sg(h, alpha)[0]

def _sg_grad(h, alpha):
    return _sg(h, alpha)[1]

def _cons_eq(x):
    return x.sum() - len(x) * 0.5

def _cons_eq_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    try:
        res = minimize(_sg_obj, h, jac=_sg_grad, args=(alpha,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lbfgs_pen(h, alpha, lam, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _lp_obj(x):
        return _sg(x, alpha)[0] + lam * (x.sum() - n * 0.5) ** 2
    def _lp_grad(x):
        g = _sg(x, alpha)[1] + 2.0 * lam * (x.sum() - n * 0.5)
        return g + 2.0 * lam * (x.sum() - n * 0.5) * np.ones(n)
    try:
        res = minimize(_lp_obj, h, jac=_lp_grad, method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent(h, max_iters, n_top, mom):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    v = np.zeros(n)
    for it in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]; cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        ws = weights.sum()
        if ws < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= ws
        g = np.zeros(n)
        for wk, k in zip(weights * dx, idx):
            sh = (n - 1) - k
            if sh >= 0:
                js = np.arange(sh, n); g[js] += wk * (1.0 - h[js - sh])
                if sh > 0:
                    jt = np.arange(n - sh); g[jt] -= wk * h[jt + sh]
            else:
                sp = -sh
                js = np.arange(sp, n); g[js] += wk * (1.0 - h[js - sp])
                jt = np.arange(n - sp); g[jt] -= wk * h[jt + sp]
        v = mom * v + g
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
            h_new = _proj(h - step * v, n * 0.5)

                    h = h_new; c5c = c5n; improved = True; break

            else:
                sp = -sh
                if sp <= i < n:
                    ci += (1.0 - h[i - sp])
                if 0 <= i <= n - 1 - sp:
                    ci -= h[i + sp]
            cj = 0.0
            if sh >= 0:
                if sh <= j < n:
                    cj += (1.0 - h[j - sh])
                if 0 <= j <= n - 1 - sh:
                    cj -= h[j + sh]
            else:
                sp = -sh
                if sp <= j < n:
                    cj += (1.0 - h[j - sp])
                if 0 <= j <= n - 1 - sp:
                    cj -= h[j + sp]
            a[k] = ci - cj
        best_d = 0.0; best_v = np.inf
        dgrid = np.linspace(dlo, dhi, 17)
        for d in dgrid:

            if v < best_v:

            c5n = _eval(h)
            if c5n < c5c - 1e-16:

            else:

    n = len(h)
    return np.repeat(h, factor)
