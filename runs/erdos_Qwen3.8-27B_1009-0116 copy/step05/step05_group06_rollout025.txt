from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_arr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    h *= target / max(s, 1e-15)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-(h))  # most room first (closest to 1)
            # Actually we want to add to those with most room: sort by (1-h) descending
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)  # remove from smallest
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _to_bin(h, n):
    """Convert to binary with exactly n//2 ones."""
    target = n // 2
    h = (h >= 0.5).astype(float)
    cnt = int(h.sum())
    if cnt == target:
        return h
    if cnt < target:
        # Need more ones: promote highest-valued zeros
        idx = np.where(h == 0)[0]
        idx = idx[np.argsort(h[idx])[::-1]]
        for k in range(target - cnt):
            h[idx[k]] = 1.0
    else:
        idx = np.where(h == 1)[0]
        idx = idx[np.argsort(h[idx])]
        for k in range(cnt - target):
            h[idx[k]] = 0.0
    return h

    """Simulated annealing on binary sequence via swaps."""
    n = len(h)
    h = _to_bin(h, n).copy()
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    no_impr = 0
    for it in range(n_iter):
        if time.time() - t0 > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5n = _eval(h)
        delta = c5n - c5
        # Acceptance: deterministic below, or random for diversity
        if delta < 0:
            c5 = c5n
            no_impr = 0
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            # Reject (or accept with small probability for exploration)
            if rng.rand() < 0.05:  # 5% random acceptance for exploration
                c5 = c5n
            else:
                h[i], h[j] = h[j], h[i]
            no_impr += 1
        if no_impr > 5000:
            # Kick: random perturbation
            for _ in range(5):
                a, b = rng.randint(n), rng.randint(n)
                if a != b and h[a] != h[b]:
                    h[a], h[b] = h[b], h[a]
            c5 = _eval(h)
            no_impr = 0
    return best_h, best_c5

    """Project h to [0,1]^n with sum = target."""
    n = len(h)
    lo, hi = -2.0, 2.0

        if s > target:
            lo = mid
        else:

    c = _corr_arr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_scaled = c * dx
    m = c_scaled.max()
    s = np.clip(alpha * (c_scaled - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:

        sh = (n - 1) - k

        if sh >= 0:

            g[i1] += wk * (1.0 - h[i1 + sh])

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

    """Continuous optimization using projected gradient + line search."""
    n = len(h_init)
    h = _repair(h_init, n).copy()
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)

        v = mu * v - step * g

            lam *= 0.5

            step *= 0.5
            if step < 1e-10:
                break
        else:
            step = min(step * 1.02, 0.01)
    return _repair(h, n)

    return np.repeat(h, factor)

    if cnt < target:
        idx = np.where(h == 0)[0]
        idx = idx[np.argsort(h[idx])[::-1]]
        for k in range(target - cnt):
            h[idx[k]] = 1.0
    else:
        idx = np.where(h == 1)[0]
        idx = idx[np.argsort(h[idx])]
        for k in range(cnt - target):
            h[idx[k]] = 0.0
    return h

def _proj_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj(h, alpha):
    c = _corr_arr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = (c * dx).max()
    s = np.clip(alpha * (c * dx - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _sa_bin(h, rng, t0, t_end):
    n = len(h)
    h = _to_bin(h, n).copy()
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    no_impr = 0
    kick_count = 0
    while time.time() - t0 < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5n = _eval(h)
        if c5n < c5:
            c5 = c5n
            no_impr = 0
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            if rng.rand() < 0.03:
                c5 = c5n
            else:
                h[i], h[j] = h[j], h[i]
            no_impr += 1
        if no_impr > 3000:
            for _ in range(4 + kick_count):
                a, b = rng.randint(n), rng.randint(n)
                if a != b and h[a] != h[b]:
                    h[a], h[b] = h[b], h[a]
            c5 = _eval(h)
            no_impr = 0
            kick_count += 1
            if kick_count > 12:
                break
    return best_h, best_c5

def _cont_opt(h_init, alpha, t0, t_end):
    n = len(h_init)
    h = _repair(h_init, n).copy()
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    step = 0.003
    mu = 0.82
    while time.time() - t0 < t_end:
        g = _smooth_grad(h, alpha)
        v = mu * v - step * g
        lam = 1.0
        improved = False
        for _ in range(18):
            h_try = _proj_sum(h - lam * v, n * 0.5)
            fn = _smooth_obj(h_try, alpha)
            if fn < f:
                h, f = h_try, fn
                improved = True
                break
            lam *= 0.45
        if not improved:
            step *= 0.5
            if step < 1e-11:
                break
        else:
            step = min(step * 1.03, 0.015)
    return _repair(h, n)

def _targeted_descent(h, t0, t_end):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    while time.time() - t0 < t_end:
        c = _corr_arr(h)
        k_peak = int(np.argmax(c))
        sh = (n - 1) - k_peak
        g = np.zeros(n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
        g -= g.mean()
        step = 0.008
        for _ in range(3):
            h_new = _proj_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c:
                h, c5c = h_new, c5n
                break
            step *= 0.5
        if step < 1e-10:
            break
    return _repair(h, n)

def _upsample(h, factor):
    return np.repeat(h, factor)

def _init_structured(n, rng):
    choices = []
    for f in [3, 5, 7, 11, 13]:
        t = np.arange(n)
        h = 0.5 + 0.35 * np.sin(2.0 * np.pi * f * t / n) + 0.15 * np.sin(2.0 * np.pi * (f * 2 + 1) * t / n)
        choices.append(_repair(h, n))
    h_rand = rng.rand(n)
    choices.append(_repair(h_rand, n))
    h_block = np.zeros(n)
    for _ in range(n // 2):
        pos = rng.randint(n)
        h_block[pos] = 1.0
        if h_block.sum() >= n / 2:
            break
    choices.append(_to_bin(h_block, n))
    return choices

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h0.copy(), _eval(h0), n0

    def _el():
        return time.time() - t0

    def _upd(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 1: Binary SA at base resolution
    h_bin = _to_bin(h0, n0)
    h_b, c5_b = _sa_bin(h_bin, rng, t0, t0 + 250)
    _upd(h_b, c5_b, n0)

    # Phase 2: Continuous polish
    if _el() < budget_s - 400:
        h_c = _cont_opt(h_b, 20000, t0, t0 + _el() + 180)
        c5_c = _eval(h_c)
        _upd(h_c, c5_c, n0)

    # Phase 3: Targeted descent on peak
    if _el() < budget_s - 350:
        h_t = _targeted_descent(best_h, t0, t0 + _el() + 120)
        c5_t = _eval(h_t)
        _upd(h_t, c5_t, best_n)

    # Phase 4: Iterate binary-continuous cycle
    for cycle in range(3):
        if _el() > budget_s - 300:
            break
        h_bin2 = _to_bin(best_h, best_n)
        h_b2, c5_b2 = _sa_bin(h_bin2, rng, t0, t0 + _el() + 100)
        _upd(h_b2, c5_b2, best_n)
        if _el() < budget_s - 250:
            h_c2 = _cont_opt(h_b2, 50000, t0, t0 + _el() + 80)
            c5_c2 = _eval(h_c2)
            _upd(h_c2, c5_c2, best_n)

    # Phase 5: Upsample to 2x resolution
    n1 = n0 * 2
    if _el() < budget_s - 200:
        h_hi = _repair(_upsample(best_h, 2), n1)
        h_b3, c5_b3 = _sa_bin(_to_bin(h_hi, n1), rng, t0, t0 + _el() + 120)
        _upd(h_b3, c5_b3, n1)
        if _el() < budget_s - 120:
            h_c3 = _cont_opt(h_b3, 80000, t0, t0 + _el() + 100)
            c5_c3 = _eval(h_c3)
            _upd(h_c3, c5_c3, n1)

    # Phase 6: Structured initializations + SA (diversification)
    if _el() < budget_s - 180:
        for init in _init_structured(n0, rng)[:4]:
            if _el() > budget_s - 150:
                break
            hb, cb = _sa_bin(_to_bin(init, n0), rng, t0, t0 + _el() + 50)
            _upd(hb, cb, n0)
            if _el() < budget_s - 130:
                hc = _cont_opt(hb, 40000, t0, t0 + _el() + 40)
                _upd(hc, _eval(hc), n0)

    # Phase 7: Final targeted + continuous polish
    if _el() < budget_s - 40:
        h_t2 = _targeted_descent(best_h, t0, t0 + _el() + 40)
        c5_t2 = _eval(h_t2)
        _upd(h_t2, c5_t2, best_n)
    if _el() < budget_s - 15:
        h_final = _cont_opt(best_h, 200000, t0, t0 + _el() + 30)
        c5_f = _eval(h_f := h_final)
        _upd(h_final, c5_f, len(h_final))

    # Phase 8: Ultra-fine pairwise at final resolution
    if _el() < budget_s - 4:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _ in range(8000):
            if _el() > budget_s - 3:
                break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = 5e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_h, best_c5 = h.copy(), c5n
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
