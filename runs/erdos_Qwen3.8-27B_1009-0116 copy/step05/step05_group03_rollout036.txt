        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _to_binary(h, target):
    b = (h >= 0.5).astype(np.float64)
    diff = int(round(target)) - int(b.sum())
    if diff > 0:
        z = np.where(b == 0)[0]
        b[z[np.argsort(-h[z])[:diff]]] = 1.0
    elif diff < 0:
        o = np.where(b == 1)[0]
        b[o[np.argsort(h[o])[:(-diff)]]] = 0.0
    return b

def _swap_search(h, dx, rng, t_end, ntries=300):
    n = len(h)
    best_c = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        cmax = float(c.max())
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 5 or len(zeros) < 5:
            break
        worst_idx = np.argsort(c)[::-1][:5]
        improved = False
        for _ in range(ntries):
            mode = rng.random()
            if mode < 0.45:
                ki = worst_idx[rng.randint(len(worst_idx))]
                s = int(abs(ki - (n - 1)))
                if 0 < s < n:
                    m = n - s
                    c1 = np.where(h[:m] == 1.0)[0]
                    c0_in = np.where(h[s:n] == 1.0)[0]
                    if len(c1) > 0:
                        i = c1[rng.randint(len(c1))]
                    else:
                        i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            elif mode < 0.7:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                # Dual swap
                li = rng.randint(len(ones) - 1)
                lj = rng.randint(len(zeros) - 1)
                i1, i2 = ones[li], ones[(li + 1)]
                j1, j2 = zeros[lj], zeros[(lj + 1)]
                if i1 == i2 or j1 == j2:
                    i1, i2 = ones[rng.randint(len(ones))], ones[rng.randint(len(ones))]
                    j1, j2 = zeros[rng.randint(len(zeros))], zeros[rng.randint(len(zeros))]
                if len(set([i1, i2, j1, j2])) < 4:
                    continue
                h[i1], h[j1] = h[j1], h[i1]
                h[i2], h[j2] = h[j2], h[i2]
                nm = float(np.max(_corr(h)))
                if nm * dx < best_c:
                    best_c = nm * dx
                    improved = True
                else:
                    h[j1], h[i1] = h[i1], h[j1]
                    h[j2], h[i2] = h[i2], h[j2]
                continue
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            nm = float(np.max(_corr(h)))
            if nm * dx < best_c:
                best_c = nm * dx
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            k = rng.randint(2, 6)
            for _ in range(k):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if i != j:
                    h[i], h[j] = h[j], h[i]
    return h, best_c

def _slsqp_smooth(h, dx, alpha, target, maxiter=300):
    n = len(h)
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = alpha * (c - m)
        s = np.clip(s, -500, 500)
        amax = s.max()
        return m + float(np.log(np.sum(np.exp(s - amax)))) / alpha + amax / alpha
    def eq(x):
        return x.sum() - target
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(r.x, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _subgrad_step(h, dx, target, lr):
    n = len(h)
    c = _corr(h)
    top = np.argsort(c)[::-1][:6]
    w = np.exp(80.0 * (c[top] - c[top].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top):
        s = abs(int(ki - (n - 1)))
        if s == 0:
            continue
        m = n - s
        g = np.zeros(n)
        g[s:n] += 1.0 - h[:m]
        g[:m] -= h[s:n]
        grad += wi * g
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return h
    h_new = _repair(np.clip(h - lr * grad / gn, 0, 1), target)
    return h_new

def _refine_continuous(h, dx, rng, t_end, target):
    n = len(h)
    best_h = h.copy()
    best_c = _c5(h, dx)
    lr = 0.008
    for _ in range(50000):
        if time.time() > t_end:
            break
        h_new = _subgrad_step(best_h, dx, target, lr)
        c = _c5(h_new, dx)
        if c < best_c:
            best_h = h_new
            best_c = c
            lr = min(lr * 1.02, 0.04)
        else:
            lr *= 0.92
            if lr < 1e-7:
                break
    h2, c2 = _slsqp_smooth(best_h, dx, 5000, target, 250)
    if c2 < best_c:
        best_h, best_c = h2, c2
    h2, c2 = _slsqp_smooth(best_h, dx, 20000, target, 200)
    if c2 < best_c:
        best_h, best_c = h2, c2
    return best_h, best_c

def _perturb_binary(h, rng, k):
    hp = h.copy()
    ones = np.where(hp == 1.0)[0]
    zeros = np.where(hp == 0.0)[0]
    for _ in range(k):
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i != j:
            hp[i], hp[j] = hp[j], hp[i]
    return hp
