import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        lo = mid if s > target else hi
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_w(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            i1 = np.arange(n - s)
            g[i1] += wk * (1.0 - h[i1 + s])
            if s > 0:
                i2 = np.arange(s, n)
                g[i2] -= wk * h[i2 - s]
        else:
            p = -s
            i1 = np.arange(p, n)
            g[i1] += wk * (1.0 - h[i1 - p])
            if p < n:
                i2 = np.arange(0, n - p)
                g[i2] -= wk * h[i2 + p]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    def _obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        return _grad_w(x, w)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lagrangian_step(h, w, n_steps, lr0, target):
    n = len(h)
    lr = lr0
    f_cur = float(np.dot(w, _corr(h)))
    for _ in range(n_steps):
        g = _grad_w(h, w)
        best_f = f_cur
        best_h = h.copy()
        for t_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = _proj_sum(h - lr * t_mult * g, target)
            f_try = float(np.dot(w, _corr(h_try)))
            if f_try < best_f:
                best_f = f_try
                best_h = h_try
        if best_f < f_cur - 1e-14:
            h = best_h
            f_cur = best_f
            lr = min(lr * 1.05, lr0 * 5)
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return h

def _softmax_weights(c, temp):
    c_shifted = c - c.max()
    w = np.exp(c_shifted / temp)

    if s < 1e-300:
        w = np.ones(len(c)) / len(c)

        w /= s
    return w
