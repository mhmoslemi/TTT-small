    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP progressive smoothing
    for alpha in [3000, 30000, 300000, 2000000]:
        if _el() > 120:
            break
        h_new = _slsqp_opt(best_h, alpha, 2500, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            best_n = len(h_new)

    # Phase 2: Multi-lag targeted descent (key innovation)
    if _el() < budget_s - 350:
        h = best_h.copy()
        n = len(h)
        for n_active in [1, 3, 5, 8, 12]:
            if _el() > budget_s - 300:
                break
            for wb in [2.0, 3.0, 5.0]:
                h, c5 = _topk_descent(h, n, wb, n_active, 200)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n

    # Phase 3: Nesterov with adaptive step
    if _el() < budget_s - 250:
        n = len(best_h)
        h = best_h.copy()
        alpha = 200000
        f = _smooth_obj(h, alpha)
        v = np.zeros(n)
        mu = 0.9
        step = 0.004
        for _s in range(400):
            if _el() > budget_s - 200:
                break
            g = _smooth_grad(h, alpha)
            h_prev = h.copy()
            v = mu * v + (1 - mu) * g
            h_new = _proj_sum(h - step * v, n * 0.5)
            f_new = _smooth_obj(h_new, alpha)
            if f_new < f:
                h = h_new
                f = f_new
                step = min(step * 1.02, 0.015)
            else:
                step *= 0.5
                h = _proj_sum(h_prev - step * v, n * 0.5)
                f = _smooth_obj(h, alpha)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: DE perturbation to escape local minima
    if _el() < budget_s - 200:
        n = len(best_h)
        pop = [best_h.copy()]
        for _i in range(11):
            hp = best_h + 0.03 * rng.randn(n)
            hp = _repair(hp, n)
            pop.append(hp)
        for _i in range(5):
            hp = best_h.copy()
            bs = rng.randint(n // 8, n // 3)
            st = rng.randint(0, n - bs)
            hp[st:st + bs] = 1.0 - hp[st:st + bs]
            hp = _repair(hp, n)
            pop.append(hp)
        pop_scores = [_eval(p) for p in pop]
        F = 0.6
        for _gen in range(30):
            if _el() > budget_s - 120:
                break
            order = np.argsort(pop_scores)
            pop = [pop[i] for i in order]
            pop_scores = [pop_scores[i] for i in order]
            ps = len(pop)
            for i in range(1, ps):
                if _el() > budget_s - 100:
                    break
                r1, r2, r3 = rng.choice(ps, 3, replace=False)
                trial = pop[r1] + F * (pop[r2] - pop[r3])
                trial = np.clip(trial, 0, 1)
                trial = _repair(trial, n)
                c_trial = _eval(trial)
                if c_trial < pop_scores[i]:
                    h_pol = _slsqp_opt(trial, 200000, 800, 1e-18)
                    c_pol = _eval(h_pol)
                    pop[i] = h_pol
                    pop_scores[i] = c_pol
                    if c_pol < best_c5:
                        best_c5 = c_pol
                        best_h = h_pol.copy()

    # Phase 5: Aggressive multi-lag descent on best
    if _el() < budget_s - 100:
        h = best_h.copy()
        n = len(h)
        for n_active in [2, 3, 5, 8]:
            if _el() > budget_s - 60:
                break
            h, c5 = _topk_descent(h, n, 2.5, n_active, 300)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 6: Final SLSQP high-alpha polish
    if _el() < budget_s - 40:
        h = _slsqp_opt(best_h, 5000000, 4000, 1e-22)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = len(h)

    # Phase 7: Ultra-fine pairwise local search
    if _el() < budget_s - 3:
        n = len(best_h)
        t_end = t0 + budget_s - 3
        h, c5 = _pairwise_search(best_h, n, t_end, 999)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
