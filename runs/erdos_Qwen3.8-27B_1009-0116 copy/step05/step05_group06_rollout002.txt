import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s)) + 1e-30) / alpha

def _smooth_grad(h, alpha):

            g[i1] += wk * (1.0 - h[i1 + sh])

def _proj_to_sum(h, target):

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha, nsteps, step0, mu):

def _block_search(h, n_iters, rng, budget_s, t0):
    n = len(h)
    c5c = _eval(h)
    best = h.copy()
    for scale in [0.03, 0.01, 0.004, 0.001]:
        for _it in range(n_iters):
            if time.time() - t0 > budget_s:
                return best
            bs = rng.randint(2, max(3, n // 20))
            pos = rng.randint(0, n - bs + 1)
            d = scale * rng.randn()
            h_new = h.copy()
            h_new[pos:pos+bs] = np.clip(h_new[pos:pos+bs] + d, 0, 1)
            h_new = _repair(h_new, n)
            c5n = _eval(h_new)
            if c5n < c5c:
                h = h_new
                c5c = c5n
                if c5n < _eval(best):
                    best = h.copy()
    return best

def _grad_informed_perturb(h, rng, scale):
    n = len(h)
    alpha = 5000
    g = _smooth_grad(h, alpha)
    g -= g.mean()
    gn = np.linalg.norm(g)

        return _repair(h + scale * rng.randn(n), n)
    g = g / gn
    hp = h - scale * g + 0.005 * rng.randn(n)
    return _repair(hp, n)

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi
