import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())

        if ni < 0 or ni > 1.0 or nj < 0 or nj > 1.0:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-16:
            h[i] = ni
            h[j] = nj
            c = new_c
            c5c = new_c5
            accepted += 1
    return h, c, c5c, accepted

def _targeted_incr(h, c, rng, n_iter, n):
    dx = 2.0 / n
    c5c = c.max() * dx
    for it in range(n_iter):
        if it % 500 == 0:
            idxs = np.argsort(c)[::-1]
            top_k = idxs[:max(1, (2 * n - 1) // 10)]
        k_star = top_k[rng.randint(len(top_k))]
        sh = (n - 1) - k_star
        if sh >= 0:
            r1 = np.arange(n - sh)
            r2 = np.arange(sh, n)
        else:
            sp = -sh
            r1 = np.arange(sp, n)
            r2 = np.arange(n - sp)
        if rng.rand() < 0.6:
            i = r1[rng.randint(len(r1))]
        else:
            i = r2[rng.randint(len(r2))]
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.008 * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1.0 or nj < 0 or nj > 1.0:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-16:
            h[i] = ni
            h[j] = nj
            c = new_c
            c5c = new_c5
    return h, c, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Momentum GD at original resolution
    for alpha in [2000, 10000, 50000, 200000, 500000]:
        if _el() > 40:
            break
        h_new = _momentum_gd(best_h.copy(), alpha, 300, 0.004)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Incremental pairwise - coarse to fine
    if _el() < budget_s - 200:
        h_ps = best_h.copy()
        c_ps = _corr(h_ps)
        n_ps = len(h_ps)
        scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        for s in scales:
            if _el() > budget_s - 150:
                break
            iters = 50000 if s > 0.001 else 80000
            h_ps, c_ps, _, _ = _incr_pair(h_ps, c_ps, rng, iters, s, n_ps)
        # Targeted refinement
        if _el() < budget_s - 120:
            h_ps, c_ps, _ = _targeted_incr(h_ps, c_ps, rng, 60000, n_ps)
        h_ps = _repair(h_ps, n_ps)
        c5 = _eval(h_ps)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_ps, n_ps

    # Phase 3: Small random restarts to escape local minima
    if _el() < budget_s - 150:
        nc = len(best_h)
        for _r in range(8):
            if _el() > budget_s - 120:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = best_h + 0.015 * rng.randn(nc)
            elif strat == 1:
                hp = best_h.copy()
                m = rng.randint(5, 15)
                idxs = rng.choice(nc, m, replace=False)
                hp[idxs] += 0.03 * rng.randn(m)
            elif strat == 2:
                hp = 1.0 - best_h
                hp += 0.01 * rng.randn(nc)
            else:
                hp = best_h.copy()
                bs = nc // 6
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] = rng.uniform(0, 1, bs)
            hp = _repair(hp, nc)
            hp = _momentum_gd(hp, 100000, 150, 0.003)
            c_tmp = _corr(hp)
            hp, c_tmp, _, _ = _incr_pair(hp, c_tmp, rng, 40000, 0.005, nc)
            hp, c_tmp, _ = _targeted_incr(hp, c_tmp, rng, 20000, nc)
            hp = _repair(hp, nc)
            c5 = _eval(hp)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hp, nc

    # Phase 4: Fine incremental on best
    if _el() < budget_s - 80:
        h_fine = best_h.copy()
        c_fine = _corr(h_fine)
        n_fine = len(h_fine)
        for s in [2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6]:
            if _el() > budget_s - 40:
                break
            h_fine, c_fine, _, _ = _incr_pair(h_fine, c_fine, rng, 100000, s, n_fine)
        # Targeted fine
        if _el() < budget_s - 20:
            h_fine, c_fine, _ = _targeted_incr(h_fine, c_fine, rng, 50000, n_fine)
        h_fine = _repair(h_fine, n_fine)
        c5 = _eval(h_fine)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fine, n_fine

    # Phase 5: Another GD pass at very high alpha (near-exact max)
    if _el() < budget_s - 40:
        h_pol = _momentum_gd(best_h.copy(), 1000000, 200, 0.001)
        c_tmp = _corr(h_pol)
        h_pol, c_tmp, _, _ = _incr_pair(h_pol, c_tmp, rng, 50000, 3e-5, len(h_pol))
        h_pol = _repair(h_pol, len(h_pol))
        c5 = _eval(h_pol)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_pol, len(h_pol)

    # Phase 6: Ultra-fine final sweep
    if _el() < budget_s - 5:
        h_fin = best_h.copy()
        c_fin = _corr(h_fin)
        n_fin = len(h_fin)
        h_fin, c_fin, _, _ = _incr_pair(h_fin, c_fin, rng, 150000, 3e-6, n_fin)
        h_fin, c_fin, _ = _targeted_incr(h_fin, c_fin, rng, 50000, n_fin)
        h_fin = _repair(h_fin, n_fin)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fin, n_fin

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
