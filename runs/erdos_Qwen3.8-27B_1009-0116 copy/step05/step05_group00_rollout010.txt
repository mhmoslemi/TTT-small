import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h0, alpha, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h0, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _proj_sum(h, n):
    target = n / 2.0
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _pgd_period(h_p, T, n, alphas, max_iters, lr_init=0.005, seed=0):
    m = n // T
    rng = np.random.RandomState(seed)
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    lr = lr_init
    momentum = np.zeros(T)
    beta = 0.85
    for alpha in alphas:
        iters = max_iters // len(alphas)
        for it in range(iters):
            h_full = np.tile(h_p, m)
            c = np.correlate(h_full, 1.0 - h_full, mode='full')
            mm = c.max()
            s = np.clip(alpha * (c - mm), -500, 500)
            w = np.exp(s - s.max())
            w /= w.sum()
            dx = 2.0 / n
            g_full = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-12:
                    continue
                sh = (n - 1) - k
                wk = w[k] * dx
                if sh >= 0:
                    i1 = np.arange(n - sh)
                    g_full[i1] += wk * (1.0 - h_full[i1 + sh])
                    if sh > 0:
                        g_full[i1 + sh] -= wk * h_full[i1]
                else:
                    sp = -sh
                    i2 = np.arange(sp, n)
                    g_full[i2] += wk * (1.0 - h_full[i2 - sp])
                    g_full[i2 - sp] -= wk * h_full[i2]
            g_p = np.zeros(T)
            for p in range(m):
                g_p += g_full[p * T:(p + 1) * T]
            g_p /= m
            momentum = beta * momentum + (1.0 - beta) * g_p
            h_new = _proj_sum(h_p - lr * momentum, T)
            c_new = _eval(np.tile(h_new, m))
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = h_new.copy()
                h_p = h_new
                if lr < 0.05:
                    lr *= 1.03
            else:
                lr *= 0.6
                if lr < 1e-6:
                    lr = lr_init
                    momentum = np.zeros(T)
    return best, best_c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _coord_descent(h, nc, n_iter, scale, seed=0):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c
