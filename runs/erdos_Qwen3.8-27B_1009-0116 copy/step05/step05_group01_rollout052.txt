    """Minimize max_k (A_k * t + B_k) for t in [lo, hi]"""
    for _ in range(50):  # 50 iterations of ternary search

        m2 = hi - (hi - lo) / 3
        f1 = np.max(A * m1 + B)
        f2 = np.max(A * m2 + B)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2

    # A[k] = coefficient of h[i] in corr[k]
    # corr[k] = sum_j h[j]*(1-h[j+k-(n-1)])
    # h[i] appears when:
    #   j = i: k-(n-1) = m, need 0<=i+m<n -> m in [-i, n-1-i]

    if order is None:

For 2-coordinate updates (i, j):

- 1D problem: min_d max_k (corr_old[k] + (A_i[k]-A_j[k])*d) s.t. constraints on d

The constraint on d: 0 ≤ h[i]+d ≤ 1 and 0 ≤ h[j]-d ≤ 1

Wait: 0 ≤ h[i]+d ≤ 1 → -h[i] ≤ d ≤ 1-h[i]
    0 ≤ h[j]-d ≤ 1 → h[j]-1 ≤ d ≤ h[j]
Combined: d ∈ [max(-h[i], h[j]-1), min(1-h[i], h[j])]

This is a 1D convex problem in d! I can solve it with ternary search.

This is essentially the same as the incremental pairwise method but with an exact 1D solve instead of a fixed step. Much more powerful!

```python
def _pair_cd(h, corr, i, j, n):
    """Update h[i] and h[j] with sum-preserving change, using exact 1D solve."""
    A_i = np.zeros(2*n-1)
    m1 = np.arange(-i, n-i)
    A_i[(n-1)+m1] += (1.0 - h[i+m1])
    m2 = np.arange(-(n-1-i), i+1)
    A_i[(n-1)+m2] += -h[i-m2]
    
    A_j = np.zeros(2*n-1)
    m1 = np.arange(-j, n-j)
    A_j[(n-1)+m1] += (1.0 - h[j+m1])
    m2 = np.arange(-(n-1-j), j+1)
    A_j[(n-1)+m2] += -h[j-m2]
    
    D = A_i - A_j  # change in corr per unit d
    
    # corr_new[k] = corr[k] + D[k]*d
    # Minimize max_k(corr[k] + D[k]*d) s.t. d_lo <= d <= d_hi
    
    d_lo = max(-h[i], h[j]-1.0)
    d_hi = min(1.0-h[i], h[j])
    
    if d_lo >= d_hi:
        return h, corr
    
    lo, hi = d_lo, d_hi
    for _ in range(50):
        t1 = lo + (hi-lo)/3
        t2 = hi - (hi-lo)/3
        if np.max(corr + D*t1) < np.max(corr + D*t2):
            hi = t2
        else:
            lo = t1
    d = (lo+hi)/2
    if abs(d) > 1e-14:
        h[i] += d
        h[j] -= d
        corr += D * d
    return h, corr
```

    m1 = np.arange(-i, n-i)
    A[(n-1)+m1] += (1.0 - h[i+m1])
    m2 = np.arange(-(n-1-i), i+1)
    A[(n-1)+m2] += -h[i-m2]
    return A

def _cd_sweep(h, corr, n, order):
    """One coordinate descent sweep."""
    for i in order:
        A = _compute_A(h, i, n)
        B = corr - A * h[i]
        lo, hi = 0.0, 1.0
        for _ in range(40):

            corr += A * d

        i = rng.randint(n)
        j = rng.randint(n)

            continue
        A_i = _compute_A(h, i, n)
        A_j = _compute_A(h, j, n)

        d_lo = max(-h[i], h[j]-1.0)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_A(h, i, n):
    A = np.zeros(2 * n - 1)
    m1 = np.arange(-i, n - i)
    A[(n - 1) + m1] += (1.0 - h[i + m1])
    m2 = np.arange(-(n - 1 - i), i + 1)
    A[(n - 1) + m2] += -h[i - m2]
    return A

def _cd_sweep(h, corr, n, order):
    for i in order:
        A = _compute_A(h, i, n)
        B = corr - A * h[i]
        lo, hi = 0.0, 1.0
        for _ in range(45):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            if np.max(A * t1 + B) < np.max(A * t2 + B):
                hi = t2
            else:
                lo = t1
        t_new = (lo + hi) * 0.5
        d = t_new - h[i]
        if abs(d) > 1e-14:
            h[i] = t_new
            corr += A * d
    return h, corr

def _pair_cd(h, corr, rng, n, n_iter):
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        A_i = _compute_A(h, i, n)
        A_j = _compute_A(h, j, n)
        D = A_i - A_j
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        lo, hi = d_lo, d_hi
        for _ in range(50):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            if np.max(corr + D * t1) < np.max(corr + D * t2):
                hi = t2
            else:
                lo = t1
        d = (lo + hi) * 0.5
        if abs(d) > 1e-14:
            h[i] += d
            h[j] -= d
            corr += D * d
    return h, corr

def _smooth_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[k] * (1.0 - h[sh:])
            g[sh:] -= w[k] * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n - sp])
            g[:n - sp] -= w[k] * h[sp:]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Aggressive CD sweeps with projection
    if _el() < budget_s - 500:
        corr = _corr(h)
        order = np.arange(n0)
        for sw in range(5000):
            if _el() > budget_s - 500:
                break
            h, corr = _cd_sweep(h, corr, n0, order)
            if abs(h.sum() - n0 * 0.5) > 1e-10:
                h = _repair(h, n0)
                corr = _corr(h)
            c5 = float(np.max(corr) * 2.0 / n0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if (sw + 1) % 100 == 0:
                rng.shuffle(order)

    # Phase 2: Pair CD (sum-preserving exact 1D)
    if _el() < budget_s - 300:
        h2 = best_h.copy()
        corr2 = _corr(h2)
        _pair_cd(h2, corr2, rng, n0, 200000)
        h2 = _repair(h2, n0)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 3: SLSQP smooth optimization
    if _el() < budget_s - 200:
        for alpha in [30000, 150000, 400000]:
            if _el() > budget_s - 200:
                break
            h3 = _slsqp_opt(best_h.copy(), alpha, 2000, n0)
            c5 = _eval(h3)
            if c5 < best_c5:
                best_c5, best_h = c5, h3.copy()

    # Phase 4: CD from perturbed multi-starts
    if _el() < budget_s - 150:
        for _r in range(10):
            if _el() > budget_s - 150:
                break
            hp = best_h + 0.03 * rng.randn(n0)
            hp = _repair(hp, n0)
            corr_p = _corr(hp)
            order_p = rng.permutation(n0)
            for sw in range(300):
                hp, corr_p = _cd_sweep(hp, corr_p, n0, order_p)
                if abs(hp.sum() - n0 * 0.5) > 1e-10:
                    hp = _repair(hp, n0)
                    corr_p = _corr(hp)
            hp = _repair(hp, n0)
            c5 = _eval(hp)
            if c5 < best_c5:
                best_c5, best_h = c5, hp.copy()

    # Phase 5: Pair CD + CD alternating polish
    if _el() < budget_s - 60:
        h5 = best_h.copy()
        corr5 = _corr(h5)
        for round in range(20):
            if _el() > budget_s - 60:
                break
            _pair_cd(h5, corr5, rng, n0, 50000)
            order5 = rng.permutation(n0)
            h5, corr5 = _cd_sweep(h5, corr5, n0, order5)
            h5 = _repair(h5, n0)
            corr5 = _corr(h5)
            c5 = float(np.max(corr5) * 2.0 / n0)
            if c5 < best_c5:
                best_c5, best_h = c5, h5.copy()

    # Phase 6: Final intensive pair CD
    if _el() < budget_s - 20:
        h6 = best_h.copy()
        corr6 = _corr(h6)
        _pair_cd(h6, corr6, rng, n0, 300000)
        h6 = _repair(h6, n0)
        c5 = _eval(h6)
        if c5 < best_c5:
            best_c5, best_h = c5, h6.copy()

    # Phase 7: Final CD polish
    if _el() < budget_s - 5:
        h7 = best_h.copy()
        corr7 = _corr(h7)
        for sw in range(200):
            h7, corr7 = _cd_sweep(h7, corr7, n0, np.arange(n0))
            h7 = _repair(h7, n0)
            corr7 = _corr(h7)
        c5 = _eval(h7)
        if c5 < best_c5:
            best_c5, best_h = c5, h7.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
