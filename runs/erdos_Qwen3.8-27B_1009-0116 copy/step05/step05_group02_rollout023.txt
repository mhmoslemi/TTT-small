    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, m, n):
    dx = 2.0 / n
    sh = m - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        g[sh:] += (1.0 - h[:n-sh])
        g[:n-sh] -= h[sh:]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _adam_lse(h, alpha, lr, iters, seed):
    """Adam optimizer with LSE-smoothed objective."""
    n = len(h)
    h = h.copy()
    m_arr = np.zeros(n)  # first moment
    v_arr = np.zeros(n)  # second moment
    t = 0
    best_c5 = _eval(h)
    best_h = h.copy()
    for it in range(iters):
        t += 1
        c = _corr(h)
        cm = c.max()
        s = np.clip(alpha * (c - cm), -400, 400)
        w = np.exp(s - s.max())
        wn = w / w.sum()
        g = np.zeros(n)
        for k in range(len(c)):
            if wn[k] < 1e-13:
                continue
            g += wn[k] * _grad_lag(h, k, n)
        m_arr = 0.9 * m_arr + 0.1 * g
        v_arr = 0.999 * v_arr + 0.001 * g * g
        m_hat = m_arr / (1 - 0.9 ** t)
        v_hat = v_arr / (1 - 0.999 ** t)
        h_new = _repair(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def _adam_max_grad(h, lr, iters, seed):
    """Adam optimizer attacking the max lag directly."""
    n = len(h)
    h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    best_c5 = _eval(h)
    best_h = h.copy()
    # Track top K lags
    K = 5
    lag_weights = np.ones(2*n-1) / (2*n-1)
    for it in range(iters):
        t += 1
        c = _corr(h)
        # Update lag weights (focus on top K)
        top_k = np.argsort(c)[-K:]
        wt = np.zeros(2*n-1)
        wt[top_k] = 1.0
        lag_weights = 0.95 * lag_weights + 0.05 * wt
        lag_weights /= lag_weights.sum()
        g = np.zeros(n)
        for k in range(2*n-1):
            if lag_weights[k] < 1e-10:
                continue
            g += lag_weights[k] * _grad_lag(h, k, n)
        m_arr = 0.9 * m_arr + 0.1 * g
        v_arr = 0.999 * v_arr + 0.001 * g * g
        m_hat = m_arr / (1 - 0.9 ** t)
        v_hat = v_arr / (1 - 0.999 ** t)
        h_new = _repair(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def _pairwise_search(h, n_iters, scale, seed):
    """Gradient-informed pairwise local search."""
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5 = _eval(h)
    c = _corr(h)
    for _ in range(n_iters):
        # Pick a lag with high correlation
        weights = np.maximum(c - c.max() * 0.999, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(2*n-1)
        weights /= weights.sum()
        k = rng.choice(2*n-1, p=weights)
        sh = k - (n-1)
        # Pick i to modify
        if sh >= 0:
            i_max = n - sh
        else:
            i_max = n + sh
        i = rng.randint(i_max)
        j1 = i
        j2 = i + abs(sh)
        if j2 >= n:
            continue
        d = scale * rng.randn()
        # Try both directions
        h[j1] += d
        h[j2] -= d
        if 0 <= h[j1] <= 1 and 0 <= h[j2] <= 1:
            c5n = _eval(h)
            if c5n < c5 - 1e-16:
                c5 = c5n
                c = _corr(h)
            else:
                h[j1] -= d
                h[j2] += d
    return h, c5
