def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pos_change(h, i, di, n):
    """Exact change in correlation vector when h[i] increases by di."""
    dc = np.zeros(2 * n - 1)
    if abs(di) < 1e-15:
        return dc
    # Positive shifts sh = 0..n-1, index k = (n-1) + sh
    for sh in range(n):
        k = (n - 1) + sh
        if sh == 0:
            dc[k] += di * (1.0 - 2.0 * h[i]) - di * di
        else:
            if i + sh < n:
                dc[k] += di * (1.0 - h[i + sh])
            if i >= sh:
                dc[k] -= di * h[i - sh]
    # Negative shifts sp = 1..n-1, index k = (n-1) - sp
    for sp in range(1, n):
        k = (n - 1) - sp
        if i >= sp:
            dc[k] += di * (1.0 - h[i - sp])
        if i + sp < n:
            dc[k] -= di * h[i + sp]
    return dc

def _incr_pair(h, c, rng, n_iter, scale, n):
    dx = 2.0 / n
    c5c = c.max() * dx
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-16:
            h[i] = ni
            h[j] = nj
            c = new_c
            c5c = new_c5
    return h, c, c5c

def _incr_triple(h, c, rng, n_iter, scale, n):
    """Triple moves: change 3 variables maintaining sum constraint."""
    dx = 2.0 / n
    c5c = c.max() * dx
    for _ in range(n_iter):
        idx = rng.choice(n, size=3, replace=False)
        i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
        # Choose random perturbation summing to 0
        d1 = scale * rng.randn()
        d2 = scale * rng.randn()
        d3 = -(d1 + d2)
        ni = h[i] + d1
        nj = h[j] + d2
        nk = h[k] + d3
        if any(v < -1e-10 or v > 1.0 + 1e-10 for v in [ni, nj, nk]):
            continue
        dc = _pos_change(h, i, d1, n) + _pos_change(h, j, d2, n) + _pos_change(h, k, d3, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-16:
            h[i] = ni
            h[j] = nj
            h[k] = nk
            c = new_c
            c5c = new_c5
    return h, c, c5c
