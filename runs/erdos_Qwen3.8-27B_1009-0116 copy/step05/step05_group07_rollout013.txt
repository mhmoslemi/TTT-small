        d = t - h.sum()
        if abs(d) > 1e-12:
            order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in order:
                if abs(d) < 1e-12:
                    break
                if d > 0:
                    a = min(d, 1.0 - h[i])
                    h[i] += a; d -= a
                else:
                    a = min(-d, h[i])
                    h[i] -= a; d += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > t:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h):
    n = len(h)
    m = k - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        nv = n - m
        if m == 0:
            g[:] = -(2.0 * h - 1.0)
        else:
            g[:nv] += (1.0 - h[m:])
            g[m:] -= h[:nv]
    else:
        s = -m
        nv = n - s
        g[s:] += (1.0 - h[:nv])
        g[:nv] -= h[s:]
    return g * (2.0 / n)

def _momentum_pgd(h, n_iter=8000, topm=15, lr0=0.006, beta0=0.3):
    n = len(h)
    target = n * 0.5
    h_c = h.copy()
    h_p = h.copy()
    f = _eval(h_c)
    beta = beta0
    for t in range(n_iter):
        if time.time() - _t0 > _budget - 15:
            break
        beta = min(beta * 0.9995, 0.8)
        y = h_c + beta * (h_c - h_p)
        y = np.clip(y, 0.0, 1.0)
        c = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k, y)
        g /= topm
        found = False
        lr = lr0 / (1.0 + 0.005 * t)
        for _ls in range(25):
            h_new = _proj(h_c - lr * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-12:
                h_p = h_c.copy()
                h_c = h_new
                f = f_new
                found = True
                break
            lr *= 0.5
        if not found:
            break
    return h_c, f

def _active_descent(h, topm, max_iter, step0=0.004):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - 15:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k, h)
        g /= topm
        found = False
        for _bt in range(40):
            h_new = _proj(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-14:
                h = h_new; f = f_new; found = True
                break
            step *= 0.5
        if not found:
            break
        step = min(step * 1.1, step0)
    return h, f

def _binary_swap(h, n_iter, seed_val):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 10:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        di = h[i] - 0.5
        dj = 0.5 - h[j]
        eps = min(di, dj) * 0.8
        h[i] -= eps
        h[j] += eps
        fn = _eval(h)
        if fn < f - 1e-15:
            f = fn
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            h[i] += eps
            h[j] -= eps
    return h, f

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-13:
                continue
            sh = k - (n - 1)
            if sh >= 0:
                g[:n-sh] += w[k] * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= w[k] * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += w[k] * (1.0 - x[:n-sp])
                g[:n-sp] -= w[k] * x[sp:]
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def _el():
    return time.time() - _t0

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: SLSQP at n0
    for alpha in [8000, 50000, 150000]:
        if _el() > 45:
            break
        hn = _slsqp(best_h, alpha, 1200, 1e-18)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5; best_h = hn.copy()

    # Phase B: Upsample to 2x
    n1 = n0 * 2
    h2 = _upsample(best_h, 2)
    if _el() < budget_s - 200:
        for alpha in [15000, 80000]:
            if _el() > budget_s - 160:
                break
            hn = _slsqp(h2, alpha, 1500, 1e-18)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5; best_h = hn.copy(); best_n = n1

    # Phase C: Nesterov momentum PGD at best resolution
    nc = len(best_h)
    if _el() < budget_s - 250:
        hn, fc = _momentum_pgd(best_h, n_iter=6000, topm=12, lr0=0.005, beta0=0.25)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    # Phase D: Active-set descent
    if _el() < budget_s - 200:
        for topm in [5, 10, 20, 35]:
            if _el() > budget_s - 160:
                break
            hn, fc = _active_descent(best_h, topm, 3000, 0.003)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase E: Higher resolution upsample
    if _el() < budget_s - 200 and best_n < 1000:
        n2 = best_n * 2
        if n2 <= 1400:
            h3 = _upsample(best_h, 2)
            hn = _slsqp(h3, 30000, 1000, 1e-17)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5; best_h = hn.copy(); best_n = n2
                if _el() < budget_s - 120:
                    hn, fc = _momentum_pgd(best_h, n_iter=4000, topm=15, lr0=0.004, beta0=0.2)
                    if fc < best_c5:
                        best_c5 = fc; best_h = hn.copy()

    # Phase F: Return to best_n and refine
    if best_n > n0 * 2 and _el() < budget_s - 150:
        x_coarse = np.linspace(0, 2, n0)
        f_int = interp1d(np.linspace(0, 2, len(best_h)), best_h, kind='cubic',
                         fill_value=(best_h[0], best_h[-1]), bounds_error=False)
        h_back = _repair(np.clip(f_int(x_coarse), 0, 1), n0)
        c5b = _eval(h_back)
        if c5b < best_c5:
            best_c5 = c5b; best_h = h_back.copy(); best_n = n0

    # Phase G: Binary push
    nc = len(best_h)
    if _el() < budget_s - 80:
        for scale in [0.002, 0.001, 0.0005]:
            for _ in range(3):
                if _el() > budget_s - 60:
                    break
                d = scale * rng.randn(nc)
                hp = _proj(best_h + d, nc * 0.5)
                hn, fc = _binary_swap(hp, 2000, seed + int(scale * 1e5))
                if fc < best_c5:
                    best_c5 = fc; best_h = hn.copy()

    # Phase H: Final SLSQP polish
    if _el() < budget_s - 60:
        hn = _slsqp(best_h, 300000, 1500, 1e-19)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5; best_h = hn.copy()

    # Phase I: Final active descent + momentum
    if _el() < budget_s - 30:
        hn, fc = _active_descent(best_h, 25, 2000, 0.002)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()
        if _el() < budget_s - 15:
            hn, fc = _momentum_pgd(best_h, n_iter=3000, topm=20, lr0=0.002, beta0=0.15)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase J: Ultra-fine pairwise
    if _el() < budget_s - 10:
        nc = len(best_h)
        for scale in [0.0008, 0.0003, 0.0001]:
            if _el() > budget_s - 5:
                break
            for _ in range(5000):
                if time.time() - _t0 > _budget - 3:
                    break
                i, j = rng.randint(nc), rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = best_h[i] + d, best_h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    best_h[i], best_h[j] = ni, nj
                    fn = _eval(best_h)
                    if fn < best_c5 - 1e-15:
                        best_c5 = fn
                    else:
                        best_h[i], best_h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
