    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lbes_obj(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    base = m + np.log(float(np.sum(np.exp(s)))) / alpha
    # Penalty for sum constraint
    base += lam * (x.sum() - n * 0.5) ** 2
    return base

def _lbes_grad(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())

    dx = 2.0 / n

    # Add penalty gradient
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

    dx = 2.0 / n
    best = h.copy()
    best_c = _eval(h)
    step = step_init
    for it in range(n_iter):
        c = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(c)
        sh = (n - 1) - k_max
        g = np.zeros(n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
        g *= dx
        # Remove mean component (to maintain sum constraint)
        g -= g.mean()
        h_new = _repair(h - step * g, n)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best = h_new.copy()
        h = h_new
        step *= step_decay
    return best, best_c

def _coord_descent(h, n_iter, rng, scales):
    n = len(h)
    c_best = _eval(h)
    for scale in scales:
        for _ in range(n_iter):
            i = rng.randint(n)

            else:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_logsumexp(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _lbes_obj(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    base = m + np.log(float(np.sum(np.exp(s)))) / alpha
    base += lam * (x.sum() - n * 0.5) ** 2
    return base

def _lbes_grad(x, alpha, lam, n):
    g = _grad_logsumexp(x, alpha)
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def _lbfgs_opt(h, alpha, lam, n_iter, n):
    bnds = [(0.0, 1.0)] * n
    res = minimize(_lbes_obj, h, jac=_lbes_grad,
                   args=(alpha, lam, n), method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': n_iter, 'ftol': 1e-18,
                                         'gtol': 1e-15, 'maxls': 50})
    return _repair(res.x, n)

def _nesterov_subgrad(h, n_iter, lr, decay):
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    h_prev = h.copy()
    best = h.copy()
    best_c = _eval(h)
    for it in range(n_iter):
        c = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        c_max = c.max()
        eps = max(0.01 * c_max, 1e-6)
        mask = c > c_max - eps
        cnt = mask.sum()
        if cnt == 0:
            mask = np.zeros(len(c), bool)
            mask[np.argmax(c)] = True
        g = np.zeros(n)
        for k in np.where(mask)[0]:
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h_curr[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= h_curr[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += (1.0 - h_curr[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= h_curr[i2 + sp]
        g *= dx / cnt
        g -= g.mean()
        momentum = 0.9 * (h_curr - h_prev)
        h_prev = h_curr.copy()
        h_curr = _repair(h_curr - lr * (g + momentum) + 0.9 * momentum, n)
        c_val = _eval(h_curr)
        if c_val < best_c:
            best_c = c_val
            best = h_curr.copy()
        lr *= decay
    return best, best_c

def _coord_descent(h, n_iter, rng, scales):
    n = len(h)
    c_best = _eval(h)
    for scale in scales:
        for _ in range(n_iter):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < c_best - 1e-16:
                c_best = c_new
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c_best

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _periodic_start(n, T, rng):
    base = rng.uniform(0.1, 0.9, T)
    h = np.tile(base, n // T)
    return _repair(h, n)

def _run_phase1(h, n, alpha_list, t0, budget):
    best = h.copy()
    best_c = _eval(h)
    for alpha in alpha_list:
        if time.time() - t0 > budget:
            break
        h_new = _lbfgs_opt(best, alpha, 100.0, 2000, n)
        c = _eval(h_new)
        if c < best_c:
            best_c = c
            best = h_new.copy()
    return best, best_c

    t0 = time.time()
