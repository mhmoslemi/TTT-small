    for step in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]:
        h_new = _repair(h - step * v, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-14:
            return h_new, c5n, v
    return h, c5c, v

def _lse_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for kk in range(2 * n - 1):
        if w[kk] < 1e-14:
            continue
        g += w[kk] * _grad_lag(h, kk)
    g *= 2.0 / n
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h0, alpha, maxit):
    n = len(h0)
    try:
        res = minimize(_lse_obj, h0, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}],
                       options={'maxiter': maxit, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0.copy(), n)

def _periodic_init(n, period):
    h = np.zeros(n)
    duty = max(1, int(period * 0.5))
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    return _repair(h, n)

def _sin_init(n, freq):
    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _tl():
        return budget_s - (time.time() - t0)
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 0: structured multi-start
    inits = [best_h.copy()]
    for p in [3, 4, 5, 6, 7, 9, 11, 13, 17]:
        if p * 2 <= n0:
            inits.append(_periodic_init(n0, p))
    for f in [2, 3, 5, 7, 11]:
        inits.append(_sin_init(n0, f))
    for _si in range(6):
        inits.append(_repair(rng.rand(n0) + best_h * 0.3, n0))
    for init_h in inits:
        if _tl() < 350:
            break
        ht = _slsqp(init_h, 3000, 600)
        c5v = _eval(ht)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = ht.copy()

    # Phase 1: momentum active-set descent
    v = np.zeros(n0)
    hw = best_h.copy()
    for it in range(3000):
        if _tl() < 280:
            break
        if it % 250 == 249:
            hw = _repair(hw + 0.004 * rng.randn(n0), n0)
            v *= 0.3
        tk = 2 + (it % 3)
        hw, _, v = _active_step(hw, tk, v, mom=0.25)
        c5w = _eval(hw)
        if c5w < best_c5:
            best_c5 = c5w
            best_h = hw.copy()

    # Phase 2: SLSQP sweep
    for alpha in [2000, 10000, 50000, 200000, 800000]:
        if _tl() < 130:
            break
        hnew = _slsqp(best_h, alpha, 1500)
        c5v = _eval(hnew)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hnew.copy()

    # Phase 3: active-set + binary push
    v = np.zeros(n0)
    hw = best_h.copy()
    for it in range(2000):
        if _tl() < 220:
            break
        if it % 300 == 299:
            msk = (hw > 0.55) | (hw < 0.45)
            hw[msk] = np.where(hw[msk] > 0.5, 1.0, 0.0)
            hw = _repair(hw, n0)
            v *= 0.1
        hw, _, v = _active_step(hw, 3, v, mom=0.2)
        c5w = _eval(hw)
        if c5w < best_c5:
            best_c5 = c5w
            best_h = hw.copy()

    # Phase 4: upsample to 2x
    if _tl() > 280:
        n1 = n0 * 2
        hh = np.interp(np.linspace(0, 1, n1, endpoint=False),
                        np.linspace(0, 1, n0, endpoint=False), best_h)
        hh = _repair(hh, n1)
        for alpha in [5000, 30000, 150000]:
            if _tl() < 130:
                break
            hh = _slsqp(hh, alpha, 1200)
            c5v = _eval(hh)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hh.copy()
                best_n = n1
        v = np.zeros(n1)
        hw = best_h.copy()
        for it in range(500):
            if _tl() < 110:
                break
            hw, _, v = _active_step(hw, 3, v, 0.2)
            c5w = _eval(hw)
            if c5w < best_c5:
                best_c5 = c5w
                best_h = hw.copy()
                best_n = n1
        best_n = len(best_h)

    # Phase 5: annealed pairwise search
    nc = best_n
    ha = best_h.copy()
    c5a = best_c5
    for scale in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        if _tl() < 30:
            break
        nit = 20000
        for _it in range(nit):
            if _tl() < 25:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = ha[i] + d, ha[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            ha[i], ha[j] = ni, nj
            c5n = _eval(ha)
            if c5n < c5a - 1e-15:
                c5a = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = ha.copy()
            else:
                ha[i], ha[j] = ni - d, nj + d

    # Phase 6: final SLSQP
    if _tl() > 18:
        hnew = _slsqp(best_h, 1000000, 2500)
        c5v = _eval(hnew)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hnew.copy()

    # Phase 7: ultra-fine
    if _tl() > 5:
        hf = best_h.copy()
        c5f = _eval(hf)
        for _it in range(50000):
            if _tl() < 2:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 1e-5 * rng.randn()
            ni, nj = hf[i] + d, hf[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                hf[i], hf[j] = ni, nj
                c5n = _eval(hf)
                if c5n < c5f - 1e-17:
                    c5f = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = hf.copy()
                else:
                    hf[i], hf[j] = hf[i] - d, hf[j] + d

    best_n = len(best_h)
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
