                if _elapsed() > budget_s - 120:

                h_new = _slsqp(h_p, alpha, 2000, 1e-19)
                c5 = _eval(h_new)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_new.copy()
            # Also try binary snap + optimize
            h_b2 = _repair(np.where(h_p > 0.5, 1.0, 0.0), best_n)
            h_b2 = _slsqp(h_b2, 100000, 2000, 1e-19)
            c5 = _eval(h_b2)
            if c5 < best_c5:
                best_c5, best_h = c5, h_b2.copy()

    # Phase 6: Nelder-Mead on periodic parameterization
    if _elapsed() < budget_s - 80:
        for T in [5, 6, 7, 8, 10, 12, 14, 16, 20, 28]:
            if _elapsed() > budget_s - 80:
                break
            if best_n % T != 0:
                continue
            h_p = _nelder_periodic(best_h, T, rng)
            c5 = _eval(h_p)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()

    # Phase 7: Try nearby n values
    if _elapsed() < budget_s - 60:
        for dn in [10, -10, 20, -20, 30, -30, 50, -50]:
            if _elapsed() > budget_s - 60:
                break
            n_try = best_n + dn
            if n_try < 50:
                continue
            xo = np.linspace(0, 2, best_n, endpoint=False)
            xn = np.linspace(0, 2, n_try, endpoint=False)
            h_r = np.interp(xn, xo, best_h)
            h_r = _repair(h_r, n_try)
            h_r = _slsqp(h_r, 100000, 2000, 1e-19)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 8: Ultra-fine L-BFGS-B + SLSQP polish
    if _elapsed() < budget_s - 20:
        for alpha in [500000, 1000000, 2000000]:
            if _elapsed() > budget_s - 20:
                break
            h_new = _lbfgsb(best_h, alpha, 5000, 1e8)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()
        h_fin = _slsqp(best_h, 2000000, 3000, 1e-20)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    # Phase 9: Pairwise coordinate descent (final micro-optimization)
    if _elapsed() < budget_s - 3:
        h_cd = best_h.copy()
        nc = len(h_cd)
        c5c = _eval(h_cd)
        for scale in [0.01, 0.005, 0.002, 0.0008, 0.0003]:
            if _elapsed() > budget_s - 3:
                break
            n_iter = 4000 if scale > 0.01 else 7000
            for _it in range(n_iter):
                if _elapsed() > budget_s - 3:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni2, nj2 = h_cd[i] + d, h_cd[j] - d
                if ni2 < 0 or ni2 > 1 or nj2 < 0 or nj2 > 1:
                    continue
                h_cd[i], h_cd[j] = ni2, nj2
                c5n = _eval(h_cd)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h_cd.copy()
                else:
                    h_cd[i], h_cd[j] = ni2 - d, nj2 + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
