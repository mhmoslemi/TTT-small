def _repair(h, n):

def _proj(h, target):

        if np.clip(h - mid, 0, 1).sum() > target:

        else:

def _grad_single(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    sc = 2.0 / n
    if s == 0:
        g[:] = (1.0 - 2.0 * h) * sc
    elif s > 0:
        g[:n - s] += (1.0 - h[s:]) * sc
        g[s:] -= h[:n - s] * sc
    else:
        p = -s
        g[p:] += (1.0 - h[:n - p]) * sc
        g[:n - p] -= h[p:] * sc
    return g

def _smooth_obj(h, alpha):
    c = _cvec(h)
    m = c.max()
    ex = np.clip(alpha * (c - m), -300, 300)
    ex -= ex.max()
    return m + np.log(np.exp(ex).sum()) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _cvec(h)
    m = c.max()
    ex = np.clip(alpha * (c - m), -300, 300)
    ex -= ex.max()
    w = np.exp(ex)
    wsum = w.sum()
    if wsum < 1e-30:
        return _grad_single(h, c.argmax())
    w /= wsum
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-10:
            g += w[k] * _grad_single(h, k)
    return g

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x): return _smooth_obj(x, alpha)
    def jac(x): return _smooth_grad(x, alpha)
    def cfun(x): return x.sum() - n * 0.5
    def cjac(x): return np.ones(len(x))
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})

        hn, fc = _proj_grad(best_h, 200000, 3000, 0.002, topm=10)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()
        hn, fc = _topm_descent(best_h, 8, 1500, 0.001)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    # Phase 8: Random micro-perturbations
    if _el() < budget_s - 8:
        rng = np.random.RandomState(seed + 999)
        nc = len(best_h)
        for _ in range(8):
            if _el() > budget_s - 5:
                break
            hp = best_h + 0.003 * rng.randn(nc)
            hp = _proj(hp, nc / 2.0)
            hn, fc = _proj_grad(hp, 50000, 500, 0.003)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
