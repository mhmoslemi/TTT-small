import numpy as np
import time
from scipy.optimize import minimize

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    h = _proj(h, n * 0.5)
    d = n * 0.5 - h.sum()
    if abs(d) > 1e-11:
        i = int(np.argmin(h)) if d > 0 else int(np.argmax(h))
        h[i] = float(np.clip(h[i] + d, 0, 1))
    return h

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_corrs(h).max())

def _g_k(k, h):
    n = len(h)
    dx = 2.0 / n
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        if sh < n:
            g[:n - sh] += 1.0 - h[sh:]
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[:n - sp] += 1.0 - h[sp:]
        g[sp:] -= h[:n - sp]
    return g * dx

def _cfun(x, n):
    return x.sum() - n * 0.5

def _cjac(x, n):
    return np.ones(len(x))

def _smooth_obj(x, n, alpha):
    c = _corrs(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.exp(s - s.max()).sum()) / alpha

def _smooth_jac(x, n, alpha):
    c = _corrs(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-14:
            g += w[k] * _g_k(k, x)
    return g

def _slsqp(h, alpha, iters):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_smooth_obj, h, jac=_smooth_jac, args=(n, alpha),
                     method='SLSQP', bounds=bnds,
                     constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac, 'args': (n,)}],
                     options={'maxiter': iters, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)

    fn = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(fn(x_new), 0, 1), n * factor)
