        else:

        sp = -sh
        g[sp:] += 1.0 - h[:n - sp]

def _smooth_grad(h, alpha, topk=0):
    n = len(h)
    c = _corr(h)
    if 0 < topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    s = np.clip(alpha * (cs - m), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()

    for ik, wk in zip(idx, w):

        g += wk * _grad_lag_vec(h, (n - 1) - ik)

def _slsqp(h, alpha, maxit=2000, ftol=1e-16, topk=0):

                                      'fun': lambda x: x.sum() - n * 0.5,

def _momentum_descent(h, alpha, topk, steps, step0):
    n = len(h)
    f = _smooth_obj(h, alpha, topk)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        g = _smooth_grad(h, alpha, topk)
        v = mu * v + (1.0 - mu) * g

def _pair_flip(h, n, n_iter, rng, scale):

    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)

            continue

            if c < best_c:

            else:

    return best_h, best_c

def _lag_descent(h, n, n_rounds):
    """Descend on single worst lag each round."""
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(n_rounds):
        c = _corr(best_h)
        k_w = int(c.argmax())

        g = _grad_lag_vec(best_h, sh)

                improved = True
                break

    return best_h, best_c

    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        g = _smooth_grad(h, alpha, topk)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_sum(h - step * v, n * 0.5)
        f_new = _smooth_obj(h_new, alpha, topk)
        if f_new < f - 1e-18:
            h, f = h_new, f_new
            step = min(step * 1.02, step0 * 3.0)
        else:
            step *= 0.6
            h = _proj_sum(h - step * v, n * 0.5)
            f = _smooth_obj(h, alpha, topk)
    return h

def _pair_flip(h, n, n_iter, rng, scale):
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            best_h[i], best_h[j] = ni, nj
            c = _eval(best_h)
            if c < best_c:
                best_c = c
            else:
                best_h[i], best_h[j] = best_h[i] - d, best_h[j] + d
    return best_h, best_c

def _lag_descent(h, n, n_rounds):
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(n_rounds):
        c = _corr(best_h)
        k_w = int(c.argmax())
        sh = (n - 1) - k_w
        g = _grad_lag_vec(best_h, sh)
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002]:
            h_new = _proj_sum(best_h - step * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-14:
                best_h, best_c = h_new, c_new
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def _binary_round(h, n):
    hb = np.where(h > 0.5, 1.0, 0.0)
    return _proj_sum(hb, n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _proj_sum(np.clip(np.array(initial_h_values, dtype=float), 0, 1), n0 * 0.5)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: SLSQP progressive alpha at original resolution
    for alpha in [3000, 15000, 75000, 400000]:
        if _elapsed(t0) > 100:
            break
        hc = _slsqp(best_h, alpha, 2500, 1e-16, 0)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 2: Topk-focused SLSQP
    if _elapsed(t0) < budget_s - 550:
        for tk in [3, 5]:
            if _elapsed(t0) > budget_s - 550:
                break
            hc = _slsqp(best_h, 300000, 2000, 1e-16, tk)
            c5 = _eval(hc)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()

    # Phase 3: Momentum gradient descent then SLSQP polish
    if _elapsed(t0) < budget_s - 400:
        h2 = _momentum_desc(best_h, 80000, 5, 250, 0.003)
        hc = _slsqp(h2, 500000, 2000, 1e-16, 0)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 4: Single worst-lag iterative descent
    if _elapsed(t0) < budget_s - 300:
        best_h, best_c5 = _lag_descent(best_h, best_n, 500)

    # Phase 5: High-alpha SLSQP polish
    if _elapsed(t0) < budget_s - 200:
        for alpha in [1000000, 3000000]:
            if _elapsed(t0) > budget_s - 200:
                break
            hc = _slsqp(best_h, alpha, 3000, 1e-16, 0)
            c5 = _eval(hc)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()

    # Phase 6: Medium pair flips
    if _elapsed(t0) < budget_s - 100:
        nit = min(int((budget_s - 100 - _elapsed(t0)) / 0.004), 25000)
        best_h, best_c5 = _pair_flip(best_h, best_n, nit, rng, 0.015)

    # Phase 7: Multi-start with diverse initializations
    if _elapsed(t0) < budget_s - 60:
        ta = np.arange(best_n, dtype=float)
        for ms in range(10):
            if _elapsed(t0) > budget_s - 60:
                break
            if ms % 5 == 0:
                hn = _proj_sum(best_h + 0.03 * rng.randn(best_n), best_n * 0.5)
            elif ms % 5 == 1:
                fn = rng.randint(2, best_n // 4 + 1)
                ph = rng.rand() * 6.2832
                hn = _proj_sum(best_h + 0.04 * np.sin(2 * np.pi * fn * ta / best_n + ph), best_n * 0.5)
            elif ms % 5 == 2:
                hn = _proj_sum(0.5 + 0.35 * np.sin(2 * np.pi * rng.randint(2, 6) * ta / best_n), best_n * 0.5)
            elif ms % 5 == 3:
                hn = _binary_round(best_h, best_n)
                hn = _proj_sum(hn + 0.01 * rng.randn(best_n), best_n * 0.5)
            else:
                hn = _binary_round(_proj_sum(rng.rand(best_n) * 1.2, best_n * 0.5), best_n)
            hc = _slsqp(hn, 200000, 1200, 1e-16, 0)
            c5 = _eval(hc)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()

    # Phase 8: Lag descent again after multi-start
    if _elapsed(t0) < budget_s - 30:
        best_h, best_c5 = _lag_descent(best_h, best_n, 300)

    # Phase 9: Ultra-fine pair swaps
    if _elapsed(t0) < budget_s - 2:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        nt = min(int((budget_s - 2 - _elapsed(t0)) / 0.002), 50000)
        for _ in range(nt):
            if _elapsed(t0) > budget_s - 2:
                break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = 1.5e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _eval(h)
                if c < c5c:
                    c5c = c
                    if c < best_c5:
                        best_c5, best_h = c, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _proj_sum(np.clip(best_h, 0, 1), best_n * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
