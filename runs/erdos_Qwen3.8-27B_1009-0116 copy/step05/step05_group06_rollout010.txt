```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _topk_obj(h, topk, alpha):
    c = _corr(h)
    idx = np.argsort(c)[::-1][:topk]
    ck = c[idx]
    m = ck.max()
    s = np.clip(alpha * (ck - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _topk_grad(h, topk, alpha):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[::-1][:topk]
    ck = c[idx]
    m = ck.max()
    s = np.clip(alpha * (ck - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _full_smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _full_smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

_CON_N = 140

def _con_fun(x):
    return x.sum() - _CON_N * 0.5

def _con_jac(x):
    return np.ones(_CON_N)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_topk(h, topk, alpha, maxiter, ftol):
    n = len(h)
    global _CON_N
    _CON_N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_topk_obj, h, args=(topk, alpha),
                       jac=lambda x, t, a: _topk_grad(x, t, a),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _slsqp_full(h, alpha, maxiter, ftol):
    n = len(h)
    global _CON_N
    _CON_N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_full_smooth_obj, h, args=(alpha,),
                       jac=lambda x, a: _full_smooth_grad(x, a),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lag_grad(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * (2.0 / n)

def _focused_minmax(h, n_active, nsteps, step0):
    n = len(h)
    c5c = _eval(h)
    step = step0
    for _ in range(nsteps):
        c = _corr(h)
        idx = np.argsort(c)[::-1][:n_active]
        g = np.zeros(n)
        for k in idx:
            g += _lag_grad(h, k)
        g /= n_active
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for _bt in range(30):
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h, c5c = h_new, c5n
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-13:
                break
        step = min(step * 1.03, step0 * 2.5)
    return h, c5c

def _local_search(h, n_iters, scale, rng):
    n = len(h)
    c5c = _eval(h)
    s = scale
    for _ in range(n_iters):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = s * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
            s = min(s * 1.04, scale)
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            s *= 0.96
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    h_hi = np.empty(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi
```
