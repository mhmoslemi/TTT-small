import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-9:
                break
    return h

def _grad_lse(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sv = k - (n - 1)
        if sv >= 0:
            g[sv:n] += wk * (1.0 - h[0:n - sv])
            if sv > 0:
                g[0:n - sv] -= wk * h[sv:n]
        else:
            p = -sv
            g[0:n - p] += wk * (1.0 - h[p:n])
            g[p:n] -= wk * h[0:n - p]
    return g * dx

def _nesterov(h0, alpha, n_iter, lr0, decay, t_end, restart_every=800, seed=0):
    n = len(h0)
    rng = np.random.RandomState(seed)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = _eval(h)
    lr = lr0
    stag = 0
    c_prev = best_c
    for it in range(n_iter):
        if time.time() > t_end:
            break
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) * (it + 1))) / 2.0
        beta = it / t_k if it > 0 else 0.0
        y = np.clip(h + beta * v, 0.0, 1.0)
        g = _grad_lse(y, alpha)
        h_new = _repair(y - lr * g, n)
        v = h_new - y
        h = h_new
        lr *= decay
        c = _eval(h)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stag = 0
        else:
            stag += 1
        if stag >= restart_every:
            h = _repair(h + 0.005 * rng.randn(n), n)
            v[:] = 0.0
            lr = lr0 * 0.4
            stag = 0
    return best_h, best_c

def _subgrad_top(h, alpha, n_iter, top_m, lr0, decay, t_end):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    best_h = h.copy()
    lr = lr0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        idx = np.argsort(c)[-top_m:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
        w /= w.sum()
        g = np.zeros(n)
        dx = 2.0 / n
        for k in idx:
            sv = k - (n - 1)
            wk = w[k] * dx
            if sv >= 0:
                g[sv:n] += wk * (1.0 - h[0:n - sv])
                if sv > 0:
                    g[0:n - sv] -= wk * h[sv:n]
            else:
                p = -sv
                g[0:n - p] += wk * (1.0 - h[p:n])
                g[p:n] -= wk * h[0:n - p]
        h_new = _repair(h - lr * g, n)
        c_new = _eval(h_new)
        if c_new < cur - 1e-14:
            h = h_new
            cur = c_new
            if cur < _eval(best_h):
                best_h = h.copy()
        else:
            h = _repair(h - lr * 0.5 * g, n)
            c_new2 = _eval(h)
            if c_new2 < cur - 1e-14:
                cur = c_new2
                if cur < _eval(best_h):

            else:

                    cur = c_new3
                    if cur < _eval(best_h):
                        best_h = h.copy()
                else:
                    lr *= 0.7
        lr = max(lr * decay, 1e-6)
    return best_h, _eval(best_h)

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _coord_descent(h0, n_iter, t_end):
    n = len(h0)
    h = h0.copy()
    cur = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = h[i]
            for trial in [0.0, 1.0, 0.5, old + 0.3, max(0, old - 0.3)]:
                if abs(trial - old) < 1e-10:
                    continue
                h[i] = trial
                h = _repair(h, n)
                c = _eval(h)
                if c < cur - 1e-13:
                    cur = c
                    improved = True
                    break
                h[i] = old
            if improved:
                improved = False
        if not improved:
            break
    return h, cur
