        c5 = _eval(hc)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    # Phase 2: Projected gradient with moderate sharpness
    if rem() > 280:
        hc = _pgd_topk(bh, 2000, 15, 800, t0 + rem() - 279)
        c5 = _eval(hc)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    # Phase 3: L-BFGS-B sharp (small k, high alpha)
    for alpha, k, pen in [(3000, 12, 1e5), (15000, 10, 1e5), (60000, 8, 1e5)]:
        if rem() < 220:
            break
        hc = _lbfgs_topk(bh, alpha, k, 5000, t0 + rem() - 219, pen)
        c5 = _eval(hc)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    # Phase 4: Balance top values
    if rem() > 160:
        hc, c5 = _balance(bh, 8, t0 + rem() - 159)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    # Phase 5: Very high alpha L-BFGS-B
    for alpha, k in [(200000, 10), (600000, 10), (1500000, 8)]:
        if rem() < 120:
            break
        hc = _lbfgs_topk(bh, alpha, k, 5000, t0 + rem() - 119, 5e5)
        c5 = _eval(hc)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    # Phase 6: Balance again with larger ntop
    if rem() > 90:
        hc, c5 = _balance(bh, 12, t0 + rem() - 89)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    # Phase 7: Perturbed restarts
    if rem() > 70:
        for _r in range(4):
            if rem() < 65:
                break
            hp = bh + 0.015 * rng.randn(bn)
            hp = _repair(hp, bn)
            hc = _lbfgs_topk(hp, 200000, 10, 3000, t0 + rem() - 64, 1e5)
            c5 = _eval(hc)
            if c5 < bc5:
                bc5, bh = c5, hc.copy()

    # Phase 8: Final balance
    if rem() > 45:
        hc, c5 = _balance(bh, 15, t0 + rem() - 44)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    # Phase 9: Coarse coordinate descent
    if rem() > 20:
        ni = min(int((rem() - 18) / 0.005), 20000)
        hc, c5 = _coord(bh, ni, t0 + rem() - 18, rng)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    # Phase 10: Ultra-fine coordinate descent
    if rem() > 10:
        ni = min(int((rem() - 9) / 0.004), 15000)
        hc, c5 = _coord(bh, ni, t0 + rem() - 9, rng)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    bh = _repair(bh, bn)
    bc5 = _eval(bh)
    return (bh, bc5, bn)
