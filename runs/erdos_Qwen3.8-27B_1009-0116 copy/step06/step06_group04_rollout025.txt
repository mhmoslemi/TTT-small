```python
def _slp_step(h, n, dx):
    c = _compute_c(h, n, dx)  # (2n-1,)
    # Compute gradient matrix
    g = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        if s >= 0:
            idx1 = np.arange(max(0, n-s))
            g[k, idx1] += dx * (1.0 - h[idx1 + s]) if s < n else 0
            idx2 = np.arange(s, n) if s > 0 else np.array([], dtype=int)
            g[k, idx2] -= dx * h[idx2 - s]
        else:
            sp = -s
            idx1 = np.arange(sp, n)
            g[k, idx1] += dx * (1.0 - h[idx1 - sp])
            idx2 = np.arange(max(0, n-sp))
            g[k, idx2] -= dx * h[idx2 + sp]
    
    # Linearization: c_k(h) ≈ c_k(h_0) + g[k,:] @ (h - h_0)
    # = g[k,:] @ h + (c_k - g[k,:] @ h_0)
    # Constraint: g[k,:] @ h + (c_k - g[k,:] @ h_0) ≤ t
    # → g[k,:] @ h - t ≤ -(c_k - g[k,:] @ h_0)
    # → g[k,:] @ h - t ≤ g[k,:] @ h_0 - c_k
    
    consts = g @ h - c  # (2n-1,)
    
    # LP: min t
    # s.t. g[k,:] @ h - t ≤ consts[k] for all k
    #      0 ≤ h ≤ 1
    #      sum(h) = n/2
    
    c_obj = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([g, -np.ones((2*n-1, 1))])
    b_ub = consts
    A_eq = np.hstack([np.ones(n), [0.0]])
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h
```

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _compute_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_mat(h, n):
    dx = 2.0 / n
    g = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        if s >= 0:
            m = n - s
            if m > 0:
                g[k, :m] += dx * (1.0 - h[s:s+m])
            if s > 0:
                g[k, s:] -= dx * h[:m]
        else:
            sp = -s
            g[k, sp:] += dx * (1.0 - h[:n-sp])
            g[k, :n-sp] -= dx * h[sp:]
    return g

def _slp_one(h, n, damping=1.0):
    c_vals = _compute_c(h, n)
    g = _grad_mat(h, n)
    consts = g @ h - c_vals
    c_obj = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([g, -np.ones((2*n-1, 1))])
    b_ub = consts
    A_eq = np.hstack([np.ones(n), [0.0]])
    b_eq = np.array([n * 0.5])
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            h_new = res.x[:n]
            h_new = (1-damping)*h + damping*h_new
            return _
