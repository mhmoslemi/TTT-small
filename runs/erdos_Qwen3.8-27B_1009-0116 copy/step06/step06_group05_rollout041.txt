        if not moved:
            break
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _make_pattern(n, kind, rng):
    h = np.zeros(n)
    if kind == 0:
        h[::2] = 1.0
    elif kind == 1:
        for i in range(n): h[i] = 1.0 if i % 3 < 2 else 0.0
    elif kind == 2:
        for i in range(n): h[i] = 1.0 if np.sin(2*np.pi*3.0*i/n) > 0 else 0.0
    elif kind == 3:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for i in range(n): h[i] = 1.0 if np.sin(2*np.pi*phi*i) > 0 else 0.0
    elif kind == 4:
        h = (rng.rand(n) < 0.5).astype(float)
    elif kind == 5:
        for i in range(n): h[i] = 1.0 if np.sin(2*np.pi*5.0*i/n + 0.3) > 0 else 0.0
    elif kind == 6:
        for i in range(n): h[i] = 1.0 if np.sin(2*np.pi*2.0*i/n) > np.sin(2*np.pi*7.0*i/n) else 0.0
    else:
        h[::3] = 1.0
    h = _repair(h, n)
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    return h

def _anneal(h, t0, t1, n_steps, rng):
    nc = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, nc)
    c5c = _eval(h)
    best = c5c
    for step in range(n_steps):
        t = t0 * (t1 / t0) ** (step / max(n_steps - 1, 1))
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j or abs(h[i] - h[j]) < 0.1: continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        h2 = _repair(h, nc)
        c5n = _eval(h2)
        if c5n < c5c or rng.rand() < np.exp(-(c5n - c5c) / max(t, 1e-7)):
            c5c = c5n
            h = h2
            if c5c < best: best = c5c
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            h = _repair(h, nc)
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)
    best_n = n0
    c5_init = best_c5

    # Phase 1: Coarse-to-fine (150s)
    for nc in [30, 50, 70, 100, 130, 170, 220]:
        if elapsed() > 130: break
        hc = _repair(_resample(h_init, nc), nc)
        for p in [4, 16]:
            if elapsed() > 130: break
            hc = _slsqp_pnorm(hc, p, 400, 1e-15)
        c5c = _eval(hc)
        if c5c < best_c5 * 1.01:
            h_ref = _repair(_resample(hc, n0), n0)
            for p in [8, 32]:
                if elapsed() > 130: break
                h_ref = _slsqp_pnorm(h_ref, p, 400, 1e-16)
            c5r = _eval(h_ref)
            if c5r < best_c5:
                best_c5, best_h, best_n = c5r, h_ref.copy(), n0

    # Phase 2: P-norm sweep (250s)
    if elapsed() < budget_s - 350:
        hp = best_h.copy()
        for p in [4, 8, 16, 32, 64, 128, 256]:
            if elapsed() > budget_s - 350: break
            hp = _slsqp_pnorm(hp, p, 4000, 1e-19)
            c5 = _eval(hp)
            if c5 < best_c5 - 1e-14:
                best_c5, best_h = c5, hp.copy()
            # Try complement
            hc = _slsqp_pnorm(_repair(1.0 - best_h, n0), p, 2000, 1e-18)
            c5c = _eval(hc)
            if c5c < best_c5 - 1e-14:
                best_c5, best_h = c5c, hc.copy()

    # Phase 3: Top-k gradient focus (200s)
    if elapsed() < budget_s - 250:
        for top_k in [1, 2, 3, 5]:
            if elapsed() > budget_s - 250: break
            h3 = best_h.copy()
            h3, c5_3 = _topk_descent(h3, 40, top_k, rng)
            h3 = _slsqp_pnorm(h3, 64, 3000, 1e-19)
            c5_3 = _eval(h3)
            if c5_3 < best_c5 - 1e-14:
                best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Diverse starts (200s)
    if elapsed() < budget_s - 200:
        for kind in range(7):
            if elapsed() > budget_s - 200: break
            hp = _make_pattern(n0, kind, rng)
            for p in [4, 16, 48]:
                if elapsed() > budget_s - 200: break
                hp = _slsqp_pnorm(hp, p, 2500, 1e-17)
            c5 = _eval(hp)
            if c5 < best_c5 - 1e-14:
                best_c5, best_h = c5, hp.copy()

    # Phase 5: Anneal + polish (180s)
    if elapsed() < budget_s - 100:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = _repair(h_bin, n0)
        t_rem = max(budget_s - 100 - elapsed(), 10)
        n_steps = max(int(t_rem * 600), 6000)
        h_a, c5_a = _anneal(h_bin, 0.3, 0.0005, n_steps, rng)
        h_a = _slsqp_pnorm(h_a, 32, 3000, 1e-18)
        c5_a = _eval(h_a)
        if c5_a < best_c5 - 1e-14:
            best_c5, best_h = c5_a, h_a.copy()

    # Phase 6: Final high-p polish + micro (80s)
    if elapsed() < budget_s - 30:
        for p in [128, 256]:
            if elapsed() > budget_s - 45: break
            h6 = _slsqp_pnorm(best_h, p, 6000, 1e-20)
            c5 = _eval(h6)
            if c5 < best_c5 - 1e-15:
                best_c5, best_h = c5, h6.copy()
        # Micro coordinate descent
        h8 = best_h.copy()
        c5c = _eval(h8)
        nc = len(h8)
        t_end = budget_s - 3
        for _it in range(max(int((t_end - elapsed()) * 400), 500)):
            if elapsed() > t_end: break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j: continue
            d = 0.0008 * rng.randn()
            ni, nj = h8[i] + d, h8[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
            h8[i], h8[j] = ni, nj
            c5n = _eval(h8)
            if c5n < c5c - 1e-17:
                c5c = c5n
            else:
                h8[i], h8[j] = ni - d, nj + d
        if c5c < best_c5 - 1e-15:
            best_c5, best_h = c5c, h8.copy()

    # Final: try a few perturbations of best
    if elapsed() < budget_s - 5:
        for _trial in range(3):
            if elapsed() > budget_s - 5: break
            hp = best_h + 0.01 * rng.randn(n0)
            hp = _repair(hp, n0)
            hp = _slsqp_pnorm(hp, 64, 2000, 1e-19)
            c5t = _eval(hp)
            if c5t < best_c5 - 1e-15:
                best_c5, best_h = c5t, hp.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
