    c = np.correlate(x, 1.0 - x, mode='full') * dx

            g[:n-sl] -= wk * x[sl:]

        x = _repair(res.x, n)
    return x

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj_grad(x, alpha, pen):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = np.exp(s - s.max()); sm_sum = sm.sum()
    w = sm / sm_sum
    obj = m + np.log(sm_sum) / alpha + pen * (x.sum() - 0.5 * n) ** 2
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = (n - 1) - k
        wk = w[k] * dx
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - x[lag:n])
            g[lag:] -= wk * x[:n - lag]
        else:
            sl = -lag
            g[sl:] += wk * (1.0 - x[:n - sl])
            g[:n - sl] -= wk * x[sl:]
    g += 2.0 * pen * (x.sum() - 0.5 * n)
    return obj, g

                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        x = _repair(res.x, n)
    return x

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_period(h_p, m, alphas, maxiter):
    T = len(h_p)
    n = T * m
    bnds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    def obj_p(v):
        t = np.tile(v, m)
        return _obj_grad(t, alphas, 0.0)[0]
    def grad_p(v):
        t = np.tile(v, m)
        _, gf = _obj_grad(t, alphas, 0.0)
        gv = np.zeros(T)
        for p in range(m):
            gv += gf[p * T:(p + 1) * T]
        return gv
    try:
        res = minimize(obj_p, h_p, jac=grad_p, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, T)
    except Exception:
        return h_p
