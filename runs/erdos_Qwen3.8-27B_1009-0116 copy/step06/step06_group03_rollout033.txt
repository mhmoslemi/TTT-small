```python
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
```

4. Update h₀
5. Repeat

```python
def _slp_refine(h, n_iters, top_lags):
    """Sequential Linear Programming refinement."""
    n = len(h)
    for it in range(n_iters):
        c = np.correlate(h, 1.0-h, mode='full') * 2.0/n
        # Find top lags
        top_idx = np.argsort(c)[-top_lags:]
        
        # Build LP: min t s.t. linearized f_k <= t
        # Variables: h[0..n-1], t
        # Objective: minimize t
        # Constraints:
        #   For k in top_lags: f_k_linearized(h) <= t
        #   0 <= h[i] <= 1
        #   sum(h) = n/2
        
        from scipy.optimize import linprog
        c_obj = np.zeros(n+1); c_obj[-1] = 1.0
        
        A_ub = []; b_ub = []
        for k in top_idx:
            sh = (n-1) - k
            grad = np.zeros(n)
            if sh >= 0:
                grad[:n-sh] += (1.0 - h[sh:])
                if sh > 0: grad[sh:] -= h[:n-sh]
            else:
                sp = -sh
                grad[sp:] += (1.0 - h[:n-sp])
                grad[:n-sp] -= h[sp:]
            grad *= 2.0/n
            f_val = c[k]  # current value at this lag
            # f_k(h) ≈ f_val + grad·(h - h_old) <= t
            # grad·h <= t - f_val + grad·h_old
            # grad·h - t <= -f_val + grad·h_old
            A_ub.append(np.append(grad, -1.0))
            b_ub.append(-f_val + np.dot(grad, h))
        
        A_ub = np.array(A_ub); b_ub = np.array(b_ub)
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        
        bounds = [(0,1)]*n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_new = res.x[:n]
            h_new = _repair(h_new, n)
            if _eval(h_new) < _eval(h):
                h = h_new
        else:
            break
    return h
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        lo, hi = -1.0, 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5 * (lo + hi), 0, 1)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    m = c.max()

    pk = 5e5 * (h.sum() - n / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + pk

                       options={'maxiter': maxiter, 'ftol': ftol})

            h_new = _repair(best - step * f * v, n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best = h_new; best_c = c_new
                step = min(step * 1.08, 0.25)
                improved = True; stuck = 0
                break
        if not improved:
            step *= 0.65; stuck += 1
            if stuck > 60:
                best = _repair(best + 0.015 * rnd.randn(n), n)
                best_c = _eval(best); step = step0; stuck = 0; v = np.zeros(n)
    return best, best_c

def _fine_search(h, n_tries, rnd):
    n = len(h); h = h.copy()
    c_cur = _eval(h); best = h.copy(); best_c = c_cur
    for _ in range(n_tries):
        i = rnd.randint(n); j = rnd.randint(n)
        if i == j:
            continue
        d = 2e-5 * rnd.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el():
        return time.time() - t0
    rnd = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy(); best_c = _eval(best_h); best_n = n0

    # Phase 1: SLP for fast structured improvement
    if _el() < 200:
        h_slp, c_slp = _slp_refine(best_h, 300, 12)
        if c_slp < best_c:
            best_c = c_slp; best_h = h_slp

    # Phase 2: SLSQP with smooth max
    for alpha in [400, 1500, 8000, 40000]:
        if _el() > 250:
            break
        hc = _slsqp_opt(best_h, alpha, 1200, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 3: SLP again with more lags
    if _el() < 400:
        h_slp, c_slp = _slp_refine(best_h, 250, 18)
        if c_slp < best_c:
            best_c = c_slp; best_h = h_slp

    # Phase 4: Subgradient exploration
    if _el() < budget_s - 350:
        tn = min(4000, int((budget_s - 350 - _el()) * 30))
        best_h, best_c = _subgrad(best_h, tn, 4, 0.018, rnd)

    # Phase 5: SLP refinement
    if _el() < budget_s - 250:
        h_slp, c_slp = _slp_refine(best_h, 200, 22)
        if c_slp < best_c:
            best_c = c_slp; best_h = h_slp

    # Phase 6: SLSQP high alpha
    for alpha in [100000, 400000]:
        if _el() > budget_s - 150:
            break
        hc = _slsqp_opt(best_h, alpha, 1500, 1e-17)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 7: Random restarts + SLP
    if _el() < budget_s - 120:
        for trial in range(3):
            if _el() > budget_s - 120:
                break
            h_try = _repair(best_h + 0.008 * rnd.randn(best_n), best_n)
            h_slp, c_slp = _slp_refine(h_try, 150, 15)
            if c_slp < best_c:
                best_c = c_slp; best_h = h_slp

    # Phase 8: Final SLP
    if _el() < budget_s - 60:
        h_slp, c_slp = _slp_refine(best_h, 150, 25)
        if c_slp < best_c:
            best_c = c_slp; best_h = h_slp

    # Phase 9: Final SLSQP
    if _el() < budget_s - 30:
        hc = _slsqp_opt(best_h, 800000, 2000, 1e-18)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 10: Fine coordinate search
    if _el() < budget_s - 8:
        n_co = min(int((budget_s - 8 - _el()) * 400), 60000)
        best_h, best_c = _fine_search(best_h, n_co, rnd)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
```
