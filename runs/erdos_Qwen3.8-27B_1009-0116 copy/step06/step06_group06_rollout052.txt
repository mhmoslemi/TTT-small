import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(h))

def _smax_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    return g

def _lbfgs(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def c_fun(x):
        return x.sum() - n * 0.5
    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    f_start = _smax_obj(h, alpha)
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxfun': maxiter * 5})
        if res.fun < f_start - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _momentum_grad(h, alpha, n_steps, t_end, lr0, mu):
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    lr = lr0
    f_prev = _smax_obj(h_cur, alpha)
    for s in range(n_steps):
        if time.time() > t_end:
            break
        g = _smax_grad(h_cur, alpha)
        v = mu * v - lr * g
        h_new = _proj(h_cur + v, n * 0.5)
        f_new = _smax_obj(h_new, alpha)
        if f_new < f_prev - 1e-15:
            h_cur = h_new
            f_prev = f_new
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.5
            v *= 0.3
    return h_cur

def _topk_grad_descent(h, n_rounds, t_end):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(c)[::-1][:8]
        g = np.zeros(n)
        for kk in top_idx:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += (1.0 - h_cur[sh:])
                if sh > 0:
                    g[sh:] -= h_cur[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h_cur[:n - sp])
                g[:n - sp] -= h_cur[sp:]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5_cur

def _coord_fine(h, n_tries, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    c5c = _eval(h_cur)
    for _it in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        scale = 3e-5 + 2e-4 * rng.rand()
        d = scale * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _eval(h_cur)
        if c5n < c5c - 1e-17:
            c5c = c5n
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
    return h_cur, c5c

def _make_diverse(h, n, rng, idx):
    if idx == 0:
        return _repair(h + 0.01 * rng.randn(n), n)
    elif idx == 1:
        shift = rng.randint(1, n // 4)
        return _repair(np.roll(h, shift), n)
    elif idx == 2:
        h2 = h.copy()
        block = n // 6
        st = rng.randint(0, n - block)
        h2[st:st + block] = 1.0 - h2[st:st + block]
        return _repair(h2, n)
    else:
        h3 = h.copy()
        h3 += 0.03 * rng.randn(n)
        return _repair(h3, n)
