def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

    """Compute sum_k ws[k] * grad_k(h) where ws is a weight vector over lags."""
    n = len(h)
    g = np.zeros(n)
    for k in range(len(ws)):
        if ws[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = ws[k] * dx
        if sh >= 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _lsexp_fun(x, alpha, lam):
    c = _correlate(x)

    s = np.clip(alpha * (c - mc), -300, 300)
    lse = mc + np.log(float(np.sum(np.exp(s)))) / alpha
    pen = lam * (x.sum() - n * 0.5) ** 2
    return lse * (2.0 / n) + pen

def _lsexp_grad(x, alpha, lam):
    n = len(x)

    c = _correlate(x)
    mc = c.max()
    s = np.clip(alpha * (c - mc), -300, 300)
    w = np.exp(s)
    w /= w.sum()
    g = _grad_all(x, w, dx)
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def _linedescent(h, g, n, c5c, max_iter=30):
    """Steepest descent with backtracking for C5 = max lags."""
    best_c5 = c5c
    best_h = h.copy()
    step = 1.0
    for _ in range(max_iter):
        step *= 0.7
        h_new = _repair(h - step * g, n)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
            best_h = h_new
        if step < 1e-8:
            break
    return best_h, best_c5

def _focused(h, n, max_rounds, top_m=5):
    c5c = _eval(h)
    for r in range(max_rounds):
        c = _correlate(h)
        # Top M lags
        m = min(top_m, len(c))
        top_idx = np.argsort(c)[-m:]
        ws = np.zeros(len(c))
        # Weight by how close to max
        for idx in top_idx:
            ws[idx] = 1.0
        ws /= m
        g = _grad_all(h, ws, 1.0)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h, c5c = _linedescent(h, g, n, c5c)
    return h, c5c

def _lbfgsb(h, alpha, lam, maxiter):

    bnds = [(0.0, 1.0)] * n
    try:

    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

    """Local search on sequence: flip single elements or blocks."""
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    c5c = _eval(h)
    improved = 0
    for _ in range(n_iter):
        # Random move: flip element or block
        r = rng.random()
        if r < 0.5:
            i = rng.randint(n)
            delta = np.zeros(n)
            delta[i] = 1.0 - 2.0 * h[i]  # flip
        else:
            bs = block_sizes[rng.randint(len(block_sizes))]
            i = rng.randint(n - bs)
            delta = np.zeros(n)
            # Flip a block: h[i:i+bs] -> 1 - h[i:i+bs]
            seg = h[i:i+bs]
            delta[i:i+bs] = 1.0 - 2.0 * seg
        h_new = _repair(h + delta, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-14:
            h = h_new
            c5c = c5n
            improved += 1
        if improved > 5000:
            break
    return h, c5c

def _pairwise_cd(h, n, rng, scale, n_iter):
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)

        d = scale * rng.randn()

        if c5n < c5c - 1e-16:

        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_all(h, ws, dx):
    n = len(h)
    g = np.zeros(n)
    for k in range(len(ws)):
        wv = ws[k]
        if wv < 1e-14:
            continue
        sh = (n - 1) - k
        wk = wv * dx
        if sh >= 0:
            if sh < n:
                g[:n - sh] += wk * (1.0 - h[sh:n])
                if sh > 0:
                    g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lsexp_fun(x, alpha, lam):
    n = len(x)
    c = _correlate(x)
    mc = c.max()
    s = np.clip(alpha * (c - mc), -300, 300)
    lse = mc + np.log(float(np.sum(np.exp(s)))) / alpha
    dev = (x.sum() - n * 0.5) / n
    return lse * (2.0 / n) + lam * dev * dev

def _lsexp_grad(x, alpha, lam):
    n = len(x)
    dx = 2.0 / n
    c = _correlate(x)
    mc = c.max()
    s = np.clip(alpha * (c - mc), -300, 300)
    w = np.exp(s)
    w /= w.sum()
    g = _grad_all(x, w, dx)
    dev = (x.sum() - n * 0.5) / n
    g += 2.0 * lam * dev / n
    return g

def _linedescent(h, g, n, c5c, max_iter=30):
    best_c5 = c5c
    best_h = h.copy()
    step = 1.0
    for _ in range(max_iter):
        step *= 0.7
        h_new = _repair(h - step * g, n)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
            best_h = h_new
        if step < 1e-8:
            break
    return best_h, best_c5

def _focused(h, n, max_rounds, top_m=5):
    c5c = _eval(h)
    for r in range(max_rounds):
        c = _correlate(h)
        m = min(top_m, len(c))
        top_idx = np.argsort(c)[-m:]
        ws = np.zeros(len(c))
        ws[top_idx] = 1.0 / m
        g = _grad_all(h, ws, 1.0)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h, c5c = _linedescent(h, g, n, c5c)
    return h, c5c

def _lbfgsb(h, alpha, lam, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lsexp_fun, h, jac=lambda x: _lsexp_grad(x, alpha, lam),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-18,
                                'maxls': 50, 'max_fun': maxiter * 5})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _binary_ls(h, n, rng, n_iter):
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    c5c = _eval(h)
    for _ in range(n_iter):
        r = rng.random()
        if r < 0.4:
            i = rng.randint(n)
            delta = np.zeros(n)
            delta[i] = 1.0 - 2.0 * h[i]
        elif r < 0.7:
            bs = [1, 2, 3, 5, 8][rng.randint(5)]
            if n < bs:
                continue
            i = rng.randint(n - bs)
            delta = np.zeros(n)
            delta[i:i + bs] = 1.0 - 2.0 * h[i:i + bs]
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            delta = np.zeros(n)
            delta[i] = 1.0 - 2.0 * h[i]
            delta[j] = 2.0 * h[j] - 1.0
        h_new = _repair(h + delta, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-14:
            h = h_new
            c5c = c5n
    return h, c5c

def _pairwise_cd(h, n, rng, scale, n_iter):
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d + d, nj + d - d
            h[i] = ni - d
            h[j] = nj + d
    return h, c5c

def _fourier_init(K, n, rng, amp=0.35):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        a = rng.uniform(-amp, amp)
        b = rng.uniform(-amp, amp)
        h += a * np.cos(np.pi * j * x) + b * np.sin(np.pi * j * x)
    return _repair(h, n)
