def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for i in idx:
                a = min(abs(diff), 1.0 - h[i]) if diff > 0 else min(abs(diff), h[i])
                h[i] += a if diff > 0 else -a
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_worst(h):

    else:
        sp = -sh
        g[sp:] += dx * (1.0 - h[:n-sp])
        g[:n-sp] -= dx * h[sp:]
    return g

def _bb_gradient(h, n, max_iter=5000, base_lr=0.08):

        g = _grad_worst(h)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        if it > 5:

            dg = g - g_prev

def _slsqp_smooth(h, alpha, maxiter=2000):

    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        sm = s - s.max()
        return m + (s.max() + np.log(np.sum(np.exp(sm)))) / alpha
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx

                g[:n-sh] += wk * (1.0 - x[sh:])

            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)

                best_c5, best_h, best_n = c_r, h_r.copy(), n_try
                h_b = (h_r > 0.5).astype(float)
                h_b = _repair(h_b, n_try)
                h_b, c_b = _binary_ls(h_b, n_try, max_rounds=30, n_trial=2000, rng=rng)
                if c_b < best_c5:
                    best_c5, best_h, best_n = c_b, h_b.copy(), n_try
            h_r2 = _slsqp_smooth(h_r, 500000, 800)
            c_r2 = _c5(h_r2)
            if c_r2 < best_c5:
                best_c5, best_h, best_n = c_r2, h_r2.copy(), n_try

    # Phase 4: Intensive SLSQP with high alpha
    if tleft() > 180:
        for alpha in [200000, 1000000, 5000000]:
            if tleft() < 80:
                break
            h_s = _slsqp_smooth(best_h, alpha, 3000)
            c_s = _c5(h_s)
            if c_s < best_c5:
                best_c5, best_h = c_s, h_s.copy()
            h_c = _repair(1.0 - best_h, best_n)
            h_c = _slsqp_smooth(h_c, alpha, 2000)
            c_c = _c5(h_c)
            if c_c < best_c5:
                best_c5, best_h = c_c, h_c.copy()

    # Phase 5: BB gradient + binary alternation
    if tleft() > 120:
        for cyc in range(8):
            if tleft() < 60:
                break
            h_bb, c_bb = _bb_gradient(best_h, best_n, max_iter=2000, base_lr=0.04)
            if c_bb < best_c5:
                best_c5, best_h = c_bb, h_bb.copy()
            h_b = (best_h > 0.5).astype(float)
            h_b = _repair(h_b, best_n)
            h_b, c_b = _binary_ls(h_b, best_n, max_rounds=40, n_trial=2500, rng=rng)
            if c_b < best_c5:
                best_c5, best_h = c_b, h_b.copy()
            if tleft() > 80:
                h_s = _slsqp_smooth(best_h, 500000, 1500)
                c_s = _c5(h_s)
                if c_s < best_c5:
                    best_c5, best_h = c_s, h_s.copy()

    # Phase 6: Final coordinate descent
    if tleft() > 15:
        h8 = best_h.copy()
        c8 = _c5(h8)
        nc = best_n
        for scale in [0.02, 0.008, 0.002, 0.0005]:
            if tleft() < 8:
                break
            for _ in range(4000):
                if tleft() < 4:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                if 0 <= h8[i] + d <= 1 and 0 <= h8[j] - d <= 1:
                    h8[i] += d
                    h8[j] -= d
                    cn = _c5(h8)
                    if cn < c8:
                        c8 = cn
                    else:
                        h8[i] -= d
                        h8[j] += d
        if c8 < best_c5:
            best_c5, best_h = c8, h8.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h)
    return (best_h, best_c5, best_n)
