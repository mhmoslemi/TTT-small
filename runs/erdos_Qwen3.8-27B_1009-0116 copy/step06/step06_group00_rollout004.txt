        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-9:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _eval(h)
    return _repair(h, n)

def _slsqp_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * 2.0 / n
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    import functools
    obj_f = functools.partial(_slsqp_obj, alpha=alpha, n=n)
    grad_f = functools.partial(_slsqp_grad, alpha=alpha, n=n)
    def _cons_fun(x):
        return x.sum() - target
    def _cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(obj_f, h, jac=grad_f,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search_3pt(h, n_tries, rng, t0, budget_s):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() - t0 > budget_s - 10:
            break
        i = rng.randint(n)
        j = (i + rng.randint(2, 8)) % n
        k = (i + rng.randint(9, 18)) % n
        d1 = 3e-5 * rng.randn()
        d2 = 3e-5 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _local_search_2pt(h, n_tries, rng, t0, budget_s):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() - t0 > budget_s - 5:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 8e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _upsample(h, f):
    return np.repeat(h, f).copy()

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Fast top-k Nesterov (k=3, cheap gradients)
    h = _nesterov_topk(h, 3, 8000, 0.002, t0, 100)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5, best_h = c5v, h.copy()

    # Phase 2: Top-k=10 Nesterov
    h = _nesterov_topk(best_h.copy(), 10, 5000, 0.001, t0, 200)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5, best_h = c5v, h.copy()

    # Phase 3: SLSQP refinement
    for alpha in [5000, 50000, 200000]:
        if time.time() - t0 > 400:
            break
        h_c = _slsqp_opt(best_h.copy(), alpha, 1500, 1e-17)
        c5v = _eval(h_c)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_c.copy()

    # Phase 4: Top-k=20 Nesterov (more faithful max approximation)
    if time.time() - t0 < budget_s - 300:
        h = _nesterov_topk(best_h.copy(), 20, 4000, 0.0008, t0, budget_s - 280)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 5: Low-res escape (try n/2 to find different basin)
    if time.time() - t0 < budget_s - 250:
        n_lo = best_n if 'best_n' in dir() else len(best_h)
        n_lo = len(best_h) // 2
        if n_lo >= 80:
            h_lo = _downsample(best_h, len(best_h) // n_lo)
            h_lo = _repair(h_lo, n_lo)
            for kt in [5, 15]:
                h_lo = _nesterov_topk(h_lo, kt, 3000, 0.002, t0, budget_s - 120)
            h_hi = _repair(_upsample(h_lo, len(best_h) // n_lo), len(best_h))
            c5v = _eval(h_hi)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_hi.copy()

    # Phase 6: High-alpha SLSQP
    if time.time() - t0 < budget_s - 120:
        h_c = _slsqp_opt(best_h.copy(), 500000, 1500, 1e-18)
        c5v = _eval(h_c)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_c.copy()
        if time.time() - t0 < budget_s - 80:
            h_c = _slsqp_opt(best_h.copy(), 2000000, 1500, 1e-18)
            c5v = _eval(h_c)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_c.copy()

    # Phase 7: Final Nesterov polish
    if time.time() - t0 < budget_s - 50:
        h = _nesterov_topk(best_h.copy(), 50, 2000, 0.0003, t0, budget_s - 30)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 8: 3-point local search
    if time.time() - t0 < budget_s - 20:
        n_tries = min(int((budget_s - 20 - (time.time() - t0)) / 0.004), 40000)
        best_h, best_c5 = _local_search_3pt(best_h, n_tries, rng, t0, budget_s)

    # Phase 9: 2-point ultra-fine polish
    if time.time() - t0 < budget_s - 3:
        n_tries = min(int((budget_s - 3 - (time.time() - t0)) / 0.002), 35000)
        best_h, best_c5 = _local_search_2pt(best_h, n_tries, rng, t0, budget_s)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
