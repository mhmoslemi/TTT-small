import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _eval_cvec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _proj(h, n):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > n * 0.5:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    h_cur = h.copy()

    mu = 0.8
    recompute = 8
    top_lags = [n - 1]
    for it in range(n_steps):
        if time.time() > t_end:
            break
        if it % recompute == 0:
            c = _eval_cvec(h_cur)
            c5_cur = float(c.max())
            if c5_cur < best_c:
                best_c = c5_cur
                best = h_cur.copy()
            top_lags = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for idx in top_lags:
            g += _grad_lag(h_cur, idx)
        g /= len(top_lags)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        v = mu * v + (1.0 - mu) * (g / gnorm)
        alpha = base_step * (1.0 + 0.15 * np.cos(it * 0.02))
        improved = False
        for _ in range(10):
            h_new = _proj(h_cur - alpha * v, n)
            c5_new = _eval(h_new)

                c5_cur = c5_new
                improved = True
                break

    return np.ones(n)

    bnds = [(0.0, 1.0)] * n
