def _s_obj(x, T, m, alpha):
    hf = np.tile(x, m)
    n = len(hf)
    c = np.correlate(hf, 1.0 - hf, mode='full')
    cm = c.max()
    s = np.clip(alpha * (c - cm) / n, -500, 500)
    return cm * 2.0 / n + np.log(float(np.sum(np.exp(s)))) / alpha * 2.0 / n

def _s_grad(x, T, m, alpha):
    hf = np.tile(x, m)
    n = len(hf)
    c = np.correlate(hf, 1.0 - hf, mode='full')
    cm = c.max()
    s = np

.clip(alpha * (c - cm) / n, -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - hf[sh:])
            if sh > 0:
                g[sh:] -= wk * hf[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - hf[:n - sp])
            g[:n - sp] -= wk * hf[sp:]
    gp = np.zeros(T)
    for p in range(m):
        gp += g[p * T:(p + 1) * T]
    return gp / m

def _coord_descent(h, n_steps, scale, deadline, seed):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        i = rng.randint(len(h))
        j = rng.randint(len(h))
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        diff = tgt - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, T):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = T * 0.5 - h.sum()
        if abs(d) < 1e-15:
            break
        h += d / T
    return np.clip(h, 0.0, 1.0)

def _pvals(h_p):
    T = len(h_p)
    vals = np.empty(T)
    for k in range(T):
        vals[k] = np.dot(h_p, 1.0 - np.roll(h_p, -k))
    return vals

def _pgrad_smooth(h_p, beta):
    T = len(h_p)
    vals = _pvals(h_p)
    m = vals.max()
    scale = max(1e-6, m * 0.01)
    w = np.exp(beta * (vals - m) / scale)
    w /= w.sum()
    g = np.zeros(T)
    for k in range(T):
        if w[k] < 1e-10:
            continue
        if k == 0:
            g += w[k] * (1.0 - 2.0 * h_p)
        else:
            g += w[k] * ((1.0 - np.roll(h_p, -k)) - np.roll(h_p, k))
    return (2.0 / T) * g

def _nesterov(h_p, niter, g0, beta, deadline):
    T = len(h_p)
    x = h_p.copy()
    y = h_p.copy()
    xp = h_p.copy()
    bc = float(_pvals(x).max()) * 2.0 / T
    bx = x.copy()
    best_t = 0
    for t in range(1, niter + 1):
        if time.time() > deadline:
            break
        g = _pgrad_smooth(y, beta)
        g -= g.mean()
        gam = g0 / np.sqrt(t + 1.0)
        xn = _proj(y - gam * g, T)
        mu = t / float(t + 2)
        yn = _proj(xn + mu * (xn - xp), T)
        xp, x, y = x, xn, yn
        c = float(_pvals(x).max()) * 2.0 / T
        if c < bc:
            bc, bx, best_t = c, x.copy(), t
        if t % 80000 == 0:
            g0 *= 0.85
            if t > best_t + 80000:
                g0 = min(g0 * 1.5, 2.0)
    return bx, bc

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        diff = tgt - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _s_obj(x, T, m, alpha):
    hf = np.tile(x, m)
    n = len(hf)
    c = np.correlate(hf, 1.0 - hf, mode='full')
    cm = c.max()
    s = np.clip(alpha * (c - cm) / n, -500, 500)
    return cm * 2.0 / n + np.log(float(np.sum(np.exp(s)))) / (alpha * n)

def _s_grad(x, T, m, alpha):
    hf = np.tile(x, m)
    n = len(hf)
    c = np.correlate(hf, 1.0 - hf, mode='full')
    cm = c.max()
    s = np.clip(alpha * (c - cm) / n, -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - hf[sh:])
            if sh > 0:
                g[sh:] -= wk * hf[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - hf[:n - sp])
            g[:n - sp] -= wk * hf[sp:]
    gp = np.zeros(T)
    for p in range(m):
        gp += g[p * T:(p + 1) * T]
    return gp / m

def _con_fun(x):
    return x.sum() - len(x) * 0.5

def _con_jac(x):
    return np.ones(len(x))

def _coord(h, n_steps, scale, deadline, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj
    return h, c5c
