Approach and rationale
We address the ines capability of gradient-based and SLSQP methods to escape local optima by combining several novel techniques: (1) inverse least-squares manifold projection, (2) multi-resolution dimensionality reduction, (3) worst-shift detection, and (4) kernel-smoothed gradient following. The key bottleneck is the curse of dimensionality in non-convex optimization; our approach reduces the effective degrees of freedom by projecting the solution along dominant harmonic components while preserving integral constraints. This allows us to escape local minima that trap gradient-only methods by exploring a quasi-one-dimensional manifold.

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - n/2.0) < 1e-12:
        return h
    # Projection scaling
    total_area = n/2.0  # satisfies ∫h dx=1
    h = h * total_area / s
    h = np.clip(np.clip(h, 0, 1), 0, 1)
    s_after = h.sum()
    # Integral restoration
    diff = total_area - s_after
    while abs(diff) > 1e-5:
        idx = np.argmax(np.abs(h.clip(0,1) - 0.5))  # median weighted choice
        if diff > 1e-5:
            h[idx] += min(diff, 1.0-h[idx])
        else:
            h[idx] -= max(-diff, h[idx])
        diff -= (1 if min(diff,1) else -1) if abs(diff)>1 else diff
        diff = total_area - h.sum()
    return h

def calculate_ghost_gradients(h, lag_range=50):
    n = len(h)
    corr_full = np.correlate(h, 1-h, mode='full')
    grad_direction = np.zeros(len(h))
    # Reject implementable lags
    lags = []
    for lag in range(1, len(corr_full)//2):
        if corr_full[lag] < 0: continue
        # Reject implausible correlations
        if abs(corr_full[lag]) > 1: continue
        lags.append(lag)
    
    # If no valid lags, use heuristics
    if not lags:
        lags = [len(corr_full)//4]
    
    # Most damaging lag
    dominant_lag = lags[np(argmin(corr_full[lags]))
    baseline = corr_full[dominant_lag]
    if baseline < -0.2: 
        dominant_lag = 0
        baseline = 0
    
    # Directional gradient calculation
    for lag in range(dominant_lag-5, dominant_lag+5):
        if lag < 0 or lag >= len(corr_full): continue
        shifted_overlap = np.correlate(h, 1-np.roll(h, lag), mode='valid')
        grad_direction[:len(shifted_overlap)] += (correlation[dominant_lag] - shifted_overlap) * (1 if lag>0 else -1)
    
    return baseline, grad_direction
