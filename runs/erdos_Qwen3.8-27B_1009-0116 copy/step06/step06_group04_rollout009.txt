So the time is mostly spent at low alpha (where all lags contribute) and decreases as alpha increases. This is actually good: the early stages are expensive but the later stages (which matter most for precision) are fast.

2. Multi-start with different initializations
3. L-BFGS-B with penalty for the sum constraint (faster than SLSQP)
4. A "top-flattening" objective variant
5. Fine-grained coordinate descent at the end

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _slsqp_run(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _adam(h0, a_sched, iters_per, lr_start, lr_end, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_x = x.copy()
    best_c = _eval(x)
    no_improve = 0
    total_iters = len(a_sched) * iters_per
    for stage, (a0, a1) in enumerate(a_sched):
        for i in range(iters_per):
            if time.time() > deadline:
                break
            t += 1
            frac = (stage * iters_per + i) / max(total_iters - 1, 1)
            alpha = a0 * (a1 / a0) ** (i / max(iters_per - 1, 1))
            lr = lr_start * (lr_end / lr_start) ** frac
            if t % 800 == 0 and no_improve > 400:
                x = x + 0.01 * rng.randn(n)
            g = _grad(x, alpha)
            m_arr = b1 * m_arr + (1 - b1) * g
            v_arr = b2 * v_arr + (1 - b2) * g * g
            mhat = m_arr / (1 - b1 ** t)
            vhat = v_arr / (1 - b2 ** t)
            x = x - lr * mhat / (np.sqrt(vhat) + eps)
            x = _repair(x, n)
            c = _eval(x)
            if c < best_c:
                best_c = c
                best_x = x.copy()
                no_improve = 0
            else:
                no_improve += 1
    return best_x, best_c

def _lbfgs_run(h, alpha, penalty, maxiter, deadline):
    n = len(h)
    def _pobj(x):
        return _obj(x, alpha) + penalty * (x.sum() - n * 0.5) ** 2
    def _pgrad(x):
        return _grad(x, alpha) + 2.0 * penalty * (x.sum() - n * 0.5)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pobj, h, jac=_pgrad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                             'gtol': 1e-14, 'maxfun': maxiter * 2})
        if time.time() > deadline:
            return h
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _paired_cd(h, n, rng, deadline, scale_init, max_steps):
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    scale = scale_init
    steps = 0
    while steps < max_steps and time.time() < deadline:
        if scale < 1e-10:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        steps += 1
        d = scale * rng.randn()
        x[i] += d
        x[j] -= d
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_x = x.copy()
            c_cur = c_new
            scale = max(scale * 0.9995, 1e-11)
        else:
            x[i] -= d
            x[j] += d
            x = _repair(x, n)
            c_cur = _eval(x)
            scale *= 0.97
    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _block_cd(h, n, block_size, rng, deadline, n_blocks):
    x = h.copy()
    best_x = x.copy()
    best_c = _eval(x)
    for _ in range(n_blocks):
        if time.time() > deadline:
            break
        t = rng.randint(n - block_size)
        for _ in range(200):
            if time.time() > deadline:
                break
            for i in range(t, t + block_size):
                d = 0.005 * rng.randn()
                x[i] += d
            x = _repair(x, np.block_size if hasattr(np, 'block_size') else n)
            x = _repair(x, len(x))
            c = _eval(x)
            if c < best_c:
                best_c = c
                best_x = x.copy()
            else:
                for i in range(t, t + block_size):
                    x[i] -= d
                x = _repair(x, len(x))
    return best_x, best_c
