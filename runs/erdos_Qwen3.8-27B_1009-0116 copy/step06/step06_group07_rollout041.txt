import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _peq_obj(x, alpha, n, T):
    h_f = np.tile(x, n // T)
    c = np.correlate(h_f, 1.0 - h_f, mode='full') * (2.0 / n)
    mm = c.max()
    s = np.clip(alpha * (c - mm), -300, 0)
    return mm + np.log(np.sum(np.exp(s))) / alpha

def _peq_grad(x, alpha, n, T):
    h_f = np.tile(x, n // T)
    nF = n
    dx = 2.0 / nF
    c_raw = np.correlate(h_f, 1.0 - h_f, mode='full')
    c = c_raw * dx
    mm = c.max()
    s = np.clip(alpha * (c - mm), -300, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(nF)
    for k in range(2 * nF - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        sh = (nF - 1) - k
        if sh >= 0:
            idx = np.arange(nF - sh)
            g[idx] += wk * (1.0 - h_f[idx + sh])
            if sh > 0:
                g[idx + sh] -= wk * h_f[idx]
        else:
            sp = -sh
            idx = np.arange(sp, nF)
            g[idx] += wk * (1.0 - h_f[idx - sp])
            g[idx - sp] -= wk * h_f[idx]
    return g.reshape(n // T, T).mean(axis=0)

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_period(h_p, n, T, alpha, maxiter):
    bnds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_peq_obj, h_p, jac=_peq_grad, args=(alpha, n, T),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_p = _repair(res.x, T)
        return h_p
    except Exception:
        return _repair(h_p, T)

def _slsqp_full(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_peq_obj, h, jac=_peq_grad, args=(alpha, n, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pgd_period(h_p, n, alphas, total_iters, lr_init, seed):
    T = len(h_p)
    m = n // T
    rng = np.random.RandomState(seed)
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    lr = lr_init
    mom = np.zeros(T)
    beta = 0.8
    iters_per_alpha = total_iters // len(alphas)
    for alpha in alphas:
        for it in range(iters_per_alpha):
            h_f = np.tile(h_p, m)
            c_raw = np.correlate(h_f, 1.0 - h_f, mode='full')
            dx = 2.0 / n
            c = c_raw * dx
            mm = c.max()
            s = np.clip(alpha * (c - mm), -300, 0)
            w = np.exp(s)
            w /= w.sum()
            g = np.zeros(n)
            for k in range(2 * n - 1):
                wk = w[k] * dx
                if wk < 1e-14:
                    continue
                sh = (n - 1) - k
                if sh >= 0:
                    idx = np.arange(n - sh)
                    g[idx] += wk * (1.0 - h_f[idx + sh])
                    if sh > 0:
                        g[idx + sh] -= wk * h_f[idx]
                else:
                    sp = -sh
                    idx = np.arange(sp, n)
                    g[idx] += wk * (1.0 - h_f[idx - sp])
                    g[idx - sp] -= wk * h_f[idx]
            g_p = g.reshape(m, T).mean(axis=0)
            mom = beta * mom + (1.0 - beta) * g_p
            h_new = np.clip(h_p - lr * mom, 0.0, 1.0)
            d = T * 0.5 - h_new.sum()
            h_new += d / T
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _repair(h_new, T)
            c_new = _eval(np.tile(h_new, m))
            if c_new < best_c - 1e-16:
                best_c = c_new
                best = h_new.copy()
                h_p = h_new
                if lr < 0.02:
                    lr *= 1.05
            else:
                lr *= 0.55
                if lr < 1e-7:
                    lr = lr_init
                    mom = np.zeros(T)
    return best, best_c

def _coord(h, n_iter, scales, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5c = _eval(h)
    for scale in scales:
        for _ in range(n_iter // len(scales)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c
