def _repair(h, n):

            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 7: Coordinate descent micro-tuning (25s)
    if time.time() - t0 < budget_s - 3:
        h8 = best_h.copy()
        c5c = _eval(h8)
        nc = len(h8)
        for scale in [0.0008, 0.0003, 0.0001]:
            if time.time() - t0 > budget_s - 3: break
            for _it in range(10000):
                if time.time() - t0 > budget_s - 3: break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j: continue
                d = scale * rng.randn()
                ni, nj = h8[i] + d, h8[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
                h8[i], h8[j] = ni, nj
                c5n = _eval(h8)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                else:
                    h8[i], h8[j] = ni - d, nj + d
        if c5c < best_c5:
            best_c5, best_h = c5c, h8.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
