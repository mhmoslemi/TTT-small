import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _full_softmax_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    c = _corr_vals(h) * dx
    cm = c.max()
    s = np.exp(np.clip((c - cm) / temp, -600, 0))
    w = s / s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        g += w[k] * _grad_lag(h, k, dx)
    return g, cm

def _topk_grad(h, k_list, dx):
    n = len(h)
    g = np.zeros(n)
    for k in k_list:
        g += _grad_lag(h, k, dx)
    return g

def _projected_momentum(h, n, t_start, t_end, temp0, temp_end, max_iter, ftol):
    dx = 2.0 / n
    h_curr = h.copy()
    h_prev = h.copy()
    mu = 0.9
    c5 = _eval(h_curr)
    best_local = c5
    best_h = h_curr.copy()
    for it in range(max_iter):
        elapsed = time.time() - t_start
        if t_end - elapsed < 5.0:
            break
        g, cm = _full_softmax_grad(h_curr, temp0 * (temp_end/temp0)**it/max(1,max_iter))
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        y = h_curr + mu * (h_curr - h_prev)
        step = 0.02 / (gn + 1e-12)
        for _shrink in range(20):
            h_new = _repair(y - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                break
            step *= 0.5
        else:
            break
        h_prev = h_curr
        h_curr = h_new
        c5 = c5n
        if c5 < best_local - ftol:
            best_local = c5
            best_h = h_curr.copy()
    return best_h, best_local

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _active_set_qp(h, n, top_k=5):
    """Identify active lags and do focused descent on them."""
    c = _corr_vals(h) * (2.0 / n)
    top_idx = np.argsort(c)[-top_k:]
    c5 = c.max()
    dx = 2.0 / n
    for _round in range(50):
        g = _topk_grad(h, list(top_idx), dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        step = 0.05
        improved = False
        for _ in range(30):
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
                improved = True
                break
            step *= 0.4
        if not improved:
            break
        c = _corr_vals(h) * dx
        top_idx = np.argsort(c)[-top_k:]
    return h, c5

def _coord_fine(h, n, t_deadline, rng, n_passes=6):
    c5 = _eval(h)
    for p in range(n_passes):
        scale = 0.005 * (0.4 ** p)
        for _ in range(3000):
            if time.time() > t_deadline:
                return h, c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            hi, hj = h[i] + d, h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
            h[i], h[j] = hi, hj
            c5n = _eval(h)
            if c5n < c5 - 1e-17:
                c5 = c5n
            else:
                h[i], h[j] = hi - d, hj + d
    return h, c5
