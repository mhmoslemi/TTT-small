import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval_full(hf):
    n = len(hf)
    return float(np.max(np.correlate(hf, 1.0 - hf, mode='full')) * 2.0 / n)

def _eval_periodic(hp, T, m):
    return _eval_full(np.tile(hp, m))

def _obj_periodic(hp, T, m, alpha):
    n = T * m
    hf = np.tile(hp, m)
    c = np.correlate(hf, 1.0 - hf, mode='full')
    mm = c.max()
    s = np.clip(alpha * (c - mm), -500, 500)
    return mm + np.log(float(np.sum(np.exp(s - s.max())))) / alpha

def _grad_periodic(hp, T, m, alpha):
    n = T * m
    hf = np.tile(hp, m)
    c = np.correlate(hf, 1.0 - hf, mode='full')
    mm = c.max()
    s = np.clip(alpha * (c - mm), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g_full = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g_full[i1] += wk * (1.0 - hf[i1 + sh])
            if sh > 0:
                g_full[i1 + sh] -= wk * hf[i1]
        else:
            sp = -sh
            g_full[sp:] += wk * (1.0 - hf[:n - sp])
            g_full[:n - sp] -= wk * hf[sp:]
    gp = np.zeros(T)
    for p in range(m):
        gp += g_full[p * T:(p + 1) * T]
    return gp

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_per(hp, T, m, alpha, maxiter, ftol):
    bnds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj_periodic, hp, jac=_grad_periodic,
                       args=(T, m, alpha), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return _repair(res.x, T)
    except Exception:
        return _repair(hp.copy(), T)

def _slsqp_full(hf, alpha, maxiter, ftol):
    n = len(hf)
    bnds = [(0.0, 1.0)] * n

    def _obj_f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mm = c.max()
        s = np.clip(alpha * (c - mm), -500, 500)
        return mm + np.log(float(np.sum(np.exp(s - s.max())))) / alpha

    def _grad_f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mm = c.max()
        s = np.clip(alpha * (c - mm), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    g[i1 + sh] -= wk * x[i1]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n - sp])
                g[:n - sp] -= wk * x[sp:]
        return g

    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj_f, hf, jac=_grad_f, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return _repair(res.x, n)
    except Exception:
        return _repair(hf.copy(), n)

def _pgd_period(hp, T, m, alpha, niter, lr_init, seed):
    n = T * m
    rng = np.random.RandomState(seed)
    best = hp.copy()
    best_c = _eval_periodic(best, T, m)
    lr = lr_init
    mom = np.zeros(T)
    beta = 0.8
    for it in range(niter):
        g = _grad_periodic(hp, T, m, alpha)
        mom = beta * mom + (1.0 - beta) * g
        hp2 = np.clip(hp - lr * mom, 0.0, 1.0)
        hp2 = _repair(hp2, T)
        c2 = _eval_periodic(hp2, T, m)
        if c2 < best_c - 1e-16:
            best_c = c2
            best = hp2.copy()
            hp = hp2
            if lr < 0.05:
                lr *= 1.02
        else:
            lr *= 0.65
            if lr < 1e-7:
                lr = lr_init
                mom = np.zeros(T)
    return best, best_c

def _coord_descent(hf, nc, n_iter, scale, seed):
    rng = np.random.RandomState(seed)
    c5c = _eval_full(hf)
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = hf[i] + d, hf[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        hf[i], hf[j] = ni, nj
        c5n = _eval_full(hf)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            hf[i], hf[j] = ni - d, nj + d
    return hf, c5c

def _proj_sum(h, T):
    target = T / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / T
    return np.clip(h, 0.0, 1.0)

def _resample(h_old, n_new):
    xo = np.linspace(0.0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
