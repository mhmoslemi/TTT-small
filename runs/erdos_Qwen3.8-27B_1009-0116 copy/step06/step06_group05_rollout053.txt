import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_shift(h, k, dx):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += dx * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= dx * h[:n - sh]
    else:
        sp = -sh
        g[sp:] += dx * (1.0 - h[:n - sp])
        g[:n - sp] -= dx * h[sp:]
    return g

def _topk_grad(h, k, temp):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(c)[-k:][::-1]
    c_top = c[top_k]
    w = np.exp(-(c_top - c_top[0]) * temp)

    g = np.zeros(n)
    for wi, ki in zip(w, top_k):
        if wi < 1e-10:
            continue
        g += wi * _grad_shift(h, ki, dx)
    return g, float(c.max() * dx)

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_shift(x, k, dx)
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_eq_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    try:
        res = minimize(_lssm_obj, h, jac=_lssm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pattern(n, pat, rng):
    h = np.zeros(n)
    if pat == 0:
        h[n // 4:3 * n // 4] = 1.0
    elif pat == 1:
        h[::2] = 1.0
    elif pat == 2:
        for i in range(n):
            h[i] = 1.0 if (i % 3) < 2 else 0.0
    elif pat == 3:
        x = np.linspace(0, 2 * np.pi, n)
        h = (0.5 + 0.4 * np.sin(2 * x + rng.rand()) > 0.5).astype(float)
    elif pat == 4:
        x = np.linspace(0, 2 * np.pi, n)
        h = (0.5 + 0.3 * np.sin(3 * x) + 0.2 * np.sin(5 * x) > 0.5).astype(float)
    elif pat == 5:
        idx = np.arange(n)
        h[idx[::3]] = 1.0
        h[idx[1::5]] = 1.0
    elif pat == 6:
        for i in range(n):
            h[i] = 1.0 if (i * 7331 + i * i) % 5 < 2 else 0.0
    else:
        h[:] = 0.5
    h = _repair(h, n)
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    return h

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _micro_cd(h, t_limit, rng):
    nc = len(h)
    c5c = _eval(h)
    t_lim = time.time() + t_limit
    for scale in [0.003, 0.001, 0.0003]:
        for _ in range(2000):
            if time.time() > t_lim:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            h = _repair(h, nc)
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h = _repair(h, nc)
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def _el():
        return time.time() - t_start

    # Phase 1: Quick SLSQP (20s)
    for alpha in [10000, 100000]:
        if _el() > 20:
            break
        hn = _slsqp(best_h, alpha, 1000, 1e-18)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 2: Top-k subgradient rounds (80s)
    if _el() < budget_s - 400:
        for round_ in range(4):
            if _el() > budget_s - 400:
                break
            t0_val = 0.04 * (0.7 ** round_)
            t1_val = 0.002 * (0.7 ** round_)
            k_s = 1 + round_
            k_e = 5 + 2 * round_
            h2, c5_2 = _subgrad(best_h, 1500, t0_val, t1_val, k_s, k_e)
            if c5_2 < best_c5:
                best_c5, best_h = c5_2, h2.copy()

    # Phase 3: SLSQP refinement (50s)
    if _el() < budget_s - 300:
        for alpha in [200000, 1000000, 5000000]:
            if _el() > budget_s - 300:
                break
            h3 = _slsqp(best_h, alpha, 3000, 1e-19)
            c5 = _eval(h3)
            if c5 < best_c5:
                best_c5, best_h = c5, h3.copy()

    # Phase 4: Try other resolutions (120s)
    for n_try in [250, 350, 450, 600]:
        if _el() > budget_s - 200:
            break
        h_r = _repair(_resample(best_h, n_try), n_try)
        h_r = _slsqp(h_r, 100000, 800, 1e-18)
        h_r, c5_r = _subgrad(h_r, 800, 0.03, 0.003, 1, 6)
        c5 = _eval(h_r)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 5: Gaussian smoothing + re-optimize (80s)
    if _el() < budget_s - 150:
        for sig in [0.8, 1.5, 3.0]:
            if _el() > budget_s - 150:
                break
            hs = _repair(gaussian_filter1d(best_h, sig, mode='wrap'), best_n)
            hs = _slsqp(hs, 200000, 2000, 1e-19)
            hs, c5_s = _subgrad(hs, 800, 0.02, 0.002, 2, 8)
            c5 = _eval(hs)
            if c5 < best_c5:
                best_c5, best_h = c5, hs.copy()

    # Phase 6: Pattern restarts (100s)
    if _el() < budget_s - 120:
        for pat in range(7):
            if _el() > budget_s - 120:
                break
            hp = _pattern(best_n, pat, rng)
            hp = _slsqp(hp, 50000, 800, 1e-18)
            hp, c5_p = _subgrad(hp, 500, 0.02, 0.002, 1, 5)
            c5 = _eval(hp)
            if c5 < best_c5:
                best_c5, best_h = c5, hp.copy()

    # Phase 7: Intensive subgradient + SLSQP cycle (100s)
    if _el() < budget_s - 60:
        for cyc in range(3):
            if _el() > budget_s - 60:
                break
            h7, c5_7 = _subgrad(best_h, 1000, 0.015, 0.001, 3, 10)
            if c5_7 < best_c5:
                best_c5, best_h = c5_7, h7.copy()
            h7b = _slsqp(best_h, 10000000, 2000, 1e-20)
            c5b = _eval(h7b)
            if c5b < best_c5:
                best_c5, best_h = c5b, h7b.copy()

    # Phase 8: Micro coordinate descent (30s)
    if _el() < budget_s - 5:
        h8, c5_8 = _micro_cd(best_h, max(budget_s - 5 - _el(), 3), rng)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8.copy()

    # Final: one more SLSQP + subgradient
    if _el() < budget_s - 2:
        hf = _slsqp(best_h, 50000000, 1000, 1e-20)
        c5f = _eval(hf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()
        hf2, c5f2 = _subgrad(best_h, 200, 0.003, 0.0005, 5, 10)
        if c5f2 < best_c5:
            best_c5, best_h = c5f2, hf2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
