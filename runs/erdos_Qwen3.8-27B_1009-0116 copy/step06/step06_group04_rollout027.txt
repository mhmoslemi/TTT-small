    """Aggressive binary local search using pair flips"""
    rng = np.random.RandomState(seed)
    x = binary_h.copy()
    # Ensure exactly n/2 ones
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    if len(ones) != n // 2:
        # Fix
        if len(ones) > n // 2:
            remove = rng.choice(ones, len(ones) - n//2, replace=False)
            x[remove] = 0
        else:
            add = rng.choice(zeros, n//2 - len(ones), replace=False)
            x[add] = 1
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
    
    c_cur = np.max(np.correlate(x, 1.0 - x, mode='full'))
    best_x = x.copy()
    best_c = c_cur
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 5000:
            break
            
        # Get current correlation
        c_arr = np.correlate(x, 1.0 - x, mode='full')
        max_c = c_arr.max()
        
        if max_c < best_c:
            best_c = max_c
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find the worst shift
        k_worst = np.argmax(c_arr)
        sh = (n - 1) - k_worst
        
        # Find positions contributing to the worst overlap
        if sh >= 0:
            h1 = x[:n-sh]
            h0 = x[sh:]
            bad_positions = np.where((h1 == 1) & (h0 == 0))[0]
        else:
            sp = -sh
            h1 = x[sp:]
            h0 = x[:n-sp]
            bad_positions = np.where((h1 == 1) & (h0 == 0))[0]
            bad_positions = bad_positions + sp  # position in x where h=1
        
        if len(bad_positions) == 0:
            break
        
        # Try flipping a bad position (1→0) and a random good position (0→1)
        # This maintains the sum constraint
        i = bad_positions[rng.randint(len(bad_positions))]
        # Find a 0 to flip to 1
        zeros_list = np.where(x == 0)[0]
        if len(zeros_list) == 0:
            break
        j = zeros_list[rng.randint(len(zeros_list))]
        
        x[i] = 0
        x[j] = 1
        new_c = np.max(np.correlate(x, 1.0 - x, mode='full'))
        
        if new_c <= max_c:
            # Accept (might improve other shifts)
            pass
        else:
            # Revert
            x[i] = 1
            x[j] = 0
    
    return best_x, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam(h0, alpha_start, alpha_end, n_iter, lr_start, lr_end, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_x = x.copy()
    best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        if t % 1000 == 0:
            x = x + 0.01 * rng.randn(n)
        g = _grad(x, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return best_x, best_c

                best_x = x.copy()

            x[j] = 0
