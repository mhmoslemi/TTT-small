            best_c5, best_h, best_n = c4, h4, len(h4)

    # Phase 5: Upsample to 2x
    if el() < budget_s - 200:
        n1 = best_n * 2
        h_up = _repair(np.repeat(best_h, 2), n1)
        h5, c5 = _deflate(h_up, 5, 200, 0.03, el, budget_s - 120)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h5, n1
        if el() < budget_s - 80:
            h5 = _nesterov(best_h.copy(), 40000, 200, 0.003, el, budget_s - 80)
            c5 = _eval(h5)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h5, len(h5)
        # Stochastic deflation at 2x
        if el() < budget_s - 50:
            h5, c5 = _stochastic_deflate(best_h.copy(), 150, 0.03, el, budget_s - 50, rng)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h5, len(h5)

    # Phase 6: Final deflation + Nesterov at best resolution
    if el() < budget_s - 60:
        h6, c6 = _deflate(best_h.copy(), 3, 150, 0.015, el, budget_s - 60)
        if c6 < best_c5:
            best_c5, best_h, best_n = c6, h6, len(h6)
    if el() < budget_s - 30:
        h6 = _nesterov(best_h.copy(), 100000, 150, 0.002, el, budget_s - 30)
        c6 = _eval(h6)
        if c6 < best_c5:
            best_c5, best_h, best_n = c6, h6, len(h6)

    # Phase 7: Multi-scale local search
    for scale in [2e-4, 5e-5, 1e-5, 3e-6, 1e-6]:
        if el() > budget_s - 8:
            break
        n_iter = min(int((budget_s - 8 - el()) / 0.004), 30000)
        if n_iter > 200:
            best_h, best_c5 = _local(best_h.copy(), n_iter, rng, el, budget_s - 3, scale)
            best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
