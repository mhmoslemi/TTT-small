Let me try: **coordinate descent with exact 1D optimization**.

For each coordinate i, the function f(h[i]) = max_k c[k] (with other h fixed) is convex (max of linear functions). I can find its minimum exactly.

But I also need to maintain the sum constraint. So instead of changing just one coordinate, I change two: h[i] += δ, h[j] -= δ.

For a given pair (i, j), the change in c[k] is:
Δc[k] = δ * (∂c[k]/∂h[i] - ∂c[k]/∂h[j])

Let a_k = ∂c[k]/∂h[i] / dx and b_k = ∂c[k]/∂h[j] / dx (normalized by dx).

Then: c[k](δ) = c[k] + δ * (a_k - b_k) * dx

max_k c[k](δ) is a convex piecewise-linear function of δ.

To minimize: check the value at all breakpoints and endpoints.

For the top M correlations (M=10), there are at most C(M,2) = 45 breakpoints.

This is very fast per pair! The question is which pairs to try.

Strategy: For each coordinate i, try pairing with a small number of "good" partners j (e.g., those that reduce the top correlations most).

    dx = 2.0/n

    best_c = c.max() * dx
    best_h = h.copy()

        c_top = c[top_idx].copy()

            # For correlation index k, shift = (n-1) - k
            # ∂c[k]/∂h[i] (raw, before *dx) = sum of contributions
            # h[i] in h[j] part: (1-h[i+sh]) if 0<=i+sh<n
            # h[i] in (1-h[j+sh]) part: -h[i-sh] if 0<=i-sh<n

            # shift_k = (n-1) - k
            shifts = (n-1) - top_idx
            
            grad_i = np.zeros(top_m)
            for m_idx in range(top_m):
                sh = shifts[m_idx]
                gi = 0.0
                if 0 <= i + sh < n:
                    gi += (1.0 - h[i + sh])
                if 0 <= i - sh < n:
                    gi -= h[i - sh]
                grad_i[m_idx] = gi
            
            # For each candidate partner j:
            # Try a few j values
            candidates = []
            # Partner that maximizes reduction in top correlations
            # We want (grad_i - grad_j) to reduce c_top the most
            # Simple heuristic: try j = (i + 1) % n, j = (i + n//2) % n, random
            for j in [(i+1)%n, (i+n//2)%n, (i+3*n//4)%n, rng.randint(n)]:
                if j == i:
                    continue
                
                shifts_j = shifts  # same shifts
                grad_j = np.zeros(top_m)
                for m_idx in range(top_m):
                    sh = shifts[m_idx]
                    gj = 0.0
                    if 0 <= j + sh < n:
                        gj += (1.0 - h[j + sh])
                    if 0 <= j - sh < n:
                        gj -= h[j - sh]
                    grad_j[m_idx] = gj

                d_hi = min(1.0 - h[i], h[j])  # h[i]+d <= 1 -> d <= 1-h[i], h[j]-d >= 0 -> d <= h[j]
                d_lo = max(d_lo, -0.5)  # limit step size
                d_hi = min(d_hi, 0.5)
                
                if d_lo >= d_hi:
                    continue
                
                # Find min of max_k (c_act[k] + d * slopes[k]) over d in [d_lo, d_hi]
                candidates_d = [d_lo, d_hi, 0.0]
                for a_idx in range(top_m):
                    for b_idx in range(a_idx+1, top_m):
                        ds = slopes[a_idx] - slopes[b_idx]
                        if abs(ds) > 1e-15:
                            bp = (c_act[b_idx] - c_act[a_idx]) / ds
                            if d_lo <= bp <= d_hi:
                                candidates_d.append(bp)

                best_val = c_act.max()
                for d in candidates_d:
                    val = (c_act + d * slopes).max()
                    if val < best_val - 1e-16:
                        best_val = val
                        best_d = d

                    c = np.correlate(h, 1.0-h, mode='full')
                    mc = c.max() * dx

                        best_h = h.copy()
                    break  # Only one move per i per sweep
        
        # Update c at end of sweep
        c = np.correlate(h, 1.0-h, mode='full')
    
    return best_h, best_c

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

    return x.sum() - 0.5 * len(x)

    return np.ones(len(x))

    h = h.copy()
    c_full = np.correlate(h, 1.0 - h, mode='full')

                if 0 <= i - sh < n:

                    if 0 <= j - sh < n:

def _fine_search(h, n_tries, rng):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    for _ in range(n_tries):
        i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j:
            continue
        d = 3e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c

def _fine_search(h, n_tries, rng, step=3e-5):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy(); best_c = c_cur
    for _ in range(n_tries):
        i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j:
            continue
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c
