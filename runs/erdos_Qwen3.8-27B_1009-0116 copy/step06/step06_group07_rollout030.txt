```python
def _max_subgrad_pgd(h, n, niter, lr_init=0.01, seed=0):
    rng = np.random.RandomState(seed)
    best = h.copy(); best_c = _eval(h)
    lr = lr_init
    for it in range(niter):
        c = np.correlate(h, 1-h, mode='full')
        kmax = c.argmax()
        sh = (n-1) - kmax
        dx = 2.0/n
        g = np.zeros(n)
        if sh >= 0:
            g[:n-sh] += (1-h[sh:])
            if sh > 0: g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            g[sp:] += (1-h[:-sp])
            g[:-sp] -= h[sp:]
        g *= dx
        h_new = _proj_sum(h - lr * g, n)
        c5 = _eval(h_new)
        if c5 < best_c:
            best = h_new.copy(); best_c = c5; h = h_new
            lr *= 1.01
        else:
            lr *= 0.95
            if lr < 1e-8: lr = lr_init
    return best, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = np.clip(h * target / s, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smobj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _smgrad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _proj_sum(h, n):
    target = n / 2.0

        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

    bnds = [(0.0, 1.0)] * n

    try:
        r = minimize(_smobj, p, jac=_smgrad, args=(alpha,), method='SLSQP',
                     bounds=bnds, constraints=cons, options={'maxiter': mi, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(p.copy(), n)

def _active_pgd(h, n, niter, lr_init=0.01, band=0.02, seed=0):
    """PGD using gradient of the max (or near-max) correlation."""
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_c = _eval(h)
    lr = lr_init
    for it in range(niter):
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        active = c >= cmax - band * (2.0 / n)
        nact = max(1, active.sum())
        g = np.zeros(n)
        dx = 2.0 / n
        for k in np.where(active)[0]:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n - sp])
                g[:n - sp] -= h[sp:]
        g *= dx / nact
        h_new = _proj_sum(h - lr * g, n)
        c5 = _eval(h_new)
        if c5 < best_c - 1e-14:
            best = h_new.copy()
            best_c = c5
            h = h_new
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.9
            if lr < 1e-9:
                lr = lr_init
    return best, best_c

def _coord_desc(h, n, niter, scales, seed=0):
    rng = np.random.RandomState(seed)
    c5 = _eval(h)
    h = h.copy()
    per_scale = niter // len(scales)
    for scale in scales:

            i = rng.randint(n)
            j = rng.randint(n)

            d = scale * rng.randn()
            hi = h[i] + d
            hj = h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:

            h[i] = hi

            if c5n < c5 - 1e-15:
                c5 = c5n
            else:
                h[i] = hi - d
                h[j] = hj + d
    return h, c5
```
