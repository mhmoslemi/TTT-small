import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    c = np.correlate(h, omh, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh == 0:
            g += wk * (omh - h)
        elif sh > 0:
            g[:n - sh] += wk * omh[sh:]
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * omh[:n - sp]
            g[:n - sp] -= wk * h[sp:]
    return g

        t_k = t_new
