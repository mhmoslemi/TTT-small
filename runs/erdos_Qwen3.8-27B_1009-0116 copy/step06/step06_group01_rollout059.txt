C₅(h) = max_k f_k(h)

If I fix all h[j] for j≠i, and vary h[i]:
f_k(h) is linear in h[i] for each k (since h[i] appears linearly in each product term).
So f_k(h[i]) = a_k * h[i] + b_k for some constants a_k, b_k.

And max_k (a_k * h[i] + b_k) is a convex piecewise-linear function of h[i].
We want to minimize this subject to 0 ≤ h[i] ≤ 1.

The minimum of a convex piecewise-linear function on [0,1] occurs at:
- One of the breakpoints (where two pieces intersect)
- Or at an endpoint (0 or 1)

The breakpoints are where a_k * h[i] + b_k = a_l * h[i] + b_l, i.e., h[i] = (b_l - b_k)/(a_k - a_l).

So for each i, I can:
1. Compute a_k and b_k for all k
2. Find all breakpoints in [0,1]
3. Evaluate the objective at each breakpoint and at 0, 1
4. Set h[i] to the best value

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _weighted_grad(h, w, n, dx):
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[:n - lag] += wk * (1.0 - h[lag:])
            g[lag:] -= wk * h[:n - lag]
        else:
            p = -lag
            g[p:] += wk * (1.0 - h[:n - p])
            g[:n - p] -= wk * h[p:]
    return g

def _lsexp_obj_and_grad(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -600, 600)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    dx = 2.0 / len(x)
    val = m + np.log(e.sum()) / alpha
    g = _weighted_grad(x, w, len(x), dx)
    return val, g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj_and_grad, h, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       args=(alpha,), options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad_descent(h, n, t_limit, alpha_init, alpha_final, n_iters_total):
    dx = 2.0 / n
    c5 = _eval(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    v = np.zeros(n)
    mu = 0.85
    step = 0.04
    best_h, best_c5 = h.copy(), c5
    for it in range(n_iters_total):
        if time.time() > t_limit:
            break
        frac = min(1.0, it / max(1, n_iters_total * 0.6))
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        w = _lsexp_obj_and_grad(h, alpha)
        w_vec = np.exp(np.clip(alpha * (c - c.max()), -600, 600))
        w_vec /= w_vec.sum()
        g = _weighted_grad(h, w_vec, n, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = mu * v - step * g / gn
        h_trial = _repair(h + v, n)
        c5_t = _eval(h_trial)
        if c5_t < c5 - 1e-17:
            h = h_trial; c5 = c5_t
            c = np.correlate(h, 1.0 - h, mode='full')
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
            step = min(step * 1.03, 0.8)
        else:
            step *= 0.6
            v *= 0.3
            if step < 1e-12:
                step, v = 0.04, np.zeros(n)
    return best_h, best_c5

def _coord_descent(h, n, t_limit, n_sweeps):
    c5 = _eval(h)
    for sweep in range(n_sweeps):
        if time.time() > t_limit:
            break
        for i in range(n):
            orig = h[i]
            for delta in [0.0, 0.02, -0.02, 0.005, -0.005, 0.05, -0.05]:
                nv = orig + delta
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                h = _repair(h, n)
                c5n = _eval(h)
                if c5n < c5 - 1e-16:
                    c5 = c5n
                else:
                    h[i] = orig
                    _repair(h, n)
                    break
    return h, c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl(rem=0):

            if c5 < best_c5 - 1e-17:

    # Phase 2: Projected gradient with adaptive alpha and momentum
    if tl(300):

                                    time.time() + (tl() - 150), 10000, 200000, 300000)
        if c5 < best_c5 - 1e-17:

            if not tl(80):

            h3 = _slsqp(best_h, alpha, 1500, 1e-20)
            c5 = _eval(h3)
            if c5 < best_c5 - 1e-17:
                best_c5, best_h = c5, h3.copy()
