def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

            x_new = _proj(x - lr * v, T)
            c_new = _eval(np.tile(x_new, m))
            if c_new < best_c - 1e-16:
                best_c = c_new
                best = x_new.copy()
                x = x_new
                lr = min(lr * 1.01, 0.05)
            else:
                lr *= 0.7
                if lr < lr0 * 0.005:
                    lr = lr0
                    v = np.zeros(T)
                x = x  # no change
    return best, best_c

def _coord_polish(h, n_iter, scale, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    cc = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < cc - 1e-16:
            cc = cn
        else:
            h[i], h[j] = ni - d, nj + d
    return h, cc

def _slsqp_run(h0, alpha, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj_sm, h0, jac=_smgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h0.copy(), n)
