                sp = -sh
                idx = np.arange(sp, n)
                g[idx] += wk * (1.0 - h[idx - sp])
                g[idx - sp] -= wk * h[idx]
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = _repair(h - lr * momentum, n)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        h = h_new
        if it > 0 and it % 200 == 0:
            lr *= 0.7
            momentum *= 0.3
    return best, best_c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _slsqp_quick(h, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def cons_fun(x):
        return x.sum() - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _perturb_binary(b, n, rng, n_flips):
    b = b.copy()
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for _ in range(n_flips):
        i, j = ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]
        b[i], b[j] = 0.0, 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float).copy()
    best_h = _repair(h0.copy(), n0)
    best_c5 = _eval(best_h)
    best_n = n0
    # Phase 1: Binary SA on n0
    b0 = _to_binary(h0, n0, rng)
    c5_b = _eval(b0)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = b0.copy()
    sa_rounds = [(0.04, 0.008, 800000), (0.02, 0.004, 800000),
                 (0.015, 0.003, 600000), (0.01, 0.002, 600000)]
    current_b = b0.copy()
    for r, (T0, T1, n_it) in enumerate(sa_rounds):
        if elapsed() > budget_s * 0.35:
            break
        cur_b, cur_c5 = _binary_sa(current_b, best_n, rng, T0, T1, n_it)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_b.copy()
            current_b = cur_b.copy()
    # Phase 2: SA with perturbation restarts
    if elapsed() < budget_s * 0.55:
        for trial in range(8):
            if elapsed() > budget_s * 0.55:
                break
            b_pert = _perturb_binary(current_b, best_n, rng, 5 + trial * 3)
            n_it = min(400000, max(100000, int((budget_s * 0.55 - elapsed()) / 8 / 0.00005)))
            if n_it < 5000:
                break
            _, T1_t, _ = sa_rounds[min(trial, 3)]
            b_t, c5_t = _binary_sa(b_pert, best_n, rng, T1_t * 1.5, T1_t * 0.3, n_it)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = b_t.copy()
                current_b = b_t.copy()
    # Phase 3: Continuous refinement on best
    if elapsed() < budget_s * 0.70:
        h_cont, c5_cont = _cont_refine(best_h.copy(), best_n, 15000, 300, 0.003)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
        h_slq, = [_slsqp_quick(best_h.copy(), best_n, 50000, 500)]
        c5_slq = _eval(h_slq)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()
    # Phase 4: Try other n values
    if elapsed() < budget_s * 0.85:
        for dn in [-20, 20, -10, 10, -30, 30]:
            if elapsed() > budget_s * 0.85:
                break
            n_try = best_n + dn
            if n_try < 100 or n_try > 700 or n_try % 2 != 0:
                continue
            h_r = _repair(_resample(best_h, n_try), n_try)
            b_r = _to_binary(h_r, n_try, rng)
            c5_r = _eval(b_r)
            b_r, c5_r = _binary_sa(b_r, n_try, rng, 0.02, 0.004, min(300000, max(50000, int((budget_s*0.85-elapsed())*10000))))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = b_r.copy()
                best_n = n_try
    # Phase 5: Final SA polish on best
    if elapsed() < budget_s * 0.92:
        n_it = max(100000, int((budget_s * 0.92 - elapsed()) / 0.00006))
        b_final, c5_final = _binary_sa(best_h.copy(), best_n, rng, 0.008, 0.001, n_it)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = b_final.copy()
    # Phase 6: Final continuous polish
    if elapsed() < budget_s - 15:
        h_final, c5_final = _cont_refine(best_h.copy(), best_n, 50000, 150, 0.002)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
