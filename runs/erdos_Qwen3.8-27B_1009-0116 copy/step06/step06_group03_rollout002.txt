        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, n_iter, step0, mu=0.9, alpha_base=2000):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    v = np.zeros(n); step = step0; stuck = 0
    for it in range(n_iter):
        alpha = alpha_base + 1000 * it
        g = _sgrad(best, alpha)
        v = mu * v + g
        nv = np.linalg.norm(v)
        if nv < 1e-14:
            break
        v /= nv
        improved = False
        for f in [1.0, 0.6, 0.4, 0.2, 0.1, 0.05, 0.02]:
            hn = _proj(best - step * f * v, 0.5 * n)
            cn = _eval(hn)
            if cn < best_c - 1e-16:
                best = hn; best_c = cn
                step = min(step * 1.08, 0.25)
                improved = True; stuck = 0
                break
        if not improved:
            step *= 0.72; stuck += 1
            if stuck > 70:
                best = _repair(best + 0.004 * np.random.randn(n), n)
                best_c = _eval(best); step = step0; stuck = 0; v = np.zeros(n)
    return best, best_c

def _fine_search(h, n_tries, rng, scale=2e-5):
    n = len(h); h = h.copy()
    c_cur = _eval(h); best = h.copy(); best_c = c_cur
    no_improve = 0
    for t in range(n_tries):
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = scale * (0.5 + 0.5 * rng.random()) * (1 if rng.random() > 0.5 else -1)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn; no_improve = 0
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
            no_improve += 1
        if no_improve > n_tries // 5:
            scale *= 0.6
            no_improve = 0
    return best, best_c

def _triple_search(h, n_tries, rng, scale=1.5e-5):
    n = len(h); h = h.copy()
    c_cur = _eval(h); best = h.copy(); best_c = c_cur
    for t in range(n_tries):
        i = rng.randint(n); j = rng.randint(n); k = rng.randint(n)
        if i == j or i == k or j == k:
            continue
        di = scale * rng.randn(); dj = scale * rng.randn()
        dk = -(di + dj)
        ni, nj, nk = h[i] + di, h[j] + dj, h[k] + dk
        if ni < 0 or ni > 1 or nj < 0 or nj > 1 or nk < 0 or nk > 1:
            continue
        h[i], h[j], h[k] = ni, nj, nk
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j], h[k] = h[i] - di, h[j] - dj, h[k] - dk
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def el():
        return time.time() - t0
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float).copy()
    h = _repair(h0, n0)
    best_h = h.copy(); best_c = _eval(best_h); best_n = n0

    # Phase 1: SLSQP with increasing smoothness (keep in basin)
    for alpha in [2000, 10000, 50000]:
        if el() > budget_s * 0.25:
            break
        hc = _slsqp(best_h, alpha, 2500, 1e-17)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 2: Subgradient with full softmax (longer exploration)
    if el() < budget_s * 0.55:
        n_iter = min(4000, int((budget_s * 0.55 - el()) / 0.035))
        best_h, best_c = _subgrad(best_h, n_iter, 0.012, mu=0.88, alpha_base=3000)

    # Phase 3: SLSQP polish with high alpha
    for alpha in [100000, 400000]:
        if el() > budget_s * 0.72:
            break
        hc = _slsqp(best_h, alpha, 2000, 1e-18)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 4: Perturb and re-optimize (escape local min)
    if el() < budget_s * 0.82:
        hc = _repair(best_h + 0.008 * rng.randn(best_n), best_n)
        hc = _slsqp(hc, 50000, 1500, 1e-17)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 5: Fine and triple searches
    if el() < budget_s - 60:
        n_tr = min(int((budget_s - 60 - el()) * 0.6 / 0.006), 30000)
        best_h, best_c = _fine_search(best_h, n_tr, rng)

    if el() < budget_s - 30:
        n_tr = min(int((budget_s - 30 - el()) / 0.008), 15000)
        best_h, best_c = _triple_search(best_h, n_tr, rng)

    # Final SLSQP polish
    if el() < budget_s - 15:
        hc = _slsqp(best_h, 1000000, 2000, 1e-18)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc; best_n = len(hc)

    if el() < budget_s - 5:
        n_tr = min(int((budget_s - 5 - el()) / 0.006), 10000)
        best_h, best_c = _fine_search(best_h, n_tr, rng, scale=8e-6)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
