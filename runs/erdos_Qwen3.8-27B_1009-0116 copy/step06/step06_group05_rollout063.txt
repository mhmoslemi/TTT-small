import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _pnorm_obj(h, p):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    return float(np.sum(c ** p)) ** (1.0 / p)

def _pnorm_grad(h, p):
    n = len(h); dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    F = np.sum(c ** p) ** (1.0 / p)
    if F < 1e-30: return np.zeros(n)
    coef = F ** (1.0 - p)
    w = np.power(c, p - 1)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18: continue
        s = k - (n - 1)
        ck = w[k] * coef * dx
        if s >= 0:
            g[:n-s] += ck * (1.0 - h[s:])
            if s > 0: g[s:] -= ck * h[:n-s]
        else:
            sp = -s
            g[sp:] += ck * (1.0 - h[:n-sp])
            g[:n-sp] -= ck * h[sp:]
    return g

def _eqf(h): return h.sum() - len(h) * 0.5
def _eqj(h): return np.ones(len(h))

def _slsqp_pn(h, p, mi):
    n = len(h)
    try:
        r = minimize(_pnorm_obj, h, jac=_pnorm_grad, args=(p,),
                     method='SLSQP', bounds=[(0.0,1.0)]*n,
                     constraints=[{'type':'eq','fun':_eqf,'jac':_eqj}],
                     options={'maxiter':mi, 'ftol':1e-18})
        return _repair(r.x, n)
    except: return _repair(h, n)

def _subgrad_mom(h, n_iter, a0, beta, p_thresh):
    n = len(h); dx = 2.0/n
    h = h.copy()
    z = np.zeros(n)
    best = _eval(h); best_h = h.copy()
    for t in range(n_iter):
        c = np.correlate(h, 1.0-h, mode='full')
        ki = int(np.argmax(c))
        s = ki - (n-1)
        g = np.zeros(n)
        if s >= 0:
            g[:n-s] += (1.0 - h[s:])
            if s > 0: g[s:] -= h[:n-s]
        else:
            sp = -s
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
        g *= dx
        alpha = a0 / (np.sqrt(t+1) + 1.0)
        z = beta * z - alpha * g
        h_new = _repair(z, n)
        c5 = _eval(h_new)
        if c5 < best: best = c5; best_h = h_new.copy()
        h = h_new
    return best_h, best

def _anneal_bin(h, t0, t1, nsteps, rng):
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, n)
    c5c = _eval(h); best = c5c; best_h = h.copy()
    for step in range(nsteps):
        t = t0 * (t1/t0) ** (step / max(nsteps-1, 1))
        i = rng.randint(n); j = rng.randint(n)
        while i == j: j = rng.randint(n)
        if h[i] == h[j]: continue
        h[i], h[j] = 1.0-h[i], 1.0-h[j]
        h2 = _repair(h, n); c5n = _eval(h2)
        if c5n < c5c or rng.rand() < np.exp(-(c5n-c5c)/max(t,1e-6)):
            h = h2; c5c = c5n
            if c5c < best: best = c5c; best_h = h.copy()
        else:
            h[i], h[j] = 1.0-h[i], 1.0-h[j]
            h = _repair(h, n)
    return best_h, best

def _coord_desc(h, n_iter, rng, scales):
    n = len(h)
    h = h.copy(); c5c = _eval(h)
    for sc in scales:
        for _ in range(n_iter):
            i = rng.randint(n); j = rng.randint(n)
            if i == j: continue
            d = sc * rng.randn()
            ni, nj = h[i]+d, h[j]-d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16: c5c = c5n
            else: h[i], h[j] = ni-d, nj+d
    return h, c5c

def _resample(h, n_new):
    no = len(h)
    return np.interp(np.linspace(0,2,n_new,False), np.linspace(0,2,no,False), h)

def _make_bin(n, p, rng):
    h = np.zeros(n)
    if p == 0:
        h[::2] = 1.0
    elif p == 1:
        h[:n//3] = 1.0; h[2*n//3:] = 1.0
    elif p == 2:
        idx = np.arange(n); h[idx[::3] < n//2*(3//2)] = 1.0
        # simpler: periodic pattern 2-1
        for i in range(n):
            h[i] = 1.0 if (i % 3) < 2 else 0.0
    elif p == 3:
        # golden ratio based
        gr = (1+5**0.5)/2
        phases = np.arange(n) * gr % n
        h[phases < n/2] = 1.0
    else:
        h[rng.choice(n, n//2, replace=False)] = 1.0
    h = (h > 0.5).astype(float)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bc, bh = _eval(h), h.copy()
    bn = n0
    
    def elapsed(): return time.time() - t_start
    def remaining(): return budget_s - elapsed()
    
    # Phase 1: SLSQP with increasing p (150s)
    for p in [20, 50, 100, 300, 1000]:
        if remaining() < 120: break
        hn = _slsqp_pn(bh, p, 3000)
        c = _eval(hn)
        if c < bc: bc, bh = c, hn.copy()
        # complement
        hc = _repair(1.0 - bh, len(bh))
        hc = _slsqp_pn(hc, p, 2000)
        cc = _eval(hc)
        if cc < bc: bc, bh = cc, hc.copy()
    
    # Phase 2: Subgradient with momentum (150s)
    if remaining() > 150:
        hs, cs = _subgrad_mom(bh, 50000, 0.05, 0.9, 0.3)
        if cs < bc: bc, bh = cs, hs.copy()
        hn = _slsqp_pn(bs if False else bh, 1000, 3000)
        c = _eval(hn)
        if c < bc: bc, bh = c, hn.copy()
    
    # Phase 3: Binary annealing (150s)
    if remaining() > 150:

        if cb < bc:
            hn = _slsqp_pn(hb, 100, 3000)
            c = _eval(hn)
            if c < bc: bc, bh, bn = c, hn.copy(), len(hn)
    
    # Phase 4: Multi-n exploration (120s)
    for nt in [200, 250, 350, 400, 500]:
        if remaining() < 100: break
        hr = _repair(_resample(bh, nt), nt)
        hr = _slsqp_pn(hr, 100, 2000)
        hr = _slsqp_pn(hr, 1000, 2000)
        c = _eval(hr)

            if remaining() < 60: break

        bh = _slsqp_pn(bh, 2000, 3000)

Let me also think about **projected gradient with majorization-minimization (MM)**. The idea is:
1. At current h, construct a quadratic upper bound Q(h, h_old) of the objective
2. Minimize Q subject to constraints (a QP)
3. Update h

For F(h) = max_k g_k(h) where g_k(h) = sum h[i](1-h[i+k]):
At h_old, the active constraint is g_{k*}(h_old).
Q(h, h_old) = g_{k*}(h) linearized at h_old + a quadratic regularizer
            ≈ g_{k*}(h_old) + ∇g_{k*}(h_old)^T (h - h_old) + 0.5μ||h - h_old||^2

The QP:
min_h g_{k*}(h_old) + ∇g_{k*}^T h - ∇g_{k*}^T h_old + 0.5μ||h - h_old||^2
s.t. 0 ≤ h ≤ 1, sum(h) = n/2

This is a convex QP! (Quadratic objective, linear constraints.) We can solve it with SLSQP or even analytically in some cases.

But SLSQP with a quadratic objective and linear constraints should converge very quickly. In fact, this is essentially one step of a proximal gradient method, which is what the subgradient method approximates.

The advantage of MM is that it guarantees monotone decrease of the objective, whereas the plain subgradient method doesn't.

Let me implement an MM step:
1. Compute g, the worst shift
2. Compute linearization of g at h
3. Solve: min (∇g)^T h + 0.5/τ * ||h - h_old||^2 s.t. 0≤h≤1, sum=n/2
4. This is equivalent to: h = project_{feasible}(h_old - τ * ∇g)
   where project_{feasible} is projection onto {0≤h≤1, sum=h=n/2}

The projection onto the box and hyperplane can be done efficiently:
- For the box projection: clip
- For the hyperplane: subtract the mean excess
- But the combined projection (box + hyperplane) requires bisection

This is essentially the same as the subgradient step with projection! So my _subgrad_mom already does this.

The key difference with MM is the step size τ. In MM, τ should be chosen such that the quadratic upper bound is valid. For the bilinear g_k, this requires:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _po(h, p):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    return float(np.sum(c ** p)) ** (1.0 / p)

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    S = float(np.sum(c ** p))
    if S < 1e-30:
        return np.zeros(n)
    scale = dx * S ** ((1.0 - p) / p)
    w = c ** (p - 1)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        s = k - (n - 1)
        ck = w[k] * scale
        if s >= 0:
            g[:n - s] += ck * (1.0 - h[s:])
            if s > 0:
                g[s:] -= ck * h[:n - s]
        else:
            sp = -s
            g[sp:] += ck * (1.0 - h[:n - sp])
            g[:n - sp] -= ck * h[sp:]
    return g

def _ef(h):
    return h.sum() - len(h) * 0.5

def _ej(h):
    return np.ones(len(h))

def _slsqp(h, p, mi):
    n = len(h)
    try:
        r = minimize(_po, h, jac=_pg, args=(p,), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _ef, 'jac': _ej}],
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, nit, a0, beta):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    v = np.zeros(n)
    bc = _eval(h)
    bh = h.copy()
    for t in range(nit):
        c = np.correlate(h, 1.0 - h, mode='full')
        ki = int(np.argmax(c))
        s = ki - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[:n - s] += (1.0 - h[s:])
            if s > 0:
                g[s:] -= h[:n - s]
        else:
            sp = -s
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
        g *= dx
        alpha = a0 / np.sqrt(t + 1.0)
        v = beta * v - alpha * g
        h = _repair(h + v, n)
        c5 = _eval(h)
        if c5 < bc:
            bc = c5
            bh = h.copy()
    return bh, bc

def _anneal(h, t0, t1, ns, rng):
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, n)
    c5c = _eval(h)
    bc = c5c
    bh = h.copy()
    for st in range(ns):
        t = t0 * (t1 / t0) ** (st / max(ns - 1, 1))
        i, j = rng.randint(n), rng.randint(n)
        while i == j:
            j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        h2 = _repair(h, n)
        c5n = _eval(h2)
        if c5n < c5c or rng.rand() < np.exp(-(c5n - c5c) / max(t, 1e-6)):
            h, c5c = h2, c5n
            if c5c < bc:
                bc, bh = c5c, h.copy()
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            h = _repair(h, n)
    return bh, bc

def _coord(h, nit, rng, scales):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for sc in scales:
        for _ in range(nit):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:

            d = sc * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:

            c5n = _eval(h)
            if c5n < c5c - 1e-16:

            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h, nn):
    return np.interp(np.linspace(0, 2, nn, False), np.linspace(0, 2, len(h), False), h)

    if p == 0:

    elif p == 1:

    elif p == 2:

    elif p == 3:

            h[i] = 1.0 if (i % 5) < 3 else 0.0
    else:
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    h = (h > 0.5).astype(float)
    return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                a = min(abs(diff), max(1.0 - h[idx], 0.0)) if diff > 0 else min(-diff, h[idx])
                h[idx] += a if diff > 0 else -a
                diff -= a if diff > 0 else -a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _po(h, p):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    cm = c.max()
    if cm < 1e-30:
        return 0.0
    return float(cm * np.sum((c / cm) ** p) ** (1.0 / p))

def _pg(h, p):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    if cm < 1e-30:
        return np.zeros(n)
    cn = c / cm
    S = float(np.sum(cn ** p))
    scale = dx * S ** ((1.0 - p) / p)
    w = cn ** (p - 1)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        ck = w[k] * scale
        if s >= 0:
            g[:n - s] += ck * (1.0 - h[s:])
            if s > 0:
                g[s:] -= ck * h[:n - s]
        else:
            sp = -s
            g[sp:] += ck * (1.0 - h[:n - sp])
            g[:n - sp] -= ck * h[sp:]
    return g

def _ef(h):
    return h.sum() - len(h) * 0.5

def _ej(h):
    return np.ones(len(h))

def _slsqp(h, p, mi):
    n = len(h)
    try:
        r = minimize(_po, h, jac=_pg, args=(p,), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _ef, 'jac': _ej}],
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, nit, a0, beta):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    v = np.zeros(n)
    bc = _eval(h)
    bh = h.copy()
    for t in range(nit):
        c = np.correlate(h, 1.0 - h, mode='full')
        ki = int(np.argmax(c))
        s = ki - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[:n - s] += (1.0 - h[s:])
            if s > 0:
                g[s:] -= h[:n - s]
        else:
            sp = -s
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
        g *= dx
        alpha = a0 / np.sqrt(t + 1.0)
        v = beta * v - alpha * g
        h = _repair(h + v, n)
        c5 = _eval(h)
        if c5 < bc:
            bc = c5
            bh = h.copy()
    return bh, bc

def _anneal(h, t0, t1, ns, rng):
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, n)
    c5c = _eval(h)
    bc, bh = c5c, h.copy()
    for st in range(ns):
        t = t0 * (t1 / t0) ** (st / max(ns - 1, 1))
        i, j = rng.randint(n), rng.randint(n)
        while i == j:
            j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        h2 = _repair(h, n)
        c5n = _eval(h2)
        if c5n < c5c or rng.rand() < np.exp(-(c5n - c5c) / max(t, 1e-6)):
            h, c5c = h2, c5n
            if c5c < bc:
                bc, bh = c5c, h.copy()
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            h = _repair(h, n)
    return bh, bc

def _coord(h, nit, rng, scales):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for sc in scales:
        for _ in range(nit):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h, nn):
    return np.interp(np.linspace(0, 2, nn, False), np.linspace(0, 2, len(h), False), h)

def _mkb(n, p, rng):
    h = np.zeros(n)
    if p == 0:
        h[::2] = 1.0
    elif p == 1:
        for i in range(n):
            h[i] = 1.0 if (i % 3) < 2 else 0.0
    elif p == 2:
        gr = (1.0 + 5.0 ** 0.5) / 2.0
        h[(np.arange(n) * gr % n) < n / 2] = 1.0
    elif p == 3:
        for i in range(n):
            h[i] = 1.0 if (i % 5) < 3 else 0.0
    else:
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    h = (h > 0.5).astype(float)
    return _repair(h, n)
