def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _lssm(x, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _lssm_grad(x, alpha):
    n = len(x); dx = 2.0/n
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15: continue
        sh = (n-1) - k; wk = w[k]*dx
        if sh >= 0:
            g[:n-sh] += wk*(1.0-x[sh:])
            if sh > 0: g[sh:] -= wk*x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-x[:n-sp])
            g[:n-sp] -= wk*x[sp:]
    return g

def _pnorm_obj(x, p, lam):
    n = len(x); dx = 2.0/n
    c = np.correlate(x, 1.0-x, mode='full') * dx
    v = np.sum(c**p)**(1.0/p)
    if lam > 0: v += lam*(x.sum()-n/2.0)**2
    return v

def _pnorm_grad(x, p, lam):
    n = len(x); dx = 2.0/n
    c = np.correlate(x, 1.0-x, mode='full')*dx
    S = np.sum(c**p)
    base = S**(1.0/p-1.0)
    dc = c**(p-1)
    g = np.zeros(n)
    for k in range(2*n-1):
        if dc[k] < 1e-30: continue
        sh = (n-1)-k; w = dc[k]*base*dx
        if sh >= 0:
            g[:n-sh] += w*(1.0-x[sh:])
            if sh > 0: g[sh:] -= w*x[:n-sh]
        else:
            sp = -sh
            g[sp:] += w*(1.0-x[:n-sp])
            g[:n-sp] -= w*x[sp:]
    if lam > 0: g += 2.0*lam*(x.sum()-n/2.0)
    return g

def _lbfgsb(h, obj, grad, p=None, lam=None, maxiter=500):
    n = len(h)
    bnds = [(0.0,1.0)]*n
    try:
        r = minimize(obj, h, jac=grad, method='L-BFGS-B',
                     args=(p, lam) if p else (), bounds=bnds,
                     options={'maxiter':maxiter,'ftol':1e-20,'gtol':1e-12})
        return _repair(r.x, n)
    except: return _repair(h, n)

def _lssm_lbfgsb(h, alpha, maxiter=800):
    n = len(h)
    bnds = [(0.0,1.0)]*n
    try:
        r = minimize(_lssm, h, jac=_lssm_grad, args=(alpha,),
                     method='L-BFGS-B', bounds=bnds,
                     options={'maxiter':maxiter,'ftol':1e-20,'gtol':1e-12})
        return _repair(r.x, n)
    except: return _repair(h, n)

def _coord_descent(h, n_passes=3, n_samples=200):
    n = len(h)
    h = h.copy()
    best = _eval(h)
    rng = np.random.RandomState(12345)
    for _ in range(n_passes):
        order = rng.permutation(n)
        for _s in range(n_samples):
            j = order[_s % n]
            t_old = h[j]
            c = np.correlate(h, 1.0-h, mode='full')
            slopes = np.zeros(2*n-1)
            for k in range(2*n-1):
                sh = (n-1)-k
                a = 0.0
                if 0 <= j < n and 0 <= j+sh < n: a += (1.0-h[j+sh])
                if 0 <= j-sh < n: a -= h[j-sh]
                slopes[k] = a
            lo, hi = -t_old, 1.0-t_old
            for _ in range(40):
                m1 = lo + (hi-lo)/3.0
                m2 = hi - (hi-lo)/3.0
                f1 = float(np.max(c + slopes*m1))
                f2 = float(np.max(c + slopes*m2))
                if f1 < f2: hi = m2
                else: lo = m1
            delta = (lo+hi)/2.0
            h_new = h.copy(); h_new[j] = t_old + delta
            h_new = _repair(h_new, n)
            c5n = _eval(h_new)
            if c5n < best - 1e-15:
                h = h_new; best = c5n
    return h, best

def _binary_targeted(h, n_iters):
    n = len(h)
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    c5c = _eval(h)
    rng = np.random.RandomState(777)
    for _ in range(n_iters):
        c = np.correlate(h, 1.0-h, mode='full')
        ki = int(np.argmax(c))
        sh = (n-1)-ki
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_delta = 0
        best_pair = None
        n_try = min(200, len(ones)*len(zeros))
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c5n = _eval(h)
            if c5n < c5c + best_delta:
                best_delta = c5n - c5c
                best_pair = (i, j)
            h[i], h[j] = 1.0, 0.0
        if best_pair is not None and best_delta < 0:
            i, j = best_pair
            h[i], h[j] = 0.0, 1.0
            h = _repair(h, n)
            c5c = _eval(h)
    return h, c5c

def _binary_3swap(h, n_iters):
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, n)
    c5c = _eval(h)
    rng = np.random.RandomState(999)
    for _ in range(n_iters):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 3 or len(zeros) < 3: break
        idx1 = rng.choice(len(ones), 3, replace=False)
        idx2 = rng.choice(len(zeros), 3, replace=False)
        a1 = ones[idx1]; a2 = zeros[idx2]
        for i in a1: h[i] = 0.0
        for j in a2: h[j] = 1.0
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5c - 1e-14:
            c5c = c5n
        else:
            for i in a1: h[i] = 1.0
            for j in a2: h[j] = 0.0
            h = _repair(h, n)
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _worst_step(h, max_rounds=30):
    nc = len(h); c5c = _eval(h); dx = 2.0/nc
    for _ in range(max_rounds):
        c = np.correlate(h, 1.0-h, mode='full')
        ki = int(np.argmax(c))
        sh = (nc-1)-ki
        g = np.zeros(nc)
        if sh >= 0:
            g[:nc-sh] += dx*(1.0-h[sh:])
            if sh > 0: g[sh:] -= dx*h[:nc-sh]
        else:
            sp = -sh
            g[sp:] += dx*(1.0-h[:nc-sp])
            g[:nc-sp] -= dx*h[sp:]
        moved = False
        for step in [0.03, 0.01, 0.005, 0.002, 0.001]:
            h_new = _repair(h - step*g, nc)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h, c5c = h_new, c5n; moved = True; break
        if not moved: break
    return h, c5c

    # Phase 1: L-BFGS-B with log-sum-exp (120s)
    for alpha in [3000, 10000, 50000, 200000]:
        if elapsed() > 120: break
        h_new = _lssm_lbfgsb(best_h, alpha, 1000)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new.copy()

    # Phase 2: p-norm L-BFGS-B (100s)
    for p, lam in [(50, 100), (200, 500), (500, 2000)]:
        if elapsed() > 220: break
        h_new = _lbfgsb(best_h, _pnorm_obj, _pnorm_grad, p, lam, 600)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new.copy()

    # Phase 3: Coordinate descent (150s)
    if elapsed() < budget_s - 300:
        h_c, c5_c = _coord_descent(best_h, n_passes=5, n_samples=300)
        if c5_c < best_c5: best_c5, best_h = c5_c, h_c.copy()

    # Phase 4: Coarse-to-fine multi-resolution (150s)
    for n_try in [200, 250, 280, 340, 360]:
        if elapsed() > budget_s - 150: break
        h_r = _repair(_resample(best_h, n_try), n_try)
        h_r = _lssm_lbfgsb(h_r, 50000, 500)
        h_r, c5_r = _worst_step(h_r, 20)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 5: Intensive L-BFGS-B + p-norm on best (150s)
    if elapsed() < budget_s - 150:
        for alpha in [100000, 500000, 2000000]:
            if elapsed() > budget_s - 150: break
            h_new = _lssm_lbfgsb(best_h, alpha, 2000)
            c5 = _eval(h_new)
            if c5 < best_c5: best_c5, best_h = c5, h_new.copy()
            h_comp = _repair(1.0-best_h, len(best_h))
            h_comp = _lssm_lbfgsb(h_comp, alpha, 1000)
            c5c = _eval(h_comp)
            if c5c < best_c5: best_c5, best_h = c5c, h_comp.copy()

    # Phase 6: Binary targeted search (200s)
    if elapsed() < budget_s - 100:
        n_it = max(int((budget_s - 100 - elapsed()) * 300), 2000)
        h_b, c5_b = _binary_targeted(best_h, n_it)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b.copy()
            h_p = _repair(h_b, len(h_b))
            h_p = _lssm_lbfgsb(h_p, 200000, 500)
            c5_p = _eval(h_p)
            if c5_p < best_c5: best_c5, best_h = c5_p, h_p.copy()

    # Phase 7: 3-swap binary optimization (100s)
    if elapsed() < budget_s - 50:
        n_it = max(int((budget_s - 50 - elapsed()) * 500), 1000)
        h_3, c5_3 = _binary_3swap(best_h, n_it)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_3.copy()
            h_p = _lssm_lbfgsb(_repair(h_3, len(h_3)), 500000, 500)
            c5_p = _eval(h_p)
            if c5_p < best_c5: best_c5, best_h = c5_p, h_p.copy()

    # Phase 8: Worst-shift gradient + polish (80s)
    if elapsed() < budget_s - 20:
        h_w, c5_w = _worst_step(best_h, 50)
        if c5_w < best_c5:
            best_c5, best_h = c5_w, h_w.copy()
            h_w = _lssm_lbfgsb(h_w, 2000000, 2000)
            c5_w = _eval(h_w)
            if c5_w < best_c5: best_c5, best_h = c5_w, h_w.copy()

    # Final micro coordinate descent
    if elapsed() < budget_s - 3:
        h_m, c5_m = _coord_descent(best_h, n_passes=2, n_samples=200)
        if c5_m < best_c5: best_c5, best_h = c5_m, h_m.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
