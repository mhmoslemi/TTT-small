def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

    return h

import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _sf(x, alpha, tk):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    nc = len(c)
    if tk and 0 < tk < nc:
        s = np.full(nc, -100.0)
        idx = np.argsort(c)[-tk:]
        s[idx] = alpha * (c[idx] - m)
    else:
        s = alpha * (c - m)
    s = np.clip(s, -300, 300)
    sm = s.max()
    e = np.exp(s - sm)
    val = m + np.log(e.sum()) / alpha
    w = e / e.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for k in range(nc):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = nm1 - k
        wk *= dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return val, g


def _adam(h, a0, a1, tk, nit, lr, seed, dl):

    rng = np.random.RandomState(seed)
    x = h.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    bx = x.copy()
    bc = _eval(x)
    for i in range(nit):
        if time.time() > dl:
            break
        t += 1
        f = i / max(nit - 1, 1)
        a = a0 * (a1 / a0) ** f
        l = lr * (0.05 ** f)
        _, g = _sf(x, a, tk)
        g -= g.mean()
        m = 0.85 * m + 0.15 * g
        v = 0.999 * v + 0.001 * g * g
        x = x - l * (m / (1 - 0.85 ** t)) / (np.sqrt(v / (1 - 0.999 ** t)) + 1e-8)
        x = _repair(x, n)
        c = _eval(x)
        if c < bc:
            bc, bx = c, x.copy()
        if t % 400 == 0:
            x += 0.003 * rng.randn(n)
            x = _repair(x, n)
    return bx, bc


def _slsqp(h, a, tk, mi):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def _f(x):
        v, _ = _sf(x, a, tk)
        return v

    def _g(x):
        _, g = _sf(x, a, tk)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_f, h, jac=_g, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mi, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)


def _lbfgs(h, a, tk, mi):
    n = len(h)
    z0 = h[:n - 1].copy()
    bnds = [(0.0, 1.0)] * (n - 1)

    def _f(z):
        x = np.empty(n)
        x[:n - 1] = z
        x[n - 1] = n * 0.5 - z.sum()
        pen = 0.0
        if x[n - 1] < 0:
            pen += 100 * x[n - 1] ** 2
            x[n - 1] = 0.0
        elif x[n - 1] > 1:
            pen += 100 * (x[n - 1] - 1) ** 2
            x[n - 1] = 1.0
        v, _ = _sf(np.clip(x, 0, 1), a, tk)
        return v + pen

    try:
        r = minimize(_f, z0, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': mi, 'ftol': 1e-20})
        x = np.empty(n)
        x[:n - 1] = r.x
        x[n - 1] = n * 0.5 - r.x.sum()
        return _repair(x, n)
    except Exception:
        return _repair(h, n)


def _resample(h, nn):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return np.interp(xn, xo, h)


def _binary_sa(h, n, rng, nit, dl):
    x = (h > 0.5).astype(float)
    x = _repair(x, n)
    cc = _eval(x)
    bc = cc
    bx = x.copy()

        else:

            sc = max(sc * 0.9995, 1e-10)
        else:

            cc = bc
            sc *= 0.94
    return bx, bc

import numpy as np
from scipy.optimize import minimize
import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _sf(x, alpha, tk):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    nc = len(c)
    if tk and 0 < tk < nc:
        s = np.full(nc, -100.0)
        idx = np.argsort(c)[-tk:]
        s[idx] = alpha * (c[idx] - m)
    else:
        s = alpha * (c - m)
    s = np.clip(s, -300, 300)
    sm = s.max()
    e = np.exp(s - sm)
    val = m + np.log(e.sum()) / alpha
    w = e / e.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for k in range(nc):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = nm1 - k
        wk *= dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return val, g


class _SLP:
    def __init__(self, a, tk):
        self.a = a
        self.tk = tk
    def f(self, x):
        return _sf(x, self.a, self.tk)[0]
    def g(self, x):
        return _sf(x, self.a, self.tk)[1]
    def eq(self, x):
        return x.sum() - len(x) * 0.5
    def eqj(self, x):
        return np.ones(len(x))


class _LBP:
    def __init__(self, n, a, tk):
        self.n = n
        self.a = a
        self.tk = tk
    def f(self, z):
        x = np.empty(self.n)
        x[:self.n - 1] = z
        x[self.n - 1] = self.n * 0.5 - z.sum()
        pen = 0.0
        if x[self.n - 1] < 0:
            pen += 200 * x[self.n - 1] ** 2
        elif x[self.n - 1] > 1:
            pen += 200 * (x[self.n - 1] - 1) ** 2
        x = np.clip(x, 0, 1)
        v, _ = _sf(x, self.a, self.tk)
        return v + pen
    def g(self, z):
        x = np.empty(self.n)
        x[:self.n - 1] = z
        x[self.n - 1] = self.n * 0.5 - z.sum()
        x = np.clip(x, 0, 1)
        _, gf = _sf(x, self.a, self.tk)
        return gf[:self.n - 1] - gf[self.n - 1]


def _slsqp(h, a, tk, mi):
    n = len(h)
    p = _SLP(a, tk)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': p.eq, 'jac': p.eqj}]
    try:
        r = minimize(p.f, h, jac=p.g, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mi, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)


def _lbfgs(h, a, tk, mi):
    n = len(h)
    p = _LBP(n, a, tk)
    z0 = h[:n - 1].copy()
    bnds = [(0.0, 1.0)] * (n - 1)
    try:
        r = minimize(p.f, z0, jac=p.g, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': mi, 'ftol': 1e-20})
        x = np.empty(n)
        x[:n - 1] = r.x
        x[n - 1] = n * 0.5 - r.x.sum()
        return _repair(x, n)
    except Exception:
        return _repair(h, n)


def _adam(h, a0, a1, tk, nit, lr, seed, dl):
    n = len(h)
    rng = np.random.RandomState(seed)
    x = h.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    bx = x.copy()
    bc = _eval(x)
    for i in range(nit):
        if time.time() > dl:
            break
        t += 1
        f = i / max(nit - 1, 1)
        a = a0 * (a1 / a0) ** f
        l = lr * (0.05 ** f)
        _, g = _sf(x, a, tk)
        g -= g.mean()
        m = 0.85 * m + 0.15 * g
        v = 0.999 * v + 0.001 * g * g
        x = x - l * (m / (1 - 0.85 ** t)) / (np.sqrt(v / (1 - 0.999 ** t)) + 1e-8)
        x = _repair(x, n)
        c = _eval(x)
        if c < bc:
            bc, bx = c, x.copy()
        if t % 400 == 0:
            x += 0.003 * rng.randn(n)
            x = _repair(x, n)
    return bx, bc


def _resample(h, nn):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return np.interp(xn, xo, h)


def _binary_sa(h, n, rng, nit, dl):
    x = (h > 0.5).astype(float)
    x = _repair(x, n)
    cc = _eval(x)
    bc = cc
    bx = x.copy()
    T = 0.01
    for _ in range(nit):
        if time.time() > dl:
            break
        i = rng.randint(n)
        x[i] = 1.0 - x[i]
        x = _repair(x, n)
        cn = _eval(x)
        if cn <= cc or rng.random() < np.exp(-(cn - cc) / max(T, 1e-10)):
            cc = cn
            if cn < bc:
                bc, bx = cn, x.copy()
        else:
            x[i] = 1.0 - x[i]
            x = _repair(x, n)
        T *= 0.9999
    return bx, bc


def _cd(h, n, rng, dl, sc0):
    x = h.copy()
    cc = _eval(x)
    bc = cc
    bx = x.copy()
    sc = sc0
    for _ in range(500000):
        if time.time() > dl or sc < 1e-9:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = sc * rng.randn()
        x[i] += d
        x[j] -= d
        x = _repair(x, n)
        cn = _eval(x)
        if cn < cc - 1e-16:
            cc = cn
            if cn < bc:
                bc, bx = cn, x.copy()
            sc = max(sc * 0.9995, 1e-10)
        else:
            x = bx.copy()
            cc = bc
            sc *= 0.94
    return bx, bc


def _try_init(h, a, tk, mi, seed, dl):
    """Quick optimize from a given initialization."""
    x, c = _adam(h, 500, 50000, tk, 2000, 0.01, seed, dl)
    if c > _eval(h):
        x, c = h, _eval(h)
    x = _slsqp(x, a, tk, mi)

The objective is `f(x) = m + log(Σ_k exp(α(c_k - m))) / α` where `m = max_j c_j`.

= ∂m/∂x + Σ_k w_k * (∂c_k/∂x - ∂m/∂x)

= ∂m/∂x + Σ_k w_k * ∂c_k/∂x - (Σ_k w_k) * ∂m/∂x

= ∂m/∂x + Σ_k w_k * ∂c_k/∂x - ∂m/∂x

= Σ_k w_k * ∂c_k/∂x
