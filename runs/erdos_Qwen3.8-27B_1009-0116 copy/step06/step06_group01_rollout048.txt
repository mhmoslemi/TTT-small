def _repair(h, n):
    # Ensure 0<=h<=1 and sum(h)=n/2
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k, dx):

    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k  # shift in terms of array indices
    if sh >= 0:
        # h[i] * (1 - h[i+sh]) for i in [0, n-sh)
        # h[i] * h[i-sh] for i in [sh, n)
        m = n - sh
        g[:m] += (1.0 - h[sh:m+sh])  # d/dh[i] of h[i](1-h[i+sh]) = (1-h[i+sh])
        if sh > 0:
            g[sh:] -= h[:n-sh]  # d/dh[i] of -h[i-sh]*h[i] = -h[i-sh]
    else:

        # h[i] * (1 - h[i-sp]) for i in [sp, n)
        # -h[i] * h[i+sp] for i in [0, n-sp)
        m = n - sp
        g[sp:] += (1.0 - h[:m])  # (1-h[i-sp])

    s = np.clip(alpha * (c - m), -500, 500)

def _smograd(h, alpha):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_lag(h, k, dx)
    return g

def _adam_step(h, g, state, lr=0.01):
    # Adam optimizer step
    if 'm' not in state:
        state['m'] = np.zeros_like(h)
        state['v'] = np.zeros_like(h)
        state['t'] = 0
    state['t'] += 1
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    state['m'] = beta1*state['m'] + (1-beta1)*g
    state['v'] = beta2*state['v'] + (1-beta2)*g*g
    mhat = state['m'] / (1 - beta1**state['t'])
    vhat = state['v'] / (1 - beta2**state['t'])
    h_new = h - lr * mhat / (np.sqrt(vhat) + eps)
    return h_new

def _slsqp(h, alpha, maxiter):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:sh+m])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh
        m = n - sp
        g[sp:] += (1.0 - h[:m])
        g[:m] -= h[sp:]
    return g * dx

def _smoobj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _smograd(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        g += w[k] * _grad_lag(x, k, dx)
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_smoobj, h, jac=_smograd, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _focused_descent(h, nc, max_rounds, top_k=3):

    dx = 2.0 / nc
    for r in range(max_rounds):

def _nesterov_descent(h, nc, max_iter, lr0, alpha):
    c5c = _eval(h)
    v = np.zeros(nc)
    y = h.copy()
    for t in range(1, max_iter + 1):
        mu = (1 + np.sqrt(1 + 4 * (t - 1) * (t - 1))) / (1 + np.sqrt(1 + 4 * t * t))
        g = _smograd(y, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-13:

            y = y_new

            y = h
