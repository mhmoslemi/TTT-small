So we want to maximize ∫f(x)f(x+k) dx for the critical k's, subject to 0 ≤ 0.5+f ≤ 1 (i.e., -0.5 ≤ f ≤ 0.5) and ∫f = 0.

This is a "maximize autocorrelation at critical lags" problem, which is more standard.

This suggests that a good h should have strong positive autocorrelation at the critical lags. This means h should have a specific periodic structure.

If the critical lags are at specific values, h should be approximately periodic with period related to those lags.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _subgrad_lag(h, k, n):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _momentum_descent(h, n, t0, time_limit, max_iter=80000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _eval(best_h)
    v = np.zeros(n)
    beta = 0.88
    base_step = 0.06
    stall = 0
    n_top = 3
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = _eval_corr(h)
        c5c = corr.max()
        if c5c < best_c5 - 5e-14:
            best_c5 = c5c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        # Adaptive lag count
        max_val = c5c
        frac = 0.92 if stall > 200 else 0.97
        active = np.where(corr >= frac * max_val)[0]
        if len(active) < 1:
            active = np.array([np.argmax(corr)])
        if len(active) > 7:
            active = np.argsort(corr)[-7:]
        if len(active) < n_top and stall > 500:
            top_all = np.argsort(corr)[-n_top:]
            active = np.unique(np.concatenate([active, top_all]))
        g = np.zeros(n)
        for k in active:
            g += _subgrad_lag(h, k, n)
        g *= dx
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        v = beta * v + (1.0 - beta) * g / g_norm
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v /= vn
        step = base_step / (1.0 + 2e-5 * it)
        if stall > 300:
            step *= 0.5
        best_try = None
        best_val = c5c
        s = step
        for _ in range(10):
            h_new = _repair(h - s * v, n)
            c5t = _eval(h_new)
            if c5t < best_val - 5e-14:
                best_val = c5t
                best_try = h_new
            else:
                s *= 0.5
        if best_try is not None:
            h = best_try
        else:
            if stall > 800:
                break
    return best_h, best_c5

def _ls_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _ls_grad(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_polish(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_ls_obj, h, jac=_ls_grad, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _pairwise_cd(h, n, rng, scale, n_iter, t0, tlim):
    c5c = _eval(h)
    for _ in range(n_iter):
        if time.time() - t0 > tlim:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5t = _eval(h)
        if c5t < c5c - 1e-14:
            c5c = c5t
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c
