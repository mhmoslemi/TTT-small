def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:

                step *= 0.5
                y_new = _proj(y - step * z, target)
                f_new = _obj(y_new, alpha)
            y = y_new
        else:
            t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * i * i))
            beta = (t_k - 1.0) / t_k
            z = g + beta * z
            y_trial = y - 0.0
            y_new = _proj(y_trial - step * z, target)
            f_new = _obj(y_new, alpha)
            if f_new > f + 1e-12:
                step *= 0.5
                y_new = _proj(y - step * z, target)
                f_new = _obj(y_new, alpha)
                if f_new > f + 1e-12:
                    step *= 0.5
            else:
                step = min(step * 1.1, 0.5)
            y = y_new
        g_prev = g
        c5 = _eval(y)
        if c5 < best_c5:
            best_c5 = c5
            best_h = y.copy()
    return best_h, best_c5

def _g_at_j(h, j):
    n = len(h)
    g = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        t1 = 1.0 - h[j + lag] if 0 <= j + lag < n else 1.0
        t2 = h[j - lag] if 0 <= j - lag < n else 0.0
        g[k] = t1 - t2
    return g

def _pair_cd(h, t_end, rng, top_k=5, n_cands=30):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _eval(best_h)
    c = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    step_base = 0.02
    for it in range(2000000):
        if time.time() > t_end:
            break
        active = np.argsort(c)[::-1][:top_k]
        max_c = c.max()
        best_delta = 0.0
        best_improve = 1e-15
        best_j = -1
        best_i = -1
        cands = rng.choice(n, size=n_cands, replace=False)
        for jj in cands:
            gi = _g_at_j(best_h, jj)
            for ii in cands:
                if ii == jj:
                    continue
                gk = _g_at_j(best_h, ii)
                dg = gi - gk
                for direction in [1, -1]:
                    for s_frac in [step_base, step_base * 0.5, step_base * 0.1]:
                        d = direction * s_frac
                        if best_h[jj] + d < 0 or best_h[jj] + d > 1:
                            continue
                        if best_h[ii] - d < 0 or best_h[ii] - d > 1:
                            continue
                        new_max = max(c[active] + d * dg[active])
                        imp = max_c - new_max
                        if imp > best_improve:
                            best_improve = imp
                            best_delta = d
                            best_j = jj
                            best_i = ii
        if best_j >= 0:
            best_h[best_j] += best_delta
            best_h[best_i] -= best_delta
            best_h = _repair(best_h, n)
            c = np.correlate(best_h, 1.0 - best_h, mode='full')
            new_c5 = float(c.max() * dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
            if it % 100 == 0:
                step_base *= 0.995
    return best_h, best_c5

def _sa_pairs(h, t_end, rng):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c_cur = _eval(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    t0v, t1v = 0.01, 1e-6
    n_iter = 2000000
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0v * (t1v / t0v) ** (i / max(1, n_iter - 1))
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.005 + 0.02 * t / 0.01)
        ov, ow = h_cur[i1], h_cur[i2]
        h_cur[i1], h_cur[i2] = ov + d, ow - d
        h_cur = _repair(h_cur, n)
        c_new = _eval(h_cur)
        if c_new < c_cur or (t > 1e-8 and rng.rand() < np.exp(-(c_new - c_cur) / t)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
        else:
            h_cur[i1], h_cur[i2] = ov, ow
            h_cur = _repair(h_cur, n)
    return best_h, best_c

def _fine_cd(h, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    c5c = _eval(h_cur)
    best_h = h_cur.copy()
    best_c = c5c
    for it in range(5000000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 1e-4 * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _eval(h_cur)
        if c5n < c5c - 1e-18:
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
    return best_h, best_c

def _multi_lag(h, t_end):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for _r in range(200):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(c)[::-1][:4]
        g = np.zeros(n)
        for kk in top_k:
            lag = kk - (n - 1)
            for j in range(n):
                t1 = 1.0 - h_cur[j + lag] if 0 <= j + lag < n else 1.0
                t2 = h_cur[j - lag] if 0 <= j - lag < n else 0.0
                g[j] += t1 - t2
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def rem():
        return max(0.0, budget_s - (time.time() - t0))
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP warmup
    for alpha in [500, 5000, 30000]:
        if rem() < 180:
            break
        h2 = _slsqp(best_h, alpha, 3000, t0 + rem() - 179)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()

    # Phase 2: Nesterov gradient
    if rem() > 250:
        h2, c2 = _nesterov(best_h, 15000, t0 + rem() - 249)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 3: Multi-lag descent
    if rem() > 220:
        h2, c2 = _multi_lag(best_h, t0 + rem() - 219)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 4: Pair coordinate descent (main workhorse)
    if rem() > 350:
        h2, c2 = _pair_cd(best_h, t0 + rem() - 349, rng, top_k=4, n_cands=20)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 5: SLSQP polish
    if rem() > 200:
        for alpha in [50000, 200000]:
            if rem() < 190:
                break
            h2 = _slsqp(best_h, alpha, 3000, t0 + rem() - 189)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()

    # Phase 6: SA
    if rem() > 180:
        h2, c2 = _sa_pairs(best_h, t0 + rem() - 170, rng)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 7: More pair CD
    if rem() > 120:
        h2, c2 = _pair_cd(best_h, t0 + rem() - 110, rng, top_k=3, n_cands=15)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 8: Fine CD polish
    if rem() > 50:
        h2, c2 = _fine_cd(best_h, t0 + rem() - 45, rng)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 9: Final SLSQP
    if rem() > 30:
        h2 = _slsqp(best_h, 100000, 2000, t0 + rem() - 25)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()

    # Phase 10: Ultra-fine CD
    if rem() > 15:
        h2, c2 = _fine_cd(best_h, t0 + rem() - 10, rng)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
