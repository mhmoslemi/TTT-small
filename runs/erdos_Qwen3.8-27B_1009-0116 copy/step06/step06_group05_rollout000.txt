def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_shift(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n-sh] += dx * (1.0 - h[sh:])
        if sh > 0: g[sh:] -= dx * h[:n-sh]
    else:

        g[sp:] += dx * (1.0 - h[:n-sp])

    idx = np.argsort(c)[::-1][:topk]

    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + float(np.log(np.sum(np.exp(s))) + s.max()) / alpha

    m = c.max()

        if cb < best_c5: best_h, best_c5, best_n = hb, cb, len(hb)

    # Phase 5: More projected grad from perturbed (150s)
    if el() < budget_s - 150:
        for rep in range(5):
            t_end = t0 + min(budget_s - 150, el() + 30)
            if el() > budget_s - 160: break
            hp = best_h + 0.01 * rng.randn(len(best_h))
            hp = _repair(hp)
            hp, cp = _proj_grad(hp, t_end, topk=5, lr0=0.05, beta=0.92)
            if cp < best_c5: best_h, best_c5 = hp, cp

    # Phase 6: SLSQP with very high alpha (100s)
    if el() < budget_s - 100:
        for a in [2000000, 10000000]:
            t_end = t0 + min(budget_s - 100, el() + 50)
            if el() > budget_s - 110: break
            hn = _slsqp(best_h, a, 5000, 1e-20)
            c = _eval(hn)
            if c < best_c5: best_h, best_c5 = hn, c

    # Phase 7: Final fine projected grad (80s)
    if el() < budget_s - 30:
        t_end = t0 + min(budget_s - 30, el() + 80)
        best_h, best_c5 = _proj_grad(best_h, t_end, topk=1, lr0=0.02, beta=0.8)
        best_h = _slsqp(best_h, 5000000, 3000, 1e-20)
        best_c5 = _eval(best_h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
