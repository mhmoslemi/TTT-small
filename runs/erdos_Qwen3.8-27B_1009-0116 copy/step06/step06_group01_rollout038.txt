def _penalty_obj(x, alpha, mu):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    base = m + np.log(float(np.sum(np.exp(s)))) / alpha
    penalty = mu * (x.sum() - n / 2.0)**2
    return base + penalty

def _penalty_grad(x, alpha, mu):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    thresh = w.max() * 1e-6
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < thresh:
            continue
        g += wk * _grad_k(x, k, n)
    g *= dx
    # Penalty gradient
    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

def _lbfgs_opt(h, alpha, mu, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_penalty_obj, h.copy(), jac=_penalty_grad,
                       args=(alpha, mu), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)
