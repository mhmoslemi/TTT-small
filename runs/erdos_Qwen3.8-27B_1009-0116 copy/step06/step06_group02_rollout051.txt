def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _slsqp_polish(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        if time.time() > t_end:
            return np.max(np.correlate(x, 1-x, 'full')) * 2.0 / n + 1.0
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def _gradFn(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        return _wgrad(x, w), _obj(x)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_gradFn, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _binary_opt(h, n, max_rounds, t_end):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = _repair(h, n)
    cur = _eval(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in ones.copy():
            done = False
            for j in zeros.copy():
                h[i] = 0.0
                h[j] = 1.0
                c2 = _eval(h)
                if c2 < cur - 1e-13:
                    cur = c2
                    improved = True
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    done = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if done:
                break
        if not improved:
            break
    return h, cur

def _pair_cd(h, n, n_iter, scales, t_end, rng_seed=123):
    rng = np.random.RandomState(rng_seed)
    h = h.copy()
    cur = _eval(h)
    for scale in scales:
        for _ in range(n_iter):
            if time.time() > t_end:
                return h, cur
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                continue
            ni = min(max(ni, 0.0), 1.0)
            nj = min(max(nj, 0.0), 1.0)
            h[i] = ni
            h[j] = nj
            cn = _eval(h)
            if cn < cur - 1e-15:
                cur = cn
            else:
                h[i] = ni - d
                h[j] = nj + d
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: PGD with increasing alpha
    for alpha in [2000, 8000, 30000, 100000]:
        if rem() < 120:
            break
        h1, c1 = _pgd(best_h, alpha, 30000, t0 + rem() - 10, lr=0.004, mom=0.8)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1
            best_n = len(h1)

    # Phase 2: Focus on worst shifts
    if rem() > 80:
        h2, c2 = _reweight(best_h, 30000, 6, t0 + rem() - 10, lr=0.002, temp=6.0)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2
        if rem() > 40:
            h2b, c2b = _reweight(best_h, 20000, 3, t0 + rem() - 10, lr=0.001, temp=10.0)
            if c2b < best_c5:
                best_c5 = c2b
                best_h = h2b

    # Phase 3: Multi-resolution search
    if rem() > 180:
        for n_try in [400, 440, 480, 520, 560]:
            if rem() < 140:
                break
            hr = _repair(_resample(best_h, n_try), n_try)
            for alpha in [8000, 40000]:
                if rem() < 130:
                    break
                hr, cr = _pgd(hr, alpha, 15000, t0 + rem() - 10, lr=0.003, mom=0.8)
            hr, cr = _reweight(hr, 10000, 5, t0 + rem() - 10, lr=0.0015, temp=7.0)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr
                best_n = n_try

    # Phase 4: SLSQP polish at best resolution
    if rem() > 80:
        h4 = _slsqp_polish(best_h, 100000, 3000, t0 + rem() - 5)
        c4 = _eval(h4)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4
            best_n = len(h4)

    # Phase 5: More reweighting + PGD
    if rem() > 50:
        h5, c5 = _reweight(best_h, 25000, 4, t0 + rem() - 10, lr=0.001, temp=12.0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h5
        if rem() > 30:
            h5b, c5b = _pgd(best_h, 200000, 20000, t0 + rem() - 10, lr=0.001, mom=0.9)
            if c5b < best_c5:
                best_c5 = c5b
                best_h = h5b

    # Phase 6: Binary optimization (try at best_n and nearby)
    if rem() > 35:
        for nb in [best_n, max(60, best_n - 40)]:
            if rem() < 30:
                break
            hb = _repair(_resample(best_h, nb), nb)
            hb, c5 = _binary_opt(hb, nb, 600, t0 + rem() - 5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hb
                best_n = nb

    # Phase 7: Pairwise coordinate descent
    if rem() > 25:
        h7, c7 = _pair_cd(best_h, best_n, 30000, [0.003, 0.001, 0.0003],
                          t0 + rem() - 5, rng_seed=42)
        if c7 < best_c5:
            best_c5 = c7
            best_h = h7

    # Phase 8: Final SLSQP with very high alpha
    if rem() > 20:
        h8 = _slsqp_polish(best_h, 500000, 5000, t0 + rem() - 3)
        c8 = _eval(h8)
        if c8 < best_c5:
            best_c5 = c8
            best_h = h8
            best_n = len(h8)

    # Phase 9: Final reweighting polish
    if rem() > 10:
        h9, c9 = _reweight(best_h, 15000, 2, t0 + rem() - 2, lr=0.0005, temp=15.0)
        if c9 < best_c5:
            best_c5 = c9
            best_h = h9

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
