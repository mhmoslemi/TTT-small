import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    if abs(s - target) < 1e-12: return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _sm_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _f1(x): return x.sum() - len(x) * 0.5
def _f2(x): return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    try:

                     method='SLSQP', bounds=[(0.,1.)]*n,
                     constraints=[{'type':'eq','fun':_f1,'jac':_f2}],
                     options={'maxiter':maxiter, 'ftol':ftol})
        return _repair(r.x, n)
    except: return _repair(h, n)

        worst = int(np.argmax(c))
        sh = (n - 1) - worst
        s = abs(sh)
        if s == 0:
            # Lag 0: c = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
            # For binary: c = n/2. Can't improve. Skip.
            i = rng.randint(n); j = rng.randint(n)
            if h[i] != h[j]:

            bnd = []
            for m in range(n - sp):

                h = _repair(h, n)
                c5n = _eval(h)
                if c5n < c5: c5 = c5n
                else: h[i], h[j] = h[j], h[i]; h = _repair(h, n)
            continue
        # Try swaps involving boundary positions
        found = False
        n_try = min(50, len(bnd) * 2)
        for _ in range(n_try):
            m = bnd[rng.randint(len(bnd))]
            if sh >= 0:
                p1, p2 = m, m + s  # h[m] != h[m+s]
            else:
                p1, p2 = m, m + (-sh)
            # p1 and p2 are a boundary pair. One is 0, other is 1.
            # Option 1: swap with a random partner
            pi = rng.randint(n)
            if pi == p1 or pi == p2 or h[pi] == h[p1]: continue
            # Try: flip p1 and pi
            h[p1], h[pi] = h[pi], h[p1]
            h2 = _repair(h, n)
            c5n = _eval(h2)
            if c5n < c5 - 1e-15:
                c5 = c5n; found = True; break
            else:
                h[p1], h[pi] = h[pi], h[p1]
                h = _repair(h, n)
        if not found:
            # Random move for diversity
            i = rng.randint(n); j = rng.randint(n)
            if i != j and h[i] != h[j]:
                h[i], h[j] = h[j], h[i]
                h = _repair(h, n)
                c5n = _eval(h)
                if c5n < c5: c5 = c5n
                else: h[i], h[j] = h[j], h[i]; h = _repair(h, n)
    return h, c5
