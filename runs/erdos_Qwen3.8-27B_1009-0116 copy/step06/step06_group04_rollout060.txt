    order = list(range(n))
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        rng.shuffle(order)
        for i in order:
            if time.time() > deadline:
                break
            j = (i + 1) % n
            old_i, old_j = x[i], x[j]

            dc_i = _grad_cv(x, i)  # dcv/dx[i] for all k
            dc_j = _grad_cv(x, j)  # dcv/dx[j] for all k
            dc_diff = dc_i - dc_j  # dcv/d(d) where dx[i]+=d, dx[j]-=d

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    cv = np.correlate(x, 1.0 - x, mode='full')
    m = cv.max()
    s = np.clip(alpha * (cv - m), -500, 500)
    return float(m + np.log(float(np.sum(np.exp(s)))) / alpha)

    n = len(x)
    cv = np.correlate(x, 1.0 - x, mode='full')
    m = cv.max()
    s = np.clip(alpha * (cv - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += w[k] * (1.0 - x[sh:n])
            if sh > 0:
                g[sh:] -= w[k] * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - x[:n-sp])
            g[:n-sp] -= w[k] * x[sp:]
    return alpha * g

            if i < n - sp:

    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: L-BFGS-B alpha sweep (0-160s)
    dl1 = t0 + 160
    for alpha, pen in [(100, 50), (500, 100), (2000, 100), (10000, 100), (50000, 200), (200000, 200)]:

    dl2 = t0 + 380
    h_cd, c_cd = _pair_cd(best_h, dl2, 400, seed)
    if c_cd < best_c5:
        best_c5, best_h = c_cd, h_cd.copy()

    # Phase 3: L-BFGS-B high alpha (380-500s)
    dl3 = t0 + 500
    for alpha in [50000, 200000, 1000000]:
        if time.time() > dl3:
            break
        h_new = _lbfgs(best_h, alpha, 200, 6000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 4: Pair CD again (500-650s)
    dl4 = t0 + 650
    h_cd, c_cd = _pair_cd(best_h, dl4, 400, seed + 1)
    if c_cd < best_c5:
        best_c5, best_h = c_cd, h_cd.copy()

    # Phase 5: Multi-n exploration (650-800s)
    dl5 = t0 + 800
    for n_try in [best_n - 20, best_n + 10, best_n + 20, 290, 330, 370]:
        if time.time() > dl5:
            break
        if n_try < 80 or n_try > 500:
            continue
        if n_try % 2 != 0:
            n_try += 1
        h_r = _repair(_resample(best_h, n_try), n_try)
        h_new = _lbfgs(h_r, 20000, 100, 3000)
        h_cd2, c_cd2 = _pair_cd(h_new, min(dl5, time.time() + 25), 100, seed + n_try)
        c5 = min(_eval(h_new), c_cd2)
        if c5 < best_c5:

    # Phase 6: Final pair CD (800-950s)
    dl6 = t0 + 950
    h_cd, c_cd = _pair_cd(best_h, dl6, 500, seed + 2)
    if c_cd < best_c5:
        best_c5, best_h = c_cd, h_cd.copy()

    # Phase 7: Final L-BFGS-B (950-1060s)
    dl7 = t0 + 1060

        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Final
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
