    except Exception:
        pass
    return _repair(h, n)

def _nesterov_grad(h, alpha, n_steps, t_end):
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    f_prev = _smax_obj(h_cur, alpha)
    step = 0.03
    mu = 0.8
    g_prev = _smax_grad(h_cur, alpha)
    for s in range(n_steps):
        if time.time() > t_end:
            break
        g = _smax_grad(h_cur, alpha)
        v = mu * v + (1.0 - mu) * g
        h_y = _proj(h_cur - step * v, n * 0.5)
        f_y = _smax_obj(h_y, alpha)
        if f_y <= f_prev + 1e-13:
            dh = h_y - h_cur
            dg = g - g_prev
            den = np.dot(dg, dg) + 1e-15
            step_bb = max(0.008, min(0.4, abs(np.dot(dh, dg)) / den))
            step = 0.6 * step + 0.4 * step_bb
            h_cur = h_y
            g_prev = g.copy()
            f_prev = f_y
        else:
            accepted = False
            for _ in range(6):
                step *= 0.5
                h_y = _proj(h_cur - step * v, n * 0.5)
                f_y = _smax_obj(h_y, alpha)
                if f_y <= f_prev + 1e-13:
                    h_cur = h_y
                    f_prev = f_y
                    g_prev = g.copy()
                    accepted = True
                    break
            if not accepted:
                step = max(0.005, step)
                if s > 200:
                    break
    return h_cur

def _multi_lag(h, n_rounds, t_end, top_n):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(c)[::-1][:top_n]
        g = np.zeros(n)
        for kk in top_k:
            g += _grad_lag(h_cur, kk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                h_cur = h_new
                c5_cur = c5_new
                improved = True
                break
        if not improved:
            break
    return h_cur, c5_cur

def _block_cd(h, t_end, n_blocks, seed_b):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    rng_b = np.random.RandomState(seed_b)
    bs = max(6, n // 50)
    for _ in range(n_blocks):
        if time.time() > t_end:
            break
        start = rng_b.randint(0, n - bs)
        idx = slice(start, start + bs)
        improved = False
        for trial in range(40):
            d = rng_b.randn(bs) * 0.015
            h_new = h_cur.copy()
            h_new[idx] = np.clip(h_new[idx] + d, 0, 1)
            h_new = _repair(h_new, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
                improved = True
                break
    return h_cur, c5_cur

def _sa(h, n_iter, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    c_cur = _eval(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    t0s, t1s = 0.015, 5e-6
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0s * (t1s / t0s) ** (i / max(1, n_iter - 1))
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        scale = 0.01 + 0.06 * t
        d = rng.randn() * scale
        old1, old2 = h_cur[i1], h_cur[i2]
        h_cur[i1] = np.clip(old1 + d, 0, 1)
        h_cur[i2] = np.clip(old2 - d, 0, 1)
        h_cur = _repair(h_cur, n)
        c_new = _eval(h_cur)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-10)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
        else:
            h_cur[i1], h_cur[i2] = old1, old2
            h_cur = _repair(h_cur, n)
    return best_h, best_c

def _coord_cd(h, n_tries, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    c_cur = _eval(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 4e-5 * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c_new = _eval(h_cur)
        if c_new < c_cur - 1e-16:
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = ni - d, nj + d
    return best_h, best_c

def _perturb(h, rng, scale):
    n = len(h)
    return _repair(h + scale * rng.randn(n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remain():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Progressive SLSQP
    for a in [500, 3000, 15000, 60000, 200000, 600000]:
        if remain() < 290:
            break
        h_new = _slsqp(best_h, a, 2500, t0 + remain() - 289)
        c = _eval(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new

    # Phase 2: Nesterov projected gradient
    if remain() > 250:
        h_new = _nesterov_grad(best_h, 30000, 600, t0 + remain() - 249)
        c = _eval(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new

    # Phase 3: Multi-lag descent (top 10)
    if remain() > 210:
        h_new, c_new = _multi_lag(best_h, 400, t0 + remain() - 209, 10)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 4: Block coordinate descent
    if remain() > 180:
        h_new, c_new = _block_cd(best_h, t0 + remain() - 179, 600, 999)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 5: Simulated annealing
    if remain() > 150:
        n_sa = min(int((remain() - 140) * 400), 100000)
        h_new, c_new = _sa(best_h, n_sa, t0 + remain() - 140, rng)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 6: Perturbation restarts
    if remain() > 120:
        for _r in range(6):
            if remain() < 110:
                break
            h_try = _perturb(best_h, rng, 0.025)
            h_try = _slsqp(h_try, 100000, 1500, t0 + remain() - 109)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try

    # Phase 7: Final SLSQP + multi-lag
    if remain() > 90:
        h_new = _slsqp(best_h, 2000000, 8000, t0 + remain() - 85)
        c = _eval(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new
    if remain() > 70:
        h_new, c_new = _multi_lag(best_h, 300, t0 + remain() - 65, 12)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 8: Coordinate descent polish
    if remain() > 20:
        n_cd = min(int((remain() - 15) * 500), 300000)
        h_new, c_new = _coord_cd(best_h, n_cd, t0 + remain() - 15, rng)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
