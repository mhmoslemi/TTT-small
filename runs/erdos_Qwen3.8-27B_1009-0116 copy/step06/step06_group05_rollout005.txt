                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:

    c = _corr(h) * (2.0 / n)
    top = np.sort(c)[::-1][:k]
    return float(top.mean())

def _topk_grad(h, k):
    n = len(h)
    c = _corr(h) * (2.0 / n)
    top_idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / k
    dx = 2.0 / n
    g = np.zeros(n)

        sh = (n - 1) - m
        wm = w[m] * dx
        if sh >= 0:
            g[:n-sh] += wm * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wm * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wm * (1.0 - h[:n-sp])
            g[:n-sp] -= wm * h[sp:]
    return g

def _sum_con(h):
    return h.sum() - len(h) * 0.5

def _sum_jac(h):
    return np.ones(len(h))

    bnds = Bounds(np.zeros(n), np.ones(n))

    try:
        res = minimize(_topk_obj, h, jac=lambda x: _topk_grad(x, k),
                       args=(k,), method='L-BFGS-B', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(-(h)) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _topk_obj(h, k):
    n = len(h)
    c = _corr(h) * (2.0 / n)
    top = np.sort(c)[::-1][:k]
    return float(top.mean())

def _topk_grad(h, k):
    n = len(h)
    c = _corr(h) * (2.0 / n)
    top_idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / k
    dx = 2.0 / n
    g = np.zeros(n)
    for m in top_idx:
        sh = (n - 1) - m
        wm = w[m] * dx
        if sh >= 0:
            g[:n-sh] += wm * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wm * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wm * (1.0 - h[:n-sp])
            g[:n-sp] -= wm * h[sp:]
    return g

def _sum_con(h):
    return h.sum() - len(h) * 0.5

def _sum_jac(h):
    return np.ones(len(h))

def _slsqp_topk(h, k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
    try:
        res = minimize(_topk_obj, h, args=(k,), jac=_topk_grad,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _directed_binary(hb, n_iters, rng):
    n = len(hb)
    h = hb.copy()
    c5 = _eval(h)
    for _ in range(n_iters):
        c = _corr(h)
        k_star = int(np.argmax(c))
        s_star = (n - 1) - k_star
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        improved = False
        vi = (zeros + s_star >= 0) & (zeros + s_star < n)
        vj = (ones + s_star >= 0) & (ones + s_star < n)
        gz = zeros[vi]
        gz = gz[h[gz + s_star] == 1] if len(gz) > 0 else np.array([], dtype=int)
        go = ones[vj]
        go = go[h[go + s_star] == 0] if len(go) > 0 else np.array([], dtype=int)
        n_try = min(30, max(3, (len(gz)+1)*(len(go)+1)))
        for _t in range(n_try):
            if len(gz) > 0 and len(go) > 0:
                i = gz[rng.randint(len(gz))]
                j = go[rng.randint(len(go))]
            elif len(zeros) > 0 and len(ones) > 0:
                i = zeros[rng.randint(len(zeros))]
                j = ones[rng.randint(len(ones))]
            else:
                break
            h[i], h[j] = 1.0, 0.0
            nc = _eval(h)
            if nc < c5 - 1e-12:
                c5 = nc; improved = True; break
            h[i], h[j] = 0.0, 1.0
        if not improved and len(zeros) > 0 and len(ones) > 0:
            for _t in range(8):
                i = zeros[rng.randint(len(zeros))]
                j = ones[rng.randint(len(ones))]
                h[i], h[j] = 1.0, 0.0
                nc = _eval(h)
                if nc < c5 - 1e-12:
                    c5 = nc; improved = True; break
                h[i], h[j] = 0.0, 1.0
    return h, c5

def _proj_grad(h, k, n_steps, lr0, rng):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    lr = lr0
    for step in range(n_steps):
        g = _topk_grad(h, k)
        h_new = _repair(h - lr * g, n)
        nc5 = _eval(h_new)
        if nc5 < c5 - 1e-15:
            h = h_new; c5 = nc5
            lr = min(lr * 1.05, 0.04)
        else:
            lr *= 0.6
            if lr < 5e-6:
                lr = lr0 * 0.4
    return h, c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _pb(h):
    n = len(h)
    hb = (h > 0.5).astype(float)
    cnt = int(hb.sum())
    target = n // 2
    if cnt > target:
        ones = np.where(hb == 1)[0]
        order = np.argsort(h[ones])
        hb[ones[order[:cnt - target]]] = 0
    elif cnt < target:
        zeros = np.where(hb == 0)[0]
        order = np.argsort(-h[zeros])
        hb[zeros[order[:target - cnt]]] = 1
    return hb

def _worst_shift_grad(h):
    n = len(h)
    c = _corr(h)
    k_star = int(np.argmax(c))
    s_star = (n - 1) - k_star
    dx = 2.0 / n
    g = np.zeros(n)
    if s_star >= 0:
        g[:n - s_star] += dx * (1.0 - h[s_star:])
        if s_star > 0:
            g[s_star:] -= dx * h[:n - s_star]
    else:
        sp = -s_star
        g[sp:] += dx * (1.0 - h[:n - sp])
        g[:n - sp] -= dx * h[sp:]
    return g

def _worst_shift_descent(h, n_steps, lr):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    for _ in range(n_steps):
        g = _worst_shift_grad(h)
        h_new = _repair(h - lr * g, n)
        nc5 = _eval(h_new)
        if nc5 < c5 - 1e-15:
            h = h_new; c5 = nc5
            lr = min(lr * 1.1, 0.03)
        else:
            lr *= 0.5
            if lr < 1e-7:
                lr = 0.005
    return h, c5
