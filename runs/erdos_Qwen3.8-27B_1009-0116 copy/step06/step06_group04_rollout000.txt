def _softmax_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _softmax_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

        if time.time() > deadline:
            break
        # Adaptive alpha
        frac = (i + 1) / n_iter
        alpha = 100.0 * (alpha_target / 100.0) ** frac
        
        # Nesterov step
        grad_y = _softmax_grad(y, alpha)
        x_new = y - lr * grad_y
        y = x_new + ((np.sqrt(1 + 4*beta*beta) - 1) / 2) * (x_new - x)
        x = x_new
        
        # Project
        x = _repair(x, n)
        y = _repair(y, n)
        beta = (1 + 1/(i+1)) / 2  # standard Nesterov beta

        if c < best_c:

            best_x = x.copy()
            
        # Occasional restart
        if i % 500 == 0 and i > 0:
            x = best_x + 0.01 * rng.randn(n)
            x = _repair(x, n)
            y = x.copy()

    try:
        res = minimize(_softmax_obj, h0, jac=lambda x: _softmax_grad(x, alpha),
                       args=(alpha,), method='L-BFGS-B', 
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)
```

Actually, I realize I can vectorize the pair evaluation. For a given i (being flipped from 1 to 0), the change in correlate is the same for all j flips. So I can:
1. Compute the "effect of flipping h[i] from 1 to 0" on the correlate vector
2. Then for each j (being flipped from 0 to 1), compute the "effect of flipping h[j] from 0 to 1"
3. The total change is the sum of both effects
4. Find the j that minimizes the max of the new correlate

This is O(n) per i instead of O(n²) per i. Total: O(n²) per round.

Let me work out the math for the correlate change:

Let c = np.correlate(h, 1-h, mode='full'). 

If we change h[i] by δ (δ = -1 for 1→0, δ = +1 for 0→1):
The new h' = h + δ*e_i

New correlate: np.correlate(h + δ*e_i, 1 - h - δ*e_i, mode='full')
= np.correlate(h, 1-h) + δ*np.correlate(e_i, 1-h) - δ*np.correlate(h, e_i) - δ²*np.correlate(e_i, e_i)

Wait, correlate is bilinear: correlate(a+b, c+d) = correlate(a,c) + correlate(a,d) + correlate(b,c) + correlate(b,d)

So: correlate(h+δe_i, 1-h-δe_i) 
= correlate(h, 1-h) + correlate(h, -δe_i) + correlate(δe_i, 1-h) + correlate(δe_i, -δe_i)
= c - δ*correlate(h, e_i) + δ*correlate(e_i, 1-h) - δ²*correlate(e_i, e_i)

Now, correlate(e_i, e_j, mode='full') is 1 at position i+j-(n-1) (or something like that) and 0 elsewhere. Actually, for mode='full':
correlate(e_i, e_j)[p] = 1 if p = i + j - (n-1)... 

Hmm, let me think more carefully. np.correlate(a, v, 'full') at position p is:
sum_k a[k] * v[p - k] (with the convention that v[m] = 0 for m outside [0, n-1])

Wait, actually the numpy documentation says:
c_k = sum_n a_{n+k} * conj(v_n)

For mode='full', k ranges from -(n-1) to n-1, and the output has length 2n-1.
The output index for k is k + (n-1).

So correlate(h, 1-h)[p] where p = 0,...,2n-2 corresponds to k = p - (n-1).
c[p] = sum_n h[n + k] * (1-h)[n] = sum_n h[n + p - n + 1] * (1-h)[n]

This is getting confusing. Let me just use the direct approach: for a pair flip (i: 1→0, j: 0→1), I'll compute the new correlate directly. For n=310, it's fast enough.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _softobj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _softgrad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            i1 = np.arange(cnt)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            cnt2 = n - sp
            i2 = np.arange(cnt2)
            g[i2] -= wk * x[i2 + sp]
    return g

def _pen_obj(x, alpha, lam):
    return _softobj(x, alpha) + lam * (x.sum() - len(x) * 0.5) ** 2

def _pen_grad(x, alpha, lam):
    n = len(x)
    g = _softgrad(x, alpha)
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def _lbfgs(h0, alpha, maxiter, deadline):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    lam = 5000.0
    try:
        res = minimize(_pen_obj, h0, jac=_pen_grad, args=(alpha, lam),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 3})
        if time.time() > deadline:
            return _repair(h0, n)
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _adam(h0, a_start, a_end, n_iter, lr_s, lr_e, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n); v = np.zeros(n); t = 0
    b1, b2, eps = 0.75, 0.999, 1e-8
    best_x = x.copy(); best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        alpha = a_start * (a_end / a_start) ** frac
        lr = lr_s * (lr_e / lr_s) ** frac
        if t % 800 == 0:
            x = x + 0.003 * rng.randn(n)
            x = _repair(x, n)
        g = _pen_grad(x, alpha, 5000.0)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c; best_x = x.copy()
    return best_x, best_c

def _binary_flip(h_init, deadline, max_rounds=200):
    n = len(h_init)
    h = (h_init > 0.5).astype(float)
    h = _repair(h, n)
    c_cur = _eval(h)
    h_best = h.copy(); c_best = c_cur
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    rng = np.random.RandomState(12345)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        # Random pairs + some deterministic
        pairs_tried = 0
        max_pairs = 500
        while pairs_tried < max_pairs:
            if time.time() > deadline:
                return h_best, c_best
            pairs_tried += 1
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0; h[j] = 1.0
            c_new = _eval(h)
            if c_new < c_cur - 1e-15:
                c_cur = c_new
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                improved = True
                if c_new < c_best:
                    c_best = c_new; h_best = h.copy()
            else:
                h[i] = 1.0; h[j] = 0.0
        if not improved:
            break
    return h_best, c_best

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)
