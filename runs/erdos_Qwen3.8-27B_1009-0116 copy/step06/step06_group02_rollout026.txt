import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        hc = np.clip(h - mid, 0.0, 1.0)
        if hc.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-11:
                break
    return h

def cvec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5val(h):
    n = len(h)
    return float(cvec(h).max()) * 2.0 / n

def lse_obj(h, alpha):
    c = cvec(h)
    mc = c.max()
    return mc + float(np.log(np.sum(np.exp(alpha * (c - mc)))) / alpha)

def lse_grad(h, alpha):
    n = len(h)
    c = cvec(h)
    mc = c.max()
    w = np.exp(alpha * (c - mc))
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            nr = n - sh
            g[:nr] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:nr]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def nesterov_run(h0, alpha, n_iter, lr_init, t_end):
    n = len(h0)
    h = h0.copy()
    h_prev = h.copy()
    best = h.copy()
    best_val = c5val(h)
    t_mom = 1.0
    lr = lr_init
    for it in range(n_iter):
        if time.time() > t_end:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) * 0.5
        beta = (t_mom - 1.0) / t_new
        h_saved = h.copy()
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        g = lse_grad(y, alpha)
        h_new = repair(y - lr * g, n)
        f_new = lse_obj(h_new, alpha)
        f_y = lse_obj(y, alpha)
        step = lr
        for _ in range(15):
            if f_new <= f_y + 1e-10:
                break
            step *= 0.5
            h_new = repair(y - step * g, n)
            f_new = lse_obj(h_new, alpha)
        h = h_new
        h_prev = h_saved
        t_mom = t_new
        v = c5val(h)
        if v < best_val:
            best_val = v
            best = h.copy()
    return best, best_val

def reweight_descent(h0, n_iter, top_m, t_end):
    n = len(h0)
    h = h0.copy()
    cur = c5val(h)
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = cvec(h)
        idx = np.argpartition(c, -top_m)[-top_m:]
        cmin, cmax = c[idx].min(), c[idx].max()
        w = np.zeros(2 * n - 1)
        if cmax > cmin + 1e-12:
            w[idx] = np.exp(8.0 * (c[idx] - cmin) / (cmax - cmin))
        else:
            w[idx] = 1.0
        w /= w.sum()
        g = np.zeros(n)
        for k in idx:
            wk = w[k] * dx
            sh = (n - 1) - k
            if sh >= 0:
                nr = n - sh
                g[:nr] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:nr]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n - sp])
                g[:n - sp] -= wk * h[sp:]
        moved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]:
            ht = repair(h - step * g, n)
            ct = c5val(ht)
            if ct < cur - 1e-14:
                h = ht; cur = ct; moved = True
                break
        if not moved:
            break
    return h, cur

def pairwise_cd(h0, n, n_iter, scales, t_end, seed=1234):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    cur = c5val(h)
    for scale in scales:
        for _ in range(n_iter):
            if time.time() > t_end:
                return h, cur
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni = h[i] + d; nj = h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i] = ni; h[j] = nj
            cn = c5val(h)
            if cn < cur - 1e-15:
                cur = cn
            else:
                h[i] = h[i] - d; h[j] = h[j] + d
    return h, cur

def resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def slsqp_polish(h0, alpha, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lse_obj, h0, jac=lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        return repair(res.x, n)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)

    best_h = h.copy()
    best_c5 = c5val(best_h)
    best_n = n0

    # Phase 1: Nesterov at original resolution with annealing

    for alpha in alphas1:

            break
        n_it = 4000 if alpha < 1000 else 2500
        lr = 0.05 / np.sqrt(alpha + 1)
        h, v = nesterov_run(best_h, alpha, n_it, lr, t0 + rem() - 15)
        if v < best_c5:
            best_c5 = v; best_h = h.copy(); best_n = len(h)

    # Phase 2: Top-m reweighting
    if rem() > 80:

        if v < best_c5:
            best_c5 = v; best_h = h2.copy(); best_n = len(h2)
    if rem() > 50:
        h2b, v = reweight_descent(best_h.copy(), 3000, 3, t0 + rem() - 10)
        if v < best_c5:
            best_c5 = v; best_h = h2b.copy()

    # Phase 3: Multi-resolution search
    if rem() > 280:
        for n_try in [400, 450, 500, 550, 600]:
            if rem() < 220:

            hr = repair(resample(best_h, n_try), n_try)
            for alpha in [50, 2000, 30000]:
