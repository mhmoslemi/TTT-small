import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha, pen):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    val = m * 2.0 / n + np.sum(np.exp(s - s.max())) / alpha
    val += pen * (h.sum() - 0.5 * n) ** 2
    return val

            g[:n - sp] -= wk * h[sp:]

        return _repair(h, n)

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:

    n = len(h)

        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]

    n = len(h); h = h.copy()

            if cn < best_c:

        else:

    return best, best_c

def _fine_search(h, n_tries, rng):
    n = len(h); h = h.copy()
    c_cur = _eval(h); best = h.copy(); best_c = c_cur
    # Multiple scales of perturbation

    for scale in scales:
        for _ in range(n_tries // 4):
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d

            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best, best_c

def _triple_search(h, n_tries, rng):
    """Try moving mass from one point to two others (or vice versa)"""
    n = len(h); h = h.copy()
    c_cur = _eval(h); best = h.copy(); best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n); j = rng.randint(n); k = rng.randint(n)
        if len(set([i, j, k])) < 3:
            continue
        d = 1e-5 * rng.randn(2)
        ni = h[i] - d[0] - d[1]
        nj = h[j] + d[0]
        nk = h[k] + d[1]
        if min(ni, nj, nk) < 0 or max(ni, nj, nk) > 1:

        cn = _eval(h)

            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j], h[k] = ni + d[0] + d[1], nj - d[0], nk - d[1]
    return best, best_c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
