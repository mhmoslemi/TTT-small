def _lsme(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _lsme_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _se(x):
    return x.sum() - len(x) * 0.5

def _sj(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_lsme, h, jac=_lsme_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lfgsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_lsme, h, jac=_lsme_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _worst_shift_grad_step(h, n_steps=50):
    nc = len(h)
    c5c = _eval(h)
    dx = 2.0 / nc
    for _ in range(n_steps):
        c = np.correlate(h, 1.0 - h, mode='full')
        ki = int(np.argmax(c))
        sh = (nc - 1) - ki
        g = np.zeros(nc)
        if sh >= 0:
            g[:nc-sh] += dx * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= dx * h[:nc-sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:nc-sp])
            g[:nc-sp] -= dx * h[sp:]
        moved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _repair(h - step * g, nc)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h, c5c = h_new, c5n
                moved = True
                break
        if not moved: break
    return h, c5c

def _perturb(h, rng, scale=0.05):
    nc = len(h)
    h_p = h + scale * rng.randn(nc)
    return _repair(h_p, nc)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _fourier_polish(h, n_harm, maxiter, alpha):
    """Optimize a few Fourier coefficients of the deviation from the current h."""
    n = len(h)
    idx = np.arange(n)
    freqs = np.arange(1, n_harm + 1)
    cos_mat = np.column_stack([np.cos(2*np.pi*f*idx/n) for f in freqs])
    sin_mat = np.column_stack([np.sin(2*np.pi*f*idx/n) for f in freqs])
    
    def make_obj_and_grad(alpha):
        def obj(params):
            h_out = h + cos_mat @ params[:n_harm] + sin_mat @ params[n_harm:]
            h_out = np.clip(h_out, 0, 1)
            s = h_out.sum()
            if s > 1e-9:
                h_out = h_out * (n/2.0)/s
                h_out = np.clip(h_out, 0, 1)
            return _lsme(h_out, alpha)
        def grad(params):
            h_out = h + cos_mat @ params[:n_harm] + sin_mat @ params[n_harm:]
            h_out = np.clip(h_out, 0, 1)
            s = h_out.sum()
            if s > 1e-9:
                h_out = h_out * (n/2.0)/s
                h_out = np.clip(h_out, 0, 1)
            gh = _lsme_grad(h_out, alpha)
            g = np.zeros(2*n_harm)
            g[:n_harm] = cos_mat.T @ gh
            g[n_harm:] = sin_mat.T @ gh
            return g
        return obj, grad

    obj, grad = make_obj_and_grad(alpha)
    x0 = np.zeros(2 * n_harm)
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = h + cos_mat @ res.x[:n_harm] + sin_mat @ res.x[n_harm:]
        return _repair(h_out, n)
    except Exception:
        return _repair(h, n)
