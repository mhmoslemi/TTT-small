import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax_f(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad_f(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _c(x): return [x.sum() - 0.5 * n]
    def _cj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c, 'jac': _cj}]
    try:
        res = minimize(_smax_f, h, jac=_sgrad_f, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_top(h, m):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-m:]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * dx

def _descend(h, n_iter, step0, top_m, rng):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    v = np.zeros(n); mu = 0.9; step = step0; stuck = 0
    for _ in range(n_iter):
        g = _grad_top(best, top_m)
        v = mu * v + g
        vn = np.linalg.norm(v)
        if vn < 1e-14:
            break
        v /= vn
        ok = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-16:
                best = h_new; best_c = c_new
                step = min(step * 1.25, 0.5)
                stuck = 0; ok = True
                break
        if not ok:
            step *= 0.6; stuck += 1
            if stuck > 40:
                best = _repair(best + 0.003 * rng.randn(n), n)
                best_c = _eval(best); step = step0; stuck = 0
                v[:] = 0
    return best, best_c

def _fine(h, n_tries, rng):
    n = len(h); h = h.copy()
    c_cur = _eval(h); best = h.copy(); best_c = c_cur
    g = _grad_top(h, 3)
    for t in range(n_tries):
        if t % 400 == 0:
            g = _grad_top(h, 3)
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        if g[i] >= g[j]:
            d = -rng.uniform(5e-6, 1.5e-4)
        else:
            d = rng.uniform(5e-6, 1.5e-4)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur - 1e-16:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c

            if cn < best_c:

        else:

    return best, best_c
