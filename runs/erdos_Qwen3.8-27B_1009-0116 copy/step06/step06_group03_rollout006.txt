def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_k(h, n, k):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _subgrad(h, n_iter, step0, top_m):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    v = np.zeros(n); mu = 0.82; step = step0
    stuck = 0
    for it in range(n_iter):
        c = np.correlate(best, 1.0 - best, mode='full')
        idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_k(best, n, k)
        g *= 2.0 / (n * len(idx))
        v = mu * v + g
        improved = False
        for f in [1.0, 0.6, 0.35, 0.15, 0.06]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best = h_new; best_c = c_new
                step = min(step * 1.12, 0.25)
                improved = True; stuck = 0
                break
        if not improved:
            step *= 0.72; stuck += 1
            if stuck > 40:
                best = _repair(best + 0.015 * np.random.randn(n), n)
                best_c = _eval(best); step = step0; stuck = 0
                v = np.zeros(n)
    return best, best_c

def _anneal(h, n_iter, t0, t1, rng):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    cur = h.copy(); cur_c = best_c
    for it in range(n_iter):
        T = t0 * (t1 / t0) ** (it / max(n_iter, 1))
        blen = rng.randint(2, max(3, n // 25))
        i = rng.randint(0, n - blen)
        old = cur[i:i + blen].copy()
        cur[i:i + blen] += rng.randn(blen) * 0.04
        cur = _repair(cur, n)
        nc = _eval(cur)
        if nc < cur_c or rng.rand() < np.exp(-(nc - cur_c) / max(T, 1e-10)):
            cur_c = nc
            if nc < best_c:
                best_c = nc; best = cur.copy()
        else:
            cur[i:i + blen] = old
            cur = _repair(cur, n)
    return best, best_c

            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c_cur:

                h[i:i+bl] = o1; h[j:j+bl] = o2; continue

            ni, nj = h[i] - d, h[j] + d
