def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                frac = np.argsort(1.0 - h)[::-1]  # most room first
            else:
                frac = np.argsort(h)[::-1]  # most value first
            for idx in frac:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * 2.0 / n

def _eval_all(h):

    c = np.correlate(h, 1.0 - h, mode='full')
    return c * 2.0 / n

def _softmax_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    cd = c * dx
    m = cd.max()
    s = np.clip(alpha * (cd - m), -500, 500)
    s -= s.max()
    return (m + np.log(np.sum(np.exp(s))) / alpha)

def _softmax_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    cd = c * dx
    m = cd.max()
    s = np.clip(alpha * (cd - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        sh = (n-1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]

import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for i in idx:
                if diff > 0:
                    a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i]); h[i] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _sobj(h, alpha):
    cv = _corr(h)
    m = cv.max()
    s = np.clip(alpha * (cv - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    cv = _corr(h)
    m = cv.max()
    s = np.clip(alpha * (cv - m), -500, 500)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

    hc = h.copy(); hp = h.copy()

        t = i / max(n_iter - 1, 1)

        y = np.clip(y, 0, 1)
        g = _sgrad(y, alpha)
        hnext = _repair(y - lr * g, n)
        hp = hc
        hc = hnext
        c = _eval(hc)
        if c < best:
            best = c; bh = hc.copy()
    return bh, best

def _slsqp(h, alpha, maxiter, ftol, deadline=None):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _f(x):
        return _sobj(x, alpha)
    def _g(x):
        return _sgrad(x, alpha)
    def _ceq(x):
        return x.sum() - n * 0.5
    def _ceq_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceq_j}]
    if deadline is not None and time.time() > deadline:
        return h
    try:
        res = minimize(_f, h, jac=_g, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if deadline is not None and time.time() > deadline:
            return h
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
