import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * 2.0 / n)

def _smooth_obj_grad(h, alpha, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    scaled = np.clip(alpha * (corr - cmax), -700, 700)
    exp_s = np.exp(scaled)
    exp_sum = exp_s.sum()
    p = exp_s / exp_sum
    logsum = cmax + np.log(exp_sum) / alpha
    sv = h.sum() - n / 2.0
    f = (2.0 / n) * logsum + beta * sv * sv
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        shift = k - (n - 1)
        a_k = max(0, k - n + 1)
        b_k = min(k, n - 1)
        # First: i in [a_k, b_k], g[i] += pk*(1-h[i-shift])
        i1 = np.arange(a_k, b_k + 1)
        g[i1] += pk * (1.0 - h[i1 - shift])
        # Second: i in [a_k-shift, b_k-shift], g[i] -= pk*h[i+shift]
        a2 = max(0, a_k - shift)
        b2 = min(n - 1, b_k - shift)
        if a2 <= b2:
            i2 = np.arange(a2, b2 + 1)
            g[i2] -= pk * h[i2 + shift]
    g *= (2.0 / n)
    g += 2.0 * beta * sv
    return f, g

def _lbfgsb(h, alpha, beta, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    res = minimize(_smooth_obj_grad, h, jac=True, args=(alpha, beta),
                   method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-12,
                            'iprint': 0, 'maxls': 50})
    if time.time() > t_end:  # won't actually check mid-iteration
        return h
    return _repair(res.x, n)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        s = np.clip(alpha * (corr - m), -500, 500)
        e = np.exp(s - s.max())
        p = e / e.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            shift = k - (n - 1)
            a_k = max(0, k - n + 1)
            b_k = min(k, n - 1)
            i1 = np.arange(a_k, b_k + 1)
            g[i1] += pk * (1.0 - x[i1 - shift])
            a2 = max(0, a_k - shift)
            b2 = min(n - 1, b_k - shift)
            if a2 <= b2:
                i2 = np.arange(a2, b2 + 1)
                g[i2] -= pk * x[i2 + shift]
        g *= 2.0 / n
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter,
                       'ftol': ftol, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _resample(h_old, n_new):

    return np.interp(xn, xo, h_old)
