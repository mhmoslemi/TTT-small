import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _grad_one(h, k, n):
    """Gradient of c_k (unnormalized correlate) w.r.t. h."""
    g = np.zeros(n)
    shift = k - (n - 1)
    if shift >= 0:
        i = np.arange(n - shift)
        g[i] += (1.0 - h[i + shift])
        if shift > 0:
            g[i + shift] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        g[i] += (1.0 - h[i - s])
        g[i - s] -= h[i]
    return g

def _project_box_hyperplane(h, n):
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    p = np.clip(h - lam, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(-p, kind='stable')
        else:
            order = np.argsort(p, kind='stable')
        for idx in order:
            a = min(abs(diff), 1.0 - p[idx]) if diff > 0 else min(abs(diff), p[idx])
            p[idx] += a if diff > 0 else -a
            diff += a if diff > 0 else -a
            if abs(diff) < 1e-12:
                break
    return p

def _smoothed_grad(h, n, alpha, top_m=10):
    c = _corr(h)
    dx = 2.0 / n
    d = c * dx
    m = d.max()
    s = np.clip(alpha * (d - m), -500, 500)
    e = np.exp(s)
    p = e / e.sum()
    obj = m + np.log(e.sum()) / alpha
    # Only use significant shifts
    thresh = 1e-10
    active = p > thresh
    g = np.zeros(n)
    for k in np.where(active)[0]:
        if p[k] < 1e-12:
            continue
        g += p[k] * dx * _grad_one(h, k, n)
    return obj, g

def _nesterov_subgrad(h, n, alphas, iters_per_alpha, t_end):
    best_h = h.copy()
    best_c = _eval(h)
    for alpha in alphas:
        y = h.copy()
        v = np.zeros(n)
        for it in range(iters_per_alpha):
            if time.time() > t_end:
                break
            obj, g = _smoothed_grad(y, n, alpha)
            c5_now = _eval(y)
            if c5_now < best_c:
                best_c = c5_now
                best_h = y.copy()
            gn = np.linalg.norm(g)
            if gn < 1e-14:

            lr = 1.0 / (alpha * gn * 0.5 + gn)
            # Backtracking
            for _bt in range(10):

                break
            v = 0.85 * v + g
            y_next = _project_box_hyperplane(y + lr * v, n)
            # Check for improvement

            else:

                if lr < 1e-10:
                    y = y_next
                    break
                y = _project_box_hyperplane(y + lr * v, n)
        h = best_h.copy()
    return best_h, best_c

def _pair_cd(h, n, n_iter, t_end):
    rng = np.random.RandomState(42)
    cur = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try small perturbation
        d = 0.01 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < cur - 1e-14:
            cur = cn
        else:
            h[i], h[j] = ni - d, nj + d

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * 2.0 / n)

def _grad_one(h, k, n):
    g = np.zeros(n)
    shift = k - (n - 1)
    if shift >= 0:
        i = np.arange(n - shift)
        g[i] += (1.0 - h[i + shift])
        if shift > 0:
            g[i + shift] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        g[i] += (1.0 - h[i - s])
        g[i - s] -= h[i]
    return g

def _project(h, n):

    for _ in range(80):

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

        else:

            p[idx] += a if diff > 0 else -a
