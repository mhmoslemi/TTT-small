            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _penalty_obj(x, alpha, pen):
    n = len(x)
    return _sm_obj(x, alpha) + pen * (x.sum() - n * 0.5) ** 2

def _penalty_grad(x, alpha, pen):
    n = len(x)
    return _sm_grad(x, alpha) + 2.0 * pen * (x.sum() - n * 0.5) * np.ones(n)

def _lbfgs(h0, alpha, pen, maxiter, deadline):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_penalty_obj, h0, jac=_penalty_grad,
                       args=(alpha, pen), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-25})
        return _repair(res.x, n)
    except Exception:
        return h0.copy()

def _adam(h0, a0, a1, ni, lr0, lr1, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    bx, bc = x.copy(), _eval(x)
    for i in range(ni):
        if time.time() > deadline:
            break
        t += 1
        fr = i / max(ni - 1, 1)
        a = a0 * (a1 / a0) ** fr
        lr = lr0 * (lr1 / lr0) ** fr
        if t % 1000 == 0:
            x = _repair(x + 0.005 * rng.randn(n), n)
        g = _sm_grad(x, a)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        x = _repair(x - lr * (m / (1 - b1**t)) / (np.sqrt(v / (1 - b2**t)) + eps), n)
        c = _eval(x)
        if c < bc:
            bc, bx = c, x.copy()
    return bx, bc

def _bin_ls(h0, n, rng, nit, deadline):
    x = (h0 > 0.5).astype(float)
    x = _repair(x, n)
    cc = _eval(x)
    bx, bc = x.copy(), cc
    for _ in range(nit):

        j = zeros[rng.randint(len(zeros))]
        x[i], x[j] = 0.0, 1.0
        cn = _eval(x)
        if cn < cc:
            cc = cn
            if cn < bc:
                bc, bx = cn, x.copy()
        else:
            x[i], x[j] = 1.0, 0.0
    return bx, bc

def _finetune(h, n, rng, deadline, sc=0.003):

    cc = _eval(x)

        if cn < cc - 1e-15:
            cc = cn
            if cn < bc:
                bc, bx = cn, x.copy()
            sc *= 0.9998
        else:
            x = bx.copy()
            cc = bc
            sc *= 0.9
    return bx, bc

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _sm_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _sm_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    ss = s - s.max()
    w = np.exp(ss)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
