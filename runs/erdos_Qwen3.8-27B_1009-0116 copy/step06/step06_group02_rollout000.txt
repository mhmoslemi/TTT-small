            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_shift(h, sh):
    n = len(h)
    g = np.zeros(n)
    if sh >= 0:
        if sh == 0:
            g[:n] += (1.0 - h)
        else:
            g[:n - sh] += (1.0 - h[sh:n])
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        if sp == 0:
            g[:] += (1.0 - h)
        else:
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g

def _focal_grad(h, top_idx, weights):
    n = len(h)
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        if w < 1e-14:
            continue
        sh = (n - 1) - k
        g += w * _grad_shift(h, sh)
    return g

def _focal_descent(h, t_end, n_iter, top_T, lr, momentum, gamma):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = _eval(h)
    v = np.zeros(n)
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[-top_T:]
        c_top = c[top_idx]
        crng = c_top.max() - c_top.min()
        if crng > 1e-12:
            w = np.exp(gamma * (c_top - c_top.min()) / crng)
        else:
            w = np.ones(top_T) / top_T
        w /= w.sum()
        g = _focal_grad(h, top_idx, w) * dx
        alpha = lr / (1.0 + 0.015 * it)
        v = momentum * v - alpha * g
        best_sh = h.copy()
        best_sc = _eval(h)
        for sm in [1.0, 0.6, 0.35, 0.18, 0.09]:
            hn = _repair(h + sm * v, n)
            cn = _eval(hn)
            if cn < best_sc:
                best_sc = cn
                best_sh = hn
        if best_sc < best_c:
            best_c = best_sc
            best_h = best_sh.copy()
        h = best_sh
    return best_h, best_c

def _fista(h, t_end, n_iter, top_T, lr):
    n = len(h)
    h = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c = _eval(h)
    t_mom = 1.0
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr(y)
        top_idx = np.argsort(c)[-top_T:]
        c_top = c[top_idx]
        crng = c_top.max() - c_top.min()
        if crng > 1e-12:
            w = np.exp(5.0 * (c_top - c_top.min()) / crng)
        else:
            w = np.ones(top_T) / top_T
        w /= w.sum()
        g = _focal_grad(y, top_idx, w) * dx
        alpha = lr / (1.0 + 0.01 * it)
        h_new = _repair(y - alpha * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = h_new + ((t_mom - 1.0) / t_new) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_mom = t_new
        cn = _eval(h)
        if cn < best_c:
            best_c = cn
            best_h = h.copy()
    return best_h, best_c

def _perturb_resample(h, n_new, seed):
    n_old = len(h)
    rng = np.random.RandomState(seed)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    h_r = np.interp(xn, xo, h)
    noise = rng.randn(n_new) * 0.05
    h_p = h_r + noise
    return _repair(h_p, n_new)

def _obj_lse(x, alpha, top_T, n):
    c = _corr(x)
    if top_T < 2 * n - 1:
        idx = np.argsort(c)[-top_T:]
        c_work = c.copy()
        c_work[np.setdiff1d(np.arange(len(c)), idx)] = c.min()
    else:
        c_work = c
    m = c_work.max()
    s = np.clip(alpha * (c_work - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad_lse(x, alpha, top_T, n):
    c = _corr(x)
    if top_T < 2 * n - 1:
        idx = np.argsort(c)[-top_T:]
        c_work = c.copy()
        c_work[np.setdiff1d(np.arange(len(c)), idx)] = c.min()
    else:
        c_work = c
    m = c_work.max()
    s = np.clip(alpha * (c_work - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(c_work)):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        g += w[k] * _grad_shift(x, sh)
    return g * dx

def _slsqp_obj(x, alpha, top_T, n):
    return _obj_lse(x, alpha, top_T, n)

def _slsqp_grad(x, alpha, top_T, n):
    return _grad_lse(x, alpha, top_T, n)

def _slsqp_eq(x, n):
    return x.sum() - n * 0.5

def _slsqp_eq_jac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, maxiter, top_T, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, top_T, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def _upd(hn, cn, nn):
        nonlocal best_h, best_c5, best_n
        if cn < best_c5:
            best_c5 = cn; best_h = hn.copy(); best_n = nn

    # Phase 1: Focal descent with momentum (main workhorse)
    t_end1 = t0 + budget_s * 0.25
    h1 = best_h.copy()
    h1, c1 = _focal_descent(h1, t_end1, 8000, top_T=6, lr=0.04, momentum=0.85, gamma=4.0)
    _upd(h1, c1, len(h1))
    h1b, c1b = _focal_descent(best_h.copy(), t_end1, 4000, top_T=3, lr=0.02, momentum=0.9, gamma=6.0)
    _upd(h1b, c1b, len(h1b))

    # Phase 2: FISTA-style acceleration
    t_end2 = t0 + budget_s * 0.40
    h2 = best_h.copy()
    h2, c2 = _fista(h2, t_end2, 6000, top_T=4, lr=0.03)
    _upd(h2, c2, len(h2))
    h2b, c2b = _fista(best_h.copy(), t_end2, 3000, top_T=2, lr=0.015)
    _upd(h2b, c2b, len(h2b))

    # Phase 3: Focus descent with very few top shifts (fine-tune)
    t_end3 = t0 + budget_s * 0.52
    h3 = best_h.copy()
    h3, c3 = _focal_descent(h3, t_end3, 10000, top_T=2, lr=0.01, momentum=0.92, gamma=8.0)
    _upd(h3, c3, len(h3))

    # Phase 4: Multi-resolution search
    t_end4 = t0 + budget_s * 0.70
    for n_try in [350, 420, 500, 560, 640]:
        if time.time() > t_end4:
            break
        hr = _perturb_resample(best_h, n_try, seed + n_try)
        hr, cr = _focal_descent(hr, min(time.time() + 40, t_end4), 3000, top_T=4, lr=0.03, momentum=0.85, gamma=4.0)
        _upd(hr, cr, n_try)

    # Phase 5: SLSQP polish on best (short, focused)
    t_end5 = t0 + budget_s * 0.82
    if time.time() < t_end5:
        for alp in [20000, 100000, 400000]:
            if time.time() > t_end5:
                break
            hs = _slsqp(best_h, alp, 600, top_T=4, t_end=t_end5)
            cs = _eval(hs)
            _upd(hs, cs, len(hs))

    # Phase 6: Final fine-grained focal descent
    t_end6 = t0 + budget_s * 0.92
    h6 = best_h.copy()
    h6, c6 = _focal_descent(h6, t_end6, 8000, top_T=2, lr=0.008, momentum=0.93, gamma=10.0)
    _upd(h6, c6, len(h6))

    # Phase 7: Final SLSQP with high alpha, conservative
    if time.time() < t0 + budget_s * 0.97:
        hf = _slsqp(best_h, 800000, 400, top_T=2, t_end=t0 + budget_s * 0.97)
        cf = _eval(hf)
        _upd(hf, cf, len(hf))

    # Final repair and evaluation
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
