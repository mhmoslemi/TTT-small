```python
def _project_to_box_sum(v, n, target):
    """Project v onto {0 <= x <= 1, sum(x) = target}"""
    # Binary search on offset t such that sum(clip(v+t, 0, 1)) = target
    lo, hi = -1.0 - v.max(), 1.0 - v.min()
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + 0.5*(lo+hi), 0.0, 1.0)
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(v, n, target=None):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    if target is None:
        target = n / 2.0
    lo, hi = -v.max() - 0.1, 1.0 - v.min() + 0.1

        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)

def _grad_lse(h, alpha, top_k=None):
    """Compute gradient of log-sum-exp smoothed objective."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -200, 200)
    w = np.exp(s - s.max())
    w /= w.sum()
    if top_k is not None and top_k < len(w):
        # Zero out small weights
        thresh = np.sort(w)[-top_k]
        w[w < thresh] = 0
        w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):

            g[:n - shift] += wk * (1.0 - h[shift:])
            if shift > 0:

        else:
            s2 = -shift
            g[s2:] += wk * (1.0 - h[:n - s2])
            g[:n - s2] -= wk * h[s2:]
    return g

def _adam_optimize(h, n, alpha, lr_init, n_iter, top_k=None):
    """Adam optimization of the LSE-smoothed objective."""
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _eval(h)
    for t in range(1, n_iter + 1):
        g = _grad_lse(h, alpha, top_k)
        lr = lr_init / np.sqrt(t)  # or lr_init
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = _project(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def _slsqp_optimize(h, alpha, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    
    def grad(x):
        return _grad_lse(x, alpha)

        return x.sum() - n * 0.5

        return np.ones(n)

                       options={'maxiter': maxiter, 'ftol': ftol})

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _focused_descent(h, n, max_rounds, top_n=5):
    """Descend on the worst lags."""
    c5c = _eval(h)
    dx = 2.0 / n
    for r in range(max_rounds):
        ov = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(ov)[-top_n:]
        g = np.zeros(n)
        for kk in top_idx:
            shift = kk - (n - 1)
            if shift >= 0:
                g[:n - shift] += dx * (1.0 - h[shift:])
                if shift > 0:
                    g[shift:] -= dx * h[:n - shift]
            else:
                s2 = -shift
                g[s2:] += dx * (1.0 - h[:n - s2])
                g[:n - s2] -= dx * h[s2:]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        # Backtracking line search
        step = 0.1

            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                found = True
                break
            step *= 0.5
        if not found:
            break
    return h, c5c
```

```python
def _grad_lse(h, alpha, top_k=None):

    c = np.correlate(h, 1.0 - h, mode='full')
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(v, n):
    target = n / 2.0
    lo, hi = -v.max() - 0.1, 1.0 - v.min() + 0.1
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)

def _grad_lse(h, alpha, top_k=None):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -200, 200)
    w = np.exp(s - s.max())
    w /= w.sum()
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(w)[-top_k:]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        w = w2 / w2.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        tau = k - (n - 1)
        wk_dx = w[k] * dx
        if tau >= 0:
            if tau < n:
                g[tau:n] += wk_dx * (1.0 - h[0:n - tau])
            if tau > 0:
                g[0:n - tau] -= wk_dx * h[tau:n]
        else:
            s2 = -tau
            if s2 < n:
                g[0:n - s2] += wk_dx * (1.0 - h[s2:n])
                g[s2:n] -= wk_dx * h[0:n - s2]
    return g

def _obj_lse(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _adam_opt(h, n, alpha, lr, n_iter, top_k=None):
    b1, b2, eps = 0.8, 0.999, 1e-8
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    best_h = h.copy()
    best_c5 = _eval(h)
    for t in range(1, n_iter + 1):
        g = _grad_lse(h, alpha, top_k)
        m_arr = b1 * m_arr + (1 - b1) * g
        v_arr = b2 * v_arr + (1 - b2) * g * g
        mh = m_arr / (1 - b1 ** t)
        vh = v_arr / (1 - b2 ** t)
        h = _project(h - lr * mh / (np.sqrt(vh) + eps), n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    def cf(x):
        return x.sum() - n * 0.5
    def cj(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(_obj_lse, h, jac=lambda x: _grad_lse(x, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _focused_descent(h, n, max_rounds, top_n=5):
    c5c = _eval(h)
    dx = 2.0 / n
    for _r in range(max_rounds):
        ov = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(ov)[-top_n:]
        g = np.zeros(n)
        for kk in top_idx:
            tau = kk - (n - 1)
            if tau >= 0:
                if tau < n:
                    g[tau:n] += dx * (1.0 - h[0:n - tau])
                if tau > 0:
                    g[0:n - tau] -= dx * h[tau:n]
            else:
                s2 = -tau
                if s2 < n:
                    g[0:n - s2] += dx * (1.0 - h[s2:n])
                    g[s2:n] -= dx * h[0:n - s2]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        step = 0.1
        found = False
        for _ in range(30):
            h_new = _project(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h, c5c = h_new, c5n
                found = True
                break
            step *= 0.5
        if not found:
            break
    return h, c5c

def _fourier_init(K, n, rng, amp=0.35):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        a = rng.uniform(-amp, amp)
        b = rng.uniform(-amp, amp)
        h += a * np.cos(np.pi * j * x) + b * np.sin(np.pi * j * x)
    return _repair(h, n)
```
