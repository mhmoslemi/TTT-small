def _lssm_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g


def _se(x):
    return x.sum() - len(x) * 0.5


def _sj(x):
    return np.ones(len(x))


def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_lssm, h, jac=_lssm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)


def _coord_descent(h, n_iter, rng):
    n = len(h)
    c_best = _eval(h)
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    while n_iter > 0:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            n_iter -= 1
            continue
        for scale in scales:
            d = scale * rng.randn()
            hi = h[i] + d
            hj = h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                break
            h[i], h[j] = hi, hj
            c_new = _eval(h)
            if c_new < c_best - 1e-16:
                c_best = c_new
                break
            h[i], h[j] = hi - d, hj + d
        n_iter -= 1
    return h, c_best


def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)


def _block_opt(h, n_top, alpha):
    n = len(h)
    c = _corr(h)
    top_shifts = np.argsort(c)[-n_top:]
    active = set()
    for k in top_shifts:
        sh = (n - 1) - k
        if sh >= 0:
            active.update(range(0, n - sh))
            if sh > 0:
                active.update(range(sh, n))
        else:
            sp = -sh
            active.update(range(sp, n))
            active.update(range(0, n - sp))
    active = sorted(active)
    if len(active) < 4:
        return h
    idx = np.array(active)
    h_sub = h[idx].copy()
    dx = 2.0 / n

    def _obj_sub(x):
        h_tmp = h.copy()
        h_tmp[idx] = x
        return _eval(h_tmp)

    bnds = [(0.0, 1.0)] * len(active)
    res = minimize(_obj_sub, h_sub, method='Nelder-Mead',
                   bounds=bnds, options={'maxiter': 300, 'xatol': 1e-10, 'fatol': 1e-14})
    h_new = h.copy()
    h_new[idx] = np.clip(res.x, 0.0, 1.0)
    return _repair(h_new, n)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def el():
        return time.time() - t_start

    # Phase 1: Initial SLSQP polish (45s)
    for alpha in [5000, 30000, 150000]:
        if el() > 45:
            break
        h_new = _slsqp(best_h, alpha, 2000, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Subgradient descent with top-m (90s)
    if el() < budget_s - 90:
        h_sg = best_h.copy()
        for m in [3, 5, 8, 12, 20]:
            for _it in range(300):
                if el() > budget_s - 90:
                    break
                g = _subgrad_top_m(h_sg, m)
                h_sg, c_sg = _proj_line_search(h_sg, g, max_iters=20)
                if c_sg < best_c5:
                    best_c5, best_h = c_sg, h_sg.copy()

    # Phase 3: Aggressive SLSQP (120s)
    if el() < budget_s - 120:
        for alpha in [500000, 2000000, 10000000, 50000000]:
            if el() > budget_s - 120:
                break
            h_new = _slsqp(best_h, alpha, 4000, 1e-20)
            c5 = _eval(h_new)
            if c5 < best_c5 - 1e-15:
                best_c5, best_h = c5, h_new.copy()

    # Phase 4: Resolution exploration (180s)
    for n_try in [250, 300, 350, 400]:
        if el() > budget_s - 120:
            break
        h_r = _repair(_resample(best_h, n_try), n_try)
        h_r = _slsqp(h_r, 30000, 1500, 1e-18)
        h_r = _slsqp(h_r, 200000, 2000, 1e-19)
        c5 = _eval(h_r)
        for _it in range(150):
            if el() > budget_s - 120:
                break
            g = _subgrad_top_m(h_r, 5)
            h_r, c5 = _proj_line_search(h_r, g, max_iters=15)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 5: Block optimization on worst shifts (120s)
    if el() < budget_s - 100:
        h_bl = best_h.copy()
        c_bl = best_c5
        for n_top in [5, 10, 15]:
            for _it in range(20):
                if el() > budget_s - 100:
                    break
                h_bl = _block_opt(h_bl, n_top, 100)
                c_bl = _eval(h_bl)
                if c_bl < best_c5:
                    best_c5, best_h = c_bl, h_bl.copy()

    # Phase 6: Another subgradient + SLSQP cycle (120s)
    if el() < budget_s - 80:
        h_new = _slsqp(best_h, 20000000, 3000, 1e-21)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
        for m in [5, 10]:
            for _it in range(200):
                if el() > budget_s - 80:
                    break
                g = _subgrad_top_m(best_h, m)
                best_h, best_c5 = _proj_line_search(best_h, g, max_iters=20)

    # Phase 7: Coordinate descent micro-tuning (80s)
    if el() < budget_s - 30:
        h_cd, c_cd = _coord_descent(best_h.copy(), 80000, rng)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Phase 8: Final SLSQP polish (25s)
    if el() < budget_s - 5:
        h_new = _slsqp(best_h, 100000000, 2000, 1e-22)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 9: Final coordinate descent (5s)
    if el() < budget_s - 1:
        h_cd, c_cd = _coord_descent(best_h.copy(), 20000, rng)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
