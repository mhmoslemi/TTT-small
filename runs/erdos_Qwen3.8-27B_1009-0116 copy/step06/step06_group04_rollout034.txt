    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cor(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_sm(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cor(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _hfc(x, args):
    n, lam = args
    h, t = x[:n], x[n]
    dx = 2.0 / n
    c = _cor(h) * dx
    v = np.maximum(0.0, c - t)
    return t + 0.5 * float(np.dot(v, v)) + lam * (h.sum() - n * 0.5) ** 2

def _hfg(x, args):
    n, lam = args
    h, t = x[:n], x[n]
    dx = 2.0 / n
    c = _cor(h) * dx
    v = np.maximum(0.0, c - t)
    gt = 1.0 - float(np.sum(v))
    sv = 2.0 * lam * (h.sum() - n * 0.5)
    gh = np.full(n, sv)
    for k in range(2 * n - 1):
        if v[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = v[k] * dx
        if sh >= 0:
            gh[:n-sh] += wk * (1.0 - h[sh:])
            gh[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            gh[sp:] += wk * (1.0 - h[:n-sp])
            gh[:n-sp] -= wk * h[sp:]
    g = np.zeros(n + 1)
    g[:n] = gh
    g[n] = gt
    return g

def _lbfgs(h0, t0, n, lam, deadline, maxiter=4000):
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    try:
        res = minimize(_hfc, x0, args=(n, lam), jac=_hfg,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 50})
        if time.time() > deadline:
            return _repair(h0.copy(), n), _eval(h0)
        hn = _repair(np.clip(res.x[:n], 0, 1).copy(), n)
        return hn, _eval(hn)
    except Exception:
        return _repair(h0.copy(), n), _eval(h0)

def _adam(h0, a0, a1, ni, lr0, lr1, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy()
    m = np.zeros(n); v = np.zeros(n); t = 0
    bx, bc = x.copy(), _eval(x)
    for i in range(ni):
        if time.time() > deadline:
            break
        t += 1
        fr = i / max(ni - 1, 1)
        alpha = a0 * (a1 / a0) ** fr
        lr = lr0 * (lr1 / lr0) ** fr
        if t % 400 == 0:
            x = x + 0.003 * rng.randn(n)
        g = _grad_sm(x, alpha)
        m = 0.85 * m + 0.15 * g
        v = 0.999 * v + 0.001 * g * g
        x = x - lr * (m / (1 - 0.85 ** t)) / (np.sqrt(v / (1 - 0.999 ** t)) + 1e-8)
        x = _repair(x, n)
        c = _eval(x)
        if c < bc:
            bc, bx = c, x.copy()
    return bx, bc

def _resample(h, nn):
    xo = np.linspace(0.0, 2.0, len(h), endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    return np.interp(xn, xo, h)

def _bin_sa(h0, n, rng, nits, deadline):
    x = (h0 > 0.5).astype(float)
    x = _repair(x, n)
    cc = _eval(x)
    bx, bc = x.copy(), cc
    T = 0.004
    for _ in range(nits):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        x[i] = 1.0 - x[i]
        x = _repair(x, n)
        cn = _eval(x)
        if cn <= cc or rng.random() < np.exp(-(cn - cc) / max(T, 1e-7)):
            cc = cn
            if cn < bc:
                bc, bx = cn, x.copy()
        else:
            x[i] = 1.0 - x[i]
            x = _repair(x, n)
        T *= 0.99997
    return bx, bc

def _sa_restart(h0, n, rng, deadline, n_restart, iters_per):
    best_x, best_c = h0.copy(), _eval(h0)
    cur = (h0 > 0.5).astype(float)
    cur = _repair(cur, n)
    for r in range(n_restart):
        if time.time() > deadline:
            break
        x, c = _bin_sa(cur, n, rng, iters_per, deadline)
        if c < best_c:
            best_c, best_x = c, x.copy()
        cur = x.copy()
        # Perturb for next restart
        for _ in range(n // 10):
            cur[rng.randint(n)] = 1.0 - cur[rng.randint(n)]
        cur = _repair(cur, n)
    return best_x, best_c

def _pcd(h, n, rng, deadline, s0):
    x = h.copy()
    cc = _eval(x)
    bx, bc = x.copy(), cc
    sc = s0
    stall = 0
    for _ in range(1000000):
        if time.time() > deadline:
            break
        if sc < 1e-10:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = sc * rng.randn()
        x[i] += d
        x[j] -= d
        x = _repair(x, n)
        cn = _eval(x)
        if cn < cc - 1e-15:
            cc = cn; stall = 0
            if cn < bc:
                bc, bx = cn, x.copy()
            sc = max(sc * 0.9997, 1e-11)
        else:
            x[i] -= d; x[j] += d
            stall += 1
            if stall > 400:
                sc *= 0.9; stall = 0
    return bx, bc
