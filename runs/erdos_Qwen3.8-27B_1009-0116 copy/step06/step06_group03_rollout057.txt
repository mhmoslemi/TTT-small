            if sh > 0:

        else:

            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

        v = x.sum() - target
        g += 2.0 * pen * v
        return g

                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lbfgs(h, alpha, pen, maxiter):
    n = len(h)
    target = 0.5 * n
    bnds = [(0.0, 1.0)] * n

    def obj(x):
        c = _corr(x)
        sm = _smax(c, alpha)
        v = x.sum() - target
        return sm + pen * v * v

    def grad(x):
        g = _sgrad(x, alpha)
        v = x.sum() - target
        g += 2.0 * pen * v
        return g

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topgrad(h, m):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-m:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * (2.0 / n) / m

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _subgrad(h, nit, s0, m, rng):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    v = np.zeros(n); mu = 0.7; step = s0; stuck = 0
    for _ in range(nit):
        g = _topgrad(best, m)
        v = mu * v + g
        improved = False
        for f in [1.0, 0.7, 0.4, 0.2, 0.1, 0.05, 0.02]:
            hnew = _proj(best - step * f * v, 0.5 * n)
            cn = _eval(hnew)
            if cn < best_c - 1e-15:
                best = hnew; best_c = cn
                step = min(step * 1.08, 0.35); improved = True; stuck = 0
                break
        if not improved:
            step *= 0.55; stuck += 1
            if stuck > 40:
                best = _repair(best + 0.008 * rng.randn(n), n)
                best_c = _eval(best); step = s0 * 0.7; stuck = 0; v = np.zeros(n)
    return best, best_c

def _fine(h, ntry, rng):
    n = len(h); h = h.copy()
    cc = _eval(h); best = h.copy(); bc = cc
    for _ in range(ntry):
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = (1e-5 + 4e-5 * rng.rand()) * (rng.rand() * 2 - 1)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < cc:
            cc = cn
            if cn < bc:
                bc = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, bc
