        return np.full(n, 0.5)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(_corr(h).max() * 2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:sh+cnt] -= wk * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:]
    return g * dx

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(h))

def _active_grad(h, n_act, beta):
    n = len(h)
    c = _corr(h)
    m = c.max()
    if n_act >= 2 * n - 1 or beta < 10:
        s = np.clip(beta * (c - m), -400, 0)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
    else:
        idx = np.argsort(c)[::-1][:n_act]
        w = np.zeros(2*n-1)
        sv = c[idx] - m
        sv = np.clip(sv * beta, -400, 0)
        sv -= sv.max()
        wn = np.exp(sv)
        wn /= wn.sum()
        w[idx] = wn
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = (n-1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:sh+cnt] -= wk * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:]
    return g * dx

def _nesterov_grad(h, alpha, n_steps, t_end, momentum=0.9):
    n = len(h)
    y = h.astype(float).copy()
    v = np.zeros(n)
    step = 0.015
    f_prev = _smooth_obj(y, alpha)
    for it in range(n_steps):
        if time.time() > t_end:
            break
        g_y = _smooth_grad(y, alpha)
        v = momentum * v + (1 - momentum) * g_y
        h_new = _proj(y - step * v, n)
        f_new = _smooth_obj(h_new, alpha)
        if f_new <= f_prev + 1e-15:
            v = momentum * (h_new - y)
            y = h_new
            f_prev = f_new
            step = min(step * 1.08, 0.4)
        else:
            for _bt in range(10):
                step *= 0.5
                h_try = _proj(y - step * v, n)
                f_try = _smooth_obj(h_try, alpha)
                if f_try <= f_prev + 1e-15:
                    v = momentum * (h_try - y)
                    y = h_try
                    f_prev = f_try
                    break
            else:
                step *= 0.3
    return y

def _active_descent(h, n_rounds, t_end, n_act=12, beta=150):
    n = len(h)
    hc = h.copy()
    c5c = _eval(hc)
    stuck = 0
    for _ in range(n_rounds):
        if time.time() > t_end:
            break
        g = _active_grad(hc, n_act, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for st in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]:
            hq = _proj(hc - st * g, n)
            cq = _eval(hq)
            if cq < c5c - 1e-16:
                hc, c5c = hq, cq
                improved = True
                stuck = 0
                break
        if not improved:
            stuck += 1
            if stuck > 15:
                hc += 0.005 * np.random.randn(n)
                hc = _proj(hc, n)
                c5c = _eval(hc)
                stuck = 0
    return hc, c5c

def _sa_opt(h, t_end, n_iter, rng, t0=0.008, t1=3e-7):
    n = len(h)
    hc = h.copy()
    cc = _eval(hc)
    bh = hc.copy()
    bc = cc
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0 * (t1/t0) ** (i / max(1, n_iter-1))
        mode = rng.randint(3)
        if mode == 0:
            i1, i2 = rng.randint(n), rng.randint(n)
            if i1 == i2: continue
            d = rng.randn() * (0.003 + 0.015*t)
            old = (hc[i1], hc[i2])
            hc[i1], hc[i2] = hc[i1]+d, hc[i2]-d
            hc = _proj(hc, n)
            cn = _eval(hc)
            if cn < cc or rng.rand() < np.exp(-(cn-cc)/max(t,1e-10)):
                cc = cn
                if cn < bc: bc, bh = cn, hc.copy()
            else:
                hc[i1], hc[i2] = old
                hc = _proj(hc, n)
        elif mode == 1:
            bl = rng.randint(3, min(20, n//4))
            st = rng.randint(0, n - bl)
            hc2 = hc.copy()
            hc2[st:st+bl] = 1.0 - hc2[st:st+bl]
            hc2 = _proj(hc2, n)
            cn = _eval(hc2)
            if cn < cc or rng.rand() < np.exp(-(cn-cc)/max(t,1e-10)):
                hc, cc = hc2, cn
                if cn < bc: bc, bh = cn, hc.copy()
            else:
                hc[:] = hc
        else:
            i1 = rng.randint(n)
            d = rng.randn() * (0.002 + 0.01*t)
            old = hc[i1]
            hc[i1] += d
            hc = _proj(hc, n)
            cn = _eval(hc)
            if cn < cc or rng.rand() < np.exp(-(cn-cc)/max(t,1e-10)):
                cc = cn
                if cn < bc: bc, bh = cn, hc.copy()
            else:
                hc[i1] = old
                hc = _proj(hc, n)
    return bh, bc

def _slsqp(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def cf(x): return x.sum() - n*0.5
    def cj(x): return np.ones(n)
    cons = [{'type':'eq','fun':cf,'jac':cj}]
    f0 = _smooth_obj(h, alpha)
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter':maxiter,'ftol':1e-16,'disp':False})
        if time.time() > t_end or not res.success:
            return _proj(h, n)
        if res.fun < f0 - 1e-13:
            return _proj(res.x, n)
    except Exception:
        pass
    return _proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def rem(): return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _proj(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0
    print(f"Start: C5={best_c5:.8f} n={best_n}")

    # Phase 1: Nesterov soft-max gradient, ramping alpha
    for alpha in [200, 800, 3000, 10000, 40000, 150000]:
        if rem() < 280: break
        t_left = rem() - 279
        ns = min(int(t_left / 0.04), 2000)
        hc = _nesterov_grad(best_h, alpha, ns, t0 + rem() - 279, momentum=0.85)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()
            print(f"  nesterov a={alpha}: {best_c5:.8f}")

    # Phase 2: Active-set descent
    if rem() > 200:
        hc, cc = _active_descent(best_h, 400, t0+rem()-199, n_act=8, beta=80)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()
            print(f"  active8: {best_c5:.8f}")
    if rem() > 160:
        hc, cc = _active_descent(best_h, 300, t0+rem()-159, n_act=16, beta=40)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()
            print(f"  active16: {best_c5:.8f}")

    # Phase 3: SLSQP
    if rem() > 150:
        hc = _slsqp(best_h, 80000, 2000, t0+rem()-149)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()
            print(f"  slsqp1: {best_c5:.8f}")

    # Phase 4: SA with block moves
    if rem() > 120:
        ni = min(int((rem()-110)/0.004), 100000)
        hc, cc = _sa_opt(best_h, t0+rem()-110, ni, rng)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()
            print(f"  SA: {best_c5:.8f}")

    # Phase 5: Re-do active descent after SA
    if rem() > 80:
        hc, cc = _active_descent(best_h, 300, t0+rem()-79, n_act=10, beta=60)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()
            print(f"  active_post: {best_c5:.8f}")

    # Phase 6: Coordinate fine-tune
    if rem() > 40:
        hc = best_h.copy()
        cc = _eval(hc)
        nt = min(int((rem()-35)/0.002), 30000)
        for _ in range(nt):
            if time.time() > t0 + rem() - 35: break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j: continue
            d = 1.5e-5 * rng.randn()
            a, b = hc[i]+d, hc[j]-d
            if a < -0.001 or a > 1.001 or b < -0.001 or b > 1.001: continue
            hc[i], hc[j] = a, b
            hc = _proj(hc, best_n)
            cq = _eval(hc)
            if cq < cc - 1e-18:
                cc = cq
                if cq < best_c5: best_c5, best_h = cq, hc.copy()
            else:
                hc[i], hc[j] = a-d, b+d
                hc = _proj(hc, best_n)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()
            print(f"  coord: {best_c5:.8f}")

    # Phase 7: Final SLSQP
    if rem() > 30:
        hc = _slsqp(best_h, 300000, 1500, t0+rem()-25)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()
            print(f"  final_slsqp: {best_c5:.8f}")

    best_h = _proj(best_h, best_n)
    best_c5 = _eval(best_h)
    print(f"FINAL: C5={best_c5:.8f} n={best_n}")
    return (best_h, best_c5, best_n)
