import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _wgrad(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[:n - sh] += wk * (-h[:n - sh])
            else:
                g[:n] += wk * (-h[:n])
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] += wk * (-h[:n - sp])
    return g

def _lse_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    return (2.0 / n) * _wgrad(x, w)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cfun(x):
        return x.sum() - n * 0.5
    def _cjac(x):
        return np.ones(n)
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _reweight(h, n_iter, t_start, t_end, top_frac=0.05, T0=0.5, Tend=0.02):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        frac = it / max(n_iter - 1, 1)
        T = T0 + (Tend - T0) * frac
        c = _eval_vec(h)
        m = max(int(np.ceil(top_frac * len(c))), 3)
        idx = np.argsort(c)[-m:]
        cmin, cmax = c[idx].min(), c[idx].max()
        if cmax - cmin > 1e-12:
            w = np.zeros(2 * n - 1)
            w[idx] = np.exp((c[idx] - cmin) / (T * (cmax - cmin)))
        else:
            w = np.zeros(2 * n - 1); w[idx] = 1.0
        w /= w.sum()
        g = _wgrad(h, w)
        dx = 2.0 / n
        # Backtracking line search
        best_h_try, best_c_try = h.copy(), cur
        step = 0.1 * dx
        for _ in range(20):
            ht = _repair(h - step * g, n)
            ct = _eval(ht)
            if ct < best_c_try:
                best_h_try, best_c_try = ht, ct
            step *= 0.6
            if step < 1e-6:
                break
        if best_c_try < cur - 1e-14:
            h, cur = best_h_try, best_c_try
        else:
            break
    return h, cur

def _pair_cd(h, n, n_iter, scales, t_end, seed=99):
    rng = np.random.RandomState(seed)
    cur = _eval(h)
    for scale in scales:
        for _ in range(n_iter):
            if time.time() > t_end:
                return h, cur
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur - 1e-15:
                cur = cn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, cur

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.clip(np.interp(xn, xo, h_old), 0.0, 1.0)
