def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_weighted(h, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        sh = (n - 1) - k
        wk = w[k]

            g[sp:] += wk * (1.0 - h[:n - sp])

    return g

            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

    """Multiplicative-weights projected gradient"""
    n = len(h)
    best = h.copy()

    w = np.exp(-c / beta_start)
    w /= w.sum()

    step = step0

        c = _corr(best)
        w = np.exp(np.clip(-c / beta, -500, 0))
        w /= w.sum()
        
        g = _grad_weighted(best, w) * (2.0 / n)
        v = mu * v - g
        step_here = step / (it + 10)

        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:

            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best = h_new
                best_c = c_new
                improved = True
                break
        if not improved:
            step *= 0.7
    
    return best, best_c

def _lbfgs_weighted(h, w, maxiter, ftol):
    """L-BFGS-B on weighted objective with constraint sum(h)=n/2"""
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        c = _corr(x)
        return float(np.dot(w, c) * dx)
    
    def grad(x):
        return _grad_weighted(x, w) * dx
    
    bnds = [(0.0, 1.0)] * n

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return _repair(h, n)

def _fine_search(h, n_tries, rng):
    n = len(h); h = h.copy()
    c_cur = _eval(h)
    best = h.copy(); best_c = c_cur
    for _ in range(n_tries):

        j = rng.randint(n)

        d = 5e-5 * rng.randn()

        h[i], h[j] = ni, nj

            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c

def _grad_w(h, w):

def _nesterov_pg(h, n_iters, beta0, beta1, step0):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c = _eval(best)
    x = h.copy()
    v = np.zeros(n)
    mu = 0.8
    step = step0
    stall = 0
    for it in range(n_iters):
        t = it / max(n_iters - 1, 1)
        beta = beta0 + (beta1 - beta0) * t
        c = _corr(x)
        w = np.exp(np.clip(-c / beta, -500, 0))
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= ws
        g = _grad_w(x, w) * dx
        v = mu * v - g
        lr = step / (1.0 + 0.1 * t)
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            x_new = _proj(x - lr * f * v, 0.5 * n)
            c_new = _eval(x_new)

                x = x_new
                best_c = c_new
                best = x.copy()
                improved = True
                stall = 0
                break

            step *= 0.8

                best_c = _eval(x)
                best = x.copy()
                v = np.zeros(n)
                step = step0
                stall = 0

def _slsqp_w(h, w, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = _corr(x)
        return float(np.dot(w, c) * dx)
    def grd(x):
        return _grad_w(x, w) * dx
    def cf(x):
        return np.array([x.sum() - 0.5 * n])
    def cj(x):
        return np.array([np.ones(n)])
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, jac=grd, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _fine(h, n_tries, rng, scale=5e-5):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur - 1e-16:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c

                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el():
        return time.time() - t0
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h0 = _repair(h0, n)
    best = h0.copy(); best_c = _eval(best); best_n = n

    # Phase 1: Nesterov PG with alpha annealing
    if _el() < budget_s - 500:
        allot = budget_s - 500 - _el()
        n_it = min(3500, int(allot / 0.022))
        best, best_c = _npg(best, max(n_it, 400), 200, 50000, 0.1, rng)

    # Phase 2: SLSQP with increasing alpha
    for alpha in [500, 3000, 20000, 100000]:
        if _el() > budget_s - 400:
            break
        w = _sw(best, alpha)
        h_new = _slsqp_w(best, w, best_n, 600, 1e-15)
        c_new = _eval(h_new)
        if c_new < best_c:
            best = h_new; best_c = c_new

    # Phase 3: NPG tighter
    if _el() < budget_s - 300:
        allot = budget_s - 300 - _el()
        n_it = min(2500, int(allot / 0.022))
        best, best_c = _npg(best, max(n_it, 300), 10000, 200000, 0.04, rng)

    # Phase 4: SLSQP very tight
    for alpha in [200000, 500000, 1000000]:
        if _el() > budget_s - 200:
            break
        w = _sw(best, alpha)
        h_new = _slsqp_w(best, w, best_n, 800, 1e-16)
        c_new = _eval(h_new)
        if c_new < best_c:
            best = h_new; best_c = c_new

    # Phase 5: Coordinate relaxation
    if _el() < budget_s - 150:
        best, best_c = _coord_relax(best, 4)

    # Phase 6: Upsample
    if _el() < budget_s - 300:
        n1 = best_n * 2
        h_up = np.repeat(best, 2)
        h_up = _repair(h_up, n1)
        for alpha in [1000, 20000, 200000]:
            if _el() > budget_s - 120:
                break
            w = _sw(h_up, alpha)
            h_up = _slsqp_w(h_up, w, n1, 500, 1e-15)
            c_new = _eval(h_up)
            if c_new < best_c:
                best = h_up; best_c = c_new; best_n = n1
        if _el() < budget_s - 60:
            n_it = min(1200, int((budget_s - 60 - _el()) / 0.035))
            best, best_c = _npg(best, max(n_it, 200), 50000, 300000, 0.02, rng)
            best_n = len(best)
        if _el() < budget_s - 30:
            best, best_c = _coord_relax(best, 2)

    # Phase 7: Final SLSQP
    if _el() < budget_s - 40:
        w = _sw(best, 2000000)
        h_new = _slsqp_w(best, w, best_n, 1200, 1e-17)
        c_new = _eval(h_new)
        if c_new < best_c:
            best = h_new; best_c = c_new; best_n = len(best)

    # Phase 8: Fine pairwise
    if _el() < budget_s - 5:
        n_tr = min(int((budget_s - 5 - _el()) / 0.0025), 50000)
        best, best_c = _fine(best, max(n_tr, 100), rng, 4e-5)
        best_n = len(best)

    best = _repair(best, best_n)
    best_c = _eval(best)
    return (best, best_c, best_n)
