def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                if diff > 0:

                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smooth_val(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx2 = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx2
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf_n, 'jac': _cj}]
    cons[0]['fun'] = _mk_cf(n)
    try:
        res = minimize(_smooth_val, h, jac=lambda x: _smooth_grad(x, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _mk_cf(n):
    return lambda x: x.sum() - 0.5 * n
def _cf_n(x): return x.sum()
def _cj(x): return np.ones(len(x))

def _topk_grad(h, top_k):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g / len(idx) * 2.0 / n

def _momentum_descent(h, n_iter, step0, top_k, mu):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    v = np.zeros(n); step = step0; stuck = 0
    for _ in range(n_iter):
        g = _topk_grad(best, top_k)
        v = mu * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-16:
                best = h_new; best_c = c_new
                step = min(step * 1.15, 0.4)
                improved = True; stuck = 0
                break
        if not improved:
            step *= 0.55; stuck += 1
            if stuck > 60:
                best = _repair(best + 0.008 * np.random.randn(n), n)
                best_c = _eval(best); step = step0; stuck = 0
                v = np.zeros(n)
    return best, best_c

def _paired_search(h, n_sweeps, rng):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy(); best_c = c_cur
    ds = [1e-4, 5e-5, 2e-5, 1e-5, 5e-6]
    for s in range(n_sweeps):
        perm = rng.permutation(n)
        for i in perm:
            j = (i + n // 3) % n
            if j == i:
                continue
            hi, hj = h[i], h[j]
            for d in ds:
                for sign in [1, -1]:
                    dd = sign * d
                    ni, nj = hi + dd, hj - dd
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    cn = _eval(h)
                    if cn < c_cur - 1e-15:
                        c_cur = cn
                        if cn < best_c:
                            best_c = cn; best = h.copy()
                    else:
                        h[i], h[j] = ni - dd, nj + dd
                        break
    return best, best_c
