def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-10:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0 and sh < n:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        elif sh < 0:
            sp = -sh
            if sp < n:
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    def _obj_wrapped(x):
        return _smooth_obj(x, alpha)
    def _jac_wrapped(x):
        return _smooth_grad(x, alpha)
    try:
        res = minimize(_obj_wrapped, h, jac=_jac_wrapped, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha, n_iter, lr0=0.05, decay=0.9995):
    n = len(h)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _eval(h)
    lr = lr0
    for i in range(n_iter):
        g = _smooth_grad(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        beta = min(0.999, 0.90 + 0.099 * i / max(n_iter, 1))
        v = beta * v - lr * g
        h_new = _repair(h + v, n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        lr *= decay
    return best_h, best_c5

def _focused_descent(h, nc, max_rounds, top_k=5):
    c5c = _eval(h)
    dx = 2.0 / nc
    for r in range(max_rounds):
        ov = np.correlate(h, 1.0 - h, mode='full')
        n_top = min(top_k, len(ov))
        top_idx = np.argsort(ov)[-n_top:]
        g = np.zeros(nc)
        for kk in top_idx:
            g += _grad_lag(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        step = 0.1
        improved = False
        for _ in range(22):
            h_new = _repair(h - step * g, nc)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
            step *= 0.5
        if not improved:
            g1 = _grad_lag(h, top_idx[-1], dx)
            gn1 = np.linalg.norm(g1)
            if gn1 < 1e-14:
                break
            g1 /= gn1
            step = 0.1
            for _ in range(22):
                h_new = _repair(h - step * g1, nc)
                c5n = _eval(h_new)
                if c5n < c5c - 1e-15:
                    h = h_new
                    c5c = c5n
                    improved = True
                    break
                step *= 0.5
            if not improved:
                break
    return h, c5c

def _binary_local(h_init, n, n_iter, rng, deadline):
    h = (h_init > 0.5).astype(float)
    h = _repair(h, n)
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    stuck = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] <= h[j]:
            h[i], h[j] = 1.0 - h[j], 1.0 - h[i]
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        h2 = _repair(h, n)
        c5n = _eval(h2)
        if c5n < c5c - 1e-14:
            c5c = c5n
            h = h2
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h.copy()
            stuck = 0
        else:
            h[i], h[j] = 1.0 - h[j], 1.0 - h[i]
            stuck += 1
            if stuck > n * 8:
                break
    return best_h, best_c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _periodic_init(n, period, rng):
    pattern = (rng.rand(period) < 0.5).astype(float)
    h = np.tile(pattern, n // period + 1)[:n]
    h += rng.randn(n) * 0.08
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Nesterov momentum at parent resolution
    if tl() > 180:
        for alpha in [5000, 20000, 80000, 200000]:
            if tl() < 50:
                break
            h_n = best_h.copy()
            h_n, c5 = _nesterov(h_n, alpha, 400, lr0=0.06, decay=0.9995)
            if c5 < best_c5:
                best_c5, best_h = c5, h_n.copy()

    # Phase 2: Focused descent with varying top_k
    if tl() > 150:
        for tk in [3, 5, 8]:
            if tl() < 45:
                break
            h_f = best_h.copy()
            h_f, c5 = _focused_descent(h_f, best_n, 500, top_k=tk)
            if c5 < best_c5:
                best_c5, best_h = c5, h_f.copy()

    # Phase 3: SLSQP polish
    if tl() > 120:
        for alpha in [50000, 200000, 600000]:
            if tl() < 40:
                break
            h_s = _slsqp(best_h, alpha, 1200, 1e-17)
            c5 = _eval(h_s)
            if c5 < best_c5:
                best_c5, best_h = c5, h_s.copy()
            h_f, c5 = _focused_descent(best_h, best_n, 300, top_k=3)
            if c5 < best_c5:
                best_c5, best_h = c5, h_f.copy()

    # Phase 4: Multi-resolution exploration
    if tl() > 220:
        for n_try in [80, 120, 160, 250, 320, 400, 500]:
            if tl() < 55:
                break
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r, c5 = _nesterov(h_r, 8000, 250, lr0=0.08, decay=0.999)
            h_r, c5 = _focused_descent(h_r, n_try, 350, top_k=5)
            h_r = _slsqp(h_r, 60000, 800, 1e-17)
            c5 = _eval(h_r)
            h_r, c5 = _focused_descent(h_r, n_try, 200, top_k=3)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 5: Binary local search at small n
    if tl() > 180:
        for n_bb in [30, 40, 60, 80, 100, 140]:
            if tl() < 45:
                break
            h_init = _resample(best_h, n_bb)
            h_b, c5 = _binary_local(h_init, n_bb, 80000, rng, t0 + tl() - 10)
            if c5 < best_c5:
                h_ref = _repair(h_b, n_bb)
                h_ref, c5 = _focused_descent(h_ref, n_bb, 300, top_k=5)
                h_ref = _slsqp(h_ref, 80000, 800, 1e-17)
                c5 = _eval(h_ref)
                h_ref, c5 = _focused_descent(h_ref, n_bb, 200, top_k=3)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h_ref.copy(), n_bb

    # Phase 6: Structured periodic multi-start
    if tl() > 130:
        for n_try in [best_n, 250, 400]:
            if tl() < 50:
                break
            for period in [4, 5, 6, 7, 9, 10]:
                if tl() < 18:
                    break
                h_p = _periodic_init(n_try, period, rng)
                h_p, c5 = _nesterov(h_p, 5000, 200, lr0=0.06, decay=0.999)
                h_p, c5 = _focused_descent(h_p, n_try, 250, top_k=5)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h_p.copy(), n_try

    # Phase 7: Basin hopping
    if tl() > 120:
        for scale in [0.03, 0.06, 0.1, 0.18]:
            if tl() < 40:
                break
            for _rep in range(3):
                if tl() < 20:
                    break
                noise = rng.randn(best_n) * scale
                h_p = _repair(best_h + noise, best_n)
                h_p, c5 = _nesterov(h_p, 12000, 200, lr0=0.04, decay=0.999)
                h_p, c5 = _focused_descent(h_p, best_n, 300, top_k=3)
                h_p = _slsqp(h_p, 100000, 800, 1e-17)
                c5 = _eval(h_p)
                h_p, c5 = _focused_descent(h_p, best_n, 200, top_k=3)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_p.copy()

    # Phase 8: Fine Nesterov polish
    if tl() > 30:
        h_fin = best_h.copy()
        h_fin, c5 = _nesterov(h_fin, 500000, 300, lr0=0.008, decay=0.9999)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    # Phase 9: Final focused descent
    if tl() > 12:
        h_fin, c5 = _focused_descent(best_h.copy(), best_n, 300, top_k=3)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
