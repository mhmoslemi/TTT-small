import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax_val(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_grad_w(c, alpha, topk=0):
    if topk > 0:
        idx = np.argsort(c)[::-1][:topk]
        w = np.zeros(len(c))
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -500, 500)
        w[idx] = np.exp(s - s.max())
        w /= w.sum()
    else:
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
    return w

def _grad_c_weighted(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _fista(h, alpha, n_steps, t_end, topk=0):
    n = len(h)
    dx = 2.0 / n
    h_x = h.copy()
    h_y = h.copy()
    f_y = _smax_val(_corr(h_y), alpha) * dx
    L = 0.5
    for i in range(n_steps):
        if time.time() > t_end:
            break
        if i > 0:
            beta_k = (i - 1) / (i + 2.0)
            h_y = _proj(h_x + beta_k * (h_x - h_y), n * 0.5)
        c_y = _corr(h_y)
        w = _smax_grad_w(c_y, alpha, topk)
        g = _grad_c_weighted(h_y, w) * dx
        h_new = _proj(h_y - (1.0 / L) * g, n * 0.5)
        f_new = _smax_val(_corr(h_new), alpha) * dx
        if f_new > f_y + (1.0 / (2.0 * L)) * np.dot(h_new - h_y, h_new - h_y):
            L *= 2.0
        else:
            L = max(L * 0.8, 0.05)
            f_y = f_new
            h_x = h_new
    return h_x

def _slsqp(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return _smax_val(_corr(x), alpha) * 2.0 / n
    def grad(x):
        c = _corr(x)
        w = _smax_grad_w(c, alpha)
        g = _grad_c_weighted(x, w) * 2.0 / n
        return g
    def cfun(x):
        return x.sum() - n * 0.5
    cons = [{'type': 'eq', 'fun': cfun, 'jac': lambda x: np.ones(n)}]
    f0 = obj(h)
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18, 'disp': False})
        if obj(res.x) < f0 - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _lag_balance(h, n_rounds, t_end, topk=8):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5 = _eval(h_cur)
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        c = _corr(h_cur)
        idx = np.argsort(c)[::-1][:topk]
        g = np.zeros(n)
        for kk in idx:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += (1.0 - h_cur[sh:])
                if sh > 0:
                    g[sh:] -= h_cur[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h_cur[:n - sp])
                g[:n - sp] -= h_cur[sp:]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h_cur, c5 = h_new, c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5

    n = len(h)
    c_cur = _eval(h)
    best_h, best_c = h.copy(), c_cur
    t0s, t1s = 0.005, 2e-6
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0s * (t1s / t0s) ** (i / max(1, n_iter - 1))

            continue

        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-10)):
            c_cur = c_new
            if c_cur < best_c:
                best_c, best_h = c_cur, h.copy()
        else:
            h[i1], h[i2] = old1, old2

    return best_h, best_c
