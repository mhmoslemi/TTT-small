import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sc_obj(x, k):
    n = len(x)
    sh = (n - 1) - k
    if sh >= 0:
        return float(np.sum(x[sh:] * (1.0 - x[:n - sh])))
    sp = -sh
    return float(np.sum(x[:n - sp] * (1.0 - x[sp:])))

def _sc_grad(x, k):
    n = len(x)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - x[sh:])
        if sh > 0:
            g[sh:] -= x[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - x[:n - sp])
        g[:n - sp] -= x[sp:]
    return g

def _se(x):
    return x.sum() - len(x) * 0.5

def _sj(x):
    return np.ones(len(x))

def _lssm(x, alpha):
    c = _corr_full(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _lssm_grad(x, alpha):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_lssm, h, jac=_lssm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _cyc_slsqp(h, n_cycles, maxit, deadline):

        pert = best_h + rng.randn(len(best_h)) * 0.05
        pert = _repair(pert, len(pert))
        hp = _cyc_slsqp(pert, 60, 400, min(dl7, time.time() + 15))
        upd(hp)
        hp = _slsqp(hp, 1000000, 2000, 1e-20)
        upd(hp)

    # Phase 8: Subgradient push-down on worst shifts (80s)
    dl8 = t0 + 790
    h8 = best_h.copy()
    n = len(h8)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    c5c = _eval(h8)
    for _it in range(200):
        if time.time() > dl8:
            break
        c = _corr_full(h8)
        thr = c.max() - 0.01
        active = np.where(c >= thr)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        g_avg = np.zeros(n)
        for k in active[:10]:
            g_avg += _sc_grad(h8, k)
        g_avg /= len(active)
        for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _repair(h8 - step * g_avg, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h8, c5c = h_new, c5n
                break
    upd(h8)

    # Phase 9: Final fine SLSQP + cycling (polish) (80s)
    dl9 = t0 + budget_s - 5
    for alpha in [2000000, 5000000]:
        if time.time() > dl9:
            break
        hf = _slsqp(best_h, alpha, 5000, 1e-21)
        upd(hf)
        hf = _cyc_slsqp(hf, 30, 400, dl9)
        upd(hf)

    # Phase 10: Micro coordinate descent (last few seconds)
    if elapsed() < budget_s - 3:
        hm = best_h.copy()
        c5m = _eval(hm)
        nc = len(hm)
        for scale in [0.001, 0.0003]:
            if elapsed() > budget_s - 3:
                break
            for _it in range(2000):
                if elapsed() > budget_s - 3:
                    break
                i, j = rng.randint(nc), rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = hm[i] + d, hm[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                hm[i], hm[j] = ni, nj
                c5n = _eval(hm)
                if c5n < c5m - 1e-16:
                    c5m = c5n
                else:
                    hm[i], hm[j] = ni - d, nj + d
        if c5m < best_c5 - 1e-14:
            best_c5, best_h = c5m, hm.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
