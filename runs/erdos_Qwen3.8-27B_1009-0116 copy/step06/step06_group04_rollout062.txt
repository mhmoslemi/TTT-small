```python
        y = _repair(y, n)
        x = x_new
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return best_x, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _subgrad(h):
    n = len(h); dx = 2.0 / n
    c = _cvec(h); ks = int(c.argmax()); sh = ks - (n - 1)
    g = np.zeros(n)

        else:

            g[sh:] -= h[:n-sh]

        t = -sh
        g[t:] += (1.0 - h[:n-t])
        g[:n-t] -= h[t:]
```

```python
def _obj_ls(x, alpha):
    c = _cvec(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad_ls(x, alpha):
    n = len(x); dx = 2.0 / n
    c = _cvec(x); m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            if sh == 0:
                g += wk * (1.0 - 2.0 * x)
            else:
                g[:n-sh] += wk * (1.0 - x[sh:])
                g[sh:] -= wk * x[:n-sh]
        else:
            t = -sh
            g[t:] += wk * (1.0 - x[:n-t])
            g[:n-t] -= wk * x[t:]
    return g
```

```python
def _adam(h0, a0, a1, niter, l0, l1, seed, dl):
    n = len(h0); rng = np.random.RandomState(seed)
    x = h0.copy(); m = np.zeros(n); v = np.zeros(n); t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    bx = x.copy(); bc = _eval(x)
    for i in range(niter):
        if time.time() > dl: break
        t += 1; f = i / max(niter - 1, 1)
        al = a0 * (a1 / a0) ** f; lr = l0 * (l1 / l0) ** f
        if t % 500 == 0: x = x + 0.005 * rng.randn(n)
```

Wait, for sh >= 0: I want to reduce c[ks] = sum_{i=0}^{n-1-sh} x[i]*(1-x[i+sh]). The "bad" terms are where x[i]=1 and x[i+sh]=0. If I swap x[i] to 0 (making it 0) or x[i+sh] to 1 (making it 1), I reduce this term.

But swapping x[i] from 1 to 0 and x[i+sh] from 0 to 1 is a valid swap (maintains sum). And it reduces c[ks] by 1 (removes one (1,0) pair and adds a (0,1) pair which doesn't contribute).

However, this swap might increase c at OTHER shifts. So I need to check the full C₅ after the swap.

The "focused" approach is: with some probability, do a focused swap (targeting the argmax), and with other probability, do a random swap (to escape local minima).
