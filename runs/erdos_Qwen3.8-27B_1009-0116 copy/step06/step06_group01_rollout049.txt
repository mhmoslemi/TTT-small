import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag_h(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if 0 <= sh < n:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    elif 0 <= -sh < n:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _build_h(y, m):
    h = np.empty(2 * m)
    h[:m] = y
    h[m:] = 1.0 - y
    return h

def _grad_lag_y(y, m, k):
    h = _build_h(y, m)
    n = 2 * m
    dx = 2.0 / n
    gh = _grad_lag_h(h, k, dx)
    return gh[:m] - gh[m:]

def _soft_obj(y, alpha):
    h = _build_h(y, len(y))
    corr = _eval_corr(h)
    mval = float(corr.max())
    s = np.clip(alpha * (corr - mval), -400, 400)
    return mval + float(np.log(np.sum(np.exp(s)))) / alpha

def _soft_grad(y, alpha):
    h = _build_h(y, len(y))
    m = len(y)
    n = len(h)
    dx = 2.0 / n
    corr = _eval_corr(h)
    mval = float(corr.max())
    s = np.clip(alpha * (corr - mval), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(m)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        gh = _grad_lag_h(h, k, dx) * w[k]
        g += gh[:m] - gh[m:]
    return g

def _slsqp_sym(y, alpha, maxiter, ftol):
    m = len(y)
    bnds = [(0.0, 1.0)] * m
    try:
        res = minimize(_soft_obj, y, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return y.copy()

def _focused_descent_sym(y, m, max_rounds, top_n=3):
    h = _build_h(y, m)
    c5c = _eval(h)
    for r in range(max_rounds):
        corr = _eval_corr(h)
        nt = min(top_n, len(corr))
        top_idx = np.argsort(corr)[-nt:]
        g = np.zeros(m)
        for kk in top_idx:
            g += _grad_lag_y(y, m, kk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        best_c5, best_y = c5c, y.copy()
        step = 0.06
        for _ in range(28):
            y_new = np.clip(y - step * g, 0.0, 1.0)
            c5n = _eval(_build_h(y_new, m))
            if c5n < best_c5 - 1e-15:
                best_c5, best_y = c5n, y_new
            else:
                step *= 0.5
        if best_c5 >= c5c - 1e-15:
            g1 = _grad_lag_y(y, m, top_idx[-1])
            gn1 = np.linalg.norm(g1)
            if gn1 < 1e-14:
                break
            g1 /= gn1
            step = 0.04
            for _ in range(28):

                c5n = _eval(_build_h(y_new, m))

                    best_c5, best_y = c5n, y_new
                else:
                    step *= 0.5
            if best_c5 >= c5c - 1e-15:
                break
        y = best_y
        h = _build_h(y, m)
        c5c = best_c5
    return y, c5c

def _nesterov_sym(y0, m, alpha, lr, n_iter, momentum=0.85):
    y = y0.copy()
    v = np.zeros(m)
    best_y = y.copy()
    best_c5 = _eval(_build_h(y, m))
    for it in range(n_iter):
        v = momentum * v - lr * _soft_grad(y, alpha)

        if c5 < best_c5:
            best_c5, best_y = c5, y_new.copy()
        y = y_new
        if it % 50 == 0:
            lr *= 0.95
    return best_y, best_c5

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _symmetrize(h):
    n = len(h)
    m = n // 2
    y = (h[:m] + (1.0 - h[m:])) / 2.0
    return _build_h(np.clip(y, 0.0, 1.0), m)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h
