import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_at_lag(h, shift, dx):
    """Gradient of the correlation at a specific shift."""
    n = len(h)
    g = np.zeros(n)
    if 0 <= shift < n:
        i = np.arange(n - shift)
        g[i] += (1.0 - h[i + shift]) * dx
        if shift > 0:
            g[shift:] -= h[:n-shift] * dx
    elif -n < shift < 0:
        s = -shift
        i = np.arange(n - s)
        g[i+s] += (1.0 - h[i]) * dx
        if s > 0:
            g[:n-s] -= h[s:] * dx
    return g

        c = _corr(x)
        m = c.max()
        if alpha < 100:
            return m + 0.5 * np.sum((c - m)**2)  # quadratic smoothing
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s - s.max())))) / alpha
    
    def grad(x):
        c = _corr(x)
        m = c.max()
        if alpha < 100:
            return 2.0 * (c - m)  # dummy, won't use this path
        s = np.clip(alpha * (c - m), -500, 500)

def _repair(h, n):

    m = c.max()

    return m + np.log(float(np.sum(np.exp(s - s.max())))) / alpha

    dx = 2.0 / n

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

        for scale in [0.02, 0.01, 0.005]:
            if time.time() > deadline - 30:
                break
            h_p = _repair(best_h + rng.randn(best_n) * scale, best_n)
            h_p = _nesterov_opt(h_p, 1500, 5000, 100000, t0, min(time.time() + 50, deadline - 20))
            h_p = _focused_cd(h_p, best_n, 200, t0, deadline - 10)
            c5 = _eval(h_p)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()

    # Phase 8: Very fine coordinate descent polish
    if time.time() < deadline - 15:
        for scale in [0.002, 0.0005, 0.0001]:
            if time.time() > deadline - 8:
                break
            best_h, c5 = _pairwise_cd(best_h, best_n, rng, scale, 8000, t0, deadline - 5)
            if c5 < best_c5:
                best_c5 = c5

    # Phase 9: Final SLSQP + focused
    if time.time() < deadline - 10:
        hf = _slsqp(best_h, 100000, 2000)
        c5 = _eval(hf)
        if c5 < best_c5:
            best_c5, best_h = c5, hf
        best_h, c5 = _focused_cd(best_h, best_n, 100, t0, deadline)
        if c5 < best_c5:
            best_c5, best_h = c5, best_h

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
