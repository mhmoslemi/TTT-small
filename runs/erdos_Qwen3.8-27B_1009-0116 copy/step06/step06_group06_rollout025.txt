import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g * dx

def _kgrad(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _adam_opt(h, alphas, steps, lr0, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    for alpha in alphas:
        if time.time() > t_end:
            break
        m_v = np.zeros(n); v_v = np.zeros(n)
        b1, b2, eps = 0.85, 0.999, 1e-8
        f_cur = _smooth_obj(h_cur, alpha)
        lr = lr0
        stall = 0
        for si in range(steps):
            if time.time() > t_end:
                break
            g = _smooth_grad(h_cur, alpha)
            t_i = si + 1
            m_v = b1 * m_v + (1 - b1) * g
            v_v = b2 * v_v + (1 - b2) * g**2
            mh = m_v / (1 - b1**t_i)
            vh = v_v / (1 - b2**t_i)
            direction = mh / (np.sqrt(vh) + eps)
            h_new = _proj(h_cur - lr * direction, n * 0.5)
            f_new = _smooth_obj(h_new, alpha)
            if f_new < f_cur - 1e-15:
                h_cur = h_new
                f_cur = f_new
                stall = 0
                lr = min(lr * 1.02, lr0 * 3)
            else:
                stall += 1
                if stall > 60:
                    h_cur += 0.005 * rng.randn(n)
                    h_cur = _repair(h_cur, n)
                    f_cur = _smooth_obj(h_cur, alpha)
                    lr = lr0
                    stall = 0
                else:
                    lr *= 0.92
    return h_cur

def _active_opt(h, n_top, steps, lr0, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    c5 = _eval(h_cur)
    lr = lr0
    stall = 0
    for si in range(steps):
        if time.time() > t_end:
            break
        c = _corr(h_cur)
        top = np.argsort(c)[::-1][:n_top]
        g = np.zeros(n)
        for kk in top:
            g += _kgrad(h_cur, kk)
        g /= n_top
        g *= 2.0 / n
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = _proj(h_cur - lr * mult * g / gn, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h_cur = h_new
                c5 = c5n
                stall = 0
                improved = True
                break
        if not improved:
            stall += 1
            lr *= 0.85
            if stall > 80:
                h_cur += 0.003 * rng.randn(n)
                h_cur = _repair(h_cur, n)
                c5 = _eval(h_cur)
                lr = lr0
                stall = 0
    return h_cur, c5

def _slsqp_fin(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def cf(x): return x.sum() - n * 0.5
    def cj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    f0 = _smooth_obj(h, alpha)
    try:
        res = minimize(_smooth_obj, h.copy(), jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'disp': False})
        if res.fun < f0 - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h.copy(), n)

def _coord_cd(h, n_iter, t_end, rng, scale=1e-5):
    n = len(h)
    h_cur = h.copy()
    c5 = _eval(h_cur)
    for i in range(n_iter):
        if time.time() > t_end:
            break
        i1 = rng.randint(n); i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = scale * rng.randn()
        ni, nj = h_cur[i1] + d, h_cur[i2] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i1], h_cur[i2] = ni, nj
        c5n = _eval(h_cur)
        if c5n < c5 - 1e-18:
            c5 = c5n
        else:
            h_cur[i1], h_cur[i2] = h_cur[i1] - d, h_cur[i2] + d
    return h_cur, c5

def _upsample(h, f):
    n = len(h)
    out = np.zeros(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

    t0 = time.time()
    rng = np.random.RandomState(seed)
