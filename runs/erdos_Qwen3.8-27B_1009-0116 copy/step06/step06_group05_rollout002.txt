    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pg_grad(h, alpha, top_m):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    if top_m > 0 and top_m < len(c):
        idx = np.argsort(c)[-top_m:]
        w = np.zeros(len(c))
        w[idx] = np.exp(alpha * (c[idx] - cm))
        w /= w.sum()
    else:
        s = np.clip(alpha * (c - cm), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for k in nz:
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _pg_optimize(h, alphas, iters, top_m, t_end):
    n = len(h)
    best = _eval(h)
    hb = h.copy()
    for alpha in alphas:
        if time.time() > t_end:
            break
        hc = hb.copy()
        mom = np.zeros(n)
        step = 0.004
        for it in range(iters):
            if time.time() > t_end:
                break
            if it > 0 and it % 400 == 0:
                step *= 0.6
            if step < 1e-9:
                break
            g = _pg_grad(hc, alpha, top_m)
            mom = 0.85 * mom - step * g
            hnew = _repair(hc + mom, n)
            c5 = _eval(hnew)
            if c5 < best - 1e-15:
                hc = hnew
                hb = hnew.copy()
                best = c5
                step = min(step * 1.4, 0.025)
            else:
                step *= 0.45
                hc = hb.copy()
                mom *= 0.3
    return hb, best

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _coord_cd(h, n_iters, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    h = h.copy()
    for it in range(n_iters):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.003 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _worst_shift_opt(h, t_end, max_rounds=30):
    n = len(h)
    c5c = _eval(h)
    dx = 2.0 / n
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        ki = int(np.argmax(c))
        sh = (n - 1) - ki
        g = np.zeros(n)
        if sh >= 0:
            g[:n-sh] += dx * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= dx * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n-sp])
            g[:n-sp] -= dx * h[sp:]
        moved = False
        for s in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            hn = _repair(h - s * g, n)
            c5n = _eval(hn)
            if c5n < c5c - 1e-15:
                h, c5c = hn, c5n
                moved = True
                break
        if not moved:
            break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP polish at native resolution
    t_end = t0 + 120
    for alpha in [20000, 100000, 500000]:
        if time.time() > t_end:
            break
        hn = _slsqp(best_h, alpha, 800)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 2: Projected gradient with momentum
    t_end = t0 + 300
    hn, c5 = _pg_optimize(best_h, [50000, 200000, 1000000], 1500, 5, t_end)
    if c5 < best_c5:
        best_c5, best_h = c5, hn.copy()

    # Phase 3: Worst-shift focused descent
    t_end = t0 + 420
    hn, c5 = _worst_shift_opt(best_h, t_end, 40)
    if c5 < best_c5:
        best_c5, best_h = c5, hn.copy()
    hn = _slsqp(best_h, 1000000, 1000)
    c5 = _eval(hn)
    if c5 < best_c5:
        best_c5, best_h = c5, hn.copy()

    # Phase 4: Multi-resolution exploration
    t_end = t0 + 750
    for n_try in [200, 256, 320, 400, 512]:
        if time.time() > t_end:
            break
        h_r = _repair(_resample(best_h, n_try), n_try)
        h_r = _slsqp(h_r, 100000, 600)
        c5 = _eval(h_r)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r.copy(), n_try
        h_r, c5 = _pg_optimize(h_r, [200000, 2000000], 800, 3, t_end)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r.copy(), n_try
        h_r, c5 = _worst_shift_opt(h_r, t_end, 20)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r.copy(), n_try
        h_r = _slsqp(h_r, 5000000, 800)
        c5 = _eval(h_r)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 5: Random continuous restarts
    t_end = t0 + 950
    for trial in range(10):
        if time.time() > t_end:
            break
        nt = int(rng.choice([250, 310, 400, 500, 600]))
        h_r = _repair(0.5 + 0.3 * rng.randn(nt), nt)
        h_r = _slsqp(h_r, 50000, 400)
        h_r, c5 = _pg_optimize(h_r, [200000], 600, 3, t_end)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r, nt
        h_r, c5 = _worst_shift_opt(h_r, t_end, 15)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r, nt

    # Phase 6: Intensive final optimization
    t_end = t0 + budget_s - 80
    hn = _slsqp(best_h, 5000000, 2000)
    c5 = _eval(hn)
    if c5 < best_c5:
        best_c5, best_h = c5, hn.copy()
    hn, c5 = _pg_optimize(best_h, [5000000, 20000000], 1500, 3, t_end)
    if c5 < best_c5:
        best_c5, best_h = c5, hn.copy()
    hn, c5 = _worst_shift_opt(best_h, t_end, 30)
    if c5 < best_c5:
        best_c5, best_h = c5, hn.copy()

    # Phase 7: Coordinate descent micro-polish
    t_end = t0 + budget_s - 5
    hn, c5 = _coord_cd(best_h, 6000, rng, t_end)
    if c5 < best_c5:
        best_c5, best_h = c5, hn.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
