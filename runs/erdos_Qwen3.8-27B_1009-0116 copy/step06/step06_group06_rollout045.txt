from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s < 1e-13:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = -min(-diff, h[idx])
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        sh = n - 1 - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g * dx

def _lbf(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    best_x = h.copy()
    best_f = _obj(h, alpha)

            res = minimize(_obj, best_x, jac=_grad, args=(alpha,),
                         method='L-BFGS-B', bounds=bnds,

def _repair(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s < 1e-13:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _objective(h, alpha, lam):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    fx = (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)
    return fx + 0.5 * lam * (h.sum() - 0.5 * n) ** 2

def _gradient(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = n - 1 - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    g += lam * (h.sum() - 0.5 * n)
    return g

def _lbfgs(h, alpha, maxiter, t_end):
    n = len(h)
    lam = 10.0
    bnds = [(0.0, 1.0)] * n
    best_x, best_f = h.copy(), _eval(h)
    for _ in range(4):
        if time.time() > t_end:
            break
        try:
            res = minimize(_objective, best_x, jac=_gradient, args=(alpha, lam),
                          method='L-BFGS-B', bounds=bnds,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13, 'disp': False})
            x = _repair(res.x, n)
            f = _eval(x)
            if f < best_f:
                best_f = f
                best_x = x
        except Exception:
            break
    return best_x, best_f

def _coord_descent(h, t_end, n_iters=50000):
    n = len(h)
    c5 = _eval(h)
    rng = np.random.RandomState(12345)
    for _ in range(n_iters):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        best_d = 0.0
        base = c5
        for d in [0.02, 0.01, 0.005, 0.002, 0.001]:
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < base - 1e-16:
                best_d = d
                c5 = c5n
                break
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5

def _sa_opt(h, t_end, n_iter):
    n = len(h)
    rng = np.random.RandomState(777)
    c_cur = _eval(h)
    best_h, best_c = h.copy(), c_cur
    t0v, t1v = 0.005, 2e-6
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0v * (t1v / t0v) ** (i / max(1, n_iter - 1))
        i1, i2 = rng.randint(n), rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.002 + 0.015 * t)
        o1, o2 = h[i1], h[i2]
        h[i1], h[i2] = o1 + d, o2 - d
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-10)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        else:
            h[i1], h[i2] = o1, o2
            h = _repair(h, n)
    return best_h, best_c

def _binary_snap(h, n):
    hb = (h > 0.5).astype(float)
    if abs(hb.sum() - 0.5 * n) > n * 0.05:
        return h
    hb = _repair(hb, n)
    return hb
