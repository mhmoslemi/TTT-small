import numpy as np
from scipy.optimize import minimize, minimize_scalar

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

                          method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-22})
            h_new = _repair(res.x, n)
            c = _eval(h_new)
            if c < best_c - 1e-15:

                best = h_new

            pass
    return best, best_c

def _nm_opt(h0, iters=20000):
    n = len(h0)
    target = n / 2.0
    def penalized_obj(x):
        h = np.clip(x, 0.0, 1.0)

def _coord_1d(h, n_iter, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    target = n / 2.0
    c5 = _eval(h)
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        if hi <= 0 and hj >= 1:
            lo, hi_lo = 0.0, 0.0
        elif hi >= 1 and hj <= 0:
            lo, hi_lo = 0.0, 0.0
        else:
            lo = max(0.0, hi - 1.0 + hj)
            hi_lo = min(1.0, hi + 1.0 - hj)
        if lo >= hi_lo:
            continue
        def obj_1d(d):
            di = hi + d
            dj = hj - d
            if di < 0 or di > 1 or dj < 0 or dj > 1:
                return 1e10
            nh = h.copy()
            nh[i], nh[j] = di, dj
            return _eval(nh)

                            options={'maxiter': 50, 'xatol': 1e-12})
        if res.success and res.fun < c5 - 1e-16:
            h[i] = hi + res.x
            h[j] = hj - res.x
            c5 = res.fun
    return h, c5

            c5 = c_try
            if c5 < best_c5 - 1e-16:
