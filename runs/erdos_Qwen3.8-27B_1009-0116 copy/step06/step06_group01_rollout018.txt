Better approach: Use Armijo condition.
- f(x + α*g) ≤ f(x) + c*α*grad·g
- where c is typically 0.01

But since g is the negative gradient (we're descending), grad·g = -||g||².
So: f(x - η*g) ≤ f(x) - c*η*||g||²

For a backtracking line search:
- Start with η = η_max
- If Armijo not satisfied, η *= 0.5
- Repeat until satisfied or η < η_min

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            if diff < 1e-12: break
    elif diff < -1e-12:
        order = np.argsort(h)
        for idx in order:
            a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if diff > -1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh); g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n); g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n); g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp); g[i2] -= h[i2 + sp]
    return g * dx

def _armijo_line(h, g, c5c, dx, eta0, max_shrinks=30):
    n = len(h)
    best_h, best_c5 = h, c5c
    eta = eta0
    for _ in range(max_shrinks):
        h_new = _repair(h - eta * g, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-16:
            return h_new, c5n
        eta *= 0.55
        if eta < 1e-7:
            break
    return best_h, best_c5

def _subgrad_momentum(h, nc, max_iters, time_limit, m_top=4, eta_scale=0.05):
    dx = 2.0 / nc
    c5c = _eval(h)
    h = h.copy()
    mom = np.zeros(nc)
    beta = 0.7
    eta = eta_scale
    best_c5, best_h = c5c, h.copy()
    stuck = 0
    t_start = time.time()
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argpartition(c, -m_top)[-m_top:]
        c_top = c[top_idx]
        w = np.exp(30.0 * (c_top - c_top.max()))
        w /= w.sum()
        g = np.zeros(nc)
        for j, idx in enumerate(top_idx):
            g += w[j] * _grad_lag(h, idx, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        mom = beta * mom - eta * g / gn
        h_new, c5n = _armijo_line(h, mom, c5c, dx, 0.012, 20)
        if c5n < c5c - 1e-15:
            h, c5c = h_new, c5n
            stuck = 0
            beta = min(0.88, beta + 0.005)
            if c5c < best_c5:
                best_c5, best_h = c5c, h.copy()
        else:
            stuck += 1
            beta *= 0.95
            eta *= 0.92
            if stuck > 300:
                break
    return best_h, best_c5

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            if sh < n:
                i1 = np.arange(n - sh); g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n); g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            if sp < n:
                i1 = np.arange(sp, n); g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp); g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x): return x.sum() - len(x) * 0.5
def _sum_jac(x): return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _coord_descent(h, nc, rng, time_limit, scale0=0.002):
    c5c = _eval(h)
    t_start = time.time()
    scale = scale0
    for _ in range(200000):
        if time.time() - t_start > time_limit: break
        i, j = rng.randint(nc), rng.randint(nc)
        if i == j: continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
            scale *= 0.9995
            if scale < 1e-5: break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: SLSQP polish on parent
    if rem() > 400:
        h1 = _slsqp(best_h, 5000, 2500, 1e-17)
        c5 = _eval(h1)
        if c5 < best_c5:

    # Phase 2: Subgradient with momentum (increasing active set)

            if rem() < 60: break
            h2, c5 = _subgrad_momentum(best_h, best_n, 800, max(30, rem()-300), m_top=m)
            if c5 < best_c5:

    # Phase 3: SLSQP with higher alpha (more peaky)
    if rem() > 300:
        for alpha in [30000, 100000, 300000]:
            if rem() < 80: break
            h3 = _slsqp(best_h, alpha, 2000, 1e-18)
            c5 = _eval(h3)
            if c5 < best_c5:
                best_c5, best_h = c5, h3.copy()

    # Phase 4: More subgradient rounds
    if rem() > 250:

        if c5 < best_c5:

    # Phase 5: Multi-resolution exploration
    if rem() > 300:

            if rem() < 80: break

            hr = _slsqp(hr, 20000, 1800, 1e-17)

            if c5 < best_c5:

    # Phase 6: Structured restarts

            if rem() < 50: break

            h_p = _repair(best_h + rng.randn(best_n) * noise_scale, best_n)

        if c5 < best_c5:
            best_c5, best_h = c5, h8.copy()

    return (best_h, best_c5, best_n)
