```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _lse_val(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500, 500)
    sm = s - s.max()
    return mx + float(np.log(np.sum(np.exp(sm)))) / alpha

def _lse_weights(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500, 500)
    sm = s - s.max()
    w = np.exp(sm)
    w /= w.sum()
    return w

def _full_grad(h, w, dx):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:sp + cnt]
    return g

def _pobj(hp, T, m, alpha):
    h = np.tile(hp, m)
    return _lse_val(h, alpha) * 2.0 / (T * m)

def _pgrad(hp, T, m, alpha):
    h = np.tile(hp, m)
    w = _lse_weights(h, alpha)
    gf = _full_grad(h, w, 2.0 / (T * m))
    g = np.zeros(T)
    for p in range(m):
        g += gf[p * T:(p + 1) * T]
    return g

def _pobj_pen(hp, T, m, alpha):
    v = _pobj(hp, T, m, alpha)
    d = hp.sum() - T * 0.5
    return v + 50.0 * d * d

def _fobj(h, alpha):
    n = len(h)
    return _lse_val(h, alpha) * 2.0 / n

def _fgrad(h, alpha):
    n = len(h)
    w = _lse_weights(h, alpha)
    return _full_grad(h, w, 2.0 / n)

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _resample(h, n_new):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
```

```python
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()

            if d > 0:

                h[i] += a; d -= a
            else:

                h[i] -= a; d += a
```
