def _repair(h, n):
    # clip and fix sum
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_single_k(h, k):
    """Gradient of c_k = sum_i h[i]*(1-h[i+k]) w.r.t. h, in original indexing."""
    n = len(h)
    g = np.zeros(n)

    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _swap_search(h, n_tries, rng):
    """Local search via pair perturbations."""
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    for _ in range(n_tries):

        cn = _eval(best)
        if cn < best_c:

        else:

    return best, best_c

def _slsqp_opt(h, alpha, maxiter):
    """SLSQP with smooth max."""
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def obj_grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        n = len(x)

                g[:n - sp] -= x[sp:]
        return g
    
    def cf(x):

        return np.array([np.ones(n)])

                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:
        return _repair(h, n)

        nt = min(int((budget_s - 30 - _el()) / 0.006), 120000)
        best_h, best_c = _swap(best_h, nt, rng, scale=3e-5)
        best_n = len(best_h)

    # Phase 7: Coordinate sweep
    if _el() < budget_s - 5:
        npass = max(1, min(3, int((budget_s - 5 - _el()) / 3.0)))
        best_h, best_c = _coord_sweep(best_h, rng, n_passes=npass)
        best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
