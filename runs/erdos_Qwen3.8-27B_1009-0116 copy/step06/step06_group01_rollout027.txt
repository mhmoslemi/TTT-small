def _grad_topk(h, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-topk:]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            m2 = n - sh
            g[:m2] += (1.0 - h[sh:sh + m2]) * dx
            if sh > 0: g[sh:] -= h[:m2] * dx
        else:
            sp = -sh; m2 = n - sp
            g[sp:] += (1.0 - h[:m2]) * dx
            g[:m2] -= h[sp:] * dx
    return g

def _sum_eq(x): return x.sum() - len(x) * 0.5
def _sum_jac(x): return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol, t0, budget):
    n = len(h)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return float(m + np.log(float(np.sum(np.exp(s)))) / alpha)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(obj, h, jac=_grad_lse, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _project(res.x, n)
    except Exception:
        return _project(h, n)

def _topk_descent(h, n, topk, n_iter, t0, budget):
    c5c = _eval(h)
    for it in range(n_iter):
        if time.time() - t0 > budget - 25: break
        g = _grad_topk(h, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-14: break
        g /= gn
        step = 0.12
        improved = False
        for _ in range(18):
            h_new = _project(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new; c5c = c5n; improved = True; break
            step *= 0.5
        if not improved:
            g1 = _grad_topk(h, 1)
            gn1 = np.linalg.norm(g1)
            if gn1 < 1e-14: break
            g1 /= gn1
            improved2 = False
            for _ in range(12):
                h_new = _project(h - step * g1, n)
                c5n = _eval(h_new)
                if c5n < c5c - 1e-14:
                    h = h_new; c5c = c5n; improved2 = True; break
                step *= 0.5
            if not improved2: break
    return h, c5c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _pairwise_cd(h, n, rng, scale, n_iter, t0, budget):
    c5c = _eval(h)
    for _ in range(n_iter):
        if time.time() - t0 > budget - 15: break
        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _project(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    best_h = h.copy(); best_c5 = _eval(h); best_n = len(h)

    # Phase 1: SLSQP continuation at parent n
    for alpha in [400, 1500, 5000, 20000, 80000]:
        if time.time() - t0 > budget_s - 200: break
        h = _slsqp(best_h, alpha, 250, 1e-15, t0, budget_s)
        c5 = _eval(h)
        if c5 < best_c5: best_c5, best_h = c5, h.copy()

    # Phase 2: Top-1 and Top-2 focused descent
    if time.time() - t0 < budget_s - 120:
        h2 = best_h.copy()
        h2, c5 = _topk_descent(h2, best_n, 1, 400, t0, budget_s)
        if c5 < best_c5: best_c5, best_h = c5, h2.copy()
        h2 = best_h.copy()
        h2, c5 = _topk_descent(h2, best_n, 2, 300, t0, budget_s)
        if c5 < best_c5: best_c5, best_h = c5, h2.copy()

    # Phase 3: High-alpha SLSQP polish
    if time.time() - t0 < budget_s - 100:
        for alpha in [200000, 500000]:
            if time.time() - t0 > budget_s - 60: break
            h3 = _slsqp(best_h, alpha, 200, 1e-16, t0, budget_s)
            c5 = _eval(h3)
            if c5 < best_c5: best_c5, best_h = c5, h3.copy()

    # Phase 4: Multi-n exploration
    for n_try in [200, 280, 400, 520, 620, 780, 1000]:
        if time.time() - t0 > budget_s - 90: break
        h_r = _project(_resample(best_h, n_try), n_try)
        h_r = _slsqp(h_r, 8000, 200, 1e-14, t0, budget_s)
        h_r, c5 = _topk_descent(h_r, n_try, 1, 150, t0, budget_s)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 5: Refine at best n
    h = best_h.copy(); n = best_n
    if time.time() - t0 < budget_s - 60:
        h = _slsqp(h, 50000, 300, 1e-15, t0, budget_s)
        c5 = _eval(h)
        if c5 < best_c5: best_c5, best_h = c5, h.copy()
        h = best_h.copy()
        h, c5 = _topk_descent(h, best_n, 1, 250, t0, budget_s)
        if c5 < best_c5: best_c5, best_h = c5, h.copy()

    # Phase 6: Pairwise coordinate descent (fine)
    if time.time() - t0 < budget_s - 30:
        h6 = best_h.copy()
        for scale in [0.005, 0.001, 0.0003]:
            if time.time() - t0 > budget_s - 15: break
            h6, c5 = _pairwise_cd(h6, best_n, rng, scale, 3000, t0, budget_s)
            if c5 < best_c5: best_c5, best_h = c5, h6.copy()

    # Phase 7: Final SLSQP + topk
    if time.time() - t0 < budget_s - 8:
        h7 = _slsqp(best_h, 200000, 150, 1e-16, t0, budget_s)
        h7, c5 = _topk_descent(h7, best_n, 1, 100, t0, budget_s)
        if c5 < best_c5: best_c5, best_h = c5, h7.copy()

    # Phase 8: Last pairwise polish
    if time.time() - t0 < budget_s - 3:
        h8, c5 = _pairwise_cd(best_h.copy(), best_n, rng, 0.0001, 2000, t0, budget_s)
        if c5 < best_c5: best_c5, best_h = c5, h8.copy()

    best_h = _project(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
