def _pairwise_step(h, j, l, c, rng):
    """Find optimal δ for h[j]+=δ, h[l]-=δ. Returns (δ, new_c5)."""
    n = len(h)
    # Compute D_k = coeff_j - coeff_l in each c_k
    # Vectorized computation
    ks = np.arange(2*n-1)
    ms = ks - (n-1)

    # m >= 0: (1-h[i-m])*[i>=m] - h[i+m]*[i+m<n]
    # m < 0: (1-h[i+m])*[i+n-1-k<n] ... let me just use the formula
    
    def coeff(i):
        # Returns array of length 2n-1, coeff of h[i] in each c_k
        ccoef = np.zeros(2*n-1)
        pos = ms >= 0
        neg = ~pos
        m_pos = ms[pos]  # 0 to n-1
        p_neg = -ms[neg]  # 1 to n-1
        
        # For m >= 0: (1-h[i-m])*[i>=m] - h[i+m]*[i+m<n]
        valid1 = i >= m_pos
        valid2 = (i + m_pos) < n
        ccoef[pos] = np.where(valid1, 1.0 - h[np.clip(i - m_pos, 0, n-1)], 0.0)
        ccoef[pos] -= np.where(valid2, h[np.clip(i + m_pos, 0, n-1)], 0.0)
        
        # For m < 0, p = -m: (1-h[i+p])*[i<n-p] - h[i-p]*[i>=p]
        valid3 = i < (n - p_neg)
        valid4 = i >= p_neg
        ccoef[neg] = np.where(valid3, 1.0 - h[np.clip(i + p_neg, 0, n-1)], 0.0)
        ccoef[neg] -= np.where(valid4, h[np.clip(i - p_neg, 0, n-1)], 0.0)
        
        return ccoef
    
    D = coeff(j) - coeff(l)
    
    # Feasible δ range
    dmin = max(-h[j], h[l] - 1.0)
    dmax = min(1.0 - h[j], h[l])
    if dmin >= dmax - 1e-14:
        return 0.0
    
    # f(δ) = max(c + δ*D), convex piecewise linear
    # Find minimum
    def fval(d):
        return np.max(c + d * D)
    
    # Slope (right derivative) at point d
    def fderiv(d):
        vals = c + d * D
        mx = vals.max()
        active = vals >= mx - 1e-14
        if not active.any():
            return 0.0
        return D[active].max()
    
    if fderiv(dmin) >= -1e-14:
        best = dmin
    elif fderiv(dmax) <= 1e-14:
        best = dmax
    else:
        # Binary search for zero crossing
        lo, hi = dmin, dmax
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if fderiv(mid) > 0:
                hi = mid
            else:
                lo = mid
        best = 0.5 * (lo + hi)
    
    # Fine-tune around best
    spread = (dmax - dmin) * 0.01
    cands = np.concatenate([[dmin, dmax], np.linspace(best-spread, best+spread, 15)])
    cands = np.clip(cands, dmin, dmax)
    best = cands[np.argmin([fval(cc) for cc in cands])]
    
    return best

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _coeff_diff(h, j, l):
    n = len(h)
    ks = np.arange(2 * n - 1)
    ms = ks - (n - 1)
    def ccoef(i):
        out = np.zeros(2 * n - 1)
        pos = ms >= 0
        neg = ~pos
        mv = ms[pos]
        if np.any(pos):
            t1 = np.where(i >= mv, 1.0 - h[np.clip(i - mv, 0, n - 1)], 0.0)
            t2 = np.where((i + mv) < n, h[np.clip(i + mv, 0, n - 1)], 0.0)
            out[pos] = t1 - t2
        pv = -ms[neg]
        if np.any(neg):
            t3 = np.where((i + pv) < n, 1.0 - h[np.clip(i + pv, 0, n - 1)], 0.0)
            t4 = np.where(i >= pv, h[np.clip(i - pv, 0, n - 1)], 0.0)
            out[neg] = t3 - t4
        return out
    return ccoef(j) - ccoef(l)

def _pairwise_opt(c, D, hj, hl):
    dmin = max(-hj, hl - 1.0)
    dmax = min(1.0 - hj, hl)
    if dmin >= dmax - 1e-14:
        return 0.0
    def fval(d):
        return np.max(c + d * D)
    def fderiv(d):
        vals = c + d * D
        mx = vals.max()
        active = vals >= mx - 1e-13
        return D[active].max()
    if fderiv(dmin) >= -1e-13:
        return dmin
    if fderiv(dmax) <= 1e-13:
        return dmax
    lo, hi = dmin, dmax
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if fderiv(mid) > 0:
            hi = mid
        else:
            lo = mid
    best = 0.5 * (lo + hi)
    spread = (dmax - dmin) * 0.05
    cands = np.unique(np.concatenate([[dmin, dmax, lo, hi, best],
                                       np.linspace(max(dmin, best - spread), min(dmax, best + spread), 9)]))
    cands = np.clip(cands, dmin, dmax)
    best = cands[np.argmin([fval(x) for x in cands])]
    return best

def _pairwise_cd(h, n, rng, max_time, eval_every=200):
    c5c = _eval(h)
    c = _corrs(h)
    t_start = time.time()
    it = 0
    stall = 0
    while time.time() - t_start < max_time:
        if it % eval_every == 0:
            c = _corrs(h)
            c5_now = _eval(h)
            if c5_now < c5c:
                c5c = c5_now
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    break
        j = rng.randint(n)
        l = rng.randint(n)
        if j == l:
            it += 1
            continue
        D = _coeff_diff(h, j, l)
        delta = _pairwise_opt(c, D, h[j], h[l])
        if abs(delta) > 1e-10:
            h[j] += delta
            h[l] -= delta
            h = _repair(h, n)
            c = _corrs(h)
        it += 1
    h = _repair(h, n)
    c5c = _eval(h)
    return h, c5c

def _slsqp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _slsqp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _grad_topk(h, c, topk, dx):
    n = len(h)
    top_idx = np.argsort(c)[-topk:]
    g = np.zeros(n)
    for k in top_idx:
        D_j_all = _coeff_diff(h, 0, 0)  # dummy to get structure
        # Just compute gradient for lag k directly
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[:n - sh] -= h[sh:]
        else:
            sp = -sh
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    g *= dx
    return g

def _momentum_descent(h, nc, max_rounds, rng):
    c5c = _eval(h)
    dx = 2.0 / nc
    v = np.zeros(nc)
    mu = 0.7
    step = 0.01
    c = _corrs(h)
    for r in range(max_rounds):
        c = _corrs(h)
        topk = min(3, 2 * nc - 1)
        g = _grad_topk(h, c, topk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        v = mu * v - step * g
        h_new = _repair(h + v, nc)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-15:
            h = h_new
            c5c = c5n
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.7
            v *= 0.5
            if step < 1e-6:
                break
    return h, c5c
