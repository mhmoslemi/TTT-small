import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _grad_corr(w, x):
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        s = abs(k - (n - 1))
        g[:n - s] += wk * (1.0 - x[s:])
        g[s:] += wk * (-x[:n - s])
    return g

def _pnorm_opt(h, p, t_end, n_iter, lr0):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    cur = _eval(h)
    lr = lr0
    mu = 0.82
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        S = float(np.sum(np.abs(c) ** p))
        if S < 1e-30:
            break
        denom = S ** ((p - 1.0) / p)
        if denom < 1e-30:
            break
        w = np.abs(c) ** (p - 1) * np.sign(c) / denom
        g = _grad_corr(w, h)
        v = mu * v + (1.0 - mu) * g
        best_h, best_c = h, cur
        for mult in [1.0, 0.5, 0.25, 0.125]:
            ht = _repair(h - lr * mult * v, n)
            ct = _eval(ht)
            if ct < best_c:
                best_h, best_c = ht, ct
        if best_c < cur - 1e-15:
            h, cur = best_h, best_c
            lr = min(lr * 1.005, 0.4)
        else:
            lr *= 0.5
    return h, cur

def _topk_descent(h, t_end, n_iter, top_k, sharpness):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        idx = np.argsort(c)[-top_k:]
        w = np.zeros(2 * n - 1)
        vals = c[idx]
        vmin, vmax = vals.min(), vals.max()
        if vmax - vmin > 1e-12:
            w[idx] = np.exp(sharpness * (vals - vmin) / (vmax - vmin))
        else:
            w[idx] = 1.0
        w[idx] /= w[idx].sum()
        g = _grad_corr(w, h)
        best_h, best_c = h, cur
        for step in [0.04, 0.02, 0.01, 0.004, 0.001]:
            ht = _repair(h - step * g, n)
            ct = _eval(ht)
            if ct < best_c:
                best_h, best_c = ht, ct
        if best_c < cur - 1e-15:
            h, cur = best_h, best_c
        else:
            break
    return h, cur

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _binary_search(h, n, t_end, max_rounds, rng):
    h = h.copy()
    h[h > 0.5] = 1.0; h[h <= 0.5] = 0.0
    h = _repair(h, n)
    cur = _eval(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        n_trials = min(3000, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            c2 = _eval(h)
            if c2 < cur - 1e-14:
                cur = c2; improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                break
            h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, cur

def _slsqp_refine(h, alpha, maxiter, ftol):
    n = len(h)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        dx = 2.0 / n
        return _grad_corr(w * dx, x)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, args=(), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: p-norm optimization (different landscape from log-sum-exp)
    for p in [4, 8, 16, 32, 64]:
        if rem() < 120:
            break
        hp = _pnorm_opt(best_h, p, t0 + rem() - 80, 2500, 0.015)
        c5 = _eval(hp)
        if c5 < best_c5:
            best_c5 = c5; best_h = hp.copy()

    # Phase 2: Top-k steepest descent
    for tk in [1, 2, 3]:
        if rem() < 80:
            break
        hs, c5 = _topk_descent(best_h, t0 + rem() - 40, 3000, tk, 6.0)
        if c5 < best_c5:
            best_c5 = c5; best_h = hs.copy()

    # Phase 3: SLSQP with high alpha (escape via different optimizer)
    for alpha in [200000, 1000000]:
        if rem() < 100:
            break
        hq = _slsqp_refine(best_h, alpha, 3000, 1e-20)
        c5 = _eval(hq)
        if c5 < best_c5:
            best_c5 = c5; best_h = hq.copy()

    # Phase 4: Multi-resolution (try different n)
    for n_try in [200, 300, 400, best_n, 600]:
        if rem() < 150:
            break
        if n_try < 50 or n_try > 1000:
            continue
        hr = _repair(_resample(best_h, n_try), n_try)
        for p in [8, 32, 128]:
            if rem() < 130:

        hr, _ = _topk_descent(hr, t0 + rem() - 30, 1500, 2, 8.0)
        c5 = _eval(hr)
        if c5 < best_c5:

        if nb < 40:

        hb, c5 = _binary_search(hb, nb, t0 + rem() - 20, 300, rng)
        if c5 < best_c5:

        if rem() < 40:
            break
        h6, _ = _pnorm_opt(best_h, 64, t0 + rem() - 20, 800, 0.005)
        c5 = _eval(h6)
