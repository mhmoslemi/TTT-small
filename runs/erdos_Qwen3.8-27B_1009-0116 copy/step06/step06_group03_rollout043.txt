            if nt < 100:

            h_half, c_half = _momentum_opt(h_half, n_half, nt, tk, st, 0.90, t4 - 20, rng)
        # Upsample and refine
        h_up = np.interp(np.linspace(0, n0-1, n0), np.linspace(0, n0-1, n_half), h_half)
        h_up = _repair(h_up, n0)
        h_up, c_up = _momentum_opt(h_up, n0, 20000, 5, 0.004, 0.90, t4 - 5, rng)
        if c_up < best_c:
            best_c = c_up; best_h = h_up.copy(); best_n = n0

    # Phase 5: Final SLSQP + high-alpha polish
    if time.time() < t5 - 30:
        for alpha in [100000, 1000000]:
            if time.time() > t5 - 15:
                break
            hc = _slsqp(best_h, alpha, 800, 1e-16)
            c = _eval(hc)
            if c < best_c:
                best_c = c; best_h = hc.copy(); best_n = len(hc)

    # Phase 6: Coordinate refinement
    if time.time() < t5 - 2:
        nt = min(int((t5 - time.time() - 2) / 0.004), 60000)
        best_h, best_c = _coord_refine(best_h, nt, rng)
        best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
