import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if 0 <= sh < n:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    elif -n < sh < 0:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _smooth_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return (m + np.log(np.exp(s).sum()) / alpha) * dx

def _smooth_grad(x, alpha, top_k=0):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -500, 500)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            g += w[ii] * _grad_lag(x, idx[ii], dx)
    else:
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-13:
                continue
            g += w[k] * _grad_lag(x, k, dx)
    return g

def _lbfgsb(x0, alpha, top_k, maxiter):
    n = len(x0)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_obj(x, alpha)
    def grad(x):
        return _smooth_grad(x, alpha, top_k)
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return _repair(res.x, n)
    except Exception:
        return _repair(x0, n)
