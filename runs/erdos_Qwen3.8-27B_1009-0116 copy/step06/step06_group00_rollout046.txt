def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                diff -= a if diff > 0 else -a
                if abs(diff) < 1e-12:
                    break
    return h

        else:

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                diff -= a if diff > 0 else -a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correlate_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_for_shifts(h, shifts, dx):
    n = len(h)
    g = np.zeros(n)
    for k in shifts:
        sh = (n - 1) - k
        wk = dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g / len(shifts)

def _topk_descent(h, n_top, max_rounds, t0, budget_s):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    for _ in range(max_rounds):
        if time.time() - t0 > budget_s - 15:
            break
        c = _correlate_vec(h)
        top_k = np.argsort(c)[-n_top:][::-1]
        g = _grad_for_shifts(h, top_k, dx)
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _slsqp_fn(x, alpha, n, dx):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha + 1e-6 * (x.sum() - n * 0.5) ** 2

def _slsqp_jac(x, alpha, n, dx):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    g += 2e-6 * (x.sum() - n * 0.5)
    return g

def _slsqp_run(h, alpha, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_fn, h, jac=_slsqp_jac,
                       args=(alpha, n, dx), method='SLSQP', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _swap_sa(h, n_iter, rng, t0, budget_s):
    n = len(h)
    c5_cur = _eval(h)
    best = h.copy()
    best_c = c5_cur
    T0, T1 = 5e-4, 5e-8
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 15:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = _eval(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-16)):
            c5_cur = c5_new
            if c5_cur < best_c:
                best_c = c5_cur
                best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_c

def _ga_coarse(n, pop_size, n_gen, seed, t0, budget_s):
    rng = np.random.RandomState(seed)
    best = np.full(n, 0.5)
    best_c = _eval(best)
    pop = []
    c5s = []
    pop.append(best.copy())
    c5s.append(best_c)
    for _ in range(pop_size - 1):
        x = np.clip(0.5 + 0.3 * rng.randn(n), 0, 1)
        x = _proj_to_sum(x, n * 0.5)
        pop.append(x)
        c5s.append(_eval(x))
    for gen in range(n_gen):
        if time.time() - t0 > budget_s - 20:
            break
        order = np.argsort(c5s)
        new_pop = [pop[order[i]].copy() for i in range(min(5, pop_size))]
        new_c5 = [c5s[order[i]] for i in range(min(5, pop_size))]
        while len(new_pop) < pop_size:
            ia = rng.randint(pop_size, 3)
            ib = rng.randint(pop_size, 3)
            ia = ia[np.argmin([c5s[i] for i in ia])]
            ib = ib[np.argmin([c5s[i] for i in ib])]
            pa, pb = pop[ia], pop[ib]
            if rng.random() < 0.7:
                ct = rng.randint(1, n - 1)
                child = np.concatenate([pa[:ct], pb[ct:]])
            else:
                child = 0.5 * pa + 0.5 * pb
            if rng.random() < 0.4:
                n_mut = max(1, int(0.1 * n))
                idx = rng.choice(n, n_mut, replace=False)
                child[idx] = np.clip(child[idx] + 0.08 * rng.randn(n_mut), 0, 1)
            child = _proj_to_sum(child, n * 0.5)
            new_pop.append(child)
            new_c5.append(_eval(child))
        pop = new_pop
        c5s = new_c5
        if c5s[0] < best_c:
            best_c = c5s[0]
            best = pop[0].copy()
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Coarse GA for diverse basin discovery
    n_coarse = max(56, n0 // 10)
    f = n0 // n_coarse
    h_c = _repair(h[:f * n_coarse].reshape(n_coarse, f).mean(axis=1), n_coarse)
    h_ga, c5_ga = _ga_coarse(n_coarse, 30, 40, seed + 1, t0, 80)
    h_up = _repair(np.repeat(h_ga, f), n0)
    c5_up = _eval(h_up)
    if c5_up < best_c5:
        best_c5 = c5_up
        best_h = h_up.copy()

    # Phase 2: SLSQP refinement
    for alpha in [5000, 50000, 200000, 800000]:
        if time.time() - t0 > 150:
            break
        h_s = _slsqp_run(best_h, alpha, 1200, 1e-16)
        c5v = _eval(h_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_s.copy()

    # Phase 3: Top-k targeted descent (multiple k values)
    for nk in [2, 3, 5, 8, 12, 20]:
        if time.time() - t0 > 350:
            break
        h_t, c5_t = _topk_descent(best_h.copy(), nk, 600, t0, 350 + (budget_s - 350) * nk / 20)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 4: Perturbed restarts
    for ri, ns in enumerate([0.03, 0.08, 0.15, 0.25]):
        if time.time() - t0 > 500:
            break
        h_p = _proj_to_sum(best_h + ns * rng.randn(n0), n0 * 0.5)
        h_p = _slsqp_run(h_p, 50000, 600, 1e-16)
        c5v = _eval(h_p)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_p.copy()
        h_t2, c5_t2 = _topk_descent(h_p, 3, 300, t0, 500)
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()

    # Phase 5: Simulated annealing with swaps
    if time.time() - t0 < budget_s - 150:
        n_sa = min(int((budget_s - 150 - (time.time() - t0)) / 0.004), 40000)
        best_h, best_c5 = _swap_sa(best_h, n_sa, rng, t0, budget_s - 150)
        best_n = len(best_h)

    # Phase 6: Final SLSQP + topk
    if time.time() - t0 < budget_s - 60:
        h_f = _slsqp_run(best_h, 500000, 2000, 1e-17)
        c5f = _eval(h_f)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = h_f.copy()
        if time.time() - t0 < budget_s - 30:
            h_t3, c5_t3 = _topk_descent(best_h, 2, 400, t0, budget_s - 30)
            if c5_t3 < best_c5:
                best_c5 = c5_t3
                best_h = h_t3.copy()

    # Phase 7: Fine 2-point polish
    if time.time() - t0 < budget_s - 5:
        h = best_h.copy()
        n = len(h)
        c5c = _eval(h)
        n_pol = min(int((budget_s - 5 - (time.time() - t0)) / 0.003), 30000)
        for _ in range(n_pol):
            if time.time() - t0 > budget_s - 3:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
            d = 5e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_h = h.copy()
                        best_n = n
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
