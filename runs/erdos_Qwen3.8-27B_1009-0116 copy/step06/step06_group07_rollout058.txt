        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha

        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())

            if w[k] < 1e-14:
                continue
            sh = k - (n - 1)  # shift

        return x.sum() - n * 0.5

        return np.ones(n)

                       bounds=bnds,

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval_full(h):
    """Returns (c5, corr_vector, argmax_index)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    idx = corr.argmax()
    return m * 2.0 / n, corr, idx

def _binary_init(h, n, seed=0):
    """Convert to binary and fix count."""
    rng = np.random.RandomState(seed)
    h = (h >= 0.5).astype(float)
    target = n // 2
    s = int(h.sum())
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if s > target:
        idx = rng.choice(ones, size=s-target, replace=False)
        h[idx] = 0
    elif s < target:
        idx = rng.choice(zeros, size=target-s, replace=False)
        h[idx] = 1
    return h

def _swap_local_search(h, n, max_rounds=100, seed=0, verbose=False):
    """Efficient binary pair-swap local search."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5, corr, max_idx = _eval_full(h)
    
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Try random swaps
        n_tries = min(n * 3, 2000)
        for _ in range(n_tries):
            a = ones[rng.randint(n_ones)]
            b = zeros[rng.randint(n_zeros)]
            h[a] = 0
            h[b] = 1
            c5n = _eval(h)
            if c5n < c5 - 1e-14:
                c5 = c5n
                improved = True
            else:
                h[a] = 1
                h[b] = 0
        if not improved:
            break
    return h, c5

def _slsqp_opt(h0, alpha, maxiter=2000, ftol=1e-15):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2*n - 1):
            if w[k] < 1e-14:
                continue
            sh = k - (n - 1)
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += w[k] * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= w[k] * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += w[k] * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= w[k] * x[i2 + sp]
        return g * (2.0 / n)
    
    def sum_eq(x):
        return x.sum() - n * 0.5
    def sum_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)
