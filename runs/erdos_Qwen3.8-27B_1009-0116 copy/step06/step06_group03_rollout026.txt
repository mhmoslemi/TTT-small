    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _get_c(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_smax(h, alpha):
    n = len(h)
    c = _get_c(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(w); w[np.argmax(c)] = 1.0
    else:
        w /= wsum
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:

        if lag >= 0:
            if lag == 0:
                g[:] += wk * (1.0 - h)
                g[:] -= wk * h

            best_h, best_c = h_new, c_new

    # Phase 7: Fine coordinate search
    if _el() < budget_s - 3:
        n_tr = min(int((budget_s - 3 - _el()) / 0.0015), 80000)
        best_h, best_c = _finetune(best_h, n_tr, rng, 5e-5)

    best_h = _proj(best_h, len(best_h) / 2.0)
    best_c = _eval(best_h)
    return (best_h, best_c, len(best_h))
