import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _coord_descend(h, time_limit):
    """Coordinate descent: for each coordinate, find optimal value via 1D search."""
    n = len(h)
    h = h.copy()
    best_c = _eval(h)
    dx = 2.0 / n
    t_start = time.time()
    
    # Precompute full correlation
    c_full = np.correlate(h, 1.0 - h, mode='full') * dx
    
    sweep = 0
    while time.time() - t_start < time_limit:
        improved = False
        for i in range(n):
            if time.time() - t_start > time_limit:
                    break
            cur = h[i]
            # Compute a_k = dc_k/dh[i] and b_k = c_k - a_k * h[i]
            # a_k: derivative of c_k w.r.t. h[i]
            # c_k = sum_m h[m]*(1-h[m+shift]) * dx where shift = k-(n-1)
            # h[i] appears when m=i: contributes (1-h[i+shift])*dx if valid
            # h[i] appears when m+shift=i, i.e., m=i-shift: contributes -h[i-shift]*dx if valid
            
            shifts = np.arange(2*n-1) - (n-1)  # ranges from -(n-1) to n-1
            # a_k = dx * [(1-h[i+shifts]) * valid1 - h[i-shifts] * valid2]
            idx1 = i + shifts
            idx2 = i - shifts
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            a = np.zeros(2*n-1)
            a[valid1] += dx * (1.0 - h[idx1[valid1]])
            a[valid2] -= dx * h[idx2[valid2]]
            
            b = c_full - a * cur
            
            # Golden section search on [0,1] to minimize max_k(a_k*x + b_k)
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            lo, hi = 0.0, 1.0
            x1 = lo + gr*(hi - lo)

        for i in range(n):
            if time.time() - t_start > time_limit:
                return h, best_c
            cur = h[i]
            idx1 = i + shifts
            idx2 = i - shifts
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            a = np.zeros(2 * n - 1)
            a[v1] += dx * (1.0 - h[idx1[v1]])
            a[v2] -= dx * h[idx2[v2]]
            b = c_full - a * cur
            lo, hi = 0.0, 1.0
            x1 = lo + gr * (hi - lo)
            x2 = hi - gr * (hi - lo)
            f1 = np.max(a * x1 + b)
            f2 = np.max(a * x2 + b)
            for _ in range(35):
                if f1 < f2:
                    hi = x2; x2 = x1; f2 = f1
                    x1 = lo + gr * (hi - lo)
                    f1 = np.max(a * x1 + b)
                else:
                    lo = x1; x1 = x2; f1 = f2
                    x2 = hi - gr * (hi - lo)
                    f2 = np.max(a * x2 + b)
            x_opt = 0.5 * (lo + hi)
            delta = x_opt - cur
            if abs(delta) > 1e-10:
                h[i] = x_opt
                c_full += a * delta
                sweep_improved = True
        c_new = _eval(h)
        if c_new < best_c - 1e-14:
            best_c = c_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3 and not sweep_improved:
                break
    return h, best_c

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _subgrad_top(h, n_iter, step0, rng):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    stuck = 0
    for _ in range(n_iter):
        c = np.correlate(best, 1.0 - best, mode='full')
        top_k = np.argsort(c)[-3:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += (1.0 - best[sh:])
                if sh > 0:
                    g[sh:] -= best[:n-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - best[:n-sp])
                g[:n-sp] -= best[sp:]
        g *= 2.0 / n
        v = mu * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best = h_new; best_c = c_new
                step = min(step * 1.15, 0.5)
                improved = True; break
        if not improved:
            step *= 0.6; stuck += 1
            if stuck > 80:
                best = _repair(best + 0.005 * rng.randn(n), n)
                best_c = _eval(best); step = step0
                stuck = 0; v = np.zeros(n)
    return best, best_c

def _fine_search(h, n_tries, rng):
    n = len(h); h = h.copy()
    c_cur = _eval(h); best = h.copy(); best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = 5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c = _eval(best_h); best_n = n0

    def _el():
        return time.time() - t0

    # Phase 1: SLSQP warm-up
    for alpha, iters in [(500, 600), (2000, 800), (10000, 1200), (50000, 1500)]:
        if _el() > 180:
            break
        hc = _slsqp(best_h, alpha, iters, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 2: Coordinate descent (exact 1D optimal per coordinate)
    if _el() < budget_s - 420:
        best_h, best_c = _coord_descend(best_h, budget_s - 420 - _el())

    # Phase 3: SLSQP refinement
    for alpha, iters in [(100000, 1500), (500000, 2000)]:
        if _el() > budget_s - 280:
            break
        hc = _slsqp(best_h, alpha, iters, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 4: Coordinate descent again
    if _el() < budget_s - 220:
        best_h, best_c = _coord_descend(best_h, budget_s - 220 - _el())

    # Phase 5: Subgradient escape
    if _el() < budget_s - 160:
        n_it = min(1500, int((budget_s - 160 - _el()) / 0.02))
        hc, c = _subgrad_top(best_h, n_it, 0.01, rng)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 6: High-precision SLSQP
    if _el() < budget_s - 80:
        hc = _slsqp(best_h, 1000000, 3000, 1e-18)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 7: Final coordinate descent
    if _el() < budget_s - 40:
        best_h, best_c = _coord_descend(best_h, budget_s - 40 - _el())

    # Phase 8: Fine pair-exchange tuning
    if _el() < budget_s - 5:
        n_tr = min(int((budget_s - 5 - _el()) / 0.003), 30000)
        best_h, best_c = _fine_search(best_h, n_tr, rng)

    best_h = _repair(best_h, len(best_h))
    best_c = _eval(best_h)
    return (best_h, best_c, len(best_h))
