import time

    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

        if abs(diff) > 1e-12:

            for idx in order:

                    h[idx] += a
                    diff -= a

                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:

        ct = c

        else:
            sp = -sh

            g[:n - sp] -= wk * h[sp:]

            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _grad_topk(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.002 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 4)) % nc
        d = 2e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _3pt_search(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + rng.randint(1, 3)) % nc
        k = (i + rng.randint(4, 8)) % nc
        d1 = 1.5e-6 * rng.randn()
        d2 = 1.5e-6 * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0<=ni<=1 and 0<=nj<=1 and 0<=nk<=1:
            h[i],h[j],h[k] = ni,nj,nk
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i],h[j],h[k] = ni-d1, nj-d2, nk+d1+d2
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 12

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP refinement with increasing alpha
    dl1 = min(deadline, t0 + 200)
    for alpha, tk, mi in [(5000, 30, 300), (20000, 20, 300), (80000, 12, 250)]:
        if time.time() > dl1:
            break
        h = _slsqp(h, alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Nesterov with sweeping alpha and topk
    dl2 = min(deadline, t0 + 500)
    for alpha, tk, st, ns in [(20000, 25, 0.004, 2000), (80000, 15, 0.002, 2000),
                               (400000, 10, 0.001, 2000), (2000000, 8, 0.0005, 2000)]:
        if time.time() > dl2:
            break
        h = _nesterov(best_h, alpha, tk, ns, st, dl2)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Block simulated annealing
    dl3 = min(deadline, t0 + 750)
    blocks = [1, 2, 3, 4, 5, 7]
    h_a, c5_a = _block_anneal(best_h.copy(), 40000, rng, dl3, 0.003, 1e-9, blocks)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()

    # Phase 4: Second round Nesterov (very high alpha, few top lags)
    dl4 = min(deadline, t0 + 920)
    for alpha, tk, st, ns in [(5000000, 6, 0.0003, 3000), (20000000, 4, 0.00015, 3000)]:
        if time.time() > dl4:
            break
        h = _nesterov(best_h, alpha, tk, ns, st, dl4)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Fine 2-point polish
    dl5 = deadline
    remaining = max(0, dl5 - time.time())
    n_pol = min(int(remaining / 0.0012), 30000)
    best_h, best_c5 = _fine_polish(best_h, n_pol, rng, dl5)

    # Phase 6: 3-point search
    if time.time() < t0 + budget_s - 3:
        rem = max(0, t0 + budget_s - 3 - time.time())
        n_3 = min(int(rem / 0.0018), 15000)
        best_h, best_c5 = _3pt_search(best_h, n_3, rng, t0 + budget_s - 3)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
