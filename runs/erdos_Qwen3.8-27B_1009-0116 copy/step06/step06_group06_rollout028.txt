def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        else:

            g[ll:] += (1.0 - h[:n-ll])

        top_k = np.argsort(c)[-K_top:]

        b_ub = np.zeros(K_top)

            A_ub[j, n] = -1.0
            b_ub[j] = gk @ h_cur - c[kk]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        c_obj[n] = 1.0  # minimize t

import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _delta_c(h, i, delta, n):
    """Compute change in correlation vector when h[i] changes by delta"""
    dc = np.zeros(2 * n - 1)
    # h[i] appears as h[m] with m=i: c[n-1+l] += delta*(1-h[i+l]) for -i<=l<=n-1-i
    # h[i] appears as 1-h[j] with j=i: c[n-1+l] -= delta*h[i-l] for i-(n-1)<=l<=i
    for l in range(-i, n - i):
        k = n - 1 + l
        if 0 <= k <= 2 * n - 2:
            dc[k] += delta * (1.0 - h[i + l])
    for l in range(i - n + 1, i + 1):
        k = n - 1 + l
        if 0 <= k <= 2 * n - 2:
            dc[k] -= delta * h[i - l]
    return dc

def _grad_c(h, l, n):
    g = np.zeros(n)
    if l >= 0:
        if l < n:
            g[:n - l] += (1.0 - h[l:])
            g[l:] -= h[:n - l]
    else:
        ll = -l
        if ll < n:
            g[ll:] += (1.0 - h[:n - ll])
            g[:n - ll] -= h[ll:]
    return g

def _proj_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slp_step(h, K, t_end):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    
    for it in range(500):
        if time.time() > t_end:
            break
        c = _corr_full(h_cur)
        top_k = np.argsort(c)[-K:]
        
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)

import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_s(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_c(h, l, n):
    g = np.zeros(n)
    if l >= 0:
        if l < n:
            g[:n - l] += (1.0 - h[l:])
            g[l:] -= h[:n - l]
    else:
        ll = -l
        if ll < n:
            g[ll:] += (1.0 - h[:n - ll])
            g[:n - ll] -= h[ll:]
    return g

def _proj_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slp_optimize(h, t_end, K, max_iters=600):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for _it in range(max_iters):
        if time.time() > t_end:
            break
        c = _corr_s(h_cur)
        top_k = np.argsort(c)[-K:]
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)
        for j, kk in enumerate(top_k):
            l = kk - (n - 1)
            gk = _grad_c(h_cur, l, n) * dx
            A_ub[j, :n] = gk
            A_ub[j, n] = -1.0
            b_ub[j] = gk @ h_cur - c[kk]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n * 0.5])
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if not res.success:
                break
            h_lp = res.x[:n]
        except Exception:
            break
        improved = False
        for damp in [0.6, 0.4, 0.25, 0.1, 0.05]:
            h_new = (1.0 - damp) * h_cur + damp * h_lp
            h_new = _proj_box_sum(h_new, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                h_cur = h_new
                c5_cur = c5_new
                improved = True
                break
        if not improved:
            break
    return h_cur, c5_cur

def _fast_coord(h, n_rounds, t_end):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for rnd in range(n_rounds):
        if time.time() > t_end:
            break
        c_base = _corr_s(h_cur)
        max_idx = np.argmax(c_base)
        improved = False
        order = np.argsort(-c_base.conj().real) if hasattr(c_base, 'conj') else np.random.permutation(n)
        for i in range(n):
            if time.time() > t_end:
                break
            old_val = h_cur[i]
            best_val = old_val
            best_c5 = c5_cur
            for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.001, -0.001]:
                nv = old_val + delta
                if nv < 0 or nv > 1:
                    continue
                h_cur[i] = nv
                dc = np.zeros(2 * n - 1)
                for l in range(-i, n - i):
                    k = n - 1 + l
                    if 0 <= k < 2 * n - 1:
                        dc[k] += delta * (1.0 - h_cur[i + l]) * (2.0 / n)
                for l in range(i - n + 1, i + 1):
                    k = n - 1 + l
                    if 0 <= k < 2 * n - 1:
                        dc[k] -= delta * h_cur[i - l] * (2.0 / n)
                c_new = c_base + dc
                mv = float(np.max(c_new))
                if mv < best_c5:
                    best_c5 = mv
                    best_val = nv
            h_cur[i] = best_val
            if best_val != old_val:
                improved = True
                c_base = _corr_s(h_cur)
                c5_cur = best_c5
    h_cur = _repair(h_cur, n)
    return h_cur, _eval(h_cur)

def _multi_grad_descent(h, n_steps, t_end, K=8):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()

            break

        top_k = np.argsort(c)[-K:]
        c_max = c.max()
        w = np.exp((c[top_k] - c_max) * 500)
        w /= w.sum()
        g = np.zeros(n)
        for j, kk in enumerate(top_k):
            l = kk - (n - 1)
            g += w[j] * _grad_c(h_cur, l, n)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = _proj_box_sum(h_cur - step * g / gn * gn, n * 0.5)
        h_new = _proj_box_sum(h_cur - step * g, n * 0.5)
        c5n = _eval(h_new)
        if c5n < c5_cur - 1e-16:
            h_cur = h_new
            c5_cur = c5n
            step = min(step * 1.1, 0.15)
        else:
            step *= 0.5
            if step < 1e-5:
                break
    return h_cur, c5_cur
