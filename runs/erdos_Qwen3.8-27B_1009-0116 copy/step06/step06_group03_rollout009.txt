def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return np.array([x.sum() - 0.5 * n])
    def _cj(x):
        return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _momentum_grad(h, n_iter, step0, momentum, top_k):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    step = step0
    stuck = 0
    
    for it in range(n_iter):
        c = np.correlate(best, 1.0 - best, mode='full')
        # Get top-k indices
        idx = np.argsort(c)[-top_k:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
        # Normalize
        w /= w.sum()
        
        # Compute gradient
        g = np.zeros(n)
        dx = 2.0 / n
        for k in idx:
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - best[sh:])
                if sh > 0:
                    g[sh:] -= wk * best[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - best[:n - sp])
                g[:n - sp] -= wk * best[sp:]
        
        # Nesterov momentum
        v = momentum * v + g
        h_new = _proj(best - step * v, 0.5 * n)
        c_new = _eval(h_new)
        
        if c_new < best_c - 1e-15:
            best = h_new
            best_c = c_new
            step = min(step * 1.05, 0.3)
            stuck = 0
        else:
            # Try smaller step
            found = False
            for f in [0.5, 0.25, 0.1, 0.05]:
                h_try = _proj(best - step * f * v, 0.5 * n)
                c_try = _eval(h_try)
                if c_try < best_c - 1e-15:
                    best = h_try
                    best_c = c_try
                    found = True
                    break
            if not found:
                step *= 0.7
                stuck += 1
                if stuck > 50:
                    best = _repair(best + 0.005 * np.random.randn(n), n)
                    best_c = _eval(best)
                    v = np.zeros(n)
                    step = step0
                    stuck = 0
    
    return best, best_c

def _fine_search(h, n_tries, rng):

    best = h.copy()
    best_c = c_cur

        cur_step = step * max(0.1, 1.0 - i / n_tries)
        
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue

        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[a], h[b] = na - d, nb + d
    
    return best, best_c

            # - Terms where j+sh=i, i.e., j=i-sh: contributes h[i-sh]*(1-h[i]) if i-sh valid

            #     - j+sh_k=i, i.e., j=i-sh_k: h[i-sh_k]*(1-h[i]) (if 0≤i-sh_k<n)

                    a = min(-diff, h[j])
                    h[j] -= a

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return np.array([x.sum() - 0.5 * n])
    def _cj(x):
        return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _momentum_grad(h, n_iter, step0, mom, top_k, rng, t_end):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    step = step0
    stuck = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(best, 1.0 - best, mode='full')
        idx = np.argsort(c)[-top_k:]
        weights = c[idx] - c[idx].min() + 1e-10
        weights /= weights.sum()
        g = np.zeros(n)
        dx = 2.0 / n
        for ii in range(len(idx)):
            k = idx[ii]
            sh = (n - 1) - k
            wk = weights[ii] * dx
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - best[sh:])
                if sh > 0:
                    g[sh:] -= wk * best[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - best[:n - sp])
                g[:n - sp] -= wk * best[sp:]
        v = mom * v + g
        h_new = _proj(best - step * v, 0.5 * n)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-15:
            best = h_new; best_c = c_new
            step = min(step * 1.03, 0.25)
            stuck = 0
        else:
            found = False
            for f in [0.5, 0.25, 0.1, 0.05]:
                h_try = _proj(best - step * f * v, 0.5 * n)
                ct = _eval(h_try)
                if ct < best_c - 1e-15:
                    best = h_try; best_c = ct; found = True; break
            if not found:
                step *= 0.65
                stuck += 1
                if stuck > 60:
                    best = _repair(best + 0.008 * rng.randn(n), n)
                    best_c = _eval(best); v = np.zeros(n)
                    step = step0; stuck = 0
    return best, best_c

def _pair_coord(h, n_pairs, rng, t_end):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy(); best_c = c_cur
    c = np.correlate(h, 1.0 - h, mode='full')
    sh_arr = np.arange(2 * n - 1) - (n - 1)
    for it in range(n_pairs):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        coeff_i = np.zeros(2 * n - 1)
        m1 = (sh_arr >= -i) & (sh_arr < n - i)
        m2 = (sh_arr > i - n) & (sh_arr <= i)
        if np.any(m1):
            coeff_i[m1] += (1.0 - h[i + sh_arr[m1]])
        if np.any(m2):
            coeff_i[m2] -= h[i - sh_arr[m2]]
        coeff_j = np.zeros(2 * n - 1)
        m1j = (sh_arr >= -j) & (sh_arr < n - j)
        m2j = (sh_arr > j - n) & (sh_arr <= j)
        if np.any(m1j):
            coeff_j[m1j] += (1.0 - h[j + sh_arr[m1j]])
        if np.any(m2j):
            coeff_j[m2j] -= h[j - sh_arr[m2j]]
        dc = coeff_i - coeff_j
        base = c
        t_lo, t_hi = -0.5, 0.5
        for _ in range(30):
            t1 = t_lo + (t_hi - t_lo) / 3
            t2 = t_hi - (t_hi - t_lo) / 3
            if np.max(base + t1 * dc) < np.max(base + t2 * dc):
                t_hi = t2
            else:
                t_lo = t1
        d_opt = 0.5 * (t_lo + t_hi)
        ni, nj = h[i] + d_opt, h[j] - d_opt
        if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
            continue
        h[i] = np.clip(ni, 0, 1)
        h[j] = np.clip(nj, 0, 1)
        actual_d = (ni - h[i]) - (nj - h[j])
        actual_d = (h[i] - (ni - d_opt))
        c_new_arr = c + (h[i] - (ni - d_opt)) * coeff_i - (h[j] - (nj + d_opt)) * coeff_j
        c_new = float(np.max(c_new_arr) * 2.0 / n)
        if c_new < c_cur - 1e-15:
            c_cur = c_new
            c = c_new_arr
            if c_new < best_c:
                best_c = c_new; best = h.copy()
        else:
            h[i] = ni - d_opt; h[j] = nj + d_opt
            h[i] = np.clip(h[i], 0, 1); h[j] = np.clip(h[j], 0, 1)
            c = c_new_arr - d_opt * dc + d_opt * dc
    return _repair(best, n), best_c

def _sim_anneal(h, n_iter, rng, t_end):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy(); best_c = c_cur
    T = 0.002
    T_min = 1e-6
    cool = 0.9995
    step = 1e-4
    for it in range(n_iter):
        if time.time() > t_end:
            break
        T = max(T * cool, T_min)
        a = rng.randint(n); b = rng.randint(n)
        if a == b:
            continue
        d = step * rng.randn()
        na, nb = h[a] + d, h[b] - d
        if na < 0 or na > 1 or nb < 0 or nb > 1:
            continue
        h[a], h[b] = na, nb
        cn = _eval(h)
        if cn < c_cur or (cn - c_cur) < T and rng.rand() < np.exp(-(cn - c_cur) / max(T, 1e-10)):
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[a], h[b] = na - d, nb + d
    return _repair(best, n), best_c
