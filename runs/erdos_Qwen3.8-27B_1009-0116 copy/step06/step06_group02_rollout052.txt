def _lse_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + float(np.sum(np.exp(s))) * (1.0 / np.exp(s.max() + np.log(len(s)))) / alpha if False else m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)

def _reweight(h, n_iter, top_m, t_end, base_rate=3.0):

    h = h.copy()
    cur = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-top_m:]
        cmin, cmax = c[idx].min(), c[idx].max()
        w = np.zeros(2 * n - 1)
        if cmax - cmin > 1e-12:
            beta = base_rate + 2.0 * it / max(1, n_iter)
            w[idx] = np.exp(beta * (c[idx] - cmin) / (cmax - cmin))
        else:
            w[idx] = 1.0
        w /= w.sum()
        g = np.zeros(n)
        dx = 2.0 / n
        active = np.where(w > 1e-14)[0]
        for k in active:

            wk = w[k] * dx

                g[sh:] -= wk * h[:cnt]
            else:

                cnt = n - sp

            for amp in [0.15, 0.25, 0.35]:
                for phase in [0, np.pi/3, 2*np.pi/3]:
                    if rem() < 60:
                        break
                    hf = _make_periodic(n0, freq, amp, phase)
                    hf = hf + 0.5 * (best_h - _repair(np.full(n0, 0.5), n0))
                    hf = _repair(hf, n0)
                    hf = _slsqp(hf, 80000, 2000, 1e-18)
                    cf = _eval(hf)
                    if cf < best_c5:
                        best_c5 = cf; best_h = hf.copy(); best_n = n0

    # Phase 6: High-alpha SLSQP final polish
    if rem() > 50:
        h6 = _slsqp(best_h, 500000, 8000, 1e-21)
        c5 = _eval(h6)
        if c5 < best_c5:
            best_c5 = c5; best_h = h6.copy(); best_n = len(h6)

    # Phase 7: Final reweight + very fine steps
    if rem() > 30:
        h7 = best_h.copy()
        h7, c5 = _reweight(h7, 3000, 2, t0 + rem() - 10, base_rate=8.0)
        if c5 < best_c5:
            best_c5 = c5; best_h = h7.copy(); best_n = len(h7)

    # Phase 8: Ultra-fine SLSQP
    if rem() > 15:
        h8 = _slsqp(best_h, 2000000, 10000, 1e-22)
        c5 = _eval(h8)
        if c5 < best_c5:
            best_c5 = c5; best_h = h8.copy(); best_n = len(h8)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
