    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-10:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break

    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for step in range(n_steps):
        t = t0 * (t1 / t0) ** (step / max(n_steps - 1, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5n = _c5max(h)
        delta = c5n - c5c
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 0.01)):
            c5c = c5n
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5n < best:
                best = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best

    n = len(h)

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        sm = s - s.max()
        return m + np.log(np.sum(np.exp(sm))) / alpha

        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)

            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g

    c5c = _c5max(h)
    while True:
        if time.time() - t_start > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for j in zeros:
            if time.time() - t_start > time_limit:
                return h, c5c
            for i in ones:
                h[j], h[i] = 1.0, 0.0
                c5n = _c5max(h)
                if c5n < c5c:
                    c5c = c5n
                    break
                h[j], h[i] = 0.0, 1.0
            if c5n < c5c:
                pass
            else:
                h[j], h[i] = 0.0, 1.0
    return h, c5c

def _greedy_best(h, time_limit, rng, max_sweeps=200):
    n = len(h)
    h = h.copy()
    t_start = time.time()
    c5c = _c5max(h)
    for _ in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_c5 = c5c
        best_pair = None
        rng.shuffle(zeros)
        rng.shuffle(ones)
        for j in zeros:
            for i in ones:
                h[j], h[i] = 1.0, 0.0
                c5n = _c5max(h)
                if c5n < best_c5:
                    best_c5 = c5n
                    best_pair = (j, i)
                h[j], h[i] = 0.0, 1.0
        if best_pair is None:
            break
        j, i = best_pair
        h[j], h[i] = 1.0, 0.0
        c5c = best_c5
    return h, c5c

def _anneal(h, t0, t1, n_steps, rng):
    n = len(h)
    h = h.copy()
    c5c = _c5max(h)
    best = c5c
    best_h = h.copy()
    for step in range(n_steps):
        t = t0 * (t1 / t0) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        c5n = _c5max(h)
        delta = c5n - c5c
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 0.005)):
            c5c = c5n
            if c5n < best:
                best = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
    return best_h, best

def _lssm_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + np.log(np.sum(np.exp(sm))) / alpha

def _lssm_grp(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _cnst_fun(x):
    return x.sum() - len(x) / 2.0

def _cnst_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cnst_fun, 'jac': _cnst_jac}]
    try:
        res = minimize(_lssm_obj, h, jac=_lssm_grp, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _kick(h, rng, n_kicks=3):
    h = h.copy()
    n = len(h)
    for _ in range(n_kicks):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(2):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
        h[i], h[j] = 1.0, 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=np.float64).copy()
    h0 = _repair(h0, n0)
    best_h = h0.copy()
    best_c5 = _c5val(h0)
    best_n = n0

    # Phase 1: Binarize + quick greedy (40s)
    h_bin = _binarize(h0, rng)
    h_bin, cm = _greedy_first(h_bin, 20, rng)
    c5b = cm * 2.0 / n0
    if c5b < best_c5:
        best_c5, best_h, best_n = c5b, h_bin.copy(), n0

    # Phase 2: Simulated annealing (150s)
    if elapsed() < budget_s - 150:
        n_steps = max(int((150 - (elapsed() - 20)) * 2000), 10000)
        h_a, cm_a = _anneal(best_h, 1.5, 0.02, n_steps, rng)
        c5a = cm_a * 2.0 / n0
        if c5a < best_c5:
            best_c5, best_h = c5a, h_a.copy()

    # Phase 3: Greedy polish after anneal (60s)
    if elapsed() < budget_s - 60:
        h_g, cm_g = _greedy_best(best_h, 40, rng, max_sweeps=500)
        c5g = cm_g * 2.0 / n0
        if c5g < best_c5:
            best_c5, best_h = c5g, h_g.copy()

    # Phase 4: Multiple anneal+greedy cycles with kicks (250s)
    if elapsed() < budget_s - 80:
        for rnd in range(8):
            if elapsed() > budget_s - 80:
                break
            h_k = _kick(best_h, rng, 2)
            h_k2, cm2 = _anneal(h_k, 0.8, 0.01, 8000, rng)
            h_k2, cm2 = _greedy_first(h_k2, 20, rng)
            c5k = cm2 * 2.0 / n0
            if c5k < best_c5:
                best_c5, best_h = c5k, h_k2.copy()

    # Phase 5: SLSQP polish (250s)
    if elapsed() < budget_s - 60:
        h_p = best_h.copy()
        for alpha in [5000, 50000, 200000, 2000000]:
            if elapsed() > budget_s - 50:
                break
            h_p = _slsqp(h_p, alpha, 2000)
            c5p = _c5val(h_p)
            if c5p < best_c5:
                best_c5, best_h = c5p, h_p.copy()

    # Phase 6: Final binary sweep + SLSQP (80s)
    if elapsed() < budget_s - 30:
        h_f = _binarize(best_h, rng)
        h_f, cm_f = _greedy_best(h_f, 30, rng, max_sweeps=300)
        c5f = cm_f * 2.0 / n0
        if c5f < best_c5:
            best_c5, best_h = c5f, h_f.copy()
        if elapsed() < budget_s - 15:
            h_f2 = _slsqp(best_h, 500000, 1500)
            c5f2 = _c5val(h_f2)
            if c5f2 < best_c5:
                best_c5, best_h = c5f2, h_f2.copy()

    # Phase 7: Coordinate micro-adjustment (remaining)
    if elapsed() < budget_s - 3:
        h_m = best_h.copy()
        c5m = _c5val(h_m)
        nc = len(h_m)
        for _ in range(20000):
            if elapsed() > budget_s - 3:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 0.001 * rng.randn()
            ni, nj = h_m[i] + d, h_m[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h_m[i], h_m[j] = ni, nj
            c5n = _c5val(h_m)
            if c5n < c5m - 1e-15:
                c5m = c5n
            else:
                h_m[i], h_m[j] = ni - d, nj + d
        if c5m < best_c5:
            best_c5, best_h = c5m, h_m.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _c5val(best_h)
    return (best_h, best_c5, best_n)
