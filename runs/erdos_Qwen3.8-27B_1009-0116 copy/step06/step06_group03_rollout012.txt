    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    smax = s.max()
    return m + (smax + np.log(np.sum(np.exp(s - smax)))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            end = n - sh
            g[:end] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:end]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:sp])
            g[:sp] -= wk * h[sp:]
    return g

    n = len(h)

    def _cf(x):
        return np.array([x.sum() - 0.5 * n])
    def _cj(x):
        return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]

        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

        best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
