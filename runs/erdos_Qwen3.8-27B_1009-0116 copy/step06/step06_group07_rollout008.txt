def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_max(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _soft_max_grad(c, h, alpha, n):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            i1 = np.arange(cnt)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                g[i1 + sh] -= wk * h[i1]
        else:
            sp = -sh
            cnt = n - sp
            i2 = np.arange(cnt)
            g[i2 + sp] += wk * (1.0 - h[i2])
            g[i2 + sp] -= wk * h[i2 + sp]
            i3 = np.arange(sp, n)
            g[i3] += wk * (1.0 - h[i3 - sp])
            g[i3 - sp] -= wk * h[i3]
    return g

        except Exception:
            pass

    # Phase 5: Simulated annealing
    h_p, c5 = _sa_period(h_p, T, n0, 30000, t0, t0 + budget_s * 0.65, seed + 50)
    if c5 < best_c5:
        best_c5 = c5
        best_h = np.tile(h_p, m).copy()

    # Phase 6: Coordinate descent polish (tight)
    for scale in [0.01, 0.003, 0.001, 0.0003, 0.0001]:
        if elapsed() > budget_s * 0.75:
            break
        h_p, c5 = _cd_period(h_p.copy(), T, n0, 30000, scale, t0, t0 + budget_s * 0.76, seed + 100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.tile(h_p, m).copy()

    # Phase 7: Another SLSQP + NM polish cycle
    if elapsed() < budget_s * 0.82:
        try:
            res = minimize(_slsqp_obj, h_p, jac=_slsqp_jac, args=(200000000, T, n0),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-19})
            h_p = _repair(res.x, T)
            c5 = _eval(np.tile(h_p, m))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_p, m).copy()
        except Exception:
            pass
    if elapsed() < budget_s * 0.88:
        res = minimize(_nm_obj, h_p, args=(T, n0), method='Nelder-Mead',
                       options={'maxiter': 4000, 'xatol': 1e-12, 'fatol': 1e-15, 'adaptive': True})
        h_opt = _proj_sum(np.clip(res.x, 0, 1), T)
        c5 = _eval(np.tile(h_opt, m))
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.tile(h_opt, m).copy()

    # Phase 8: Full 310-point SLSQP
    if elapsed() < budget_s * 0.92:
        nb = [(0.0, 1.0)] * n0
        ns_cons = [{'type': 'eq', 'fun': _slsqp_sum_eq, 'jac': _slsqp_sum_jac, 'args': (n0,)}]
        for alpha in [500000, 5000000]:
            if elapsed() > budget_s * 0.94:
                break
            try:
                res = minimize(_slsqp_obj, best_h, jac=_slsqp_jac, args=(alpha, n0, n0),
                               method='SLSQP', bounds=nb, constraints=ns_cons,
                               options={'maxiter': 800, 'ftol': 1e-17})
                h_new = _repair(res.x, n0)
                c5 = _eval(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    best_n = n0
            except Exception:
                pass

    # Phase 9: Final CD at full res
    if elapsed() < budget_s - 15:
        for scale in [0.003, 0.0005]:
            if elapsed() > budget_s - 8:
                break
            h_c = best_h.copy()
            c5c = _eval(h_c)
            rng2 = np.random.RandomState(seed + 999)
            for _ in range(15000):
                if elapsed() > budget_s - 5:
                    break
                i, j = rng2.randint(best_n), rng2.randint(best_n)
                if i == j:
                    continue
                d = scale * rng2.randn()
                ni, nj = h_c[i] + d, h_c[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h_c[i], h_c[j] = ni, nj
                c5n = _eval(h_c)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                else:
                    h_c[i], h_c[j] = ni - d, nj + d
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h_c.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
