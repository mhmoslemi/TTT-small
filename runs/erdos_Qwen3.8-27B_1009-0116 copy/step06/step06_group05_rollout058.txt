def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)

def _lssm(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _lssm_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')

    s = np.clip(alpha * (c - m), -500, 500)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_at_k(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0: g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _lssm(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _lssm_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _se(x): return x.sum() - len(x) * 0.5
def _sj(x): return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_lssm, h, jac=_lssm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lp_refine(h, n, tol_frac=0.02, max_rounds=10):
    h = h.copy()
    for _ in range(max_rounds):
        c = _corr(h)
        dx = 2.0 / n
        cmax = c.max()
        active = np.where(c >= cmax - tol_frac * cmax)[0]
        a_rows, b_rows = [], []
        for k in active:
            g = _grad_at_k(h, k, n) / dx
            a_rows.append(np.append(g, -1.0))
            b_rows.append(-c[k] + g @ h / dx)
        A = np.array(a_rows)
        b = np.array(b_rows)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [n / 2.0]
        bounds = [(0, 1)] * n + [(0, None)]
        try:
            res = linprog(np.zeros(n + 1), A_ub=A, b_ub=b, A_eq=A_eq,
                          b_eq=b_eq, bounds=bounds, method='highs')
            if res.success and res.x is not None:
                h_new = _repair(res.x[:n], n)
                if _eval(h_new) < _eval(h) - 1e-14:
                    h = h_new
                else:
                    break
            else:
                break
        except Exception:
            break
    return h

def _subgrad(h, n, t_max, t_start):
    h = h.copy()
    c5_cur = _eval(h)
    c5_best = c5_cur
    h_best = h.copy()
    step = 0.03
    while time.time() - t_start < t_max:
        c = _corr(h)
        ki = int(np.argmax(c))
        g = _grad_at_k(h, ki, n)
        moved = False
        for s in [step, step*0.5, step*0.25, step*0.1, step*0.05]:
            h_new = _repair(h - s * g, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                h, c5_cur = h_new, c5_new
                if c5_cur < c5_best:
                    c5_best, h_best = c5_cur, h.copy()
                moved = True
                break
        if not moved:
            step *= 0.85
            if step < 1e-7: break
    return h_best, c5_best

def _binary_anneal(h, n, t_max, t_start, rng):
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, n)
    c5_cur = _eval(h)
    c5_best = c5_cur
    h_best = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    n_steps = 999999
    step = 0
    while step < n_steps and time.time() - t_start < t_max:
        step += 1
        t = 0.4 * (0.0003 / 0.4) ** (step / max(n_steps - 1, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        h = _repair(h, n)
        c5_new = _eval(h)
        if c5_new < c5_cur or rng.rand() < np.exp(-(c5_new - c5_cur) / max(t, 1e-7)):
            c5_cur = c5_new
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if c5_cur < c5_best:
                c5_best, h_best = c5_cur, h.copy()
        else:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
            h = _repair(h, n)
    return h_best, c5_best

def _kick(h, n, sigma, rng):
    h_k = h + sigma * rng.randn(n)
    return _repair(h_k, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def elapsed(): return time.time() - t0

    # Phase 1: SLSQP chain (250s)
    for alpha in [5000, 20000, 100000, 500000, 2000000, 10000000]:
        if elapsed() > 250: break
        h_new = _slsqp(best_h, alpha, 3000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: LP active-set refinement (100s)
    if elapsed() < budget_s - 550:
        h_lp = _lp_refine(best_h, best_n, 0.02, 15)
        c5_lp = _eval(h_lp)
        if c5_lp < best_c5:

        h_lp2 = _lp_refine(best_h, best_n, 0.005, 20)
        c5_lp2 = _eval(h_lp2)
        if c5_lp2 < best_c5:
            best_c5, best_h = c5_lp2, h_lp2.copy()

    # Phase 3: Kick + re-optimize (150s)
    for kick_sigma in [0.05, 0.02, 0.01, 0.005]:
        if elapsed() > budget_s - 400: break
        h_k = _kick(best_h, best_n, kick_sigma, rng)
        h_k = _slsqp(h_k, 200000, 2000, 1e-19)
        h_k = _lp_refine(h_k, best_n, 0.01, 8)
        c5_k = _eval(h_k)
        if c5_k < best_c5:
            best_c5, best_h = c5_k, h_k.copy()

    # Phase 4: Binary annealing (200s)
    if elapsed() < budget_s - 200:
        t_rem_a = budget_s - 200 - elapsed()
        h_a, c5_a = _binary_anneal(best_h, best_n, t_rem_a, t0, rng)
        h_a = _slsqp(h_a, 500000, 2000, 1e-19)

    # Phase 5: Subgradient descent (100s)

    for n_try in [308, 312, 250, 350, 200, 400]:

                        np.linspace(0, 2, best_n, endpoint=False), best_h)
        h_r = _repair(h_r, n_try)

        if c5_f < best_c5:

    # Phase 8: Micro coordinate descent (5s)
    if elapsed() < budget_s - 1:
        h8 = best_h.copy()
        c5c = _eval(h8)
        nc8 = len(h8)
        for scale in [0.002, 0.0005, 0.0001]:

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
