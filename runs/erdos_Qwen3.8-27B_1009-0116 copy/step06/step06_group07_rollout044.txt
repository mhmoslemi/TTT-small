import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    w = np.exp(s)

    dx = 2.0 / n
    g = np.zeros(n)

        lag = k - (n - 1)

            g[idx] += wk * (1.0 - x[idx - l2])
            g[idx - l2] -= wk * x[idx]
    return g

def _slsqp(h0, alpha, maxiter=5000, ftol=1e-18):

    except Exception:
        pass

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _coord_descent(h, n_iter, scale, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c
