```python
def _obj_and_grad(h, K, alpha):
    """Compute soft-max objective and gradient using top-K shifts."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    
    # Get top K
    top_idx = np.argpartition(c, -K)[-K:]
    top_c = c[top_idx]
    
    # Soft weights over top K
    m = top_c.max()
    w = np.exp(alpha * (top_c - m))
    w /= w.sum()
    
    # Objective: m + log(sum(w*exp(alpha*(c-m))))/alpha ≈ soft max over top K
    # Actually just use m + log(sum(exp(alpha*(top_c-m))))/alpha
    logsumexp = np.log(np.sum(np.exp(alpha * (top_c - m))))
    obj = m + logsumexp / alpha
    
    # Gradient
    g = np.zeros(n)
    for ik, ki in enumerate(top_idx):
        wk = w[ik] * dx
        lag = int(ki) - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[:n-lag] += wk * (1.0 - h[lag:])
            g[lag:] -= wk * h[:n-lag]
        else:
            p = -lag
            g[p:] += wk * (1.0 - h[:n-p])
            g[:n-p] -= wk * h[p:]
    
    return obj, g
```

```python
def _slsqp_polish(h, alpha, maxiter):
    n = len(h)
    def _obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        s = np.clip(alpha*(c-m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s))))/alpha
    def _grad(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        s = np.clip(alpha*(c-m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0/n
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15:
                continue
            sh = (n-1) - k
            wk = w[k]*dx
            if sh >= 0:
                g[:n-sh] += wk*(1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk*x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk*(1.0 - x[:n-sp])
                g[:n-sp] -= wk*x[sp:]
        return g
    bnds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':lambda x: x.sum()-n*0.5,'jac':lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter':maxiter,'ftol':1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)
```

    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _binary_sa(h0, n, deadline, seed):
    rng = np.random.RandomState(seed)
    x = _repair((h0 > 0.5).astype(float), n)
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    T_sa = 0.004
    for _ in range(800000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        x[i] = 1.0 - x[i]
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new <= c_cur or rng.random() < np.exp(-(c_new - c_cur) / T_sa):
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
        else:
            x[i] = 1.0 - x[i]
            x = _repair(x, n)
        T_sa *= 0.99999
    return best_x, best_c
