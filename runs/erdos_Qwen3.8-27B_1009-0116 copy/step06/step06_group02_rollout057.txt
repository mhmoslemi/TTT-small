def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break

                break

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cor(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _contrib_1d(h, j, n):
    contrib = np.zeros(2 * n - 1)
    quad = np.zeros(2 * n - 1)
    if j < n:
        quad[n-1] = -1.0
        contrib[n-1] = 1.0
    for s in range(1, n):
        k = (n - 1) + s
        i1 = j - s
        if 0 <= i1 < n:
            contrib[k] += 1.0 - h[i1]
        i2 = j + s
        if 0 <= i2 < n:
            contrib[k] -= h[i2]
        k2 = (n - 1) - s
        i3 = j + s
        if 0 <= i3 < n:
            contrib[k2] += 1.0
