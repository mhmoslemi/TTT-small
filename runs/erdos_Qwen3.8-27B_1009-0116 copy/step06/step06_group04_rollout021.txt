    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()

                g[i2] -= wk * x[i2 - sh]

            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]

    x = x.copy()
    c_arr = _eval_arr(x)
    c_cur = c_arr.max()
    best_x = x.copy(); best_c = c_cur
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        top_k = np.argsort(c_arr)[-3:]
        k_star = top_k[rng.randint(len(top_k))]
        sh = (n - 1) - k_star
        good_i = []
        good_j = []
        if sh >= 0:
            for i in range(n - sh):
                if x[i] > 0.5 and x[i + sh] < 0.5:
                    good_i.append(i)
            for j in range(sh, n):
                if x[j] < 0.5 and x[j - sh] > 0.5:
                    good_j.append(j)
        else:
            sp = -sh
            for i in range(sp, n):
                if x[i] > 0.5 and x[i - sp] < 0.5:
                    good_i.append(i)
            for j in range(n - sp):
                if x[j] < 0.5 and x[j + sp] > 0.5:
                    good_j.append(j)
        if len(good_i) == 0 or len(good_j) == 0:
            if rng.random() < 0.3:
                ones = np.where(x > 0.5)[0]
                zeros = np.where(x < 0.5)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                good_i = [ones[rng.randint(len(ones))]]
                good_j = [zeros[rng.randint(len(zeros))]]
            else:
                c_arr = np.roll(c_arr, 1)
                stall += 1
                if stall > 50:
                    break
                continue
        stall = 0
        i = good_i[rng.randint(len(good_i))]
        j = good_j[rng.randint(len(good_j))]
        if i == j:
            continue
        x[i] = 1.0 - x[i]
        x[j] = 1.0 - x[j]
        x = _repair(x, n)
        c_arr = _eval_arr(x)
        c_new = c_arr.max()
        if c_new <= c_cur + 1e-12:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new; best_x = x.copy()
        else:
            x[i] = 1.0 - x[i]
            x[j] = 1.0 - x[j]
            x = _repair(x, n)
            c_arr = _eval_arr(x)
            c_cur = c_arr.max()
    return best_x, best_c

def _binary_search_random(x, n, rng, deadline, max_iters=3000000):
    x = x.copy()
    ones = np.where(x > 0.5)[0]
    zeros = np.where(x < 0.5)[0]
    c_cur = _eval(x)
    best_x = x.copy(); best_c = c_cur
    for it in range(max_iters):
        if time.time() > deadline:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        x[a] = 0.0; x[b] = 1.0
        c_new = _eval(x)
        if c_new <= c_cur:
            c_cur = c_new
            ones = (ones[ones != a]).tolist()
            zeros = (zeros[zeros != b]).tolist()
            ones = (np.array(ones) if isinstance(ones, list) else ones).tolist()
            zeros = (np.array(zeros) if isinstance(zeros, list) else zeros).tolist()
            ones = list(np.setdiff1d(ones, [a]))
            zeros = list(np.setdiff1d(zeros, [b]))
            ones.append(b); zeros.append(a)
            ones = np.array(ones); zeros = np.array(zeros)
            if c_new < best_c:
                best_c = c_new; best_x = x.copy()
        else:
            x[a] = 1.0; x[b] = 0.0
    return best_x, best_c

def _fw_coord(x, n, deadline, max_iters=500000):
    x = x.copy()
    c_arr = _eval_arr(x)
    c_cur = c_arr.max()
    best_x = x.copy(); best_c = c_cur
    rng = np.random.RandomState(12345)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        cur = x[i]
        for trial in [0.0, 1.0]:
            if abs(trial - cur) < 1e-9:
                continue
            x[i] = trial
            x = _repair(x, n)
            c_arr = _eval_arr(x)
            c_new = c_arr.max()
            if c_new < c_cur - 1e-12:
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new; best_x = x.copy()
                break
        else:
            x[i] = cur
            x = _repair(x, n)
    return best_x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Adam with increasing alpha (0-120s)
    dl1 = t0 + 120
    for (a0, a1, ni, l0, l1) in [(200, 2000, 2000, 0.03, 0.003),
                                   (2000, 20000, 2000, 0.02, 0.002),
                                   (20000, 200000, 2000, 0.01, 0.001)]:
        if time.time() > dl1:
            break
        x, c = _adam_opt(best_h, a0, a1, ni, l0, l1, seed, dl1)
        if c < best_c5:
            best_c5, best_h = c, x.copy()

    # Phase 2: SLSQP polish (120-220s)
    dl2 = t0 + 220
    for alpha in [50000, 500000, 5000000]:
        if time.time() > dl2:
            break
        h_new = _slsqp(best_h, alpha, 3000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Binary targeted search (220-500s)
    dl3 = t0 + 500
    h_bin = (best_h > 0.5).astype(float)
    h_bin = _repair(h_bin, best_n)
    x_b, c_b = _binary_search_targeted(h_bin, best_n, rng, dl3)
    if c_b < best_c5:
        best_c5, best_h = c_b, x_b.copy()

    # Phase 4: Multi-n exploration (500-700s)
    dl4 = t0 + 700
    for n_try in [best_n - 10, best_n, best_n + 10, best_n + 20, best_n + 40]:
        if time.time() > dl4:
            break
        if n_try < 60 or n_try > 800:
            continue
        if n_try % 2 != 0:
            n_try += 1
        h_r = _repair(_resample(best_h, n_try), n_try)
        if time.time() > dl4:
            break
        x_a, c_a = _adam_opt(h_r, 500, 50000, 1500, 0.01, 0.0005, seed + n_try, dl4)
        if c_a < best_c5:
            best_c5, best_h, best_n = c_a, x_a.copy(), n_try
        if time.time() > dl4:
            break
        h_bin2 = (x_a > 0.5).astype(float)
        h_bin2 = _repair(h_bin2, n_try)
        x_b2, c_b2 = _binary_search_targeted(h_bin2, n_try, rng, dl4)
        if c_b2 < best_c5:
            best_c5, best_h, best_n = c_b2, x_b2.copy(), n_try

    # Phase 5: Random binary search on best (700-850s)
    dl5 = t0 + 850
    h_bin3 = (best_h > 0.5).astype(float)
    h_bin3 = _repair(h_bin3, best_n)
    x_r, c_r = _binary_search_random(h_bin3, best_n, rng, dl5, 2000000)
    if c_r < best_c5:
        best_c5, best_h = c_r, x_r.copy()

    # Phase 6: Frank-Wolfe coordinate on continuous (850-950s)
    dl6 = t0 + 950
    x_fw, c_fw = _fw_coord(best_h, best_n, dl6, 300000)
    if c_fw < best_c5:
        best_c5, best_h = c_fw, x_fw.copy()

    # Phase 7: Final high-alpha Adam + SLSQP (950-1050s)
    dl7 = t0 + 1050
    x, c = _adam_opt(best_h, 50000, 1000000, 3000, 0.003, 0.00005, seed + 111, dl7)
    if c < best_c5:
        best_c5, best_h = c, x.copy()
    if time.time() < dl7:
        h_fin = _slsqp(best_h, 10000000, 3000)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    # Phase 8: Final binary targeted (1050-1090s)
    dl8 = t0 + 1090
    h_final_bin = (best_h > 0.5).astype(float)
    h_final_bin = _repair(h_final_bin, best_n)
    x_fb, c_fb = _binary_search_targeted(h_final_bin, best_n, rng, dl8, 500000)
    if c_fb < best_c5:
        best_c5, best_h = c_fb, x_fb.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
