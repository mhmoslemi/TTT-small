```python
def _grad_smax(h, alpha):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    r = np.correlate(h, u, mode='full')  # or FFT version
    ar = alpha * r * dx
    w = np.exp(ar - ar.max())
    w /= w.sum()
    
    N = 1
    while N < 4 * n:
        N *= 2
    
    f_W = np.fft.rfft(w, N)
    f_u = np.fft.rfft(u, N)
    f_h = np.fft.rfft(h, N)
    
    conv_Wu = np.fft.irfft(f_W * f_u, N)  # W_pad * u
    corr_Wh = np.fft.irfft(f_W * np.conj(f_h), N)  # W_pad ⋆ h
    
    grad = np.zeros(n)
    grad[:] = dx * (conv_Wu[n-1:n-1+n] - corr_Wh[n-1:n-1+n][::-1])
    return grad
```
