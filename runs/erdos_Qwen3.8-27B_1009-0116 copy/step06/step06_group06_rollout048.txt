def _block_cd(h, n, t_end, block_size, rng, n_rounds):
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    bnds_orig = [(0.0, 1.0)] * n
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        bnds = bnds_orig[:n]
        for start in range(0, n - block_size + 1, block_size):
            if time.time() > t_end:
                break
            end = min(start + block_size, n)
            for i in range(end):
                if i >= start and i < end:
                    bnds[i] = (0.0, 1.0)
                else:
                    bnds[i] = (h_cur[i], h_cur[i])
        def c_fun(x):
            return x.sum() - n * 0.5
        cons = [{'type': 'eq', 'fun': c_fun, 'jac': lambda x: np.ones(x.shape)}]
        try:
            res = minimize(_smax_obj, h_cur, jac=_smax_grad, args=(50000,),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-14, 'disp': False})
            if res.fun < c5_cur * len(h_cur) / 2.0 - 1e-10:
                h_try = _repair(res.x, n)
                c5n = _eval(h_try)
                if c5n < c5_cur - 1e-15:
                    h_cur = h_try
                    c5_cur = c5n
        except Exception:
            pass
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end_final = t0 + budget_s
    rng = np.random.RandomState(seed)

    def rem():
        return t_end_final - time.time()

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Top-k SLSQP to smooth the landscape
    for k in [50, 20, 10, 5]:
        if rem() < 200:
            break
        h_try = _topk_slsqp(best_h, k, 2000, t0 + rem() - 195)
        c5 = _eval(h_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 2: SLSQP with progressive smax alpha
    for alpha in [500, 3000, 15000, 80000, 300000]:
        if rem() < 180:
            break
        h_try = _slsqp_opt(best_h, alpha, 4000, t0 + rem() - 175)
        c5 = _eval(h_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 3: Projected gradient with smax
    if rem() > 200:
        h_try, _ = _proj_grad(best_h, _smax_obj, _smax_grad, 600,
                              t0 + rem() - 195, step_init=0.015)
        h_try = _repair(h_try, len(h_try))
        c5 = _eval(h_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 4: Multi-lag descent
    if rem() > 180:
        h_try, c5_try = _multi_lag_descent(best_h, 400, t0 + rem() - 175, top_k=10)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Upsample to 2x
    if rem() > 400:
        n1 = n0 * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for k in [20, 10, 5]:
            if rem() < 370:
                break
            h_hi = _topk_slsqp(h_hi, k, 1500, t0 + rem() - 365)
        for alpha in [1000, 10000, 100000, 500000]:
            if rem() < 340:
                break
            h_hi = _slsqp_opt(h_hi, alpha, 3000, t0 + rem() - 335)
        c5_hi = _eval(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n1
        if rem() > 280:
            h_hi, _ = _proj_grad(h_hi, _smax_obj, _smax_grad, 400,
                                  t0 + rem() - 275, step_init=0.01)
            h_hi = _repair(h_hi, n1)
            c5_hi = _eval(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n1
        if rem() > 220:
            h_try, c5_try = _multi_lag_descent(h_hi, 300, t0 + rem() - 215, top_k=10)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n1

    # Phase 6: Block coordinate descent
    if rem() > 150:
        bs = max(20, best_n // 10)
        h_try, c5_try = _block_cd(best_h, best_n, t0 + rem() - 145, bs, rng, 200)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 7: SA with structured moves
    if rem() > 100:
        n_iter = min(int((rem() - 90) / 0.003), 50000)
        h_try, c5_try = _sa_fine(best_h, best_n, t0 + rem() - 85, n_iter, rng, 0.015)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 8: Final SLSQP polish with very high alpha
    if rem() > 60:
        h_try = _slsqp_opt(best_h, 500000, 6000, t0 + rem() - 55)
        c5 = _eval(h_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 9: Ultra-fine coordinate descent
    if rem() > 20:
        h_cur = best_h.copy()
        nc = len(h_cur)
        c5c = _eval(h_cur)
        n_tries = min(int((rem() - 15) / 0.0012), 80000)
        for _it in range(n_tries):
            if rem() < 15:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 3e-5 * rng.randn()
            ni, nj = h_cur[i] + d, h_cur[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h_cur[i], h_cur[j] = ni, nj
            c5n = _eval(h_cur)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h_cur.copy()
            else:
                h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d

    # Phase 10: Final tiny SA
    if rem() > 10:
        h_try, c5_try = _sa_fine(best_h, best_n, t0 + rem() - 5, 5000, rng, 2e-5)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
