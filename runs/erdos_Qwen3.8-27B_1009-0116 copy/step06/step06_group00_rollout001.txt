        g, cmax_y, _ = _topk_grad(y, k)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_sum(y - step * v, target)
        f_new = _eval(h_new)
        if f_new < f:
            h_prev = h.copy()
            h = h_new
            f = f_new
            step = min(step * 1.05, step0 * 5.0)
        else:
            step *= 0.6
            h_new = _proj_sum(y - step * v, target)
            f_new2 = _eval(h_new)
            if f_new2 < f:
                h_prev = h.copy()
                h = h_new
                f = f_new2
    return _repair(h, n), f

def _sim_anneal(h0, n_iter, T0, T1, rng, budget_fn):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    fc = _eval(h)
    best_h = h.copy()
    best_f = fc
    for it in range(n_iter):
        if budget_fn() < 15:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter - 1, 1))
        n_pix = rng.randint(3, 12)
        idx = rng.choice(n, size=n_pix, replace=False)
        deltas = 0.02 * rng.randn(n_pix)
        total = deltas.sum()
        deltas[list(idx)[0]] -= total
        h_t = h.copy()
        h_t[list(idx)] += deltas
        h_t = _proj_sum(h_t, target)
        fc_new = _eval(h_t)
        delta = fc_new - fc
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h = _repair(h_t, n)
            fc = fc_new
            if fc < best_f:
                best_f = fc
                best_h = h.copy()
    return best_h, best_f

def _coord_desc(h0, n_sweeps, rng, budget_fn):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    fc = _eval(h)
    best_h = h.copy()
    best_f = fc
    for s in range(n_sweeps):
        if budget_fn() < 10:
            break
        for _ in range(n // 2):
            if budget_fn() < 10:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n // 2)) % n
            d = 5e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                fc_n = _eval(h)
                if fc_n < fc:
                    fc = fc_n
                    if fc < best_f:
                        best_f = fc
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_f, best_f

def _slsqp_opt(h, alpha, maxiter, budget_fn):
    n = len(h)
    dx = 2.0 / n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for ki in range(2 * n - 1):
            if w[ki] < 1e-14:
                continue
            sh = (n - 1) - ki
            wk = w[ki] * dx
            if sh >= 0:
                m2 = n - sh
                g[:m2] += wk * (1.0 - x[sh:sh+m2])
                if sh > 0:
                    g[sh:n] -= wk * x[0:m2]
            else:
                sp = -sh
                g[sp:n] += wk * (1.0 - x[0:n-sp])
                g[0:n-sp] -= wk * x[sp:n]
        return g
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    cons = [{'type': 'eq', 'fun': _cf, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _remaining():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Nesterov top-k subgradient at full resolution
    for k_top in [6, 10, 15, 8]:
        if _remaining() < 120:
            break
        h_c, f_c = _nesterov_topk(best_h, k_top, 800, 0.008, _remaining)
        f_c = _eval(h_c)
        if f_c < best_c5:
            best_c5 = f_c
            best_h = h_c.copy()

    # Phase 2: Simulated annealing for exploration
    if _remaining() > 200:
        n_iter = int(min(_remaining() - 180, 300) / 0.004)
        n_iter = max(n_iter, 2000)
        h_sa, f_sa = _sim_anneal(best_h, n_iter, 0.003, 1e-6, rng, _remaining)
        f_sa = _eval(h_sa)
        if f_sa < best_c5:
            best_c5 = f_sa
            best_h = h_sa.copy()

    # Phase 3: Nesterov again after annealing
    if _remaining() > 100:
        for k_top in [5, 8, 12]:
            if _remaining() < 60:
                break
            h_c, f_c = _nesterov_topk(best_h, k_top, 500, 0.005, _remaining)
            f_c = _eval(h_c)
            if f_c < best_c5:
                best_c5 = f_c
                best_h = h_c.copy()

    # Phase 4: SLSQP refinement
    if _remaining() > 80:
        h_s = _slsqp_opt(best_h, 50000, 500, _remaining)
        f_s = _eval(h_s)
        if f_s < best_c5:
            best_c5 = f_s
            best_h = h_s.copy()
        if _remaining() > 40:
            h_s2 = _slsqp_opt(best_h, 200000, 400, _remaining)
            f_s2 = _eval(h_s2)
            if f_s2 < best_c5:
                best_c5 = f_s2
                best_h = h_s2.copy()

    # Phase 5: Coordinate descent polish
    if _remaining() > 20:
        n_sweeps = int(min(_remaining() - 15, 60) / 0.5)
        h_cd, f_cd, _ = _coord_desc(best_h, max(n_sweeps, 10), rng, _remaining)
        f_cd = _eval(h_cd)
        if f_cd < best_c5:
            best_c5 = f_cd
            best_h = h_cd.copy()

    # Phase 6: Very fine 2-point local search
    if _remaining() > 3:
        h = best_h.copy()
        nc = len(h)
        fc = _eval(h)
        n_tries = int((_remaining() - 2) / 0.001)
        n_tries = min(n_tries, 50000)
        for _it in range(n_tries):
            if _remaining() < 2:
                break
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 4)) % nc
            d = 3e-6 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                fc_n = _eval(h)
                if fc_n < fc - 1e-18:
                    fc = fc_n
                    if fc < best_c5:
                        best_c5 = fc
                        best_h = h.copy()
                        best_n = nc
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
