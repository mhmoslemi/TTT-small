import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _top_lags(h, M):
    ov = np.correlate(h, 1.0 - h, mode='full')
    return ov[np.argsort(ov)[-M:]], ov

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _active_set_descent(h, nc, max_rounds, M=5):
    """Descent using gradients of top-M active lags."""
    c5c = _eval(h)
    dx = 2.0 / nc
    for r in range(max_rounds):
        ov = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(ov)[-M:]
        g = np.zeros(nc)
        for kk in top_idx:
            g += _grad_lag(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        # Line search
        step = 0.1
        best_c5 = c5c
        best_h = h
        for _ in range(30):
            h_new = _repair(h - step * g, nc)
            c5n = _eval(h_new)
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
                best_h = h_new
            else:
                step *= 0.5
        if best_c5 >= c5c - 1e-15:
            break
        h = best_h
        c5c = best_c5
    return h, c5c

def _soft_max_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _soft_max_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_soft_max_obj, h, jac=_soft_max_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _sim_anneal(h, nc, rng, t_start, t_end, n_iter):
    """Simulated annealing by moving mass between pairs."""
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for i in range(n_iter):
        t = t_start * (t_end / t_start) ** (i / n_iter)
        a = rng.randint(nc)
        b = rng.randint(nc)
        if a == b:
            continue
        d = rng.uniform(-0.05, 0.05)
        na, nb = h[a] + d, h[b] - d
        if na < 0 or na > 1 or nb < 0 or nb > 1:
            continue
        h[a], h[b] = na, nb
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        elif rng.rand() < np.exp((c5c - c5n) / max(t, 1e-10)):
            c5c = c5n
        else:
            h[a], h[b] = na - d, nb + d
    return best_h, best_c5

def _fine_cd(h, nc, rng, scales=None, iters_per_scale=5000):
    """Fine coordinate descent with multiple scales."""
    if scales is None:
        scales = [0.002, 0.0005, 0.0001, 0.00002]
    c5c = _eval(h)
    for scale in scales:
        for _ in range(iters_per_scale):
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _binary_search_n(h_best, best_n, n_target, rng):
    """Optimize at a specific grid size."""
    h_r = _repair(_resample(h_best, n_target), n_target)
    # SLSQP
    h_r = _slsqp(h_r, 10000, 1500, 1e-17)
    # Active set descent
    h_r, c5 = _active_set_descent(h_r, n_target, 300, M=5)
    # Fine CD
    h_r, c5 = _fine_cd(h_r, n_target, rng, [0.001, 0.0003], 3000)
    # Final SLSQP
    h_r = _slsqp(h_r, 50000, 1000, 1e-18)
    c5 = _eval(h_r)
    # Final active set
    h_r, c5 = _active_set_descent(h_r, n_target, 200, M=3)
    return h_r, c5, n_target
