        idx = np.argsort(c)[-top_k:]
        g = _active_grad(best, idx)
        # Project: move opposite to gradient, maintain sum constraint
        g_mean = g.mean()
        g = g - g_mean  # zero-sum (maintains integral constraint)
        new_h = best - lr * g
        new_h = _repair(new_h, n)
        nc = _eval(new_h)
        if nc < best_c - 1e-15:
            best = new_h
            best_c = nc
            if lr < 0.01:
                lr *= 1.02
        else:
            lr *= 0.7
            if lr < 1e-8:
                lr = 0.001
    return best, best_c

    """Pairwise coordinate descent"""

        if c5n < c5c - 1e-16:

        else:

    return h, c5c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    n = len(h)
