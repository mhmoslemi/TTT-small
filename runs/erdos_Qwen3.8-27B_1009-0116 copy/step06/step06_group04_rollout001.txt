import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_obj(x, alpha, dx):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -600, 600)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _soft_grad(x, alpha, dx):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -600, 600)
    w = np.exp(s - s.max())
    w /= w.sum()
    d = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    for i in range(n):
        ip = i + d
        im = i - d
        ipv = (ip >= 0) & (ip < n)
        imv = (im >= 0) & (im < n)
        wi = np.zeros(2 * n - 1)
        wi[:] = 1.0
        wi[~ipv] = 0.5
        wi[ipv] -= x[ip[ipv]]
        wi[~imv] -= 0
        wi[imv] -= x[im[imv]]
        g[i] = dx * np.dot(w, wi)
    return g

                return best_x, best_c

def _slsqp_polish(h, alpha, maxiter):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    def _sum_eq(x):
        return x.sum() - n * 0.5
    def _sum_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_soft_obj, h, jac=lambda x: _soft_grad(x, alpha, dx),
                       args=(), method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

            best_c5, best_h, best_n = c_r, x_r.copy(), n_try

    # Phase 6: Random restarts + optimization (800-950s)
    dl6 = t0 + 930
    for trial in range(6):
        if time.time() > dl6:
            break
        h_rand = _random_binary(best_n, seed + 1000 + trial * 7)
        x_r, c_r = _adam_opt(h_rand, [200, 2000, 20000, 100000], 8000, 0.02, 0.001, seed + 2000 + trial, dl6)
        x_r, c_r = _paired_cd(x_r, best_n, rng, 300000, dl6)
        if c_r < best_c5:
            best_c5, best_h = c_r, x_r.copy()
    # Also try complement
    h_comp = _repair(1.0 - best_h, best_n)
    c5c = _eval(h_comp)
    if c5c < best_c5:
        best_c5, best_h = c5c, h_comp.copy()

    # Phase 7: Final high-alpha Adam + paired CD + SLSQP (930-1080s)
    dl7 = t0 + 1070
    x, c = _adam_opt(best_h, [50000, 200000, 1000000], 15000, 0.005, 0.00003, seed + 5555, dl7)
    if c < best_c5:
        best_c5, best_h = c, x.copy()
    x, c = _paired_cd(best_h, best_n, rng, 3000000, dl7)
    if c < best_c5:
        best_c5, best_h = c, x.copy()
    if time.time() < dl7 - 10:
        h_fin = _slsqp_polish(best_h, 5000000, 5000)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    # Final repair
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
