            c = _c5(h_temp)
            if c < best_c - 1e-14:
                best_c = c
                best_v = v
        h[j] = best_v
        h = _project(h, target)
        c5c = _c5(h)
    return h, c5c

def _slsqp_fast(h, alpha, maxiter):
    n = len(h)
    target = n / 2.0
    def obj(x):
        ov = np.correlate(x, 1.0 - x, mode='full')
        m = ov.max()
        s = np.clip(alpha * (ov - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def jac(x):
        return _grad_lse(x, alpha)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _project(res.x, target)
    except Exception:
        return _project(h, target)

def _resample(h_old, n_new):
    xo = np.linspace(0.0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _fourier_init(K, n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        h += rng.uniform(-0.25, 0.25) * np.cos(np.pi * j * x)
        h += rng.uniform(-0.25, 0.25) * np.sin(np.pi * j * x)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _project(np.array(initial_h_values, dtype=float).copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _c5(best_h)
    best_n = n0

    # Phase 1: Projected gradient with sweeping alpha at native n
    alphas = [2000, 5000, 15000, 40000, 100000, 250000]
    h1 = best_h.copy()
    h1, c1 = _projected_grad_descent(h1, alphas, 300, tl)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: SLSQP polish
    if tl() > 120:
        h2 = _slsqp_fast(best_h, 80000, 1500)
        c2 = _c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Fine projected gradient
    if tl() > 150:
        h3 = best_h.copy()
        h3, c3 = _projected_grad_descent(h3, [50000, 120000, 300000, 600000], 250, tl)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Multi-n exploration
    if tl() > 200:
        for n_try in [400, 500, 700, 800]:
            if tl() < 80:
                break
            hr = _project(_resample(best_h, n_try), n_try / 2.0)
            hr, cr = _projected_grad_descent(hr, [5000, 20000, 80000], 250, tl)
            if tl() > 30:
                hr = _slsqp_fast(hr, 60000, 1000)
            cr = _c5(hr)
            if cr < best_c5:
                best_c5, best_h, best_n = cr, hr, n_try

    # Phase 5: Multi-start Fourier at best_n
    if tl() > 200:
        for K in [4, 7, 10, 14]:
            if tl() < 60:
                break
            for _tr in range(3):
                hf = _fourier_init(K, best_n, rng)
                hf, cf = _projected_grad_descent(hf, [3000, 10000, 50000], 200, tl)
                if tl() > 20:
                    hf = _slsqp_fast(hf, 50000, 800)
                cf = _c5(hf)
                if cf < best_c5:
                    best_c5, best_h = cf, hf.copy()

    # Phase 6: Final SLSQP at high alpha
    if tl() > 100:
        h6 = _slsqp_fast(best_h, 200000, 2000)
        c6 = _c5(h6)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    # Phase 7: Ultra-fine projected gradient
    if tl() > 60:
        h7 = best_h.copy()
        h7, c7 = _projected_grad_descent(h7, [200000, 500000, 1000000, 2000000], 150, tl)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    # Phase 8: Coordinate descent
    if tl() > 40:
        h8 = best_h.copy()
        h8, c8 = _coord_descent(h8, 3 * best_n, tl)
        if c8 < best_c5:
            best_c5, best_h = c8, h8.copy()

    # Phase 9: Final polish
    if tl() > 15:
        h9 = _slsqp_fast(best_h, 500000, 500)
        c9 = _c5(h9)
        if c9 < best_c5:
            best_c5, best_h = c9, h9
        h9 = best_h.copy()
        h9, c9 = _projected_grad_descent(h9, [500000, 1000000], 80, tl)
        if c9 < best_c5:
            best_c5, best_h = c9, h9

    best_h = _project(best_h, best_n / 2.0)
    best_c5 = _c5(best_h)
    return (best_h, best_c5, best_n)
