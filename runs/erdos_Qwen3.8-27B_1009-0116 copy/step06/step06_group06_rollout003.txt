def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):

        else:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(h))

def _soft_grad(h, alpha, topK):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    w = np.exp(-alpha * (m - c) / max(1.0, alpha))
    if w.sum() < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w.sum()
    idx = np.argsort(w)[::-1][:topK]
    g = np.zeros(n)
    dx = 2.0 / n
    for k in idx:
        wk = w[k]
        if wk < 1e-12:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    return g

def _grad_desc(h, alpha, n_steps, t_end, topK):
    n = len(h)
    h_cur = h.copy()
    f_prev = _soft_obj(h_cur, alpha)
    v = np.zeros(n)
    mu = 0.85
    lr = 0.03
    for _s in range(n_steps):
        if time.time() > t_end:
            break
        g = _soft_grad(h_cur, alpha, topK)
        v = mu * v + (1.0 - mu) * g
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = v / (np.linalg.norm(v) + 1e-30)
        accepted = False
        for _bt in range(10):
            h_new = _proj(h_cur - lr * v, n * 0.5)
            f_new = _soft_obj(h_new, alpha)
            if f_new <= f_prev + 1e-14:
                h_cur = h_new
                f_prev = f_new
                accepted = True
                lr = min(lr * 1.08, 0.4)
                break
            lr *= 0.5
        if not accepted:
            v *= 0.5
            lr *= 0.4
    return h_cur

def _slsqp_opt(h, alpha, maxiter, t_end, topK):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def c_fun(x):
        return x.sum() - n * 0.5
    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    f0 = _soft_obj(h, alpha)
    try:
        res = minimize(_soft_obj, h, jac=lambda x: _soft_grad(x, alpha, topK),
                       args=(), method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        if res.fun < f0 - 1e-13:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _lag_grad_single(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _multi_lag(h, n_rounds, t_end, n_top):

    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(c)[::-1][:n_top]
        cmax = c[top_k[0]]
        w = np.exp(-80.0 * (cmax - c[top_k]))
        w /= w.sum()
        g = np.zeros(n)
        for i in range(n_top):
            g += w[i] * _lag_grad_single(h_cur, top_k[i])
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn
        improved = False
        for step in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5_cur

def _gauss_smooth(h, sigma):
    n = len(h)
    rad = int(4 * sigma)
    kernel = np.zeros(n)
    for d in range(-rad, rad + 1):
        kernel[d % n] = np.exp(-d * d / (2.0 * sigma * sigma))
    kernel /= kernel.sum()
    h_f = np.fft.rfft(h)
    k_f = np.fft.rfft(kernel)
    return np.fft.irfft(h_f * k_f, n=n)

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _sa_opt(h, t_end, rng, n_iter):
    n = len(h)
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = 0.008 * max(0.05, 1.0 - i / max(1, n_iter))
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.003 + 0.025 * t)
        oi1, oi2 = h[i1], h[i2]
        h[i1], h[i2] = oi1 + d, oi2 - d
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        else:
            h[i1], h[i2] = oi1, oi2
            h = _repair(h, n)
    return best_h, best_c
