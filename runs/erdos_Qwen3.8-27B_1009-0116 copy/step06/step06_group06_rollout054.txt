import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        rem = abs(diff)
        for idx in order:
            if diff > 0:
                a = min(rem, 1.0 - h[idx])
                h[idx] += a
                rem -= a
            else:
                a = min(rem, h[idx])
                h[idx] -= a
                rem -= a
            if rem < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    lag = k - (n - 1)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n - lag])

    else:
        l2 = -lag
        g[:n - l2] += (1.0 - h[l2:])
        g[l2:] -= h[:n - l2]
    return g

def _smootheval(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    s = np.clip(alpha * (c - cm), -500, 500)
    s -= s.max()
    return (cm + np.log(np.sum(np.exp(s))) / alpha) * 2.0 / n

def _smoothgrad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    s = np.clip(alpha * (c - cm), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        g += wk * _lag_grad(h, k)
    return g * dx

def _lbfgs_opt(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def cfun(x):
        return x.sum() - n * 0.5

    def cjac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(_smootheval, h, jac=_smoothgrad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-16,
                                'maxls': 50, 'disp': False})
        if res.fun < _smootheval(h, alpha) - 1e-12:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _nesterov(h, n_steps, t_end, lr0=0.03, mom=0.9, thresh=0.85):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c = _eval(h_cur)
    for s in range(n_steps):
        if time.time() > t_end:
            break
        t = s + 1
        y = h_cur + mom * (h_cur - h_prev)
        y = np.clip(y, 0, 1)
        c = np.correlate(y, 1.0 - y, mode='full') * dx
        cm = c.max()
        mask = c > thresh * cm
        cnt = mask.sum()
        if cnt == 0:
            mask[np.argmax(c)] = True
            cnt = 1
        g = np.zeros(n)
        for k in np.where(mask)[0]:
            g += _lag_grad(y, k)
        g *= dx / cnt
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        lr = lr0 / (t ** 0.5)
        h_new = _proj(y - lr * g / gn, n * 0.5)
        h_prev = h_cur.copy()
        h_cur = h_new
        c_cur = _eval(h_cur)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
    return best_h, best_c

        if _rem() < 200:
            break
        h_c = _lbfgs_opt(best_h, alpha, 4000, t0 + _rem() - 195)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c

    # Phase 2: Nesterov subgradient
    if _rem() > 160:
        h2, c2 = _nesterov(best_h, 1000, t0 + _rem() - 155, lr0=0.025, mom=0.9, thresh=0.82)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 3: Top-m subgradient
    if _rem() > 140:
        h3, c3 = _topm_subgrad(best_h, 600, t0 + _rem() - 135, m=10)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3

    # Phase 4: High-alpha L-BFGS
    if _rem() > 110:
        h4 = _lbfgs_opt(best_h, 500000, 6000, t0 + _rem() - 105)
        c5 = _eval(h4)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h4

    # Phase 5: Multi-restart with perturbations
    if _rem() > 180:
        scales = [0.04, 0.08, 0.03, 0.12, 0.06]
        for sc in scales:
            if _rem() < 40:
                break
            h_r = _perturb_restart(best_h, best_n, rng, sc)
            h_r = _lbfgs_opt(h_r, 30000, 2000, t0 + _rem() - 35)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r
                best_n = best_n

    # Phase 6: Nesterov again after restarts
    if _rem() > 100:
        h5, c5 = _nesterov(best_h, 600, t0 + _rem() - 95, lr0=0.015, mom=0.92, thresh=0.75)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h5

    # Phase 7: Upsample to 2x
    if _rem() > 350:
        n1 = n0 * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha in [500, 5000, 30000, 150000]:
            if _rem() < 300:
                break
            h_hi = _lbfgs_opt(h_hi, alpha, 3000, t0 + _rem() - 295)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi
                best_n = n1
        if _rem() > 250:
            h6, c6 = _nesterov(best_h, 500, t0 + _rem() - 245, lr0=0.015, mom=0.9, thresh=0.85)
            if c6 < best_c5:
                best_c5 = c6
                best_h = h6
                best_n = len(best_h)
        if _rem() > 200:
            h7 = _lbfgs_opt(best_h, 400000, 4000, t0 + _rem() - 195)
            c5 = _eval(h7)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h7
                best_n = len(best_h)

    # Phase 8: Final L-BFGS polish
    if _rem() > 60:
        h8 = _lbfgs_opt(best_h, 800000, 6000, t0 + _rem() - 55)
        c5 = _eval(h8)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h8

    # Phase 9: Top-m fine subgradient
    if _rem() > 40:
        h9, c9 = _topm_subgrad(best_h, 200, t0 + _rem() - 35, m=12)
        if c9 < best_c5:
            best_c5 = c9
            best_h = h9

    # Phase 10: Coordinate descent
    if _rem() > 20:
        h_cd = best_h.copy()
        nc = len(h_cd)
        c_cd = _eval(h_cd)
        n_t = min(int((_rem() - 15) / 0.001), 40000)
        for _ in range(n_t):
            if _rem() < 15:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 5e-5 * rng.randn()
            h_cd[i] += d
            h_cd[j] -= d
            h_cd = _repair(h_cd, nc)
            c_n = _eval(h_cd)
            if c_n < c_cd:
                c_cd = c_n
                if c_n < best_c5:
                    best_c5 = c_n
                    best_h = h_cd.copy()
            else:
                h_cd = best_h.copy()
                c_cd = best_c5
                h_cd[i] -= d
                h_cd[j] += d
                h_cd = _repair(h_cd, nc)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
