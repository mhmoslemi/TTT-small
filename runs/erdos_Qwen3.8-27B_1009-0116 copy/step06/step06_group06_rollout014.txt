import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-15:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, r):
    """Gradient of correlate(h,1-h,'full')[r] w.r.t. h. Fixed version."""
    n = len(h)
    s = r - (n - 1)  # shift
    g = np.zeros(n)
    if s > 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    elif s < 0:
        m = -s
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def _smooth_max_obj(h, alpha, pen):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    smax = (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)
    cv = (h.sum() - 0.5 * n) * 1.0 / n
    return smax + pen * cv * cv

def _smooth_max_grad(h, alpha, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        g += w[r] * _grad_lag(h, r)
    g *= dx
    cv = (h.sum() - 0.5 * n) * 1.0 / n
    g += 2.0 * pen * cv / n
    return g

def _lbfgs(h, alpha, pen, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_max_obj, h.copy(), jac=_smooth_max_grad,
                       args=(alpha, pen), method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'bfactor': 1e-10, 'disp': False, 'maxfun': maxiter})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_nesterov(h, n_iter, t_end, alpha_top=20, thresh=0.005):
    n = len(h)
    dx = 2.0 / n
    h_col = h.copy()
    h_prev = h.copy()
    c5_cur = _eval(h_col)
    best_h = h_col.copy()
    best_c5 = c5_cur
    step = 0.03
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h_col, 1.0 - h_col, mode='full')
        cmax = c.max()
        active = np.where(c >= cmax - thresh)[0]
        if len(active) == 0:
            idx = np.argsort(c)[::-1][:alpha_top]
            active = idx
        g = np.zeros(n)
        for r in active:
            g += _grad_lag(h_col, r)
        g *= dx / max(1.0, float(len(active)))
        # Nesterov momentum
        h_new = h_col - step * g
        h_new = _proj(h_new, 0.5 * n)
        # Line search
        improved = False
        for s_factor in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            h_try = _proj(h_col - step * s_factor * g, 0.5 * n)
            c5_try = _eval(h_try)
            if c5_try < c5_cur - 1e-16:
                h_cur_backup = h_col.copy()
                h_col = h_try
                c5_cur = c5_try
                improved = True
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_col.copy()
                step = min(step * 1.02, 0.15)
                break
        if not improved:
            step = max(step * 0.9, 0.0005)
    return best_h, best_c5

def _pair_cd(h, n_iter, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 1e-4 * rng.randn()
        hi, hj = h_cur[i], h_cur[j]
        ni, nj = hi + d, hj - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _eval(h_cur)
        if c5n < c5_cur - 1e-17:
            c5_cur = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = hi, hj
    return best_h, best_c5

    h_hi = np.repeat(h, factor)
