            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
        else:
            ss = -s
            g[:n-ss] += (1.0 - h[ss:])
            g[ss:] -= h[:n-ss]
    g -= g.mean()
    return g, c, idx

def _grad_desc(h, K, n_iters, time_limit, t_start):
    c5 = _c5(h)
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        g, c, top_idx = _topK_grad(h, K)
        # Adaptive step size
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _repair(h - step * g, len(h))
            c5_new = _c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                break
    return h, c5

def _topK_lssm(h, K, alpha):
    """Smooth objective: alpha * (max + soft top-K)"""
    c = _corr(h)
    m = c.max()
    # Soft top-K: sum of exp(alpha*c) / sum of all... 
    # Actually just use: m + log(sum(exp(alpha*(c-m))))/alpha
    # This is the log-sum-exp smoothing of max
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _topK_lssm_grad(h, K, alpha):
    c = _corr(h)
    n = len(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15: continue
        sh = (n-1) - k
        if sh == 0:
            g += w[k] * (1.0 - 2.0*h)
        elif sh > 0:
            g[sh:] += w[k] * (1.0 - h[:n-sh])
            g[:n-sh] -= w[k] * h[sh:]
        else:
            sp = -sh
            g[:n-sp] += w[k] * (1.0 - h[sp:])
            g[sp:] -= w[k] * h[:n-sp]
    g -= g.mean()
    return g

def _slsqp_obj(h, alpha):
    return _topK_lssm(h, 0, alpha)

def _slsqp_jac(h, alpha):
    return _topK_lssm_grad(h, 0, alpha)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_jac, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _anneal_continuous(h, t0_temp, t1_temp, n_steps, time_limit, t_start, rng):
    c5 = _c5(h)
    n = len(h)
    for step in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        t = t0_temp * (t1_temp / t0_temp) ** (step / max(n_steps-1, 1))
        # Random perturbation: adjust a few coordinates
        n_pert = rng.randint(2, 8)
        idx = rng.choice(n, n_pert, replace=False)
        delta = rng.randn(n_pert) * 0.05
        h_new = h.copy()
        h_new[idx] += delta
        h_new = _repair(h_new, n)
        c5_new = _c5(h_new)
        if c5_new < c5 or rng.rand() < np.exp(-(c5_new - c5) / max(t, 1e-8)):
            h = h_new
            c5 = c5_new
    return h, c5
