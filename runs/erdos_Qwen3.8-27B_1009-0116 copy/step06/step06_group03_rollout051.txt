            i = rng.randint(n - 1); j = i + 1
        elif sh > 0:
            if n - sh <= 0: continue
            i = rng.randint(n - sh); j = i + sh
        else:
            sp = -sh
            if n - sp <= 0: continue
            j = rng.randint(n - sp); i = j + sp
        d = step * rng.randn()
        ni = h[i] + d; nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        oi, oj = h[i], h[j]
