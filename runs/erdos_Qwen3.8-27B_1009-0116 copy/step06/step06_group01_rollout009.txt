    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    y = h.copy()
    t_k = 1.0
    best = h.copy()

    for i in range(iters):

        g = _lse_grad(y, alpha)
        # Lipschitz estimate
        gn = np.linalg.norm(g)

        L = gn + 1.0  # crude Lipschitz

        # FISTA update

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _lse_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    if diff > 0:
        for _ in range(60):
            mid = (lo + hi) / 2.0
            s = np.clip(h - mid, 0.0, 1.0).sum() - target
            if s > 0:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    else:
        for _ in range(60):
            mid = (lo + hi) / 2.0
            s = np.clip(h + mid, 0.0, 1.0).sum() - target
            if s < 0:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    d2 = target - h.sum()
    if abs(d2) > 1e-14:
        h += d2 / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _momentum_descent(h, n, iters, alpha_start, alpha_end, mu=0.85, lr_init=0.04):
    best = h.copy()
    best_c5 = _eval(h)
    x = h.copy()
    v = np.zeros(n)
    t_k = 1.0
    for i in range(iters):
        alpha = alpha_start * (alpha_end / alpha_start) ** (i / max(iters - 1, 1))
        g = _lse_grad(x, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        lr = lr_init / (1.0 + 0.001 * i)
        y = x + ((t_k - 1.0) / (t_k + 1.0)) * (x - (x - v))
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x + ((t_k - 1.0) / t_next) * v
        v = np.zeros(n)
        gv = _lse_grad(y, alpha)
        for shrink in range(10):
            x_new = _proj(y - lr * gv, n)
            fn = _eval(x_new)
            if fn < _eval(y) - 1e-16 or shrink == 0:
                break
            lr *= 0.5
        if fn < best_c5 - 1e-15:
            best_c5 = fn
            best = x_new.copy()
        v = mu * v + (lr * gv)
        x = x_new
        t_k = t_next
    return best, best_c5

def _focused_descent(h, n, max_rounds, max_K=10):
    c5c = _eval(h)
    dx = 2.0 / n
    for r in range(max_rounds):
        ov = np.correlate(h, 1.0 - h, mode='full')
        K = min(2 + r // 25, max_K)
        n_top = min(K, len(ov))
        top_idx = np.argsort(ov)[-n_top:]
        g = np.zeros(n)
        for kk in top_idx:
            g += _grad_lag(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        best_c5, best_h = c5c, h
        step = 0.04
        for _ in range(22):
            h_new = _proj(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < best_c5 - 1e-15:
                best_c5, best_h = c5n, h_new
            else:
                step *= 0.5
            if step < 1e-9:
                break
        if best_c5 >= c5c - 1e-15:
            g1 = _grad_lag(h, top_idx[-1], dx)
            gn1 = np.linalg.norm(g1)
            if gn1 < 1e-14:
                break
            g1 /= gn1
            step = 0.04
            for _ in range(22):
                h_new = _proj(h - step * g1, n)
                c5n = _eval(h_new)
                if c5n < best_c5 - 1e-15:
                    best_c5, best_h = c5n, h_new
                else:
                    step *= 0.5
                if step < 1e-9:
                    break
            if best_c5 >= c5c - 1e-15:
                break
        h = best_h
        c5c = best_c5
    return h, c5c

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _sq(x):
        return x.sum() - len(x) * 0.5
    def _sj(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': _sq, 'jac': _sj}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _coord_cd(h, n, rng, n_iter, scales):
    c5c = _eval(h)
    for scale in scales:
        for _ in range(n_iter):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c
