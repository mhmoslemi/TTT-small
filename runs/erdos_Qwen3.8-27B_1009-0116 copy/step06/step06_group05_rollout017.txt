        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

                    h[idx] -= a

def _eval(h):

def _proj_feature(h, k, n):

    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m) * dx, -500, 500)
    sm = s - s.max()
    return m * dx + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _sm_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m) * dx, -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _sc_fun(x, n):
    return x.sum() - n * 0.5

def _sc_jac(x, n):
    return np.ones(n)

def _slsqp_smooth(h, n, alpha, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc_fun, 'jac': _sc_jac, 'args': (n,)}]
    args = (n, alpha)

        res = minimize(_sm_obj, h, jac=_sm_grad, args=args,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
