    hp2 = hp.copy()
    hp2 = _lbfgs_T(hp2, T, n0, [500000, 5000000], 1e6, 3000, 0.22, t0, budget_s)
    hp2 = _repair(hp2, T)
    c5 = _eval(np.tile(hp2, m))
    if c5 < best_c5:
        best_c5 = c5; best_h = np.tile(hp2, m).copy(); best_n = n0
        hp = hp2.copy()

    # Phase 3: Nesterov PGD exploration on T=31
    hp3 = hp.copy()
    hp3, c5 = _nesterov(hp3, T, n0, [20000, 200000, 2000000], 6000, 0.003, seed)
    if c5 < best_c5:
        best_c5 = c5; best_h = np.tile(hp3, m).copy(); best_n = n0
        hp = hp3.copy()
    # Re-optimize the best with L-BFGS-B after PGD
    hp_best = hp.copy()
    hp_best = _lbfgs_T(hp_best, T, n0, [100000, 1000000], 1e6, 2000, 0.35, t0, budget_s)
    hp_best = _repair(hp_best, T)
    c5 = _eval(np.tile(hp_best, m))
    if c5 < best_c5:
        best_c5 = c5; best_h = np.tile(hp_best, m).copy(); best_n = n0
        hp = hp_best.copy()

    # Phase 4: Try T=62
    if elapsed() < budget_s * 0.55:
        T62 = 62
        if n0 % T62 == 0:
            m62 = n0 // T62
            hp62 = _proj_sum(best_h.reshape(m62, T62).mean(axis=0), T62)
            hp62 = _lbfgs_T(hp62, T62, n0, [10000, 100000, 500000], 5e5, 2000, 0.55, t0, budget_s)
            hp62 = _repair(hp62, T62)
            c5 = _eval(np.tile(hp62, m62))
            if c5 < best_c5:
                best_c5 = c5; best_h = np.tile(hp62, m62).copy(); best_n = n0

    # Phase 5: Full L-BFGS-B on n=310
    if elapsed() < budget_s * 0.72:
        for alpha, mu in [(50000, 5e4), (500000, 5e5)]:
            if elapsed() > budget_s * 0.73:
                break
            try:
                hf = best_h[:].copy()
                res = minimize(_obj_full, hf, jac=_jac_full, args=(alpha, n0, n0, mu),
                               method='L-BFGS-B', bounds=[(0.0, 1.0)] * n0,
                               options={'maxiter': 1500, 'ftol': 1e-19})
                hf = _repair(res.x, n0)
                c5 = _eval(hf)
                if c5 < best_c5:
                    best_c5 = c5; best_h = hf.copy(); best_n = n0
            except Exception:
                pass

    # Phase 6: Perturbation + re-optimize
    if elapsed() < budget_s * 0.82:
        for trial in range(5):
            if elapsed() > budget_s * 0.83:
                break
            scale = 0.03 + 0.02 * trial
            hp_p = hp + scale * rng.randn(T)
            hp_p = _proj_sum(hp_p, T)
            hp_p = _lbfgs_T(hp_p, T, n0, [100000, 1000000], 1e6, 1500, 0.84, t0, budget_s)
            hp_p = _repair(hp_p, T)
            c5 = _eval(np.tile(hp_p, m))
            if c5 < best_c5:
                best_c5 = c5; best_h = np.tile(hp_p, m).copy(); best_n = n0
                hp = hp_p.copy()

    # Phase 7: Coordinate descent polish (multi-scale)
    if elapsed() < budget_s - 15:
        for sc in [0.008, 0.003, 0.001, 0.0004]:
            if elapsed() > budget_s - 8:
                break
            h9, c5 = _coord(best_h.copy(), 4000, sc, seed + 200 + int(sc * 10000))
            if c5 < best_c5:
                best_c5 = c5; best_h = h9.copy()

    # Phase 8: Try nearby n
    if elapsed() < budget_s - 25:
        for dn in [-10, 10, -31, 31]:
            if elapsed() > budget_s - 12:
                break
            nt = best_n + dn
            if nt < 100 or nt > 500:
                continue
            hr = _repair(_resample(best_h, nt), nt)
            try:
                res = minimize(_obj_full, hr, jac=_jac_full, args=(50000, nt, nt, 5e4),
                               method='L-BFGS-B', bounds=[(0.0, 1.0)] * nt,
                               options={'maxiter': 800, 'ftol': 1e-18})
                hr = _repair(res.x, nt)
            except Exception:
                pass
            c5 = _eval(hr)
            if c5 < best_c5:
                best_c5 = c5; best_h = hr.copy(); best_n = nt

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
