import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-12:
        h = h * tgt / s
        h = np.clip(h, 0.0, 1.0)
    diff = tgt - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _proj(h, tgt):
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / a) * (2.0 / n)

def _sm_grad(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    nz = np.where(w > 1e-13)[0]
    for k in nz:
        wk = w[k]
        sh = (n - 1) - k

            if sh > 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    return g

    v = _sm_val(h, a)
    return v + mu * (h.sum() - n * 0.5) ** 2

    g = _sm_grad(h, a)
    g += 2.0 * mu * (h.sum() - n * 0.5) * np.ones(n)
    return g

    n = len(h)

    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:

        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:

        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
