import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lssm_obj(h, alpha, beta):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    lse = m + (float(np.sum(np.exp(sm))) + s.max()) / alpha
    pen = beta * (h.sum() - n / 2.0) ** 2
    return lse + pen

def _lssm_grad(h, alpha, beta):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2.0 * beta * (h.sum() - n / 2.0)
    return g

def _topk_obj(h, K, beta):
    n = len(h)
    c = _corr(h)
    topk = np.sort(c)[-K:]
    f = topk.mean() + 0.1 * (c.max() - topk.mean())
    pen = beta * (h.sum() - n / 2.0) ** 2
    return f + pen

def _topk_grad(h, K, beta):
    n = len(h)
    c = _corr(h)
    topk_idx = np.argsort(c)[-K:]
    w = np.zeros(2 * n - 1)
    w[topk_idx] = 1.0 / K
    # Also add weight to the max

def _lbgfgs(h, obj_fn, jac_fn, maxiter):
    n = len(h)
