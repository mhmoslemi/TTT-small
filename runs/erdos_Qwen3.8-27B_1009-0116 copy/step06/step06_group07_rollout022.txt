def _proj(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

        else:
            hi = mid

    n = len(h)

def _coord_descent(h, n_iter, scales, seed=0):

    n = len(h)

    """Optimize periodic pattern using minimize with trust-constr or SLSQP"""
