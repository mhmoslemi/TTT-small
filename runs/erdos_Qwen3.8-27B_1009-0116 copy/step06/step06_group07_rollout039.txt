import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12: break
    return h

def _obj(p, T, m, alpha, pen):
    h = np.tile(p, m)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    ex = alpha * (c - cm)
    ex -= ex.max()
    ss = p.sum() - T * 0.5
    return cm + np.log(np.sum(np.exp(ex))) / alpha + pen * ss * ss

def _grad(p, T, m, alpha, pen):
    n = T * m
    h = np.tile(p, m)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    ex = alpha * (c - cm)
    ex -= ex.max()
    w = np.exp(ex); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * 2.0 / n
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    gp = np.zeros(T)
    for i in range(T):
        gp[i] = g[i::T].sum()
    ss = p.sum() - T * 0.5
    return gp + 2.0 * pen * ss

    x = np.clip(x, 0.0, 1.0)
    return _eval(np.tile(x, m)) + 1e8 * (x.sum() - T * 0.5) ** 2

                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * T,
                           options={'maxiter': 400, 'ftol': 1e-22})
                best = _repair(r.x, T)
            except Exception:
                pass
            c5 = _eval(np.tile(best, m))
            if c5 < bc:
                bc = c5
    return best, bc

def _opt_slsqp(p0, T, m):
    best = p0.copy()
    bc = _eval(np.tile(best, m))
    for alpha in [5000, 30000, 150000, 600000]:
        for pen in [1e5, 1e8, 1e10]:
