```python
def _softmax_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sc = np.clip(alpha * (c - m), -500, 500)
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        offset = (n - 1) - k
        if offset >= 0:
            # c[k] = sum_{i=0}^{n-1-offset} h[i] * (1-h[i+offset])
            # ∂c/∂h[j] = (1-h[j+offset])*[j<=n-1-offset] - h[j-offset]*[j>=offset]
            if offset == 0:
                g[:] += wk * (2.0 * h[:] - 1.0)
            else:
                # For j in [0, n-1-offset]: (1-h[j+offset])
                g[:n-offset] += wk * (1.0 - h[offset:])
                # For j in [offset, n-1]: -h[j-offset]
                g[offset:] -= wk * h[:n-offset]
        else:
            p = -offset  # p = k - (n-1)
            # c[k] = sum_{i=p}^{n-1} h[i] * (1-h[i-p])
            # ∂c/∂h[j] = (1-h[j-p])*[j>=p] - h[j+p]*[j<=n-1-p]
            g[p:] += wk * (1.0 - h[:n-p])
            g[:n-p] -= wk * h[p:]
    
    g *= dx
    return g
```

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _softmax_obj(h, alpha, lam):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sc = np.clip(alpha * (c - m), -500, 500)
    sc -= sc.max()
    return (m + np.log(np.sum(np.exp(sc))) / alpha) * 2.0 / n + lam * (h.sum() - n / 2) ** 2

def _softmax_grad(h, alpha, lam):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sc = np.clip(alpha * (c - m), -500, 500)
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        off = (n - 1) - k
        if off >= 0:
            if off == 0:
                g += wk * (2.0 * h - 1.0)
            else:
                g[:n - off] += wk * (1.0 - h[off:])
                g[off:] -= wk * h[:n - off]
        else:
            p = -off
            g[p:] += wk * (1.0 - h[:n - p])
            g[:n - p] -= wk * h[p:]
    g *= dx
    g += 2.0 * lam * (h.sum() - n / 2)
    return g

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _lbfgs(h, alpha, lam, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_softmax_obj, h, jac=_softmax_grad, args=(alpha, lam),
                       method='L-BFGS-B', bounds=bnds, maxiter=maxiter,
                       options={'ftol': 1e-17, 'gtol': 1e-15, 'maxls': 40})
        if time.time() <= t_end:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _proj_grad(h, alpha, n_steps, t_end):
    n = len(h)
    f_prev = _softmax_obj(h, alpha, 0.0)
    h_arr = h.copy()
    v = np.zeros(n)
    mu = 0.75
    step = 0.015
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        g = _softmax_grad(h_arr, alpha, 0.0)
        v = mu * v + (1.0 - mu) * g
        accepted = False
        for _t in range(10):
            h_new = _proj(h_arr - step * v, n * 0.5)
            f_new = _softmax_obj(h_new, alpha, 0.0)
            if f_new <= f_prev + 1e-14:
                h_arr = h_new
                f_prev = f_new
                step = min(step * 1.04, 0.3)
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step *= 0.3
    return h_arr

def _upsample(h, factor):
    return np.repeat(h, factor)

def _fourier_perturb(h, rng, amp, n_pert):
    n = len(h)
    H = np.fft.rfft(h.astype(np.float64))
    half = n // 2 + 1
    idx = rng.choice(range(1, half), size=min(n_pert, half - 1), replace=False)
    H[idx] += amp * (rng.randn(len(idx)) + 1j * rng.randn(len(idx)))
    out = np.fft.irfft(H, n)
    return _repair(out, n)

def _coord_descent(h, n_tries, rng, t_end, scale=3e-5):
    n = len(h)
    c5c = _eval(h)
    h_cur = h.copy()
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _eval(h_cur)
        if c5n < c5c - 1e-17:
            c5c = c5n
        else:
            h_cur[i], h_cur[j] = ni - d, nj + d
    return h_cur, c5c

def _coarse_search(n_coarse, rng, t_end, n_gen=80, pop=25):
    from scipy.optimize import differential_evolution
    def obj(x):
        hx = _repair(x, n_coarse)
        return _eval(hx)
    bnds = [(0.0, 1.0)] * n_coarse
    try:
        res = differential_evolution(obj, bnds, seed=rng.randint(10000),
                                     maxiter=n_gen, popsize=pop, tol=1e-12,
                                     polish=False, init='latinhypercube')
        if time.time() <= t_end:
            return _repair(res.x, n_coarse), _eval(_repair(res.x, n_coarse))
    except Exception:
        pass
    return None, np.inf
