def _periodic_obj(h_p, T, alpha):
    """Soft-max objective for periodic problem"""
    A = np.zeros(T)
    for r in range(T):
        A[r] = np.sum(h_p * (1 - np.roll(h_p, r)))
    m = A.max()
    s = np.clip(alpha * (A - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _periodic_grad(h_p, T, alpha):

    for r in range(T):
        A[r] = np.sum(h_p * (1 - np.roll(h_p, r)))
    m = A.max()
    s = np.clip(alpha * (A - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(T)
    for r in range(T):
        if w[r] < 1e-15:
            continue
        # dA_r/dh[j] = (1 - h[(j+r) % T]) - 1 (from the -h[j]*h[(j+r)%T] term... wait)
        # A_r = sum_j h[j]*(1-h[(j+r)%T]) = sum_j h[j] - sum_j h[j]*h[(j+r)%T]
        # dA_r/dh[j] = 1 - h[(j+r)%T] - h[(j-r)%T]  (the second term from when j appears in the second factor)
        # Actually: d/dh[j] [sum_i h[i]*h[(i+r)%T]] = h[(j-r)%T] + (if r=0: 2*h[j], else 0... no)
        # Let me be more careful:

        #   i=j: h[j]*h[(j+r)%T] → derivative: h[(j+r)%T]

def _periodic_grad(h_p, T, alpha):

    for r in range(T):
        A[r] = np.dot(h_p, 1 - np.roll(h_p, r))

    def _full_grad(x, a):
        n2 = len(x)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(a * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx2 = 2.0 / n2
        g = np.zeros(n2)
        for k in range(2 * n2 - 1):
            if w[k] < 1e-12:
                continue
            sh = (n2 - 1) - k
            wk = w[k] * dx2
            if sh >= 0:
                i1 = np.arange(n2 - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n2)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n2)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n2 - sp)
                g[i2] -= wk * x[i2 + sp]
        return g

    def _full_eq(x, n3):
        return x.sum() - n3 * 0.5

    def _full_jac(x, n3):
        return np.ones(n3)

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _full_eq, 'jac': _full_jac, 'args': (n,)}]
    try:
        res = minimize(_full_obj, h0, jac=_full_grad, args=(alpha,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval_full(best_h)
    best_n = n0

    # Phase 1: Optimize T=31 periodic pattern (n0=310, m=10)
    T1, m1 = 31, n0 // 31
    h_p = h.reshape(m1, T1).mean(axis=0)
    h_p = _proj_sum(h_p, T1)
    t1_end = t0 + budget_s * 0.20
    h_p, c5 = _optimize_T(h_p, T1, m1, t1_end, seed)
    if c5 < best_c5:
        best_c5 = c5
        best_h = np.tile(h_p, m1).copy()
        best_n = n0

    # Phase 2: Subgradient refinement on T=31
    if elapsed() < budget_s * 0.35:
        t2_end = t0 + budget_s * 0.35
        h_p2, c5_2 = _subgrad_periodic(h_p, T1, 0.01, 5000, t2_end, seed + 1)
        if c5_2 < best_c5:
            h_p = h_p2
            best_c5 = c5_2
            best_h = np.tile(h_p, m1).copy()
        # Re-optimize from subgradient result
        if elapsed() < budget_s * 0.40:
            h_p3, c5_3 = _optimize_T(h_p2, T1, m1, t0 + budget_s * 0.40, seed + 2)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = np.tile(h_p3, m1).copy()
                h_p = h_p3

    # Phase 3: Try T=31 with m=14 (n=434) for better accuracy
    m_alt = 14
    n_alt = T1 * m_alt
    if elapsed() < budget_s * 0.55:
        t3_end = t0 + budget_s * 0.55
        h_alt, c5_alt = _optimize_T(h_p, T1, m_alt, t3_end, seed + 3)
        c5_alt_full = _eval_full(np.tile(h_alt, m_alt))
        if c5_alt_full < best_c5:
            best_c5 = c5_alt_full
            best_h = np.tile(h_alt, m_alt).copy()
            best_n = n_alt

    # Phase 4: Try other periods T
    for T_try in [37, 43, 47, 53, 61, 71]:
        if elapsed() > budget_s * 0.70:
            break
        m_try = max(4, (500 // T_try) * 2)
        m_try = max(m_try, 4)
        if m_try * T_try > 900:
            m_try = 900 // T_try
        if m_try < 4:
            continue
        h_init_T = np.zeros(T_try)
        h_init_T[:T_try // 2] = 1.0
        h_init_T[T_try // 2:] = 0.0
        h_init_T = _repair(h_init_T + 0.3 * rng.randn(T_try), T_try)
        t4_end = t0 + budget_s * 0.70
        h_T, c5_T = _optimize_T(h_init_T, T_try, m_try, t4_end, seed + T_try)
        if c5_T < best_c5:
            best_c5 = c5_T
            best_h = np.tile(h_T, m_try).copy()
            best_n = m_try * T_try

    # Phase 5: Full-resolution SLSQP on best solution
    if elapsed() < budget_s * 0.85:
        for alpha_f in [50000, 200000]:
            if elapsed() > budget_s * 0.85:
                break
            h_new = _slsqp_full(best_h.copy(), alpha_f, 3000)
            c5_new = _eval_full(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = len(h_new)

    # Phase 6: Coordinate descent polish
    if elapsed() < budget_s - 15:
        for scale in [0.003, 0.001, 0.0004]:
            if elapsed() > budget_s - 5:
                break
            h_cd, c5_cd = _coord_descent(best_h.copy(), best_n, 4000, scale, seed + 200)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Final repair and evaluation
    best_h = _repair(best_h, best_n)
    best_c5 = _eval_full(best_h)
    return (best_h, best_c5, best_n)
