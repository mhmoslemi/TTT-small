def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _pnorm_obj(x, p):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / len(x))
    return float(np.sum(c**p) ** (1.0/p))

def _pnorm_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cp = c**p
    sp = np.sum(cp)
    if sp < 1e-30:
        return np.zeros(n)
    # d/dx (sum c^p)^(1/p) = (1/p) * (sum c^p)^(1/p - 1) * p * sum(c^(p-1) * dc/dx)
    # = (sum c^p)^(1/p - 1) * sum(c^(p-1) * dc/dx)
    factor = sp**(1.0/p - 1.0)
    # dc_k/dx for each lag
    g = np.zeros(n)
    for k in range(2*n - 1):
        ck = c[k]
        if ck < 1e-15:
            continue
        cpk = ck**(p-1)
        sh = (n-1) - k
        if sh >= 0:
            if sh < n:
                i1 = np.arange(n - sh)
                g[i1] += cpk * (1.0 - x[i1+sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= cpk * x[i2-sh]
        else:

            if sp_ < n:
                i1 = np.arange(sp_, n)
                g[i1] += cpk * (1.0 - x[i1-sp_])
                i2 = np.arange(n - sp_)
                g[i2] -= cpk * x[i2+sp_]

            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _lsexp_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_lag(x, k, dx)
    return g

def _pnorm_obj(x, p):
    c = _corr(x)
    return float(np.sum(c**p) ** (1.0 / p))

def _pnorm_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    sp = np.sum(c**p)
    if sp < 1e-30:
        return np.zeros(n)
    factor = sp ** (1.0 / p - 1.0)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        ck = c[k]
        if ck < 1e-15:
            continue
        w = ck ** (p - 1)
        g += w * _grad_lag(x, k, dx)
    return factor * g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_smax(h, alpha, maxiter, ftol, use_pnorm=False, p=30):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    if use_pnorm:
        try:
            res = minimize(_pnorm_obj, h, jac=_pnorm_grad, args=(p,),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            return _repair(res.x, n)
        except Exception:
            return _repair(h, n)
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _line_search(h, g, nc, c5c, max_shrinks=30):
    best_c5 = c5c
    best_h = h
    step = 0.1
    for _ in range(max_shrinks):
        h_new = _repair(h - step * g, nc)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
            best_h = h_new
        else:
            step *= 0.5
    return best_h, best_c5

def _active_descent(h, nc, max_rounds, top_m=5):
    c5c = _eval(h)
    dx = 2.0 / nc
    for r in range(max_rounds):
        ov = _corr(h)
        n_top = min(top_m, len(ov))
        top_idx = np.argsort(ov)[-n_top:]
        g = np.zeros(nc)
        for kk in top_idx:
            g += _grad_lag(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h_new, c5n = _line_search(h, g, nc, c5c)
        if c5n >= c5c - 1e-15:
            g1 = _grad_lag(h, top_idx[-1], dx)
            gn1 = np.linalg.norm(g1)
            if gn1 < 1e-14:
                break
            g1 /= gn1
            h_new, c5n = _line_search(h, g1, nc, c5c)
            if c5n >= c5c - 1e-15:
                break
        h = h_new
        c5c = c5n
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _fourier_eval(theta, n, K):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(K):
        a = theta[2 * j]
        b = theta[2 * j + 1]
        h += a * np.cos((j + 1) * np.pi * x) + b * np.sin((j + 1) * np.pi * x)
    h = _repair(h, n)
    return h

def _fourier_obj(theta, n, K):
    h = _fourier_eval(theta, n, K)
    return _eval(h)

def _pairwise_cd(h, nc, rng, scale, n_iter):
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        h = _repair(h, nc)
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i] = ni - d
            h[j] = nj + d
            h = _repair(h, nc)
    return h, c5c

def _symmetry_penalty(h, n):
    """Encourage h[i] + h[(i+n//2)%n] ≈ 1"""
    half = n // 2
    diff = h[:half] + h[half:] - 1.0
    return np.sum(diff**2)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP smooth-max polish
    if tl() > 300:
        h1 = _slsqp_smax(best_h, 12000, 2500, 1e-18)
        c5 = _eval(h1)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: Active-set descent (top 5)
    if tl() > 300:
        h2 = best_h.copy()
        h2, c5 = _active_descent(h2, best_n, 500, top_m=5)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 3: p-norm SLSQP
    if tl() > 250:
        for p in [20, 50]:
            if tl() < 80:
                break
            h3 = _slsqp_smax(best_h, 0, 2000, 1e-18, use_pnorm=True, p=p)
            c5 = _eval(h3)
            if c5 < best_c5:
                best_c5, best_h = c5, h3.copy()

    # Phase 4: Another active descent round
    if tl() > 250:
        h4 = best_h.copy()
        h4, c5 = _active_descent(h4, best_n, 600, top_m=7)
        if c5 < best_c5:
            best_c5, best_h = c5, h4.copy()

    # Phase 5: Fourier global search via dual_annealing
    if tl() > 350:
        K = 15
        dim = 2 * K
        bounds_fa = [(-0.4, 0.4)] * dim
        n_ft = best_n
        try:
            def fa_obj_wrapper(theta):
                return _fourier_obj(theta, n_ft, K)
            fa_res = dual_annealing(fa_obj_wrapper, bounds_fa, maxiter=200,
                                    seed=seed, initial_temp=1.0, max_eval=600)
            if fa_res.fun < best_c5 - 1e-6:
                h_f = _fourier_eval(fa_res.x, n_ft, K)
                c5 = _eval(h_f)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_f.copy()
                    # Polish
                    h_f2 = _slsqp_smax(best_h, 50000, 1500, 1e-18)
                    c5 = _eval(h_f2)
                    if c5 < best_c5:
                        best_c5, best_h = c5, h_f2.copy()
                    h_f2, c5 = _active_descent(h_f2, n_ft, 300, top_m=5)
                    if c5 < best_c5:
                        best_c5, best_h = c5, h_f2.copy()
        except Exception:
            pass

    # Phase 6: Multi-resolution search
    if tl() > 300:
        n_list = [150, 250, 350, 500]
        if best_n not in n_list:
            n_list.append(best_n)
        for n_try in n_list:
            if tl() < 80:
                break
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r = _slsqp_smax(h_r, 15000, 1500, 1e-17)
            h_r, c5 = _active_descent(h_r, n_try, 250, top_m=5)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try
                h_r = _slsqp_smax(best_h, 80000, 1000, 1e-18)
                c5 = _eval(h_r)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_r.copy()
                h_r, c5 = _active_descent(h_r, n_try, 200, top_m=5)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_r.copy()

    # Phase 7: Perturbation + re-optimize
    if tl() > 200:
        for _rep in range(4):
            if tl() < 60:
                break
            noise = rng.randn(best_n) * 0.015
            h_p = _repair(best_h + noise, best_n)
            h_p, c5 = _active_descent(h_p, best_n, 300, top_m=5)
            h_p = _slsqp_smax(h_p, 100000, 1200, 1e-18)
            c5 = _eval(h_p)
            h_p, c5 = _active_descent(h_p, best_n, 250, top_m=7)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()

    # Phase 8: Binary search on small grid then upsample
    if tl() > 200:
        for n_bb in [60, 100, 150]:
            if tl() < 70:
                break
            h_bb = _repair(_resample(best_h, n_bb), n_bb)
            h_bb = np.where(h_bb > 0.5, 1.0, 0.0)
            h_bb = _repair(h_bb, n_bb)
            c5c = _eval(h_bb)
            stuck = 0
            for _it in range(20000):
                if tl() < 20:
                    break
                i = rng.randint(n_bb)
                h_bb[i] = 1.0 - h_bb[i]
                h_t = _repair(h_bb, n_bb)
                c5n = _eval(h_t)
                if c5n < c5c - 1e-14:
                    c5c = c5n
                    h_bb = h_t
                    stuck = 0
                else:
                    h_bb[i] = 1.0 - h_bb[i]
                    stuck += 1
                    if stuck > 5000:
                        break
            if c5c < best_c5 + 0.005:
                h_up = _repair(_resample(h_bb, best_n), best_n)
                h_up = _slsqp_smax(h_up, 30000, 2000, 1e-17)
                h_up, c5 = _active_descent(h_up, best_n, 300, top_m=5)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_up.copy()

    # Phase 9: Final ultra-fine pairwise CD
    if tl() > 20:
        h9 = best_h.copy()
        nc = len(h9)
        for scale in [0.002, 0.0005, 0.0001]:
            if tl() < 8:
                break
            h9, c5c = _pairwise_cd(h9, nc, rng, scale, 5000)
            if c5c < best_c5:
                best_c5, best_h = c5c, h9.copy()

    # Phase 10: Final active descent
    if tl() > 10:
        h_fin, c5 = _active_descent(best_h.copy(), best_n, 150, top_m=5)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
