            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _slsqp_pass(h, alpha, maxiter, t_end):

    bnds = [(0.0, 1.0)] * n

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if time.time() > t_end:

        return _repair(res.x, n)
    except Exception:

def _targeted_grad_descent(h, n, t_end, top_m=4, n_iter=5000):
    # Focus gradient on top-m worst shifts only
    cur = _eval(h)
    h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-top_m:]
        wm = np.zeros(2 * n - 1)
        wm[idx] = 1.0
        wm /= wm.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if wm[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = wm[k] * dx
            if sh >= 0:
                i1 = np.arange(max(0, n - sh))
                g[i1] += wk * (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * h[i2 + sp]
        moved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002]:
            ht = _repair(h - step * g, n)
            ct = _eval(ht)
            if ct < cur - 1e-15:
                h = ht
                cur = ct
                moved = True
                break
        if not moved:
            break
    return h, cur

def _binary_search(h, n, t_end, max_rounds=500):
    # Binary local search: try all (i,j) swaps
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    cur = _eval(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        # For each 1-position, try moving to each 0-position
        for i in ones:
            if time.time() > t_end:
                break
            h[i] = 0.0
            # For each 0-position as target
            for j in zeros:
                h[j] = 1.0
                c2 = _eval(h)
                if c2 < cur - 1e-14:
                    cur = c2
                    improved = True
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    break
                h[j] = 0.0
            h[i] = 1.0
            if improved:
                break
        if not improved:
            break
    return h, cur

    xn = np.linspace(0, 2.0, n_new, endpoint=False)

        cur_max = ov.max()
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        improved = False
        best_gain = 1e-15
        best_pair = None
        for i in ones:
            if time.time() > t_end:
                break
            vi = np.where((i - np.arange(-(n-1), n) >= 0) & (i - np.arange(-(n-1), n) < n),
                         h[(i - np.arange(-(n-1), n))[(i - np.arange(-(n-1), n)>=0) & (i - np.arange(-(n-1), n)<n)]], 0.0) + \
                 np.where((i + np.arange(-(n-1), n) >= 0) & (i + np.arange(-(n-1), n) < n),
                         h[(i + np.arange(-(n-1), n))[(i + np.arange(-(n-1), n)>=0) & (i + np.arange(-(n-1), n)<n)]], 0.0)
            for j in zeros:
                sj = np.arange(-(n-1), n)
                vj = np.where((j - sj >= 0) & (j - sj < n),
                             h[(j - sj)[(j-sj>=0)&(j-sj<n)]], 0.0) + \
                     np.where((j + sj >= 0) & (j + sj < n),
                             h[(j + sj)[(j+sj>=0)&(j+sj<n)]], 0.0)
                delta = (vi - vj) * dx
                new_max = np.max(ov + delta)
                if new_max < cur_max - 1e-14:
                    gain = cur_max - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0; h[j] = 1.0
            cur = _eval(h)
            improved = True
        if not improved:
            break
    return h, cur

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end_total = t0 + budget_s
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def rem():
        return t_end_total - time.time()

    # Phase 1: SLSQP polish at original n
    for alpha in [8000, 40000, 150000]:
        if rem() < 180:
            break
        hn = _slsqp_pass(best_h, alpha, 2500, t0 + rem() - 160)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5; best_h = hn.copy()

    # Phase 2: Targeted GD focusing on top 3 worst shifts
    if rem() > 120:
        h2 = best_h.copy()
        h2, c5 = _targeted_gd(h2, len(h2), t0 + rem() - 100, top_m=3, n_iter=8000)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2.copy()

    # Phase 3: Binary local search at several resolutions
    for nb in [best_n, 400, 200, 300, 250, 350, 280]:
        if rem() < 90:
            break
        if nb < 80 or nb > 800 or nb % 2 == 1:
            nb += 1
        if nb == 1:
            continue
        if len(best_h) == nb:
            hb = (best_h > 0.5).astype(float)
        else:
            hb = (_resample(best_h, nb) > 0.5).astype(float)
        hb = _repair(hb, nb)
        hb, c5 = _binary_local_search(hb, nb, t0 + rem() - 70, max_rounds=400)
        if c5 < best_c5:
            best_c5 = c5; best_h = hb.copy(); best_n = nb

    # Phase 4: Continuous refinement of best binary
    if rem() > 70:
        h4 = best_h.copy()
        for alpha in [30000, 100000, 400000]:
            if rem() < 60:
                break
            h4 = _slsqp_pass(h4, alpha, 2000, t0 + rem() - 50)
        c5 = _eval(h4)
        if c5 < best_c5:
            best_c5 = c5; best_h = h4.copy(); best_n = len(h4)

    # Phase 5: Multi-resolution with targeted GD
    for n_try in [100, 140, 180, 220, 260, 340, 380, 420, 480, 520, 560, 640, 800]:
        if rem() < 60:
            break
        hr = _repair(_resample(best_h, n_try), n_try)
        if rem() > 55:
            hr, c5 = _targeted_gd(hr, n_try, t0 + rem() - 50, top_m=3, n_iter=2000)
            if c5 < best_c5:
                best_c5 = c5; best_h = hr.copy(); best_n = n_try

    # Phase 6: Another binary round at current best n
    if rem() > 50:
        hb = (best_h > 0.5).astype(float)
        hb = _repair(hb, best_n)
        hb, c5 = _binary_local_search(hb, best_n, t0 + rem() - 35, max_rounds=300)
        if c5 < best_c5:
            best_c5 = c5; best_h = hb.copy()

    # Phase 7: Final SLSQP + targeted GD
    if rem() > 20:
        hf = _slsqp_pass(best_h, 300000, 4000, t0 + rem() - 15)
        c5 = _eval(hf)
        if c5 < best_c5:
            best_c5 = c5; best_h = hf.copy(); best_n = len(hf)
        if rem() > 10:
            hf2, c5 = _targeted_gd(best_h.copy(), len(best_h), t0 + rem() - 5, top_m=2, n_iter=3000)
            if c5 < best_c5:
                best_c5 = c5; best_h = hf2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
