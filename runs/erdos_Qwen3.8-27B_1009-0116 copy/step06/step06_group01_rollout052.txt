import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def compute_overlaps(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

    return h

def lse_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def lse_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13: continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh); g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n); g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh; i1 = np.arange(sp, n); g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp); g[i2] -= wk * x[i2 + sp]
    return g

def sum_eq(x): return x.sum() - len(x) * 0.5
def sum_jac(x): return np.ones(len(x))

def slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}]
    try:
        res = minimize(lse_obj, h, jac=lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

def resample(h_old, n_new):

    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

    return repair(h, n)

            if tl() < 80: break
            h_r = repair(resample(best_h, n_try), n_try)
            h_r = slsqp_opt(h_r, 30000, 2000, 1e-17)
            h_r, c5 = coord_descent(h_r, n_try, 100, rng, tl)
            h_r, c5 = fine_coord_descent(h_r, n_try, 50, rng, tl, [0.002, 0.0005])
            h_r, c5 = active_grad_descent(h_r, n_try, 150, tl, top_m=3)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 7: Fourier multi-start
    if tl() > 250:
        for K in [3, 5, 8]:
            for tr in range(3):
                if tl() < 60: break
                h_f = fourier_init(K, best_n, rng)
                h_f = slsqp_opt(h_f, 20000, 1500, 1e-16)
                h_f, c5 = coord_descent(h_f, best_n, 80, rng, tl)
                h_f, c5 = fine_coord_descent(h_f, best_n, 40, rng, tl, [0.002, 0.0005])
                if c5 < best_c5: best_c5, best_h = c5, h_f

    # Phase 8: Perturbation + re-optimize
    if tl() > 150:
        for rep in range(4):
            if tl() < 50: break
            noise = rng.randn(best_n) * 0.008
            h_p = repair(best_h + noise, best_n)
            h_p, c5 = coord_descent(h_p, best_n, 80, rng, tl)
            h_p = slsqp_opt(h_p, 80000, 2000, 1e-18)
            h_p, c5 = fine_coord_descent(h_p, best_n, 40, rng, tl, [0.001, 0.0003])
            h_p, c5 = active_grad_descent(h_p, best_n, 100, tl)
            if c5 < best_c5: best_c5, best_h = c5, h_p

    # Phase 9: Final ultra-fine pass
    if tl() > 10:
        h_fin, c5 = fine_coord_descent(best_h.copy(), best_n, 50, rng, tl,
                                       [0.0001, 0.00003, 0.000005])
        if c5 < best_c5: best_c5, best_h = c5, h_fin

    # Phase 10: One last active grad
    if tl() > 5:
        h_fin2, c5 = active_grad_descent(best_h.copy(), best_n, 50, tl)
        if c5 < best_c5: best_c5, best_h = c5, h_fin2

    best_h = repair(best_h, best_n)
    best_c5 = eval_c5(best_h)
    return (best_h, best_c5, best_n)
