    step = 0.06
    for _ in range(max_shrinks):
        h_new = _repair(h - step * g, nc)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-15:
            best_c5, best_h = c5n, h_new
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return best_h, best_c5

def _adaptive_descent(h, nc, max_rounds, time_limit):
    c5c, dx = _eval(h), 2.0 / nc
    t_start = time.time()
    for r in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        ov = np.correlate(h, 1.0 - h, mode='full')
        top5 = np.argsort(ov)[-5:]
        vals = ov[top5]
        n_top = 3 if (vals[-1] - vals[0]) < 0.002 * max(vals[-1], 1e-10) else 1
        top_idx = top5[-n_top:]
        g = sum(_grad_lag(h, kk, dx) for kk in top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h_new, c5n = _line_search(h, g, nc, c5c)
        if c5n >= c5c - 1e-15:
            improved = False
            for kk in top5:
                g1 = _grad_lag(h, kk, dx)
                if np.linalg.norm(g1) < 1e-14:
                    continue
                g1 /= np.linalg.norm(g1)
                h_try, c5_try = _line_search(h, g1, nc, c5c)
                if c5_try < c5c - 1e-15:
                    h, c5c, improved = h_try, c5_try, True
                    break
            if not improved:
                break
        else:
            h, c5c = h_new, c5n
    return h, c5c

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0 and sh < n:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        elif sh < 0 and -sh < n:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter=3000, ftol=1e-17):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _cyclic_cd(h, nc, rng, scale, n_iter, t_start, t_limit):
    c5c = _eval(h)
    for it in range(n_iter):
        if time.time() - t_start > t_limit:
            break
        i = it % nc
        d = scale * rng.randn()
        h[i] += d
        h = _repair(h, nc)
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i] -= d
            h = _repair(h, nc)
    return h, c5c

def _fourier_init(K, n, rng, amp=0.25):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        h += rng.uniform(-amp, amp) * np.cos(np.pi * j * x)
        h += rng.uniform(-amp, amp) * np.sin(np.pi * j * x)
    return _repair(h, n)

def _hill_climb_small(h, nc, t_start, t_limit):
    c5c = _eval(h)
    improved = True
    while improved and time.time() - t_start < t_limit:
        improved = False
        for i in range(nc):
            if time.time() - t_start > t_limit:
                break
            old = h[i]
            h[i] = 1.0 - old
            h = _repair(h, nc)
            c5n = _eval(h)
            if c5n < c5c - 1e-14:
                c5c, improved = c5n, True
            else:
                h = _repair(np.concatenate([h[:i], h[i+1:]]), nc) if False else h
                h[i] = old
                h = _repair(h, nc)
            if time.time() - t_start > t_limit:
                break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: Adaptive descent on parent
    if tl() > 180:
        h1, c5 = _adaptive_descent(best_h.copy(), best_n, 600, 100)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: SLSQP with increasing alpha
    if tl() > 280:
        for alpha in [5000, 30000, 150000, 600000]:
            if tl() < 60:
                break
            h2 = _slsqp(best_h, alpha)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h = c5, h2.copy()

    # Phase 3: More adaptive descent
    if tl() > 200:
        h3, c5 = _adaptive_descent(best_h.copy(), best_n, 800, 120)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()

    # Phase 4: Multi-resolution
    if tl() > 300:
        for n_try in [80, 120, 160, 200, 280, 360, 500]:
            if tl() < 60:
                break
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r = _slsqp(h_r, 50000, 2500, 1e-17)
            h_r, c5 = _adaptive_descent(h_r, n_try, 400, 25)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 5: Fourier multi-start at small n
    if tl() > 280:
        for n_small in [60, 100, 150, 200, 300]:
            if tl() < 80:
                break
            for K in [2, 4, 6, 9]:
                if tl() < 40:
                    break
                for _tr in range(3):
                    h_f = _fourier_init(K, n_small, rng, amp=0.2)
                    h_f = _slsqp(h_f, 80000, 2500, 1e-17)
                    h_f, c5 = _adaptive_descent(h_f, n_small, 300, 12)
                    if c5 < best_c5:
                        best_c5, best_h, best_n = c5, h_f.copy(), n_small

    # Phase 6: Structured perturbation escape
    if tl() > 220:
        nc = best_n
        for _rep in range(8):
            if tl() < 40:
                break
            x = np.linspace(0, 2.0, nc, endpoint=False)
            pert = np.zeros(nc)
            for j in range(1, 5):
                f = j * rng.choice([1, 2, 3])
                amp = rng.uniform(0.003, 0.015)
                pert += amp * (rng.uniform(-1, 1) * np.cos(np.pi * f * x) +
                               rng.uniform(-1, 1) * np.sin(np.pi * f * x))
            h_p = _repair(best_h + pert, nc)
            h_p, c5 = _adaptive_descent(h_p, nc, 300, 15)
            h_p = _slsqp(h_p, 120000, 2000, 1e-17)
            h_p, c5 = _adaptive_descent(h_p, nc, 200, 12)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()

    # Phase 7: Cyclic CD at multiple scales
    if tl() > 100:
        nc = best_n
        h7 = best_h.copy()
        c5c = _eval(h7)
        for scale in [0.01, 0.004, 0.001, 0.0003, 0.00005]:
            if tl() < 15:
                break
            h7, c5c = _cyclic_cd(h7, nc, rng, scale, 6000, time.time(), 18)
            if c5c < best_c5:
                best_c5, best_h = c5c, h7.copy()

    # Phase 8: Hill climbing on small n
    if tl() > 80 and best_n <= 200:
        h8, c5 = _hill_climb_small(best_h.copy(), best_n, time.time(), 40)
        if c5 < best_c5:
            best_c5, best_h = c5, h8.copy()

    # Phase 9: Final SLSQP + descent
    if tl() > 40:
        h_fin = _slsqp(best_h, 300000, 3000, 1e-18)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()
    if tl() > 15:
        h_fin2, c5 = _adaptive_descent(best_h.copy(), best_n, 200, 10)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
