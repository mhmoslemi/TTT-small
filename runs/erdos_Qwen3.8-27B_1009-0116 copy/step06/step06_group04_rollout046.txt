    h_bin = _repair((best_h > 0.5).astype(float), best_n)
    x_b, c_b = _binary_swap(h_bin, best_n, rng, 80000, dl7)
    if c_b < best_c5:
        best_c5 = c_b; best_h = x_b.copy()
    x_p2, c_p2 = _adam(x_b, 2000, 100000, 1000, 0.004, dl7)
    if c_p2 < best_c5:
        best_c5 = c_p2; best_h = x_p2.copy()

    # Phase 8: Final Adam + L-BFGS + perturb (960-1085s)
    dl8 = t0 + 1080
    x, c = _adam(best_h, 50000, 2000000, 2500, 0.002, dl8)
    if c < best_c5:
        best_c5 = c; best_h = x.copy()
    if time.time() < dl8 - 30:
        h_fin = _lbfgs(best_h, 5000000, 2000)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_fin.copy()
    x_p3, c_p3 = _perturb(best_h, best_n, rng, 2000, dl8)
    if c_p3 < best_c5:
        best_c5 = c_p3; best_h = x_p3.copy()
    x_cd2, c_cd2 = _paired_cd(best_h, best_n, rng, dl8, 30000)
    if c_cd2 < best_c5:
        best_c5 = c_cd2; best_h = x_cd2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
