    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        # Bisection projection
        lo, hi = -2.0, 2.0
        for _ in range(60):
            mid = (lo + hi) * 0.5

            if s2 > target:

            else:
                hi = mid

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    return _proj(np.clip(h, 0.0, 1.0), n * 0.5)

def _lag_grad(h, sh, n):
    g = np.zeros(n)
    if sh >= 0:
        i = np.arange(n - sh)
        g[i] += 1.0 - h[i + sh]
        if sh > 0:
            j = np.arange(sh, n)
            g[j] -= h[j - sh]
    else:
        s = -sh
        i = np.arange(s, n)
        g[i] += 1.0 - h[i - s]
        j = np.arange(n - s)
        g[j] -= h[j + s]
    return g

def _grad_top(h, n_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-n_top:]
    g = np.zeros(n)
    for k in idx:
        g += _lag_grad(h, (n - 1) - k, n)
    return g * dx / n_top
