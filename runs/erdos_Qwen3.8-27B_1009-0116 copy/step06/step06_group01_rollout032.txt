import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_subgrad_nesterov(h, n_iters, alpha_init, alpha_decay, beta):
    """Projected subgradient with Nesterov momentum and softmax weighting."""
    n = len(h)
    dx = 2.0 / n
    y = h.copy()
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    for it in range(n_iters):
        c = _corr(y)
        # Softmax weights
        m = c.max()
        alpha = alpha_init * alpha_decay ** it
        s = np.clip(alpha * (c - m), -300, 300)
        w = np.exp(s)
        w /= w.sum()
        # Weighted subgradient
        g = np.zeros(n)
        # Compute gradient of weighted sum of correlations
        # c_k = sum over valid (i,j) pairs where j-i gives lag k
        # For mode='full' with vectors of length n:
        # c_k = sum_{i} a[i] * b[i + (n-1-k)] for valid indices
        # where a = h, b = 1-h
        # Actually let me just use the formula directly
        for k in range(2*n-1):
            if w[k] < 1e-10:
                continue
            sh = (n-1) - k
            wk = w[k]
            if sh >= 0:
                # c_k = sum_{i=0}^{n-1-sh} h[i] * (1-h[i+sh])
                idx = np.arange(n - sh)
                g[idx] += wk * (1.0 - h[idx + sh])
                if sh > 0:
                    g[sh:] -= wk * h[:n-sh]
            else:
                sp = -sh
                # c_k = sum_{i=0}^{n-1-sp} h[i+sp] * (1-h[i])
                g[sp:] += wk * (1.0 - h[:n-sp])
                g[:n-sp] -= wk * h[sp:]

        y = _repair(y - step * g, n)
        c5_new = _eval(y)
        h = (1+beta)/beta * y - 1.0/beta * h if it > 0 else y
        h = _repair(h, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = y.copy()
    return best_h, best_c5
