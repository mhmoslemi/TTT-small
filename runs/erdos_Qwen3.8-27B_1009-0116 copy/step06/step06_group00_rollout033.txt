```python
    """Compute g[k] = dc_k/dh[i] - dc_k/dh[j] for all lags k."""
    dx = 2.0 / n
    g = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        sh = k - (n - 1)
        # dc_k/dh[pos] = (1-h[pos+sh])*[0<=pos+sh<n] - h[pos-sh]*[0<=pos-sh<n]
        # For sh >= 0:
        #   dc/dh[p] = (1-h[p+sh]) if p+sh < n else 0, minus h[p-sh] if sh>0 and p-sh >= 0 else 0
        # For sh < 0, sp = -sh:
        #   dc/dh[p] = (1-h[p-sp]) if p-sp >= 0 else 0, minus h[p+sp] if p+sp < n else 0
        
        def _dc(dh_pos, sh):
            if sh >= 0:
                val = 0.0
                if dh_pos + sh < n: val += (1.0 - h[dh_pos + sh])
                if sh > 0 and dh_pos - sh >= 0: val -= h[dh_pos - sh]
            else:

                val = 0.0
                if dh_pos - sp >= 0: val += (1.0 - h[dh_pos - sp])
                if dh_pos + sp < n: val -= h[dh_pos + sp]
            return val
        g[k] = (_dc(i, sh) - _dc(j, sh)) * dx
    return g
```

```python
    sh = np.arange(2*n-1) - (n-1)  # shifts from -(n-1) to (n-1)

    # Vectorized:
    def _grad_at(p):

        # first term: p+sh in [0,n-1]
        idx1 = p + sh
        mask1 = (idx1 >= 0) & (idx1 < n)
```

```python
import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_at_pos(h, p, n):
    """Compute gradient of all c_k w.r.t. h[p], length 2n-1."""
    g = np.zeros(2 * n - 1)
    dx = 2.0 / n
    # First term: (1-h[p+sh]) for k in [max(0,n-1-p), min(2n-2, 2n-2-p)]
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    # Second term: -h[p-sh] for k in [max(0,p), min(2n-2, p+n-1)]
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * dx

def _min_max_1d(c, g, d_lo, d_hi, niter=50):
    """Minimize max_k(c_k + d*g_k) over d in [d_lo, d_hi]."""
    for _ in range(niter):
        d_mid = 0.5 * (d_lo + d_hi)
        vals = c + d_mid * g
        k_max = np.argmax(vals)
        gm = g[k_max]
        if gm > 1e-15:
            d_hi = d_mid
        elif gm < -1e-15:
            d_lo = d_mid
        else:
            break
    return 0.5 * (d_lo + d_hi)

def _pair_cd(h, n_iter, rng, el_fn, budget_s):
    """Pairwise coordinate descent with exact 1D min-max."""
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    no_improve = 0
    for it in range(n_iter):
        if el_fn() > budget_s - 15:
            break
        if it % 500 == 0 and no_improve > 2000:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        g = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi:

        d_opt = _min_max_1d(c, g, d_lo, d_hi)
        h[i] += d_opt
        h[j] -= d_opt
        c5_new = _eval(h)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            no_improve = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            no_improve += 1
    return best_h, best_c
```

For j=i+sh: ∂c_k/∂h[i] = 1 - h[j] - h[i-sh]
∂c_k/∂h[j] = 1 - h[j+sh] - h[j-sh] = 1 - h[j+sh] - h[i]
g_k = (1-h[j]-h[i-sh]) - (1-h[j+sh]-h[i]) = h[i] - h[j] - h[i-sh] + h[j+sh]

The linear part of Δc_k is d * g_k = d*(h[i]-h[j]-h[i-sh]+h[j+sh])

But I computed the linear part as -d(h[j]-h[i]) = d(h[i]-h[j]). 

These differ by d*(-h[i-sh]+h[j+sh]). The extra terms come from the non-interacting parts (h[i-sh] and h[j+sh] are unchanged, so their contribution to the gradient is still there).

OK so the full picture:
Δc_k = d * g_k + d² * q_k

where q_k = 1 if i and j interact at lag k (i.e., |i-j|=|sh_k| and the right one), else 0.

Specifically, q_k = 1 if |i-j| = |k-(n-1)| AND the correct directional condition holds.

For the 1D optimization, the objective is:
f(d) = max_k (c_k + d*g_k + d²*q_k)

This is the max of a set of quadratic functions (some linear where q_k=0, some quadratic where q_k=1). It's still convex in d (max of convex functions, and each quadratic with q_k=1 has positive coefficient d², so is convex).

So I can still use subgradient/bisection, just with the modified function.

For the subgradient at d:
f'(d) = g_{k*} + 2*d*q_{k*} where k* = argmax

- If subgradient > 0: d_hi = d_mid
- If subgradient < 0: d_lo = d_mid

```python
import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * (2.0 / n)

def _pair_cd(h, n_iter, rng, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 400 == 0 and stable > 3000:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        g = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        for _ in range(45):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < 2 * n - 1:
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if (kmax == k1 or kmax == k2):
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-13:
            stable += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < 2 * n - 1:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < 2 * n - 1:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < 2 * n - 1:
                c[k2] -= d_opt * d_opt
            stable += 1
    return best_h, best_c

def _subgrad_descent(h, rng, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    m_top = 3
    for it in range(5000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-m_top:]
        g = np.zeros(n)
        for k in top_k:
            sh = k - (n - 1)
            for p in range(n):
                val = 0.0
                if sh >= 0:
                    if p + sh < n:
                        val += (1.0 - h[p + sh])
                    if sh > 0 and p - sh >= 0:
                        val -= h[p - sh]
                else:
                    sp = -sh
                    if p - sp >= 0:
                        val += (1.0 - h[p - sp])
                    if p + sp < n:
                        val -= h[p + sp]
                g[p] += val
        g *= dx / m_top
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = min(0.01, 0.5 / gn)
        for s_mult in [1.0, 0.5, 0.2, 0.1, 0.05]:
            h_new = _proj_to_sum(h - s_mult * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                break
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
    return best_h, best_c

def _fine_polish(h, rng, t0, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(500000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-18:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Try perturbations for better start
    for ps in [0.03, 0.08, 0.15, 0.3]:
        hp = np.clip(h + ps * rng.randn(n0), 0, 1)
        hp = _proj_to_sum(hp, n0 * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 1: Subgradient (30s budget)
    dl1 = t0 + min(40, budget_s * 0.08)
    h1, c5_1 = _subgrad_descent(best_h.copy(), rng, t0, dl1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Pairwise CD - main optimization (largest budget)
    dl2 = t0 + min(budget_s * 0.65, budget_s - 120)
    h2, c5_2 = _pair_cd(best_h.copy(), 80000, rng, t0, dl2)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Another round of pairwise CD (different rng stream)
    dl3 = t0 + min(budget_s * 0.82, budget_s - 80)
    h3, c5_3 = _pair_cd(best_h.copy(), 60000, rng, t0, dl3)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Subgradient again on the improved solution
    dl4 = t0 + min(budget_s * 0.90, budget_s - 50)
    h4, c5_4 = _subgrad_descent(best_h.copy(), rng, t0, dl4)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Final fine polish
    dl5 = t0 + (budget_s - 5)
    h5, c5_5 = _fine_polish(best_h.copy(), rng, t0, dl5)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Final verification
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
```
