import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _obj_full(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad_full(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_opt(h0, alpha, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj_full, h0, jac=_grad_full, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _powell_periodic(h_p, T, m, n, fniters=200):
    # Powell's method on periodic pattern with sum constraint handled by repair
    def obj_p(p):
        h = _repair(np.tile(p, m), n)
        return _eval(h)
    h_p = h_p.copy()
    res = minimize(obj_p, h_p, method='Powell', bounds=[(0,1)]*T,
                   options={'maxiter': fniters, 'ftol': 1e-16})
    return _repair(res.x, T)

def _active_set_opt(h, n, alpha, n_active, maxiter):
    # Minimize max of top n_active correlations
    c = _corr(h)
    top_idx = np.argsort(c)[-n_active:]
    # Minimize sum of top correlations (proxy for max)
    def obj_act(x, idx):
        c = _corr(x)
        return np.sum(c[idx]) / len(idx)
    def grad_act(x, idx):
        n_l = len(x)
        g = np.zeros(n_l)
        dx = 2.0 / n_l
        for k in idx:
            sh = (n_l - 1) - k
            wk = dx / len(idx)
            if sh >= 0:
                i1 = np.arange(n_l - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n_l)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n_l)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n_l - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(lambda x: obj_act(x, top_idx), h,
                       jac=lambda x: grad_act(x, top_idx),
                       args=(), method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _pgd_period(h_p, T, n, alphas, max_iters, lr_init, seed=0):
    m = n // T
    rng = np.random.RandomState(seed)
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    lr = lr_init
    momentum = np.zeros(T)
    beta = 0.9
    for alpha in alphas:
        iters = max_iters // len(alphas)
        for it in range(iters):
            h_full = np.tile(h_p, m)
            c = _corr(h_full)
            mm = c.max()
            s = np.clip(alpha * (c - mm), -400, 400)
            w = np.exp(s - s.max())
            w /= w.sum()
            dx = 2.0 / n
            g_full = np.zeros(n)
            for k in range(len(w)):
                if w[k] < 1e-10:
                    continue
                sh = (n - 1) - k
                wk = w[k] * dx
                if sh >= 0:
                    i1 = np.arange(n - sh)
                    g_full[i1] += wk * (1.0 - h_full[i1 + sh])
                    if sh > 0:
                        g_full[i1 + sh] -= wk * h_full[i1]
                else:
                    sp = -sh
                    i2 = np.arange(sp, n)
                    g_full[i2] += wk * (1.0 - h_full[i2 - sp])
                    g_full[i2 - sp] -= wk * h_full[i2]
            g_p = np.zeros(T)
            for p in range(m):
                g_p += g_full[p * T:(p + 1) * T]
            g_p /= m
            momentum = beta * momentum + (1.0 - beta) * g_p
            h_new = np.clip(h_p - lr * momentum, 0, 1)
            # Project onto sum constraint
            target = T / 2.0
            d = target - h_new.sum()
            h_new += d / T
            h_new = np.clip(h_new, 0, 1)
            # Final repair for exact sum
            h_new = _repair(h_new, T)
            c_new = _eval(np.tile(h_new, m))
            if c_new < best_c - 1e-15:

                best = h_new.copy()
                h_p = h_new
                if lr < 0.05:
                    lr *= 1.02
            else:
                lr *= 0.7
                if lr < 1e-7:
                    lr = lr_init

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _coord_descent(h, n_iter, scales, seed=0):

    n = len(h)
    c5c = _eval(h)
    for scale in scales:
        for _ in range(n_iter):
            i = rng.randint(n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _soft_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _soft_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_opt(h0, alpha, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_soft_obj, h0, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

    for alpha in alphas:

        for _ in range(n_iter):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP on T=31 periodic (fast)
    if n0 % 31 == 0 and elapsed() < budget_s * 0.15:
        m = n0 // 31
        h_p = h.reshape(m, 31).mean(axis=0)
        h_p = _repair(h_p, 31)
        for alpha in [50000, 300000, 1500000, 5000000]:
            if elapsed() > budget_s * 0.15:
                break
            h_p = _slsqp_opt(h_p, alpha, 3000)
            c5 = _eval(np.tile(h_p, m))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_p, m).copy()

    # Phase 2: Adam PGD on T=31 periodic
    if n0 % 31 == 0 and elapsed() < budget_s * 0.30:
        m = n0 // 31
        h_p = best_h.reshape(m, 31).mean(axis=0) if best_n == n0 else _repair(_resample(best_h, n0).reshape(m, 31).mean(axis=0), 31)
        h_p = _repair(h_p, 31)
        h_p, c5 = _adam_pgd(h_p, 31, n0, [50000, 200000, 800000, 3000000], 5000, seed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.tile(h_p, m).copy()

    # Phase 3: Nelder-Mead on periodic pattern (derivative-free)
    if n0 % 31 == 0 and elapsed() < budget_s * 0.42:
        m = n0 // 31
        h_p = best_h.reshape(m, 31).mean(axis=0) if best_n == n0 else _repair(_resample(best_h, n0).reshape(m, 31).mean(axis=0), 31)
        h_p = _repair(h_p, 31)
        def obj_nm(p):
            return _eval(np.tile(_repair(p, 31), m))
        try:
            res = minimize(obj_nm, h_p, method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-14, 'fatol': 1e-14})
            h_p = _repair(res.x, 31)
            c5 = _eval(np.tile(h_p, m))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_p, m).copy()
        except Exception:
            pass

    # Phase 4: Full-resolution SLSQP
    if elapsed() < budget_s * 0.62:
        for alpha in [50000, 400000, 2000000]:
            if elapsed() > budget_s * 0.64:
                break
            h_new = _slsqp_opt(best_h, alpha, 2500)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(h_new)

    # Phase 5: Active-set refinement (min top-k correlations)
    if elapsed() < budget_s * 0.75:
        for n_act in [1, 3, 5, 8]:
            if elapsed() > budget_s * 0.77:
                break
            h_new = _active_opt(best_h, best_n, n_act, 2000)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 6: Try other periods / n values
    if elapsed() < budget_s * 0.85:
        for n_try, T_try in [(300, 30), (330, 33), (350, 35), (280, 28)]:
            if elapsed() > budget_s * 0.87:
                break
            h_r = _repair(_resample(best_h, n_try), n_try)
            if n_try % T_try == 0:
                m_t = n_try // T_try
                h_pt = h_r.reshape(m_t, T_try).mean(axis=0)
                h_pt = _repair(h_pt, T_try)
                h_pt = _slsqp_opt(h_pt, 500000, 2000)
                c5 = _eval(np.tile(h_pt, m_t))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = np.tile(h_pt, m_t).copy()
                    best_n = n_try

    # Phase 7: Random restart on periodic + SLSQP
    if n0 % 31 == 0 and elapsed() < budget_s * 0.92:
        m = n0 // 31
        for trial in range(3):
            if elapsed() > budget_s * 0.93:
                break
            h_pert = best_h[:31] + 0.03 * rng.randn(31)
            h_pert = _repair(h_pert, 31)
            h_pert = _slsqp_opt(h_pert, 1000000, 2000)
            c5 = _eval(np.tile(h_pert, m))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_pert, m).copy()

    # Phase 8: Coordinate descent polish
    if elapsed() < budget_s - 8:
        h_new, c5 = _coord_descent(best_h.copy(), 4000, [0.008, 0.002, 0.0004], seed + 77)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
