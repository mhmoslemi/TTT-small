    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(-(1.0 - h)) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:K]
    topk = c[idx]
    m = topk.max()
    s = alpha * (topk - m)
    sm = s - s.max()
    val = m + np.log(np.sum(np.exp(sm))) / alpha
    val += penalty * (h.sum() - n * 0.5) ** 2
    return val

    g += 2 * penalty * (h.sum() - n * 0.5)
    return g

                       args=(K, alpha, penalty), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(-(1.0 - h)) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_obj(x, K, alpha, mu):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(c)[::-1][:K]
    tk = c[idx]

    s = alpha * (tk - m)
    sm = s - s.max()

            if cvr < best_c5 - 1e-14:
                best_c5, best_h, best_n = cvr, hr.copy(), nt

    # Phase 4: Binary annealing + polish (150s)
    if el() < budget_s - 150:
        t_rem = budget_s - 150 - el()
        nsteps = max(int(t_rem * 1200), 2000)
        hb, cvb = _anneal_binary(best_h, nsteps, rng)
        hr = _opt(hb, 5, 200, 1.0, 500)
        cvr = _eval(hr)
        if cvr < best_c5 - 1e-14:
            best_c5, best_h = cvr, hr.copy()
        else:
            cvb2 = _eval(_opt(hb, 3, 500, 5.0, 400))
            if cvb2 < best_c5 - 1e-14:
                best_c5, best_h = cvb2, _opt(hb, 3, 500, 5.0, 400)

    # Phase 5: Aggressive final optimization (80s)
    if el() < budget_s - 30:
        for K, a, mu in [(3,5000,10.0),(5,2000,5.0),(10,500,2.0),(3,2000,10.0)]:
            if el() > budget_s - 30: break
            hx = _opt(best_h, K, a, mu, 600)
            cv = _eval(hx)
            if cv < best_c5 - 1e-14:
                best_c5, best_h = cv, hx.copy()
        hx, cv = _proj_desc(best_h, 3, 500, 200)
        if cv < best_c5 - 1e-14:
            best_c5, best_h = cv, hx.copy()
        # Try complement
        hc = _repair(1.0 - best_h, len(best_h))
        hc = _opt(hc, 5, 200, 1.0, 400)
        cvc = _eval(hc)
        if cvc < best_c5 - 1e-14:
            best_c5, best_h = cvc, hc.copy()

    # Phase 6: Diverse restarts (30s)
    if el() < budget_s - 5:
        for kd in range(4):
            if el() > budget_s - 5: break
            hi = _make_init(best_n, kd, rng)
            hi = _opt(hi, 5, 200, 1.0, 300)
            if _eval(hi) < best_c5 - 1e-14:
                best_c5, best_h = _eval(hi), hi.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
