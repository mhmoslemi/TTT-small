    h = h_init.copy()
    y = h.copy()
    target = n / 2.0
    dx = 2.0 / n
    cur_obj = _eval(h)
    L = alpha * dx * n  # rough Lipschitz constant
    if eta0 is None:
        eta = 1.0 / L
    else:
        eta = eta0
    
    for t in range(n_iter):
        if time.time() > t_end:
            break
        # Compute gradient at y
        c = np.correlate(y, 1.0 - y, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        ws = np.exp(s - s.max())
        ws /= ws.sum()
        
        # Gradient
        g = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if ws[k_idx] < 1e-14:
                continue
            k = k_idx - (n - 1)
            wk = ws[k_idx] * dx
            if k >= 0:
                if k > 0:
                    g[k:] += wk * (1.0 - y[:n-k])
                g[:n-k] -= wk * y[k:n]
            else:
                sp = -k
                g[:n-sp] += wk * (1.0 - y[sp:])
                g[sp:] -= wk * y[:n-sp]
        
        # Nesterov step
        if t == 0:
            beta = 0.0
        else:
            beta = (1.0 - np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        
        # Actually standard Nesterov: 
        # y_new = x_new + (t/(t+1+gamma)) * (x_new - x_prev)

        beta = 0.9  # fixed momentum

            # Backward extrapolation
            y = x_new + beta * (x_new - h)
            y = np.clip(y, 0, 1)
            h = x_new
            cur_obj = obj_new
        else:
            # Restart momentum
            y = x_new
            # Decrease step size
            eta *= 0.9

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(x, target):
    n = len(x)
    # Project onto {h: 0<=h<=1, sum(h)=target} using bisection on offset
    lo = x.min() - 0.1
    hi = x.max() + 0.1
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_lse(x, alpha, dx):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    ws = np.exp(s - s.max())
    ws /= ws.sum()
    # Compute gradient
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = ws[k_idx]
        if w < 1e-15:
            continue
        k = k_idx - (n - 1)
        wk = w * dx
        if k >= 0:
            if k > 0:
                g[k:n] += wk * (1.0 - x[0:n-k])
            if n - k > 0:
                g[0:n-k] -= wk * x[k:n]
        else:
            sp = -k
            g[0:n-sp] += wk * (1.0 - x[sp:n])
            if sp < n:
                g[sp:n] -= wk * x[0:n-sp]
    return g, ws

def _nesterov(h_init, alpha_start, t_end, n_restarts=3):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _eval(h)
    
    for restart in range(n_restarts):
        if time.time() > t_end - 30:
            break
        # Occasionally add noise
        if restart > 0:
            rng = np.random.RandomState(42 + restart)
            h = h + 0.02 * rng.randn(n)
            h = _proj(h, target)
        
        y = h.copy()
        eta = 5.0 / (alpha_start * dx * n)  # initial step size
        beta_mom = 0.85
        cur_c5 = _eval(h)
        
        for t in range(30000):
            if time.time() > t_end - 30:
                break
            g, ws = _grad_lse(y, alpha_start, dx)
            # GradNorm for step size
            gn = np.linalg.norm(g)
            if gn < 1e-12:
                break
            
            x_new = _proj(y - eta * g, target)
            c5_new = _eval(x_new)
            
            if c5_new < cur_c5 - 1e-15:
                # Accept
                y = x_new + beta_mom * (x_new - h)
                y = np.clip(y, 0, 1)
                h = x_new.copy()
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                # Occasionally increase momentum
                if t % 5000 == 0 and t > 0:
                    beta_mom = min(0.95, beta_mom + 0.05)
            else:
                # Reject, decrease step, restart momentum
                eta *= 0.8
                y = h.copy()
                if eta < 1e-10:
                    break
    
    return best_h, best_c5

def _run_slsqp(h, alpha, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s - s.max())))) / alpha
    
    def slsqp_h(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        ws = np.exp(s - s.max()); ws /= ws.sum()
        return _grad_lse(x, alpha, 2.0/n)[0]
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=slsqp_h, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _proj(res.x, n/2.0)
    except:
        return _proj(h, n/2.0)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _fourier_init(n, seed=0):
    """Initialize with Fourier modes"""
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.5)
    # Period 3 structure (frequency 3*n/2 in the n-point DFT over [0,2])
    # Actually, period 2/3 means frequency 3 in [0,2], i.e., 3 cycles in [-1,1] or 3 in [0,2]
    # DFT frequency: 3 cycles over the interval [0,2]
    for freq in [3, 6, 9]:
        amp = rng.uniform(0.1, 0.4) * (3.0/freq)
        phase = rng.uniform(0, 2*np.pi)
        j = np.arange(n)
        h += amp * np.cos(2*np.pi*freq*j/n + phase)
    h = _proj(h, n/2.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(x, target):
    n = len(x)
    lo = x.min() - 0.5
    hi = x.max() + 0.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_lse(x, alpha, dx, top_m=None):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    if top_m is not None and top_m < len(c):
        idx = np.argsort(c)[-top_m:]
        cmask = np.zeros(len(c))
        cmask[idx] = 1.0
        c = c * cmask + (1 - cmask) * (c.min() - 1.0)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    ws = np.exp(s - s.max())
    ws /= ws.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = ws[k_idx]
        if w < 1e-15:
            continue
        k = k_idx - (n - 1)
        wk = w * dx
        if k >= 0:
            if k > 0:
                g[k:n] += wk * (1.0 - x[0:n-k])
            g[0:n-k] -= wk * x[k:n]
        else:
            sp = -k
            g[0:n-sp] += wk * (1.0 - x[sp:n])
            g[sp:n] -= wk * x[0:n-sp]
    return g, ws, m + np.log(float(ws.sum() * np.exp(-(s - s.max()).min()))) / alpha

def _nesterov(h_init, alpha, t_end, top_m=None, max_steps=20000, noise_amp=0.0):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    if noise_amp > 0:
        rng = np.random.RandomState(int(alpha * 7 + n))
        h = _proj(h + noise_amp * rng.randn(n), target)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    y = h.copy()
    eta = 0.8 / (alpha * dx * n + 1.0)
    beta_mom = 0.8
    cur_c5 = best_c5
    stall = 0
    for t in range(max_steps):
        if time.time() > t_end:
            break
        g, ws, _ = _grad_lse(y, alpha, dx, top_m)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        x_new = _proj(y - eta * g, target)
        c5_new = _eval(x_new)
        if c5_new < cur_c5 - 1e-16:
            y = x_new + beta_mom * (x_new - h)
            y = np.clip(y, 0.0, 1.0)
            h = x_new.copy()
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            if t % 4000 == 0 and t > 0:
                beta_mom = min(0.93, beta_mom + 0.04)
                eta *= 1.1
        else:
            stall += 1
            eta *= 0.85
            y = h.copy()
            if stall > 200 or eta < 1e-12:
                break
    return best_h, best_c5

def _slsqp_polish(h, alpha, maxiter, n):
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return float(m + np.log(float(np.sum(np.exp(s - s.max())))) / alpha)
    def jac_fn(x):
        return _grad_lse(x, alpha, dx)[0]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=jac_fn, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _proj(res.x, n / 2.0)
    except Exception:
        return _proj(h, n / 2.0)

def _targeted_cd(h, n_iter, t_end, top_m=5):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _eval(h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-top_m:]
        improved = False
        for k_idx in idx:
            k = k_idx - (n - 1)
            if abs(k) >= n:
                continue
            if k >= 0:
                pairs = [(i, i + k) for i in range(n - k)]
            else:
                sp = -k
                pairs = [(i - sp, i) for i in range(sp, n)]
            if not pairs:
                continue
            contribs = np.array([h[a] * (1 - h[b]) for a, b in pairs])
            order = np.argsort(contribs)[::-1]
            for oi in order[:8]:
                a, b = pairs[oi]
                if h[a] > h[b]:
                    d = min(0.1, (h[a] - h[b]) * 0.5, h[a] - 1e-4)
                    if d < 1e-6:
                        continue
                    h[a] -= d; h[b] += d
                    c_new = _eval(h)
                    if c_new < cur - 1e-15:
                        cur = c_new
                        improved = True
                    else:
                        h[a] += d; h[b] -= d
                else:
                    d = min(0.1, (h[b] - h[a]) * 0.5, h[b] - 1e-4)
                    if d < 1e-6:
                        continue
                    h[b] -= d; h[a] += d
                    c_new = _eval(h)
                    if c_new < cur - 1e-15:
                        cur = c_new
                        improved = True
                    else:
                        h[b] += d; h[a] -= d
        if not improved:
            break
    h = _proj(h, target)
    return h, _eval(h)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    j = np.arange(n)
    h = np.full(n, 0.5)
    for freq in [3, 5, 7, 9]:
        amp = rng.uniform(0.15, 0.35) * (3.0 / freq)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * j / n + phase)
    return _proj(h, n / 2.0)
