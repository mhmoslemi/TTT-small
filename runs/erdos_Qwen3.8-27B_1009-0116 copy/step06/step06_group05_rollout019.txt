def _cgrad(h, lag, n):
    j = np.arange(n)
    sh = lag - (n - 1)
    g = np.zeros(n)
    i1 = j - sh
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += 1.0 - h[i1[m1]]
    i2 = j + sh
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max()) * 2.0 / n

def _eval_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lssm_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _lssm_jac(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        g += w[k] * dx * _cgrad(x, k, n)
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_eq_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    try:
        res = minimize(_lssm_obj, h, jac=_lssm_jac, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _project(res.x, n * 0.5)
    except Exception:
        return _project(h, n * 0.5)

def _heavyball(h0, n, T, M, eta0, mu, rng, deadline):

        corr = _eval_corr(h)

def _coord_descent(h, n_iter, scales, rng, deadline):
    n = len(h)
    h = h.copy()

    best = h.copy()
    best_c5 = c5c

            i = rng.randint(n)

            ni, nj = h[i] + d, h[j] - d

    # Phase 2: Heavy-ball subgradient (350s)
    deadline2 = t0 + 390
    h2 = best_h.copy()
    n2 = len(h2)
    h2, c5_2 = _heavyball(h2, n2, 500000, 4, 0.3, 0.9, rng, deadline2)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: SLSQP polish after subgradient (100s)
    deadline3 = deadline2 + 100
    for alpha in [50000, 500000, 5000000]:
        if elapsed() > deadline3:
            break
        h_new = _slsqp(best_h, alpha, 3000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5, best_h = c5, h_new.copy()
        # Complement
        h_comp = _project(1.0 - best_h, len(best_h) * 0.5)
        h_comp = _slsqp(h_comp, alpha, 1500, 1e-18)
        c5c = _eval(h_comp)
        if c5c < best_c5 - 1e-15:
            best_c5, best_h = c5c, h_comp.copy()

    # Phase 4: Worst-shift focused descent (80s)
    deadline4 = deadline3 + 80
    h4 = best_h.copy()
    c5_4 = best_c5
    for _rep in range(200):
        if elapsed() > deadline4:
            break
        h4, c5_4 = _worst_shift_step(h4, 5)
        if c5_4 < best_c5 - 1e-15:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Coordinate descent (80s)
    deadline5 = deadline4 + 80
    h5, c5_5 = _coord_descent(best_h, 20000, [0.002, 0.0005, 0.0001], rng, deadline5)
    if c5_5 < best_c5 - 1e-15:
        best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Multi-n exploration (150s)
    deadline6 = deadline5 + 150
    for n_try in [250, 300, 320, 350, 400]:
        if elapsed() > deadline6:
            break
        h_r = _project(_resample(best_h, n_try), n_try * 0.5)
        h_r = _slsqp(h_r, 100000, 1500, 1e-18)
        h_r, c5_r = _heavyball(h_r, n_try, 50000, 3, 0.2, 0.85, rng, min(deadline6, elapsed() + 25))
        c5_r = _eval(h_r)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 7: Diverse restarts with heavy-ball (100s)
    deadline7 = deadline6 + 100
    for pat in range(5):
        if elapsed() > deadline7:
            break
        hb = _make_diverse(best_n, pat, rng)
        hb = _slsqp(hb, 50000, 1000, 1e-17)
        hb, c5_h = _heavyball(hb, best_n, 30000, 4, 0.25, 0.9, rng, min(deadline7, elapsed() + 18))
        c5_h = _eval(hb)
        if c5_h < best_c5 - 1e-12:
            best_c5, best_h = c5_h, hb.copy()

    # Phase 8: Final SLSQP + coord descent polish (40s)
    if elapsed() < budget_s - 5:
        h8 = _slsqp(best_h, 5000000, 3000, 1e-20)
        c5_8 = _eval(h8)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8.copy()
        h8, c5_8 = _coord_descent(best_h, 15000, [0.001, 0.0003, 0.0001], rng, budget_s - 3)
        if c5_8 < best_c5 - 1e-15:
            best_c5, best_h = c5_8, h8.copy()

    best_h = _project(best_h, best_n * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
