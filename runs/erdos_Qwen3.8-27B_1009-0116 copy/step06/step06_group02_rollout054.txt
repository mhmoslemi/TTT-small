def _grad_topM(h, M, alpha=3.0):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[-M:]
    vals = c[top_idx]
    vmin, vmax = vals.min(), vals.max()
    if vmax - vmin > 1e-14:
        w = np.exp(alpha * (vals - vmin) / (vmax - vmin))
    else:
        w = np.ones(M)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(M):
        k = top_idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            i1 = np.arange(max(0, n - sh))
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]

            i1 = np.arange(max(0, n - sh))
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _obj_lse(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad_lse(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    nc = len(c)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(nc):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(max(0, n - sh))
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _con_sum(x):
    return x.sum() - _CON_N[0] * 0.5

def _con_sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    _CON_N[0] = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_sum, 'jac': _con_sum_jac}]
    try:
        res = minimize(_obj_lse, h, jac=_grad_lse, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _proj_grad(h, n, n_iter, M, alpha, t_end):
    h = h.copy()
    cur = _eval(h)
    step = 0.04
    stalled = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = _grad_topM(h, M, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.05, step * 0.01]:
            h_try = _repair(h - s * g, n)
            val = _eval(h_try)
            if val < cur - 1e-14:
                h = h_try
                cur = val
                improved = True
                break
        if improved:
            stalled = 0
            step = min(step * 1.08, 0.15)
        else:
            stalled += 1
            if stalled > 60:
                break
            step *= 0.5
    return h, cur

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _perturb(h, n, scale, rng):
    h = h.copy()
    h += rng.randn(n) * scale
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Projected gradient, M=1 (focus on worst shift)
    if _remaining(t0, budget_s) > 180:
        h1, c1 = _proj_grad(best_h, n0, 30000, 1, 1.0, t0 + _remaining(t0, budget_s) - 160)
        if c1 < best_c5:
            best_c5 = c1; best_h = h1.copy()

    # Phase 2: Projected gradient, M=3
    if _remaining(t0, budget_s) > 140:
        h2, c2 = _proj_grad(best_h, n0, 30000, 3, 3.0, t0 + _remaining(t0, budget_s) - 120)
        if c2 < best_c5:
            best_c5 = c2; best_h = h2.copy()

    # Phase 3: SLSQP with moderate alpha
    if _remaining(t0, budget_s) > 120:
        h3 = _slsqp(best_h, 30000, 3000, 1e-18)
        c3 = _eval(h3)
        if c3 < best_c5:
            best_c5 = c3; best_h = h3.copy()

    # Phase 4: Projected gradient, M=5, higher alpha
    if _remaining(t0, budget_s) > 100:
        h4, c4 = _proj_grad(best_h, n0, 20000, 5, 5.0, t0 + _remaining(t0, budget_s) - 80)
        if c4 < best_c5:
            best_c5 = c4; best_h = h4.copy()

    # Phase 5: SLSQP with high alpha
    if _remaining(t0, budget_s) > 100:
        h5 = _slsqp(best_h, 100000, 4000, 1e-19)
        c5 = _eval(h5)
        if c5 < best_c5:
            best_c5 = c5; best_h = h5.copy()

    # Phase 6: Multi-resolution search
    if _remaining(t0, budget_s) > 200:
        for n_try in [300, 400, 600, 800]:
            if _remaining(t0, budget_s) < 130:
                break
            hr = _repair(_resample(best_h, n_try), n_try)
            hr, cr = _proj_grad(hr, n_try, 10000, 5, 3.0, t0 + _remaining(t0, budget_s) - 110)
            if cr < best_c5:
                best_c5 = cr; best_h = hr.copy(); best_n = n_try
            if _remaining(t0, budget_s) > 60:
                hr = _slsqp(hr, 50000, 2000, 1e-18)
                cr = _eval(hr)
                if cr < best_c5:
                    best_c5 = cr; best_h = hr.copy(); best_n = n_try

    # Phase 7: Random perturbations to escape local optima
    if _remaining(t0, budget_s) > 120:
        for trial in range(6):
            if _remaining(t0, budget_s) < 80:
                break
            scale = 0.03 / (trial + 1)
            hp = _perturb(best_h, best_n, scale, rng)
            hp, cp = _proj_grad(hp, best_n, 8000, 3, 3.0, t0 + _remaining(t0, budget_s) - 60)
            if cp < best_c5:
                best_c5 = cp; best_h = hp.copy()

    # Phase 8: Final SLSQP polish
    if _remaining(t0, budget_s) > 50:
        h8 = _slsqp(best_h, 300000, 5000, 1e-20)
        c8 = _eval(h8)
        if c8 < best_c5:
            best_c5 = c8; best_h = h8.copy()

    # Phase 9: Final projected gradient with small M
    if _remaining(t0, budget_s) > 20:
        h9, c9 = _proj_grad(best_h, best_n, 10000, 2, 5.0, t0 + _remaining(t0, budget_s) - 5)
        if c9 < best_c5:
            best_c5 = c9; best_h = h9.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
