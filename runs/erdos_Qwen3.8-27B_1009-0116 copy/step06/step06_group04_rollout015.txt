Objective: _smm(c, alpha) + mu * (sum(h) - n/2)² - beta * sum(log(h) + log(1-h))

This is smooth and can be optimized with L-BFGS-B (with bounds [0,1]) or even plain gradient descent.

```python
def _obj_barrier(x, alpha, mu, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    obj = _smm(c, alpha)
    s = x.sum() - len(x) * 0.5
    obj += mu * s * s
    # Log barrier for [0,1]
    obj -= beta * (np.log(x).sum() + np.log(1.0 - x).sum())
    return obj

def _grad_barrier(x, alpha, mu, beta):
    g = _grad_alpha(x, alpha)
    s = x.sum() - len(x) * 0.5
    g += 2 * mu * s
    g -= beta * (1.0 / x - 1.0 / (1.0 - x))
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            o = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in o:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_alpha(x, alpha, topk=None):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    if topk and topk < len(w):
        idx = np.argsort(w)[::-1][:topk]
    else:
        idx = np.arange(len(w))
    g = np.zeros(n)
    for k in idx:
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _obj_slsqp(x, alpha, topk):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _sum_f(x):
    return x.sum() - len(x) * 0.5

def _sum_j(x):
    return np.ones(len(x))

def _slsqp_run(h, alpha, maxiter, topk):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_f, 'jac': _sum_j}]
    try:
        res = minimize(_obj_slsqp, h, jac=lambda x: _grad_alpha(x, alpha, topk),
                       args=(alpha, topk), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam_run(h0, alpha_seq, lr, n_per, seed, deadline, topk_start, topk_end):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy()
    best_x = x.copy()
    best_c = _eval(x)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    for al in alpha_seq:
        for _ in range(n_per):
            if time.time() > deadline:
                return best_x, best_c
            t += 1
            frac = (al in alpha_seq) and alpha_seq.index(al) / max(len(alpha_seq) - 1, 1)
            topk = int(topk_start + (topk_end - topk_start) * frac)
            if topk < 20:
                topk = 20
            g = _grad_alpha(x, al, topk)
            m_v = b1 * m_v + (1 - b1) * g
            v_v = b2 * v_v + (1 - b2) * g * g
            mhat = m_v / (1 - b1 ** t)
            vhat = v_v / (1 - b2 ** t)
            x = x - lr * mhat / (np.sqrt(vhat) + eps)
            x = _repair(x, n)
            c = _eval(x)
            if c < best_c:
                best_c = c
                best_x = x.copy()
        if t % 2000 == 0:
            x = _repair(x + 0.003 * rng.randn(n), n)
    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _binary_local(h0, n, rng, n_iters, deadline):
    x = (h0 > 0.5).astype(float)
    x = _repair(x, n)
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        x[i] = 1.0 - x[i]
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new <= c_cur:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
        else:
            x[i] = 1.0 - x[i]
            x = _repair(x, n)
    return best_x, best_c

def _coord_descent(h, n, rng, deadline, scale):
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    for _ in range(500000):
        if time.time() > deadline or scale < 1e-8:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        x[i] += d
        x[j] -= d
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new < c_cur - 1e-16:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
            scale = max(scale * 0.9995, 1e-9)
        else:
            x = best_x.copy()
            c_cur = best_c
            scale *= 0.9
    return best_x, best_c
```
