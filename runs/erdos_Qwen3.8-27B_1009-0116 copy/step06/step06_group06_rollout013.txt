    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(c)[::-1][:topk]
    # Normalize weights to sum to 1
    w = np.zeros(2 * n - 1)
    w[top] = 1.0 / topk
    
    def f(x):
        cx = np.correlate(x, 1.0 - x, mode='full')
        return np.sum(w * cx) * (2.0 / n)
    
    def g(x):
        cx = np.correlate(x, 1.0 - x, mode='full')
        # recompute weights based on current x
        top_x = np.argsort(cx)[::-1][:topk]
        wx = np.zeros(2 * n - 1)
        wx[top_x] = 1.0 / topk
        return _grad_wcorr(wx, x)
    
    # Use SLSQP with the active-set objective
    bnds = [(0.0, 1.0)] * n
    def cfun(x): return x.sum() - n * 0.5
    def cjac(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _grad_wcorr(w, h):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-15: continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    g *= dx
    return g

def _coord_desc(h, n_step, rng, t_end):

        if c5n < c5:

        else:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h
