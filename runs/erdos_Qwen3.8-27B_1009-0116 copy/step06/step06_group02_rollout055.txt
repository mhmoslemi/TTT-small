    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    diff = target - x.sum()
    if abs(diff) < 1e-14:
        return x
    if diff > 0:
        room = 1.0 - x
        tr = room.sum()
        if tr > 1e-14:
            x += (diff / tr) * room
    else:
        room = x.copy()
        tr = room.sum()
        if tr > 1e-14:
            x += (diff / tr) * room

from scipy.optimize import minimize
import time

def _sigmoid(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -500, 500)))

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(x, n):
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    diff = target - x.sum()
    if abs(diff) < 1e-14:
        return x
    if diff > 0:
        room = 1.0 - x
        tr = room.sum()
        if tr > 1e-14:
            x += (diff / tr) * room
    else:
        room = x
        tr = room.sum()
        if tr > 1e-14:
            x -= (-diff / tr) * room
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - x[idx]); x[idx] += a; diff -= a
            else:
                a = min(-diff, x[idx]); x[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return x

def _lse(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad_h(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        if w[ki] < 1e-14:
            continue
        shift = ki - (n - 1)
        wk = w[ki] * dx
        if shift >= 0:
            g[shift:] += wk * (1.0 - h[:n - shift])
            g[:n - shift] -= wk * h[shift:]
        else:
            sp = -shift
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g

    h = _sigmoid(z)
    dx = 2.0 / n
    return _lse(h, alpha) * dx + mu * (h.sum() - n * 0.5) ** 2

def _z_grad(z, n, alpha, mu):
    h = _sigmoid(z)
    sh = h * (1.0 - h)
    gh = _lse_grad_h(h, alpha)
    pg = 2.0 * mu * (h.sum() - n * 0.5)
    return (gh + pg) * sh

def _h_to_z(h):
    h = np.clip(h, 1e-8, 1 - 1e-8)
    return 2.0 * np.arctanh(2.0 * h - 1.0)

    z0 = _h_to_z(h_init)
    best_h = h_init.copy()
    best_c = _eval(best_h)
    for alpha in alpha_list:
        if time.time() > t_end:
            break
        try:
            res = minimize(_z_obj, z0, jac=_z_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         options={'maxiter': maxiter, 'ftol': 1e-16,

                best_h = h_new.copy()

        except Exception:

    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
