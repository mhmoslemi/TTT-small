        if time.time() - t0 > time_lim:
            break
        c = _corr_all(y) * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for kk in range(len(c)):
            if w[kk] < 1e-14:
                continue
            g += w[kk] * _grad_lag(y, kk, dx)
        y_new = _repair(y - step * g, n)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = y_new + ((t_k - 1.0) / t_new) * (y_new - y)
        y = np.clip(y, 0.0, 1.5)
        t_k = t_new
        v = _eval(y)
        if v < best_v - 1e-15:
            best_v = v
            best_h = y.copy()
        if it > 5 and best_v > v + 5e-6:
            step *= 0.5
    return _repair(best_h, n), best_v

def _active_descent(h, n_top, max_it, time_lim, t0, step0=0.04):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5c = _eval(h)
    step = step0
    for it in range(max_it):
        if time.time() - t0 > time_lim:
            break
        c = _corr_all(h)
        top_idx = np.argsort(c)[-n_top:]
        g = np.zeros(n)
        for kk in top_idx:
            g += _grad_lag(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for _ in range(22):
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
            step *= 0.5
        if not improved:
            step = max(step0 * 0.3, 0.001)
            if step < 5e-5:
                break
        else:
            step = min(step * 1.1, step0 * 2)
    return h, c5c

def _waterfill_grad(h, n_top=5):
    """Weighted gradient focusing more on highest correlations."""
    n = len(h)
    dx = 2.0 / n
    c = _corr_all(h)
    top_idx = np.argsort(c)[-n_top:]
    vals = c[top_idx]
    above = np.maximum(vals - vals.mean(), 0)
    wts = above / (above.sum() + 1e-15)
    g = np.zeros(n)
    for i, kk in enumerate(top_idx):
        g += wts[i] * _grad_lag(h, kk, dx)
    return g, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _block_init(n, rng, n_blocks):
    """Create a block-structured initial guess."""
    h = np.zeros(n)
    per = n // n_blocks
    for b in range(n_blocks):
        if b % 2 == 0:
            h[b*per:(b+1)*per] = 1.0
    h = _repair(h, n)
    return h

def _slsqp_quick(h, alpha, maxit):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = _corr_all(x) * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def grad(x):
        c = _corr_all(x) * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for kk in range(len(c)):
            if w[kk] < 1e-13:
                continue
            g += w[kk] * _grad_lag(x, kk, dx)
        return g
    from scipy.optimize import minimize
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Quick active descent on parent
    if tl() > 250:
        h1, c5 = _active_descent(best_h, 3, 600, 120, t0, 0.05)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: FISTA with high alpha (focus on max)
    if tl() > 250:
        h2, c5 = _fista(best_h, 30000, 300, 100, t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 3: Active descent refined
    if tl() > 200:
        h3, c5 = _active_descent(best_h, 4, 500, 80, t0, 0.03)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()

    # Phase 4: SLSQP polish
    if tl() > 250:
        h4 = _slsqp_quick(best_h, 50000, 1500)
        c5 = _eval(h4)
        if c5 < best_c5:
            best_c5, best_h = c5, h4.copy()

    # Phase 5: Active descent after SLSQP
    if tl() > 150:
        h5, c5 = _active_descent(best_h, 3, 500, 70, t0, 0.02)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Phase 6: Multi-resolution search
    for n_try in [310, 434, 500, 550, 700, 800]:
        if tl() < 140:
            break
        h_r = _repair(_resample(best_h, n_try), n_try)
        h_r, c5a = _active_descent(h_r, 3, 200, 40, t0, 0.04)
        h_r, c5b = _fista(h_r, 20000, 100, 35, t0)
        h_r, c5c = _active_descent(h_r, 4, 200, 30, t0, 0.02)
        c5 = min(c5a, c5b, c5c)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 7: Block structured initial + optimize
    if tl() > 200:
        for nb in [4, 6, 8]:
            if tl() < 60:
                break
            hb = _block_init(best_n, rng, nb)
            hb, c5 = _fista(hb, 10000, 150, 50, t0)
            hb, c5 = _active_descent(hb, 3, 200, 40, t0, 0.03)
            hb, c5 = _active_descent(hb, 5, 150, 30, t0, 0.01)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hb.copy(), best_n

    # Phase 8: Structured perturbation + refine
    for _rep in range(8):
        if tl() < 50:
            break
        mode = rng.randint(3)
        if mode == 0:
            h_p = _repair(best_h + rng.randn(best_n) * 0.008, best_n)
        elif mode == 1:
            h_p = best_h.copy()
            idx = rng.choice(best_n, size=best_n // 10, replace=False)
            h_p[idx] += rng.choice([-1, 1], size=len(idx)) * 0.1
            h_p = _repair(h_p, best_n)
        else:
            blk = best_n // 4
            h_p = _repair(np.roll(best_h, rng.randint(blk), axis=0), best_n)
        h_p, c5 = _active_descent(h_p, 3, 200, 25, t0, 0.03)
        h_p, c5 = _fista(h_p, 20000, 80, 20, t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h_p.copy()

    # Phase 9: Ultra-fine final polish
    if tl() > 25:
        h9 = best_h.copy()
        n = len(h9)
        dx = 2.0 / n
        c5c = _eval(h9)
        for scale in [0.003, 0.001, 0.0004]:
            if tl() < 8:
                break
            for _it in range(3000):
                if time.time() - t0 > budget_s - 5:
                    break
                c = _corr_all(h9)
                top2 = np.argsort(c)[-2:]
                g = np.zeros(n)
                for kk in top2:
                    g += _grad_lag(h9, kk, dx)
                gn = np.linalg.norm(g)
                if gn < 1e-15:
                    break
                g /= gn
                st = scale
                ok = False
                for _ in range(14):
                    hn = _repair(h9 - st * g, n)
                    cn = _eval(hn)
                    if cn < c5c - 1e-16:
                        h9 = hn
                        c5c = cn
                        ok = True
                        break
                    st *= 0.5
                if not ok:
                    break
        if c5c < best_c5:
            best_c5, best_h = c5c, h9.copy()

    # Phase 10: Final active descent
    if tl() > 12:
        hf, c5 = _active_descent(best_h, 3, 200, 10, t0, 0.005)
        if c5 < best_c5:
            best_c5, best_h = c5, hf.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
