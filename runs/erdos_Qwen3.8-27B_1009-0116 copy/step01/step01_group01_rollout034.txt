def repair(h, n):

# Globals for Nelder-Mead
_nm_n = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            residual = target - h.sum()
            if abs(residual) > 1e-14:
                idx = int(np.argmin(h)) if residual > 0 else int(np.argmax(h))
                h[idx] += residual
    return h

def coord_eval(x, la, lb, cp, c0b, oj):
    if len(la) > 0:
        mx = float(np.max(la * x + lb))
    else:
        mx = -1e10
    if mx < cp:
        mx = cp
    c0v = (c0b - oj + oj * oj) + x - x * x
    if c0v > mx:
        mx = c0v
    return mx

def golden_1d(la, lb, cp, c0b, oj):
    phi_c = 0.61803398875
    lo, hi = 0.0, 1.0
    x1 = lo + phi_c * (hi - lo)
    x2 = hi - phi_c * (hi - lo)
    f1 = coord_eval(x1, la, lb, cp, c0b, oj)
    f2 = coord_eval(x2, la, lb, cp, c0b, oj)
    for _ in range(50):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, x2, f2 = x2, x1, f1
            x1 = lo + phi_c * (hi - lo)
            f1 = coord_eval(x1, la, lb, cp, c0b, oj)
        else:
            lo, x1, f1 = x1, x2, f2
            x2 = hi - phi_c * (hi - lo)
            f2 = coord_eval(x2, la, lb, cp, c0b, oj)
    xo = (lo + hi) * 0.5
    fo = coord_eval(xo, la, lb, cp, c0b, oj)
    # Check boundaries
    f0 = coord_eval(0.0, la, lb, cp, c0b, oj)
    f1v = coord_eval(1.0, la, lb, cp, c0b, oj)
    if f0 < fo:
        xo, fo = 0.0, f0
    if f1v < fo:
        xo, fo = 1.0, f1v
    return xo, fo

def coord_descent(h, n, max_sweeps=200):
    h = h.copy()
    c_full = np.correlate(h, 1.0 - h, mode='full')
    best = float(np.max(c_full))
    m_all = np.arange(-(n - 1), n, dtype=float)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            oj = h[j]
            # Validity
            valid1 = (m_all >= j - (n - 1)) & (m_all <= j)
            valid2 = (m_all >= -j) & (m_all <= n - 1 - j)
            nonzero = (m_all != 0)
            
            # Coefficients for m ≠ 0
            coeff = np.zeros(2 * n - 1)
            m_v1n = valid1 & nonzero
            if m_v1n.any():
                idx_v1 = (j - m_all[m_v1n]).astype(int)
                coeff[m_v1n] += (1.0 - h[idx_v1])
            m_v2n = valid2 & nonzero
            if m_v2n.any():
                idx_v2 = (j + m_all[m_v2n]).astype(int)
                coeff[m_v2n] -= h[idx_v2]
            
            # Affected nonzero indices
            aff_nz = m_v1n | m_v2n
            nz_not_aff = nonzero & ~aff_nz
            
            if nz_not_aff.any():
                cp = float(np.max(c_full[nz_not_aff]))
            else:
                cp = -1e10
            
            la = coeff[aff_nz]
            lb = c_full[aff_nz] - oj * coeff[aff_nz]
            c0b = c_full[n - 1]  # lag 0 is at index n-1
            
            xo, fo = golden_1d(la, lb, cp, c0b, oj)
            
            if fo < best - 1e-13:
                h[j] = xo
                # Update c_full
                c_full[aff_nz] += (xo - oj) * coeff[aff_nz]
                c_full[n - 1] = c0b + (xo - oj) - (xo * xo - oj * oj)
                best = fo
                improved = True
        if not improved:
            break
    h = repair(h, n)
    return h, compute_c5(h)

def nm_obj(x):
    return compute_c5(repair(x, _nm_n))

def smooth_obj(x, t):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(c)
    return (cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    global _nm_n
    np.random.seed(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    def time_ok(frac):
        return (budget_s * frac - elapsed()) > 5.0
    
    # Initialize
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = repair(h0, n0)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Coordinate descent at n0
    if time_ok(0.15):
        h, v = coord_descent(best_h, n0, max_sweeps=200)
        if v < best_c5:
            best_c5, best_h = v, h
    
    # Phase 2: Nelder-Mead polish at n0
    if time_ok(0.25) and n0 <= 500:
        _nm_n = n0
        res = minimize(nm_obj, best_h.copy(), method='Nelder-Mead',
                       options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-12})
        h = repair(res.x, n0)
        v = compute_c5(h)
        if v < best_c5:
            best_c5, best_h = v, h
    
    # Phase 3: Higher resolution
    for mult in [2, 3, 4]:
        n = n0 * mult
        if n > 3000 or not time_ok(0.6):
            break
        xo = np.linspace(0.0, 2.0, len(best_h))
        xn = np.linspace(0.0, 2.0, n)
        h_init = np.interp(xn, xo, best_h)
        h_init = repair(h_init, n)
        
        h, v = coord_descent(h_init, n, max_sweeps=100)
        if v < best_c5:
            best_c5, best_h = v, h
        
        if time_ok(0.55 + 0.1 * mult) and n <= 1000:
            _nm_n = n
            res = minimize(nm_obj, best_h.copy(), method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-13})
            h = repair(res.x, n)
