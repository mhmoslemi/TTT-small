import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(4):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            m = h < 1.0 - 1e-10
        else:
            m = h > 1e-10
        c = int(np.sum(m))
        if c == 0:
            break
        h[m] += diff / c
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_topk(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    w = 1.0 / top_k
    for j in idx:
        m = j - (n - 1)
        if m >= 0:
            grad[:n - m] += w * dx * (1.0 - h[m:])
            grad[m:] -= w * dx * h[:n - m]
        else:
            ma = -m
            grad[ma:] += w * dx * (1.0 - h[:n - ma])
            grad[:n - ma] -= w * dx * h[ma:]
    return grad

def _subgrad(h0, n_iter, lr_start, lr_end, tk_start, tk_end, t0, budget_frac=0.9):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5(h)
    vel = np.zeros(n)
    beta = 0.9
    deadline = t0 + (time.time() - t0) * 0 + budget_frac * n_iter * 0.001
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = it / max(1, n_iter - 1)
        lr = lr_start * (lr_end / lr_start) ** (t ** 0.7)
        tk = max(2, int(tk_start - (tk_start - tk_end) * t))
        grad = _grad_topk(h, tk)
        vel = beta * vel + (1 - beta) * grad
        h_new = _project(h - lr * vel, target)
        c = _c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _coord_descent(h0, t0, max_s):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_c = _c5(h)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    neigh = [-4, -3, -2, -1, 1, 2, 3, 4]
    for d in steps:
        if time.time() - t0 > max_s:
            break
        for _ in range(8):
            if time.time() - t0 > max_s:
                break
            improved = False
            order = np.arange(n)
            np.random.shuffle(order)
            for i in order:
                if time.time() - t0 > max_s:
                    break
                for dj in neigh:
                    j = i + dj
                    if j < 0 or j >= n:
                        continue
                    h2 = h.copy()
                    h2[i] += d
                    h2[j] -= d
                    if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                        c2 = _c5(h2)
                        if c2 < best_c - 1e-13:
                            h = h2
                            best_c = c2
                            improved = True
                    h2 = h.copy()
                    h2[i] -= d
                    h2[j] += d
                    if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                        c2 = _c5(h2)
                        if c2 < best_c - 1e-13:
                            h = h2
                            best_c = c2
                            improved = True
            if not improved:
                break
    return h, best_c

def _anneal(h0, t0, max_s):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5(h)
    T = 0.008
    cnt = 0
    while time.time() - t0 < max_s:
        cnt += 1
        T *= 0.9998
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        d = np.random.uniform(-0.06, 0.06)
        if abs(d) < 0.005:
            continue
        h2 = h.copy()
        h2[i] += d
        h2[j] -= d
        if not (0 <= h2[i] <= 1 and 0 <= h2[j] <= 1):
            continue
        c2 = _c5(h2)
        delta = c2 - best_c
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
            h = h2
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
    return best_h, best_c

def _interp_parent(parent, n):
    pn = len(parent)
    if n == pn:
        return parent.copy()
    x_old = (np.arange(pn) + 0.5) / pn * 2.0
    x_new = (np.arange(n) + 0.5) / n * 2.0
    h = np.interp(x_new, x_old, parent)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)

    pn = len(parent)
    best_h = parent.copy()
    best_c5 = _c5(parent)
    best_n = pn
    print(f"Start: n={pn}, C5={best_c5:.6f}")

    n_list = [70, 80, 90, 100, 120, 60]
    for n in n_list:
        remaining = budget_s - (time.time() - t0)
        if remaining < 70:
            break

        # Phase A: subgradient from interpolated parent
        h_init = _interp_parent(parent, n)
        alloc = min(remaining * 0.25, 200)
        n_iter = int(alloc * 4000)
        n_iter = min(n_iter, 80000)
        h1, c1 = _subgrad(h_init, n_iter, 0.004, 0.00005, 15, 3, t0, budget_frac=alloc / remaining)
        if c1 < best_c5:
            best_c5 = c1; best_h = h1; best_n = n
            print(f"  n={n} sg_A: C5={c1:.6f}")

        # Phase B: sharper subgradient continuing from A
        h2, c2 = _subgrad(h1, int(n_iter * 0.7), 0.002, 0.00002, 8, 2, t0, budget_frac=remaining * 0.15 / remaining)
        if c2 < best_c5:
            best_c5 = c2; best_h = h2; best_n = n
            print(f"  n={n} sg_B: C5={c2:.6f}")

        # Phase C: coordinate descent
        remaining = budget_s - (time.time() - t0)
        cd_time = min(remaining * 0.15, 60)
        h3, c3 = _coord_descent(h2, t0, t0 + cd_time)
        if c3 < best_c5:
            best_c5 = c3; best_h = h3; best_n = n
            print(f"  n={n} cd: C5={c3:.6f}")

        # Phase D: simulated annealing
        remaining = budget_s - (time.time() - t0)
        if remaining > 40:
            sa_time = min(remaining * 0.12, 50)
            h4, c4 = _anneal(h3, t0, t0 + sa_time)
            if c4 < best_c5:
                best_c5 = c4; best_h = h4; best_n = n
                print(f"  n={n} sa: C5={c4:.6f}")

        # Phase E: second start with perturbation
        remaining = budget_s - (time.time() - t0)
        if remaining > 80:
            rng = np.random.RandomState(seed + n * 7 + 1)
            h_pert = h_init + rng.normal(0, 0.08, n)
            h_pert = _project(h_pert, n / 2.0)
            h5, c5v = _subgrad(h_pert, int(remaining * 0.15 * 4000), 0.003, 0.0001, 10, 3, t0, budget_frac=remaining * 0.15 / remaining)
            remaining = budget_s - (time.time() - t0)

                    best_c5 = c6; best_h = h6; best_n = n

    # Ensure constraint
