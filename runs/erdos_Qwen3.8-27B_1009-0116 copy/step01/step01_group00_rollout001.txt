import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(80):
        h = h * (target / max(s, 1e-15))
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = min(-diff, h[idx])
                a = -a
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-13:
                break
    return h

def _ls_obj(h, alpha, lam, target):
    corr = _corr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(h) - target) ** 2

def _run_lbfgs(h_start, alpha, lam, target, n, bounds, maxiter, t0, budget):
    if time.time() - t0 > budget - 20:
        return h_start
    try:
        res = minimize(_ls_obj, h_start, args=(alpha, lam, target),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return _repair(res.x, n)
    except Exception:
        return h_start

    n = len(initial_h_values)

    best_c5 = _c5(best_h, dx)

    def remaining():

    # Phase 1: Progressive L-BFGS-B with increasing smoothness and penalty
    schedule = [
        (15, 5, 500), (40, 10, 500), (80, 20, 600), (150, 30, 600),
        (300, 50, 700), (600, 80, 800), (1200, 120, 900), (2500, 200, 1000)
    ]
    h = best_h.copy()
    for alpha, lam, mit in schedule:
        if remaining() < 60:
            break
        h_new = _run_lbfgs(h, alpha, lam, target, n, bounds, mit, t0, budget_s)
        c5 = _c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
        h = h_new

    # Phase 2: Diverse restarts with different perturbation styles
    restarts = [
        ('gauss', 0.15), ('gauss', 0.3), ('gauss', 0.5),

                break
            improved = False
            for _ in range(300):

                if i == j:
                    continue
                si, sj = h[i], h[j]
                for d in deltas:
                    ni, nj = si + d, sj - d

            best_h = h_new

    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
