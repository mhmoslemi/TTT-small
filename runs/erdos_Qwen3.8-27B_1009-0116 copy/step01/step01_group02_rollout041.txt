def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            rem = diff
            for idx in order:
                if rem > 0:
                    add = min(rem, 1.0 - h[idx])
                    h[idx] += add
                    rem -= add
                else:
                    red = min(-rem, h[idx])
                    h[idx] -= red
                    rem += red
                if abs(rem) < 1e-15:
                    break
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')

    grad = np.zeros(n)

                    grad[j] -= h[j - m]
                if m < 0 and (j - m) >= 0:
                    pass
    h_new = h - step_size * grad
