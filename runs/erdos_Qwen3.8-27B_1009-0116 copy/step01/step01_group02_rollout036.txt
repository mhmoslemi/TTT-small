import time

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def subgradient_at(h, n, m_idx):
    shift = m_idx - (n - 1)
    grad = np.zeros(n)
    js = np.arange(n)
    lo1 = np.maximum(0, shift)
    hi1 = np.minimum(n-1, n-1+shift)
    valid1 = (js >= lo1) & (js <= hi1)
    # (1-h[j-shift]) for valid j-shift
    idx1 = js[valid1] - shift
    grad[valid1] += (1.0 - h[idx1])
    valid2 = (js >= -shift) & (js <= n-1-shift)
    idx2 = js[valid2] + shift
    grad[valid2] -= h[idx2]
    return grad

h[i] appears in:
- Term k=i: h[i] * (1-h[i-shift]) → contributes h[i] * (1-h[i-shift])
- Term k=i+shift: h[i+shift] * (1-h[i]) → contributes h[i+shift] * (1-h[i]) = -h[i+shift]*h[i] + h[i+shift]

So the contribution of h[i] to corr[m] is:
h[i] * (1-h[i-shift]) - h[i+shift] * h[i] + h[i+shift]
= h[i] * [(1-h[i-shift]) - h[i+shift]] + h[i+shift]

This is linear in h[i]! So for a fixed shift m, corr[m] is linear in h[i].

The max over m is then a max of linear functions of h[i], which is convex in h[i].

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def subgrad(h, n, m_idx):
    shift = m_idx - (n - 1)
    js = np.arange(n)
    g = np.zeros(n)
    v1 = (js >= max(0, shift)) & (js <= min(n - 1, n - 1 + shift))
    g[v1] += (1.0 - h[js[v1] - shift])
    v2 = (js >= max(0, -shift)) & (js <= min(n - 1, n - 1 - shift))
    g[v2] -= h[js[v2] + shift]
    return g

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-10:
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = int(m.sum())
        if c > 0:
            h[m] += d / c
            h = np.clip(h, 0.0, 1.0)
            d2 = target - h.sum()
            if abs(d2) > 1e-12:
                m2 = (h > 1e-10) & (h < 1.0 - 1e-10)
                c2 = int(m2.sum())
                if c2 > 0:
                    h[m2] += d2 / c2
                    h = np.clip(h, 0.0, 1.0)
    return h

def polish_slsqp(h, n, mi=200):
    dx = 2.0 / n
    tgt = n / 2.0
    bd = [(0.0, 1.0)] * n
    def f(x):
        return float(np.max(np.correlate(x, 1.0 - x, mode='full'))) * dx
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        r = minimize(f, h, method='SLSQP', bounds=bd, constraints=con,
                     options={'maxiter': mi, 'ftol': 1e-14})
        return repair(r.x, n)
    except Exception:
        return h

def lse_c5(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def init_sawtooth(n):
    x = np.arange(n)
    h = np.where((x * 7) % n < n // 2, 0.85, 0.15)
    return repair(h, n)

def init_block(n):
    h = np.zeros(n)
    h[:n // 4] = 1.0
    h[3 * n // 4:] = 1.0
    return repair(h, n)

def init_fourier(n, freqs):
    x = np.arange(n)
    h = np.zeros(n)
    for f in freqs:
        h += np.cos(2.0 * np.pi * f * x / n)
    h = 0.5 + 0.25 * h
    return repair(h, n)

def subgrad_descent(h, n, max_iter, t0, budget):
    dx = 2.0 / n
    step = 0.04
    c_cur = compute_c5(h, n)
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > budget - 25:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mi = int(np.argmax(corr))
        g = subgrad(h, n, mi)
        step = max(step * 0.997, 0.0002)
        h_new = repair(h - step * g * dx, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_cur - 1e-15:
            h, c_cur = h_new, c_new
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
    return h, c_cur

def lse_polish(h, n, betas, t0, budget):
    tgt = n / 2.0
    bd = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    c_cur = compute_c5(h, n)
    for beta in betas:
        if time.time() - t0 > budget - 12:
            break
        try:
            r = minimize(lambda x: lse_c5(x, n, beta), h,
                         method='SLSQP', bounds=bd, constraints=con,
                         options={'maxiter': 120, 'ftol': 1e-13})
            hc = repair(r.x, n)
            c = compute_c5(hc, n)
            if c < c_cur:
                h, c_cur = hc, c
        except Exception:
            pass
    return h, c_cur

    rng = np.random.RandomState(seed)

    init = globals().get('initial_h_values', None)

    n_main = n0 if init is not None else 70

        for fn in [init_sawtooth, init_block,

            except Exception:
                pass

    best_h = starts[0][1].copy()
    best_n = starts[0][0]

        c_s = compute_c5(h_s, n_s)

        h_cur = h_s.copy()
        # Phase 1: Subgradient

        # Phase 2: LSE smooth

        # Phase 3: Exact SLSQP
        if time.time() - t0 < budget_s - 10:

            cs = compute_c5(hs, n_s)
            if cs < c_cur:

        # Phase 4: Second subgradient pass from SLSQP result

        if time.time() - t0 < budget_s - 8:
            hs2 = polish_slsqp(h_cur, n_s, mi=200)
            cs2 = compute_c5(hs2, n_s)
