def eval_c5(h, dx):

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            if d2 > 0:

            else:
                idx = int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))

def opt_slsqp(h, n, dx, maxiter=1000, ftol=1e-14):
