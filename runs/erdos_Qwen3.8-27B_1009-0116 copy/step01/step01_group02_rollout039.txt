def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    # Scale to hit target
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            # Distribute differently
            if diff > 0:
                mask = (h < 1.0 - 1e-8)
            else:
                mask = (h > 1e-8)
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_c5(h, n, beta):

    scaled = beta * corr
    m = np.max(scaled)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    # Find active set (within tolerance)
    tol = 1e-6
    active = np.where(corr >= np.max(corr) - tol)[0]
    
    grad = np.zeros(n)
    for k_idx in active:
        lag = k_idx - (n - 1)
        if lag >= 0 and lag < n:

            i_range = np.arange(nr)
            grad[i_range + lag] += (1.0 - h[i_range])
            grad[i_range] += (-h[i_range + lag])

                j_range = np.arange(nr)
                grad[j_range] += (-h[j_range + t])
                grad[j_range + t] += (1.0 - h[j_range])

    grad *= dx
    return obj, grad

            h = np.clip(h, 0.0, 1.0)

    s = beta * corr
    m = np.max(s)
    return float((m + np.log(np.sum(np.exp(s - m)))) / beta * dx)

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _fun_lse(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    s = beta * corr
    m = np.max(s)
    return float((m + np.log(np.sum(np.exp(s - m)))) / beta * dx)

def _jac_lse(x, n, beta):
    dx = 2.0 / n
    m_len = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    s = beta * corr
    mx = np.max(s)
    w = np.exp(s - mx)
    w /= np.sum(w)
    grad = np.zeros(n)
    for ki in range(m_len):
        wi = w[ki]
        if wi < 1e-10:
            continue
        lag = ki - (n - 1)
        if lag >= 0 and lag < n:
            nr = n - lag
            ir = np.arange(nr)
            grad[ir + lag] += wi * (1.0 - x[ir])
            grad[ir] += wi * (-x[ir + lag])
        elif lag < 0:
            t = -lag
            if t < n:
                nr = n - t
                jr = np.arange(nr)
                grad[jr] += wi * (-x[jr + t])
                grad[jr + t] += wi * (1.0 - x[jr])
    return grad * dx

def _fun_exact(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr)) * dx

def _jac_exact(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(corr)
    active = np.where(corr >= mx - 1e-7)[0]
    grad = np.zeros(n)
    for ki in active:
        lag = ki - (n - 1)
        if lag >= 0 and lag < n:
            ir = np.arange(n - lag)
            grad[ir + lag] += (1.0 - x[ir])
            grad[ir] += (-x[ir + lag])
        elif lag < 0:
            t = -lag
            if t < n:
                jr = np.arange(n - t)
                grad[jr] += (-x[jr + t])
                grad[jr + t] += (1.0 - x[jr])
    grad /= max(len(active), 1)
    return grad * dx

def _con_eq(x, target):
    return np.sum(x) - target

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    init = globals().get('initial_h_values', None)
    if init is not None:
        n = len(init)
        h_best = repair(init, n)
    else:
        n = 70
        h_best = np.full(n, 0.5)

    best_c5 = compute_c5(h_best, n)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con_fn = partial(_con_eq, target=target)
    cons = [{'type': 'eq', 'fun': con_fn}]

    def slsqp_exact(h0, maxit, ftol):
        r = minimize(partial(_fun_exact, n=n), h0,
                     jac=partial(_jac_exact, n=n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': ftol})
        return repair(r.x, n)

    def slsqp_lse(h0, beta, maxit, ftol):
        r = minimize(partial(_fun_lse, n=n, beta=beta), h0,
                     jac=partial(_jac_lse, n=n, beta=beta), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': ftol})
        return repair(r.x, n)

    for b in [4.0, 8.0, 15.0, 30.0, 60.0]:
