import numpy as np
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = x.sum()
    if abs(s - tgt) < 1e-13:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(50):
        t = 0.5 * (lo + hi)
        if np.clip(x - t, 0.0, 1.0).sum() > tgt:
            lo = t
        else:
            hi = t
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def get_B(p, x, n):
    k = np.arange(2 * n - 1)
    j = p + k - (n - 1)
    i = p - k + (n - 1)
    B = np.zeros(2 * n - 1)
    vj = (j >= 0) & (j < n)
    vi = (i >= 0) & (i < n) & (k != n - 1)
    B[vj] = 1.0 - x[np.clip(j, 0, n - 1)]
    if vi.any():
        B[vi] -= x[np.clip(i, 0, n - 1)]
    B[n - 1] -= x[p]
    return B

def min_1d(A, B, tau):
    t = 0.5
    for _ in range(35):
        z = (A + B * t) / tau
        z -= z.max()
        w = np.exp(z)
        sw = w.sum()
        fp = np.dot(B, w) / sw
        if abs(fp) < 1e-11:
            break
        fp2 = np.dot(B * B, w) / sw - fp * fp
        if fp2 > 1e-14:
            t_new = t - fp / fp2
        else:
            t_new = t - fp * tau
        if t_new < -0.1 or t_new > 1.1:
            lo, hi = 0.0, 1.0
            for _ in range(35):
                mid = 0.5 * (lo + hi)
                z2 = (A + B * mid) / tau
                z2 -= z2.max()
                w2 = np.exp(z2)
                if np.dot(B, w2) / w2.sum() > 0:
                    hi = mid
                else:
                    lo = mid
            t = 0.5 * (lo + hi)
            break
        t = t_new
    return max(0.0, min(1.0, t))

def coord_cd(x0, n, tau, max_sweeps=300, time_limit=None, t_start=0):
    x = x0.copy()
    corr = np.correlate(x, 1.0 - x, mode="full")
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        max_chg = 0.0
        for p in range(n):
            B = get_B(p, x, n)
            A = corr - x[p] * B
            t_new = min_1d(A, B, tau)
            chg = abs(t_new - x[p])
            if chg > max_chg:
                max_chg = chg
            old = x[p]
            x[p] = t_new
            corr += (t_new - old) * B
        x = project_feasible(x, n)
        if sweep % 20 == 19:
            corr = np.correlate(x, 1.0 - x, mode="full")
        if max_chg < 1e-10:
            break
    return x

def project_up(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, x_src, kind='linear', fill_value='extrapolate')(xd)
    return project_feasible(xp, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0_raw = np.array(initial_h_values, dtype=float)
    else:
        h0_raw = np.full(95, 0.5)
    n0 = len(h0_raw)
    dx0 = 2.0 / n0
    h0 = project_feasible(h0_raw, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    def try_update(x, n):
        nonlocal best_h, best_n, best_val
        dx = 2.0 / n
        v = compute_c5(x, dx)
        if v < best_val:
            best_val = v
            best_h = x.copy()
            best_n = n
        return v

    # Phase 1: Coordinate descent at base grid with decreasing tau
    taus_coarse = [1.0, 0.5, 0.25, 0.15, 0.1]
    for tau in taus_coarse:
        if time.time() - t_start > budget_s * 0.45:
            break
        x = coord_cd(best_h, n0, tau, max_sweeps=400, time_limit=budget_s * 0.45, t_start=t_start)
        try_update(x, n0)

    # Phase 2: Fine tau at base grid
    taus_fine = [0.06, 0.04, 0.025, 0.015, 0.01]
    for tau in taus_fine:
        if time.time() - t_start > budget_s * 0.60:
            break
        x = coord_cd(best_h, n0, tau, max_sweeps=600, time_limit=budget_s * 0.60, t_start=t_start)
        try_update(x, n0)

    # Phase 3: Random restarts at base grid
    if time.time() - t_start < budget_s * 0.72:
        n_restarts = 3
        for r in range(n_restarts):
            remaining = budget_s * 0.72 - (time.time() - t_start)
            if remaining < 15:
                break
            x_pert = project_feasible(best_h + 0.08 * rng.standard_normal(n0), n0)
            x = coord_cd(x_pert, n0, 0.1, max_sweeps=200, time_limit=budget_s * 0.72, t_start=t_start)
            x = coord_cd(x, n0, 0.02, max_sweeps=300, time_limit=budget_s * 0.72, t_start=t_start)
            try_update(x, n0)

    # Phase 4: Project to 2x+1 grid
    if time.time() - t_start < budget_s * 0.78:
        n1 = 2 * best_n + 1
        x_p = project_up(best_h, best_n, n1)
        x = coord_cd(x_p, n1, 0.5, max_sweeps=200, time_limit=budget_s * 0.88, t_start=t_start)
        x = coord_cd(x, n1, 0.15, max_sweeps=300, time_limit=budget_s * 0.88, t_start=t_start)
        x = coord_cd(x, n1, 0.05, max_sweeps=400, time_limit=budget_s * 0.92, t_start=t_start)
        x = coord_cd(x, n1, 0.015, max_sweeps=400, time_limit=budget_s * 0.95, t_start=t_start)
        try_update(x, n1)

    # Phase 5: Final polish at best grid
    remaining = budget_s - (time.time() - t_start) - 5
    if remaining > 20:
        x = coord_cd(best_h, best_n, 0.008, max_sweeps=500, time_limit=budget_s - 3, t_start=t_start)
        try_update(x, best_n)
        if best_n < 400:
            remaining2 = budget_s - (time.time() - t_start) - 3
            if remaining2 > 15:
                n_fin = 2 * best_n + 1
                x_f = project_up(best_h, best_n, n_fin)
                x_f = coord_cd(x_f, n_fin, 0.03, max_sweeps=300, time_limit=budget_s - 3, t_start=t_start)
                x_f = coord_cd(x_f, n_fin, 0.008, max_sweeps=300, time_limit=budget_s - 3, t_start=t_start)
                try_update(x_f, n_fin)

    # Final evaluation
    best_h = project_feasible(best_h, best_n)
    dx_f = 2.0 / best_n
    final_val = compute_c5(best_h, dx_f)
    return (best_h, final_val, best_n)
