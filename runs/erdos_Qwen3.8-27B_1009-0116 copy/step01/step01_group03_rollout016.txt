import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    """Return the raw correlation (without dx factor)."""
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_c5(h, beta):
    corr = compute_corr(h)
    dx = 2.0 / len(h)
    m = corr.max()
    shifted = (corr - m) * beta
    return dx * (m + np.log(np.sum(np.exp(shifted))) / beta)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def to_binary(h, ts):
    """Convert to binary solution with exact sum."""
    h_b = np.round(h)
    n_ones = h_b.sum()
    if n_ones > ts:
        # Need to flip some 1s to 0s
        frac = h_b - np.round(h)  # actually these are distances to 0.5
        # Flip the 1s closest to 0.5 (i.e., with smallest h value among 1s)
        ones_idx = np.where(h_b == 1)[0]
        need_flip = int(n_ones - ts)
        # Flip the ones with smallest h[i] (closest to 0)
        order = np.argsort(h[ones_idx])
        for idx in order[:need_flip]:
            h_b[ones_idx[idx]] = 0
    elif n_ones < ts:
        zeros_idx = np.where(h_b == 0)[0]
        need_flip = int(ts - n_ones)
        order = np.argsort(-h[zeros_idx])
        for idx in order[:need_flip]:
            h_b[zeros_idx[idx]] = 1
    return h_b.astype(float)

def swap_local_search(h, ts, max_sweeps=500, time_limit=60):
    """Binary local search using swap moves."""
    n = len(h)
    t0 = time.time()
    c = compute_corr(h)
    m = c.max()
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                # Try swap: h[i] 1->0, h[j] 0->1
                h[i] = 0
                h[j] = 1
                c_new = compute_corr(h)
                m_new = c_new.max()
                if m_new < m - 1e-12:
                    m = m_new
                    c = c_new
                    improved = True
                    # Update lists
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    break
                else:
                    h[i] = 1
                    h[j] = 0
            # Re-fetch if we broke out
            if improved:
                break
        if not improved:
            break
    c5 = compute_c5(h)
    return h, c5

        mn = float(cn.max())
        delta = mn - m
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-9)):
            m = mn
            corr = cn
            if m < best_m:
                best_m = m
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, float(best_m * dx)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def rem():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n_init = 140
        rng0 = np.random.RandomState(seed)
        h_init = np.zeros(n_init)
        h_init[rng0.choice(n_init, n_init // 2, replace=False)] = 1.0

    n = len(h_init)
    ts = n / 2.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with annealed beta
    h = best_h.copy()
    for beta in [2.0, 5.0, 10.0, 20.0, 50.0, 120.0, 300.0, 800.0]:
        if rem() < 40:
            break
        h = slsqp(h, ts, beta, maxiter=350)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Binary local search if near-binary
    if rem() > 100:
        h_b = to_binary_clean(best_h, ts)
        h_b, c5_b = binary_lsearch(h_b, ts, max_time=min(70, rem() - 80))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 3: Simulated annealing
    if rem() > 90:
        h_sa = best_h.copy()
        if np.minimum(h_sa, 1 - h_sa).sum() / n > 0.15:
            h_sa = to_binary_clean(h_sa, ts)
        h_sa, c5_sa = sim_anneal(h_sa, ts, t_hot=0.4, t_cold=0.003,
                                 max_time=min(80, rem() - 70), seed=seed)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Follow-up binary local search on SA result
        if rem() > 60:
            h_sa2, c5_sa2 = binary_lsearch(best_h.copy(), ts, max_time=min(40, rem() - 50))
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()

    # Phase 4: SLSQP polish (continuous) from best
    if rem() > 60:
        h = best_h.copy()
        for beta in [80.0, 250.0, 600.0]:
            if rem() < 30:
                break
            h = slsqp(h, ts, beta, maxiter=300)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 5: Random restarts
    if rem() > 80:
        rng = np.random.RandomState(seed + 7777)
        for _ in range(6):
            if rem() < 25:
                break
            h_r = best_h + rng.normal(0, 0.12, n)
            h_r = project_feasible(h_r, ts)
            h_r = slsqp(h_r, ts, 15.0, maxiter=250)
            h_r = slsqp(h_r, ts, 150.0, maxiter=250)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                # Quick binary polish
                if rem() > 20:
                    hb = to_binary_clean(best_h, ts)
                    hb, c5_b = binary_lsearch(hb, ts, max_time=min(20, rem() - 15))
                    if c5_b < best_c5:
                        best_c5 = c5_b
                        best_h = hb

    # Phase 6: Final SLSQP polish
    if rem() > 25:
        h = best_h.copy()
        h = slsqp(h, ts, 500.0, maxiter=300)
        h = slsqp(h, ts, 1500.0, maxiter=200)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
