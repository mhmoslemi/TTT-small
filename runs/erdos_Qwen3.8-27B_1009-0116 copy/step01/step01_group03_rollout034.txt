import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        # Redistribute
        for k in range(len(idx)):
            room = (1.0 - h[idx[k]]) if diff2 > 0 else h[idx[k]]
            adj = min(room, abs(diff2)) * (1 if diff2 > 0 else -1)
            h[idx[k]] += adj
            diff2 -= adj
            if abs(diff2) < 1e-14:
                break
    return h

def smooth_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp_optimize(h, n, ts, beta, maxiter=800):
    res = minimize(lambda x: smooth_obj(x, n, beta), h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return h

        delta = rng.normal(0, sigma)
        h[i] = np.clip(h[i] + delta, 0, 1)

        c5_new = compute_c5(h)
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / max(T, 1e-10)):
            c5_cur = c5_new

                best_h = h.copy()

            else:

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def _smooth_obj(x):
    n = _g['n']
    beta = _g['beta']
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def _sum_fun(x):
    return x.sum() - _g['ts']

def slsqp_opt(h, n, ts, beta, maxiter=1000):
    _g['n'] = n
    _g['beta'] = beta
    _g['ts'] = ts
    try:
        res = minimize(_smooth_obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sum_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except Exception:
        pass
    return h

            continue
        h[i] = new_i
        h[j] = new_j
        c5_new = compute_c5(h)
        d = c5_new - c5_cur
        if d < 0 or (T > 1e-10 and rng.random() < np.exp(-d / T)):

                no_improve = 0
            else:
                no_improve += 1
        else:

            h[j] = h[j] + delta
            no_improve += 1
        T = max(T * 0.99997, 1e-7)
        sigma = max(sigma * 0.99999, 0.003)
        if no_improve > 80000:
            break
    return best_h, best_c5
