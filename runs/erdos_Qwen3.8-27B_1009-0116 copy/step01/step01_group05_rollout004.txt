import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def _coord_minimax(h, i, n, dx):
    """For coordinate i, find x that minimizes max_k f_k where f_k = dx * sum_j h[j+k]*(1-h[j])
    with h[i] replaced by x."""
    # f_k as function of x: f_k = const_k + dx * x * coeff_k
    # where coeff_k = (1-h[i-k]) - h[i+k] (with validity)
    nfull = 2*n - 1
    offsets = np.arange(-(n-1), n)  # lags s
    coeffs = np.zeros(nfull)
    for idx, s in enumerate(offsets):
        # h[i] appears as h[j+s] when j = i-s
        if 0 <= i-s < n:
            coeffs[idx] += (1.0 - h[i-s])
        # h[i] appears as h[j] when j = i, so h[j+s] = h[i+s]
        if 0 <= i+s < n:
            coeffs[idx] -= h[i+s]
    
    # f_k = base_k + dx * x * coeffs[k]
    # base_k = f_k(h) - dx * h[i] * coeffs[k]
    # Actually: f_k(h) = dx * sum_j h[j+k]*(1-h[j])
    # When we change h[i] from h[i] to x:
    # f_k(x) = f_k(h) + dx * (x - h[i]) * coeffs[k]
    
    base = _corr(h, dx)  # current f_k values
    h_i = h[i]
    # f_k(x) = base[k] + dx * (x - h_i) * coeffs[k]
    #         = (base[k] - dx*h_i*coeffs[k]) + dx*x*coeffs[k]
    a = dx * coeffs  # coefficient of x
    b = base - dx * h_i * coeffs  # constant term
    
    # Minimize max_k (a[k]*x + b[k]) over x in [0,1]
    # This is: minimize max_k (a[k]*x + b[k])
    # The optimal x is where the upper envelope of lines is minimized
    # If all a[k] >= 0: optimal x = 0
    # If all a[k] <= 0: optimal x = 1
    # Otherwise: x* is an intersection point
    
    # Simple approach: ternary search on [0,1] (since it's 1D convex)

        if f1 < f2:

        else:

    return 0.5 * (lo + hi)

    h = h.copy()
    best_c5 = _c5(h, dx)
    best_h = h.copy()

            best_h = h.copy()
        # Check convergence
        if sweep > 100 and best_c5 - c5 < 1e-8:

import numpy as np
from scipy.optimize import minimize
import time


def _corr_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _eval_c5(h, dx):
    return _corr_vec(h, dx).max()


def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _solve_1d_max(a, b):
    """Solve min_{x in [0,1]} max_k (a[k]*x + b[k]) via ternary search."""
    if np.all(a >= -1e-14):
        return 0.0
    if np.all(a <= 1e-14):
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) * 0.3333333
        m2 = lo + (hi - lo) * 0.6666667
        if np.max(a * m1 + b) <= np.max(a * m2 + b):
            hi = m2
        else:
            lo = m1
    return 0.5 * (lo + hi)


def _coord_descent(h, dx, target, t_end, max_sweeps=50000, perturb_every=500):
    n = len(h)
    h = h.copy()
    f = _corr_vec(h, dx)
    best_c5 = f.max()
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if sweep > 0 and sweep % perturb_every == 0:
            h = _proj(h + np.random.normal(0, 0.03, n), target)
            f = _corr_vec(h, dx)
            stall = 0
        for i in range(n):
            offsets = np.arange(-(n - 1), n)
            jm = i - offsets
            valid_m = (jm >= 0) & (jm < n)
            jp = i + offsets
            valid_p = (jp >= 0) & (jp < n)
            c = np.zeros(2 * n - 1)
            c[valid_m] += (1.0 - h[jm[valid_m]])
            c[valid_p] -= h[jp[valid_p]]
            a = dx * c
            b = f - dx * h[i] * c
            x_new = _solve_1d_max(a, b)
            if abs(x_new - h[i]) > 1e-13:
                f += dx * (x_new - h[i]) * c
                h[i] = x_new
        h = _proj(h, target)
        f = _corr_vec(h, dx)
        c5 = f.max()
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 1000:
            break
    return best_h, best_c5


def _soft_obj(h, dx, eps):
    cv = _corr_vec(h, dx)
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))


def _soft_grad(h, dx, eps):
    cv = _corr_vec(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        wt = w[idx]
        if wt < 1e-15:
            continue
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * dx * h[jp[vp]]
    return g


def _slsqp(h0, dx, n, target, eps, maxiter=300):
    def obj(x):
        return _soft_obj(x, dx, eps)

    def grad(x):
        return _soft_grad(x, dx, eps)

    def eq_fun(x):
        return x.sum() - target

    def eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h0, jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _proj(res.x, target)
    except Exception:
        return h0.copy()


def _subgrad_step(h, dx, target, lr):
    cv = _corr_vec(h, dx)
    mx = cv.max()
    active = np.where(cv >= mx - 1e-10)[0]
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    g /= max(len(active), 1)
    return _proj(h - lr * g, target)


def _run_n(h_init, n, dx, target, t0, deadline, rng):
    best_h = h_init.copy()
    best_c5 = _eval_c5(best_h, dx)

    # Phase 1: Coordinate descent (45%)
    t1 = t0 + (deadline - t0) * 0.45
    h, c5 = _coord_descent(best_h, dx, target, t1, max_sweeps=30000, perturb_every=400)
    if c5 < best_c5:
        best_h, best_c5 = h, c5

    # Phase 2: SLSQP with decreasing eps (25%)
    t2 = t0 + (deadline - t0) * 0.70
    for eps in [0.08, 0.04, 0.02, 0.01, 0.004, 0.001, 0.0004, 0.0001]:
        if time.time() > t2 - 1:
            break
        h2 = _slsqp(best_h, dx, n, target, eps, maxiter=250)
        c2 = _eval_c5(h2, dx)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Coordinate descent + subgradient + perturb (20%)
    t3 = t0 + (deadline - t0) * 0.90
    while time.time() < t3:
        h3, c3 = _coord_descent(best_h, dx, target, min(time.time() + 8, t3),
                                max_sweeps=8000, perturb_every=300)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3
        # Subgradient polish
        lrr = 0.1
        for _ in range(200):
            if time.time() > t3 - 1:
                break
            h3 = _subgrad_step(best_h, dx, target, lrr)
            c3 = _eval_c5(h3, dx)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()
                lrr = min(lrr * 1.05, 1.0)
            else:
                lrr = max(lrr * 0.92, 0.01)
        # Perturbation restart
        amp = rng.uniform(0.02, 0.12)
        hp = _proj(best_h + rng.normal(0, amp, n), target)
        cp = _eval_c5(hp, dx)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    # Phase 4: Final polish (10%)
    for eps in [0.0003, 0.0001, 0.00005]:
        if time.time() > deadline - 2:
            break
        h4 = _slsqp(best_h, dx, n, target, eps, maxiter=300)
        c4 = _eval_c5(h4, dx)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Final coord descent
    h5, c5_val = _coord_descent(best_h, dx, target, deadline - 0.5, max_sweeps=10000)
    if c5_val < best_c5:
        best_h, best_c5 = h5, c5_val

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
