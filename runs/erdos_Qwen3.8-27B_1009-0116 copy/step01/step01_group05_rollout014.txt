import numpy as np
from scipy.optimize import minimize
import time

_DX = 0.0
_EPS = 0.01
_N = 70
_TGT = 35.0

def _obj(x):
    cv = np.correlate(x, 1.0 - x, mode="full") * _DX
    mx = cv.max()
    return mx + _EPS * np.log(np.sum(np.exp((cv - mx) / _EPS)))

def _grad(x):
    cv = np.correlate(x, 1.0 - x, mode="full") * _DX
    mx = cv.max()
    w = np.exp((cv - mx) / _EPS)
    w /= w.sum()
    n = _N
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * _DX
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _cfun(x):
    return x.sum() - _TGT

def _cjac(x):
    return np.ones(_N)

def _c5(h, dx):
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _slsqp(h0, dx, n, target, eps, maxiter=500):
    global _DX, _EPS, _N, _TGT
    _DX, _EPS, _N, _TGT = dx, eps, n, target
    try:
        res = minimize(_obj, h0.copy(), jac=_grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-14})
        return _proj(res.x, target)
    except Exception:
        return h0

def _substep(h, dx, target, lr, tol):
    cv = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = cv.max()
    act = np.where(cv >= mx - tol)[0]
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in act:
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    g /= max(len(act), 1)
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h.copy()
    g /= gn
    return _proj(h - lr * g, target)

def _coord_refine(h, dx, n, target, rng, t_end):

    for it in range(2000):

        j = rng.randint(n)

        lo_t = max(0.0, s - 1.0)
        hi_t = min(1.0, s)
        if hi_t - lo_t < 1e-10:
            continue

        best_tc = bc

                best_t = t
        h[i], h[j] = best_t, s - best_t
        c = _c5(h, dx)
        if c < bc:
            bc = c
            best = h.copy()
    return best, bc
