```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / len(h))

def _smooth_max_obj(h, alpha, dx, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - np.max(corr)
    return (np.max(corr) + np.log(np.sum(np.exp(alpha * shifted)))/alpha) * dx

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-14:
        h *= target / s
    return np.clip(h, 0.0, 1.0)

def _pair_swap_opt(h, target, n_sweeps=20, time_limit=1.0):
    n = len(h)
    best = _c5(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        # Try random pairs
        for trial in range(200):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            # Move small amount from i to j
            for step in (0.01, 0.02, 0.05, 0.1):
                if h[i] < step or h[j] > 1.0 - step:
                    continue
                h2 = h.copy()
                h2[i] -= step
                h2[j] += step
                c = _c5(h2)
                if c < best - 1e-9:
                    h = h2
                    best = c
                    improved = True
                    break
                # Also try reverse
                if h[i] > step and h[j] < 1.0 - step:
                    h2[i] += step
                    h2[j] -= step
                    c = _c5(h2)
                    if c < best - 1e-9:
                        h = h2
                        best = c
                        improved = True
                        break
        if not improved:
            break
    return h, best
```

```python
def _subgradient_minmax(h, n, target, n_iter=5000, lr0=0.01):
    h = h.copy()
    for it in range(n_iter):
        # Compute all O(k)
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        # corr has length 2n-1, centered at index n-1
        # O(k) for lag k is corr[n-1+k] * dx... but we just need the max
        O = corr  # in "count" units, multiply by dx later for C5
        max_val = np.max(O)
        # Find top-k lags
        top_indices = np.argsort(O)[-min(10, len(O)):]  # top 10 lags
        # Compute negative gradient for each active lag
        grad = np.zeros(n)
        for idx in top_indices:
            lag = idx - (n-1)  # actual lag
            # O(lag) = sum_i h[i]*(1-h[i+lag]) for valid i
            # dO/dh[j] = sum over terms involving h[j]
            # h[j] appears as: (1-h[j+lag])*h[j] and h[j-lag]*(1-h[j])
            # = (1-h[j+lag]) - h[j-lag] (boundary: if j+lag out of range, term is 0)
            if lag >= 0:
                # h[i] for i in [0, n-1-lag], paired with h[i+lag] for i+lag in [lag, n-1]
                for j in range(n):
                    term = 0.0
                    if j + lag < n:  # h[j] appears as the "first" factor
                        term += (1.0 - h[j+lag])
                    if j - lag >= 0:  # h[j] appears as the "second" factor (h[j]=h[(j-lag)+lag])
                        term -= h[j-lag]
                    grad[j] += term
            else:
                lag2 = -lag
                for j in range(n):
                    term = 0.0
                    if j + lag2 < n:
                        term += (1.0 - h[j+lag2])
                    if j - lag2 >= 0:
                        term -= h[j-lag2]
                    grad[j] += term
        
        # Average gradient
        grad /= len(top_indices)
        
        # Step size (decreasing)
        lr = lr0 / (1 + it * 0.01)
        
        # Update
        h_new = h - lr * grad
        # Project: clip to [0,1] and fix sum
        h_new = np.clip(h_new, 0, 1)
        s = np.sum(h_new)
        if s > 1e-14:
            h_new *= target / s
        h_new = np.clip(h_new, 0, 1)
        
        h = h_new
    
    return h
```

```python
    """Gradient of O(lag) = sum_i h[i]*(1-h[i+lag]) w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        # O(0) = sum h[i](1-h[i]) = sum h[i] - sum h[i]^2
        # dO/dh[j] = 1 - 2*h[j]
        grad = 1.0 - 2.0 * h
    elif lag > 0:

            grad[i + lag] -= h[i]

        for i in range(lag, n):

            grad[i - lag] -= h[i]
```

```python
    h = h.astype(float).copy()
    for _ in range(15):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(free):
            h = np.clip(h, 0.0, 1.0)
            break
        step = diff / np.sum(free)
        step = np.clip(step, -0.99, 0.99)
        h[free] += step
    return np.clip(h, 0.0, 1.0)

def _interp_to_n(parent, pn, n):
    if n == pn:
        return parent.astype(float).copy()
    xp = np.linspace(0.0, 2.0, pn, endpoint=False) + 1.0 / pn
    fp = parent
    xq = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    return np.interp(xq, xp, fp)

def _grad_O(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        if lag >= n:
            return grad
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] -= h[i]
    else:
        lag = -lag
        if lag >= n:
            return grad
        for i in range(lag, n):
            grad[i] += (1.0 - h[i - lag])
            grad[i - lag] -= h[i]
    return grad

def _minmax_subgrad(h, n, target, n_iter=8000, lr0=0.004, n_top=5, time_limit=30.0):
    h = h.astype(float).copy()
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        total_grad = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            total_grad += _grad_O(h, lag)
        total_grad /= n_top
        norm_g = np.linalg.norm(total_grad)
        if norm_g < 1e-14:
            break
        lr = lr0 / np.sqrt(1.0 + it * 0.0005)
        h = h - lr * total_grad
        h = _project(h, target)
    return h

def _smooth_obj(x, alpha, dx, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -50.0 / alpha, 50.0 / alpha)
    return (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * dx

def _slsqp_opt(h, n, target, time_limit=60.0):
    t0 = time.time()
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    best_h = h.copy()
    best_c = _c5(h)
    alphas = [50, 150, 400, 1000, 3000]
    for al in alphas:
        if time.time() - t0 > time_limit - 5:
            break
        def obj(x):
            return _smooth_obj(x, al, dx, target)
        try:
            mi = 80 if n <= 90 else 50
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': mi, 'ftol': 1e-12})
            hn = _project(res.x, target)
            cn = _c5(hn)
            if cn < best_c:
                best_h, best_c = hn, cn
        except Exception:
            pass
    return best_h

def _grad_pair_refine(h, target, n_iter=30000, seed=0, time_limit=20.0):
    n = len(h)
    h = h.astype(float).copy()
    best = _c5(h)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-3:]
        grad_sum = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            grad_sum += _grad_O(h, lag)
        grad_sum /= 3.0
        # Find best two-coordinate move
        gi = np.argmin(grad_sum)
        gj = np.argmax(grad_sum)
        if gi == gj:
            continue
        # Move mass from j (increase grad → decrease h) to i (decrease grad → increase h)
        for step in (0.02, 0.05, 0.1, 0.01):
            if h[j] < step or h[i] > 1.0 - step:
                continue
            h2 = h.copy()
            h2[i] += step
            h2[j] -= step
            c = _c5(h2)
            if c < best - 1e-12:
                h = h2
                best = c
                break
        # Also try random pair
        if it % 5 == 0:
            ri, rj = rng.randint(n), rng.randint(n)
            if ri != rj:
                for step in (0.02, 0.05, 0.1):
                    if h[ri] < step or h[rj] > 1.0 - step:
                        continue
                    h2 = h.copy()
                    h2[ri] += step
                    h2[rj] -= step
                    c = _c5(h2)
                    if c < best - 1e-12:
                        h = h2
                        best = c
                        break
    return h, best

def _fix_feasibility(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    n_fix = 20
    for _ in range(n_fix):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(np.sum(free))
        if cnt == 0:
            h += diff / n
            break
        h[free] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-15:
        idx = np.argsort(np.abs(h - 0.5))[:int(np.abs(diff2) * 10) + 1]
        for i in idx:
            adj = np.sign(-diff2) * min(abs(diff2), abs(1.0 if diff2 < 0 else h[i]))
            h[i] += np.sign(diff2) * min(abs(diff2), 1e-4)
            diff2 = target - np.sum(h)
            if abs(diff2) < 1e-14:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)

    pn = len(parent)
    best_h = parent.copy()
    best_c5 = _c5(parent)
    best_n = pn

    n_values = sorted(set([pn, 64, 72, 80, 84, 90, 96, 100, 108, 112, 120, 128, 140, 160]))

    for n in n_values:
        if time.time() - t0 > budget_s - 40:
            break
        target = n / 2.0

        # Multiple initializations
        n_inits = 3 if n <= 90 else 2
        for init in range(n_inits):
            if time.time() - t0 > budget_s - 25:
                break
            if n == pn and init == 0:
                h = parent.astype(float).copy()
            elif init == 0:
                h = _interp_to_n(parent, pn, n)
            else:
                rng0 = np.random.RandomState(seed + n * 17 + init * 31)
                h = _interp_to_n(parent, pn, n)
                h += rng0.uniform(-0.08, 0.08, n)
            h = _project(h, target)

            # Phase A: Subgradient min-max
            rem = budget_s - (time.time() - t0)
            tl_sg = min(25.0, rem * 0.25)
            if tl_sg > 5:
                h = _minmax_subgrad(h, n, target, n_iter=10000, lr0=0.004,
                                    n_top=5, time_limit=tl_sg)
            c = _c5(h)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n
                print(f"  subgrad n={n} init={init}: C5={c:.6f}")

            # Phase B: SLSQP
            rem = budget_s - (time.time() - t0)
            tl_sq = min(50.0, rem * 0.3)
            if tl_sq > 8:
                h = _slsqp_opt(h, n, target, time_limit=tl_sq)
            c = _c5(h)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n
                print(f"  slsqp   n={n} init={init}: C5={c:.6f}")

            # Phase C: Gradient-guided pair refinement
            rem = budget_s - (time.time() - t0)
            tl_pr = min(25.0, rem * 0.2)
            if tl_pr > 5:
                h, c = _grad_pair_refine(h, target, n_iter=40000,
                                         seed=seed + n + init, time_limit=tl_pr)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n
                print(f"  pairref n={n} init={init}: C5={c:.6f}")

    # Final feasibility fix
    best_h = _fix_feasibility(best_h, best_n)
    best_c5 = _c5(best_h)
    print(f"Final: n={best_n}, C5={best_c5:.6f}")
    return best_h, best_c5, best_n
```
