import numpy as np
from scipy.optimize import minimize, linprog
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def _obj_func(x, dx):
    return eval_c5(x, dx)

def _con_eq_fun(x):
    return x.sum() - len(x) / 2.0

def _con_eq_jac(x):
    return np.ones(len(x))

def _mk_cons(n):
    return [{'type': 'eq', 'fun': _con_eq_fun, 'jac': _con_eq_jac}]

def slsqp_opt(h, n, dx, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = _mk_cons(n)
    try:
        res = minimize(_obj_func, h, args=(dx,), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def build_A(n, coeff):
    num_lags = 2 * n - 1
    A = np.zeros((num_lags, n))
    for k in range(num_lags):
        shift = n - 1 - k
        lo = max(0, -shift)
        hi = min(n, n - shift)
        if lo < hi:
            A[k, lo:hi] = coeff[lo + shift: hi + shift]
    return A

def solve_lp(n, coeff):
    num_lags = 2 * n - 1
    A = build_A(n, coeff)
    A_ub = np.hstack([A, -np.ones((num_lags, 1))])
    b_ub = np.zeros(num_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([float(n) / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, 1)] * n + [(0, None)], method='highs')
        if res.success:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, float('inf')

def alternating_lp(h_init, n, dx, max_iter=300, tol=1e-12):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)
    stalled = 0
    for it in range(max_iter):
        v = 1.0 - h
        x_new, _ = solve_lp(n, v)
        if x_new is not None:
            h_new = project_feasible(x_new, n)
            c5_new = eval_c5(h_new, dx)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                stalled = 0
            else:
                stalled += 1
            h = h_new
        v = 1.0 - h
        x_new, _ = solve_lp(n, h)
        if x_new is not None:
            h_new = 1.0 - x_new
            h_new = project_feasible(h_new, n)
            c5_new = eval_c5(h_new, dx)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                stalled = 0
            else:
                stalled += 1
            h = h_new
        if stalled > 20:
            break
    return project_feasible(best_h, n), eval_c5(project_feasible(best_h, n), dx)

def interpolate_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    return np.interp(x_dst, x_src, h_src)

def optimize_at_n(h_init, n, dx, t_start, budget_s):
    elapsed = time.time() - t_start
    if elapsed > budget_s - 40:
        return h_init, eval_c5(h_init, dx)
    h, c = slsqp_opt(h_init, n, dx, maxiter=2000, ftol=1e-15)
    elapsed = time.time() - t_start
    if elapsed < budget_s - 30:
        h2, c2 = alternating_lp(h, n, dx, max_iter=500)
        if c2 < c:
            h, c = h2, c2
    elapsed = time.time() - t_start
    if elapsed < budget_s - 15:
        h3, c3 = slsqp_opt(h, n, dx, maxiter=1500, ftol=1e-16)
        if c3 < c:
            h, c = h3, c3
    return h, c
