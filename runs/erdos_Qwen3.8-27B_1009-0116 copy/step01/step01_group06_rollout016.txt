            al = -lag
            jmax = n - al
            for j in range(jmax):
                grad[j] -= c * h[j + al]
                grad[j + al] += c * (1.0 - h[j])
    return grad

def slsqp_opt(h, n, dx, maxiter=1000, ftol=1e-14):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return eval_c5(x, dx)
    def ceq(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def lbfgs_smooth(h, n, dx, T=100, maxiter=2000):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        v = smooth_val(x, n, dx, T)
        pen = 500.0 * (x.sum() - target) ** 2
        return v + pen
    def grad(x):
        g = smooth_grad(x, n, dx, T)
        g += 1000.0 * (x.sum() - target) * np.ones(n)
        return g
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 50, 'maxfun': 50000})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def de_optimize(n, dx, x0, t0, budget_s):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        h = project_feasible(x, n)
        return eval_c5(h, dx)
    remaining = budget_s - 20 - (time.time() - t0)
    if remaining < 5:
        return x0, eval_c5(x0, dx)
    maxiter = min(80, int(remaining // 3))
    popsize = min(30, max(15, n // 2))
    try:
        res = differential_evolution(obj, bounds, seed=123, max_iter=maxiter,
                                      popsize=popsize, tol=1e-10, polish=False,
                                      mutation=(0.3, 0.8), recombination=0.7,
                                      init='sohal', updating='deferred')
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return x0, eval_c5(x0, dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    n = 70
    dx = 2.0 / n
    
    best_h = project_feasible(np.array(initial_h_values, dtype=float), n)
    best_c5 = eval_c5(best_h, dx)
    
    # Phase 1: SLSQP quick polish
    if time.time() - t0 < budget_s - 300:
        h, c = slsqp_opt(best_h, n, dx, maxiter=1500, ftol=1e-15)
        if c < best_c5:
            best_h, best_c5 = h, c
    
    # Phase 2: L-BFGS-B with smoothified objective, annealing T
    T_list = [20, 40, 80, 160, 320, 640]
    for idx, T in enumerate(T_list):
        if time.time() - t0 < budget_s - 150:
            h, c = lbfgs_smooth(best_h, n, dx, T=T, maxiter=3000)
            if c < best_c5:
                best_h, best_c5 = h, c
            # SLSQP after each smooth refine
            if time.time() - t0 < budget_s - 120:
                h, c = slsqp_opt(best_h, n, dx, maxiter=800, ftol=1e-15)
                if c < best_c5:
                    best_h, best_c5 = h, c
    
    # Phase 3: DE global search
    if time.time() - t0 < budget_s - 80:
        h, c = de_optimize(n, dx, best_h, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = h, c
        # Polish DE result
        if time.time() - t0 < budget_s - 40:
            h, c = lbfgs_smooth(best_h, n, dx, T=200, maxiter=2000)
            if c < best_c5:
                best_h, best_c5 = h, c
            if time.time() - t0 < budget_s - 20:
                h, c = slsqp_opt(best_h, n, dx, maxiter=1500, ftol=1e-15)
                if c < best_c5:
                    best_h, best_c5 = h, c
    
    # Phase 4: Perturbation + local optimization
    attempt = 0
    while time.time() - t0 < budget_s - 3:
        attempt += 1
        if attempt % 6 == 0:
            scale = 0.02
        elif attempt % 6 == 1:
            scale = 0.05
        elif attempt % 6 == 2:
            scale = 0.10
        elif attempt % 6 == 3:
            scale = 0.20
        elif attempt % 6 == 4:
            scale = 0.35
        else:
            scale = 0.08
        h0 = best_h + scale * np.random.randn(n)
        h0 = project_feasible(h0, n)
        if scale <= 0.10 and time.time() - t0 < budget_s - 10:
            h, c = lbfgs_smooth(h0, n, dx, T=150, maxiter=1000)
            if c < best_c5:
                h, c = slsqp_opt(h, n, dx, maxiter=500, ftol=1e-14)
                if c < best_c5:
                    best_h, best_c5 = h, c
        else:
            h, c = slsqp_opt(h0, n, dx, maxiter=400, ftol=1e-13)
            if c < best_c5:
                best_h, best_c5 = h, c
    
    # Final: one more SLSQP polish
    if time.time() - t0 < budget_s - 1:
        h, c = slsqp_opt(best_h, n, dx, maxiter=1000, ftol=1e-16)
        if c < best_c5:
            best_h, best_c5 = h, c
    
    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
