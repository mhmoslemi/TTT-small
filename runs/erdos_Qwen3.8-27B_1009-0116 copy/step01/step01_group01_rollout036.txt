import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-13:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def subgrad_iter(h, n, lr):
    """One subgradient iteration. Returns new h and the max corr value."""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    m = k_star - (n - 1)
    
    g = np.zeros(n)
    if m >= 0:
        g[:n-m] += (1.0 - h[m:])
        g[m:] += -h[:n-m]
    else:
        p = -m
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] += -h[p:]
    
    h_new = h - lr * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair(h_new, n)
    return h_new, float(np.max(c) * 2.0 / n)

def subgrad_descent(h, n, n_iter, lr_init=0.05, decay=0.999):
    h = h.copy()
    best_h = h.copy()
    best = compute_c5(h)
    lr = lr_init
    
    for it in range(n_iter):
        h, c5 = subgrad_iter(h, n, lr)
        if c5 < best:
            best = c5
            best_h = h.copy()
        lr *= decay
    
    return best_h, best

def binary_polish(h, n, n_passes=3):
    """Try 0/1 for each coordinate, adjusting partner to maintain sum."""
    h = h.copy()
    best = compute_c5(h)
    target = n * 0.5
    
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            j = (i + 1) % n
            s = h[i] + h[j]
            old_i, old_j = h[i], h[j]
            best_local = best
            best_i = old_i
            
            # Try: 0, 1, and a few fractions
            for vi in (0.0, 1.0):
                vj = s - vi
                if 0.0 <= vj <= 1.0 and abs(vi - old_i) > 1e-9:
                    h[i], h[j] = vi, vj
                    v = compute_c5(h)
                    if v < best_local - 1e-12:
                        best_local = v
                        best_i = vi
            
            if best_i != old_i:
                h[j] = s - best_i
                best = best_local
                improved = True
            else:
                h[i], h[j] = old_i, old_j
        
        if not improved:
            break
    
    return repair(h, n), best

def interleaved_polish(h, n, n_passes=2):
    """Try intermediate values for coordinates not at 0/1."""
    h = h.copy()

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-13:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def get_subgrad(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    m = k_star - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[:n - m] += (1.0 - h[m:])
        g[m:] += -h[:n - m]
    else:
        p = -m
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] += -h[p:]
    return g, c
