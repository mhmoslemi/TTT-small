import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smooth_val_grad(h, dx, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = c.max()
    sc = alpha * (c - cmax)
    exp_sc = np.exp(sc)
    s_exp = exp_sc.sum()
    p = exp_sc / s_exp
    val = cmax + np.log(s_exp) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-12:
            continue
        off = k - (n - 1)
        if off >= 0:
            n1 = n - off
            if n1 <= 0:
                continue
            grad[:n1] += w * (1.0 - h[off:])
            grad[off:] -= w * h[:n1]
        else:
            oa = -off
            n1 = n - oa
            if n1 <= 0:
                continue
            grad[oa:] += w * (1.0 - h[:n1])
            grad[:n1] -= w * h[oa:]
    grad *= dx
    return val, grad

def adam_pgd(h_init, n, dx, alphas, iters, t_end=None):
    h = h_init.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.08
    for alpha in alphas:
        for _ in range(iters):
            t += 1
            if t_end is not None and time.time() > t_end:
                return best_h, best_c5
            val, g = smooth_val_grad(h, dx, n, alpha)
            m_vec = b1 * m_vec + (1 - b1) * g
            v_vec = b2 * v_vec + (1 - b2) * g * g
            m_hat = m_vec / (1 - b1 ** t)
            v_hat = v_vec / (1 - b2 ** t)
            lr = lr0 / (1.0 + 0.05 * t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_feasible(h_new, n)
            c = eval_c5(h_new, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def lbgfsb_smooth(h, n, dx, alpha, maxiter=200, penalty=50.0):
    target = float(n) / 2.0
    def f(x):
        v, _ = smooth_val_grad(x, dx, n, alpha)
        return v + penalty * (x.sum() - target) ** 2
    def df(x):
        _, g = smooth_val_grad(x, dx, n, alpha)
        return g + 2.0 * penalty * (x.sum() - target) * np.ones(n)
    try:
        res = minimize(f, h, jac=df, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_out = project_feasible(res.x, n)
        return h_out, eval_c5(h_out, dx)
    except Exception:
        return h, eval_c5(h, dx)

def local_refine(h, n, dx):
    """Small-step coordinate perturbations to squeeze more out."""
    h = h.copy()
    best_c5 = eval_c5(h, dx)
    target = float(n) / 2.0
    for _ in range(200):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = 0.02 * np.random.choice([-1.0, 1.0])
        h[i] += delta
        h[j] += (target - h.sum())
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)
        h = project_feasible(h, n)
        c = eval_c5(h, dx)
        if c < best_c5:
            best_c5 = c
        else:
            h[i] -= delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h = project_feasible(h, n)
    return h, best_c5

    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    n = 70
    dx = 2.0 / n
    try:
        h0 = np.array(initial_h_values, dtype=float)
        if len(h0) != n:
            h0 = np.full(n, 0.5)
    except Exception:
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0, n)

    best_n = n

    # Phase 1: L-BFGS-B with moderate smoothing
    if time.time() < t_end - 200:
        h1, c1 = lbgfsb_smooth(best_h, n, dx, alpha=20, maxiter=300)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Annealed Adam PGD (main workhorse)
    alphas = [5, 10, 20, 50, 100, 300, 1000, 3000]
    if time.time() < t_end - 100:
        h2, c2 = adam_pgd(best_h, n, dx, alphas, 60, t_end=t_end - 100)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: L-BFGS-B with high alpha
    if time.time() < t_end - 80:
        h3, c3 = lbgfsb_smooth(best_h, n, dx, alpha=5000, maxiter=200, penalty=100.0)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Local coordinate refinement
    if time.time() < t_end - 60:
        h4, c4 = local_refine(best_h, best_n, 2.0 / best_n)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: Try other n values with structured starts
    for nt in [60, 80, 100, 120]:
        if time.time() > t_end - 80:
            break
        dxt = 2.0 / nt
        starts = []
        starts.append((np.arange(nt) % 2 == 0).astype(float))
        starts.append((np.sin(np.linspace(0, 4 * np.pi, nt)) > 0).astype(float))
        starts.append(np.random.rand(nt))
        for ht in starts:
            if time.time() > t_end - 40:

            ht = project_feasible(ht, nt)
            ct = eval_c5(ht, dxt)
            if ct < best_c5:
                best_h, best_c5, best_n = ht.copy(), ct, nt
            ht2, ct2 = adam_pgd(ht, nt, dxt, [10, 50, 200, 1000], 40, t_end=t_end - 40)
            if ct2 < best_c5:
                best_h, best_c5, best_n = ht2, ct2, nt

    # Phase 6: Random restarts on best_n
    attempts = 0
    dx_best = 2.0 / best_n
    while time.time() < t_end - 30:
        attempts += 1
        scale = [0.2, 0.1, 0.05, 0.02][attempts % 4]
        hr = best_h + scale * np.random.randn(best_n)
        hr = project_feasible(hr, best_n)
        hr2, cr2 = adam_pgd(hr, best_n, dx_best, [50, 200, 1000], 40, t_end=t_end - 30)
        if cr2 < best_c5:
            best_h, best_c5 = hr2, cr2

    # Phase 7: Final polish
    if time.time() < t_end - 5:
        hf, cf = lbgfsb_smooth(best_h, best_n, dx_best, alpha=10000, maxiter=100, penalty=200.0)
        if cf < best_c5:
            best_h, best_c5 = hf, cf
        hf2, cf2 = local_refine(best_h, best_n, dx_best)
        if cf2 < best_c5:
            best_h, best_c5 = hf2, cf2

    best_h = project_feasible(best_h, best_n)
    best_c5 = eval_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
