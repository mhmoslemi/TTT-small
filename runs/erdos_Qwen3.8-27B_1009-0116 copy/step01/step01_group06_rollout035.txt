def project_feasible(h, n):

    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))

    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def grad_smooth_max(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= weights.sum()
    n = len(h)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        lag = int(k_idx) - (n - 1)
        for mm in range(n):
            if 0 <= mm - lag < n:
                grad[mm] += w * (1.0 - h[mm - lag])
            if 0 <= mm + lag < n:

            for t in cands:
                v = np.max(a + t * d)
                if v < best_v - 1e-15:
                    best_v = v
                    best_t = t
            if abs(best_t - old) > 1e-12:
                h[i] = best_t
                diff_sum = target - h.sum()
                if abs(diff_sum) > 1e-10:
                    if diff_sum > 0:
                        room = 1.0 - h
                        room[i] = 0.0
                        s = room.sum()
                        if s > 1e-14:
                            h += diff_sum * room / s
                    else:
                        msk = h.copy()
                        msk[i] = 0.0
                        s = msk.sum()
                        if s > 1e-14:
                            h += diff_sum * msk / s
                    h = np.clip(h, 0.0, 1.0)
                    d2 = target - h.sum()
                    if abs(d2) > 1e-10:
                        h[i] = np.clip(h[i] + d2, 0.0, 1.0)
                        h = np.clip(h, 0.0, 1.0)
                any_change = True
        c5 = eval_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
        if not any_change:
            break
    return h, best_c5
