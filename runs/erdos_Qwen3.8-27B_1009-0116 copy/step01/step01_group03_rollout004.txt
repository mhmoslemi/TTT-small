def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    adj = diff / len(idx)
    h[idx] = np.clip(h[idx] + adj, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        for i in idx:
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(h[i] + diff2, 1.0)
                diff2 -= (h[i] - max(h[i]-diff2, 0))
            elif diff2 < 0 and h[i] > 0.0:
                old = h[i]
                h[i] = max(h[i] + diff2, 0.0)
                diff2 += (old - h[i])
            if abs(diff2) < 1e-10:
                break
    return h

def slsqp_direct(vars, n):

        return t - corr * (2.0 / n)  # >= 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Fine-tune
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def slsqp_solve_direct(n, h_init, maxiter=2000):
    """Direct min-t formulation."""
    dx = 2.0 / n
    t_init = compute_c5(h_init) + 0.01
    x0 = np.concatenate([h_init, [t_init]])
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr * dx
    
    def eq(x):
        return x[:-1].sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq}]
    
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x[:-1], n/2.0)
    return project_feasible(h_init, n/2.0)

def simulated_annealing(h, n, ts, t_start, t_end, n_iter, seed):
    """SA on continuous [0,1]^n with sum constraint."""
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    
    for i in range(n_iter):
        T = t_start * (t_end / t_start) ** (i / n_iter)
        # Perturb 1-3 random coordinates
        n_pert = rng.randint(1, 4)
        idx = rng.choice(n, n_pert, replace=False)
        new_h = cur_h.copy()
        new_h[idx] += rng.uniform(-0.5, 0.5, n_pert)
        new_h = project_feasible(new_h, ts)
        new_c5 = compute_c5(new_h)
        
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10)):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        # else: reject, keep cur_h
    
    return best_h, best_c5

def pair_swap_search(h, n, ts, rng, max_sweeps=50):
    """Local search with pair-swap moves."""
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    ones = np.where(cur_h > 0.99)[0]
    zeros = np.where(cur_h < 0.01)[0]
    
    for sweep in range(max_sweeps):
        improved = False
        # Pair swaps: move a 1 to a 0 position
        for i in ones.copy():
            for j in zeros.copy():
                if i == j:
                    continue
                cur_h[i] = 0.0
                cur_h[j] = 1.0
                c5 = compute_c5(cur_h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                    ones = np.where(cur_h > 0.99)[0]
                    zeros = np.where(cur_h < 0.01)[0]
                else:
                    cur_h[i] = 1.0
                    cur_h[j] = 0.0
        
        # Single flips (for non-binary solutions)
        for i in range(n):
            old = cur_h[i]
            if old > 0.99 or old < 0.01:
                continue
            # Try setting to 0 or 1
            for target in (0.0, 1.0):
                cur_h[i] = target
                cur_h = project_feasible(cur_h, ts)
                c5 = compute_c5(cur_h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                else:
                    cur_h[i] = old
                    cur_h = project_feasible(cur_h, ts)
        
        if not improved:
            break
    
    return cur_h, best_c5

def make_periodic(n, period, ones_per_period, seed):
    """Create a periodic pattern."""
    rng = np.random.RandomState(seed)
    # Create one period
    p = period
    pattern = np.zeros(p)
    positions = rng.choice(p, min(ones_per_period, p), replace=False)
    pattern[positions] = 1.0
    # Tile to fill n
    h = np.tile(pattern, n // p + 1)[:n]
    # Adjust to have exactly n/2 ones
    h = project_feasible(h, n/2.0)
    # Round to binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Fix sum
    n_ones = int(h.sum())
    if n_ones != n//2:
        if n_ones > n//2:
            excess = n_ones - n//2
            idx = np.where(h == 1.0)[0]
            h[idx[:excess]] = 0.0
        else:
            deficit = n//2 - n_ones
            idx = np.where(h == 0.0)[0]
            h[idx[:deficit]] = 1.0
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def slsqp_smooth(h, n, ts, beta, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        shifted = (corr - m) * beta
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def slsqp_direct(n, h_init, maxiter):
    dx = 2.0 / n
    t0 = compute_c5(h_init) + 0.005
    x0 = np.concatenate([h_init, [t0]])
    def obj(x):
        return x[-1]
    def ineq(x):
        h = x[:-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return x[-1] - corr * dx
    def eq(x):
        return x[:-1].sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq}]
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x[:-1], n / 2.0)
    return project_feasible(h_init, n / 2.0)

def simulated_anneal(h, n, ts, t_hi, t_lo, n_iter, seed):
    rng = np.random.RandomState(seed)
    best_h, best_c5 = h.copy(), compute_c5(h)
    cur_h, cur_c5 = h.copy(), best_c5
    for i in range(n_iter):
        T = t_hi * (t_lo / t_hi) ** (i / max(n_iter - 1, 1))
        npert = rng.randint(1, 4)
        idx = rng.choice(n, npert, replace=False)
        new_h = cur_h.copy()
        new_h[idx] += rng.uniform(-0.6, 0.6, npert)
        new_h = project_feasible(new_h, ts)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h.copy()
    return best_h, best_c5

def pair_swap_search(h, n, ts, rng, max_sweeps):
