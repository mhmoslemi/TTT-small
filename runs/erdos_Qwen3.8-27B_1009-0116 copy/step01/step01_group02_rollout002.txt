    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    """Compute subgradient of overlap at shift s w.r.t. h."""
    # shift s means we compute sum_i h[i] * (1 - h[i+s]) for valid i
    # s ranges from -(n-1) to n-1
    # In np.correlate(h, 1-h, 'full'), the index is s + (n-1)
    # Let's compute directly
    
    g = np.zeros(n)
    # For valid i: 0 <= i < n and 0 <= i+s < n
    # i ranges from max(0, -s) to min(n-1, n-1-s)
    i_start = max(0, -s)
    i_end = min(n, n - s)  # exclusive
    
    for i in range(i_start, i_end):
        j = i  # h[i]
        k = i + s  # h[i+s]
        # d/dh[j] of h[j]*(1-h[k]) = (1-h[k])
        # d/dh[k] of h[j]*(1-h[k]) = -h[j]
        if s != 0:
            g[j] += (1.0 - h[k])
            g[k] += -h[j]
        else:
            # h[i]*(1-h[i]), derivative = 1 - 2*h[i]
            g[j] += (1.0 - 2.0 * h[i])
    return g

def subgradient_descent(h, n, max_iter=5000, lr_init=0.01):

    h = repair(h, n)
    c5 = compute_c5(h, n)

        if c5_new < c5:
            h = h_new
            c5 = c5_new
    
    return h, c5

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 0.005) & (h < 0.995)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    if abs(target - np.sum(h)) > 1e-10:
        h[:] += (target - np.sum(h)) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_idx(h, n, k_idx):
    g = np.zeros(n)
    t_start = max(0, -k_idx)
    t_end = min(n, n - k_idx)

    t = np.arange(t_start, t_end)
    a = t + k_idx
    b = t

        g[a] = 1.0 - 2.0 * h[a]
    else:
        g[a] += (1.0 - h[b])
        g[b] += -h[a]

        g += wi * grad_corr_idx(h, n, idx)
        w += wi

        return np.sum(x) - target

    try:

                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return h
