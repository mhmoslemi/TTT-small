    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def flip_delta(h, j):
    n = len(h)
    m_vals = np.arange(-(n - 1), n)
    jp = j + m_vals
    jm = j - m_vals
    mp = (jp >= 0) & (jp < n)
    mm = (jm >= 0) & (jm < n)
    hp = np.where(mp, h[np.clip(jp, 0, n - 1)], 0.0)
    hm = np.where(mm, h[np.clip(jm, 0, n - 1)], 0.0)
    return 1.0 - hp - hm

def binary_flip_search(h, max_sweeps=200, deadline=1e18, temperature=0.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    c = get_corr(h)
    cmax = float(np.max(c))
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            d = flip_delta(h, j)
            if h[j] > 0.5:
                new_c = c - d
            else:
                new_c = c + d
            new_max = float(np.max(new_c))
            if temperature > 0 and new_max > cmax:
                if np.random.rand() > np.exp(-(new_max - cmax) * dx / temperature):
                    continue
            if new_max < cmax - 1e-12:
                h[j] = 1.0 - h[j]
                c = new_c
                cmax = new_max
                improved = True
        if not improved:
            break
    return h, cmax * dx

def pgd_optimize(h, n, T_list, max_iter, deadline):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    for T in T_list:
        h = best_h.copy()
        lr = 0.5
        for it in range(max_iter):
            if time.time() > deadline:
                break
            c = get_corr(h)
            cmax = np.max(c)
            exp_c = np.exp((c - cmax) * T * dx)
            w = exp_c / exp_c.sum()
            js = np.arange(n)[:, None]
            ms = (np.arange(2 * n - 1) - (n - 1))[None, :]
            idx1 = js + ms
            idx2 = js - ms
            m1 = (idx1 >= 0) & (idx1 < n)
            m2 = (idx2 >= 0) & (idx2 < n)
            t1 = np.where(m1, h[np.clip(idx1, 0, n - 1)], 0.0) @ w
            t2 = np.where(m2, h[np.clip(idx2, 0, n - 1)], 0.0) @ w
            g = (1.0 - t1 - t2) * dx
            h_new = np.clip(h - lr * g, 0.0, 1.0)
            h_new = repair(h_new, n)
            v = compute_c5(h_new)
            if v < best_c:
                best_c = v
                best_h = h_new.copy()
                lr = min(lr * 1.04, 2.0)
            else:
                lr *= 0.8
            h = h_new
    return best_h, best_c

def slsqp_polish(h, n, deadline):
    dx = 2.0 / n
    T_s = 400.0

    def obj(x):
        c = get_corr(x)
        cm = np.max(c)
        return (cm + np.log(np.sum(np.exp((c - cm) * T_s))) / T_s) * dx

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    maxit = max(50, int((deadline - time.time()) / 0.5))
    res = minimize(obj, h.copy(), method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxit, 'ftol': 1e-13})
    h_r = repair(res.x, n)
    return h_r, compute_c5(h_r)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Binary search from rounded initial
    h_bin = (h0 > 0.5).astype(float)
    h_bin = repair(h_bin, n0)
    dl1 = t0 + budget_s * 0.12
    h_b, c_b = binary_flip_search(h_bin, 300, dl1)
    if c_b < best_c5:
        best_c5, best_h = c_b, h_b.copy()

    # Phase 2: Multiple random restarts with binary + annealing
    dl2 = t0 + budget_s * 0.35
    for restart in range(12):
        if time.time() > dl2:
            break
        noise = np.random.randn(n0) * 0.3
        h_try = best_h + noise
        h_try = repair(h_try, n0)
        h_rounded = np.round(h_try)
        h_rounded = repair(h_rounded, n0)
        dl_r = min(dl2, time.time() + budget_s * 0.02)
        h_r, c_r = binary_flip_search(h_rounded, 100, dl_r, temperature=0.02)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r.copy()
        h_rounded2 = np.round(h_try + np.random.randn(n0) * 0.1)
        h_rounded2 = repair(h_rounded2, n0)
        dl_r2 = min(dl2, time.time() + budget_s * 0.015)
        h_r2, c_r2 = binary_flip_search(h_rounded2, 80, dl_r2, temperature=0.01)
        if c_r2 < best_c5:
            best_c5, best_h = c_r2, h_r2.copy()

    # Phase 3: PGD continuous refinement
    if time.time() < t0 + budget_s * 0.55:
        dl3 = t0 + budget_s * 0.55
        h_p, c_p = pgd_optimize(best_h.copy(), n0, [150.0, 300.0], 400, dl3)
        if c_p < best_c5:
            best_c5, best_h = c_p, h_p.copy()

    # Phase 4: SLSQP polish
    if time.time() < t0 + budget_s * 0.78:
        dl4 = t0 + budget_s * 0.78
        h_s, c_s = slsqp_polish(best_h, n0, dl4)
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s.copy()

    # Phase 5: Binary search from polished continuous
    if time.time() < t0 + budget_s * 0.90:
        h_bin5 = np.round(best_h)
        h_bin5 = repair(h_bin5, n0)
        dl5 = t0 + budget_s * 0.90
        h_b5, c_b5 = binary_flip_search(h_bin5, 300, dl5)
        if c_b5 < best_c5:
            best_c5, best_h = c_b5, h_b5.copy()

    # Phase 6: Try higher resolution (interpolate + optimize)
    if time.time() < t0 + budget_s * 0.95 and n0 <= 400:
        n_h = n0 * 2
        xo = np.linspace(0.0, 2.0, n0)
        xn = np.linspace(0.0, 2.0, n_h)
        h_interp = np.interp(xn, xo, best_h)
        h_interp = repair(h_interp, n_h)
        h_rounded_h = np.round(h_interp)
        h_rounded_h = repair(h_rounded_h, n_h)
        dl6 = t0 + budget_s * 0.97
        h_h, c_h = binary_flip_search(h_rounded_h, 60, dl6)
        if c_h < best_c5:
            best_c5, best_h = c_h, h_h.copy()

    # Final repair and evaluation
    n_final = len(best_h)
    best_h = repair(best_h, n_final)
    c5_final = compute_c5(best_h)
    best_c5 = min(best_c5, c5_final)

    return (best_h, best_c5, n_final)
