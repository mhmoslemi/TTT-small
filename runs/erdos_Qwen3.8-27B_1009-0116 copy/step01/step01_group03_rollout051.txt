            if time_rem(t0, budget_s) > 12:
                h_p = slsqp_solve(best_h, nr, nr / 2.0, 400.0, maxiter=400)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p

    # Final polish
    if time_rem(t0, budget_s) > 8:
        nr = len(best_h)
        h_p = slsqp_solve(best_h, nr, nr / 2.0, 1200.0, maxiter=400)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
