```python
def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:

        else:

            if s > 1e-14:
```

Actually, let me think about this more carefully. For the discrete problem:

corr_full[idx] = sum_i h[i] * (1-h[i+lag]) for valid i, where lag = idx-(n-1)

If I change h[j] from old to new:
- For lags where j appears in the "first" factor (i=j): corr[idx] changes by (new-old)*(1-h[j+lag])
- For lags where j appears in the "second" factor (i+lag=j, i=j-lag): corr[idx] changes by -new*h[j-lag] + old*h[j-lag] = (old-new)*h[j-lag]

So the change in corr[idx] is:
Δcorr[idx] = (new-old)*[(1-h[j+lag]) - h[j-lag]]

for the specific lag where both j and j±lag are in range.

This means for a given j, each lag is affected by the same coefficient (new-old), just with different multiplicands.

For coordinate descent, the objective as a function of v = h[j]:
C₅(v) = max_k [corr_k + (v - h[j]) * a_k]
where a_k = (1-h[j+lag_k]) - h[j-lag_k] (for the lag where j is involved)

This is the max of linear functions in v, so it's convex in v! We can find the optimal v by binary search or by checking the breakpoints.

The optimal v minimizes max_k [corr_k + (v-h[j])*a_k] subject to 0 ≤ v ≤ 1 and maintaining the sum constraint.

If we ignore the sum constraint, the optimum is where the active constraints cross. With the sum constraint, we'd need to adjust another coordinate.

This suggests a good local improvement: coordinate descent with optimal step in each coordinate, followed by a projection to fix the sum.

```python
def coord_descent(h, n, dx, max_sweeps=100, t0=None, budget=None):
    h = h.copy()
    best_c5 = eval_c5(h, dx)
    target = n / 2.0
    for sweep in range(max_sweeps):
        if t0 is not None and budget is not None and time.time() - t0 > budget:
            break
        improved = False
        for j in range(n):
            # Compute the effect of changing h[j]
            # corr[k] for each k, and how it changes with h[j]
            # For lag = idx - (n-1):
            #   h[j] in first factor: valid if j+lag in [0,n), i.e., lag in [-j, n-1-j]
            #   h[j] in second factor: valid if j-lag in [0,n), i.e., lag in [j-(n-1), j]
            #   Combined: lag in [j-(n-1), n-1-j] intersect [j-(n-1), j]
            #   = [j-(n-1), min(j, n-1-j)]
            
            # For each lag, the coefficient a_lag:
            # First factor: (1 - h[j+lag]) if j+lag in [0,n)
            # Second factor: -h[j-lag] if j-lag in [0,n)
            
            # The "base" corr values:
            corr = np.correlate(h, 1.0-h, mode='full') * dx
            n_full = 2*n-1
            lags = np.arange(n_full) - (n-1)
            
            # Coefficients: for each lag, how much does corr[lag] change per unit change in h[j]?
            # From first factor: (1-h[j+lag]) if j+lag in [0,n)
            # From second factor: -h[j-lag] if j-lag in [0,n)
            coeff = np.zeros(n_full)
            for idx in range(n_full):
                lag = lags[idx]
                c = 0.0
                if 0 <= j+lag < n:
                    c += (1.0 - h[j+lag])
                if 0 <= j-lag < n:
                    c -= h[j-lag]
                coeff[idx] = c
            
            # C5(v) = max_k [corr[k] + (v - h[j]) * coeff[k]]
            # We want to find v in [0,1] that minimizes this
            # This is the lower envelope of lines, or rather the minimum of the upper envelope
            # Actually: minimize max_k [corr[k] + (v-h[j])*coeff[k]]
            # = minimize max_k [corr[k] + coeff[k]*v - coeff[k]*h[j]]
            # The second term is just a shift, so: minimize max_k [corr[k] + coeff[k]*v]
            # Let's define f_k(v) = corr[k] + coeff[k]*v
            # We want min_v max_k f_k(v) for v in [0,1]
            
            # This is a 1D minimax problem. The function g(v) = max_k f_k(v) is convex (max of linear).
            # Its minimum is where the subgradient contains 0, or at a boundary.
            
            # Simple approach: ternary search or grid search on [0,1]
            # Since it's convex, ternary search works
            def g(v):
                if v < 0 or v > 1:
                    v = np.clip(v, 0, 1)
                vals = corr + coeff * v
                return np.max(vals)
            
            # Use more efficient: find the optimal v
            # g(v) is convex piecewise-linear. Minimum is at a breakpoint or boundary.
            # Breakpoints are where two lines cross: corr[i]+coeff[i]*v = corr[j]+coeff[j]*v
            # v = (corr[j]-corr[i])/(coeff[i]-coeff[j])
            
            # For efficiency, just do a fine grid + local search
            v_candidates = np.linspace(0, 1, 101)
            g_vals = g(v_candidates)
            v_best = v_candidates[np.argmin(g_vals)]
            
            # Refine around v_best
            v_fine = np.linspace(max(0, v_best-0.01), min(1, v_best+0.01), 50)
            g_fine = g(v_fine)
            v_best = v_fine[np.argmin(g_fine)]
            
            if g(v_best) < best_c5 - 1e-10:
                h[j] = v_best
                # Fix sum
                h = project_feasible(h, n)
                c5_new = eval_c5(h, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        
        if not improved:
            break
    
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def compute_smooth_grad(h, dx, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(temp * (corr - m), -50, 50)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    grad = np.zeros(n)
    n_full = 2 * n - 1
    lags = np.arange(n_full) - (n - 1)
    for idx in range(n_full):
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = lags[idx]
        if lag >= 0:
            jlo, jhi = 0, n - lag
        else:
            jlo, jhi = -lag, n
        if jlo < jhi:
            grad[jlo:jhi] += w * (1.0 - h[jlo + lag:jhi + lag])
        if lag >= 0:
            jlo2, jhi2 = lag, n
        else:
            jlo2, jhi2 = 0, n + lag
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= w * h[jlo2 - lag:jhi2 - lag]
    return grad

def coord_descent(h0, n, dx, max_sweeps=80, t_limit=None):
```
