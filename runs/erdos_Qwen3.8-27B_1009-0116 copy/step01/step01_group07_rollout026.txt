def _eq_fun(x):
    return np.sum(x) - _g_target

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -500.0 / _g_alpha, 500.0 / _g_alpha)
    return (m + np.log(np.sum(np.exp(_g_alpha * sh))) / _g_alpha) * _g_dx

def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / len(h))

def _norm(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-13:
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _init(n, target, rng, kind):
    x = np.arange(n, dtype=float)
    if kind == 0:
        h = np.full(n, 0.5)
    elif kind == 1:
        f = rng.randint(1, 5)
        h = 0.5 + 0.3 * np.sin(2.0 * np.pi * f * x / n)
    elif kind == 2:
        ph = rng.uniform(0, 2.0 * np.pi)
        h = 0.5 + 0.25 * np.sin(4.0 * np.pi * x / n + ph) + 0.15 * np.cos(6.0 * np.pi * x / n)
    elif kind == 3:
        h = 0.5 + 0.35 * (2.0 * x / n - 1.0)
    else:
        h = 0.5 + 0.3 * np.abs(2.0 * x / n - 1.0) - 0.15
    return _norm(h, target)

def _cd_cycle(h, n_passes, n_trials=7):
    n = len(h)
    best = _c5(h)
    fracs = np.array([0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0])[:n_trials]
    for _p in range(n_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                a, b = h[i], h[j]
                lo = max(a - 1.0, -b)
                hi = min(a, 1.0 - b)
                if hi - lo < 1e-12:
                    continue
                span = hi - lo
                bd, bv = 0.0, best
                for t in fracs:
                    d = lo + t * span
                    h[i] -= d
                    h[j] += d
                    c = _c5(h)
                    if c < bv:
                        bv, bd = c, d
                    h[i] += d
                    h[j] -= d
                if bd != 0.0:
                    h[i] -= bd
                    h[j] += bd
                    best = bv
                    improved = True
        if not improved:
            break
    return h, best

def _cd_random(h, n_iter, rng):
    n = len(h)
    best = _c5(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        a, b = h[i], h[j]
        lo = max(a - 1.0, -b)
        hi = min(a, 1.0 - b)
        if hi - lo < 1e-12:
            continue
        span = hi - lo
        bd, bv = 0.0, best
        for _t in range(6):
            d = rng.uniform(lo, hi)
            h[i] -= d
            h[j] += d
            c = _c5(h)
            if c < bv:
                bv, bd = c, d
            h[i] += d
            h[j] -= d
        if bd != 0.0:
            lo2, hi2 = max(lo, bd - span * 0.3), min(hi, bd + span * 0.3)
            for _t in range(5):
                d = rng.uniform(lo2, hi2)
                h[i] -= d
                h[j] += d
                c = _c5(h)
                if c < bv:
                    bv, bd = c, d
                h[i] += d
                h[j] -= d
            h[i] -= bd
            h[j] += bd
            best = bv
    return h, best

def _sa(h, n_iter, rng, t_start, t_end):
    n = len(h)
    best, bh = _c5(h), h.copy()
    T = t_start
    cool = (t_end / t_start) ** (1.0 / max(1, n_iter))
    for _ in range(n_iter):
        T *= cool
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.05, 0.05)
        ni, nj = h[i] + delta, h[j] - delta
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c = _c5(h)
        if c < best or rng.uniform() < np.exp(-(c - best) / max(T, 1e-12)):
            if c < best:
                best, bh = c, h.copy()
        else:
            h[i], h[j] = ni - delta, nj + delta
    return bh, best

def _polish(h, n_passes):
    n = len(h)
    best = _c5(h)
    fracs = np.linspace(0.0, 1.0, 21)
    for _p in range(n_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                a, b = h[i], h[j]
                lo = max(a - 1.0, -b)
                hi = min(a, 1.0 - b)
                if hi - lo < 1e-12:
                    continue
                span = hi - lo
                bd, bv = 0.0, best
                for t in fracs:
                    d = lo + t * span
                    h[i] -= d
                    h[j] += d
                    c = _c5(h)
                    if c < bv:
                        bv, bd = c, d
                    h[i] += d
                    h[j] -= d
                if bd != 0.0:
                    h[i] -= bd
                    h[j] += bd
                    best = bv
                    improved = True
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    global _g_target, _g_alpha, _g_dx
    t0 = time.time()
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)
    pn = len(parent)
    best_c5, best_h, best_n = _c5(parent), parent.copy(), pn
    res_list = sorted(set([pn, 60, 80, 90, 100, 120]))
    res_list = [r for r in res_list if 30 <= r <= 150]

    for n in res_list:
        if time.time() - t0 > budget_s - 50:
            break
        target, dx = n / 2.0, 2.0 / n
        _g_target, _g_dx = target, dx
        bounds = [(0.0, 1.0)] * n
        rng = np.random.RandomState(seed + n * 17)
        inits = []
        if n == pn:
            inits.append(parent.copy())
        else:
            xp, xn = np.linspace(0, 2, pn), np.linspace(0, 2, n)
            inits.append(_norm(np.interp(xn, xp, parent), target))
        for k in range(4):
            inits.append(_init(n, target, rng, k))

        for h0 in inits:
            if time.time() - t0 > budget_s - 20:
                break
            h = h0.copy()
            for alpha in [80, 400, 1500]:
                if time.time() - t0 > budget_s - 15:
                    break
                _g_alpha = alpha
                try:
                    mi = 50 if n <= 90 else 30
                    res = minimize(_smooth_obj, h, method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': _eq_fun}],
                                   options={'maxiter': mi, 'ftol': 1e-12})
                    h = _norm(res.x, target)
                except Exception:
                    pass
            h, c = _cd_cycle(h, 8 if n <= 90 else 5)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n
                print(f"n={n} cdc: {c:.6f}")
            if time.time() - t0 < budget_s - 30:
                h, c = _sa(h, 1500, np.random.RandomState(seed + n + 31), 0.002, 0.00005)
                if c < best_c5:
                    best_c5, best_h, best_n = c, h.copy(), n
                    print(f"n={n} sa: {c:.6f}")
            if time.time() - t0 < budget_s - 15:
                h, c = _cd_random(h, 200, np.random.RandomState(seed + n + 41))
                if c < best_c5:
                    best_c5, best_h, best_n = c, h.copy(), n
                    print(f"n={n} cdr: {c:.6f}")
            if time.time() - t0 < budget_s - 10:
                h, c = _cd_cycle(h, 3)
                if c < best_c5:
                    best_c5, best_h, best_n = c, h.copy(), n
                    print(f"n={n} cdc2: {c:.6f}")

    # Final polish on best solution
    if time.time() - t0 < budget_s - 5:
        best_h, best_c5 = _polish(best_h, 4)
        print(f"polish: {best_c5:.6f}")

    h_ret = best_h.copy()
    tgt = best_n / 2.0
    d = tgt - np.sum(h_ret)
    if abs(d) > 1e-13:
        m = (h_ret > 1e-10) & (h_ret < 1.0 - 1e-10)
        cnt = int(np.sum(m))
        if cnt > 0:
            h_ret[m] += d / cnt
        h_ret = np.clip(h_ret, 0.0, 1.0)
    c5f = _c5(h_ret)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return h_ret, c5f, best_n
