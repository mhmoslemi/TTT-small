def project_simplex(v, target):
    # Project v onto {0 <= x <= 1, sum(x) = target}
    # Binary search on threshold
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

import numpy as np
import time

def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / len(h))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} via bisection on threshold."""
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_smooth(h, alpha):
    n = len(h)
    g = 1.0 - h
    corr = _corr(h)
    cs = corr - corr.max()
    w = np.exp(alpha * cs)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wl = w[d + n - 1]
        if wl < 1e-14:
            continue
        if d >= 0:
            grad[:n-d] += wl * g[d:]
            grad[d:] -= wl * h[:n-d]
        else:
            e = -d
            grad[e:] += wl * g[:n-e]
            grad[:n-e] -= wl * h[e:]
    return grad * dx

def _pgd(h0, target, max_iter=200, alpha=50.0):
    """Projected gradient descent with backtracking."""
    h = h0.copy()
    l_r = _c5(h)
    lr = 0.1
    for it in range(max_iter):
        grad = _grad_smooth(h, alpha)
        for _ in range(20):
            h_new = _project(h - lr * grad, target)
            l_new = _c5(h_new)
            if l_new < l_r - 1e-12:
                break
            lr *= 0.5
        if l_new < l_r:
            h = h_new
            l_r = l_new
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.5
        if l_new >= l_r:
            break
    return h, l_r

def _coord_descent(h, target, sweeps=20):
    """Coordinate descent: try all single-point adjustments."""
    n = len(h)
    l_r = _c5(h)
    for sw in range(sweeps):
        improved = False
        order = np.arange(n)
        for i in order:
            best_d = 0.0
            best_l = l_r
            for d in (-0.02, 0.02, -0.05, 0.05, -0.1, 0.1):
                h2 = h.copy()
                h2[i] += d
                h2 = _project(h2, target)
                if np.max(np.abs(h2 - h)) < 1e-8:
                    continue
                l2 = _c5(h2)
                if l2 < best_l:
                    best_l = l2
                    best_d = d
                    h_new = h2
            if best_d != 0:
                h = h_new
                l_r = best_l
                improved = True
        if not improved:
            break
    return h, l_r

def _pair_moves(h, target, sweeps=5):
    """Try pair moves: h[i]+d, h[j]-d."""
    n = len(h)
    l_r = _c5(h)
    for sw in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                for d in (0.01, -0.01, 0.03, -0.03, 0.05, -0.05):
                    h2 = h.copy()
                    h2[i] += d
                    h2[j] -= d
                    if h2[i] < 0 or h2[i] > 1 or h2[j] < 0 or h2[j] > 1:

                    if l2 < l_r - 1e-10:
                        h = h2
                        l_r = l2
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, l_r

    print(f"Final: n={best_n}, C5={best_c5:.6f}")
    return best_h, best_c5, best_n
