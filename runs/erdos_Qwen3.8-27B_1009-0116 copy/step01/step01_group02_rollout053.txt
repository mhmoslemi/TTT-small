import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_c(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

def lse_grad(h, n, beta):

    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    for si in range(len(corr)):
        wi = w[si]
        if wi < 1e-12:
            continue
        s = si - (n - 1)
        for j in range(n):
            d = 0.0
            if 0 <= j + s < n:
                d += 1.0 - h[j + s]
            if 0 <= j - s < n:
                d -= h[j - s]
            grad[j] += wi * d
    grad *= dx
    return grad

                d += 1.0 - h[j + s]
            if 0 <= j - s < n:
                d -= h[j - s]
            grad[j] += d
    grad *= dx / top_k
    return grad

        if not improved:

    best_c5 = compute_c5(h_best, n)
