from scipy.optimize import minimize
import time

_n_holder = [0]
_opt_params = {'n': 0, 'dx': 0, 't': 1}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-10:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def _cons_fun(x):
    return x.sum() - _n_holder[0] * 0.5

def _cons_jac(x):
    return np.ones(_n_holder[0])

def _obj(x):
    n, dx, t = _opt_params['n'], _opt_params['dx'], _opt_params['t']
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(c)
    return (cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t) * dx

def corr_grad(h, n, k):
    grad = np.zeros(n)
    lo = max(0, -k)
    hi = min(n, n - k)
    if hi <= lo:
        return grad
    i = np.arange(lo, hi)
    grad[i] += (1.0 - h[i + k])
    grad[i + k] += (-h[i])
    return grad

def subgrad_step_multi(h, n, dx, lr, n_active=3):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    active = np.where(c > cmax - 0.01 * cmax)[0]
    active = active[:n_active]
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        grad += corr_grad(h, n, k)
    grad /= len(active)
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair(h_new, n)
    return h_new, float(np.max(c) * dx)

def binary_flip_search(h, n, n_iter, rng):
    h = h.copy()
    best = compute_c5(h)
    target = n * 0.5
    dx = 2.0 / n
    for it in range(n_iter):
        i = rng.randint(n)
        old = h[i]
        cand = 1.0 if old < 0.5 else 0.0
        if abs(cand - old) < 1e-6:
            h[i] = rng.uniform(0, 1)
            cand = h[i]
        h[i] = cand
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            val = compute_c5(h)
            if val >= best - 1e-12:
                h[i] = old
            else:
                best = val
            continue
        mask = (h < 1.0 - 1e-9) if diff > 0 else (h > 1e-9)
        candidates = np.where(mask)[0]
        candidates = candidates[candidates != i]
        if len(candidates) == 0:
            h[i] = old
            continue
        best_j = -1
        best_val = best
        for trial in range(min(5, len(candidates))):
            j = candidates[rng.randint(len(candidates))]
            old_j = h[j]
            h[j] = np.clip(old_j + diff, 0.0, 1.0)
            val = compute_c5(h)
            if val < best_val:

                best_j = j
            h[j] = old_j
        if best_j >= 0:
            h[best_j] = np.clip(h[best_j] + diff, 0.0, 1.0)
            best = best_val
        else:
            h[i] = old
    return h, best

def make_kronecker(n, alpha):
    x = np.arange(n) * alpha
    frac = x - np.floor(x)
    return (frac < 0.5).astype(float)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return h

    bnds = [(0.0, 1.0)] * n

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    return repair(res.x, n)
