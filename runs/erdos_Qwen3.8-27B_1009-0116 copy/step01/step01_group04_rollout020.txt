            if time_left(t0, budget_s, 0.98) < 3:
                break
            xp = optimize_slq(best_h, best_n, dx_b, tau, mi=500)
            v = compute_c5(xp, dx_b)
            if v < best_val:
                best_val = v
                best_h = xp.copy()
        if time_left(t0, budget_s, 0.98) > 5:
            xp = optimize_trf(best_h, best_n, dx_b, 5e-05, mi=200)
            v = compute_c5(xp, dx_b)
            if v < best_val:
                best_val = v
                best_h = xp.copy()

    dx_f = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)
    return (best_h, final_val, best_n)
