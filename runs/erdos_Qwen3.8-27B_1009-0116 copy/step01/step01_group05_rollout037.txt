_N = 70
_EPS = 0.01
_TGT = 35.0

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return float(_corr(h, dx).max())

def _sobj(x):
    cv = _corr(x, _DX)
    mx = cv.max()
    return mx + _EPS * np.log(np.sum(np.exp((cv - mx) / _EPS)))

def _sgrad(x):
    cv = _corr(x, _DX)
    mx = cv.max()
    w = np.exp((cv - mx) / _EPS)
    w /= w.sum()
    n = _N
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * _DX
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    return x.sum() - _TGT

def _ejac(x):
    return np.ones(len(x))

    _N = n
    _EPS = float(eps)
    _TGT = float(target)
    res = minimize(
        _sobj, h0, jac=_sgrad,
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
        method='SLSQP',
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )

        base = corr - old * slope
        matrix = base[None, :] + np.outer(vs, slope)
        obj = matrix.max(axis=1)
        best_idx = obj.argmin()
        h[j] = vs[best_idx]
        corr = base + h[j] * slope
    return _proj(h, target)

def _subgrad_step(h, dx, target, lr):
    cv = _corr(h, dx)
    mx = cv.max()
    active = np.where(cv >= mx - 1e-10)[0]
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    g /= max(len(active), 1)
    return _proj(h - lr * g, target)

def _optimize(h0, dx, n, target, rng, t_end):
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)
    h = h0.copy()

    for eps in [0.1, 0.05, 0.01, 0.005, 0.001, 0.0005, 0.0002]:
        if time.time() > t_end - 15:
            break
        try:
            h_new = _slsqp(h, dx, n, target, eps, maxiter=300)
            c = _c5(h_new, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass

    for _ in range(300):
        if time.time() > t_end - 10:
            break
        h_new = _coord_pass(h, dx, n, target)
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new

    if time.time() < t_end - 8:
        for eps in [0.0005, 0.0002, 0.0001]:
            if time.time() > t_end - 4:
                break
            try:
                h_new = _slsqp(best_h, dx, n, target, eps, maxiter=400)
                c = _c5(h_new, dx)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_new.copy()
            except Exception:
                pass

    h = best_h.copy()
    for _ in range(500):
        if time.time() > t_end - 2:
            break
        h_new = _coord_pass(h, dx, n, target, ngrid=100)
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.RandomState(seed)

    ns_to_try = [70, 60, 80]
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_init = len(initial_h_values)
        if n_init not in ns_to_try:
            ns_to_try.insert(0, n_init)
        else:
            ns_to_try.remove(n_init)
            ns_to_try.insert(0, n_init)

    best_h = None
    best_c5 = float('inf')
    best_n = 70

    for ni, n in enumerate(ns_to_try):
        if time.time() > deadline - 40:
            break
        dx = 2.0 / n
        target = n / 2.0

        if 'initial_h_values' in globals() and initial_h_values is not None and n == len(initial_h_values):
            h0 = np.array(initial_h_values, dtype=float).copy()
        elif 'initial_h_values' in globals() and initial_h_values is not None:
            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(initial_h_values)), initial_h_values)
        else:
            h0 = rng.uniform(0.2, 0.8, n)
        h0 = _proj(h0, target)

        frac = 0.7 if ni == 0 else 0.5
        time_left = deadline - time.time()
        t_end = t0 + (deadline - t0) * (1.0 - frac * (len(ns_to_try) - ni - 1) / max(len(ns_to_try) - ni, 1))
        t_end = max(t0 + 30, min(t_end, deadline))

        h_opt, c5_opt = _optimize(h0, dx, n, target, rng, t_end)

        n_restarts = 4 if c5_opt < 0.39 else 6
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            amp = rng.uniform(0.02, 0.12) * (1.0 + 0.3 * r)
            h_pert = _proj(h_opt + rng.normal(0, amp, n), target)
            c_pert = _c5(h_pert, dx)
            if c_pert < c5_opt:
                h_opt = h_pert.copy()
                c5_opt = c_pert
            t_end2 = min(time.time() + min(50, deadline - time.time() - 10), deadline)
            if t_end2 - time.time() > 5:
                h2, c2 = _optimize(h_pert, dx, n, target, rng, t_end2)
                if c2 < c5_opt:
                    h_opt, c5_opt = h2, c2

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n

    return best_h, best_c5, best_n
