def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12 and abs(s - tgt) > 1e-10:
        # Try to adjust
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)

        cnt = np.sum(mask)
        if cnt > 0:
            xp[mask] += diff / cnt

    c = np.correlate(h, 1.0 - h, mode="full")
    cm = np.max(c)

    t2 = np.correlate(p[::-1], h, mode="full")

    return obj, grad

def pg_descent(h0, n, dx, tau, n_iter, t_end, alpha0=1.0):

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = alpha0
    for it in range(n_iter):
        if time.time() > t_end:
            break

        accepted = False
        for b in range(30):

            h_new = project_feasible(h - a * grad, n)
            obj_new, _ = smooth_obj_grad(h_new, tau, dx)

        c5v = compute_c5(h, dx)

            best_h = h.copy()

def slsqp_polish(h0, n, dx, tau, t_end, maxiter=200):
    def objective(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        cm = np.max(c)
        sh = np.clip((c - cm) / tau, -50.0, 0.0)
        return (cm + tau * np.log(np.sum(np.exp(sh)))) * dx
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        cm = np.max(c)
        sh = np.clip((c - cm) / tau, -50.0, 0.0)
        exp_sh = np.exp(sh)
        S = np.sum(exp_sh)
        p = exp_sh / S
        obj = (cm + tau * np.log(S)) * dx
        t1 = np.correlate(p, 1.0 - x, mode="full")
        t2 = np.correlate(p[::-1], x, mode="full")
        grad = (dx / tau) * (t1[:n] - t2[:n])
        return obj, grad
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    def constraint_jac(x):
        return np.ones(n)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        xp = np.clip(x - mid, 0.0, 1.0)
        if np.sum(xp) > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    xp = np.clip(x - mid, 0.0, 1.0)
    diff = target - np.sum(xp)
    if abs(diff) > 1e-10:
        mask = (xp > 1e-10) & (xp < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            xp[mask] += diff / cnt
    return np.clip(xp, 0.0, 1.0)

def smooth_obj(h, tau, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = np.max(c)
    sh = np.clip((c - cm) / tau, -50.0, 0.0)

def smooth_obj_grad(h, tau, dx):
    n = len(h)

    cm = np.max(c)
    sh = np.clip((c - cm) / tau, -50.0, 0.0)

    t2 = np.correlate(p[::-1], h, mode="full")
    grad = dx * (t1[:n] - t2[:n])
    return obj, grad

def pg_descent(h0, n, dx, tau, n_iter, t_end, alpha0=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    if alpha0 is None:
        alpha0 = 0.5 / dx
    alpha = alpha0
    stuck = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        obj, grad = smooth_obj_grad(h, tau, dx)
        accepted = False
        for b in range(25):
            a = alpha * (0.5 ** b)
            h_new = project_feasible(h - a * grad, n)
            obj_new = smooth_obj(h_new, tau, dx)
            if obj_new <= obj - 1e-14:
                h = h_new
                if b <= 2:
                    alpha = min(alpha * 1.05, 10.0 / dx)
                accepted = True
                break
        if not accepted:
            alpha *= 0.5
            if alpha < 1e-12:
                break
            h = project_feasible(h - alpha * grad, n)
        c5v = compute_c5(h, dx)
        if c5v < best_c5 - 1e-12:
            best_c5 = c5v
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5

        c = np.correlate(x, 1.0 - x, mode="full")
