            # contribution from -h[j+lag]
            if 0 <= j + lag < n:

def project_feasible(h, ts):
    """Project h to {0<=h<=1, sum(h)=ts}"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Water-filling
    if diff > 0:
        # Need to add mass
        free = h < 1.0 - 1e-12
        if free.sum() == 0:
            return h
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    else:
        # Need to remove mass
        free = h > 1e-12
        if free.sum() == 0:
            return h
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    # Second pass for precision
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            free = h < 1.0 - 1e-14
        else:
            free = h > 1e-14
        if free.sum() > 0:
            idx = np.where(free)[0]
            # Add to one element
            h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

    corr = np.correlate(h, 1.0 - h, mode="full")
    v = beta * corr
    v -= v.max()

    ps = e / e.sum()

def pgd_optimize(h_init, n, ts, t_end):
    h = project_to_box_sum(h_init.copy(), ts)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in [5.0, 20.0, 80.0, 300.0, 1200.0]:
        t0 = time.time()
        step = 0.8
        prev_h = h.copy()
        for it in range(20000):

            vel = h - prev_h
            h_new = h + 0.4 * vel - step * g
            h_new = project_to_box_sum(h_new, ts)
            prev_h = h
            h = h_new
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if it % 20 == 0:
                f_check, _ = smooth_max_grad(h, beta)
                if f_check > f + 0.005:
                    step *= 0.6
                else:
                    step = min(step * 1.08, 2.5)
    return best_h, best_c5

def subgrad_polish(h, n, ts, max_iters=8000, t_end=None):
    best = h.copy()
    best_c5 = compute_c5(h)
    step = 0.3
    h_curr = h.copy()
    for it in range(max_iters):
        if t_end is not None and time.time() > t_end:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:n] += (1.0 - h_curr[:n - lag])
            grad[0:n - lag] -= h_curr[lag:n]
        else:
            ll = -lag
            grad[0:n - ll] += (1.0 - h_curr[ll:n])
            grad[ll:n] -= h_curr[:n - ll]
        gn = np.linalg.norm(grad)
        if gn < 1e-10:
            break
        h_new = h_curr - step * grad / gn
        h_new = project_to_box_sum(h_new, ts)
        c5_new = compute_c5(h_new)

def push_binary(h, ts):
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    s = h_bin.sum()
    ts_int = int(round(ts))
    if s > ts_int:
        excess = int(s - ts_int)
        fracs = np.where(h_bin == 1.0, 1.0 - h, 0.0)
        removes = np.argsort(fracs)[:excess]
        h_bin[removes] = 0.0
    elif s < ts_int:

        fracs = np.where(h_bin == 0.0, h, 0.0)
        adds = np.argsort(-fracs)[:deficit]
        h_bin[adds] = 1.0
    return h_bin

        best_h = h_sg.copy()

    # Phase 3: Push to binary and pair-search
    h_bin, c5_bin = binary_pair_search(h_sg, n, ts, t_end, max_sweeps=200)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 4: PGD polish on the binary solution (relax then re-optimize)
    h_relax = best_h + 0.05 * rng.normal(0, 1, n)
    h_relax = project_to_box_sum(h_relax, ts)
    h_pg2, c5_pg2 = pgd_optimize(h_relax, n, ts, t_end)
    if c5_pg2 < best_c5:
        best_c5 = c5_pg2
        best_h = h_pg2.copy()

    # Phase 5: Final binary pair search
    h_bin2, c5_bin2 = binary_pair_search(h_pg2, n, ts, t_end, max_sweeps=400)
    if c5_bin2 < best_c5:
        best_c5 = c5_bin2
        best_h = h_bin2.copy()

    # Phase 6: Try a random perturbation + re-optimize
    if time.time() < t_end - 30:
        h_r = best_h + 0.15 * rng.normal(0, 1, n)
        h_r = project_to_box_sum(h_r, ts)
        h_pr, c5_pr = pgd_optimize(h_r, n, ts, t_end)
        h_br, c5_br = binary_pair_search(h_pr, n, ts, t_end, max_sweeps=200)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()

    # Phase 7: Try different random binary start
    if time.time() < t_end - 40:
        h_rand = np.zeros(n)
        h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        h_pr2, c5_pr2 = pgd_optimize(h_rand, n, ts, t_end)
        h_br2, c5_br2 = binary_pair_search(h_pr2, n, ts, t_end, max_sweeps=200)
        if c5_br2 < best_c5:
            best_c5 = c5_br2
            best_h = h_br2.copy()

    # Final binary refinement
    if time.time() < t_end - 5:
        h_fin, c5_fin = binary_pair_search(best_h, n, ts, t_end - 2, max_sweeps=100)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_to_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
