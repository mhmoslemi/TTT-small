import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:

def compute_subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = 1e-9

        else:

            grad[L:] -= h[:n-L]

        return np.ones((1, n))

        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)

from scipy.optimize import minimize

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def compute_subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = 1e-9
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k_idx in active:
        L = abs(int(k_idx) - (n - 1))
        if L == 0:
            grad += (1.0 - 2.0 * h)
        else:
            grad[:n - L] += (1.0 - h[L:])
            grad[L:] -= h[:n - L]
    grad /= float(len(active))
    return dx * grad

def slsqp_opt(h, n, maxiter=500, ftol=1e-12):

        return eval_c5(x, dx)

        return np.ones((1, n))
