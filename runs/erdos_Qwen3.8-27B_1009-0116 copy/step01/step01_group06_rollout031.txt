def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0

        if abs(diff) < 1e-14:

            if s < 1e-14:

            if s < 1e-14:

    """Approximate max with log-sum-exp for smoother gradients."""
    shifted = corr / temperature
    shifted = shifted - np.max(shifted)
    return temperature * (np.max(shifted) + np.log(np.sum(np.exp(shifted))))

def softmax_grad(corr, temperature=0.01):
    """Gradient of the soft-max."""
    shifted = corr / temperature
    shifted = shifted - np.max(shifted)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    return weights / temperature  # This gives softmax probabilities

def binary_local_search(h, n, dx, max_iters=1000000, t0=None, budget_frac=0.5):
    """Local search on binary vectors by swapping 1s and 0s."""
    h = (h > 0.5).astype(float)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_corr = np.max(np.correlate(h, 1-h, mode='full'))
    best_h = h.copy()
    
    for it in range(max_iters):
        # Check time
        if t0 is not None and time.time() - t0 > budget_frac * 1100:
            break
        
        # Try random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        
        h[i] = 0
        h[j] = 1
        
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = np.max(corr)
        
        if max_corr <= best_corr:
            best_corr = max_corr
            best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    return best_h, best_corr * dx

- The subgradient of max_k corr[k] * dx with respect to h[m]:

For the correlation d(corr[k])/dh[m]:

- If k ≤ m ≤ n-1: contributes -h[m-k]

So for all lags:
d(soft)/dh[m] = dx * (1/|A|) * sum_{k in A} [(1-h[m+|k|]) * I(m+|k| < n) - h[m-|k|] * I(m-|k| ≥ 0)]

where I is the indicator.

This can be computed in O(|A| * n) time, which for small |A| (usually 1-3 active lags) is fast.

def subgrad_max(h, dx, n, tol=1e-6):
    """Compute subgradient of max_k corr[k] * dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < tol)[0]
    n_active = len(active)
    grad = np.zeros(n)
    for j in active:
        lag = int(j) - (n - 1)
        if lag >= 0:
            # corr = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            for i in range(n - lag):
                grad[i] += (1.0 - h[i + lag])
                grad[i + lag] -= h[i]
        else:
            lag_neg = -lag
            for i in range(lag_neg, n):
                grad[i] += (1.0 - h[i - lag_neg])
                grad[i - lag_neg] -= h[i]
    return grad / n_active * dx, max_val * dx

def grad_descent_max(h, n, dx, maxiter=50000, lr0=0.01):

    best_c5 = eval_c5(h, dx)
    best_h = h.copy()
    lr = lr0
    n_corr = 2 * n - 1
    for it in range(maxiter):

        max_val = np.max(corr)
        tol = max(1e-6, 1e-4 * max_val)

                    grad[i + lag] -= h[i]

                ln = -lag
                for i in range(ln, n):
                    grad[i] += (1.0 - h[i - ln])
                    grad[i - ln] -= h[i]
        grad /= n_act
        # Step
        h_new = h - lr * grad * dx
        h_new = np.clip(h_new, 0.0, 1.0)

        h[excess] = 0
    elif len(ones_idx) < target_ones:
        zeros_idx = np.where(h == 0)[0]
        deficit = np.random.choice(zeros_idx, target_ones - len(ones_idx), replace=False)
        h[deficit] = 1
    
    best_h = h.copy()
    best_corr = corr_max(h)
    
    for round_i in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        improved = False
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        # Randomize swap order
        order_i = np.random.permutation(len(ones_list))
        order_j = np.random.permutation(len(zeros_list))
        for ii in order_i:
            if deadline and time.time() > deadline:
                break
            for jj in order_j:
                i, j = ones_list[ii], zeros_list[jj]
                if h[i] == 0 or h[j] == 1:
                    continue
                h[i], h[j] = 0.0, 1.0
                nc = corr_max(h)
                if nc < best_corr:
                    best_corr = nc
                    best_h = h.copy()
                    improved = True
                    # Restart inner loops with new sets
                    ones_list = np.where(h == 1)[0]
                    zeros_list = np.where(h == 0)[0]
                    order_i = np.random.permutation(len(ones_list))
                    order_j = np.random.permutation(len(zeros_list))
                    ii = -1
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return best_h, best_corr * dx

def grad_descent(h, n, dx, maxiter=20000, lr0=0.01, deadline=None):

        return ho, eval_c5(ho, dx)
    except Exception:
