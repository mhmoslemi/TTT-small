import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Spread adjustment across all elements
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # Fix residual
        residual = target - h.sum()
        if abs(residual) > 1e-14:
            if residual > 0:
                idx = np.where(h < 1.0)[0]
            else:
                idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += residual
                h[idx[0]] = np.clip(h[idx[0]], 0, 1)
    return h

def coordinate_descent_fine(h, n, n_sweeps, n_scan, time_limit):
    """Coordinate descent with fine 1D scan, adjusting pair (i, j)"""
    t_start = time.time()
    target = n * 0.5
    best_val = compute_c5(h)
    
    for sweep in range(n_sweeps):
        if time.time() - t_start > time_limit:
            break
        for i in range(n):
            hi = h[i]
            j = (i + 1) % n
            hj = h[j]
            fij = hi + hj
            
            lo = max(0.0, fij - 1.0)
            hi_bound = min(1.0, fij)
            if lo >= hi_bound:
                continue
            
            scan = np.linspace(lo, hi_bound, n_scan)
            cur_val = best_val
            best_ci = hi
            best_cj = hj
            
            for ci in scan:
                cj = fij - ci
                if cj < -1e-12 or cj > 1 + 1e-12:
                    continue
                h[i] = ci
                h[j] = cj
                val = compute_c5(h)
                if val < cur_val:
                    cur_val = val
                    best_ci = ci
                    best_cj = cj
            
            h[i] = best_ci
            h[j] = best_cj
            best_val = min(best_val, cur_val)
    
    return h, best_val

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        residual = target - h.sum()
        if abs(residual) > 1e-13:
            mask = (h < 1.0) if residual > 0 else (h > 0.0)
            if mask.any():
                h[mask] += residual / mask.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def cd_scan(h, n, n_sweeps, n_pts, time_limit):

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(c)
    shifted = (c - cmax) * t
    shifted -= np.max(shifted)

    w_sum = np.sum(w)
    f = (cmax + np.log(w_sum) / t) * dx
    if np.all(w == 0):
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum
    # Gradient: df/dh[i] = dx * sum_m w_m * dc_m/dh[i]
    # dc_m/dh[i] = (1-x[i-m+n-1])*[0<=i-m+n-1<n] - x[i+m-n+1]*[0<=i+m-n+1<n and i+m-n+1!=i]
    grad = np.zeros(n)
    nm1 = n - 1
    for m in range(len(c)):
        if w[m] < 1e-12:
            continue
        # Term 1: (1 - x[i - m + n - 1]) for valid index
        j1 = np.arange(n) - m + nm1
        mask1 = (j1 >= 0) & (j1 < n)
        grad[mask1] += (1.0 - x[j1[mask1]]) * w[m]
        # Term 2: -x[i + m - n + 1] for valid index, i+m-n+1 != i i.e. m != n-1

            mask2 = (j2 >= 0) & (j2 < n)
