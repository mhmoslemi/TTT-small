    h = h.copy()
    target = n * 0.5
    best = compute_c5(h)
    for sweep in range(500):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            oj = h[j]
            if oj < 1e-6 or oj > 1-1e-6:
                continue
            cands = [0.0, 1.0, 0.5]
            bc, bv = oj, best
            for v in cands:
                if abs(v - oj) < 1e-6:
                    continue
                h[j] = v
                d = target - h.sum()
                if abs(d) > 1e-14:
                    # Find the index with most room
                    if d > 0:
                        room = 1.0 - h
                    else:
                        room = h.copy()
                    # Don't use j itself
                    best_r = np.argmax(room)
                    h[best_r] = min(max(h[best_r] + d, 0), 1)
                    # If still off, spread
                    d2 = target - h.sum()
                    if abs(d2) > 1e-13:
                        if d2 > 0:
                            m = h < 1.0
                        else:
                            m = h > 0.0
                        h[m] += d2 / m.sum()
                        h = np.clip(h, 0, 1)
                val = compute_c5(h)
                if val < bv:
                    bc, bv = v, val
            if bv < best - 1e-13:
                # Re-apply best
                h[j] = bc
                d = target - h.sum()
                if abs(d) > 1e-14:
                    # Find best reservoir
                    if d > 0:
                        room = 1.0 - h
                    else:
                        room = h.copy()
                    best_r = int(np.argmax(room))
                    h[best_r] = min(max(h[best_r] + d, 0), 1)
                    d2 = target - h.sum()
                    if abs(d2) > 1e-13:
                        m = (h < 1.0) if d2 > 0 else (h > 0.0)
                        h[m] += d2 / m.sum()
                        h = np.clip(h, 0, 1)
                best = bv
                improved = True
            else:
                h[j] = oj
