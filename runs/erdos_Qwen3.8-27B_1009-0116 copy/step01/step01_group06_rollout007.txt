def proj_feasible(h, n):

    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

        return np.ones(n)

def smooth_obj_and_grad(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Log-sum-exp for numerical stability
    mc = np.max(corr)
    exp_corr = np.exp(tau * (corr - mc))
    log_sum = mc + np.log(exp_corr.sum())
    obj = dx * log_sum / tau
    
    # Softmax weights
    w = exp_corr / exp_corr.sum()
    
    # Gradient
    grad = np.zeros(len(h))
    n = len(h)
    for k_idx in range(2*n-1):
        if w[k_idx] < 1e-15:
            continue
        lag = k_idx - (n - 1)
        # corr[k_idx] = sum_i h[i+lag] * (1-h[i]) for valid i
        for m in range(n):
            # d/dh[m] of h[i+lag]*(1-h[i]) where i+lag=m => i=m-lag
            # term 1: h[m] appears as h[i+lag] when i = m-lag
            if 0 <= m - lag < n:
                i = m - lag
                grad[m] += w[k_idx] * (1 - h[i])
            # term 2: h[m] appears as h[i] when i = m
            grad[m] += w[k_idx] * (-h[m + lag]) * (1 if 0 <= m + lag < n else 0)
    
    grad *= dx
    return obj, grad

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smooth_obj_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = tau * (corr - mc)
    exp_c = np.exp(shifted)
    sum_exp = exp_c.sum()
    obj = dx * (mc + np.log(sum_exp)) / tau
    w = exp_c / sum_exp
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # (1-h[m+lag]) for m in [0, n-1-lag]
            grad[:n-lag] += wk * omh[lag:n]
            # -h[m-lag] for m in [lag, n)
            grad[lag:] -= wk * h[:n-lag]
        else:
            a = -lag
            # (1-h[m-a]) for m in [a, n)
            grad[a:] += wk * omh[:n-a]
            # -h[m+a] for m in [0, n-1-a]
            grad[:n-a] -= wk * h[a:n]
    grad *= dx
    return obj, grad

def lbfgs_smooth(h, n, dx, tau, maxiter, ftol):
    def obj(x):
        o, _ = smooth_obj_grad(x, dx, tau)
        return o
    def grad(x):
        _, g = smooth_obj_grad(x, dx, tau)
        return g
    # Add penalty for constraint
    def obj_pen(x):
        o, g = smooth_obj_grad(x, dx, tau)
        penalty = 100.0 * (x.sum() - n/2.0)**2
        pen_grad = 200.0 * (x.sum() - n/2.0)
        return o + penalty, g + pen_grad * np.ones(n)
    
    # Use the penalty approach
    bounds = [(0.0, 1.0)] * n
    try:
        # Initial: project to satisfy constraint
        h0 = proj_feasible(h, n)
        res = minimize(lambda x: smooth_obj_grad(x, dx, tau)[0], h0,
                       jac=lambda x: smooth_obj_grad(x, dx, tau)[1],
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        ho = proj_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def slsqp_smooth(h, n, dx, tau, maxiter):
    def obj(x):
        return smooth_obj_grad(x, dx, tau)[0]
    def grad(x):
        return smooth_obj_grad(x, dx, tau)[1]
    def ceq(x):
        return x.sum() - n/2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        ho = proj_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def slsqp_max(h, n, dx, maxiter):
    def obj(x):
        return eval_c5(x, dx)
    def ceq(x):
        return x.sum() - n/2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        ho = proj_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n = 70
    dx = 2.0 / n
    best_h = proj_feasible(np.array(initial_h_values, dtype=float), n)
    best_c5 = eval_c5(best_h, dx)
    
    # Phase 1: Progressive smooth max with SLSQP
    for tau in [30, 60, 100, 200, 400, 800, 1600]:
        if time.time() - t0 > budget_s - 200:
            break
        h, c = slsqp_smooth(best_h, n, dx, tau, maxiter=300)
        if c < best_c5:
            best_h, best_c5 = h, c
    
    # Phase 2: L-BFGS-B with high tau
    for tau in [400, 800, 1600, 3200]:
        if time.time() - t0 > budget_s - 150:
            break
        h, c = lbfgs_smooth(best_h, n, dx, tau, maxiter=500, ftol=1e-16)
        if c < best_c5:
            best_h, best_c5 = h, c
    
    # Phase 3: SLSQP with actual max for polishing
    if time.time() - t0 < budget_s - 80:
        h, c = slsqp_max(best_h, n, dx, maxiter=1000)
        if c < best_c5:
            best_h, best_c5 = h, c
    
    # Phase 4: Random restarts
    attempt = 0
    while time.time() - t0 < budget_s - 5:
        attempt += 1
        if attempt <= 20:
            scale = 0.15 * max(1.0, 1.0 / (1 + attempt * 0.05))
        elif attempt <= 50:
            scale = 0.05
        else:
            scale = 0.02
        
        if attempt % 7 == 0:
            h0 = np.random.rand(n)
        elif attempt % 3 == 0:
            h0 = np.where(np.random.rand(n) < 0.5, 0.0, 1.0)
        else:
            h0 = best_h + scale * np.random.randn(n)
        h0 = proj_feasible(h0, n)
        
        # Quick smooth optimization
        h, c = slsqp_smooth(h0, n, dx, tau=200, maxiter=150)
        if c < best_c5:
            best_h, best_c5 = h, c
            # Polish
            h2, c2 = slsqp_max(best_h, n, dx, maxiter=300)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2
    
    # Final
    best_h = proj_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
