import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _project(h, n):
    # Project onto {0<=h<=1, sum(h)=n/2}
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Bisection on lambda for sum constraint
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            h = np.maximum(h - 0.001, 0)
        else:
            h = np.minimum(h + 0.001, 1)
    # Final adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _subgrad_step(h, dx, step, m=5):
    corr = _corr(h)
    # Top m lags
    top = np.argsort(corr)[-m:]
    # Average subgradient
    grad = np.zeros_like(h)
    n = len(h)
    for k in top:
        # k in [0, 2n-2], lag = k - (n-1)
        lag = k - (n - 1)
        # For each j, contribution to grad from lag:
        # ∂c_lag/∂h[j] = (1-h[j+lag]) - h[j-lag] (valid indices only)
        for j in range(n):
            g = 0.0
            if 0 <= j + lag < n:
                g += (1 - h[j + lag])
            if 0 <= j - lag < n:
                g -= h[j - lag]
            grad[j] += g
    grad /= m
    h_new = h - step * grad
    return _project(h_new, n)

    if lag >= 0:

        g[:n-lag] += (1 - h[lag:])

        # (1 - h[j+lag]) = (1 - h[j-lag_abs]) for j where j-lag_abs >= 0

def _topm_grad(h, corr, m):

    n = len(h)

    g = np.zeros(n)
    for k in top:

        if lag == 0:
            g += 1.0 - 2.0 * h
        elif lag > 0:
            g[:n-lag] += 1.0 - h[lag:]
            g[lag:] -= h[:n-lag]
        else:
            la = -lag
            g[la:] += 1.0 - h[:n-la]
            g[:n-la] -= h[la:]
    return g / m

def _sgd_phase(h, target, dx, rng, t_end, step0=0.015, m=5, beta=0.8):
    """Subgradient descent with momentum."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_val(h, dx)
    vel = np.zeros(n)
    step = step0
    for it in range(20000):
        if time.time() > t_end:
            break
        corr = _corr(h)
        g = _topm_grad(h, corr, m)
        vel = beta * vel + (1.0 - beta) * g
        h = _project(h - step * vel, target)
        c5 = _c5_val(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step *= 0.9997
    return best_h, best_c5

def _coord_search(h, target, dx, rng, t_end, n_passes=50):
    """Coordinate-pair local search."""
    n = len(h)
    h = h.copy()
    best_c5 = _c5_val(h, dx)
    deltas = np.array([0.002, 0.005, 0.01, 0.02, 0.05,
                        -0.002, -0.005, -0.01, -0.02, -0.05])
    for p in range(n_passes):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            js = rng.choice(n, size=20, replace=False)
            for j in js:
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < -1e-12 or ni > 1+1e-12 or nj < -1e-12 or nj > 1+1e-12:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5 = _c5_val(h, dx)
                    if c5 < best_c5 - 1e-10:
                        best_c5 = c5
                        improved = True
                    else:
                        h[i] = hi
                        h[j] = hj
        if not improved:
            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr(h))) * dx

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a; diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a; diff += a
        if abs(diff) < 1e-14:
            break
    return h

def _grad_at(h, pos):
    n = len(h)
    k = np.arange(2 * n - 1)
    lag = k - (n - 1)
    g = np.zeros(2 * n - 1)
    idx1 = pos + lag
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += 1.0 - h[idx1[m1]]
    idx2 = pos - lag
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _exhaustive_pair(h, target, dx, t_end):
    n = len(h)
    corr = _corr(h)
    best_c5 = float(np.max(corr)) * dx
    grad_cache = [_grad_at(h, i) for i in range(n)]
    hi_arr = h  # shorthand
    
    deltas = np.array([0.001, 0.002, 0.005, 0.01, 0.02, 0.05])
    all_deltas = np.concatenate([deltas, -deltas])
    
    sweeps = 0
    while time.time() < t_end:
        improved = False
        for i in range(n):

                g = grad_cache[i] - grad_cache[j]
                hi, hj = h[i], h[j]
                best_d = 0.0
                best_nm = best_c5
                for d in all_deltas:
                    ni, nj = hi + d, hj - d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    nm = float(np.max(corr + d * g)) * dx
                    if nm < best_nm - 1e-12:
