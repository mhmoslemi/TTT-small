from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        s = np.sum(x)
        if abs(s - tgt) < 1e-13:
            break
        mask = (x > 1e-9) & (x < 1.0 - 1e-9)
        if mask.sum() == 0:
            x = x * (tgt / max(s, 1e-12))
            x = np.clip(x, 0, 1)
            continue
        x[mask] += (tgt - s) / mask.sum()
        x = np.clip(x, 0, 1)
    return x

def project_hypersimplex(v, n):
    target = n / 2.0
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

def grad_lag(h, lag):
    n = len(h)
    l = abs(lag)
    if l == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - l] += 1.0
        grad[:n - l] -= h[l:]
        grad[l:] -= h[:n - l]
    else:
        grad[l:] += 1.0
        grad[l:] -= h[:n - l]
        grad[:n - l] -= h[l:]
    return grad

def adam_update(h, grad, m, v, t, lr, n, b1=0.9, b2=0.999):
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * grad * grad
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    h_new = h - lr * mh / (np.sqrt(vh) + 1e-8)
    h_new = project_hypersimplex(h_new, n)
    return h_new, m, v

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, x_src, kind='linear', fill_value='extrapolate')(xd)
    return repair(xp, n_dst)

def slsqp_obj(x, corr_cache, n, tau, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    return (m + tau * np.log(np.sum(np.exp(sh)))) * dx

def slsqp_constraint_fun(x, n):
    return np.sum(x) - n / 2.0

def slsqp_polish(x0, n, time_limit, maxiter=300):
    dx = 2.0 / n
    t0 = time.time()
    taus = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    best = x0.copy()
    best_val = compute_c5(x0, dx)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': lambda x: slsqp_constraint_fun(x, n), 'jac': lambda x: np.ones(n)}]
    for tau in taus:
        if time.time() - t0 > time_limit - 0.5:
            break
        global _slsqp_tau, _slsqp_dx
        _slsqp_tau = tau
        _slsqp_dx = dx
        try:
            r = minimize(_make_slsqp_obj(x, n, tau, dx), best, method='SLSQP',
                         bounds=bd, constraints=cs,
                         options={'maxiter': maxiter, 'ftol': 1e-12})
            if r.success or r.fun < best_val:
                cand = repair(r.x, n)
                cv = compute_c5(cand, dx)
                if cv < best_val:
                    best_val = cv
                    best = cand
        except Exception:
            pass
    return best, best_val

_slsqp_tau = 0.01
_slsqp_dx = 0.01

def _make_slsqp_obj(x, n, tau, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    return (m + tau * np.log(np.sum(np.exp(sh)))) * dx

def adam_loop(h0, n, time_limit, k_top=5, lr0=3.0, restart_every=2000, restart_scale=0.05):
    t0 = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    patience = 0
    best_patience_val = best_val
    
    while time.time() - t0 < time_limit - 2.0:
        # Periodic restart
        if t_count > 0 and t_count % restart_every == 0:
            h = repair(h + restart_scale * np.random.default_rng(t_count).standard_normal(n), n)
            m = np.zeros(n)
            v = np.zeros(n)
            patience = 0
            best_patience_val = best_val
            continue
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-k_top:]
        active = corr[top_idx].copy()
        base = active.max()
        spread = base - active.min() + 1e-15
        weights = np.exp((active - base) / spread)
        weights /= weights.sum()
        
        grad = np.zeros(n)

            lag = kk - (n - 1)
            grad += weights[i] * grad_lag(h, lag)
        
        lr = lr0 / (np.sqrt(t_count + 1) + 1.0)
