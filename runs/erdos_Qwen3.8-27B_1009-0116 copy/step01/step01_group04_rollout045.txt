    """Project x onto {0 <= y <= 1, sum(y) = S}"""
    def f(t):
        return np.sum(np.clip(x - t, 0, 1)) - S
    # Bisection
    lo, hi = -1, 2
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(x - t, 0, 1)

    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12 and abs(s - tgt) > 1e-10:

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12 and abs(s - tgt) > 1e-10:
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)
    s2 = np.sum(x)
    if abs(s2 - tgt) > 1e-10:
        diff = tgt - s2
        mask = (x > 1e-9) & (x < 1.0 - 1e-9)
        if mask.any():
            adj = diff / np.sum(mask)
            x[mask] = np.clip(x[mask] + adj, 0.0, 1.0)
    return x

def smooth_max_val(x, dx, tau):
    c5 = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(c5)
    if tau < 1e-10:
        return m
    sh = np.clip((c5 - m) / tau, -50.0, 0.0)
    return m + tau * np.log(np.sum(np.exp(sh)))

def slsqp_opt(x0, n, dx, tau, maxiter=800, ftol=1e-15):
    def obj(x):
        return smooth_max_val(x, dx, tau)
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        r = minimize(obj, x0, method='SLSQP', bounds=bd, constraints=cs,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return repair(r.x, n)

        return x0.copy()

def lcos_step(x, n, dx, tau, steps=300, lr=0.003):
    """Projected subgradient with L-cosine step schedule"""
    x = x.copy()
    for s in range(steps):
        c5 = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(c5)
        w = np.exp(np.clip((c5 - m) / tau, -50, 0))
        w /= np.sum(w)
        grad = np.zeros(n)
        for mi in range(2 * n - 1):
            ki = mi - (n - 1)
            wi = w[mi]
            if wi < 1e-12:
                continue
            lo1 = max(0, -ki)
            hi1 = min(n - 1, n - 1 - ki)
            if lo1 <= hi1:
                grad[lo1:hi1 + 1] += wi * (1.0 - x[lo1 + ki:hi1 + ki + 1])
            lo2 = max(0, ki)
            hi2 = min(n - 1, n - 1 + ki)
            if lo2 <= hi2:
                grad[lo2:hi2 + 1] -= wi * x[lo2 - ki:hi2 - ki + 1]
        step = lr * (1.0 + s / steps) ** (-1.5)
        x = repair(x - step * grad, n)
    return x
