def _smooth_max(s, alpha):

    return m + np.log(np.sum(np.exp(alpha * (s - m)))) / alpha

def _smooth_obj_and_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    s = e / np.sum(e)
    val = m + np.log(np.sum(e)) / alpha
    n = len(h)
    # gradient computation

    # Q[i] = sum_{j=0}^{i} s[n-1+i-j] * h[j]

import numpy as np
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    se = np.sum(e)
    s = e / se
    val = m + np.log(se) / alpha
    # Gradient
    grad = np.zeros(n)
    # s has 2n-1 elements, indexed 0..2n-2
    for i in range(n):
        p = 0.0
        for j in range(n):
            idx1 = n - 1 + i - j  # for first sum
            idx2 = n - 1 + j - i  # for second sum
            p += s[idx1] * (1.0 - h[j]) - s[idx2] * h[j]
        grad[i] = p
    return val, grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(30):
        s = h.sum()
        if s < 1e-14:
            return np.full(n, target / n)
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _grad_step(h, grad, step, n):
    return _repair(h - step * grad, n)

def _binary_search(h_bin, n, rng, max_iters=20000):
    dx = 2.0 / n
    best_c5 = _compute_c5(h_bin.astype(float), dx)
    best = h_bin.copy()
    for _ in range(max_iters):
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        best[i] = 0
        best[j] = 1
        c5 = _compute_c5(best.astype(float), dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
        else:
            best[i] = 1
            best[j] = 0
    return best.astype(float), best_c5

def _make_binary(h, n, rng):
    h = h.copy()
    h = np.round(h)
    h = np.clip(h, 0, 1)
    diff = n / 2.0 - h.sum()

        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:int(diff)]] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:int(-diff)]] = 0
    return h
