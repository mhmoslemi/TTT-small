    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2[0]] = np.clip(h[idx2[0]] + diff2, 0, 1)

    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

        if time.time() - t0 > t_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > t_limit:
                break
            for ds in steps:
                found = False
                for s in (1, -1):
                    nh = h.copy()
                    nh[i] = np.clip(nh[i] + s * ds, 0, 1)
                    nh = project_feasible(nh, ts)
                    c5 = compute_c5(nh)
                    if c5 < best_c5 - 1e-15:
                        h = nh
                        best_c5 = c5
                        improved = True
                        found = True
                        break
                if found:
                    break
        if not improved:
            break
    return h, best_c5

def targeted_binary(h, n, ts, t_limit=40):
    best_c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(200):
        if time.time() - t0 > t_limit:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode="full")
        ak = int(np.argmax(corr))
        k_lag = ak - (n - 1)
        if k_lag >= 0:
            pairs = [(i, i + k_lag) for i in range(n - k_lag)]
        else:
            pairs = [(i + (-k_lag), i) for i in range(n + k_lag)]
        rng_local = np.random.RandomState(hash(ak) % 2**31)
        rng_local.shuffle(pairs)
        for i, j in pairs:
            if time.time() - t0 > t_limit:
                break
            for idx in (i, j):
                for tv in (0.0, 1.0):
                    if abs(tv - h[idx]) < 1e-9:
                        continue
                    old = h[idx]
                    h[idx] = tv
                    h = project_feasible(h, ts)
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-15:
                        best_c5 = c5
                        improved = True
                    else:
                        h[idx] = old
                        h = project_feasible(h, ts)
        if not improved:
            break
    return h, best_c5

def optimize_at_n(h_start, n, ts, t_remaining):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # SLSQP sweep
    alphas = [5, 20, 100, 500, 2000, 8000, 30000]
    for beta in alphas:
        if t_remaining - (time.time() - (t_remaining - 1100)) > 0:

        if time.time() - (t_remaining - (t_remaining)) < t_remaining:

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def rem():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(140)
        h_init[rng.choice(140, 70, replace=False)] = 1.0

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Deep optimization at initial n
    n = n_init
    ts = n // 2
    h = h_init.copy()
    for beta in [5, 20, 100, 500, 2000, 8000, 30000]:
        if rem() < 40:
            break
        h = slsqp_run(h, n, beta, ts, maxiter=600)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    if rem() > 30:
        h, c5 = binary_search(best_h.copy(), n, ts, t_limit=min(50, rem() // 5))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    if rem() > 25:
        h, c5 = targeted_binary(best_h.copy(), n, ts, t_limit=min(30, rem() // 6))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    if rem() > 20:
        h, c5 = fine_local(best_h.copy(), n, ts, t_limit=min(25, rem() // 5))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    if rem() > 20:
        h = slsqp_run(best_h.copy(), n, 100000, ts, maxiter=800)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if rem() > 15:
            h, c5 = binary_search(h, n, ts, t_limit=15)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 2: Try other n values
    other_ns = [120, 100, 160, 200, 280]
    for n_try in other_ns:
        if rem() < 70:
            break
        ts_try = n_try // 2
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(xn, xo, best_h)
        h_try = project_feasible(h_try, ts_try)

        for beta in [50, 500, 5000]:
            if rem() < 25:
                break
            h_try = slsqp_run(h_try, n_try, beta, ts_try, maxiter=400)
            c5_t = compute_c5(h_try)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_try.copy()

        if rem() > 30:
            h_try, c5_t = binary_search(h_try, n_try, ts_try, t_limit=min(30, rem() // 4))
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_try.copy()

    # Phase 3: Random restarts around best
    if rem() > 60:
        n_b = len(best_h)
        ts_b = n_b // 2
        n_rest = max(3, int(rem() // 25))
        for _ in range(n_rest):
            if rem() < 25:
                break
            scale = rng.choice([0.01, 0.02, 0.05, 0.1])
            h_r = best_h.copy() + rng.normal(0, scale, n_b)
            h_r = project_feasible(h_r, ts_b)
            h_r = slsqp_run(h_r, n_b, 1000, ts_b, maxiter=400)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            if rem() > 20:
                h_r, c5_r = binary_search(h_r, n_b, ts_b, t_limit=min(15, rem() // 4))
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()

    # Phase 4: Final polish
    if rem() > 15:
        n_f = len(best_h)
        ts_f = n_f // 2
        h_f = slsqp_run(best_h.copy(), n_f, 50000, ts_f, maxiter=500)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if rem() > 10:
            h_f, c5_f = binary_search(h_f, n_f, ts_f, t_limit=min(10, rem() // 2))
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    best_h = project_feasible(best_h, len(best_h) // 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
