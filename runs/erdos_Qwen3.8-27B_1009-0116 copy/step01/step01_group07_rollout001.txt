from scipy.interpolate import interp1d

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _normalize(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * target / s
    return np.clip(h, 0.0, 1.0)

def _interp(parent, n_new):
    n_old = len(parent)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    f = interp1d(x_old, parent, kind='cubic')
    return _normalize(f(x_new), n_new / 2.0)

def _slsqp_opt(h, n, alpha, maxiter=80):
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        sh = np.clip(c - m, -1.0, 1.0)
        return (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * dx
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return _normalize(res.x, target)

def _coord_descent(h, n_rounds=30, steps=(0.02, 0.05, 0.1)):

    target = n / 2.0
    best_c = _c5(h)

                    h2[i] += sign * d

                        if 0 <= h2[j] <= 1:
                            c2 = _c5(h2)
                            if c2 < best_c - 1e-12:

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _normalize(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * target / s
    return np.clip(h, 0.0, 1.0)

def _interp_parent(parent, n_new):
    n_old = len(parent)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    f = interp1d(x_old, parent, kind='cubic', bounds_error=False, fill_value='extrapolate')
    return _normalize(f(x_new), n_new / 2.0)

def _slsqp_run(h_init, n, alpha, maxiter):
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def _fun(x):
        return np.sum(x) - target
    def _jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _fun, 'jac': _jac}]

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        sh = np.clip(c - m, -2.0, 2.0)
        return (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * dx

    res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
    return _normalize(res.x, target)

def _coord_descend(h, n_rounds):
    n = len(h)
    target = n / 2.0
    best_c = _c5(h)
    steps = (0.01, 0.03, 0.06, 0.1)
    for _ in range(n_rounds):
        improved = False
        for i in range(n):
            for d in steps:
                for sgn in (1, -1):
                    delta = sgn * d
                    if h[i] + delta < -1e-12 or h[i] + delta > 1.0 + 1e-12:
                        continue
                    # Try partner j
                    for j in ((i+1) % n, (i+n-1) % n, (i+n//2) % n):
                        if j == i:
                            continue
                        h2 = h.copy()
                        h2[i] += delta
                        h2[j] -= delta
                        if h2[i] < 0 or h2[i] > 1 or h2[j] < 0 or h2[j] > 1:
                            continue
                        c2 = _c5(h2)
                        if c2 < best_c - 1e-13:
                            h = h2
                            best_c = c2
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def _anneal(h, target, t_hi, t_lo, n_iter, seed_val):
    n = len(h)
    rng = np.random.RandomState(seed_val)
    cur_c = _c5(h)
    best = h.copy()
    best_c = cur_c
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        t = t_hi * (t_lo / t_hi) ** frac
        amp = 0.02 + 0.12 * (1.0 - frac)
        h_new = h.copy()
        nm = rng.randint(1, 4)
        idxs = rng.choice(n, nm, replace=False)
        for ix in idxs:
            h_new[ix] += rng.uniform(-amp, amp)
        h_new = _normalize(h_new, target)
        c_new = _c5(h_new)
        accept = (c_new < cur_c) or (rng.random() < np.exp(-(c_new - cur_c) / max(t, 1e-12)))
        if accept:
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best = h_new.copy()
                best_c = c_new
    return best, best_c

def _gen_perturb(base, n, rng):
    hp = base.copy()
    for k in range(rng.randint(1, 6)):
        freq = rng.randint(1, max(2, n // 3))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.03, 0.2)
        hp += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return _normalize(hp, n / 2.0)

        for h_init in inits:

                if c_sa < best_c5:

    c5_final = _c5(h_ret)
    print(f"Final: n={best_n}, C5={c5_final:.6f}")
    return h_ret, c5_final, best_n
