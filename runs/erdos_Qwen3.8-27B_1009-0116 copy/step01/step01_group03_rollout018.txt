```python
def subgradient_descent(h, n, time_limit=100):
    ts = n / 2.0
    t0 = time.time()
    dx = 2.0 / n
    h = h.copy()
    f_prev = compute_c5(h)
    
    for t in range(100000):
        if time.time() - t0 > time_limit:
            break
        
        # Compute subgradient
        corr = np.correlate(h, 1.0 - h, mode="full")
        f_raw = corr.max()
        active = np.where(corr >= f_raw - 1e-10)[0]
        
        # Compute subgradient for active lags
        grad = np.zeros(n)
        for m_idx in active:
            m = m_idx - (n - 1)  # lag offset
            for i in range(n):
                # (1-h[i-m]) if i-m valid
                j1 = i - m
                if 0 <= j1 < n:
                    grad[i] += (1 - h[j1])
                # -h[i+m] if i+m valid
                j2 = i + m
                if 0 <= j2 < n:
                    grad[i] -= h[j2]
        
        grad *= dx / len(active)
        
        # Step size
        alpha = 0.01 / math.sqrt(1 + t * 0.01)
        
        # Update
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, ts)
        
        f_new = compute_c5(h_new)
        if f_new < f_prev:
            h = h_new
            f_prev = f_new
        # else: keep h, maybe reduce alpha
        
    return h, f_prev
```

```python
import numpy as np
from scipy.optimize import minimize
import time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def interp_to_n(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.astype(float).copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def subgradient_step(h, n, ts):
    """One subgradient step on the max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f_max = corr.max()
    active = np.where(corr >= f_max - 1e-12)[0]
    
    grad = np.zeros(n)
    for m_idx in active:
        m = m_idx - (n - 1)
        for i in range(n):
            j1 = i - m
            if 0 <= j1 < n:
                grad[i] += (1.0 - h[j1])
            j2 = i + m
            if 0 <= j2 < n:
                grad[i] -= h[j2]
    
    grad *= dx / len(active)
    return grad, f_max * dx

def subgradient_descent(h, n, time_limit=120):
    ts = n / 2.0
    t0 = time.time()
    h = h.copy()
    f_curr = compute_c5(h)
    alpha = 0.5
    no_improve = 0
    
    for t in range(200000):
        if time.time() - t0 > time_limit:
            break
        
        grad, _ = subgradient_step(h, n, ts)
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, ts)
        f_new = compute_c5(h_new)
        
        if f_new < f_curr - 1e-15:
            h = h_new
            f_curr = f_new
            no_improve = 0
            alpha = min(alpha * 1.05, 5.0)
        else:
            no_improve += 1
            alpha *= 0.95
            if no_improve > 100:
                alpha = 0.5
                no_improve = 0
        
        # Also try diminishing step
        if t % 500 == 0 and t > 0:
            alpha = max(0.01, 1.0 / math.sqrt(1.0 + t * 0.001))
    
    return h, f_curr

def coordinate_descent(h, n, time_limit=120, n_candidates=7):
    ts = n / 2.0
    t0 = time.time()
    h = h.copy()
    f_curr = compute_c5(h)
    rng = np.random.RandomState(123)
    
    for sweep in range(200):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            best_val = h[i]
            best_f = f_curr
            # Try multiple candidates
            candidates = {h[i], 0.0, 1.0, 1.0 - h[i]}
            if 0 < h[i] < 1:
                candidates.update([h[i]*0.5, min(1.0, h[i]*1.5), 0.25, 0.75])
            for cand in candidates:
                cand = min(1.0, max(0.0, cand))
                if abs(cand - h[i]) < 1e-10:
                    continue
                h[i] = cand
                h = project_feasible(h, ts)
                f_test = compute_c5(h)
                if f_test < best_f:
                    best_f = f_test
                    best_val = cand
            h[i] = best_val
            h = project_feasible(h, ts)
            if best_f < f_curr:
                f_curr = best_f
                improved = True
        if not improved:
            break
    
    return h, f_curr

def pair_local_search(h, n, time_limit=120):
    """Local search using pair moves (swap and transfer)."""
    ts = n / 2.0
    t0 = time.time()
    h = h.copy()
    f_curr = compute_c5(h)
    rng = np.random.RandomState(456)
    
    for sweep in range(1000):
        if time.time() - t0 > time_limit:
            break
        improved = False
        # Try all pairs in random order
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        rng.shuffle(pairs)
        # Limit to a subset for speed
        n_pairs = min(len(pairs), 2000)
        for (i, j) in pairs[:n_pairs]:
            if abs(h[i] - h[j]) < 1e-10:
                continue
            # Try swap
            h[i], h[j] = h[j], h[i]
            f_test = compute_c5(h)
            if f_test < f_curr - 1e-14:
                f_curr = f_test
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
            # Try partial transfer
            delta = (h[j] - h[i]) * 0.5
            h[i] += delta
            h[j] -= delta
            h = project_feasible(h, ts)
            f_test = compute_c5(h)
            if f_test < f_curr - 1e-14:
                f_curr = f_test
                improved = True
            else:
                h[i] -= delta
                h[j] += delta
        
        if not improved:
            break
    
    return h, f_curr

def simulated_annealing(h, n, time_limit=200, seed=42):
    """SA with structured moves."""
    ts = n / 2.0
    t0 = time.time()
    h = h.copy()
    f_curr = compute_c5(h)
    f_best = f_curr
    h_best = h.copy()
    rng = np.random.RandomState(seed)
    
    T = 0.01  # initial temperature
    T_min = 1e-8
    cooling = 0.9997
    
    while time.time() - t0 < time_limit:
        T = max(T * cooling, T_min)
        
        # Generate move
        move_type = rng.randint(4)
        h_new = h.copy()
        
        if move_type == 0:  # Single flip
            i = rng.randint(n)
            h_new[i] = 1.0 - h[i]
        elif move_type == 1:  # Pair swap
            i, j = rng.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h[j], h[i]
        elif move_type == 2:  # Transfer
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.uniform(-0.3, 0.3)
            h_new[i] += delta
            h_new[j] -= delta
        else:  # Gaussian perturbation
            idx = rng.choice(n, rng.randint(2, 5), replace=False)
            h_new[idx] += rng.normal(0, 0.1, len(idx))
        
        h_new = project_feasible(h_new, ts)
        f_new = compute_c5(h_new)
        
        delta_f = f_new - f_curr
        if delta_f < 0 or rng.random() < math.exp(-delta_f / T):
            h = h_new
            f_curr = f_new
            if f_curr < f_best:
                f_best = f_curr
                h_best = h.copy()
    
    return h_best, f_best

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def tl():
        return budget_s - (time.time() - t_start)
    
    # Get initial

            if c5 < best_c5:

        # Phase 5: Final coordinate descent polish

            if c5 < best_c5:

            if c5 < best_c5:

                best_h = h2.copy()
    
    # Final projection
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
```

```python
    corr = np.correlate(h, 1.0 - h, mode="full")

    max_corr = corr.max()
```

```python
def smoothed_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    max_corr = corr.max()
    exp_corr = np.exp(beta * (corr - max_corr))
    weights = exp_corr / exp_corr.sum()
    
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-15:
            continue
        k = j - (n - 1)
        if k > 0:
            a = np.concatenate((np.zeros(k), one_minus_h[:n-k]))
            b = np.concatenate((h[k:], np.zeros(k)))
        elif k < 0:
            m = -k
            a = np.concatenate((one_minus_h[m:], np.zeros(m)))
            b = np.concatenate((np.zeros(m), h[:n-m]))
        else:
            a = one_minus_h
            b = h
        grad += w * (a - b)
    
    grad *= dx
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def interp_to_n(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.astype(float).copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def smoothed_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mxc = corr.max()
    ec = np.exp(beta * (corr - mxc))
    wts = ec / ec.sum()
    omh = 1.0 - h
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = wts[j]
        if w < 1e-14:
            continue
        k = j - (n - 1)
        if k > 0:
            a = np.concatenate((np.zeros(k), omh[:n-k]))
            b = np.concatenate((h[k:], np.zeros(k)))
        elif k < 0:
            m = -k
            a = np.concatenate((omh[m:], np.zeros(m)))
            b = np.concatenate((np.zeros(m), h[:n-m]))
        else:
            a = omh
            b = h
        grad += w * (a - b)
    grad *= dx
    return grad

def grad_descent(h, n, time_limit=150):
    ts = n / 2.0
    t0 = time.time()
    h = h.copy()
    f_curr = compute_c5(h)
    alpha = 0.1
    beta = 10.0
    no_imp = 0
    for t in range(500000):
        if time.time() - t0 > time_limit:
            break
        beta = 10.0 + t * 0.05
        g = smoothed_grad(h, min(beta, 500.0))
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        h_new = project_feasible(h - alpha * g, ts)
        f_new = compute_c5(h_new)
        if f_new < f_curr - 1e-15:
            h = h_new
            f_curr = f_new
            no_imp = 0
            alpha = min(alpha * 1.02, 2.0)
        else:
            no_imp += 1
            alpha *= 0.9
            if no_imp > 200:
                alpha = 0.5 / (1.0 + 0.01 * t)
                no_imp = 0
    return h, f_curr

def coord_descent(h, n, time_limit=120):
    ts = n / 2.0
    t0 = time.time()
    h = h.copy()
    f_curr = compute_c5(h)
    rng = np.random.RandomState(77)
    for sweep in range(300):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            best_val = h[i]
            best_f = f_curr
            cands = [h[i], 0.0, 1.0, 1.0 - h[i]]
            if 0.01 < h[i] < 0.99:
                cands += [h[i]*0.4, h[i]*1.6, 0.25, 0.75, 0.1, 0.9]
            for cv in cands:
                cv = min(1.0, max(0.0, cv))
                if abs(cv - h[i]) < 1e-10:
                    continue
                h[i] = cv
                h = project_feasible(h, ts)
                ft = compute_c5(h)
                if ft < best_f:
                    best_f = ft
                    best_val = cv
            h[i] = best_val
            h = project_feasible(h, ts)
            if best_f < f_curr:
                f_curr = best_f
                improved = True
        if not improved:
            break
    return h, f_curr

def pair_search(h, n, time_limit=100):
    ts = n / 2.0
    t0 = time.time()
    h = h.copy()
    f_curr = compute_c5(h)
    rng = np.random.RandomState(99)
    for sweep in range(500):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for _ in range(500):
            i, j = rng.choice(n, 2, replace=False)
            if abs(h[i] - h[j]) < 1e-10:
                continue
            hi, hj = h[i], h[j]
            h[i], h[j] = hj, hi
            ft = compute_c5(h)
            if ft < f_curr - 1e-14:
                f_curr = ft
                improved = True
            else:
                h[i], h[j] = hi, hj
            mid = (hi + hj) * 0.5
            h[i] = mid + (hi - mid) * 0.5
            h[j] = mid - (hj - mid) * 0.5
            h = project_feasible(h, ts)
            ft = compute_c5(h)
            if ft < f_curr - 1e-14:
                f_curr = ft
                improved = True
            else:
                h[i], h[j] = hi, hj
        if not improved:
            break
    return h, f_curr

def sa_search(h, n, time_limit=200, seed=42):
    ts = n / 2.0
    t0 = time.time()
    h = h.copy()
    f_curr = compute_c5(h)
    f_best = f_curr
    h_best = h.copy()
    rng = np.random.RandomState(seed)
    T = 0.008
    T_min = 1e-9
    cool = 0.99985
    stuck = 0
    while time.time() - t0 < time_limit:
        T = max(T * cool, T_min)
        mt = rng.randint(5)
        h_new = h.copy()
        if mt == 0:
            i = rng.randint(n)
            h_new[i] = 1.0 - h[i]
        elif mt == 1:
            i, j = rng.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h[j], h[i]
        elif mt == 2:
            i, j = rng.choice(n, 2, replace=False)
            d = rng.uniform(-0.4, 0.4)
            h_new[i] += d
            h_new[j] -= d
        elif mt == 3:
            i, j, k = rng.choice(n, 3, replace=False)
            h_new[[i, j, k]] = h[[j, k, i]]
        else:
            idx = rng.choice(n, rng.randint(2, 6), replace=False)
            h_new[idx] += rng.normal(0, 0.08, len(idx))
        h_new = project_feasible(h_new, ts)
        f_new = compute_c5(h_new)
        df = f_new - f_curr
        if df < 0 or rng.random() < math.exp(-df / T):
            h = h_new
            f_curr = f_new
            stuck = 0
            if f_curr < f_best:
                f_best = f_curr
                h_best = h.copy()
        else:
            stuck += 1
            if stuck > 500:
                stuck = 0
    return h_best, f_best

def push_binary(h, n):
    ts = n / 2.0
    f_curr = compute_c5(h)
    h2 = h.copy()
    non_ext = np.where((h2 > 1e-6) & (h2 < 1.0 - 1e-6))[0]
    rng = np.random.RandomState(55)
    rng.shuffle(non_ext)
    for i in non_ext:
        for target in [0.0, 1.0]:
            h2[i] = target
            h2 = project_feasible(h2, ts)
            ft = compute_c5(h2)
            if ft <= f_curr + 1e-14:
                f_curr = ft
                break
            else:
                h2[i] = h[i]
                h2 = project_feasible(h2, ts)
    return h2, f_curr

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def tl():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        rng0 = np.random.RandomState(seed)
        h_parent = np.zeros(140)
        h_parent[rng0.choice(140, 70, replace=False)] = 1.0

    best_h = h_parent.astype(float).copy()
    best_c5 = compute_c5(best_h)

    n_list = [len(h_parent), 100, 70, 160, 200]

    for n in n_list:
        if tl() < 80:
            break
        h = h_parent.copy() if n == len(h_parent) else interp_to_n(h_parent, n)
        ts = n / 2.0
        h = project_feasible(h, ts)

        if tl() > 200:
            h, c5 = grad_descent(h, n, time_limit=min(tl() * 0.18, 150))
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

        if tl() > 150:
            h, c5 = coord_descent(h, n, time_limit=min(tl() * 0.22, 140))
            if c5 < best_c5:
                best_c5, best_h = c5, n // 2 and h.copy()
                best_h = h.copy()

        if tl() > 120:
            h, c5 = pair_search(h, n, time_limit=min(tl() * 0.2, 100))
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

        if tl() > 150:
            h, c5 = sa_search(h, n, time_limit=min(tl() * 0.25, 200), seed=seed + n)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

        if tl() > 60:
            h, c5 = push_binary(h if len(h) == n else best_h, n)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

        if tl() > 40:
            hp = best_h.copy() if len(best_h) == n else h
            hp, c5 = grad_descent(hp, n, time_limit=min(tl() * 0.12, 60))
            if c5 < best_c5:
                best_c5, best_h = c5, hp.copy()

        if tl() > 30:
            hp = best_h.copy() if len(best_h) == n else h
            hp, c5 = coord_descent(hp, n, time_limit=min(tl() * 0.1, 50))
            if c5 < best_c5:
                best_c5, best_h = c5, hp.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
