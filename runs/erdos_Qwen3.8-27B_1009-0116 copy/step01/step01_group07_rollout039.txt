import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        if np.sum(mask) == 0:
            return h
        room = 1.0 - h[mask]
        tot_room = room.sum()
        if tot_room < 1e-14:
            return h
        frac = min(diff / tot_room, 1.0)
        h[mask] += room * frac
    else:
        mask = h > 1e-12
        if np.sum(mask) == 0:
            return h
        h[mask] += diff * h[mask] / np.sum(h[mask])
    return np.clip(h, 0.0, 1.0)

def _grad_smooth_max(h, alpha, dx, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(corr - m, -20.0 / alpha, 0.0)
    ew = np.exp(alpha * shifted)
    sw = ew.sum()
    w = ew / sw
    grad = np.zeros(n)
    # Vectorized: use the fact that grad[i] = sum_j w[j] * [(1-h[i-(j-n+1)]) - h[i+(j-n+1)]]
    # where valid indices are in [0,n)
    # Let lags = np.arange(2*n-1) - n + 1 (shape (2n-1,))
    # For position i, term1 = sum_j w[j] * (1-h[i-lags[j]]) for 0<=i-lags[j]<n
    # term2 = sum_j w[j] * h[i+lags[j]] for 0<=i+lags[j]<n

    # term1[i] = (1-h) * w convolved at offset i... 
    # Actually: term1 = cross_correlation of w with (1-h)
    # np.correlate(w, 1-h, 'full')[n-1+1-(n-1)] ... hmm
    # Let me just use the loop for correctness. n is small.
    lags = np.arange(2 * n - 1) - (n - 1)
    for i in range(n):
        a = i - lags  # = i - lag for each lag
        b = i + lags  # = i + lag for each lag
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[i] = np.sum(w[m1] * (1.0 - h[a[m1]])) - np.sum(w[m2] * h[b[m2]])
    grad *= dx
    grad += 2.0 * 500.0 * (np.sum(h) - target)

def _psolve(n, h0, target, max_iter=300, dx=None):
    if dx is None:
        dx = 2.0 / n
    best_h = h0.copy()
    best_c = _c5(h0)
    h = h0.copy()
    alpha = 5.0
    lr = 0.1
    for it in range(max_iter):
        grad = _grad_smooth_max(h, alpha, dx, target)
        # Backtracking line search
        for trial in range(20):
            step = lr * (0.5 ** trial)
            h_new = _project(h - step * grad, target)
            c_new = _c5(h_new)
            if c_new < best_c or c_new < _c5(h) - 1e-12:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
        else:
            # No improvement, increase alpha
            alpha = min(alpha * 1.5, 500.0)
            lr = 0.05
    return best_h, best_c

    n = len(h)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        if not np.any(mask):
            return h
        room = 1.0 - h[mask]
        tr = room.sum()
        if tr < 1e-14:
            return h
        h[mask] += room * min(diff / tr, 1.0)
    else:
        mask = h > 1e-12
        if not np.any(mask):
            return h
        h[mask] += diff * h[mask] / np.sum(h[mask])
    return np.clip(h, 0.0, 1.0)

def _grad_c5_smooth(h, alpha, dx, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(corr - m, -50.0 / alpha, 0.0)
    ew = np.exp(alpha * shifted)

        b = i + lags

def _project_exact(h, target):
    n = len(h)
    target = float(target)
    # First handle the box constraint
    # We need: sum(clip(h - tau, 0, 1)) = target
    # This is monotone decreasing in tau
    lo, hi = -1.0, 1.0
    
    def f(tau):
        return np.sum(np.clip(h - tau, 0.0, 1.0)) - target
    
    if f(0) == 0 or abs(f(0)) < 1e-14:
        return np.clip(h, 0, 1)
    
    if f(0) > 0:
        # Need to decrease: tau > 0
        lo, hi = 0.0, 1.0 + max(h) - target/n
    else:
        # Need to increase: tau < 0
        lo, hi = -(1.0 - min(h)), 0.0
    
    for _ in range(60):

        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _proj_exact(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_smooth(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -50.0 / alpha, 0.0)
    ew = np.exp(alpha * sh)
    w = ew / ew.sum()
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.empty(n)
    for i in range(n):
        a = i - lags
        b = i + lags
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[i] = np.sum(w[m1] * (1.0 - h[a[m1]])) - np.sum(w[m2] * h[b[m2]])
    return grad * dx

def _pgd(h0, target, dx, max_iter, t_limit):

    best = h.copy()
    bestc = _c5(h)
    alpha = 5.0
    lr = 0.5
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > t_limit:
            break
        g = _grad_smooth(h, alpha, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        step = lr / max(gn, 0.01)
        ok = False
        for _ in range(20):
            h_new = _proj_exact(h - step * g, target)
            c = _c5(h_new)
            if c < bestc - 1e-14:
                h = h_new
                bestc = c
                best = h.copy()
                ok = True
                break
            if c < _c5(h) - 1e-14:
                h = h_new
                ok = True
                break
            step *= 0.5
        if not ok:
            alpha = min(alpha * 1.7, 2000.0)
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0
            if alpha < 200:
                alpha *= 1.08
    return best, bestc

def _coord(h_in, target, max_rd, t_lim):
    t0 = time.time()
    n = len(h_in)
    h = h_in.copy()
    bc = _c5(h)
    ds = (0.3, -0.3, 0.15, -0.15, 0.08, -0.08, 0.04, -0.04, 0.02, -0.02, 0.01, -0.01)
    joffs = (3, 11, 1, 7, 17, 5)
    for rd in range(max_rd):
        if time.time() - t0 > t_lim:
            break
        mv = False
        for i in range(n):
            hi0 = h[i]
            for d in ds:
                nv = hi0 + d
                if nv < 0 or nv > 1:
                    continue
                for jo in joffs:
                    j = (i + jo) % n
                    hj0 = h[j]
                    nj = hj0 - d
                    if nj < 0 or nj > 1:

                    h[i] = nv
                    h[j] = nj
                    c = _c5(h)
                    if c < bc - 1e-13:
                        bc = c
                        mv = True
                        break
                    h[i] = hi0
                    h[j] = hj0
                if mv:
                    break
            if mv:
                break
        if not mv:
            break
    return _proj_exact(h, target), bc

def _slsqp_solve(h0, n, target, dx, mi=60):
    bnds = [(0.0, 1.0)] * n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        s = np.clip(c - m, -0.005, 0.0)
        return (m + np.log(np.sum(np.exp(1000.0 * s))) / 1000.0) * dx + 80.0 * (x.sum() - target) ** 2

    def eqf(x):
        return x.sum() - target

    def eqj(x):
        return np.ones(n)

    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bnds,
                     constraints=[{'type': 'eq', 'fun': eqf, 'jac': eqj}],
                     options={'maxiter': mi, 'ftol': 1e-13})
        return _proj_exact(r.x, target)
    except Exception:
        return h0
