import numpy as np
from scipy.optimize import minimize
import time

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_corr_arr(h)) * dx)

def _smooth_obj(x, alpha):
    corr = _corr_arr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _eq_con(x, target):
    return np.sum(x) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-9:
                break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _eval_c5_pair(h, i, j, d, dx):
    hi, hj = h[i], h[j]
    h[i] = hi + d
    h[j] = hj - d
    c5 = _compute_c5(h, dx)
    h[i] = hi
    h[j] = hj
    return c5

def _golden_min_pair(h, i, j, dx, d_lo, d_hi, n_iter=30):
    hi0, hj0 = h[i], h[j]
    invphi = 0.6180339887
    a, b = d_lo, d_hi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = _eval_c5_pair(h, i, j, c, dx)
    fd = _eval_c5_pair(h, i, j, d, dx)
    for _ in range(n_iter):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = _eval_c5_pair(h, i, j, c, dx)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = _eval_c5_pair(h, i, j, d, dx)
    best_d = (a + b) / 2.0
    c5_best = _eval_c5_pair(h, i, j, best_d, dx)
    c5_zero = _eval_c5_pair(h, i, j, 0.0, dx)
    if c5_best < c5_zero - 1e-14:
        h[i] = hi0 + best_d
        h[j] = hj0 - best_d
        return c5_best
    return c5_zero

def _pair_sweep(h, n, dx, t_start, budget_s, max_sweeps, rng, shuffle=True):
    best_c5 = _compute_c5(h, dx)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for sweep in range(max_sweeps):
        if time.time() - t_start > budget_s - 25:
            break
        if shuffle:
            rng.shuffle(pairs)
        improved = False
        for (i, j) in pairs:
            if time.time() - t_start > budget_s - 25:
                break
            hi, hj = h[i], h[j]
            d_lo = max(-hi, hj - 1.0) + 1e-12
            d_hi = min(1.0 - hi, hj) - 1e-12
            if d_hi <= d_lo:
                continue
            c5_new = _golden_min_pair(h, i, j, dx, d_lo, d_hi)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
        if not improved:
            break
    return h, best_c5

def _slsqp_step(h, n, dx, alpha, maxiter):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target_sum,)}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        c5 = _compute_c5(hc, dx)
        return hc, c5
    except Exception:
        return h, _compute_c5(h, dx)

    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    # Phase 2: Main pair golden-section sweeps (shuffled)
    best_h, best_c5 = _pair_sweep(best_h, n, dx, t_start, budget_s,
                                  max_sweeps=300, rng=rng, shuffle=True)

    # Phase 3: SLSQP refinement from improved point
    if time.time() - t_start < budget_s - 100:

            if time.time() - t_start > budget_s - 80:

            hc, c5 = _slsqp_step(best_h, n, dx, alpha, 300)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()

    # Phase 4: More pair sweeps (deterministic order this time)
    best_h, best_c5 = _pair_sweep(best_h, n, dx, t_start, budget_s,
                                  max_sweeps=300, rng=rng, shuffle=False)

    # Phase 5: Final SLSQP + pair polish
    if time.time() - t_start < budget_s - 50:
        hc, c5 = _slsqp_step(best_h, n, dx, 1000, 400)
        if c5 < best_c5:

    best_h, best_c5 = _pair_sweep(best_h, n, dx, t_start, budget_s,

    # Final repair and evaluation
    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
