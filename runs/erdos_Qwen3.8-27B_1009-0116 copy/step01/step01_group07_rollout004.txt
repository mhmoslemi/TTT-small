                best_c5 = cp
                best_h = h_pert.copy()
                print(f"  ILS it={it}: C5={cp:.6f}")

    # Upsample and polish
    if time.time() < deadline - 35 and best_n < 250:
        n_up = min(300, best_n * 2)
        target_up = n_up / 2.0
        h_up = _norm(_interp_to(best_h, n_up), target_up)
        for alpha in [500, 2000, 8000]:
            if time.time() > deadline - 15:
                break
            h_up = _slsqp(h_up, n_up, target_up, alpha, 5000.0, 40)
        h_up, cup = _forced_coord(h_up, target_up, 2000, min(deadline - 5, time.time() + 15), 42)
        if cup < best_c5:
            best_c5 = cup
            best_h = h_up.copy()
            best_n = n_up
            print(f"  Upsample n={n_up}: C5={cup:.6f}")

    # Final normalization
    tgt = best_n / 2.0
    diff = tgt - np.sum(best_h)
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            best_h[mask] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)

    c5f = _c5(best_h)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return best_h, c5f, best_n
