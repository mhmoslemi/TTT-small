import numpy as np
from scipy.optimize import minimize
import time


def _get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _compute_c5(h, dx):
    return np.max(_get_corr(h)) * dx


def _soft_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def _slsqp_smooth(h_start, alpha, n, bounds, target):
    def obj(x):
        return _soft_max(_get_corr(x), alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 400, 'ftol': 1e-15})
    return res.x


def _local_search(h, n, dx, best_c5, t0, budget_s, reserve, deltas):
    h = h.copy()
    for sweep in range(500):
        if time.time() - t0 > budget_s - reserve:
            break
        improved = False
        for i in range(n):
            hi = h[i]
            for j in range(i + 1, n):
                hj = h[j]
                hi = h[i]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = _compute_c5(h, dx)
                    if c5 < best_c5 - 1e-11:
                        best_c5 = c5
                        improved = True
                        hj = nj
                    else:
                        h[i], h[j] = hi, hj
        if not improved:
            break
    return h, best_c5
