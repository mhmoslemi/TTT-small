def _project(y, target, n):
    """Project y onto {x: 0<=x[i]<=1, sum(x)=target}."""
    # Binary search for lambda
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo+hi)/2, 0, 1)
    # Small correction
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
        else:
            order = np.argsort(x)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - x[idx])
                x[idx] += add
                diff -= add
            else:
                sub = min(-diff, x[idx])
                x[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return x

def _fista(h0, alpha, target, n, n_iter=2000, step_size=None):
    h = h0.copy()
    z = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_obj = _compute_c5(h, 2.0/n)
    
    if step_size is None:
        # Estimate step size
        step_size = 1.0 / (2 * n * alpha)  # rough Lipschitz estimate
    
    for it in range(n_iter):
        # Compute gradient at z
        _, grad = _smooth_obj_and_grad(z, alpha)
        
        # Gradient step
        y = z - step_size * grad
        
        # Project
        x_new = _project(y, target, n)
        
        # FISTA momentum
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        z = x_new + ((t - 1) / t_new) * (x_new - h)
        z = _project(z, target, n)  # Ensure z is feasible
        
        h = x_new
        t = t_new
        
        obj = _compute_c5(h, 2.0/n)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
    
    return best_h, best_obj

                h[i] = ni
                h[j] = nj
                c = _c5(h, dx)
                if c < bc - 1e-12:
                    bc = c
                    imp = True
                    break
                h[i] = hi0
                h[j] = hj0
            else:
                h[i] = hi0
                h[j] = hj0
        if not imp:
            break
    return h, bc

def _slsqp_obj(x, a):
    return _sobj(x, a)

def _slsqp_con(x, target):
    return float(np.sum(x)) - target

def _rem_time(t0, budget_s):
    return budget_s - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0
    rng = np.random.RandomState(seed)

    h0 = _proj(np.array(initial_h_values, dtype=float), tgt)
    bh = h0.copy()
    bc = _c5(bh, dx)

    # Phase 1: FISTA with increasing alpha from initial
    for a in [30, 60, 120, 250, 500, 1000, 2000]:
        if _rem_time(t0, budget_s) < 50:
            break
        h, c = _fista(bh, a, tgt, iters=4000)
        if c < bc:
            bc = c
            bh = h

    # Phase 2: Intensive coordinate descent
    if _rem_time(t0, budget_s) > 20:
        h, c = _cd(bh, dx, n, 500, rng)
        if c < bc:
            bc = c
            bh = h

    # Phase 3: Structured + random restarts
    sigmas = [0.1, 0.2, 0.35, 0.5, 0.7, 0.15, 0.4, 0.6, 0.25, 0.55,
              0.8, 0.3, 0.45, 0.65, 0.2, 0.5, 0.3, 0.7, 0.15, 0.4]
    for s in sigmas:
        if _rem_time(t0, budget_s) < 60:
            break
        hp = _proj(bh + s * rng.randn(n), tgt)
        h, c = _fista(hp, 300, tgt, iters=3000)
        if c < bc:
            bc = c
            bh = h
            if _rem_time(t0, budget_s) > 30:
                h2, c2 = _cd(bh, dx, n, 150, rng)
                if c2 < bc:
                    bc = c2
                    bh = h2

    # Phase 4: Structured initial guesses (sinusoidal)
    if _rem_time(t0, budget_s) > 100:
        for freq in [1, 2, 3, 4]:
            for amp in [0.1, 0.2, 0.3, 0.4]:
                if _rem_time(t0, budget_s) < 60:
                    break
                phase = rng.rand() * 2.0 * np.pi
                hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
                hs = _proj(hs, tgt)
                h, c = _fista(hs, 200, tgt, iters=2000)
                if c < bc:
                    bc = c
                    bh = h
                    h2, c2 = _cd(bh, dx, n, 100, rng)
                    if c2 < bc:
                        bc = c2
                        bh = h2

    # Phase 5: SLSQP polish (multiple attempts)
    bnds = [(0.0, 1.0)] * n
    for a_polish in [2000, 5000]:
        if _rem_time(t0, budget_s) < 30:
            break
        try:
            r = minimize(_slsqp_obj, bh, args=(a_polish,), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (tgt,)}],
                         options={'maxiter': 600, 'ftol': 1e-15})
            hf = _proj(r.x, tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc = cf
                bh = hf
        except Exception:
            pass

    # Phase 6: More random restarts with smaller sigma (fine-tuning)
    if _rem_time(t0, budget_s) > 100:
        for s in [0.05, 0.08, 0.1, 0.06, 0.04, 0.07, 0.05, 0.03, 0.06, 0.04]:
            if _rem_time(t0, budget_s) < 50:
                break
            hp = _proj(bh + s * rng.randn(n), tgt)
            h, c = _fista(hp, 500, tgt, iters=3000)
            if c < bc:
                bc = c
                bh = h
                h2, c2 = _cd(bh, dx, n, 100, rng)
                if c2 < bc:
                    bc = c2
                    bh = h2

    # Phase 7: Final intensive coordinate descent
    if _rem_time(t0, budget_s) > 15:
        h, c = _cd(bh, dx, n, 1000, rng)
        if c < bc:
            bc = c
            bh = h

    # Phase 8: Final SLSQP
    if _rem_time(t0, budget_s) > 20:
        try:
            r = minimize(_slsqp_obj, bh, args=(10000,), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (tgt,)}],
                         options={'maxiter': 800, 'ftol': 1e-16})
            hf = _proj(r.x, tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc = cf
                bh = hf
        except Exception:
            pass

    # Final CD after SLSQP
    if _rem_time(t0, budget_s) > 8:
        h, c = _cd(bh, dx, n, 500, rng)
        if c < bc:
            bc = c
            bh = h

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
