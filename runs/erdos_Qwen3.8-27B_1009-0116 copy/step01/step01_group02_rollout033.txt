def subgradient_descent(h, n, alpha, n_iter):
    for it in range(n_iter):
        corr = np.correlate(h, 1-h, mode='full')
        max_val = np.max(corr)
        # Find active lags
        active = corr >= max_val - 1e-8
        # Compute subgradient
        grad = np.zeros(n)
        for m in range(2*n-1):
            if not active[m]:
                continue
            k = m - (n-1)  # the actual lag
            # ∂f_k/∂h[j] for valid j
            # First term: (1-h[j-k]) where j-k is valid i and j is valid
            # Second term: -h[j+k] where j is valid i and j+k is valid
            for j in range(n):
                g = 0.0
                # Contribution from h[j] being in the "h[i+k]" slot
                i = j - k
                if 0 <= i < n:
                    # Check if this i is in the valid range for this m
                    # Valid i range: max(0,-k) to min(n-1, n-1-k)
                    if max(0, -k) <= i <= min(n-1, n-1-k):

                # Contribution from h[j] being in the "(1-h[i])" slot

        # Project: grad - mean(grad) (to make it orthogonal to the constraint direction)
        grad -= np.mean(grad)
        h = h - alpha * grad
        h = repair(h, n)
        # Adaptive step
        alpha *= 0.999
    return h

        j_start = i_start + k  # = max(k, 0) when k>0, 0 when k<0... 
        # Actually j = i + k, so j_start = i_start + k, j_end = i_end + k
        # And we need 0 <= j < n

            # First term: for j=k to n-1, grad[j] += (1-h[j-k])
            grad[k:] += one_minus_h[:n-k]
            # Second term: for i=0 to n-1-k, grad[i] -= h[i+k]
            grad[:n-k] -= h[k:]

            grad[kp:] -= h[:n-kp]
    grad /= len(active_lags)
    # Project off the constraint direction
    grad -= np.mean(grad)
    return grad

                # Symmetric pattern
                half = n // 2
                h0h = rng.uniform(0.1, 0.9, half)
                h0 = np.concatenate([h0h, h0h[::-1]])[:n]
            h0 = repair(h0, n)
            hc = slsqp_opt(h0, n, maxiter=200, ftol=1e-11)
            c = compute_c5(hc, n)
            if c < best_c5:
                best_c5, best_h, best_n = c, hc.copy(), n

    # Phase 2: Subgradient descent with periodic polish
    t_p2 = budget_s * 0.65
    if best_h is not None and elapsed() < t_p2:
        n = best_n
        h = best_h.copy()
        alpha = 0.02
        no_improve = 0
        c_curr = best_c5
        while elapsed() < t_p2:
            h_new, c_new, _ = subgrad_step(h, n, tol_frac=0.004, alpha=alpha)
            if c_new < c_curr - 1e-13:
                h, c_curr = h_new, c_new
                alpha = min(alpha * 1.02, 0.12)
                no_improve = 0
                if c_curr < best_c5:
                    best_c5, best_h = c_curr, h.copy()
            else:
                alpha *= 0.97
                no_improve += 1
                if no_improve % 300 == 0:
                    h = slsqp_opt(h, n, maxiter=150, ftol=1e-12)
                    c_curr = compute_c5(h, n)
                    if c_curr < best_c5:
                        best_c5, best_h = c_curr, h.copy()
                        no_improve = 0
                if no_improve > 1500:
                    h = repair(h + rng.normal(0, 0.15, n), n)
                    c_curr = compute_c5(h, n)
                    alpha = 0.04
                    no_improve = 0

    # Phase 3: Intensive local search with diverse moves
    t_p3 = budget_s * 0.88
    if best_h is not None and elapsed() < t_p3:
        n = best_n
        h = best_h.copy()
        stall = 0
        while elapsed() < t_p3:
            move_type = stall % 5
            if move_type < 2:
                sigma = 0.01 + 0.06 * min(stall / 8.0, 5.0)
                h_pert = repair(h + rng.normal(0, sigma, n), n)
            elif move_type == 2:
                h_pert = block_shift(h, n, rng)
            elif move_type == 3:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    h_pert = h.copy()
                else:
                    eps = float(rng.choice([0.02, 0.05, 0.08, 0.12]))
                    h_pert = h.copy()
                    h_pert[i] += eps
                    h_pert[j] -= eps
                    h_pert = repair(h_pert, n)
            else:
                # Top-K guided
                h_pert = h + rng.normal(0, 0.03, n)
                h_pert = repair(h_pert, n)
                try:
                    target_sum = n / 2.0
                    def tk(x):
                        return topk_obj(x, n, K=5)
                    def eqc(x):
                        return np.sum(x) - target_sum
                    r = minimize(tk, h_pert, method='SLSQP',
                                 bounds=[(0.0, 1.0)] * n,
                                 constraints=[{'type': 'eq', 'fun': eqc}],
                                 options={'maxiter': 150, 'ftol': 1e-11})
                    h_pert = repair(r.x, n)
                except Exception:
                    pass

            hc = slsqp_opt(h_pert, n, maxiter=350, ftol=1e-14)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-14:
                best_c5, best_h, h = c, hc.copy(), hc
                stall = 0
            else:
                stall += 1
                if stall > 30:
                    # Aggressive restart from best with large noise
                    h = repair(best_h + rng.normal(0, 0.15, n), n)
                    stall = 15

    # Phase 4: Final polish
    if best_h is not None and elapsed() < budget_s - 10:
        n = best_n
        for _ in range(5):
            if elapsed() > budget_s - 5:
                break
            hc = slsqp_opt(best_h, n, maxiter=500, ftol=1e-15)
            c = compute_c5(hc, n)
            if c < best_c5:
                best_c5, best_h = c, hc

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
