def compute_c5(h, dx):

def compute_grad(h, n, tau, dx):
    """Compute gradient of soft-max objective."""
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    # soft-max weights
    shifted = corr * dx / tau
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    p = exp_shifted / np.sum(exp_shifted)  # weights
    
    # For each shift d (index k in corr), d = k - (n-1)
    # grad[j] += p[k] * ((1-h[j+d]) - h[j-d]) with boundary handling
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        d = k - (n - 1)
        # (1-h[j+d]) for valid j+d
        if d >= 0:
            # j+d valid when j < n-d
            grad[:n-d] += p[k] * (1.0 - h[d:])
        else:
            # j+d valid when j > -d
            grad[-d:] += p[k] * (1.0 - h[:n+d])
        # -h[j-d] for valid j-d
        if d >= 0:
            # j-d valid when j >= d
            grad[d:] -= p[k] * h[:n-d]
        else:
            # j-d valid when j < n+d
            grad[:n+d] -= p[k] * h[-d:]
    return dx * grad

def project_feasible(x, n):
    """Project x onto {0<=h<=1, sum(h)=n/2}."""
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(x)
    if abs(s - target) < 1e-12:
        return x
    # Bisection on lambda: find lambda s.t. sum(clip(x + lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    if s > target:
        # Need to decrease
        lo, hi = -max(x), 0.0
    else:
        lo, hi = 0.0, 1.0 - min(x)
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        trial = np.clip(x + mid, 0.0, 1.0)
        trial_sum = np.sum(trial)
        if trial_sum > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(result)
    # Distribute among non-saturated elements
    free = (result > 1e-10) & (result < 1.0 - 1e-10)
    if np.any(free):
        result[free] += diff / np.sum(free)
        result = np.clip(result, 0.0, 1.0)
    return result

    """Adam optimization with projection."""

    h = h0.copy()

    v = np.zeros(n)  # second moment

    eps = 1e-8
    t = 0
    
    t_start = time.time()

    best_val = compute_c5(h, dx)

        lr = 0.5 * tau  # learning rate scaled with tau
        stuck = 0
        while time.time() - t_start < time_limit:
            t += 1
            grad = compute_grad(h, n, tau, dx)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_feasible(h_new, n)
            
            val = compute_c5(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
            
            if stuck > 200:
                break
    
    return best_h, best_val

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_grad(h, n, tau, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = corr * dx / tau
    ms = np.max(shifted)
    e = np.exp(shifted - ms)
    p = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        d = k - (n - 1)
        w = p[k]
        if d >= 0:
            grad[:n-d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n-d]
        else:
            j = -d
            grad[j:] += w * (1.0 - h[:n-j])
            grad[:n-j] -= w * h[j:]
    return dx * grad

def project_feasible(x, n):

    s = np.sum(x)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_grad(h, n, tau, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx / tau
    ms = np.max(scaled)
    e = np.exp(scaled - ms)
    p = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        s = n - 1 - k
        if s >= 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        else:
            j0 = -s
            grad[j0:] += w * (1.0 - h[:n - j0])
            grad[:n - j0] -= w * h[j0:]
    return dx * grad

def project_feasible(x, n):
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(x)
    if abs(s - target) < 1e-12:
        return x.copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        trial = np.clip(x + mid, 0.0, 1.0)
        if np.sum(trial) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(result)
    free = (result > 1e-5) & (result < 1.0 - 1e-5)
    if np.any(free):
        result[free] += diff / np.sum(free)
    return np.clip(result, 0.0, 1.0)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if s > 1e-12:
        x = x * (tgt / s)
    return project_feasible(np.clip(x, 0.0, 1.0), n)

def adam_optimize(h0, n, dx, taus, time_limit, seed=42, lr=0.05):
    t_start = time.time()
    h = project_feasible(h0.copy(), n)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    tc = 0
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    for tau in taus:
        if time.time() - t_start > time_limit - 1:
            break
        stale = 0
        cur_lr = lr * min(1.0, max(0.1, tau / 0.3))
        while time.time() - t_start < time_limit - 1:
            tc += 1
            g = compute_grad(h, n, tau, dx)
            m1 = b1 * m1 + (1 - b1) * g
            m2 = b2 * m2 + (1 - b2) * g * g
            mh = m1 / (1 - b1 ** tc)
            vh = m2 / (1 - b2 ** tc)
            h_new = h - cur_lr * mh / (np.sqrt(vh) + eps)
            h = project_feasible(h_new, n)
            val = compute_c5(h, dx)
            if val < best_val - 1e-13:
                best_val = val
                best_h = h.copy()
                stale = 0
            else:
                stale += 1
            if stale > 800:
                break
    return best_h, best_val

def slsqp_polish(x0, n, dx, tau, maxiter=200):
    cN = n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        sh = np.clip((corr - m) / tau, -50.0, 0.0)
        return (m + tau * np.log(np.sum(np.exp(sh)))) * dx
    def cfun(x):
        return np.sum(x) - cN / 2.0
    def cjac(x):
        return np.ones(cN)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        r = minimize(obj, x0, method='SLSQP', bounds=bd, constraints=cs,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0_raw = np.asarray(initial_h_values, dtype=float)
    else:
        h0_raw = np.full(96, 0.5)
    n = len(h0_raw)
    dx = 2.0 / n
    h0 = repair(h0_raw, n)

    best_h = h0.copy()
    best_val = compute_c5(h0, dx)

    # Phase 1: Primary Adam with multi-scale tau
    taus1 = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
    p1_time = budget_s * 0.40
    x1, v1 = adam_optimize(h0, n, dx, taus1, p1_time, seed=seed, lr=0.06)
    if v1 < best_val:

        best_h = x1

        h_p = project_feasible(best_h + 0.06 * rng.standard_normal(n), n)
        taus2 = [0.5, 0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003]
        p2_time = budget_s * 0.18
        x2, v2 = adam_optimize(h_p, n, dx, taus2, p2_time, seed=seed + 1, lr=0.04)
        if v2 < best_val:
            best_val = v2
            best_h = x2

    # Phase 3: Another diverse restart (smaller perturbation for refinement)
    if time.time() - t_start < t_start + budget_s * 0.80:
        h_p3 = project_feasible(best_h + 0.03 * rng.standard_normal(n), n)
        taus3 = [0.2, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001]
        p3_time = budget_s * 0.15

        if v3 < best_val:
            best_val = v3
            best_h = x3

    # Phase 4: SLSQP polish at very small tau
    remaining = budget_s - (time.time() - t_start) - 10
    if remaining > 40:
        h_slq = slsqp_polish(best_h, n, dx, 0.001, maxiter=300)
        v_slq = compute_c5(h_slq, dx)
        if v_slq < best_val:
            best_val = v_slq
            best_h = h_slq

    # Phase 5: Final Adam micro-polish
    remaining2 = budget_s - (time.time() - t_start) - 5
    if remaining2 > 20:
        taus5 = [0.005, 0.002, 0.001]
        x5, v5 = adam_optimize(best_h, n, dx, taus5, remaining2 * 0.6,
                               seed=seed + 3, lr=0.02)
        if v5 < best_val:
            best_val = v5
            best_h = x5

    # Final repair and evaluation
    best_h = project_feasible(best_h, n)
    final_val = compute_c5(best_h, dx)
    return (best_h, final_val, n)
