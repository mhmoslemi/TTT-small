import numpy as np
from scipy.optimize import minimize
import time

def _c5v(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _c5c(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c)), c

def _fix(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def _local_pairs(h, n_sweeps=300, t0=None, tlim=None):
    n = len(h)
    bc, _ = _c5c(h)
    for sw in range(n_sweeps):
        if t0 is not None and tlim is not None and time.time() - t0 > tlim:
            break
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                if hi < 1.0 and hj > 0.0:
                    d = min(hj, 1.0 - hi)
                    h[i] += d
                    h[j] -= d
                    c, _ = _c5c(h)
                    if c < bc - 1e-13:
                        bc = c
                        imp = True
                    else:
                        h[i] -= d
                        h[j] += d
                if h[i] > 0.0 and h[j] < 1.0:
                    d2 = min(h[i], 1.0 - h[j])
                    h[i] -= d2
                    h[j] += d2
                    c2, _ = _c5c(h)
                    if c2 < bc - 1e-13:
                        bc = c2
                        imp = True
                    else:
                        h[i] += d2
                        h[j] -= d2
        if not imp:
            break
    return h, bc

def _local_small(h, n_sweeps=100, t0=None, tlim=None):
    n = len(h)
    bc, _ = _c5c(h)
    steps = [0.01, 0.02, 0.04, 0.08]
    for sw in range(n_sweeps):
        if t0 is not None and tlim is not None and time.time() - t0 > tlim:
            break
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                for d in steps:
                    if h[i] + d > 1.0 or h[j] - d < 0.0:
                        continue
                    h[i] += d
                    h[j] -= d
                    c, _ = _c5c(h)
                    if c < bc - 1e-12:
                        bc = c
                        imp = True
                    else:
                        h[i] -= d
                        h[j] += d
                    if h[i] - d < 0.0 or h[j] + d > 1.0:
                        continue
                    h[i] -= d
                    h[j] += d
                    c2, _ = _c5c(h)
                    if c2 < bc - 1e-12:
                        bc = c2
                        imp = True
                    else:
                        h[i] += d
                        h[j] -= d
        if not imp:
            break
    return h, bc

def _slsqp(h, n, target, alphas, t0, tlim):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        a = alphas[0] if len(alphas) == 1 else _slsqp_cur_alpha[0]
        sh = np.clip(corr - m, -500.0 / a, 500.0 / a)
        return (m + np.log(np.sum(np.exp(a * sh))) / a) * dx
    def eqf(x):
        return np.sum(x) - target
    def eqj(x):
        return np.ones_like(x)
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    best = h.copy()
    bestc = _c5v(h)
    _slsqp_cur_alpha = [20.0]
    for al in alphas:
        if time.time() - t0 > tlim - 5:
            break
        _slsqp_cur_alpha[0] = al
        try:
            mi = 80 if n <= 80 else 50
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': mi, 'ftol': 1e-12})
            hn = _fix(res.x, target)
            cn = _c5v(hn)
            if cn < bestc:
                best = hn.copy()
                bestc = cn
        except Exception:
            pass
    return best, bestc

_slsqp_cur_alpha = [20.0]

def run(seed=42, budget_s=1100, **kwargs):

            if cf2 < best_c5:

            ht, cgf2 = _local_pairs(ht, 200, t0, tlim)

    # Final correction for exact sum
    h_ret = best_h.copy()
    tgt = best_n / 2.0
    diff = tgt - np.sum(h_ret)
    if abs(diff) > 1e-13:
        m = (h_ret > 1e-12) & (h_ret < 1.0 - 1e-12)
        cnt = int(np.sum(m))
        if cnt > 0:
            h_ret[m] += diff / cnt

    return h_ret, c5f, best_n
