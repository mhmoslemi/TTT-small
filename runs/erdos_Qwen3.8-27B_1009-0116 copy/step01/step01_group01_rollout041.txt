import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h


def obj_lse(h, t, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    lse = cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t
    penalty = mu * (h.sum() - n * 0.5) ** 2
    return (lse + penalty) * dx


def obj_focused(h, t, mu, boost_idx, boost_amt):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    c[boost_idx] += boost_amt
    cmax = np.max(c)
    lse = cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t
    penalty = mu * (h.sum() - n * 0.5) ** 2
    return (lse + penalty) * dx


def local_search(h, n, max_sweeps=30):
    h = h.copy()
    target = n * 0.5
    best = compute_c5(h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_h = h[i]
            candidates = [0.0, 1.0, 0.5, old_h * 0.5, 1.0 - old_h]
            candidates.append(max(0.0, min(1.0, old_h + 0.1)))
            candidates.append(max(0.0, min(1.0, old_h - 0.1)))
            best_val = best
            best_cand = old_h
            for cand in candidates:
                if abs(cand - old_h) < 1e-10:
                    continue
                h[i] = cand
                diff = target - h.sum()
                j = (i + 7) % n
                h[j] = np.clip(h[j] + diff, 0.0, 1.0)
                val = compute_c5(h)
                if val < best_val - 1e-12:
                    best_val = val
                    best_cand = cand
            if best_cand != old_h:
                h[i] = best_cand
                diff = target - h.sum()
                j = (i + 7) % n
                h[j] = np.clip(h[j] + diff, 0.0, 1.0)
                best = best_val
                improved = True
            else:
                h[i] = old_h
        if not improved:
            break
    return repair(h, n), best


def interpolate_h(h_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    h_dst = np.interp(xd, xs, h_src)
    return repair(h_dst, n_dst)
