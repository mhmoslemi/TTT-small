import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-10:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        r = d
        for i in o:
            if abs(r) < 1e-10:
                break
            if r > 0:
                a = min(r, 1.0 - h[i])
                h[i] += a; r -= a
            else:
                a = min(-r, h[i])
                h[i] -= a; r += a
    return h

def _soft_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter=400):
    t = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - t}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(_soft_obj, h, args=(alpha,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return _repair(res.x, n)

def _pair_ls(h, dx, n, rng, deadline):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    deltas = np.array([0.003, 0.007, 0.01, 0.02, 0.04, 0.07, 0.12, 0.2, 0.35])
    count = 0
    no_improve = 0
    while time.time() < deadline:
        d = rng.choice(deltas)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        ni, nj = hi + d, hj - d
        if ni > 1.0 or nj < 0.0:
            ni, nj = hi - d, hj + d
            if ni < 0.0 or nj > 1.0:
                continue
        best_h[i] = ni
        best_h[j] = nj
        c = _c5(best_h, dx)
        count += 1
        if c < best_c5 - 1e-12:
            best_c5 = c
            no_improve = 0
        else:
            best_h[i] = hi
            best_h[j] = hj
            no_improve += 1
        if no_improve > 80000:
            best_h += 0.08 * rng.randn(n)
            best_h = _repair(best_h, n)
            best_c5 = _c5(best_h, dx)
            no_improve = 0
    return best_h, best_c5

def _guided_ls(h, dx, n, rng, deadline):
    """Local search biased toward reducing the active max correlation."""
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    no_improve = 0
    while time.time() < deadline:
        corr = _corr(best_h)
        kmax = int(np.argmax(corr))
        # Gradient of corr[kmax] w.r.t. h: for each i, contribution
        # corr[kmax] = sum_m h[m]*(1-h[m+kmax-(n-1)])
        lag = kmax - (n - 1)
        grad = np.zeros(n)
        for m in range(n):
            val = 1.0
            ip = m + lag
            if 0 <= ip < n:
                val = 1.0 - best_h[ip]
            grad[m] += val
            iq = m - lag
            if 0 <= iq < n:
                grad[m] += best_h[iq]
        # Try to decrease h where grad is positive, increase where negative
        # Pick i with high grad (reduce h[i]) and j with low grad (increase h[j])
        hi_idx = np.argsort(grad)[-5:]
        lo_idx = np.argsort(grad)[:5]
        trials = 0
        for _ in range(3000):
            if time.time() > deadline:
                break
            i = rng.choice(hi_idx)
            j = rng.choice(lo_idx)
            if i == j:
                continue
            d = rng.uniform(0.005, 0.15)
            ni, nj = best_h[i] - d, best_h[j] + d
            if ni < -1e-12 or nj > 1.0 + 1e-12:
                continue
            best_h[i] = max(ni, 0.0)
            best_h[j] = min(nj, 1.0)
            c = _c5(best_h, dx)
            trials += 1
            if c < best_c5 - 1e-12:
                best_c5 = c
                no_improve = 0
            else:
                best_h[i], best_h[j] = best_h[i] + d, best_h[j] - d
                no_improve += 1
            if no_improve > 5000 and trials > 100:
                break
        if no_improve > 5000 and trials == 0:
            no_improve = 0
    return best_h, best_c5

def _optimize_n(n, rng, deadline, h_init=None):
    dx = 2.0 / n
    if h_init is not None:
        best_h = _repair(h_init.copy(), n)
    else:
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        rng.shuffle(best_h)
        best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    # SLSQP phases
    for alpha in [30, 80, 200, 500]:
        if time.time() > deadline:
            break
        try:
            hc = _slsqp(best_h, n, alpha, maxiter=300)
            c = _c5(hc, dx)
            if c < best_c5:
                best_c5, best_h = c, hc
        except Exception:
            pass
    # Local search
    dl = min(deadline, time.time() + (deadline - time.time()) * 0.5)
    best_h, best_c5 = _pair_ls(best_h, dx, n, rng, dl)
    # Guided search
    dl2 = min(deadline, time.time() + (deadline - time.time()) * 0.6)
    best_h, best_c5 = _guided_ls(best_h, dx, n, rng, dl2)
    return best_h, best_c5, n
