def softmax_gradient(h, dx, n, temp):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    w = np.exp(temp * (corr - cmax))
    ws = w.sum()

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def softmax_gradient(h, dx, n, temp):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    w = np.exp(temp * (corr - cmax))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-14:
            continue
        lag = ki - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            l = -lag
            grad[l:] += wk * (1.0 - h[:n - l])
            grad[:n - l] -= wk * h[l:]
    return dx * grad

def nesterov_opt(h0, n, dx, maxiter, temp_start, temp_end, lr0):
    h = h0.copy()
    h_prev = h.copy()
    mu = 1.0
    lr = lr0
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)
    for it in range(maxiter):
        frac = it / max(float(maxiter), 1.0)
        temp = temp_start * (temp_end / temp_start) ** frac
        if it > 0:
            mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
            y = h + ((mu / mu_new) - 1.0) * (h - h_prev)
            y = project_feasible(y, n)
            mu = mu_new
        else:
            y = h
        g = softmax_gradient(y, dx, n, temp)
        h_new = y - lr * g
        h_new = project_feasible(h_new, n)
        c5 = eval_c5(h_new, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
        h_prev = h
        h = h_new
        if it % 8000 == 0 and it > 0:
            lr *= 0.97
    return best_h, best_c5

def slsqp_opt(h, n, dx, maxiter=500):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return eval_c5(x, dx)
    def ceq(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def swap_local_search(h, n, dx, max_iter):
    h_s = h.copy()
    sorted_h = np.sort(h_s)[::-1]
    thresh = sorted_h[n // 2 - 1]
    h_s[h_s >= thresh] = 1.0
    h_s[h_s < thresh] = 0.0
    while h_s.sum() > n / 2:
        ones = np.where(h_s == 1.0)[0]
        h_s[np.random.choice(ones)] = 0.0
    while h_s.sum() < n / 2:
        zeros = np.where(h_s == 0.0)[0]
        h_s[np.random.choice(zeros)] = 1.0
    best_c5 = eval_c5(h_s, dx)
    best_h = h_s.copy()
    for _ in range(max_iter):
        ones = np.where(h_s == 1.0)[0]
        zeros = np.where(h_s == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_s[i], h_s[j] = h_s[j], h_s[i]
        c5 = eval_c5(h_s, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_s.copy()
        else:
            h_s[i], h_s[j] = h_s[j], h_s[i]
    return best_h, best_c5

def upsample(h_old, n_old, n_new):
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    x_old = np.arange(n_old) * dx_old
    x_new = np.arange(n_new) * dx_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def time_left(t0, budget_s):
    return budget_s - (time.time() - t0)
