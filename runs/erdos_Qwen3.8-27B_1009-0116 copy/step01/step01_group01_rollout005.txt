                # Recompute correlation
                c = compute_corr(h)
                best = float(np.max(c) * dx)
                improved = True
            # else: don't update, but c is still valid
            
        if not improved:
            break
    
    return h, best

c_k = Σ_j h[j] * (1 - h[j+k-lag])

- The term involving h[i] as first factor: h[i+δ] * (1-h[...]) = (h[i]+δ)(1-h[...]) = h[i](1-h[...]) + δ(1-h[...])

- Similarly for h[s] with -δ.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_descent_fast(h, n, max_sweeps=300, rng=None):
    h = h.astype(float).copy()
    dx = 2.0 / n
    c = compute_corr(h)
    best = float(np.max(c) * dx)
    if rng is None:
        rng = np.random.RandomState(42)
    lags = np.arange(0, 2 * n - 1) - (n - 1)
    for sweep in range(max_sweeps):
        improved = False
        order = rng.permutation(n)
        for i in order:
            s = (i + n // 2) % n
            if s == i:
                s = (i + 1) % n
            h_i, h_s = h[i], h[s]
            d_lo = max(-h_i, h_s - 1.0)
            d_hi = min(1.0 - h_i, h_s)
            if d_hi - d_lo < 1e-14:
                continue
            idx_ip, idx_im = i + lags, i - lags
            idx_sp, idx_sm = s + lags, s - lags
            mp, mm = (idx_ip >= 0) & (idx_ip < n), (idx_im >= 0) & (idx_im < n)
            sp, sm = (idx_sp >= 0) & (idx_sp < n), (idx_sm >= 0) & (idx_sm < n)
            m = np.zeros(2 * n - 1)
            m[mp] += 1.0 - h[idx_ip[mp]]
            m[mm] -= h[idx_im[mm]]
            m[sp] -= 1.0 - h[idx_sp[sp]]
            m[sm] += h[idx_sm[sm]]
            gr = 0.6180339887
            a, b = d_lo, d_hi
            c1, c2 = b - gr * (b - a), a + gr * (b - a)
            f1, f2 = np.max(c + c1 * m), np.max(c + c2 * m)
            for _ in range(20):
                if f1 < f2:
                    b, c2, f2 = c2, c1, f1
                    c1 = b - gr * (b - a)
                    f1 = np.max(c + c1 * m)
                else:
                    a, c1, f1 = c1, c2, f2
                    c2 = a + gr * (b - a)
                    f2 = np.max(c + c2 * m)
            d_star = (a + b) / 2
            new_max = float(np.max(c + d_star * m) * dx)
            if new_max < best - 1e-13:
                h[i] += d_star
                h[s] -= d_star
                h = np.clip(h, 0.0, 1.0)
                c = c + d_star * m
                best = new_max
                improved = True
        if not improved:
            break
    h = repair(h, n)
    best = compute_c5(h)
    return h, best

def subgradient_psm(h, n, steps=20000, rng=None):
    h = h.astype(float).copy()
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(123)
    lags = np.arange(0, 2 * n - 1) - (n - 1)
    c = compute_corr(h)
    best = float(np.max(c) * dx)
    best_h = h.copy()
    lb = 0.37
    for st in range(steps):
        cvals = c
        top_k = np.argsort(cvals)[-3:]
        weights = np.ones(len(top_k))
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            lag = k - (n - 1)
            idx_p, idx_m = np.arange(n) + lag, np.arange(n) - lag
            mp, mm = (idx_p >= 0) & (idx_p < n), (idx_m >= 0) & (idx_m < n)
            g = np.zeros(n)
            g[mp] += (1.0 - h[idx_p[mp]])
            g[mm] -= h[idx_m[mm]]
            grad += w * g
        cmax = float(np.max(c) * dx)
        step = (cmax - lb) / (np.dot(grad, grad) + 1e-12)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair(h_new, n)
        c_new = compute_corr(h_new)
        val = float(np.max(c_new) * dx)
        if val < best:
            best, best_h, h, c = val, h_new.copy(), h_new.copy(), c_new
        else:
            h, c = h_new.copy(), c_new
        lb = max(lb, best - 0.01)
    return best_h, best

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_ok(frac):
        return (budget_s * frac - elapsed()) > 3.0

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()

    if time_ok(0.3):

        if v1 < best_c5:

    # Phase 2: PSM at original resolution
    if time_ok(0.45):
        h2, v2 = subgradient_psm(best_h, n0, steps=15000, rng=rng)
        if v2 < best_c5:
            best_c5, best_h = v2, h2.copy()
        h3, v3 = coord_descent_fast(best_h, n0, max_sweeps=100, rng=rng)
        if v3 < best_c5:
            best_c5, best_h = v3, h3.copy()

    # Phase 3: Higher resolution
    for mult in [2, 4]:
        n = n0 * mult
        if n > 2000 or not time_ok(0.85):
            break
        xo = np.linspace(0.0, 2.0, len(best_h))
        xn = np.linspace(0.0, 2.0, n)
        h_init = np.interp(xn, xo, best_h)
        h_init = repair(h_init, n)
        hv, vv = coord_descent_fast(h_init, n, max_sweeps=150, rng=rng)
        if vv < best_c5:
            best_c5, best_h = vv, hv.copy()
        # Downsample if n > n0
        if n > n0 and len(best_h) == n:
            xd = np.linspace(0.0, 2.0, n0)
            hd = np.interp(xd, xn, best_h)
            hd = repair(hd, n0)
            vd = compute_c5(hd)
            if vd < best_c5:
                best_c5, best_h = vd, hd.copy()

    # Phase 4: SLSQP polish
    if time_ok(0.92) and len(best_h) <= 800:
        n = len(best_h)
        dx = 2.0 / n
        def obj_smooth(x, t=150.0):
            cv = np.correlate(x, 1.0 - x, mode='full')
            cm = np.max(cv)
            return float((cm + np.log(np.sum(np.exp((cv - cm) * t))) / t) * dx)
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                 'jac': lambda x: np.ones(n)}]
        bnds = [(0.0, 1.0)] * n
        t_rem = max(5.0, budget_s * 0.10 - elapsed())
        maxiter = max(50, int(t_rem / 0.4))
        res = minimize(obj_smooth, best_h.copy(), method='SLSQP',
                       bounds=bnds, constraints=cons,

            best_c5, best_h = v4, h4.copy()

    # Phase 5: Final CD polish

        h5, v5 = coord_descent_fast(best_h, n, max_sweeps=50, rng=rng)
        if v5 < best_c5:
            best_c5, best_h = v5, h5.copy()

    best_h = repair(best_h, len(best_h))
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, len(best_h))
