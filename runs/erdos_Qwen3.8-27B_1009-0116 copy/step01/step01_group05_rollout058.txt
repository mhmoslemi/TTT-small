import time

_N = 0
_EPS = 0.1

def _cx(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

from scipy.optimize import minimize
import time

_DX = 0.0
_N = 0
_EPS = 0.1
_TGT = 0.0

def _cx(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _cx(h, dx).max()

def _proj(h, tgt):
    lo, hi = -3.0, 4.0
    for _ in range(70):
        m = 0.5 * (lo + hi)
        if np.clip(h - m, 0, 1).sum() > tgt:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _gv(h, dx, n, eps):
    cv = _cx(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        wt = wi * dx
        a = abs(idx - (n - 1))
        if a == 0:
            g += wt * (1.0 - 2.0 * h)
        else:
            g[:n-a] += wt * (1.0 - h[a:])
            g[a:] -= wt * h[:n-a]
    return g

def _oi(x):
    cv = _cx(x, _DX)
    mx = cv.max()
    return mx + _EPS * np.log(np.sum(np.exp((cv - mx) / _EPS)))

def _gi(x):
    return _gv(x, _DX, _N, _EPS)

def _ef(x):
    return x.sum() - _TGT

def _ej(x):
    return np.ones(len(x))

def _slsqp(h0, dx, n, tgt, eps, mi=200):
    global _DX, _N, _EPS, _TGT
    _DX, _N, _EPS, _TGT = dx, n, float(eps), float(tgt)
    try:
        r = minimize(_oi, h0, jac=_gi,
                     bounds=[(0.0, 1.0)]*n,
                     constraints=[{'type':'eq','fun':_ef,'jac':_ej}],
                     method='SLSQP', options={'maxiter':mi,'ftol':1e-16})
        return _proj(r.x, tgt)
    except Exception:
        return h0.copy()

    mx = cv.max()
    act = np.where(cv >= mx - tol)[0]

        a = abs(idx - (n-1))

        else:
            g[:n-a] += dx * (1.0 - h[a:])
            g[a:] -= dx * h[:n-a]

    return _proj(h - lr * g, tgt)

                best = hn.copy()
    return best, bc

def _lp_step(h, dx, n, tgt, tol=1e-6, max_lp=100):
    cv = _cx(h, dx)
    C = cv.max()
    active = set(np.where(cv >= C - tol)[0])
    
    # For each active k, compute gradient of f_k at h
    grads = []
    rhs = []
    for idx in active:
        g_k = np.zeros(n)
        a = abs(idx - (n-1))
        if a == 0:
            g_k[:] = dx * (1.0 - 2.0 * h)
        else:
            g_k[:n-a] += dx * (1.0 - h[a:])
            g_k[a:] -= dx * h[:n-a]
        c_k = cv[idx]
        # Linearized: c_k + g_k · (h_new - h) <= t
        # => g_k · h_new <= t + c_k - g_k · h
        grads.append(g_k)
        rhs.append(c_k - np.dot(g_k, h))

    # s.t. g_k · h - t <= rhs_k for all k in active
    #      sum(h) = tgt
    #      0 <= h <= 1

    # min z[0]
    # s.t. A_in @ z <= b_in
    #      A_eq @ z = b_eq
    #      lb <= z <= ub
    
    m = len(grads)
    A = np.zeros((m, n+1))
    for i, g_k in enumerate(grads):
        A[i, 0] = -1  # -t
        A[i, 1:] = g_k  # g_k · h
    b_in = np.array(rhs)

    A_eq = np.zeros((1, n+1))

                  bounds=list(zip(lb, ub)), method='highs')
