```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        # If still not exact, adjust individual elements
        diff = target - np.sum(h)
        if abs(diff) > 1e-14:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def subgradient_descent(h, n, alpha, steps):
    """Projected subgradient descent for the max objective."""
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for step in range(steps):
        corr = compute_corr(h, n)
        c_max = np.max(corr) * dx
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        
        # Active set: lags within 5% of max
        threshold = c_max * 0.95
        active = np.where(corr * dx >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Compute subgradient: average gradient over active lags
        # For lag k in correlate 'full' output, index = k + (n-1)
        # corr[k + n - 1] = sum_i h[i] * (1-h[i+k]) for k >= 0
        # or sum_i h[i+k] * (1-h[i]) for k < 0
        
        grad = np.zeros(n)
        for lag_idx in active:
            k = lag_idx - (n - 1)  # actual lag
            # ∂/∂h[j] of sum_i h[i](1-h[i+k]) for k >= 0:
            # For each i where both i and i+k are in [0,n):
            #   ∂/∂h[j]: if j==i: (1-h[i+k]); if j==i+k: -h[i]
            # For k < 0, let k' = -k:
            #   sum_i h[i+k'](1-h[i])
            #   ∂/∂h[j]: if j==i: -h[i+k']; if j==i+k': (1-h[i])
            
            if k >= 0:
                # g_k = sum_{i=0}^{n-k-1} h[i](1-h[i+k])
                # ∂g_k/∂h[j] = (1-h[j+k]) for j < n-k (first term)
                #              - h[j-k] for j >= k (second term)
                for j in range(n - k):
                    if j + k < n:
                        grad[j] += (1 - h[j + k])
                for j in range(k, n):
                    grad[j] -= h[j - k]
            else:
                # k < 0, let kp = -k > 0
                # g_k = sum_{i=0}^{n-kp-1} h[i+kp](1-h[i])
                # ∂g_k/∂h[j] = -h[j-kp] for j >= kp (from -h[i] term where i=j)
                #              (1-h[j-kp]) for j-kp >= 0 (from h[i+kp] term where i+kp=j)
                kp = -k
                for j in range(kp, n):
                    grad[j] -= h[j - kp]
                for j in range(kp, n):
                    grad[j] += (1 - h[j - kp])
        
        # Average over active lags
        grad /= len(active)
        
        # Step
        h_new = h - alpha * grad
        
        # Project onto constraints
        h_new = np.clip(h_new, 0.0, 1.0)
        s = np.sum(h_new)
        if abs(s - target) > 1e-10:
            h_new = h_new * (target / s)
            h_new = np.clip(h_new, 0.0, 1.0)
            # Fix residual
            diff = target - np.sum(h_new)
            if abs(diff) > 1e-12:
                mask = (h_new > 0.01) & (h_new < 0.99)
                if np.sum(mask) > 0:
                    h_new[mask] += diff / np.sum(mask)
                    h_new = np.clip(h_new, 0.0, 1.0)
        
        h = h_new
    
    return best_h, best_c
```

```python
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        frac = diff / n
        h = h + frac
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_active(h, n, active_indices):
    """Compute average subgradient over active lags."""
    grad = np.zeros(n)
    n_full = 2 * n - 1
    for idx in active_indices:
        k = idx - (n - 1)
        if k >= 0:
            kk = min(k, n)
            if kk < n:
                grad[:n - kk] += (1.0 - h[kk:])
                grad[kk:] -= h[:n - kk]
        else:
            kk = min(-k, n)
            if kk < n:
                grad[kk:] += (1.0 - h[:n - kk])
                grad[:n - kk] -= h[kk:]
    grad /= max(len(active_indices), 1)
    return grad

def subgrad_step(h, n, alpha, frac_active=0.1):
    """One projected subgradient step."""
    corr = compute_corr(h, n)
    c_max = np.max(corr)
    threshold = c_max * (1.0 - frac_active)
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = grad_active(h, n, active)
    h_new = h - alpha * grad
    h_new = repair(h_new, n)
    return h_new, float(np.max(corr))

    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx
```

```python
    for idx in active_indices:
```

```python
    return np.correlate(h, 1.0 - h, mode='full')

    t = n / 2.0
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)

        h = np.clip(h + d / n, 0.0, 1.0)

    g = np.zeros(n)
    for idx in act:
        k = idx - (n - 1)
        if k >= 0:
            if 0 < k < n:
                g[:n-k] += 1.0 - h[k:]
                g[k:] -= h[:n-k]
        else:
            kk = -k
            if 0 < kk < n:
                g[kk:] += 1.0 - h[:n-kk]
                g[:n-kk] -= h[kk:]
    return g / max(len(act), 1)

    mx = cr.max()
    act = np.where(cr >= mx * (1.0 - frac))[0]
    g = _ggrad(h, n, act)
    return _repair(h - alpha * g, n), float(mx)

            h_b[i] = 0.0

        for i in idx[:target - ones]:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_gN = 70
_gB = 50

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    t = n / 2.0
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    return h

def _ggrad(h, n, act):
    g = np.zeros(n)
    for idx in act:
        k = idx - (n - 1)
        if k == 0:
            g += 1.0 - 2.0 * h
        elif 0 < k < n:
            g[:n-k] += 1.0 - h[k:]
            g[k:] -= h[:n-k]
        elif -n < k < 0:
            kp = -k
            g[kp:] += 1.0 - h[:n-kp]
            g[:n-kp] -= h[kp:]
    return g / max(len(act), 1)

def _sgrad(h, n, alpha, frac):
    cr = np.correlate(h, 1.0 - h, mode='full')
    mx = cr.max()
    act = np.where(cr >= mx * (1.0 - frac))[0]
    if len(act) == 0:
        act = np.array([np.argmax(cr)])
    g = _ggrad(h, n, act)
    return _repair(h - alpha * g, n)

def _lse(x):
    n, b = _gN, _gB
    dx = 2.0 / n
    cr = np.correlate(x, 1.0 - x, mode='full')
    s = b * cr
    m = s.max()
    return (m + np.log(np.sum(np.exp(s - m)))) / b * dx

def _eqf(x):
    return x.sum() - _gN / 2.0

def _eqj(x):
    return np.ones(_gN)

def _init_h(n, rng):
    h = np.where(np.arange(n) % 2 == 0, 0.8, 0.2).astype(float)
    h += rng.normal(0, 0.15, n)
    return _repair(h, n)

def _binary_local(h, n, rng, iters, tlimit):
    hb = np.where(h > 0.5, 1.0, 0.0)
    ones = int(hb.sum())
    tgt = n // 2
    if ones > tgt:
        idx1 = np.where(hb == 1.0)[0]
        for i in idx1[:ones - tgt]:
            hb[i] = 0.0
    elif ones < tgt:
        idx0 = np.where(hb == 0.0)[0]
        for i in idx0[:tgt - ones]:
            hb[i] = 1.0
    best = _c5(hb, n)
    best_hb = hb.copy()
    for _ in range(iters):
        if time.time() > tlimit:
            break
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if hb[i] == hb[j]:
            continue
        hb[i], hb[j] = hb[j], hb[i]
        c = _c5(hb, n)
        if c < best:
            best = c
            best_hb = hb.copy()
        else:
            hb[i], hb[j] = hb[j], hb[i]
    return best_hb, best

def _slsqp(h, n, betas, maxiter, tleft):
    global _gN, _gB
    _gN = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eqf, 'jac': _eqj}]
    for b in betas:
        if time.time() - tleft > 1:
            break
        _gB = b
        try:
            r = minimize(_lse, h, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h = _repair(r.x, n)
        except Exception:
            pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _gN, _gB
    t0 = time.time()
    rng = np.random.RandomState(seed)
    tleft_check = t0 + budget_s

    init = globals().get('initial_h_values', None)
    best_h, best_c5 = None, np.inf

    n_list = []
    if init is not None:
        n_list.append(len(init))
    for nv in [70, 100, 50, 140]:
        if nv not in n_list:
            n_list.append(nv)
    n_list.sort()

    for n in n_list:
        if time.time() > tleft_check - 120:
            break
        _gN = n

        if init is not None and n == len(init):
            h = _repair(np.array(init, dtype=float).copy(), n)
        else:
            h = _init_h(n, rng)

        c = _c5(h, n)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

        # Subgradient phase
        alpha, stall, lb = 0.06, 0, c
        phase_end = tleft_check - 120
        for step in range(10000):
            if time.time() > phase_end:
                break
            h_new = _sgrad(h, n, alpha, 0.05)
            c_new = _c5(h_new, n)
            if c_new < lb - 1e-12:
                h, lb, stall = h_new, c_new, 0
                if c_new < best_c5:
                    best_c5, best_h = c_new, h.copy()
            elif rng.random() < np.exp(-max(0.0, c_new - lb) / 0.003):
                h, stall = h_new, stall + 1
            else:
                stall += 1
            if stall > 120:
                alpha *= 0.93
                stall = 0
            if stall > 500:
                h = _repair(h + rng.normal(0, 0.3, n), n)
                alpha, stall = 0.06, 0
                lb = _c5(h, n)
                if lb < best_c5:
                    best_c5, best_h = lb, h.copy()
                break

        # SLSQP polish on log-sum-exp
        if time.time() < tleft_check - 50:
            h = _slsqp(h, n, [30, 80, 200, 500], 300, tleft_check - 20)
            c2 = _c5(h, n)
            if c2 < best_c5:
                best_c5, best_h = c2, h.copy()

        # Binary local search (only for even n)
        if time.time() < tleft_check - 35 and n % 2 == 0:
            hb, cb = _binary_local(h, n, rng, 4000, tleft_check - 30)
            if cb < best_c5:
                best_c5, best_h = cb, hb.copy()
            if cb < c2:
                h = _slsqp(hb, n, [50, 150], 200, tleft_check - 15)
                c3 = _c5(h, n)
                if c3 < best_c5:
                    best_c5, best_h = c3, h.copy()

        # Coordinate moves
        if time.time() < tleft_check - 20:
            for _ in range(4000):
                if time.time() > tleft_check - 20:
                    break
                i, j = int(rng.randint(n)), int(rng.randint(n))
                if i == j:
                    continue
                for eps in (0.02, 0.05, 0.1, 0.15):
                    hp = h.copy()
                    hp[i] += eps
                    hp[j] -= eps
                    if 0.0 <= hp[i] <= 1.0 and 0.0 <= hp[j] <= 1.0:
                        c4 = _c5(hp, n)
                        if c4 < best_c5 - 1e-13:
                            best_c5, best_h, h = c4, hp.copy(), hp
                            break

    # Final deep SLSQP
    if time.time() < tleft_check - 3 and best_h is not None:
        nf = len(best_h)
        best_h = _slsqp(best_h, nf, [200, 500, 1000], 400, tleft_check)
        best_c5 = _c5(best_h, nf)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5(best_h, len(best_h))
    return (best_h, best_c5, len(best_h))
```
