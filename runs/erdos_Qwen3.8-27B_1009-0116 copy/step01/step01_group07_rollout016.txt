import numpy as np
from scipy.optimize import minimize, linprog
import time

_dx_g = 1.0
_alpha_g = 100.0
_target_g = 35.0

def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / len(h))

def _norm(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    cnt = int(np.sum(interior))
    if cnt > 0:
        h[interior] += diff / cnt
    else:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def _obj_sm(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    lo = -50.0 / _alpha_g
    sh = np.clip(c - m, lo, -lo)
    return (m + np.log(np.sum(np.exp(_alpha_g * sh))) / _alpha_g) * _dx_g

def _obj_pen(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    sh = np.clip(c - m, -0.5, 0.5)
    sm = m + np.log(np.sum(np.exp(3000.0 * sh))) / 3000.0
    return sm * _dx_g + 1000.0 * (np.sum(x) - _target_g) ** 2

def _eq_fun(x):
    return np.sum(x) - _target_g

def _eq_jac(x):
    return np.ones_like(x)

def _sine_init(n, target, rng, ns=3, amax=0.3):
    h = np.full(n, 0.5)
    for _ in range(ns):
        f = rng.randint(1, max(2, n // 4))
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.05, amax)
        h += a * np.sin(2 * np.pi * f * np.arange(n) / n + p)
    return _norm(h, target)

def _subgrad_step(h, target, eta=0.004):
    n = len(h)
    c_full = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c_full))
    m = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        jm = j + m
        if 0 <= jm < n:
            grad[j] += (1.0 - h[jm])
        j2 = j - m
        if 0 <= j2 < n:
            grad[j] += (-h[j2])
    grad -= np.mean(grad)
    h_new = h - eta * grad
    return _project(h_new, target)

def _coord_desc(h, target, n_iter, seed):
    n = len(h)
    bc = _c5(h)
    rng_c = np.random.RandomState(seed)
    for _ in range(n_iter):
        improved = False
        for i in rng_c.permutation(n):
            found = False
            for _t in range(5):
                j = rng_c.randint(0, n)
                if j == i:
                    continue
                d = rng_c.uniform(-0.06, 0.06)
                if abs(d) < 0.004:
                    continue
                h2 = h.copy()
                h2[i] += d
                h2[j] -= d
                if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                    c = _c5(h2)
                    if c < bc - 1e-12:
                        h = h2
                        bc = c
                        improved = True
                        found = True
                        break
            if found:
                pass
        if not improved:
            break
    return h, bc

def _lp_cut(h0, n, target, max_iter=10):
    h = h0.copy()
    best_c = _c5(h)
    for _ in range(max_iter):
        c_all = np.correlate(h, 1.0 - h, mode='full')
        grads = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            m = k - (n - 1)
            for j in range(n):
                g = 0.0
                if 0 <= j + m < n:
                    g += (1.0 - h[j + m])
                if 0 <= j - m < n:
                    g += (-h[j - m])
                grads[k, j] = g
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in range(2 * n - 1):
            row = np.zeros(n + 1)
            row[:n] = grads[k]
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(float(c_all[k] - np.dot(grads[k], h)))
        for i in range(n):
            r1 = np.zeros(n + 1); r1[i] = 1.0
            r2 = np.zeros(n + 1); r2[i] = -1.0
            A_rows.extend([r1, r2])
            b_rows.extend([1.0, 0.0])
        A_ineq = np.array(A_rows)
        b_ineq = np.array(b_rows)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bl = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
        try:
            res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                          bounds=bl, method='highs', options={'presolve': True})
            if res.status == 0:
                h_new = np.clip(res.x[:n], 0.0, 1.0)
                h_new = _project(h_new, target)
                c_new = _c5(h_new)
                if c_new < best_c:
                    h = h_new
                    best_c = c_new
                else:
                    h = 0.7 * h + 0.3 * h_new
                    h = _project(h, target)
        except Exception:
            break
    return h, _c5(h)

def _do_slsqp(h0, n, target, alpha, maxiter=80):
    global _dx_g, _alpha_g, _target_g
    _dx_g = 2.0 / n
    _alpha_g = alpha
    _target_g = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    res = minimize(_obj_sm, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    return _project(res.x, target)

def _do_lbfgs(h0, n, target, maxiter=300):
    global _dx_g, _alpha_g, _target_g
    _dx_g = 2.0 / n
    _target_g = target
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_pen, h0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return _project(res.x, target)

def _try_n(best_h, best_n, new_n, target_new, rng):
    n_old = len(best_h)
    if new_n == n_old:
        return best_h.copy()

    t_new = np.linspace(0, 2, new_n, endpoint=False) + (2.0/new_n)/2
    h_interp = np.interp(t_new, t_old, best_h, period=2.0)
    h_interp = _norm(h_interp, target_new)
    return h_interp
