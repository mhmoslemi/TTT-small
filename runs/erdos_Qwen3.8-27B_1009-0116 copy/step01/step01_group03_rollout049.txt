def compute_c5(h):

    dx = 2.0 / n

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-6)[0]
    else:

        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta)
