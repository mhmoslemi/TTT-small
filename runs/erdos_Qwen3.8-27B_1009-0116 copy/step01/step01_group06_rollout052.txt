    elif r == 3:
        h = 0.5 + 0.25*np.sin(2*np.pi*3*np.arange(n)/n) + 0.15*np.cos(2*np.pi*5*np.arange(n)/n) + 0.05*rng.randn(n)
    elif r == 4:
        h = rng.rand(n)
    elif r == 5:
        h = np.linspace(0.15, 0.85, n) + 0.1 * rng.randn(n)
    elif r == 6:
        h = 0.5 + 0.4*np.cos(2*np.pi*np.arange(n)/(n//3)) + 0.05*rng.randn(n)
    else:
        h = np.tile([0.8, 0.2, 0.6, 0.4], n // 4 + 1)[:n] + 0.05 * rng.randn(n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    n = 70
    dx = 2.0 / n

    best_h = project_feasible(np.array(initial_h_values, dtype=float), n)
    best_c5 = eval_c5(best_h, dx)

    # Phase 1: L-BFGS-B log-sum-exp anneal from parent
    if budget_s - (time.time() - t0) > 150:
        h, c = lbfgs_lse(best_h, n, dx, budget_s - (time.time() - t0) - 80)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 2: SLSQP exact-constraint polish
    if time.time() - t0 < budget_s - 80:
        h, c = slsqp_opt(best_h, n, dx, maxiter=1200)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: Fine L-BFGS-B with low tau
    if time.time() - t0 < budget_s - 60:
        h, c = lbfgs_lse(best_h, n, dx, budget_s - (time.time() - t0) - 40,
                         taus=[1.2, 1.0, 0.8, 0.6], pen=500.0)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 4: Random restarts
    attempt = 0
    while time.time() - t0 < budget_s - 8:
        attempt += 1
        rem = budget_s - (time.time() - t0) - 8
        if rem < 12:
            break
        m = attempt % 6
        if m == 0:
            h0 = struct_init(n, rng)
        elif m == 1:
            h0 = best_h + 0.06 * rng.randn(n)
        elif m == 2:
            h0 = best_h + 0.02 * rng.randn(n)
        elif m == 3:
            h0 = best_h.copy()
            i, j = rng.choice(n, 2, replace=False)
            d = rng.uniform(-0.25, 0.25)
            h0[i] += d
            h0[j] -= d
        elif m == 4:
            h0 = struct_init(n, rng)
            h0 = 0.5 * h0 + 0.5 * best_h
        else:
            h0 = best_h + 0.12 * rng.randn(n)
        h0 = project_feasible(h0, n)

        tau_pick = rng.choice([5.0, 3.0, 2.0, 1.5])
        try:
            res = minimize(lse_obj, h0, args=(n, dx, tau_pick, 300.0),
                           jac=lse_jac, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 250, 'gtol': 1e-9, 'factr': 1e10})
        except Exception:
            continue
        ho = project_feasible(res.x, n)
        c = eval_c5(ho, dx)
        if c < best_c5:
            best_h, best_c5 = ho, c
            if time.time() - t0 < budget_s - 35:
                h2, c2 = slsqp_opt(ho, n, dx, maxiter=400)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2
                    if time.time() - t0 < budget_s - 20:
                        h3, c3 = lbfgs_lse(ho, n, dx, budget_s - (time.time()-t0) - 15,
                                            taus=[1.0, 0.8], pen=500.0)
                        if c3 < best_c5:
                            best_h, best_c5 = h3, c3

    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
