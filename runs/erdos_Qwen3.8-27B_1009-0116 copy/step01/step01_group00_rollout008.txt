import numpy as np
from scipy.optimize import minimize
import time


def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _smooth_grad(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        if w[k_idx] < 1e-15:
            continue
        wk = w[k_idx]
        lag = k_idx - (n - 1)
        for j in range(n):
            ip = j + lag
            if 0 <= ip < n:
                grad[j] += wk * (1.0 - x[j])
                grad[ip] += -wk * x[j]
    return grad


def _project_feasible(x, target_sum):
    x = x.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - proj)
        else:
            order = np.argsort(proj)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - proj[idx])
                proj[idx] += add
                diff -= add
            else:
                sub = min(-diff, proj[idx])
                proj[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return proj


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, target / n)
    return _project_feasible(h, target)


def _pgd_optimize(h, n, alpha, max_iter, t_start, budget_s):
    target = n / 2.0
    lr = 0.02
    best_x = h.copy()
    best_obj = _smooth_obj(h, alpha)
    prev_obj = best_obj
    stall = 0
    for it in range(max_iter):
        if time.time() - t_start > budget_s - 25:
            break
        g = _smooth_grad(h, alpha)
        f0 = _smooth_obj(h, alpha)
        step = lr
        best_step_x = h.copy()
        best_step_f = f0
        for _ls in range(8):
            x_new = h - step * g
            x_new = _project_feasible(x_new, target)
            f_new = _smooth_obj(x_new, alpha)
            if f_new < f0 - 1e-15:
                best_step_x = x_new
                best_step_f = f_new
                break
            if f_new < best_step_f:
                best_step_x = x_new
                best_step_f = f_new
            step *= 0.5
        h = best_step_x
        if h is best_step_x and best_step_f >= f0 - 1e-15:
            stall += 1
            if stall > 30:
                break
        else:
            stall = max(0, stall - 1)
        if best_step_f < best_obj - 1e-14:
            best_obj = best_step_f
            best_x = h.copy()
        lr *= 0.995
    return best_x


def _coord_search(h, n, dx, t_start, budget_s, rng, max_pass=800):
    best_local = _compute_c5(h, dx)
    deltas = np.array([0.002, 0.005, 0.01, 0.015, 0.03, 0.05,
                       -0.002, -0.005, -0.01, -0.015, -0.03, -0.05])
    n_pass = 0
    no_improve = 0
    while n_pass < max_pass and time.time() - t_start < budget_s - 10:
        improved = False
        i = rng.randint(n)
        js = rng.choice(n, size=min(n, 50), replace=False)
        for j in js:
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in deltas:
                ni, nj = hi + d, hj - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                h[i], h[j] = ni, nj
                c5 = _compute_c5(h, dx)
                if c5 < best_local - 5e-10:
                    best_local = c5
                    improved = True
                else:
                    h[i], h[j] = hi, hj
                if time.time() - t_start > budget_s - 10:
                    break
        n_pass += 1
        if not improved:
            no_improve += 1
            if no_improve > 80:
                break
        else:
            no_improve = 0
    if best_local < _compute_c5(h, dx) + 1e-12:
        h_dup = h.copy()
        h_dup = _repair(h_dup, n)
        c5_check = _compute_c5(h_dup, dx)
        if c5_check < best_local + 1e-12:
            best_local = c5_check
            h = h_dup
    return h, best_local
