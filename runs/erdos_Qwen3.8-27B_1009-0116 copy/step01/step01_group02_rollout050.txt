def repair(h, n):

            hf = repair(r.x, n)
            cf = compute_c5(hf, n)
            if cf < best_c5:
                h_best, best_c5 = hf, cf
        except Exception:
            pass

    # Tiny final coordinate sweep
    if time.time() < t0 + budget_s - 1:
        for _ in range(200):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for eps in (0.002, 0.005, 0.01):
                hp = h_best.copy()
                hp[i] += eps
                hp[j] -= eps
                if hp[i] > 1.0 or hp[j] < 0.0:
                    continue
                c = compute_c5(hp, n)
                if c < best_c5 - 1e-15:
                    h_best, best_c5 = hp, c
                    break

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
