from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
        space = 1.0 - h[idx]
    else:
        idx = np.where(h > 1e-10)[0]
        space = h[idx]
    if len(idx) == 0:
        return h
    # Distribute proportionally
    total_space = space.sum()
    if total_space < abs(diff):
        h[idx] += np.sign(diff) * space
        diff = 0
    else:
        h[idx] += diff * space / total_space
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        for i in idx:
            adj = np.clip(diff2, -(h[i]), 1.0-h[i])
            h[i] += adj
            diff2 -= adj
            if abs(diff2) < 1e-12:
                break
    return h

def smooth_objective(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = np.clip((corr - m) * beta, -700, 700)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def slsqp_optimize(h, ts, beta, maxiter=500, ftol=1e-12):
    n = len(h)
    res = minimize(
        lambda x: smooth_objective(x, beta),
        h, method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project(res.x, ts)
    return project(h, ts)

    n = len(h)
    t0 = time.time()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(12345)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, ts):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        for i in idx:
            if diff2 > 0:
                adj = min(diff2, 1.0 - h[i])
            else:
                adj = max(diff2, -h[i])
            h[i] += adj
            diff2 -= adj
            if abs(diff2) < 1e-12:
                break
    return h

def smooth_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = np.clip((corr - m) * beta, -700, 700)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

    res = minimize(
        lambda x: smooth_obj(x, beta),
        h, method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project(res.x, ts)
    return project(h, ts)
