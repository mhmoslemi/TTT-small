        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s

        if abs(d2) > 1e-14:

    return h

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    # log-sum-exp
    shifted = beta * (corr - m)
    return (m + np.log(np.sum(np.exp(shifted))) / beta)

def lse_gradient(h, dx, beta):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = beta * (corr - m)
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        wk = w[k_idx]
        # contributions
        for m_idx in range(n):
            if 0 <= m_idx + lag < n:
                grad[m_idx] += wk * (1.0 - h[m_idx + lag])
            if 0 <= m_idx - lag < n:
                grad[m_idx] -= wk * h[m_idx - lag]

_target_sum = 35.0
_n_pts = 70
_de_deadline = 0.0

def eval_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * _dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            if d2 > 0:
                free = np.where(h < 1.0 - 1e-15)[0]
                if len(free) > 0:
                    h[free[0]] = min(1.0, h[free[0]] + d2)
            else:
                free = np.where(h > 1e-15)[0]
                if len(free) > 0:
                    h[free[0]] = max(0.0, h[free[0]] + d2)
    return h

def _cfun(x):
    return x.sum() - _target_sum

def _cjac(x):
    return np.ones(_n_pts)

def c5_obj(x):
    return eval_c5(x)

def lse_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full') * _dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(_beta * (corr - m)))) / _beta

def lse_grad(x):
    n = _n_pts
    corr = np.correlate(x, 1.0 - x, mode='full') * _dx
    m = np.max(corr)
    w = np.exp(_beta * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-12:
            continue
        L = k_idx - (n - 1)
        wk = w[k_idx]
        for i in range(n):
            if 0 <= i + L < n:
                grad[i] += wk * (1.0 - x[i + L])
            if 0 <= i - L < n:
                grad[i] -= wk * x[i - L]
    return grad

def de_obj(x):
    return eval_c5(project_feasible(x, _n_pts))

def de_cb(xk, convergence):
    return time.time() > _de_deadline

def do_slsqp(h, maxiter):
    global _n_pts, _target_sum, _dx
    bounds = [(0.0, 1.0)] * _n_pts
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        res = minimize(c5_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, _n_pts)
    except Exception:
        return h

def do_lse(h, betas, maxper, deadline):
    global _beta, _n_pts, _target_sum, _dx
    bounds = [(0.0, 1.0)] * _n_pts
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    for b in betas:
        if time.time() > deadline:
            break
        _beta = b
        try:
            res = minimize(lse_obj, h, jac=lse_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxper, 'ftol': 1e-12})
            h = project_feasible(res.x, _n_pts)
        except Exception:
            pass
    return h

def structured_start(n, rng):
    t = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    h[::2] = 1.0
    jitter = 0.15 * np.random.RandomState(123).randn(n)
    h += jitter
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _dx, _beta, _target_sum, _n_pts, _de_deadline
    np.random.seed(seed)
    t0 = time.time()

    _n_pts = 70
    _dx = 2.0 / _n_pts
    _target_sum = float(_n_pts) / 2.0

    try:
        h_init = np.array(initial_h_values, dtype=float)
        if len(h_init) != _n_pts:
            h_init = np.random.rand(_n_pts)
    except Exception:
        h_init = np.random.rand(_n_pts)

    best_h = project_feasible(h_init, _n_pts)
    best_c5 = eval_c5(best_h)

    # Phase 1: SLSQP quick refinement
    if time.time() - t0 < budget_s - 350:
        best_h = do_slsqp(best_h, 500)
        best_c5 = eval_c5(best_h)

    # Phase 2: LSE smoothing (primary refinement)
    if time.time() - t0 < budget_s - 250:
        h = do_lse(best_h, [20, 50, 100, 200, 500, 1000, 2000], 400, t0 + budget_s * 0.48)
        c = eval_c5(h)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: DE for global exploration
    _de_deadline = t0 + budget_s * 0.72
    if time.time() - t0 < budget_s - 180:
        bounds_de = [(0.0, 1.0)] * _n_pts
        try:
            de_res = differential_evolution(de_obj, bounds_de, seed=seed + 7,
                maxiter=200, popsize=14, tol=1e-6,
                mutation=(0.3, 1.0), recombination=0.8,
                updating='deferred', polish=False, workers=1, callback=de_cb)
            h = project_feasible(de_res.x, _n_pts)
            c = eval_c5(h)
            if c < best_c5:
                best_h, best_c5 = h, c
        except Exception:
            pass

    # Phase 4: LSE with very high beta (nearly exact max)
    if time.time() - t0 < budget_s - 80:
        h = do_lse(best_h, [1000, 2000, 5000, 10000], 600, t0 + budget_s * 0.88)
        c = eval_c5(h)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 5: SLSQP final polish
    if time.time() - t0 < budget_s - 15:
        best_h = do_slsqp(best_h, 3000)
        best_c5 = eval_c5(best_h)

    # Phase 6: LSE restarts from perturbed best
    while time.time() - t0 < budget_s - 8:
        scale = 0.02 + 0.03 * (1.0 - (time.time() - t0) / budget_s)
        h0 = best_h + scale * np.random.randn(_n_pts)
        h0 = project_feasible(h0, _n_pts)
        h = do_lse(h0, [500, 1500, 5000], 250, t0 + budget_s - 10)
        c = eval_c5(h)
        if c < best_c5:
            best_h, best_c5 = h, c
        else:
            h0 = 0.5 * best_h + 0.5 * np.random.rand(_n_pts)
            h0 = project_feasible(h0, _n_pts)
            h = do_lse(h0, [100, 500, 2000], 200, t0 + budget_s - 10)
            c = eval_c5(h)
            if c < best_c5:
                best_h, best_c5 = h, c
            else:
                break

    best_h = project_feasible(best_h, _n_pts)
    best_c5 = eval_c5(best_h)
    return (best_h, best_c5, _n_pts)
