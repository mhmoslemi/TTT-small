            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad

                best_h = ht.copy()
                h = ht
                break
            lr *= 0.5
            if lr < 1e-14:
                break

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-10:
            if d2 > 0:
                idx = int(np.argmax(1.0 - h))
                h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
            else:
                idx = int(np.argmin(h))
                h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def compute_grad(h, dx, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -50.0, 50.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    p = exp_vals / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
