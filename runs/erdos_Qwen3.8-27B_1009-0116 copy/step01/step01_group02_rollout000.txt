import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_values(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def subgradient_step(h, n, k_star, lr):
    grad = np.zeros(n)
    k = abs(k_star)
    if k >= n:
        k = 0
    if k == 0:
        for j in range(n):
            grad[j] = (1.0 - h[j]) - 0.0
        grad -= h
    elif k_star >= 0:
        for j in range(k, n):
            grad[j] += (1.0 - h[j - k])
        for j in range(0, n - k):
            grad[j] -= h[j + k]
    else:
        for j in range(0, n - k):
            grad[j] += (1.0 - h[j + k])
        for j in range(k, n):
            grad[j] -= h[j - k]
    h_new = h - lr * grad
    return repair(h_new, n)

def binary_project(h, n):
    target = n / 2.0
    floor_t = int(np.floor(target))
    frac = target - floor_t
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:floor_t]] = 1.0
    if frac > 0 and floor_t < n:
        h_bin[sorted_idx[floor_t]] = frac
    return h_bin

def make_weighted_obj(n, weights):
    def wobj(x):
        dx_local = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.sum(weights * corr)) * dx_local
    return wobj

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = globals().get('initial_h_values', None)
    if init is not None:
        n = len(init)
        h_best = np.array(init, dtype=float).copy()
    else:
        n = 70
        h_best = np.full(n, 0.5)

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    n_corr = 2 * n - 1

    def eq_con(x):
        return np.sum(x) - target_sum
    cons = [{'type': 'eq', 'fun': eq_con}]

    # Phase 1: Multiplicative weights + SLSQP (main loop)
    weights = np.ones(n_corr) / n_corr
    it = 0
    stall = 0
    max_it = 400

    while time.time() - t0 < budget_s - 60 and it < max_it:
        it += 1
        cvals = lag_values(h_best, n)
        max_c = float(np.max(cvals))
        if max_c > 1e-12:
            weights = np.maximum(cvals, 1e-8) ** 2
            sw = float(np.sum(weights))
            if sw > 1e-15:
                weights = weights / sw
            else:
                weights = np.ones(n_corr) / n_corr
        else:
            weights = np.ones(n_corr) / n_corr

        # SLSQP on weighted objective
        wobj = make_weighted_obj(n, weights)
        improved = False
        try:
            r = minimize(wobj, h_best, method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': 200, 'ftol': 1e-13})
            hc = repair(r.x, n)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-14:
                h_best, best_c5, improved = hc, c, True
        except Exception:
            pass

        # Subgradient on worst lag (every 4 iters)
        if not improved and it % 4 == 0 and time.time() - t0 < budget_s - 60:
            for _ in range(40):
                cvals = lag_values(h_best, n)
                k_worst = int(np.argmax(cvals))
                k_star = k_worst - (n - 1)
                lr = 0.5 / (1.0 + 0.1 * it)
                h_sg = subgradient_step(h_best, n, k_star, lr)
                c_sg = compute_c5(h_sg, n)
                if c_sg < best_c5 - 1e-14:
                    h_best, best_c5 = h_sg, c_sg
                    improved = True

        # Binary projection + re-optimize (every 7 iters)
        if it % 7 == 0 and time.time() - t0 < budget_s - 80:
            h_bin = binary_project(h_best, n)
            try:
                r2 = minimize(wobj, h_bin, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 200, 'ftol': 1e-13})
                hc2 = repair(r2.x, n)
                c2 = compute_c5(hc2, n)
                if c2 < best_c5 - 1e-14:
                    h_best, best_c5 = hc2, c2
                    improved = True
            except Exception:
                pass

        # Perturbation restart when stalled
        if stall > 10 and time.time() - t0 < budget_s - 80:
            sigma = 0.05 + 0.02 * min(stall - 10, 15)
            h_pert = repair(h_best + rng.normal(0, sigma, n), n)
            try:
                r3 = minimize(wobj, h_pert, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 300, 'ftol': 1e-13})
                hc3 = repair(r3.x, n)
                c3 = compute_c5(hc3, n)
                if c3 < best_c5 - 1e-14:
                    h_best, best_c5 = hc3, c3
                    improved = True
                    stall = -5
            except Exception:
                pass

        # Nelder-Mead derivative-free (every 12 iters when stalled)
        if stall > 15 and it % 12 == 0 and time.time() - t0 < budget_s - 80:
            h_nm = repair(h_best + rng.normal(0, 0.03, n), n)
            try:
                r4 = minimize(lambda x: compute_c5(x, n), h_nm, method='Nelder-Mead',
                            bounds=bounds,
                            options={'maxiter': 500, 'xatol': 1e-9, 'fatol': 1e-12})
                hc4 = repair(np.clip(r4.x, 0, 1), n)
                c4 = compute_c5(hc4, n)
                if c4 < best_c5 - 1e-14:
                    h_best, best_c5 = hc4, c4
                    improved = True
                    stall = -5
            except Exception:
                pass

        if improved:
            stall = 0
        else:
            stall += 1

    # Phase 2: Direct max objective polish
    if time.time() - t0 < budget_s - 20:
        try:
            r_f = minimize(lambda x: compute_c5(x, n), h_best, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-15})
            hf = repair(r_f.x, n)
            cf = compute_c5(hf, n)
            if cf < best_c5:
                h_best, best_c5 = hf, cf
        except Exception:
            pass

    # Phase 3: Fine coordinate moves
    if time.time() - t0 < budget_s - 5:
        for _ in range(500):
            if time.time() - t0 > budget_s - 3:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for eps in (0.003, 0.007, 0.015, 0.04):
                hp = h_best.copy()
                hp[i] += eps
                hp[j] -= eps
                if hp[i] > 1.0 or hp[j] < 0.0:
                    continue
                hp = repair(hp, n)
                c = compute_c5(hp, n)
                if c < best_c5 - 1e-14:
                    h_best, best_c5 = hp, c
                    break

    # Phase 4: One more SLSQP polish
    if time.time() - t0 < budget_s - 3:
        try:
            r_p = minimize(lambda x: compute_c5(x, n), h_best, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-15})
            hp2 = repair(r_p.x, n)
            cp2 = compute_c5(hp2, n)
            if cp2 < best_c5:
                h_best, best_c5 = hp2, cp2
        except Exception:
            pass

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
