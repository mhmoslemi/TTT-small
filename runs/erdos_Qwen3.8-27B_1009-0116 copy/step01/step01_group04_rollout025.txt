from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)
    # Fix sum exactly
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-10:
        # Distribute diff evenly where possible
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
    # Final exact fix
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-10:
        idx = np.argmax(x < 1.0) if diff > 0 else np.argmin(x > 0.0)
        x[idx] += diff
        x = np.clip(x, 0.0, 1.0)
    return x

def smooth_max_obj(h, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / tau, -50, 0)
    return m + tau * np.log(np.sum(np.exp(shifted)))

def slsqp_optimize(x0, n, dx, tau, maxiter=300):
    def obj(x):
        return smooth_max_obj(x, tau)
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(res.x, n)
    except:
        return x0.copy()

def binary_local_search(h, n, dx, max_sweeps=50, time_limit=10):
    """Local search on binary solutions using swap moves."""
    t0 = time.time()
    # Round to binary
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    # Fix sum
    s = np.sum(h_bin)
    tgt = n / 2.0
    if s > tgt:
        # Turn off some 1s
        idx_ones = np.where(h_bin == 1.0)[0]
        rng = np.random.default_rng(42)
        rng.shuffle(idx_ones)
        for i in range(int(s - tgt)):
            h_bin[idx_ones[i]] = 0.0
    elif s < tgt:
        idx_zeros = np.where(h_bin == 0.0)[0]
        rng = np.random.default_rng(42)
        rng.shuffle(idx_zeros)
        for i in range(int(tgt - s)):
            h_bin[idx_zeros[i]] = 1.0
    
    best = h_bin.copy()
    best_val = compute_c5(best, dx)
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        # Try random subset of swaps
        n_tries = min(500, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            val = compute_c5(h_bin, dx)
            if val < best_val - 1e-12:
                best_val = val
                best = h_bin.copy()
                improved = True
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        if not improved:

            any_improve = False
            for i in ones:
                for j in zeros:
                    h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
                    val = compute_c5(h_bin, dx)
                    if val < best_val - 1e-12:
                        best_val = val
                        best = h_bin.copy()
                        any_improve = True
                        improved = True
                        break
                    h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
                if improved:
                    break
            if not any_improve:
                break
    return best, best_val

def project_feasible(x, n):
    """Project x onto {0 <= x_i <= 1, sum(x) = n/2}"""
    tgt = n / 2.0
    # Bisection on lambda: x_i = clip(x_i - lambda, 0, 1), sum = tgt
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(x - mid, 0, 1)
        s = np.sum(proj)
        if s > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)
```

def subgradient_step(h, n, dx, step_size):
    corr = np.correlate(h, 1-h, mode="full")
    k_star = np.argmax(corr)
    # Gradient of c[k_star] w.r.t. h
    # c[k_star] = sum_j h[j+k_star] * (1-h[j])
    # ∂/∂h[i]: (1-h[i-k_star]) if valid - h[i+k_star] if valid
    grad = np.zeros(n)
    k = k_star - (2*n - 2) // 2  # Convert to signed lag
    # Actually, in "full" mode, the output index k corresponds to lag k - (n-1)
    # So lag = k_star - (n-1)
    lag = k_star - (n - 1)
    # ∂c[lag]/∂h[i] = (1-h[i-lag]) if 0<=i-lag<n - h[i+lag] if 0<=i+lag<n
    # Which is: from the "shifted" term: (1-h[i-lag]) when i-lag is valid
    #          from the "unshifted" term: -h[i+lag] when i+lag is valid
    
    # h[j+lag] * (1-h[j]):
    # ∂/∂h[i]: 
    #   if i = j+lag (j=i-lag): derivative w.r.t. h[i] is (1-h[i-lag])
    #   if i = j: derivative w.r.t. h[i] is -h[i+lag]
    
    mask1 = np.arange(n)
    valid1 = (mask1 - lag >= 0) & (mask1 - lag < n)
    valid2 = (mask1 + lag >= 0) & (mask1 + lag < n)
    
    grad[valid1] += (1 - h[mask1[valid1] - lag])
    grad[valid2] += (-h[mask1[valid2] + lag])
    
    # Step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if abs(s - tgt) > 1e-12:
        # Proportional scaling with clipping
        x = x * (tgt / s) if s > 0 else np.full(n, 0.5)
        x = np.clip(x, 0.0, 1.0)
        # Fix residual
        for _ in range(3):
            diff = tgt - np.sum(x)
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                mask = x < 1.0
            else:
                mask = x > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_obj(h, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    return m + tau * np.log(np.sum(np.exp(sh)))

def slsqp(h0, n, tau, maxiter):
    def obj(x):
        return smooth_obj(x, tau)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(r.x, n)
    except:
        return h0.copy()

def coord_descent(h, n, dx, max_iter=1000, time_limit=30):
    """Coordinate descent with exact max and sum constraint via rescaling."""
    t0 = time.time()
    h = h.copy()
    best_val = compute_c5(h, dx)
    best_h = h.copy()
    tgt = n / 2.0
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            # Save current
            old_val = h[i]
            # Try to optimize h[i] while maintaining sum
            # h[i] can range such that sum stays at tgt
            # Other sum = tgt - h[i], so h[i] is free in [0,1]
            # But we need sum of others = tgt - h[i], which must be in [0, n-1]
            # So h[i] in [max(0, tgt-(n-1)), min(1, tgt)]
            lo = max(0.0, tgt - (n-1))
            hi = min(1.0, tgt)
            if lo >= hi:
                continue
            
            # Evaluate objective at a few points
            # The max correlation as function of h[i] is max of linear functions
            # Find the kink points
            corr = np.correlate(h, 1.0 - h, mode="full")
            # h[i] appears in corr at specific lags
            # Let's just do a 1D minimization via grid
            n_grid = 50
            candidates = np.linspace(lo, hi, n_grid)
            best_c = np.inf
            best_v = lo
            for v in candidates:
                h[i] = v
                # Rescale others to maintain sum
                others_sum = tgt - v
                idx = np.arange(n) != i
                s_others = h[idx].sum()
                if s_others > 0:
                    h[idx] = h[idx] * (others_sum / s_others)
                    h[idx] = np.clip(h[idx], 0.0, 1.0)
                # After clipping, fix sum again (approximately)
                diff = tgt - h.sum()
                if abs(diff) > 1e-10 and idx.any():
                    mask = (h < 1.0) & (np.arange(n) != i) if diff > 0 else (h > 0.0) & (np.arange(n) != i)
                    if mask.sum() > 0:
                        h[mask] += diff / mask.sum()
                        h = np.clip(h, 0.0, 1.0)
                        h[i] = v  # Restore
                
                c = compute_c5(h, 1.0)  # Without dx for comparison
                if c < best_c:
                    best_c = c
                    best_v = v
            
            h[i] = best_v
            # Rescale
            idx = np.arange(n) != i
            others_sum = tgt - best_v
            s_others = h[idx].sum()
            if s_others > 0 and abs(s_others - others_sum) > 1e-10:
                h[idx] = h[idx] * (others_sum / s_others)
                h = np.clip(h, 0.0, 1.0)
                h[i] = best_v
            
            val = compute_c5(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()
                improved = True
            
            if not improved and it > 50:
                # Check if we're stuck
                pass
        
        if not improved and it > 20:
            break
    
    return best_h, best_val
```

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if s < 1e-15:
        return np.full(n, 0.5)
    x *= tgt / s
    x = np.clip(x, 0.0, 1.0)
    for _ in range(5):
        d = tgt - np.sum(x)
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = x < 1.0
        else:
            m = x > 0.0
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
        x = np.clip(x, 0.0, 1.0)
    return x

def smooth_obj(h, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    return m + tau * np.log(np.sum(np.exp(sh)))

def slsqp_opt(h0, n, tau, maxiter):
    tg = n / 2.0
    def obj(x):
        return smooth_obj(x, tau)
    def eq(x):
        return np.sum(x) - tg
    def eqjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq, 'jac': eqjac}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(r.x, n)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    
    # Get initial
    if 'initial_h_values' in globals():

    best_n = n0

    grids = sorted(set([n0, n0-1, n0+1, 96, 100, 120, 144, 160]))

        if alloc < 10:

        else:

            for tau in taus:

                x_cur = slsqp_opt(x_cur, n, tau, 400)
                val = compute_c5(x_cur, dx)
                if val < best_val:

                    best_h = x_cur.copy()
                    best_n = n
        
        # Binary local search

                if not improved:
