            g[k:] += (1.0 - h[:n-k])
            # j-shift = j+k valid: j in [0, n-1-k]

    return h

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_fast(h, n, tol=1e-7):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mv = float(corr.max())
    active = np.where(corr >= mv - tol)[0]
    g = np.zeros(n)
    for s in active:
        shift = int(s - (n - 1))
        if shift >= 0:
            g[:n - shift] += (1.0 - h[shift:])
            g[shift:] -= h[:n - shift]
        else:
            k = -shift
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
    g /= len(float(active))
    return g * dx, mv

def slsqp_polish(h, n):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(c)) * dx
    def eq_fun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': 500, 'ftol': 1e-15})
        hc = repair(r.x, n)
        return hc, compute_c5(hc, n)
    except:
        return h, compute_c5(h, n)

def optimize_one(h, n, t_end, rng):

    alpha = 0.1
    best_local = c_h

The key observation: at optimality, the "active" lags (those achieving the max correlation) should be balanced. If we know which lags are active, we can set up a system:

- For non-active lags: corr_s(h) ≤ C
- Constraint: 0 ≤ h[i] ≤ 1, sum(h) = n/2

This is a system of nonlinear equations. We can try to solve it with scipy's fsolve or by iterative methods.

Minimize: max_s corr_s + μ * (sum(h) - n/2)² + ν * max(0, h-1)² + ν * max(0, -h)²

Or better, use the log-sum-exp smoothing with very high beta:

f(h) = (1/β) * log(∑_s exp(β * corr_s * dx))

This approximates max_s corr_s * dx as β → ∞. It's smooth and we can use gradient descent.

The gradient is:
∇f = (1/β) * (Σ_s exp(β*corr_s*dx) * ∂(corr_s*dx)/∂h) / (Σ_s exp(β*corr_s*dx))
= (Σ_s p_s * ∂corr_s/∂h) * dx

where p_s = exp(β*corr_s*dx) / Σ exp(β*corr_s*dx) are softmax weights.

This is a weighted average of the subgradients, which is exactly the subgradient with softmax weighting! So my subgradient approach with tol=0 (or small tol) is a special case.

def repair(h, n):

        return np.full(n, 0.5)

        cnt = mask.sum()
        if cnt > 0:

            h = np.clip(h, 0.0, 1.0)

            h += diff2 / n
            h = np.clip(h, 0.0, 1.0)
