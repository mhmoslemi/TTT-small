    target = n * 0.5
    h = h.copy()
    
    for step in range(steps):
        c = compute_corr(h)
        # Weight the top few lags
        cmax = np.max(c)
        # Use softmax weighting for smoothness
        weights = np.exp((c - cmax) * 20)
        weights /= weights.sum()
        
        # Gradient of weighted sum w.r.t. h
        # c[k] = sum_j h[j] * (1 - h[j + k - n + 1])
        # d c[k] / d h[i] = (1-h[i+k-n+1]) - h[i-k+n-1]  (with validity)
        
        grad = np.zeros(n)
        for k_idx in range(2*n-1):
            wk = weights[k_idx]
            if wk < 1e-10:
                continue
            m = k_idx - n + 1  # actual lag
            for i in range(n):
                # contribution to grad[i] from lag m
                j1 = i + m  # h[i] * (1-h[i+m]) → deriv w.r.t h[i] is (1-h[i+m])
                j2 = i - m  # h[i-m] * (1-h[i]) → deriv w.r.t h[i] is -h[i-m]

                if 0 <= j2 < n:

        alpha = alpha_init / (1 + step * 0.001)

        # Fix sum
        diff = target - h_new.sum()
        h_new += diff / n
        h_new = np.clip(h_new, 0, 1)
        
        h = h_new
    
    return h

def slsqp_optimize(h, n, t_param, maxiter=300):
    dx = 2.0 / n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n * 0.5,
             'jac': lambda x, n=n: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h.copy(), args=(n, t_param),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    h_out = repair(res.x, n)
    return h_out


def slsqp_obj_factory(n, t_param):
    def _obj(x):
        return smooth_obj(x, n, t_param)
    return _obj


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def time_left(frac):
        return budget_s * frac - elapsed()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Subgradient momentum at base resolution
    if time_left(0.35) > 5:
        steps = min(8000, int(time_left(0.35) * 15))
        h_s, v_s = subgradient_momentum(best_h, n0, steps, 50.0, 200.0, 0.05)
        h_s = repair(h_s, n0)
        if v_s < best_c5:
            best_c5, best_h = v_s, h_s

    # Phase 2: SLSQP polish at base resolution
    if time_left(0.45) > 10:
        for t_trial in [80.0, 200.0, 500.0]:
            if time_left(0.45) < 5:
                break
            h_s = slsqp_optimize(best_h, n0, t_trial, maxiter=200)
            v = compute_c5(h_s)
            if v < best_c5:
                best_c5, best_h = v, h_s.copy()

    # Phase 3: Alternating LP to find structured solution
    if time_left(0.65) > 10:
        T_try = best_c5 * 0.995
        h_lp, v_lp = alternating_lp(best_h, n0, T_try, max_iter=40)
        h_lp = repair(h_lp, n0)
        v_lp = compute_c5(h_lp)
        if v_lp < best_c5:
            best_c5, best_h = v_lp, h_lp
        # Try slightly tighter
        if time_left(0.65) > 10:
            T_try2 = best_c5 * 0.99
            h_lp2, v_lp2 = alternating_lp(best_h, n0, T_try2, max_iter=30)
            h_lp2 = repair(h_lp2, n0)
            v_lp2 = compute_c5(h_lp2)
            if v_lp2 < best_c5:
                best_c5, best_h = v_lp2, h_lp2

    # Phase 4: Upscale to 2x and optimize
    n1 = n0 * 2
    if n1 <= 800 and time_left(0.85) > 20:
        xo = np.linspace(0.0, 2.0, n0)
        xn = np.linspace(0.0, 2.0, n1)
        h_init = np.interp(xn, xo, best_h)
        h_init = repair(h_init, n1)
        # Subgradient on upscaled
        steps2 = min(5000, int(time_left(0.75) * 10))
        h_u, v_u = subgradient_momentum(h_init, n1, steps2, 50.0, 300.0, 0.03)
        h_u = repair(h_u, n1)
        v_u = compute_c5(h_u)
        if v_u < best_c5:
            best_c5, best_h = v_u, h_u
        # SLSQP on upscaled
        if time_left(0.85) > 10:
            h_u2 = slsqp_optimize(best_h, n1, 200.0, maxiter=150)
            v2 = compute_c5(h_u2)
            if v2 < best_c5:
                best_c5, best_h = v2, h_u2

    # Phase 5: Random restarts + SLSQP at base res
    if time_left(0.93) > 10:
        n_cur = len(best_h)
        for attempt in range(4):
            if time_left(0.93) < 5:
                break
            noise = np.random.randn(n_cur) * 0.08
            h_try = best_h + noise
            h_try = repair(h_try, n_cur)
            h_t = slsqp_optimize(h_try, n_cur, 300.0, maxiter=150)
            v_t = compute_c5(h_t)
            if v_t < best_c5:
                best_c5, best_h = v_t, h_t

    # Phase 6: Final subgradient polish
    if time_left(0.99) > 5:
        n_f = len(best_h)
        steps_f = min(3000, int(time_left(0.99) * 20))
        h_f, v_f = subgradient_momentum(best_h, n_f, steps_f, 100.0, 500.0, 0.02)
        h_f = repair(h_f, n_f)
        v_f = compute_c5(h_f)
        if v_f < best_c5:
            best_c5, best_h = v_f, h_f

    # Final: ensure constraint
    n_final = len(best_h)
    best_h = repair(best_h, n_final)
    c5_final = compute_c5(best_h)

    return (best_h, c5_final, n_final)
